```python
def repair_vector(v, target_sum, max_attempts=10000):
    "Adjust vector to ensure sum(v) == target_sum"
    n = len(v)
    attempts = 0
    while abs(target_sum - np.sum(v)) > 1e-12 and attempts < max_attempts:
        i = np.random.randint(n)
        if target_sum > np.sum(v):
            # Need to add to this element
            available_to_add = min(target_sum - np.sum(v), 1 - v[i])
            if available_to_add > 0:
                add_amount = np.random.random() * available_to_add
                v[i] += add_amount
        else:
            # Need to remove from this element
            available_to_remove = max(0, v[i])
            remove_amount = np.random.random() * available_to_remove
            if remove_amount > 0:
                v[i] -= remove_amount
```

```python
        # Cap child values between 0 and 1
        low = max(low, 0)
        high = min(high, 1)
        child[i] = np.random.uniform(low, high)
    return child

def cauchy_mutation(v, scale=0.1, rate=0.1):
    "Apply Cauchy mutation to all elements in v"
    n = len(v)
    mutated = np.copy(v)
    for i in range(n):
        if np.random.random() < rate:
            # Cauchy distribution
            delta = np.random.standard_cauchy() * scale
            mutated[i] = np.clip(v[i] + delta, 0, 1)
    return mutated

def real_valued_ga(n_points, target_sum=0.5, max_attempts=1000, time_budget=300):
    "Genetic algorithm for real-valued optimization"
    n_pop = 50
    N = n_points  # length of vector
    initial_sum = N/2.0
    
    # Initialize population with sum constraint
    pop = []
    while len(pop) < n_pop:
        candidate = np.random.random(N)  # Sample [0,1]
        candidate = repair_vector(candidate, initial_sum)[0]
        if candidate not in pop:  # Avoid duplicates
            pop.append(candidate)
    
    pop_arr = np.array(pop)
    
    def fitness_func(x):
        "Penalty for constraint violation"
        sum_val = np.sum(x)
        penalty = 0 if abs(sum_val - target_sum) < 1e-5 else 1000 * abs(sum_val - target_sum)
        return penalty + compute_c5(x)
    
    best_idx = np.argmin([fitness_func(x) for x in pop_arr])
    best_fitness = fitness_func(pop_arr[best_idx])
    best_vector = pop_arr[best_idx].copy()
    
    start_time = time.time()
    
    for gen in range(max_attempts):
        if time.time() - start_time > time_budget:
            break
            
        # Selection
        fitnesses = np.array([fitness_func(x) for x in pop_arr])
        probs = fitnesses / fitnesses.sum()  # Proportional selection
        selected_idx = np.random.choice(n_pop, 2, p=probs)
        parent1, parent2 = pop_arr[selected_idx[0]], pop_arr[selected_idx[1]]
        
        # Crossover
        child = blended_crossover(parent1, parent2, n_points)
        
        # Mutation
        child_mutated = cauchy_mutation(child, scale=0.05)
        
        # Repair and repair parents if needed
        child_repaired, child_sum = repair_vector(child_mutated, initial_sum)
        parent1_repaired, p1_sum = repair_vector(parent1, initial_sum)
        parent2_repaired, p2_sum = repair_vector(parent2, initial_sum)
        
        # Calculate fitness
        child_fit = fitness_func(child_repaired)
        
        # Elitism and selection
        if child_fit < best_fitness:
            best_idx = gen_index  # Track best found
            best_fitness = child_fit
            best_vector = child_repaired.copy()
        
        # Replace worst with best found so far
        worst_idx = np.argmax(fitnesses)
        pop_arr[worst_idx] = child_repaired
        
    return best_vector, best_fitness
```

```python
def ga_refinement(n_points, time_budget=200):
    "Refine solution with simulated annealing"
    n = n_points
    dim = n
    initial_sum = dim/2.0
    init_pos = np.random.random(dim)
    init_pos = repair_vector(init_pos, initial_sum)[0]

    # SA parameters
    T0 = 0.1 / np.sqrt(dim)
    T_min = 1e-5
    n_iterations = int(np.log(T0/T_min) / np.log(0.99)) * 2
    
    best_pos = init_pos.copy()
    best_val = compute_c5(best_pos)
    
    t = T0
    pos = init_pos.copy()
    best_pos = pos.copy()
    best_val = compute_c5(best_pos)
    
    for i in range(n_iterations):
        if time.time() - start_time > time_budget:
            break
            
        # Generate new position by adding Gaussian noise
        perturbation = np.random.normal(0, 0.05, dim)
        new_pos = np.clip(pos + perturbation, 0, 1)
        new_pos = repair_vector(new_pos, initial_sum)[0]
        
        new_val = compute_c5(new_pos)
        
        # Apply simulated annealing
        delta = new_val - best_val
        if delta < 1e-10 or np.random.random() < np.exp(-delta * 1000 / t):
            pos = new_pos
            current_val = new_val
            
        if current_val < best_val:
            best_val = current_val
            best_pos = pos.copy()
            
        t *= 0.99
        
    return best_pos, best_val
```
