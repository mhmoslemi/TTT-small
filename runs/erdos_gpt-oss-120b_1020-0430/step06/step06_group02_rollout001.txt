import time, numpy as np
from scipy.optimize import minimize

# ---------- global vars for constraints ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- helper functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == n/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple iterative repair – fast for n=512
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift (correct version)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    # forward contribution (1 - h[i+k]) for i <= n-1-k
    grad[: n - k] += 1.0 - h[k:]
    # backward contribution (-h[i-k]) for i >= k
    grad[k:] -= h[: n - k]
    return grad

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    """SLSQP minimisation of the soft‑max approximation of C5."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair any tiny sum error
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x

def thue_morse_bits(n: int) -> np.ndarray:
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]

# ---------- main routine ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target = n / 2.0

    # ---- generate initial candidates ----
    candidates = []
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        candidates.append((h0, compute_c5(h0)))
    # a few Thue‑Morse variants
    for sz in (256, 512):
        tm = thue_morse_bits(sz)
        if sz != n:
            tm = np.interp(np.linspace(0, sz, n, endpoint=False), np.arange(sz), tm)
        candidates.append((tm, compute_c5(tm)))
    # random balanced binary vectors
    for _ in range(4):
        hb = np.zeros(n)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        candidates.append((hb, compute_c5(hb)))
    # random real‑valued vectors
    for _ in range(3):
        hr = rng.random(n)
        hr = enforce_sum(hr, rng)
        candidates.append((hr, compute_c5(hr)))

    best_h, best_c = min(candidates, key=lambda t: t[1])
    best_h = best_h.copy()

    # ---- Phase 1: subgradient descent with decreasing step ----
    it = 0
    no_improve = 0
    step0 = 0.25
    decay = 0.9997
    while time.time() - start < budget_s * 0.45:
        step = step0 * (decay ** it)
        if step < 1e-13:
            break
        grad = worst_gradient(best_h)
        h_new = best_h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 2000:
            # small random nudge to escape plateau
            i = rng.integers(0, n)
            eps = rng.random() * 0.1
            h_new = best_h.copy()
            h_new[i] = np.clip(h_new[i] + eps, 0.0, 1.0)
            h_new = enforce_sum(h_new, rng)
            best_h, best_c = h_new, compute_c5(h_new)
            no_improve = 0
        it += 1

    # ---- Phase 2: simulated annealing ----
    T = 0.001
    decay_T = 0.9999
    while time.time() - start < budget_s * 0.75:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        eps = rng.random() * max_transfer * 0.3
        h_new = best_h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        if c_new < best_c or rng.random() < np.exp(-(c_new - best_c) / T):
            best_h, best_c = h_new, c_new
        T *= decay_T
        if T < 1e-9:
            T = 1e-9

    # ---- Phase 3: SLSQP refinements with increasing beta ----
    for beta in (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0, 200000.0):
        if time.time() - start > budget_s * 0.90:
            break
        h_opt = slsqp_optimize(best_h, beta, rng, max_iter=2000)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- Phase 4: fine local gradient sweeps ----
    for step in (0.05, 0.02, 0.01, 0.005, 0.002):
        if time.time() - start > budget_s * 0.98:
            break
        for _ in range(400):
            grad = worst_gradient(best_h)
            h_new = best_h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, rng)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_h, best_c = h_new, c_new

    # final sanity enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
