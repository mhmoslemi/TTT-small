```python
def repair_vector(v, target_sum, max_attempts=10000):
    "Adjust vector to ensure sum(v) == target_sum"
    n = len(v)
    attempts = 0
    while abs(target_sum - np.sum(v)) > 1e-12 and attempts < max_attempts:
        i = np.random.randint(n)
        if target_sum > np.sum(v):
            # Need to add to this element
            available_to_add = min(target_sum - np.sum(v), 1 - v[i])
            if available_to_add > 0:
                add_amount = np.random.random() * available_to_add
                v[i] += add_amount
        else:
            # Need to remove from this element
            available_to_remove = max(0, v[i])
            remove_amount = np.random.random() * available_to_remove
            if remove_amount > 0:
                v[i] -= remove_amount
        attempts += 1
    return np.clip(v, 0, 1), np.sum(v)
```

```python
def blended_crossover(parent1, parent2, alpha=1.5, n_points=128):
    "Perform blended crossover between two parents"
    child = np.zeros_like(parent1)
    n = len(parent1)
    for i in range(n):
        r1, r2 = parent1[i], parent2[i]
        low = min(r1, r2) - alpha * abs(r1 - r2)
        high = max(r1, r2) + alpha * abs(r1 - r2)
        # Cap child values between 0 and 1
        low = max(low, 0)
        high = min(high, 1)
        child[i] = np.random.uniform(low, high)
    return child

def cauchy_mutation(v, scale=0.1, rate=0.1):
    "Apply Cauchy mutation to all elements in v"
    n = len(v)
    mutated = np.copy(v)
    for i in range(n):
        if np.random.random() < rate:
            # Cauchy distribution
            delta = np.random.standard_cauchy() * scale
            mutated[i] = np.clip(v[i] + delta, 0, 1)
    return mutated
```

```python
def ga_refinement(n_points, time_budget=200):

    t = T0
    pos = init_pos.copy()
    best_pos = pos.copy()
    best_val = compute_c5(best_pos)

        if time.time() - start_time > time_budget:
            break
            
        # Generate new position by adding Gaussian noise
        perturbation = np.random.normal(0, 0.05, dim)
        new_pos = np.clip(pos + perturbation, 0, 1)
        new_pos = repair_vector(new_pos, initial_sum)[0]
        
        new_val = compute_c5(new_pos)
        
        # Apply simulated annealing
        delta = new_val - best_val
        if delta < 1e-10 or np.random.random() < np.exp(-delta * 1000 / t):
            pos = new_pos
            current_val = new_val
            
        if current_val < best_val:
            best_val = current_val
            best_pos = pos.copy()

    return best_pos, best_val
```

```python
            initial_vec = None
            if parent_available and results[resolution//2] is not None:
                # Upsample previous resolution solution
                initial_vec = upsample_vector(results[resolution//2], resolution)
                initial_vec = repair_vector(initial_vec, resolution/2.0)[0]
            
            # Run GA phase
            vec, c5_val = real_valued_ga(
                n_points=resolution,
                time_budget=time_limit
            )
            
            # Apply local refinement
            best_vec, best_c5 = ga_refinement(
                n_points=resolution,
                time_budget=time_limit//2
            )
            
            results[resolution] = (best_vec, best_c5)
            
        except Exception as e:
            print(f"Error at {resolution}: {e}")
            continue
    
    # Select best solution across all resolutions
    best_result = None
    best_c5 = float('inf')
    
    for resolution, (vec, c5_val) in results.items():
        if c5_val < best_c5:
            best_c5 = c5_val
            best_result = vec
    
    # Enforce constraints one final time
    best_vec = repair_vector(best_vec, best_vec.size/2)[0]
    
    # Calculate final C5
    final_c5 = compute_c5(best_vec)
    
    return best_vec.tolist(), float(final_c5), int(best_vec.size)
```

**Key Components**:
- Repair function maintains integral constraint while allowing exploration
- Blended crossover preserves diversity while maintaining feasibility
- Cauchy mutation provides global exploration power
- Fitness proportional selection avoids local optima
- Local refinement with simulated annealing improves local fitness
- Multi-resolution strategy leverages GA's global search ability at lower resolutions then refines solutions at higher resolutions
</implementation-procedure>
