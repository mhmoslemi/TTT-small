import time, math, numpy as np

def _corr_max(h):
    """Return integer max of np.correlate(h,1-h,'full') for binary h."""
    return int(np.correlate(h, 1 - h, mode='full').max())

def _worst_shift(corr, n):
    """Return shift k corresponding to max correlation."""
    idx = int(np.argmax(corr))
    return idx - (n - 1)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        # binarise, keep exact half ones
        h = (h > 0.5).astype(np.int8)
        target = n // 2
        ones = int(h.sum())
        if ones > target:
            idx = np.flatnonzero(h)
            flip = rng.choice(idx, ones - target, replace=False)
            h[flip] = 0
        elif ones < target:
            idx = np.flatnonzero(1 - h)
            flip = rng.choice(idx, target - ones, replace=False)
            h[flip] = 1
    else:
        n = 512
        target = n // 2
        h = np.zeros(n, dtype=np.int8)
        ones_idx = rng.choice(n, target, replace=False)
        h[ones_idx] = 1

    cur_corr = np.correlate(h, 1 - h, mode='full')
    cur_max = int(cur_corr.max())
    best_h = h.copy()
    best_max = cur_max

    arr = np.arange(n)
    start_temp = 10.0
    end_temp = 0.001

    # ----- main simulated‑annealing loop -----------------------------
    while time.time() - start < budget_s * 0.99:
        # temperature linear in elapsed fraction
        t_frac = (time.time() - start) / (budget_s * 0.99)
        T = start_temp * (1 - t_frac) + end_temp

        # worst shift for this configuration
        shift = _worst_shift(cur_corr, n)

        # 80 % of the time try a directed swap that improves the worst shift
        if rng.random() < 0.8:
            if shift >= 0:
                i_mask = (h[:n - shift] == 1) & (h[shift:] == 0)
                i_cand = np.flatnonzero(i_mask)
                if i_cand.size:
                    j_mask = (h == 0) & (arr + shift < n) & (h[arr + shift] == 1)
                    j_cand = np.flatnonzero(j_mask)
                    if j_cand.size:
                        i = rng.choice(i_cand)
                        j = rng.choice(j_cand)
                        new_h = h.copy()
                        new_h[i] = 0
                        new_h[j] = 1
                        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
                        new_max = int(new_corr.max())
                        delta = new_max - cur_max
                        if delta <= 0 or rng.random() < math.exp(-delta / T):
                            h = new_h
                            cur_corr = new_corr
                            cur_max = new_max
                            if cur_max < best_max:
                                best_max = cur_max
                                best_h = h.copy()
                        continue  # go to next iteration

        # ----- fallback: random 1↔0 swap ------------------------------
        # pick a random 1
        a = rng.integers(n)
        while h[a] != 1:
            a = rng.integers(n)
        # pick a random 0
        b = rng.integers(n)
        while h[b] != 0:
            b = rng.integers(n)
        new_h = h.copy()
        new_h[a] = 0
        new_h[b] = 1
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_max = int(new_corr.max())
        delta = new_max - cur_max
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            h = new_h
            cur_corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()

    # ----- final evaluation -----------------------------------------
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_max = int(final_corr.max())
    final_c = final_max * (2.0 / n)

    # ensure proper format
    best_h = np.clip(best_h, 0, 1)
    return best_h.tolist(), float(final_c), int(n)
