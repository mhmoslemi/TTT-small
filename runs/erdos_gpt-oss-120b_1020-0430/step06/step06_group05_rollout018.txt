def slsqp_refine(h, target, rng, beta, max_iter):

    global CURRENT_TARGET, CURRENT_BETA

    CURRENT_TARGET = target

    CURRENT_BETA = beta

    from scipy.optimize import minimize

    n = h.size

    bounds = [(0.0, 1.0)] * n

    cons = {'type':'eq', 'fun': sum_eq_constraint}

    res = minimize(softmax_obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':max_iter,'ftol':1e-12,'disp':False})

    x = np.clip(res.x, 0.0, 1.0)

    # enforce sum

    diff = target - x.sum()

    if abs(diff) > 1e-12:

        x = enforce_sum(x, target, rng)

    return x

def gradient_refine(h, target, rng, step=0.015, max_iter=10):

        _, grad = worst_shift_and_grad(h)

def random_perturb(h, rng, scale=0.2):

    n = h.size

    i = rng.integers(0,n)

    j = rng.integers(0,n)

    if i == j:

        return h

    max_eps = min(h[i], 1.0 - h[j])

    if max_eps <= 0:

        return h

    eps = max_eps * rng.random() * scale

    h2 = h.copy()

    h2[i] -= eps

    h2[j] += eps

    return h2

def enforce_sum(h, target, rng):

    diff = target - h.sum()

    n = h.size

    # while diff not zero

    while abs(diff) > 1e-12:

        i = rng.integers(0,n)

        if diff > 0:

            cap = 1.0 - h[i]

            if cap <= 0:

                continue

            delta = min(cap, diff)

        else:

            cap = -h[i]  # negative capacity (how much we can decrease)

            if cap >= 0:

                continue

            delta = max(cap, diff)  # both negative

        h[i] += delta

        diff = target - h.sum()

    return h

Potential infinite loop if diff > 0 but all h[i]==1 (cap=0). But diff <= n - sum(h). If sum(h) < target and all entries are 1, sum(h) equals n > target, not possible. If sum(h) < target, some entry must be <1, so cap>0 exists. So fine.

    init_h = enforce_sum(init_h, target_sum, rng)

import time
import numpy as np

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            delta = min(cap, diff)
        else:
            cap = -h[i]
            if cap >= 0.0:
                continue
            delta = max(cap, diff)  # both negative
        h[i] += delta
        diff = target - h.sum()
    return h


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount of mass while preserving total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def worst_shift_and_grad(h: np.ndarray):
    """Return shift k delivering max overlap and corresponding gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # shift index (can be negative)
    k = idx
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        kk = -k
        grad[kk:] -= h[: n - kk]
        grad[: kk] -= h[kk:]
    return k, grad


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Few steps of gradient descent along worst‑shift direction."""
    for i in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** i)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int) -> np.ndarray:
    """One SLSQP pass on the smooth max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize

    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


def tournament_select(fitness: np.ndarray, rng: np.random.Generator,
                      tournament_size: int = 5) -> int:
    """Return index of the best individual among a random subset."""
    n = len(fitness)
    cand = rng.choice(n, tournament_size, replace=False)
    return cand[np.argmin(fitness[cand])]


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        init_h = np.array(initial_h_values, dtype=float)
        init_h = np.clip(init_h, 0.0, 1.0)
    else:
        n = 512
        init_h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        init_h[idx] = 1.0
    target_sum = n / 2.0
    init_h = enforce_sum(init_h, target_sum, rng)

    # Population
    pop_size = 200
    pop = [init_h]
    while len(pop) < pop_size:
        h = rng.random(n)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / h.sum())
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target_sum, rng)
        pop.append(h)

    c_vals = np.empty(pop_size)
    for i, h in enumerate(pop):
        c_vals[i] = compute_c5(h)

    best_idx = int(np.argmin(c_vals))
    best_h = pop[best_idx].copy()
    best_c = c_vals[best_idx]

    tournament_sz = 5

    # Main evolutionary loop
    while time.time() - start < budget_s * 0.95:
        p1_idx = tournament_select(c_vals, rng, tournament_sz)
        p2_idx = tournament_select(c_vals, rng, tournament_sz)
        p1, p2 = pop[p1_idx], pop[p2_idx]

        cut = rng.integers(1, n)
        child = np.concatenate([p1[:cut], p2[cut:]])
        child = enforce_sum(child, target_sum, rng)
        child = random_perturb(child, rng, scale=0.15)
        child = gradient_refine(child, target_sum, rng, step=0.02, max_iter=5)
        c_child = compute_c5(child)

        worst_idx = int(np.argmax(c_vals))
        if c_child < c_vals[worst_idx]:
            pop[worst_idx] = child
            c_vals[worst_idx] = c_child
            if c_child < best_c:
                best_c = c_child
                best_h = child.copy()

        if best_c < 0.38078:
            break

    # Longer gradient refinement on the best found solution
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        best_h = gradient_refine(best_h, target_sum, rng,
                                 step=0.015, max_iter=30)
        best_c = compute_c5(best_h)
        remaining = budget_s - (time.time() - start)

    # Final SLSQP polishing if time permits
    if remaining > 5.0:
        for beta in (500.0, 2000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.99:
                break
            h_s = slsqp_refine(best_h, target_sum, rng, beta=beta, max_iter=400)
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()

    # Ensure feasibility before returning
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
