        return h

        total = free.sum()

        if total > 0:

def binary_local_search(h_init, rng, max_iters, time_limit):
    # copy h to avoid modifying input? We'll modify copy.
    h = h_init.copy()
    n = len(h)
    # compute G
    G = np.correlate(h, 1 - h, mode='full')
    G = G[n-1:]
    best_Gmax = G.max()
    best_h = h.copy()
    start = time.time()
    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        k = int(G.argmax())
        limit = n - k
        # find candidates
        idx = np.arange(limit)
        # A: h[i]=1, h[i+k]=0
        A_mask = (h[idx] == 1) & (h[idx + k] == 0)
        B_mask = (h[idx] == 0) & (h[idx + k] == 1)
        # if both non-empty, choose swap
        if A_mask.any() and B_mask.any():
            i = rng.choice(idx[A_mask])
            j = rng.choice(idx[B_mask])
        else:
            # fallback random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # compute delta
        delta = np.zeros(n, dtype=int)
        # i removal
        maxk_i = n - 1 - i
        if maxk_i >= 0:
            delta[:maxk_i+1] -= (1 - h[i + np.arange(maxk_i+1)])
        # j addition
        maxk_j = n - 1 - j
        if maxk_j >= 0:
            delta[:maxk_j+1] += (1 - h[j + np.arange(maxk_j+1)])
        # perform swap
        h[i] = 0
        h[j] = 1
        # update G
        G += delta
        current_Gmax = G.max()
        if current_Gmax < best_Gmax:
            best_Gmax = current_Gmax
            best_h = h.copy()
            # early break if target satisfied
            # compute C5 = (2.0/n) * best_Gmax
            if (2.0 / n) * best_Gmax <= 0.3808:
                return best_h, best_Gmax, True
    return best_h, best_Gmax, False

def enforce_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        free = 1.0 - h
        total = free.sum()
        if total > 0:
            h += free * (diff / total)
    else:
        free = h
        total = free.sum()
        if total > 0:
            h += free * (diff / total)
    # Clip for safety
    np.clip(h, 0.0, 1.0, out=h)
    # fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # adjust a random entry within its bounds
        idx = np.random.randint(0, h.size)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

import time
import numpy as np

# ---------- basic utilities ----------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function (binary or continuous) on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float) -> np.ndarray:
    """
    Scale h so that its sum equals target while keeping 0≤h≤1.
    Simple water‑filling projection; works for both binary and fractional h.
    """
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        free = 1.0 - h
        total = free.sum()
        if total > 0:
            h += free * (diff / total)
    else:
        free = h
        total = free.sum()
        if total > 0:
            h += free * (diff / total)   # diff is negative
    np.clip(h, 0.0, 1.0, out=h)

    # tiny residual fix (rare)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        idx = np.random.randint(0, h.size)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑overlap objective.
    Returns d max_k O_k / dh (O_k defined by the full correlation).
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)                 # index in 0 .. 2n-2
    s = idx - (n - 1)                     # signed shift
    grad = np.zeros_like(h, dtype=float)

    if s > 0:
        # (1 - h_{j+s}) term
        grad[: n - s] += (1.0 - h[s:])
        # -h_{j-s} term
        grad[s:] -= h[: n - s]
    elif s < 0:
        s = -s
        # (1 - h_{j-s}) term (since shift negative)
        grad[s:] += (1.0 - h[: n - s])
        # -h_{j+s} term
        grad[: n - s] -= h[s:]
    else:                                 # s == 0 -> O_0 = 0, gradient = 1-2h
        grad = 1.0 - 2.0 * h
    return grad


# ---------- binary local search ------------------------------------------------
def binary_local_search(h_init: np.ndarray, rng: np.random.Generator,
                       max_iters: int, time_limit: float):
    """
    Heuristic that swaps a 1 with a 0 to reduce the worst overlap.
    Returns (best_h, best_Gmax, success_flag).
    """
    h = h_init.copy()
    n = h.size
    G = np.correlate(h, 1 - h, mode="full")
    G = G[n - 1:]                 # G[k] = sum_i h_i (1-h_{i+k})
    best_G = G.max()
    best_h = h.copy()
    start = time.time()
    delta = np.zeros(n, dtype=int)

    while True:
        if time.time() - start > time_limit:
            break
        # worst shift
        k = int(G.argmax())
        limit = n - k
        if limit <= 0:
            # no overlapping region – make a random balanced swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i, j = rng.choice(ones), rng.choice(zeros)
        else:
            idx = np.arange(limit)
            A_mask = (h[idx] == 1) & (h[idx + k] == 0)
            B_mask = (h[idx] == 0) & (h[idx + k] == 1)
            if A_mask.any() and B_mask.any():
                i = rng.choice(idx[A_mask])
                j = rng.choice(idx[B_mask])
            else:
                # fallback random balanced swap
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i, j = rng.choice(ones), rng.choice(zeros)

        # compute delta G for the swap (i:1→0, j:0→1)
        delta.fill(0)

        # remove i
        maxk_i = n - 1 - i
        if maxk_i >= 0:
            delta[: maxk_i + 1] -= (1 - h[i + np.arange(maxk_i + 1)])

        # add j
        maxk_j = n - 1 - j
        if maxk_j >= 0:
            delta[: maxk_j + 1] += (1 - h[j + np.arange(maxk_j + 1)])

        # perform the swap
        h[i] = 0
        h[j] = 1
        G += delta

        cur_G = G.max()
        if cur_G < best_G:
            best_G = cur_G
            best_h = h.copy()
            # early success if we already beat the target
            if (2.0 / n) * best_G <= 0.3808:
                return best_h, best_G, True
    return best_h, best_G, False


# ---------- main entry point ---------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- determine problem size ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 512
        h0 = None

    target = n / 2.0

    # ----- start from the provided construction if any -------------------------
    best_h = None
    best_c5 = float("inf")
    if h0 is not None:
        np.clip(h0, 0.0, 1.0, out=h0)
        enforce_sum(h0, target)
        c0 = compute_c5(h0)
        best_h = h0.copy()
        best_c5 = c0

    # ----- binary local‑search phase (≈70 % of the budget) -----------------------
    while time.time() - start_time < budget_s * 0.70:
        # random balanced binary start
        h_rand = np.zeros(n, dtype=np.int8)
        ones_idx = rng.choice(n, n // 2, replace=False)
        h_rand[ones_idx] = 1

        remaining = budget_s * 0.70 - (time.time() - start_time)
        per_search = min(0.5, remaining)  # seconds
        local_h, Gmax, succeeded = binary_local_search(
            h_rand, rng, max_iters=5000, time_limit=per_search
        )
        c_local = (2.0 / n) * Gmax
        if c_local < best_c5:
            best_c5 = c_local
            best_h = local_h.astype(float)      # switch to float for refinement
        if best_c5 <= 0.3808 or succeeded:
            break

    # ----- continuous refinement using worst‑gradient descent (≈30 %) ---------
    if best_h is None:
        # fallback: a simple balanced binary if everything failed
        best_h = np.zeros(n, dtype=float)
        best_h[rng.choice(n, n // 2, replace=False)] = 1.0

    h = best_h.copy()
    step = 0.2
    max_iter = 20000
    while (time.time() - start_time) < budget_s * 0.98 and max_iter > 0:
        grad = worst_gradient(h)
        h_new = h - step * grad
        np.clip(h_new, 0.0, 1.0, out=h_new)
        enforce_sum(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            if best_c5 <= 0.3808:
                break
        step *= 0.995
        max_iter -= 1
        if step < 1e-8:
            break

    # ----- final sanity & return ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.size)
