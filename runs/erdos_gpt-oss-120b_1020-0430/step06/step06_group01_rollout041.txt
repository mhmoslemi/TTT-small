def binary_hill_climb(n, rng, time_budget, init_h):
    sub_start = n - 1 - np.arange(n)
    h = init_h.copy()
    c = np.correlate(h, 1 - h, mode='full')
    current_max = np.max(c)
    best_h = h.copy()
    best_c = current_max
    ones = np.flatnonzero(h).astype(int)
    zeros = np.flatnonzero(1 - h).astype(int)
    # dictionaries
    pos_one = np.empty(n, dtype=int)
    pos_one[ones] = np.arange(len(ones))
    pos_zero = np.empty(n, dtype=int)
    pos_zero[zeros] = np.arange(len(zeros))
    start = time.time()
    it = 0
    last_improve_it = 0
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break
        # temperature
        T = 5.0 * (1e-4 / 5.0) ** (elapsed / time_budget)
        # random swap
        i = ones[rng.integers(0, len(ones))]
        j = zeros[rng.integers(0, len(zeros))]
        # compute delta
        delta = np.zeros(c.size)
        delta[i:i+n] += h
        delta[sub_start[i]:sub_start[i]+n] -= (1 - h)
        delta[j:j+n] -= h
        delta[sub_start[j]:sub_start[j]+n] += (1 - h)
        new_max = np.max(c + delta)
        accept = False
        if new_max < current_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - current_max) / max(T, 1e-12)):
                accept = True
        if accept:
            # apply
            h[i] = 0.0
            h[j] = 1.0
            # update c
            c += delta
            current_max = new_max
            # update ones/zeros and dicts
            i_pos = pos_one[i]
            j_pos = pos_zero[j]
            # swap entries in ones and zeros
            ones[i_pos] = j
            zeros[j_pos] = i
            pos_one[j] = i_pos
            pos_zero[i] = j_pos
            # remove old positions (optional). Not needed further.
            # better set pos_one[i] = -1; pos_zero[j] = -1
            # but not needed.
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
                last_improve_it = it
        it += 1
        # stagnation check
        if it - last_improve_it > 20000:
            # restart from best
            h = best_h.copy()
            c = np.correlate(h, 1 - h, mode='full')
            current_max = np.max(c)
            ones = np.flatnonzero(h).astype(int)
            zeros = np.flatnonzero(1 - h).astype(int)
            pos_one.fill(-1)
            pos_zero.fill(-1)
            pos_one[ones] = np.arange(len(ones))
            pos_zero[zeros] = np.arange(len(zeros))
            last_improve_it = it
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- basic helpers -------------------

def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function on [0,2]."""
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h)=len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ------------------- binary hill climb -------------------

def binary_hill_climb(n: int, rng: np.random.Generator,
                     time_budget: float, init_h: np.ndarray):
    """Greedy/SA hill climb on binary vectors (sum = n/2)."""
    sub_start = n - 1 - np.arange(n)          # start of -(1‑h) slice for each index
    c = np.correlate(init_h, 1.0 - init_h, mode="full")
    cur_max = np.max(c)
    best_h = init_h.copy()
    best_c = cur_max

    # index sets and position maps
    ones = np.flatnonzero(init_h).astype(int)
    zeros = np.flatnonzero(1.0 - init_h).astype(int)
    pos_one = np.empty(n, dtype=int)
    pos_zero = np.empty(n, dtype=int)
    pos_one.fill(-1)
    pos_zero.fill(-1)
    pos_one[ones] = np.arange(ones.size)
    pos_zero[zeros] = np.arange(zeros.size)

    start = time.time()
    it = 0
    last_improve = 0
    c_len = c.size
    delta = np.empty(c_len, dtype=float)

    while True:
        now = time.time()
        if now - start >= time_budget:

        # temperature schedule (annealing)
        elapsed = now - start
        T = 5.0 * (1e-4 / 5.0) ** (elapsed / time_budget)

        # propose a random 1↔0 swap
        i = ones[rng.integers(0, ones.size)]
        j = zeros[rng.integers(0, zeros.size)]

        delta.fill(0.0)
        delta[i:i + n] += init_h
        delta[sub_start[i]:sub_start[i] + n] -= (1.0 - init_h)

CURRENT_TARGET = None
CURRENT_BETA = None


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Simple hill climb using the worst‑shift gradient."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))   # donor (decrease)
        j = int(np.argmin(grad))   # receiver (increase)
        if i != j:
            eps = min(h[i], 1.0 - h[j]) * rng.random() * step
            if eps > 0:
                h[i] -= eps
                h[j] += eps
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        step = max(step * 0.97, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 8) -> np.ndarray:
    """Fine‑tune using decreasing steps along worst‑gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # quick redistribution
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

                        time_budget: float, init_h: np.ndarray = None):
    """Run hill‑climb + SLSQP on a continuous vector."""
    start = time.time()
    if init_h is None:
        h = rng.random(n)
    else:
        h = init_h.copy()
    h = enforce_sum(h, rng)

    # brief guided hill‑climb
    max_iters = int(min(15000, (time_budget) * 400))
    h, c_val = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.7)
    best_h, best_c = h.copy(), c_val

    # SLSQP passes with increasing β
    beta_seq = (20., 80., 200., 800., 3000., 8000., 20000., 50000.)
    for beta in beta_seq:
        if time.time() - start > time_budget * 0.95:
            break
        iters = 2000 if beta >= 200 else 1000
        h = slsqp_optimize(best_h, beta, rng, max_iter=iters)
        c_val = compute_c5(h)
        if c_val < best_c:
            best_h, best_c = h.copy(), c_val

    # final gradient tidy‑up
    if time.time() - start < time_budget * 0.98:
        h = gradient_refine(best_h, rng, step=0.01, max_iter=8)
        c_val = compute_c5(h)
        if c_val < best_c:
            best_h, best_c = h.copy(), c_val

    return best_h, best_c


# ------------------- main entry point -------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    total_budget = budget_s
    start_total = time.time()
    n = 512
    dx = 2.0 / n

    # ----- initialise binary start -----
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        idx_top = np.argpartition(init, -n // 2)[-n // 2:]
        binary_start = np.zeros(n, dtype=float)
        binary_start[idx_top] = 1.0
    else:
        binary_start = np.zeros(n, dtype=float)
        binary_start[: n // 2] = 1.0
        rng.shuffle(binary_start)

    # ----- binary hill climb -----
    binary_time = total_budget * 0.35
    best_h_bin, best_c_bin = binary_hill_climb(n, rng, binary_time, binary_start)
    best_c5 = best_c_bin * dx
    best_h = best_h_bin.copy()

    # ----- continuous refinement -----
    elapsed = time.time() - start_total
    remaining = total_budget - elapsed
    if remaining > 5.0:
        h0 = best_h.copy()
        h0 += rng.normal(scale=0.02, size=n)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        h_cont, c_cont = continuous_optimize(n, rng, remaining, init_h=h0)
        c5_cont = c_cont * dx
        if c5_cont < best_c5:
            best_h = h_cont.copy()
            best_c5 = c5_cont

    # ----- final minute stochastic tweak -----
    while time.time() - start_total < total_budget * 0.99:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            eps = min(best_h[i], 1.0 - best_h[j]) * rng.random() * 0.01
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c5:
                best_h = nh
                best_c5 = nc

    # ----- sanitise and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(best_h.size)
