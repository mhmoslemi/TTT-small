def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    k = int(h_bin.sum())
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # keep lists of positions of 1 and 0
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    T0 = max(1e-5, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        c_new = compute_c5(cur_h)
        # temperature decay softly
        T = T0 * (0.9995 ** it)
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / max(T,1e-12)):
            cur_c = c_new
            # update lists
            ones[ones == i] = j
            zeros[zeros == j] = i
            if c_new < best_c:
                best_c = c_new
                best_h = cur_h.copy()
                it = 0
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        # stop if stagnation
        if it > 20000:
            break
    return best_h, best_c

def gradient_smoothmax(h, beta):
    n = h.size
    v = get_corr(h)
    bc = beta * v
    m = bc.max()
    exp_term = np.exp(bc - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    # loop over shifts
    # w index 0 corresponds to shift = -(n-1)
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            # contributions where i + shift < n
            g[: n - shift] += wi * one_minus_h[shift:]
            g[shift:]   -= wi * h[: n - shift]
        else:
            s = -shift
            g[s:] += wi * one_minus_h[: n - s]
            g[: n - s] -= wi * h[s:]
    return g

def project_onto_box_sum(v, target):

    # Returns projection onto [0,1] box and sum = target
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    # final proj
    proj = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # due to rounding, might be slightly off; adjust small residual
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        # distribute diff over non-extreme elements
        # find indices where 0 < proj < 1
        free = np.where((proj > 1e-12) & (proj < 1-1e-12))[0]
        if free.size > 0:
            # add equal share
            add = diff / free.size
            proj[free] += add
    return proj

_corr_cache = {}

def _get_corr(v):
    n = v.shape[0]
    N = _corr_cache.get(n)
    if N is None:
        N = 1 << ((2*n - 1).bit_length())
        _corr_cache[n] = N
    # zero pad
    fv = np.fft.rfft(np.concatenate([v, np.zeros(N - n)]))
    fv2 = np.fft.rfft(np.concatenate([1 - v, np.zeros(N - n)]))
    corr = np.fft.irfft(fv * fv2, n=N)
    return corr[: 2*n - 1]

def compute_c5(h):
    v = _get_corr(h)
    return float(v.max() * (2.0 / h.size))

def gradient_smoothmax(h, beta):
    n = h.size
    v = _get_corr(h)
    bc = beta * v
    m = bc.max()
    exp_term = np.exp(bc - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * one_minus_h[shift:]
            g[shift:] -= wi * h[: n - shift]
        else:
            s = -shift
            g[s:] += wi * one_minus_h[: n - s]
            g[: n - s] -= wi * h[s:]
    return g

def project_onto_box_sum(v, target):
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    proj = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # Small numeric drift fix
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-12) & (proj < 1 - 1e-12))[0]
        if free.size:
            proj[free] += diff / free.size
    return proj

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    T0 = max(1e-6, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c_new = compute_c5(cur_h)
        T = T0 * (0.9995 ** it)
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-12)):
            cur_c = c_new
            # update lists

            zeros[zeros == j] = i
            if c_new < best_c:
                best_c = c_new
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
