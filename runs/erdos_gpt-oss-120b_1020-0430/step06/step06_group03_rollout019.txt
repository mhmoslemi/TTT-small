def adam_optimize(h, beta, epochs, lr, eps=1e-8, b1=0.9, b2=0.999):
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    target = h.size / 2.0
    for t in range(1, epochs+1):
        grad = softmax_gradient(h, beta)
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*(grad*grad)
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, target)
    return h

def compute_c5(h):
    n = h.size
    L = 2*n - 1
    # FFT length = next power of two >= L
    Lfft = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, n=Lfft)
    Hc = np.fft.rfft(1 - h, n=Lfft)
    corr = np.fft.irfft(H * np.conj(Hc), n=Lfft)
    # Only the first L entries correspond to full linear convolution
    corr = corr[:L]
    return np.max(corr) * (2.0 / n)

def compute_c5(h):
    n = h.shape[0]
    L = 2*n - 1
    Lfft = 1 << (L-1).bit_length()
    H = np.fft.rfft(h, n=Lfft)
    Hc = np.fft.rfft(1.0 - h, n=Lfft)
    corr = np.fft.irfft(H * np.conj(Hc), n=Lfft)
    corr = corr[:L]
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo+hi)*0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo+hi)*0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1-1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)

            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

        grad[k:] -= wk * h[: n - k]
    return grad
