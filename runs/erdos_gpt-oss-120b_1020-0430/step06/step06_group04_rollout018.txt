    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i:]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii:]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii:]
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    for beta in betas:
        for it in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue

    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target

        v = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * v
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

    n = h_bin.size

    current_c = compute_c5(current_h)
    best_h = current_h.copy()

    zeros = list(np.flatnonzero(current_h == 0.0))
