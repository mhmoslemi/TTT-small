import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


def linear_corr_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) correlation a⋆b using FFT. Returns length len(a)+len(b)-1."""
    n = a.shape[0] + b.shape[0] - 1
    L = next_pow2(n)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    conv = np.fft.irfft(A * B, L)
    return conv[:n]


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    corr = linear_corr_fft(h, 1.0 - h)  # length 2n-1
    return float(corr.max() * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000):
    """Modify h in‑place so that sum(h) == n/2 exactly, staying in [0,1]."""
    n = h.size
    target = n * 0.5
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- gradient for smooth‑max ----------
def gradient_smoothmax(h: np.ndarray, beta: float) -> np.ndarray:
    """∇ of (1/β)log∑exp(β·v_k) where v_k = corr(h,1-h)."""
    n = h.size
    v = linear_corr_fft(h, 1.0 - h)          # size 2n‑1
    bv = beta * v
    m = np.max(bv)
    w = np.exp(bv - m)
    w /= w.sum()                             # soft‑max weights, length 2n‑1

    # A_i = Σ_k w_k (1‑h_{i+k})
    A_full = linear_corr_fft(w, 1.0 - h)
    A = A_full[n - 1 : n - 1 + n]

    # B_i = Σ_k w_k h_{i‑k}
    B_full = linear_corr_fft(w[::-1], h)
    B = B_full[n - 1 : n - 1 + n]

    return A - B


# ---------- worst‑shift sub‑gradient ----------
def worst_shift_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    v = linear_corr_fft(h, 1.0 - h)
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:]    -= h[: n - k]
        else:
            kk = -k
            g[kk:]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
    g /= max(1, len(idxs))
    return g


# ---------- projected gradient descent (smooth‑max) ----------
def grad_descent_smoothmax(h, betas, steps_per_beta, base_step, rng, deadline):
    n = h.size
    for beta in betas:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            cur_c = compute_c5(h)
            best_h, best_c = h, cur_c
            for mult in (1.0, 0.5, 0.25, 0.75, 0.33, 2.0):

                c = compute_c5(cand)

            h = best_h
    return h

def worst_shift_refine(h, rng, base_step, iters, deadline):

# ---------- SLSQP polish with gradient ----------
def slsqp_smoothmax_opt(h0, beta, rng, max_iter=1500):
    n = h0.size
    target = n * 0.5

    def obj(x):
        v = linear_corr_fft(x, 1.0 - x)
        bv = beta * v
        m = np.max(bv)
        return (m + np.log(np.sum(np.exp(bv - m)))) / beta

    def jac(x):
        return gradient_smoothmax(x, beta)

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        jac=jac,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):

    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c

    zeros = list(np.flatnonzero(cur_h == 0.0))

    it = 0

            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c, best_h = c, cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
        if it > 8000:
            break
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    # ---- choose discretisation size ----
    base_n = 960                     # even, moderate resolution
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
        xt = np.linspace(0.0, 2.0, base_n, endpoint=False)
        parent_interp = np.interp(xt, xs, parent)
    else:
        parent_interp = None

    restarts = 5
    for r in range(restarts):
        if time.time() > deadline:
            break
        n = base_n
        if r == 0 and parent_interp is not None:
            h = parent_interp.copy()
        else:
            h = rng.random(n)
            h *= (n * 0.5) / h.sum()
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        # ----- smooth‑max gradient descent -----
        betas = [10, 50, 200, 800, 3000, 12000, 50000]
        h = grad_descent_smoothmax(
            h,
            betas,
            steps_per_beta=40,
            base_step=0.45,
            rng=rng,
            deadline=deadline,
        )

        # ----- worst‑shift refinement -----
        h = worst_shift_refine(h, rng, base_step=0.15, iters=30, deadline=deadline)

        # ----- SLSQP polish (highest β) -----
        for beta in (50000, 120000, 300000):
            if time.time() > deadline:
                break
            h = slsqp_smoothmax_opt(h, beta, rng, max_iter=1200)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

        # ----- binary rounding + SA -----
        h_bin = np.rint(h).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n

    # ----- optional up‑sampling if time left -----
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        h_up = grad_descent_smoothmax(
            h_up,
            betas[:5],
            steps_per_beta=30,
            base_step=0.35,
            rng=rng,
            deadline=deadline,
        )
        h_up = worst_shift_refine(h_up, rng, base_step=0.12, iters=20, deadline=deadline)
        for beta in (50000, 120000, 300000):
            if time.time() > deadline:
                break
            h_up = slsqp_smoothmax_opt(h_up, beta, rng, max_iter=800)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

        # binary SA on up‑sampled version
        h_bin = np.rint(h_up).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n_up

    # ----- fallback (should never trigger) -----
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
