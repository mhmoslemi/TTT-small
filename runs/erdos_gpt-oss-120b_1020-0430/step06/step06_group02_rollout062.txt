```
def compute_corr_delta(h, i, j):
    n = h.size
    a = h[i]
    b = h[j]
    # swapped values: new_i = b, new_j = a
    # delta for each shift s
    delta = np.zeros(2*n-1, dtype=int)  # int because binary
    # s from -(n-1) to n-1
    for s in range(-n+1, n):
        idx = s + n - 1
        # source contribution at index i (if valid)
        # for any s, source exists if i >= max(0, -s) and i < min(n, n - s) i.e., i in [max(0,-s), min(n, n - s) - 1]
        # This condition can be rephrased as 0 <= i + s < n (if s>=0) else 0 <= i < n + s (if s<0). Simpler: check if 0 <= i + s < n.
        if 0 <= i + s < n:
            delta[idx] += (b - a) * (1 - h[i + s])
        # source contribution at index j
        if 0 <= j + s < n:
            delta[idx] += (a - b) * (1 - h[j + s])
        # target contribution where i is the target:
        # term with p = i - s is source index, target index i = p + s. That term exists if 0 <= i - s < n and 0 <= i < n. Equivalent to 0 <= i - s < n and 0 <= i < n always true for i, so just check 0 <= i - s < n.
        if 0 <= i - s < n:
            delta[idx] += h[i - s] * (a - b)  # h[p] * ((1 - new_i) - (1 - old_i)) = h[p] * (old_i - new_i) = h[p]*(a - b)
        # target contribution for j
        if 0 <= j - s < n:
            delta[idx] += h[j - s] * (b - a)
    return delta
```

def annealing_hill_climb(h0, rng, time_limit, best_corr_global):
    n = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = int(corr.max())
    best_h = h.copy()

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # temperature schedule: linearly decreasing from 1.0 to 0.0
        elapsed = time.time() - start
        T = max(0.001, 1.0 - elapsed / time_limit)  # simple linear
        # pick random i (1) and j (0)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        j = rng.choice(zeros)
        delta = compute_corr_delta(h, i, j)
        new_max_corr = int((corr + delta).max())

        accept = False

            if max_corr < best_corr:

                best_h = h.copy()
                if best_corr <= 97:
                    break
        it += 1
    return best_h, best_corr

import time
import numpy as np

def compute_corr(h):
    """Full aperiodic correlation of h with (1-h). Returns int32 array."""
    return np.correlate(h, 1 - h, mode="full").astype(np.int32)


def compute_corr_delta(h, i, j):
    """Delta of full correlation after swapping h[i] (a) and h[j] (b)."""
    n = h.shape[0]
    a = h[i]
    b = h[j]
    one_minus_h = 1 - h
    rev_h = h[::-1]
    delta = np.zeros(2 * n - 1, dtype=np.int32)
    # source contributions
    delta[n - 1 - i : n - 1 - i + n] += (b - a) * one_minus_h
    delta[n - 1 - j : n - 1 - j + n] += (a - b) * one_minus_h
    # target contributions
    delta[i : i + n] += (a - b) * rev_h
    delta[j : j + n] += (b - a) * rev_h
    return delta


def binary_from_continuous(h_cont, rng):
    """Turn a continuous vector into a binary one with exact sum = n/2."""
    n = h_cont.size
    h_bin = (h_cont > 0.5).astype(np.int32)
    target = n // 2
    cur = int(h_bin.sum())
    while cur > target:
        idx = rng.integers(0, n)
        if h_bin[idx] == 1:
            h_bin[idx] = 0
            cur -= 1
    while cur < target:
        idx = rng.integers(0, n)
        if h_bin[idx] == 0:
            h_bin[idx] = 1
            cur += 1
    return h_bin


def annealing_hill_climb(h0, rng, time_limit, best_known_corr):
    """Metropolis annealing on binary vectors with swap moves."""
    n = h0.shape[0]
    h = h0.copy()
    corr = compute_corr(h)
    max_corr = int(corr.max())
    best_h = h.copy()
    best_corr = max_corr
    start = time.time()
    T0 = 5.0                     # initial temperature
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = max(0.001, T0 * (1.0 - elapsed / time_limit))
        # pick a random 1 and a random 0
        i = rng.integers(0, n)
        while h[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] == 1:
            j = rng.integers(0, n)
        delta = compute_corr_delta(h, i, j)
        new_max_corr = int((corr + delta).max())
        delta_c = new_max_corr - max_corr
        accept = False
        if delta_c <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta_c / T):
                accept = True
        if accept:
            h[i], h[j] = h[j], h[i]
            corr += delta
            max_corr = new_max_corr
            if max_corr < best_corr:
                best_corr = max_corr
                best_h = h.copy()

                    break
        if best_corr <= 97:
            break
    return best_h, best_corr
