import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def compute_c5_and_worst(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    max_val = corr[idx]
    k = idx - (n - 1)
    c5 = max_val * (2.0 / n)
    return c5, k, corr

def get_bad_indices(h: np.ndarray, k: int):
    n = h.size
    if k >= 0:
        i = np.arange(n - k)
        mask = (h[i] == 1) & (h[i + k] == 0)
        return i[mask]
    else:
        s = -k
        i = np.arange(s, n)
        mask = (h[i] == 1) & (h[i - s] == 0)
        return i[mask]

def get_good_zero_indices(h: np.ndarray, k: int):
    n = h.size
    zeros = np.where(h == 0)[0]
    if k >= 0:
        non_overlap = zeros[zeros >= n - k]
        overlap = zeros[zeros < n - k]
        good_overlap = overlap[(h[overlap + k] == 1)]
        return np.concatenate((non_overlap, good_overlap))
    else:
        s = -k
        non_overlap = zeros[zeros < s]
        overlap = zeros[zeros >= s]
        good_overlap = overlap[(h[overlap - s] == 1)]
        return np.concatenate((non_overlap, good_overlap))

def directed_swap(h: np.ndarray, k: int, rng: np.random.Generator):
    bad = get_bad_indices(h, k)
    if bad.size == 0:
        return None
    i = rng.choice(bad)
    good = get_good_zero_indices(h, k)
    if good.size == 0:
        return None
    if good.size == 1 and good[0] == i:
        return None
    if i in good:
        good = good[good != i]
        if good.size == 0:
            return None
    j = rng.choice(good)
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return h_new

def random_swap(h: np.ndarray, rng: np.random.Generator):
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return None
    i = rng.choice(ones)
    j = rng.choice(zeros)
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return h_new

def init_random_h(n: int, target: int, rng: np.random.Generator):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, target, replace=False)
    h[idx] = 1.0
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 256
    target = n // 2

    # initialise h, preferring the provided construction if present
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = (h >= 0.5).astype(float)      # binarise
        cur = int(round(h.sum()))
        if cur != target:
            diff = target - cur
            if diff > 0:
                zeros = np.where(h == 0)[0]
                h[rng.choice(zeros, diff, replace=False)] = 1.0
            else:
                ones = np.where(h == 1)[0]
                h[rng.choice(ones, -diff, replace=False)] = 0.0
    else:
        h = init_random_h(n, target, rng)

    c5, k, _ = compute_c5_and_worst(h)
    best_h = h.copy()
    best_c5 = c5

    iter_no_improve = 0
    restart_interval = 5000
    max_total_iters = 2_000_000
    it = 0

    while time.time() - start < budget_s * 0.99 and it < max_total_iters:
        it += 1
        if iter_no_improve >= restart_interval:
            h = init_random_h(n, target, rng)
            c5, k, _ = compute_c5_and_worst(h)
            iter_no_improve = 0
            continue

        c5, k, _ = compute_c5_and_worst(h)

        # directed move
        h_new = directed_swap(h, k, rng)
        improved = False
        if h_new is not None:
            c5_new = compute_c5(h_new)
            if c5_new + 1e-14 < c5:
                h, c5 = h_new, c5_new
                improved = True

        if not improved:

            if h_new is not None:

                if c5_new + 1e-14 < c5:

                else:
                    # simulated annealing acceptance
                    elapsed = time.time() - start
                    T = 0.01 * max(0.0, 1.0 - elapsed / (budget_s * 0.99))

    # final greedy refinement on the best solution

    final_c5 = compute_c5(best_h)
