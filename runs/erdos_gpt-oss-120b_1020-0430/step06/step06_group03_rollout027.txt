def grad_softmax_corr(h, beta):

import time
import numpy as np

def project_onto_box_sum(v, target, max_iter=60):
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_c5(h):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * G, n=N)[:2 * n - 1]
    return corr.max() * (2.0 / n)

def grad_softmax_corr(h, beta):
    n = h.shape[0]
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * G, n=N)[:2 * n - 1]
    max_corr = corr.max()
    w = np.exp(beta * (corr - max_corr))
    w /= w.sum()
    w_pad = np.r_[w, 0.0]
    W = np.fft.rfft(w_pad)
    conv_wh = np.fft.irfft(W * H, n=N)
    w_rev = w[::-1]
    w_rev_pad = np.r_[w_rev, 0.0]
    W_rev = np.fft.rfft(w_rev_pad)
    conv_wrev_h = np.fft.irfft(W_rev * H, n=N)
    start = n - 1
    A = conv_wh[start:start + n]
    B = conv_wrev_h[start:start + n]
    return (1.0 - A - B) / beta

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c
