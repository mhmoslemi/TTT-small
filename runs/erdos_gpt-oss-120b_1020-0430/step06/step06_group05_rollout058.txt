import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng):
    # similar as before 
    diff = target - h.sum()
    n = h.size
    # while loop reduces diff
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h

def softmax_grad(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    max_O = np.max(O)
    # stabilize: subtract max_O for exponentials
    exp_vals = np.exp(beta * (O - max_O))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp  # length 2n-1
    # compute convs
    conv2 = np.convolve(w, h, mode='full')
    conv1 = np.convolve(w[::-1], h, mode='full')

    G1 = conv1[offset:offset + n]  # sum_k w_k * h[i + k]
    grad = 1.0 - (G1 + G2)
    return grad, O

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, atol: float = 1e-12) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > atol:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def softmax_grad(h: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max_k O_k."""
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")          # length 2n‑1
    max_O = np.max(O)
    exp_vals = np.exp(beta * (O - max_O))
    w = exp_vals / np.sum(exp_vals)                   # normalized weights, length 2n‑1

    conv2 = np.convolve(w, h, mode="full")            # w * h
    conv1 = np.convolve(w[::-1], h, mode="full")      # w_rev * h
    off = n - 1
    G2 = conv2[off:off + n]                           # Σ w_k h[i‑k]
    G1 = conv1[off:off + n]                           # Σ w_k h[i+k]
    grad = 1.0 - (G1 + G2)                            # ∂/∂h_i of Σ w_k O_k
    return grad, O


def worst_shift_and_grad(h: np.ndarray):
    """Sub‑gradient for the current worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)   # shift giving max O_k (can be negative)
    k = idx
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        kp = -k
        grad[kp:] -= h[: n - kp]
        grad[: n - kp] -= h[kp:]
    return k, grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    n = 512
    target = n / 2.0

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
    else:
        h = rng.random(n)
        h = enforce_sum(h, target, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- soft‑max gradient descent ------------------------------------
    betas = np.logspace(0, np.log10(20000), num=12)   # 1 … 20000
    total_iters = 0
    for beta in betas:
        inner = int(2000 + (beta / betas[-1]) * 10000)   # up to ~12000 steps
        for it in range(inner):
            total_iters += 1
            if time.time() - start > budget_s * 0.98:
                break

            grad, _ = softmax_grad(h, beta)
            lr = 0.12 / (np.sqrt(total_iters) + 1.0)   # diminishing step
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            diff = target - h_new.sum()
            if abs(diff) > 1e-12:
                h_new = enforce_sum(h_new, target, rng)

            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            # accept if improvement, otherwise keep old h
            if c_new < compute_c5(h):
                h = h_new
            else:
                # occasional random shake to escape plateaus
                if rng.random() < 0.008:
                    i, j = rng.integers(0, n, size=2)
                    if i != j:
                        max_eps = min(h[i], 1.0 - h[j])
                        if max_eps > 0:
                            eps = max_eps * rng.random() * 0.05
                            h[i] -= eps
                            h[j] += eps
                            h = enforce_sum(h, target, rng)

        if time.time() - start > budget_s * 0.98:
            break

    # ----- sub‑gradient descent on the actual max -----------------------
    sub_iters = 20000
    h_sub = best_h.copy()
    c_sub = best_c
    for it in range(sub_iters):
        if time.time() - start > budget_s * 0.99:
            break
        k, grad = worst_shift_and_grad(h_sub)
        step = 0.5 / np.sqrt(it + 1.0)
        h_new = h_sub - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            h_new = enforce_sum(h_new, target, rng)

        c_new = compute_c5(h_new)
        if c_new < c_sub:
            c_sub = c_new
            h_sub = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        else:
            # keep current h_sub
            pass

    # ----- final sanity enforcement ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
