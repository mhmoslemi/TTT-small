import time
import numpy as np

def project_onto_box_sum(v, target):
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_c5_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    fh = np.fft.rfft(h, L)
    f1 = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(fh * np.conj(f1), L)
    return corr

def gradient_for_shift(k, h):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        t = -k
        grad[t:] += 1.0 - h[: n - t]
        grad[: n - t] -= h[t:]
    return grad

def worst_gradient(h):
    n = h.shape[0]
    corr = compute_c5_fft(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)
    return gradient_for_shift(k, h)

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh
