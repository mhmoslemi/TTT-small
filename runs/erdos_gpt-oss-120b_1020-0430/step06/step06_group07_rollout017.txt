def softmax_grad(h, beta, probs):
    n = h.size
    grad = np.zeros_like(h)
    # Probabilities for positive shifts (including zero to n-1) and negative shifts (then rest)
    # corr length = 2n-1, indices 0..2n-2, shift = idx - (n-1)
    # Let's separate probs into w_pos (size n) and w_neg (size n-1)
    w_pos = probs[n-1:]  # shifts 0..n-1
    w_neg = probs[:n-1][::-1]  # shifts -1..-(n-1) (reversed)
    # Positive part
    for s in range(n):
        w = w_pos[s]
        if w == 0.0:  # skip
            continue
        # part1: (1 - h[i+s]) for i=0..n-1-s
        grad[:n-s] += w * (1 - h[s:])
        # part2: -h[i] for i=s..n-1
        grad[s:] -= w * h[:n-s]
    # Negative part (t from 1..n-1)
    for t in range(1, n):
        w = w_neg[t-1]  # corresponds to shift -t
        if w == 0.0:
            continue
        # part1: (1 - h[i - t]) for i = t..n-1
        grad[t:] += w * (1 - h[:n-t])
        # part2: -h[i] for i = 0..n-1-t
        grad[:n-t] -= w * h[t:]
    return grad

import time, numpy as np

def project_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n, rng):
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_and_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # part1
        grad[:n-s] += 1 - h[s:]
        # part2
        grad[s:] -= h[:n-s]
    else:
        t = -shift
        grad[t:] += 1 - h[:n-t]
        grad[:n-t] -= h[t:]
    return shift, grad, corr[idx]

def subgradient_optimize(h0, target, time_budget):
    start = time.time()
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        _, grad, _ = worst_shift_and_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c

def softmax_obj_and_grad(h, beta):
    # compute corr
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(beta * corr)
    # For numeric stability:
    exp_term = np.exp(beta * corr - m)
    Z = exp_term.sum()
    value = (np.log(Z) + m) / beta
    probs = exp_term / Z  # length 2n-1
    # gradient
    n = h.size
    grad = np.zeros_like(h)
    # split probs
    w_pos = probs[n-1:]  # shifts 0..n-1
    w_neg = probs[:n-1][::-1]  # shifts -1..-(n-1)
    # positive shifts
    for s in range(n):
        w = w_pos[s]
        if w == 0.0:
            continue
        # (1 - h[i+s]) for i = 0..n-1-s
        grad[:n-s] += w * (1 - h[s:])
        # -h[i] for i = s..n-1
        grad[s:] -= w * h[:n-s]
    # negative shifts
    for t in range(1, n):
        w = w_neg[t-1]
        if w == 0.0:
            continue
        grad[t:] += w * (1 - h[:n-t])
        grad[:n-t] -= w * h[t:]
    return value, grad

def projected_gradient_descent_softmax(h0, target, beta, max_iters, time_budget, rng):
    start = time.time()
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    t = 0
    # step size schedule: start with lr = 0.5 maybe
    lr0 = 0.5
    while t < max_iters and (time.time() - start) < time_budget:
        t += 1
        value, grad = softmax_obj_and_grad(h, beta)
        # step

            # reduce time if reached target

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        shift, grad, _ = worst_shift_and_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:

            i = rng.integers(0, h.size)
            j = rng.integers(0, h.size)
            max_eps = min(h[i], 1.0 - h[j])

            best_c, best_h, h = c, nh.copy(), nh

            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

def upsample(h, rng):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)

import time
import numpy as np

def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_shift_and_gradient(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        grad[: n - s] += 1 - h[s:]
        grad[s:] -= h[: n - s]
    else:
        t = -shift
        grad[t:] += 1 - h[: n - t]
        grad[: n - t] -= h[t:]
    return shift, grad, corr[idx]

def subgradient_optimize(h0: np.ndarray, target: float, time_budget: float):
    start = time.time()
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        _, grad, _ = worst_shift_and_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c

def softmax_from_corr(corr: np.ndarray, h: np.ndarray, beta: float):
    m = np.max(beta * corr)
    exp_term = np.exp(beta * corr - m)
    Z = exp_term.sum()
    value = (np.log(Z) + m) / beta
    probs = exp_term / Z
    n = h.size
    grad = np.zeros_like(h)
    w_pos = probs[n - 1 :]        # shifts 0 .. n-1
    w_neg = probs[: n - 1][::-1]  # shifts -1 .. -(n-1)
    for s in range(n):
        w = w_pos[s]
        if w == 0.0:
            continue
        grad[: n - s] += w * (1 - h[s:])
        grad[s:] -= w * h[: n - s]
    for t in range(1, n):
        w = w_neg[t - 1]
        if w == 0.0:
            continue
        grad[t:] += w * (1 - h[: n - t])
        grad[: n - t] -= w * h[t:]
    return value, grad

def projected_gradient_descent_softmax(h0: np.ndarray, target: float,
                                      beta: float, max_iters: int,
                                      time_budget: float):
    start = time.time()
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    lr0 = 0.5
    t = 0
    while t < max_iters and (time.time() - start) < time_budget:
        t += 1
        corr = np.correlate(h, 1 - h, mode="full")
        c = corr[np.argmax(corr)] * (2.0 / h.size)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= 0.3808:
                break
        _, grad = softmax_from_corr(corr, h, beta)
        lr = lr0 / np.sqrt(t + 10)
        h = project_box_sum(h - lr * grad, target)
    final_c = compute_c5(h)
    if final_c < best_c:
        best_h, best_c = h, final_c
    return best_h, best_c

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        _, grad, _ = worst_shift_and_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            i = rng.integers(0, h.size)
            j = rng.integers(0, h.size)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

def upsample(h: np.ndarray, rng: np.random.Generator):
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    return project_box_sum(h2, n2 / 2.0)
