def gradient_refine(h, rng, step=0.02, max_iter=30):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr) - (n-1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1 - 2*h
        else:
            # g[i] = (1 - h[i+k]) - h[i] for i < n-k
            g[:n-k] = 1 - h[k:]
            g[k:] -= h[:n-k]
        # line search: compute best step in [ -step, step]
        # We'll try step * (0.5**it) to shrink.
        s = step * (0.5**it)
        h_new = h - s * g
        # clip
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # enforce sum
        diff = (n/2.0) - h_new.sum()
        if abs(diff) > 1e-12:
            # redistribute diff
            # add to random entries up to capacity
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h_new[i])
                else:
                    add = max(diff, -h_new[i])
                h_new[i] += add
                diff = (n/2.0) - h_new.sum()
        h = h_new
    return h
```

def enforce_sum(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h

def binary_sa(n, rng, time_budget, h0=None):
    start = time.time()
    # init binary h
    if h0 is None:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n//2, replace=False)
        h[ones] = 1.0
    else:
        # round to nearest binary
        h = np.rint(h0).astype(float)
        # ensure exact n/2 ones
        target = n // 2
        current = int(h.sum())
        if current > target:
            # too many ones
            ones_idx = np.where(h == 1.0)[0]
            drop = rng.choice(ones_idx, current - target, replace=False)
            h[drop] = 0.0
        elif current < target:
            zeros_idx = np.where(h == 0.0)[0]
            add = rng.choice(zeros_idx, target - current, replace=False)
            h[add] = 1.0
    # maintain lists
    ones_arr = np.where(h == 1.0)[0]
    zeros_arr = np.where(h == 0.0)[0]
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    # SA loop
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(1e-8, 0.5 * (1 - elapsed / time_budget))  # linear cooling
        # pick random swap
        if len(ones_arr) == 0 or len(zeros_arr) == 0:
            break
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = ones_arr[i_pos]
        j = zeros_arr[j_pos]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            c_curr = c_new
            # update arrays
            ones_arr[i_pos] = j
            zeros_arr[j_pos] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # reject revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

import time
import numpy as np

# -------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def binary_sa(n: int, rng: np.random.Generator, time_budget: float, h0: np.ndarray = None):
    """Simulated annealing on the {0,1} hypercube with exact sum n/2."""
    start = time.time()
    # initialise binary vector
    if h0 is None:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
    else:
        h = np.rint(h0).astype(float)
        target = n // 2
        cur = int(h.sum())
        if cur > target:
            ones = np.where(h == 1.0)[0]
            drop = rng.choice(ones, cur - target, replace=False)
            h[drop] = 0.0
        elif cur < target:
            zeros = np.where(h == 0.0)[0]
            add = rng.choice(zeros, target - cur, replace=False)
            h[add] = 1.0
    ones_arr = np.where(h == 1.0)[0]
    zeros_arr = np.where(h == 0.0)[0]
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr

    T0, T_end = 1.0, 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # random swap
        if len(ones_arr) == 0 or len(zeros_arr) == 0:
            break
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = ones_arr[i_pos]
        j = zeros_arr[j_pos]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            ones_arr[i_pos] = j
            zeros_arr[j_pos] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 30) -> np.ndarray:
    """Gradient descent on the worst shift with diminishing steps."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        s = step * (0.5 ** it)
        h_new = h - s * g
        np.clip(h_new, 0.0, 1.0, out=h_new)
        diff = (n / 2.0) - h_new.sum()
        if abs(diff) > 1e-12:
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                add = min(diff, 1.0 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
                h_new[i] += add
                diff = (n / 2.0) - h_new.sum()
        h = h_new
    return h


# -------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # ----- initial solution -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0.copy()
        best_c = compute_c5(best_h)
    else:
        n0 = 512
        h0 = np.zeros(n0, dtype=float)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0
        best_h = h0
        best_c = compute_c5(best_h)

    # ----- Phase 1: binary SA on current resolution -------------------------
    remaining = total - (time.time() - start)
    if remaining > 15.0:
        t_phase = min(remaining * 0.25, 300.0)
        h_sa, c_sa = binary_sa(best_h.size, rng, t_phase, h0=best_h)
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa

    # ----- Phase 2: upsample to 1024 ---------------------------------------
    remaining = total - (time.time() - start)
    if remaining > 10.0 and best_h.size < 1024:
        h_up = upsample_h(best_h, 1024)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        best_h = h_up
        best_c = compute_c5(best_h)

    # ----- Phase 3: binary SA on 1024 points --------------------------------
    remaining = total - (time.time() - start)
    if remaining > 20.0:
        t_phase = min(remaining * 0.35, 500.0)
        h_sa2, c_sa2 = binary_sa(1024, rng, t_phase, h0=best_h)
        if c_sa2 < best_c:
            best_h, best_c = h_sa2.copy(), c_sa2

    # ----- Phase 4: gradient refinement -------------------------------------
    remaining = total - (time.time() - start)
    if remaining > 5.0:
        h_ref = gradient_refine(best_h, rng, step=0.02, max_iter=30)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_h, best_c = h_ref.copy(), c_ref

    # ----- Final sanitisation ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
