def binary_hill_climb(h, rng, max_iters, patience):
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    indices_one = np.flatnonzero(h)
    indices_zero = np.flatnonzero(1 - h)
    no_improve = 0
    for it in range(max_iters):
        # compute worst gradient
        grad = worst_gradient(h)  # uses same implementation for any h value
        # choose i among ones with max grad
        i = indices_one[np.argmax(grad[indices_one])]
        # choose j among zeros with min grad
        j = indices_zero[np.argmin(grad[indices_zero])]
        if i == j:
            # fallback random
            i = rng.choice(indices_one)
            j = rng.choice(indices_zero)
        # swap
        h[i], h[j] = 0, 1
        # update indices sets
        # (swap values)
        # Actually after swap, i becomes zero, j becomes one. So we need to adjust arrays:
        # simply recompute indices each iteration? That is O(n) each, okay for n=256 and few iterations.
        new_c = compute_c5(h)
        if new_c < cur_c - 1e-12:
            cur_c = new_c
            indices_one = np.flatnonzero(h)
            indices_zero = np.flatnonzero(1 - h)
            no_improve = 0
        else:
            # revert
            h[i], h[j] = 1, 0
            no_improve += 1
            if no_improve >= patience:
                # break if stuck
                break
    return h, cur_c

def reduce_max_deficit(h, target_deficit, rng, max_iter, time_end):
    n = h.size
    target = n // 2
    for it in range(max_iter):
        if time.time() > time_end:
            break
        # compute O and deficit
        O = np.correlate(h, h, mode='full')
        min_O = O.min()
        deficit_max = target - min_O
        if deficit_max <= target_deficit:
            break
        shift_idx = np.argmin(O) - (n - 1)  # signed shift where min O occurs
        # generate candidates
        if shift_idx >= 0:
            k = shift_idx
            i_range = np.arange(n - k)
            i_cand = i_range[(h[i_range] == 1) & (h[i_range + k] == 0)]
            j_cand = i_range[(h[i_range] == 0) & (h[i_range + k] == 1)]
        else:
            k = -shift_idx
            i_range = np.arange(k, n)
            i_cand = i_range[(h[i_range] == 1) & (h[i_range - k] == 0)]
            j_cand = i_range[(h[i_range] == 0) & (h[i_range - k] == 1)]
        if i_cand.size == 0 or j_cand.size == 0:
            break
        i = rng.choice(i_cand)
        j = rng.choice(j_cand)
        # swap
        h[i], h[j] = 0.0, 1.0
    # final deficit stats
    O = np.correlate(h, h, mode='full')
    min_O = O.min()
    deficit_max = target - min_O
    return h, deficit_max

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (maximal) overlap term w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def _max_deficit_stats(h: np.ndarray):
    """Return (deficit_max, shift_of_max_deficit) for binary h."""
    n = h.size
    target = n // 2
    O = np.correlate(h, h, mode="full")
    min_O = O.min()
    deficit = target - min_O
    shift = np.argmin(O) - (n - 1)
    return deficit, shift

def _reduce_max_deficit(h: np.ndarray, target_deficit: int,
                       rng: np.random.Generator, max_iter: int,
                       time_end: float) -> (np.ndarray, int):
    """Try to lower the maximal deficit to target_deficit using directed swaps."""
    n = h.size
    for _ in range(max_iter):
        if time.time() >= time_end:
            break
        deficit, shift = _max_deficit_stats(h)
        if deficit <= target_deficit:
            break
        if shift >= 0:
            k = shift
            rngange = np.arange(n - k)
            i_cand = rngange[(h[rngange] == 1) & (h[rngange + k] == 0)]
            j_cand = rngange[(h[rngange] == 0) & (h[rngange + k] == 1)]
        else:
            k = -shift
            rngange = np.arange(k, n)
            i_cand = rngange[(h[rngange] == 1) & (h[rngange - k] == 0)]
            j_cand = rngange[(h[rngange] == 0) & (h[rngange - k] == 1)]
        if i_cand.size == 0 or j_cand.size == 0:
            break
        i = rng.choice(i_cand)
        j = rng.choice(j_cand)
        h[i], h[j] = 0.0, 1.0
    deficit, _ = _max_deficit_stats(h)
    return h, deficit

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a low C5 value."""
    rng = np.random.default_rng(seed)
    start = time.time()
    time_limit = start + budget_s * 0.95  # reserve a small margin

    n = 512
    target = n // 2

    best_h = None
    best_c = float("inf")

    # start from the provided construction if available
    if "initial_h_values" in globals():
        hv = np.array(initial_h_values, dtype=float)
        bin_h = (hv >= 0.5).astype(float)
        diff = int(target - bin_h.sum())
        if diff > 0:
            zero_idx = np.flatnonzero(bin_h == 0)
            bin_h[rng.choice(zero_idx, diff, replace=False)] = 1.0
        elif diff < 0:
            one_idx = np.flatnonzero(bin_h == 1)
            bin_h[rng.choice(one_idx, -diff, replace=False)] = 0.0
        c0 = compute_c5(bin_h)
        if c0 < best_c:
            best_c = c0
            best_h = bin_h.copy()

    # main stochastic search
    while time.time() < time_limit:
        # random binary start
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, target, replace=False)] = 1.0
        cur_c = compute_c5(h)

        # guided hill climbing with swaps
        patience = 200
        no_improve = 0
        max_iters = 4000
        for _ in range(max_iters):
            if time.time() >= time_limit:
                break
            grad = worst_gradient(h)
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)
            if ones.size == 0 or zeros.size == 0:
                break
            i = ones[np.argmax(grad[ones])]
            j = zeros[np.argmin(grad[zeros])]
            h[i], h[j] = 0.0, 1.0
            new_c = compute_c5(h)
            if new_c < cur_c - 1e-12:
                cur_c = new_c
                no_improve = 0
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
                no_improve += 1
                if no_improve >= patience:
                    break

        # attempt directed reduction of the worst deficit
        deficit_target = 97  # corresponds to C5 = 97/256 ≈ 0.3789
        if cur_c > (deficit_target * 2.0 / n):
            h, final_deficit = _reduce_max_deficit(
                h,
                deficit_target,
                rng,
                max_iter=2000,
                time_end=time_limit,
            )
            cur_c = final_deficit * 2.0 / n
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            if final_deficit <= deficit_target:
                # reached the target; we can stop early
                time_limit = time.time() + 0.01  # break outer loop soon
                break

        # occasional random perturbation to escape local minima
        if time.time() < time_limit:
            for _ in range(5):
                i = rng.choice(np.flatnonzero(h))
                j = rng.choice(np.flatnonzero(1 - h))
                h[i], h[j] = 0.0, 1.0
                new_c = compute_c5(h)
                if new_c < cur_c:
                    cur_c = new_c
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                else:
                    h[i], h[j] = 1.0, 0.0

    # fallback in case nothing was found
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[:target] = 1.0
        best_c = compute_c5(best_h)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = int(round(best_h.sum()))
    if cur_sum != target:
        diff = target - cur_sum
        if diff > 0:
            zero_idx = np.flatnonzero(best_h == 0)
            best_h[rng.choice(zero_idx, diff, replace=False)] = 1.0
        elif diff < 0:
            one_idx = np.flatnonzero(best_h == 1)
            best_h[rng.choice(one_idx, -diff, replace=False)] = 0.0

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
