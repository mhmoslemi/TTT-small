def compute_c5(h):
    n = h.size
    # compute correlation via FFT
    # cor[i] = sum_j h[j] * (1 - h[(j+i) mod n])? Wait correlation is linear shift not circular; we need linear correlation, not cyclic. Using FFT we can compute linear convolution with zero padding.
    a = h
    b = 1.0 - h
    # linear correlation = convolution of a and reversed b
    # conv_len = 2*n -1
    # Use FFT size = next pow2 of conv_len.
    n_fft = 1 << (conv_len - 1).bit_length()
    # Compute FFT
    fa = np.fft.rfft(a, n=n_fft)
    fb = np.fft.rfft(b[::-1], n=n_fft) # reversed b
    conv = np.fft.irfft(fa*fb, n=n_fft)[:2*n-1]
    # correlation values correspond to conv indices i = (n-1)+shift
    # So max over conv
    max_val = np.max(conv)
    return float(max_val * (2.0/n))

    n = h.size
    # v via np.correlate
    v = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(beta * v)

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):

    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300, deadline=None):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    # add time control: use callback to check time.
    # SciPy minimize does not accept a callback with deadline directly. Use options maxiter and check time after.
    # We'll just call minimize, but ensure we limit max_iter small.
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # fix sum exactly
    adjust_sum_exact(x, rng)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    """Maximum overlap integral (discrete)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Force sum(h) == n/2 by moving mass on random coordinates."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- gradients ----------
def gradient_smoothmax(h, beta):
    """Gradient of (1/β) log-sum-exp of overlaps."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h)
    one_minus_h = 1.0 - h
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += wi * one_minus_h[shift:]
            g[shift:]     -= wi * h[: n - shift]
        else:
            ii = -shift
            g[ii:]        += wi * one_minus_h[: n - ii]
            g[: n - ii]   -= wi * h[ii:]
    return g

def worst_shift_gradient(h):
    """Sub‑gradient for the current worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)
        if shift >= 0:
            g[: n - shift] += 1.0 - h[shift:]
            g[shift:]      -= h[: n - shift]
        else:
            ii = -shift
            g[ii:]         += 1.0 - h[: n - ii]
            g[: n - ii]    -= h[ii:]
    g /= max(1, len(idxs))
    return g

# ---------- descent loops ----------
def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    n = h.size
    for beta in betas:
        step = base_step
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 2.0):
                cand = h - step * mult * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
            step *= 0.98
    return h

def worst_shift_refine(h, rng, base_step, iters, deadline):
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = worst_shift_gradient(h)
        if np.linalg.norm(g) < 1e-14:
            break
        best_h = h
        best_c = compute_c5(h)
        for mult in (1.0, 0.5, 2.0):
            cand = h - step * mult * g
            np.clip(cand, 0.0, 1.0, out=cand)
            adjust_sum_exact(cand, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand
        h = best_h
        step *= 0.97
    return h

# ---------- SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300, deadline=None):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

# ---------- binary hill climb ----------
def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 12000:
            break
    return best_h, best_c

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):

    # base resolution
    n_base = 1080
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
        xt = np.linspace(0.0, 2.0, n_base, endpoint=False)
        parent_interp = np.interp(xt, xs, parent)
    else:
        parent_interp = None

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------ restarts ------------
    restart_kinds = ["parent", "random", "block", "binary"]
    for r in range(5):
        if time.time() > deadline:
            break
        n = n_base
        kind = restart_kinds[r % len(restart_kinds)]
        if kind == "parent" and parent_interp is not None:
            h = parent_interp.copy()
        elif kind == "block":
            h = np.zeros(n)
            h[: n // 2] = 1.0
        elif kind == "binary":
            h = np.zeros(n)
            ones = rng.choice(n, n // 2, replace=False)
            h[ones] = 1.0
        else:  # random
            h = rng.random(n)
            h = h / h.sum() * (n / 2.0)
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        # smooth‑max gradient descent
        betas = [5, 20, 80, 200, 500, 2000, 8000, 30000, 100000, 300000]

        # worst‑shift refinement
        h = worst_shift_refine(h, rng, base_step=0.15, iters=2000, deadline=deadline)

        # SLSQP polish (high β)
        for beta in (300000, 100000, 30000):
            if time.time() > deadline:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=400, deadline=deadline)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

        # binary hill‑climb (optional)
        h_bin = np.rint(h).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.15)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n

    # ------------ up‑sample if time permits ------------
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        # brief refinement on up‑sampled version
        h_up = gradient_descent_smoothmax(
            h_up, betas[:6], base_step=0.4, it_per_beta=40, rng=rng, deadline=deadline
        )
        h_up = worst_shift_refine(h_up, rng, base_step=0.12, iters=800, deadline=deadline)
        for beta in (300000, 100000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=300, deadline=deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up
        # binary polish on up‑sampled
        h_bin = np.rint(h_up).astype(float)
        adjust_sum_exact(h_bin, rng)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.1)
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
        if c_bin < best_c:

    # final safety
    if best_h is None:

        best_h = np.zeros(n)

        adjust_sum_exact(best_h, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
