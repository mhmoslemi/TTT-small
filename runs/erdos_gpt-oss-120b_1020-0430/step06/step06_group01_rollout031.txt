def binary_sa(n, rng, time_budget):
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n//2, replace=False)
    zeros = np.setdiff1d(np.arange(n), ones, assume_unique=True)
    h[ones] = 1.0
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr

    pos_one = np.empty(n, dtype=int)
    pos_zero = np.empty(n, dtype=int)
    for idx, v in enumerate(ones):
        pos_one[v] = idx
    for idx, v in enumerate(zeros):
        pos_zero[v] = idx

    T0 = 0.2
    T_end = 1e-6

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # pick random indices
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = ones[idx_i]
        j = zeros[idx_j]

        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr

        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            # accept
            c_curr = c_new
            # update lists
            ones[idx_i] = j
            zeros[idx_j] = i
            pos_one[j] = idx_i
            pos_zero[i] = idx_j
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    break
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap at the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # shift index (negative means left)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ---------- guided hill climbing ----------
def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 20000, step_scale: float = 0.9):
    """Gradient‑directed hill climbing with adaptive step size."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.01, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 1200) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- binary simulated annealing ----------
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    zeros = np.setdiff1d(np.arange(n, dtype=int), ones, assume_unique=True)
    h[ones] = 1.0
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr

    pos_one = np.empty(n, dtype=int)
    pos_zero = np.empty(n, dtype=int)
    for idx, v in enumerate(ones):
        pos_one[v] = idx
    for idx, v in enumerate(zeros):
        pos_zero[v] = idx

    T0, T_end = 0.2, 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)

        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = ones[idx_i]
        j = zeros[idx_j]

        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            ones[idx_i] = j
            zeros[idx_j] = i
            pos_one[j] = idx_i
            pos_zero[i] = idx_j
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    break
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# ---------- continuous optimisation pipeline ----------
def continuous_optimize(n, rng, time_budget, init_h=None):
    start = time.time()
    best_h, best_c = None, np.inf

    # initial vector
    h = init_h.copy() if init_h is not None else rng.random(n)
    h = enforce_sum(h, rng)

    # aggressive hill‑climb
    max_iters = int(min(30000, (time_budget) * 400))
    h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.9)
    best_h, best_c = h, c

    # soft‑max SLSQP with increasing β
    beta_seq = (20., 80., 200., 800., 3000., 8000., 20000.)
    for beta in beta_seq:
        if time.time() - start > time_budget * 0.96:
            break
        max_iter = 2000 if beta >= 200 else 1000
        h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # final gradient tidy‑up
    if time.time() - start < time_budget * 0.98:
        h = gradient_refine(best_h, rng, step=0.005, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # short random restarts
    while time.time() - start < time_budget * 0.995:
        if best_h is not None and rng.random() < 0.5:
            h = best_h.copy()
            for _ in range(8):
                i = rng.integers(0, n)
                delta = (rng.random() - 0.5) * 0.25
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = rng.random(n)
            h = enforce_sum(h, rng)

        # brief hill‑climb
        max_iters = int(min(8000, (time_budget - (time.time() - start)) * 250))
        h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.9)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # quick SLSQP passes
        for beta in (20., 80., 200.):
            if time.time() - start > time_budget * 0.995:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=500)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
    return best_h, best_c


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    n = 512  # work dimension

    # time allocation
    binary_time = total_budget * 0.55
    cont_time = total_budget * 0.38
    final_time = total_budget - binary_time - cont_time

    # binary simulated annealing
    h_bin, c_bin = binary_sa(n, rng, binary_time)

    best_h, best_c, best_n = h_bin, c_bin, n

    # optional initial_h_values handling
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        if c0 < best_c:
            best_h, best_c, best_n = h0, c0, h0.size

    # continuous refinement
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > cont_time:
        remaining = cont_time
    if remaining > 0.0:
        h_cont, c_cont = continuous_optimize(n, rng, remaining, init_h=best_h)
        if c_cont < best_c:
            best_h, best_c, best_n = h_cont, c_cont, n

    # final stochastic tweaking
    while time.time() - start < total_budget - 0.001:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps > 0:
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # sanitise output
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
