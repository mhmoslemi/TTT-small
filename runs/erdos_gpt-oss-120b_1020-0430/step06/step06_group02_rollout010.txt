def proj_simplex_cap(v, target):
    # Clip
    v = np.clip(v, 0.0, 1.0)
    # If sum close to target, done
    s = v.sum()
    if abs(s - target) <= 1e-12:
        return v
    # Use bisection
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def compute_O_and_grad(h, beta, L, ones_fft):
    n = h.size
    # FFT of h and h_rev
    h_fft = np.fft.fft(h, L)
    h_rev = h[::-1]
    h_rev_fft = np.fft.fft(h_rev, L)
    # rev_one_minus_h_fft = ones_fft - h_rev_fft
    rev_one_minus_h_fft = ones_fft - h_rev_fft
    O_full = np.real(np.fft.ifft(h_fft * rev_one_minus_h_fft))[:2*n-1]
    # Softmax weights
    z = beta * O_full
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()
    # w_fft
    w_fft = np.fft.fft(w, L)
    # conv_w_h
    conv_w_h = np.real(np.fft.ifft(w_fft * h_fft))[:3*n-2]
    # conv_w_rev_h
    conv_w_rev_h = np.real(np.fft.ifft(w_fft * h_rev_fft))[:3*n-2]
    # conv2 = conv_w_h[(n-1):(n-1+n)]
    conv2 = conv_w_h[(n-1):(n-1+n)]
    # conv1 = conv_w_rev_h[(2*n - 2) - np.arange(n)]
    idx = (2*n - 2) - np.arange(n)
    conv1 = conv_w_rev_h[idx]
    grad = np.ones(n) - conv1 - conv2
    return O_full, grad

def proj_simplex_cap(v, target):
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= 1e-12:
        return v
    # bisection for lambda
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust small residual
    residual = target - w.sum()
    if abs(residual) > 1e-12:
        # Find an index where w not at bound to add residual
        idx = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if idx.size > 0:
            i = idx[0]
            w[i] = np.clip(w[i] + residual, 0.0, 1.0)
        else:
            # fallback: add to any index not at 1
            i = np.argmax(w)
            w[i] = np.clip(w[i] + residual, 0.0, 1.0)
    return w
```

import time
import numpy as np

# ---------- utility functions -------------------------------------------------

def proj_simplex_cap(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {0<=x<=1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= 1e-12:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target - w.sum()
    if abs(residual) > 1e-12:
        # try to distribute the tiny residual
        free = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if free.size:
            i = free[0]
        else:
            i = np.argmax(w)
        w[i] = np.clip(w[i] + residual, 0.0, 1.0)
    return w


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence (0/1)."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


def resample(arr: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of arr (defined on [0,2]) to n points."""
    n0 = arr.size
    t0 = np.linspace(0, 2, n0, endpoint=False)
    t1 = np.linspace(0, 2, n, endpoint=False)
    return np.interp(t1, t0, arr)


def compute_O_full(h: np.ndarray, L: int, ones_fft: np.ndarray) -> np.ndarray:
    """Full correlation O_k = sum h_i (1‑h_{i+k}) for k=-(n-1)…(n-1)."""
    n = h.size
    h_fft = np.fft.fft(h, L)
    h_rev_fft = np.fft.fft(h[::-1], L)
    rev_one_minus_h_fft = ones_fft - h_rev_fft
    O_full = np.real(np.fft.ifft(h_fft * rev_one_minus_h_fft))[: 2 * n - 1]
    return O_full


def compute_c5_from_O(O_full: np.ndarray, n: int) -> float:
    """C5 given O_k values."""
    return np.max(O_full) * (2.0 / n)


def compute_O_and_grad(h: np.ndarray, beta: float, L: int, ones_fft: np.ndarray):
    """Soft‑max weights, O_k values and gradient of the smoothed max."""
    n = h.size
    # FFTs of h and its reverse
    h_fft = np.fft.fft(h, L)
    h_rev_fft = np.fft.fft(h[::-1], L)

    # O_k (full correlation)
    rev_one_minus_h_fft = ones_fft - h_rev_fft
    O_full = np.real(np.fft.ifft(h_fft * rev_one_minus_h_fft))[: 2 * n - 1]

    # soft‑max weights
    z = beta * O_full
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()

    # FFT of weights
    w_fft = np.fft.fft(w, L)

    # Convolutions needed for gradient
    conv_w_h = np.real(np.fft.ifft(w_fft * h_fft))[: 3 * n - 2]
    conv_w_rev_h = np.real(np.fft.ifft(w_fft * h_rev_fft))[: 3 * n - 2]

    # conv2 = sum_k w_k * h_{j‑k}
    conv2 = conv_w_h[(n - 1) : (n - 1 + n)]

    # conv1 = sum_k w_k * h_{j+k}
    idx = (2 * n - 2) - np.arange(n)
    conv1 = conv_w_rev_h[idx]

    grad = np.ones(n) - conv1 - conv2
    return O_full, grad


# ---------- main routine -----------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # resolution (choose a power‑of‑two‑ish size for FFT efficiency)
    n = 768                     # 2 · 384, fits easily in the time budget
    target = n / 2.0
    L = 1 << ((3 * n - 2).bit_length())      # next power of two ≥ 3n‑2
    ones_fft = np.zeros(L, dtype=complex)
    ones_fft[0] = n

    # ---- generate several start vectors ------------------------------------
    candidates = []

    if "initial_h_values" in globals():
        h0 = resample(np.array(initial_h_values, dtype=float), n)
        h0 = proj_simplex_cap(h0, target)
        O0 = compute_O_full(h0, L, ones_fft)
        c0 = compute_c5_from_O(O0, n)
        candidates.append((h0, c0))

    # random balanced binary
    rand_bin = np.zeros(n)
    rand_bin[rng.choice(n, n // 2, replace=False)] = 1.0
    rand_bin = proj_simplex_cap(rand_bin, target)
    O1 = compute_O_full(rand_bin, L, ones_fft)
    c1 = compute_c5_from_O(O1, n)
    candidates.append((rand_bin, c1))

    # thue‑Morse up‑sampled (take 1024 bits, then slice)
    tm = thue_morse_bits(1024)[:n]
    tm = proj_simplex_cap(tm, target)
    O2 = compute_O_full(tm, L, ones_fft)
    c2 = compute_c5_from_O(O2, n)
    candidates.append((tm, c2))

    # pick the best seed
    best_h, best_c = min(candidates, key=lambda t: t[1])

    # ---- progressive soft‑max optimisation (Adam with projection) ---------
    betas = [1.0, 5.0, 20.0, 80.0, 320.0, 1280.0, 5120.0, 20480.0]
    beta_index = 0

    # Adam state
    m = np.zeros_like(best_h)
    v = np.zeros_like(best_h)

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break

        beta = betas[beta_index]
        # step size heuristics: smaller for larger beta
        step = 0.6 / (np.sqrt(beta) + 1.0)

        # one Adam update
        O_full, grad = compute_O_and_grad(best_h, beta, L, ones_fft)
        c = compute_c5_from_O(O_full, n)

        if c < best_c:
            best_c = c
            # keep momentum consistent with new point
            m = np.zeros_like(best_h)
            v = np.zeros_like(best_h)

        # Adam moments
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        t = beta_index + 1  # a dummy iteration counter for bias correction
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        best_h = best_h - step * m_hat / (np.sqrt(v_hat) + 1e-9)
        best_h = proj_simplex_cap(best_h, target)

        # occasional random jitter to escape plateaus
        if (beta_index % 2) == 0 and rng.random() < 0.15:
            eps = rng.random() * 0.02
            h_tmp = best_h + eps * (rng.random(n) - 0.5)
            h_tmp = proj_simplex_cap(h_tmp, target)
            O_tmp = compute_O_full(h_tmp, L, ones_fft)
            c_tmp = compute_c5_from_O(O_tmp, n)
            if c_tmp < best_c:
                best_h, best_c = h_tmp, c_tmp
                m = np.zeros_like(best_h)
                v = np.zeros_like(best_h)

        # move to next beta if progress stalls
        if best_c * (2.0 / n) < 0.3808:   # reached target (conservative check)
            break
        if beta_index < len(betas) - 1:
            beta_index += 1
        else:
            # keep cycling with the largest beta for final fine tuning
            beta_index = len(betas) - 1

    # ---- final local refinement with smaller steps -------------------------
    step_vals = (0.04, 0.02, 0.01, 0.005)
    for step in step_vals:
        for _ in range(300):
            if time.time() - start > budget_s * 0.99:
                break
            O_full, grad = compute_O_and_grad(best_h, betas[-1], L, ones_fft)
            c = compute_c5_from_O(O_full, n)
            if c < best_c:
                best_c = c
            best_h = best_h - step * grad
            best_h = proj_simplex_cap(best_h, target)

    # ---- final sanity check -------------------------------------------------
    best_h = proj_simplex_cap(best_h, target)
    final_c = compute_c5_from_O(compute_O_full(best_h, L, ones_fft), n)

    return best_h.tolist(), float(final_c), int(n)
