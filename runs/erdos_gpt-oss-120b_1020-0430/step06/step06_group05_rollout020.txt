def compute_c5_fft(h):
    n = h.shape[0]
    g = 1.0 - h
    m = 2 * n
    H = np.fft.fft(h, m)
    G = np.fft.fft(g, m)
    corr = np.fft.ifft(H * np.conj(G)).real
    # take first (2*n-1) entries
    max_val = np.max(corr[:2*n-1])
    return max_val * (2.0 / n)

def local_improve(h):
    n = h.shape[0]
    while True:
        # compute correlation array
        n = h.shape[0]
        m = 2 * n
        H = np.fft.fft(h, m)
        G = np.fft.fft(1.0 - h, m)
        corr = np.fft.ifft(H * np.conj(G)).real
        corr = corr[:2*n-1]
        max_val = corr.max()
        if max_val < 1e-12:
            break
        idx = corr.argmax()
        shift = idx - (n - 1)
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            h_i = h[i_range]
            h_i_shift = h[i_range + shift]
        else:
            shift_abs = -shift
            i_range = np.arange(shift_abs, n)
            h_i = h[i_range]
            h_i_shift = h[i_range + shift]  # shift negative, so i+shift >=0
        # Find A and B
        A_mask = (h_i == 1) & (h_i_shift == 0)
        B_mask = (h_i == 0) & (h_i_shift == 1)
        # A and B indices in i_range
        A_inds = i_range[A_mask]
        B_inds = i_range[B_mask]
        if A_inds.size == 0 or B_inds.size == 0:
            break
        # choose first
        i_swap = A_inds[0]
        j_swap = B_inds[0]
        # swap values
        h[i_swap], h[j_swap] = h[j_swap], h[i_swap]
    return h

import time, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 using np.correlate."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_fft(h: np.ndarray) -> float:
    """Fast C5 via FFT (same objective as compute_c5)."""
    n = h.shape[0]
    m = 2 * n
    H = np.fft.fft(h, m)
    G = np.fft.fft(1.0 - h, m)
    corr = np.fft.ifft(H * np.conj(G)).real
    max_val = np.max(corr[: 2 * n - 1])
    return max_val * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to satisfy Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Return (shift, gradient) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        if k > 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            k = -k
            grad[: n - k] -= h[: n - k]
            grad[k:] -= h[k:]
    return k, grad


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 15) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def local_improve(h: np.ndarray) -> np.ndarray:
    """Greedy swaps that reduce the current worst shift."""
    n = h.shape[0]
    while True:
        m = 2 * n
        H = np.fft.fft(h, m)
        G = np.fft.fft(1.0 - h, m)
        corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
        max_val = np.max(corr)
        if max_val < 1e-12:
            break
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        if shift >= 0:
            rng_i = np.arange(0, n - shift)
            hi = h[rng_i]
            hk = h[rng_i + shift]
        else:
            s = -shift
            rng_i = np.arange(s, n)
            hi = h[rng_i]
            hk = h[rng_i + shift]  # negative shift
        A_mask = (hi == 1) & (hk == 0)
        B_mask = (hi == 0) & (hk == 1)
        A = rng_i[A_mask]
        B = rng_i[B_mask]
        if A.size == 0 or B.size == 0:
            break
        i_swap = A[0]
        j_swap = B[0]
        h[i_swap], h[j_swap] = h[j_swap], h[i_swap]
    return h
