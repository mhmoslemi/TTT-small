def project_sum_cap(x, target):
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo+hi)/2
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2, 0.0, 1.0)

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def softmax_weights(O: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max weights for the overlap vector O."""
    s = beta * O
    m = s.max()
    e = np.exp(s - m)
    return e / e.sum()


def project_sum_cap(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the set {z | sum(z)=target , 0≤z≤1}.
    Uses a simple bisection on the shift applied before clipping.
    """
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


# ---------- gradient of smooth C5 ----------
def smooth_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the soft‑max C5 (beta>0)."""
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    w = softmax_weights(O, beta)

    # split weights
    w_pos = w[n - 1 :]          # k = 0,…,n‑1   (len n)
    w_neg = w[: n - 1][::-1]   # L = 1,…,n‑1   (len n‑1)

    # positive‑shift contributions
    term_pos_plus = np.correlate(w_pos, h, mode="full")[:n]
    term_pos_minus = np.convolve(w_pos, h, mode="full")[:n]

    # negative‑shift contributions (loop over L)
    term_neg_plus = np.zeros(n)
    term_neg_minus = np.zeros(n)
    for L, wL in enumerate(w_neg, start=1):
        if wL == 0.0:
            continue
        term_neg_plus[:-L] += wL * h[L:]
        term_neg_minus[L:] += wL * h[:-L]

    grad = 1.0 - term_pos_plus - term_pos_minus - term_neg_plus - term_neg_minus
    return grad


# ---------- main optimisation ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512                     # number of discretisation points
    target_sum = n / 2.0

    # ---- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, h0)
        else:
            h = h0.copy()
    else:
        h = rng.random(n)

    h = project_sum_cap(h, target_sum)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- parameters ---------------------------------------------------
    beta = 1.0
    beta_max = 2000.0
    beta_inc_iter = 200          # increase beta every this many steps
    step0 = 0.5
    max_iters = 200000           # large; loop stops by time

    it = 0
    # ---- main loop ----------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        grad = smooth_gradient(h, beta)

        # step size schedule
        step = step0 * (1.0 - it / max_iters)
        if step < 1e-4:
            step = 1e-4

        h_new = h - step * grad
        h_new = project_sum_cap(h_new, target_sum)

        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            if best_c <= 0.3808:               # early exit if already good enough
                break

        h = h_new
        it += 1

        if it % beta_inc_iter == 0:
            beta = min(beta * 1.5, beta_max)

        # occasional random shake to escape plateaus
        if it % 5000 == 0:
            h = best_h + rng.normal(scale=0.01, size=n)
            h = project_sum_cap(h, target_sum)
            beta = 1.0

    # ---- final refinement (high beta, tiny steps) --------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        refine_iters = int(min(2000, remaining * 200))
        h = best_h.copy()
        beta = beta_max
        for _ in range(refine_iters):
            grad = smooth_gradient(h, beta)
            h = h - 0.001 * grad
            h = project_sum_cap(h, target_sum)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # ---- final sanitisation -------------------------------------------
    best_h = project_sum_cap(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
