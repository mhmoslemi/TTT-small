    rng = np.random.default_rng(seed)
    start = time.time()
    n = 1024
    # generate initial h
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        # maybe resample to size n via interpolation or tiling?
        # If initial length is 512, we can double it by repeating each entry.
        if init.size != n:
            # repeat each entry twice
            h = np.repeat(init, n // init.size)
        else:
            h = init.copy()
        # The repeated version may not be exactly balanced; enforce sum.
        # We'll binarize (h>0.5) and adjust:
        h = (h > 0.5).astype(np.float64)
        # ensure sum == n/2
        current = int(h.sum())
        target = n // 2
        diff = target - current
        while diff != 0:
            idx = rng.choice(n, size=abs(diff), replace=False)
            if diff > 0:
                # set zeros to 1
                zeros = np.where(h[idx] == 0)[0]
                set_idx = idx[zeros][:diff]
                h[set_idx] = 1
                diff = target - int(h.sum())
            else:
                ones = np.where(h[idx] == 1)[0]
                set_idx = idx[ones][: -diff]
                h[set_idx] = 0
                diff = target - int(h.sum())
    else:
        # random balanced binary
        h = np.zeros(n, dtype=np.float64)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

def thue_morse_bits(n):
    bits = np.zeros(n, dtype=np.int8)
    for i in range(n):
        bits[i] = bin(i).count('1') % 2
    return bits

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def thue_morse_bits(n: int) -> np.ndarray:
    """First n bits of the Thue–Morse sequence (0/1)."""
    bits = np.empty(n, dtype=np.float64)
    for i in range(n):
        bits[i] = bin(i).count("1") & 1
    return bits


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    n = 1024                         # number of samples (even)
    offset = n - 1                   # index shift for correlation array

    # ---------- initial candidates ----------
    candidates = []

    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        if init.size != n:
            factor = n // init.size
            init = np.repeat(init, factor)
        init = (init > 0.5).astype(np.float64)
        target = n // 2
        diff = int(target - init.sum())
        if diff != 0:
            zeros = np.where(init == 0)[0]
            ones = np.where(init == 1)[0]
            if diff > 0:
                idx = rng.choice(zeros, diff, replace=False)
                init[idx] = 1.0
            else:
                idx = rng.choice(ones, -diff, replace=False)
                init[idx] = 0.0
        candidates.append((init, compute_c5(init)))

    # Thue–Morse
    tm = thue_morse_bits(n)
    candidates.append((tm, compute_c5(tm)))

    # Random balanced binary
    rand = np.zeros(n, dtype=np.float64)
    rand[rng.choice(n, n // 2, replace=False)] = 1.0
    candidates.append((rand, compute_c5(rand)))

    best_h, best_c = min(candidates, key=lambda t: t[1])
    best_corr = np.max(np.correlate(best_h, 1.0 - best_h, mode="full"))

    # ---------- Simulated annealing ----------
    T = 0.5
    T_end = 1e-6
    cooling = 0.9999
    delta = np.empty(2 * n - 1, dtype=np.float64)
    cur_h = best_h.copy()
    cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
    cur_max = cur_corr.max()
    best_overall_h = best_h.copy()
    best_overall_corr = best_corr
    no_improve = 0
    it = 0

    while time.time() - start < budget_s * 0.94:
        # pick a 1 and a 0
        i = rng.integers(n)
        while cur_h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] == 1:
            j = rng.integers(n)

        delta.fill(0.0)
        tmp = 1.0 - cur_h
        si = offset - i
        sj = offset - j
        delta[si : si + n] -= tmp
        delta[sj : sj + n] += tmp
        rev = cur_h[::-1]
        sri = i + offset - (n - 1)
        srj = j + offset - (n - 1)
        delta[sri : sri + n] += rev
        delta[srj : srj + n] -= rev

        new_corr = cur_corr + delta
        new_max = new_corr.max()
        d_corr = new_max - cur_max
        d_c5 = (2.0 / n) * d_corr

        accept = d_c5 <= 0.0 or rng.random() < np.exp(-d_c5 / T)
        if accept:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            cur_corr = new_corr
            cur_max = new_max
            if cur_max < best_overall_corr:

                best_overall_h = cur_h.copy()
                no_improve = 0
            else:

        else:
            no_improve += 1

        if T > T_end:

        if no_improve > 200_000:

            cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            cur_max = cur_corr.max()
            no_improve = 0
            T = 0.5

        it += 1
        if it % 10_000 == 0 and time.time() - start > budget_s * 0.94:
            break

    # ---------- Final greedy descent (zero temperature) ----------
    cur_h = best_overall_h.copy()
    cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
    cur_max = cur_corr.max()
    max_iter = 200_000
    for _ in range(max_iter):
        if time.time() - start > budget_s * 0.99:
            break
        i = rng.integers(n)
        while cur_h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)

            j = rng.integers(n)

        sj = offset - j

        delta[sj : sj + n] += tmp
        rev = cur_h[::-1]

        delta[srj : srj + n] -= rev

        new_max = new_corr.max()

            cur_max = new_max

    return cur_h.tolist(), float(final_c), int(n)
