import time
import numpy as np

def adjust_sum_exact(h, rng, max_iter=2000):
    """Force sum(h) == len(h)/2 exactly by moving mass between entries."""
    n = h.size
    target = n * 0.5
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def next_pow_two(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def compute_corr_fft(h: np.ndarray, n: int, M: int) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT, length 2*n-1."""
    H = np.fft.rfft(h, n=M)
    G = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(H * np.conj(G), n=M)[: 2 * n - 1]
    return corr

def compute_c5_fft(h: np.ndarray, n: int, M: int) -> float:
    """C5 = max_overlap * (2/n) using FFT correlation."""
    corr = compute_corr_fft(h, n, M)
    return float(corr.max() * (2.0 / n))

            if up_it % 1000 == 0 and time.time() > deadline:
                break

        if up_best_c < best_c:
            best_c = up_best_c
            best_h = up_best_h.copy()
            best_n = up_n
            M = up_M
        else:
            best_n = base_n
            M = next_pow_two(2 * best_n)

    # ---- final safety checks ----
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5_fft(best_h, n, next_pow_two(2 * n))
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5_fft(best_h, best_n, next_pow_two(2 * best_n))

    return best_h.tolist(), float(final_c), int(best_n)
