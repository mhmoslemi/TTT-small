def gradient_descent_opt(h0, beta_seq, rng, start_time, budget_s):
    h = h0.copy()
    N = h.size
    target = N/2.0
    best_h = h.copy()
    # compute initial objective
    c = np.correlate(h, 1-h, mode='full')
    maxc = c.max()
    sum_exp = np.exp(beta_seq[0] * (c - maxc)).sum()
    best_obj = (maxc + np.log(sum_exp)) / beta_seq[0]
    best_h = h.copy()
    # for each beta
    for beta in beta_seq:
        # adapt max iterations based on remaining time
        remaining = start_time + budget_s - time.time()
        if remaining < 0.1:
            break
        # set iteration limit: e.g., 2000 or based on remaining / (approx time per iter)
        max_iter = 2000
        step = 0.1
        obj = best_obj
        for it in range(max_iter):
            # compute c and w
            c = np.correlate(h, 1-h, mode='full')
            maxc = c.max()
            exp_vals = np.exp(beta * (c - maxc))
            sum_exp = exp_vals.sum()
            w = exp_vals / sum_exp
            # compute conv1 and conv2 via np.convolve
            conv1 = np.convolve(w[::-1], h, mode='full')
            conv2 = np.convolve(w, h, mode='full')
            grad = 1.0 - conv1[N-1:N-1+N] - conv2[N-1:N-1+N]
            # line search: try step, adjust if not improving
            improved = False
            for _ in range(10):
                nh = h - step * grad
                nh = np.clip(nh, 0.0, 1.0)
                diff = target - nh.sum()
                if abs(diff) > 1e-12:
                    # simple correction: distribute among free indices
                    # find indices not at bound
                    free = np.where((nh > 1e-12) & (nh < 1 - 1e-12))[0]
                    if free.size > 0:
                        # distribute diff equally
                        nh[free] += diff / free.size
                    else:
                        # fallback to random adjustments
                        for _ in range(200):
                            i = rng.integers(N)
                            if diff > 0:
                                add = min(diff, 1.0 - nh[i])
                            else:
                                add = max(diff, -nh[i])

                    obj = obj_new
                    improved = True
                    break

                    step *= 0.5

                        break

        c = np.correlate(h, 1-h, mode='full')
        maxc = c.max()
        sum_exp = np.exp(beta * (c - maxc)).sum()
        best_obj = (maxc + np.log(sum_exp)) / beta
    return best_h

def conv_fft(a,b):
    n = len(a) + len(b) - 1
    m = 1 << (n-1).bit_length()

    B = np.fft.rfft(b, n=m)

    conv = np.fft.irfft(C, n=m)
    return conv[:n]

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Soft‑max objective for SLSQP."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- gradient descent on soft‑max ----------
def gradient_descent_opt(h0: np.ndarray, beta_seq, rng,
                        start_time: float, budget_s: float) -> np.ndarray:
    """Projected gradient descent on log‑sum‑exp of correlations."""
    h = h0.copy()
    N = h.size
    target = N / 2.0
    best_h = h.copy()
    best_obj = np.inf
    for beta in beta_seq:
        step = 0.1
        # initial objective at this beta
        c = np.correlate(h, 1.0 - h, mode="full")
        maxc = c.max()
        obj = (maxc + np.log(np.exp(beta * (c - maxc)).sum())) / beta
        it = 0
        while True:
            if time.time() > start_time + budget_s * 0.85:
                break
            if it >= 800:
                break
            it += 1
            # gradient of log‑sum‑exp
            c = np.correlate(h, 1.0 - h, mode="full")
            maxc = c.max()
            exp_vals = np.exp(beta * (c - maxc))
            w = exp_vals / exp_vals.sum()
            conv1 = np.convolve(w[::-1], h, mode="full")
            conv2 = np.convolve(w, h, mode="full")
            grad = 1.0 - conv1[N - 1:N - 1 + N] - conv2[N - 1:N - 1 + N]
            # trial step
            nh = h - step * grad
            nh = np.clip(nh, 0.0, 1.0)
            nh = enforce_sum(nh, rng)
            # new objective
            c_new = np.correlate(nh, 1.0 - nh, mode="full")
            maxc_new = c_new.max()
            obj_new = (maxc_new + np.log(np.exp(beta * (c_new - maxc_new)).sum())) / beta
            if obj_new < obj:
                h = nh
                obj = obj_new
                step *= 1.02
                if obj < best_obj:
                    best_obj = obj
                    best_h = h.copy()
            else:
                step *= 0.5
                if step < 1e-8:
                    break
        # carry the final h to next beta
    return best_h
