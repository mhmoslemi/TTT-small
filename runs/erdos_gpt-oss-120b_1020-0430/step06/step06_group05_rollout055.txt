import time
import numpy as np
import cvxpy as cp


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Solve a small MIP for the Erdős C5 constant.

    The MIP directly minimises the worst‐case uncovered measure
    T = max_k Σ_{i=0}^{n-1-k} h_i (1−h_{i+k}) and returns the step function.
    """
    start = time.time()
    target_c = 0.3808  # we only need a bound ≤ this value
    best_h, best_c, best_n = None, float("inf"), None

    # candidate discretisations (small → fast)
    n_candidates = [64, 80, 96, 112, 128]

    # utility to evaluate the discrete C5 for a given h
    def compute_c5(arr: np.ndarray) -> float:
        corr = np.correlate(arr, 1.0 - arr, mode="full")
        return np.max(corr) * (2.0 / arr.size)

    for n in n_candidates:
        # stop if we are out of time
        if time.time() - start > budget_s * 0.94:
            break

        # build the MIP for this n
        x = cp.Variable(n, boolean=True)          # h_i ∈ {0,1}
        T = cp.Variable()                         # worst uncovered count
        constraints = [cp.sum(x) == n / 2, T >= 0]

        for k in range(1, n):
            m = n - k
            y = cp.Variable(m, boolean=True)      # y_i = x_i * x_{i+k}
            # linearisation of the product
            constraints += [y <= x[:m],
                            y <= x[k:],
                            y >= x[:m] + x[k:] - 1]
            # T ≥ Σ x_i – Σ y_i  (i runs over 0…n-1-k)
            constraints += [T >= cp.sum(x[:m]) - cp.sum(y)]

        prob = cp.Problem(cp.Minimize(T), constraints)

        # allocate time for this solve (≈90 % of remaining budget)
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining <= 0:
            break
        time_limit = max(1.0, remaining * 0.9)   # seconds

        solved = False
        # try a sequence of MIP solvers, respecting the time limit when possible
        try:
            prob.solve(
                solver=cp.GLPK_MI,
                glpk={"tm_lim": int(time_limit * 1000)},
                warm_start=True,
                verbose=False,
            )
            solved = True
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(
                    solver=cp.CBC,
                    cbc={"seconds": time_limit},
                    warm_start=True,
                    verbose=False,
                )
                solved = True
            except Exception:
                pass
        if not solved:
            try:
                prob.solve(
                    solver=cp.ECOS_BB,
                    warm_start=True,
                    verbose=False,
                )
                solved = True
            except Exception:
                pass
        if not solved:
            # could not solve for this n, move on
            del x, T, prob
            continue

        if prob.status not in ("optimal", "optimal_inaccurate"):
            # not optimal – still try to use the obtained values
            pass

        T_val = T.value
        if T_val is None:
            del x, T, prob
            continue

        # obtain a binary step function from the solution
        x_val = x.value
        if x_val is None:
            del x, T, prob
            continue
        h = np.rint(np.clip(x_val, 0.0, 1.0)).astype(float)

        # enforce the exact half‑mass condition (tiny rounding errors)
        target_sum = n // 2
        diff = target_sum - h.sum()
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) >= diff:
                h[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.where(h == 1)[0]
            if len(ones) >= -diff:
                h[ones[: -diff]] = 0.0

        if h.sum() != target_sum:
            # failed to repair; skip this candidate
            del x, T, prob
            continue

        c5_val = compute_c5(h)

        if c5_val < best_c:
            best_c, best_h, best_n = c5_val, h.copy(), n

        if best_c <= target_c:
            # good enough – stop searching
            del x, T, prob
            break

        del x, T, prob  # free memory before next n

    # ----------------------------------------------------------------------
    # fallback if MIP never succeeded
    if best_h is None:
        if "initial_h_values" in globals():
            best_h = np.array(initial_h_values, dtype=float)
            best_n = best_h.size
            best_c = compute_c5(best_h)
        else:
            # simple half‑ones construction
            best_n = 512
            best_h = np.zeros(best_n)
            best_h[: best_n // 2] = 1.0
            rng = np.random.default_rng(seed)
            best_h = rng.permutation(best_h)
            best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h.tolist(), float(best_c), int(best_n)
