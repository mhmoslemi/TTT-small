def optimal_delta(h, grad):
    # returns delta vector of same shape
    n = h.shape[0]
    delta = np.zeros_like(h)
    # Positive grads (P)
    pos_mask = grad > 1e-12
    # Negative grads (N)
    neg_mask = grad < -1e-12
    # If both empty return zeros
    if not pos_mask.any() or not neg_mask.any():
        return delta
    # Extract arrays
    h_pos = h[pos_mask]
    grad_pos = grad[pos_mask]
    h_neg = h[neg_mask]
    grad_neg = grad[neg_mask]  # negative values

    # removal capacities
    rem_caps = h_pos.copy()
    # addition capacities
    add_caps = (1.0 - h_neg)

    total_remove = rem_caps.sum()
    total_add = add_caps.sum()
    T = min(total_remove, total_add)

    # Determine which side is limiting
    if total_remove <= total_add:
        # remove all from P
        delta[pos_mask] = -rem_caps
        # add partially to N up to T (which = total_remove)
        # allocate to largest |grad| first
        # sort indices of N by descending -grad (i.e., absolute)
        order = np.argsort(-grad_neg)  # descending absolute
        remaining = T
        for idx in order:
            cap = add_caps[idx]
            take = min(cap, remaining)
            delta[neg_mask][idx] = take
            remaining -= take
            if remaining <= 1e-14:
                break
    else:
        # add all to N
        delta[neg_mask] = add_caps
        # remove partially from P up to T (total_add)
        order = np.argsort(-grad_pos)  # descending grad
        remaining = T
        for idx in order:
            cap = rem_caps[idx]
            take = min(cap, remaining)
            delta[pos_mask][idx] = -take
            remaining -= take
            if remaining <= 1e-14:
                break
    # ensure sum delta approx 0; small rounding diff
    return delta

def optimal_delta(h, grad):
    n = h.shape[0]
    delta = np.zeros_like(h)
    pos_idx = np.where(grad > 1e-12)[0]
    neg_idx = np.where(grad < -1e-12)[0]
    if pos_idx.size == 0 or neg_idx.size == 0:
        return delta
    # capacities
    rem_caps = h[pos_idx]  # removal capacity
    add_caps = 1.0 - h[neg_idx]  # addition capacity
    total_remove = rem_caps.sum()
    total_add = add_caps.sum()
    T = min(total_remove, total_add)
    # If total_remove <= total_add, remove all from pos, add limited to T to neg sorted
    if total_remove <= total_add:
        # remove all
        delta[pos_idx] = -rem_caps
        # allocate addition to neg sorted by absolute grad (descending)
        grad_abs_neg = -grad[neg_idx]  # positive values
        order = np.argsort(-grad_abs_neg)  # descending
        remaining = T
        for idx in order:
            i = neg_idx[idx]
            cap = add_caps[idx]
            take = min(cap, remaining)
            delta[i] = take
            remaining -= take
            if remaining <= 1e-14:
                break
    else:
        # add all to neg
        delta[neg_idx] = add_caps
        # allocate removal from pos sorted by grad descending
        grad_pos = grad[pos_idx]
        order = np.argsort(-grad_pos)  # descending
        remaining = T
        for idx in order:
            i = pos_idx[idx]
            cap = rem_caps[idx]
            take = min(cap, remaining)
            delta[i] = -take
            remaining -= take
            if remaining <= 1e-14:
                break
    return delta

def random_perturb(h, rng, scale=0.1):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    # Transfer from i to j preserving sum
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

Now enforce_sum:

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h

def local_search(h_init, target, rng, time_deadline):
    h = h_init.copy()
    c = compute_c5(h)
    # Repeat up to some max iterations or until time expires
    for _ in range(200):

            break

        delta = optimal_delta(h, grad)

import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kp = -k
        grad[kp:] = 1.0 - h[: n - kp]
        grad[: n - kp] -= h[kp:]
    return k, grad

def optimal_delta(h, grad):
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return np.zeros_like(h)
    rem = h[pos]
    add = 1.0 - h[neg]
    tot_rem = rem.sum()
    tot_add = add.sum()
    T = min(tot_rem, tot_add)
    delta = np.zeros_like(h)
    if tot_rem <= tot_add:
        delta[pos] = -rem
        grad_abs_neg = -grad[neg]
        order = np.argsort(-grad_abs_neg)
        left = T
        for idx in order:
            i = neg[idx]
            cap = add[idx]
            take = min(cap, left)
            delta[i] = take
            left -= take
            if left <= 1e-14:
                break
    else:
        delta[neg] = add
        order = np.argsort(-grad[pos])
        left = T
        for idx in order:
            i = pos[idx]
            cap = rem[idx]
            take = min(cap, left)
            delta[i] = -take
            left -= take
            if left <= 1e-14:
                break
    return delta

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h

def random_perturb(h, rng, scale=0.1):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    maxi = min(h[i], 1.0 - h[j])
    if maxi <= 0:
        return h
    eps = maxi * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

def local_search(h0, target, rng, deadline):
    h = h0.copy()
    c = compute_c5(h)
    for _ in range(200):
        if time.time() > deadline:
            break

        delta = optimal_delta(h, grad)

import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    else:
        kp = -k
        grad[kp:] = 1.0 - h[: n - kp]
        grad[: n - kp] -= h[kp:]
    return k, grad

def optimal_delta(h, grad):
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return np.zeros_like(h)
    rem = h[pos]
    add = 1.0 - h[neg]
    tot_rem = rem.sum()
    tot_add = add.sum()
    T = min(tot_rem, tot_add)
    delta = np.zeros_like(h)
    if tot_rem <= tot_add:
        delta[pos] = -rem
        grad_abs_neg = -grad[neg]
        order = np.argsort(-grad_abs_neg)
        left = T
        for idx in order:
            i = neg[idx]
            cap = add[idx]
            take = min(cap, left)
            delta[i] = take
            left -= take
            if left <= 1e-14:
                break
    else:
        delta[neg] = add
        order = np.argsort(-grad[pos])
        left = T
        for idx in order:
            i = pos[idx]
            cap = rem[idx]
            take = min(cap, left)
            delta[i] = -take
            left -= take
            if left <= 1e-14:
                break
    return delta

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h

def random_perturb(h, rng, scale=0.1):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    maxi = min(h[i], 1.0 - h[j])
    if maxi <= 0:
        return h
    eps = maxi * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2

    h = h0.copy()

    for _ in range(200):

        if np.allclose(delta, 0, atol=1e-12):

        h_new = h + delta
        c_new = compute_c5(h_new)
        if c_new < c - 1e-12:

        improved = False
        for _ in range(10):

            c2 = compute_c5(h2)
            if c2 < c - 1e-12:
                h, c = h2, c2
                improved = True
                break
        if not improved:
            break
    return enforce_sum(h, target, rng), compute_c5(h)

            if c_s < best_c:
                best_h, best_c = h_s, c_s
        except Exception:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    return best_h.tolist(), float(compute_c5(best_h)), int(best_h.size)
