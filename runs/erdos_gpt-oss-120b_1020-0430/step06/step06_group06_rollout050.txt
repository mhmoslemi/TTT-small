import time
import numpy as np
from scipy.optimize import minimize

# ----- global helpers for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple proportional correction first
    if abs(diff) > 1e-12:
        # add diff/n to all, then clip and recompute diff
        add_all = diff / n
        h = np.clip(h + add_all, 0.0, 1.0)
        diff = target - h.sum()
    # finish with random one‑by‑one corrections
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 5) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear up‑sampling from old_h (on [0,2)) to new_n points."""
    old_n = old_h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # ----- choose resolution -----
    N = 1024  # higher resolution gives finer control

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h0 = upsample_h(np.array(initial_h_values, dtype=float), N)
    else:
        # sin pattern: half a sine wave from 0 to 2
        i = np.arange(N)
        h0 = 0.5 + 0.5 * np.sin(np.pi * i / N)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- random restarts with low‑β SLSQP -----
    beta_seq_low = (20.0, 200.0, 2000.0)
    while time_left() > budget_s * 0.3:
        # random starting point
        rand_h = rng.random(N)
        rand_h = enforce_sum(rand_h, rng)
        h_cur = rand_h.copy()
        for beta in beta_seq_low:
            if time_left() <= 0:
                break
            h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=200)
        c_cur = compute_c5(h_cur)
        if c_cur < best_c:
            best_h, best_c = h_cur.copy(), c_cur

    # ----- final refinement on the best candidate -----
    beta_seq_high = (20000.0, 200000.0, 500000.0)
    for beta in beta_seq_high:
        if time_left() <= budget_s * 0.1:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=500)
        best_c = compute_c5(best_h)

    # ----- gradient polishing -----
    if time_left() > budget_s * 0.05:
        refined = gradient_refine(best_h, rng, step=0.02, max_iter=8)
        rc = compute_c5(refined)
        if rc < best_c:
            best_h, best_c = refined.copy(), rc

    # ----- final repair & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
