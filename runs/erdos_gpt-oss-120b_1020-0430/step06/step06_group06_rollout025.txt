import time
import numpy as np
from scipy.optimize import minimize

# ----- global placeholders for constraints -----
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple iterative correction
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 200) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def random_move(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Swap two entries or transfer a random amount while preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    if rng.random() < 0.2:  # pure swap
        nh = h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        return nh
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 8) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

    G_deadline = start + budget_s * 0.85
    while time.time() < G_deadline:
        refined = gradient_refine(best_h, rng, step=0.03, max_iter=5)
        rc = compute_c5(refined)
        if rc < best_c:
            best_h, best_c = refined.copy(), rc
        # occasional small random improvement attempts
        for _ in range(100):
            nh = random_move(best_h, rng, scale=0.1)
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh.copy(), nc

    # ----- soft‑max SLSQP passes -----
    S_deadline = start + budget_s * 0.98
    beta_seq = (50.0, 500.0, 5000.0, 20000.0, 200000.0)
    for beta in beta_seq:
        if time.time() > S_deadline:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=150)
        best_c = compute_c5(best_h)

    # ----- final repair & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
