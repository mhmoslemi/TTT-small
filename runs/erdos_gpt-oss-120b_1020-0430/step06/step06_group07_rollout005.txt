def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine n_points
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # convert to binary: sort descending
        order = np.argsort(-h0)
        h_bin = np.zeros(n, dtype=np.int8)
        h_bin[order[:n//2]] = 1
    else:
        n = 512
        # random binary half 1s
        perm = rng.permutation(n)
        h_bin = np.zeros(n, dtype=np.int8)
        h_bin[perm[:n//2]] = 1

    # compute c = np.correlate(h_bin, h_bin, mode='full')
    c = np.correlate(h_bin, h_bin, mode='full')
    best_h = h_bin.copy()
    best_c_min = c.min()
    # compute best_c5
    best_c5 = 1 - 2 * best_c_min / n

    # maintain lists
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    # set parameters
    max_iters = 5000000   # huge but will break on time
    iteration = 0
    # temperature schedule
    T0 = 0.0005  # initial temperature (C5 units)
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.94:
            break
        iteration += 1
        # sample i,j
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute new c
        start_i = n - 1 - i
        start_j = n - 1 - j
        new_c = c.copy()
        new_c[start_i:start_i+n] -= h_bin
        new_c[start_j:start_j+n] += h_bin
        # compute new min_c
        new_c_min = new_c.min()
        new_c5 = 1 - 2 * new_c_min / n
        # acceptance
        cur_c5 = 1 - 2 * c.min() / n
        # temperature decreasing
        T = T0 * (1 - iteration / max_iters)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # if T>0, accept with prob
            if T > 1e-12:
                prob = math.exp((cur_c5 - new_c5) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            # apply swap
            h_bin[i] = 0
            h_bin[j] = 1
            # update lists
            # remove i from ones, add to zeros; remove j from zeros, add to ones
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            # update c
            c = new_c
            # compare to best
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                # optionally early stop if below target.
                if best_c5 <= 0.3808:
                    break

    # final sanitization
    best_h = np.clip(best_h, 0, 1).astype(float)
    # ensure sum constraint: sum should be n/2; if off due to rounding, fix
    # compute sum
    current_sum = best_h.sum()
    target_sum = n / 2
    if abs(current_sum - target_sum) > 1e-6:
        # adjust via adding or removing ones
        diff = int(round(target_sum - current_sum))
        if diff > 0:
            # set diff zeros to 1
            zero_idxs = np.where(best_h < 0.5)[0]
            best_h[zero_idxs[:diff]] = 1.0
        elif diff < 0:
            one_idxs = np.where(best_h > 0.5)[0]
            best_h[one_idxs[: -diff]] = 0.0

    final_c5 = compute_c5(best_h)  # Use provided function compute_c5 if imported; else compute ourselves

import time, math
import numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def worst_gradient(h):
    n = h.size
    # full correlation h*h
    cross = np.correlate(h, h, mode='full')[n - 1:]  # shifts 0..n-1
    k = int(np.argmin(cross))  # shift with minimal self‑overlap → maximal O
    grad = np.ones_like(h)
    if k != 0:
        # for positions where both h[i] and h[i+k] are defined (i from 0 to n-k-1)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def subgradient_optimize(h0, target, time_budget, rng):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c
```

import time, math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# basic evaluation
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ----------------------------------------------------------------------
# projection onto {0≤x≤1, sum x = target}
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

# ----------------------------------------------------------------------
# worst‑shift gradient (used by sub‑gradient descent)
def worst_gradient(h: np.ndarray):
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # shifts 0..n-1
    k = int(np.argmin(cross))                        # shift giving max overlap
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

# ----------------------------------------------------------------------
# projected sub‑gradient descent on max‑overlap
def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10.0)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c

# ----------------------------------------------------------------------
# soft‑max SLSQP optimiser (final polishing)
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # smooth approximation of max_k O_k
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # --------------------------------------------------------------
    # problem size and initial binary construction
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n = h_init.size
    else:
        n = 512
        h_init = None
    half = n // 2

    if h_init is not None:
        order = np.argsort(-h_init, kind="mergesort")
        ones_idx = order[:half]
        zeros_idx = order[half:]
    else:
        perm = rng.permutation(n)
        ones_idx = perm[:half]
        zeros_idx = perm[half:]

    h = np.zeros(n, dtype=np.int8)
    h[ones_idx] = 1

    # permutation / position scheme for O(1) swaps
    perm = np.empty(n, dtype=np.int32)
    perm[:half] = ones_idx
    perm[half:] = zeros_idx
    pos = np.empty(n, dtype=np.int32)
    pos[perm] = np.arange(n, dtype=np.int32)

    # correlation of the binary vector with itself
    c = np.correlate(h, h, mode="full").astype(np.int64)
    cur_c5 = 1.0 - 2.0 * c.min() / n
    best_h = h.copy()
    best_c5 = cur_c5

    # --------------------------------------------------------------
    # simulated‑annealing on the binary representation
    max_iters = 5_000_000
    T0 = 7e-4
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.85:
            break
        if it >= max_iters:
            break
        it += 1

        # pick random one and zero via permutation
        idx_one = rng.integers(0, half)
        i = int(perm[idx_one])
        idx_zero = rng.integers(half, n)
        j = int(perm[idx_zero])

        # tentative correlation after swapping i↔j
        new_c = c.copy()
        start_i = n - 1 - i
        start_j = n - 1 - j
        new_c[start_i:start_i + n] -= h
        new_c[start_j:start_j + n] += h
        new_c_min = new_c.min()
        new_c5 = 1.0 - 2.0 * new_c_min / n

        T = T0 * (1.0 - it / max_iters)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        elif T > 0.0:
            if rng.random() < math.exp((cur_c5 - new_c5) / T):
                accept = True

        if accept:
            # perform swap
            h[i] = 0
            h[j] = 1
            perm[idx_one], perm[idx_zero] = perm[idx_zero], perm[idx_one]
            pos[i] = idx_zero
            pos[j] = idx_one
            c = new_c
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                if best_c5 <= 0.3808:
                    break

    # --------------------------------------------------------------
    # convert the best binary vector to float for continuous polishing
    h_float = best_h.astype(np.float64)

    # --------------------------------------------------------------
    # continuous sub‑gradient descent (remaining time)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        sub_time = remaining * 0.90
        h_ref, c_ref = subgradient_optimize(h_float, n / 2.0, sub_time, rng)
        final_h = h_ref
        final_c = c_ref
    else:
        final_h = h_float
        final_c = compute_c5(final_h)

    # --------------------------------------------------------------
    # final SLSQP polishing if time permits
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        h_opt = slsqp_optimize(final_h, beta=8000.0, rng=rng, max_iter=200)
        final_c = compute_c5(h_opt)
        final_h = h_opt

    # --------------------------------------------------------------
    # sanitise output
    final_h = np.clip(final_h, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = final_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        diff = int(round(target_sum - cur_sum))
        if diff > 0:
            zero_idx = np.where(final_h < 0.5)[0]
            final_h[zero_idx[:diff]] = 1.0
        elif diff < 0:
            one_idx = np.where(final_h > 0.5)[0]
            final_h[one_idx[: -diff]] = 0.0
    final_c = compute_c5(final_h)

    return final_h.tolist(), float(final_c), int(n)
