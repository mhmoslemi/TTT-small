def greedy_binary_reduce(h, rng, deadline):
    n = h.size
    dx = 2.0 / n
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        c = max_val * dx
        if c <= target_c:
            break
        idx = np.argmax(corr)
        k = idx - (n - 1)
        if k == 0:
            # shift zero not relevant for binary (c=0)
            # break?
            break
        # compute both1, both0
        if k > 0:
            h1 = h[:n - k]
            h2 = h[k:]
            both1 = np.where((h1 == 1) & (h2 == 1))[0]
            both0 = np.where((h1 == 0) & (h2 == 0))[0]
            # map indices
            if both1.size > 0 and both0.size > 0:
                i = both1[rng.integers(both1.size)]
                j = both0[rng.integers(both0.size)] + k
                h[i] = 0
                h[j] = 1
                continue
        else:  # k < 0
            k_abs = -k
            h1 = h[k_abs:]
            h2 = h[:n - k_abs]
            both1 = np.where((h1 == 1) & (h2 == 1))[0]
            both0 = np.where((h1 == 0) & (h2 == 0))[0]

                h[j] = 1

        j = rng.integers(n)

def continuous_sa(h0, rng, deadline):
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = 0.005
    T_end = 1e-8
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # compute T
        T = T0 * (T_end / T0) ** (elapsed / total_sa_time)
        # compute scale
        scale = T / T0  # small near end
        # propose move: choose i,j
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i: j += 1
        max_eps = min(h[i], 1.0 - h[j]) if rng.random() < 0.5 else min(h[j], 1.0 - h[i])
        eps = max_eps * scale * rng.random()
        if eps <= 0:
            continue
        h_new = h.copy()
        if rng.random() < 0.5:
            h_new[i] -= eps
            h_new[j] += eps
        else:
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - best_c
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # accept with Metropolis
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
    return best_h, best_c

def greedy_binary_reduce(h, rng, deadline, target_c):
    n = h.size
    dx = 2.0 / n
    target_overlap = target_c / dx
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        if max_val <= target_overlap:
            break
        idx = np.argmax(corr)
        k = idx - (n - 1)
        if k == 0:
            # zero shift, ignore
            break
        if k > 0:
            h1 = h[:n - k]
            h2 = h[k:]
            both1 = np.where((h1 == 1) & (h2 == 1))[0]
            both0 = np.where((h1 == 0) & (h2 == 0))[0]
            if both1.size and both0.size:
                i = both1[rng.integers(both1.size)]
                j = both0[rng.integers(both0.size)] + k
                h[i] = 0
                h[j] = 1
                continue
        else:
            k_abs = -k
            h1 = h[k_abs:]
            h2 = h[:n - k_abs]
            both1 = np.where((h1 == 1) & (h2 == 1))[0]
            both0 = np.where((h1 == 0) & (h2 == 0))[0]
            if both1.size and both0.size:
                i = both1[rng.integers(both1.size)] + k_abs
                j = both0[rng.integers(both0.size)]
                h[i] = 0
                h[j] = 1
                continue
        # fallback random swap of mismatched bits
        i = rng.integers(n)
        j = rng.integers(n)
        while h[i] == h[j]:
            j = rng.integers(n)
        h[i], h[j] = h[j], h[i]
    final_corr = np.correlate(h, 1.0 - h, mode='full')
    final_c = final_corr.max() * dx
    return h, final_c

    dx = 2.0 / n
    best_h = h.copy()

        elapsed = time.time() - start_sa
        if total_sa > 0:

        else:
            T = T_end
        scale = 0.2 * (T / T0)  # similar scaling
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # decide direction
        if rng.random() < 0.5:

            break
        idx = np.argmax(corr)
        k = idx - (n - 1)
        if k == 0:
            break
        if k > 0:
            h1 = h[: n - k]
            h2 = h[k:]
            both1 = np.where((h1 == 1) & (h2 == 1))[0]
            both0 = np.where((h1 == 0) & (h2 == 0))[0]
            if both1.size and both0.size:
                i = both1[rng.integers(both1.size)]
                j = both0[rng.integers(both0.size)] + k
                h[i] = 0
                h[j] = 1
                continue
        else:
            k_abs = -k
            h1 = h[k_abs:]
            h2 = h[: n - k_abs]
            both1 = np.where((h1 == 1) & (h2 == 1))[0]
            both0 = np.where((h1 == 0) & (h2 == 0))[0]
            if both1.size and both0.size:
                i = both1[rng.integers(both1.size)] + k_abs
                j = both0[rng.integers(both0.size)]
                h[i] = 0
                h[j] = 1
                continue
        # fallback random swap of mismatched bits
        i = rng.integers(n)
        j = rng.integers(n)
        while h[i] == h[j]:
            j = rng.integers(n)
        h[i], h[j] = h[j], h[i]
    final_c = compute_c5(h)
    return h, final_c

# Continuous simulated annealing
def continuous_sa(h0: np.ndarray, rng: np.random.Generator,
                 deadline: float) -> tuple:
    """SA on the continuous feasible set (sum fixed, bounds [0,1])."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start_sa = time.time()
    total_sa = max(deadline - start_sa, 0.0)
    if total_sa <= 0.0:
        return best_h, best_c
    T0, T_end = 0.005, 1e-8
    while time.time() < deadline:
        elapsed = time.time() - start_sa
        T = T0 * (T_end / T0) ** (elapsed / total_sa) if total_sa > 0 else T_end
        scale = 0.2 * (T / T0)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            if eps <= 0.0:
                continue
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
        c_new = compute_c5(h_new)
        delta = c_new - best_c
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
    return best_h, best_c

# Main driver
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    def update_candidate(h):
        nonlocal best_h, best_c, best_n
        c = compute_c5(h)
        if c < best_c:
            best_h = h.copy()
            best_c = c
            best_n = h.size

    # ---- 1. initial solution if provided ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        update_candidate(h0)

    # ---- 2. Binary greedy search on 512 points ----
    n = 512
    remaining = deadline - time.time()
    if remaining > 30.0:
        t_bin = min(remaining * 0.5, 200.0)
        dead_bin = time.time() + t_bin
        bin_best_h = None
        bin_best_c = np.inf
        while time.time() < dead_bin:
            h_bin = np.zeros(n, dtype=np.int8)
            ones = rng.choice(n, n // 2, replace=False)
            h_bin[ones] = 1
            h_bin, c_bin = greedy_binary_reduce(h_bin, rng, dead_bin, TARGET_C)
            if c_bin < bin_best_c:
                bin_best_c = c_bin
                bin_best_h = h_bin.copy()
                if bin_best_c <= TARGET_C:
                    break
        if bin_best_h is not None:
            update_candidate(bin_best_h.astype(float))

    # Fallback simple construction if nothing yet
    if best_h is None:
        h_simple = np.concatenate([np.ones(n // 2), np.zeros(n // 2)]).astype(float)
        h_simple = enforce_sum(h_simple, rng)
        update_candidate(h_simple)

    # ---- 3. Optional small perturbation for binary start ----
    if best_h is not None and np.all((best_h == 0) | (best_h == 1)):
        eps = 1e-4
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size and zeros.size:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            best_h[i] -= eps
            best_h[j] += eps
            best_h = enforce_sum(best_h, rng)

    # ---- 4. SLSQP soft‑max refinement on 512 points ----
    if best_h is not None and best_h.size == 512:
        remaining = deadline - time.time()
        if remaining > 20.0:
            h = best_h.copy()
            for beta in (10., 40., 150., 500., 1500., 5000., 15000., 50000.):
                if time.time() >= deadline - 5.0:
                    break
                max_iter = 800 if beta < 2000 else 500
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c = h.copy(), c
                if best_c <= TARGET_C:
                    break
            # gradient tidy‑up
            if time.time() < deadline - 5.0:
                h = gradient_refine(h, rng, step=0.015, max_iter=15)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c = h.copy(), c

    # ---- 5. Upsample to 1024 if needed ----
    if best_h is not None and best_h.size < 1024:
        remaining = deadline - time.time()
        if remaining > 15.0:
            h_up = upsample_h(best_h, 1024)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = enforce_sum(h_up, rng)
            update_candidate(h_up)

    # ---- 6. Continuous SA on the finest grid ----
    if best_h is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            h_sa, c_sa = continuous_sa(best_h, rng, deadline)
            if c_sa < best_c:
                best_h, best_c = h_sa.copy(), c_sa
            # final gradient tidy‑up
            if time.time() < deadline - 2.0:
                h_sa = gradient_refine(best_h, rng, step=0.01, max_iter=10)
                c_sa = compute_c5(h_sa)
                if c_sa < best_c:
                    best_h, best_c = h_sa.copy(), c_sa

    # ---- Final sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
