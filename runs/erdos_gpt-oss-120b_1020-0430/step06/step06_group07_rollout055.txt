def build_basis(n, M):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    basis = [np.ones_like(x)]

        basis.append(np.cos(m * np.pi * x))
        basis.append(np.sin(m * np.pi * x))
    return np.vstack(basis)  # shape (dim, n)

    from scipy.optimize import differential_evolution

    start = time.time()
    total_budget = budget_s
    rng = np.random.default_rng(seed)

    # Helper functions defined above (project_box_sum, compute_c5, etc.)
    # (Will be defined outside but in same file.)

        h_initial = np.clip(h_initial, 0.0, 1.0)
        h_initial = project_box_sum(h_initial, n_fine / 2.0)
        init_c = compute_c5(h_initial)
        best_h = h_initial
        best_c = init_c
    else:
        # create random feasible fine
        best_h = random_feasible(n_fine, rng)
        best_c = compute_c5(best_h)

    # Coarse DE stage
    remaining = total_budget - (time.time() - start)
    if remaining > 20.0:

            return compute_c5(h)

        de_start = time.time()

            return (time.time() - de_start) >= de_time

            de_obj,

            strategy='best1bin',

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} via binary search on a shift."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def worst_gradient(h: np.ndarray) -> tuple:
    """Gradient of the worst shift overlap O_k = Σ h_i (1-h_{i+k})."""
    n = h.size
    # cross-correlation of h with itself; central part gives Σ h_i h_{i+k}
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))  # shift giving largest O_k
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def subgradient_optimize(h0: np.ndarray, target: float,
                         time_budget: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on max‑overlap (via worst shift)."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while time.time() - start < time_budget:
        t += 1
        grad, _ = worst_gradient(h)
        alpha = 0.5 / np.sqrt(t + 10)
        h_new = project_box_sum(h - alpha * grad, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
            if best_c <= 0.3808:
                break
        h = h_new
    return best_h, best_c


def build_basis(n: int, M: int) -> np.ndarray:
    """Fourier‑type basis (constant + M cosine + M sine) on [0,2)."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    rows = [np.ones_like(x)]
    for m in range(1, M + 1):
        rows.append(np.cos(m * np.pi * x))
        rows.append(np.sin(m * np.pi * x))
    return np.vstack(rows)  # shape (1+2M, n)


def h_from_coeff(a: np.ndarray, basis: np.ndarray) -> np.ndarray:
    """Construct h from coefficients, clip to [0,1] and enforce sum."""
    raw = np.dot(a, basis)
    raw = np.clip(raw, 0.0, 1.0)
    raw = project_box_sum(raw, raw.size / 2.0)
    return raw


def upsample_to(h: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from size len(h) to target_n with projection."""
    n_old = h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_box_sum(h_new, target_n / 2.0)
    return h_new
