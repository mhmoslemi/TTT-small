def project_onto_capped_simplex(v, target):
    n = v.shape[0]
    # Ensure target is between 0 and n
    assert 0 <= target <= n
    low = -np.max(v) - 1.0
    high = np.max(v)
    for _ in range(50):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # due to rounding may not exactly hit target, adjust by small correction distributing among entries not at bounds
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff equally to non-boundary entries
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
        else:
            # all at bounds, adjust smallest adjustment
            # find idx with smallest distance to bound
            idx = np.argmin(np.abs(x - (0.5)))  # arbitrary
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def compute_correlation(h):
    n = h.size
    size = 1 << (2*n-1).bit_length()  # next power of two
    # but for simplicity use size = 2*n
    size = 2*n
    h_fft = np.fft.rfft(h, n=size)
    rev = (1.0 - h)[::-1]
    rev_fft = np.fft.rfft(rev, n=size)
    conv = np.fft.irfft(h_fft * rev_fft, n=size)
    # conv length size; we need first 2*n-1 entries
    return conv[:2*n-1]

        target = n/2.0

    def compute_c5_corr(h):
        # returns (c5, corr)
        size = 2 * n
        h_f = np.fft.fft(h, size)
        rev_f = np.fft.fft((1.0 - h)[::-1], size)
        conv = np.fft.ifft(h_f * rev_f).real[:2*n-1]

# globals for the SLSQP soft‑max refinement
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    def compute_c5_corr(h):
        """Return (C5, full correlation) for current n."""
        size = 2 * n
        h_f = np.fft.fft(h, size)
        rev_f = np.fft.fft((1.0 - h)[::-1], size)

        c = np.max(conv) * (2.0 / n)
        return c, conv

            tau = cssv[cond][-1] / rho
        else:
            tau = 0.0
        w = np.clip(v - tau, 0.0, 1.0)
