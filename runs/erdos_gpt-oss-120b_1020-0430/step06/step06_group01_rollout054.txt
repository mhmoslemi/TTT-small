def project_sum_box(h, target):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < target:
        # need to add diff
        cap = 1 - h
        total_cap = cap.sum()
        if total_cap == 0:
            return h
        # add proportionally
        h += cap * ((target - s) / total_cap)
    else:
        # need to subtract diff
        # available amount to subtract is h (since lower bound 0)
        total_h = h.sum()
        # subtract proportionally
        h -= h * ((s - target) / total_h)
    # after this, some might exceed due to rounding? Clip again.
    h = np.clip(h, 0, 1)
    # S might still be marginally off due to numeric; we can do a small final correction
    s = h.sum()
    if abs(s - target) > 1e-9:
        # fallback to redistributive loop
        diff = target - s
        # random redistribution
        n = h.size
        while abs(diff) > 1e-12:
            i = rng.integers(0, n)
            add = min(diff, 1 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h
```

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_box(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust sum to target."""
    h = np.clip(h, 0.0, 1.0)

    if abs(s - target) < 1e-12:
        return h
    if s < target:
        cap = 1.0 - h
        total_cap = cap.sum()
        if total_cap > 0:
            h = h + cap * ((target - s) / total_cap)

        if total_h > 0:

    best_h = project_sum_box(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
