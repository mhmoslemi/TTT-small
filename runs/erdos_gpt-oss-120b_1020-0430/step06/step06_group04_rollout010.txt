import time, math, numpy as np

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5_fft(h, dx):
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    B = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * np.conj(B), m)[: 2 * n - 1]
    return float(np.max(corr) * dx)

def compute_c5_fft_pre(h, dx, m):
    H = np.fft.rfft(h, m)
    B = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * np.conj(B), m)[: 2 * h.shape[0] - 1]
    return float(np.max(corr) * dx)

def propose_move(h, rng):

    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    max_fwd = min(h[i], 1.0 - h[j])   # i -> j
    max_bwd = min(h[j], 1.0 - h[i])   # j -> i
    if max_fwd <= 0 and max_bwd <= 0:
        return None
    if max_fwd > 0 and max_bwd > 0:
        if rng.random() < 0.5:
            eps = rng.random() * max_fwd
            return i, j, -eps, eps
        else:
            eps = rng.random() * max_bwd
            return i, j, eps, -eps

        return i, j, -eps, eps
