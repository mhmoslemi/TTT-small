def compute_corr_fft(h):
    n = h.shape[0]
    size = 2 * n
    # rfft approach
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * G, n=size)
    return corr[:2*n-1]  # length 2n-1

def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear convolution h * (1-h) using FFT (full length)."""

    return np.max(corr) * (2.0 / h.size)


def enforce_sum_precise(h: np.ndarray, rng: np.random.Generator, target: float) -> np.ndarray:
    """Adjust h so that sum(h)=target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # fast adjustments on non‑boundary entries
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
        else:
            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # clip any tiny out‑of‑bounds values
    h[:] = np.clip(h, 0.0, 1.0)
    return h


def gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Subgradient of S_k = Σ h_i (1‑h_{i+k}) for the worst shift."""
    n = h.shape[0]
    if k >= 0:
        h_plus = np.concatenate((np.zeros(k), h[: n - k]))
        h_minus = np.concatenate((h[k:], np.zeros(k)))
    else:
        kk = -k
        h_plus = np.concatenate((h[kk:], np.zeros(kk)))
        h_minus = np.concatenate((np.zeros(kk), h[: n - kk]))
    return 1.0 - (h_plus + h_minus)


def subgradient_descent(h0: np.ndarray, rng: np.random.Generator,
                        max_iters: int, alpha0: float) -> tuple:
    """Subgradient descent on max‑overlap objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    alpha = alpha0

    for it in range(max_iters):
        k, _, _ = None, None, None
        # find worst shift
        corr = compute_corr_fft(h)
        idx = int(np.argmax(corr))
        k = idx - (n - 1)

        # gradient step
        g = gradient_for_shift(h, k)
        y = h - alpha * g
        y = np.clip(y, 0.0, 1.0)
        h_new = enforce_sum_precise(y, rng, target)

        # evaluate
        corr_new = compute_corr_fft(h_new)
        c_new = np.max(corr_new) * (2.0 / n)

        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new

        if c_new <= c:
            h, c = h_new, c_new
        else:
            alpha *= 0.5
            if alpha < 1e-6:
                alpha = alpha0

        # occasional stochastic perturbation
        if rng.random() < 0.01:
            eps = rng.random(n) * 0.02
            h_pert = np.clip(h + eps - eps.mean(), 0.0, 1.0)
            h_pert = enforce_sum_precise(h_pert, rng, target)
            c_pert = compute_c5(h_pert)
            if c_pert < c:
                h, c = h_pert, c_pert
                if c_pert < best_c:
                    best_h, best_c = h_pert.copy(), c_pert

    return best_h, best_c


# ---------- SLSQP soft‑max refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 200) -> np.ndarray:
    """SLSQP on the smoothed max‑overlap (softmax) objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact‑sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        # upsample to 1024 points if needed
        if init.size < 1024:
            factor = 1024 // init.size
            if 1024 % init.size == 0:
                init = np.tile(init, factor)
            else:
                reps = int(np.ceil(1024 / init.size))
                init = np.tile(init, reps)[:1024]
        n = init.size
        init = enforce_sum_precise(init, rng, n / 2.0)
        best_h = init.copy()
        best_c = compute_c5(best_h)
    else:
        n = 1024
        h0 = rng.random(n)
        best_h = enforce_sum_precise(h0, rng, n / 2.0)
        best_c = compute_c5(best_h)

    # ---- stochastic subgradient search ----
    time_limit = start + budget_s * 0.85
    max_iters_per = 400
    alpha0 = 0.5
    restarts = 0
    while time.time() < time_limit and restarts < 3000:
        # random start
        h_start = rng.random(n)
        h_start = enforce_sum_precise(h_start, rng, n / 2.0)
        h_opt, c_opt = subgradient_descent(h_start, rng, max_iters_per, alpha0)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt
        restarts += 1

    # ---- optional SLSQP polishing (if time permits) ----
    while time.time() < start + budget_s * 0.97:
        for beta in (500, 5000, 50000, 200000):
            if time.time() > start + budget_s * 0.97:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)
            best_c = compute_c5(best_h)

    # ---- final repair & return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_precise(best_h, rng, n / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
