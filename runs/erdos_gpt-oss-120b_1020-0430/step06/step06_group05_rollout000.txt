import time
import numpy as np
from scipy.optimize import minimize

# ---------- global vars for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ---------- problem specific helpers ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 30) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int) -> np.ndarray:
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


# ---------- GA helpers ----------
def repair_binary(b: np.ndarray, target: int, rng: np.random.Generator) -> np.ndarray:
    """Flip bits to achieve exact sum = target."""
    diff = int(b.sum() - target)
    if diff > 0:
        idx = rng.choice(np.flatnonzero(b), size=diff, replace=False)
        b[idx] = 0
    elif diff < 0:
        idx = rng.choice(np.flatnonzero(1 - b), size=-diff, replace=False)
        b[idx] = 1
    return b


def crossover_one_point(p1: np.ndarray, p2: np.ndarray, rng: np.random.Generator):
    n = p1.size
    cp = rng.integers(1, n)  # 1 … n‑1
    c1 = np.concatenate([p1[:cp], p2[cp:]])
    c2 = np.concatenate([p2[:cp], p1[cp:]])
    return c1, c2


def mutate_swap(b: np.ndarray, rng: np.random.Generator):
    """Swap a 1 with a 0 (preserves sum)."""
    ones = np.flatnonzero(b)
    zeros = np.flatnonzero(1 - b)
    if ones.size == 0 or zeros.size == 0:
        return b
    i = rng.choice(ones)
    j = rng.choice(zeros)
    b[i] = 0
    b[j] = 1
    return b

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # resolution
    n = 1024
    if n % 2:
        n += 1
    target = n // 2

    # GA parameters
    pop_size = 80
    elite_size = 2
    max_gens = 300
    tsize = 5
    pc = 0.7   # crossover prob
    pm = 0.3   # mutation prob

    # initialise population (binary)
    population = []
    # use parent construction if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        b0 = (h0 > 0.5).astype(np.uint8)
        b0 = repair_binary(b0, target, rng)
        population.append(b0)
    while len(population) < pop_size:
        b = np.zeros(n, dtype=np.uint8)
        idx = rng.choice(n, target, replace=False)
        b[idx] = 1
        population.append(b)

    # evaluate
    def fitness(b):
        return compute_c5(b.astype(np.float64))

    fitness_vals = np.array([fitness(b) for b in population])
    best_idx = np.argmin(fitness_vals)
    best_b = population[best_idx].copy()
    best_c = fitness_vals[best_idx]

    # GA loop
    for gen in range(max_gens):
        # time guard
        if time.time() - start > budget_s * 0.95:
            break

        # sort by fitness
        order = np.argsort(fitness_vals)
        new_pop = [population[i].copy() for i in order[:elite_size]]  # elitism
        # fill rest
        while len(new_pop) < pop_size:
            # tournament selection
            idx1 = rng.choice(pop_size, size=tsize, replace=False)
            p1 = population[idx1[np.argmin(fitness_vals[idx1])]]
            idx2 = rng.choice(pop_size, size=tsize, replace=False)
            p2 = population[idx2[np.argmin(fitness_vals[idx2])]]

            if rng.random() < pc:
                c1, c2 = crossover_one_point(p1, p2, rng)
            else:
                c1, c2 = p1.copy(), p2.copy()

            if rng.random() < pm:
                c1 = mutate_swap(c1, rng)
            if rng.random() < pm:
                c2 = mutate_swap(c2, rng)

            c1 = repair_binary(c1, target, rng)
            c2 = repair_binary(c2, target, rng)

            new_pop.append(c1)
            if len(new_pop) < pop_size:
                new_pop.append(c2)

        population = new_pop
        fitness_vals = np.array([fitness(b) for b in population])
        cur_best_idx = np.argmin(fitness_vals)
        cur_best_c = fitness_vals[cur_best_idx]
        if cur_best_c < best_c:
            best_c = cur_best_c
            best_b = population[cur_best_idx].copy()
            if best_c < 0.38079:
                break

    # ---------- continuous refinement ----------
    h = best_b.astype(np.float64)
    # gradient descent
    if time.time() - start < budget_s * 0.95:
        h = gradient_refine(h, target, rng, step=0.02, max_iter=30)
        best_c = compute_c5(h)

    # SLSQP soft‑max passes
    if time.time() - start < budget_s * 0.90:
        for beta in (500.0, 2000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.95:
                break
            h = slsqp_refine(h, target, rng, beta=beta, max_iter=300)
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c = cur_c

    # final enforcement
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    final_c = compute_c5(h)

    return h.tolist(), float(final_c), int(n)
