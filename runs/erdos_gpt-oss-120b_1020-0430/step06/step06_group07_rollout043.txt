def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0≤x≤1, sum x = target} via binary search on λ."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# --------------------------------------------------------------
def compute_c5_fft(h: np.ndarray) -> float:
    """Fast evaluation of C5 using FFT (linear correlation of h and 1‑h)."""
    n = h.size
    a = h
    b = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(A * B, n=L)[: 2 * n - 1]  # linear convolution
    return float(conv.max() * (2.0 / n))


# --------------------------------------------------------------
def anneal_optimize(h0: np.ndarray, rng: np.random.Generator,
                   time_budget: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing on the move “transfer ε from i to j”."""
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5_fft(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    start_temp = max(0.002, 0.05 * cur_c)   # initial temperature

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        temp = start_temp * (1.0 - elapsed / time_budget)  # linear cooling
        i = rng.integers(n)
        if h[i] <= 1e-12:
            continue
        j = rng.integers(n)
        if h[j] >= 1.0 - 1e-12:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * max(0.01, temp / start_temp)
        if eps <= 1e-12:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5_fft(h_new)
        delta = c_new - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / (temp + 1e-15)):
            h, cur_c = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
                if best_c <= 0.3808:
                    break
    return best_h, best_c


# --------------------------------------------------------------
def greedy_hill_climb(h0: np.ndarray, rng: np.random.Generator,
                      time_budget: float) -> tuple[np.ndarray, float]:
    """Greedy improvement: only accept moves that lower C5."""
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5_fft(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    while time.time() - start < time_budget:
        i = rng.integers(n)
        if h[i] <= 1e-12:
            continue
        j = rng.integers(n)
        if h[j] >= 1.0 - 1e-12:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.5
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5_fft(h_new)
        if c_new < cur_c:
            h, cur_c = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
                if best_c <= 0.3808:
                    break
    return best_h, best_c


# --------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- decide grid size -------------------------------------------------
    # 1024 points gives a finer bound if time permits
    n = 1024 if budget_s > 80 else 512

    # ---- initialise -------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
    else:
        h0 = rng.random(n)

    target = n / 2.0
    h0 = project_box_sum(h0, target)

    best_h, best_c = h0, compute_c5_fft(h0)

    # ---- multiple annealing restarts --------------------------------------
    remaining = budget_s - (time.time() - start)
    restarts = 3
    for r in range(restarts):
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining <= 5.0:
            break
        # diversify start point a bit
        h_start = best_h.copy()
        for _ in range(12):
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            max_eps = min(h_start[i], 1.0 - h_start[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * 0.5
            h_start[i] -= eps
            h_start[j] += eps
        h_start = project_box_sum(h_start, target)

        run_time = max(5.0, remaining / (restarts - r + 1))
        h_opt, c_opt = anneal_optimize(h_start, rng, run_time)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        if best_c <= 0.3808:
            break

    # ---- final greedy polishing -------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        h_opt, c_opt = greedy_hill_climb(best_h, rng, remaining)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- sanitise and return ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
