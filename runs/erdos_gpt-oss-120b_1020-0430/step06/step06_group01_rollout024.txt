import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.7):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.015, max_iter=10):
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = opt.minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa(n, rng, time_budget):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.2, 1e-4
    # start with random binary vector
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    ones_list = list(ones)
    zeros_list = [i for i in range(n) if h[i] == 0.0]
    dict_one = {i: idx for idx, i in enumerate(ones_list)}
    dict_zero = {i: idx for idx, i in enumerate(zeros_list)}
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    it = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # pick random 1 and 0
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        # try swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept

            idx_i = dict_one[i]
            idx_j = dict_zero[j]
            # update lists
            ones_list[idx_i] = j
            zeros_list[idx_j] = i
            # update dicts

            dict_zero.pop(j)
            dict_one[j] = idx_i
            dict_zero[i] = idx_j
            if c_new < best_c:

                if best_c <= 0.3808:
                    break
        else:

            h[j] = 0.0
        it += 1
        # occasional restart if no improvement for many steps
        if it % 50000 == 0 and time.time() - start < time_budget:
            # reinitialize
            h[:] = 0.0
            ones = rng.choice(n, n // 2, replace=False)
            h[ones] = 1.0
            ones_list = list(ones)
            zeros_list = [i for i in range(n) if h[i] == 0.0]
            dict_one = {i: idx for idx, i in enumerate(ones_list)}
            dict_zero = {i: idx for idx, i in enumerate(zeros_list)}
            c_curr = compute_c5(h)
    return best_h, best_c

    start = time.time()
    best_h = init_h.copy()
    best_c = compute_c5(best_h)
    # hill climb

    # SLSQP passes

        c = compute_c5(h)
        if c < best_c:

    # gradient refinement
    if time.time() - start < time_budget * 0.97:
        h = gradient_refine(best_h, rng, step=0.01, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
    # random restarts
    while time.time() - start < time_budget * 0.99:
        # perturb current best or random
        if rng.random() < 0.5:
            h = best_h.copy()
            for _ in range(10):
                i = rng.integers(0, n)
                delta = (rng.random() - 0.5) * 0.3
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = rng.random(n)
            h = enforce_sum(h, rng)
        # short hill climb

        h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.6)
        if c < best_c:

        # quick SLSQP

            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
    return best_h, best_c

        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa(n, rng, time_budget):
    start = time.time()
    best_h, best_c = None, np.inf
    T0, T_end = 0.2, 1e-4
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    ones_list = list(ones)
    zeros_list = [i for i in range(n) if h[i] == 0.0]
    dict_one = {i: idx for idx, i in enumerate(ones_list)}
    dict_zero = {i: idx for idx, i in enumerate(zeros_list)}
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    it = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        h[i], h[j] = 0.0, 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            idx_i = dict_one[i]
            idx_j = dict_zero[j]
            ones_list[idx_i] = j
            zeros_list[idx_j] = i
            dict_one.pop(i)
            dict_zero.pop(j)
            dict_one[j] = idx_i
            dict_zero[i] = idx_j
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= 0.3808:
                    break
        else:
            h[i], h[j] = 1.0, 0.0
        it += 1
        if it % 50000 == 0 and time.time() - start < time_budget:
            h[:] = 0.0
            ones = rng.choice(n, n // 2, replace=False)
            h[ones] = 1.0
            ones_list = list(ones)
            zeros_list = [i for i in range(n) if h[i] == 0.0]
            dict_one = {i: idx for idx, i in enumerate(ones_list)}
            dict_zero = {i: idx for idx, i in enumerate(zeros_list)}
            c_curr = compute_c5(h)
    return best_h, best_c

def continuous_optimize(n, rng, time_budget, init_h):
    start = time.time()
    best_h, best_c = init_h.copy(), compute_c5(init_h)
    max_iters = int(min(50000, time_budget * 400))
    h, c = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.7)
    best_h, best_c = h, c
    beta_seq = (10., 30., 80., 200., 800., 3000., 8000., 20000., 50000.)
    for beta in beta_seq:
        if time.time() - start > time_budget * 0.95:
            break
        max_iter = 2000 if beta < 200 else 1200
        h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
    if time.time() - start < time_budget * 0.97:
        h = gradient_refine(best_h, rng, step=0.01, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
    while time.time() - start < time_budget * 0.99:
        if rng.random() < 0.5:
            h = best_h.copy()
            for _ in range(10):
                i = rng.integers(0, n)
                delta = (rng.random() - 0.5) * 0.3
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = rng.random(n)
            h = enforce_sum(h, rng)
        max_iters = int(min(15000, (time_budget - (time.time() - start)) * 300))
        h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.6)
        if c < best_c:
            best_h, best_c = h.copy(), c
        for beta in (30., 80., 200.):
            if time.time() - start > time_budget * 0.99:
                break
            h = slsqp_optimize(best_h, beta, rng, max_iter=500)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s
    n = 1024
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        init_h = np.interp(x_new, x_old, h0)
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = enforce_sum(init_h, rng)
    else:
        init_h = rng.random(n)
        init_h = enforce_sum(init_h, rng)
    binary_time = total_budget * 0.15
    h_bin, c_bin = binary_sa(n, rng, binary_time)
    if c_bin < compute_c5(init_h):
        best_h, best_c = h_bin.copy(), c_bin
    else:
        best_h, best_c = init_h.copy(), compute_c5(init_h)
    remaining = total_budget - (time.time() - start)
    if remaining > 0:
        h_cont, c_cont = continuous_optimize(n, rng, remaining, best_h)
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont
    while time.time() - start < total_budget * 0.999:
        grad = worst_gradient(best_h)
        i, j = int(np.argmax(grad)), int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
