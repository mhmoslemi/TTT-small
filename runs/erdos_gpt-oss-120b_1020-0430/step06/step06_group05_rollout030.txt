def proj_capped_simplex(v, target, eps=1e-12, max_iter=100):
    lo = -np.max(v) - 1.0   # ensure v - lo > 1
    hi = np.max(v)           # ensure v - hi <= 0
    for _ in range(max_iter):

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _project_capped_simplex(v: np.ndarray, target: float, eps: float = 1e-12,
                            max_iter: int = 80) -> np.ndarray:
    """Project v onto {0<=x<=1, sum(x)=target} by bisection on a threshold."""
    lo = -np.max(v) - 1.0          # makes v-lo > 1 for all entries
    hi = np.max(v)                 # makes v-hi <= 0 for all entries
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
        if hi - lo < eps:
            break
    return np.clip(v - lo, 0.0, 1.0)


def _compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1-h) using FFT (length 2n-1)."""
    n = h.shape[0]
    m = _next_pow2(2 * n - 1)
    f = np.fft.rfft(h, n=m)
    g = np.fft.rfft((1.0 - h)[::-1], n=m)
    conv = np.fft.irfft(f * g, n=m)[: 2 * n - 1]
    return conv[::-1]  # same ordering as np.correlate(full)


def _compute_c5_fft(h: np.ndarray) -> float:
    """Discrete C5 value for step function h on [0,2]."""
    corr = _compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


def _worst_shift_and_grad(h: np.ndarray):
    """
    Return k (non‑negative worst shift), gradient for that shift,
    and the current C5 value.
    """
    n = h.shape[0]
    corr = _compute_corr_fft(h)
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    cur_c = np.max(corr) * (2.0 / n)
    return k, grad, cur_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # problem size – larger gives a tighter bound, still cheap with FFT
    n = 2048
    target_sum = n / 2.0

    # ----- initialise h ----------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        init = np.asarray(init, dtype=float)
        if init.size != n:
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, init)
        else:
            h = init.copy()
        h = np.clip(h, 0.0, 1.0)
        h = _project_capped_simplex(h, target_sum)
    else:
        h = rng.random(n)
        h = _project_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_c = _compute_c5_fft(best_h)

    # ----- main subgradient loop -------------------------------------------
    step0 = 2.0               # initial step magnitude
    it = 0
    cur_c = best_c
    while True:
        if time.time() - start > budget_s * 0.99:
            break
        if best_c < 0.3808:
            break

        k, grad, cur_c = _worst_shift_and_grad(h)

        # step size ~ 1/sqrt(iter) scaled by gradient norm
        grad_norm = np.linalg.norm(grad) + 1e-12
        step = step0 / (grad_norm * np.sqrt(it + 1))

        v = h - step * grad
        h_new = _project_capped_simplex(v, target_sum)
        c_new = _compute_c5_fft(h_new)

        # keep the best ever seen
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

        # accept if improvement (strict)
        if c_new < cur_c - 1e-12:
            h = h_new
            cur_c = c_new
        else:
            # occasional uphill move to escape plateaus
            if rng.random() < 0.001:
                h = h_new
                cur_c = c_new

        # occasional random perturbation to inject diversity
        if it % 5000 == 0 and it > 0:
            pert = h + rng.normal(scale=0.02, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = _project_capped_simplex(pert, target_sum)
            c_pert = _compute_c5_fft(pert)
            if c_pert < best_c:
                best_c = c_pert
                best_h = pert.copy()
            if c_pert < cur_c:
                h = pert
                cur_c = c_pert

        it += 1

    # ----- final polishing (tiny gradient descent) ------------------------
    # a few extra refinement steps with a reduced step size
    for _ in range(12):
        k, grad, _ = _worst_shift_and_grad(best_h)
        step = 0.05 / (np.linalg.norm(grad) + 1e-12)
        v = best_h - step * grad
        best_h = _project_capped_simplex(v, target_sum)

    final_c = _compute_c5_fft(best_h)

    # ensure feasibility one last time
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_capped_simplex(best_h, target_sum)
    final_c = _compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
