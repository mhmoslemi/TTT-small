def compute_c5_corr(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    return max_val * (2.0 / n), max_idx, corr
```

def compute_c5_corr_fft(h):
    n = len(h)
    # zero pad to length 2n
    size = 1
    while size < 2*n:
        size <<= 1
    H = np.fft.rfft(h, size)
    Hc = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(H * Hc, size)[:2*n-1]
    max_idx = np.argmax(corr)
    c = corr[max_idx] * (2.0 / n)

import time, math, numpy as np

def compute_c5_corr(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    c = float(corr[max_idx] * (2.0 / n))
    return c, max_idx, corr

def project_onto_box_sum(v, target):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

                kk = -k
                grad[kk:] -= h[: n - kk]
                grad[: n - kk] -= h[kk:]
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        eps = max_eps
        improved = False
        while eps > 1e-12:
            h[i] -= eps
            h[j] += eps
            c_new, max_idx_new, _ = compute_c5_corr(h)
            if c_new <= c:
                c, max_idx = c_new, max_idx_new
                improved = True
                if c < best_c:
                    best_c, best_h = c, h.copy()
                break
            h[i] += eps
            h[j] -= eps
            eps *= 0.5
        if not improved:
            break

    # ---------------------------------------------------------
    # Final projection (ensure exact constraints)
    # ---------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5_corr(best_h)[0]

    return best_h.tolist(), float(final_c), int(best_h.size)
