def binary_sa(n, rng, time_budget):
    start = time.time()
    h = np.zeros(n, dtype=float)
    # randomly set n//2 ones
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    T0, T_end = 0.5, 1e-5  # high temperature
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # sample a random pair i,j with opposite values
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h)
        delta = c_new - best_c
        if c_new < best_c or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_c
```

    if target is None:
        target = h.size / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # find indices with room
    if diff > 0:
        # need to add diff
        candidates = np.where(h < 1.0)[0]
        if candidates.size == 0:
            return h  # cannot adjust

        i = rng.choice(candidates)

def project_sum_bounds(v, target, n_iter=50):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(n_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def worst_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1 - 2 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def binary_sa(n, rng, time_budget):
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    c_curr = best_c
    T0, T_end = 0.5, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

def project_to_feasible(v):

    # initial clipping
    v = np.clip(v, 0.0, 1.0)

        s = proj.sum()

            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np

# ------------------ core utilities ------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the feasible set:
        0 ≤ v_i ≤ 1,   sum(v) = n/2
    using a binary search on the Lagrange multiplier.
    """
    n = v.size
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    lo = np.min(v) - 1.0   # ensures sum > target
    hi = np.max(v)         # ensures sum < target
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift (maximal correlation)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ------------------ binary simulated annealing ------------------

def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Anneal a binary (0/1) vector with sum = n/2."""
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    c_curr = best_c
    T0, T_end = 0.5, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c


# ------------------ gradient descent optimizer ------------------

def gradient_descent_optimize(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Projected gradient descent using the worst‑shift gradient."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    lr = 0.02
    it = 0
    while time.time() - start < time_budget:
        grad = worst_gradient(h)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if c_new < c_curr:
            h, c_curr = h_new, c_new
        else:
            if rng.random() < 0.001:
                h, c_curr = h_new, c_new
        it += 1

    return best_h, best_c


# ------------------ guided hill climb (final tidy‑up) ------------------

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Local improvement using the worst‑shift gradient (mass swaps)."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ------------------ upsampling helper ------------------

def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n samples."""
    old_n = h.size
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
