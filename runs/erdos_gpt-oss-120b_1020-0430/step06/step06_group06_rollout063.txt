def project_onto_simplex(v, target):
    # v: ndarray float
    # returns x = np.clip(v + lam, 0, 1) such that sum(x) = target.
    lo = -np.max(v)  # ensures v+lo <=0
    hi = 1 - np.min(v)  # ensures v+hi >=1
    # binary search for lam
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0, 1)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v + (lo+hi)/2.0, 0, 1)
    # due to numeric issues, adjust small diff
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # adjust with greedy: add diff to entries not at bounds
        # diff may be small - apply to random indices
        idx = np.where((x > 1e-12) & (x < 1-1e-12))[0]
        if idx.size > 0:
            # equally distribute
            inc = diff / idx.size
            x[idx] += inc
            # clip again
            x = np.clip(x, 0, 1)
        else:
            # fallback, adjust any
            x[0] = np.clip(x[0] + diff, 0, 1)
    return x

import time
import numpy as np

# ---------- projection onto the box-simplex ----------
def _proj_simplex(v, target):
    """Project v onto {x in [0,1]^n, sum x = target} using a bisection on lambda."""
    lo = -np.max(v)          # makes all entries <=0
    hi = 1.0 - np.min(v)      # makes all entries >=1
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny numerical fix
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
        else:
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

# ---------- main routine ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---- initialise h ----
    DESIRED_N = 1024
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n0 = h0.size
        if DESIRED_N % n0 == 0:
            factor = DESIRED_N // n0
            h = np.repeat(h0, factor)
            n = DESIRED_N
        else:
            h = h0.copy()
            n = n0
    else:
        n = DESIRED_N
        target = n / 2.0
        v = rng.random(n)
        h = _proj_simplex(v, target)

    target = n / 2.0
    if abs(h.sum() - target) > 1e-12:
        h = _proj_simplex(h, target)

    size_fft = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n-1

    # ---- helper to compute C5 from h (via auto‑correlation) ----
    def _c5_from_h(vec):
        fft_h = np.fft.fft(vec, size_fft)
        conv = np.real(np.fft.ifft(fft_h * np.conj(fft_h)))
        conv_d = conv[n - 1 : n - 1 + n]
        return (target - conv_d.min()) * 2.0 / n

    best_h = h.copy()
    best_c = _c5_from_h(best_h)

    # ---- gradient descent on smooth‑max objective ----
    betas = [2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0,
             1000.0, 2000.0, 4000.0]
    eval_every = 5
    outer_break = False

    for beta in betas:
        step = max(0.5 / beta, 1e-4)
        max_iters = 800
        for it in range(max_iters):
            if time.time() - start > budget_s - 5.0:
                outer_break = True
                break

            # ----- correlation of current h -----
            fft_h = np.fft.fft(h, size_fft)
            conv_full = np.real(np.fft.ifft(fft_h * np.conj(fft_h)))
            conv_d = conv_full[n - 1 : n - 1 + n]          # length n
            O_d = target - conv_d                         # overlap values (unscaled)

            # ----- soft‑max weights -----
            betaO = beta * O_d
            max_betaO = np.max(betaO)
            w = np.exp(betaO - max_betaO)
            w /= w.sum()

            # ----- symmetric kernel for gradient -----
            w_sym = np.zeros(size_fft, dtype=np.float64)
            w_sym[n - 1 :] = w
            w_sym[: n - 1] = w[1:][::-1]                 # mirror (skip centre)

            # ----- gradient g = 1 - (h * w_sym) -----
            fft_w = np.fft.fft(w_sym, size_fft)
            conv_hw = np.real(np.fft.ifft(fft_h * fft_w))
            g = 1.0 - conv_hw[n - 1 : n - 1 + n]

            # ----- gradient step & projection -----
            h_new = h - step * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h = _proj_simplex(h_new, target)

            # ----- occasional evaluation -----
            if (it % eval_every) == 0:
                cur_c = _c5_from_h(h)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if outer_break:
            break

    # ---- final local random refinement (use remaining time) ----
    while time.time() - start < budget_s - 1.0:
        cand = best_h + rng.normal(scale=0.02, size=n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = _proj_simplex(cand, target)
        cur_c = _c5_from_h(cand)
        if cur_c < best_c:
            best_c = cur_c
            best_h = cand.copy()

    # ---- final repair and C5 computation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _proj_simplex(best_h, target)
    final_c = _c5_from_h(best_h)

    return best_h.tolist(), float(final_c), int(n)
