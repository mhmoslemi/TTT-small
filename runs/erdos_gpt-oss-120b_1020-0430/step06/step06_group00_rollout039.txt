def softmax_grad(x, beta):
    N = x.size
    a = 1.0 - x
    c = np.correlate(x, a, mode='full')
    maxc = np.max(c)
    expb = np.exp(beta * (c - maxc))
    Z = np.sum(expb)
    w = expb / Z
    term1 = np.zeros(N)
    term2 = np.zeros(N)
    # iterate over s_idx with weight
    # To speed up, maybe use vectorized operations using slicing for all s >=0 and s<0 handle in two loops albeit 2N-1 loops; that's okay.
    # Instead we could precompute contributions for all s >=0 using vectorized sliding window with stride tricks; but loop is fine.
    for s_idx, ws in enumerate(w):
        if ws == 0.0:
            continue
        s = s_idx - (N - 1)
        if s >= 0:
            term1[:N - s] += ws * a[s:]
            term2[s:] += ws * x[:N - s]
        else:
            t = -s
            term1[t:] += ws * a[:N - t]
            term2[:N - t] += ws * x[t:]
    grad = term1 - term2
    return grad

def softmax_grad_fft(x, beta):
    N = x.size
    a = 1.0 - x
    c = np.correlate(x, a, mode='full')
    maxc = np.max(c)
    expb = np.exp(beta * (c - maxc))
    Z = np.sum(expb)
    w = expb / Z
    # Reverse w
    w_rev = w[::-1]
    # Determine size for FFT: at least len(w_rev) + len(a) - 1 = (2N-1) + N - 1 = 3N-2
    nfft = 1 << ((3 * N - 2 - 1).bit_length())  # next power of 2
    # Compute FFTs
    W = np.fft.fft(w_rev, nfft)
    A = np.fft.fft(a, nfft)
    X = np.fft.fft(x, nfft)
    conv_a = np.fft.ifft(W * A).real
    conv_x = np.fft.ifft(W * X).real
    # Extract valid part: indices from N-1 to N-1+N-1 = 2N-2 inclusive => length N
    start = N - 1
    end = start + N
    term1 = conv_a[start:end]
    term2 = conv_x[start:end]
    grad = term1 - term2
    return grad

def slsqp_optimize_softmax(h0, beta, max_iter, rng):
    N = h0.size
    global _GLOBAL_TARGET
    _GLOBAL_TARGET = N / 2.0
    bounds = [(0.0, 1.0)] * N
    cons = {'type': 'eq', 'fun': _sum_eq_constraint, 'jac': _sum_eq_jac}
    res = minimize(softmax_obj, h0, args=(beta,), jac=softmax_grad,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum EXACT
    diff = _GLOBAL_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution as before
        n = N
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = _GLOBAL_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Global for equality constraint
_GLOBAL_TARGET = 0.0


def _sum_eq_constraint(x):
    return np.sum(x) - _GLOBAL_TARGET


def _sum_eq_jac(x):
    return np.ones_like(x)


# ------------------------------------------------------------
# Core utilities
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/N)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] = 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 15) -> np.ndarray:
    """Iterative refinement using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

# Binary simulated annealing (0/1 vectors)
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Binary SA with exactly n/2 ones."""
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c

# Soft‑max objective (with FFT gradient)
def softmax_obj(x: np.ndarray, beta: float) -> float:
    c = np.correlate(x, 1.0 - x, mode="full")
    maxc = np.max(c)
    return maxc + np.log(np.sum(np.exp(beta * (c - maxc)))) / beta


def softmax_grad(x: np.ndarray, beta: float) -> np.ndarray:
    N = x.size
    a = 1.0 - x
    c = np.correlate(x, a, mode="full")
    maxc = np.max(c)
    expb = np.exp(beta * (c - maxc))
    Z = np.sum(expb)
    w = expb / Z
    w_rev = w[::-1]
    nfft = 1 << ((3 * N - 2 - 1).bit_length())
    W = np.fft.fft(w_rev, nfft)
    A = np.fft.fft(a, nfft)
    X = np.fft.fft(x, nfft)
    conv_a = np.fft.ifft(W * A).real
    conv_x = np.fft.ifft(W * X).real
    start = N - 1
    term1 = conv_a[start:start + N]
    term2 = conv_x[start:start + N]
    return term1 - term2


def slsqp_optimize_softmax(h0: np.ndarray, beta: float,
                          max_iter: int, rng: np.random.Generator) -> np.ndarray:
    N = h0.size
    global _GLOBAL_TARGET
    _GLOBAL_TARGET = N / 2.0
    bounds = [(0.0, 1.0)] * N
    cons = {"type": "eq", "fun": _sum_eq_constraint, "jac": _sum_eq_jac}
    res = minimize(softmax_obj, h0, args=(beta,), jac=softmax_grad,
                   method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = _GLOBAL_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, N)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = _GLOBAL_TARGET - x.sum()
    return x
