    start = time.time()

    m = n//2
    best_c = np.inf
    best_a = None

    # function definitions: compute_c5, softmax_gradient, compute_c5_a, worst_gradient_a

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discretised step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))


def softmax_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the soft‑max approximation of max_k corr_k."""
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    Omax = O.max()
    w = np.exp(beta * (O - Omax))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum

    grad = np.zeros_like(h)
    wk = w[n - 1]                      # k = 0
    grad += wk * (1.0 - 2.0 * h)

    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk                     # add wk to all entries
        grad[: n - k] -= wk * h[k:]    # subtract wk*h[i+k] where i+k < n
        grad[k:] -= wk * h[: n - k]    # subtract wk*h[i-k] where i-k >= 0
    return grad


def compute_c5_a(a: np.ndarray) -> float:
    """C5 for a symmetric function built from a."""
    h = np.concatenate([a, 1.0 - a[::-1]])
    return compute_c5(h)


def worst_gradient_a(a: np.ndarray) -> np.ndarray:
    """Gradient of the worst (largest) correlation w.r.t. a."""
    m = a.size
    n = m * 2
    h = np.concatenate([a, 1.0 - a[::-1]])
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)

    grad_h = np.ones_like(h)
    if k == 0:
        grad_h[:] = 1.0 - 2.0 * h
    else:
        grad_h[: n - k] -= h[k:]
        grad_h[k:] -= h[: n - k]

    # map back to a
    grad_a = grad_h[:m] - grad_h[-1:-m - 1:-1]
    return grad_a


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # discretisation size (must be even for our symmetry)
    n = 256
    m = n // 2

    # initialise best record
    best_c = np.inf
    best_a = None

    # build an initial a – if the parent solution exists we borrow its first half
    if 'initial_h_values' in globals():
        ih = np.asarray(initial_h_values, dtype=float)
        if ih.size >= m:
            a0 = ih[:m].copy()
        else:
            a0 = rng.random(m)
    else:
        a0 = rng.random(m)

    # schedule of smoothness (beta) and iterations per beta
    beta_schedule = [(5, 1500), (20, 1500), (100, 1500),
                     (500, 1500), (2000, 1500), (8000, 1500), (20000, 1500)]

    # Main time‑bounded loop: multiple restarts
    while time.time() - start < budget_s * 0.95:
        # start from a fresh point (occasionally reuse a0)
        if rng.random() < 0.3 and best_a is not None:
            a = best_a.copy()
        else:
            a = a0.copy()
            # randomise a a little to avoid identical starts
            a += rng.normal(scale=0.02, size=m)
            a = np.clip(a, 0.0, 1.0)

        # Adam state
        mt = np.zeros_like(a)
        vt = np.zeros_like(a)
        t_global = 0

        # soft‑max descent
        for beta, iters in beta_schedule:
            for _ in range(iters):
                if time.time() - start > budget_s * 0.95:
                    break
                h = np.concatenate([a, 1.0 - a[::-1]])
                grad_h = softmax_gradient(h, beta)
                grad_a = grad_h[:m] - grad_h[-1:-m - 1:-1]

                t_global += 1
                mt = 0.9 * mt + 0.1 * grad_a
                vt = 0.999 * vt + 0.001 * (grad_a ** 2)

                m_hat = mt / (1.0 - 0.9 ** t_global)

                a = a - step
                a = np.clip(a, 0.0, 1.0)

            a_local[i] += eps

            c_now = compute_c5_a(a_local)

                best_a = a_local.copy()

    # final symmetric function
    best_h = np.concatenate([best_a, 1.0 - best_a[::-1]])
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
