def project_onto_sum_box(x, target, tol=1e-9, max_iter=100):
    lo = -1.0  # lower bound for lambda
    hi = 1.0   # upper bound for lambda
    # Actually lambda can be between -max(x) (to push all to zero) and + (1 - min(x)) (to push to one). Use lo = -max(x) -1; hi = 1 - min(x) +1 to be safe.
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo+hi)/2.0, 0.0, 1.0)

def compute_c5_fast(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    # compute (1-h) reversed
    b_rev = (1.0 - h)[::-1]
    # FFT
    H = np.fft.fft(h, L)
    B = np.fft.fft(b_rev, L)
    conv = np.fft.ifft(H * B).real
    # correlation length 2n-1
    corr = conv[:2*n-1][::-1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

Note: conv values may have small imaginary parts due to floating errors; we use .real.

def smooth_obj_and_grad(h, beta, target, L_full, L_grad):
    # h length n
    n = h.size
    # compute conv_hh for overlaps
    H = np.fft.fft(h, L_full)
    conv_hh = np.fft.ifft(H * H).real[:2*n-1]  # length 2n-1
    overlaps = target - conv_hh
    # compute smooth max
    max_ov = np.max(overlaps)
    exp_term = np.exp(beta * (overlaps - max_ov))
    sum_exp = exp_term.sum()
    lse = max_ov + np.log(sum_exp)
    val = lse / beta

    # weights
    w = exp_term / sum_exp  # length 2n-1

    # gradient
    # conv1 = w * h (conv)
    conv1_full = np.fft.ifft(np.fft.fft(w, L_grad) * np.fft.fft(h, L_grad)).real
    # extract conv1_i
    start = n - 1
    conv1_i = conv1_full[start:start + n]

    # b = 1 - h
    b = 1.0 - h
    w_rev = w[::-1]
    conv2_full = np.fft.ifft(np.fft.fft(w_rev, L_grad) * np.fft.fft(b, L_grad)).real
    conv2_i = conv2_full[start:start + n]

    grad = 1.0 - (conv1_i + conv2_i)

    return val, grad

def adam_optimize(h, beta_seq, target, L_full, L_grad, rng, time_budget, start_time, total_budget):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for beta in beta_seq:
        # compute steps allowed based on remaining time
        # step count maybe 5000 or less depending.
        max_steps = 5000
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        for step in range(1, max_steps+1):
            if time.time() - start_time > total_budget * 0.95:
                return best_h, best_c
            val, grad = smooth_obj_and_grad(h, beta, target, L_full, L_grad)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            lr = 0.2 / np.sqrt(beta)  # might scale with beta
            step_h = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step_h
            # Project onto [0,1] and sum using project_onto_sum_box
            h = project_onto_sum_box(h, target)
            # Periodically evaluate
            if step % 500 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
                # early exit
                if best_c <= 0.3808:
                    return best_h, best_c
        # after finishing beta, evaluate again.
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap C5 for step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst shift overlap (sub‑gradient of C5)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def project_onto_sum_box(x: np.ndarray, target: float,
                         tol: float = 1e-12, max_iter: int = 50) -> np.ndarray:
    """Projection onto {0≤x_i≤1, sum x_i = target} using a bisection on the Lagrange multiplier."""
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + mid, 0.0, 1.0)

# ---------- smooth‑max objective with Adam ----------
def smooth_obj_and_grad(h: np.ndarray, beta: float, target: float,
                        L_full: int, L_grad: int):
    n = h.size
    # overlaps via convolution h*h (zero‑padded)
    H = np.fft.fft(h, L_full)
    conv_hh = np.fft.ifft(H * H).real[: 2 * n - 1]
    overlaps = target - conv_hh
    max_ov = np.max(overlaps)
    exp_term = np.exp(beta * (overlaps - max_ov))
    sum_exp = exp_term.sum()
    lse = max_ov + np.log(sum_exp)
    val = lse / beta

    w = exp_term / sum_exp          # weights, length 2n‑1
    # gradient: 1 - (conv(w,h) + conv(w_rev,1‑h)) at appropriate slice
    start = n - 1
    conv1_full = np.fft.ifft(np.fft.fft(w, L_grad) * np.fft.fft(h, L_grad)).real
    conv1_i = conv1_full[start : start + n]

    b = 1.0 - h
    w_rev = w[::-1]
    conv2_full = np.fft.ifft(np.fft.fft(w_rev, L_grad) * np.fft.fft(b, L_grad)).real
    conv2_i = conv2_full[start : start + n]

    grad = 1.0 - (conv1_i + conv2_i)
    return val, grad

                  L_full: int, L_grad: int,
                  rng: np.random.Generator,
                  start_time: float, total_budget: float):

    for beta in beta_seq:
        max_steps = 2500
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        for t in range(1, max_steps + 1):
            if time.time() - start_time > total_budget * 0.95:
                return best_h, best_c
            _, grad = smooth_obj_and_grad(h, beta, target, L_full, L_grad)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.2 / (np.sqrt(beta) + 1e-12)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            h = project_onto_sum_box(h, target)
            if t % 500 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if best_c <= 0.3808:
            break
    return best_h, best_c
