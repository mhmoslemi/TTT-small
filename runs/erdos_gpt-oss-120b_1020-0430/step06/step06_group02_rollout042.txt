import time
import numpy as np
from scipy.optimize import minimize

# -------------------- utilities -----------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_sum_and_bounds(y: np.ndarray, target: float) -> np.ndarray:
    """Project y onto {0 <= x <= 1, sum x = target} by a simple bisection."""
    lo = -np.max(y)          # makes y+lo <= 0 -> sum 0
    hi = 1.0 - np.min(y)     # makes y+hi >= 1 -> sum n
    for _ in range(30):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the max overlap for the current worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # signed shift giving worst value
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]      -= h[: n - k]
    return grad


# ---------------------- Adam optimizer ---------------------------------

def adam_optimize(h0: np.ndarray, target: float, max_iters: int,
                  time_deadline: float, stop_c5: float = 0.3808):
    """Projected Adam on the worst‑shift objective."""
    h = project_onto_sum_and_bounds(np.clip(h0, 0.0, 1.0), target)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    while t < max_iters:
        t += 1
        if time.time() > time_deadline:
            break
        # occasional evaluation to track improvements
        if t % 10 == 0:
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
                if best_c <= stop_c5:
                    break
        g = worst_gradient(h)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_sum_and_bounds(h, target)
    # final check
    final_c = compute_c5(h)
    if final_c < best_c:
        best_c, best_h = final_c, h.copy()
    return best_h, best_c


# ---------------------- SLSQP with soft‑max ---------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1500) -> np.ndarray:
    """Run SLSQP on the soft‑max objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact projection to fix any tiny sum error
    x = project_onto_sum_and_bounds(x, CURRENT_TARGET)
    return x

                break

    # ---- final sanity check --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_sum_and_bounds(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
