def deterministic_hill_climb(h, target, max_outer_iters=1000):
    h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_outer_iters):
        grad = worst_gradient(h)  # gradient of Omax for worst shift.
        i = int(np.argmax(grad))  # Largest gradient -> smallest s_i
        j = int(np.argmin(grad))  # Smallest gradient -> largest s_j
        if i == j:
            break
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            break
        # Move full
        h[i] -= max_eps
        h[j] += max_eps
        # Project sum may be off? But we moved mass within same total, so sum unchanged.
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
        else:
            # If not improved, revert and break
            h[i] += max_eps
            h[j] -= max_eps
            break
    return h, best_c

import time, math, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def deterministic_hill_climb_full(h, target, max_iters=1000):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for _ in range(max_iters):
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        eps = min(best_h[i], 1.0 - best_h[j])
        if eps <= 0.0:
            break
        best_h[i] -= eps
        best_h[j] += eps
        c = compute_c5(best_h)
        if c < best_c - 1e-12:
            best_c = c
        else:
            # revert and stop
            best_h[i] += eps
            best_h[j] -= eps
            break
    return best_h, best_c

def softmax_gradient(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def softmax_descent(h, beta, target, iters, rng, step0=0.5):
    n = h.size
    for it in range(iters):
        grad = softmax_gradient(h, beta)
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        step = step0 * (1.0 - it / iters) / norm
        v = h - step * grad
        h = project_onto_box_sum(v, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # Initialise problem size and target sum
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # 1) strong deterministic hill climb on the starting point
    best_h, best_c = deterministic_hill_climb_full(best_h, target, max_iters=500)

    # 2) softmax based gradient descent with a rich beta schedule
    beta_schedule = [20, 100, 500, 2000, 8000, 20000, 50000, 100000]
    for beta in beta_schedule:
        if time.time() - start > budget_s * 0.78:
            break
        iters = int(600 * 256 / n)
        iters = max(iters, 120)
        step0 = 0.5 if beta < 2000 else 0.2
        best_h = softmax_descent(best_h, beta, target, iters, rng, step0=step0)
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c

    # 3) final deterministic polishing
    best_h, best_c = deterministic_hill_climb_full(best_h, target, max_iters=800)

    # 4) random restarts with quick deterministic polishing
    while time.time() - start < budget_s * 0.95:

        best_c = compute_c5(best_h)

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
