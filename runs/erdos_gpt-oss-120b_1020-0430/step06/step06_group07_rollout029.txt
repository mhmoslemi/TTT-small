import time
import numpy as np

# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} using binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


# ------------------------------------------------------------
def softmax_gradient_descent(
    h0: np.ndarray,
    beta_schedule,
    max_iters_per_beta: int,
    rng: np.random.Generator,
    start_time: float,
    total_budget: float,
):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    # best so far
    overlap = np.correlate(h, h, mode="full")[n - 1 :]
    O = target - overlap
    best_c = O.max() * (2.0 / n)
    best_h = h.copy()
    global_iter = 0

    for beta in beta_schedule:
        for _ in range(max_iters_per_beta):
            if time.time() - start_time > total_budget - 0.5:
                return best_h, best_c

            # overlap & O
            overlap = np.correlate(h, h, mode="full")[n - 1 :]
            O = target - overlap
            c = O.max() * (2.0 / n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                # early stop if already very good
                if best_c <= 0.3808:
                    pass

            # soft‑max weights
            max_O = O.max()
            log_w = beta * (O - max_O)
            w = np.exp(log_w)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / n)
            else:
                w /= w_sum

            # gradient of smooth max
            g = np.zeros_like(h)
            w0 = w[0]
            g -= 2.0 * w0 * h
            # vectorised slice loop (O(n²) work but fast in C)
            for k in range(1, n):
                wk = w[k]
                if wk == 0.0:
                    continue
                g[: n - k] -= wk * h[k:]
                g[k :] -= wk * h[: n - k]

            # step size
            global_iter += 1
            alpha = 0.5 / np.sqrt(global_iter)

            h = project_box_sum(h - alpha * g, target)

    return best_h, best_c


# ------------------------------------------------------------
def worst_gradient(h: np.ndarray):
    """Gradient of the worst (largest) overlap shift."""
    n = h.size
    cross = np.correlate(h, h, mode="full")[n - 1 :]  # length n
    k = int(np.argmin(cross))
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    step_scale: float = 0.6,
):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ------------------------------------------------------------
def upsample_h(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    n2 = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    return project_box_sum(h2, n2 / 2.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    beta_sched = [1.0, 5.0, 20.0, 80.0, 300.0, 1200.0, 5000.0]

    # ---- guided hill‑climb -----------------------------------------

    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
