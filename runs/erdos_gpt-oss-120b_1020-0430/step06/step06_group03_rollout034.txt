def softmax_gradient(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad
```

import time, numpy as np, scipy.signal as signal

def project_onto_box_sum(v, target):
    # water-filling via bisection
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        lam = (lo+hi)/2
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # adjust small residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1-1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.argmin(np.abs(v - (lo+hi)/2))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def compute_corr(h):
    # convolution of h and reversed (1-h)
    return signal.fftconvolve(h, (1.0 - h)[::-1], mode='full')

def compute_c5(h):
    N = h.size
    corr = compute_corr(h)
    return float(corr.max() * (2.0 / N))

def softmax_gradient(h, beta):
    N = h.size
    corr = compute_corr(h)
    Omax = corr.max()
    w = np.exp(beta * (corr - Omax))
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros_like(h)
    wk = w[N-1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, N):
        wk = w[N-1 + k] + w[N-1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: N - k] -= wk * h[k:]
        grad[k:] -= wk * h[: N - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def random_swap_perturb(h, rng, scale=1.0):
    N = h.size
    i, j = rng.integers(N, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        # swap direction
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import time, math, numpy as np
from scipy import signal

def project_onto_box_sum(v, target):
    lo, hi = -1.0, 2.0
    for _ in range(40):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.argmin(np.abs(v - lam))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def compute_corr(h):
    return signal.fftconvolve(h, (1.0 - h)[::-1], mode='full')

def compute_c5(h):
    N = h.size
    corr = compute_corr(h)
    return float(corr.max() * (2.0 / N))

def softmax_gradient(h, beta):
    N = h.size
    corr = compute_corr(h)
    Omax = corr.max()
    w = np.exp(beta * (corr - Omax))
    w /= w.sum()
    grad = np.zeros_like(h)
    wk = w[N - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, N):
        wk = w[N - 1 + k] + w[N - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: N - k] -= wk * h[k:]
        grad[k:] -= wk * h[: N - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def worst_gradient(h):
    N = h.size
    corr = compute_corr(h)
    idx = np.argmax(corr) - (N - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: N - k] -= h[k:]
        grad[k:] -= h[: N - k]
    return grad

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def random_swap_perturb(h, rng, scale=1.0):
    N = h.size
    i, j = rng.integers(N, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initial vector -----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        N0 = h0.size
        N = N0 * 4
        h = np.repeat(h0, 4)
    else:
        N = 2048
        h = rng.random(N)
        h = np.clip(h, 0.0, 1.0)

    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- Phase 1: smoothed gradient descent -----
    beta_schedule = [10, 40, 160, 640, 2560, 10240]
    for beta in beta_schedule:
        if time.time() - start > budget_s * 0.70:
            break
        iters = int(2000 * 1024 / N)
        step0 = 0.5 / np.sqrt(beta)
        h_cur = best_h.copy()
        for it in range(iters):
            if time.time() - start > budget_s * 0.70:
                break
            grad = softmax_gradient(h_cur, beta)
            norm = np.linalg.norm(grad)
            if norm == 0:
                step = 0.0
            else:
                step = step0 * (1.0 - it / iters) / norm
            v = h_cur - step * grad
            h_cur = project_onto_box_sum(v, target)
        c = compute_c5(h_cur)
        if c < best_c:
            best_h, best_c = h_cur.copy(), c

    # ----- Phase 2: simulated annealing -----
    h_cur = best_h.copy()
    cur_c = best_c
    T0, T_end = 0.02, 0.0002
    total_sa = int(8000 * 1024 / N)
    for t in range(total_sa):
        if time.time() - start > budget_s * 0.90:
            break
        T = T0 + (T_end - T0) * (t / total_sa)
        nh = random_swap_perturb(h_cur, rng, scale=0.4)
        nc = compute_c5(nh)
        delta = nc - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h_cur, cur_c = nh, nc
            if cur_c < best_c:
                best_h, best_c = h_cur.copy(), cur_c

    # guided hill climb after SA
    if time.time() - start < budget_s * 0.94:
        hc, cc = guided_hill_climb(best_h, rng,
                                   max_iters=int(2000 * 1024 / N),
                                   step_scale=0.4)
        if cc < best_c:
            best_h, best_c = hc.copy(), cc

    # ----- Phase 3: final high‑beta polish -----
    if time.time() - start < budget_s * 0.99:
        beta = 50000
        iters = int(1500 * 1024 / N)
        step0 = 0.2 / np.sqrt(beta)
        for it in range(iters):
            if time.time() - start > budget_s * 0.99:
                break
            grad = softmax_gradient(best_h, beta)
            norm = np.linalg.norm(grad)
            if norm == 0:
                step = 0.0
            else:
                step = step0 * (1.0 - it / iters) / norm
            v = best_h - step * grad
            best_h = project_onto_box_sum(v, target)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
