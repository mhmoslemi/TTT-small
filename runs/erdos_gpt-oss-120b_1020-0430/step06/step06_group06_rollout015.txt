def project_box_sum(v, target, low=0.0, high=1.0):
    # find lambda such that sum(clipped) = target
    # v may be any real array
    # lambda bounds: low - max(v) <= lambda <= high - min(v)

    lam_high = high - np.min(v)

    for _ in range(60):

        s = clipped.sum()
        if s > target:
            lam_high = lam

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # compute correlation
    H = np.fft.rfft(x, n=2*x.size)
    B = np.fft.rfft(1.0 - x, n=2*x.size)
    corr = np.fft.irfft(H * np.conj(B), n=2*x.size)[:2*x.size-1]
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable logsumexp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def correlation_fft(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(B), n=L)
    return corr[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    return np.max(correlation_fft(h)) * (2.0 / h.size)


def worst_gradient_full(h: np.ndarray, k: int) -> np.ndarray:
    n = h.size
    grad = np.ones(n)
    if k == 0:
        grad -= 2 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def project_onto_box_sum(v: np.ndarray, target: float,
                         low: float = 0.0, high: float = 1.0) -> np.ndarray:
    lam_low = low - np.max(v)
    lam_high = high - np.min(v)
    for _ in range(60):
        lam = (lam_low + lam_high) / 2.0
        proj = np.clip(v + lam, low, high)
        s = proj.sum()
        if s > target:
            lam_high = lam
        else:
            lam_low = lam
    return np.clip(v + lam_low, low, high)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    n = x.size
    L = 2 * n
    H = np.fft.rfft(x, n=L)
    B = np.fft.rfft(1.0 - x, n=L)
    corr = np.fft.irfft(H * np.conj(B), n=L)[: 2 * n - 1]
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    if abs(CURRENT_TARGET - x.sum()) > 1e-12:
        x = project_onto_box_sum(x, CURRENT_TARGET)
    return x
