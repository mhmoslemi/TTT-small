def project_onto_box_sum(v, target):
    # ensure sum(v) = target after clipping to [0,1]
    # Approach: Clip v to [0,1], denote u = np.clip(v,0,1). Then adjust via lambda to meet sum.
    # Equivalent to solving min ||x - v||^2 s.t. 0<=x<=1, sum x = target.
    # This can be solved via sorting; lambda such that sum(clip(v - lambda,0,1)) = target.
    # Use algorithm: sort v, compute cumulative sums to find lambda.
    n = len(v)
    lo = -1.0 # lower bound on lambda
    hi = 1.0 # upper bound? Actually lambda between min(v)-1 and max(v)
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        lam = (lo+hi)/2
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # due to numerical, adjust small residual if needed
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # Distribute diff equally among non-border entries
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            # Find index with maximal slack
            i = np.argmin(np.abs(v - (lo+hi)/2))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:

        if mask.any():

            x[mask] += add

            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
