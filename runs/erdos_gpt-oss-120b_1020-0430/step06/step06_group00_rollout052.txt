import time
import numpy as np
from scipy.optimize import minimize

# -------------------- core utilities --------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 using FFT (linear convolution)."""
    n = h.size
    N = 2 * n - 1
    H = np.fft.fft(h, N)
    O = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(H * O).real
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to have sum = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n points on [0,2)."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


# -------------------- gradient ascent on min self‑correlation --------------------
def grad_min_corr(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of min self‑correlation for a given integer shift."""
    n = h.size
    if shift > 0:
        left = np.concatenate([np.zeros(shift), h[: n - shift]])
        right = np.concatenate([h[shift:], np.zeros(shift)])
    elif shift < 0:
        s = -shift
        left = np.concatenate([h[s:], np.zeros(s)])
        right = np.concatenate([np.zeros(s), h[: n - s]])
    else:
        left = h
        right = h
    return left + right


def projected_ascent(
    h0: np.ndarray, rng: np.random.Generator, max_iter: int = 400, init_step: float = 0.05
):
    """Projected ascent maximizing minimal self‑correlation."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    step = init_step
    best_c5 = compute_c5(h)
    for _ in range(max_iter):
        N = 2 * n - 1
        H = np.fft.fft(h, N)
        conv = np.fft.ifft(H * H).real
        idx = np.argmin(conv)
        shift = idx - (n - 1)
        grad = grad_min_corr(h, shift)
        h += step * grad
        np.clip(h, 0.0, 1.0, out=h)
        # simple uniform redistribution for sum, followed by repair if needed
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h = enforce_sum(h, rng)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        step *= 0.99
    return h, best_c5


def random_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector."""
    h = rng.random(n)
    np.clip(h, 0.0, 1.0, out=h)
    return enforce_sum(h, rng)


# -------------------- soft‑max SLSQP --------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 800):
    """SLSQP on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# -------------------- continuous simulated annealing --------------------
def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """SA on the continuous feasible set."""
    start = time.time()
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    N = h.size
    T0, T_end = 0.005, 1e-7
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        scale = 0.3 * (T / T0)
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * scale
            h_new = h.copy()
            h_new[j] -= eps
            h_new[i] += eps
        c_new = compute_c5(h_new)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c
