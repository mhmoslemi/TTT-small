import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    n = h.size
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = None
            best_c = np.inf
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                cand = h - base_step * mult * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, best_h = c, cand
            if best_h is not None:
                h = best_h
    return h

def greedy_worst_shift_refine(h, rng, max_iters, deadline):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    eps = 1e-12
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        v = np.correlate(h, 1.0 - h, mode="full")
        k = int(v.argmax()) - (n - 1)
        if k >= 0:
            right = np.concatenate([h[k:], np.zeros(k)])
            left = np.concatenate([np.zeros(k), h[:n - k]])
        else:
            kk = -k
            right = np.concatenate([np.zeros(kk), h[:n - kk]])
            left = np.concatenate([h[kk:], np.zeros(kk)])
        neighbor = right + left
        inc_idx = np.where(h < 1 - eps)[0]
        dec_idx = np.where(h > eps)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i_inc = inc_idx[np.argmax(neighbor[inc_idx])]
        i_dec = dec_idx[np.argmin(neighbor[dec_idx])]
        delta = min(1.0 - h[i_inc], h[i_dec])
        if delta <= eps:
            break
        h[i_inc] += delta
        h[i_dec] -= delta
        c = compute_c5(h)
        if c + 1e-12 < best_c:
            best_c, best_h = c, h.copy()
    h[:] = best_h
    return h, best_c

def interpolate_parent(parent, n):
    xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(xt, xs, parent)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    parent = globals().get("initial_h_values", None)

    n_choices = [360, 540, 720, 960]
    betas = [1, 5, 15, 45, 135, 405, 1215, 3645, 10935, 32805]

    restarts = 6
    for _ in range(restarts):
        if time.time() > deadline:
            break
        n = int(rng.choice(n_choices))
        if parent is not None:
            if len(parent) == n:
                h = np.array(parent, copy=True, dtype=float)
            else:
                h = interpolate_parent(parent, n)
        else:
            h = rng.random(n)
            h = h / h.sum() * (n / 2.0)
            np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)

        # smooth‑max gradient descent

            betas,

        # aggressive greedy refinement on worst shift
        h, c = greedy_worst_shift_refine(
            h, rng, max_iters=2500, deadline=deadline

        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n

        # optional short binary simulated‑annealing if time permits
        if time.time() < deadline:
            h_bin = np.rint(h).astype(float)
            adjust_sum_exact(h_bin, rng)
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
            # simple hill‑climb: swap 1↔0 if it improves C5
            cur_h = h_bin.copy()
            cur_c = compute_c5(cur_h)
            ones = list(np.flatnonzero(cur_h == 1))
            zeros = list(np.flatnonzero(cur_h == 0))
            it = 0
            while time.time() < sub_dead and it < 8000 and ones and zeros:

                j = rng.choice(zeros)
                cur_h[i] = 0
                cur_h[j] = 1
                c_new = compute_c5(cur_h)
                if c_new < cur_c:
                    cur_c = c_new
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    if c_new < best_c:

                    it = 0
                else:
                    cur_h[i] = 1
                    cur_h[j] = 0
                    it += 1
        # end of this restart

    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
