def binary_sa_incremental(n, rng, time_budget, init_h=None):
    start = time.time()
    # Initialize binary h
    if init_h is not None:
        # init_h is float array length n, sum = n/2
        probs = init_h / init_h.sum()
        ones = rng.choice(n, n//2, replace=False, p=probs)
    else:
        ones = rng.choice(n, n//2, replace=False)
    h = np.zeros(n, dtype=np.int8)
    h[ones] = 1

    # O via correlation
    corr = np.correlate(h.astype(np.int64), 1 - h.astype(np.int64), mode='full')
    O = corr[n-1:].astype(np.int32)  # length n
    O[0] = 0
    current_max = int(O.max())
    best_h = h.copy()
    best_max = current_max

    # Build index arrays
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    idx_in_one = np.full(n, -1, dtype=np.int32)
    idx_in_zero = np.full(n, -1, dtype=np.int32)
    idx_in_one[ones_idx] = np.arange(len(ones_idx), dtype=np.int32)
    idx_in_zero[zeros_idx] = np.arange(len(zeros_idx), dtype=np.int32)

    # target
    target_max = int(np.floor(0.3808 * n / 2))

    T0 = 30.0
    T_end = 1e-4

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # pick i from ones, j from zeros
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        # Apply O updates (using old h)
        # i contributions
        O[:n - i] += h[i:]
        O[:i + 1] += h[i::-1]
        # j contributions (subtract)
        O[:n - j] -= h[j:]
        O[:j + 1] -= h[j::-1]
        O[0] = 0  # enforce zero for k=0
        new_max = int(O.max())
        delta = new_max - current_max
        accept = False
        if delta <= 0:
            accept = True
        else:
            # acceptance probability
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True
        if accept:
            # accept swap
            # update h
            h[i] = 0
            h[j] = 1
            # update index arrays and ones_idx/zeros_idx
            pos_i = idx_in_one[i]
            pos_j = idx_in_zero[j]
            # swap positions
            ones_idx[pos_i] = j
            zeros_idx[pos_j] = i
            # update idx maps
            idx_in_one[i] = -1
            idx_in_one[j] = pos_i
            idx_in_zero[j] = -1
            idx_in_zero[i] = pos_j
            current_max = new_max
            # update best
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
                if best_max <= target_max:
                    break
        else:
            # revert O
            O[:n - i] -= h[i:]
            O[:i + 1] -= h[i::-1]
            O[:n - j] += h[j:]
            O[:j + 1] += h[j::-1]
            O[0] = 0
        # (optional) early exit if time nearly over
    return best_h.astype(float), best_max
```

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

import time
import numpy as np

# -------------------- core utilities --------------------

def compute_c5(h: np.ndarray) -> float:
    """C5 for discrete step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 20000,
    step_scale: float = 0.7,
):
    """Deterministic gradient‑directed hill climbing."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    step: float = 0.015,
    max_iter: int = 10,
) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# -------------------- incremental binary SA --------------------

def binary_sa_incremental(
    n: int,
    rng: np.random.Generator,
    time_budget: float,
    init_h: np.ndarray | None = None,
):
    """Simulated annealing on binary vectors with O(n) update."""
    start = time.time()
    # ---- initialise binary vector ----
    if init_h is not None:
        probs = init_h / init_h.sum()
        ones = rng.choice(n, n // 2, replace=False, p=probs)
    else:
        ones = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=np.int8)
    h[ones] = 1

    # ---- compute O_k (k >= 0) ----
    corr = np.correlate(h.astype(np.int64), (1 - h).astype(np.int64), mode="full")
    O = corr[n - 1 :].astype(np.int32)  # length n, O[0] = 0
    O[0] = 0
    current_max = int(O.max())
    best_h = h.copy()
    best_max = current_max

    # ---- index bookkeeping ----
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    idx_in_one = np.full(n, -1, dtype=np.int32)
    idx_in_zero = np.full(n, -1, dtype=np.int32)
    idx_in_one[ones_idx] = np.arange(len(ones_idx), dtype=np.int32)
    idx_in_zero[zeros_idx] = np.arange(len(zeros_idx), dtype=np.int32)

    target_max = int(np.floor(0.3808 * n / 2))

    T0, T_end = 30.0, 1e-4

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)

        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]

        # ---- apply O updates using old h (binary) ----
        O[: n - i] += h[i:]
        O[: i + 1] += h[i::-1]
        O[: n - j] -= h[j:]
        O[: j + 1] -= h[j::-1]
        O[0] = 0

        new_max = int(O.max())
        delta = new_max - current_max
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True

        if accept:
            # ---- commit swap ----
            h[i] = 0
            h[j] = 1

            pos_i = idx_in_one[i]
            pos_j = idx_in_zero[j]
            ones_idx[pos_i] = j
            zeros_idx[pos_j] = i
            idx_in_one[i] = -1
            idx_in_one[j] = pos_i
            idx_in_zero[j] = -1
            idx_in_zero[i] = pos_j

            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
                if best_max <= target_max:
                    break
        else:
            # ---- revert O ----
            O[: n - i] -= h[i:]
            O[: i + 1] -= h[i::-1]
            O[: n - j] += h[j:]
            O[: j + 1] += h[j::-1]
            O[0] = 0

    return best_h.astype(float), best_max

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    n = 960  # <1000 points, fine resolution

    # ----- parent construction if present -----
    if "initial_h_values" in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent.size != n:
            x_old = np.linspace(0.0, 2.0, h_parent.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_parent = np.interp(x_new, x_old, h_parent)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = enforce_sum(h_parent, rng)
    else:
        h_parent = None

    # ----- binary simulated annealing (major chunk) -----
    binary_time = total_budget * 0.55
    h_bin, _ = binary_sa_incremental(n, rng, binary_time, init_h=h_parent)
    best_h = h_bin
    best_c = compute_c5(best_h)

    # ----- early exit if already good enough -----
    if best_c <= 0.3808:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        final_c = compute_c5(best_h)
        return best_h.tolist(), float(final_c), int(best_h.size)

    # ----- continuous refinement with gradient hill‑climb -----
    remaining = total_budget - (time.time() - start)
    if remaining > 0.1:
        h_cont = best_h.astype(float)
        max_iters = int(min(60000, remaining * 700))
        h_cont, c_cont = guided_hill_climb(h_cont, rng, max_iters=max_iters, step_scale=0.7)
        c_cont = compute_c5(h_cont)
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont

    # ----- final stochastic tweaking -----
    while time.time() - start < total_budget * 0.999:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # ----- sanitise final result -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
