import time
import numpy as np

# ---------------------------------------------------------
# Basic utilities
def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum x = target} via binary search on lambda."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in the cube [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---------------------------------------------------------
# Soft‑max projected gradient descent (FFT‑based)
def softmax_grad_descent(
    h0: np.ndarray, target: float, budget_s: float, rng: np.random.Generator
):
    """Maximize min autocorrelation via smooth max and projected GD."""
    start = time.time()
    n = h0.size
    n_fft = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    scale = 2.0 / n

    h = h0.copy()
    best_h = h.copy()
    # initial C5
    H = np.fft.rfft(h, n=n_fft)
    conv_hh = np.fft.irfft(H * H, n=n_fft)[: 2 * n - 1]
    cross = conv_hh[n - 1 :]
    O = 1.0 - scale * cross
    best_c = O.max()

    # hyper‑parameters
    beta = 5.0
    beta_max = 4000.0
    beta_inc = 1.02
    base_lr = 0.7
    mu = 0.9
    v = np.zeros_like(h)
    it = 0
    last_imp = 0

    while time.time() - start < budget_s:
        it += 1

        # ----- compute O_k ----------
        H = np.fft.rfft(h, n=n_fft)
        conv_hh = np.fft.irfft(H * H, n=n_fft)[: 2 * n - 1]
        cross = conv_hh[n - 1 :]
        O = 1.0 - scale * cross
        cur_c = O.max()
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            last_imp = it
            if best_c <= 0.38075:  # early exit if we beat target comfortably
                break

        # ----- soft‑max weights ----------
        if beta < beta_max:
            beta = min(beta_max, beta * beta_inc)
        maxO = O.max()
        w = np.exp(beta * (O - maxO))
        w /= w.sum()

        # ----- gradient via FFT ----------
        W = np.fft.rfft(w, n=n_fft)

        # right shift contribution: conv(w, h)
        right_conv = np.fft.irfft(W * H, n=n_fft)[: 2 * n - 1]
        right_sum = right_conv[n - 1 :]

        # left shift contribution: conv(w, h_rev)
        h_rev = h[::-1]
        H_rev = np.fft.rfft(h_rev, n=n_fft)
        left_conv = np.fft.irfft(W * H_rev, n=n_fft)[: 2 * n - 1]
        left_sum = left_conv[n - 1 :][::-1]

        grad = -scale * (right_sum + left_sum)

        # ----- step with momentum ----------
        lr = base_lr / np.sqrt(it)
        v = mu * v - lr * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_box_sum(h_new, target)
        h = h_new

        # ----- occasional random kick ----------
        if it - last_imp > 2000:
            # add a small random perturbation to escape stagnation
            h += rng.normal(scale=0.01, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, target)
            last_imp = it

    return best_h, best_c

    rng = np.random.default_rng(seed)
