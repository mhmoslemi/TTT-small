def compute_corr(h):
    n = h.shape[0]
    size = 2*n
    # use rfft (real-to-complex) for speed
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    conv = np.fft.irfft(H * G, size)[:2*n-1]
    return conv
```

def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    max_iter = 5000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:

        # adjust with capacity
        if diff > 0:
            add = min(diff, 1.0 - h[i])

            add = max(diff, -h[i])

            h[i] += add
            diff = target - h.sum()

    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift (fast version)."""
    n = h.shape[0]
    conv = np.fft.irfft(
        np.fft.rfft(h, 2 * n) * np.fft.rfft(1.0 - h, 2 * n), 2 * n
    )[: 2 * n - 1]
    idx = np.argmax(conv) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def anneal_binary(h: np.ndarray, rng: np.random.Generator,
                  start_time: float, budget_s: float) -> tuple:
    """Simulated‑annealing on binary balanced vectors."""
    n = h.shape[0]
    dx = 2.0 / n
    # initial correlation and objective
    conv = np.fft.irfft(
        np.fft.rfft(h, 2 * n) * np.fft.rfft(1.0 - h, 2 * n), 2 * n
    )[: 2 * n - 1]
    cur_c = np.max(conv) * dx
    best_h = h.copy()
    best_c = cur_c

    # index lists for O(1) swaps
    ones = [int(i) for i in range(n) if h[i] > 0.5]
    zeros = [int(i) for i in range(n) if h[i] < 0.5]

    T0 = 0.02                     # initial temperature
    decay = 0.9995                # exponential decay factor
    i = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.90:
            break
        T = T0 * (decay ** i)

        # propose a swap
        idx1 = rng.integers(len(ones))
        idx0 = rng.integers(len(zeros))
        i1 = ones[idx1]
        i0 = zeros[idx0]

        # execute swap
        h[i1] = 0.0
        h[i0] = 1.0

        # recompute C5
        conv = np.fft.irfft(
            np.fft.rfft(h, 2 * n) * np.fft.rfft(1.0 - h, 2 * n), 2 * n
        )[: 2 * n - 1]
        new_c = np.max(conv) * dx
        delta = new_c - cur_c

        accept = False
        if delta <= 0:
            accept = True
        elif T > 1e-12:
            accept = rng.random() < np.exp(-delta / T)

        if accept:
            cur_c = new_c
            # update the index lists
            ones[idx1] = i0
            zeros[idx0] = i1
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i1] = 1.0
            h[i0] = 0.0

        i += 1

    return best_h, best_c


# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            h = thue_morse_bits(n)
        else:
            h = (h0 > 0.5).astype(float)
    else:
        h = thue_morse_bits(n)

    h = enforce_balanced_binary(h, rng, n)

    # ----- annealing phase (≈90% of budget) -------------------------------
    h_best, c_best = anneal_binary(h, rng, start, budget_s)

    # ----- continuous refinement (remaining time) -------------------------
    h_cont = h_best.copy()
    h_cont = enforce_sum(h_cont, rng)

    step0 = 0.05
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.999:
            break
        step = step0 * (0.99 ** it)
        grad = worst_gradient(h_cont)
        nh = h_cont - step * grad
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum(nh, rng)
        c = compute_c5(nh)
        if c < c_best:
            c_best, h_cont = c, nh.copy()
            h_best = h_cont.copy()
        it += 1

    # ----- final sanity enforcement ---------------------------------------
    final_h = enforce_sum(h_best, rng)
    final_c = compute_c5(final_h)

    return final_h.tolist(), float(final_c), n
