def linear_correlation_fft(a, b):
    n = a.size
    size = 2 * n
    # pad to size
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b[::-1], n=size)
    conv = np.fft.irfft(A * B, n=size)
    # conv length size, linear convolution; linear correlation equals conv[0:2*n-1]
    return conv[:2*n-1]
```

def compute_corr_fft(h):
    n = h.size
    size = 2*n
    # FFT of h (real)
    H = np.fft.rfft(h, n=size)
    # (1 - h)[::-1]
    rev = (1.0 - h)[::-1]
    R = np.fft.rfft(rev, n=size)
    conv = np.fft.irfft(H * R, n=size)
    return conv[:2*n-1]
```

import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) via FFT (full length)."""
    n = h.size
    size = 2 * n
    H = np.fft.rfft(h, n=size)
    rev = (1.0 - h)[::-1]
    R = np.fft.rfft(rev, n=size)
    conv = np.fft.irfft(H * R, n=size)
    return conv[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with N points."""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def softmax_obj(x: np.ndarray) -> float:
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
