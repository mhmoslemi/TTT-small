def project_box_sum(v, total):

   # find lambda such that sum(clip(v + lambda,0,1)) = total

   # Since the clipping function is monotonic in lambda, we can binary search.

   lo = -1.0  # lambda likely in reasonable range
   hi = 1.0
   for _ in range(50):
       mid = (lo+hi)/2
       x = np.clip(v + mid, 0, 1)
       s = x.sum()
       if s > total:
           hi = mid
       else:
           lo = mid
   # after loop, we can return x
   return np.clip(v + (lo+hi)/2, 0, 1)

import time
import numpy as np
import scipy.optimize as opt

# ---------- helper functions ----------

def evaluate_c5(h):
    """Return the max overlap integral for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def project_sum_box(v, total):
    """
    Project vector v onto the box [0,1]^len(v) while enforcing sum(v)=total.
    Solves for λ in x = clip(v+λ,0,1) with Σx = total using bisection.
    """
    lo = -np.max(v)               # ensures v+λ >= 0
    hi = 1.0 - np.min(v)           # ensures v+λ <= 1
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def upsample_projection(old_h, new_n):
    """Interpolate old_h to length new_n and project onto feasibility."""
    old_n = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    return project_sum_box(new_h, new_n / 2.0)


def slsqp_optimize(h0, time_allow):
    """Run several SLSQP attempts from h0 within the given time budget."""
    n = h0.shape[0]
    best_h = h0.copy()
    best_val = evaluate_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - n / 2.0}
    start = time.time()
    attempts = 3
    for att in range(attempts):
        if time.time() - start > time_allow:
            break
        # random perturbation for restarts (skip on first attempt)
        if att == 0:
            init = best_h.copy()
        else:
            noise = np.random.uniform(-0.2, 0.2, size=n)
            init = np.clip(h0 + noise, 0.0, 1.0)
            init = project_sum_box(init, n / 2.0)
        try:
            res = opt.minimize(
                evaluate_c5,
                init,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 250, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue
        cand = project_sum_box(np.clip(res.x, 0.0, 1.0), n / 2.0)
        val = evaluate_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand
    return best_h, best_val


def anneal_optimize(h_start, time_allow):
    """Simple simulated annealing that respects the sum constraint."""
    n = h_start.shape[0]
    best_h = h_start.copy()
    best_val = evaluate_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()

    # annealing schedule
    T0 = 0.2
    T1 = 1e-4
    max_iter = 20000
    for it in range(max_iter):
        if time.time() - start > time_allow:
            break
        # linear cooling
        T = T0 * ((T1 / T0) ** (it / max_iter))
        # propose move: transfer ε from j to i
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]
        max_up = 1.0 - hi
        max_down = hj
        if max_up <= 0 or max_down <= 0:
            continue
        eps = np.random.rand() * min(max_up, max_down) * 0.5
        new_h = cur_h.copy()
        new_h[i] += eps
        new_h[j] -= eps
        new_val = evaluate_c5(new_h)
        delta = new_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
    return best_h, best_val
