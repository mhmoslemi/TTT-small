import numpy as np
import time, math

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def derivative_for_shift(h, k):
    n = h.size
    deriv = np.zeros(n)
    if k >= 0:
        # pos term for i >= k
        idx = np.arange(k, n)
        deriv[idx] += 1 - h[idx - k]
        # neg term for i + k < n
        idx_n = np.arange(0, n - k)
        deriv[idx_n] -= h[idx_n + k]
    else:
        kk = -k
        idx = np.arange(0, n - kk)
        deriv[idx] += 1 - h[idx + kk]
        idx_n = np.arange(kk, n)
        deriv[idx_n] -= h[idx_n - kk]
    return deriv

def random_feasible_vector(n, target_sum):
    # binary plus one fractional remainder
    f = int(np.floor(target_sum))
    rem = target_sum - f
    x = np.zeros(n)
    perm = np.arange(n)
    np.random.shuffle(perm)
    x[perm[:f]] = 1.0
    if rem > 1e-12:
        x[perm[f]] = rem
    return x

import numpy as np
import time
import math

def compute_c5(h, dx):
    """Maximum overlap integral for a discretized step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def derivative_for_shift(h, k):
    """Gradient of the overlap at the shift k that attains the maximum."""
    n = h.size
    d = np.zeros(n)
    if k >= 0:
        idx = np.arange(k, n)
        d[idx] += 1 - h[idx - k]
        idx2 = np.arange(0, n - k)
        d[idx2] -= h[idx2 + k]
    else:
        kk = -k
        idx = np.arange(0, n - kk)
        d[idx] += 1 - h[idx + kk]
        idx2 = np.arange(kk, n)
        d[idx2] -= h[idx2 - kk]
    return d

def random_feasible_vector(n, target):
    """Binary vector with a single fractional entry to meet the sum constraint."""
    whole = int(np.floor(target))
    frac = target - whole
    v = np.zeros(n)
    perm = np.arange(n)
    np.random.shuffle(perm)
    v[perm[:whole]] = 1.0
    if frac > 1e-12:
        v[perm[whole]] = frac
    return v
