def c5_from_h(h):
    dx = 2.0 / len(h)
    # compute correlation as ints maybe:

def optimize(N=200, time_budget=1080): (floating seconds less for overhead)
    dx = 2.0 / N
    # initial vector: use random binary with exactly N/2 ones.
    h = np.zeros(N, dtype=np.int8)
    ones_idx = np.random.choice(N, N//2, replace=False)
    h[ones_idx] = 1
    cur_c5, cur_max = c5_from_h(h)
    best_h = h.copy()
    best_c5 = cur_c5
    best_max = cur_max

    start = time.time()
    # maybe compute initial corr array store
    corr = np.correlate(h, 1 - h, mode='full')
    # not needed except for max
    # iteration loop
    iter = 0
    while time.time() - start < time_budget:
        iter += 1
        # find lag with max correlation
        idx_max = np.argmax(corr)
        L = idx_max - (N - 1)  # can be negative
        # find mismatch masks
        if L >= 0:
            len_overlap = N - L
            if len_overlap <= 0:
                # ignore; should not happen because L in range
                continue
            mask_i = (h[:len_overlap] == 1) & (h[L:] == 0)
            mask_j = (h[:len_overlap] == 0) & (h[L:] == 1)
            idx_i_candidates = np.where(mask_i)[0]
            idx_j_candidates = np.where(mask_j)[0]
        else:
            K = -L
            len_overlap = N - K
            if len_overlap <= 0:
                continue
            mask_i = (h[K:] == 1) & (h[:len_overlap] == 0)
            mask_j = (h[K:] == 0) & (h[:len_overlap] == 1)
            idx_i_candidates = np.where(mask_i)[0] + K  # shift indices
            idx_j_candidates = np.where(mask_j)[0] + K

        if idx_i_candidates.size == 0 or idx_j_candidates.size == 0:
            # fallback: random swap
            # choose i random from ones, j random from zeros.
            one_idxs = np.where(h == 1)[0]
            zero_idxs = np.where(h == 0)[0]
            i = one_idxs[np.random.randint(one_idxs.size)]
            j = zero_idxs[np.random.randint(zero_idxs.size)]
        else:
            i = idx_i_candidates[np.random.randint(idx_i_candidates.size)]
            j = idx_j_candidates[np.random.randint(idx_j_candidates.size)]

        # perform swap
        h[i], h[j] = h[j], h[i]   # swap
        # compute new correlation
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = new_corr.max()
        if new_max <= cur_max:
            # accept, update current state
            corr = new_corr
            cur_max = new_max
            cur_c5 = new_max * dx
            # update best if better
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_max = cur_max
                # early exit if meet target
                if best_c5 <= 0.3808:
                    break
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

        # maybe occasional random perturb to escape
        if iter % 10000 == 0:
            # small random swap
            # else continue
            pass

    return best_h.astype(float), best_c5, N

import time, math, random
import numpy as np

def _c5_and_max(h):
    """Return (C5 value, maximal correlation count) for a step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    max_cnt = int(corr.max())
    return max_cnt * (2.0 / n), max_cnt, corr

def run(seed=42, budget_s=1100, **kwargs):
    # -------------------------------------------------------------
    # basic settings
    # -------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)

    # use the supplied initial solution if present
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, copy=True, dtype=np.float64)
        n0 = h0.shape[0]
        best_h = h0.copy()
        best_c5, best_max, best_corr = _c5_and_max(best_h)
    else:
        best_h = None
        best_c5 = 1e9
        best_max = None
        n0 = None

    # we will work with an even length to keep the sum integral.
    N = 200                     # even, < 1000 as requested
    dx = 2.0 / N

    # -----------------------------------------------------------------
    # initialise a binary candidate (exactly N/2 ones)
    # -----------------------------------------------------------------
    h = np.zeros(N, dtype=np.float64)
    ones = np.random.choice(N, N // 2, replace=False)
    h[ones] = 1.0

    cur_c5, cur_max, cur_corr = _c5_and_max(h)

    # keep the best seen across all restarts
    if cur_c5 < best_c5:
        best_h = h.copy()
        best_c5, best_max = cur_c5, cur_max

    start = time.time()
    T0 = 2.0               # initial temperature for simulated annealing
    no_improve = 0
    max_no_improve = 5000  # restart after this many non‑improving steps
    iteration = 0

    while time.time() - start < budget_s:
        iteration += 1

        # temperature schedule (linear cooling)
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = T0 * (1.0 - frac)

        # ---------------------------------------------------------
        # pick a swap; try to attack the worst lag
        # ---------------------------------------------------------
        idx_max = int(cur_corr.argmax())
        L = idx_max - (N - 1)               # lag (may be negative)

        # find positions that contribute to the maximal count
        if L >= 0:
            ov = N - L
            mask_i = (h[:ov] == 1) & (h[L:] == 0)   # mismatched ones
            mask_j = (h[:ov] == 0) & (h[L:] == 1)   # opposite side
            i_cand = np.where(mask_i)[0]
            j_cand = np.where(mask_j)[0]
        else:
            K = -L

        if i_cand.size and j_cand.size:

            ones_idx = np.where(h == 1)[0]
