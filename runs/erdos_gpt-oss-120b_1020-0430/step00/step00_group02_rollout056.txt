import time
import numpy as np

# -------------------------------------------------
# Helper functions
# -------------------------------------------------
def project_onto_simplex_box(v, target, max_iter=200):
    """Project v onto the intersection of the box [0,1]^n and the hyperplane sum=v_target."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            idx = np.where(x > 0)[0]
            if not idx.size:
                break
            delta = (s - target) / idx.size
            x[idx] -= delta
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.where(x < 1)[0]
            if not idx.size:
                break
            delta = (target - s) / idx.size
            x[idx] += delta
            x = np.clip(x, 0.0, 1.0)
    return x

def random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    v = rng.random(n)
    # force the mean to be 0.5 roughly, then project
    v = v - v.mean() + 0.5
    v = np.clip(v, 0.0, 1.0)
    v = project_onto_simplex_box(v, target)
    return v

def sinusoid_h(n):
    """Sinusoidal starting shape h(x)=0.5+0.5*sin(pi*x) sampled on n points."""
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    h = 0.5 + 0.5 * np.sin(np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_simplex_box(h, n / 2.0)
    return h

def compute_c5(h):
    """Return the Erdős C5 value for a discretised step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr_and_shift_max(h):
    """Return (shift, correlation array, max correlation) for h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)  # integer shift corresponding to the lag
    return shift, corr, corr[idx]

def compute_gradient(h, shift):
    """Gradient of the correlation at a given integer shift."""
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    else:
        s = -shift
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    return g

# Core local‑search routine
# -------------------------------------------------
def local_search(start_h, time_budget, seed):
    rng = np.random.default_rng(seed)
    n = len(start_h)
    h = start_h.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    step = 0.2            # maximal mass moved in a single transfer
    no_improve = 0
    start = time.time()
    while time.time() - start < time_budget:
        shift, _, _ = compute_corr_and_shift_max(h)
        grad = compute_gradient(h, shift)

        # indices that can donate (positive gradient) and receive (negative gradient)
        pos_idx = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg_idx = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]

        moved = False
        if pos_idx.size and neg_idx.size:
            i = rng.choice(pos_idx)
            j = rng.choice(neg_idx)
            max_eps = min(step, h[i], 1.0 - h[j])
            if max_eps > 1e-12:
                eps = max_eps * rng.random()
                nh = h.copy()
                nh[i] -= eps
                nh[j] += eps
                new_score = compute_c5(nh)
                if new_score <= cur_score + 1e-12:          # accept non‑worsening
                    h = nh
                    cur_score = new_score
                    moved = True
                    if new_score < best_score - 1e-12:
                        best_h = nh.copy()
                        best_score = new_score
                        no_improve = 0
        if not moved:
            # generic random transfer
            donors = np.where(h > 1e-12)[0]
            receivers = np.where(h < 1.0 - 1e-12)[0]
            if donors.size and receivers.size:
                i = rng.choice(donors)
                j = rng.choice(receivers)
                max_eps = min(step, h[i], 1.0 - h[j])
                if max_eps > 1e-12:
                    eps = max_eps * rng.random()
                    nh = h.copy()
                    nh[i] -= eps
                    nh[j] += eps
                    new_score = compute_c5(nh)
                    if new_score <= cur_score + 1e-12:
                        h = nh
                        cur_score = new_score
                        moved = True
                        if new_score < best_score - 1e-12:
                            best_h = nh.copy()
                            best_score = new_score
                            no_improve = 0
        if not moved:
            no_improve += 1
        # step‑size adaptation
        if no_improve >= 5000:
            step = max(step * 0.5, 1e-5)
            no_improve = 0
        # occasional perturbation to escape a deep plateau
        if step <= 1e-5 and no_improve > 20000:
            noise = (rng.random(n) - 0.5) * 0.02
            nh = best_h + noise
            nh = np.clip(nh, 0.0, 1.0)
            nh = project_onto_simplex_box(nh, n / 2.0)
            h = nh
            cur_score = compute_c5(nh)
            step = 0.2
            no_improve = 0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # -----------------------------------------------------------------
    # 1. Obtain an initial feasible vector
    # -----------------------------------------------------------------
    init_h = globals().get('initial_h_values')
    if init_h is not None:
        h0 = np.asarray(init_h, dtype=float).copy()
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_simplex_box(h0, len(h0) / 2.0)
    else:
        n0 = 45
        h0 = random_feasible(n0, rng)

    best_h = h0
    best_score = compute_c5(best_h)

    # -----------------------------------------------------------------
    # 2. Define the sequence of resolutions (45 → 90 → 180)
    # -----------------------------------------------------------------
    target_sizes = [len(best_h)]
    while target_sizes[-1] < 180:
        target_sizes.append(min(180, target_sizes[-1] * 2))

    # allocate rough time slices (200 s, 300 s, rest)
    stage_times = [200, 300]  # for all but the final stage
    # -----------------------------------------------------------------
    # 3. Optimisation at each resolution
    # -----------------------------------------------------------------
    for idx, n_target in enumerate(target_sizes):
        # up‑sample if needed
        if len(best_h) != n_target:
            factor = n_target // len(best_h)
            best_h = np.repeat(best_h, factor)

        # recompute remaining budget
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining <= 0:
            break

        # decide how much time to spend on this stage
        if idx < len(stage_times):
            stage_time = min(stage_times[idx], remaining)
        else:
            stage_time = remaining

        # ----------- multi‑start local search for this stage -------------
        # 1) start from the up‑sampled best solution
        bh, bs = local_search(best_h, stage_time * 0.6, seed + idx * 1000 + 1)

        # 2) start from a fresh random feasible vector
        rand_start = random_feasible(n_target, np.random.default_rng(seed + idx * 1000 + 2))
        bh2, bs2 = local_search(rand_start, stage_time * 0.2, seed + idx * 1000 + 3)

        # 3) start from a sinusoidal pattern
        sinus_start = sinusoid_h(n_target)
        bh3, bs3 = local_search(sinus_start, stage_time * 0.2, seed + idx * 1000 + 4)

        # pick the best among the three
        candidates = [(bh, bs), (bh2, bs2), (bh3, bs3)]
        bh, bs = min(candidates, key=lambda x: x[1])

        if bs < best_score - 1e-12:
            best_h = bh
            best_score = bs

    # -----------------------------------------------------------------
    # 4. Final sanity projection (ensuring exact sum)
    # -----------------------------------------------------------------
    n_final = len(best_h)
    target_sum = n_final / 2.0
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h = project_onto_simplex_box(best_h, target_sum)

    final_score = compute_c5(best_h)

    return best_h, float(final_score), n_final
