def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

        n = 80

    target_sum = n / 2.0
    dx = 2.0 / n

    def project_to_sum_box(x):

        x = np.clip(x, 0.0, 1.0)

        # binary search for λ s.t. sum clip(x+λ,0,1) = target_sum
        lo, hi = -1.0, 1.0

        def shifted_sum(l):
            return np.clip(x + l, 0.0, 1.0).sum()
