def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    # Choose n
    n = 200
    target_sum = n // 2
    dx = 2.0 / n
    # Function to compute max mismatch using circular correlation
    def max_mismatch(h):
        H = np.fft.rfft(h)
        G = np.fft.rfft(1 - h)
        corr = np.fft.irfft(H * np.conj(G), n=n)
        return corr.max()
    # Initialize best
    best_h = None
    best_val = float('inf')
    # Number of restarts or attempts
    max_restarts = 5
    for restart in range(max_restarts):
        # Random binary vector with exactly target_sum ones
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=target_sum, replace=False)
        h[ones] = 1.0
        cur_val = max_mismatch(h) * dx
        cur_max = cur_val / dx  # store raw mismatch for comparisons (maybe int)
        # optional: compute initial best
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # Simulated annealing loop
        iter_count = 0
        # parameters
        max_iter = 200000  # large
        while time.time() - start < budget_s - 2:
            if iter_count > max_iter:
                break
            # temperature schedule linear decay
            # use proportion of time used
            elapsed = time.time() - start
            frac = elapsed / budget_s
            T = max(0.01, 1.0 * (1 - frac))  # from 1 down to 0.01
            # choose random swap
            ones_idx = np.flatnonzero(h == 1)
            zeros_idx = np.flatnonzero(h == 0)
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            # swap
            h[i] = 0.0
            h[j] = 1.0
            new_max_raw = max_mismatch(h)  # raw mismatch count (not scaled)
            # accept
            if new_max_raw < cur_max or rng.random() < math.exp(-(new_max_raw - cur_max) / (T + 1e-12)):
                cur_max = new_max_raw

                    best_h = h.copy()

                h[j] = 0.0

import numpy as np, time, math

def _circular_max_mismatch(h):
    """Maximum of h*(1-h) correlation for cyclic shifts."""
    n = h.shape[0]
    H = np.fft.rfft(h, n=n)
    G = np.fft.rfft(1.0 - h, n=n)
    corr = np.fft.irfft(H * np.conj(G), n=n)
    return float(np.max(corr))

def _project_to_sum(v, target):
    """Project v onto the box [0,1] with exact sum = target."""
    # initial clipping
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h
