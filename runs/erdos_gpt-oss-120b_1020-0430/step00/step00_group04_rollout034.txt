def proj_bound_sum(v, S, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    # initial clip
    v = np.clip(v, lower, upper)
    # if sum matches S return
    current = v.sum()
    if abs(current - S) <= tol:
        return v
    # binary search for dual variable lambda
    # We want to find w_i = min(max(v_i - lambda, lower), upper) such that sum w_i = S.
    # lambda can be positive or negative.
    # Use bisection on lambda.
    # Lower bound for lambda: -M where M large; Upper bound similarly.
    # Because w_i = lower if lambda very large positive; w_i = upper if lambda very negative.
    # So start lo = -max(v-i) - something; hi = max(v_i - lower) + something.
    lo = -1e5
    hi = 1e5
    # But we can bound more tightly: lambda_min = min(v_i - upper) (if lambda <= v_i - upper then w_i=upper)
    # lambda_max = max(v_i - lower) (if lambda >= v_i - lower then w_i=lower)
    # Actually w_i = upper if v_i - lambda >= upper => lambda <= v_i - upper
    # So for lambda very small (negative large magnitude), w_i saturates at upper.
    # For lambda very large (positive), w_i saturates at lower.
    # So we set lo = min(v - upper) - 1 maybe; hi = max(v - lower) + 1
    lo = np.min(v - upper) - 1.0
    hi = np.max(v - lower) + 1.0
    for _ in range(100):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return w

import numpy as np, time

def proj_bound_sum(v, S, lower=0.0, upper=1.0, tol=1e-9):
    v = np.clip(v, lower, upper)
    total = v.sum()
    if abs(total - S) <= tol:
        return v
    # binary search on lambda
    lo = np.min(v - upper) - 1.0
    hi = np.max(v - lower) + 1.0
    for _ in range(100):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return w

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(c)

def worst_shift_and_grad(h):
    n = h.shape[0]
    c = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(c)
    shift = idx - (n - 1)
    g = np.ones_like(h)
    if shift >= 0:
        # subtract h[i+shift] from first part
        g[:n-shift] -= h[shift:]
        # subtract h[i-shift] from second part
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        # i+shift = i - s
        g[s:] -= h[:n-s]
        # i-shift = i + s
        g[:n-s] -= h[s:]
    return shift, g

import time
import numpy as np
from scipy.interpolate import interp1d


def _proj_bound_sum(v: np.ndarray, S: float, lo: float = 0.0, hi: float = 1.0, tol: float = 1e-9) -> np.ndarray:
    """Euclidean projection of v onto {x : lo<=x<=hi, sum(x)=S}."""
    v = np.clip(v, lo, hi)
    cur = v.sum()
    if abs(cur - S) <= tol:
        return v
    # binary search for the dual variable λ
    lam_low = np.min(v - hi) - 1.0
    lam_high = np.max(v - lo) + 1.0
    for _ in range(100):
        lam = (lam_low + lam_high) / 2.0
        w = np.clip(v - lam, lo, hi)
        s = w.sum()
        if s > S:
            lam_low = lam
        else:
            lam_high = lam
        if lam_high - lam_low < tol:
            break
    return w


def _c5_value(h: np.ndarray) -> float:
    """Return C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _worst_shift_and_grad(h: np.ndarray):
    """Find the shift giving maximal overlap and the gradient of that term."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)          # actual integer shift
    g = np.ones_like(h)

    if shift >= 0:
        g[: n - shift] -= h[shift:]
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        g[s:] -= h[: n - s]
        g[: n - s] -= h[s:]

    return shift, g


def _improve_h(h0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Local optimisation (projected sub‑gradient) within a time budget."""
    h = _proj_bound_sum(h0, h0.size / 2.0)
    best_h = h.copy()
    best_score = _c5_value(h)

    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        _, grad = _worst_shift_and_grad(h)

        # a simple diminishing step size
        alpha = 0.2 * np.exp(-0.0005 * it)

        h_new = _proj_bound_sum(h - alpha * grad, h.size / 2.0)
        score = _c5_value(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()

        # accept only improving steps (tiny random acceptance to escape plateaus)
        if score <= _c5_value(h) or rng.random() < 0.01:
            h = h_new

        it += 1
    return best_h, best_score
