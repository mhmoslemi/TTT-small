def project_box_sum(v, target):
    # binary search for lambda
    lo = -1.0
    hi = 2.0
    # expand bounds suitable
    # Use large bounds to guarantee feasible region
    lo = np.min(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust tiny diff
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        # distribute among free variables
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust all equally
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def subgradient_update(h, target_sum, step):
    n = h.shape[0]
    # compute correlation for (h, 1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)  # can be negative
    d = abs(shift)
    if d == 0:
        # no useful direction, pick next largest shift
        # compute second max maybe; for simplicity skip update
        return h
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[:n-d] += h[d:]
        grad[d:] += h[:n-d]
    else:
        grad[d:] += h[:n-d]
        grad[:n-d] += h[d:]
    # step in positive direction (increase overlap)
    new_h = h + step * grad
    # project to feasible set
    new_h = project_box_sum(new_h, target_sum)
    return new_h

def optimize_h(initial_h, max_iter, step0):
    n = initial_h.shape[0]
    target_sum = n / 2.0
    h = initial_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(max_iter):
        # decreasing step
        step = step0 / (1.0 + it * 0.01)
        h = subgradient_update(h, target_sum, step)
        cur = compute_c5(h)
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
    return best_h, best_val

import numpy as np
import time
import random

# ------------------------------------------------------------
# helper: project a vector onto the intersection of the box [0,1]^n
# with the affine hyperplane sum(x)=target
# ------------------------------------------------------------
def _project_box_sum(v, target):
    """
    Euclidean projection of v onto {x | 0<=x_i<=1, sum(x)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / x.size
            x = np.clip(x, 0.0, 1.0)
    return x


# ------------------------------------------------------------
# compute C5 for a discrete vector h
# ------------------------------------------------------------
def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


# ------------------------------------------------------------
# one subgradient step aimed at reducing the maximal overlap
# ------------------------------------------------------------
def _subgrad_step(h, target_sum, step):
    n = h.size
    # correlation array (raw, not scaled)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # shift in [-n+1, n-1]
    d = abs(shift)
    if d == 0:                     # unlikely, but safeguard
        return h
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[: n - d] += h[d:]
        grad[d:]       += h[: n - d]
    else:  # shift < 0   (use symmetry)
        grad[d:]       += h[: n - d]
        grad[: n - d]  += h[d:]
    # ascent direction for the inner product, which
    # reduces the max correlation
    h_new = h + step * grad
    # project back to feasible set
    h_new = _project_box_sum(h_new, target_sum)
    return h_new


# ------------------------------------------------------------
# local optimisation starting from a feasible h
# ------------------------------------------------------------
def _optimize_local(h0, max_iter, step0):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _compute_c5(h)
    for it in range(max_iter):
        # simple diminishing step size
        step = step0 / (1.0 + it * 0.02)
        h = _subgrad_step(h, target, step)
        cur = _compute_c5(h)
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # ----- decide on the discretisation size -----------------
    if "initial_h_values" in globals():
        init_vec = np.array(initial_h_values, dtype=float)
        n = init_vec.size
        target = n / 2.0
        # make sure the supplied vector respects the constraints
        if abs(np.sum(init_vec) - target) > 1e-9:
            init_vec = _project_box_sum(init_vec, target)
        starters = [init_vec]
    else:
        n = 200                 # default size when no parent is given
        target = n / 2.0
        starters = []

    # additional random starts (always keep at least 4 attempts)
    while len(starters) < 4:
        rnd = np.random.rand(n)
        rnd = _project_box_sum(rnd, target)
        starters.append(rnd)

    best_h = None
    best_c5 = float("inf")

    # ----- main optimisation loop (multiple restarts) --------
    # allocate a modest number of iterations per start; adapt to remaining time
    base_iter = 2500
    step0 = 0.4

    for start_vec in starters:
        if time.time() - start_time > budget_s * 0.95:
            break
        # allocate iterations proportionally to remaining budget
        remaining = budget_s - (time.time() - start_time)
        max_iter = int(min(base_iter, remaining * 30))   # rough 30 iter per second
        h_opt, c5_opt = _optimize_local(start_vec, max_iter, step0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # ----- final polishing: tiny stochastic tweaks ---------
    # keep attempting random perturbations as long as time permits
    n = best_h.size
    target = n / 2.0

    return ([float(v) for v in best_h], float(best_c5), int(n))
