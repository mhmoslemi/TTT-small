def project_onto_simplex_box(a, S):
    # a: np.ndarray, shape (n,)
    # returns x satisfying sum(x)=S, 0<=x<=1.
    # Use bisection for lambda:
    lo = -np.max(a) - 1.0 # ensure a_i + lo <=0
    hi = 1 - np.min(a) + 1.0 # ensures a_i+hi >=1
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(a + mid, 0.0, 1.0)
        total = x.sum()
        if total > S:
            hi = mid
        else:
            lo = mid
    return np.clip(a + (lo+hi)/2.0, 0.0, 1.0)

def project_box_sum(v, target):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        total = proj.sum()
        if total > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # small adjustment to meet exact sum due to numerical tolerance
    diff = target - proj.sum()
    if abs(diff) > 1e-10:

        free = (proj > 0) & (proj < 1)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0) # ensure bounds
    return proj

def project_box_sum(v, target):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        total = proj.sum()
        if total > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # Adjust for numerical tolerance
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0) & (proj < 1)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)

    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    return conv.max() * dx, conv

    n = h.shape[0]
    max_val = conv.max()
    idx_max = np.where(conv == max_val)[0]
    g = np.zeros_like(h)
    i = np.arange(n)

        j = i + shift

        g[mask] += 1 - h[j[mask]]

        mask2 = (k >= 0) & (k < n)
        g[mask2] -= h[k[mask2]]
    if len(idx_max) > 1:

    return g

import numpy as np
import time
import random

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto the set {0<=x<=1, sum(x)=target}."""
    lo = -np.max(v) - 1.0           # makes v+lo <= 0  → sum=0
    hi = 1.0 - np.min(v) + 1.0      # makes v+hi >= 1  → sum=n
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_score(h: np.ndarray) -> tuple[float, np.ndarray]:
    """Return C5 value and the full correlation array."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx, conv


def _subgrad_from_conv(h: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k ∫h(1‑h_k) for the shifts attaining the maximum."""
    n = h.size
    max_val = conv.max()
    idx_max = np.where(conv == max_val)[0]
    g = np.zeros_like(h)
    i = np.arange(n)
    for idx in idx_max:
        shift = idx - (n - 1)                     # integer shift
        j = i + shift
        mask_j = (j >= 0) & (j < n)
        g[mask_j] += 1.0 - h[j[mask_j]]
        k = i - shift
        mask_k = (k >= 0) & (k < n)
        g[mask_k] -= h[k[mask_k]]
    if len(idx_max) > 1:
        g /= len(idx_max)
    return g


def _random_feasible(n: int, target: float) -> np.ndarray:
    """Generate a random point in the feasible set."""
    v = np.random.rand(n)
    return _project_box_sum(v, target)


def _upsample_initial(init, n_target: int) -> np.ndarray:
    """Linearly up‑sample a possibly existing initial vector."""
    init = np.asarray(init, dtype=float)
    n0 = init.size
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x, x0, init)


def run(seed: int = 42, budget_s: int = 1100):
    """Search for a step function with a low C5 value."""
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # 1.  Choose discretisation size and an initial feasible vector
    # --------------------------------------------------------------
    n = 120                     # even ⇒ sum = n/2 exactly
    target_sum = n / 2.0

    if "initial_h_values" in globals():
        h = _upsample_initial(initial_h_values, n)
    else:
        # a cosine shape (peaks at the ends, valley in the middle)
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        h = 0.5 + 0.5 * np.cos(np.pi * xs)   # values in [0,1]
    h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_score, best_conv = _c5_score(best_h)

    start_time = time.time()
    outer = 0

    # --------------------------------------------------------------
    # 2.  Main optimisation loop (projected sub‑gradient + SA)
    # --------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.95:
        # start point for this outer iteration
        if np.random.rand() < 0.3:
            cur_h = best_h.copy()
        else:
            cur_h = _random_feasible(n, target_sum)

        cur_score, cur_conv = _c5_score(cur_h)

        # number of inner steps – modest, but many outer cycles fit the budget
        inner_limit = 1000
        inner = 0
        while inner < inner_limit and time.time() - start_time < budget_s * 0.95:
            g = _subgrad_from_conv(cur_h, cur_conv)

            # simple diminishing step size
            step = 0.4 / np.sqrt(outer * inner_limit + inner + 1.0)

            cand = cur_h - step * g
            cand = _project_box_sum(cand, target_sum)

            cand_score, cand_conv = _c5_score(cand)

            if cand_score < cur_score:
                cur_h, cur_score, cur_conv = cand, cand_score, cand_conv
                if cand_score < best_score:
                    best_h, best_score, best_conv = cand.copy(), cand_score, cand_conv
            else:

                if np.random.rand() < np.exp(-(cand_score - cur_score) / temp):

            inner += 1
        outer += 1

    # final safety projection and exact scoring
    best_h = _project_box_sum(best_h, target_sum)
    final_score, _ = _c5_score(best_h)

    return (best_h, float(final_score), n)
