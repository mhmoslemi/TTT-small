def project_box_hyperplane(y, target, low=0.0, high=1.0, tol=1e-9, max_iter=100):
    # Determine lambda such that sum_i clip(y_i - lam, low, high) = target.
    # Use bisection.
    # Since sum_i clip(y_i - lam) is decreasing in lam.
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        lam = (lo+hi)/2
        proj = np.clip(y - lam, low, high)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - lam, low, high)

import time, math
import numpy as np
from scipy.optimize import minimize

def compute_raw_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full'))

def compute_c5(h):
    n = h.size
    return compute_raw_c5(h) * (2.0 / n)

def proj_box_sum(y, target, low=0.0, high=1.0, tol=1e-9, max_iter=100):
    # Bisection on lambda such that sum(clip(y - lam)) = target
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(y - lam, low, high)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return proj

def grad_for_shift(h, s):
    n = h.size
    grad = np.ones_like(h)
    if s >= 0:
        if s < n:
            grad[:n - s] -= h[s:]
            grad[s:] -= h[:n - s]
    else:
        t = -s
        if t < n:
            grad[t:] -= h[:n - t]
            grad[:n - t] -= h[t:]
    return grad

def gradient_max_corr(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    s = idx - (n - 1)
    return grad_for_shift(h, s)

def random_feasible(n, target, rng):
    y = rng.random(n)
    return proj_box_sum(y, target)

def subgradient_descent(h0, target, rng, max_iter=2000, step0=0.4):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_raw = compute_raw_c5(h)
    for it in range(1, max_iter + 1):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = int(np.argmax(corr))
        s = idx - (n - 1)
        grad = grad_for_shift(h, s)
        step = step0 / math.sqrt(it)
        h_new = proj_box_sum(h - step * grad, target)
        new_raw = compute_raw_c5(h_new)
        # accept if improvement
        if new_raw < np.max(corr):
            h = h_new
        # keep best overall
        if new_raw < best_raw:
            best_raw = new_raw
            best_h = h_new.copy()
    return best_h, best_raw

def hill_climb(h0, target, rng, steps=1500):
    h = h0.copy()
    n = h.size
    best_raw = compute_raw_c5(h)
    best_h = h.copy()
    for _ in range(steps):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_dec = h[i]
        max_inc = 1.0 - h[j]
        max_delta = min(max_dec, max_inc)
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_raw = compute_raw_c5(h_new)
        if new_raw < best_raw:
            best_raw = new_raw
            best_h = h_new.copy()
            h = h_new
    return best_h, best_raw

def slsqp_refine(h0, target, maxiter=200):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    def obj(x):
        return compute_raw_c5(x)
    def jac(x):
        return gradient_max_corr(x)
    try:
        res = minimize(obj, h0, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.success:
            x = proj_box_sum(res.x, target)
            return x, compute_raw_c5(x)
    except Exception:
        pass
    return h0, compute_raw_c5(h0)

    else:
        h0 = random_feasible(n, target, rng)
        best_h, best_raw = slsqp_refine(h0, target)
    # main loop

        # early break if very good

import time, math
import numpy as np
from scipy.optimize import minimize

def compute_raw_c5(h):
    """Maximum (unscaled) overlap integral on the discrete grid."""
    return np.max(np.correlate(h, 1 - h, mode="full"))


def compute_c5(h):
    """Scaled C5 = max overlap * dx, where dx = 2 / n."""
    n = h.size
    return compute_raw_c5(h) * (2.0 / n)


def proj_box_sum(y, target, low=0.0, high=1.0, tol=1e-9, max_iter=100):
    """
    Projection of a vector y onto the intersection of the box [low,high]^n
    and the hyperplane sum(x) = target by a simple bisection on a Lagrange
    multiplier λ.
    """
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(y - lam, low, high)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - lam, low, high)


def grad_for_shift(h, s):
    """Gradient of the raw overlap for a fixed shift s."""
    n = h.size
    g = np.ones_like(h)
    if s >= 0:
        if s < n:
            g[: n - s] -= h[s:]
            g[s:] -= h[: n - s]
    else:
        t = -s
        if t < n:
            g[t:] -= h[: n - t]
            g[: n - t] -= h[t:]
    return g


def gradient_max_corr(h):
    """Gradient of the max overlap (unscaled) w.r.t. the vector h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)
    return grad_for_shift(h, s)


def random_feasible(n, target, rng):
    """Generate a random vector in [0,1]^n with prescribed sum = target."""
    y = rng.random(n)
    return proj_box_sum(y, target)


def subgradient_descent(h0, target, rng, max_iter=2000, step0=0.5):
    """Simple sub‑gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_raw = compute_raw_c5(h)
    old_raw = best_raw
    for it in range(1, max_iter + 1):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = int(np.argmax(corr))
        s = idx - (n - 1)
        grad = grad_for_shift(h, s)
        step = step0 / math.sqrt(it)
        h_new = proj_box_sum(h - step * grad, target)
        new_raw = compute_raw_c5(h_new)
        # monotone descent step
        if new_raw < old_raw:
            h = h_new
            old_raw = new_raw
        if new_raw < best_raw:
            best_raw = new_raw
            best_h = h_new.copy()
    return best_h, best_raw


def hill_climb(h0, target, rng, steps=1500):
    """Random pairwise mass transfers that keep the sum fixed."""
    h = h0.copy()
    n = h.size
    best_raw = compute_raw_c5(h)
    best_h = h.copy()
    for _ in range(steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        dec = h[i]
        inc = 1.0 - h[j]
        delta = min(dec, inc) * rng.random()
        if delta <= 0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_raw = compute_raw_c5(h_new)
        if new_raw < best_raw:
            best_raw = new_raw
            best_h = h_new.copy()
            h = h_new
    return best_h, best_raw


def slsqp_refine(h0, target, maxiter=200):
    """SLSQP local refinement of a feasible point."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def obj(x):
        return compute_raw_c5(x)

        return gradient_max_corr(x)

    try:
        res = minimize(

            method="SLSQP",
            jac=jac,

        )
        if res.success:
            x = proj_box_sum(res.x, target)
            return x, compute_raw_c5(x)
    except Exception:
        pass
    return h0, compute_raw_c5(h0)

            for _ in range(rng.integers(1, 4)):

        h_hc, raw_hc = hill_climb(h_sd, target, rng, steps=800)

        # final SLSQP polish
        h_opt, raw_opt = slsqp_refine(h_hc, target, maxiter=150)

        if raw_opt < best_raw:

            # optional early stop if we already beat the target bound
            if best_raw * (2.0 / n) < 0.3808:
