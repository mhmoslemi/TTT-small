        return np.max(np.correlate(v, 1.0 - v, mode='full')) * dx

    # initialize from provided initial_h_values if it matches size n

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------
    # Parameters
    # ---------------------------------------------------------
    n = 200                     # number of sample points (must be even)
    target = n // 2             # required sum of h_i (so that integral = 1)
    dx = 2.0 / n                # discretisation step

    # ---------------------------------------------------------
    # Helper: compute the overlap bound C5 for a given vector
    # ---------------------------------------------------------
    def compute_c5(v):
        # correlation of h with (1‑h), full mode, scaled by dx
        return np.max(np.correlate(v, 1.0 - v, mode='full')) * dx

    # ---------------------------------------------------------
    # Start from a provided construction if possible
    # ---------------------------------------------------------
    best_h = None
    best_c5 = np.inf
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size == n and np.isclose(init.sum() * dx, 1.0, atol=1e-12):
            c0 = compute_c5(init)
            best_h = init.copy()
            best_c5 = float(c0)

    # ---------------------------------------------------------
    # Otherwise begin from a random binary vector with the correct sum
    # ---------------------------------------------------------
    if best_h is None:
        cur_h = np.zeros(n, dtype=float)
        cur_h[:target] = 1.0

    if cur_c5 < best_c5:

        best_c5 = cur_c5

    # ---------------------------------------------------------
    # Simulated‑annealing style hill climbing on binary swaps
    # ---------------------------------------------------------
    T = 0.02            # initial temperature
    decay = 0.9999      # exponential cooling factor

    while True:
        if time.time() - start > budget_s * 0.95:
