def propose_move(h, step):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # Determine allowable delta: limited by 0/1 constraints on both
    max_inc_i = 1 - h[i]
    max_dec_i = h[i]
    max_inc_j = 1 - h[j]
    max_dec_j = h[j]
    # We'll move delta from j to i: delta limited by max_inc_i and max_dec_j
    delta_max = min(max_inc_i, max_dec_j)
    delta_min = -min(max_inc_j, max_dec_i)   # moving opposite direction (i->j)
    # randomly choose delta within [-step, step] intersect [delta_min, delta_max]
    lo = max(-step, delta_min)
    hi = min(step, delta_max)
    if lo >= hi:
        return None
    delta = np.random.uniform(lo, hi)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def enforce_sum_constraint(h, target_sum, rng):
    s = h.sum()
    diff = target_sum - s
    eps = 1e-12
    attempts=0
    max_attempts=10000
    while abs(diff) > 1e-9 and attempts < max_attempts:
        if diff > 0:
            # need to increase: pick index where h[i] < 1
            idx = rng.choice(np.where(h < 1 - 1e-12)[0])
            max_inc = 1 - h[idx]
            delta = min(diff, max_inc)
            h[idx] += delta
            diff -= delta
        else:
            idx = rng.choice(np.where(h > 1e-12)[0])
            max_dec = h[idx]
            delta = min(-diff, max_dec)
            h[idx] -= delta
            diff += delta
        attempts+=1
    # In case cannot adjust (should not happen)
    return h

def propose_move(h, step, rng):
    n = len(h)
    if n < 2:
        return None
    i, j = rng.choice(n, size=2, replace=False)
    # compute allowed delta
    hi, hj = h[i], h[j]
    # we allow moving mass from j to i (+delta to i, -delta to j)
    max_inc_i = 1.0 - hi
    max_dec_j = hj
    max_move_pos = min(max_inc_i, max_dec_j)
    # also allow moving opposite direction: from i to j (negative delta)
    max_inc_j = 1.0 - hj
    max_dec_i = hi
    max_move_neg = min(max_inc_j, max_dec_i)
    # delta can be in [-max_move_neg, max_move_pos]
    lo = max(-step, -max_move_neg)
    hi = min(step, max_move_pos)
    if lo >= hi:
        return None
    delta = rng.uniform(lo, hi)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------

def erdos_objective(h):
    """Return C5 bound for step function h (discrete version)."""
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def propose_move(h, step, rng):
    """Return a new feasible h after moving mass between two indices."""
    n = len(h)
    if n < 2:
        return None
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]

    # mass from j to i (positive delta)
    max_inc_i = 1.0 - hi
    max_dec_j = hj
    max_pos = min(max_inc_i, max_dec_j)

    # mass from i to j (negative delta)
    max_inc_j = 1.0 - hj
    max_dec_i = hi
    max_neg = min(max_inc_j, max_dec_i)

    lo = max(-step, -max_neg)
    hi_ = min(step, max_pos)
    if lo >= hi_:
        return None
    delta = rng.uniform(lo, hi_)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h


def enforce_sum_constraint(h, target, rng):
    """Adjust h in-place so that sum(h) == target while staying in [0,1]."""
    s = h.sum()
    diff = target - s
    n = len(h)
    eps = 1e-12
    attempts = 0
    max_attempts = 5 * n
    while abs(diff) > 1e-12 and attempts < max_attempts:
        if diff > 0:
            # need to increase total mass
            idx_candidates = np.where(h < 1 - eps)[0]
            if idx_candidates.size == 0:
                break
            i = rng.choice(idx_candidates)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            # need to decrease total mass
            idx_candidates = np.where(h > eps)[0]
            if idx_candidates.size == 0:
                break
            i = rng.choice(idx_candidates)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
        attempts += 1

    # If still off (unlikely), apply a tiny uniform correction and clip
    if abs(diff) > 1e-12:
        adj = diff / n
        h += adj
        np.clip(h, 0.0, 1.0, out=h)
        # final tiny fix using one more transfer if needed
        s = h.sum()
        diff = target - s
        if abs(diff) > 1e-12:
            # give up fractional error
            h *= (target / s)
            np.clip(h, 0.0, 1.0, out=h)
    return h


def upsample_h(h, new_n, rng):
    """Linear interpolation of h to a larger grid of size new_n."""
    n = len(h)
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    np.clip(h_new, 0.0, 1.0, out=h_new)
    target = new_n / 2.0
    enforce_sum_constraint(h_new, target, rng)
    return h_new


def local_search(start_h, max_iter=20000, max_step=0.5, time_limit=1e9, start_time=0.0):
    """Simple Simulated‑Annealing style hill‑climb."""
    n = len(start_h)
    target_sum = n / 2.0
    cur_h = start_h.copy()
    cur_val = erdos_objective(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    for it in range(max_iter):
        # time guard
        if time.time() - start_time > time_limit:
            break
        frac = it / max_iter
        step = max_step * (1.0 - frac)
        temp = step  # temperature proportional to step size
        new_h = propose_move(cur_h, step, rng)
        if new_h is None:
            continue
        new_val = erdos_objective(new_h)
        # acceptance rule
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (temp + 1e-12)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
        # occasional random jitter around the best point
        if (it + 1) % 1000 == 0 and rng.random() < 0.1:
            cur_h = best_h + rng.normal(scale=0.05, size=n)
            np.clip(cur_h, 0.0, 1.0, out=cur_h)
            enforce_sum_constraint(cur_h, target_sum, rng)
            cur_val = erdos_objective(cur_h)
    return best_h, best_val
