def project_onto_box_sum(y, S, max_iter=50):

        else:

    return x

def _evaluate_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    # linear correlation with zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

    """
    Euclidean projection of ``v`` onto the set
        {x | 0 ≤ x_i ≤ 1,  Σ x_i = target_sum }.
    Uses a simple bisection on the scalar shift τ such that
        x_i = clip(v_i - τ, 0, 1)  satisfies the sum constraint.
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):          # enough for double precision tolerance
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0.0, 1.0)


def _gradient_step(h: np.ndarray, alpha: float, beta: float) -> np.ndarray:
    """
    One gradient‑ascent step for the soft‑minimum of the autocorrelation.
    Returns the projected candidate.
    """
    n = h.size
    # autocorrelation c_s = Σ_i h_i h_{i+s}
    c = np.correlate(h, h, mode="full")          # length 2n‑1, symmetric
    # soft‑min weights (stable softmax)
    c_max = c.max()
    w = np.exp(-alpha * (c - c_max))
    p = w / w.sum()                              # p_s ≥ 0, Σ p_s = 1

    # convolution p * h gives Σ_s p_s h_{i+s}
    conv_full = np.convolve(p, h, mode="full")
    conv = conv_full[n - 1 : n - 1 + n]          # exactly n entries

    dx = 2.0 / n
    step = beta * 2.0 * dx * conv                # direction that raises soft‑min
    cand = h + step
    # enforce box & integral constraints
    cand = _project_onto_box_sum(cand, n / 2.0)
    return cand


def _optimize_from_start(h_start: np.ndarray,
                         time_limit: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected gradient ascent with a simple cooling schedule."""
    n = h_start.size
    target_sum = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_val = _evaluate_c5(h)

    alpha = 5.0        # initial softness
    beta = 1.0         # step‑size multiplier
    it = 0

    while True:
        if time.time() - start_time > time_limit * 0.95:
            break

        cand = _gradient_step(h, alpha, beta)
        cand_val = _evaluate_c5(cand)

        # keep the best ever seen
        if cand_val < best_val:
            best_val = cand_val
            best_h = cand.copy()

        # accept if we made progress; otherwise shrink the step size a bit
        cur_val = _evaluate_c5(h)
        if cand_val <= cur_val:
            h = cand
        else:
            beta *= 0.9
            if beta < 1e-4:
                beta = 1e-4

        # anneal the softness parameter slowly
        if it % 200 == 0 and it > 0:
            alpha = min(alpha * 1.3, 200.0)

        it += 1

    # final local random perturbations (small‑scale hill climbing)
    remain = time_limit - (time.time() - start_time)
    pert_iters = int(min(2000, remain / 0.001))  # cheap tiny steps
    sigma = 0.05
    for _ in range(pert_iters):
        delta = rng.normal(scale=sigma, size=n)
        cand = best_h + delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_onto_box_sum(cand, target_sum)
        cand_val = _evaluate_c5(cand)
        if cand_val < best_val:
            best_val = cand_val
            best_h = cand
            sigma = 0.05            # reset step size after improvement
        else:
            sigma *= 0.998          # slowly shrink exploration radius

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize the Erdős minimum‑overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Choose a starting feasible vector
    # ------------------------------------------------------------------
    try:
        # ``initial_h_values`` is supplied by the harness (if present)
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h0 = _project_onto_box_sum(h0, n / 2.0)
    except NameError:
        # No supplied seed – create a random feasible start (even length)
        n = 100                     # a modest, even size
        h0 = rng.random(n)
        h0 = _project_onto_box_sum(h0, n / 2.0)

    # ------------------------------------------------------------------
    # Run several independent restarts until time expires
    # ------------------------------------------------------------------
    best_h = None
    best_val = float('inf')
    remaining = budget_s
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.97:
            break

        # allocate a slice of the remaining budget for this trial
        trial_budget = min(remaining, 200.0)   # at most 200 s per trial
        h_opt, val_opt = _optimize_from_start(h0, trial_budget, start, rng)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

        # prepare a fresh random start for the next trial
        h0 = rng.random(n)
        h0 = _project_onto_box_sum(h0, n / 2.0)

        remaining = budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    if best_h is None:                     # should never happen
        best_h = h0
        best_val = _evaluate_c5(best_h)

    # guarantee feasibility one last time
    best_h = _project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c5 = _evaluate_c5(best_h)

    return best_h, float(final_c5), best_h.size
