def project_to_bounds_sum(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > target:
        # Need to reduce sum
        # sort indices by h descending
        idx = np.argsort(-h)
        for i in idx:
            if s <= target: break
            reduction = min(h[i], s - target)
            h[i] -= reduction
            s -= reduction
    elif s < target:
        idx = np.argsort(h)  # smallest first
        for i in idx:
            if s >= target: break
            addition = min(1 - h[i], target - s)
            h[i] += addition
            s += addition
    return h

import time, math, numpy as np

def evaluate_h(arr, dx):
    # Compute max overlap integral for vector arr (0/1 or real)
    # arr assumed 1D numpy array
    # Use np.correlate
    # Convert to float to avoid overflow
    a = arr.astype(np.float64)
    b = (1.0 - a)
    # np.correlate returns array of length 2n-1
    raw = np.correlate(a, b, mode='full')
    return raw.max() * dx

def binary_random_solution(n, half, rng):
    h = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1
    return h

def hill_climb_binary(n, half, rng, time_limit, init_h=None):
    dx = 2.0 / n
    if init_h is not None:
        h = init_h.astype(np.int8).copy()
    else:
        h = binary_random_solution(n, half, rng)
    cur_val = evaluate_h(h, dx)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    # parameters
    max_temp = 0.07
    min_temp = 0.001
    max_no_improve = 15000
    no_improve = 0
    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        # temperature schedule
        frac = elapsed / time_limit
        T = max_temp * (1 - frac) + min_temp * frac
        # recompute indices
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            # reinitialize
            h = binary_random_solution(n, half, rng)
            cur_val = evaluate_h(h, dx)
            no_improve = 0
            continue
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i] = 0
        h[j] = 1
        new_val = evaluate_h(h, dx)
        delta = new_val - cur_val
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 1e-12:
                prob = math.exp(-delta / T)
                if rng.random() < prob:
                    accept = True

            h[j] = 0
            no_improve += 1

            # random restart
            h = binary_random_solution(n, half, rng)
            cur_val = evaluate_h(h, dx)
            no_improve = 0
    return best_h, best_val

import time, math, numpy as np

def evaluate_h(arr, dx):
    """Return max_k ∫ h(x)(1–h(x+k))dx for the discrete array."""
    a = arr.astype(np.float64)
    b = 1.0 - a
    raw = np.correlate(a, b, mode="full")
    return raw.max() * dx


def binary_random_solution(n, half_int, rng):
    """Uniform random binary vector with exactly half_int ones."""
    h = np.zeros(n, dtype=np.int8)
    h[rng.choice(n, size=half_int, replace=False)] = 1
    return h


def hill_climb_binary(n, half_int, rng, time_limit, init_h=None):
    """Simulated‑annealing on binary vectors (swap 1↔0)."""
    dx = 2.0 / n
    # start vector
    if init_h is not None:
        h = init_h.astype(np.int8).copy()
    else:
        h = binary_random_solution(n, half_int, rng)

    cur_raw = np.correlate(h.astype(np.float64), (1 - h).astype(np.float64), mode="full")
    cur_val = cur_raw.max() * dx

    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    max_temp, min_temp = 0.07, 0.001
    max_no_improve = 15000
    no_improve = 0
    p_direct = 0.22                           # probability to use a directed swap

    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        frac = elapsed / time_limit
        T = max_temp * (1 - frac) + min_temp * frac

        # indices of ones and zeros (cheap for n≈70)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        if ones.size == 0 or zeros.size == 0:
            h = binary_random_solution(n, half_int, rng)
            cur_raw = np.correlate(h.astype(np.float64), (1 - h).astype(np.float64), mode="full")
            cur_val = cur_raw.max() * dx
            no_improve = 0
            continue

        # decide which swap to try
        if rng.random() < p_direct:
            # direct move based on the currently worst shift
            max_idx = cur_raw.argmax()
            shift = max_idx - (n - 1)               # shift with maximal overlap
            if shift >= 0:
                mism = np.where((h[: n - shift] == 1) & (h[shift:] == 0))[0]
            else:
                s = -shift
                mism = np.where((h[s:] == 1) & (h[: n - s] == 0))[0] + s
            if mism.size:
                i = rng.choice(mism)                # a 1 participating in the worst overlap
                j = rng.choice(zeros)                # any 0
            else:
                i = rng.choice(ones)
                j = rng.choice(zeros)
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # perform swap
        h[i] = 0
        h[j] = 1

        new_raw = np.correlate(h.astype(np.float64), (1 - h).astype(np.float64), mode="full")
        new_val = new_raw.max() * dx
        delta = new_val - cur_val

        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0:
                if rng.random() < math.exp(-delta / T):
                    accept = True

        if accept:
            cur_val = new_val
            cur_raw = new_raw
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            no_improve += 1

        if no_improve >= max_no_improve:
            # random restart
            h = binary_random_solution(n, half_int, rng)
            cur_raw = np.correlate(h.astype(np.float64), (1 - h).astype(np.float64), mode="full")
            cur_val = cur_raw.max() * dx
            no_improve = 0

    return best_h, best_val


def continuous_refine(n, half, rng, time_limit, start_h):
    """Small mass‑transfer moves on a real vector while keeping the sum = half."""
    dx = 2.0 / n
    h = start_h.astype(np.float64).copy()

    # tiny correction to meet the exact sum constraint
    diff = half - h.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(n)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)

    cur_val = evaluate_h(h, dx)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    max_temp, min_temp = 0.04, 0.0005

    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        frac = elapsed / time_limit
        T = max_temp * (1 - frac) + min_temp * frac

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # amount we are allowed to move from i to j
        max_decrease = h[i]
        max_increase = 1.0 - h[j]
        max_move = min(max_decrease, max_increase)
        if max_move <= 0:
            continue

        step_factor = 0.5 * (1 - frac) + 0.01    # slowly shrinks
        delta = rng.random() * max_move * step_factor

        h[i] -= delta
        h[j] += delta
        new_val = evaluate_h(h, dx)
        delta_obj = new_val - cur_val

        accept = False
        if delta_obj <= 0:
            accept = True
        else:
            if T > 0:
                if rng.random() < math.exp(-delta_obj / T):
                    accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta

    # final projection to enforce the sum exactly (tiny adjustments)
    diff = half - best_h.sum()
    if abs(diff) > 1e-9:
        adjustable = np.where((best_h > 0) & (best_h < 1))[0]
        if adjustable.size:
            add = diff / adjustable.size
            best_h[adjustable] = np.clip(best_h[adjustable] + add, 0.0, 1.0)
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    best_h = np.clip(best_h, 0.0, 1.0)
    final_val = evaluate_h(best_h, dx)
    return best_h, final_val


def run(seed=42, budget_s=1100, **kwargs):

        # fix the exact sum
        diff = half - start_cont.sum()
        if abs(diff) > 1e-12:
            start_cont[0] = np.clip(start_cont[0] + diff, 0.0, 1.0)

        best_cont_h, best_cont_val = continuous_refine(
            n, half, rng, t_cont, start_cont
        )
        if best_cont_val < best_bin_val:
            best_h, best_val = best_cont_h, best_cont_val
        else:
            best_h, best_val = best_bin_h.astype(np.float64), best_bin_val
    else:
        best_h, best_val = best_bin_h.astype(np.float64), best_bin_val

    # --- final sanity check ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = half - best_h.sum()
    if abs(diff) > 1e-9:
        adjustable = np.where((best_h > 0) & (best_h < 1))[0]
        if adjustable.size:
            add = diff / adjustable.size
            best_h[adjustable] = np.clip(best_h[adjustable] + add, 0.0, 1.0)
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_val = evaluate_h(best_h, dx)

    return (best_h, final_val, n)
