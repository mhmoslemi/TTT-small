def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def anneal(h, budget_s):
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    current = h.copy()
    current_c = evaluate(current)
    best = current.copy()
    best_c = current_c
    # parameters
    T0 = 0.1
    T_end = 1e-4
    # total iterations estimate based on budget S: maybe 200k
    iter = 0
    while time.time() - start < budget_s * 0.9: # keep some margin
        # schedule
        t = (time.time() - start) / (budget_s * 0.9)
        T = T0 * (T_end / T0) ** t
        # pick random i,j
        i, j = random.sample(range(n), 2)
        hi = current[i]
        hj = current[j]
        # feasible delta interval
        delta_low = max(-hi, hj - 1.0)  # because delta >= -hi (so hi+delta>=0), delta >= hj-1 (so hj-delta<=1)
        delta_high = min(1.0 - hi, hj)   # hi+delta <=1 and hj-delta >=0
        if delta_low >= delta_high:
            continue
        # Choose delta small magnitude with probability; we allow both small and large
        # maybe restrict to a maximum step size
        max_step = 0.1
        low = max(delta_low, -max_step)
        high = min(delta_high, max_step)
        if low >= high:
            # if small step not possible, allow full
            low = delta_low
            high = delta_high
        delta = random.uniform(low, high)
        # update
        current[i] = hi + delta
        current[j] = hj - delta
        new_c = evaluate(current)
        delta_c = new_c - current_c
        if delta_c < 0 or random.random() < math.exp(-delta_c / T):
            # accept
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = current.copy()
        else:
            # revert
            current[i] = hi
            current[j] = hj
    return best, best_c
```

import time
import math
import random
import numpy as np

def _c5_metric(h):
    """Return the C_5 value for a step function sampled by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _initial_vector():
    """Create an initial feasible vector."""
    # use provided construction if present, otherwise a uniform start
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n0 = h0.size
        # target size – a modest increase gives more freedom without heavy cost
        n = max(100, ((n0 + 9) // 10) * 10)   # round up to a multiple of 10, ≥100
        if n % 2:
            n += 1
        # linear interpolation onto the new grid
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        # fallback – uniform half‑density
        n = 200
        h = np.full(n, 0.5, dtype=np.float64)

    # enforce the sum constraint exactly (sum h = n/2)
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur               # simple scaling keeps ratios
    # clip to the box and correct any remaining deficit
    np.clip(h, 0.0, 1.0, out=h)
    deficit = target - h.sum()
    if abs(deficit) > 1e-12:
        # distribute the tiny remainder to a random admissible entry
        for idx in np.random.permutation(n):
            lo, hi = 0.0, 1.0
            if lo <= h[idx] + deficit <= hi:
                h[idx] += deficit
                break
    return h


def _anneal(start_h, max_seconds):
    """Simulated‑annealing improvement of a step function."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_metric(best_h)
    cur_c = best_c

    start = time.time()
    # temperature schedule
    T0 = 0.04          # start temperature (in C‑unit scale)
    T_end = 1e-6
    max_step = 0.15    # maximum absolute delta per elementary move

    while time.time() - start < max_seconds:
        # linear temperature decay in log‑space
        progress = (time.time() - start) / max_seconds
        T = T0 * (T_end / T0) ** progress

        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        # feasible interval for delta (i receives, j loses)
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        # restrict to modest steps most of the time
        low_cap = max(low, -max_step)
        high_cap = min(high, max_step)
        if low_cap < high_cap:
            delta = random.uniform(low_cap, high_cap)
        else:
            delta = random.uniform(low, high)

        h[i] = hi + delta
        h[j] = hj - delta

        new_c = _c5_metric(h)
        dC = new_c - cur_c
        if dC < 0 or random.random() < math.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h, its C_5 bound and number of samples."""
    random.seed(seed)
    np.random.seed(seed)

    #--- initialise ---------------------------------------------------------
    h0 = _initial_vector()
    n = h0.size

    #--- allocate time -------------------------------------------------------
    total_budget = float(budget_s)
    start_all = time.time()
    # primary annealing (majority of budget)
    primary_time = total_budget * 0.85
    best_h, best_c = _anneal(h0, primary_time)

    #--- a few random restarts (remaining budget) ---------------------------
    remaining = total_budget - (time.time() - start_all)
    while remaining > 5.0:                # at least a few seconds per restart
        # generate a binary starting point (exactly n/2 ones)
        target = n / 2.0
        ones = int(target)
        frac = target - ones
        h_rand = np.zeros(n, dtype=np.float64)
        idx = np.random.permutation(n)
        h_rand[idx[:ones]] = 1.0
        if frac > 0:
            # one fractional entry
            h_rand[idx[ones]] = frac
        # anneal this start for a short slice of the time left
        slice_time = min(remaining * 0.3, 30.0)   # no more than 30 s per slice
        cand_h, cand_c = _anneal(h_rand, slice_time)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h
        remaining = total_budget - (time.time() - start_all)

    #--- final polishing with a few deterministic refinement passes ----------
    # simple hill‑climbing: try tiny redistributions between random pairs
    for _ in range(3):
        improved = False
        for _ in range(2000):
            i, j = random.sample(range(n), 2)
            hi, hj = best_h[i], best_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:

            delta = random.uniform(low, high) * 0.05  # very small tweak
            best_h[i] = hi + delta
            best_h[j] = hj - delta
            new_c = _c5_metric(best_h)
            if new_c + 1e-9 < best_c:
                best_c = new_c
                improved = True
            else:
                best_h[i] = hi
                best_h[j] = hj
        if not improved:
            break

    #--- final safeguard: enforce exact sum (tiny rounding errors) ----------
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        for idx in np.random.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    # compute final bound (use the provided helper if it exists)
    try:
        final_c = evaluate_erdos_solution(best_h)
    except Exception:
        final_c = _c5_metric(best_h)

    return best_h, float(final_c), int(n)
