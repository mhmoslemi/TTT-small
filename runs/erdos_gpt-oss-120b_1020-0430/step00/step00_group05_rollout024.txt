def solve_mip(N, max_time):
    import cvxpy as cp, numpy as np, time
    t_start = time.time()
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == N//2, t >= 0]
    w_vars = []
    for k in range(1, N):
        size = N - k
        w = cp.Variable(size, boolean=True)
        w_vars.append(w)
        # Link constraints
        constraints += [w <= h[:size], w <= h[k:], w >= h[:size] + h[k:] - 1]
        # forward mismatch count
        forward = N//2 - cp.sum(w)
        backward = cp.sum(h[k:]) - cp.sum(w)
        constraints += [t >= forward, t >= backward]
    prob = cp.Problem(cp.Minimize(t), constraints)
    # select solver
    for solver in [cp.CBC, cp.GLPK_MI, cp.SCIP, cp.ECOS_BB]:
        try:
            prob.solve(solver=solver, verbose=False, warm_start=True)
            break
        except Exception:
            continue
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    h_val = np.rint(h.value).astype(int)
    bound = evaluate(h_val)
    return h_val, bound

import time, math, random
import numpy as np

def _evaluate(h: np.ndarray) -> float:
    """Return C5 bound for a binary step function h."""
    n = h.shape[0]
    # correlation of h with (1-h), full mode
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _hill_climb(n: int, h0: np.ndarray, max_iters: int,
                start_time: float, deadline: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style hill climbing on binary vectors."""
    h = h0.astype(np.int8).copy()
    cur_val = _evaluate(h)
    best_h = h.copy()
    best_val = cur_val

    # maintain explicit index lists for fast random swaps
    ones = [i for i, v in enumerate(h) if v == 1]
    zeros = [i for i, v in enumerate(h) if v == 0]

    # starting temperature – a small fraction of the typical score range
    start_T = 0.001
    for it in range(max_iters):
        # enforce overall time budget
        if time.time() >= deadline - 0.1:
            break

        i = random.choice(ones)
        j = random.choice(zeros)

        # perform the swap
        h[i], h[j] = h[j], h[i]
        new_val = _evaluate(h)
        delta = new_val - cur_val

        # linear cooling schedule
        T = start_T * (1.0 - it / max_iters)
        accept = False
        if delta < -1e-12:
            accept = True
        elif T > 0 and random.random() < math.exp(-delta / T):
            accept = True

        if accept:
            cur_val = new_val
            # update index lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = h.copy()
        else:
            # revert the swap
            h[i], h[j] = h[j], h[i]

    return best_h, best_val
