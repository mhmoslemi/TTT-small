def compute_c5(alpha, p, n):
    if alpha < 0 or alpha > 0.5: return large
    if p < 0 or p > 1: return large
    if 1 - alpha == 0: return large
    c = (0.5 - alpha * p) / (1 - alpha)

    h = np.full(n, c, dtype=np.float64)
    h[:left_len] = p
    h[-left_len:] = p
    # compute C5
    corr = np.correlate(h, 1 - h, mode='full')
    t = corr.max() * (2.0 / n)
    return t

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    from scipy.optimize import minimize

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # resolution (must be even to satisfy the integral exactly)
    n_points = 200
    dx = 2.0 / n_points

    # ------------------------------------------------------------
    # evaluation of C5 for a given h (numpy array, values in [0,1])
    # ------------------------------------------------------------
    def eval_c5(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------
    # 1. try the provided parent construction (if any)
    # ------------------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        if init.shape[0] == n_points:
            # enforce the integral (tiny numerical drift is tolerated)
            init = np.clip(init, 0.0, 1.0)
            val = eval_c5(init)
            if val < best_val:
                best_val, best_h = val, init.copy()
    except Exception:
        pass

    # ------------------------------------------------------------
    # 2. search among simple three–level step functions
    #    left/right blocks of length α, height p,
    #    middle part of height c chosen to satisfy the integral.
    # ------------------------------------------------------------
    max_left = n_points // 2
    p_grid = np.linspace(0.0, 1.0, 21)          # 0, 0.05, … , 1
    for left_len in range(0, max_left + 1):
        if time.time() - start_time > budget_s * 0.85:
            break
        alpha = left_len * dx                    # length of each outer block
        if left_len == 0:
            # constant case (p irrelevant, c = 0.5)
            h = np.full(n_points, 0.5, dtype=float)
            val = eval_c5(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
            continue
        denom = 1.0 - alpha
        if denom <= 0.0:
            continue
        for p in p_grid:
            c = (0.5 - alpha * p) / denom
            if not (0.0 <= c <= 1.0):
                continue
            h = np.full(n_points, c, dtype=float)
            h[:left_len] = p
            h[-left_len:] = p
            val = eval_c5(h)
            if val < best_val:
                best_val, best_h = val, h.copy()

    # ------------------------------------------------------------
    # 3. refine the best outer‑block length with a 1‑D continuous
    #    optimisation over p (the outer height).
    # ------------------------------------------------------------
    if best_h is not None:
        # estimate the outer block size from the current best vector
        # (count leading entries that differ from the interior value)
        tol = 1e-8
        left_guess = 0
        for i in range(n_points):
            if abs(best_h[i] - best_h[0]) > tol:
                break
            left_guess += 1
        # try a few nearby lengths
        candidates = {left_guess}
        if left_guess > 0:
            candidates.update({left_guess - 1, left_guess + 1})
        for left_len in list(candidates):
            if not (0 <= left_len <= max_left):
                continue
            if time.time() - start_time > budget_s * 0.90:
                break
            alpha = left_len * dx
            denom = 1.0 - alpha
            if denom <= 0.0:
                continue

            def objective(p_arr):
                p_val = float(p_arr[0])
                c_val = (0.5 - alpha * p_val) / denom
                if not (0.0 <= c_val <= 1.0):
                    return 1e3
                h = np.full(n_points, c_val, dtype=float)
                if left_len > 0:
                    h[:left_len] = p_val
                    h[-left_len:] = p_val
                return eval_c5(h)

            p0 = float(best_h[0]) if left_len > 0 else 1.0

                           options={"maxiter": 60, "ftol": 1e-12})
            p_opt = float(res.x[0])

    # ------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n_points, 0.5, dtype=float)
        best_val = eval_c5(best_h)

    return (best_h, float(best_val), int(n_points))
