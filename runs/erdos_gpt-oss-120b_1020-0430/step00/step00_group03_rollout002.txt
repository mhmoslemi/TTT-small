def subgradient_refine(h, max_iters=2000, step0=0.1):
    best = h.copy()
    best_score = compute_c5(best)
    for t in range(1, max_iters+1):
        # compute correlation vector
        corr = np.correlate(best, 1 - best, mode='full')
        idx = np.argmax(corr)
        # compute lag
        lag = idx - (n_points - 1)
        # compute gradient
        grad = np.zeros_like(best)
        if lag >= 0:
            grad[:n_points - lag] += 1 - best[lag:]
            grad[lag:] += -best[:n_points - lag]
        else:
            p = -lag
            grad[p:] += 1 - best[:n_points - p]
            grad[:n_points - p] += -best[p:]
        # step size decreasing
        step = step0 / np.sqrt(t)
        new_h = best - step * grad
        new_h = project_boxed_simplex(new_h, target_sum, 1.0)
        new_score = compute_c5(new_h)
        if new_score < best_score:
            best = new_h
            best_score = new_score
    return best, best_score
```

import time
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Find a step function h on [0,2] that yields a low C₅ bound.

    Returns
    -------
    h_values : np.ndarray
        The discretized step function, length n_points, values in [0,1].
    c5_bound : float
        The computed C₅ value for h.
    n_points : int
        Number of discretization points (length of h_values).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimensions and initial feasible vector (if supplied)
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        n = int(init.shape[0])
        init_h = np.array(init, dtype=np.float64)
    else:
        n = 70
        init_h = None

    target_sum = n / 2.0          # ∑h = n/2  (since dx = 2/n)
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Projection onto the capped simplex {0≤x_i≤1, Σx_i = target_sum}
    # ------------------------------------------------------------------
    def project_boxed_simplex(v: np.ndarray) -> np.ndarray:
        lo = np.min(v) - 1.0      # makes all entries =1 → sum = n
        hi = np.max(v)            # makes all entries =0 → sum = 0
        # binary search on the threshold τ
        for _ in range(45):
            tau = (lo + hi) * 0.5
            x = np.clip(v - tau, 0.0, 1.0)
            s = x.sum()
            if s > target_sum:
                lo = tau
            else:
                hi = tau
        return np.clip(v - tau, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Utility helpers
    # ------------------------------------------------------------------
    def random_feasible() -> np.ndarray:
        v = rng.random(n)
        return project_boxed_simplex(v)

    def compute_c5(h: np.ndarray) -> float:
        corr = np.correlate(h, 1.0 - h, mode="full")
        return corr.max() * dx

    CR = 0.9
