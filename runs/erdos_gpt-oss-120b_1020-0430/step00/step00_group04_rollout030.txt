def compute_c5_linear(h):
    N = len(h)
    dx = 2.0 / N
    # compute correlation via FFT
    L = 1 << ((2*N - 1).bit_length())  # next power of two
    # reversed (1 - h)
    g = 1 - h
    # For convolution: correlation = convolve(h, g[::-1], mode='full')
    # So we will compute FFT of h and reversed g.
    f_h = np.fft.rfft(np.concatenate([h, np.zeros(L - N)]))
    f_g = np.fft.rfft(np.concatenate([g[::-1], np.zeros(L - N)]))
    conv = np.fft.irfft(f_h * f_g, n=L)
    # The relevant region is length 2N-1, indices 0..2N-2.
    conv = conv[:2*N-1]
    max_corr = conv.max()
    return max_corr * dx

def gradient_for_lag(v, lag):
    n = v.shape[0]
    if lag == 0:
        return 2 * v
    if lag > 0:
        grad = np.zeros(n)
        grad[:n-lag] += v[lag:]
        grad[lag:] += v[:n-lag]
        return grad
    else: # lag <0 treat same abs
        return gradient_for_lag(v, -lag)

    idx_desc = np.argsort(-grad)  # high to low

import numpy as np
import time
import math

# ----- helper functions (top level) ---------------------------------

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] while fixing its sum to target."""
    lo, hi = -5.0, 5.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        u = np.clip(v - lam, 0.0, 1.0)
        if u.sum() > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _c5_value(vec: np.ndarray, dx: float) -> float:
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx approximated by full correlation."""
    corr = np.correlate(vec, 1 - vec, mode="full")
    return float(corr.max() * dx)


def _lag_of_min_autocorr(vec: np.ndarray, n: int) -> int:
    """Return signed lag where the linear autocorrelation of vec is minimal."""
    auto = np.correlate(vec, vec, mode="full")
    idx = int(auto.argmin())
    return idx - (n - 1)


def _gradient_for_lag(vec: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the autocorrelation at a given lag (linear, zero‑padded)."""
    n = vec.shape[0]
    l = abs(lag)
    if l == 0:
        return 2.0 * vec
    g = np.zeros_like(vec)
    # forward part
    g[: n - l] += vec[l:]
    # backward part
    g[l:] += vec[: n - l]
    return g


# ----- main entry point ----------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a step function h on [0,2] (as a numpy array),
    its C₅ bound, and the number of discretisation points.
    """
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    try:
        init_vals = initial_h_values  # provided by the harness if any
        n0 = len(init_vals)
    except NameError:
        init_vals = None
        n0 = 0

    # use at least 200 points (still well below the 1 000 limit)
    n = max(200, n0)
    if n > 1000:
        n = 1000

    # ----- initial construction -----------------------------------------
    if init_vals is not None and n0 == n:
        h = np.array(init_vals, dtype=float)
    elif init_vals is not None:
        # interpolate the parent construction onto the new grid
        src_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        tgt_x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(tgt_x, src_x, init_vals)
    else:
        # smooth cosine start (mean 0.5, respects the sum constraint)
        i = np.arange(n, dtype=float)
        h = 0.5 + 0.5 * np.cos(2.0 * np.pi * i / n)

    h = _project_to_sum(h, target_sum)

    step_initial = 0.2          # max amount moved between two cells

    while True:
        now = time.time()
        if now - start_time > budget_s - 0.5:
            break

        fraction = 1.0 - elapsed / budget_s
        step = max(step_initial * fraction, 1e-6)
        temperature = max(T0 * fraction, 1e-12)

        # compute the lag where autocorrelation is currently smallest
        lag = _lag_of_min_autocorr(h, n)

        # decide which kind of move to propose
        if rng.random() < p_targeted:

            src_candidates = np.where((grad == grad.min()) & (h > 1e-12))[0]

                src = src[rng.integers(len(src))] if src.size else None
            else:
                src = src_candidates[0]

            # target: high gradient & room to increase
            tgt_candidates = np.where((grad == grad.max()) & (h < 1 - 1e-12))[0]
            if tgt_candidates.size == 0:
                tgt = np.where(h < 1 - 1e-12)[0]
                tgt = tgt[rng.integers(len(tgt))] if tgt.size else None
            else:
                tgt = tgt_candidates[0]

            if src is None or tgt is None or src == tgt:
                # fallback to a simple random pair move
                i, j = rng.integers(0, n, size=2)
                while i == j:
                    j = rng.integers(0, n)
                src, tgt = i, j
        else:

            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            src, tgt = i, j

        # maximum transferable amount respecting the box constraints
        max_transfer = min(h[src], 1.0 - h[tgt])
        if max_transfer <= 0.0:

        h_new[src] -= delta
        h_new[tgt] += delta
