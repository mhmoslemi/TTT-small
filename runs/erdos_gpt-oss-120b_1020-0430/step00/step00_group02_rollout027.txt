def hill_climb(h_start, max_iters, time_limit, best_global_c5):
   h = h_start.copy()
   cur_c5 = evaluate_c5(h)
   best_h_local = h.copy()
   best_c5_local = cur_c5

   n = len(h)
   # We can optionally compute active shift index = np.argmax(np.correlate(...))
   for iter in range(max_iters):
       if time.time() >= time_limit: break
       # choose two distinct indices
       i, j = rng.choice(n, size=2, replace=False)
       # compute feasible delta range
       lower = max(-h[i], h[j] - 1.0)
       upper = min(1.0 - h[i], h[j])
       if lower >= upper: continue
       # sample delta uniform in feasible range
       delta = rng.uniform(lower, upper)
       # apply move
       h_new = h.copy()
       h_new[i] += delta
       h_new[j] -= delta
       new_c5 = evaluate_c5(h_new)
       # acceptance if improvement
       if new_c5 < cur_c5 - 1e-12:
           h = h_new
           cur_c5 = new_c5
           if new_c5 < best_c5_local:
               best_c5_local = new_c5
               best_h_local = h_new.copy()
               # If new is also globally best, update global
               if new_c5 < best_global_c5 - 1e-12:
                   # store externally? We'll handle after function returns.
                   pass
   return best_h_local, best_c5_local

    n = len(h)
    g = np.ones(n, dtype=np.float64)

def evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def compute_gradient(h, offset):
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if offset >= 0:
        if offset < n:
            g[:n-offset] -= h[offset:]
            g[offset:] -= h[:n-offset]
    else:
        k = -offset
        if k < n:
            g[k:] -= h[:n-k]
            g[:n-k] -= h[k:]
    return g

def generate_start(base_h, rng):
    n = base_h.shape[0]
    delta = rng.normal(size=n)
    delta -= delta.mean()
    pos = delta > 0
    neg = delta < 0
    max_step = np.inf
    if np.any(pos):
        max_step = min(max_step, np.min((1 - base_h[pos]) / delta[pos]))
    if np.any(neg):
        max_step = min(max_step, np.min((-base_h[neg]) / delta[neg]))
    if not np.isfinite(max_step) or max_step <= 0:
        return base_h.copy()
    step = max_step * rng.uniform(0.2, 0.9)
    h_new = base_h + step * delta
    # ensure sum unchanged (should). If small numeric drift, correct.
    # Clip to bounds
    np.clip(h_new, 0, 1, out=h_new)
    return h_new

def descent_local(h0, rng, time_limit, global_best_c5):
    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    no_improve = 0
    max_no_improve = 3000
    while time.time() < time_limit:
        # compute correlation
        conv = np.correlate(h, 1 - h, mode='full')
        cur_c5 = np.max(conv) * dx
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        # gradient descent
        idx_max = np.argmax(conv)
        offset = idx_max - (n - 1)
        grad = compute_gradient(h, offset)
        d = -(grad - np.mean(grad))
        # compute step size
        pos = d > 1e-15
        neg = d < -1e-15
        max_step = np.inf
        if np.any(pos):
            max_step = min(max_step, np.min((1 - h[pos]) / d[pos]))
        if np.any(neg):
            max_step = min(max_step, np.min((-h[neg]) / d[neg]))
        if not np.isfinite(max_step) or max_step <= 0:
            # can't move along gradient
            # fallback to random
            improved = False
        else:
            step = max_step * 0.9
            # backtrack attempts
            improved = False
            trial = step
            for _ in range(5):
                if trial < 1e-12:
                    break
                h_new = h + trial * d
                conv_new = np.correlate(h_new, 1 - h_new, mode='full')
                new_c5 = np.max(conv_new) * dx
                if new_c5 < cur_c5 - 1e-12:
                    h = h_new
                    improved = True
                    break
                trial *= 0.5
        if improved:
            no_improve = 0
            continue
        # random swap move
        i, j = rng.choice(n, size=2, replace=False)
        lower = max(-h[i], h[j] - 1.0)
        upper = min(1.0 - h[i], h[j])
        moved = False
        if lower < upper:
            delta = rng.uniform(lower, upper)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            conv_new = np.correlate(h_new, 1 - h_new, mode='full')
            new_c5 = np.max(conv_new) * dx
            if new_c5 < cur_c5 - 1e-12:
                h = h_new
                moved = True
        if moved:
            no_improve = 0
            continue
        no_improve += 1
        if no_improve > max_no_improve:
            break
    # return best seen
    return best_h, best_c5

import time
import numpy as np

def evaluate_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def compute_gradient(h, offset):
    """
    Gradient of the overlap for the active shift `offset`
    (offset = index_of_max - (n-1)).
    """
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if offset >= 0:
        if offset < n:
            g[: n - offset] -= h[offset:]
            g[offset:] -= h[: n - offset]
    else:
        k = -offset
        if k < n:
            g[k:] -= h[: n - k]
            g[: n - k] -= h[k:]
    return g


def generate_start(base_h, rng):
    """
    Produce a random feasible vector close to `base_h`.
    The sum stays equal to target = len(base_h)/2 and
    values stay in [0,1].
    """
    n = base_h.shape[0]
    delta = rng.normal(size=n)
    delta -= delta.mean()
    pos = delta > 0
    neg = delta < 0
    max_step = np.inf
    if np.any(pos):
        max_step = min(max_step, np.min((1 - base_h[pos]) / delta[pos]))
    if np.any(neg):
        max_step = min(max_step, np.min((-base_h[neg]) / delta[neg]))
    if not np.isfinite(max_step) or max_step <= 0:
        return base_h.copy()
    step = max_step * rng.uniform(0.2, 0.9)
    h_new = base_h + step * delta
    np.clip(h_new, 0, 1, out=h_new)
    return h_new
