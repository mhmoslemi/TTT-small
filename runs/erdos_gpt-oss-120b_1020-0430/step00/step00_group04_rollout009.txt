def hill_climb(h, max_iter, temp_schedule=None):
    n = len(h)
    cur = h.copy()
    cur_score, cur_raw = evaluate(cur)
    best_local = cur.copy()
    best_score_local = cur_score
    best_raw = cur_raw

    for it in range(max_iter):
        # If time exceed global budget break
        i = np.random.randint(n)
        # choose j of opposite value
        if cur[i] == 1:
            # find zero index
            zero_indices = np.where(cur == 0)[0]
            if len(zero_indices) == 0: break
            j = zero_indices[np.random.randint(len(zero_indices))]
        else:
            one_indices = np.where(cur == 1)[0]
            if len(one_indices) == 0: break
            j = one_indices[np.random.randint(len(one_indices))]
        # swap
        new = cur.copy()
        new[i], new[j] = new[j], new[i]
        new_score, new_raw = evaluate(new)
        # simple hill climbing: accept if lower
        if new_score < cur_score - 1e-12:
            cur = new
            cur_score = new_score
            cur_raw = new_raw
            if new_score < best_score_local - 1e-12:
                best_local = new
                best_score_local = new_score
                best_raw = new_raw
        else:
            # optionally accept with probability (annealing)
            pass
        # early termination if cur_score already low enough (maybe 0.3808)
    return best_local, best_score_local

def hill_climb_initial(h0, time_limit):
    start = time.time()
    n = h0.shape[0]
    cur = h0.copy()
    cur_score = evaluate_h(cur)
    best = cur.copy()
    best_score = cur_score
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))

    max_iters = 200000 # large
    iter = 0
    T_start = cur_score * 0.2 + 0.01  # start temperature relative
    T_end = 0.0001
    while time.time() - start < time_limit and iter < max_iters:
        iter += 1
        # choose random swap
        i = random.choice(ones)
        j = random.choice(zeros)
        # swap
        cur[i] = 0
        cur[j] = 1
        new_score = evaluate_h(cur)
        delta = new_score - cur_score
        # temperature schedule linear
        T = T_start * (1 - iter / max_iters) + T_end
        if delta < -1e-12 or random.random() < np.exp(-delta / (T + 1e-12)):
            # accept
            cur_score = new_score
            # update lists
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # reject revert
            cur[i] = 1
            cur[j] = 0
        # if no improvement for many steps, maybe restart: but not needed.

    return best, best_score

import time
import random
import numpy as np

# ---------- evaluation ----------
def _c5_value(h):
    """Return C5 = max_{k} ∫ h(x)(1-h(x+k))dx for the discrete vector h."""
    n = h.shape[0]
    # FFT based cross‑correlation (full length 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    max_corr = conv.max()
    return max_corr * (2.0 / n)


# ---------- helper constructions ----------
def _thue_morse(n):
    """Binary Thue–Morse word of length n (n must be a power of two)."""
    # parity of popcount
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                       dtype=float)


# ---------- binary hill climbing (swap) ----------
def _binary_hill_climb(h0, time_limit):
    """Simple simulated‑annealing on a binary 0/1 vector respecting sum = n/2."""
    start = time.time()
    n = h0.shape[0]
    cur = h0.copy()
    cur_score = _c5_value(cur)
    best = cur.copy()
    best_score = cur_score

    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))

    max_iters = 200_000
    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1
        i = random.choice(ones)
        j = random.choice(zeros)

        # swap
        cur[i] = 0
        cur[j] = 1
        new_score = _c5_value(cur)

        # simple greedy acceptance (occasionally accept uphill)
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] = 1
            cur[j] = 0

        # occasional restart from the best seen so far
        if it % 10000 == 0 and cur_score > best_score + 1e-6:
            cur = best.copy()
            cur_score = best_score
            ones = list(np.flatnonzero(cur == 1))
            zeros = list(np.flatnonzero(cur == 0))

    return best, best_score


# ---------- continuous hill climbing (mass transfer) ----------
def _continuous_hill_climb(h0, time_limit):
    """Mass‑transfer SA on a real vector in [0,1] with fixed total sum."""
    start = time.time()
    n = h0.shape[0]
    cur = h0.copy()
    cur_score = _c5_value(cur)
    best = cur.copy()
    best_score = cur_score

    idx = list(range(n))
    max_iters = 800_000
    it = 0
    while time.time() - start < time_limit and it < max_iters:
        it += 1
        i, j = random.sample(idx, 2)
        ai, aj = cur[i], cur[j]

        # feasible delta range so that both stay in [0,1]
        delta_max = min(ai, 1.0 - aj)
        delta_min = -min(aj, 1.0 - ai)
        if delta_max <= delta_min:
            continue
        # a modest step inside the feasible interval
        delta = random.uniform(delta_min, delta_max) * 0.5

        cur[i] = ai - delta
        cur[j] = aj + delta
        new_score = _c5_value(cur)

        # temperature linear schedule
        T = max(1e-4, cur_score * 0.1 * (1.0 - it / max_iters))
        if new_score < cur_score - 1e-12 or random.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] = ai
            cur[j] = aj

    return best, best_score


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    overall_start = time.time()

    # ----------------- decide problem size and start vector -----------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        # project to a feasible binary vector (half ones)
        order = np.argsort(-init)               # descending
        h_start = np.zeros(n, dtype=float)
        h_start[order[: n // 2]] = 1.0
    else:
        # choose a modest even size when no parent is supplied
        n = 100
        h_start = np.zeros(n, dtype=float)
        h_start[: n // 2] = 1.0
        np.random.shuffle(h_start)

    # Ensure exact sum = n/2 (numerical safety)
    target = n / 2.0
    diff = h_start.sum() - target
    if abs(diff) > 1e-12:
        # simple projection onto the simplex with bounds [0,1]
        # (here diff is tiny because we forced half ones)
        h_start -= diff / n
        np.clip(h_start, 0.0, 1.0, out=h_start)

    # ----------------- allocate time -----------------
    time_used = time.time() - overall_start
    remaining = max(0.0, budget_s - time_used)

    # 60 % of the remaining budget for binary optimisation,
    # the rest for continuous refinement.
    binary_budget = remaining * 0.6
    cont_budget = remaining - binary_budget

    # ----------------- binary optimisation (multiple restarts) -----------------
    best_h = h_start.copy()
    best_score = _c5_value(best_h)
    binary_start = time.time()
    while time.time() - binary_start < binary_budget:
        # give each restart at most 5 s (or the remaining slice)
        slice_time = min(5.0,
                        binary_budget - (time.time() - binary_start))
        if slice_time < 0.5:
            break

        # try a varied initial point: use Thue-Morse when applicable
        if (n & (n - 1)) == 0 and random.random() < 0.3:
            h0 = _thue_morse(n)
        else:
            # random balanced binary vector
            h0 = np.zeros(n, dtype=float)
            h0[: n // 2] = 1.0
            np.random.shuffle(h0)

        h_opt, s_opt = _binary_hill_climb(h0, slice_time)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt.copy()

    # ----------------- continuous refinement -----------------
    if cont_budget > 0.5:
        h_opt, s_opt = _continuous_hill_climb(best_h, cont_budget)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt.copy()

    # final bound (re‑evaluate to be safe)
    final_c5 = float(_c5_value(best_h))

    return (best_h.tolist(), final_c5, int(n))
