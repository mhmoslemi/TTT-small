def project_to_simplex_box(v, S):
    # Solves min||x - v||^2 subject to 0 <= x <= 1, sum x = S.
    # Using water-filling: find λ such that sum_i clip(v_i - λ, 0, 1) = S.
    lo = -1.0  # lower bound lambda less than min(v)-? Actually if λ very negative, v_i - λ >>1 => clip 1.
    hi = 1.0   # if λ very positive, v_i - λ <=0 => clip 0.
    # Ensure sum at lo >= S and sum at hi <= S.
    # For safety, compute sum at lo and hi and expand bounds if needed.
    # Use binary search for accuracy ~1e-9.
    for _ in range(40):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

        g[:n - k] -= h[k:]
        g[k:]     -= h[:n - k]
    else:
        kp = -k
        g[kp:]    -= h[:n - kp]
        g[:n - kp] -= h[kp:]
    return g

import time, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_box_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto {0≤x_i≤1, sum x_i = target_sum}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    v = np.clip(v, 0.0, 1.0)                     # enforce box first
    # find bounds for lambda
    lo, hi = -1.0, 1.0
    # expand bounds if necessary
    while np.clip(v - lo, 0.0, 1.0).sum() < target_sum:
        lo -= 1.0
    while np.clip(v - hi, 0.0, 1.0).sum() > target_sum:
        hi += 1.0
    for _ in range(30):                         # enough for double precision
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] (discretised) that
    minimises  C5 = max_k ∫ h(x)*(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # initialise h
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)  # may be provided
        n = h0.size
        h = h0.copy()
    except Exception:
        n = 200                     # fall‑back size (still well below 1k)
        h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0           # because ∑h_i*dx = 1  ⇒  ∑h_i = n/2
    h = _project_box_simplex(h, target_sum)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    # simulated‑annealing parameters
    max_iter = 2_000_000
    init_temp = 0.12
    final_temp = 1e-4
    temp = init_temp
    decay = (final_temp / init_temp) ** (1.0 / max_iter)

    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.96:
            break

        # propose a transfer of mass from j → i
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # allowable amount respecting bounds
        max_amount = min(cur_h[j], 1.0 - cur_h[i])
        if max_amount <= 0.0:
            continue
        delta = max_amount * np.random.rand()
        if delta == 0.0:
            continue
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_c5 = _compute_c5(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or np.random.rand() < math.exp(-(new_c5 - cur_c5) / temp):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

        temp *= decay

        # occasional diversification – jitter around the best solution
        if (it + 1) % 50000 == 0:
            jitter = np.random.normal(scale=0.018, size=n)
            cur_h = np.clip(best_h + jitter, 0.0, 1.0)
            cur_h = _project_box_simplex(cur_h, target_sum)
            cur_c5 = _compute_c5(cur_h)

    # final projection to guarantee feasibility
    best_h = _project_box_simplex(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(best_c5), n
