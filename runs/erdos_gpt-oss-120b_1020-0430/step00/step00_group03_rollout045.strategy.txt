import numpy as np

def repair(h, desired_sum=35):
    """
    Ensures ∑h[i] == desired_sum (default 35) by adjusting last element.
    """
    s = np.sum(h[:-1])  # Sum of first 69 elements
    adjustment = desired_sum - s
    adjustment = max(0, min(1.0, adjustment))  # Clip in [0,1]
    return np.concatenate([h[:-1], [adjustment]])

            idx = np.random.randint(0, len(self.members))
        return idx

def evolve_evolutionary(run_info):
    """
    Parameters: run_info - dict containing global variables including 'initial_h_values'
    """
    dx = 2.0 / 70.0
    
    def objective(h):
        """
        Main objective with constraint penalty.
        """
        total = np.sum(h)
        desired_sum = 70/2  # 35.0
        if abs(total - desired_sum) > 1e-2:  # Allow 0.01 tolerance before penalty
            # Calculate penalty = (distance from target) * (big number)
            return 1e9 * abs(total - desired_sum)/desired_sum
        
        # If within tolerance, just use correlation
        correlation = np.correlate(h, 1-h, mode='full')
        return np.max(correlation) * dx
    
    def mutation(parent, cycle_count):
        """
        Cyclical mutation rate: ramps up mutation rate as evolution progresses
        """
        # Calculate mutation probability based on current cycle
        cycle_factor = 0.9  # Peserve some diversity
        if cycle_count < 50:
            mutation_rate = 0.3
        else:
            mutation_rate = 0.7
        
        child = np.copy(parent)
        # Mutate a random number of components
        num_muts = np.random.poisson(mutation_rate)
        if num_muts > 0:
            indices = np.random.choice(70, num_muts, replace=False)
            for i in indices:
                # Apply relative perturbation
                current = child[i]
                scale = 0.5
                perturbation = np.random.uniform(-scale, scale)
                new_val = current * (1 + perturbation)
                new_val = max(0.0, min(1.0, new_val))
                child[i] = new_val
        return child
    
    def crossover(parent1, parent2):
        """
        One-point crossover followed by repair
        """
        child = np.copy(parent1)
        crossover_point = np.random.randint(1, 70)
        # Usually prefer crossover from both parents
        child[:crossover_point] = parent2[:crossover_point]
        child = repair(child)
        return child
    
    # Create initial population based on whether initial_h_values is available
    start_pop_size = 80 if 'initial_h_values' in globals() else 40
    
    # If initial is available, make it part of the population
    initial_values = 'initial_h_values' in globals()
    
    if initial_values:
        pop = Population(size=start_pop_size, initial=run_info['initial_h_values'])
    else:
        pop = Population(size=start_pop_size)
    
    # Score population
    pop.evaluate()
    
    best_candidate = pop.best_member()
    best_fitness = np.min(pop.fitness)
    pop_size = pop.members.shape
    
    # Track lifetime best
    lifetime_best = (best_candidate, best_fitness)
    
    start_time = time.time()
    evaluations = 0
    max_evaluations = 20000  # Initial upper bound based on time budget
    
    if initial_values:
        print("Using provided initial solution")
    else:
        print("Initializing random population")
    
    try:
        while (time.time() - start_time) < run_info['budget_s'] and evaluations < max_evaluations - 30:
            # Sample two parents with antisampling to prevent premature convergence
            parent_idx1 = pop.selection_index(np.argmin(pop.fitness)) if pop_size > 1 else 0
            parent_idx2 = pop.selection_index(parent_idx1)
            
            parent1 = pop.members[parent_idx1]
            parent2 = pop.members[parent_idx2]
            
            # Apply adaptive mutation
            if np.random.rand() < np.min(['initial_h_values' not in globals(), 0.7]) or pop_size == 0:
                child = mutation(parent1, evaluations // 30)
            else:
                child = crossover(parent1, parent2)
            
            # Evaluate ifتأسيس viable
            fitness_child = objective(child)
            evaluations += 1
            
            # For statistical purposes, keep track of best
            if fitness_child < lifetime_best[1]:
                lifetime_best = (child, fitness_child)
            
            # Maintain population capacity but prioritize lifetime best
            if evaluations % 5 == 0:
                # Cross on current best
                current_best = lifetime_best[0]
                child = crossover(parent1, current_best)
                fitness_child = objective(child)
                evaluations += 1
            
            # Independent mutation of current best sometimes
            if np.random.rand() < 0.1 and evaluations % 10 == 0:
                child_temp = mutation(lifetime_best[0], evaluations // 30)
                fitness_child_temp = objective(child_temp)
                if fitness_child_temp < lifetime_best[1]:
                    lifetime_best = (child_temp, fitness_child_temp)
                    evaluations += 1
            
            # Update population
            pop.members[parent_idx2] = lifetime_best[0]  # Overwrite inf is necessary to keep diversity
            pop.evaluate()
            
            # Calculate average fitness improvement
            improvement = np.min(pop.fitness) - np.mean(pop.fitness)
            
            # Termination conditions
            if evaluations >= max_evaluations - 10 or improvement < 5e-4:
                break
        
        # Final repair
        if not initial_values and 'initial_h_values' not in globals():  # Only repair if not started with initial
            lifetime_best = repair(lifetime_best[0])
            
        if initial_values:  # Maintain the initial structure if available
            current_fitness = objective(lifetime_best[0])
            lifetime_best = (lifetime_best[0], lifetime_best[0])
        else:
            # Excuse me the lifetime best possibly being inconsistent
            if best_candidate is None:
                best_candidate = np.random.rand(70)
                repair(best_candidate)
            
        print("Terminated evaluations:", evaluations)
        
    except KeyboardInterrupt:  # Allow safe stopping
        pass
    
    # Ensure we use repair even if initial_h_values was provided
    lifetime_best = repair(lifetime_best[0])
    
    return (best_candidate.tolist(), lifetime_best[1].item(), 70)
