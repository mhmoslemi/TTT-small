import time
import numpy as np

def compute_max_corr(h: np.ndarray) -> int:
    """Maximum value of the correlation h * (1-h) over all integer shifts."""
    # h is integer 0/1 array
    return int(np.max(np.correlate(h, 1 - h, mode="full")))

def compute_c5_bound(h: np.ndarray) -> float:
    """Return the C5 bound for a given binary step function."""
    n = h.shape[0]
    max_corr = compute_max_corr(h)
    return max_corr * (2.0 / n)

def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random 0/1 vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1
    return h

def upsample_vector(h: np.ndarray, factor: int) -> np.ndarray:
    """Repeatedly copy each entry `factor` times (integer factor)."""
    if factor == 1:
        return h.copy()
    return np.repeat(h, factor)

def improve_sequence(h: np.ndarray, time_budget: float,
                    rng: np.random.Generator) -> (np.ndarray, int):
    """
    Greedy local search that tries to lower the maximum correlation.
    Operates on a binary (0/1) vector `h` of even length.
    Returns the best vector found and its integer max correlation.
    """
    n = h.shape[0]
    target_int = int(np.ceil(0.38085 * (n / 2) - 1e-12))
    start = time.time()
    best_h = h.copy()
    best_corr = compute_max_corr(best_h)

    while time.time() - start < time_budget:
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = int(corr.max())
        if max_corr < best_corr:
            best_corr = max_corr
            best_h = h.copy()
        if max_corr <= target_int:
            # already good enough – stop early
            best_h = h.copy()
            best_corr = max_corr
            break

        # shifts that attain the current maximum
        max_idxs = np.where(corr == max_corr)[0]
        shift_idx = rng.choice(max_idxs)
        shift = shift_idx - (n - 1)          # integer shift value in [-n+1, n-1]

        # find a mismatched (1,0) pair for this shift
        if shift >= 0:
            left = h[: n - shift]
            right = h[shift:]
            mism_i = np.where((left == 1) & (right == 0))[0]          # indices in `left`
            # candidate zeros that would NOT create a new (1,0) for the same shift
            partner = np.zeros(n, dtype=np.int8)
            partner[: n - shift] = h[shift:]        # partner[j] = h[j+shift] when defined
            cand_j = np.where((h == 0) & (partner == 1))[0]
        else:
            s = -shift
            left = h[s:]
            right = h[: n - s]
            mism_i = np.where((left == 1) & (right == 0))[0] + s       # actual indices in `h`
            partner = np.zeros(n, dtype=np.int8)
            partner[s:] = h[: n - s]                # partner[j] = h[j-shift] when defined
            cand_j = np.where((h == 0) & (partner == 1))[0]

        if mism_i.size == 0:
            # no direct mismatches for this shift – perform a random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = rng.choice(mism_i)
            if cand_j.size > 0:
                j = rng.choice(cand_j)
            else:
                zeros = np.where(h == 0)[0]
                j = rng.choice(zeros)

        # perform the swap
        h[i], h[j] = 0, 1
        new_corr = compute_max_corr(h)
        if new_corr <= max_corr:
            # keep the change (it never makes the max larger)
            if new_corr < best_corr:
                best_corr = new_corr
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1, 0

    return best_h, best_corr
