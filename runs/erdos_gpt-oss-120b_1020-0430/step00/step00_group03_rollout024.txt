def improve_once(h):

    max_corr = corr.max()
    # find worst shifts
    max_lags = np.where(corr == max_corr)[0]
    # pick one (maybe random)
    idx = max_lags[np.random.randint(len(max_lags))]
    lag = idx - (N - 1)
    # compute gradient
    # build arrays h_shift_left = h[lag:] for indices i s.t. i+lag<N
    # can compute with vectorized with np.roll? but handle boundaries:

    grad = np.ones(N)  # initial 1 from derivative of sum_i h_i

        # i-lag in range for i=lag..N-1
        grad[lag:] -= h[:N-lag]  # subtract h_{i-lag}

        l = -lag
        # i+lag = i - l
        # For i >= l: i+lag in range
        grad[l:] -= h[:N-l]
        # i-lag = i + l
        # For i <= N-l-1: i-lag in range
        grad[:N-l] -= h[l:]
    # Now grad holds derivative constant (without dx scaling). We may ignore dx factor.
    # Find i with largest positive gradient, j with most negative gradient
    pos_idx = np.argmax(grad)  # max gradient maybe >0
    neg_idx = np.argmin(grad)  # min gradient maybe <0
    # If grad[pos] <= 0 or grad[neg] >= 0, no improvement possible
    if grad[pos_idx] <= 0 or grad[neg_idx] >= 0:
        return False  # no improvement possible
    # determine epsilon
    max_eps = min(h[pos_idx], 1 - h[neg_idx])
    # also limited by grad values maybe
    eps = min(max_eps, 0.01) # small step
    if eps <= 0:
        return False
    # update
    h[pos_idx] -= eps
    h[neg_idx] += eps
    return True

def evaluate_h(h):
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * (2.0 / N)

def compute_gradient(h, lag):
    N = h.shape[0]
    grad = np.ones(N)
    if lag >= 0:
        L = lag
        if L < N:
            grad[:N-L] -= h[L:]
            grad[L:] -= h[:N-L]
    else:
        L = -lag
        if L < N:
            grad[L:] -= h[:N-L]
            grad[:N-L] -= h[L:]
    return grad

def project_onto_bounds_and_sum(x, target_sum):

    x = np.clip(x, 0.0, 1.0)
    # Iteratively adjust
    for _ in range(10):

        if abs(diff) < 1e-12:

        # Determine active indices that can be moved in needed direction
        if diff > 0:

import time
import random
import numpy as np

def evaluate_h(h):
    """Return C5 for a given step vector h."""
    N = h.shape[0]
    # correlation of h with (1-h) using zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / N)


def compute_gradient(h, lag):
    """Gradient of  C(k)=∑ h_i (1‑h_{i+lag})  w.r.t. h, for a fixed integer lag."""
    N = h.shape[0]
    grad = np.ones(N)
    if lag >= 0:
        L = lag
        if L < N:
            grad[: N - L] -= h[L:]          # term from h_i * h_{i+lag}
            grad[L:] -= h[: N - L]          # term from h_{i-lag} * h_i
    else:
        L = -lag
        if L < N:
            grad[L:] -= h[: N - L]
            grad[: N - L] -= h[L:]
    return grad


def project_onto_bounds_and_sum(x, target):
    """
    Project vector x onto the box [0,1]^n while keeping the sum exactly `target`.
    Simple water‑filling style iterative projection.
    """
    x = np.clip(x, 0.0, 1.0)
    for _ in range(20):
        cur = x.sum()
        diff = cur - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to reduce sum – can reduce any element > 0
            active = x > 0.0
        else:
            # need to increase sum – can increase any element < 1
            active = x < 1.0
        cnt = active.sum()
        if cnt == 0:
            break
        adj = diff / cnt
        x[active] -= adj
        x = np.clip(x, 0.0, 1.0)
