import time, math, numpy as np
from scipy.optimize import minimize

def _project_onto_box_and_sum(x, target, iters=60):
    """
    Project vector x onto the intersection of the box [0,1]^n and the
    hyperplane sum(x)=target.  Uses a simple binary search on the Lagrange
    multiplier.
    """
    lo, hi = -1e5, 1e5
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h):
    """compute C5 for a step vector h (dx = 2/len(h))."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    x = rng.random(n)
    target = n * 0.5
    return _project_onto_box_and_sum(x, target)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] (represented by a vector) that yields a
    small C5 value.  Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise a feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals() and len(initial_h_values) > 0:
        h_curr = np.asarray(initial_h_values, dtype=float)
        n = h_curr.size
        target_sum = n * 0.5
        # ensure feasibility (tiny numerical drift may exist)
        h_curr = _project_onto_box_and_sum(h_curr, target_sum)
    else:
        n = 45                      # default resolution
        target_sum = n * 0.5
        h_curr = _random_feasible(n, rng)

    best_h = h_curr.copy()
    best_c = _c5_of(best_h)

    # ------------------------------------------------------------------
    # 2. Helper for constrained local optimisation (SLSQP)
    # ------------------------------------------------------------------
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n

    def _objective(x):
        # direct max‑overlap objective (SLSQP uses finite‑difference gradients)
        return np.max(np.correlate(x, 1.0 - x, mode="full")) * (2.0 / n)

    # ------------------------------------------------------------------
    # 3. Local optimisation from the current best point
    # ------------------------------------------------------------------
    try:
        res = minimize(_objective, best_h, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": 250, "ftol": 1e-10, "disp": False})
        if res.success:
            cand = _project_onto_box_and_sum(np.clip(res.x, 0.0, 1.0), target_sum)
            cand_c = _c5_of(cand)
            if cand_c < best_c:
                best_h, best_c = cand, cand_c
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing / stochastic hill‑climbing phase
    # ------------------------------------------------------------------
    # Parameters (tuned for speed vs quality)
    max_iter = 200_000
    sigma0 = 0.5          # initial noise scale
    temp0 = 0.2           # initial temperature
    cur_h = best_h.copy()
    cur_c = best_c

    it = 0
    while True:

        if time.time() - start_time > budget_s * 0.94:
            break
        if it >= max_iter:
            break

        frac = it / max_iter
        sigma = sigma0 * (1.0 - frac) + 1e-12
        T = temp0 * (1.0 - frac) + 1e-12

        # propose a perturbed point and project back to feasibility
        prop = cur_h + sigma * rng.standard_normal(n)
        prop = _project_onto_box_and_sum(prop, target_sum)

        prop_c = _c5_of(prop)

        # Metropolis acceptance

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

        if res.success:

            cand_c = _c5_of(cand)

    except Exception:

    # ------------------------------------------------------------------

    best_h = _project_onto_box_and_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    final_c = _c5_of(best_h)

    return (best_h.tolist(), float(final_c), int(n))
