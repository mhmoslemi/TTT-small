def compute_f_and_grad(h, alpha=50):
    N = len(h)
    dx = 2.0/N
    # compute f array for each shift s
    # using np.correlate
    f = np.correlate(h, 1 - h, mode='full') * dx # this yields length 2N-1
    # compute weights
    max_f = np.max(f) # for numerical stability
    exp_alpha = np.exp(alpha * (f - max_f))  # subtract max for stability
    sum_exp = np.sum(exp_alpha)
    w = exp_alpha / sum_exp
    # gradient
    grad = np.zeros_like(h)
    # compute weighted sum of -(h_{j-s}+h_{j+s}) times dx * w_s
    # for s from -(N-1) to (N-1). index offset = s + N -1
    for offset, ws in enumerate(w):

        # contributions to grad:
        # for each j, term: -dx * ws * (h_{j-s} + h_{j+s})
        # Equivalent to adding -dx*ws*(..., when indices in range)

def compute_f_and_grad(h, alpha=30):
    N = len(h)
    dx = 2.0 / N
    f = np.correlate(h, 1 - h, mode='full') * dx
    # smoothing weights
    max_f = np.max(f)
    exp_alpha = np.exp(alpha*(f - max_f))
    sum_exp = np.sum(exp_alpha)
    w = exp_alpha / sum_exp
    # conv w & h
    conv1 = np.convolve(w, h, mode='full')
    # conv_rev
    w_rev = w[::-1]
    conv2 = np.convolve(w_rev, h, mode='full')
    # slice
    start = N-1
    conv1_slice = conv1[start:start+N]
    conv2_slice = conv2[start:start+N]
    grad = -dx * (conv1_slice + conv2_slice)
    return f, grad

def grad_smoothed_max(h, alpha=10):
    N = len(h)
    dx = 2.0/N
    f = np.correlate(h, 1-h, mode='full') * dx
    max_f = np.max(f)
    # Use stable softmax
    exp_a = np.exp(alpha * (f - max_f))
    sum_exp = np.sum(exp_a)

import time, math
import numpy as np

# ---------- helper utilities ----------
def project_onto_bounds_and_sum(v, target):
    """Projection onto [0,1] box with a fixed sum using a simple bisection."""
    lo, hi = -2.0, 2.0
    # enlarge bounds if needed
    while np.clip(v + lo, 0, 1).sum() > target:
        lo -= 1.0
    while np.clip(v + hi, 0, 1).sum() < target:
        hi += 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(v + mid, 0, 1)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0, 1)


def compute_C(h):
    """Exact C5 value for a discrete step vector."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def grad_smoothed_max(h, alpha):
    """Gradient of the soft‑max of the overlap integral."""
    n = len(h)
    dx = 2.0 / n
    f = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(f)
    w = np.exp(alpha * (f - m))
    w /= w.sum()
    conv1 = np.convolve(w, h, mode="full")
    conv2 = np.convolve(w[::-1], h, mode="full")
    start = n - 1
    g = -dx * (conv1[start:start + n] + conv2[start:start + n])
    return g, f


def gradient_optimize(h0, max_iter=200, alpha_start=5.0, alpha_end=30.0, lr=1.0):
    """Smoothed‑max gradient descent with a linear schedule for alpha."""
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    for it in range(max_iter):
        a = alpha_start + (alpha_end - alpha_start) * it / max(1, max_iter - 1)
        g, _ = grad_smoothed_max(h, a)
        step = lr / (1.0 + 0.005 * it)
        h = h - step * g
        h = project_onto_bounds_and_sum(h, target)
    return h, compute_C(h)


def local_search(h, C0, max_iter=500, delta=0.1):
    """Simple two‑coordinate hill‑climb that preserves the sum."""
    n = len(h)
    best_h, best_C = h.copy(), C0
    for _ in range(max_iter):
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        inc = min(1.0 - best_h[i], best_h[j], delta)
        if inc <= 0:
            continue
        d = inc * np.random.rand()
        cand = best_h.copy()
        cand[i] += d
        cand[j] -= d
        Cc = compute_C(cand)
        if Cc < best_C:
            best_h, best_C = cand, Cc
    return best_h, best_C


def resample_h(old_h, new_n):
    """Linear interpolation of a step vector onto a new grid."""
    old_n = len(old_h)
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    new_h = np.interp(x_new, x_old, old_h)
    return project_onto_bounds_and_sum(new_h, new_n / 2.0)
