import numpy as np, time, math

def _c5_value(h):
    # compute overlap constant for a given step function h
    # dx = 2 / len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / h.size)

def _project_sum_constraint(h, target):
    # ensure sum equals target while staying in [0,1] bounds
    # simple greedy adjustment
    cur = h.sum()
    diff = cur - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to reduce total
        idxs = np.where(h > 0)[0]
        for i in idxs:
            dec = min(diff, h[i])
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    else:
        need = -diff
        idxs = np.where(h < 1)[0]
        for i in idxs:
            inc = min(need, 1.0 - h[i])
            h[i] += inc
            need -= inc
            if need <= 1e-12:
                break
    return h

def run(seed=42, budget_s=1100, **_):
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        N = h.size
    else:
        N = 70
        h = np.zeros(N, dtype=np.float64)
        ones = rng.choice(N, N // 2, replace=False)
        h[ones] = 1.0

    # feasibility repair
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    h = _project_sum_constraint(h, target)

    cur_val = _c5_value(h)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()

        T = T0 * (1.0 - it / max_iters)

import numpy as np, time, math

def _c5_value(h: np.ndarray) -> float:
    """Return the overlap constant for the step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def _project_sum_constraint(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h so that sum(h)=target while keeping 0≤h≤1."""
    cur = h.sum()
    diff = cur - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:                      # need to decrease the sum
        idx = np.where(h > 0)[0]
        for i in idx:
            dec = min(diff, h[i])
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    else:                             # need to increase the sum
        need = -diff
        idx = np.where(h < 1)[0]
        for i in idx:
            inc = min(need, 1.0 - h[i])
            h[i] += inc
            need -= inc
            if need <= 1e-12:
                break
    return h

def run(seed: int = 42, budget_s: int = 1100, **_):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) initialise a feasible h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        N = h.size
    else:
        N = 70
        h = np.zeros(N, dtype=np.float64)
        ones = rng.choice(N, N // 2, replace=False)
        h[ones] = 1.0

    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    h = _project_sum_constraint(h, target)

    # ------------------------------------------------------------------
    # 2) simulated annealing on pairwise mass transfers
    # ------------------------------------------------------------------
    cur_val = _c5_value(h)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    max_iters = 500_000                 # safe upper bound; actual stop by time
    T0 = 0.02                           # initial temperature
    max_step = 0.10                     # maximal transfer at T0

    for it in range(max_iters):
        if time.time() - start > budget_s - 3.0:
            break

        # linear cooling, never below a tiny epsilon
        T = max(T0 * (1.0 - it / max_iters), 1e-9)

        # pick two distinct indices
        i, j = rng.integers(N, size=2)
        if i == j:
            continue

        # choose a direction for the transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        delta_max = min(h[src], 1.0 - h[dst])
        if delta_max <= 0.0:
            continue

        # step size shrinks with temperature
        step_limit = max_step * (T / T0)
        if step_limit < 1e-9:
            step_limit = 1e-9
        delta = rng.random() * min(delta_max, step_limit)
        if delta <= 0.0:
            continue

        # apply tentative move
        h[src] -= delta
        h[dst] += delta

        new_val = _c5_value(h)
        dE = new_val - cur_val
        accept = (dE <= 0.0) or (rng.random() < math.exp(-dE / T))

        if accept:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[src] += delta
            h[dst] -= delta

    # ------------------------------------------------------------------
    # 3) deterministic refinement targeting the worst shift
    # ------------------------------------------------------------------
    for _ in range(5_000):
        if time.time() - start > budget_s - 1.0:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (N - 1)

        if shift >= 0:
            i0, i1 = 0, N - shift
            j0, j1 = shift, N
        else:
            i0, i1 = -shift, N
            j0, j1 = 0, N + shift

        diffs = best_h[i0:i1] - best_h[j0:j1]
        max_diff = np.max(diffs)
        if max_diff <= 1e-9:
            break
        pos = np.argmax(diffs)
        i = i0 + pos
        j = j0 + pos
        delta_max = min(best_h[i], 1.0 - best_h[j])
        if delta_max <= 0.0:
            break
        delta = min(delta_max, max_diff * 0.5, 0.02)
        if delta <= 0.0:
            break

        # trial transfer
        best_h[i] -= delta
        best_h[j] += delta
        new_val = _c5_value(best_h)
        if new_val < best_val - 1e-12:
            best_val = new_val
        else:
            # revert
            best_h[i] += delta
            best_h[j] -= delta

    # ------------------------------------------------------------------
    # 4) final projection to meet constraints exactly
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum_constraint(best_h, target)
    best_h = np.clip(best_h, 0.0, 1.0)

    final_val = _c5_value(best_h)
    return best_h, float(final_val), N
