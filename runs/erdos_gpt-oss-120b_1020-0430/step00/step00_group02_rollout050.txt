         if s > target:
             lo = mid
         else:
             hi = mid
    return np.clip(v - (lo+hi)/2, 0,1)

import time
import numpy as np

# ----- helper functions -----------------------------------------------------

def project_to_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x<=1, sum(x)=target} by binary searching the Lagrange multiplier."""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def upscale_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function to a higher resolution."""
    old_n = len(old_h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    return project_to_capped_simplex(new_h, new_n / 2.0)


def compute_bound(h: np.ndarray) -> float:
    """Exact C5 bound for a discretised step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full')
    return 1.0 - corr.min() * dx


def gradient_step(h: np.ndarray, gamma: float) -> np.ndarray:
    """Gradient of the smoothed max-overlap loss."""
    n = len(h)
    # autocorrelation
    corr = np.correlate(h, h, mode='full')
    # soft‑min weights
    tmp = -gamma * corr
    m = np.max(tmp)
    w = np.exp(tmp - m)
    p = w / w.sum()                # size 2*n-1, sum to 1

    grad = np.zeros_like(h)
    off = n - 1                     # index of zero shift in p
    for idx, coeff in enumerate(p):
        if coeff == 0.0:
            continue
        s = idx - off               # shift value
        if s >= 0:
            if s < n:
                grad[:n - s] += coeff * h[s:]
            if s > 0:
                grad[s:] += coeff * h[:n - s]
        else:
            sn = -s
            if sn < n:
                grad[sn:] += coeff * h[:n - sn]
            if sn > 0:
                grad[:n - sn] += coeff * h[sn:]
    return -grad                    # derivative of loss = -soft‑min


def optimize_h(h0: np.ndarray, time_limit: float, start_time: float) -> tuple:
    """
    Projected gradient descent on the smoothed max–overlap objective.
    Returns (best_h, best_bound) found before time runs out.
    """
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_b = compute_bound(h)

    # schedule parameters
    total_iters = 10 ** 6
    for it in range(total_iters):
        # stop when the global budget is almost exhausted
        if time.time() - start_time > time_limit * 0.99:
            break

        # linearly anneal learning rate and smoothing parameter
        t = it / total_iters
        lr = 0.5 * (1.0 - t) + 0.02 * t       # 0.5 → 0.02
        gamma = 5.0 + 35.0 * t                # 5  → 40

        # gradient and step
        g = gradient_step(h, gamma)
        h_new = h - lr * g
        h = project_to_capped_simplex(h_new, target)

        # occasional random jitter to escape flat regions
        if (it & 127) == 0:
            jitter = np.random.normal(scale=0.001, size=n)
            h = project_to_capped_simplex(h + jitter, target)

        # keep best exact bound
        if (it % 20) == 0:
            b = compute_bound(h)
            if b < best_b:
                best_b = b
                best_h = h.copy()

    # final check
    b = compute_bound(h)
    if b < best_b:
        best_b, best_h = b, h.copy()
    return best_h, best_b
