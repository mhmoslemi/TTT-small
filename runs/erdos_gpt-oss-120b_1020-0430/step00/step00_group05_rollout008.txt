import time, math
import numpy as np

def _max_corr(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx, c

def _binary_from_initial(init):
    n = len(init)
    half = n // 2
    order = np.argsort(init)[::-1]          # descending
    b = np.zeros(n, dtype=float)
    b[order[:half]] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ------------------------------------------------------------------ #
    # Choose number of points (must be even). Prefer the length of the
    # parent construction if it exists, otherwise use a modest size.
    # ------------------------------------------------------------------ #
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        if n % 2:
            n += 1
        # start from a binary version of the parent (if it has the right size)
        try:
            h = _binary_from_initial(np.asarray(initial_h_values, float))
        except Exception:
            h = np.zeros(n, float)
            h[rng.choice(n, n // 2, replace=False)] = 1.0
    else:
        n = 200                     # a convenient default (<1000)
        h = np.zeros(n, float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    cur_val, cur_corr = _max_corr(h)
    best_h = h.copy()
    best_val = cur_val

    # ------------------------------------------------------------------ #
    # Simulated‑annealing / hill‑climbing parameters
    # ------------------------------------------------------------------ #
    total_time = budget_s * 0.95
    T0, Tf = 0.02, 5e-4          # temperature schedule
    it = 0
    # main optimisation loop
    while time.time() - start < total_time:
        it += 1
        # current worst shift
        max_idx = int(np.argmax(cur_corr))
        k_shift = max_idx - (n - 1)          # signed shift giving max overlap
        # ---------------------------------------------------------------- #
        # Propose a swap.
        # With some probability we try a targeted swap that directly
        # reduces the contribution at the worst shift.
        # ---------------------------------------------------------------- #
        if rng.random() < 0.4 and k_shift != 0:
            k = k_shift
            if k > 0:
                mask = (h[:n - k] == 1.0) & (h[k:] == 0.0)
                idx = np.nonzero(mask)[0]
                if idx.size:
                    i = idx[0]
                    j = i + k
                else:
                    i = rng.choice(np.where(h == 1.0)[0])
                    j = rng.choice(np.where(h == 0.0)[0])
            else:                                   # k < 0
                kp = -k
                mask = (h[kp:] == 1.0) & (h[:n - kp] == 0.0)
                idx = np.nonzero(mask)[0]
                if idx.size:
                    j = idx[0]
                    i = j + kp
                else:
                    i = rng.choice(np.where(h == 1.0)[0])
                    j = rng.choice(np.where(h == 0.0)[0])
        else:
            i = rng.choice(np.where(h == 1.0)[0])
            j = rng.choice(np.where(h == 0.0)[0])

        # perform the swap
        h[i], h[j] = 0.0, 1.0
        new_val, new_corr = _max_corr(h)

        # temperature based acceptance
        elapsed = time.time() - start
        frac = elapsed / total_time
        T = T0 * (1.0 - frac) + Tf * frac
        if new_val <= cur_val or rng.random() < math.exp((cur_val - new_val) / T):
            cur_val, cur_corr = new_val, new_corr
            if new_val < best_val:

        else:

        # occasional full recomputation to avoid drift from rounding
        if it % 5000 == 0:
            cur_val, cur_corr = _max_corr(h)

    # final verification
    best_val, _ = _max_corr(best_h)
    return (best_h.tolist(), float(best_val), n)
