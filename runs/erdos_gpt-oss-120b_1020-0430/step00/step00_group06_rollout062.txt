def project_to_bounds_and_sum(x):
    n = len(x)
    target = n/2
    x = np.clip(x, 0.0, 1.0)
    # water-filling
    while True:
        total = x.sum()
        diff = target - total
        # indices not at bounds
        interior = (x > 0) & (x < 1)
        count = interior.sum()
        if abs(diff) < 1e-12 or count == 0:

def project_simple(h, target_sum):

    h = np.clip(h, 0.0, 1.0)

    while True:

def evaluate_h(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_simple(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    # water-filling to satisfy sum
    # We'll use iterative loop as described
    while True:
        total = h.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        interior = (h > 0) & (h < 1)
        count = interior.sum()

            h = np.clip(h, 0.0, 1.0)
            break
        delta = diff / count
        h[interior] += delta
        # re-clamp
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, dx, lr, target_sum):

    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (N - 1)

        term_plus = np.ones(N)
        term_plus[:N-s] = 1 - h[s:]
        term_minus = np.zeros(N)

    h_new = h - lr * grad
    h_new = project_simple(h_new, target_sum)

        return np.clip(1 - np.abs(xs - 1.0), 0, 1)

import numpy as np
import time

def _evaluate_h(h, dx):
    """C5 value for a given step function h."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def _project(h, target_sum):
    """Project h onto the box [0,1] with fixed sum (=target_sum)."""
    h = np.clip(h, 0.0, 1.0)
    # simple water-filling iteration
    while True:
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        interior = (h > 0.0) & (h < 1.0)
        cnt = interior.sum()
        if cnt == 0:
            # no free variables – spread diff uniformly and clip again
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            break
        delta = diff / cnt
        h[interior] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def _subgradient_step(h, dx, lr, target_sum):
    """One sub‑gradient step for the max‑overlap objective."""
    N = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (N - 1)                     # shift that attains the max

    # compute the sub‑gradient of the active term
    if s >= 0:
        term_plus = np.ones(N)
        term_plus[:N - s] = 1 - h[s:]
        term_minus = np.zeros(N)
        term_minus[s:] = h[:N - s]
    else:
        t = -s
        term_plus = np.ones(N)
        term_plus[t:] = 1 - h[: N - t]
        term_minus = np.zeros(N)
        term_minus[: N - t] = h[t:]
    grad = term_plus - term_minus

    h_new = h - lr * grad
    h_new = _project(h_new, target_sum)
    val = _evaluate_h(h_new, dx)
    return h_new, val

def _make_h_from_a(a, N):
    """One‑parameter family: ramps of length a, plateau of width 2‑2a."""
    dx = 2.0 / N
    if a <= 0.0:                     # flat 0.5
        return np.full(N, 0.5)
    if a >= 1.0:                     # pure triangle
        x = (np.arange(N) + 0.5) * dx
        return np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
    H = 1.0 / (2.0 - a)               # height of plateau
    b = 2.0 - 2.0 * a                # plateau width
    x = (np.arange(N) + 0.5) * dx
    h = np.empty(N, dtype=float)

    left = x < a
    h[left] = H * x[left] / a

    middle = (x >= a) & (x < a + b)
    h[middle] = H

    right = x >= a + b
    h[right] = H * (2.0 - x[right]) / a

    # enforce constraints exactly (tiny discretisation error possible)
    target = N / 2.0
    return _project(h, target)

def _random_feasible(N, rng):
    """Random binary/half‑binary feasible vector."""
    h = np.zeros(N, dtype=float)
    half = N // 2
    ones = rng.choice(N, half, replace=False)
    h[ones] = 1.0
    if N % 2:
        # need an extra 0.5 to hit the exact sum N/2
        free = np.where(h == 0.0)[0]
        h[rng.choice(free)] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- resolution ---------------------------------------------------------
    N = 200                     # number of discretisation points
    dx = 2.0 / N
    target_sum = N / 2.0

    best_h = None
    best_val = np.inf

    # --- quick scan of the one‑parameter family -----------------------------
    a_vals = np.linspace(0.0, 1.0, 401)
    for a in a_vals:
        h = _make_h_from_a(a, N)
        val = _evaluate_h(h, dx)
        if val < best_val:
            best_val, best_h = val, h.copy()
        # early exit if record already beaten
        if best_val <= 0.3808:
            break
        if time.time() - start > budget_s * 0.5:
            break

    # --- sub‑gradient optimisation -----------------------------------------
    # parameters for the descent
    lr0 = 0.7
    max_iters = 5000
    it = 0
    h_cur = best_h.copy()
    lr = lr0

    while time.time() - start < budget_s - 0.5:
        h_cur, cur_val = _subgradient_step(h_cur, dx, lr, target_sum)
        it += 1
        if cur_val + 1e-12 < best_val:
            best_val, best_h = cur_val, h_cur.copy()
            if best_val <= 0.3808:
                break
        # learning‑rate schedule
        if it % 200 == 0:
            lr = max(lr * 0.9, 0.02)
        # occasional random restart to escape plateaus
        if it % 1000 == 0:
            h_cur = _random_feasible(N, rng)
            lr = lr0
        if it >= max_iters:
            # start fresh again but keep time budget in mind
            it = 0
            h_cur = _random_feasible(N, rng)
            lr = lr0

    # final safety projection (ensures exact constraints)
    best_h = _project(best_h, target_sum)
    final_val = _evaluate_h(best_h, dx)
    return best_h, final_val, N
