def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n_points
    try: 
        init_h = np.array(initial_h_values, dtype=np.float64)
        n = init_h.shape[0]
    except NameError:
        n = 80
        init_h = None
    dx = 2.0 / n
    target_sum = n / 2

    def clip_and_norm(h):
        # Ensure within [0,1] and sum target
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s != target_sum:
            # adjust by adding same constant to all non-saturated entries; simple scaling:
            if s > 0:
                h = h * (target_sum / s)
                # after scaling may exceed 1; clip again and iterate a few times
                h = np.clip(h, 0.0, 1.0)
                # small residual error adjust linearly
                diff = target_sum - h.sum()
                if abs(diff) > 1e-12:
                    # distribute diff uniformly over free entries
                    free = (h > 0) & (h < 1)
                    cnt = free.sum()
                    if cnt > 0:
                        h[free] += diff / cnt
                        h = np.clip(h, 0.0, 1.0)
            else:
                # all zero, distribute equally
                h[:] = target_sum / n
        return h

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # generate triangle
    def triangle_h():
        x = (np.arange(n) + 0.5) * dx
        h = np.where(x <= 1, x, 2 - x)   # linear piecewise
        # this yields values in [0,1] and integral approx 1; now normalize
        h = clip_and_norm(h)
        return h

    # candidate best
    best_h = None
    best_val = np.inf

    if init_h is not None:
        # ensure constraints (might already satisfy)
        h0 = clip_and_norm(init_h.copy())
        val0 = evaluate(h0)
        if val0 < best_val:
            best_val = val0
            best_h = h0.copy()
    # triangle candidate
    h_tri = triangle_h()
    val_tri = evaluate(h_tri)
    if val_tri < best_val:
        best_val = val_tri
        best_h = h_tri.copy()

    # start local search from best_h
    cur_h = best_h.copy()
    cur_val = best_val

    # annealing schedule
    max_iter = 1000000  # will break by time
    # compute temperature schedule gradually from initial temperature to near 0
    # set initial temperature based on typical improvement magnitude
    # we can set initial temperature as 0.01 maybe
    init_temp = 0.02
    final_temp = 0.0001
    # compute cooling factor
    # use exponential cooling
    total_time = budget_s - (time.time() - start)
    # We'll use time to control loops.

    # function to get temperature based on elapsed fraction
    def temperature(frac):
        # exponential cooling
        return init_temp * (final_temp / init_temp) ** frac

    # main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        # compute fraction of time used relative to budget
        frac = elapsed / budget_s
        temp = temperature(frac)

        # propose move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # Transfer epsilon from i to j or reverse direction random
        if rng.random() < 0.5:
            # move from i to j
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
        else:
            max_eps = min(cur_h[j], 1.0 - cur_h[i])
            i, j = j, i  # swap direction so always move from i -> j after possible swap?

        if max_eps <= 0:

        eps = rng.random() * max_eps
        # apply move
        new_h = cur_h.copy()
        new_h[i] -= eps

        # evaluate
        new_val = evaluate(new_h)
        # accept if better or with probability
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    # after loop, final projection for safety
    best_h = clip_and_norm(best_h)
    best_val = evaluate(best_h)

    return (best_h, float(best_val), int(n))

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises
        C5 = max_k ∫ h(x)*(1-h(x+k)) dx
    The returned tuple is (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # Helper functions
    # ------------------------------------------------------------------
    def _project(h, target_sum):
        """Project h onto [0,1] and the hyperplane sum(h)=target_sum."""
        h = np.clip(h, 0.0, 1.0)
        for _ in range(10):
            diff = target_sum - h.sum()
            if abs(diff) < 1e-12:
                break
            free = (h > 0.0) & (h < 1.0)
            cnt = free.sum()
            if cnt == 0:
                # all at a bound – scale globally
                h = h * (target_sum / h.sum())
                h = np.clip(h, 0.0, 1.0)
                break
            delta = diff / cnt
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
        return h

    def _evaluate(h):
        """C5 for a candidate h (vector of length n)."""
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def _shift_arrays(arr, lag):
        """Return (arr shifted forward by lag, arr shifted backward by lag)."""
        if lag >= 0:
            # forward shift: arr[i+lag] (zero where out of range)
            fwd = np.empty_like(arr)
            fwd[:n - lag] = arr[lag:]
            fwd[n - lag:] = 0.0
            # backward shift: arr[i-lag]
            bwd = np.empty_like(arr)
            bwd[lag:] = arr[:n - lag]
            bwd[:lag] = 0.0
        else:
            l = -lag
            fwd = np.empty_like(arr)
            fwd[l:] = arr[:n - l]
            fwd[:l] = 0.0
            bwd = np.empty_like(arr)
            bwd[:n - l] = arr[l:]
            bwd[n - l:] = 0.0
        return fwd, bwd

    # Problem dimensions and constraints
    # ------------------------------------------------------------------
    try:
        init_vec = np.asarray(initial_h_values, dtype=float)

    except Exception:
        n = 80
        init_vec = None

    dx = 2.0 / n
    target_sum = n / 2.0  # because ∫h = 1  ⇒  sum(h)*dx = 1

    # ------------------------------------------------------------------
    # Initial candidates
    # ------------------------------------------------------------------
    best_h = None
    best_val = np.inf

    # 1) use the supplied construction if any
    if init_vec is not None:
        h0 = _project(init_vec.copy(), target_sum)
        v0 = _evaluate(h0)
        best_h, best_val = h0, v0

    # 2) symmetric triangular shape (the continuous optimal candidate)
    x = (np.arange(n) + 0.5) * dx  # centre points
    tri = np.where(x <= 1.0, x, 2.0 - x)
    tri = _project(tri, target_sum)
    v_tri = _evaluate(tri)
    if v_tri < best_val:
        best_h, best_val = tri.copy(), v_tri

    # ------------------------------------------------------------------
    # Local optimisation (simulated‑annealing + gradient‑guided moves)
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_val = best_val

    # annealing schedule (exponential)
    init_temp = 2e-2
    final_temp = 1e-4

    def _temp(frac):
        return init_temp * (final_temp / init_temp) ** frac

    # parameters
    max_move = 0.2          # maximal absolute change per transfer
    grad_thresh = 1e-6      # sensitivity for picking gradient signs

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        # temperature based on elapsed fraction
        frac = elapsed / budget_s
        temp = _temp(frac)

        # correlation and the lag giving the current maximum
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx_max = int(np.argmax(corr))
        lag = idx_max - (n - 1)

        # Build gradient of the active lag (sub‑gradient of the max‑correlation)
        fwd, bwd = _shift_arrays(cur_h, lag)
        grad = 1.0 - fwd - bwd    # dF/dh_i for the current active term

        # indices where decreasing h_i would help (grad>0) and increasing helps (grad<0)
        pos_idx = np.where(grad > grad_thresh)[0]
        neg_idx = np.where(grad < -grad_thresh)[0]

        if pos_idx.size and neg_idx.size:
            i = rng.choice(pos_idx)
            j = rng.choice(neg_idx)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps > 0:
                eps = max_eps * rng.random() * 0.5
                new_h = cur_h.copy()
                new_h[i] -= eps
                new_h[j] += eps
                new_val = _evaluate(new_h)
                if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
                    cur_h, cur_val = new_h, new_val
                    if new_val < best_val:
                        best_h, best_val = new_h.copy(), new_val
                continue  # move to next iteration

        # fallback: random mass transfer between two arbitrary indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # random direction of transfer
        if rng.random() < 0.5:
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            src, dst = i, j
        else:
            max_eps = min(cur_h[j], 1.0 - cur_h[i])
            src, dst = j, i
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[src] -= eps
        new_h[dst] += eps
        new_val = _evaluate(new_h)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val

    # final clean‑up: ensure constraints are exactly met
    best_h = _project(best_h, target_sum)
    best_val = float(_evaluate(best_h))

    return (best_h, best_val, int(n))
