def hill_climb_binary(n, max_iter):
    # start with a random balanced binary vector
    idx = np.arange(n)
    np.random.shuffle(idx)
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    best = h.copy()
    best_val = compute_c5(best)
    for it in range(max_iter):
        # pick two positions i (with value 1) and j (value 0) to swap
        ones = np.where(best==1)[0]
        zeros = np.where(best==0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        new_h = best.copy()
        new_h[i] = 0
        new_h[j] = 1
        val = compute_c5(new_h)
        if val < best_val - 1e-8: # improvement
            best = new_h
            best_val = val
        # maybe accept worse with prob? skip for simplicity
    return best, best_val
```

    return np.max(corr) * dx

def _project_capped_simplex(v, target):
    """Project vector v onto the box [0,1] with prescribed sum = target."""
    # water‑filling on a shifted vector: y_i = clip(v_i - lam, 0, 1)
    lo, hi = -1e5, 1e5
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _binary_hill_climb(start_h, max_iters, deadline):
    """Local search on binary vectors (0/1) preserving the exact half‑mass."""
    h = start_h.copy()
    best_h = h.copy()
    best_val = _compute_c5(best_h)
    cur_val = best_val
    n = h.shape[0]
    # pre‑compute index sets for speed
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        val = _compute_c5(new_h)
        if val < cur_val - 1e-12:
            best_h = new_h
            cur_val = val
            if val < best_val:
                best_val = val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_val = np.inf
    best_n = None

    # initialise from parent if it exists
    try:
        init = np.array(initial_h_values, dtype=float)
        val = _compute_c5(init)
        best_h, best_val, best_n = init, val, init.shape[0]
    except Exception:
        pass

    # candidate grid sizes (even for binary constructions)
    cand_ns = [48, 64, 80, 96, 112, 128, 160, 192]

    # Main optimisation loop over grid sizes
    for n in cand_ns:
        if time.time() - start_time > 0.95 * budget_s:
            break
        target_sum = n / 2.0
        # ---- 1) SLSQP local optimisation from several starts ----
        for restart in range(5):
            if time.time() - start_time > 0.95 * budget_s:
                break
            # random start: half binary, half continuous alternately
            if restart % 2 == 0:
                # binary start
                perm = np.random.permutation(n)
                x0 = np.zeros(n, dtype=float)
                x0[perm[:n // 2]] = 1.0
            else:
                # continuous start projected onto feasible set
                rnd = np.random.rand(n)
                x0 = _project_capped_simplex(rnd, target_sum)
            # constraints for SLSQP
            cons = ({'type': 'eq',
                     'fun': lambda x, t=target_sum: np.sum(x) - t},)
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(_compute_c5, x0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-9,
                                        'disp': False})
            except Exception:
                continue
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = _project_capped_simplex(h_opt, target_sum)
            val = _compute_c5(h_opt)
            if val < best_val - 1e-12:
                best_h, best_val, best_n = h_opt, val, n
        # ---- 2) Binary hill‑climb refinement (if size matches) ----
        # use a fresh binary start if we have no good binary of this size yet
        if n == best_n and best_h is not None and best_h.shape[0] == n:
            start_h = best_h.copy()
        else:
            perm = np.random.permutation(n)
            start_h = np.zeros(n, dtype=float)
            start_h[perm[:n // 2]] = 1.0
        # 3k–5k swaps are usually enough for modest n
        hill_iters = 4000
        deadline = start_time + budget_s * 0.95
        h_bin, val_bin = _binary_hill_climb(start_h, hill_iters, deadline)
        if val_bin < best_val - 1e-12:
            best_h, best_val, best_n = h_bin, val_bin, n
        # optional tiny SLSQP polish after binary improvement
        if best_n == n:
            try:
                cons = ({'type': 'eq',
                         'fun': lambda x, t=target_sum: np.sum(x) - t},)
                bounds = [(0.0, 1.0)] * n
                res = minimize(_compute_c5, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-9,
                                        'disp': False})
                h_opt = np.clip(res.x, 0.0, 1.0)
                h_opt = _project_capped_simplex(h_opt, target_sum)
                val = _compute_c5(h_opt)
                if val < best_val - 1e-12:
                    best_h, best_val, best_n = h_opt, val, n
            except Exception:
                pass

    # final sanity projection
    if best_h is not None:
        best_h = _project_capped_simplex(best_h, best_n / 2.0)
        best_val = _compute_c5(best_h)

    return best_h.tolist(), float(best_val), int(best_n)
