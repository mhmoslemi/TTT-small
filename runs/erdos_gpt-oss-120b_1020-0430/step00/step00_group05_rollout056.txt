    delta = (rng.random() - 0.5) * step
    # Ensure resultant values in [0,1]
    # adjust delta so that 0 <= h_i + delta <=1 and 0 <= h_j - delta <=1
    max_delta_i = 1 - h[i]

def local_search(h, max_seconds):
    start_loc = time.time()
    best_h = h.copy()
    best_val = evaluate(best_h, n)
    cur_h = h.copy()
    cur_val = best_val
    it = 0
    while time.time() - start_loc < max_seconds:
        # propose move
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j: continue
        delta = (rng.random() - 0.5) * step_scale
        # compute bounds
        low = max(-cur_h[i], -(1 - cur_h[j]))
        high = min(1 - cur_h[i], cur_h[j])
        if delta < low: delta = low
        elif delta > high: delta = high
        # skip if low > high (should not)

        new_h[j] -= delta
        new_val = evaluate(new_h, n)
        # acceptance

            j = i + s
            g[i] += 1 - h[j]
            g[j] -= h[i]
        # step size schedule

        # update
        h_new = h - step * g
        # project onto bounds and sum
        h_new = np.clip(h_new, 0.0, 1.0)
        # ensure sum (project to box+sum)
        h_new = project_box_sum(h_new, target_sum)
        # evaluate
        new_val = evaluate_corr(h_new, n)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        # accept new_h for next iteration (always move)
        h = h_new
    return best_h, best_val

import time
import numpy as np

def evaluate_corr(h, n):
    """Maximum overlap integral for a given discretisation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(x, target):
    """Project onto 0≤x≤1 with exact sum=target (L2 optimal)."""
    # Bisection on lambda for clipping x-lambda into [0,1]
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    return y

def subgradient_optimize(h0, time_limit, rng):
    """Sub‑gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_val = evaluate_corr(h, n)

    start = time.time()
    it = 0
    step0 = 0.5  # initial step magnitude

    while time.time() - start < time_limit:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        s = idx - (n - 1)          # integer lag
        # sub‑gradient for this lag
        g = np.zeros_like(h)
        if s >= 0:
            k = n - s
            g[:k] += 1.0 - h[s:]
            g[s:] -= h[:k]
        else:
            s = -s
            k = n - s
            g[s:] += 1.0 - h[:k]
            g[:k] -= h[s:]

        it += 1
        step = step0 / np.sqrt(it)
        # gradient step, then projection onto box & sum constraint
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_box_sum(h_new, target)

        new_val = evaluate_corr(h_new, n)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

        h = h_new

    return best_h, best_val

def initial_triangle(n):
    """Triangular step function with peak 1 at the centre."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    return h

def initial_sine(n):
    """Scaled half‑sine shape (area normalised to 1)."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * xs / 2.0)
    # normalise area to exactly 1
    area = h.sum() * dx
    if area != 0:
        h *= (1.0 / area)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- determine discretisation -------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        m = base.shape[0]
        n = max(200, int(m * 4))
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, base)
    else:
        n = 400
        h0 = initial_triangle(n)

    # enforce constraints
    target_sum = n / 2.0
    h0 = project_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = evaluate_corr(best_h, n)

    # ---- main optimisation phase --------------------------------------------------
    # primary sub‑gradient descent from the provided (or triangle) start
    remaining = budget_s - (time.time() - start_time) - 0.5
    if remaining > 5.0:

        if val_opt < best_val:

    # ---- additional random restarts (small time slices) ---------------------------

        if val_loc < best_val:

    # final projection to guarantee feasibility (numerical drift)
    best_h = project_box_sum(best_h, target_sum)
    best_val = evaluate_corr(best_h, n)

    return (best_h.tolist(), float(best_val), int(n))
