import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size and an initial feasible vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n = h0.shape[0]
    else:
        n = 80
        h0 = np.full(n, 0.5, dtype=np.float64)      # uniform feasible start

    target_sum = n / 2.0                               # because ∫h = 1  and dx = 2/n
    np.clip(h0, 0.0, 1.0, out=h0)

    # simple projection onto the (0,1) simplex
    cur = h0.sum()
    if abs(cur - target_sum) > 1e-12:
        h0 *= target_sum / cur
        np.clip(h0, 0.0, 1.0, out=h0)
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-12:
            free = (h0 > 0.0) & (h0 < 1.0)
            if free.any():
                h0[free] += diff / free.sum()
                np.clip(h0, 0.0, 1.0, out=h0)
            else:                                 # fallback scaling
                h0 *= target_sum / h0.sum()

    dx = 2.0 / n

    # --------------------------------------------------------------
    # Objective: C₅ = max_k ∫ h(x)(1‑h(x+k))dx
    # --------------------------------------------------------------
    def c5_of(vec):
        # correlation of h with (1‑h) using full mode, then scale by dx
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return np.max(corr) * dx

    cur_h = h0.copy()
    cur_score = c5_of(cur_h)

    best_h = cur_h.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # Simulated‑annealing / stochastic local search
    # --------------------------------------------------------------
    start = time.time()
    # leave a small margin for finalisation
    deadline = start + budget_s * 0.97

    T0 = 0.02                     # initial temperature
    max_step = 0.5                # biggest mass transfer allowed
    min_step = 1e-4

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (budget_s * 0.97)   # 0 → 1

        # annealing schedules (linear)
        T = max(1e-6, T0 * (1.0 - progress))
        step = min_step + (max_step - min_step) * (1.0 - progress)

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:

        if rng.random() < 0.5:

            if max_delta <= 0.0:

            old_i, old_j = cur_h[i], cur_h[j]
            cur_h[i] = old_i - delta
            cur_h[j] = old_j + delta
        else:

        # Metropolis acceptance
