def compute_subgradient(h, active_shifts):
    n = len(h)
    grad = np.zeros(n)
    for s in active_shifts:
        if s >= 0:
            # subtract h[i+s] for i in [0, n-s-1]
            grad[:n-s] -= h[s:]
            # subtract h[i-s] for i in [s, n-1]
            grad[s:] -= h[:n-s]
            # add 1 for all i
            grad += 1.0
        else:
            s_abs = -s
            # For s negative treat similarly
            grad[s_abs:] -= h[:n-s_abs]
            grad[:n-s_abs] -= h[s_abs:]
            grad += 1.0
    grad /= len(active_shifts)
    return grad

            grad[:n - s] -= h[s:]
            # subtract shifted backward
            grad[s:] -= h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] -= h[:n - s_abs]
            grad[:n - s_abs] -= h[s_abs:]
    grad /= len(shifts)
    return grad

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the hyperplane sum(x)=target with box bounds
def _project_sum_box(v, target, lo=0.0, hi=1.0, maxit=60):
    """Project v onto {x: lo<=x<=hi, sum(x)=target} using bisection on the
    Lagrange multiplier lambda for the equality constraint."""
    lo_lambda = -2.0
    hi_lambda = 2.0
    for _ in range(maxit):
        lam = (lo_lambda + hi_lambda) * 0.5
        x = np.clip(v + lam, lo, hi)
        s = x.sum()
        if s < target:
            lo_lambda = lam
        else:
            hi_lambda = lam
    lam = (lo_lambda + hi_lambda) * 0.5
    return np.clip(v + lam, lo, hi)

# ------------------------------------------------------------
# Helper: full linear correlation using FFT (real-valued, fast)
def _full_corr(h):
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    conv = np.fft.irfft(H * G, n=size)[: 2 * n - 1]
    return conv

# ------------------------------------------------------------
# Helper: compute C5 (max overlap integral) for a given h
def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = _full_corr(h)
    return conv.max() * dx

# ------------------------------------------------------------
# Helper: subgradient of max_k  ∑ h_i (1-h_{i+k})
def _subgradient(h, active_shifts):
    n = h.shape[0]
    g = np.zeros(n, dtype=np.float64)
    for s in active_shifts:
        g += 1.0                     # the constant term of the gradient
        if s >= 0:
            if s < n:
                g[: n - s] -= h[s:]          # term -h_{i+s}
                g[s:] -= h[: n - s]          # term -h_{i-s}
        else:
            sa = -s
            if sa < n:
                g[sa:] -= h[: n - sa]        # term -h_{i-s}
                g[: n - sa] -= h[sa:]        # term -h_{i+s}
    g /= len(active_shifts)
    return g

    np.random.seed(seed)

    n_list = [n for n in n_list if n >= 30]          # guard against tiny sizes
    n_list = sorted(set(n_list))

    best_h = None
    best_val = np.inf

    # --------------------------------------------------------------------
    for n in n_list:

        if time.time() - start_time > budget_s * 0.95:

        else:

    final_h = _project_sum_box(final_h, final_n / 2.0)
