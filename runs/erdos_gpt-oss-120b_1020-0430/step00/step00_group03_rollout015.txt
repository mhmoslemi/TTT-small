def subgradient_max(h, dx):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)  # index of max correlation
    k = idx - (n - 1)
    # Compute shifted versions
    shifted_plus = np.empty_like(h)
    shifted_minus = np.empty_like(h)
    if k >= 0:
        shifted_plus[:n - k] = h[k:]
        shifted_plus[n - k:] = 0.0
        shifted_minus[k:] = h[:n - k]
        shifted_minus[:k] = 0.0
    else:
        p = -k
        shifted_plus[p:] = h[:n - p]
        shifted_plus[:p] = 0.0
        shifted_minus[:n - p] = h[p:]
        shifted_minus[n - p:] = 0.0
    grad = dx * (np.ones_like(h) - shifted_plus - shifted_minus)
    return grad, k

def random_pair_move(h, rng):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    # compute feasible delta range
    # new hi = hi + delta, new hj = hj - delta
    lo = max(-hi, hj - 1.0)
    hi_delta = min(1.0 - hi, hj)
    if lo >= hi_delta:
        return None  # no feasible move
    delta = rng.uniform(lo, hi_delta)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # Clip just in case numeric issues
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # perhaps adjust the sum to exact S using projection but should be exact.
    return new_h

import numpy as np, time, math

def compute_c5(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr * dx).max()

def proj_capped_simplex(v, target_sum):
    # Project onto [0,1]^n intersect {sum = target_sum}
    lower = 0.0
    upper = 1.0
    v = np.clip(v, lower, upper)
    # If already satisfies sum, return
    s = v.sum()
    if abs(s - target_sum) <= 1e-12:
        return v
    # binary search on lambda
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, lower, upper)

def subgradient_max(h, dx):
    n = h.shape[0]
    # compute correlation vector (without scaling)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)  # lag
    # compute shifted arrays
    shifted_plus = np.zeros_like(h)
    shifted_minus = np.zeros_like(h)
    if k >= 0:
        if k < n:
            shifted_plus[:n - k] = h[k:]
            shifted_minus[k:] = h[:n - k]
    else:
        p = -k
        if p < n:
            shifted_plus[p:] = h[:n - p]
            shifted_minus[:n - p] = h[p:]
    grad = dx * (np.ones_like(h) - shifted_plus - shifted_minus)
    return grad, k

def random_pair_move(h, rng):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi = h[i]
    hj = h[j]
    lo = max(-hi, hj - 1.0)
    hi_delta = min(1.0 - hi, hj)
    if lo >= hi_delta:
        return None
    delta = rng.uniform(lo, hi_delta)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    # Clip small numerical deviations
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h

import time, math
import numpy as np

# ---------- helper functions ----------
def compute_c5(h, dx):
    """Return max_k  ∫ h(x)*(1-h(x+k)) dx approximated on a grid."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (corr * dx).max()


def proj_capped_simplex(v, target):
    """Project v onto the set {0≤x_i≤1, Σx_i = target}."""
    lower, upper = 0.0, 1.0
    v = np.clip(v, lower, upper)
    s = v.sum()
    if abs(s - target) <= 1e-12:
        return v
    lo, hi = -1.0, 1.0           # λ lives in this interval for [0,1] box
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, lower, upper)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, lower, upper)


def subgradient_max(h, dx):
    """Sub‑gradient of the active lag (the one attaining the max)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)                     # signed lag
    # shifted copies, zero when index out of range
    plus = np.zeros_like(h)
    minus = np.zeros_like(h)
    if k >= 0:
        if k < n:
            plus[: n - k] = h[k:]
            minus[k:] = h[: n - k]
    else:
        p = -k
        if p < n:
            plus[p:] = h[: n - p]
            minus[: n - p] = h[p:]
    grad = dx * (np.ones_like(h) - plus - minus)
    return grad, k


def random_pair_move(h, rng):
    """Move a small amount of mass from one position to another, keeping the sum."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    lo = max(-hi, hj - 1.0)
    hi_delta = min(1.0 - hi, hj)
    if lo >= hi_delta:
        return None
    delta = rng.uniform(lo, hi_delta)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h


# ---------- main optimisation ----------
def run(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)

    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 70
        S = n / 2.0
        h0 = proj_capped_simplex(rng.random(n), S)

    dx = 2.0 / n
    S = n / 2.0
    h0 = proj_capped_simplex(h0, S)

    best_h = h0.copy()
    best_score = compute_c5(best_h, dx)
    cur_h = best_h.copy()
    cur_score = best_score

    start = time.time()
    step0 = 5.0
    prob_grad = 0.65
    T0 = best_score * 0.02 + 1e-4
    it = 0

    while True:
        if time.time() - start >= budget_s - 0.1:
            break
        it += 1

        # ---- propose a new candidate ----
        if rng.random() < prob_grad:
            grad, _ = subgradient_max(cur_h, dx)
            step = step0 / math.sqrt(it)
            cand_best = None
            cand_best_score = math.inf
            for factor in (1.0, 0.5, 2.0):
                cand = cur_h - step * factor * grad
                cand = np.clip(cand, 0.0, 1.0)
                cand = proj_capped_simplex(cand, S)
                sc = compute_c5(cand, dx)
                if sc < cand_best_score:
                    cand_best_score = sc
                    cand_best = cand
            cand, cand_score = cand_best, cand_best_score
        else:
            cand = random_pair_move(cur_h, rng)
            if cand is None:
                continue
            cand_score = compute_c5(cand, dx)

        # ---- acceptance ----
        if cand_score < cur_score - 1e-12:
            cur_h, cur_score = cand, cand_score
            if cand_score < best_score - 1e-12:
                best_h, best_score = cand.copy(), cand_score
        else:
            T = max(1e-8, T0 * (1.0 - (time.time() - start) / budget_s))
            if rng.random() < math.exp(-(cand_score - cur_score) / T):
                cur_h, cur_score = cand, cand_score

        # ---- occasional diversification from the best found ----
        if it % 5000 == 0:
            pert = best_h.copy()
            for _ in range(5):
                tmp = random_pair_move(pert, rng)
                if tmp is not None:
                    pert = tmp
            sc = compute_c5(pert, dx)
            if sc < best_score:
                best_h, best_score = pert.copy(), sc
                cur_h, cur_score = pert.copy(), sc

    return best_h.tolist(), float(best_score), int(n)
