import numpy as np
import time

def _project_to_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n with sum == target (Euclidean projection)."""
    # Clip to box first
    y = np.clip(x, 0.0, 1.0)
    # Iteratively adjust the sum by distributing the deficit/excess over free variables
    for _ in range(10):
        s = y.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        free = (y > 0.0) & (y < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # nothing free, cannot fix exactly – break
            break
        y[free] -= diff / cnt
        y = np.clip(y, 0.0, 1.0)
    return y


def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step function h represented by samples."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / N))


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize
        C5 = max_k ∫ h(x) (1 - h(x+k)) dx .
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # Initial construction
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.array(init, dtype=float)
    else:
        # fallback: 120 points, uniform random respecting the sum constraint
        N = 120
        h = np.random.rand(N)
        h = _project_to_bounds_and_sum(h, N / 2.0)

    N = h.shape[0]
    target_sum = N / 2.0
    h = _project_to_bounds_and_sum(h, target_sum)

    best_h = h.copy()
    best_score = _c5_value(best_h)
    curr_h = best_h.copy()
    curr_score = best_score

    # ------------------------------------------------------------------
    # Parameters for the stochastic search
    # ------------------------------------------------------------------
    max_step_factor = 0.5       # scales the random transfer amount
    alpha_grad = 0.08           # step size for gradient-like moves
    T0 = 0.02                   # initial temperature for simulated annealing

    it = 0
    # main loop
    while time.time() - start_time < budget_s * 0.92:
        it += 1
        # --------------------------------------------------------------
        # 1) Random transfer between two positions (keeps the total sum)
        # --------------------------------------------------------------
        i, j = np.random.randint(N, size=2)
        while i == j:
            i = np.random.randint(N)
        # feasible delta such that both stay in [0,1]
        lo = max(-curr_h[i], -(1.0 - curr_h[j]))
        hi = min(1.0 - curr_h[i], curr_h[j])
        if hi <= lo:
            continue
        delta = np.random.uniform(lo, hi) * max_step_factor
        if abs(delta) < 1e-12:
            continue
        new_h = curr_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_score = _c5_value(new_h)

        # acceptance test (simulated annealing)
        T = T0 * (1.0 - (time.time() - start_time) / budget_s)
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            if T > 0:
                prob = np.exp((curr_score - new_score) / (T + 1e-12))
                if np.random.rand() < prob:
                    accept = True
        if accept:
            curr_h = new_h
            curr_score = new_score
            if new_score < best_score:

                best_h = new_h.copy()

        # --------------------------------------------------------------

        if it % 5000 == 0:

            full_corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")
            max_idx = np.argmax(full_corr)
            k = max_idx - (N - 1)          # integer shift (can be negative)

            # gradient of O(k) = sum(h) - sum_i h_i h_{i+k}
            g = np.ones_like(curr_h)
            if k >= 0:
                # forward part
                g[: N - k] -= curr_h[k:]
                # backward part
                g[k:] -= curr_h[: N - k]
            else:
                kk = -k
                g[kk:] -= curr_h[: N - kk]
                g[: N - kk] -= curr_h[kk:]

            # take a step opposite to the subgradient, then project
            trial = _project_to_bounds_and_sum(curr_h - alpha_grad * g, target_sum)
            trial_score = _c5_value(trial)

            # accept if improvement (or with Metropolis prob.)
            accept = False
            if trial_score < curr_score:
                accept = True
            else:
                if T > 0:

                    if np.random.rand() < prob:

            if accept:

                if trial_score < best_score:
