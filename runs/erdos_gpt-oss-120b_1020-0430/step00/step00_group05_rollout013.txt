def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    rng = np.random.default_rng(seed)
    # Choose n: we can try to pick moderate n, maybe 200 or 300 determined by time.
    n = 200
    # maybe variable: if time remaining > some threshold, we can increase n gradually.
    # but we can just use 200.

    best_h = None
    best_val = float('inf')
    # We'll do multiple restarts: while time left, start new random h and hillclimb.
    while time.time() - start < budget_s * 0.9:  # keep margin
        # random initial binary vector
        perm = rng.permutation(n)
        h = np.zeros(n, dtype=np.float64)
        h[perm[:n//2]] = 1.0
        cur_val = evaluate_erdos_solution(h)
        # local search loops: attempt many swaps
        # Determine number of iterations based on time left.
        # We'll set inner loop for a certain count, e.g., 20000,
        # but break if time low.
        max_iter = 20000
        # Precompute indices lists for ones and zeros for quick random picks
        for it in range(max_iter):
            # compute time check each maybe 1000 iterations
            if it % 1000 == 0:
                if time.time() - start >= budget_s * 0.95:
                    break
            # pick random positions
            # we need indices where h==1 and h==0
            # We can maintain arrays of indices of ones and zeros.
            # Simpler: choose random i from [0,n), j equal to random but ensure h[i]!=h[j].
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            if h[i] == h[j]:
                continue
            # Flip them:
            h[i], h[j] = h[j], h[i]
            new_val = evaluate_erdos_solution(h)
            improve = new_val < cur_val - 1e-12  # tolerance
            if improve:
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
            else:
                # revert
                h[i], h[j] = h[j], h[i]
        # optional early exit if best_val is already low enough.
        if best_val <= 0.3808:
            break
    return best_h, float(best_val), n

def hill_climb(h, max_iter, rng):
    n = len(h)
    cur_val = compute_c5(h)
    best_h_local = h.copy()
    best_val_local = cur_val
    for it in range(max_iter):
        # pick swap
        # Choose indices of opposite values.
        idx_one = rng.integers(n)
        while h[idx_one] != 1.0:
            idx_one = rng.integers(n)
        idx_zero = rng.integers(n)
        while h[idx_zero] != 0.0:
            idx_zero = rng.integers(n)
        # perform swap
        h[idx_one] = 0.0
        h[idx_zero] = 1.0
        new_val = compute_c5(h)
        if new_val <= cur_val:
            cur_val = new_val
            # record best local if improved
            if new_val < best_val_local:
                best_val_local = new_val
                best_h_local = h.copy()
        else:
            # revert
            h[idx_one] = 1.0
            h[idx_zero] = 0.0
        # optional early exit if cur_val <= target threshold
    return best_h_local, best_val_local

import time
import math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Exact same evaluation as the judge uses."""
    n = h.shape[0]
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)))


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points)."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------
    # 1)  Choose a starting vector (binary, sum = n/2)
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
        # force binary representation
        h = (h0 >= 0.5).astype(np.float64)
        target_ones = n // 2
        cur_ones = int(h.sum())
        diff = target_ones - cur_ones
        if diff > 0:
            zeros = np.where(h == 0)[0]
            pick = rng.choice(zeros, size=diff, replace=False)
            h[pick] = 1.0
        elif diff < 0:
            ones = np.where(h == 1)[0]
            pick = rng.choice(ones, size=-diff, replace=False)
            h[pick] = 0.0
    else:
        n = 200                     # a modest size, well below the 1 k limit
        h = np.zeros(n, dtype=np.float64)
        perm = rng.permutation(n)
        h[perm[: n // 2]] = 1.0

    best_h = h.copy()
    best_val = _c5_value(best_h)

    # -------------------------------------------------------------
    # 2)  Local search / simulated annealing
    # -------------------------------------------------------------
    target = 0.3808   # we stop as soon as we beat this value

    def _time_left() -> float:
        return budget_s - (time.time() - start)

    while _time_left() > 2.0:
        # occasionally restart from a fresh random configuration
        if rng.random() < 0.2:
            cur = np.zeros(n, dtype=np.float64)
            perm = rng.permutation(n)
            cur[perm[: n // 2]] = 1.0
        else:
            cur = best_h.copy()

        cur_val = _c5_value(cur)

        # number of attempted swaps for this run – scales with remaining time
        max_iter = int(15000 * (_time_left() / budget_s)) + 2000
        max_iter = max(2000, min(max_iter, 60000))

        for it in range(max_iter):
            if it % 2000 == 0 and _time_left() < 0.5:
                break

            # temperature linearly decreasing to zero
            T = 0.02 * (1.0 - it / max_iter)

            # pick a random 1‑index and a random 0‑index
            i = rng.integers(n)
            while cur[i] != 1.0:
                i = rng.integers(n)
            j = rng.integers(n)
            while cur[j] != 0.0:
                j = rng.integers(n)

            # perform the swap
            cur[i] = 0.0
            cur[j] = 1.0
            new_val = _c5_value(cur)
            delta = new_val - cur_val

            # accept if improvement or with Metropolis probability
            if delta <= 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = cur.copy()
                    if best_val <= target:
                        break
            else:
                # revert the swap
                cur[i] = 1.0
                cur[j] = 0.0

        if best_val <= target:
            break
        if _time_left() < 1.0:
            break

    # final safety evaluation (ensures the returned bound matches the vector)
    final_val = _c5_value(best_h)
    return best_h, float(final_val), n
