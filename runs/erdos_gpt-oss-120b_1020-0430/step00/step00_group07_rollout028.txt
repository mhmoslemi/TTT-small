import numpy as np
import time

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the set { w : 0 <= w_i <= 1, sum(w) = target }.
    Uses a simple bisection on the Lagrange multiplier λ such that
        w_i = clip(v_i - λ, 0, 1)
    """
    lo = np.min(v) - 1.0  # ensures w_i = 1 for all i when λ = lo
    hi = np.max(v)        # ensures w_i = 0 for all i when λ = hi
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    return w

def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute C5 = max_k ∫ h(x) (1‑h(x+k)) dx using discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _worst_shift_and_gradient(h: np.ndarray):
    """
    Return the worst shift (as index offset from -n+1) and the corresponding gradient
    of the overlap term for that shift.
    """
    n = h.shape[0]
    # correlation (without dx)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(corr.argmax())
    shift = abs(max_idx - (n - 1))          # absolute offset in samples

    if shift == 0:
        grad = 2.0 * h
    else:
        grad = np.zeros_like(h)
        # contribution from h_{i+shift}
        grad[: n - shift] += h[shift:]
        # contribution from h_{i-shift}
        grad[shift:] += h[: n - shift]
    return shift, grad

def _random_feasible(n: int, target: float) -> np.ndarray:
    """Generate a random point inside the capped simplex."""
    v = np.random.rand(n)
    return _project_capped_simplex(v, target)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 0.5   # a small safety margin

    # Determine the number of points
    if 'initial_h_values' in globals():
        n = int(initial_h_values.shape[0])
    else:
        n = 80                      # fallback size
    dx = 2.0 / n
    target_sum = n / 2.0           # because sum(h)*dx = 1

    # ------------------------------------------------------------------
    # Build a few deterministic starting candidates
    candidates = []

    # 1) use the provided solution (if any)
    if 'initial_h_values' in globals():
        h0 = np.clip(np.asarray(initial_h_values, dtype=float), 0.0, 1.0)
        if abs(h0.sum() - target_sum) > 1e-9:
            h0 = _project_capped_simplex(h0, target_sum)
        candidates.append(h0)

    # 2) a triangular “tent” shape (peak = 1 at the centre)
    x = np.linspace(dx * 0.5, 2.0 - dx * 0.5, n)
    h_tri = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
    if abs(h_tri.sum() - target_sum) > 1e-9:
        h_tri = _project_capped_simplex(h_tri, target_sum)
    candidates.append(h_tri)

    # Evaluate initial candidates and keep the best
    best_h = None
    best_val = np.inf
    for cand in candidates:
        val = _c5_value(cand, dx)
        if val < best_val:
            best_val = val
            best_h = cand.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop (gradient‑step + occasional random restart)
    h = best_h.copy()
    step = 0.2                     # initial step magnitude
    no_improve = 0
    RESTART_AFTER = 3000          # iterations without improvement before restart

    while time.time() < end_time:
        # current C5 and the worst shift
        cur_val = _c5_value(h, dx)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        shift, grad = _worst_shift_and_gradient(h)

        # Try a gradient step with the current step size
        improved = False
        for _ in range(5):
            if step < 1e-12:
                break
            v = h + step * grad
            h_new = _project_capped_simplex(v, target_sum)
            new_val = _c5_value(h_new, dx)
            if new_val < cur_val - 1e-12:
                h = h_new
                step = min(step * 1.1, 2.0)   # slightly increase step after success
                improved = True
                break
            else:
                step *= 0.5                    # reduce step and retry

        if not improved:
            # small random jitter to escape flat regions
            jitter = (np.random.rand(n) - 0.5) * 0.02
            h = _project_capped_simplex(h + jitter, target_sum)
            step = max(step * 0.9, 1e-8)

        # restart if stuck for a while
        if no_improve >= RESTART_AFTER:
            h = _random_feasible(n, target_sum)
            step = 0.2
            no_improve = 0

    # ------------------------------------------------------------------
    # Final evaluation of the best solution found
    final_val = _c5_value(best_h, dx)

    # Return as required: (h_values, c5_bound, n_points)
    return (best_h, float(final_val), n)
