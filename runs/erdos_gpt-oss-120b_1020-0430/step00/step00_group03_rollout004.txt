import time, random, numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    # set random seeds
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # load or create initial vector h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.shape[0]
    else:
        n = 70
        h = np.full(n, 0.5, dtype=np.float64)

    desired_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - desired_sum) > 1e-12:
        diff = desired_sum - cur_sum
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        cur_sum = h.sum()
        diff2 = desired_sum - cur_sum
        if abs(diff2) > 1e-12:
            h[1] = np.clip(h[1] + diff2, 0.0, 1.0)

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # helper to evaluate the objective C5 = max_k ∫h(x)(1‑h(x+k))dx
    # using the identity C5 = 1 - min autocorrelation * dx
    # ------------------------------------------------------------------
    def score_vec(v):
        corr = np.correlate(v, v, mode='full')
        return 1.0 - np.min(corr) * dx

    best_h = h.copy()
    best_score = score_vec(best_h)
    cur_h = best_h.copy()
    cur_score = best_score

    start = time.time()
    deadline = start + budget_s - 0.2   # safety margin

    # annealing parameters
    step_start = 0.08
    step_end   = 0.001
    temp_start = 0.015
    temp_end   = 0.0002

    while True:
        now = time.time()
        if now >= deadline:
            break
        prog = (now - start) / (budget_s - 0.2)
        step = step_start * (1 - prog) + step_end * prog
        temp = temp_start * (1 - prog) + temp_end * prog

        # --------------------------------------------------------------
        # find the lag giving the smallest autocorrelation (worst overlap)
        # --------------------------------------------------------------
        corr_full = np.correlate(cur_h, cur_h, mode='full')
        min_idx = int(np.argmin(corr_full))
        L = min_idx - (n - 1)            # shift corresponding to worst lag

        # --------------------------------------------------------------
        # propose a move
        # --------------------------------------------------------------
        if random.random() < 0.7:   # gradient‑guided move
            g = np.zeros(n, dtype=np.float64)
            if L >= 0:
                g[:n - L] += cur_h[L:]
                g[L:]     += cur_h[:n - L]
            else:
                Lp = -L
                g[Lp:]    += cur_h[:n - Lp]
                g[:n - Lp] += cur_h[Lp:]

            dec = np.where(cur_h > 1e-12)[0]
            inc = np.where(cur_h < 1 - 1e-12)[0]
            if dec.size and inc.size:
                src = dec[np.argmin(g[dec])]
                tgt = inc[np.argmax(g[inc])]
                if src == tgt:
                    src, tgt = dec[0], inc[0]
                max_move = min(step, cur_h[src], 1.0 - cur_h[tgt])
                delta = random.random() * max_move if max_move > 0 else 0.0
            else:
                delta = 0.0
        else:                       # pure random transfer
            src = random.randrange(n)
            tgt = random.randrange(n - 1)
            if tgt >= src:
                tgt += 1
            if random.random() < 0.5:
                s, t = src, tgt
            else:
                s, t = tgt, src
            max_move = min(step, cur_h[s], 1.0 - cur_h[t])
            delta = random.random() * max_move if max_move > 0 else 0.0
            src, tgt = s, t

        if delta == 0.0:
            continue

        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[tgt] += delta

                cur_h, cur_score = new_h, new_score

    # ------------------------------------------------------------------

            best_h[1] = np.clip(best_h[1] + diff2, 0.0, 1.0)

    final_score = score_vec(best_h)
    return best_h, float(final_score), int(n)
