    n = len(h0)

    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable lse
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    return np.max(corr) * (2.0 / h.size)

    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})

                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the optimizer callbacks (avoid closures / lambdas)
CURRENT_TARGET = None   # will be set to n/2 before each SLSQP run
CURRENT_BETA = None     # will be set to the beta value for soft‑max objective


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_{k} Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C_5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def slsqp_optimize(h0, beta, max_iter=200):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # Enforce the integral exactly (numerical drift may appear)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution loop
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = np.full(n, 0.5)
    # random pairwise transfers to mix the vector
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Simple deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # slowly shrink the step size
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c
