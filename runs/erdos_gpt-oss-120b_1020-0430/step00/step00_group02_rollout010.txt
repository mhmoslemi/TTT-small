import numpy as np
import time

def compute_max_overlap(h: np.ndarray) -> float:
    """
    Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation.
    """
    n = h.shape[0]
    # next power of two for FFT size
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    G = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(H * G, N)[: 2 * n - 1]
    # conv contains sums of products; multiply by dx = 2/n
    return float(np.max(conv) * (2.0 / n))


def propose_swap(h: np.ndarray, eps: float):
    """
    Propose a feasible move that conserves the total sum:
    transfer a small amount delta from index i to index j.
    Returns (i, j, delta) or None if no feasible pair is found.
    """
    # Positions that can give away mass (h>0) and receive mass (h<1)
    donors = np.where(h > 1e-12)[0]
    receivers = np.where(h < 1.0 - 1e-12)[0]
    if donors.size == 0 or receivers.size == 0:
        return None
    i = int(np.random.choice(donors))
    j = int(np.random.choice(receivers))
    if i == j:
        return None
    hi, hj = h[i], h[j]
    # feasible interval for delta (move from i to j)
    lower = max(-eps, hi - 1.0, -hj)
    upper = min(eps, hi, 1.0 - hj)
    if lower >= upper:
        return None
    delta = float(np.random.uniform(lower, upper))
    return i, j, delta

        h = np.clip(h * scale, 0.0, 1.0)
        # fix any residual discrepancy by adjusting the first element
        diff = target_sum - h.sum()
        h[0] = min(max(h[0] + diff, 0.0), 1.0)

    # --- optimisation -------------------------------------------------
    best_h = h.copy()
    best_val = compute_max_overlap(best_h)
    cur_h = h.copy()
    cur_val = best_val

    # annealing parameters
    T0 = 0.30          # initial temperature
    eps0 = 0.5         # initial maximal delta
    eps_min = 0.001    # final minimal delta

    # main simulated‑annealing loop (90% of allocated time)
    while time.time() - start_time < 0.9 * time_limit:
        elapsed = time.time() - start_time
        frac = elapsed / (0.9 * time_limit)
        T = T0 * (1.0 - frac) + 1e-12
        eps = eps0 * (1.0 - frac) + eps_min * frac

        proposal = propose_swap(cur_h, eps)
        if proposal is None:
            continue
        i, j, delta = proposal

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_val = compute_max_overlap(new_h)

        if new_val < cur_val or np.random.rand() < np.exp((cur_val - new_val) / max(T, 1e-12)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
                if best_val < 0.3808:      # early exit if we beat target
                    break

    # greedy refinement with very small moves until time runs out
    while time.time() - start_time < time_limit:
        proposal = propose_swap(cur_h, eps_min)
        if proposal is None:
            continue
        i, j, delta = proposal
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = compute_max_overlap(new_h)
        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
                if best_val < 0.3808:
                    break

    # final correction of the integral (tiny numerical drift)
    final_sum = best_h.sum()
    if abs(final_sum - target_sum) > 1e-9:
        diff = target_sum - final_sum
        # safely adjust a single entry without breaking bounds
        idx = np.argmax(best_h) if diff > 0 else np.argmin(best_h)
        best_h[idx] = min(max(best_h[idx] + diff, 0.0), 1.0)

    final_val = compute_max_overlap(best_h)

    # return as required by the harness
    return best_h.tolist(), float(final_val), n
