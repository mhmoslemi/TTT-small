    lo = np.min(x) - 1.0

def project_to_sum_box(x, target):
    # binary search on lambda
    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0, 1)
        s = np.sum(y)
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo+hi)/2.0, 0, 1)

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def worst_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n-1)
    return shift, corr[idx]

def gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        # overlapping region length n-shift
        g[:n-shift] += h[shift:]
        g[shift:] += h[:n-shift]
    else:
        s = -shift
        g[:n-s] += h[s:]
        g[s:] += h[:n-s]
    return g

def _project_to_box_sum(x, target, iters=60):
    """Euclidean projection of x onto {0<=y<=1, sum(y)=target}."""
    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(iters):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = np.sum(y)
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def _c5_of(h):
    """Return the C5 value for a step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def _worst_shift(h):
    """Index shift producing the maximal overlap (zero‑lag is at n‑1)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return idx - (n - 1)


def _gradient_for_shift(h, shift):
    """Gradient of the product sum for a given shift (to increase the product)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        g[:n - s] += h[s:]
        g[s:] += h[:n - s]
    else:
        s = -shift
        g[:n - s] += h[s:]
        g[s:] += h[:n - s]
    return g


def _improve_one_start(start_h, target, time_limit):
    """Projected gradient ascent from a start point, limited by time_limit seconds."""
    cur = start_h.copy()
    cur_val = _c5_of(cur)
    best_h = cur.copy()
    best_val = cur_val
    n = cur.shape[0]
    step_sizes = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    start = time.time()
    for it in range(2000):
        if time.time() - start > time_limit:
            break
        shift = _worst_shift(cur)
        g = _gradient_for_shift(cur, shift)
        if np.allclose(g, 0):
            break

        cand_h = None
        cand_val = np.inf
        for eta in step_sizes:
            h_tmp = cur + eta * g
            h_tmp = _project_to_box_sum(h_tmp, target)
            val = _c5_of(h_tmp)
            if val < cand_val:
                cand_val = val
                cand_h = h_tmp
        if cand_val < cur_val - 1e-12:
            cur = cand_h
            cur_val = cand_val
            if cur_val < best_val:
                best_h = cur
                best_val = cur_val
        else:
            break

    # fallback random local moves after gradient stalls
    for _ in range(50):
        if time.time() - start > time_limit:
            break
        i, j = np.random.choice(n, 2, replace=False)
        max_ij = min(cur[i], 1.0 - cur[j])
        max_ji = min(cur[j], 1.0 - cur[i])
        if max_ij <= 0 and max_ji <= 0:
            continue
        if np.random.rand() < 0.5 and max_ij > 0:
            delta = max_ij * np.random.rand()
            new_h = cur.copy()
            new_h[i] -= delta
            new_h[j] += delta
        elif max_ji > 0:
            delta = max_ji * np.random.rand()
            new_h = cur.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            continue
        new_val = _c5_of(new_h)
        if new_val < best_val:
            cur = new_h
            cur_val = new_val
            best_h = new_h
            best_val = new_val
    return best_h, best_val


def _random_feasible(n, target):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    x = np.random.rand(n)
    return _project_to_box_sum(x, target)
