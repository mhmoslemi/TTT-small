import time
import numpy as np

def _compute_c5(h):
    """Return C5 for a step function sampled in h (array of shape (n,))."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h) using zero padding (mode='full')
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def _make_two_level(n, e, a):
    """
    Build a vector of length n with two levels:
    * first/last e entries = a (edge height)
    * remaining middle entries = b chosen so that sum(h) = n/2.
    Returns None if b would be outside [0,1].
    """
    if e * 2 > n:                     # impossible, no interior left
        return None
    # target total sum (integral = 1)
    target = n / 2.0
    # remaining slots after the two edge blocks
    interior = n - 2 * e
    # solve for interior height b:
    # 2*e*a + interior*b = target  ->  b = (target - 2*e*a) / interior
    b = (target - 2.0 * e * a) / interior
    if b < 0.0 or b > 1.0:
        return None
    h = np.empty(n, dtype=np.float64)
    if e > 0:
        h[:e] = a
        h[-e:] = a
    if interior > 0:
        h[e:n - e] = b
    return h

def _initial_candidates(n):
    """
    Generate a set of simple two‑level candidates.
    Edge length e runs from 0 to n//4,
    edge height a runs over a modest grid.
    """
    cand = []
    a_grid = np.linspace(0.5, 1.0, 11)   # 0.5 … 1.0 inclusive
    for e in range(0, n // 4 + 1):
        for a in a_grid:
            h = _make_two_level(n, e, a)
            if h is not None:
                cand.append(h)
    return cand

def _random_feasible_vector(n, rng):
    """Draw a random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    # start from exponential variates (positive, less concentrated than uniform)
    v = rng.exponential(scale=1.0, size=n)
    # scale to target sum
    v = v * (target / v.sum())
    # clip and rebalance if any entry exceeds 1
    while True:
        over = v > 1.0
        if not np.any(over):
            break
        excess = v[over] - 1.0
        v[over] = 1.0
        # redistribute the excess uniformly among the non‑saturated entries
        free = ~over
        if not np.any(free):
            # all entries are 1, which can't happen because target<n
            raise RuntimeError("Infeasible random start")
        v[free] -= excess.sum() / free.sum()
    # final tiny numerical correction
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        v += diff / n
    np.clip(v, 0.0, 1.0, out=v)
    return v

def _simulated_annealing(start_h, max_iter, time_limit, rng):
    """
    Simple SA that preserves the sum constraint by adjusting two entries at a time.
    Returns the best vector seen and its C5 value.
    """
    h = start_h.copy()
    n = h.size
    target_sum = n / 2.0
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    start_time = time.time()
    # linear cooling schedule
    for it in range(1, max_iter + 1):
        if time.time() - start_time > time_limit:
            break

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        hi, hj = h[i], h[j]

        # feasible delta so that both stay in [0,1] after transfer:
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue

        # cooling factor (range 0‑1)
        frac = 1.0 - it / max_iter
        # restrict step size gradually
        delta = delta_min + rng.random() * (delta_max - delta_min) * frac

        # propose move
        h[i] = hi + delta
        h[j] = hj - delta

        new_score = _compute_c5(h)

        # temperature for Metropolis acceptance
        T = max(0.001, 0.02 * frac)   # modest starting temperature
        accept = new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with values in [0,1] and integral 1
    that minimises C5 = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_vector, c5_value, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose a reasonable resolution (even number simplifies the sum constraint)
    n = 200                      # 200 samples gives dx = 0.01
    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 1) Simple grid of two‑level step functions (edges of height a, interior b)
    # ------------------------------------------------------------------
    best_h = None
    best_score = np.inf

    for cand in _initial_candidates(n):
        score = _compute_c5(cand)
        if score < best_score:
            best_score = score
            best_h = cand
        # early exit if we already beat the target record
        if best_score <= 0.3808:
            break

    # ------------------------------------------------------------------
    # 2) If the simple grid did not reach the target, improve with SA.
    #    Use a few random restarts, each limited by the remaining budget.
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if best_score > 0.3808 and remaining > 5.0:
        # allocate the rest of the time to simulated annealing
        # number of restarts depends on remaining budget
        restarts = max(1, int(remaining // 30))   # about 30 s per SA run
        iters_per_run = 300_000                  # enough iterations for convergence
        for _ in range(restarts):
            # initialise either from the best candidate or a fresh random feasible vector
            if rng.random() < 0.3 and best_h is not None:
                start_vec = best_h.copy()
                # add a tiny random perturbation to escape possible plateaus
                perturb = rng.normal(scale=0.01, size=n)
                start_vec += perturb
                np.clip(start_vec, 0.0, 1.0, out=start_vec)
                # re‑project to the exact sum
                diff = target_sum - start_vec.sum()
                start_vec += diff / n
                np.clip(start_vec, 0.0, 1.0, out=start_vec)
            else:
                start_vec = _random_feasible_vector(n, rng)

            # run SA with a time limit respecting the global budget
            now = time.time()
            time_for_this = min(remaining, 30.0)   # at most 30 s per SA call
            vec, score = _simulated_annealing(start_vec, iters_per_run, time_for_this, rng)

            if score < best_score:
                best_score = score
                best_h = vec

            # update remaining time
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            if remaining < 5.0 or best_score <= 0.3808:
                break

    # ------------------------------------------------------------------
    # 3) Final tidy‑up: enforce constraints exactly (clipping + sum correction)
    # ------------------------------------------------------------------
    h = np.clip(best_h, 0.0, 1.0)
    # correct any tiny drift in the integral
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute the correction uniformly, then clip again
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
        # a second pass to ensure the sum is exact (tiny residuals are harmless)
        corr = target_sum - h.sum()
        if abs(corr) > 1e-12:
            # distribute the remaining error over a random subset of indices that are not at bounds
            free = (h > 1e-12) & (h < 1 - 1e-12)
            if free.any():
                h[free] += corr / free.sum()
                np.clip(h, 0.0, 1.0, out=h)

    final_score = _compute_c5(h)

    return (h, final_score, n)
