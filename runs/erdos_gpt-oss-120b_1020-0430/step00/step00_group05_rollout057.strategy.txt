def compute_gradient(h, k, n_points):
    """Compute the gradient of the correlation for lag k at point h."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for j in range(n_points):
        # valid_forward includes h[j] if j+k is in the grid
        if 0 <= j + k < n_points:
            grad[j] += dx * (1.0 - h[j + k])
        # valid_backward includes h[j] if j-k is in the grid
        if 0 <= j - k < n_points:
            grad[j] -= dx * h[j - k]
    return grad


def score(h, n_points):
    """Compute score for a given state h at grid size n_points."""
    dx = 2.0 / n_points
    # Compute correlation (mode='full' gives lags from 1-n_points to n_points-1)
    corr_full = np.correlate(h, 1 - h, mode='full')
    # The score is the maximum correlation value normalized by dx
    return dx * corr_full.max()


def project(h, desired_sum, dx, n_points):
    """Project h to unit interval and adjust to have desired sum near desired_sum."""
    # Respect box constraints first
    h_proj = np.clip(h, 0.0, 1.0)

        return h_proj
