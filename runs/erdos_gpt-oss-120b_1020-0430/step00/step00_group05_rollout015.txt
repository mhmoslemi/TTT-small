Let a=1, b=0 before swap. After swap, at i1 we have 0, at i2 we have 1.

Define delta array d[i] for change in h at index i: Δh[i] = new - old. So Δh[i1] = -1, Δh[i2] = +1.

Then new correlation C'[k] = Σ_i (h_i + Δh_i) * (1 - (h_{i+k} + Δh_{i+k})).

Expand: C'[k] = Σ_i (h_i * (1 - h_{i+k})) + Σ_i Δh_i * (1 - h_{i+k}) - Σ_i h_i * Δh_{i+k} - Σ_i Δh_i * Δh_{i+k}.

Simplify because Δh_i nonzero only at i1 and i2.

def run(seed=42,budget_s=1100,**kwargs):
    import numpy as np, time, math, random
    start = time.time()
    np.random.seed(seed)
    random.seed(seed)

    N = 200  # may adjust if enough time
    half = N//2
    # initialize with random binary
    h = np.zeros(N, dtype=np.float64)
    h[:half] = 1.0
    np.random.shuffle(h)
    # function to compute max overlap count
    def max_overlap_count(arr):
        # arr is binary float array
        c = np.correlate(arr, 1.0 - arr, mode='full')
        return int(c.max())
    cur_max = max_overlap_count(h)
    cur_score = cur_max * (2.0/N)

    best_h = h.copy()
    best_max = cur_max
    best_score = cur_score

    # parameters
    max_no_improve = 20000
    no_improve_counter = 0
    iteration = 0

    while True:
        if time.time() - start > budget_s - 0.2:
            break
        # schedule temperature
        # progress fraction of total time
        progress = (time.time() - start) / (budget_s - 0.2)
        temp = max(0.5 * (1 - progress), 0.001) * 5  # fudge factor
        # propose swap
        i1 = np.random.randint(N)
        i2 = np.random.randint(N)
        if h[i1] == h[i2]:
            continue
        # swap
        h[i1], h[i2] = h[i2], h[i1]
        new_max = max_overlap_count(h)
        # decide accept
        if new_max <= cur_max:
            accept = True
        else:
            delta = new_max - cur_max
            # temperature scaling: use temp as like 1.0 => moderate acceptance
            # but we can adjust acceptance probability
            accept = (math.exp(-delta / temp) > random.random())
        if accept:
            cur_max = new_max
            cur_score = new_max * (2.0/N)
            # reset no_improve?
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                best_score = cur_score
                no_improve_counter = 0
            else:
                no_improve_counter += 1
        else:
            # revert swap
            h[i1], h[i2] = h[i2], h[i1]
            no_improve_counter += 1
        iteration += 1
        # restart if stagnation
        if no_improve_counter >= max_no_improve:
            # random restart
            h = np.zeros(N)
            h[:half] = 1
            np.random.shuffle(h)
            cur_max = max_overlap_count(h)
            cur_score = cur_max * (2.0/N)
            no_improve_counter = 0
    # final evaluation
    # ensure best_h meets constraints
    assert abs(best_h.sum() - half) < 1e-6
    # convert to list of floats

    c5_bound = best_score

import time, math
import numpy as np

def _max_overlap_count(arr):
    """Return integer max overlap count for binary/float array."""
    # correlation h * (1-h)  (full mode, zero padding)
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return int(corr.max()), corr

def _mismatches_for_shift(h, shift):
    """Indices i where h[i]==1 and h[i+shift]==0 (for given shift)."""
    n = h.shape[0]
    if shift >= 0:
        idx = np.arange(n - shift)
        bad = (h[idx] == 1) & (h[idx + shift] == 0)
        return idx[bad]
    else:
        d = -shift
        idx = np.arange(d, n)
        bad = (h[idx] == 1) & (h[idx - d] == 0)
        return idx[bad]

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    start = time.time()
    np.random.seed(seed)

    # target resolution – larger values give more flexibility.
    N = 200                     # must be even so that N/2 is integer
    half = N // 2
    dx = 2.0 / N

    # ---------- initial construction ----------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        # Resample to length N if needed
        if init.size != N:
            from scipy.interpolate import interp1d
            xs_old = np.linspace(0, 2, init.size, endpoint=False)
            xs_new = np.linspace(0, 2, N, endpoint=False)
            f = interp1d(xs_old, init, kind="linear", fill_value="extrapolate")
            init = f(xs_new)
        # choose the half largest entries to be 1, rest 0
        order = np.argsort(-init)
        h = np.zeros(N, dtype=float)
        h[order[:half]] = 1.0
    else:
        h = np.zeros(N, dtype=float)
        h[:half] = 1.0
        np.random.shuffle(h)

    cur_max, cur_corr = _max_overlap_count(h)
    cur_score = cur_max * dx
    best_h = h.copy()
    best_max = cur_max
    best_score = cur_score

    # ---------- optimization parameters ----------
    target_c5 = 0.3808                 # stop early if we beat this
    target_max = int(math.floor(target_c5 * N / 2))
    temp0 = 5.0                        # initial temperature (in count units)
    min_temp = 0.05
    p_targeted = 0.75                  # probability to use a targeted move
    stagnation_limit = 40000          # restart after this many non‑improving steps
    steps_without_improve = 0

    # Pre‑compute index lists for ones and zeros (updated lazily)
    while time.time() - start < budget_s - 0.2:
        # linear cooling schedule
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 0.2)
        temp = max(temp0 * (1.0 - progress), min_temp)

        use_targeted = (np.random.rand() < p_targeted) and (cur_max > best_max)
        if use_targeted:
            # use the shift that currently attains the maximum
            idx_max = int(cur_corr.argmax())
            shift = idx_max - (N - 1)
            mism = _mismatches_for_shift(h, shift)
            if mism.size == 0:
                # fallback to random move
                use_targeted = False
            else:
                i = int(np.random.choice(mism))
                zeros = np.where(h == 0)[0]
                j = int(np.random.choice(zeros))
        if not use_targeted:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))

        # perform the swap
        h[i], h[j] = h[j], h[i]

        new_max, new_corr = _max_overlap_count(h)

        if new_max <= cur_max:
            accept = True
        else:
            delta = new_max - cur_max
            accept = (math.exp(-delta / temp) > np.random.rand())

        if accept:
            cur_max, cur_corr = new_max, new_corr
            cur_score = cur_max * dx
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                best_score = cur_score
                steps_without_improve = 0
                # early exit if we already beat the target
                if best_max <= target_max:
                    break
            else:
                steps_without_improve += 1
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            steps_without_improve += 1

        if steps_without_improve >= stagnation_limit:
            # random restart to escape possible plateau
            h = np.zeros(N, dtype=float)
            h[:half] = 1.0
            np.random.shuffle(h)
            cur_max, cur_corr = _max_overlap_count(h)
            cur_score = cur_max * dx
            steps_without_improve = 0

    # ---------- final polishing (simple greedy pass) ----------
    # try a few random improving swaps even after main loop
    for _ in range(2000):
        if best_max <= target_max:
            break
        # focus on current worst shift
        idx_max = int(cur_corr.argmax())
        shift = idx_max - (N - 1)
        mism = _mismatches_for_shift(best_h, shift)
        if mism.size == 0:

        i = int(np.random.choice(mism))

        new_max, _ = _max_overlap_count(best_h)

            best_score = best_max * dx
        else:

    # final sanity checks
    assert abs(best_h.sum() - half) < 1e-9

    return best_h.tolist(), best_score, N
