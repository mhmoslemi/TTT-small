import time
import math
import numpy as np

def _corr_unscaled(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1-h) without the dx factor."""
    return np.correlate(h, 1.0 - h, mode="full")


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search a step function h on [0,2] with ∫h=1 that minimizes
    C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    try:
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.size
    except Exception:  # initial_h_values not present
        n = 95                         # a reasonable small even/odd size
        target_sum = n / 2.0
        h = np.zeros(n, dtype=np.float64)
        ones = int(math.floor(target_sum))
        h[:ones] = 1.0
        rest = target_sum - ones
        if rest > 0:
            h[ones] = rest
        np.random.shuffle(h)

    # guarantee feasibility (numeric safety)
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple projection onto the hyperplane while respecting box constraints
        diff = target_sum - cur_sum
        # distribute diff over non‑saturated coordinates
        for i in range(n):
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0:
                sub = min(-diff, h[i])

                diff += sub
            if abs(diff) < 1e-12:

    # ------------------------------------------------------------------

    return (best_h.tolist(), float(final_c5), int(n))
