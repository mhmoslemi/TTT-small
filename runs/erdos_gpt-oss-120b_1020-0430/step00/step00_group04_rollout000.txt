def propose_move(h, eps):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # amount to transfer from i to j
    # ensure after move h[i] - eps >=0 and h[j] + eps <=1
    max_eps = min(eps, h[i], 1 - h[j])
    # might be zero; try opposite direction also.
    if max_eps <= 0:
        # choose different indices or reverse
        max_eps = min(eps, h[j], 1 - h[i])
        if max_eps <= 0:
            return h, 0
        # swap direction
        i, j = j, i
    delta = np.random.rand() * max_eps  # random within bound
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    # sum unchanged
    return new_h, delta

import time
import numpy as np

def _project_box_simplex(y, target):
    """
    Euclidean projection of y onto the set {x | 0<=x_i<=1, sum x_i = target}.
    Uses a simple bisection on the threshold lambda.
    """
    lo = y.min() - 1.0
    hi = y.max()
    # binary search for lambda
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    # fix tiny residual sum error
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            idx = np.flatnonzero(free)[0]
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
        else:
            # all at bounds, adjust a random index (will stay at bound)
            idx = np.random.randint(len(x))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

        parent_n = init_parent.size
        # interpolated version on our grid
        xs_parent = np.linspace(0.0, 2.0, parent_n, endpoint=False)
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xs, xs_parent, init_parent)
        # project onto feasible set
        h = _project_box_simplex(h, target_sum)
    except Exception:
        # fallback: random binary pattern (exactly half ones)
        h = np.zeros(n, dtype=float)
        h[:n // 2] = 1.0
        np.random.shuffle(h)

    best_h = h.copy()
    # evaluate via the provided helper (fallback to our own if missing)
    def _eval(vec):
        # the harness supplies evaluate_erdos_solution; use it if available
        try:
            return evaluate_erdos_solution(vec)
        except Exception:
            corr = np.correlate(vec, 1.0 - vec, mode="full")
            return corr.max() * dx

    best_val = _eval(best_h)

    # ----- optimisation loop -----
    iteration = 0
    step_initial = 0.5
    decay = 0.9997
    top_k = 5                     # examine a few worst shifts for stochasticity
    while time.time() - start_time < budget_s * 0.95:
        # correlation and current objective
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_val = corr.max() * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            # early exit if we already beat the target record
            if best_val < 0.3808:
                break

        # choose a shift among the top‑k worst ones
        k = min(top_k, corr.size)
        idxs = np.argpartition(-corr, k - 1)[:k]
        idx = np.random.choice(idxs)
        shift = idx - (n - 1)          # shift in index units (can be negative)

        # sub‑gradient of the worst shift term
        grad = np.zeros_like(h)
        i_start = max(0, -shift)
        i_end = min(n, n - shift)
        if i_end > i_start:
            left = np.arange(i_start, i_end)
            right = left + shift
            grad[left] += (1.0 - h[right])
            grad[right] -= h[left]

        # step size decays with the number of iterations
        step = step_initial * (decay ** iteration)
        y = h - step * grad
        h = _project_box_simplex(y, target_sum)

        # occasional random jitter to escape shallow plateaus
        if iteration % 500 == 0 and iteration > 0:
            jitter = (np.random.rand(n) - 0.5) * 0.02
            h = _project_box_simplex(h + jitter, target_sum)

        iteration += 1

    # final evaluation of the best vector found
    final_val = _eval(best_h)

    return (best_h, float(final_val), n)
