import numpy as np
import time
import math
import random

def project_to_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n while enforcing sum(v)=target."""
    # binary search for λ such that sum(clip(v+λ,0,1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction for numerical drift
    err = target - y.sum()
    if abs(err) > 1e-12:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += err / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a step function represented by h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def gradient_max_shift(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k ∫h(x)(1‑h(x+k))dx w.r.t. the discrete vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
    shifts = idxs - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[: n - k] += (1.0 - h[k:])
            grad[k:] -= h[: n - k]
        else:
            l = -k
            grad[l:] += (1.0 - h[: n - l])
            grad[: n - l] -= h[l:]
    if len(shifts):
        grad /= len(shifts)
    return grad

    """Generate a random vector in [0,1]^n with prescribed sum."""
    v = np.random.rand(n)
    return project_to_bounds_sum(v, target)
