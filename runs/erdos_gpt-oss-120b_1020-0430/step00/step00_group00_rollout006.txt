    T = T0

    iter_count = 0

    target = 0.3808

def compute_c5_fft(h):
    N = h.shape[0]
    g = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)
    dx = 2.0 / N
    return np.max(conv) * dx

import time
import math
import numpy as np

def _compute_c5_fft(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1−h(x+k)) using FFT (linear correlation)."""
    n = h.shape[0]
    g = 1.0 - h
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g[::-1], n=L)          # reverse for correlation
    conv = np.fft.irfft(H * G, n=L)
    dx = 2.0 / n
    return float(conv.max() * dx)


def _compute_c5_and_worst_lag(h: np.ndarray):
    """Return (max value, lag achieving it, full correlation array)."""
    n = h.shape[0]
    g = 1.0 - h
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)
    idx = int(conv.argmax())
    lag = idx - (n - 1)                     # lag ∈ [-(n-1), n-1]
    dx = 2.0 / n
    return float(conv[idx] * dx), lag, conv


def _random_binary_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a {0,1}-valued vector with exactly n/2 ones (integral =1)."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def _replicate(arr: np.ndarray, factor: int) -> np.ndarray:
    """Upsample by integer replication (preserves integral exactly)."""
    return np.repeat(arr, factor)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimises C₅."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---------- choose resolution ----------
    target_n = 360                      # desired number of points (fast enough)
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base_n = base.shape[0]
        factor = max(1, target_n // base_n)
        n = base_n * factor
        h_current = _replicate(base, factor)
    else:
        n = target_n
        h_current = _random_binary_start(n, rng)

    # Ensure the integral constraint holds exactly (it does for the above constructions)
    dx = 2.0 / n
    # tiny numerical fix if needed
    total = h_current.sum()
    desired = n / 2.0
    if not np.isclose(total, desired):
        h_current *= desired / total
        np.clip(h_current, 0.0, 1.0, out=h_current)

    # Best solution tracking
    best_h = h_current.copy()
    best_c5 = _compute_c5_fft(best_h)

    # Annealing parameters
    T0 = 5e-3
    T = T0
    cooling = 0.9999
    target = 0.3808                      # goal to beat
    iteration = 0

    # Main optimisation loop
    while time.time() - start_time < budget_s * 0.97:
        # compute current worst lag and its correlation value
        cur_c5, worst_k, _ = _compute_c5_and_worst_lag(h_current)
        # early exit if already good enough
        if best_c5 <= target:
            break

        # ----- decide move type -----
        if rng.random() < 0.1:
            # pure random move
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
        else:
            # targeted move based on the lag giving the current maximum
            if worst_k >= 0:
                idxs = np.arange(n - worst_k)
                prod = h_current[idxs] * (1.0 - h_current[idxs + worst_k])
            else:
                offset = -worst_k
                idxs = np.arange(offset, n)
                prod = h_current[idxs] * (1.0 - h_current[idxs + worst_k])
            i = idxs[int(np.argmax(prod))]
            # pick a distinct partner at random
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

        # feasible signed transfer amount (move mass from i to j)
        # we try decreasing h[i] (most common) ; if impossible try opposite direction
        eps_max = min(h_current[i], 1.0 - h_current[j])
        direction = -1  # -1 : i -> j (decrease i)
        if eps_max <= 0.0:
            # try opposite direction
            eps_max = min(h_current[j], 1.0 - h_current[i])
            direction = +1  # j -> i (increase i)
            if eps_max <= 0.0:
                continue

        # sample a positive epsilon within feasible range
        epsilon = rng.random() * eps_max
        h_new = h_current.copy()
        if direction == -1:
            h_new[i] -= epsilon
            h_new[j] += epsilon
        else:
            h_new[i] += epsilon
            h_new[j] -= epsilon

        # evaluate new state
        new_c5 = _compute_c5_fft(h_new)
        dE = new_c5 - cur_c5

        # Metropolis acceptance
        if dE <= 0.0 or rng.random() < math.exp(-dE / T):
            h_current = h_new
            cur_c5 = new_c5
            # update best if improved
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

        # temperature schedule

        iteration += 1

        # occasional random restart of the *current* state (keep best)
        if iteration % 30000 == 0:
            # keep best, but wander from a fresh start
            h_current = _random_binary_start(n, rng) if rng.random() < 0.5 else best_h.copy()
            T = T0

    final_c5 = _compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c5), n
