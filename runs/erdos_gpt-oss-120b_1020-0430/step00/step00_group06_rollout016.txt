    def anneal(start_vec):

        h = start_vec.copy()

            if len(ones) == 0 or len(zeros) == 0:

import time
import random
import numpy as np

def _c5_value(vec):
    """Compute C5 for a binary (or [0,1]) vector."""
    n = vec.shape[0]
    max_corr = np.max(np.correlate(vec, 1.0 - vec, mode="full"))
    return max_corr * (2.0 / n)


def _improve_once(h):
    """One improvement step: try a targeted swap that reduces the worst shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    worst_idx = int(np.argmax(corr))
    s = worst_idx - (n - 1)                # shift giving the current maximum
    # domain of indices that actually participate in this shift
    if s >= 0:
        region = np.arange(0, n - s)
    else:
        region = np.arange(-s, n)
    shifted = region + s

    # positions where we have a 1 that matches a 0 after the shift
    bad = region[(h[region] == 1) & (h[shifted] == 0)]
    # positions where we have a 0 that matches a 1 after the shift
    good = region[(h[region] == 0) & (h[shifted] == 1)]

    if bad.size and good.size:
        i = random.choice(bad)
        j = random.choice(good)
        h[i], h[j] = h[j], h[i]            # perform the swap
        # keep the swap only if it improves the objective
        new_c5 = _c5_value(h)
        old_c5 = (corr.max()) * (2.0 / n)
        if new_c5 < old_c5:
            return True, new_c5
        # revert
        h[i], h[j] = h[j], h[i]

    # fallback: a random feasible swap
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if ones.size == 0 or zeros.size == 0:
        return False, None
    i = random.choice(ones)
    j = random.choice(zeros)
    h[i], h[j] = h[j], h[i]
    new_c5 = _c5_value(h)
    old_c5 = (corr.max()) * (2.0 / n)
    if new_c5 < old_c5:
        return True, new_c5
    h[i], h[j] = h[j], h[i]                # revert
    return False, None


def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function on [0,2] with integral 1 that minimises
    max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:
        init = initial_h_values          # supplied by the harness
        n = int(init.shape[0])
        # Convert to a binary feasible vector (the problem allows any [0,1] values,
        # but binary vectors are easier to manipulate while still respecting the constraints)
        h = (init > 0.5).astype(float)
        target_ones = n // 2
        cur = int(h.sum())
        if cur > target_ones:            # too many ones – flip random excess to zero
            idx = np.where(h == 1)[0]
            to_zero = np.random.choice(idx, size=cur - target_ones, replace=False)
            h[to_zero] = 0.0
        elif cur < target_ones:          # too few – turn random zeros into ones
            idx = np.where(h == 0)[0]
            to_one = np.random.choice(idx, size=target_ones - cur, replace=False)
            h[to_one] = 1.0
    except NameError:
        # No supplied construction – start from scratch.
        n = 200                     # a comfortable size (<1000) giving fine resolution
        if n % 2:
            n += 1
        target_ones = n // 2
        h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, target_ones, replace=False)
        h[ones] = 1.0

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop (time‑bounded)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # leave a safety margin

    best_h = h.copy()
    best_c5 = _c5_value(best_h)
    target_c5 = 0.38085

    # quick exit if the starter already meets the target
    if best_c5 <= target_c5:
        return best_h, best_c5, n

    # parameters for simple restart strategy
    no_improve_limit = 5000
    no_improve_cnt = 0

    while time.time() < deadline:
        improved, new_val = _improve_once(h)
        if improved:
            no_improve_cnt = 0
            if new_val < best_c5:
                best_c5 = new_val
                best_h = h.copy()
                if best_c5 <= target_c5:
                    break
        else:
            no_improve_cnt += 1

        # stagnation handling – random perturbation around the best found so far
        if no_improve_cnt >= no_improve_limit:
            # keep the current best and shuffle a few bits to escape a local trap
            h = best_h.copy()
            # perform a handful of random pairwise swaps (still respecting the weight)
            for _ in range(10):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i = random.choice(ones)
                j = random.choice(zeros)
                h[i], h[j] = h[j], h[i]
            no_improve_cnt = 0

    # ------------------------------------------------------------------
    # 3.  Return the best solution discovered
    # ------------------------------------------------------------------
    final_c5 = _c5_value(best_h)
    return best_h, final_c5, n
