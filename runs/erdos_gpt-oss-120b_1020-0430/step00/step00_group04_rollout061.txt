def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n_points
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        # Ensure shape
        # We'll create binary rounded version
        h = np.where(init > 0.5, 1.0, 0.0)
    else:
        n = 100
        # create random binary vector with half ones
        h = np.zeros(n, dtype=float)
        h[:n//2] = 1.0
        rng.shuffle(h)
    # Ensure sum constraint exactly n/2
    target = n // 2
    current_sum = int(h.sum())
    # Fix if not correct
    if current_sum != target:
        # adjust
        # convert to list indices
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        while current_sum > target:
            # turn a random 1 into 0
            idx = rng.choice(ones)
            h[idx] = 0.0
            ones = np.delete(ones, np.where(ones == idx))
            zeros = np.append(zeros, idx)
            current_sum -= 1
        while current_sum < target:
            idx = rng.choice(zeros)
            h[idx] = 1.0
            zeros = np.delete(zeros, np.where(zeros == idx))
            ones = np.append(ones, idx)
            current_sum += 1
    # Compute initial value
    best_h = h.copy()
    best_val = compute_c5(best_h)
    # Local search
    # Precompute 1-h for speed?
    # We'll use loop with time limited
    max_iter = 10000000  # large but break on time
    # Pre compute index lists for ones and zeros
    ones_idx = np.flatnonzero(best_h == 1)
    zeros_idx = np.flatnonzero(best_h == 0)
    no_improve = 0
    # Simple simulated annealing schedule
    # We'll define temperature that decays slowly
    while time.time() - start < budget_s * 0.95:
        # propose swap
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # apply swap to candidate
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = compute_c5(cand)
        # accept if better
        if val < best_val - 1e-12:
            best_h = cand
            best_val = val
            # update lists
            # swap i and j in indices
            # remove i from ones, add to zeros; remove j from zeros, add to ones
            ones_idx = ones_idx[ones_idx != i]
            zeros_idx = zeros_idx[zeros_idx != j]
            ones_idx = np.append(ones_idx, j)
            zeros_idx = np.append(zeros_idx, i)
            no_improve = 0
        else:
            no_improve += 1

            cand = np.zeros(n)
            cand[:target] = 1.0
            rng.shuffle(cand)
            val = compute_c5(cand)
            if val < best_val:
                best_h = cand
                best_val = val
                ones_idx = np.flatnonzero(best_h == 1)
                zeros_idx = np.flatnonzero(best_h == 0)
            no_improve = 0
    # return list, bound, n_points
    return best_h.tolist(), float(best_val), n

    """Clip to [0,1] and adjust linearly so that sum = target."""
    v = np.clip(vec, 0.0, 1.0)

        diff = target - v.sum()
        if abs(diff) < 1e-12:

        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)

        c5_bound = float(evaluate_erdos_solution(best_h))
    else:
        c5_bound = float(_objective(best_h))

    return best_h.tolist(), c5_bound, n
