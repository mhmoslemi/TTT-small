    import time, numpy as np
    try:
        from scipy.optimize import minimize
        has_scipy_opt = True
    except Exception:
        has_scipy_opt = False
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine n_points
    # Prefer to use n=200 to provide enough resolution
    n_desired = 200
    # If initial_h_values exists with length > n_desired, use it? But we can upscale.
    if 'initial_h_values' in globals():
        init_len = len(initial_h_values)
        if init_len > n_desired:
            n_desired = init_len
    # Ensure n is even
    if n_desired % 2 != 0:
        n_desired += 1
    n = n_desired
    target_sum = n / 2.0
    dx = 2.0 / n

    def evaluate(h):
        # compute max overlap
        # use np.correlate
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    # Prepare starting vectors
    starts = []

    alt = np.zeros(n, dtype=np.float64)
    alt[::2] = 1.0

            alt[:diff] = 1.0
        elif diff < 0:
            alt[: -diff] = 0.0
    starts.append(alt.copy())
    # if initial_h exists, upscale to n via interpolation

        init = np.array(initial_h_values, dtype=np.float64)
        x_old = np.linspace(0, 2, init.size, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
