def project_to_bounds_sum(x, target, lower=0., upper=1., max_iter=50):
    lo = -np.max(x) - 10  # ensure all x+lo <= lower
    hi = upper - np.min(x) + 10 # ensure all x+hi >= upper
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, lower, upper)
        s = y.sum()
        if abs(s - target) < 1e-9:
            return y
        if s > target:
            # need to reduce sum: decrease lambda
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo+hi)/2.0, lower, upper)

    n = len(h)
    grad = np.zeros(n)
    for l in max_lags:
        if l >= 0:
            right = np.concatenate((np.zeros(l), h[:-l]))
            left = np.concatenate((h[l:], np.zeros(l)))

    best_h = h.copy()
    best_cost = eval_h(best_h)
    n = len(h)
    target_sum = n/2
    step = 0.1 # initial step size
    decay = 0.999
    min_step = 1e-5
    # iterate until time or iter limit
    for it in range(max_iter):
        # check time
        if time.time() - start > remaining_time:
            break
        # compute corr and max lag(s)
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        max_idxs = np.where(corr == max_val)[0]  # could be many
        # compute lags
        lags = max_idxs - (n - 1)
        # compute gradient sum
        grad = np.zeros(n)
        for l in lags:
            # compute left and right shift arrays
            if l >= 0:
                # left shift (h[t + l])

    # Use bisection.
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Adjust small numeric residual by distributing evenly among non-saturated entries.
    # Optionally enforce exact sum.
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        # add diff equally to all entries not at bound
        free = (y > 0) & (y < 1)
        if free.any():
            y[free] += diff / free.sum()
            # Clip again
            y = np.clip(y,0,1)
    return y

def gradient_descent(h, max_iter, time_limit, initial_step=0.5, decay=0.999, min_step=1e-6):
    n = len(h)
    target_sum = n / 2
    step = initial_step
    best_h = h.copy()
    best_cost = compute_c5(best_h)
    cur_h = h.copy()
    cur_cost = best_cost
    start = time.time()
    for it in range(max_iter):
        now = time.time()
        if now - start > time_limit:
            break
        # compute correlation (max idx)
        n = len(cur_h)
        L = 1 << ((2*n-1).bit_length())
        H = np.fft.fft(cur_h, n=L)
        B = np.fft.fft(1 - cur_h, n=L)
        corr = np.fft.ifft(H * np.conj(B)).real[:2*n-1]
        dx = 2.0/n
        corr_scaled = corr * dx
        max_idx = int(np.argmax(corr_scaled))
        lag = max_idx - (n - 1)
        # compute gradient
        if lag >= 0:
            left = np.concatenate((cur_h[lag:], np.zeros(lag, dtype=cur_h.dtype)))
            right = np.concatenate((np.zeros(lag, dtype=cur_h.dtype), cur_h[:n-lag]))
        else:
            ll = -lag
            left = np.concatenate((np.zeros(ll, dtype=cur_h.dtype), cur_h[:n-ll]))
            right = np.concatenate((cur_h[ll:], np.zeros(ll, dtype=cur_h.dtype)))
        grad = 1.0 - left - right
        # gradient step
        h_new = cur_h - step * grad
        # projection
        h_new = project_to_bounds_sum(h_new, target_sum)
        # compute new cost
        H_new = np.fft.fft(h_new, n=L)
        B_new = np.fft.fft(1 - h_new, n=L)
        corr_new = np.fft.ifft(H_new * np.conj(B_new)).real[:2*n-1] * dx
        new_cost = corr_new.max()
        # update best
        if new_cost < best_cost - 1e-12:
            best_cost = new_cost
            best_h = h_new.copy()
        # accept step regardless (could implement accept if new_cost < cur_cost else maybe not)
        cur_h = h_new
        cur_cost = new_cost
        # reduce step
        step = max(step * decay, min_step)
    return best_h, best_cost

import time
import numpy as np

# ------- Helper Functions --------------------------------------------------- #
def project_to_bounds_sum(x, target, lower=0.0, upper=1.0, max_iter=60):
    """Project vector x onto the box [lower,upper] while enforcing sum(x)=target."""
    # Find lambda such that sum(clip(x+lambda, lower, upper)) = target.
    lo = -np.max(x) - 10.0
    hi = upper - np.min(x) + 10.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, lower, upper)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) * 0.5, lower, upper)
    # tiny correction for numerical drift
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > lower) & (y < upper)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, lower, upper)
        else:
            # all at bounds – adjust the first element safely
            idx = np.argmax(y)
            y[idx] = np.clip(y[idx] + diff, lower, upper)
    return y


def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for a discrete step function."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # power of two ≥ 2n-1
    H = np.fft.fft(h, n=L)
    B = np.fft.fft(1.0 - h, n=L)
    corr = np.fft.ifft(H * np.conj(B)).real[: 2 * n - 1]
    dx = 2.0 / n
    return (corr * dx).max()


def max_lag_index(h):
    """Return the lag (in index offset) attaining the maximum correlation."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, n=L)
    B = np.fft.fft(1.0 - h, n=L)
    corr = np.fft.ifft(H * np.conj(B)).real[: 2 * n - 1]
    dx = 2.0 / n
    corr_scaled = corr * dx
    idx = int(np.argmax(corr_scaled))
    lag = idx - (n - 1)
    return lag


def gradient_step(h, lag):
    """Compute the subgradient of the max‑correlation w.r.t. h for a given lag."""
    n = h.shape[0]
    if lag >= 0:
        left = np.concatenate((h[lag:], np.zeros(lag, dtype=h.dtype)))
        right = np.concatenate((np.zeros(lag, dtype=h.dtype), h[: n - lag]))
    else:
        ll = -lag
        left = np.concatenate((np.zeros(ll, dtype=h.dtype), h[: n - ll]))
        right = np.concatenate((h[ll:], np.zeros(ll, dtype=h.dtype)))
    return 1.0 - left - right  # derivative of ∑ h_i(1‑h_{i+lag})


def gradient_descent(
    h0,
    max_iter,
    time_limit,
    step0=0.5,
    decay=0.999,
    min_step=1e-6,
):
    """Simple projected sub‑gradient descent for the C5 problem."""
    n = h0.shape[0]
    target = n / 2.0
    h = project_to_bounds_sum(h0, target)
    best_h = h.copy()
    best_val = compute_c5(best_h)

    step = step0
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        lag = max_lag_index(h)
        grad = gradient_step(h, lag)

        h_new = h - step * grad
        h_new = project_to_bounds_sum(h_new, target)

        val_new = compute_c5(h_new)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()

        h = h_new
        step = max(step * decay, min_step)

    return best_h, best_val


def upscale_h(h, factor):
    """Upscale a step vector by integer factor using repetition."""
    if factor == 1:
        return h.copy()
    return np.repeat(h, factor)


# ------- Main Optimization Routine ------------------------------------------ #
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ----- initial vector -------------------------------------------------- #
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    if init_h is None or init_h.size == 0:
        n0 = 45
        init_h = np.full(n0, 0.5, dtype=float)
    else:
        n0 = init_h.shape[0]

    # Ensure feasibility of the starting point
    target0 = n0 / 2.0
    init_h = project_to_bounds_sum(init_h, target0)

    best_h = init_h
    best_val = compute_c5(best_h)

    # ----- resolution schedule --------------------------------------------- #
    # Factors are integers so we can repeat the vector without interpolation.
    factors = [1, 2, 4, 8]          # 45 → 90 → 180 → 360 points
    max_iters = {45: 4000, 90: 6000, 180: 8000, 360: 12000}
    restarts = 3                    # random restarts per level (except level 1)

    for factor in factors:
        # time allocation check
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= 2.0:
            break

        n = n0 * factor
        target = n / 2.0

        # upscale current best solution
        h_cur = upscale_h(best_h, factor)

        # baseline descent on the upscaled vector
        allowed = remaining * 0.7
        iters = max_iters.get(n, 5000)
        h_opt, val_opt = gradient_descent(
            h_cur,
            max_iter=iters,
            time_limit=allowed,
            step0=0.5,
            decay=0.998,
            min_step=1e-5,
        )
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()

        # a few random restarts for diversity
        for _ in range(restarts):
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            if remaining <= 1.0:
                break
            # Random feasible start
            rand = np.random.rand(n)
            rand = project_to_bounds_sum(rand, target)
            # Shorter run for a random start
            allowed = remaining * 0.3
            iters_r = max(2000, iters // 3)
            h_r, v_r = gradient_descent(
                rand,
                max_iter=iters_r,
                time_limit=allowed,
                step0=0.5,
                decay=0.998,
                min_step=1e-5,
            )
            if v_r < best_val:
                best_val, best_h = v_r, h_r.copy()

    # Final evaluation (just to be sure)
    final_c5 = compute_c5(best_h)

    # Return as plain list for compatibility
    return (best_h.tolist(), float(final_c5), best_h.shape[0])
