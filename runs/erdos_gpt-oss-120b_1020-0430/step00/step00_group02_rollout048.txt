import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.convolve(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def _make_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random binary vector with exactly n/2 ones (n assumed even)."""
    k = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:k] = 1.0
    rng.shuffle(v)
    return v


def _upsample(h_coarse: np.ndarray, n_fine: int, rng: np.random.Generator) -> np.ndarray:
    """Linear interpolation from a coarse grid to a finer grid, then project to the feasible set."""
    n_coarse = h_coarse.size
    x_coarse = np.linspace(0.0, 2.0, n_coarse, endpoint=False)
    x_fine = np.linspace(0.0, 2.0, n_fine, endpoint=False)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine = np.clip(h_fine, 0.0, 1.0)

    # enforce the exact integral constraint
    target = n_fine / 2.0
    cur = h_fine.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        h_fine += diff / n_fine
        h_fine = np.clip(h_fine, 0.0, 1.0)
        # If clipping changed the sum, correct the remaining discrepancy by pairwise transfers.
        cur = h_fine.sum()
        diff = target - cur
        while abs(diff) > 1e-12:
            if diff > 0:
                cand = np.where(h_fine < 1.0)[0]
                if cand.size == 0:
                    break
                i = rng.choice(cand)
                inc = min(diff, 1.0 - h_fine[i])
                h_fine[i] += inc
                diff -= inc
            else:
                cand = np.where(h_fine > 0.0)[0]
                if cand.size == 0:
                    break
                i = rng.choice(cand)
                dec = min(-diff, h_fine[i])
                h_fine[i] -= dec
                diff += dec
    return h_fine


def _hill_climb(start_h: np.ndarray,
                start_temp: float,
                end_temp: float,
                max_iters: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style local search on the feasible polytope."""
    h = start_h.copy()
    n = h.size
    cur_c5 = _c5_value(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # linear temperature schedule
        t = start_temp + (end_temp - start_temp) * (it / max_iters)

        # pick two distinct indices
        i, j = rng.integers(n, size=2)
        while i == j:
            j = rng.integers(n)

        # decide direction of transfer
        if rng.random() < 0.5:
            max_inc = 1.0 - h[i]
            max_dec = h[j]
            if max_inc <= 0.0 or max_dec <= 0.0:
                continue
            delta = max_inc if max_inc < max_dec else max_dec
            delta *= rng.random() * t
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            max_inc = 1.0 - h[j]
            max_dec = h[i]
            if max_inc <= 0.0 or max_dec <= 0.0:
                continue
            delta = max_inc if max_inc < max_dec else max_dec
            delta *= rng.random() * t
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_c5 = _c5_value(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / (t + 1e-12)):
            h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

    return best_h, best_c5
