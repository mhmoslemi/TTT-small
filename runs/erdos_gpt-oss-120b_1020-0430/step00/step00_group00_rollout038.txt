def project_onto_bounds_and_sum(v, target_sum):
    lo = -v.max() - 1.0
    hi = 1.0 - v.min() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # due to rounding, adjust slight difference:
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        # add diff to an element that is not at bound
        for i in range(len(x)):
            if diff > 0 and x[i] < 1 - 1e-12:
                delta = min(diff, 1 - x[i])
                x[i] += delta
                diff -= delta
            elif diff < 0 and x[i] > 1e-12:
                delta = min(-diff, x[i])
                x[i] -= delta
                diff += delta
            if abs(diff) < 1e-12:
                break
    return x

def hill_climb(h0, max_iters=5000, init_step=0.5, decay=0.99, tol=1e-6, budget_end=None):
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    best_val = evaluate_c5(h)
    best_h = h.copy()
    step = init_step
    for it in range(max_iters):
        # time check
        if budget_end is not None and time.time() > budget_end:
            break
        _, lag, _ = max_corr_and_lag(h)
        grad = compute_gradient(h, lag)
        improved = False
        for factor in (1.0, 0.5, 0.25):
            α = step * factor
            cand = project_onto_bounds_and_sum(h - α * grad, target_sum)
            val = evaluate_c5(cand)
            if val < best_val - 1e-12:
                h = cand
                best_h = cand
                best_val = val
                improved = True
                break
        if not improved:
            step *= decay
            if step < tol:
                break
    return best_h, best_val

import time
import numpy as np
from scipy.optimize import differential_evolution

# ---------- helper functions (top level) ---------------------------------

def project_onto_bounds_and_sum(v, target):
    """Euclidean projection of v onto the set {0<=x<=1, sum x = target}."""
    lo = -v.max() - 1.0
    hi = 1.0 - v.min() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for i in range(len(x)):
            if diff > 0 and x[i] < 1.0 - 1e-12:
                d = min(diff, 1.0 - x[i])
                x[i] += d
                diff -= d
            elif diff < 0 and x[i] > 1e-12:
                d = min(-diff, x[i])
                x[i] -= d
                diff += d
            if abs(diff) < 1e-12:
                break
    return x


def evaluate_c5(h):
    """Compute the Erdős C5 value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_gradient(h, lag):
    """Gradient of the worst correlation (c = sum h - sum h_i h_{i+lag})."""
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if lag >= 0:
        l = lag
        if l < n:
            g[: n - l] -= h[l:]
            g[l:] -= h[: n - l]
    else:
        l = -lag
        if l < n:
            g[l:] -= h[: n - l]
            g[: n - l] -= h[l:]
    return g


def max_corr_and_lag(h):
    """Return (max correlation value, lag achieving it, full correlation)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (len(h) - 1)
    return corr[idx], lag, corr


def hill_climb(h0, max_iters=5000, step0=0.5, decay=0.99, tol=1e-6,
               time_limit=None):
    """Simple projected gradient descent on the max‑correlation objective."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_val = evaluate_c5(h)
    best_h = h.copy()
    step = step0
    for _ in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        _, lag, _ = max_corr_and_lag(h)
        grad = compute_gradient(h, lag)
        improved = False
        for fac in (1.0, 0.5, 0.25):
            cand = project_onto_bounds_and_sum(h - step * fac * grad, target)
            val = evaluate_c5(cand)
            if val < best_val - 1e-12:
                h = cand
                best_h = cand
                best_val = val
                improved = True
                break
        if not improved:
            step *= decay
            if step < tol:
                break
    return best_h, best_val


def random_feasible(n, rng):
    """Create a random feasible vector (sum = n/2, entries in [0,1])."""
    v = rng.random(n)
    return project_onto_bounds_and_sum(v, n / 2.0)


def intervals_to_h(starts, n):
    """Build h from 5 unit‑length intervals with given left ends."""
    dx = 2.0 / n
    x = np.arange(n) * dx
    h = np.zeros(n, dtype=np.float64)

        h[mask] += 1.0 / 5.0
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_bounds_and_sum(h, n / 2.0)
    return h
