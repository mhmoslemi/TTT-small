    # Temperature schedule parameters:
    T0 = 0.1  # initial temperature
    while elapsed() < time_limit:

        t_frac = elapsed() / time_limit

        accept = False

        elif T > 0:
            # accept with probability exp(-delta / T)
            if np.random.rand() < np.exp(-delta / T):
                accept = True
        if accept:

                    break

def SA_optimize(N, time_limit, init_h=None, target=0.3808):
    # If init_h length doesn't match N, ignore
    if init_h is None or len(init_h) != N:
        # create random balanced vector
        h = np.zeros(N, dtype=np.float64)
        h[:N//2] = 1.0
        np.random.shuffle(h)
    else:
        # ensure balanced: sum may be half; but if not, adjust randomly
        h = init_h.astype(np.float64).copy()
        cur_sum = int(h.sum())
        needed = N//2 - cur_sum
        if needed >0:
            zeros_idx = np.where(h==0)[0]
            if len(zeros_idx) >= needed:
                chosen = np.random.choice(zeros_idx, needed, replace=False)
                h[chosen] = 1.0
        elif needed <0:
            ones_idx = np.where(h==1)[0]
            chosen = np.random.choice(ones_idx, -needed, replace=False)
            h[chosen] = 0.0

    dx = 2.0 / N
    # compute current bound
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    cur_bound = max_corr * dx
    best_h = h.copy()
    best_bound = cur_bound
    best_max_corr = max_corr

    start = time.time()
    # while time left
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        cur_bound = max_corr * dx

        # update best
        if cur_bound < best_bound:
            best_bound = cur_bound
            best_h = h.copy()
            best_max_corr = max_corr
            if best_bound <= target:
                break

        # get shift with max
        idx_max = int(np.argmax(corr))
        shift = idx_max - (N - 1)

        # Build candidate lists
        contribute = []
        safe_zero = []
        # compute contributions list
        for i in range(N):
            if h[i] == 1:
                idx = i - shift
                if 0 <= idx < N and h[idx] == 0:

def _targeted_swap_step(h, N, dx):
    # compute correlation and current max count
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()

    # construct candidate indices lists
    contribute = []
    safe_zero = []

    # compute contributions
    for i in range(N):
        if h[i] == 1:
            idx = i - shift
            if 0 <= idx < N and h[idx] == 0:
                contribute.append(i)
    for j in range(N):
        if h[j] == 0:
            idx = j - shift
            if not (0 <= idx < N) or h[idx] == 1:

        j = np.random.choice(safe_zero)

        return new_max_corr

import time

def _bound_and_max(h):
    """Return (C5 bound, integer max correlation) for binary vector h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = int(corr.max())
    return max_corr * dx, max_corr


def _optimize_N(N, time_limit, init_h, target):
    """
    Heuristic search for a balanced 0/1 vector of length N.
    Returns (best_vector, best_bound).
    """
    # ----- initialise vector -------------------------------------------------
    if init_h is not None and init_h.shape[0] == N:
        h = init_h.astype(np.float64).copy()
        cur = int(h.sum())
        need = N // 2 - cur
        if need > 0:
            zeros = np.where(h == 0)[0]
            h[np.random.choice(zeros, need, replace=False)] = 1.0
        elif need < 0:
            ones = np.where(h == 1)[0]
            h[np.random.choice(ones, -need, replace=False)] = 0.0
    else:
        h = np.zeros(N, dtype=np.float64)
        h[: N // 2] = 1.0
        np.random.shuffle(h)

    dx = 2.0 / N
    best_h = h.copy()
    best_bound, best_max = _bound_and_max(h)

    start = time.time()
    T0 = 2.0                     # initial temperature for simulated annealing

    while time.time() - start < time_limit:
        # current correlation and shift attaining the max
        corr_full = np.correlate(h, 1 - h, mode="full")
        cur_max = int(corr_full.max())
        idx_max = int(np.argmax(corr_full))
        shift = idx_max - (N - 1)

        # update best record
        cur_bound = cur_max * dx
        if cur_bound < best_bound - 1e-12:
            best_bound = cur_bound
            best_h = h.copy()
            best_max = cur_max
            if best_bound <= target:
                break

        # ----- choose a swap -------------------------------------------------
        # indices that contribute to the current worst shift
        contribute = []
        safe_zero = []

        for i in range(N):
            if h[i] == 1:
                j = i - shift
                if 0 <= j < N and h[j] == 0:
                    contribute.append(i)

        for j in range(N):
            if h[j] == 0:
                i = j - shift
                if not (0 <= i < N) or h[i] == 1:
                    safe_zero.append(j)

        if contribute and safe_zero:

            j = np.random.choice(safe_zero)

    # final bound recomputation
    final_c = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * (2.0 / best_N)

    return best_h, float(final_c), int(best_N)
```
