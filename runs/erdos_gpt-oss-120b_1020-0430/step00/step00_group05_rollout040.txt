def compute_fmax(h):
    n = len(h)
    total = n / 2.0   # sum of h values (since sum = n/2)
    corr = np.correlate(h, h, mode='full')
    # exclude zero shift index n-1
    # Note: h is floats; use float64
    # find min correlation for k !=0:
    left_min = corr[:n-1].min()
    right_min = corr[n:].min()
    min_corr = left_min if left_min < right_min else right_min
    fmax = total - min_corr
    return fmax

import time
import random
import math
import numpy as np

def _dx(n):
    return 2.0 / n

def _c5_value(h):
    """return C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * _dx(n))

def _prepare_initial(h):
    """Project arbitrary float vector onto the feasible set."""
    n = h.shape[0]
    target = n / 2.0
    # clip first
    h = np.clip(h, 0.0, 1.0)
    # fix sum by uniform adjustment, repeat a few times
    for _ in range(5):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    cur = h.sum()
    if abs(cur - target) > 1e-9:
        # distribute the remaining error to the most interior entries
        idx = np.argsort(h)  # smallest first
        for i in idx:
            add = min(1.0 - h[i], target - cur) if cur < target else min(h[i], cur - target)
            if add == 0:
                continue
            h[i] += add if cur < target else -add
            cur = h.sum()
            if abs(cur - target) < 1e-12:
                break
    return h

def _make_shape(shape, n):
    """Create a feasible shape (array) of length n."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    if shape == "linear":
        h = x / 2.0                     # ramps from 0 to 1
    elif shape == "triangle":
        h = 1.0 - np.abs(x - 1.0)       # peak 1 at centre, 0 at ends
    elif shape == "cosine":
        h = (1.0 + np.cos(np.pi * x)) / 2.0   # 1 at ends, 0 in middle
    else:                               # random uniform in [0,1]
        h = np.random.rand(n)
    return _prepare_initial(h)

def _target_shift(corr, n):
    """Return the shift (integer) where the overlap is maximal."""
    idx = int(np.argmax(corr))
    return idx - (n - 1)   # shift value (positive, negative or 0)

def _local_search(start_h, time_limit):
    """Improve a single start vector under the given time budget."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_val = _c5_value(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    eps = 1e-12
    step = 0.2               # initial transfer size
    it = 0
    start_time = time.time()
    while time.time() - start_time < time_limit:
        # compute worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        shift = _target_shift(corr, n)
        off = abs(shift)

        new_h = cur_h.copy()
        improved = False

        if off > 0:
            left = cur_h[: n - off]
            right = cur_h[off:]
            if shift > 0:
                # contribution = left * (1-right), reduce by moving from left to right
                cand = np.where((left > eps) & (right < 1.0 - eps))[0]
                if cand.size:
                    i = random.choice(cand.tolist())
                    delta = min(step, left[i], 1.0 - right[i])
                    if delta > 0:
                        new_h[i] -= delta
                        new_h[i + off] += delta
                        improved = True
            else:  # shift < 0
                # contribution = right * (1-left), reduce by moving from right to left
                cand = np.where((right > eps) & (left < 1.0 - eps))[0]
                if cand.size:
                    i = random.choice(cand.tolist())
                    delta = min(step, right[i], 1.0 - left[i])
                    if delta > 0:
                        new_h[i + off] -= delta
                        new_h[i] += delta
                        improved = True

        if not improved:
            # random transfer preserving sum
            i, j = random.sample(range(n), 2)
            if cur_h[i] < 1.0 - eps and cur_h[j] > eps:
                delta = min(step, 1.0 - cur_h[i], cur_h[j])
                new_h[i] += delta
                new_h[j] -= delta
                improved = True
            elif cur_h[j] < 1.0 - eps and cur_h[i] > eps:
                delta = min(step, 1.0 - cur_h[j], cur_h[i])
                new_h[j] += delta
                new_h[i] -= delta
                improved = True

        if not improved:
            # nothing feasible – break out
            break

        new_val = _c5_value(new_h)

        if new_val + 1e-12 < cur_val:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        # schedule step size decay
        it += 1
        if it % 1000 == 0:
            step = max(step * 0.9, 0.001)

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős minimum‑overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # Choose a working resolution
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        n = base.shape[0]
        starts = [_prepare_initial(base.copy())]
    else:
        n = 200                     # default size when no parent is supplied
        starts = []

    # add a few handcrafted starts (all have exact integral 1)
    for name in ("linear", "triangle", "cosine"):
        starts.append(_make_shape(name, n))
    # a few random starts
    for _ in range(2):
        starts.append(_make_shape("random", n))

    # time split among starts (roughly equal)
    total_time = budget_s * 0.97   # keep a little margin
    per_start = total_time / len(starts)

    best_h = None
    best_val = math.inf

    for s in starts:
        h_opt, val_opt = _local_search(s, per_start)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt

    # final feasibility enforcement (tiny adjustments)
    target = n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        best_h += (target - cur) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    c5 = _c5_value(best_h)

    return (best_h.tolist(), float(c5), n)
