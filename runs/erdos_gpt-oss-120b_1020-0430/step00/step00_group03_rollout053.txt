def project_onto_capped_simplex(v, S):
    # Find λ s.t. sum(clip(v - λ, 0, 1)) = S.
    lo = np.min(v) - 1.0  # lower bound for λ
    hi = np.max(v)        # upper bound
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)*0.5, 0, 1)

def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0/n
    # compute correlation v
    v_full = np.correlate(h, 1 - h, mode='full') * dx   # length 2n-1
    # compute softmax weights
    m = v_full.max()
    exps = np.exp(beta*(v_full - m))
    w = exps / exps.sum()   # shape 2n-1
    # For gradient, compute contributions via loops
    grad = np.zeros_like(h)
    w_rev = w[::-1]
    # Loop over ℓ from -(n-1) to (n-1)
    for idx in range(2*n-1):
        ell = idx - (n-1)
        wp = w[idx]
        wm = w_rev[idx]  # weight for -ℓ
        if wp == 0 and wm == 0:
            continue
        if ell >= 0:
            # overlap indices j = 0..n-ell-1
            j_slice = slice(0, n-ell)
            h_shift = h[ell:]   # length n-ell
            grad[j_slice] += wp * (1 - h_shift) - wm * h_shift
        else:
            shift = -ell
            j_slice = slice(shift, n)
            h_shift = h[:n-shift]
            grad[j_slice] += wp * (1 - h_shift) - wm * h_shift
    grad *= dx
    return grad, v_full

import numpy as np
import time

# ---------- Helper functions ----------
def evaluate_h(h):
    """Exact C5 evaluation for a step function."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def project_to_capped_simplex(v, S, tol=1e-12, max_iter=60):
    """
    Project vector v onto the set {x: 0 <= x_i <= 1, sum x_i = S}.
    Finds λ such that x = clip(v - λ, 0, 1) satisfies sum x = S.
    """
    lo = np.min(v) - 1.0  # λ lower bound
    hi = np.max(v)        # λ upper bound
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def gradient_softmax_max(h, beta):
    """
    Compute gradient of smooth max of the overlap values.
    Returns gradient (same shape as h) and the full vector of overlap values.
    """
    n = h.size
    dx = 2.0 / n
    # overlap values for all lags
    conv = np.correlate(h, 1 - h, mode="full") * dx          # length 2n-1
    # softmax weights
    m = conv.max()
    exp_shift = np.exp(beta * (conv - m))
    w = exp_shift / exp_shift.sum()                         # shape (2n-1,)
    w_rev = w[::-1]                                          # w_{-ell}
    grad = np.zeros_like(h)

    # ℓ runs from -(n-1) to (n-1)
    for idx in range(2 * n - 1):
        ell = idx - (n - 1)
        wp = w[idx]          # weight for +ℓ
        wm = w_rev[idx]      # weight for -ℓ
        if wp == 0.0 and wm == 0.0:
            continue
        if ell >= 0:
            # indices j = 0 .. n-ell-1
            end = n - ell
            h_shift = h[ell:]                     # length = end
            grad[:end] += wp * (1.0 - h_shift) - wm * h_shift
        else:
            shift = -ell
            # indices j = shift .. n-1
            end = n - shift
            h_shift = h[:end]                     # length = end
            grad[shift:] += wp * (1.0 - h_shift) - wm * h_shift

    grad *= dx
    return grad, conv


def adam_optimize(initial_h, time_budget):
    """
    Perform Adam optimisation on the smooth‑max objective.
    Returns the best vector found and its exact C5 value.
    """
    n = initial_h.size
    S = n / 2.0
    h = project_to_capped_simplex(initial_h, S)

    best_h = h.copy()
    best_score = evaluate_h(best_h)

    # Adam parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 5.0                     # initial learning rate
    beta_smooth = 1.0            # initial soft‑max inverse temperature
    it = 0
    start = time.time()

    while time.time() - start < time_budget:
        it += 1
        # gradient of smooth max
        grad, conv = gradient_softmax_max(h, beta_smooth)
        # exact objective for tracking
        cur_score = conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h_new = h - step
        h = project_to_capped_simplex(h_new, S)

        # schedule adjustments
        beta_smooth = min(50.0, beta_smooth * 1.001 + 0.001)
        if it % 200 == 0:
            lr *= 0.9

    final_score = evaluate_h(best_h)
    return best_h, final_score


def hill_climb(base_h, base_score, time_budget):
    """
    Simple random‑pair hill climbing to polish a solution.
    Returns an improved vector and its score.
    """
    n = base_h.size
    S = n / 2.0
    h = base_h.copy()
    best_h = h.copy()
    best_score = base_score
    start = time.time()
    rng = np.random.default_rng()

    while time.time() - start < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a small amount δ from j to i
        delta_max = min(0.05, 1.0 - h[i], h[j])
        if delta_max <= 0:
            continue
        delta = rng.uniform(-delta_max, delta_max)
        # apply move
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
        # re‑project only to fix possible tiny sum drift
        cur_sum = h_new.sum()
        if abs(cur_sum - S) > 1e-9:
            # simple fix: adjust the first non‑boundary element
            diff = S - cur_sum
            # find an index not at a bound
            idx = np.where((h_new > 1e-12) & (h_new < 1 - 1e-12))[0]
            if idx.size:
                k = idx[0]
                h_new[k] = np.clip(h_new[k] + diff, 0.0, 1.0)
            else:
                # fall back to projection if needed
                h_new = project_to_capped_simplex(h_new, S)

        new_score = evaluate_h(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
            h = h_new
        else:
            # occasional acceptance to escape plateaus
            if rng.random() < 0.001:
                h = h_new

    return best_h, best_score


def optimise_one_start(init_h, time_budget):
    """
    Run Adam optimisation and a short hill‑climb.
    Returns best vector and its C5 value.
    """
    # split time between gradient phase and polishing
    t_adam = time_budget * 0.85
    t_hc = time_budget - t_adam
    h_adam, score_adam = adam_optimize(init_h, t_adam)
    h_final, score_final = hill_climb(h_adam, score_adam, t_hc)
    return h_final, score_final


# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_all = time.time()
    # Determine dimensionality
    try:
        init_vec = np.array(initial_h_values, dtype=np.float64)
        n = init_vec.size
    except NameError:
        n = 70
        init_vec = np.random.rand(n)

    S = n / 2.0
    # Ensure the starter respects the constraints
    init_vec = project_to_capped_simplex(init_vec, S)

    best_h = None
    best_score = np.inf

    # Number of independent restarts (adjusted by remaining time)
    restarts = 6
    for r in range(restarts):
        elapsed = time.time() - start_all
        remaining = budget_s - elapsed
        if remaining <= 5.0:
            break
        # allocate a slice of the remaining time
        alloc = remaining * 0.9 / (restarts - r)
        if r == 0:
            h0 = init_vec.copy()
        else:
            h0 = np.random.rand(n)
            h0 = project_to_capped_simplex(h0, S)

        cand_h, cand_score = optimise_one_start(h0, alloc)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()

    # Final safety projection (enforces exact sum)
    best_h = project_to_capped_simplex(best_h, S)

    # Compute final bound using the provided evaluator (if present)
    try:
        final_bound = evaluate_erdos_solution(best_h)
    except NameError:
        final_bound = evaluate_h(best_h)

    return best_h, float(final_bound), n
