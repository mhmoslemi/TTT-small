def project_onto_bounds_and_sum(v, target):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mu = (lo + hi) / 2.0
        proj = np.clip(v - mu, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    return np.clip(v - ((lo + hi) / 2.0), 0.0, 1.0)

import time
import numpy as np

def _project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x: 0<=x_i<=1, sum x_i = target}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mu = (lo + hi) * 0.5
        proj = np.clip(v - mu, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mu          # need larger mu to shrink the sum
        else:
            hi = mu
    return np.clip(v - ((lo + hi) * 0.5), 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for the discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def _max_shift_and_grad(h: np.ndarray):
    """Find shift giving the maximal overlap and return (shift, gradient)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(corr.argmax())
    shift = idx - (n - 1)               # shift = k (positive → right)
    # gradient of the maximal term c_k = sum_i h_i (1-h_{i+shift})
    grad = np.ones_like(h)
    if shift >= 0:
        grad[: n - shift] -= h[shift:]
        grad[shift:]     -= h[: n - shift]
    else:
        k = -shift
        grad[k:]        -= h[: n - k]
        grad[: n - k]   -= h[k:]
    return shift, grad


def _subgradient_descent(start_h: np.ndarray, iters: int,
                         target_sum: float, time_limit: float,
                         start_time: float):
    """Run a simple projected sub‑gradient descent."""
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_value(h)

    for t in range(1, iters + 1):
        if time.time() - start_time > time_limit:
            break
        shift, grad = _max_shift_and_grad(h)
        # diminishing step size
        step = 0.5 / np.sqrt(t)
        h_new = h - step * grad
        h = _project_to_bounds_and_sum(h_new, target_sum)

        # occasional small random shake to escape plateaus
        if t % 250 == 0:
            sigma = 0.06 / np.sqrt(t)
            h = np.clip(h + np.random.uniform(-sigma, sigma, size=n), 0.0, 1.0)
            h = _project_to_bounds_and_sum(h, target_sum)

        val = _c5_value(h)
        if val < best_val - 1e-10:
            best_val = val
            best_h = h.copy()
    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h on [0,2] with a low Erdős C₅ bound."""
    start = time.time()
    np.random.seed(seed)

    # ----- choose discretisation size -------------------------------------------------
    N = 256                     # enough points for a fine discretisation
    target_sum = N / 2.0

    # ----- build an initial vector ----------------------------------------------------
    try:
        # use the supplied construction if present, up‑sample it

        n_small = small.shape[0]
        x_small = np.linspace(0.0, 2.0, n_small, endpoint=False)
        x_big = np.linspace(0.0, 2.0, N, endpoint=False)
        h0 = np.interp(x_big, x_small, small)
    except NameError:
        # no supplied construction – start from a random point
        h0 = np.random.rand(N)

    h0 = _project_to_bounds_and_sum(h0, target_sum)
    best_h = h0.copy()
    best_val = _c5_value(best_h)

    # ----- optimisation loop ----------------------------------------------------------
    # we keep trying new starting points until the budget is almost exhausted
    time_budget = budget_s * 0.95
    restart = 0
    while time.time() - start < time_budget:
        # run a descent from the current start point
