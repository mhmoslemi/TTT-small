def solve_binary_sequence(n):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n//2]
    for l in range(1, n):
        # i indices
        # Index slice off
        idx_range = n - l
        if idx_range <= 0: continue
        y = cp.Variable(idx_range, boolean=True)
        # y_i <= h[i]
        constraints.append(y <= h[:idx_range])
        # y_i <= h[l:l+idx_range]
        constraints.append(y <= h[l:])
        # y_i >= h[i] + h[l_i] -1
        constraints.append(y + 1 >= h[:idx_range] + h[l:])
        constraints.append(cp.sum(y) >= t)
    # t is lower bound, but also we can bound t.
    constraints.append(t >= 0)
    # t ≤ maximal possible (n//2)
    constraints.append(t <= n//2)
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLPK_MI, warm_start=True)
    return h.value, t.value, prob.status
```

def solve_milp_binary(n):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n//2]
    # For each lag
    for l in range(1, n):
        idx_range = n - l
        if idx_range <= 0:
            continue
        y = cp.Variable(idx_range, boolean=True)
        constraints.append(y <= h[:idx_range])
        constraints.append(y <= h[l:])
        constraints.append(y >= h[:idx_range] + h[l:] - 1)
        constraints.append(cp.sum(y) >= t)
    constraints.append(t >= 0)
    constraints.append(t <= n//2)
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.GLPK_MI, glpk={'msg_lev': 'GLP_MSG_OFF'})
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, None
    h_val = np.rint(h.value).astype(float)  # keep as float for correlation
    t_val = int(round(t.value))
    return h_val, t_val

import numpy as np, time, cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode='full') * dx
    return float(O.max())

def solve_binary_sequence(n, max_time=None):
    # Build MILP to maximize minimal auto-correlation
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    cons = [cp.sum(h) == n // 2]
    for lag in range(1, n):
        length = n - lag
        if length <= 0:
            continue
        y = cp.Variable(length, boolean=True)
        cons.append(y <= h[:length])
        cons.append(y <= h[lag:])
        cons.append(y >= h[:length] + h[lag:] - 1)
        cons.append(cp.sum(y) >= t)
    cons.append(t >= 0)
    cons.append(t <= n // 2)
    prob = cp.Problem(cp.Maximize(t), cons)
    solve_kwargs = dict(solver=cp.CBC, verbose=False)
    # Try to limit time if supported
    if max_time is not None:
        # CBC accepts "seconds" parameter
        solve_kwargs["seconds"] = max_time
    try:
        prob.solve(**solve_kwargs)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
        except Exception:
            return None
    if prob.status not in ("optimal", "optimal_inaccurate", "feasible"):
        return None
    h_val = np.rint(h.value).astype(float)
    # repair sum if necessary
    target = n // 2
    cur = int(round(h_val.sum()))
    if cur != target:
        diff = cur - target
        if diff > 0:
            idx = np.where(h_val > 0.5)[0]
            if len(idx) >= diff:
                np.random.shuffle(idx)
                h_val[idx[:diff]] = 0.0
        else:
            idx = np.where(h_val < 0.5)[0]
            if len(idx) >= -diff:
                np.random.shuffle(idx)
                h_val[idx[: -diff]] = 1.0
    return h_val

    # attempt MILP
    h = solve_binary_sequence(n, max_time=remaining)
    if h is None:

        try:
            h = np.array(initial_h_values, copy=False, dtype=float)
            # ensure sum constraint (if not satisfied, normalize)
            target_sum = n * 0.5
            cur_sum = h.sum()
            if abs(cur_sum - target_sum) > 1e-9:

                # distribute excess uniformly
                if excess > 0:
                    # reduce some entries
                    idx = np.where(h > 0)[0]
                    np.random.shuffle(idx)
                    for i in idx:
                        red = min(h[i], excess / len(idx))
                        h[i] -= red
                        excess -= red
                        if excess <= 0:

                else:

import numpy as np
import time
import cvxpy as cp

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode="full") * dx
    return float(O.max())

def _solve_binary_sequence(n, max_seconds=None):
    """Return a {0,1} vector of length n with sum n/2 that maximises the
    minimal self‑overlap.  If solving fails, None is returned."""
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    cons = [cp.sum(h) == n // 2]

    for lag in range(1, n):
        length = n - lag
        y = cp.Variable(length, boolean=True)
        cons.append(y <= h[:length])
        cons.append(y <= h[lag:])
        cons.append(y >= h[:length] + h[lag:] - 1)
        cons.append(cp.sum(y) >= t)

    cons.append(t >= 0)
    cons.append(t <= n // 2)

    prob = cp.Problem(cp.Maximize(t), cons)

    solve_kwargs = {"solver": cp.CBC, "verbose": False}
    if max_seconds is not None:
        # CBC accepts a 'seconds' argument; ignore if unknown.
        solve_kwargs["seconds"] = max_seconds
    try:
        prob.solve(**solve_kwargs)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate", "feasible"):
        return None

    hv = np.rint(h.value).astype(float)  # round to {0,1}
    target = n // 2
    cur = int(round(hv.sum()))
    if cur != target:
        diff = cur - target
        if diff > 0:
            idx = np.where(hv > 0.5)[0]
            if len(idx) >= diff:
                np.random.shuffle(idx)
                hv[idx[:diff]] = 0.0
        else:
            idx = np.where(hv < 0.5)[0]
            if len(idx) >= -diff:
                np.random.shuffle(idx)
                hv[idx[: -diff]] = 1.0
    return hv

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()

    # Determine the dimension to work with.
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        if n % 2:
            n += 1
    else:
        n = 70

    # Reserve a bit of time for housekeeping.
    remaining = max(budget_s - (time.time() - start) - 0.5, 0.5)

    # Try the MILP approach.
    h = _solve_binary_sequence(n, max_seconds=remaining)

    # Fallback to the provided construction if MILP failed.
    if h is None:
        if "initial_h_values" in globals():
            h = np.array(initial_h_values, dtype=float, copy=False)
            # Ensure the integral constraint (possible tiny drift).
            target_sum = n / 2.0
            cur_sum = h.sum()
            if abs(cur_sum - target_sum) > 1e-9:
                h = np.clip(h, 0.0, 1.0)
                scale = target_sum / cur_sum
                h *= scale
                # Simple correction for rounding error.
                diff = target_sum - h.sum()
                if abs(diff) > 1e-8:
                    h += diff / len(h)
        else:
            # Simple deterministic construction: half ones, half zeros.
            h = np.zeros(n, dtype=float)
            h[: n // 2] = 1.0
            np.random.shuffle(h)

    # Final sanity checks / repairs.
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    cur = h.sum()
    if abs(cur - target) > 1e-6:
        # Proportional scaling followed by a tiny uniform adjustment.
        h *= target / cur
        diff = target - h.sum()
        if abs(diff) > 1e-9:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    c5 = _compute_c5(h)
