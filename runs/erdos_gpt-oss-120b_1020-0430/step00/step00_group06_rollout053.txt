def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr  # also return full corr if needed

def project_to_sum(h, target_sum, max_iter=10):
    """Clip h to [0,1] and adjust so that sum equals target_sum."""
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            # ensure we don't exceed bound
            max_inc = np.min(1.0 - h[free])
            if inc > max_inc:
                inc = max_inc
            h[free] += inc
        else:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            dec = (-diff) / free.size
            max_dec = np.min(h[free])
            if dec > max_dec:
                dec = max_dec
            h[free] -= dec
    return h

def compute_c5(h):
    """Return (C5, full correlation array) for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def project_to_sum(h, target, iters=15):
    """Clip h to [0,1] and adjust so that sum(h) = target."""
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(iters):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            max_inc = np.min(1.0 - h[free])
            inc = min(inc, max_inc)
            h[free] += inc
        else:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            dec = (-diff) / free.size
            max_dec = np.min(h[free])
            dec = min(dec, max_dec)
            h[free] -= dec
    return h
