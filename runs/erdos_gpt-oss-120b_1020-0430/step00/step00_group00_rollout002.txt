import time, math
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Simulated‑annealing search for a step function h on [0,2] that minimizes
    C₅ = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (list_of_h, C5_value, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # 1.  Initialise h
    # ------------------------------------------------------------
    try:
        # Use the parent solution if it exists.
        init_h = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        # No parent – build a feasible start with the required integral.
        n = 45
        target_sum = n / 2.0
        init_h = np.zeros(n, dtype=float)
        full_ones = int(math.floor(target_sum))
        init_h[:full_ones] = 1.0
        remainder = target_sum - full_ones
        if remainder > 1e-12:
            init_h[full_ones] = remainder
        rng.shuffle(init_h)
    else:
        n = init_h.size
        target_sum = n / 2.0

    # Ensure the integral constraint exactly (up to floating round‑off).
    cur_sum = init_h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        init_h = init_h * (target_sum / cur_sum)
        init_h = np.clip(init_h, 0.0, 1.0)

    dx = 2.0 / n

    # ------------------------------------------------------------
    # 2.  Helper to evaluate C₅ for a vector h
    # ------------------------------------------------------------
    def evaluate(h):
        # Using the same definition as the harness.
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    # ------------------------------------------------------------
    # 3.  Simulated‑annealing loop
    # ------------------------------------------------------------
    best_h = init_h.copy()
    best_val = evaluate(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    # SA schedule parameters
    max_iter = 8_000_000          # upper bound – will be cut by the time budget
    start_T = 0.05
    end_T = 1e-5
    log_ratio = math.log(end_T / start_T)

    start_step = 0.5
    end_step = 0.01

    start_time = time.time()
    # check every few thousand iterations to avoid excessive overhead
    check_interval = 4096

    for it in range(max_iter):
        # ---- time guard -------------------------------------------------
        if (it & (check_interval - 1)) == 0:
            if time.time() - start_time > budget_s * 0.97:
                break

        prog = it / max_iter
        T = start_T * math.exp(log_ratio * prog)          # exponential cooling
        step = start_step * (1.0 - prog) + end_step * prog

        # ---- propose a move that keeps the sum unchanged -----------------
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        hi, hj = cur_h[i], cur_h[j]

        # feasible delta range respecting bounds and current step size
        lower = max(-hj, hi - 1.0, -step)
        upper = min(hi, 1.0 - hj, step)
        if lower >= upper:
            continue

        delta = rng.uniform(lower, upper)

        cand = cur_h.copy()
        cand[i] = hi - delta
        cand[j] = hj + delta

        cand_val = evaluate(cand)

        # ---- Metropolis acceptance ---------------------------------------
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / T):
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val
                # optional early stop when we already beat a target
                if best_val <= 0.3808:
                    # keep searching a bit for possible further improvement
                    pass

    # ------------------------------------------------------------
    # 4.  Project the best vector onto the exact integral hyperplane
    # ------------------------------------------------------------
    final_sum = best_h.sum()
    diff = target_sum - final_sum
    if abs(diff) > 1e-12:

        for idx in range(n):
            if abs(diff) < 1e-12:

            else:

                dec = min(-diff, cap)
                best_h[idx] -= dec
                diff += dec

    # 5.  Return the result
    # ------------------------------------------------------------
    return (best_h.tolist(), float(final_val), int(n))
