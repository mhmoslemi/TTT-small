def propose_move(h, step=0.1):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    max_transfer = min(step, h[i], 1 - h[j])
    if max_transfer <= 0: return None
    delta = np.random.uniform(0, max_transfer)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def project_to_feasible(v, target):
    # find lambda such that sum_clipped = target
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return proj

def gradient_step(h, lag, eta):
    # compute gradient g = 1 - (h shifted forward + h shifted backward)
    n = len(h)
    if lag >= 0:
        forward = np.concatenate([np.zeros(lag), h[:n-lag]])
        backward = np.concatenate([h[lag:], np.zeros(lag)])
    else:
        l = -lag
        forward = np.concatenate([h[l:], np.zeros(l)])
        backward = np.concatenate([np.zeros(l), h[:n-l]])
    g = 1.0 - (forward + backward)  # shape n
    # Determine source (g>0) and target (g<0)
    src_candidates = np.where(g > 1e-9)[0]
    tgt_candidates = np.where(g < -1e-9)[0]
    if src_candidates.size == 0 or tgt_candidates.size == 0:
        return h, False
    src = src_candidates[np.argmax(g[src_candidates])]
    tgt = tgt_candidates[np.argmin(g[tgt_candidates])]
    # compute max delta to maintain bounds
    max_delta = min(eta, h[src], 1.0 - h[tgt])
    if max_delta <= 0:
        return h, False
    new_h = h.copy()
    new_h[src] -= max_delta
    new_h[tgt] += max_delta
    return new_h, True

    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # ensure we transfer from i to j, direction can be either
    # We'll attempt both directions because both may be feasible.
    # Choose direction that gives positive max_transfer.
    # Compute delta_max1 = min(max_step, h[i], 1 - h[j]) (transfer i->j)

        return None

import time, math
import numpy as np

def _project_to_capped_simplex(v, target, max_iter=60):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum x_i = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _score(h):
    """C5 score for a step function given by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return (c * dx).max()


def _gradient_worst_lag(h):
    """Return (lag, gradient) for the lag attaining the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)                     # lag in samples, positive = shift right
    # forward = h shifted by +lag (zero‑padded)
    if lag >= 0:
        forward = np.concatenate([np.zeros(lag), h[:n - lag]])
        backward = np.concatenate([h[lag:], np.zeros(lag)])
    else:
        l = -lag
        forward = np.concatenate([np.zeros(l), h[:n - l]])
        backward = np.concatenate([h[l:], np.zeros(l)])
    grad = 1.0 - (forward + backward)       # derivative of f_k (dx factor omitted)
    return lag, grad


def _transfer_move(h, rng, max_delta):
    """Return a new feasible vector obtained by moving mass between two coordinates."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2, endpoint=False)
    while i == j:
        j = rng.integers(0, n)
    # try i -> j
    d1 = min(max_delta, h[i], 1.0 - h[j])
    # try j -> i
    d2 = min(max_delta, h[j], 1.0 - h[i])
    if d1 <= 0 and d2 <= 0:
        return None
    if d1 >= d2:
        delta = rng.random() * d1
        new = h.copy()
        new[i] -= delta
        new[j] += delta
    else:
        delta = rng.random() * d2
        new = h.copy()
        new[j] -= delta
        new[i] += delta
    return new


def _gradient_move(h, lag, grad, rng, max_delta):
    """Move mass from a positive‑gradient index to a negative‑gradient index."""
    src = np.where(grad > 1e-9)[0]
    tgt = np.where(grad < -1e-9)[0]
    if src.size == 0 or tgt.size == 0:
        return None
    i = src[np.argmax(grad[src])]
    j = tgt[np.argmin(grad[tgt])]
    d = min(max_delta, h[i], 1.0 - h[j])
    if d <= 0:
        return None
    delta = rng.random() * d
    new = h.copy()
    new[i] -= delta
    new[j] += delta
    return new


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] (sampled uniformly) to minimise
    C5 = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # initialise ------------------------------------------------------- #
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, copy=True)
        n = h0.shape[0]
    else:
        n = 200                     # reasonable resolution when no start is given
        h0 = np.full(n, 0.5)       # uniform start (will be projected)

    target_sum = n / 2.0            # ∑h = n/2  ⇔  ∫ h = 1
    h0 = _project_to_capped_simplex(h0, target_sum)

    best_h = h0.copy()
    best_score = _score(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    # ------------------------------------------------------------------ #
    # simulated annealing ------------------------------------------------ #
    start = time.time()
    deadline = start + budget_s - 0.05

    max_delta = 0.05               # maximal amount transferred in a single move
    T0 = 0.04
    Tmin = 1e-6
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        it += 1
        T = max(Tmin, T0 * (decay ** it))

        # choose move type
        if rng.random() < 0.7:
            lag, grad = _gradient_worst_lag(cur_h)
            cand = _gradient_move(cur_h, lag, grad, rng, max_delta)
            if cand is None:
                cand = _transfer_move(cur_h, rng, max_delta)
        else:
            cand = _transfer_move(cur_h, rng, max_delta)

        if cand is None:
            continue

        cand_score = _score(cand)
        delta = cand_score - cur_score

        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_h = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score

        # occasional re‑heat around the best solution
        if it % 5000 == 0:
            pert = np.clip(best_h + rng.uniform(-max_delta, max_delta, size=n), 0.0, 1.0)
            cur_h = _project_to_capped_simplex(pert, target_sum)
            cur_score = _score(cur_h)

    # final projection (tiny numeric drift may appear)
    best_h = _project_to_capped_simplex(best_h, target_sum)
    final_score = _score(best_h)
    if final_score < best_score:
        best_score = final_score

    return best_h, float(best_score), n
