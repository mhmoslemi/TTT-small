import time

import scipy.optimize

def _compute_c5(h):
    """Return C5 for step function h (array of values)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _grad_c5(h):
    """Sub‑gradient of C5 at h (uses the worst shift)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                     # shift can be negative
    grad = np.ones(n)
    if shift >= 0:
        ov = n - shift
        # contribution of h[i] * h[i+shift]
        grad[:ov] -= h[shift:shift + ov]
        grad[shift:shift + ov] -= h[:ov]
    else:
        s = -shift
        ov = n - s
        grad[s:] -= h[:ov]
        grad[:ov] -= h[s:]
    return grad * dx

def _project_box_hyperplane(v, target_sum, tol=1e-9, max_it=60):
    """Project v onto {0 ≤ x ≤ 1, sum(x)=target_sum}."""
    lo, hi = -2.0, 2.0                     # enough to drive all entries to 0/1
    for _ in range(max_it):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - w.sum()
    if abs(diff) > tol:
        interior = (w > 1e-12) & (w < 1 - 1e-12)
        idx = np.where(interior)[0]
        if idx.size > 0:
            w[idx] += diff / idx.size
            np.clip(w, 0.0, 1.0, out=w)
    return w

    """
    Linear shape h(x)=alpha + (1-2*alpha)*|x-1| on [0,2].
    alpha∈[0,0.5] guarantees integral 1.
    """
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    dist = np.abs(x - 1.0)                     # in [0,1]
    beta = 1.0 - 2.0 * alpha
    h = alpha + beta * dist
    # projection to exact constraints (tiny discretisation error)
    return _project_box_hyperplane(h, n / 2.0)

def _make_power_shape(n, alpha, p):
    """
    Shape h(x)=alpha + beta*|x-1|^p  with integral 1.
    beta computed from the integral condition.
    """
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    dist = np.abs(x - 1.0)
    beta = (1.0 - 2.0 * alpha) * (p + 1.0) / 2.0
    h = alpha + beta * (dist ** p)
    return _project_box_hyperplane(h, n / 2.0)

def _make_edge_const_shape(n, a_frac):
    """
    Two blocks of value 1 on each side, interior constant.
    a_frac = total fraction of points belonging to the two edge blocks (0..0.5).
    """
    a_frac = max(0.0, min(0.5, a_frac))
    n_edge = int(round(n * a_frac))
    # split equally on both sides
    left = n_edge // 2
    right = n_edge - left
    interior = n - left - right
    if interior == 0:
        c = 0.0
    else:
        c = (n / 2.0 - (left + right)) / interior
    h = np.full(n, c)
    if left > 0:
        h[:left] = 1.0
    if right > 0:
        h[-right:] = 1.0
    return _project_box_hyperplane(h, n / 2.0)

def _interpolate_initial(old_h, n):
    """Resize old_h (len m) to n points by linear interpolation."""
    m = len(old_h)
    dx_old = 2.0 / m
    x_old = (np.arange(m) + 0.5) * dx_old
    dx_new = 2.0 / n
    x_new = (np.arange(n) + 0.5) * dx_new
    new_h = np.interp(x_new, x_old, old_h, left=0.0, right=0.0)
    return _project_box_hyperplane(new_h, n / 2.0)

def _slsqp_optimize(start_h, max_iter=80):
    n = start_h.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target},)

    def obj(x):
        return _compute_c5(x)

    def jac(x):
        return _grad_c5(x)

    try:
        res = scipy.optimize.minimize(obj, start_h, method='SLSQP',
                                      jac=jac, bounds=bounds,
                                      constraints=cons,
                                      options={'maxiter': max_iter,
                                               'ftol': 1e-10,
                                               'disp': False})
        if res.success:
            return res.x, _compute_c5(res.x)
    except Exception:
        pass
    # fallback – return the start if optimisation fails
    return start_h, _compute_c5(start_h)
