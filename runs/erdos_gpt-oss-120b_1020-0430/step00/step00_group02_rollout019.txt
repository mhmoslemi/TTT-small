def _evaluate_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1−h(x+k))dx = max(conv)*dx."""
    n = h.shape[0]
    dx = 2.0 / n
    m = 2 * n
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * np.conj(G), n=m)[: 2 * n - 1]
    return conv.max() * dx


def _worst_shift(h: np.ndarray):
    """Return (value, shift) for the shift attaining the maximal overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    m = 2 * n
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * np.conj(G), n=m)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    value = conv[idx] * dx
    shift = idx - (n - 1)          # shift ∈ {-(n‑1),…,n‑1}
    return value, shift


def _subgradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the worst‑shift term w.r.t. h."""
    n = h.shape[0]
    g = np.ones_like(h)
    if shift > 0:
        g[: n - shift] -= h[shift:]          # -h_{i+shift}
        g[shift:]      -= h[: n - shift]    # -h_{i-shift}
    elif shift < 0:
        s = -shift
        g[s:]          -= h[: n - s]        # -h_{i-shift}
        g[: n - s]     -= h[s:]            # -h_{i+shift}
    else:                                    # shift == 0
        g -= 2.0 * h
    return g


def _project_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, Σx = target} via a simple water‑filling (binary search)."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger number of points."""
    old_n = h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function giving a low C₅ bound."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Starting vector
    init = kwargs.get('initial_h_values')
    if init is None:
        # simple “U” shape: high at the ends, low in the middle
        n0 = 45
        x = np.linspace(0.0, 2.0, n0, endpoint=False)
        r = np.abs(x - 1.0)                    # distance from centre
        a = (n0 / 2.0) / r.sum()               # scale to satisfy the integral
        h0 = np.clip(a * r, 0.0, 1.0)
        h0 = _project_bounds_sum(h0, n0 / 2.0)
        init = h0
    else:
        init = np.asarray(init, dtype=float)

    # ------------------------------------------------------------------
    # Candidate resolutions (must stay ≤1000 points)
    candidates = [45, 70, 105, 150, 210, 315, 450]
    candidates = [n for n in candidates if n <= 1000]

    best_h, best_score, best_n = None, np.inf, None
    total_iters = 0

    for n in candidates:
        # time guard
        if time.time() - start > budget_s * 0.95:
            break

        # initialise at this resolution
        if n == init.shape[0]:
            h = init.copy()
        else:
            h = _upsample(init, n)

        target = n / 2.0
        h = _project_bounds_sum(h, target)

        cur_score = _evaluate_c5(h)
        if cur_score < best_score:
            best_h, best_score, best_n = h.copy(), cur_score, n

        # ------------------------------------------------------------------
        # Gradient‑based descent with occasional random jitter
        max_iter = int(3e4 * (500.0 / n))          # more steps for smaller n
        max_iter = max(2000, min(max_iter, 40000))
        alpha0 = 0.12                               # initial step size
        for it in range(max_iter):
            # global time limit
            if time.time() - start > budget_s * 0.99:
                break

            # current worst shift
            worst_val, shift = _worst_shift(h)

            # sub‑gradient for that shift
            g = _subgradient(h, shift)

            # decreasing step size
            alpha = alpha0 / (1.0 + it * 1e-4)

            # propose a new point
            h_new = h - alpha * g
            np.clip(h_new, 0.0, 1.0, out=h_new)
            h_new = _project_bounds_sum(h_new, target)

            new_score = _evaluate_c5(h_new)
            total_iters += 1

            # Accept if improvement, else keep the better of the two
            if new_score < cur_score:
                h, cur_score = h_new, new_score
            else:
                # occasional escape move
                if np.random.rand() < 0.001:
                    h = h_new
                    cur_score = new_score

            # record best overall
            if cur_score < best_score:
                best_h, best_score, best_n = h.copy(), cur_score, n

            # every few hundred iterations, add a tiny random jitter
            if it % 500 == 0 and it > 0:
                i, j = np.random.randint(0, n, size=2)
                delta = (np.random.rand() - 0.5) * 0.04
                # enforce bounds on the fly
                delta = min(delta, 1.0 - h[i], h[j]) if delta > 0 else max(
                    delta, -h[i], -(1.0 - h[j])
                )
                if abs(delta) > 1e-12:
                    h[i] += delta
                    h[j] -= delta
                    # projection guarantees feasibility
                    h = _project_bounds_sum(h, target)
                    cur_score = _evaluate_c5(h)
                    if cur_score < best_score:
                        best_h, best_score, best_n = h.copy(), cur_score, n

    # final safety projection
    if best_h is not None:
        best_h = _project_bounds_sum(best_h, best_n / 2.0)
        best_score = _evaluate_c5(best_h)

    # fallback (should never happen)
    if best_h is None:
        best_n = 45
        best_h = np.full(best_n, 0.5)
        best_score = _evaluate_c5(best_h)

    return best_h, float(best_score), int(best_n)
