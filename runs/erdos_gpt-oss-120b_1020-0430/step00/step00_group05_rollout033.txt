def hill_climb(n, rng, max_iters):
   # start with random binary vector with n/2 ones
   h = np.zeros(n, dtype=np.int8)
   h[:n//2] = 1
   rng.shuffle(h)
   best = h.copy()
   bestC = compute_C(h)
   for it in range(max_iters):
      # propose swap
      ones = np.where(h==1)[0]
      zeros = np.where(h==0)[0]
      i = rng.choice(ones)
      j = rng.choice(zeros)
      # swap
      h[i], h[j] = 0, 1
      C = compute_C(h)
      if C <= bestC:
          bestC = C
          best = h.copy()
      else:
          # maybe accept with probability exp(-(C-bestC)/temp)
          # but we can optionally reject. We'll do simple: keep if better, else revert
          h[i], h[j] = 1, 0
   return best, bestC
```

def anneal(n, rng, time_limit):
    # returns best_h, best_c5
    # start random binary
    h = np.zeros(n, dtype=float)
    h[:n//2] = 1.0
    rng.shuffle(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    cur_h = h
    cur_c5 = best_c5
    # temperature schedule parameter
    T0 = 0.5
    # iterations defined roughly based on time limit
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # propose swap
        i = rng.integers(n)
        while cur_h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] > 0.5:
            j = rng.integers(n)
        # swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c5 = evaluate_erdos_solution(cur_h)
        # acceptance
        # use simulated annealing decreasing temperature
        temp = T0 * (1 - it / 1000000)  # large schedule
        if new_c5 <= cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / (temp + 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_c5

    # Sturmian (Beatty) construction with golden‑ratio slope
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    frac = (np.arange(n, dtype=float) * phi) % 1.0
    h = (frac < 0.5).astype(float)

    # Enforce exact sum = n/2 (required by the problem)
    target = n // 2
    cur = int(round(h.sum()))
    while cur > target:
        idx = rng.choice(np.where(h == 1.0)[0])
        h[idx] = 0.0
        cur -= 1
    while cur < target:
        idx = rng.choice(np.where(h == 0.0)[0])
        h[idx] = 1.0
        cur += 1
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Perform a time‑bounded simulated‑annealing style optimisation.
    Returns a tuple (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -------------------------------------------------------------
    # Determine the grid size – use the length of the supplied start
    # vector if it exists, otherwise fall back to 200 points.
    # -------------------------------------------------------------
    try:
        n_points = len(initial_h_values)
    except NameError:
        n_points = 200

    # -------------------------------------------------------------
    # Initialise h
    # -------------------------------------------------------------
    h_cur = _initial_vector(n_points, rng)
    best_h = h_cur.copy()
    best_c5 = _eval_c5(best_h)

    # -------------------------------------------------------------
    # Simulated annealing parameters (tuned for the given budget)
    # -------------------------------------------------------------
    # total number of proposals – we'll stop early if time runs out
    max_iters = 800_000
    T0 = 0.5                # initial temperature
    step_max = 0.4          # initial maximal delta for a move

    # Pre‑compute masks for fast index selection
    indices = np.arange(n_points)

    it = 0
    while it < max_iters:
        # time guard
        if time.time() - start_time > budget_s - 0.2:
            break
        it += 1

        # -----------------------------------------------------------------
        # Propose a move: transfer a small amount of mass from index j
        # (where h is >0) to index i (where h is <1).  This keeps the sum
        # invariant and respects box constraints.
        # -----------------------------------------------------------------
        i = rng.integers(n_points)
        while h_cur[i] >= 0.999:
            i = rng.integers(n_points)

        j = rng.integers(n_points)
        while h_cur[j] <= 0.001:
            j = rng.integers(n_points)

        # maximal admissible transfer respecting the box constraints
        max_delta = min(1.0 - h_cur[i], h_cur[j], step_max) * rng.random()
        if max_delta <= 0.0:
            continue

        # Apply the move
        h_cur[i] += max_delta
        h_cur[j] -= max_delta

        new_c5 = _eval_c5(h_cur)

        # -------------------------------------------------------------
        # Acceptance rule – Metropolis with a linear cooling schedule.
        # -------------------------------------------------------------
        temperature = T0 * (1.0 - it / max_iters)
        if new_c5 <= best_c5:
            # immediate acceptance of any improvement over the global best
            best_c5 = new_c5
            best_h = h_cur.copy()
        if new_c5 <= _eval_c5(best_h) or rng.random() < math.exp(-(new_c5 - best_c5) / (temperature + 1e-12)):
            # keep the move (it may be worse than the current best but better than the last state)
            pass
        else:
            # reject – revert
            h_cur[i] -= max_delta
            h_cur[j] += max_delta

        # ----------- gradually shrink the allowed step size ----------------
        if it % 5000 == 0:
            step_max = max(0.02, step_max * 0.95)

    # final evaluation (guard against tiny floating‑point drift)
    final_c5 = _eval_c5(best_h)

    # -----------------------------------------------------------------
    # Return the result in the required format
    # -----------------------------------------------------------------
    return (best_h, float(final_c5), int(n_points))
