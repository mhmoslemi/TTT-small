        return True
    else:

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    from scipy.optimize import differential_evolution
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_score = np.inf
    best_n = None

    def compute_c5(h):
        n = h.shape[0]
        dx = 2.0 / n
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def project(v, target):
        lo, hi = -1.0, 1.0
        for _ in range(40):
            lam = (lo + hi) / 2.0
            proj = np.clip(v + lam, 0.0, 1.0)
            s = proj.sum()
            if s > target:
                hi = lam
            else:
                lo = lam
        return np.clip(v + lo, 0.0, 1.0)

    # Candidate sizes
    n_list = [45, 60, 80, 100, 120, 150]
    for n in n_list:
        if time.time() - start > budget_s * 0.95:
            break
        target_sum = n / 2.0
        # initial guess from triangular shape
        dx = 2.0 / n
        x = (np.arange(n) + 0.5) * dx
        h0 = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
        h0 = project(h0, target_sum)
        # Build initial population for DE
        popsize = 15
        # Random population
        init_pop = np.random.rand(popsize, n)
        init_pop[0] = h0
        for i in range(popsize):
            init_pop[i] = project(init_pop[i], target_sum)
        # objective uses projection to enforce sum
        def obj(v):
            h = project(v, target_sum)
            return compute_c5(h)
        # run DE
        result = differential_evolution(obj, [(0.0, 1.0)] * n,
                                        init=init_pop,
                                        maxiter=2000,
                                        popsize=popsize,
                                        mutation=(0.5, 1.0),
                                        recombination=0.7,
                                        polish=False,
                                        seed=seed,
                                        updating='deferred')
        h_opt = project(result.x, target_sum)
        score_opt = compute_c5(h_opt)
        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()
            best_n = n
        # quick local refinement
        h = h_opt.copy()
        cur_score = score_opt
        step = 0.2
        # simple annealing like loop
        for it in range(5000):
            if time.time() - start > budget_s * 0.95:
                break
            # choose random pair
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            if np.random.rand() < 0.5:
                # move mass from i to j
                max_inc = min(step, 1.0 - h[j])
                max_dec = min(step, h[i])
                delta = np.random.rand() * min(max_inc, max_dec)
                if delta <= 0:
                    continue
                h_i_new = h[i] - delta
                h_j_new = h[j] + delta
                h[i], h[j] = h_i_new, h_j_new
            else:
                max_inc = min(step, 1.0 - h[i])
                max_dec = min(step, h[j])
                delta = np.random.rand() * min(max_inc, max_dec)
                if delta <= 0:
                    continue
                h_i_new = h[i] + delta
                h_j_new = h[j] - delta
                h[i], h[j] = h_i_new, h_j_new
            new_score = compute_c5(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
                    best_n = n

import numpy as np, time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_score = np.inf
    best_n = None
    target_bound = 0.3808

    def compute_c5(h):
        n = h.shape[0]
        dx = 2.0 / n
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def project(v, target):
        lo, hi = -1.0, 1.0
        for _ in range(40):
            lam = (lo + hi) / 2.0
            proj = np.clip(v + lam, 0.0, 1.0)
            s = proj.sum()
            if s > target:
                hi = lam
            else:
                lo = lam
        return np.clip(v + lo, 0.0, 1.0)

    # candidate discretizations (even length)
    n_candidates = [45, 60, 80, 100, 120, 150]
    for n in n_candidates:
        if time.time() - start > budget_s * 0.95:
            break
        target_sum = n / 2.0

        iv = globals().get('initial_h_values')
        if iv is not None and iv.shape[0] == n:

import time
import numpy as np
from scipy.optimize import differential_evolution

def _compute_c5(arr):
    """C5 for a given discretisation."""
    n = arr.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(arr, 1.0 - arr, mode="full")) * dx

def _project_to_feasible(v, target_sum):
    """
    Euclidean projection of a vector onto the set
    {x | 0 ≤ x_i ≤ 1 , sum x_i = target_sum}.
    """
    lo, hi = -1.0, 1.0
    for _ in range(40):  # binary search for Lagrange multiplier
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        if proj.sum() > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(v + lo, 0.0, 1.0)
