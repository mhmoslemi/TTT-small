def simulated_annealing(h, target_sum, time_limit):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    curr_h = h.copy()
    curr_val = best_val
    start = time.time()
    # SA parameters
    T_start = 0.001
    T_end = 1e-8
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T_start * (1 - frac) + T_end * frac
        # pick i,j
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)
        # compute feasible delta
        d_low = max(-curr_h[i], curr_h[j] - 1.0)
        d_high = min(1.0 - curr_h[i], curr_h[j])
        if d_low >= d_high:
            continue
        delta = np.random.uniform(d_low, d_high)
        # apply
        curr_h[i] += delta
        curr_h[j] -= delta
        new_val = compute_c5(curr_h)
        # acceptance
        if new_val < curr_val or np.random.rand() < np.exp(-(new_val - curr_val) / max(T, 1e-12)):
            curr_val = new_val
            # update best
            if new_val < best_val:
                best_val = new_val
                best_h = curr_h.copy()
        else:
            # revert
            curr_h[i] -= delta
            curr_h[j] += delta
    return best_h, best_val
```

import time, random, math
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _enforce_sum_and_bounds(h, target):
    """Clip to [0,1] and adjust to hit exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Greedy redistribution
    n = len(h)
    # increase if diff>0, decrease otherwise
    for _ in range(2 * n):
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            caps = 1.0 - h
            i = np.argmax(caps)
            if caps[i] <= 0.0:
                break
            delta = min(caps[i], diff)
            h[i] += delta
            diff -= delta
        else:
            caps = h
            i = np.argmax(caps)
            if caps[i] <= 0.0:
                break
            delta = min(caps[i], -diff)
            h[i] -= delta
            diff += delta
    return np.clip(h, 0.0, 1.0)

def _random_feasible(n, target):
    """Generate a random vector in [0,1]^n with exact sum `target`."""
    h = np.random.rand(n)
    diff = target - h.sum()
    # Simple redistribution loop
    while abs(diff) > 1e-12:
        if diff > 0.0:
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            i = random.choice(idxs.tolist())
            cap = 1.0 - h[i]
            delta = min(cap, diff)
            h[i] += delta
            diff -= delta
        else:
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            i = random.choice(idxs.tolist())
            cap = h[i]
            delta = min(cap, -diff)
            h[i] -= delta
            diff += delta
    return h

def _simulated_annealing(start_h, target_sum, time_limit):
    """SA that preserves sum via pairwise transfers."""
    n = len(start_h)
    curr = start_h.copy()
    best = curr.copy()
    curr_val = _compute_c5(curr)
    best_val = curr_val
    start = time.time()
    # SA schedule parameters
    T0, T1 = 1e-3, 1e-8
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + T1 * frac
        # pick distinct indices
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)
        # feasible delta range that keeps both entries in [0,1]
        d_low = max(-curr[i], curr[j] - 1.0)
        d_high = min(1.0 - curr[i], curr[j])
        if d_low >= d_high:
            continue
        delta = np.random.uniform(d_low, d_high)
        # apply move
        curr[i] += delta
        curr[j] -= delta
        new_val = _compute_c5(curr)
        improve = new_val < curr_val
        if improve or random.random() < math.exp(-(new_val - curr_val) / max(T, 1e-12)):
            curr_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = curr.copy()
                if best_val <= 0.3808:      # early exit condition
                    break
        else:
            # revert
            curr[i] -= delta
            curr[j] += delta
    return best, best_val

def _refine_with_slsqp(h, target_sum, max_iter=1500):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    try:
        res = minimize(_compute_c5, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
    except Exception:
        return h, _compute_c5(h)
    if not res.success:
        return h, _compute_c5(h)
    refined = _enforce_sum_and_bounds(res.x, target_sum)
    return refined, _compute_c5(refined)

def _upsample_h(old_h, new_n):
    old_n = len(old_h)
    # map points linearly onto new grid (treated as step function)
    x_old = np.arange(old_n) / old_n
    x_new = np.arange(new_n) / new_n
    new_h = np.interp(x_new, x_old, old_h, left=old_h[0], right=old_h[-1])
    return _enforce_sum_and_bounds(new_h, new_n / 2.0)

    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    total_budget = float(budget_s)

    # Initial vector
    try:

    phase1_time = total_budget * 0.55

        best_h, n0 / 2.0, phase1_time)
    if val_candidate < best_val:
        best_h, best_val = h_candidate, val_candidate

    # optional local refinement
    refined_h, refined_val = _refine_with_slsqp(best_h, n0 / 2.0)
    if refined_val < best_val:

    elapsed = time.time() - start_time
    if best_val > 0.3808 and (total_budget - elapsed) > total_budget * 0.30:
