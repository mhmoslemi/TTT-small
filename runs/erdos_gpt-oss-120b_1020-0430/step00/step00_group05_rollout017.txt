def project_to_sum_and_bounds(x, target):
    # Clip x to box after adding shift lambda
    # Use bisection to find lambda
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = lam
        else:
            low = lam
    y = np.clip(x + (low + high) / 2.0, 0.0, 1.0)
    # Due to numerical issues, sum may be slightly off. Adjust by small redistribution
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        # adjust smallest non-bound entries
        # find indices where 0 < y < 1
        free = np.where((y > 0) & (y < 1))[0]
        if len(free) > 0:
            # distribute diff equally among free
            y[free] += diff / len(free)
            # clip again after distribution
            y = np.clip(y, 0.0, 1.0)
        else:

import numpy as np
import time
import math

def _project_sum_bounds(x: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of x onto the set {y | 0≤y≤1, sum(y)=target}."""
    # Find lambda such that sum(clip(x+lambda,0,1)) = target.
    lo = -np.max(x) - 1.0      # ensures all x+lam ≤ 0
    hi = 1.0 - np.min(x) + 1.0  # ensures all x+lam ≥ 1
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

    # Fix tiny residual error (at most 1e-12) by adjusting a free coordinate.
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = np.where((y > 0.0) & (y < 1.0))[0]
        if free.size == 0:               # all at bounds – adjust any index safely
            i = 0
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
        else:
            i = free[0]
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y


def _c5_value(h: np.ndarray) -> float:
    """Compute C5 for a discrete step function h."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _worst_shift(h: np.ndarray):
    """Return (shift, correlation array) where shift gives the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    i_max = int(np.argmax(corr))
    shift = i_max - (n - 1)           # integer lag: positive => right shift
    return shift, corr


def _gradient_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of C_i (the worst shift) w.r.t. h."""
    n = h.shape[0]
    grad = np.ones_like(h)
    if shift >= 0:
        s = shift
        grad[: n - s] -= h[s:]          # term h[j+shift]
        grad[s:]   -= h[: n - s]        # term h[j-shift]
    else:
        s = -shift
        grad[s:]   -= h[: n - s]
        grad[: n - s] -= h[s:]
    return grad


def _random_pair_move(h: np.ndarray, delta: float):
    """Perform a random ε‑move on a pair of coordinates preserving the sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # Feasible ε interval so that both remain in [0,1].
    low  = max(-h[i], h[j] - 1.0)
    high = min(1.0 - h[i], h[j])
    if low >= high:
        return h, None   # no feasible move found
    # optionally bound the step size
    bound = delta
    low = max(low, -bound)
    high = min(high, bound)
    if low >= high:
        return h, None
    eps = np.random.uniform(low, high)
    h_new = h.copy()
    h_new[i] += eps
    h_new[j] -= eps
    return h_new, eps


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except NameError:
        # no parent construction – start from a simple balanced binary vector
        n = 200
        h0 = np.zeros(n)
        idx = np.random.choice(n, n // 2, replace=False)
        h0[idx] = 1.0
    else:
        n = h0.shape[0]

    target_sum = n / 2.0
    # enforce exact constraints immediately
    h0 = _project_sum_bounds(h0, target_sum)

    cur_h = h0
    cur_c5 = _c5_value(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # parameters for the stochastic search
    max_delta = 0.3          # max move magnitude for random pair updates
    min_delta = 1e-5
    lr0 = 0.2                # initial learning rate for gradient steps
    T0 = 0.006               # initial temperature for SA
    iteration = 0

    # main optimisation loop
    while time.time() - start < budget_s * 0.96:
        # compute the worst shift and its gradient
        shift, _ = _worst_shift(cur_h)
        grad = _gradient_worst_shift(cur_h, shift)

        # decide on move type
        if np.random.rand() < 0.25:                     # gradient step
            # linear decay of learning rate
            lr = lr0 * (1.0 - iteration / 500000.0)
            if lr < 1e-4:
                lr = 1e-4
            h_candidate = cur_h - lr * grad
            h_candidate = _project_sum_bounds(h_candidate, target_sum)
        else:                                           # random pair move
            # temperature‑controlled max step size
            elapsed = time.time() - start
            frac = elapsed / (budget_s * 0.96)
            delta = max_delta * (1.0 - frac) + min_delta * frac
            h_candidate, _ = _random_pair_move(cur_h, delta)
            if h_candidate is None:
                iteration += 1
                continue  # no feasible random move; try again

        cand_c5 = _c5_value(h_candidate)

        # acceptance rule (simulated annealing)
        if cand_c5 < cur_c5:
            accept = True
        else:
            # temperature linearly decreasing
            T = T0 * (1.0 - frac)
            prob = math.exp(-(cand_c5 - cur_c5) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_h = h_candidate
            cur_c5 = cand_c5

        # keep the best ever seen
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()

        iteration += 1

    # final polish: a few small gradient steps with very tiny learning rate
    for _ in range(200):
        shift, _ = _worst_shift(best_h)
        grad = _gradient_worst_shift(best_h, shift)
        h_candidate = best_h - 0.001 * grad
        h_candidate = _project_sum_bounds(h_candidate, target_sum)
        cand_c5 = _c5_value(h_candidate)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = h_candidate

    # ensure constraints are satisfied exactly before returning
    best_h = _project_sum_bounds(best_h, target_sum)

    final_c5 = _c5_value(best_h)      # precise value for the output
    return best_h, float(final_c5), n
