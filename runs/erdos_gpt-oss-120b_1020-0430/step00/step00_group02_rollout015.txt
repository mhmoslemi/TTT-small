def project_onto_box_simplex(x, target_sum, tol=1e-9, max_iter=50):
    # binary search for lambda
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # adjust residual tiny
    diff = target_sum - y.sum()
    if abs(diff) > tol:
        # distribute diff to interior points
        interior = (y > 0.0 + tol) & (y < 1.0 - tol)

import time
import numpy as np
from scipy.optimize import differential_evolution

# Global state used by the optimizer callbacks
CURRENT_N = None
CURRENT_TARGET_SUM = None
START_TIME = None
BUDGET = None
BEST_SCORE = np.inf
BEST_H = None
BEST_N = None


def _project_onto_box_simplex(x, target_sum, tol=1e-12, max_iter=60):
    """
    Euclidean projection of vector x onto the set {0<=y_i<=1, sum y_i = target_sum}.
    Uses a simple binary search for the Lagrange multiplier.
    """
    lo = -np.max(x) - 1.0   # ensures x+lo <= 0 -> all clipped to 0
    hi = 1.0 - np.min(x) + 1.0   # ensures x+hi >= 1 -> all clipped to 1
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

    # Tiny residual correction (rare)
    diff = target_sum - y.sum()
    if abs(diff) > tol:
        interior = (y > tol) & (y < 1.0 - tol)
        if interior.any():
            y[interior] += diff / interior.sum()
        else:  # all at bounds – distribute uniformly (will be re‑clipped)
            y += diff / len(y)
        y = np.clip(y, 0.0, 1.0)
    return y


def _c5_of_h(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for the discrete vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx


def _de_objective(x):
    """Objective for differential evolution – works on the global CURRENT_*."""
    h = _project_onto_box_simplex(x, CURRENT_TARGET_SUM)
    return _c5_of_h(h)


def _de_callback(xk, _):
    """Called after each DE generation – updates the best solution and respects budget."""
    if time.time() - START_TIME > BUDGET:
        return True
    h = _project_onto_box_simplex(xk, CURRENT_TARGET_SUM)
    c5 = _c5_of_h(h)
    global BEST_SCORE, BEST_H, BEST_N
    if c5 < BEST_SCORE:
        BEST_SCORE = c5
        BEST_H = h.copy()
        BEST_N = CURRENT_N
    return False


def _random_feasible(n):
    """Draw a random point in the feasible polytope."""
    return _project_onto_box_simplex(np.random.rand(n), n / 2.0)


def _adjust_to_exact_sum(h):
    """Final fine‑tuning to satisfy the integral exactly (sum = n/2)."""
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute diff over interior entries (those not at a hard bound)
    interior = (h > 1e-12) & (h < 1.0 - 1e-12)
    if not interior.any():
        interior = np.arange(n)
    else:
        interior = np.where(interior)[0]
    h[interior] += diff / len(interior)
    h = np.clip(h, 0.0, 1.0)
    # recurse if still off (rare)
    if abs(h.sum() - target) > 1e-12:
        return _adjust_to_exact_sum(h)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low C₅ value."""
    global START_TIME, BUDGET, BEST_SCORE, BEST_H, BEST_N, CURRENT_N, CURRENT_TARGET_SUM

    np.random.seed(seed)
    START_TIME = time.time()
    BUDGET = float(budget_s)

    BEST_SCORE = np.inf
    BEST_H = None
    BEST_N = None

    # ------------------------------------------------------------------
    # Determine which discretisations to explore
    # ------------------------------------------------------------------
    n_candidates = []

    # Use the supplied initial construction if present
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        init_vec = np.asarray(init_vec, dtype=float)
        n_candidates.append(len(init_vec))

    # Additional even sizes (larger give more freedom)
    for n in (50, 80, 100, 150, 200):
        if n not in n_candidates:
            n_candidates.append(n)

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for n in n_candidates:
        if time.time() - START_TIME > BUDGET:
            break

        CURRENT_N = n
        CURRENT_TARGET_SUM = n / 2.0

        # If we have a matching‑size initial vector, evaluate it straight away
        if init_vec is not None and len(init_vec) == n:
            h0 = _project_onto_box_simplex(init_vec, CURRENT_TARGET_SUM)
            c0 = _c5_of_h(h0)
            if c0 < BEST_SCORE:
                BEST_SCORE = c0
                BEST_H = h0.copy()
                BEST_N = n

        # ---------- Differential Evolution ----------
        bounds = [(0.0, 1.0)] * n

        # modest population; iterations scaled with size
        if n <= 80:
            maxiter = 500
        elif n <= 150:
            maxiter = 400
        else:
            maxiter = 300
        popsize = 12

        # build a custom initial population that seeds the known construction (if applicable)
        init_pop = np.random.rand(popsize, n)
        if init_vec is not None and len(init_vec) == n:
            init_pop[0] = _project_onto_box_simplex(init_vec, CURRENT_TARGET_SUM)

        try:
            differential_evolution(
                _de_objective,
                bounds,
                strategy="best1bin",
                maxiter=maxiter,
                popsize=popsize,
                tol=1e-6,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=seed,
                callback=_de_callback,
                polish=False,
                updating="deferred",
                init=init_pop,
                disp=False,
            )
        except Exception:
            # callback may have raised True to stop early – ignore
            pass

        # ---------- Simple random search (extra exploration) ----------
        rand_trials = 2000
        for _ in range(rand_trials):
            if time.time() - START_TIME > BUDGET:
                break
            h = _random_feasible(n)
            c = _c5_of_h(h)
            if c < BEST_SCORE:
                BEST_SCORE = c
                BEST_H = h.copy()
                BEST_N = n

    # ------------------------------------------------------------------
    # Fallback – should never happen but keep code safe
    # ------------------------------------------------------------------
    if BEST_H is None:
        n = n_candidates[0] if n_candidates else 2
        BEST_H = np.full(n, 0.5)
        BEST_N = n
        BEST_SCORE = _c5_of_h(BEST_H)

    # Final projection to guarantee the exact integral constraint
    BEST_H = _adjust_to_exact_sum(BEST_H)
    final_score = _c5_of_h(BEST_H)

    return (BEST_H, final_score, BEST_N)
