def refine_with_slsqp(h0):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2
    # convert to variable x
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    # equality constraint sum(x) = target
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
    if res.success:
        return res.x, objective(res.x)
    else:
        return h0, objective(h0)

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (must be top level – no closures)
# ----------------------------------------------------------------------


def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a step function sampled by h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_to_sum(v: np.ndarray, low: float = 0.0, high: float = 1.0, total: float = None) -> np.ndarray:
    """
    Project vector v onto the box [low,high] together with the linear equality
    sum(x)=total (total must be feasible, i.e. low*len(v) <= total <= high*len(v)).
    Uses a simple bisection on the Lagrange multiplier.
    """
    if total is None:
        total = low * len(v)
    lo = -1e6
    hi = 1e6
    for _ in range(80):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, low, high)
        s = np.sum(x)
        if s > total:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, low, high)


def simulated_annealing(
    h0: np.ndarray,
    time_budget: float,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Simple pairwise-exchange SA that respects the sum constraint exactly.
    Returns the best vector found and its C5 value.
    """
    rng = np.random.default_rng(seed)
    n = h0.shape[0]
    target = n / 2.0

    h_curr = h0.copy()
    c_curr = compute_c5(h_curr)

    h_best = h_curr.copy()
    c_best = c_curr

    start = time.time()
    # initial temperature – a small fraction of the gap to a naive bound
    T0 = max(0.02, c_best - 0.35)

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1.0 - frac)

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        hi = h_curr[i]
        hj = h_curr[j]

        # feasible delta interval that keeps both entries inside [0,1]
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue

        # choose delta
        # 10 % of the time make an extreme move (to a bound)
        if rng.random() < 0.10:
            delta = delta_max if rng.random() < 0.5 else delta_min
        else:
            # regular move, size shrinks linearly with time
            factor = max(0.01, 1.0 - frac)
            interval = delta_max - delta_min
            delta = delta_min + rng.random() * factor * interval

        if abs(delta) < 1e-14:
            continue

        h_new = h_curr.copy()
        h_new[i] += delta
        h_new[j] -= delta

        c_new = compute_c5(h_new)

        # Metropolis acceptance
        if c_new < c_curr or rng.random() < math.exp(-(c_new - c_curr) / (T + 1e-15)):
            h_curr = h_new
            c_curr = c_new
            if c_new < c_best:
                h_best = h_new.copy()
                c_best = c_new

    return h_best, c_best


def refine_with_slsqp(h_start: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    A short SLSQP refinement; the objective is the (non‑smooth) max‑correlation,
    but the optimizer often still improves the solution.
    """
    n = h_start.shape[0]
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return float(np.max(corr) * dx)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n

    # give the optimizer a generous iteration limit but stop early if time runs out
    max_iter = max(30, int(time_budget * 20))  # rough scaling: ~20 iters per second
    opts = {"maxiter": max_iter, "ftol": 1e-10, "disp": False}

    try:
        res = minimize(obj, h_start, method="SLSQP", bounds=bounds, constraints=cons, options=opts)
        if res.success:
            h_opt = project_to_sum(res.x, total=target)
            return h_opt, obj(h_opt)
    except Exception:
        pass
    # fall back to the start if anything goes wrong
    return h_start, obj(h_start)

# Main entry point required by the harness
# ----------------------------------------------------------------------


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where
        h_values : list of step values on a uniform grid of n_points samples,
        c5_bound : the achieved upper bound for C5,
        n_points : the number of samples (grid size).
    """
    # ------------------------------------------------------------------
    # Initialise a feasible starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # default resolution if no parent solution is supplied
        h0 = np.full(45, 0.5, dtype=float)

    n = h0.shape[0]
    target = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_sum(h0, total=target)

    # ------------------------------------------------------------------
    # Allocate time: most for simulated annealing, the rest for a quick SLSQP pass
    # ------------------------------------------------------------------
    start_total = time.time()
    sa_budget = budget_s * 0.85
    remain = budget_s - (time.time() - start_total)

    # Simulated annealing
    h_best, c_best = simulated_annealing(h0, sa_budget, seed + 1)

    # Optional refinement if time permits
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining > 0.5:
        h_refined, c_refined = refine_with_slsqp(h_best, remaining)
        if c_refined < c_best:
            h_best, c_best = h_refined, c_refined

    # Final safety projection (ensures exact sum)
    h_best = project_to_sum(h_best, total=target)
    c_best = compute_c5(h_best)

    return (h_best.tolist(), float(c_best), n)
