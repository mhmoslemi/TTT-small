import time
import math
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Simulated‑annealing / hill‑climbing optimizer for the Erdős C5 constant.
    Returns a feasible step function h, its C5 value, and the number of samples.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise a feasible h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
        # enforce sum constraint (tiny numerical tolerance possible)
        target_sum = n / 2.0
        cur_sum = h.sum()
        if not math.isclose(cur_sum, target_sum, rel_tol=1e-12, abs_tol=1e-12):
            # simple rescaling with clipping, then a small correction
            h = np.clip(h * (target_sum / cur_sum), 0.0, 1.0)
            diff = target_sum - h.sum()
            if abs(diff) > 1e-12:
                # distribute the tiny remainder uniformly among non‑boundary entries
                mask = (h > 0.0) & (h < 1.0)
                if mask.any():
                    adjust = diff / mask.sum()
                    h[mask] += adjust
                    h = np.clip(h, 0.0, 1.0)
    else:
        # a simple binary start: half ones, the rest zeros (odd n gets a single 0.5)
        n = 90
        h = np.zeros(n, dtype=np.float64)
        ones = n // 2
        idx = rng.permutation(n)
        h[idx[:ones]] = 1.0
        if n % 2:
            h[idx[ones]] = 0.5

    # ------------------------------------------------------------------
    # auxiliary routine
    # ------------------------------------------------------------------
    def c5_of(vec):
        """return C5 for a given vector (full correlation, max, multiplied by dx)."""
        nloc = vec.shape[0]
        dx = 2.0 / nloc
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        return float(corr.max() * dx)

    best_h = h.copy()
    best_c5 = c5_of(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    n = cur_h.shape[0]
    target_sum = n / 2.0
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # simulated‑annealing loop
    # ------------------------------------------------------------------
    max_iter = 2_000_000               # generous upper bound, will stop by time
    T_start = 0.02                     # initial temperature
    max_step_start = 0.5               # initial maximal delta for a move

    # keep the last correlation to avoid a duplicated compute when possible
    corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
    max_idx = int(corr.argmax())
    shift = max_idx - (n - 1)          # shift that currently attains the maximum

    it = 0
    while it < max_iter:
        # time check (leave a small margin)
        if time.time() - start_time > budget_s * 0.96:
            break

        # temperature and step size schedules (linear decay)
        T = T_start * (1.0 - it / max_iter)
        max_step = max_step_start * (1.0 - it / max_iter) + 1e-3

        # ------------------------------------------------------------------
        # directed move: try to lower the correlation at the worst shift
        # ------------------------------------------------------------------
        # overlapping index range for the current worst shift
        if shift >= 0:
            i_start = 0
            i_end = n - shift
        else:
            i_start = -shift
            i_end = n
        idx_range = np.arange(i_start, i_end)               # indices that participate
        # weights w_i = 1 - h[i+shift] (the factor appearing in the correlation)
        w = 1.0 - cur_h[idx_range + shift]

        # feasible candidates for increasing / decreasing the value
        inc_mask = (cur_h[idx_range] < 1.0 - 1e-12)
        dec_mask = (cur_h[idx_range] > 1e-12)
        inc_idx = idx_range[inc_mask]
        dec_idx = idx_range[dec_mask]

        move_made = False
        if inc_idx.size and dec_idx.size:
            # choose i with small w (good to increase) and j with large w (good to decrease)
            w_inc = w[inc_mask]
            w_dec = w[dec_mask]
            i = inc_idx[np.argmin(w_inc)]
            j = dec_idx[np.argmax(w_dec)]
            hi, hj = cur_h[i], cur_h[j]
            # feasible delta range respecting bounds
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low < high:
                # restrict magnitude
                low = max(low, -max_step)
                high = min(high, max_step)
                if low < high:
                    delta = rng.uniform(low, high)
                    cur_h[i] = hi + delta
                    cur_h[j] = hj - delta
                    new_c5 = c5_of(cur_h)
                    if new_c5 < cur_c5 or rng.random() < math.exp((cur_c5 - new_c5) / (T + 1e-12)):
                        cur_c5 = new_c5
                        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
                        max_idx = int(corr.argmax())
                        shift = max_idx - (n - 1)
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = cur_h.copy()
                        move_made = True
                    else:
                        # revert
                        cur_h[i] = hi
                        cur_h[j] = hj
        # ------------------------------------------------------------------
        # fallback random exchange move (if directed move was impossible or rejected)
        # ------------------------------------------------------------------
        if not move_made:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                hi, hj = cur_h[i], cur_h[j]
                low = max(-hi, hj - 1.0)
                high = min(1.0 - hi, hj)
                if low < high:
                    # limit step size
                    low = max(low, -max_step)
                    high = min(high, max_step)
                    if low < high:
                        delta = rng.uniform(low, high)
                        cur_h[i] = hi + delta
                        cur_h[j] = hj - delta
                        new_c5 = c5_of(cur_h)
                        if new_c5 < cur_c5 or rng.random() < math.exp((cur_c5 - new_c5) / (T + 1e-12)):
                            cur_c5 = new_c5
                            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
                            max_idx = int(corr.argmax())
                            shift = max_idx - (n - 1)
                            if new_c5 < best_c5:
                                best_c5 = new_c5
                                best_h = cur_h.copy()
                        else:
                            cur_h[i] = hi
                            cur_h[j] = hj
        it += 1

    # ------------------------------------------------------------------
    # final short deterministic polishing (pure improvements only)
    # ------------------------------------------------------------------
    for _ in range(5):
        improved = False
        for __ in range(5000):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            hi, hj = best_h[i], best_h[j]
            low = max(-hi, hj - 1.0)
            high = min(1.0 - hi, hj)
            if low >= high:
                continue
            delta = rng.uniform(low, high)
            best_h[i] = hi + delta
            best_h[j] = hj - delta
            new_c5 = c5_of(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                improved = True
            else:
                best_h[i] = hi
                best_h[j] = hj
        if not improved:
            break

    return best_h.tolist(), float(best_c5), n
