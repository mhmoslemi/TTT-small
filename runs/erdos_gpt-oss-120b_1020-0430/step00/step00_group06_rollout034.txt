    n = len(h)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # compute overlaps
        conv = np.correlate(h, 1 - h, mode='full')
        overlaps = conv * dx
        k_max = np.argmax(overlaps)
        # compute shift offset: shift = k_max - (n-1)
        shift = k_max - (n-1)
        # find mismatched indices: i such that i+shift within [0,n), and h[i] > h[i+shift]
        if shift >= 0:
            idx_i = np.arange(0, n-shift)
            idx_j = idx_i + shift
        else:
            shift_abs = -shift
            idx_i = np.arange(shift_abs, n)
            idx_j = idx_i - shift_abs
        diff = h[idx_i] - h[idx_j]  # positive diff indicates h_i larger than counterpart.
        # choose indices where diff>0, also ensure h[idx_i] > 0, h[idx_j] <1.
        candidates = np.where(diff > 1e-8)[0]  # indices in idx_i array
        if candidates.size == 0:
            # pick random pair?
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
        else:
            # choose weighted by diff
            probs = diff[candidates]
            probs = probs / probs.sum()

        max_move = min(h[i], 1 - h[j])

def full_corr(a, b):
    n = len(a)
    m = 2*n -1
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    corr = np.fft.irfft(fa * np.conj(fb), m)
    return corr

def full_corr(a, b):
    n = a.shape[0]
    m = 2*n - 1
    # compute FFT with m length
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    corr = np.fft.irfft(fa * np.conj(fb), m)
    return corr

import numpy as np
import time

def _full_corr(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """linear full correlation a * b (zero‑padded) using FFT."""
    n = a.size
    m = 2 * n - 1
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    return np.fft.irfft(fa * np.conj(fb), m)


def _project_box_sum(z: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto 0≤x≤1 and sum(x)=target_sum (one‑dimensional water‑filling)."""
    lo, hi = -np.max(z) - 2.0, 1.0 - np.min(z) + 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(z + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Maximum overlap for the current step function."""
    conv = _full_corr(h, 1.0 - h)
    return float((conv * dx).max())


def _improve_one_step(h: np.ndarray, dx: float, current_c5: float) -> tuple:
    """Perform a local move that tries to reduce the current worst overlap."""
    n = h.size
    conv = _full_corr(h, 1.0 - h)
    overlaps = conv * dx
    max_idx = overlaps.argmax()
    max_val = overlaps[max_idx]
    shift = max_idx - (n - 1)          # integer shift k

    # indices that actually overlap for this shift
    if shift >= 0:
        i_src = np.arange(0, n - shift)
        i_tgt = i_src + shift
    else:
        s = -shift
        i_src = np.arange(s, n)
        i_tgt = i_src - s

    diff = h[i_src] - h[i_tgt]        # positive diff => source larger than target
    pos = np.where(diff > 1e-12)[0]

    if pos.size:
        probs = diff[pos]
        probs /= probs.sum()
        chosen = np.random.choice(pos, p=probs)
        i = i_src[chosen]
        j = i_tgt[chosen]
        # move at most what would equalise the two entries and respect bounds
        max_move = min(h[i] - h[j], h[i], 1.0 - h[j])
        if max_move > 1e-12:
            delta = max_move * np.random.rand() * 0.5
            h[i] -= delta
            h[j] += delta
    else:
        # no beneficial direction for this shift – try a random tiny transfer
        i, j = np.random.randint(0, n, size=2)
        max_move = min(h[i], 1.0 - h[j])
        if max_move > 1e-12:
            delta = max_move * np.random.rand() * 0.3
            h[i] -= delta
            h[j] += delta

    # enforce feasibility
    h = _project_box_sum(h, n / 2.0)
    new_c5 = _c5_value(h, dx)
    # keep the move only if it does not worsen the max overlap
    if new_c5 <= current_c5:
        return h, new_c5
    # revert
    return h, current_c5

        it = 0
        while time.time() - start_time < budget_s * 0.95:
            # try a directed improvement step
            h, cur_c5 = _improve_one_step(h, dx, cur_c5)

            # keep the global best
            if cur_c5 < best_c5 - 1e-12:
                best_c5, best_h = cur_c5, h.copy()

            it += 1

            # occasional random jitter to escape shallow plateaus
            if it % 800 == 0:
                i, j = np.random.randint(0, N, size=2)
                max_move = min(h[i], 1.0 - h[j])
                if max_move > 1e-12:
                    delta = max_move * np.random.rand() * 0.2
                    h[i] -= delta
                    h[j] += delta
                    h = _project_box_sum(h, target_sum)
                    cur_c5 = _c5_value(h, dx)
                    if cur_c5 < best_c5:
                        best_c5, best_h = cur_c5, h.copy()

            # stop early if the budget is nearly exhausted
            if time.time() - start_time > budget_s * 0.99:
                break

    # final verification & return
    final_c5 = _c5_value(best_h, dx)
    return best_h.tolist(), float(final_c5), N
