import time
import random
import numpy as np

def _c5_value(h):
    """Return C5 for a given h (array of floats in [0,1])."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear (zero‑padded) correlation
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _project_to_sum_and_box(v):
    """
    Euclidean projection of vector v onto the set
        { w : 0 ≤ w_i ≤ 1 , sum w_i = len(v)/2 }.
    Uses a simple bisection on the Lagrange multiplier λ in
        w_i = clip(v_i - λ , 0 , 1).
    """
    n = v.shape[0]
    target = n * 0.5
    # initial clipping
    v = np.clip(v, 0.0, 1.0)

    lo, hi = -2.0, 2.0          # bounds for λ
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction to hit the sum exactly
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        mask = (w > 0.0) & (w < 1.0)
        cnt = int(mask.sum())
        if cnt > 0:
            w[mask] += diff / cnt
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback – distribute onto a single free component
            i = np.argmin(np.abs(w - 0.5))
            w[i] = np.clip(w[i] + diff, 0.0, 1.0)
    return w


def _anneal(h0, max_iters, time_budget):
    """
    Simulated‑annealing style local search.
    Keeps the total sum exactly constant by moving mass between two coordinates.
    """
    start = time.time()
    n = h0.shape[0]
    cur = h0.copy()
    cur_val = _c5_value(cur)
    best = cur.copy()
    best_val = cur_val

    # temperature schedule (linear decay)
    init_temp = max(cur_val * 0.5, 0.02)

    for it in range(max_iters):
        if time.time() - start >= time_budget:
            break

        i, j = random.sample(range(n), 2)
        hi, hj = cur[i], cur[j]

        # admissible delta: add to i, remove from j
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue

        # step size gradually shrinks
        step_factor = 0.5 * (1.0 - it / max_iters)
        delta = random.uniform(delta_min, delta_max) * step_factor

        cur[i] = hi + delta
        cur[j] = hj - delta

        new_val = _c5_value(cur)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # Metropolis acceptance
            temp = init_temp * (1.0 - it / max_iters)
            if temp > 0 and random.random() < np.exp((cur_val - new_val) / temp):
                cur_val = new_val
            else:
                # revert
                cur[i] = hi
                cur[j] = hj

    return best, best_val
