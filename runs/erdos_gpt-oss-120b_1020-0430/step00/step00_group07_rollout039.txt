Pseudo:

def search_periodic(N, max_period=20):
    best_h = None
    best_val = np.inf
    dx = 2.0 / N
    for p in range(2, max_period+1, 2):
        if N % p != 0:
            continue
        half = p // 2
        # generate combos of half positions
        for combo in combinations(range(p), half):
            pattern = np.zeros(p, dtype=np.float64)
            pattern[list(combo)] = 1.0
            h = np.tile(pattern, N // p)
            val = evaluate(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    return best_h, best_val

def propose(h, step):
    i = np.random.randint(N)
    j = np.random.randint(N)
    while i == j:
        j = np.random.randint(N)
