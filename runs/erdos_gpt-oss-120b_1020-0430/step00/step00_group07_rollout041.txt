import numpy as np
import time

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 bound for a given step function h."""
    n = h.size
    dx = 2.0 / n
    # linear correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    """
