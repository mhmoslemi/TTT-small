def simulated_annealing(initial_h, max_time):
    h = initial_h.copy()
    cur_obj = compute_obj(h)
    best_h = h.copy()
    best_obj = cur_obj
    start = time.time()
    n = len(h)
    iter = 0
    while time.time() - start < max_time:
        iter += 1
        # schedule 
        T = max(1e-4, (max_time - (time.time() - start)) / max_time) # linearly decreasing maybe
        # randomly pick indices i!=j
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)
        # compute max feasible delta
        # direction: move from j to i (increase i, decrease j) with probability .5 else opposite
        if np.random.rand() < 0.5:
            # move from j to i
            delta_max = min(1 - h[i], h[j])
            if delta_max <= 0:
                continue
            delta = delta_max * np.random.rand()
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            delta_max = min(1 - h[j], h[i])
            if delta_max <= 0:
                continue

                h_new[i] -= delta
                h_new[j] += delta
                obj = compute_obj(h_new, dx)
                if obj < compute_obj(h, dx) - 1e-9:
                    h = h_new
                    improved = True
        passes += 1
    return h

import time, math, random
import numpy as np

def _compute_obj(h, dx):
    """Maximum overlap integral for a given step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_binary(n):
    """Binary vector with exactly n/2 ones (n is even)."""
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 and small max overlap.

    Returns
    -------
    h_values : np.ndarray
        The best vector found (length = n_points).
    c5_bound : float
        The corresponding overlap constant.
    n_points : int
        Number of discretisation points.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise solution (use provided parent if present)
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        n = init.size
        best_h = init.copy()
    except Exception:
        n = 80
        best_h = _random_binary(n)

    dx = 2.0 / n
    best_obj = _compute_obj(best_h, dx)

    # ------------------------------------------------------------------
    # Simulated‑annealing core
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_obj = best_obj

    start = time.time()
    time_limit = budget_s * 0.95        # keep a safety margin
    T_start = 0.05                      # initial temperature
    steps_since_best = 0
    # probability of taking a full move (delta = delta_max)
    p_full = 0.12

    while time.time() - start < time_limit:
        # pick two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        cand = cur_h.copy()
        if np.random.rand() < 0.5:               # move mass from j to i
            delta_max = min(1.0 - cand[i], cand[j])
            if delta_max <= 0.0:
                continue
            delta = delta_max if np.random.rand() < p_full else np.random.rand() * delta_max
            cand[i] += delta
            cand[j] -= delta
        else:                                      # move mass from i to j
            delta_max = min(1.0 - cand[j], cand[i])
            if delta_max <= 0.0:
                continue
            delta = delta_max if np.random.rand() < p_full else np.random.rand() * delta_max
            cand[i] -= delta
            cand[j] += delta

        new_obj = _compute_obj(cand, dx)

        # temperature schedule (linear cooling)
        elapsed = time.time() - start
        T = max(1e-7, T_start * (1.0 - elapsed / time_limit))

        # Metropolis acceptance
        if new_obj < cur_obj or np.random.rand() < math.exp((cur_obj - new_obj) / T):
            cur_h, cur_obj = cand, new_obj
            if new_obj < best_obj:
                best_h, best_obj = cand.copy(), new_obj
                steps_since_best = 0
            else:
                steps_since_best += 1
        else:
            steps_since_best += 1

        # ------------------------------------------------------------------
        # If we have been stagnant for a while, restart near the best solution
        # ------------------------------------------------------------------
        if steps_since_best > 200_000:
            cur_h = best_h.copy()
            # apply a few small random perturbations
            for _ in range(12):
                a = np.random.randint(0, n)
                b = np.random.randint(0, n - 1)
                if b >= a:
                    b += 1

                cur_h[b] -= delta
