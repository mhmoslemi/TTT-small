import time
import numpy as np

def _raw_overlap(h):
    """Maximum of h * (1-h) correlation (raw, without dx factor)."""
    h = np.asarray(h, dtype=float)
    return np.max(np.correlate(h, 1.0 - h, mode="full"))

def _binarize(vals):
    """Return a balanced 0/1 vector with the same length as vals."""
    n = len(vals)
    m = n // 2
    idx = np.argsort(-vals)  # descending
    h = np.zeros(n, dtype=int)
    h[idx[:m]] = 1
    return h

def _hill_climb_binary(h0, end_time, max_iter=50000):
    """
    Simple simulated‑annealing hill climbing for a binary step function.
    Returns the best vector found and its raw overlap.
    """
    n = len(h0)
    best_h = h0.copy()
    best_raw = _raw_overlap(best_h)
    cur_h = h0.copy()
    cur_raw = best_raw

    # pre‑compute positions of ones / zeros for fast random swaps
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)

    start = time.time()
    for it in range(max_iter):
        if time.time() >= end_time:
            break
        # linear temperature schedule
        t = 1.0 * (1 - it / max_iter) + 1e-4 * (it / max_iter)

        # pick a random 1 and a random 0
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # propose swap
        cur_h[i], cur_h[j] = 0, 1
        new_raw = _raw_overlap(cur_h)

        # accept if improvement or with Metropolis probability
        if new_raw <= cur_raw or np.random.rand() < np.exp(-(new_raw - cur_raw) / (t + 1e-12)):
            cur_raw = new_raw
            # update index lists after an accepted move
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = 1, 0

    return best_h, best_raw

def _continuous_refine(h0, end_time, max_iter=20000):
    """
    Small continuous adjustments while preserving the total integral.
    """
    h = h0.astype(float).copy()
    n = len(h)
    cur_raw = _raw_overlap(h)
    best_h = h.copy()
    best_raw = cur_raw

    for it in range(max_iter):
        if time.time() >= end_time:
            break
        # feasible indices for increase / decrease
        inc_idx = np.where(h < 1.0 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = np.random.choice(inc_idx)
        j = np.random.choice(dec_idx)
        max_delta = min(1.0 - h[i], h[j])
        if max_delta <= 0:

        h[i] += delta
        h[j] -= delta

                best_raw = cur_raw
                best_h = h.copy()
        else:

            h[i] -= delta
            h[j] += delta

    return best_h, best_raw

    if factor == 1:

    return np.repeat(h, factor)

    # choose a modest set of upscale factors (powers of two)
    factors = [1, 2, 4, 8, 16]          # final size = n0 * 16  (≈720 for n0=45)
    cur_h = base_h.copy()
    cur_factor = 1

    for idx, f in enumerate(factors):
        # allocate a fair share of the remaining budget
        elapsed = time.time() - start_time
        remaining = max(0.0, budget_s - elapsed)
        if remaining <= 1.0:
            break
        remaining_factors = len(factors) - idx
        stage_budget = max(5.0, remaining / remaining_factors)

        # up‑sample if needed
        if f != cur_factor:
            cur_h = _upsample(cur_h, f // cur_factor)
            cur_factor = f

        n = len(cur_h)

        # ---- binary hill‑climbing ------------------------------------
        stage_end = time.time() + stage_budget * 0.7   # leave some time for refinement
        best_h_bin, best_raw_bin = _hill_climb_binary(cur_h, stage_end)

        # ---- optional continuous refinement -------------------------
        refine_end = time.time() + stage_budget * 0.3
        best_h_cont, best_raw_cont = _continuous_refine(best_h_bin, refine_end)

        # choose the better of the two for this resolution
        if best_raw_cont < best_raw_bin:
            cur_best_h, cur_best_raw = best_h_cont, best_raw_cont
        else:
            cur_best_h, cur_best_raw = best_h_bin, best_raw_bin

        # update global best if improved
        if cur_best_raw < best_overall_raw:
            best_overall_raw = cur_best_raw
            best_overall_h = cur_best_h.copy()
            best_n = n

        # prepare for next level (use current best as seed)
        cur_h = cur_best_h.copy()

        # early exit if we already beat the target threshold comfortably
        target_raw = 0.1904 * n   # corresponds to C5 ≈ 0.3808
        if best_overall_raw <= target_raw:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final reporting
    # ------------------------------------------------------------------
    c5_bound = best_overall_raw * (2.0 / best_n)

    # Ensure constraints exactly (tiny numerical drift may appear after refinement)
    # Force the sum to be n/2 by a uniform rescaling; then clip to [0,1].
    total = best_overall_h.sum()
    desired = best_n / 2
    if abs(total - desired) > 1e-9:
        scale = desired / total
        best_overall_h = np.clip(best_overall_h * scale, 0.0, 1.0)

    return best_overall_h.tolist(), float(c5_bound), int(best_n)
