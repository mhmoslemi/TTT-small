import time
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] that minimizes
    C5 = max_k ∫ h(x) (1 - h(x+k)) dx,
    approximated by a discrete correlation on n points.

    Returns:
        (h_values, c5_bound, n_points)
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise a binary vector respecting the constraints
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.shape[0]
        init = np.clip(init, 0.0, 1.0)
        target = n // 2                     # required number of ones
        order = np.argsort(-init)           # descending
        h = np.zeros_like(init)
        h[order[:target]] = 1.0
    else:
        n = 200                              # modest size – fast evaluations
        target = n // 2
        h = np.zeros(n, dtype=np.float64)
        h[rng.choice(n, size=target, replace=False)] = 1.0

    # Helper: correlation value (raw, not scaled) and full array
    def corr_raw(vec):
        c = np.correlate(vec, 1.0 - vec, mode='full')
        return c.max(), c

    # ------------------------------------------------------------------
    # Local optimisation – simulated‑annealing style swaps
    # ------------------------------------------------------------------
    cur_raw, cur_corr = corr_raw(h)
    best_raw, best_corr = cur_raw, cur_corr
    best_h = h.copy()

    # indices of ones / zeros for fast random selection
    def indices(v):
        ones = np.flatnonzero(v > 0.5)
        zeros = np.flatnonzero(v <= 0.5)
        return ones, zeros

    ones_idx, zeros_idx = indices(h)

    # temperature schedule (linear)
    init_T = max(1.0, 0.2 * cur_raw)
    final_T = 1e-4

    # restart control
    restart_every = 200_000          # iterations without improvement → fresh start
    last_improve = 0
    it = 0
    max_iters = 10_000_000

    while (time.time() - start_time) < budget_s * 0.97 and it < max_iters:
        # locate the worst shift (the one giving the current max)
        cur_raw, cur_corr = corr_raw(h)
        max_idx = cur_corr.argmax()
        shift = max_idx - (n - 1)               # can be negative

        # --------------------------------------------------------------
        # Choose a 1‑position that actually contributes to this worst shift
        # --------------------------------------------------------------
        i_candidate = None
        if shift >= 0:
            length = n - shift
            bad = (h[:length] > 0.5) & (h[shift:] < 0.5)
            bad_pos = np.flatnonzero(bad)
            if bad_pos.size:
                i_candidate = rng.choice(bad_pos)
        else:
            d = -shift
            length = n - d
            bad = (h[d:] > 0.5) & (h[:length] < 0.5)
            bad_pos = np.flatnonzero(bad)
            if bad_pos.size:
                i_candidate = rng.choice(bad_pos) + d

        if i_candidate is None:
            # fallback – any 1
            i_candidate = rng.choice(ones_idx)

        # pick a 0 somewhere (any zero works)
        j_candidate = rng.choice(zeros_idx)

        # Perform the swap
        h[i_candidate], h[j_candidate] = h[j_candidate], h[i_candidate]

        new_raw, _ = corr_raw(h)

        # Temperature based on elapsed fraction
        frac = (time.time() - start_time) / budget_s
        T = init_T * (1.0 - frac) + final_T * frac

        # Accept if better, otherwise with Metropolis probability
        if new_raw < cur_raw or rng.random() < np.exp(-(new_raw - cur_raw) / (T + 1e-12)):
            # keep the change
            cur_raw = new_raw
            ones_idx, zeros_idx = indices(h)

            # record best solution
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_h = h.copy()
                last_improve = it
        else:
            # revert swap
            h[i_candidate], h[j_candidate] = h[j_candidate], h[i_candidate]

        # ------------------------------------------------------------------
        # Periodic restart if we have been stagnant for a while
        # ------------------------------------------------------------------
        if it - last_improve > restart_every:
            h[:] = 0.0
            h[rng.choice(n, size=target, replace=False)] = 1.0
            cur_raw, _ = corr_raw(h)
            ones_idx, zeros_idx = indices(h)
            last_improve = it

        it += 1

    # ------------------------------------------------------------------
    # Final bound (scale by dx = 2 / n)
    # ------------------------------------------------------------------
    c5_bound = float(best_raw * (2.0 / n))
    return (best_h.tolist(), c5_bound, n)
