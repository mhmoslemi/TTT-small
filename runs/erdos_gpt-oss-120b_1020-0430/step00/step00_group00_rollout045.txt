import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Compute the Erdős C5 value for a step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1‑h) over the full range
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound and the number of samples.

    The algorithm starts from the supplied ``initial_h_values`` (if any),
    then performs a simple hill‑climbing search that preserves the
    integral constraint exactly.
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Acquire a feasible starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        # fallback: 45‑point symmetric block of ones (meets the integral)
        n_fallback = 45
        h = np.zeros(n_fallback, dtype=float)
        ones = n_fallback // 2
        start = (n_fallback - ones) // 2
        h[start:start + ones] = 1.0

    n = h.size
    target_sum = n / 2.0

    # enforce bounds and the integral exactly
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if not np.isclose(cur_sum, target_sum):
        # simple rescaling (keeps shape) then a tiny correction if needed
        if cur_sum > 0:
            h *= target_sum / cur_sum
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            # distribute the remaining mass on the first non‑saturated entry
            for idx in range(n):
                slack_up = 1.0 - h[idx]
                slack_down = h[idx]
                if diff > 0 and slack_up >= diff:
                    h[idx] += diff
                    break
                if diff < 0 and slack_down >= -diff:
                    h[idx] += diff
                    break

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ------------------------------------------------------------------
    # 2.  Simple hill‑climbing (mass transfer between two coordinates)
    # ------------------------------------------------------------------
    time_limit = budget_s - 0.2          # safety margin
    while time.time() - start_time < time_limit:
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # feasible amount that can be moved from i to j
        max_down = best_h[i]               # cannot go below 0
        max_up = 1.0 - best_h[j]           # cannot exceed 1
        max_delta = min(max_down, max_up)
        if max_delta <= 0.0:
            continue
        # random transfer amount (up to the feasible bound)
        delta = np.random.rand() * max_delta
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = _c5_value(h_new)

        # accept only improvements (strict to avoid endless ties)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new

    # ------------------------------------------------------------------
    # 3.  Return the best found solution
    # ------------------------------------------------------------------
    return best_h, float(best_c5), n
