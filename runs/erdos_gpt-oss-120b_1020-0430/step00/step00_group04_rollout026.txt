def repair_vector(v, target):
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    # Distribute diff uniformly among free entries (not at bound)
    if abs(diff) < 1e-12:
        return v
    # Simple iteration
    for _ in range(10):
        mask_low = v < 1.0
        mask_high = v > 0.0
        if diff > 0:
            free = mask_low
        else:
            free = mask_high
        count = free.sum()
        if count == 0:
            break
        delta = diff / count
        # limit delta not to cross bound
        if diff > 0:
            # ensure v + delta <= 1
            delta = min(delta, 1.0 - v[free].max())
        else:
            delta = max(delta, -v[free].min())
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
    return v
```

import time
import math
import random
import numpy as np

def _repair_vector(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp to [0,1] and adjust the sum to target_sum by a simple
    redistribution loop.  The adjustment is tiny, so a few passes are enough."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # free entries are those that can still move in the required direction
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        free = np.where(mask)[0]
        if free.size == 0:
            break
        add = diff / free.size
        # prevent overshoot
        if diff > 0:
            max_add = (1.0 - v[free]).min()
            add = min(add, max_add)
        else:
            max_sub = v[free].min()
            add = max(add, -max_sub)
        v[free] += add
        v = np.clip(v, 0.0, 1.0)
    return v


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute the C5 value for a step function given by the array h."""
    # np.correlate works with float arrays; result length 2*n-1
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a low‑C₅ step function.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values – 1‑D numpy array of the step values on the uniform grid,
        c5_bound – the achieved C₅ value (float),
        n_points – number of discretisation points (int).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    max_time = budget_s * 0.95          # leave a small safety margin

    # ------------------------------------------------------------------
    # Choose a discretisation size.  200 points gives a good trade‑off
    # between resolution and speed.
    # ------------------------------------------------------------------
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0                # because ∫h = 1 ⇒ Σh = n/2

    # ------------------------------------------------------------------
    # Initialise a feasible vector.
    # If the harness supplied an initial construction we start from it,
    # otherwise we create a random balanced binary array.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.shape[0]
        dx = 2.0 / n
        target_sum = n / 2.0
        h = _repair_vector(h, target_sum)
    else:
        # random binary with exactly n/2 ones
        h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    cur_h = h.copy()
    cur_c5 = _c5_value(cur_h, dx)

    # keep the best solution seen so far
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search.
    # Swaps of a 0↔1 pair keep the sum constraint exact.
    # ------------------------------------------------------------------
    # temperature schedule parameters
    T_start = 0.02
    T_end = 1e-6

    while True:
        now = time.time()
        if now - start_time > max_time:
            break
        # linear temperature decay
        progress = (now - start_time) / max_time
        T = T_start * (1.0 - progress) + T_end * progress

        # propose a swap of a 0 and a 1
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if cur_h[i] == cur_h[j]:
            continue                     # not a 0↔1 swap
        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = _c5_value(cur_h, dx)

        # acceptance test
        if new_c5 < cur_c5 or np.random.rand() < math.exp(-(new_c5 - cur_c5) / (T + 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # reject – revert the swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # ------------------------------------------------------------------
    # Final clean‑up: guarantee the constraints exactly (tiny numerical drift
    # may have appeared if a non‑binary start was used).
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_vector(best_h, target_sum)

    final_c5 = _c5_value(best_h, dx)
    return best_h, final_c5, n
