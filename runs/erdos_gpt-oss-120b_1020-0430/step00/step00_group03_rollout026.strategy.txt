     h = np.clip(h,0,1)
     s = np.sum(h)
     if abs(s-35)<1e-4: 
         return h
     # Adjust when sum too high (avoid zeros)
     if s>35:
         # Sort and adjust largest elements first
         indices = np.argsort(h)
         i=0
         while s>35 and i<len(h):
             add = h[i] - 0  # cannot decrease below existing value
             h[i] = 0  # push to bound
             s -= add
             if s<=35: return np.clip(h,0,1)
             i+=1
         # Force entire adjustment on last component
         h[-1] -= s -35
         return np.clip(h,0,1)
     else:  # too low
         # Find smallest elements to raise first
         indices = np.argsort(-h)
         i=0
         while s<35 and i<len(h):
             cap = 1 - h[i]  # available increase
             fill = np.min(35-s, cap+1e-7)
             h[i] += np.clip(fill,0,cap)
             s += fill
             i+=1
         # Final adjustment to meet exactly
         h[-1] += 35 - s
         return np.clip(h,0,1)
