import time
import numpy as np

# ------------------------------------------------------------
# projection onto the box [0,1] intersected with the hyperplane sum = target
def _project_box_simplex(z, target):
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(z - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(z - lam, 0.0, 1.0)


# ------------------------------------------------------------
# value of the objective C5 for a given step function
def _c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# ------------------------------------------------------------
# gradient of a smoothed max_{k} I(k)  where I(k)=∫h(1−h(·+k))
def _grad_smoothed_max(h, dx, alpha, shifts):
    # I(k) values
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    I = corr
    m = I.max()
    w = np.exp(alpha * (I - m))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    n = h.size
    g = np.zeros_like(h)
    # loop over all shifts
    for ws, s in zip(w, shifts):
        if ws == 0.0:
            continue
        if s >= 0:                               # forward shift
            if s < n:
                g[: n - s] += ws * (1.0 - h[s:])   # (1 - h[i+s])
                g[s:]     -= ws * h[: n - s]      # - h[i-s]
        else:                                     # negative shift
            t = -s
            if t < n:
                g[t:]     += ws * (1.0 - h[: n - t])  # (1 - h[i+s]) where i+s = i-t
                g[: n - t] -= ws * h[t:]               # - h[i-s] where i-s = i+t
    return g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # ---- initial vector -------------------------------------------------
    try:

    dx = 2.0 / n
    shifts = np.arange(-n + 1, n, dtype=int)

    # ensure feasibility of the starting point
    h0 = _project_box_simplex(h0, target_sum)

    cur = h0.copy()
    best = cur.copy()
    best_val = _c5_value(best, dx)

    start = time.time()
    max_iters = 200000          # an upper bound; loop will exit by time
    alpha_start, alpha_end = 5.0, 80.0

    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > budget_s - 0.5:
            break

        # schedules
        frac = it / max_iters
        if frac > 1.0:
            frac = 1.0
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lr = 0.5 * (1.0 - frac) + 0.01

        # gradient step
        g = _grad_smoothed_max(cur, dx, alpha, shifts)
        nxt = cur - lr * g

        # occasional jitter to escape plateaus
        if it % 1000 == 0 and it > 0:
            nxt += np.random.normal(scale=0.02, size=n)

        nxt = _project_box_simplex(nxt, target_sum)

        val = _c5_value(nxt, dx)

        if val < best_val:
            best_val = val
            best = nxt.copy()
            no_improve = 0
        else:
            no_improve += 1

        # if stuck, re‑seed near the current best

            cur = _project_box_simplex(cur, target_sum)
            no_improve = 0

        cur = nxt

    # final projection (tiny numerical drift may appear)
    best = _project_box_simplex(best, target_sum)
    final_val = _c5_value(best, dx)
