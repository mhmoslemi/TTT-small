def project_bounds_sum(x, target):
    # clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # then adjust sum
    s = x.sum()
    diff = target - s
    # adapt using water-filling:
    # if diff >0, need to increase some entries not at 1
    # if diff <0, need to decrease some entries not at 0
    # We'll iterate adjust equally among eligible indices
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = x < 1.0
            # number of eligible
            if not np.any(idx):
                break
            inc = diff / np.count_nonzero(idx)
            # ensure not exceed 1
            inc = np.minimum(inc, 1.0 - x[idx])
            if np.all(inc == 0):
                break
            x[idx] += inc
        else:
            idx = x > 0.0
            if not np.any(idx):
                break
            dec = diff / np.count_nonzero(idx) # diff negative, so dec negative
            dec = np.maximum(dec, -x[idx]) # can't go below 0
            if np.all(dec == 0):
                break
            x[idx] += dec
        s = x.sum()
        diff = target - s
    # final clip just in case
    x = np.clip(x, 0.0, 1.0)
    # after final clipping sum may be slightly off, adjust small residual by adding to first non-saturated entry
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-8:
        # adjust first element within bounds
        i = 0
        while i < len(x):
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
                if abs(diff) < 1e-12: break
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
                if abs(diff) < 1e-12: break
            i += 1
    return x

import time
import numpy as np

# ---------- helper functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dz = 1.0 - h
    m = 2 * n
    # correlation using FFT (real signals)
    fh = np.fft.rfft(h, m)
    fz = np.fft.rfft(dz, m)
    corr = np.fft.irfft(fh * np.conj(fz), m)
    max_corr = np.max(corr[: 2 * n - 1])
    return max_corr * (2.0 / n)


def project_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clamp to [0,1] and adjust to meet the exact sum `target`."""
    v = np.clip(v, 0.0, 1.0)
    eps = 1e-12
    for _ in range(200):  # enough for convergence
        s = v.sum()
        diff = target - s
        if abs(diff) < eps:
            break
        if diff > 0:
            idx = v < 1.0 - eps
            if not np.any(idx):
                break
            inc = diff / np.count_nonzero(idx)
            # don't overshoot any bound
            inc = np.minimum(inc, (1.0 - v[idx]))
            v[idx] += inc
        else:
            idx = v > eps
            if not np.any(idx):
                break
            dec = diff / np.count_nonzero(idx)  # diff negative
            dec = np.maximum(dec, -v[idx])
            v[idx] += dec
    # final tiny correction (if still off)
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    diff = target - s
    if abs(diff) > eps:
        # distribute the residual on the first non‑saturated entry
        for i in range(len(v)):
            if diff > 0 and v[i] < 1.0:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
            elif diff < 0 and v[i] > 0.0:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
            if abs(diff) < eps:
                break
    return v


def propose_move(h: np.ndarray, step_frac: float, target_sum: float) -> np.ndarray | None:
    """Create a new feasible vector.
    With 5 % probability performs a global perturbation,
    otherwise transfers a small amount of mass between two entries.
    """
    n = len(h)
    # occasional global random move
    if np.random.rand() < 0.05:
        noise = np.random.normal(scale=step_frac, size=n)
        h_new = h + noise
        return project_bounds_sum(h_new, target_sum)

    i, j = np.random.choice(n, 2, replace=False)
    if np.random.rand() < 0.5:
        # move from i to j
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            return None
        delta = np.random.uniform(0.0, max_transfer * step_frac)
        if delta <= 0:
            return None
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        # move from j to i
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0:
            return None
        delta = np.random.uniform(0.0, max_transfer * step_frac)
        if delta <= 0:
            return None
        h_new = h.copy()
        h_new[j] -= delta
        h_new[i] += delta
    return h_new


def optimize_h(initial: np.ndarray, target_sum: float, time_budget: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style search for a low C₅ value."""
    h_cur = project_bounds_sum(initial.copy(), target_sum)
    cur_val = compute_c5(h_cur)

    best_h = h_cur.copy()
    best_val = cur_val

    start = time.time()
    max_temp, min_temp = 0.12, 0.001
    max_step, min_step = 0.5, 0.02

    while True:
        if time.time() - start >= time_budget:
            break
        # linear cooling based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_budget
        temp = max_temp * (1 - frac) + min_temp
        step = max_step * (1 - frac) + min_step

        h_new = propose_move(h_cur, step, target_sum)
        if h_new is None:
            continue
        new_val = compute_c5(h_new)
        delta = new_val - cur_val

        if new_val < best_val:
            best_h, best_val = h_new.copy(), new_val

        if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
            h_cur, cur_val = h_new, new_val

    # final safety projection
    best_h = project_bounds_sum(best_h, target_sum)
    best_val = compute_c5(best_h)
    return best_h, best_val


def upscale(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a higher resolution followed by projection."""
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False) + (2.0 / old_n) / 2.0
    x_new = np.linspace(0, 2, new_n, endpoint=False) + (2.0 / new_n) / 2.0
    h_up = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    target = new_n / 2.0
    # simple scaling then projection
    if h_up.sum() != 0:
        h_up = h_up * (target / h_up.sum())
    h_up = project_bounds_sum(h_up, target)
    return h_up

    """Return (h_values, c5_bound, n_points) with C₅ as small as possible."""
    np.random.seed(seed)
    start_total = time.time()

    # ------------------------------------------------------------------
    # starting vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: half‑ones, half‑zeros shuffled
        n0 = 42

        h0[: n0 // 2] = 1.0
        np.random.shuffle(h0)

    # explore a few resolutions (keeping within the time budget)
    factors = [1, 3, 5]            # 42, 126, 210 points (all ≤ 1000)

    remaining_factors = len(factors)
    for f in factors:

        n = n0 * f
        if n > 1000:

        # build initial vector for this resolution
        if f == 1:
            h_start = h0.copy()
        else:

        time_left = budget_s - (time.time() - start_total)
        if time_left <= 0:

        # keep a safety margin for later loops
        sub_budget = time_left * 0.85 / remaining_factors

    # pick the best candidate (fallback to the smallest one if none)
    if candidates:
