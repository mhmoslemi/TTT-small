def project_capped_simplex(v, target, max_iter=100, tol=1e-9):
     lo = -np.max(v) - 1.0  # ensure v+lo <=0
     hi = 1 - np.min(v) + 1.0  # ensure v+hi >=1
     for _ in range(max_iter):
         mid = (lo + hi) / 2.0
         x = np.clip(v + mid, 0, 1)
         s = np.sum(x)
         if abs(s - target) < tol:
             return x
         if s > target:
             hi = mid
         else:
             lo = mid
     return np.clip(v + (lo+hi)/2.0, 0, 1)

def project_bounds_sum(v, target):
    # Clip to [0,1] for now
    v_clipped = np.clip(v, 0, 1)
    # If sum of clipped equals target, done.
    s = v_clipped.sum()
    if abs(s - target) < 1e-9:
        return v_clipped
    # Otherwise need to adjust values that are not at bound to achieve sum.
    # Use lambda method:
    # Solve for lambda: sum_i clip(v_i + lambda, 0, 1) = target.
    lo = -np.max(v) - 1  # small enough
    hi = 1 - np.min(v) + 1
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0, 1)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0, 1)

def compute_max_overlap(h):
    # Compute correlation
    conv = np.correlate(h, 1 - h, mode='full')

def compute_gradient_max_shift(h):
    conv = np.correlate(h, 1 - h, mode='full')
    k = np.argmax(conv) # index of max
    # Convert index to shift: With mode='full', length 2n-1:
    # Index offset = n - 1 corresponds to shift 0; positive indices > n-1 correspond to shift positive.
    shift = k - (len(h) - 1)  # shift can be negative
    # We'll treat shift as integer; but maybe there are multiple same max; use first.
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        # For i in [0, n-1-shift], term includes h_i * (1 - h_{i+shift})
        # derivative w.r.t h_i: (1 - h_{i+shift})
        # derivative w.r.t h_{i+shift}: -h_i
        i_range = np.arange(0, n - shift)
        g[i_range] += (1 - h[i_range + shift])
        g[i_range + shift] -= h[i_range]
    else:
        # shift negative: shift = -d
        d = -shift
        # Overlap with i from d to n-1: term h_i * (1 - h_{i-d})
        i_range = np.arange(d, n)
        g[i_range] += (1 - h[i_range - d])
        g[i_range - d] -= h[i_range]
    return g, shift, conv

import numpy as np
import time
import math

def project_bounds_sum(v, target, tol=1e-9, max_iter=60):
    # Clip initial to bounds
    x = np.clip(v, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) <= tol:
        return x
    # Binary search for lambda
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    return x

def max_overlap(h):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * (2.0 / n)

def gradient_of_worst(h):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    g = np.zeros_like(h)
    if shift >= 0:
        end = n - shift
        i = np.arange(end)
        g[i] += (1.0 - h[i + shift])
        g[i + shift] -= h[i]
    else:
        d = -shift
        i = np.arange(d, n)
        g[i] += (1.0 - h[i - d])
        g[i - d] -= h[i]
    return g, shift, conv

def random_feasible(n, target):
    v = np.random.rand(n)
    v = v * (target / v.sum())
    return project_bounds_sum(v, target)

# ---------- projection onto the box [0,1] with a fixed sum ----------
def _proj_capped_simplex(v, target, tol=1e-9, max_iter=60):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    x = np.clip(v, 0.0, 1.0)
    if abs(x.sum() - target) <= tol:
        return x
    lo = -np.max(v) - 1.0          # makes v+lo <= 0
    hi = 1.0 - np.min(v) + 1.0     # makes v+hi >= 1
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

# ---------- objective (max overlap) ----------
def _c5_value(h):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)

# ---------- gradient of the worst (max) shift ----------
def _worst_shift_gradient(h):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(conv.argmax())
    shift = idx - (n - 1)               # integer shift, can be negative
    g = np.zeros_like(h)

    if shift >= 0:
        end = n - shift
        i = np.arange(end)
        g[i] += (1.0 - h[i + shift])
        g[i + shift] -= h[i]
    else:
        d = -shift
        i = np.arange(d, n)
        g[i] += (1.0 - h[i - d])
        g[i - d] -= h[i]
    return g, shift, conv

# ---------- generate a random feasible vector ----------
def _random_feasible(n, target):
    v = np.random.rand(n)
    v = v * (target / v.sum())
    return _proj_capped_simplex(v, target)

# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.95          # safety margin

    # ---- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h0 = _proj_capped_simplex(h0, n * 0.5)
    else:
        n = 84
        h0 = None
    target_sum = n * 0.5

    # ---- build a small pool of starting points ------------------------
    starts = []
    if h0 is not None:
        starts.append(h0)
        for _ in range(4):
            mix = np.random.rand()
            rnd = np.random.rand(n)
            cand = mix * h0 + (1.0 - mix) * rnd
            starts.append(_proj_capped_simplex(cand, target_sum))
    else:
        for _ in range(6):
            starts.append(_random_feasible(n, target_sum))

    best_h = None
    best_val = np.inf

    # ---- local optimisation on each start -----------------------------
    for base in starts:
        h = base.copy()
        cur_val = _c5_value(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= 0.3808:
                return best_h.tolist(), best_val, n

        it = 0
        while time.time() - start_time < time_limit:
            it += 1
            grad, _, _ = _worst_shift_gradient(h)
            step = 0.8 / math.sqrt(it)          # diminishing step
            h_new = h - step * grad
            h_new = _proj_capped_simplex(h_new, target_sum)
            new_val = _c5_value(h_new)

            if new_val <= cur_val:
                h, cur_val = h_new, new_val
                if new_val < best_val:
                    best_val, best_h = new_val, h_new.copy()
                    if best_val <= 0.3808:
                        return best_h.tolist(), best_val, n
            else:
                # occasional uphill move (simulated‑annealing style)
                temp = 0.01 + 0.001 * it
                if np.random.rand() < math.exp(-(new_val - cur_val) / temp):
                    h, cur_val = h_new, new_val

            # small random tweak every few hundred iterations to shake out of plateaus
            if it % 300 == 0:
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * 0.1
                # keep both indices inside [0,1]
                delta = min(delta, h[i])          # don't go below 0 at i
                delta = min(delta, 1.0 - h[j])    # don't exceed 1 at j
                h[i] -= delta
                h[j] += delta
                # re‑project only to fix possible tiny sum drift
                h = _proj_capped_simplex(h, target_sum)

    # ---- fallback (should rarely happen) -------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_val = _c5_value(best_h)

    return best_h.tolist(), best_val, n
