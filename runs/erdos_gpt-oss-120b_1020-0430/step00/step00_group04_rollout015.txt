import numpy as np, random, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    # adjust using simple fill: compute diff, add to first non-boundary element
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # find index where moving diff doesn't violate bounds
    for idx in range(len(h)):
        if diff > 0:
            max_inc = 1.0 - h[idx]
            inc = min(diff, max_inc)
            h[idx] += inc
            diff -= inc
        else:
            max_dec = h[idx]
            dec = min(-diff, max_dec)
            h[idx] -= dec
            diff += dec
        if abs(diff) < 1e-12:
            break
    # In rare case diff not zero due to all bounds, we just ignore small diff.
    return h

def initial_from_interpolation(base, n):
    # base: array length m

    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    h_new = project_sum(h_new, target)
    return h_new

import numpy as np
import random
import time
import math


def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with zero padding (np.correlate uses the linear convention)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def _project_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clamp to [0,1] and then adjust the total mass to exactly *target*."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # greedy redistribution – modify the first entries that can still move
    for i in range(v.size):
        if diff > 0:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        if abs(diff) < 1e-12:
            break
    return v


def _interpolate(base: np.ndarray, n: int) -> np.ndarray:
    """Linearly interpolate a base vector to a longer length *n*."""
    m = base.size
    xb = np.linspace(0.0, 2.0, m, endpoint=False)
    xn = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(xn, xb, base)
    h = _project_sum(h, n / 2.0)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a step function h on [0,2] that (hopefully) improves the known bound
    for the Erdős minimum overlap problem constant C5.

    The returned tuple is (h_values, c5_bound, n_points):
        h_values – numpy array of shape (n_points,)
        c5_bound – the achieved C5 value (float)
        n_points – number of samples (int)
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Choose a starting discretisation and initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        # pick a modestly larger grid (3× the original, at least 120 points)
        n = max(120, base.size * 3)
        h = _interpolate(base, n)
    else:
        n = 120
        h = np.random.rand(n)
        h = _project_sum(h, n / 2.0)

    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 2.  Record the current best solution
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _c5_value(best_h)

    cur_h = h.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # 3.  Simulated‑annealing style pairwise “mass‑transfer” search
    # ------------------------------------------------------------------
    start_time = time.time()
    stagnation = 0          # iterations without improvement
    while True:
        # stop a few seconds before the hard budget limit
        if time.time() - start_time > budget_s - 5.0:
            break

        elapsed = time.time() - start_time
        # linear annealing of step size (from 1 → 0.01)
        step_frac = max(0.01, 1.0 - elapsed / (budget_s - 5.0))

        # ---- propose a transfer between two coordinates ----------------
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random direction: increase i, decrease j  OR  vice‑versa
        if random.random() < 0.5:
            up, down = i, j
        else:
            up, down = j, i

        max_up = 1.0 - cur_h[up]     # how much we can add to “up”
        max_down = cur_h[down]      # how much we can remove from “down”
        max_delta = min(max_up, max_down)
        if max_delta <= 0.0:
            continue

        delta = max_delta * step_frac * random.random()
        if delta <= 0.0:
            continue

        new_h = cur_h.copy()
        new_h[up] += delta
        new_h[down] -= delta

        new_score = _c5_value(new_h)

        # ---- acceptance ------------------------------------------------
        if new_score < cur_score:
            cur_h, cur_score = new_h, new_score
            stagnation = 0
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
        else:
            # occasional uphill moves (temperature decays to ~0)
            temperature = max(1e-6, 0.01 * (1.0 - elapsed / (budget_s - 5.0)))
            if random.random() < math.exp(-(new_score - cur_score) / temperature):
                cur_h, cur_score = new_h, new_score
                stagnation = 0
            else:
                stagnation += 1

        # ---- restart when stuck ---------------------------------------
        if stagnation > 30000:
            # jitter around the best solution found so far
            jitter = np.random.randn(n) * 0.02
            cand = best_h + jitter
            cand = _project_sum(cand, target_sum)
            cur_h, cur_score = cand, _c5_value(cand)
            stagnation = 0

    # ------------------------------------------------------------------
    # 4.  Final clean‑up: enforce the exact integral constraint
    # ------------------------------------------------------------------
    best_h = _project_sum(best_h, target_sum)
    best_score = _c5_value(best_h)

    return best_h, best_score, n
