import time
import math
import numpy as np

# ---------- helper functions ----------

def proj_capped_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the set
    {x | 0 <= x_i <= 1, sum(x) = s }.
    Uses a binary search for the Lagrange multiplier.
    """
    n = v.size
    if s <= 0:
        return np.zeros(n)
    if s >= n:
        return np.ones(n)

    low = np.min(v) - 1.0          # ensures all clipped to 1 when lambda <= low
    high = np.max(v)               # ensures all clipped to 0 when lambda >= high

    for _ in range(50):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        sumx = x.sum()
        if sumx > s:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def evaluate_c5(h: np.ndarray, dx: float) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx  (discretised)"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def gradient_at_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """
    Sub‑gradient of the correlation at a given lag:
    ∂/∂h_i  Σ_j h_j (1‑h_{j+lag})  = (1‑h_{i+lag}) - h_{i‑lag}
    (indices out of range contribute 0).
    """
    n = h.size
    g = np.zeros_like(h)

    if lag >= 0:
        # term (1 - h[i+lag]) for i=0..n‑lag‑1
        g[: n - lag] = 1.0 - h[lag:]
        # term -h[i‑lag] for i=lag..n‑1
        g[lag:] -= h[: n - lag]
    else:
        L = -lag
        # term (1 - h[i‑L]) for i=L..n‑1
        g[L:] = 1.0 - h[: n - L]
        # term -h[i+L] for i=0..n‑L‑1
        g[: n - L] -= h[L:]
    return g


# ---------- main optimisation routine ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ----- problem size -----
    # Use a finer discretisation (200 points) – more freedom → tighter bound.
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0

    # ----- initial construction (if supplied) -----
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        # upscale by linear interpolation to the chosen resolution
        x_small = np.linspace(0.0, 2.0, base.size, endpoint=False)
        x_big   = np.linspace(0.0, 2.0, n, endpoint=False)
        h_initial = np.interp(x_big, x_small, base)
        h_initial = np.clip(h_initial, 0.0, 1.0)
        h_initial = proj_capped_simplex(h_initial, target_sum)
    else:
        h_initial = None

    # ----- bookkeeping -----
    best_h = None
    best_val = float('inf')
    start_time = time.time()
    time_margin = 0.25          # keep a small safety margin

    # ----- helper for a quick evaluation of a candidate -----
    def maybe_update_best(candidate, val):
        nonlocal best_h, best_val
        if val < best_val:
            best_val = val
            best_h = candidate.copy()

    # ----- main search loop (multiple restarts) -----
    while time.time() - start_time < budget_s - time_margin:
        # start point for this restart
        if h_initial is not None and rng.random() < 0.3:
            h = h_initial.copy()

            h = proj_capped_simplex(rng.random(n), target_sum)

            if now - start_time > budget_s - time_margin:

            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            hi, hj = h[i], h[j]
            delta_low  = max(-hi,          hj - 1.0)
