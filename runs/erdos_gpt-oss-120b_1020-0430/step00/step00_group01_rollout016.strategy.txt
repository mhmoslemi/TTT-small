           return constraint(x, n_points)
       res = minimize(lambda x: objective(x, n_points), 
                      h_initial, method='SLSQP',
                      bounds=bounds,
                      constraints={'type': 'eq', 'fun': cons},
                      options={'maxiter': max_iter, 'ftol': 1e-8})
       if res.success:
           return res.x, res.fun
       else:
           # Fall back to last feasible point
           h_feas = h_initial.copy()
           return h_feas, compute_C5(h_feas, n_points)
