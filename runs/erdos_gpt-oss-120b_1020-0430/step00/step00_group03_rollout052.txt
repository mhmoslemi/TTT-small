def random_swap_move(h):
    n = len(h)
    i = rng.integers(n)
    # ensure h[i] > 0
    while h[i] <= 1e-12:
        i = rng.integers(n)
    j = rng.integers(n)
    while h[j] >= 1 - 1e-12:
        j = rng.integers(n)
    max_delta = min(h[i], 1 - h[j])
    if max_delta <= 0:
        return h  # no change
    delta = rng.uniform(0, max_delta)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

import time
import numpy as np

def _evaluate_c5(h, dx):
    """Return C5 for a given step function h."""
    # correlation of h with (1-h), full mode
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _proj_box_simplex(v, target):
    """
    Euclidean projection of vector v onto the set
    {h | 0 <= h_i <= 1, sum(h) = target}.
    Uses a simple bisection on the offset tau such that
    h_i = clip(v_i - tau, 0, 1) and sum h = target.
    """
    lo = np.min(v) - 1.0   # guarantees h_i = 1 for all i when tau = lo
    hi = np.max(v) + 1.0   # guarantees h_i = 0 for all i when tau = hi
    for _ in range(60):
        tau = (lo + hi) * 0.5
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) * 0.5
    h = np.clip(v - tau, 0.0, 1.0)
    # tiny adjustment to hit the target exactly (helps for rounding)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the small residue uniformly on non‑boundary entries
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _random_swap_move(h, rng):
    """Return a new vector where a random amount is moved from i to j."""
    n = h.size
    # pick i with h_i > 0
    idx_i = rng.integers(n)
    attempts = 0
    while h[idx_i] <= 1e-12 and attempts < 20:
        idx_i = rng.integers(n)
        attempts += 1
    # pick j with h_j < 1
    idx_j = rng.integers(n)
    attempts = 0
    while h[idx_j] >= 1.0 - 1e-12 and attempts < 20:
        idx_j = rng.integers(n)
        attempts += 1
    # if unable to find suitable pair, return a copy
    if h[idx_i] <= 1e-12 or h[idx_j] >= 1.0 - 1e-12:
        return h.copy()
    max_delta = min(h[idx_i], 1.0 - h[idx_j])
    if max_delta <= 0.0:
        return h.copy()
    delta = rng.random() * max_delta
    new_h = h.copy()
    new_h[idx_i] -= delta
    new_h[idx_j] += delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        # fall back to a random feasible vector
        n = 70
        target = n / 2.0
        h0 = rng.random(n)
        h0 = _proj_box_simplex(h0 * (target / h0.sum()), target)

    dx = 2.0 / n
    target_sum = n / 2.0

    # current best solution
    best_h = _proj_box_simplex(h0, target_sum)
    best_val = _evaluate_c5(best_h, dx)

    # current working solution (starts from best)
    cur_h = best_h.copy()
    cur_val = best_val

    # algorithm parameters
    start_time = time.time()
    max_iters = 2_000_000          # generous upper bound
    patience = 5000                # steps without improvement before a perturbation
    no_improve = 0
    step0 = 0.5                    # base step size for gradient moves

    for it in range(1, max_iters + 1):
        # time check
        if time.time() - start_time > budget_s * 0.97:
            break

        # decide which move to use
        if rng.random() < 0.2:   # gradient‑based move 20% of the time
            # compute the worst lag for the current vector
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            max_idx = np.argmax(corr)
            lag = max_idx - (n - 1)

            # sub‑gradient of the active lag (scaled without dx)
            g = np.zeros(n, dtype=float)
            if lag >= 0:
                L = lag
                # part where h_i appears as the first factor
                g[:n - L] += (1.0 - cur_h[L:])
                # part where h_i appears as the second factor
                g[L:] -= cur_h[:n - L]
            else:
                L = -lag
                # equivalently treat a negative shift
                g[L:] += (1.0 - cur_h[:n - L])
                g[:n - L] -= cur_h[L:]

            # diminishing step size
            step = step0 / np.sqrt(it)
            cand = cur_h - step * g
            cand = _proj_box_simplex(cand, target_sum)
            cand_val = _evaluate_c5(cand, dx)
        else:
            # simple random swap move (preserves sum and bounds)
            cand = _random_swap_move(cur_h, rng)

    # enforce the sum exactly (tiny numerical drift)
    best_h = _proj_box_simplex(best_h, target_sum)

    return best_h, best_val, n
