def compute_gradient(h, L):
    n = h.shape[0]
    g = np.zeros_like(h)
    if L >= 0:
        if L < n:
            g[:n-L] += (1 - h[L:])
            g[L:] -= h[:n-L]
    else:
        k = -L
        if k < n:
            g[k:] += (1 - h[:n-k])
            g[:n-k] -= h[k:]
    return g
```

import time, math, random
import numpy as np

def _project_capped_simplex(v, target_sum, max_iter=60, eps=1e-9):
    """Project v onto the set {x: 0<=x<=1, sum(x)=target_sum}."""
    lo = np.min(v) - 1.0          # ensures sum(x) >= target_sum
    hi = np.max(v)                # ensures sum(x) <= target_sum
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
        if hi - lo < eps:
            break
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _overlaps_and_lag(h):
    """Return raw overlaps (without dx) and the lag attaining the max."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    max_idx = conv.argmax()
    lag = max_idx - (len(h) - 1)
    return conv, lag


def _gradient_weighted(h, weights, lags):
    """Weighted gradient = sum_{L} w_L * grad_O_L."""
    n = h.size
    g = np.zeros_like(h)
    for w, L in zip(weights, lags):
        if w == 0.0:
            continue
        if L >= 0:
            if L < n:
                g[:n - L] += w * (1.0 - h[L:])
                g[L:] -= w * h[:n - L]
        else:
            k = -L
            if k < n:
                g[k:] += w * (1.0 - h[:n - k])
                g[:n - k] -= w * h[k:]
    return g

    # adaptive parameters
    alpha0 = 0.12
    beta_max = 35.0

    while time.time() - start_time < budget_s * 0.95:
        # current overlaps
        conv, lag = _overlaps_and_lag(h)
        cur_val = conv.max() * dx
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
            if best_val <= 0.3808:
                break
        else:
            no_improve += 1

        # weighted (smooth) max parameters
        beta = min(beta_max, 5.0 + iteration * 0.002)
        # stabilize exponentials
        O = conv  # raw overlaps (no dx)
        O_max = O.max()
        w = np.exp(beta * (O - O_max))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient for the smooth objective
        g = _gradient_weighted(h, w, lags)

        # step size decays like 1/sqrt(iter)
        alpha = alpha0 / math.sqrt(iteration + 1)
        h_new = h - alpha * g

        # ensure feasibility
        h = _project_capped_simplex(h_new, target_sum)

        # occasional random perturbation to escape plateaus
        if iteration % 7000 == 0 and iteration > 0:
            noise = np.random.uniform(-0.06, 0.06, size=n)
            h = np.clip(h + noise, 0.0, 1.0)
            h = _project_capped_simplex(h, target_sum)

        # restart if stagnated for long
        if no_improve > 25000:
            h = np.random.rand(n)
            h = _project_capped_simplex(h, target_sum)
            no_improve = 0

        iteration += 1

    # final evaluation
    final_conv, _ = _overlaps_and_lag(best_h)
    best_val = final_conv.max() * dx

    return best_h.tolist(), float(best_val), n
