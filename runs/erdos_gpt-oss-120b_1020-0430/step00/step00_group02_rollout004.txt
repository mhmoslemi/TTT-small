    lower = 0.0
    upper = 1.0
    # lambda search
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def gradient_step(h, step):
    # compute worst shift and gradient
    n = len(h)
    total = np.sum(h)
    # compute correlation of h with itself
    # using FFT for efficiency
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, nfft)
    corr = np.fft.irfft(H * np.conj(H), nfft)[:2*n-1]
    # product term for shift s: corr_val = sum_i h_i * h_{i+s}
    # g_s = total - corr_val (since integrates overlapped region) -> maximize g_s
    g = total - corr
    idx = np.argmax(g)
    s = idx - (n - 1)
    # compute gradient for that shift
    grad = np.ones_like(h)
    if s >= 0:
        # i from 0 to n-s-1 have h[i+s]
        grad[:n - s] -= h[s:]
        # i from s to n-1 have h[i-s]

        grad[s_abs:] -= h[:n - s_abs]
        grad[:n - s_abs] -= h[s_abs:]
    # step descent
    new_h = h - step * grad
    # clip and project
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = project_box_sum(new_h, target = n/2)

import time
import numpy as np

def _project_box_sum(v: np.ndarray, target: float, lower: float = 0.0, upper: float = 1.0) -> np.ndarray:
    """Project v onto the set {x: lower<=x_i<=upper, sum x_i = target}."""
    # Bisection on lambda for the projection x_i = clip(v_i - lam, lower, upper)
    lo, hi = -10.0, 10.0  # wide enough to saturate at bounds
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, lower, upper)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo + hi) * 0.5, lower, upper)

    # tiny correction if rounding left a small residual
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if free.any():
            add = diff / free.sum()
            proj[free] += add
            proj = np.clip(proj, lower, upper)
    return proj


def _c5_score(h: np.ndarray) -> float:
    """Return the C5 value for a given step vector h."""
    n = h.shape[0]
    one_minus = 1.0 - h
    # length for linear correlation (next power of two)
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    B = np.fft.rfft(one_minus, n=nfft)
    corr = np.fft.irfft(H * np.conj(B), nfft)[: 2 * n - 1]
    dx = 2.0 / n
    return float(corr.max() * dx)


def _worst_shift_and_grad(h: np.ndarray):
    """Find the shift maximizing the overlap and return that shift and its subgradient."""
    n = h.shape[0]
    total = h.sum()
    # autocorrelation of h with itself (linear)
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    corr = np.fft.irfft(H * np.conj(H), nfft)[: 2 * n - 1]
    # g(s) = total - corr(s)
    g = total - corr
    idx = int(g.argmax())
    shift = idx - (n - 1)                     # shift in index units
    grad = np.ones_like(h)

    if shift >= 0:
        grad[: n - shift] -= h[shift:]
        grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        grad[s:] -= h[: n - s]
        grad[: n - s] -= h[s:]
    return shift, g[idx], grad


def _anneal_optimize(
    h0: np.ndarray,
    max_iters: int,
    init_temp: float,
    final_temp: float,
    deadline: float,
):
    """Simple simulated‑annealing optimizer starting from h0."""
    n = h0.shape[0]
    target = n / 2.0
    cur = h0.copy()
    cur_score = _c5_score(cur)
    best_h, best_score = cur.copy(), cur_score

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # temperature schedule (exponential)
        tau = it / max_iters
        temp = init_temp * ((final_temp / init_temp) ** tau)

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # amount we can move from i to j while staying in [0,1]
        max_move = min(cur[i], 1.0 - cur[j])
        if max_move <= 0.0:
            continue
        # propose a random transfer (scaled by temperature)
        delta = np.random.uniform(-max_move, max_move) * (0.5 + 0.5 * temp / init_temp)
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_score = _c5_score(new)

        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / (temp + 1e-12)):
            cur, cur_score = new, new_score
            if new_score < best_score:
                best_h, best_score = new.copy(), new_score

    return best_h, best_score


def _gradient_refine(h: np.ndarray, steps: int):
    """A few sub‑gradient steps on the worst shift."""
    n = h.shape[0]
    target = n / 2.0
    cur = h.copy()
    for t in range(steps):
        _, _, grad = _worst_shift_and_grad(cur)
        step = 0.4 / np.sqrt(t + 1.0)
        cur = cur - step * grad
        cur = np.clip(cur, 0.0, 1.0)
        cur = _project_box_sum(cur, target)
    return cur
