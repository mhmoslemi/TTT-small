def optimize(N, max_iter=5000, step0=0.1):
    h = np.full(N, 0.5) # uniform
    h = project_to_feasible(h) (but already feasible)
    best_h = h.copy(); best_c = compute_c5(h)
    for it in range(max_iter):
        # compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        max_idx = np.argmax(corr)
        # determine lag ℓ: ℓ = max_idx - (N-1)
        lag = max_idx - (N-1)
        # compute gradient
        shift_left = shift_vec(h, lag)
        shift_right = shift_vec(h, -lag)
        grad = 1.0 - (shift_left + shift_right)   # gradient of max Cℓ
        # step size schedule
        step = step0 / np.sqrt(it+1)
        h_new = h - step * grad
        h_new = project_to_feasible(h_new)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c; best_h = h_new.copy()
        h = h_new
    return best_h, best_c

import numpy as np
import time

def _shift_vec(v, lag):
    """Shift vector v by lag with zero padding. Positive lag => v[i+lag]."""
    n = v.shape[0]
    if lag > 0:
        return np.concatenate((np.zeros(lag, dtype=v.dtype), v[: n - lag]))
    if lag < 0:
        lag = -lag
        return np.concatenate((v[lag:], np.zeros(lag, dtype=v.dtype)))
    return v


def _project_to_feasible(x):
    """Euclidean projection of x onto {0<=h_i<=1, sum h_i = n/2}."""
    n = x.shape[0]
    target = n * 0.5
    lo = -np.max(x) - 1.0   # makes all entries 1 after clipping
    hi = np.max(x) + 1.0    # makes all entries 0 after clipping
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _binary_from_continuous(h):
    """Turn a continuous h into a binary vector with exactly n/2 ones."""
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]  # indices of the largest k entries
    hb = np.zeros(n, dtype=float)
    hb[idx] = 1.0
    return hb


def _random_feasible(n):
    """Generate a random feasible vector (continuous, not binary)."""
    x = np.random.rand(n)
    return _project_to_feasible(x)


def _random_binary(n):
    """Generate a random binary feasible vector (exactly n/2 ones)."""
    hb = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    hb[ones] = 1.0
    return hb


def _optimize_start(h0, start_time, budget_s):
    """Sub‑gradient descent from h0, with simple projections."""
    n = h0.shape[0]
    target_sum = n * 0.5
    h = _project_to_feasible(h0.copy())
    best_h = h.copy()
    best_c = _compute_c5(best_h)

    max_iter = 20000
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.96:
            break

        # correlation and worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)

        # gradient of the worst correlation term
        fwd = _shift_vec(h, lag)
        bwd = _shift_vec(h, -lag)
        grad = (1.0 - fwd) - bwd  # ∂C/∂h

        # step size (simple diminishing schedule)
        step = 0.8 / np.sqrt(it + 1.0)

        # stochastic perturbation every few hundred steps
        if (it % 500) == 0 and it > 0:
            h = _project_to_feasible(h + np.random.normal(scale=0.02, size=n))

        h_new = h - step * grad
        h = _project_to_feasible(h_new)

        c = _compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # try a binary rounding of the best continuous vector
    hb = _binary_from_continuous(best_h)
    c_bin = _compute_c5(hb)
    if c_bin < best_c:
        best_c = c_bin
        best_h = hb

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine base size and candidate sizes
    # ------------------------------------------------------------------
    base_h = globals().get("initial_h_values")
    base_n = len(base_h) if base_h is not None else 60
    cand_n = {base_n}
    for mul in (2, 3, 4):
        nn = base_n * mul
        if nn % 2:
            nn += 1
        cand_n.add(nn)
    # add a few hand‑picked sizes
    cand_n.update([80, 120, 150])
    cand_n = sorted([n for n in cand_n if n > 0])

    best_h = None
    best_c = float("inf")

    # ------------------------------------------------------------------
    # Main search over different resolutions
    # ------------------------------------------------------------------
    for n in cand_n:
        if time.time() - start > budget_s * 0.95:
            break

        # build a pool of starting vectors
        starts = []

        if base_h is not None:
            bh = np.asarray(base_h, dtype=float)
            if n == base_n:
                starts.append(bh)
            else:
                # upscale by simple repetition when possible
                if n % base_n == 0:
                    starts.append(np.repeat(bh, n // base_n))
                else:
                    # linear interpolation to the new grid
                    xs = np.linspace(0, 2, base_n, endpoint=False)
                    xn = np.linspace(0, 2, n, endpoint=False)
                    starts.append(np.interp(xn, xs, bh))

        # few random continuous starts
        for _ in range(5):
            starts.append(_random_feasible(n))

        # few random binary starts (only for even n)
        if n % 2 == 0:
            for _ in range(3):
                starts.append(_random_binary(n))

        # optimise each start
        for h0 in starts:
            h_opt, c_opt = _optimize_start(h0, start, budget_s)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                # early exit if we beat the target record
                if best_c < 0.3808:
                    break
        if best_c < 0.3808:
            break

    # Fallback in the unlikely case no solution was found
    if best_h is None:
        n = 60
        best_h = np.full(n, 0.5)
        best_c = _compute_c5(best_h)

    # final safety projection and evaluation
    best_h = _project_to_feasible(best_h)
    best_c = _compute_c5(best_h)

    return best_h.tolist(), best_c, len(best_h)
