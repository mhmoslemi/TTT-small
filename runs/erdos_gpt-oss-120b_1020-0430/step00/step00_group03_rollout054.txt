import time
import math
import numpy as np

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return C5 for the given step values."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def project_constraints(vec: np.ndarray, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the linear sum constraint."""
    np.clip(vec, 0.0, 1.0, out=vec)
    diff = target_sum - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    # indices that can still move
    free = np.where((vec > 1e-12) & (vec < 1.0 - 1e-12))[0]
    if free.size == 0:
        # forced adjustment on a single entry
        idx = np.argmax(vec) if diff < 0 else np.argmin(vec)
        vec[idx] = np.clip(vec[idx] + diff, 0.0, 1.0)
        return vec

    for _ in range(10):
        inc = diff / free.size
        new_vals = vec[free] + inc
        over = new_vals > 1.0
        under = new_vals < 0.0
        if over.any():
            actual = 1.0 - vec[free[over]]
            vec[free[over]] = 1.0
            diff -= actual.sum()
        elif under.any():
            actual = -vec[free[under]]
            vec[free[under]] = 0.0
            diff -= actual.sum()
        else:
            vec[free] += inc
            diff = target_sum - vec.sum()
            break
        free = np.where((vec > 1e-12) & (vec < 1.0 - 1e-12))[0]
        if free.size == 0:
            break

    if abs(diff) > 1e-10:
        idx = np.argmax(vec) if diff < 0 else np.argmin(vec)
        vec[idx] = np.clip(vec[idx] + diff, 0.0, 1.0)
    return vec


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst correlation term w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[: n - shift] = 1.0 - h[shift:]
            g[shift:] -= h[: n - shift]
    else:
        a = -shift
        g[a:] = 1.0 - h[: n - a]
        g[: n - a] -= h[a:]
    return g


def random_transfer(h: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """Return a new vector after moving some mass from one entry to another."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    if rng.random() < 0.5:
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            return None
        delta = rng.random() * max_delta
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0.0:
            return None
        delta = rng.random() * max_delta
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    return new_h
