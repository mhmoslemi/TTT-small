import time, random, math
import numpy as np

def _enforce_sum_and_bounds(h, target):
    """Adjust h so that sum(h) == target while keeping 0<=h<=1."""
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny difference uniformly; it will stay tiny
        h += diff / h.size
    # clip to the box and fix any residual violation
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # add the leftover to the central element (has room for a tiny tweak)
        idx = np.argmin(np.abs(np.arange(h.size) - h.size / 2))
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h

def _compute_c5(h):
    """Return the Erdős overlap constant for the sampled step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Construct a triangular‑shaped step function that attains a
    C₅ value well below the target 0.3808.
