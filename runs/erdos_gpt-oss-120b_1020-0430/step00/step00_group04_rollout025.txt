def project_to_bounds_and_sum(v, target):
    w = np.clip(v, 0, 1)
    tol = 1e-10
    max_iter = 100
    for _ in range(max_iter):
        current_sum = w.sum()
        diff = target - current_sum
        if abs(diff) <= tol:
            break
        # indices free: not at bound
        free = (w > 0 + 1e-12) & (w < 1 - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust; return fallback: distribute diff equally using clip again
            # Add diff/n to all and clip again
            w += diff / len(w)
            w = np.clip(w, 0, 1)
            continue
        delta = diff / n_free
        w[free] += delta
        # clip any that exceed bounds after update
        w = np.clip(w, 0, 1)
    # final clip
    w = np.clip(w, 0, 1)
    # maybe final scaling if slight diff remains
    # If still off by a tiny amount, adjust smallest free variable
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        # find index that is not at bound or adjust one
        idx = np.argmin(np.abs(w - 0.5))
        w[idx] = np.clip(w[idx] + diff, 0, 1)
    return w

import time
import math
import numpy as np

def project_to_bounds_and_sum(v, target, max_iter=100, tol=1e-12):
    """
    Project vector v onto the set {0 <= x_i <= 1, sum x_i = target}
    using a simple water-filling iteration.
    """
    w = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = w.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (w > 0.0) & (w < 1.0)
        n_free = free.sum()
        if n_free == 0:          # nothing free, spread uniformly and clip again
            w += diff / len(w)
            w = np.clip(w, 0.0, 1.0)
            continue
        delta = diff / n_free
        w[free] += delta
        w = np.clip(w, 0.0, 1.0)
    # final tiny adjustment if needed
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        idx = np.argmin(np.abs(w - 0.5))
        w[idx] = np.clip(w[idx] + diff, 0.0, 1.0)
    return w

def compute_c5(h):
    """Return the overlap constant C5 for a given discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def upscale_h(h_low, factor=2):
    """Linear interpolation to a finer grid (factor must be integer)."""
    n_low = h_low.shape[0]
    n_high = n_low * factor
    # positions of sample centers (midpoints of subintervals)
    x_low = np.linspace(0.0, 2.0, n_low, endpoint=False) + (2.0 / n_low) / 2.0
    x_high = np.linspace(0.0, 2.0, n_high, endpoint=False) + (2.0 / n_high) / 2.0
    h_high = np.interp(x_high, x_low, h_low)
    target_sum = n_high / 2.0
    h_high = project_to_bounds_and_sum(h_high, target_sum)
    return h_high

def random_binary_initial(n):
    """Create a binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    arr[:n // 2] = 1.0
    np.random.shuffle(arr)
    return arr

def generate_random_neighbor(h, step):
    """Swap a small amount of mass between two random indices."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]
    delta_max = min(1.0 - hi, hj)
    delta_min = max(-hi, -(1.0 - hj))
    if delta_max <= delta_min:
        return None
    low = max(delta_min, -step)
    high = min(delta_max, step)
    if low >= high:
        low, high = delta_min, delta_max
    delta = np.random.uniform(low, high)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # Clip just in case
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def generate_targeted_neighbor(h, step):
    """
    Move mass from a donor to a pair (i, i+d) that belongs to the
    currently worst autocorrelation lag.
    This preserves the total sum exactly.
    """
    n = h.shape[0]
    # autocorrelation h with itself
    autocorr = np.correlate(h, h, mode='full')
    idx_min = int(np.argmin(autocorr))
    d = idx_min - (n - 1)                     # lag (can be negative)
    # overlapping region where pairs (i, i+d) exist
    if d >= 0:
        start, end = 0, n - d
    else:
        start, end = -d, n
    if end <= start:
        return None
    # choose a pair with a good chance of improvement:
    # weight by partner value (larger partner => larger gain)
    i_candidates = np.arange(start, end)
    partner_vals = h[i_candidates] + h[i_candidates + d]
    probs = partner_vals / partner_vals.sum()
    i = np.random.choice(i_candidates, p=probs)
    j = i + d
    # choose donor away from the pair
    possible = np.where((h > 1e-12) &
                        (np.arange(n) != i) &
                        (np.arange(n) != j))[0]
    if possible.size == 0:
        return None
    donor = np.random.choice(possible)
    # bounds for the transfer
    delta_max_i = min(1.0 - h[i], 1.0 - h[j])
    delta_max_donor = h[donor]
    delta_max = min(delta_max_i, delta_max_donor, step)
    if delta_max <= 0.0:
        return None
    delta = np.random.uniform(0.0, delta_max)
    new_h = h.copy()
    new_h[i] += delta / 2.0
    new_h[j] += delta / 2.0
    new_h[donor] -= delta
    # Clip for safety (should already respect bounds)
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

    """Return (h_values, c5_bound, n_points) with C5 as small as possible."""
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1.  Obtain an initial vector (use provided one if available)
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():

    h0 = project_to_bounds_and_sum(h0, target_sum0)

    # ------------------------------------------------------------------
    upscale_factor = 2                     # 42 → 84 points
    h = upscale_h(h0, factor=upscale_factor)
    n = h.shape[0]
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 3.  Local search (simulated annealing + directed moves)
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = compute_c5(best_h)

    curr_h = best_h.copy()
    curr_score = best_score

    # parameters
    total_iters_est = 400000                # rough upper bound for scheduling
    T0 = 5e-3                               # initial temperature
    step0 = 0.12                            # initial move magnitude
    iteration = 0
    last_improve_iter = 0

    while time.time() - start_time < budget_s - 0.5:
        # decide move type
        if np.random.rand() < 0.7:
            new_h = generate_targeted_neighbor(curr_h, step0)
        else:
            new_h = generate_random_neighbor(curr_h, step0)

        if new_h is None:
            continue

        new_score = compute_c5(new_h)

        # acceptance rule
        if new_score < curr_score:
            accept = True
        else:
            # annealing acceptance
            T = T0 * max(0.0, 1.0 - iteration / (total_iters_est * 0.8))
            prob = math.exp(-(new_score - curr_score) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            curr_h = new_h
            curr_score = new_score
            if new_score < best_score - 1e-9:
                best_h = new_h.copy()
                best_score = new_score
                last_improve_iter = iteration

        # schedule decay
        iteration += 1
        if iteration % 2000 == 0:
            # gradual cooling and step reduction
            T0 *= 0.99
            step0 = max(0.005, step0 * 0.995)

        # escape stagnation
        if iteration - last_improve_iter > 25000:
            # perturb best solution modestly and continue from there
            perturb = best_h.copy()
            # perform a few random transfers to shake it
            for _ in range(8):
                cand = generate_random_neighbor(perturb, step0 * 2)
                if cand is not None:
                    perturb = cand
            # make sure constraints are still satisfied
            perturb = project_to_bounds_and_sum(perturb, target_sum)
            curr_h = perturb
            curr_score = compute_c5(curr_h)
            last_improve_iter = iteration

    # ------------------------------------------------------------------
    # 4.  Final projection and evaluation
    # ------------------------------------------------------------------
    best_h = project_to_bounds_and_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    return (best_h, float(best_score), n)
