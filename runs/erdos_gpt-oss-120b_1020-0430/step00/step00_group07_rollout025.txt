import numpy as np
import time
import math

def _c5_value(h, dx):
    """Return the overlap integral C5 for a given step function."""
    # np.correlate implements linear (zero‑padded) convolution.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _repair_sum(h, target_sum, eps=1e-12):
    """Adjust h minimally so that its sum equals target_sum while keeping bounds."""
    s = np.sum(h)
    diff = target_sum - s
    while abs(diff) > eps:
        if diff > 0:
            idx = np.where(h < 1.0 - eps)[0]
            if len(idx) == 0:
                break
            inc = min(diff, 1.0 - h[idx[0]])
            h[idx[0]] += inc
        else:
            idx = np.where(h > eps)[0]
            if len(idx) == 0:
                break
            dec = min(-diff, h[idx[0]])
            h[idx[0]] -= dec
        diff = target_sum - np.sum(h)
    # final clip for safety
    np.clip(h, 0.0, 1.0, out=h)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] that minimizes the Erdős C₅ constant.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values – NumPy array with the step heights,
        c5_bound – the achieved upper bound on C₅,
        n_points – number of discretisation points used.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------ #
    # 1.  Initialise a feasible h.
    # ------------------------------------------------------------------ #
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        # default: 80 points, half‑ones half‑zeros shuffled
        n_default = 80
        h = np.zeros(n_default, dtype=np.float64)
        h[: n_default // 2] = 1.0
        np.random.shuffle(h)

    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility (bounds + exact sum)
    np.clip(h, 0.0, 1.0, out=h)
    h = _repair_sum(h, target_sum)

    # ------------------------------------------------------------------ #
    # 2.  Baseline evaluation.
    # ------------------------------------------------------------------ #
    cur_h = h.copy()
    cur_c5 = _c5_value(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------ #
    # 3.  Simulated‑annealing style local search.
    # ------------------------------------------------------------------ #
    start_time = time.time()
    max_iters = 2_000_000          # hard cap; time limit wins.
    it = 0
    eps = 1e-12

    # Temperature schedule (linear cooling)
    T0 = 0.02
    T_min = 1e-5

    # Step size for mass transfers (gradually shrinks)
    step = 0.02
    step_min = 0.001

    # Helper to pick a random feasible source/target pair
    def _pick_pair(arr):
        src_idx = np.where(arr > eps)[0]
        tgt_idx = np.where(arr < 1.0 - eps)[0]
        if src_idx.size == 0 or tgt_idx.size == 0:
            return None, None
        i = np.random.choice(src_idx)
        j = np.random.choice(tgt_idx)
        while i == j:
            j = np.random.choice(tgt_idx)
        return i, j

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        if it >= max_iters:
            break

        # temperature for this iteration
        T = max(T_min, T0 * (1.0 - it / max_iters))

        # -------------------------------------------------------------- #
        # 3a. Random mass transfer between two points.
        # -------------------------------------------------------------- #
        i, j = _pick_pair(cur_h)
        if i is not None:
            delta = min(step, cur_h[i], 1.0 - cur_h[j])
            if delta > eps:
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta

                new_c5 = _c5_value(new_h, dx)
                accept = False
                if new_c5 < cur_c5 - 1e-14:
                    accept = True
                else:
                    # Metropolis acceptance
                    prob = math.exp(-(new_c5 - cur_c5) / (T + eps))
                    if np.random.rand() < prob:
                        accept = True
                if accept:
                    cur_h = new_h
                    cur_c5 = new_c5
                    if new_c5 < best_c5 - 1e-14:
                        best_c5 = new_c5
                        best_h = new_h.copy()

        # -------------------------------------------------------------- #
        # 3b. Every few hundred steps, apply a directed greedy move
        #     based on the currently worst shift.
        # -------------------------------------------------------------- #
        if it % 500 == 0:
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            k_idx = np.argmax(corr)
            d = k_idx - (n - 1)            # integer shift amount

            if d >= 0:
                idxs = np.arange(n - d)                 # i where i+d in range
                partner = cur_h[idxs + d]
                self_idx = idxs
            else:
                idxs = np.arange(-d, n)                  # i where i+d >=0
                partner = cur_h[idxs + d]
                self_idx = idxs

            # Sort overlapping positions by partner value
            order = np.argsort(partner)
            # consider a few extremes to keep it cheap
            for attempt in range(3):
                src = self_idx[order[0]]        # smallest partner
                tgt = self_idx[order[-1]]       # largest partner
                if cur_h[src] <= eps or cur_h[tgt] >= 1.0 - eps:
                    # no room to move, try next extreme
                    order = order[1:-1]
                    if order.size == 0:
                        break
                    continue
                delta = min(step, cur_h[src], 1.0 - cur_h[tgt])
                if delta <= eps:
                    break
                new_h = cur_h.copy()
                new_h[src] -= delta
                new_h[tgt] += delta
                new_c5 = _c5_value(new_h, dx)
                if new_c5 < cur_c5 - 1e-14:
                    cur_h = new_h
                    cur_c5 = new_c5
                    if new_c5 < best_c5 - 1e-14:
                        best_c5 = new_c5
                        best_h = new_h.copy()
                # only one directed move per check
                break

        # gradually shrink step size
        if it % 1000 == 0 and step > step_min:
            step = max(step_min, step * 0.99)

        it += 1

    # ------------------------------------------------------------------ #
    # 4.  Final repair of the best solution (exact sum).
    # ------------------------------------------------------------------ #
    best_h = _repair_sum(best_h, target_sum)

    # Re‑evaluate to be safe
    best_c5 = _c5_value(best_h, dx)

    # Return as required
    return best_h, best_c5, n
