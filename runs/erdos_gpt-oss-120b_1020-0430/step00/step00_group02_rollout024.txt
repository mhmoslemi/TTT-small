def project_onto_bounds_and_sum(v, target, tol=1e-9, max_iter=100):
    low = -1.0
    high = 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        proj = np.clip(v + mid, 0, 1)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s < target:
            low = mid
        else:
            high = mid
    return proj
```

import numpy as np, time
from scipy.optimize import minimize

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def project_to_bounds_and_sum(v, target):
    # water-filling as above
    low = -1.0
    high = 1.0
    for _ in range(100):
        mid = (low + high) / 2.0
        proj = np.clip(v + mid, 0, 1)
        s = proj.sum()
        if abs(s - target) < 1e-12:
            return proj
        if s < target:
            low = mid
        else:
            high = mid
    return proj

def upscale_h(h, new_n):
    old_n = h.size
    old_x = np.linspace(0, 2, old_n, endpoint=False)  # grid points
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    # linear interpolation
    new_h = np.interp(new_x, old_x, h, period=2.0)  # periodic? maybe not important
    # Ensure sum constraint
    target = new_n / 2.0
    new_h = project_to_bounds_and_sum(new_h, target)
    return new_h

def local_optimize(h0, max_iter=200, maxfev=5000):
    n = h0.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
    res = minimize(lambda x: compute_c5(x), h0,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-8, 'disp': False})
    # Ensure feasibility
    x = res.x
    # project onto feasible set just in case
    x = project_to_bounds_and_sum(x, target_sum)
    return x, compute_c5(x)
```

import numpy as np, time
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def project_to_bounds_and_sum(v, target, tol=1e-9, max_iter=100):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < tol:
            return proj
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) / 2.0, 0.0, 1.0)

def upscale_h(h, new_n):
    old_n = h.size
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, h, period=2.0)
    target = new_n / 2.0
    new_h = project_to_bounds_and_sum(new_h, target)
    return new_h

def local_optimize(h0, max_iter=200):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = ({'type':'eq','fun':lambda x: x.sum() - target})
    res = minimize(lambda x: compute_c5(x), h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter':max_iter, 'ftol':1e-8, 'disp':False})
    x = res.x
    x = project_to_bounds_and_sum(x, target)
    return x, compute_c5(x)

def hill_climb_random(h0, rng, max_iters, time_limit, start_time):
    n = h0.size
    best = h0.copy()
    best_val = compute_c5(best)
    for _ in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # choose i to increase, j to decrease
        inc_idx = np.where(best < 1 - 1e-12)[0]
        dec_idx = np.where(best > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i = rng.choice(inc_idx)
        j = rng.choice(dec_idx)
        max_delta = min(1.0 - best[i], best[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_val = compute_c5(cand)
        if cand_val < best_val - 1e-12:
            best = cand
            best_val = cand_val
    return best, best_val

def get_initial():
    try:
        arr = np.array(initial_h_values, dtype=float)
        return arr
    except NameError:
        n = 60
        return np.full(n, 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    h = get_initial()
    n0 = h.size
    target0 = n0 / 2.0
    h = project_to_bounds_and_sum(h, target0)
    best_h, best_val = local_optimize(h, max_iter=150)
    # progressive refinements
    levels = [2, 3, 4]  # scaling factors
    for factor in levels:
        if time.time() - start > budget_s - 30:
            break
        new_n = n0 * factor
        h_up = upscale_h(best_h, new_n)
        # local refinement
        h_opt, val_opt = local_optimize(h_up, max_iter=200)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt
        # random hill climbing within remaining time for this level
        remaining = budget_s - (time.time() - start) - 5
        if remaining > 5:
            h_hc, val_hc = hill_climb_random(best_h, rng, max_iters=20000,
                                             time_limit=remaining, start_time=time.time())
            if val_hc < best_val:
                best_h, best_val = h_hc, val_hc
    # final random hill climbing with whatever time left
    remaining = budget_s - (time.time() - start)
    if remaining > 5:
        best_h, best_val = hill_climb_random(best_h, rng, max_iters=50000,
                                             time_limit=remaining, start_time=time.time())
    # ensure feasibility
    best_h = project_to_bounds_and_sum(best_h, best_h.size / 2.0)
    best_val = compute_c5(best_h)
    return best_h.tolist(), float(best_val), best_h.size

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap integral for step function h."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def project_to_bounds_and_sum(v, target, tol=1e-10, max_iter=100):
    """Project v onto [0,1]^n while forcing sum(v)=target."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s < target:
            low = mid
        else:
            high = mid
    # final fallback
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)

def upscale_h(h, new_n):
    """Linear interpolation of h to a larger grid of size new_n."""
    old_n = h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, h)
    target = new_n / 2.0
    return project_to_bounds_and_sum(new_h, target)

def local_optimize(h0, max_iter=200):
    """SLSQP refinement of h0 under bounds and sum constraint."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    res = minimize(lambda x: compute_c5(x), h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    x = project_to_bounds_and_sum(res.x, target)
    return x, compute_c5(x)

def random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with prescribed sum n/2."""
    v = rng.random(n)
    return project_to_bounds_and_sum(v, n / 2.0)

def hill_climb_random(h0, rng, max_iters, time_limit, start_time):
    """Simple hill‐climbing by moving mass between two coordinates."""
    n = h0.size
    best = h0.copy()
    best_val = compute_c5(best)
    for _ in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        inc = np.where(best < 1.0 - 1e-12)[0]
        dec = np.where(best > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        max_delta = min(1.0 - best[i], best[j])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_val = compute_c5(cand)
        if cand_val < best_val:
            best, best_val = cand, cand_val
    return best, best_val

def get_initial():
    """Retrieve the global initial_h_values or fallback to uniform."""
    try:
        return np.array(initial_h_values, dtype=float)
    except NameError:
        return np.full(60, 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- baseline ---------------------------------------------------------
    h = get_initial()
    n0 = h.size
    h = project_to_bounds_and_sum(h, n0 / 2.0)
    best_h, best_val = local_optimize(h, max_iter=150)

    # ---- progressive refinements -----------------------------------------
    # scaling factors relative to the original resolution
    for factor in (2, 3, 4):
        if time.time() - start_time > budget_s - 30:
            break
        n = n0 * factor
        # upscale current best solution
        h_up = upscale_h(best_h, n)

        # deterministic refinement
        h_opt, val_opt = local_optimize(h_up, max_iter=250)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

        # a few random starts + local refinement
        for _ in range(3):
            rand_h = random_feasible(n, rng)
            h_opt, val_opt = local_optimize(rand_h, max_iter=150)
            if val_opt < best_val:
                best_h, best_val = h_opt, val_opt

        # hill climbing while time permits at this resolution
        remaining = budget_s - (time.time() - start_time) - 5
        if remaining > 5:
            hc_h, hc_val = hill_climb_random(
                best_h, rng, max_iters=20000,
                time_limit=remaining, start_time=time.time()
            )
            if hc_val < best_val:
                best_h, best_val = hc_h, hc_val

    # ---- final exhaustive hill climbing ------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5:
        best_h, best_val = hill_climb_random(
            best_h, rng, max_iters=50000,
            time_limit=remaining, start_time=time.time()
        )

    # final feasibility projection and exact evaluation
    best_h = project_to_bounds_and_sum(best_h, best_h.size / 2.0)
    best_val = compute_c5(best_h)

    return best_h.tolist(), float(best_val), best_h.size
