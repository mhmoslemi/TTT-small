import numpy as np
import time
from scipy.optimize import minimize

def _c5_value(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all lags
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_feasible(y: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of y onto the set
        {x ∈ ℝⁿ : 0 ≤ x_i ≤ 1, Σ x_i = target_sum }.
    Implemented by a bisection on the Lagrange multiplier λ:
        x_i = clip(y_i - λ, 0, 1)
    """
    lo, hi = -10.0, 10.0               # bounds wide enough for λ
    for _ in range(60):                # enough for double precision
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h:[0,2]→[0,1] with ∫h = 1 that minimizes
    C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------
    # initialise from possible parent construction
    init_h = globals().get("initial_h_values")
    if init_h is not None:
        init_h = np.asarray(init_h, dtype=float)
        n0 = init_h.shape[0]
    else:
        n0 = 45
        init_h = np.full(n0, 0.5)

    # candidate discretisations (keep them modest for speed)
    cand_n = [n0]
    if n0 * 2 <= 200:
        cand_n.append(n0 * 2)
    if n0 * 3 <= 200:
        cand_n.append(n0 * 3)
    cand_n = sorted(set(cand_n))

    best_h = None
    best_c5 = np.inf

    # ------------------------------------------------------------
    # helper to build starting points
    def _make_starts(base_vec, n, count):
        target = n / 2.0
        base = _project_to_feasible(base_vec, target)
        starts = [base]
        for _ in range(count - 1):
            pert = base + 0.2 * np.random.randn(n)
            pert = np.clip(pert, 0.0, 1.0)
            starts.append(_project_to_feasible(pert, target))
        return starts

    # ------------------------------------------------------------
    # main optimisation over different resolutions
    for n in cand_n:
        # create a base vector of length n (interpolated from init_h when needed)
        if n == init_h.shape[0]:
            base_vec = init_h.copy()
        else:
            x_old = np.linspace(0.0, 2.0, init_h.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            base_vec = np.interp(x_new, x_old, init_h)

        # number of random starts (more for smaller n)
        nstarts = 8 if n <= 90 else 6
        starts = _make_starts(base_vec, n, nstarts)

        target = n / 2.0
        eq_con = {"type": "eq", "fun": lambda x, t=target: np.sum(x) - t}
        bounds = [(0.0, 1.0)] * n

        for x0 in starts:
            # stop if we run out of time
            if time.time() - start_time > 0.94 * budget_s:

            # SLSQP local optimisation
            try:

                               bounds=bounds,

            except Exception:

            cand_c5 = _c5_value(cand)
            if cand_c5 < best_c5:

        # early exit if time is nearly exhausted
        if time.time() - start_time > 0.94 * budget_s:
            break

    # ------------------------------------------------------------
    # Fallback: guarantee a feasible solution even if optimisation failed
    if best_h is None:
        best_h = _project_to_feasible(init_h, init_h.shape[0] / 2.0)
        best_c5 = _c5_value(best_h)

    # ------------------------------------------------------------
    # Simple stochastic refinement (random local moves)
    n_best = best_h.shape[0]
    target = n_best / 2.0
    step = 0.12

        cand = np.clip(cand, 0.0, 1.0)

    # final exact projection and C5 recomputation
    best_h = _project_to_feasible(best_h, best_h.shape[0] / 2.0)
    best_c5 = _c5_value(best_h)

    return best_h, float(best_c5), int(best_h.shape[0])
