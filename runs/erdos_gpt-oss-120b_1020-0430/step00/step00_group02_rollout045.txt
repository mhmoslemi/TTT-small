def adjust_sum(h, target):
    # simple iterative correction
    total = np.sum(h)
    # small tolerance
    for _ in range(10):
        diff = target - total
        if abs(diff) < 1e-9:
            break
        # capacity to add: where h < 1, we can add up to 1 - h
        # capacity to subtract: where h > 0, we can subtract up to h.
        if diff > 0:
            caps = 1 - h
            cap_total = np.sum(caps)
            if cap_total < 1e-12: break
            inc = diff * caps / cap_total
            h = np.minimum(h + inc, 1.0)
        else:
            caps = h
            cap_total = np.sum(caps)
            if cap_total < 1e-12: break
            dec = -diff * caps / cap_total
            h = np.maximum(h - dec, 0.0)
        total = np.sum(h)
    return h
```

    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    for _ in range(max_iter):
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            caps = 1.0 - h
            total = caps.sum()
            if total == 0:
                break
            inc = diff * caps / total
            h = h + inc
        else:
            caps = h
            total = caps.sum()
            if total == 0:
                break
            dec = -diff * caps / total
            h = h - dec
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
    # final tiny correction if needed
    cur = h.sum()

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def adjust_sum(h: np.ndarray, target_sum: float, max_iter: int = 20) -> np.ndarray:
    """Clip to [0,1] and modify the vector so that its sum equals target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    for _ in range(max_iter):
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            caps = 1.0 - h
            cap_total = caps.sum()
            if cap_total == 0.0:
                break
            inc = diff * caps / cap_total
            h = h + inc
        else:
            caps = h
            cap_total = caps.sum()
            if cap_total == 0.0:
                break
            dec = -diff * caps / cap_total
            h = h - dec
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
    # Tiny uniform correction if needed
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-9:
        if diff > 0.0:
            mask = h < 1.0
            if mask.any():
                add = diff / mask.sum()
                h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 0.0
            if mask.any():
                sub = (-diff) / mask.sum()
                h[mask] = np.maximum(h[mask] - sub, 0.0)
        cur = h.sum()
        if abs(cur - target_sum) > 1e-8:
            factor = target_sum / cur if cur != 0.0 else 0.0
            h = np.clip(h * factor, 0.0, 1.0)
            h = adjust_sum(h, target_sum, max_iter=5)
    return h


def resample_to_target(h_low: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of a low‑resolution step function to a higher resolution."""
    n_low = h_low.shape[0]
    if n_low == n_target:
        return h_low.copy()
    x_low = np.linspace(0.0, 2.0, n_low, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_tgt = np.interp(x_tgt, x_low, h_low)
    target_sum = n_target / 2.0
    return adjust_sum(h_tgt, target_sum)

    n = init_h.shape[0]
    best_h = init_h.copy()
    best_c5 = init_c5
    cur_h = init_h.copy()
    cur_c5 = init_c5
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        it += 1
        # temperature slowly decays
        temp = max(0.001, 0.2 * (0.999 ** it))

        # propose moving a tiny amount of mass from j to i
        i, j = rng.choice(n, 2, replace=False)
        low = max(-cur_h[i], cur_h[j] - 1.0)
        high = min(1.0 - cur_h[i], cur_h[j])
        if high <= low:
            continue
        delta = low + rng.random() * (high - low)
        max_step = temp
        if delta > max_step:
            delta = max_step
        elif delta < -max_step:
            delta = -max_step

        new_h[i] += delta
        new_h[j] -= delta
        if not (0.0 <= new_h[i] <= 1.0 and 0.0 <= new_h[j] <= 1.0):
            continue

        new_c5 = compute_c5(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
        else:

            if rng.random() < prob:
                cur_h, cur_c5 = new_h, new_c5

        if new_c5 < best_c5:
