import time
import numpy as np

# ---------- helpers ----------
def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate gives sum_i h[i]*(1-h[i+shift])
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def project_sum(h, target):
    """Adjust h to have sum == target while staying in [0,1]."""
    # simple proportional redistribution
    eps = 1e-12
    s = h.sum()
    delta = target - s
    if abs(delta) < eps:
        return h
    # iterative redistribution
    for _ in range(10):
        if abs(delta) < eps:
            break
        if delta > 0:
            free = 1.0 - h
            mask = free > eps
            cap = free[mask].sum()
            if cap <= 0:
                break
            inc = delta * free[mask] / cap
            h[mask] += inc
        else:
            free = h
            mask = free > eps
            cap = free[mask].sum()
            if cap <= 0:
                break
            dec = delta * free[mask] / cap
            h[mask] += dec
        delta = target - h.sum()
    # final tiny fix (should be tiny)
    delta = target - h.sum()
    if abs(delta) > eps:
        if delta > 0:
            idx = np.where(h < 1.0 - eps)[0][0]
            h[idx] = min(1.0, h[idx] + delta)
        else:
            idx = np.where(h > eps)[0][0]
            h[idx] = max(0.0, h[idx] + delta)
    return h


def base_seeds(n, rng, base_vec=None):
    """Generate a list of initial vectors of length n."""
    seeds = []
    x = np.linspace(0, 2, n, endpoint=False)

    # 1. upsample existing vector if provided
    if base_vec is not None:
        xb = np.linspace(0, 2, base_vec.size, endpoint=False)
        up = np.interp(x, xb, base_vec)
        seeds.append(up)

    # 2. pure sine (integral =1 by construction)
    a = np.pi / 4.0
    sin_h = a * np.sin(np.pi * x / 2.0)
    seeds.append(sin_h)

    # 3. random binary / half‑integer patterns
    for _ in range(3):
        h = np.zeros(n)
        ones = n // 2
        idx = rng.choice(n, size=ones, replace=False)
        h[idx] = 1.0
        if n % 2:  # need a 0.5 to hit exact average
            rest = list(set(range(n)) - set(idx))
            h[rng.choice(rest)] = 0.5
        seeds.append(h)

    return seeds


def local_search(h0, max_iter, time_limit, start_time, rng):
    """Simple simulated‑annealing style improvement."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    cur = compute_c5(h)
    best_h, best_val = h.copy(), cur
    max_step = 0.25
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        # propose a transfer between two positions
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            delta = rng.uniform(0, max_step)
            delta = min(delta, h[i], 1.0 - h[j])
            if delta <= 0:
                continue
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            delta = rng.uniform(0, max_step)
            delta = min(delta, h[j], 1.0 - h[i])
            if delta <= 0:
                continue
            h_new = h.copy()
            h_new[j] -= delta
            h_new[i] += delta

        new_val = compute_c5(h_new)

        # Metropolis acceptance
        if new_val < cur or rng.random() < np.exp(-(new_val - cur) / (0.05 * max_step + 1e-12)):
            h, cur = h_new, new_val
            if new_val < best_val:
                best_h, best_val = h_new.copy(), new_val

        # anneal step size
        if it % 400 == 0 and it:
            max_step *= 0.85

        # early stop if already good enough
        if best_val <= 0.3808:
            break

    return best_h, best_val
