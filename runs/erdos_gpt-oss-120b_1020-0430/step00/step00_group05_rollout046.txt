def project_box_sum(y, target, tol=1e-9, max_iter=100):
    lo = -np.max(y) - 1.0 # ensures y_i - lo >> 1 => clipped to 1
    hi = np.max(y) + 1.0 # ensures y_i - hi << 0 => clipped to 0

    for _ in range(max_iter):

        if abs(s - target) < tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid

        eta = 0.12 / math.sqrt(it + 0.1)
        y = h - eta * g
        h_new = project_box_sum(y, target_sum)

        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new

    # ----------------------------------------------------------------------
    # Phase 2 – stochastic local improvements
    current_h = best_h.copy()
    current_val = best_val
    while time.time() - start_time < budget_s * 0.99:
        # pick a source (decrease) and a destination (increase)
        src_candidates = np.where(current_h > 1e-12)[0]
        dst_candidates = np.where(current_h < 1.0 - 1e-12)[0]
        if src_candidates.size == 0 or dst_candidates.size == 0:
            break
        src = int(rng.choice(src_candidates))
        possible = dst_candidates[dst_candidates != src]
        if possible.size == 0:
            continue
        dst = int(rng.choice(possible))

        max_move = min(current_h[src], 1.0 - current_h[dst])
        if max_move <= 0:
            continue

        # sometimes move the whole amount (swap), otherwise a small fraction
        if rng.random() < 0.15:
            eps = max_move
        else:
            eps = max_move * rng.random() * 0.5

        new_h = current_h.copy()
        new_h[src] -= eps
        new_h[dst] += eps

        new_val = compute_c5(new_h)
        if new_val < current_val:
            current_h = new_h
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        else:
            # occasional uphill step to escape plateaus
            if rng.random() < 0.001:
                current_h = new_h
                current_val = new_val

    # ----------------------------------------------------------------------
    # Final sanity fix (tiny rounding errors)
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    diff = s - target_sum
    if abs(diff) > 1e-10:
        for i in range(n):
            if 0.0 <= best_h[i] - diff <= 1.0:
                best_h[i] -= diff
                break

    best_val = compute_c5(best_h)
    return (best_h.tolist(), best_val, n)
