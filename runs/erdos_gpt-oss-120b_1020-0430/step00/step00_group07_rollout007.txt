import time
import numpy as np

def _objective(h, dx):
    """Return C5 for a given step vector h."""
    # correlation of h with (1-h) over all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h, target_sum):
    """Clip to [0,1] and enforce the exact sum (up to floating error)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        h[:] = target_sum / len(h)
        return h
    # simple scaling
    h *= target_sum / cur
    h = np.clip(h, 0.0, 1.0)
    # adjust any remaining discrepancy by tweaking the first element(s)
    diff = target_sum - h.sum()
    i = 0
    while abs(diff) > 1e-12 and i < len(h):
        add = np.clip(diff, -h[i], 1.0 - h[i])
        h[i] += add
        diff = target_sum - h.sum()
        i += 1
    return h

    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h
