import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _evaluate(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x) (1-h(x+k)) dx for the discrete vector."""
    n = h.size
    dx = 2.0 / n
    # correlation h with (1-h) (zero padding outside)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_to_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the box [0,1] with the additional
    equality constraint sum(v)=target_sum.  The solution has the form
    clip(v+λ,0,1) with a suitable λ found by bisection.
    """
    lo = -np.max(v) - 2.0          # guarantee all entries <=0
    hi = 1.0 - np.min(v) + 2.0     # guarantee all entries >=1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) * 0.5
    return np.clip(v + λ, 0.0, 1.0)


def _best_shift_contributions(h: np.ndarray, lag: int):
    """
    Return (indices, contributions) for the lag that attains the current
    maximal correlation.  `indices` are the positions i where the term
    h[i]*(1-h[i+lag]) participates.
    """
    n = h.size
    if lag >= 0:
        idx = np.arange(n - lag, dtype=int)
        contrib = h[idx] * (1.0 - h[idx + lag])
        src = idx
        dst = idx + lag
    else:
        shift = -lag
        idx = np.arange(shift, n, dtype=int)
        contrib = h[idx] * (1.0 - h[idx - shift])
        src = idx
        dst = idx - shift
    return src, dst, contrib


# ----------------------------------------------------------------------
# Core local search
# ----------------------------------------------------------------------
def _local_optimize(h0: np.ndarray, time_limit: float, rng: np.random.Generator):
    """
    Simple hill‑climbing that repeatedly tries to move mass from a
    position contributing heavily to the worst lag to its partner.
    Occasionally random perturbations are added to escape plateaus.
    """
    n = h0.size
    target = n / 2.0                     # required integral = 1 -> sum = n/2
    h = h0.copy()
    best_h = h.copy()
    best_val = _evaluate(best_h)

    start = time.time()
    while time.time() - start < time_limit:
        # current correlation (raw, without dx)
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(corr.argmax())
        lag = max_idx - (n - 1)

        src, dst, contrib = _best_shift_contributions(h, lag)
        # pick the index with the largest contribution
        i_local = int(contrib.argmax())
        i = src[i_local]
        j = dst[i_local]

        # propose moving a bit of mass from i to j (reduces the bad term)
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max > 1e-12:
            delta = delta_max * rng.random()
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # fallback: random pair move
            a, b = rng.integers(n, size=2)
            while a == b:
                b = rng.integers(n)
            delta_max = min(1.0 - h[a], h[b])
            if delta_max <= 1e-12:
                continue
            delta = delta_max * rng.random()
            if rng.random() < 0.5:
                h_new = h.copy()
                h_new[a] += delta
                h_new[b] -= delta
            else:
                h_new = h.copy()
                h_new[a] -= delta
                h_new[b] += delta

        val = _evaluate(h_new)
        if val + 1e-12 < best_val:
            best_val = val
            best_h = h_new.copy()
            h = h_new      # keep the improvement
        else:
            # occasional uphill move to avoid local traps
            if rng.random() < 0.001:
                h = h_new

        # tiny random jitter every so often
        if rng.random() < 0.002:
            noise = rng.normal(scale=0.02, size=n)
            h_jit = np.clip(h + noise, 0.0, 1.0)
            h_jit = _project_to_sum(h_jit, target)
            h = h_jit

    return best_h, best_val

# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # number of discretisation points – use the provided one if present
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 45                     # a modest default
        h0 = np.linspace(0.0, 1.0, n, dtype=float)

    # Make sure the starting vector satisfies the constraints exactly
    target_sum = n / 2.0
    h0 = _project_to_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = _evaluate(best_h)

    overall_start = time.time()
    # Allocate most of the budget to several independent local searches
    while time.time() - overall_start < budget_s * 0.94:
        time_left = budget_s - (time.time() - overall_start) - 0.2
        if time_left <= 0.1:
            break

        # start from either the known construction or a random point
        if "initial_h_values" in globals() and rng.random() < 0.5:
            start_vec = np.asarray(initial_h_values, dtype=float)
            start_vec = _project_to_sum(start_vec, target_sum)
        else:
            rand_vec = rng.random(n)
            start_vec = _project_to_sum(rand_vec, target_sum)

        cand_h, cand_val = _local_optimize(start_vec, time_left, rng)

        if cand_val < best_val:
            best_val = cand_val
            best_h = cand_h.copy()

        # early exit if we already beat the target
        if best_val <= 0.3808:
            break

    # final sanity check – ensure constraints are met
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum(best_h, target_sum)

    return (best_h.tolist(), float(best_val), n)
