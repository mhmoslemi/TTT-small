import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_to_sum_bounds(x: np.ndarray, target_sum: float, max_iter: int = 60) -> np.ndarray:
    """
    Project vector x onto the intersection of the box [0,1]^n and the affine
    hyperplane sum(x) = target_sum using a simple binary search for the Lagrange
    multiplier.
    """
    lo, hi = -2.0, 2.0  # wide enough bounds for the shift
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def max_overlap_and_shifts(h: np.ndarray, dx: float):
    """Return (max_value, array_of_shifts) for the current h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = corr.max()
    # shifts are indices offset by (len(h)-1)
    idx = np.where(np.abs(corr - max_val) <= 1e-12)[0]
    shifts = idx - (len(h) - 1)
    return max_val, shifts


def add_shift_gradient(grad: np.ndarray, h: np.ndarray, delta: int, dx: float):
    """Accumulate gradient contribution of a single active shift."""
    N = h.shape[0]
    if delta >= 0:
        k = N - delta
        if k > 0:
            grad[:k] += (1.0 - h[delta:]) * dx
            grad[delta:] += -h[:k] * dx
    else:
        d = -delta
        k = N - d
        if k > 0:
            grad[d:] += (1.0 - h[:k]) * dx
            grad[:k] += -h[d:] * dx


def gradient_active_shifts(h: np.ndarray, shifts: np.ndarray, dx: float) -> np.ndarray:
    """Compute (average) sub‑gradient of the max‑overlap functional."""
    grad = np.zeros_like(h)
    for d in shifts:
        add_shift_gradient(grad, h, int(d), dx)
    if shifts.size > 0:
        grad /= float(shifts.size)
    return grad


def upscale_initial(old_h: np.ndarray, N: int) -> np.ndarray:
    """Linearly interpolate a previous solution to a new grid size N."""
    old_N = old_h.shape[0]
    # points are taken at left edges of intervals for consistency with np.correlate
    old_x = np.linspace(0.0, 2.0, old_N, endpoint=False)
    new_x = np.linspace(0.0, 2.0, N, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    new_h = np.clip(new_h, 0.0, 1.0)
    target = N / 2.0
    new_h = project_to_sum_bounds(new_h, target)
    return new_h

    Returns (best_h, best_val, N) possibly improving the incoming best_sofar.
    """
    start_time = time.time()
    dx = 2.0 / N
    target_sum = N / 2.0
    best_h, best_val, _ = best_sofar

    # ------------------------------------------------------------------
    # Prepare starting points

    starts = []

    # 1) use the provided initial_h_values if present
    init = globals().get("initial_h_values", None)

    # Local optimisation for each start
    # ------------------------------------------------------------------
    for h0 in starts:
        h = h0.copy()
        lr = 0.2  # initial learning rate

        max_iter = 5000
        for it in range(max_iter):

            # record global best
