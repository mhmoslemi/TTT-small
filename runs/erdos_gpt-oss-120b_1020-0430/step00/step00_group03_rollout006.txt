def move_mass(h, ell, eps):
  n = len(h)
  # compute gradient per index: g[i] = (h[i-ell] if i-ell>=0 else 0) + (h[i+ell] if i+ell<n else 0)
  # (the effect of inc h[i] on R(ell))
  g = np.zeros_like(h)
  if ell > 0:
    g[ell:] += h[:n-ell]  # for i >= ell, i-ell within bounds
    g[:n-ell] += h[ell:]  # for i < n-ell, i+ell within bounds
  elif ell < 0:
    # shift negative: same as positive with -ell
    ell = -ell
    g[ell:] += h[:n-ell]
    g[:n-ell] += h[ell:]
  else: # ell == 0
    g = 2*h  # derivative of sum h_i^2.
  # Now choose i_low: where g is smallest and h[i]>eps
  # choose i_high: where g is largest and h[i]<1-eps
  i_low = np.argmin(g + 1e6 * (h < eps))  # penalize if not enough mass
  i_high = np.argmax(g - 1e6 * (h > 1-eps)) # penalize if cannot add
  # Actually more robust: filter candidates.
  low_candidates = np.where(h > eps)[0]
  high_candidates = np.where(h < 1-eps)[0]
  if low_candidates.size == 0 or high_candidates.size == 0:
    return h
  i_low = low_candidates[np.argmin(g[low_candidates])]
  i_high = high_candidates[np.argmax(g[high_candidates])]
  # transfer eps
  h_new = h.copy()
  delta = min(eps, h[i_low], 1 - h[i_high])
  h_new[i_low] -= delta
  h_new[i_high] += delta
  return h_new

def evaluate(h, dx):
    # compute max correlation
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_to_sum_and_bounds(x, target):
    # Simple water-filling binary search for tau.
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(50):
        tau = (lo + hi) / 2
        y = np.clip(x - tau, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(x - ((lo + hi) / 2), 0.0, 1.0)
    # ensure sum close target
    return y

def random_h(n, target):
    # generate uniform random then project
    x = np.random.rand(n)
    return project_to_sum_and_bounds(x, target)

def neighbor_transfer(h, epsilon=0.02):
    n = len(h)
    # choose i where h[i] > epsilon and j where h[j] < 1-epsilon
    idx_pos = np.where(h > epsilon)[0]
    idx_neg = np.where(h < 1 - epsilon)[0]
    if len(idx_pos) == 0 or len(idx_neg) == 0:
        return h.copy()
    i = np.random.choice(idx_pos)
    j = np.random.choice(idx_neg)
    delta = min(epsilon, h[i], 1 - h[j])
    # randomize delta maybe smaller
    delta = delta * np.random.rand()
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def improve_by_gradient(h):
    # produce a slightly improved h via moving mass based on worst shift
    n = len(h)
    target = n/2
    # compute cross = h*(1-h) correlation
    cross = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(cross)
    ell = idx - (n - 1)  # shift index
    # compute gradient of R(ell) w.r.t h: g[i] = (h[i+ell] if i+ell < n else 0) + (h[i-ell] if i-ell>=0 else 0)
    g = np.zeros_like(h)
    if ell > 0:
        g[ell:] += h[:n-ell]   # i with i-ell >=0 => h[i-ell] adds to g[i]
        g[:n-ell] += h[ell:]   # i+ell < n => h[i+ell] adds
    elif ell < 0:
        ell2 = -ell
        g[ell2:] += h[:n-ell2]
        g[:n-ell2] += h[ell2:]
    else:
        g = 2 * h
    # choose i_low with smallest gradient and h[i] > eps
    eps = 1e-6
    low_candidates = np.where(h > eps)[0]
    high_candidates = np.where(h < 1 - eps)[0]
    if len(low_candidates) == 0 or len(high_candidates) == 0:
        return h
    i_low = low_candidates[np.argmin(g[low_candidates])]
    i_high = high_candidates[np.argmax(g[high_candidates])]

    delta = min(0.01, h[i_low], 1 - h[i_high])
    if delta <= 0:
        return h

    h_new[i_low] -= delta
    h_new[i_high] += delta
    return h_new

import time
import numpy as np

def _project_to_sum(x, target):
    """Euclidean projection of x onto {0<=y<=1, sum(y)=target}."""
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(45):
        tau = (lo + hi) * 0.5
        y = np.clip(x - tau, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # fix tiny sum error
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(y < 1.0 - 1e-12)[0]
            if idx.size:
                y[idx[0]] += diff
        else:
            idx = np.where(y > 1e-12)[0]
            if idx.size:
                y[idx[0]] += diff
    return np.clip(y, 0.0, 1.0)


def _evaluate(h, dx):
    """C5 value for a candidate h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _worst_shift(h):
    """Return integer lag that attains max C (worst overlap)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    n = len(h)
    return idx - (n - 1)  # shift index


def _gradient_move(h, ell, eps=0.01):
    """Move a small mass toward indices that improve correlation for lag ell."""
    n = len(h)
    g = np.zeros_like(h)
    if ell > 0:
        g[ell:] += h[: n - ell]
        g[: n - ell] += h[ell:]
    elif ell < 0:
        ell = -ell
        g[ell:] += h[: n - ell]
        g[: n - ell] += h[ell:]
    else:
        g = 2.0 * h
    # candidates respecting bounds
    low = np.where(h > eps)[0]
    high = np.where(h < 1.0 - eps)[0]
    if low.size == 0 or high.size == 0:
        return h
    i_low = low[np.argmin(g[low])]
    i_high = high[np.argmax(g[high])]
    delta = min(eps, h[i_low], 1.0 - h[i_high])
    if delta <= 0:
        return h
    h2 = h.copy()
    h2[i_low] -= delta
    h2[i_high] += delta
    return h2


def _local_improve(h, dx, steps=3):
    """Apply a few gradient moves for the current worst shift."""
    for _ in range(steps):
        ell = _worst_shift(h)
        h2 = _gradient_move(h, ell, eps=0.02)
        # maintain feasibility
        h2 = np.clip(h2, 0.0, 1.0)
        # sum unchanged by gradient move, but clip may disturb slightly
        if abs(h2.sum() - h.sum()) > 1e-9:
            h2 = _project_to_sum(h2, h.sum())
        if _evaluate(h2, dx) < _evaluate(h, dx):
            h = h2
        else:
            break
    return h


def _random_candidate(n, target):
    """Generate a random feasible h."""
    x = np.random.rand(n)
    return _project_to_sum(x, target)


def _mutate(h, target):
    """Generate a neighbour by transferring a random mass."""
    n = len(h)
    # pick two distinct indices
    i, j = np.random.choice(n, 2, replace=False)
    if h[i] < 1e-12 or h[j] > 1.0 - 1e-12:
        return h.copy()
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h.copy()
    delta = max_delta * np.random.rand()
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return h2

    # ----- initialise -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project_to_sum(h0, target)
    except Exception:
        n = 70
        target = n / 2.0
        h0 = _random_candidate(n, target)

    dx = 2.0 / n
    best_h = h0.copy()
    best_score = _evaluate(best_h, dx)

    # small pool of elite solutions
    elite = [(best_score, best_h.copy())]

    # ----- simulated‑annealing loop ----------------------------------
    cur_h = best_h.copy()
    cur_score = best_score
    T = 0.08  # initial temperature
    it = 0
    while True:
        if time.time() - start > budget_s * 0.94:
            break
        it += 1

        # generate neighbour
        if np.random.rand() < 0.7:
            cand = _mutate(cur_h, target)
        else:
            # noise + projection
            sigma = np.random.uniform(0.01, 0.15)
            cand = cur_h + np.random.normal(scale=sigma, size=n)
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_to_sum(cand, target)

        # local refinement
        cand = _local_improve(cand, dx, steps=2)

        cand_score = _evaluate(cand, dx)

        # SA acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_h, cur_score = cand, cand_score

        # update global best
        if cand_score < best_score - 1e-9:
            best_h, best_score = cand.copy(), cand_score
            elite.append((best_score, best_h.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:5]

        # temperature schedule
        if it % 500 == 0:
            T *= 0.92
            # occasional random restart from elite or fresh
            if np.random.rand() < 0.25:
                base = elite[np.random.randint(len(elite))][1]
                cur_h = base.copy()
                cur_score = _evaluate(cur_h, dx)
            if np.random.rand() < 0.1:
                cur_h = _random_candidate(n, target)
                cur_score = _evaluate(cur_h, dx)

        # occasional big mutation from best
        if it % 2000 == 0 and np.random.rand() < 0.3:
            cand = best_h + np.random.normal(scale=0.2, size=n)
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_to_sum(cand, target)
            cand = _local_improve(cand, dx, steps=3)
            cand_score = _evaluate(cand, dx)
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score
                elite.append((best_score, best_h.copy()))
                elite.sort(key=lambda x: x[0])
                elite = elite[:5]

    # final projection & evaluation
    best_h = _project_to_sum(np.clip(best_h, 0.0, 1.0), target)
    best_score = _evaluate(best_h, dx)

    return best_h, float(best_score), n
