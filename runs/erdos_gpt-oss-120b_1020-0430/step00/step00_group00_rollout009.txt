def propose_move(h, step):
    n = len(h)
    i, j = np.random.randint(n, size=2)
    if i == j:
        return None, None, None
    delta = (np.random.rand() * 2 - 1) * step # random in [-step, step]
    # Adjust delta so that do not violate bounds:
    # we will do: new_h_i = h_i - delta; new_h_j = h_j + delta
    # Because sum still constant.
    # So we need 0 <= h_i - delta <= 1, and 0 <= h_j + delta <= 1.
    # So delta_min = -h_i (if delta too negative subtracting negative => adding)
    # Actually constraint:
    #   h_i - delta >= 0 => delta <= h_i
    #   h_i - delta <= 1 => delta >= h_i - 1
    #   h_j + delta >= 0 => delta >= -h_j
    #   h_j + delta <= 1 => delta <= 1 - h_j
    delta_low = max(h_i - 1, -h_j)
    delta_high = min(h_i, 1 - h_j)
    # Clip delta to [delta_low, delta_high]
    delta = np.clip(delta, delta_low, delta_high)
    if abs(delta) < 1e-12:
        return None, None, None
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h, i, j, delta

import time
import numpy as np
from scipy.optimize import minimize

def _c5_raw(h):
    """Raw (unscaled) max correlation sum_i h_i (1-h_{i+ℓ}) for ℓ∈[-n+1,n-1]."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))

def _c5_scaled(h):
    """Scaled C5 value according to the problem statement."""
    n = h.shape[0]
    dx = 2.0 / n
    return _c5_raw(h) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------ #
    # initial feasible vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
    else:
        n = 45
        h0 = np.full(n, 0.5, dtype=np.float64)

    target_sum = n * 0.5  # integral = 1  →  sum(h)*dx = 1  →  sum(h)=n/2

    # project onto hyperplane if needed (very rare)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h0 *= target_sum / cur_sum
        h0 = np.clip(h0, 0.0, 1.0)
        # adjust any remaining tiny error
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-12:
            # distribute over non‑saturated entries
            free = np.where((h0 > 0) & (h0 < 1))[0]
            if free.size:
                h0[free] += diff / free.size
                h0 = np.clip(h0, 0.0, 1.0)
            else:
                h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)

    # ------------------------------------------------------------------ #
    # local stochastic search (pairwise mass transfer)
    cur_h = h0.copy()
    cur_val = _c5_scaled(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    max_step = 0.5               # maximal amount moved between two points
    stagnation = 0                # iterations without improvement
    max_iters_without_improve = 200_000

    # main loop
    while time.time() - start_time < budget_s * 0.95:
        # cooling schedule (linear)
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.95)
        step = max_step * (1.0 - frac) + 1e-4
        # temperature for Metropolis acceptance
        temp = max(1e-4, (1.0 - frac) * 0.08)

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]

        low = max(hi - 1.0, -hj)      # ensure both stay in [0,1]
        high = min(hi, 1.0 - hj)
        if low >= high:
            continue
        delta = (np.random.rand() * 2.0 - 1.0) * step
        delta = np.clip(delta, low, high)
        if abs(delta) < 1e-14:
            continue

        new_h = cur_h.copy()
        new_h[i] = hi - delta
        new_h[j] = hj + delta

        new_val = _c5_scaled(new_h)

        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
                stagnation = 0
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - cur_val) / temp):
                cur_h, cur_val = new_h, new_val

        stagnation += 1
        if stagnation > max_iters_without_improve:
            # perturb the current best and restart local walk
            cur_h = best_h.copy()
            cur_val = best_val
            # small random shake
            shake = (np.random.rand(n) - 0.5) * step * 0.2
            cur_h += shake
            cur_h = np.clip(cur_h, 0.0, 1.0)
            # re‑project onto the sum hyperplane
            diff = target_sum - cur_h.sum()
            free = np.where((cur_h > 0) & (cur_h < 1))[0]
            if free.size:
                cur_h[free] += diff / free.size
                cur_h = np.clip(cur_h, 0.0, 1.0)
            else:
                cur_h[0] = np.clip(cur_h[0] + diff, 0.0, 1.0)
            cur_val = _c5_scaled(cur_h)
            stagnation = 0

    # ------------------------------------------------------------------ #
    # optional refinement with a constrained nonlinear optimizer
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30.0:
        # Combine h and auxiliary variable t into one vector
        def objective(v):
            # minimize t (the raw max correlation)
            return v[-1]

        def eq_cons(v):
            h = v[:-1]
            return np.sum(h) - target_sum

        def ineq_cons(v):
            h = v[:-1]
            t = v[-1]
            # raw correlations (no scaling)
            return np.correlate(h, 1.0 - h, mode="full") - t

        # initial guess
        t0 = _c5_raw(best_h)
        v0 = np.concatenate([best_h, [t0]])

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        constraints = [
            {"type": "eq", "fun": eq_cons},
            {"type": "ineq", "fun": ineq_cons},
        ]

        options = {"maxiter": 2000, "ftol": 1e-12}
        try:
            res = minimize(
                objective,
                v0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options=options,
            )
            if res.success:
                h_opt = res.x[:-1]
                val_opt = _c5_scaled(h_opt)
                if val_opt < best_val:
                    best_h, best_val = h_opt, val_opt
        except Exception:
            # any issue – keep current best
            pass

    # ------------------------------------------------------------------ #
    # final post‑processing: enforce exact feasibility (tiny numerical errors)
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 0) & (best_h < 1))[0]
        if free.size:
            best_h[free] += diff / free.size
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    final_val = _c5_scaled(best_h)
    return best_h, float(final_val), int(n)
