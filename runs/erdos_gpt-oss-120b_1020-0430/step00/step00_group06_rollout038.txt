def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c5 = float('inf')
    # try multiple n values list
    for N in [96, 128, 80, 100]:
        # ensure N even for binary
        if time.time() - start > budget_s - 5:
            break
        # generate initial random balanced binary vector
        h = np.zeros(N, dtype=np.float64)
        # choose N/2 indices to be 1
        ones = rng.choice(N, size=N//2, replace=False)
        h[ones] = 1.0
        # compute c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Simulated annealing loop
        # schedule: total iterations maybe 50000 per N, but adapt to time
        # T start at maybe 0.5 * (some) relative to worst-case diff.
        # We'll use simple exponential cooling.
        T0 = 0.5
        T_min = 1e-4
        # total iterations allowed: maybe 200000 per N but limited by time
        iter = 0
        max_iter = 500000  # generous
        while True:
            if time.time() - start > budget_s - 5:
                break
            if iter >= max_iter:
                break
            # compute temperature
            # simple linear cooling: T = T0 * (1 - iter / max_iter)
            T = max(T0 * (1 - iter / max_iter), T_min)
            # propose random swap

import time
import numpy as np

def _c5_and_shift(h):
    """Return (C5 value, shift giving the max overlap) for binary/real h."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    i = int(np.argmax(corr))
    max_raw = float(corr[i])
    shift = i - (N - 1)          # integer shift (positive → right)
    return max_raw * (2.0 / N), shift


def _c5(h):
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / N)
