def gradient_for_lag(h, lag):

    N = len(h)

    g = np.zeros(N)

    if lag >=0:
        L = N - lag

        g[:L] += 1 - h[lag:]   # forward part

import numpy as np
import time
import random

def _project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n with sum equal to target using a simple
    bisection on the offset tau so that  sum(clamp(x - tau,0,1)) = target."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        tau = (lo + hi) * 0.5
        proj = np.clip(x - tau, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            hi = tau
        else:
            lo = tau
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _corr_bound(h: np.ndarray) -> float:
    """Return the C5 bound for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / h.size))


def _gradient_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the correlation term at a given lag (full‑mode index)."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        L = n - lag
        # forward part: derivative w.r.t. h[i] where i+lag is inside
        g[:L] += 1.0 - h[lag:]
        # backward part: derivative w.r.t. h[i] where i-lag was the first index
        g[lag:] -= h[:L]
    else:
        k = -lag
        L = n - k
        g[k:] += 1.0 - h[:L]
        g[:L] -= h[k:]
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Find a step function h on [0,2] with sum(h)=n/2 that gives a small
    Erdős C5 bound. Returns (h_values, c5_bound, n_points)."""
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise a feasible h.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.size
    else:
        # default to 200 points linear ramp (feasible by construction)
        n = 200
        h = np.arange(n, dtype=np.float64) / (n - 1)

    target_sum = n / 2.0
    h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_c = _corr_bound(best_h)

    # ------------------------------------------------------------------
    # Simple sub‑gradient descent on the worst lag.
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0
    max_iter = 20000  # safety guard
    while time.time() - start_time < budget_s * 0.95 and it < max_iter:
        # current bound and worst lag
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(corr_full.argmax())
        cur_c = float(corr_full[max_idx] * (2.0 / n))
        lag = max_idx - (n - 1)

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # gradient of the worst term
        g = _gradient_for_lag(h, lag)

        # simple diminishing step size
        step = 0.5 / np.sqrt(it + 1.0)

        # propose a new point and project back to the feasible set
        h_new = h - step * g
        h_new = _project_box_sum(h_new, target_sum)

        # evaluate the proposal
        new_c = _corr_bound(h_new)

        # accept if not worse (or with a tiny probability to escape)
        if new_c <= cur_c or random.random() < np.exp(-(new_c - cur_c) / max(step, 1e-8)):
            h = h_new

        # occasional random perturbation if we seem stuck
        if no_improve > 2000:
            # re‑randomise a small fraction of entries
            ind = np.random.choice(n, size=max(1, n // 30), replace=False)
            perturb = np.random.uniform(-0.2, 0.2, size=ind.shape)
            h[ind] = np.clip(h[ind] + perturb, 0.0, 1.0)
            h = _project_box_sum(h, target_sum)
            no_improve = 0

        it += 1

    final_c = _corr_bound(best_h)
    return best_h.tolist(), final_c, n
