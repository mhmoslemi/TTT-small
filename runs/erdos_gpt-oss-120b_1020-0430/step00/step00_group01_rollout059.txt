def search(N, total_time, start_h=None):
   start_time = time.time()
   best_h = None
   best_c5 = float('inf')
   # We'll use "restart" loops until time used
   while time.time() - start_time < total_time:
       # initialize h_f
       if start_h is not None:
            h = start_h.copy()
            # If we haven't used start_h many times, we may keep, else generate random
            start_h = None  # use only once
       else:
            # random binary
            h = np.zeros(N, dtype=np.float64)
            h[:N//2] = 1.0
            np.random.shuffle(h)
       # hill climb
       cur_c5 = compute_c5(h)
       # choose iteration budget for this restart
       sub_budget = min(0.5, (total_time - (time.time() - start_time)) / 2)  # allocate half time for each restart
       iter_start = time.time()
       while time.time() - iter_start < sub_budget:
            # propose random swap
            i = np.random.randint(N)
            if h[i] < 0.5:  # we need i with value 1; else pick new
                continue
            j = np.random.randint(N)
            if h[j] > 0.5:
                continue
            # swap
            h[i], h[j] = 0.0, 1.0   # after swap
            new_c5 = compute_c5(h)
            if new_c5 < cur_c5 - 1e-12:
                cur_c5 = new_c5
            else:
                # revert
                h[i], h[j] = 1.0, 0.0
       # after hill climbing, if cur_c5 better update global best
       if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
   return best_h, best_c5

def run(seed=42, budget_s=1100):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = None

    # duplicate initial function for projecting onto sum constraint
    def project(v, target):
        lo = -1e5
        hi = 1e5
        for _ in range(60):
            mid = (lo + hi) / 2.0
            w = np.clip(v + mid, 0.0, 1.0)
            s = w.sum()
            if s > target:
                hi = mid
            else:
                lo = mid
        return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

    def compute_c5(h):
        N = len(h)
        dx = 2.0 / N
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # function to compute gradient for smoothed max with given alpha
    def gradient_smooth(h, alpha):
        N = len(h)
        # correlation without dx
        corr = np.correlate(h, 1.0 - h, mode='full')
        # compute weights
        exp_a = np.exp(alpha * corr)
        w = exp_a / exp_a.sum()
        # compute gradient
        grad = np.zeros(N, dtype=np.float64)
        shifts = np.arange(-(N-1), N)  # length 2N-1
        # vectorized loops
        for idx, s in enumerate(shifts):
            ws = w[idx]
            if s >= 0:
                if N - s > 0:
                    grad[:N - s] += ws * (1.0 - h[s:])
                if s < N:
                    grad[s:] -= ws * h[:N - s]
            else:
                k = -s
                if N - k > 0:
                    grad[k:] += ws * (1.0 - h[:N - k])
                    grad[:N - k] -= ws * h[k:]
        return grad

    # initial h from initial_h_values if present
    try:
        init_h = initial_h_values
        init_exists = True
    except NameError:
        init_exists = False

    Ns = [80, 120, 200, 300]

        # initialize h

import time, numpy as np

def project_to_sum(v, target):
    # binary search lambda
    lo = -3.0
    hi = 3.0
    # expand bounds if needed
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def gradient_smooth(h, alpha):
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    expa = np.exp(alpha * (corr - maxc))
    w = expa / expa.sum()
    grad = np.zeros(N, dtype=np.float64)
    shifts = np.arange(-(N-1), N)
    for idx, s in enumerate(shifts):
        ws = w[idx]
        if s >= 0:
            if N - s > 0:
                grad[:N - s] += ws * (1.0 - h[s:])
            if s < N:
                grad[s:] -= ws * h[:N - s]
        else:
            k = -s
            if N - k > 0:
                grad[k:] += ws * (1.0 - h[:N - k])
                grad[:N - k] -= ws * h[k:]
    return grad

def upscale_h(h_old, N_new):
    N_old = len(h_old)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, N_new / 2.0)
    return h_new

def project_to_sum(v, target):
    # v is float array
    lo = -5.0
    hi = 5.0
    for _ in range(60):
        lam = (lo + hi) / 2
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

import time
import numpy as np

# ---------- helper functions ----------
def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] while forcing sum(v)=target."""
    lo, hi = -5.0, 5.0        # wide enough to span the whole feasible interval
    for _ in range(60):      # enough for double precision
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _smooth_weights(corr: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max on the correlation vector."""
    maxc = np.max(corr)
    ex = np.exp(alpha * (corr - maxc))
    return ex / ex.sum()


def _grad_smooth(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the smooth‑max objective w.r.t. h."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = _smooth_weights(corr, alpha)

    grad = np.zeros(N, dtype=np.float64)
    shifts = np.arange(-(N - 1), N)       # length 2N‑1, matches corr order
    for idx, s in enumerate(shifts):
        ws = w[idx]
        if s >= 0:
            if N - s > 0:
                grad[: N - s] += ws * (1.0 - h[s:])
            if s < N:
                grad[s:] -= ws * h[: N - s]
        else:
            k = -s
            if N - k > 0:
                grad[k:] += ws * (1.0 - h[: N - k])
                grad[: N - k] -= ws * h[k:]
    return grad


def _upscale(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function to a finer grid."""
    old_n = old.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    v = np.interp(x_new, x_old, old)
    v = np.clip(v, 0.0, 1.0)
    return _project_to_sum(v, new_n / 2.0)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    np.random.seed(seed)
    start_time = time.time()
    total_budget = float(budget_s)

    # ----- decide on grid size -----
    init_exists = "initial_h_values" in globals()
    if init_exists:
        init_vec = np.clip(np.asarray(initial_h_values, dtype=np.float64), 0.0, 1.0)
        base_n = init_vec.shape[0]
    else:
        init_vec = None
        base_n = 100

    # we want a reasonably fine grid; 256 works well and stays fast
    N = max(200, (base_n // 2) * 2)   # make it even
    if N % 2:
        N += 1

    # ----- initial point -----
    if init_exists and init_vec.shape[0] == N:
        h = _project_to_sum(init_vec.copy(), N / 2.0)
    elif init_exists:
        h = _upscale(init_vec, N)
    else:
        # random binary start (exactly N/2 ones)
        h = np.zeros(N, dtype=np.float64)
        h[: N // 2] = 1.0
        np.random.shuffle(h)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ----- optimisation parameters -----
    lr = 0.7               # learning rate
    alpha = 2.0            # smoothing strength
    no_imp = 0
    IMP_LIMIT = 2000

    # ----- main optimisation loop -----
    while time.time() - start_time < total_budget * 0.97:
        grad = _grad_smooth(h, alpha)
        h_new = h - lr * grad
        h_new = _project_to_sum(h_new, N / 2.0)

        c5_new = _c5_value(h_new)
        if c5_new < best_c5 - 1e-9:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 1.2)      # modestly increase step size after improvement
            no_imp = 0
        else:
            no_imp += 1
            lr *= 0.92
            # occasional random restart to avoid stagnation
            if lr < 1e-4 or no_imp > IMP_LIMIT:
                # restart from the best found so far with a small random jitter
                jitter = np.random.uniform(-0.12, 0.12, size=N)
                h = best_h + jitter
                h = np.clip(h, 0.0, 1.0)
                h = _project_to_sum(h, N / 2.0)
                lr = 0.7
                no_imp = 0

        # anneal the smoothing; larger alpha focuses more on the true max
        alpha = min(200.0, alpha * 1.003)

    # final projection & evaluation
    best_h = _project_to_sum(best_h, N / 2.0)
    final_c5 = _c5_value(best_h)

    return best_h.tolist(), float(final_c5), N
