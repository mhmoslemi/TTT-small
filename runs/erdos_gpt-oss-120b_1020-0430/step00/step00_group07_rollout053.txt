def simulated_annealing(initial_h, time_limit):
    cur_h = initial_h.copy()
    cur_val = max_overlap(cur_h)
    best_h_local = cur_h.copy()
    best_val_local = cur_val
    n = len(cur_h)
    start = time.time()
    iteration = 0
    while time.time() - start < time_limit:
        # choose random i, j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # Determine delta bounds
        # delta added to i, subtracted from j
        max_inc_i = 1.0 - cur_h[i]
        max_dec_i = cur_h[i]
        max_inc_j = 1.0 - cur_h[j]
        max_dec_j = cur_h[j]
        # delta must satisfy: -max_dec_i <= delta <= max_inc_i and -max_inc_j <= delta <= max_dec_j
        lo = -max_dec_i
        hi = max_inc_i
        lo = max(lo, -max_inc_j)
        hi = min(hi, max_dec_j)
        if lo >= hi:
            continue
        # Temperature schedule
        elapsed = time.time() - start
        fraction = elapsed / time_limit
        T = max(1e-4, (1 - fraction) * 0.05)  # linear cooling from 0.05 to near 0
        # Choose delta: maybe scaled by T to make smaller moves later
        # we can for simplicity choose uniform in [lo, hi]
        delta = np.random.uniform(lo, hi)
        # Apply
        cur_h[i] += delta
        cur_h[j] -= delta
        # Evaluate
        new_val = max_overlap(cur_h)
        # Accept?
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_val = new_val
            if new_val < best_val_local:
                best_val_local = new_val
                best_h_local = cur_h.copy()
        else:
            # revert
            cur_h[i] -= delta
            cur_h[j] += delta
        iteration += 1
    return best_h_local, best_val_local

import numpy as np, time, math

def project_to_simplex(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # iterative redistribute excess/deficit
    # Use water-filling algorithm: raise below target proportionally
    # Simplest: while diff not zero allocate through free entries
    diff = target - total
    for _ in range(100):
        free = (v > 0) & (v < 1)
        if not free.any():
            break
        # distribute diff equally among free entries
        delta = diff / free.sum()
        v[free] += delta
        # clip again
        v = np.clip(v, 0.0, 1.0)
        new_total = v.sum()
        diff = target - new_total
        if abs(diff) < 1e-12:
            break
    # if still not exact, scale whole vector (rare)
    if abs(v.sum() - target) > 1e-9:
        v = v * (target / v.sum())
        v = np.clip(v, 0.0, 1.0)
    return v

def compute_val_and_shift(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    val = corr[idx] * dx
    shift = idx - (len(h) - 1)
    return val, shift

    h[j] -= delta

    """Clip to [0,1] and adjust the sum to the required target."""
    v = np.clip(v, 0.0, 1.0)

    if abs(v.sum() - target) > 1e-9:
        v = v * (target / v.sum())
        v = np.clip(v, 0.0, 1.0)
    return v


def _value_and_shift(h: np.ndarray, dx: float):
    """Return (C5 value, shift causing the maximum) for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * dx, idx - (len(h) - 1)


def _random_move(h: np.ndarray):
    """Perform a zero‑sum random move on two indices."""
    n = h.shape[0]
    i, j = np.random.randint(n, size=2)
    if i == j:
        return None
    # delta such that 0 ≤ h[i]+δ ≤ 1 and 0 ≤ h[j]‑δ ≤ 1
    delta_low = max(-h[i], h[j] - 1.0)
    delta_high = min(1.0 - h[i], h[j])
    if delta_low >= delta_high:
        return None
    delta = np.random.uniform(delta_low, delta_high)
    h[i] += delta
    h[j] -= delta
    return (i, j, delta)


def _revert_random_move(h: np.ndarray, info):
    if info is None:
        return
    i, j, delta = info
    h[i] -= delta
    h[j] += delta


def _target_move(h: np.ndarray, shift: int):
    """Increase the two entries participating in the worst shift
    and compensate by decreasing another entry."""
    n = h.shape[0]
    if shift == 0:
        return None
    # valid index interval for i such that i+shift lies inside the array
    if shift > 0:
        lo, hi = 0, n - shift - 1
    else:
        lo, hi = -shift, n - 1
    if lo > hi:
        return None

    idxs = np.arange(n)
    for _ in range(20):
        i = np.random.randint(lo, hi + 1)
        j2 = i + shift
        if h[i] >= 1.0 or h[j2] >= 1.0:
            continue
        # donor index (must have some mass to give)
        donors = np.where((h > 1e-12) & (idxs != i) & (idxs != j2))[0]
        if donors.size == 0:
            continue
        j = np.random.choice(donors)
        delta_max = min(1.0 - h[i], 1.0 - h[j2], h[j] / 2.0)
        if delta_max <= 1e-12:
            continue
        delta = np.random.uniform(0.0, delta_max)
        h[i] += delta
        h[j2] += delta
        h[j] -= 2.0 * delta
        return (i, j2, j, delta)
    return None


def _revert_target_move(h: np.ndarray, info):
    if info is None:
        return
    i, j2, j, delta = info
    h[i] -= delta
    h[j2] -= delta
    h[j] += 2.0 * delta


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h that (approximately) minimises C₅."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 200                     # number of samples on [0,2]
    dx = 2.0 / n
    target_sum = n / 2.0        # integral = 1  ⇒  Σh = n/2

    # ------------------------------------------------------------------
    # Initial solution (use provided construction if available)
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=np.float64)   # may be absent
        if init.shape[0] != n:
            old_n = init.shape[0]
            x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, init)
        else:
            h0 = init.copy()
        h0 = _project_to_simplex(h0, target_sum)
    except Exception:  # no initial_h_values or problem converting it
        h0 = np.zeros(n, dtype=np.float64)
        h0[np.random.choice(n, n // 2, replace=False)] = 1.0

    best_h = h0.copy()
    best_val, _ = _value_and_shift(best_h, dx)

    # ------------------------------------------------------------------
    # Simulated‑annealing style optimisation with many short episodes
    # ------------------------------------------------------------------
    min_episode = 5.0           # seconds per optimisation episode (minimum)

    while True:
        now = time.time()
        elapsed = now - start
        remaining = budget_s - elapsed
        if remaining < min_episode:
            break

        # Choose start point for this episode
        if np.random.rand() < 0.3:
            cur = best_h.copy()
            cur += 0.05 * (np.random.rand(n) - 0.5)
            cur = _project_to_simplex(cur, target_sum)
        else:
            cur = np.zeros(n, dtype=np.float64)
            cur[np.random.choice(n, n // 2, replace=False)] = 1.0

        cur_val, cur_shift = _value_and_shift(cur, dx)

        # Length of this episode (leave a tiny safety margin)
        ep_time = min(remaining - 0.5, 20.0)
        ep_start = time.time()

        while True:
            # global time guard
            if time.time() - start >= budget_s * 0.99:
                break
            # episode time guard
            if time.time() - ep_start >= ep_time:
                break

            # ----------------------------------------------------------------
            # Propose a move (random or targeted)
            # ----------------------------------------------------------------
            if np.random.rand() < 0.35:
                info = _target_move(cur, cur_shift)
                move_type = "target"
            else:
                info = _random_move(cur)
                move_type = "random"

            if info is None:
                continue

            # Evaluate new situation
            new_val, new_shift = _value_and_shift(cur, dx)

            # --------------------------------------------------------------
            # Acceptance test (linear cooling inside the episode)
            # --------------------------------------------------------------
            frac = (time.time() - ep_start) / ep_time
            T = (1.0 - frac) * 0.04 + 0.0005      # from ~0.04 down to 5·10⁻⁴
            if (new_val < cur_val) or (np.random.rand() <
                                       math.exp(-(new_val - cur_val) / max(T, 1e-12))):
                # accept
                cur_val, cur_shift = new_val, new_shift
                if new_val < best_val:
                    best_val = new_val
                    best_h = cur.copy()
            else:
                # reject → revert
                if move_type == "target":
                    _revert_target_move(cur, info)
                else:
                    _revert_random_move(cur, info)

    # final safety projection & value recomputation
    best_h = _project_to_simplex(best_h, target_sum)
    final_val, _ = _value_and_shift(best_h, dx)

    return best_h, final_val, n
