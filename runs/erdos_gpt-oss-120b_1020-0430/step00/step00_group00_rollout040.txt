def project_box_sum(x, target):
    lo = -1.0 - np.max(x)  # low enough
    hi = 1.0 - np.min(x)  # high enough
    for _ in range(50):
        mid = (lo+hi)/2
        y = np.clip(x + mid, 0, 1)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid

        if free.any():

            y = np.clip(y, 0, 1)

            pass

def gradient_worst(h, lag):
    n = len(h)
    g = np.ones(n)
    if lag >=0:
        g[lag:] -= h[:n-lag]  # subtract h[i-lag]
        g[:n-lag] -= h[lag:]  # subtract h[i+lag]
    else:
        p = -lag
        g[:n-p] -= h[p:]      # subtract h[i-lag] = h[i + p]
        g[p:] -= h[:n-p]      # subtract h[i+lag] = h[i - p]
    return g

        free = (y > 0) & (y < 1)
        if np.any(free):
            y[free] += diff / free.sum()

import time
import math
import random
import numpy as np


def _evaluate_c5(h):
    """Return (c5_value, full correlation array) for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx), conv


def _project_feasible(x):
    """
    Euclidean projection of x onto the set
        { h : 0 <= h_i <= 1,  sum_i h_i = n/2 }.
    """
    n = x.shape[0]
    target = n / 2.0
    lo, hi = -1.5, 1.5  # wide enough for all cases
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for rounding errors
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def _gradient_worst(h, lag):
    """Sub‑gradient of the worst (max) correlation term."""
    n = h.shape[0]
    g = np.ones(n)
    if lag >= 0:
        L = lag
        if L == 0:
            g -= 2.0 * h
        else:
            g[L:] -= h[: n - L]       # term h[i‑L]
            g[: n - L] -= h[L:]       # term h[i+L]
    else:
        p = -lag
        g[: n - p] -= h[p:]           # i‑lag = i + p
        g[p:] -= h[: n - p]           # i+lag = i – p
    return g


def _local_optimize(start_h, max_iters, time_limit):
    """
    Simple projected sub‑gradient descent from start_h.
    Returns (best_h, best_score) found before time_limit.
    """
    h = _project_feasible(start_h.copy())
    n = h.shape[0]
    dx = 2.0 / n
    best_h, best_score = h.copy(), _evaluate_c5(h)[0]

    step0 = 0.5
    for it in range(max_iters):
        if time.time() > time_limit:
            break

        # current correlation
        conv = np.correlate(h, 1.0 - h, mode="full")
        max_idx = conv.argmax()
        cur_score = float(conv[max_idx] * dx)
        lag = max_idx - (n - 1)

        # keep the best ever encountered
        if cur_score < best_score:
            best_h, best_score = h.copy(), cur_score

        # sub‑gradient w.r.t. the worst lag
        g = _gradient_worst(h, lag)
        step = step0 / math.sqrt(it + 1.0)

        # simple back‑tracking line search
        while True:
            h_new = _project_feasible(h - step * g)
            new_score = _evaluate_c5(h_new)[0]
            if new_score <= cur_score or step < 1e-6:
                break
            step *= 0.5

        if new_score < cur_score:
            h = h_new
            continue

        # occasional random perturbation to escape plateaus
        i, j = random.sample(range(n), 2)
        delta = (random.random() - 0.5) * 0.2  # small move
        # adjust delta so bounds stay satisfied
        delta = max(delta, -h[i])
        delta = min(delta, 1.0 - h[i])
        delta = max(delta, -(1.0 - h[j]))
        delta = min(delta, h[j])
        if abs(delta) > 1e-12:
            h_pert = h.copy()
            h_pert[i] += delta
            h_pert[j] -= delta
            h_pert = _project_feasible(h_pert)
            pert_score = _evaluate_c5(h_pert)[0]
            if pert_score < cur_score:
                h = h_pert
    return best_h, best_score

    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------

    for cand in candidates:
