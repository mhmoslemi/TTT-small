def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = len(h)
        target_sum = n / 2.0
        # ensure sum exactly target
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) > 1e-8:
            # adjust first entry
            diff = target_sum - cur_sum
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    else:
        n = 200
        # binary vector half ones
        h = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=n//2, replace=False)
        h[idx] = 1.0
    dx = 2.0 / n

    def compute_c5(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return corr.max() * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # initial step size
    max_step = 0.2
    # temperature schedule
    T0 = 0.01
    # iteration
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.5:  # keep margin
            break
        # annealing schedule
        fraction = elapsed / budget_s
        T = T0 * (1.0 - fraction)  # linear cooling
        max_step = 0.2 * (1.0 - fraction) + 0.005  # reduce

        # choose two indices
        i, j = rng.choice(n, size=2, replace=False)

        # compute feasible delta
        lower = max(current_h[i] - 1.0, -current_h[j])
        upper = min(current_h[i], 1.0 - current_h[j])
        if lower >= upper:
            continue
        delta = rng.uniform(-max_step, max_step)
        delta = max(lower, min(delta, upper))
        if abs(delta) < 1e-10:
            continue

        new_h = current_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_c5 = compute_c5(new_h)

        # acceptance
        if new_c5 < current_c5 or rng.random() < math.exp(-(new_c5 - current_c5) / (T + 1e-12)):
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5

        it += 1

        # optional occasional restart: if not improving for many iterations, random perturbation
        # but skip for now.

    # final slight cleanup: ensure sum exact
    # Adjust any rounding error:
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        # add diff to a random element within bounds
        idx = rng.choice(n)
        best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)

    return (best_h, best_c5, n)

import time, math
import numpy as np

def _random_start_vector(n, rng):
    """Create a feasible vector of length n with sum n/2."""
    half = n // 2
    vec = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=half, replace=False)
    vec[idx] = 1.0
    if n % 2 == 1:
        # one element gets the remaining 0.5
        remaining = np.setdiff1d(np.arange(n), idx, assume_unique=True)
        vec[rng.choice(remaining)] = 0.5
    return vec

def _neighbor_array(h, shift):
    """Return h shifted by `shift` (zero‑padding outside)."""
    n = h.size
    if shift >= 0:
        if shift == 0:
            return h.copy()
        return np.concatenate([h[shift:], np.zeros(shift, dtype=np.float64)])
    s = -shift
    return np.concatenate([np.zeros(s, dtype=np.float64), h[:n - s]])

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)

        diff = target_sum - h0.sum()
        if abs(diff) > 1e-12:
            # tiny correction keeping bounds
            h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)
        starters = [h0]
    else:
        n = 200                      # even size – allows binary starts
        starters = []

    # add a few random starts (binary or half‑binary)
    for _ in range(4):
        starters.append(_random_start_vector(n, rng))

    dx = 2.0 / n
    eps = 1e-12

    def _c5(vec):
        """Maximum overlap integral for a given vector."""
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return corr.max() * dx

    # ---- optimisation helpers ----------------------------------------------
    def _greedy_step(vec, max_step):
        """Move mass from a position with small neighbour to one with large neighbour."""
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        neigh = _neighbor_array(vec, shift)

        pos = np.where(vec > eps)[0]
        neg = np.where(vec < 1.0 - eps)[0]
        if pos.size == 0 or neg.size == 0:
            return vec, _c5(vec)

        i = pos[np.argmin(neigh[pos])]
        j = neg[np.argmax(neigh[neg])]

        max_delta = min(vec[i], 1.0 - vec[j], max_step)
        if max_delta <= 0:
            return vec, _c5(vec)

        delta = rng.random() * max_delta          # random fraction of feasible move
        new_vec = vec.copy()
        new_vec[i] -= delta
        new_vec[j] += delta
        return new_vec, _c5(new_vec)

    def _random_step(vec, max_step):
        """Transfer a random amount between two entries, respecting bounds."""
        i, j = rng.choice(n, size=2, replace=False)
        lo = max(vec[i] - 1.0, -vec[j])
        hi = min(vec[i], 1.0 - vec[j])

        new_vec[j] += delta
        return new_vec, _c5(new_vec)
