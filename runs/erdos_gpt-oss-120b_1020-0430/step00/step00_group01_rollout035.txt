def project_box_sum(v, target):
    lo = -2.0
    hi = 2.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, 0.0, 1.0)

import time, math, numpy as np

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x_i<=1, sum x_i = target}."""
    # Use bisection on lambda where x_i = clip(v_i - lam, 0, 1)
    lo = -max(v) - 3.0
    hi = max(v) + 3.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Compute C5 = max_{k} ∫ h(x)(1-h(x+k))dx using the prescribed discretisation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

    g = np.zeros_like(h)
    if lag >= 0:
        m = n - lag
        # +1 for i where i+lag is inside the array
        g[:m] += 1.0
        # -h[i+lag] for the same i
        g[:m] -= h[lag:]
        # -h[i-lag] for i >= lag
        g[lag:] -= h[:m]
    else:  # lag < 0
        L = -lag
        m = n - L
        # +1 for i where i-L >= 0
        g[L:] += 1.0
        # -h[i-L]
        g[L:] -= h[:m]
        # -h[i+L] for i < m
        g[:m] -= h[L:]
    return g


def _run_one(n: int, rng: np.random.Generator,
             time_limit: float, start: float,
             best_h: np.ndarray, best_c5: float) -> tuple:
    """
    Perform a short projected sub‑gradient optimisation for a given size n.
    Returns possibly improved (h, c5) pair.
    """
    # initial shape – symmetric triangular “tent”
    x = (np.arange(n) + 0.5) * (2.0 / n)          # sample locations
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))   # tent, integral ≈1
    target = n / 2.0
    h = _project_box_sum(h, target)

    # a few random restarts to escape poor local minima
    for restart in range(3):
        # add a tiny random perturbation then re‑project
        h += 0.03 * rng.standard_normal(n)
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, target)

        cur_c5 = _c5_value(h)
        if cur_c5 < best_c5:
            best_h, best_c5 = h.copy(), cur_c5

        it = 0
        max_iters = 3000
        while it < max_iters and (time.time() - start) < time_limit:
            # correlation and the lag that is currently worst
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = int(np.argmax(corr))
            lag = idx - (n - 1)

            # gradient for this lag
            g = _grad_at_lag(h, lag)

            # decreasing step size
            step = 0.6 / math.sqrt(it + 1.0)

            improved = False
            # simple backtracking line search
            for _ in range(8):
                h_new = _project_box_sum(h - step * g, target)
                new_c5 = _c5_value(h_new)
                if new_c5 < cur_c5 - 1e-12:
                    h, cur_c5 = h_new, new_c5
                    improved = True
                    if cur_c5 < best_c5:
                        best_h, best_c5 = h.copy(), cur_c5
                    break
                step *= 0.5
            if not improved:
                # small move even if not improving – helps to escape flat regions
                step *= 0.5
                h = _project_box_sum(h - step * g, target)
                cur_c5 = _c5_value(h)
            it += 1
            # early exit if we already beat the target record
            if best_c5 < 0.3807:
                return best_h, best_c5
        # after a restart we keep the best found so far
        if best_c5 < 0.3807:
            break
    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c5 = np.inf
    best_n = None

    # if a parent solution is provided, use it as the baseline
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        c0 = _c5_value(h0)
        best_h, best_c5, best_n = h0.copy(), c0, len(h0)

    # a sequence of problem sizes – finer grids can give tighter bounds
    sizes = [94, 188, 376, 752]               # all even, below the 1k limit
    for n in sizes:
        if (time.time() - start_time) >= budget_s:
            break
        # allocate a fraction of the remaining time to this size
        remaining = budget_s - (time.time() - start_time)
        # give roughly 20 % of the remaining wall‑time to each size (adjusted)
        alloc = max(5.0, remaining * 0.20)

        # run the optimiser for this resolution
        best_h, best_c5 = _run_one(n, rng, alloc, start_time,
                                   best_h if best_h is not None else np.empty(0),
                                   best_c5)

        best_n = n
        # if we have already passed the target, stop early
        if best_c5 < 0.3807:
            break

    # fallback – uniform half‑density if everything failed
    if best_h is None:
        n = 200
        best_h = np.full(n, 0.5)
        best_c5 = _c5_value(best_h)
        best_n = n

    # ensure the returned vector obeys the constraints (numerical safety)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, best_n / 2.0)

    return best_h.tolist(), float(best_c5), int(best_n)
