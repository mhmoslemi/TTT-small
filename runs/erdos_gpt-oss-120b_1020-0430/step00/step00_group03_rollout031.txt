    max_c = c.max()

    i = np.random.choice(mism)

def guided_swap(h):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    max_val = corr.max()
    L = np.argmax(corr)
    shift = L - (n-1)
    # compute mismatch positions
    if shift >= 0:
        mism = np.where((h[:n-shift]==1) & (h[shift:]==0))[0]
    else:
        s = -shift
        mism = np.where((h[s:]==1) & (h[:n-s]==0))[0] + s  # first index i where h_i=1 and h_{i-s}=0
    if mism.size == 0:
        # no mism; perfect maybe
        return None, None, max_val, shift
    i = np.random.choice(mism)
    # candidate zeros
    zeros = np.where(h==0)[0]
    # attempt to find j satisfying condition
    # define condition function:
    if shift >= 0:
        # need j+shift >= n or h[j+shift]==1
        cond = lambda j : (j+shift >= n) or (h[j+shift]==1)
    else:
        s = -shift
        cond = lambda j: (j < s) or (h[j-s]==1)
    # random sample attempts:
    attempts = 0
    while attempts < 30:
        j = np.random.choice(zeros)
        if cond(j):
            return i, j, max_val, shift
        attempts += 1
    # fallback: random swap
    # choose random i from ones, random j from zeros (ensuring i != j)
    ones = np.where(h==1)[0]
    i_rand = np.random.choice(ones)
    j_rand = np.random.choice(zeros)
    return i_rand, j_rand, max_val, shift
