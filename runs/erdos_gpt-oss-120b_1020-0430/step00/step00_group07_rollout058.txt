def repair(h):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    target = len(h)/2
    diff = target - h.sum()
    # adjust by adding diff proportionally to entries not at bounds
    # We'll iterate:
    # Determine mask of free entries (0< h <1). While diff != 0 and mask nonempty:
    # Compute delta = diff / number_free
    # Add delta to free entries, clip again, recompute diff, repeat.
    # This converges quickly.
    mask = (h > 0) & (h < 1)
    # While diff absolute > 1e-12 and mask any.
    while abs(diff) > 1e-12 and np.any(mask):
        num = mask.sum()
        delta = diff / num
        # Add delta
        h[mask] += delta
        # Clip again (some may exceed bounds)
        over = h > 1
        under = h < 0
        if np.any(over):
            h[over] = 1
        if np.any(under):
            h[under] = 0
        diff = target - h.sum()
        mask = (h > 0) & (h < 1)
    # final clip
    h = np.clip(h, 0, 1)
    return h

import numpy as np
import time
import math

def evaluate_h(h, dx):
    """Return C5 for step function h discretized with spacing dx."""
    # correlation of h with (1-h) using full mode (zero padding)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def proj_box_sum(v, target, max_iter=60):
    """
    Euclidean projection of vector v onto the set {x in [0,1]^n : sum(x)=target}.
    Uses a bisection on the Lagrange multiplier.
    """
    low = -2.0
    high = 2.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

def worst_shift_and_grad(h):
    """
    Compute the shift s (as integer relative to centre) that attains the
    maximal overlap integral and return also the sub‑gradient of that max.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)          # shift index (negative to positive)
    # sub‑gradient of C(s) = sum h_i (1 - h_{i-s})
    grad = np.ones_like(h)
    if s >= 0:
        # i-s is valid for i >= s
        grad[: n - s] -= h[s:]          # term h_{i+s}
        grad[s: ]   -= h[: n - s]       # term h_{i-s}
    else:
        s_abs = -s
        grad[s_abs: ] -= h[: n - s_abs]
        grad[: n - s_abs] -= h[s_abs: ]
    # also return the raw correlation maximum (without dx)
    max_corr = np.max(corr)
    return s, grad, max_corr

    init_h = proj_box_sum(init_h, target_sum)

    best_h = init_h.copy()
    best_score = evaluate_h(best_h, dx)

    # ------------------------------------------------------------
    # 1) Gradient‑descent phase (≈60 % of the time budget)
    # ------------------------------------------------------------
    gd_budget = budget_s * 0.6
    cur_h = best_h.copy()
    cur_score = best_score

    while time.time() - start < gd_budget:
        # worst shift and sub‑gradient
        _, grad, _ = worst_shift_and_grad(cur_h)

        # linearly decreasing step size
        elapsed = time.time() - start
        frac = elapsed / gd_budget
        step = max(1e-4, 0.5 * (1.0 - frac))

        new_h = cur_h - step * grad
        new_h = proj_box_sum(new_h, target_sum)

        new_score = evaluate_h(new_h, dx)

        # deterministic improvement, otherwise occasional uphill move
        if new_score < cur_score:
            cur_h, cur_score = new_h, new_score
        else:
            T = max(1e-8, 0.01 * (1.0 - frac))
            if np.random.rand() < math.exp(-(new_score - cur_score) / T):
                cur_h, cur_score = new_h, new_score

        if new_score < best_score:
            best_score, best_h = new_score, new_h.copy()

    # ------------------------------------------------------------
    # 2) Simulated‑annealing fine tuning (remaining budget)
    # ------------------------------------------------------------
    cur_h, cur_score = best_h.copy(), best_score
    T0 = max(0.01, best_score * 0.1)

    while time.time() - start < budget_s * 0.99:
        elapsed = time.time() - start
        total_frac = elapsed / budget_s
        T = max(1e-8, T0 * (1.0 - total_frac))
        move_factor = max(0.01, 0.5 * (1.0 - total_frac))

        # propose a mass‑transfer move between two coordinates
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        up = min(cur_h[i], 1.0 - cur_h[j])
        if up <= 0.0:
            up = min(cur_h[j], 1.0 - cur_h[i])
            if up <= 0.0:
                continue
            delta = np.random.rand() * up * move_factor
            new_h = cur_h.copy()
            new_h[j] -= delta
            new_h[i] += delta
        else:
            delta = np.random.rand() * up * move_factor
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_score = evaluate_h(new_h, dx)
        diff = new_score - cur_score
        if diff < 0.0 or np.random.rand() < math.exp(-diff / T):
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_score, best_h = new_score, new_h.copy()

        # occasional random restart to avoid deep traps
        if np.random.rand() < 0.001:
            rand_h = np.zeros(n, dtype=np.float64)
            rand_h[: n // 2] = 1.0
            np.random.shuffle(rand_h)
            cur_h = proj_box_sum(rand_h, target_sum)
            cur_score = evaluate_h(cur_h, dx)

    # ------------------------------------------------------------
    # Final projection and return
    # ------------------------------------------------------------
    best_h = proj_box_sum(best_h, target_sum)
    final_score = evaluate_h(best_h, dx)

    return best_h, float(final_score), n
