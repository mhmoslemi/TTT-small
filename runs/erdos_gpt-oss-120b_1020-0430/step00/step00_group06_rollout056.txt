def discretize_h(L,G,n_points):
    dx = 2.0 / n_points
    xs = np.arange(n_points) * dx  # left edge of each cell
    # For each cell, compute overlap length with intervals
    a,b = intervals_from_LG(L,G)
    h = np.zeros(n_points)
    # iterate over intervals
    for start,end in zip(a,b):
        # find cell indices where xs < end and xs+dx > start.
        # Overlap length = min(end, xs+dx) - max(start, xs) clipped to >=0
        # vectorized:
        lo = np.maximum(start, xs)
        hi = np.minimum(end, xs + dx)
        overlap = np.maximum(0.0, hi - lo)
        h += overlap / dx  # each cell contributions from intervals (no overlap as intervals non overlapping)
    # ensure values within [0,1] (should be)
    return h
```

    starts = np.empty(m)
    ends = np.empty(m)

        ends[i] = pos + L[i]
        pos = ends[i] + G[i+1]

def compute_c5(L, G):
