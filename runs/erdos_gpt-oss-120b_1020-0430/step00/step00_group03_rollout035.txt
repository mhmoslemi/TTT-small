def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # compute correlation
    # np.correlate returns length 2*n-1, where index shift = -(n-1) .. (n-1)
    # We multiply by dx at the end
    # For cross-correlation, we need to compute h and (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)

def simulated_annealing(n, budget_s, seed):
    # parameters
    np.random.seed(seed)
    random.seed(seed)
    dx = 2.0/n
    target_sum = n/2
    # initial candidate: random binary
    cur_h = random_binary_h(n)
    cur_max = compute_max_corr(cur_h)
    best_h = cur_h.copy()
    best_max = cur_max
    start = time.time()
    T0 = 5.0  # start temperature
    while time.time() - start < budget_s * 0.9:
        # propose swap
        # find indices where h[i]==1 and h[j]==0
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones)==0 or len(zeros)==0:
            # cannot swap, maybe restart
            cur_h = random_binary_h(n)
            cur_max = compute_max_corr(cur_h)
            continue
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_max = compute_max_corr(new_h)
        delta = new_max - cur_max
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / (budget_s*0.9))
        if delta < 0 or random.random() < math.exp(-delta / max(T, 1e-12)):
            cur_h = new_h
            cur_max = new_max
            # update best
            if new_max < best_max:
                best_h = new_h.copy()
                best_max = new_max
        # maybe occasional random restart if no improvement for many steps
    return best_h, best_max

    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')

    s = max_idx - (n - 1)

    order = np.argsort(-contrib)

        j = j_range[idx]
        max_transfer = min(h[i], 1.0 - h[j])

import time, math, random
import numpy as np

def _max_corr(h: np.ndarray) -> float:
    """Raw maximum of the correlation h * (1-h) (zero padding)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def _c5_bound(h: np.ndarray) -> float:
    """C5 value for a step function represented by h."""
    n = h.shape[0]
    return _max_corr(h) * (2.0 / n)

def _random_binary(n: int) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones (n must be even)."""
    k = n // 2
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, k, replace=False)
    h[idx] = 1.0
    return h

def _directed_move(h: np.ndarray):
    """Propose a move that reduces the contribution of the worst shift.
    Returns (i, j, delta) where mass delta is moved from i to j, or None."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)                      # shift giving the current maximum
    if s >= 0:
        i_idx = np.arange(0, n - s)
        j_idx = i_idx + s
    else:
        i_idx = np.arange(-s, n)
        j_idx = i_idx + s
    # contributions for this shift
    contrib = h[i_idx] * (1.0 - h[j_idx])
    # index of the largest contribution
    order = np.argsort(-contrib)               # descending
    for pos in order:
        i, j = i_idx[pos], j_idx[pos]
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer > 1e-12:
            # move a random fraction of the feasible amount
            delta = max_transfer * np.random.rand()
            return int(i), int(j), float(delta)
    return None

def _random_transfer(h: np.ndarray):
    """Propose a generic transfer between two random positions."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]
    # feasible transfer from i to j
    max_pos = min(hi, 1.0 - hj)
    # feasible transfer from j to i (negative delta)
    max_neg = min(hj, 1.0 - hi)
    if max_pos + max_neg < 1e-12:
        return None
    delta = np.random.uniform(-max_neg, max_pos)
    if abs(delta) < 1e-12:
        return None
    return int(i), int(j), float(delta)

def _apply_move(h: np.ndarray, i: int, j: int, delta: float):
    """Return a new vector after moving delta from i to j (preserves sum)."""
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    # safety clipping (should already be within bounds)
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h

    Returns (h_values, c5_bound, n_points).
    """

    np.random.seed(seed)

    # target length for the search – even for binary feasibility
    n = 200
    if n % 2:

    init = globals().get("initial_h_values")

        np.clip(h0, 0.0, 1.0, out=h0)

        cur_sum = h0.sum()
        diff = (h0.shape[0] / 2.0) - cur_sum

        # work only with binary values – swaps keep feasibility
        h = best_h.copy()
        raw = best_raw
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        max_swaps = 20000
        t0 = time.time()
        for _ in range(max_swaps):
            if time.time() - t0 > (budget_s * 0.07):
                break
            if ones.size == 0 or zeros.size == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            trial = h.copy()
            trial[i], trial[j] = trial[j], trial[i]
            trial_raw = _max_corr(trial)
            if trial_raw < raw:
                h, raw = trial, trial_raw
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
        if raw < best_raw:
            best_h, best_raw = h, raw

    # final bound and return
    final_bound = best_raw * dx
    return best_h, final_bound, n
