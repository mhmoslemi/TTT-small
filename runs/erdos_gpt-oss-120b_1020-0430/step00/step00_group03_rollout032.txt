def proj_capped_simplex(v, target, max_val=1.0, tol=1e-9, max_iter=100):
    low = -max_val - np.max(v) - 1.0
    high = np.max(v) + max_val + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0, max_val)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - (low+high)/2.0, 0, max_val)
    # due to tolerance may not match exactly; adjust small residual
    residual = target - w.sum()
    if abs(residual) > 1e-6:
        # adjust among non-bound indices
        free = (w > 0) & (w < max_val)
        if free.any():
            w[free] += residual / free.sum()
            w = np.clip(w, 0, max_val)
    return w
```

def project_to_box_sum(v, target, max_val=1.0):
    # binary search
    low = -max_val - np.max(v) - 1.0
    high = np.max(v) + max_val + 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0, max_val)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - (low+high)/2.0, 0, max_val)
    # adjust residual
    residual = target - w.sum()
    if abs(residual) > 1e-8:
        # try to distribute residual to non-bound values
        free = (w > 0) & (w < max_val)
        if free.any():
            w[free] += residual / free.sum()
            w = np.clip(w, 0, max_val)
        else:
            # all at bounds; just adjust one element within bounds
            # find index not at max if residual>0 else not at zero
            if residual > 0:
                idx = np.argmax(w < max_val)
                if w[idx] < max_val:
                    w[idx] = min(max_val, w[idx] + residual)
            else:
                idx = np.argmax(w > 0)
                if w[idx] > 0:
                    w[idx] = max(0, w[idx] + residual)
    return w
```

import time

def _project_capped_simplex(v: np.ndarray, target: float, max_val: float = 1.0) -> np.ndarray:
    """
    Project vector v onto the set { w | 0 <= w_i <= max_val, sum(w) = target }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -max_val - np.max(v) - 1.0
    hi = np.max(v) + max_val + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, max_val)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, max_val)
    # tiny correction to meet the sum exactly (within numeric noise)
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        free = (w > 0.0) & (w < max_val)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, max_val)
        else:
            # all at a bound – adjust a single element safely
            idx = 0
            if diff > 0:
                # need to increase somewhere, pick first not at max
                candidates = np.where(w < max_val)[0]
                if candidates.size:
                    idx = candidates[0]
            else:
                # need to decrease, pick first not at zero
                candidates = np.where(w > 0.0)[0]
                if candidates.size:
                    idx = candidates[0]
            w[idx] = max(0.0, min(max_val, w[idx] + diff))
    return w

    """Return the discrete C5 value for a step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    # pre‑compute masks for quick sampling
    def _indices_above(val):
        return np.where(h > val)[0]

    def _indices_below(val):
        return np.where(h < val)[0]

    it = 0
    while time.time() - start_time < budget_s * 0.94 and it < max_iter:
        # temperature schedule (linear)
        progress = (time.time() - start_time) / (budget_s * 0.94)
        T = T0 * (1.0 - progress) + T_end * progress

        # pick a source (decrease) and a target (increase)
        src_candidates = np.where(h > 1e-12)[0]
        tgt_candidates = np.where(h < 1.0 - 1e-12)[0]
        if src_candidates.size == 0 or tgt_candidates.size == 0:
            break
        src = int(rnd.choice(src_candidates))
        tgt = int(rnd.choice(tgt_candidates))
        max_move = min(h[src], 1.0 - h[tgt])
        if max_move <= 0.0:
            continue
        # random transfer amount (occasionally full move)
        eps = max_move * rnd.random() * move_factor
        # occasionally allow a full transfer to explore more aggressively
        if rnd.random() < 0.01:
            eps = max_move

        h_new = h.copy()
        h_new[src] -= eps
        h_new[tgt] += eps

        # evaluate new candidate
        new_c = _c5_value(h_new, dx)

        # acceptance rule
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            if T > 0.0:
                prob = np.exp(-(new_c - cur_c) / T)
                if rnd.random() < prob:
                    accept = True

        if accept:
            h = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                # early exit if we beat the target
                if best_c <= 0.3808:
                    break

        # occasional random perturbation to escape plateaus
        if it % 2000 == 0 and rnd.random() < 0.03:
            # perform a random swap of two indices (could be full binary swap)
            i, j = rnd.choice(n, size=2, replace=False)
            hi, hj = h[i], h[j]
            h[i], h[j] = hj, hi
            cur_c = _c5_value(h, dx)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()

        # occasional small projection to keep numeric drift in check
        if it % 5000 == 0:
            h = _project_capped_simplex(h, target_sum)

        it += 1

    # final projection to guarantee feasibility
    best_h = _project_capped_simplex(best_h, target_sum)
    best_c = _c5_value(best_h, dx)

    return (best_h.tolist(), float(best_c), n)
