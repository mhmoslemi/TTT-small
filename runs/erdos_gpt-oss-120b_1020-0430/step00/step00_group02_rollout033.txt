def project(h):
    n = len(h)
    target = n / 2.0
    # binary search lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo+hi)/2.0, 0.0, 1.0)
    # due to rounding errors, adjust again with small diff
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        # distribute diff over indices not at bounds
        free = (h_proj > 1e-12) & (h_proj < 1 - 1e-12)
        if free.any():
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

import time
import numpy as np

def _project_to_constraints(v):
    """Project v onto [0,1]^n and sum(v)=len(v)/2."""
    n = v.size
    target = n * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        free = (v > 1e-12) & (v < 1 - 1e-12)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v


def _c5_value(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx, discretised."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _local_search(start_h, time_budget):
    """Simple directed SA on a fixed‑size vector."""
    n = start_h.size
    h = start_h.copy()
    cur = _c5_value(h)
    best_h, best = h.copy(), cur
    start = time.time()
    T = 0.012
    step = 0.5
    it = 0
    while time.time() - start < time_budget:
        # worst shift
        full = np.correlate(h, 1.0 - h, mode="full")
        k = np.argmax(full) - (n - 1)
        kabs = abs(k)

        if kabs == 0:
            i_out, i_in = np.random.choice(n, 2, replace=False)
        else:
            # neighbour sum for the worst shift
            fwd = np.zeros(n)
            bwd = np.zeros(n)
            if kabs < n:
                fwd[: n - kabs] = h[kabs:]
                bwd[kabs:] = h[: n - kabs]
            neigh = fwd + bwd
            pct = max(1, int(0.1 * n))
            low = np.argpartition(neigh, pct)[:pct]
            high = np.argpartition(-neigh, pct)[:pct]
            i_out = low[np.random.randint(low.size)]
            i_in = high[np.random.randint(high.size)]

        max_delta = min(h[i_out], 1.0 - h[i_in])
        if max_delta <= 1e-12:
            # fallback random pair
            i_out, i_in = np.random.choice(n, 2, replace=False)
            max_delta = min(h[i_out], 1.0 - h[i_in])
            if max_delta <= 1e-12:
                it += 1
                continue
        delta = max_delta * np.random.rand() * step
        if delta < 1e-14:
            it += 1
            continue

        nh = h.copy()
        nh[i_out] -= delta
        nh[i_in] += delta
        nh = np.clip(nh, 0.0, 1.0)          # keep within box
        # sum is preserved analytically; minor drift corrected occasionally
        if abs(nh.sum() - h.sum()) > 1e-12:
            nh = _project_to_constraints(nh)

        new = _c5_value(nh)

        # Metropolis acceptance
        accept = False
        if new < cur:
            accept = True
        else:
            prob = np.exp(-(new - cur) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            h, cur = nh, new
            if new < best:
                best_h, best = nh.copy(), new

        it += 1
        if it % 500 == 0:
            T = max(T * 0.9, 1e-7)
            step = max(step * 0.95, 1e-4)

    return best_h, best


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # fetch possible parent construction
    try:
        parent = np.array(initial_h_values, dtype=float)
    except Exception:
        parent = None

    # candidate sizes (start with parent length, then a few larger ones)
    sizes = [45, 60, 80, 100]
    if parent is not None:
        sizes = [parent.size] + [s for s in sizes if s != parent.size]

    overall_best_h = None
    overall_best_score = np.inf
    overall_n = None
    start_total = time.time()

    for n in sizes:
        # time left for this size
        elapsed = time.time() - start_total
        if elapsed >= budget_s * 0.95:
            break
        time_for_n = (budget_s - elapsed) * 0.9 / (len(sizes) - sizes.index(n))
        # initialise
        if parent is not None and parent.size == n:
            h0 = _project_to_constraints(parent.copy())
        else:
            h0 = np.random.rand(n)
            h0 = _project_to_constraints(h0)

        # local optimisation
        h_opt, sc_opt = _local_search(h0, time_for_n)

        # keep the best overall
        if sc_opt < overall_best_score:

            overall_best_score = sc_opt
            overall_n = n

    # final safety projection
    overall_best_h = _project_to_constraints(overall_best_h)
    final_score = _c5_value(overall_best_h)
