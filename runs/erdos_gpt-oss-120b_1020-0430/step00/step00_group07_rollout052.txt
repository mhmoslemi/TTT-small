def project_onto_simplex_box(x, total):
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > total:
            hi = lam
        else:
            lo = lam
    y = np.clip(x + (lo+hi)/2.0, 0.0, 1.0)
    return y

Works because sum(y) monotonic w.r.t lambda (if lambda increases, each term x_i+lambda increments until saturates at 1). So find lambda giving exact sum total.

        h_new = cur_h - alpha * grad
        h_new = project_onto_simplex_box(h_new, total)

        new_val = max_overlap(h_new, dx)
        if new_val <= cur_val:
            cur_h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
        # else keep the old point

        smooth_T = max(min_smooth_T, smooth_T * 0.9995)

    # ------------------------------------------------------------------
    # 3) Final refinement with a bound‑constrained optimizer (≈20%)
    # ------------------------------------------------------------------
    # Use L‑BFGS‑B with a heavy penalty for violating the sum constraint.
    bounds = [(0.0, 1.0)] * n
    penalty_factor = 1e6

    def penalised_obj(x):
        f = max_overlap(x, dx)
        g = x.sum() - total
        return f + penalty_factor * (g * g)

    remaining_time = budget_s - (time.time() - start_time)
    maxiter_opt = int(500 * max(1.0, remaining_time / 30.0))

    try:
        res = opt.minimize(
            penalised_obj,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter_opt, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_simplex_box(h_opt, total)
            val_opt = max_overlap(h_opt, dx)
            if val_opt < best_val:
                best_h = h_opt
                best_val = val_opt
    except Exception:
        pass  # optimiser failure – keep current best

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_simplex_box(best_h, total)
    final_val = max_overlap(best_h, dx)

    return best_h, float(final_val), n
