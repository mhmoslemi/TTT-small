def project_box_sum(v, lower, upper, target_sum, max_iter=60):

def gradient_for_lag(h, lag):
    n = len(h)
    # compute gradient vector as described: (1 - h[i+lag]) - h[i-lag] with zero padding
    g = np.zeros_like(h)
    if lag >= 0:
        # i+lag valid for i in [0, n-1-lag]
        i_range = np.arange(n - lag)
        g[i_range] = 1 - h[i_range + lag]
        # i-lag valid for i in [lag, n-1] => i-lag index is i - lag
        i_range2 = np.arange(lag, n)
        g[i_range2] -= h[i_range2 - lag]
    else:
        # lag negative, shift other direction
        lag_pos = -lag
        # i+lag = i - lag_pos
        i_range = np.arange(lag_pos, n)
        g[i_range] = 1 - h[i_range - lag_pos]
        # i - lag = i + lag_pos
        i_range2 = np.arange(0, n - lag_pos)
        g[i_range2] -= h[i_range2 + lag_pos]
    return g

def gradient_for_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        # h[i+lag] exists for i up to n-1-lag
        idx = np.arange(n - lag)
        g[idx] = 1 - h[idx + lag]
        # h[i - lag] exists for i >= lag
        idx2 = np.arange(lag, n)
        g[idx2] -= h[idx2 - lag]
    else:
        # lag < 0: shift other direction
        k = -lag
        # h[i+lag] = h[i - k] exists for i >= k
        idx = np.arange(k, n)
        g[idx] = 1 - h[idx - k]
        # h[i - lag] = h[i + k] exists for i up to n-1-k
        idx2 = np.arange(0, n - k)
        g[idx2] -= h[idx2 + k]
    return g

def project_box_sum(v, lower, upper, target, max_iter=60):
    # Find lambda such that sum_i clip(v_i + lambda) = target.
    lo = -1e5
    hi = 1e5
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + (lo + hi) / 2.0, lower, upper)
    return w

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """C5 value for a discretised step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_sum(v: np.ndarray, lower: float, upper: float, target_sum: float,
                    max_iter: int = 60) -> np.ndarray:
    """
    Euclidean projection of v onto the hyperplane sum(x)=target_sum
    intersected with box constraints lower <= x_i <= upper.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + (lo + hi) * 0.5, lower, upper)
    return w


def worst_lag(h: np.ndarray) -> int:
    """Return the lag (negative allowed) where the overlap integral is maximal."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return idx - (n - 1)   # lag = index - (n-1)


def gradient_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the worst‐lag overlap with respect to h."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        # term (1 - h[i+lag])
        valid = np.arange(n - lag)
        g[valid] = 1.0 - h[valid + lag]
        # term -h[i-lag]
        valid = np.arange(lag, n)
        g[valid] -= h[valid - lag]
    else:
        k = -lag
        # (1 - h[i-k])
        valid = np.arange(k, n)
        g[valid] = 1.0 - h[valid - k]
        # -h[i+k]
        valid = np.arange(0, n - k)
        g[valid] -= h[valid + k]
    return g


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linearly interpolate a coarse initial vector to length n and enforce constraints."""
    n0 = init.size
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h_new = np.interp(x_new, x0, init)
    h_new = np.clip(h_new, 0.0, 1.0)
    target_sum = n / 2.0
    # scale to exact integral, then final projection to respect bounds
    cur_sum = h_new.sum()
    if cur_sum > 0:
        h_new *= target_sum / cur_sum
        h_new = np.clip(h_new, 0.0, 1.0)
    # final projection (handles any tiny deviation)
    h_new = project_box_sum(h_new, 0.0, 1.0, target_sum)
    return h_new


def random_binary(n: int) -> np.ndarray:
    """Random 0/1 vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    indices = np.arange(n)
    np.random.shuffle(indices)
    half = n // 2
    h[indices[:half]] = 1.0
    return h

# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a tuple (h_values, c5_bound, n_points) within the time budget."""
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s - 0.3      # small safety margin

    # target number of points – a compromise between speed and accuracy
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2.0    # because integral of h must be 1

    # ------------------------------------------------------------------
    # Prepare an initial feasible vector
    # ------------------------------------------------------------------
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        # up‑sample the supplied initial vector
        h0 = upsample_initial(np.asarray(init_vec, dtype=float), n_points)
    else:
        # start from a random binary pattern
        h0 = random_binary(n_points)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    # one outer restart consists of a deterministic gradient descent phase,
    # periodically interleaved with small random perturbations.
    max_restarts = 50
    max_iters_per_restart = 25000

    # learning‑rate schedule (largest at start of a restart)
    lr_start = 0.5
    # temperature for occasional uphill moves (simulated‑annealing flavour)
    temp_start = 0.02

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for restart in range(max_restarts):
        # time check
        if time.time() - start_time > max_time:
            break

        # start a new run – either from the current best (perturbed) or from a fresh random binary
        if np.random.rand() < 0.4:
            h = best_h.copy()
            # add a tiny random jitter to escape shallow basins
            jitter = np.random.uniform(-0.02, 0.02, size=n_points)
            h = np.clip(h + jitter, 0.0, 1.0)
            h = project_box_sum(h, 0.0, 1.0, target_sum)
        else:
            h = random_binary(n_points)

        cur_c5 = compute_c5(h)

        # deterministic gradient descent phase
        for it in range(max_iters_per_restart):
            # time escape
            if time.time() - start_time > max_time:
                break

            # 1) identify worst lag and its gradient
            lag = worst_lag(h)
            grad = gradient_for_lag(h, lag)

            # 2) step size (decays linearly inside a restart)
            lr = lr_start * (1.0 - it / max_iters_per_restart)

            # 3) take a projected gradient step
            v = h - lr * grad
            h_new = project_box_sum(v, 0.0, 1.0, target_sum)

            # 4) evaluate
            new_c5 = compute_c5(h_new)

            # 5) acceptance (simple hill‑climbing with occasional uphill moves)
            if new_c5 < cur_c5:
                h, cur_c5 = h_new, new_c5
                if new_c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), new_c5
            else:
                # simulated‑annealing acceptance
                temp = temp_start * (1.0 - it / max_iters_per_restart) + 1e-12
                prob = math.exp(-(new_c5 - cur_c5) / temp)
                if np.random.rand() < prob:
                    h, cur_c5 = h_new, new_c5

            # occasional random local swap to inject diversity
            if it % 400 == 0 and np.random.rand() < 0.07:
                i, j = np.random.choice(n_points, 2, replace=False)
                # feasible epsilon for moving mass from i to j
                lower = max(-h[i], h[j] - 1.0)
                upper = min(1.0 - h[i], h[j])
                if lower < upper:
                    eps = np.random.uniform(-0.1, 0.1)
                    eps = np.clip(eps, lower, upper)
                    h2 = h.copy()
                    h2[i] += eps
                    h2[j] -= eps
                    # projection is not needed – eps already respects bounds and sum
                    cand_c5 = compute_c5(h2)
                    if cand_c5 < cur_c5:
                        h, cur_c5 = h2, cand_c5
                        if cand_c5 < best_c5:
                            best_h, best_c5 = h2.copy(), cand_c5

        # end of one restart
        # keep the best known solution across restarts
        if best_c5 <= 0.3808:
            break   # already good enough, can stop early

    # final safety projection (guarantees exact integral)
    best_h = project_box_sum(best_h, 0.0, 1.0, target_sum)

    # recompute final bound with a fresh evaluation
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_points
