def hill_climb(h, steps, best_val):
    N = len(h)
    for _ in range(steps):
        # propose swapping two entries
        i, j = np.random.choice(N, 2, replace=False)
        # ensure they are different values to maintain sum constraint after swap
        if h[i] == h[j]:
            continue
        # propose swap: temporary copy
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        val = compute_max_overlap(h_new)
        if val < best_val:
            h[:] = h_new
            best_val = val
        # optionally accept with probability
    return h, best_val

def _max_overlap(h):
    """C5 value for a given vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1‑h), zero‑padded outside the interval
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr  # also return the whole correlation


def _project_sum_one(h, target):
    """Project h onto the box [0,1] while fixing sum(h)=target.
    Simple water‑filling: shift by a constant λ and then clip."""
    # Find λ such that sum(clip(h-λ,0,1)) = target.
    # Binary search on λ.
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    return np.clip(h - lo, 0.0, 1.0)


def _targeted_move(h, shift):
    """Propose a small mass transfer that reduces the worst shift.
    Return a new vector (or None if no feasible transfer)."""
    n = h.shape[0]
    if shift >= 0:
        i_idx = np.arange(0, n - shift)
        j_idx = i_idx + shift
    else:                              # shift < 0
        s = -shift
        i_idx = np.arange(s, n)
        j_idx = i_idx - s
    hi = h[i_idx]
    hj = h[j_idx]
    # contribution of each pair to the overlap for this shift
    contrib = hi * (1.0 - hj)
    if contrib.size == 0:
        return None
    k = int(np.argmax(contrib))
    i = int(i_idx[k])
    j = int(j_idx[k])
    # max amount we can move from i to j without breaking bounds
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 1e-12:

    delta = max_delta * np.random.rand() * 0.5
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

        if abs(h[i] - h[j]) > 1e-9:
            h_swapped = h.copy()
            h_swapped[i], h_swapped[j] = h_swapped[j], h_swapped[i]
            new_val, _ = _max_overlap(h_swapped)
            if new_val < cur_val - 1e-12:
                h = h_swapped
                continue

        # ------------------------------------------------------------------
        # 4) Occasionally perform a small random perturbation of a single entry
        # ------------------------------------------------------------------
        if random.random() < 0.01:
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.2  # [-0.1,0.1]
            # keep sum unchanged: compensate with another random index
            other = (idx + np.random.randint(1, n)) % n
            # Limit changes so we stay inside [0,1]
            d1 = min(delta, 1.0 - h[idx]) if delta > 0 else max(delta, -h[idx])
            d2 = min(-d1, h[other]) if d1 < 0 else min(-d1, 1.0 - h[other])
            # Apply feasible part
            if abs(d1) > 1e-12 and abs(d2) > 1e-12:
                h_tmp = h.copy()
                h_tmp[idx] += d1
                h_tmp[other] += d2
                new_val, _ = _max_overlap(h_tmp)
                if new_val < cur_val - 1e-12:
                    h = h_tmp
                    continue

        # occasional restart to avoid being stuck
        if random.random() < 0.0005:
            h = np.zeros(n, dtype=float)
            h[: n // 2] = 1.0
            np.random.shuffle(h)

    # ----------------------------------------------------------------------
    # Final repair (numerical noise may have drifted the sum a tiny bit)
    # ----------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    if not np.isclose(best_h.sum(), target_sum, atol=1e-9):
        best_h = _project_sum_one(best_h, target_sum)

    # Return as plain Python list for safety
    return best_h.tolist(), float(best_val), int(n)
