def hill_climb(h, best_val, time_limit):
    cur = h.copy()
    cur_val = best_val
    n = len(cur)
    while time.time() < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur[i], cur[j]
        delta_max_pos = min(1 - hi, hj)
        delta_max_neg = min(hi, 1 - hj)
        if delta_max_pos + delta_max_neg == 0:
            continue
        delta = rng.uniform(-delta_max_neg, delta_max_pos)
        if abs(delta) < 1e-12:
            continue
        cur[i] = hi + delta
        cur[j] = hj - delta
        new_val = compute_c5(cur)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur.copy()
        else:
            # revert
            cur[i] = hi
            cur[j] = hj
    return best_h, best_val
```
