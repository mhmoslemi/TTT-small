def project_onto_bounds_and_sum(v, target_sum):
    # Clip into [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Now adjust to meet sum constraint while staying in bounds.
    # Use iterative water-filling.
    total = v.sum()
    if abs(total - target_sum) < 1e-12:
        return v
    # Determine direction.
    if total > target_sum:
        # need to reduce sum, can only reduce values above 0.
        # We'll iteratively subtract equally from non-zero entries.
        while total > target_sum + 1e-12:
            # indices that are > 0
            mask = v > 1e-12
            if not np.any(mask):
                break
            # compute amount to subtract per free variable
            excess = total - target_sum
            # distribute equally among free
            # compute reduction = excess / count
            cnt = mask.sum()
            reduction = excess / cnt
            # Ensure not reduce below 0
            reduction = min(reduction, v[mask].min())
            v[mask] -= reduction
            total = v.sum()
    else:
        # need to increase sum, can only increase values < 1.
        while total < target_sum - 1e-12:
            mask = v < 1 - 1e-12
            if not np.any(mask):
                break
            deficit = target_sum - total
            cnt = mask.sum()
            addition = deficit / cnt
            addition = min(addition, 1 - v[mask].max())
            v[mask] += addition
            total = v.sum()
    # final clip just in case
    v = np.clip(v, 0.0, 1.0)
    return v

def project_onto_bounds_and_sum(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    diff = total - target_sum
    # tolerance
    eps = 1e-12
    if abs(diff) <= eps:
        return v
    if diff > 0:
        # need decrease
        while diff > eps:
            free = v > 0
            cnt = free.sum()
            if cnt == 0:
                break
            # max reduction possible per free entry = its current value
            max_reduc = v[free]
            # we need to reduce total by diff.
            reduc_per = diff / cnt
            # limit by max avoid negative
            reduc = np.minimum(reduc_per, max_reduc)
            v[free] -= reduc
            total = v.sum()
            diff = total - target_sum
    else:
        # need increase
        while diff < -eps:
            free = v < 1
            cnt = free.sum()
            if cnt == 0:
                break
            max_inc = 1 - v[free]
            inc_per = -diff / cnt
            inc = np.minimum(inc_per, max_inc)
            v[free] += inc
            total = v.sum()
            diff = total - target_sum
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v

import time
import numpy as np

def _project_onto_bounds_and_sum(v, target, tol=1e-9, max_iter=60):
    """
    Project vector v onto the set {x| 0<=x_i<=1 , sum(x)=target}
    using a simple binary search for the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0
    # binary search for lambda
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction to hit the sum more accurately
    diff = target - x.sum()
    if abs(diff) > tol:
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt > 0:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _max_overlap(h):
    """return max_{k} ∫ h(x)(1‑h(x+k)) dx for the current discretisation"""
    n = h.size
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode="full")
    return ov.max() * dx, ov

def _gradient_of_overlap(h, shift):
    """gradient of the overlap corresponding to the given integer shift"""
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        m = n - shift
        if m > 0:
            grad[:m] += 1.0 - h[shift:]
        if shift < n:
            grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        m = n - s
        if m > 0:
            grad[s:] += 1.0 - h[:m]
        if s < n:
            grad[:m] -= h[s:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise problem size and starting vectors
    # ------------------------------------------------------------------
    try:

    except NameError:
        n = 128                     # fallback size
        init = None
        use_initial = False

    target_sum = n * 0.5           # because ∫h = 1 and dx = 2/n

    # ------------------------------------------------------------------
    # helpers for evaluation
    # ------------------------------------------------------------------
    def evaluate(h):
        c5, _ = _max_overlap(h)
        return float(c5)

    # ------------------------------------------------------------------
    # optimisation loop (sub‑gradient descent + random swaps)
    # ------------------------------------------------------------------
    start_time = time.time()
    best_h = None
    best_c5 = np.inf

    # parameters

    swap_prob = 0.12               # probability of doing a random swap instead of a gradient step
    swap_eps = 0.07                # max amount moved in a swap

    for restart in range(max_restarts):
        # time guard
        if time.time() - start_time > budget_s * 0.96:
            break

        if restart == 0 and use_initial:
            h = init.copy()
        else:
            h = rng.random(n)
        h = _project_onto_bounds_and_sum(h, target_sum)

        cur_c5 = evaluate(h)
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()

        it = 0
        while True:

            # time check each 200 iterations
            if it % 200 == 0 and (time.time() - start_time > budget_s * 0.96):
