def project_to_constraints(h):
    # Clip to [0,1], then adjust sum to target using a simple method
    h = np.clip(h, 0, 1)
    target = len(h) / 2.0
    current = h.sum()
    # adjust by adding uniform offset then clip; iterate a little if needed
    if abs(current - target) > 1e-8:
        # compute deficit
        diff = target - current
        # distribute diff proportionally to free entries (not at bounds)
        # Simple approach: add diff/n to all and clip again, repeat.
        # ensure we converge.
        for _ in range(10):
            free = (h > 0) & (h < 1)
            n_free = free.sum()
            if n_free == 0:
                break
            add = diff / n_free
            # update within bounds
            h[free] += add
            # clip
            h = np.clip(h, 0, 1)
            new_current = h.sum()
            diff = target - new_current

def project_to_constraints(h, target_sum):
    # Clip to [0,1]
    h_clipped = np.clip(h, 0, 1)
    # Simple iterative water-filling to fix sum.
    # Compute current sum.
    cur = h_clipped.sum()
    diff = target_sum - cur
    # Iterate adjusting free variables away from bounds.
    # We'll use while loop limited 100 iterations.
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        # Determine indices that are not at bounds (strictly inside)
        free = (h_clipped > 0) & (h_clipped < 1)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust; give up
            break
        delta = diff / n_free
        # add delta to free entries
        h_clipped[free] += delta
        # clip again
        h_clipped = np.clip(h_clipped, 0, 1)
        # recompute diff
        cur = h_clipped.sum()
        diff = target_sum - cur

import time
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(h):
    """Return the overlap constant for a step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_to_constraints(x, target_sum, eps=1e-12):
    """
    Project a vector onto the feasible set:
    0 <= x_i <= 1  and  sum(x) = target_sum.
    Simple water‑filling iteration; works well for the modest dimensions used here.
    """
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target_sum - cur
    # Distribute the remaining mass uniformly over the free variables.
    for _ in range(50):
        if abs(diff) < eps:
            break
        free = (x > 0.0) & (x < 1.0)
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target_sum - cur
    # Final tiny correction (may slightly violate bounds, then re‑clip)
    if abs(diff) > eps:
        x += diff / x.size
        x = np.clip(x, 0.0, 1.0)
    return x

        h0 = np.full(n, 0.5)       # uniform start satisfies the integral automatically

    target_sum = n / 2.0
    h0 = project_to_constraints(h0, target_sum)   # guarantee feasibility

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # objective with a light penalty for the integral constraint
    # ------------------------------------------------------------------
    penalty_weight = 15.0

    def objective(x):
        # penalty on the integral (sum) constraint
        s = np.sum(x)
        pen = penalty_weight * (s - target_sum) ** 2
        return compute_c5(x) + pen

    bounds = [(0.0, 1.0)] * n

    # ------------------------------------------------------------------
    # global search with Differential Evolution
    # ------------------------------------------------------------------
    # The DE call itself is fast for this dimension; we set modest
    # population size and iterations so that we stay well inside the
    # time budget.
    de_max_iter = 400          # generations
    de_popsize = 10            # each generation holds popsize * dim individuals
    result = differential_evolution(
        objective,
        bounds,
        maxiter=de_max_iter,
        popsize=de_popsize,
        seed=seed,
        polish=False,
        disp=False,
    )
    cand = project_to_constraints(result.x, target_sum)
    cand_val = compute_c5(cand)
    if cand_val < best_val:
        best_h, best_val = cand, cand_val

    # ------------------------------------------------------------------
    # short local refinement (pairwise mass transfers)
    # ------------------------------------------------------------------
    # Simple hill‑climbing that preserves the total mass.
    max_local_iters = 120000
    max_step_initial = 0.3
    elapsed = lambda: time.time() - start_time

    cur_h = best_h.copy()
    cur_val = best_val
    max_step = max_step_initial
    no_improve = 0

    for it in range(max_local_iters):
        if elapsed() > budget_s * 0.97:
            break

        # linearly shrink the step size
        frac = it / max_local_iters
        step = max_step_initial * (1.0 - frac)

        # pick two distinct indices and move a random amount delta from one to the other
        i, j = np.random.choice(n, 2, replace=False)
        if np.random.rand() < 0.5:
            # move from i to j
            max_transfer = min(cur_h[i], step, 1.0 - cur_h[j])
            if max_transfer <= 0.0:
                continue
            delta = np.random.rand() * max_transfer
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move from j to i
            max_transfer = min(cur_h[j], step, 1.0 - cur_h[i])
            if max_transfer <= 0.0:
                continue
            delta = np.random.rand() * max_transfer
            new_h = cur_h.copy()
            new_h[j] -= delta
            new_h[i] += delta

        new_val = compute_c5(new_h)

        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
                # early exit if we beat the target bound
                if best_val <= 0.3808:
                    break
        else:
            no_improve += 1
            # occasionally accept a worse move to escape flat regions
            if no_improve > 2000:
                # random shake
                shake = np.random.uniform(-step, step, size=n)
                shaken = cur_h + shake
                shaken = np.clip(shaken, 0.0, 1.0)
                shaken = project_to_constraints(shaken, target_sum)
                shaken_val = compute_c5(shaken)
                if shaken_val < cur_val:
                    cur_h, cur_val = shaken, shaken_val
                no_improve = 0

    # final projection to fix any tiny drift
    best_h = project_to_constraints(best_h, target_sum)
    best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Return result as required by the harness
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_val), int(n))
