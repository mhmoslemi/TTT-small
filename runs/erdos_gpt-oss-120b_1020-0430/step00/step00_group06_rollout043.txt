def project_onto_sum_and_bounds(v, target):
    # find lambda using bisection so that sum(clip(v-lambda,0,1)) = target
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    # ensure lo and hi produce sums beyond target:
    # sum at lo should be > target, at hi should be < target
    # Because as lambda -> -inf, clip(v - lambda) -> 1 => sum = n
    # as lambda -> +inf, sum = 0.
    # But we can just set lo = -1e5, hi = 1e5.

    for _ in range(50):
        lam = (lo + hi) / 2
        w = np.clip(v - lam, 0, 1)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2, 0, 1)
    # may not hit exact target due to clipping at bounds; adjust if needed
    # compute difference
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        # distribute diff over non-bound variables
        # Find indices where 0 < w_i < 1
        free = (w > 0) & (w < 1)
        if np.any(free):
            w[free] += diff / free.sum()
            # re-clamp
            w = np.clip(w, 0, 1)
        else:
            # all bound, fallback to scaling (will violate bounds possibly)
            w = np.clip(w + diff / len(w), 0, 1)
    # after final adjust, sum might be slightly off; we accept.
    return w

def project_onto_sum_and_bounds(v, target):
    # Project v onto box [0,1] intersect hyperplane sum=v = target.
    # using lambda to shift values.
    lo = -1.0
    hi = 2.0
    # ensure enclose solution
    for _ in range(70):
        lam = (lo + hi) / 2
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2, 0.0, 1.0)
    # If due to clipping sum deviates slightly, we can fine-tune: adjust free variables
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        # distribute diff equally among non-bound positions
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback: adjust uniformly then clip (might break sum slightly)
            w += diff / len(w)
            w = np.clip(w, 0.0, 1.0)
    return w

import time, math, random
import numpy as np

def _project_onto_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the hyperplane sum(v)=target with box 0<=v<=1."""
    lo, hi = -1.0, 2.0
    # bisection for the Lagrange multiplier
    for _ in range(70):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment to hit the sum exactly
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w += diff / w.size
            w = np.clip(w, 0.0, 1.0)
    return w

def _raw_max_corr(vec: np.ndarray) -> float:
    """Maximum of the discrete overlap integral (unnormalised by dx)."""
    return np.max(np.correlate(vec, 1.0 - vec, mode='full'))

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # choose a moderate number of points (even, <1000)
    n = 200
    # if a parent construction is supplied and not too large, reuse its length
    if 'initial_h_values' in globals():
        init_len = len(initial_h_values)
        if init_len <= 500:
            n = init_len
    if n % 2:
        n += 1

    target_sum = n / 2.0
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # 1) Simple two-parameter family: outer blocks of height 1,
    #    middle region of constant height c.
    # ------------------------------------------------------------------
    best_raw = float('inf')
    best_h = None

    steps = 200
    for step in range(1, steps):
        # L is the length of each outer block (0 < L < 0.5)
        L = (step / steps) * 0.49  # avoid the pathological L=0.5
        c = (0.5 - L) / (1.0 - L) if L < 1.0 else 0.0
        # build the piecewise constant vector
        h = np.full(n, c, dtype=float)
        block_pts = int(round(L * n / 2.0))
        if block_pts > 0:
            h[:block_pts] = 1.0
            h[-block_pts:] = 1.0
        h = _project_onto_sum_and_bounds(h, target_sum)
        raw = _raw_max_corr(h)
        if raw < best_raw:
            best_raw, best_h = raw, h.copy()
        # early exit if already good enough
        if best_raw * dx <= 0.3809:
            break

    # ------------------------------------------------------------------
    # 2) Local optimisation: hill‑climbing on a binary representation.
    #    Start from the best candidate found above (rounded to 0/1).
    # ------------------------------------------------------------------
    # create a binary seed from best_h (keep the top half as 1)
    binary = np.zeros(n, dtype=float)
    ones_idx = np.argsort(-best_h)[:n // 2]
    binary[ones_idx] = 1.0
    h = binary.copy()
    cur_raw = _raw_max_corr(h)
    if cur_raw < best_raw:
        best_raw, best_h = cur_raw, h.copy()

    # parameters for the stochastic search
    temperature = 2.0
    cooling = 0.9999
    no_improve = 0
    max_iter = 10_000_000  # safety bound
    iter_cnt = 0

    while time.time() - start < budget_s * 0.65 and iter_cnt < max_iter:
        # pick a random 1 and a random 0 and swap them
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = random.choice(ones.tolist())
        j = random.choice(zeros.tolist())
        h_new = h.copy()
        h_new[i], h_new[j] = 0.0, 1.0
        new_raw = _raw_max_corr(h_new)
        delta = new_raw - cur_raw
        accept = False
        if delta < 0:
            accept = True
        else:
            if temperature > 1e-12 and random.random() < math.exp(-delta / temperature):
                accept = True
        if accept:
            h = h_new
            cur_raw = new_raw
            no_improve = 0 if delta < 0 else no_improve + 1
            if cur_raw < best_raw:
                best_raw, best_h = cur_raw, h.copy()
                if best_raw * dx <= 0.38085:
                    break
        else:
            no_improve += 1

        temperature *= cooling
        if no_improve > 5000:
            # random restart to avoid stagnation
            h = np.zeros(n)
            idx = np.random.choice(n, n // 2, replace=False)
            h[idx] = 1.0
            cur_raw = _raw_max_corr(h)
            no_improve = 0
        iter_cnt += 1

    # ------------------------------------------------------------------
    # 3) Fine‑grained projected sub‑gradient descent starting from best_h.
    # ------------------------------------------------------------------
    h = best_h.copy()
    cur_raw = best_raw
    it = 0
    temp2 = 0.5
    while time.time() - start < budget_s * 0.95:
        # correlation vector (full mode)
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(np.argmax(corr))
        k = idx_max - (n - 1)        # shift that attains the maximum
        # gradient of that worst shift
        g = np.ones_like(h)
        if k >= 0:
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
        else:
            kk = -k
            g[kk:] -= h[:n - kk]
            g[:n - kk] -= h[kk:]
        # step size decreasing with iteration count
        step = 0.6 / (1 + it * 0.001)
        h_new = h - step * g
        h_new = _project_onto_sum_and_bounds(h_new, target_sum)
        new_raw = _raw_max_corr(h_new)
        if new_raw < best_raw:
            best_raw, best_h = new_raw, h_new.copy()
            if best_raw * dx <= 0.38085:
                break
        if new_raw <= cur_raw:
            h, cur_raw = h_new, new_raw
        else:
            # occasional uphill acceptance
            if random.random() < math.exp(-(new_raw - cur_raw) / temp2):
                h, cur_raw = h_new, new_raw
        temp2 *= 0.99995
        it += 1
        if it > 500000:
            break

    c5_bound = best_raw * dx
    return best_h.tolist(), float(c5_bound), n
