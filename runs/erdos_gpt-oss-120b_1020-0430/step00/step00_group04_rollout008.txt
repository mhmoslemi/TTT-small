def repair_binary(h):
    # ensure exactly half ones
    n = len(h)

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_bounds_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    # simple water-filling
    delta = target - s
    # iterate
    for _ in range(1000):
        if abs(delta) < 1e-12:
            break
        if delta > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        add = delta / cnt
        # Update only masked entries
        v[mask] = np.clip(v[mask] + add, 0.0, 1.0)
        s = v.sum()
        delta = target - s
    # final slight adjust by distributing remainder to some feasible positions
    if abs(v.sum() - target) > 1e-9:
        diff = target - v.sum()
        for i in range(len(v)):
            if 0 <= v[i] + diff <= 1:
                v[i] += diff
                break
    return v

def random_binary(n):
    h = np.zeros(n, dtype=float)
    h[:n//2] = 1.0
    np.random.shuffle(h)
    return h

def random_continuous(n, target):
    # start from uniform and project
    h = np.random.rand(n)
    return project_to_bounds_sum(h, target)

def swap_move(h):
    # binary swap: pick a 1 and 0 and swap
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return h_new

        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):

            cur = val
            if val < best:
                best = val
                best_h = h.copy()
        if best <= 0.3808:
            break
    return best_h, best

def upsample_vector(v, target_n):
    m = len(v)
    if m == target_n:
        return v.astype(float)
    # repeat each element floor times
    repeat = target_n // m
    rem = target_n - repeat * m
    arr = np.repeat(v, repeat)
    if rem > 0:
        arr = np.concatenate([arr, v[:rem]])
    return arr.astype(float)

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector onto the box [0,1]^n while fixing the sum to *target*."""
    v = np.clip(v, 0.0, 1.0)
    delta = target - v.sum()
    if abs(delta) < 1e-12:
        return v
    # simple water‑filling: distribute the remaining mass uniformly among
    # components that are not at a bound.
    for _ in range(1000):
        if abs(delta) < 1e-12:
            break
        if delta > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        add = delta / cnt
        v[mask] = np.clip(v[mask] + add, 0.0, 1.0)
        delta = target - v.sum()
    # tiny residual fix (unlikely to be needed)
    if abs(v.sum() - target) > 1e-9:
        diff = target - v.sum()
        for i in range(v.size):
            if 0.0 <= v[i] + diff <= 1.0:
                v[i] += diff
                break
    return v


def random_binary(n: int) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


def random_continuous(n: int, target: float) -> np.ndarray:
    """Random real vector in [0,1] with the prescribed sum."""
    h = np.random.rand(n)
    return _project_bounds_sum(h, target)


def _delta_move(h: np.ndarray, max_delta: float = 0.2) -> np.ndarray | None:
    """Move a small amount of mass from one coordinate to another."""
    n = h.size
    i, j = np.random.choice(n, size=2, replace=False)
    vi, vj = h[i], h[j]
    lower = -min(vj, max_delta)
    upper = min(1.0 - vi, max_delta)
    if upper <= lower:
        return None
    delta = np.random.uniform(lower, upper)
    h_new = h.copy()
    h_new[i] = vi + delta
    h_new[j] = vj - delta
    return h_new


def simulated_annealing_continuous(
    start_h: np.ndarray,
    max_moves: int,
    start_T: float,
    end_T: float,
    max_delta: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Simple SA that preserves the integral and box constraints."""
    h = start_h.copy()
    cur_val = compute_c5(h)
    best_h, best_val = h.copy(), cur_val
    for k in range(max_moves):
        T = start_T + (end_T - start_T) * (k / max_moves)
        # try a feasible delta move; give a few attempts before skipping
        move = None
        for _ in range(3):
            mv = _delta_move(h, max_delta)
            if mv is not None:
                move = mv
                break
        if move is None:
            continue
        val = compute_c5(move)
        d = val - cur_val
        if d < 0 or np.random.rand() < np.exp(-d / (T + 1e-12)):
            h, cur_val = move, val
            if val < best_val:
                best_h, best_val = h.copy(), val
        if best_val <= 0.3808:
            break
    return best_h, best_val


def _upsample_vector(v: np.ndarray, target_n: int) -> np.ndarray:
    """Repeat entries to reach *target_n* (last part may be truncated)."""
    m = v.size
    if m == target_n:
        return v.astype(float)
    repeat = target_n // m
    remainder = target_n - repeat * m
    out = np.repeat(v, repeat)
    if remainder:
        out = np.concatenate([out, v[:remainder]])
    return out.astype(float)

    best_val = compute_c5(best_h)
