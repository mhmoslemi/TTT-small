import time
import numpy as np
from scipy.optimize import minimize

# Global variable used by constraint functions (set before each optimization)
CURRENT_TARGET = None  # sum of h (integral constraint)

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a given step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def eq_sum_constraint(x: np.ndarray) -> float:
    """Equality: sum of h (all but the last entry) must equal CURRENT_TARGET."""
    return float(np.sum(x[:-1]) - CURRENT_TARGET)

def ineq_corr_constraint(x: np.ndarray) -> np.ndarray:
    """Inequalities: t - correlation(k) >= 0 for all integer shifts k."""
    h = x[:-1]
    t = x[-1]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr  # vector, each entry must be non‑negative

def objective_t(x: np.ndarray) -> float:
    """Objective for SLSQP formulation: minimise the auxiliary variable t."""
    return float(x[-1])

def random_feasible_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random vector of length n with entries in [0,1] and
    sum equal to n/2 (i.e. integral = 1 on [0,2]).
    Simple sequential sampling respecting remaining sum and bounds.
    """
    target = n / 2.0
    h = np.empty(n)
    remaining = target
    for i in range(n - 1):
        low = max(0.0, remaining - (n - i - 1) * 1.0)
        high = min(1.0, remaining)
        h[i] = rng.uniform(low, high)
        remaining -= h[i]
    h[-1] = remaining
    rng.shuffle(h)
    return h

def local_optimize_slsqp(h0: np.ndarray, maxiter: int = 200) -> tuple:
    """
    Perform a local optimisation starting from h0.
    Returns (h_opt, t_opt) where t_opt is the achieved C5 bound.
    """
    global CURRENT_TARGET
    n = h0.shape[0]
    CURRENT_TARGET = n / 2.0
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]  # last bound for t
    cons = [
        {'type': 'eq', 'fun': eq_sum_constraint},
        {'type': 'ineq', 'fun': ineq_corr_constraint}
    ]

    res = minimize(
        objective_t,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    CURRENT_TARGET = None

    if not res.success:
        return h0, t0

    h_opt = np.clip(res.x[:-1], 0.0, 1.0)
    # enforce exact sum (tiny numeric drift)
    diff = h_opt.sum() - n / 2.0
    if abs(diff) > 1e-12:
        # adjust uniformly; stays inside bounds because diff is tiny
        h_opt -= diff / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    t_opt = float(res.x[-1])
    return h_opt, t_opt

    """

    start_time = time.time()

    # ------------------------------------------------------------------

        if s != 0:
