def _project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the intersection of the box [0,1]^n
    and the affine hyperplane sum(v) = target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 2.0          # wide enough bounds for the multiplier
    for _ in range(60):        # enough for double precision
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(v: np.ndarray) -> float:
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx  (discrete version)."""
    n = v.shape[0]
    dx = 2.0 / n
    conv = np.correlate(v, 1.0 - v, mode='full')
    return float(conv.max() * dx)


def _max_y_shift(v: np.ndarray):
    """
    Returns the maximal overlap value (without the dx factor) and the
    integer shift k at which it occurs.
    """
    n = v.shape[0]
    y = np.correlate(v, 1.0 - v, mode='full')
    idx = int(np.argmax(y))
    max_val = float(y[idx])
    shift = idx - (n - 1)          # negative shifts are allowed
    return max_val, shift


def _gradient_of_shift(v: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the overlap functional with respect to v for a fixed shift k.
    The functional for a shift k is:
        y_k(v) = Σ_i v_i (1 - v_{i+k})
    whose gradient is 1 - (v_{i+k} + v_{i‑k}) (with out‑of‑range terms omitted).
    """
    n = v.shape[0]
    g = np.ones_like(v)
    if k >= 0:
        if k != 0:
            # subtract v_{i+k}
            g[: n - k] -= v[k:]
            # subtract v_{i‑k}
            g[k:] -= v[: n - k]
        else:                     # k == 0  →  gradient = 1 - 2*v_i
            g -= 2.0 * v
    else:
        kk = -k
        if kk != 0:
            g[: n - kk] -= v[kk:]
            g[kk:] -= v[: n - kk]
        else:
            g -= 2.0 * v
    return g


def _subgradient_descent(v0: np.ndarray,
                         max_iter: int = 800,
                         step0: float = 0.15,
                         decay: float = 0.995) -> tuple:
    """
    Simple projected sub‑gradient descent for the C₅ functional.
    Returns the best vector encountered and its C₅ value.
    """
    n = v0.shape[0]
    target = n / 2.0
    v = v0.copy()
    best_v = v.copy()
    best_val = _c5_value(v)
    step = step0

    for _ in range(max_iter):
        # current worst shift and gradient
        _, k = _max_y_shift(v)
        g = _gradient_of_shift(v, k)

        # descent step and projection onto feasible set
        v = v - step * g
        v = _project_sum_box(v, target)

        # evaluation
        cur_val = _c5_value(v)
        if cur_val < best_val:
            best_val = cur_val
            best_v = v.copy()

        step *= decay

    return best_v, best_val


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return _project_sum_box(v, n / 2.0)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
      - h_values is a numpy array of step‑function values,
      - c5_bound is the achieved upper bound for C₅,
      - n_points is the number of sampled points (length of h_values).
    The function respects the time budget by stopping early if needed.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Choose the discretisation size.
    # If a parent construction is supplied, keep its size (it is already
    # a reasonably small grid).  Otherwise fall back to a modest size.
    # -----------------------------------------------------------------
    try:
        h_initial = np.asarray(initial_h_values, dtype=np.float64)
        n = h_initial.shape[0]
    except NameError:
        n = 200                     # a safe default when no parent is given
        h_initial = _random_feasible(n, rng)

    # Ensure feasibility (projection also fixes tiny rounding errors)
    h_initial = _project_sum_box(h_initial, n / 2.0)

    # Best solution seen so far
    best_h = h_initial.copy()
    best_val = _c5_value(best_h)

    # -------------------------------------------------------------
    # Parameters for the local search
    # -------------------------------------------------------------
    # Number of gradient iterations per restart – moderate to stay fast.
    iters_per_restart = 1200 if n <= 100 else 800
    # Initial step size – scaled with grid size.
    step0 = 0.2 if n <= 100 else 0.12
    decay = 0.996

    # -------------------------------------------------------------
    # Main optimisation loop (time‑bounded)
    # -------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.95:
        # Choose a starting point:
        #   – occasionally reuse the current best with a small random perturbation,
        #   – otherwise start from an independent random feasible vector.
        if rng.random() < 0.35:
            # perturb best_h a little while preserving the sum
            # pick two indices and transfer a random amount between them
            v = best_h.copy()
            i, j = rng.choice(n, size=2, replace=False)
            # maximal amount we can move from i to j without violating bounds
            max_move = min(v[i], 1.0 - v[j])
            if max_move > 1e-12:
                delta = rng.uniform(0, max_move)
                v[i] -= delta
                v[j] += delta
        else:
            v = _random_feasible(n, rng)

        # Run the gradient descent from this start
        v_opt, val_opt = _subgradient_descent(v,
                                              max_iter=iters_per_restart,
                                              step0=step0,
                                              decay=decay)

        # Keep the best overall result
        if val_opt < best_val:
            best_val = val_opt
            best_h = v_opt.copy()
            # Very good result – we can stop early
            if best_val <= 0.3808:
                break

    # Final projection to be absolutely safe about the constraints
    best_h = _project_sum_box(best_h, n / 2.0)
    final_c5 = _c5_value(best_h)

    # Return the solution
    return best_h, float(final_c5), n
