        s = x.sum()
        if s > target:

    diff = target - x.sum()

        free = (x > 0) & (x < 1)

            # all at bounds, cannot adjust: ignore
            pass
    return x

import numpy as np, time

def project_to_sum(v, target):
    # binary search λ
    lo = -np.max(v) - 5.0
    hi = np.max(v) + 5.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # small correction
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff to non-boundary elements
        free = (x > 0) & (x < 1)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            np.clip(x, 0.0, 1.0, out=x)
    return x

def max_corr(h, dx):
    return float(np.correlate(h, 1 - h, mode='full').max() * dx)

    it = 0

    while time.time() - start < budget_s * 0.96:
        if rng.random() < prob_grad:
            # ----- gradient move (sub‑gradient of the worst shift) -----
            corr_full = np.correlate(h, 1.0 - h, mode='full')
            max_idx = int(np.argmax(corr_full))
            lag = max_idx - (n - 1)

            grad = np.ones_like(h)
            if lag > 0:
                grad[: n - lag] -= h[lag:]
                grad[lag:]       -= h[: n - lag]
            elif lag < 0:
                l = -lag
                grad[l:]   -= h[: n - l]
                grad[: l]  -= h[l:]
            else:                     # lag == 0
                grad -= 2.0 * h

            cand = h - lr * grad
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_to_sum(cand, target_sum)

            cand_c = _max_corr(cand, dx)
            cur_c  = _max_corr(h, dx)

            # accept if better, otherwise accept with a tiny probability
            if cand_c < cur_c or rng.random() < 0.02:
                h = cand
        else:
            # ----- random swap move preserving the sum exactly ----------
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            hi, hj = h[i], h[j]

            # feasible delta so that both stay in [0,1]
            lo = max(-hi,     hj - 1.0)
            hi_ = min(1.0 - hi, hj)
            if lo >= hi_:
                continue
            delta = rng.uniform(lo, hi_)
            cand = h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta

            cand_c = _max_corr(cand, dx)
            cur_c  = _max_corr(h, dx)

            if cand_c < cur_c or rng.random() < 0.01:
                h = cand

        # keep the best ever seen
        cur_c = _max_corr(h, dx)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        # learning‑rate schedule
        lr *= decay
        it += 1

        # occasional re‑randomisation from the current best
        if it % 5000 == 0:
            h = best_h + 0.05 * rng.standard_normal(n)
            h = np.clip(h, 0.0, 1.0)
            h = _project_to_sum(h, target_sum)

    # final projection and evaluation
    best_h = _project_to_sum(best_h, target_sum)
    final_c = _max_corr(best_h, dx)

    return best_h.tolist(), float(final_c), n
