def propose_move(h):
    n = len(h)
    i = rng.integers(n)
    j = rng.integers(n)
    while i == j:
        j = rng.integers(n)
    # Determine max delta such that h[i] + delta ∈ [0,1] and h[j] - delta ∈ [0,1].
    max_up = 1 - h[i]
    max_down = h[j]
    # We want delta in [-max_down, max_up] (since we add delta to i, subtract delta from j)
    # Choose delta drawn from uniform [-step, step] then clip to allowed range
    step = 0.1  # can adapt

    delta = np.clip(delta, -max_down, max_up)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h, delta

import time
import numpy as np

def _calc_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function h on [0,2] using the
    evaluation rule of the harness."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _random_swap_move(h: np.ndarray, rng: np.random.Generator, max_step: float):
    """Return a new vector obtained by moving a small amount of mass from one
    coordinate to another while keeping the total sum unchanged."""
    n = h.shape[0]
    i, j = rng.integers(n, size=2)
    while i == j:
        j = rng.integers(n)

    # allowable increase at i and decrease at j
    up_i = 1.0 - h[i]
    down_j = h[j]

    # draw a raw delta in [-max_step, max_step] and clamp to feasible range
    delta = (rng.random() * 2.0 - 1.0) * max_step
    delta = np.clip(delta, -down_j, up_i)

    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimize a step function for the Erdős C5 constant.

    Returns a tuple (h_values, c5_bound, n_points) where h_values is a
    NumPy array of the step heights, c5_bound its achieved C5 value and
    n_points the number of discretization points.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # start from the provided construction if present; otherwise use a
    # simple linear ramp (which already satisfies the integral constraint)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        n = 100
        # linear ramp h(x)=x/2 on [0,2]  -> discrete version
        idx = np.arange(n, dtype=np.float64) + 0.5
        h = idx / n
    n = h.shape[0]

    # ensure feasibility (sum must be n/2)
    target_sum = n / 2.0
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-12:
        # renormalize linearly while keeping bounds respected
        h = h * (target_sum / current_sum)

    best_h = h.copy()
    best_score = _calc_c5(best_h)
    current_h = h.copy()
    current_score = best_score

    # ------------------------------------------------------------------
    # Simple simulated‑annealing style local search
    # ------------------------------------------------------------------
    start_time = time.time()
    max_step = 0.2          # maximal mass moved in a single swap
    init_temp = best_score * 0.5
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # linear cooling schedule
        temp = init_temp * (1.0 - elapsed / (budget_s * 0.95))

        # propose a move
        new_h = _random_swap_move(current_h, rng, max_step)
        new_score = _calc_c5(new_h)

        if new_score < current_score:
            # always accept improvement
            current_h = new_h
            current_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
        else:
            # maybe accept a worsening move (simulated annealing)
            if temp > 0.0:
                prob = np.exp(-(new_score - current_score) / temp)
                if rng.random() < prob:
                    current_h = new_h
                    current_score = new_score

    # final safety: enforce exact integral (tiny rounding errors)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny correction to a non‑boundary element
        for i in range(n):
            if 0.0 < best_h[i] < 1.0:
                best_h[i] += diff
                break

    best_h = np.clip(best_h, 0.0, 1.0)   # final safety clamp
    final_c5 = _calc_c5(best_h)

    return best_h, final_c5, n
