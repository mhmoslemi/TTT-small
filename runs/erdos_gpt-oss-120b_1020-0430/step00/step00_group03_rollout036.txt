def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, random, math
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    total_budget = budget_s

    N = 100
    w = N // 2
    # generate random balanced binary vector
    h = np.zeros(N, dtype=np.int8)
    h[:w] = 1
    np.random.shuffle(h)
    # compute initial score
    max_conv = compute_max_conv(h)
    best_h = h.copy()
    best_score = max_conv

    # Prepare indices for ones and zeros
    ones_idx = np.flatnonzero(h == 1)
    zeros_idx = np.flatnonzero(h == 0)

    # Simulated annealing parameters
    # Target bound: log late stop
    # Temperature schedule linear decay
    T0 = 5.0
    # loops
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= total_budget * 0.95:
            break
        # propose swap
        # pick random one and zero index
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)
        # swap
        h[i], h[j] = 0, 1
        # compute new score
        new_score = compute_max_conv(h)
        # decide accept
        delta = new_score - max_conv
        # temperature linear
        T = max(0.1, T0 * (1 - elapsed / total_budget))
        accept = False
        if delta <= 0:
            accept = True
        else:
            if random.random() < math.exp(-delta / T):
                accept = True
        if accept:
            max_conv = new_score
            # update indices
            # i becomes zero, j becomes one
            # update arrays efficiently using list?
            # Use sets for O(1) removal and addition:
            # Actually we can just recompute ones_idx and zeros_idx occasionally.
            # But more efficient: maintain python sets.
        else:
            # revert swap
            h[i], h[j] = 1, 0
        # Update best
        if max_conv < best_score:
            best_score = max_conv
            best_h = h.copy()
            # maybe print progress
        # maybe occasionally recompute indices in case of stale
        it += 1

    # fast container for indices of 1s and 0s
    ones = set(int(i) for i, v in enumerate(h) if v)
    zeros = set(int(i) for i, v in enumerate(h) if not v)

    # simulated‑annealing schedule
    T0 = 5.0
    it_without_imp = 0
    max_without_imp = 200_000   # restart after many non‑improving steps

    total_time = budget_s
    while True:
        elapsed = time.time() - start
        if elapsed >= total_time * 0.95:
            break

        # pick a random 1–0 pair and propose a swap
        i = random.sample(ones, 1)[0]
        j = random.sample(zeros, 1)[0]

        # perform swap
        h[i] = 0
        h[j] = 1
        new_max = _max_overlap(h)

        delta = new_max - cur_max
        # temperature linearly decreasing
        T = max(0.1, T0 * (1.0 - elapsed / total_time))
        accept = False
        if delta <= 0:
            accept = True
        else:
            if random.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_max = new_max
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            it_without_imp = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                # early exit if target reached
                if best_max <= target_max:
                    break
        else:
            # revert swap
            h[i] = 1
            h[j] = 0

        it_without_imp += 1
        # occasional full recompute of index sets (cheap) to avoid drift
        if it_without_imp % 5000 == 0:
            ones = set(int(i) for i, v in enumerate(h) if v)
            zeros = set(int(i) for i, v in enumerate(h) if not v)
        # restart if stuck too long
        if it_without_imp > max_without_imp:
            h = np.zeros(N, dtype=np.int8)
            h[:half] = 1
            np.random.shuffle(h)
            cur_max = _max_overlap(h)
            ones = set(int(i) for i, v in enumerate(h) if v)
            zeros = set(int(i) for i, v in enumerate(h) if not v)
            it_without_imp = 0

    # final bound
    bound = best_max * dx
    # return as Python list of floats
    return (best_h.astype(float).tolist(), float(bound), N)
