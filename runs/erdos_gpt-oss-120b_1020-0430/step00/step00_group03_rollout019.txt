def attempt_directed_move(h, k):
    L = n - abs(k)
    if L <= 0:
        return None, None
    i_range = np.arange(start, start + L)
    partners = i_range + k
    coeff = 1 - h[partners]  # shape L
    # donor candidates: where h[i] > eps and coeff large
    # recipient: where h[i] < 1-eps and coeff small
    donor_candidates = np.where(h[i_range] > 1e-6)[0]
    recipient_candidates = np.where(h[i_range] < 1 - 1e-6)[0]
    if len(donor_candidates) == 0 or len(recipient_candidates) == 0:
        return None, None
    # choose i_max = donor with max coeff
    donor_idx_rel = donor_candidates[np.argmax(coeff[donor_candidates])]
    recipient_idx_rel = recipient_candidates[np.argmin(coeff[recipient_candidates])]
    i = i_range[donor_idx_rel]
    j = i_range[recipient_idx_rel]
    # compute epsilon
    epsilon = min(h[i], 1 - h[j])
    if epsilon <= 1e-9:
        return None, None
    # optionally limit epsilon to e.g., 0.1 to avoid huge jumps
    epsilon = min(epsilon, 0.1)
    # apply move
    h_new = h.copy()
    h_new[i] -= epsilon
    h_new[j] += epsilon
    # ensure numeric bounds
    h_new = np.clip(h_new, 0.0, 1.0)
    # adjust sum maybe small due to clipping? Should be preserved exactly if we only moved epsilon not clipped; but after clip epsilon already ensures within bounds.
    return h_new, (i,j,epsilon)
```

def get_partner_sum(h, k):
    n = len(h)
    ps = np.zeros(n)
    if k >= 0:
        # for i where i + k < n: add h[i + k] to ps[i]
        i = np.arange(0, n - k)
        ps[i] += h[i + k]
        # for i where i - k >=0: i>=k: add h[i - k]
        i2 = np.arange(k, n)

        i2 = np.arange(0, n - kp)
        ps[i2] += h[i2 + kp]

            i1 = np.arange(0, n - k)
            ps[i1] += h[i1 + k]
            # i such that i - k >=0 => i >= k
            i2 = np.arange(k, n)
            ps[i2] += h[i2 - k]
        else:
            kp = -k
            # i such that i - kp >=0 => i >= kp
            i1 = np.arange(kp, n)
            ps[i1] += h[i1 - kp]
            # i such that i + kp < n => i < n - kp
            i2 = np.arange(0, n - kp)
            ps[i2] += h[i2 + kp]

        # donors: lower ps (small partner sum) with h > eps
        eps = 1e-8
        donor_candidates = np.where(h > eps)[0]
        recipient_candidates = np.where(h < 1 - eps)[0]
        if donor_candidates.size == 0 or recipient_candidates.size == 0:
            break
        donor_idx = donor_candidates[np.argmin(ps[donor_candidates])]
        recipient_idx = recipient_candidates[np.argmax(ps[recipient_candidates])]
        if donor_idx == recipient_idx:
            # pick random alternative
            # skip this iteration
            continue
        # step size factor
        step_factor = 0.3  # could vary
        epsilon = min(h[donor_idx], 1 - h[recipient_idx]) * step_factor
        if epsilon < 1e-10:
            # try random move
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            delta = min(h[i], 1 - h[j]) * step_factor
            if delta <= 0:
                continue
            donor_idx, recipient_idx = i, j
            epsilon = delta

        h_new = h.copy()
        h_new[donor_idx] -= epsilon
        h_new[recipient_idx] += epsilon
        # ensure bounds
        h_new = np.clip(h_new, 0.0, 1.0)
        # optional small fix to sum
        # sum is unchanged theoretically
        # but due to rounding:
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-12:
            # adjust a non-saturated index
            free = np.where((h_new > eps) & (h_new < 1 - eps))[0]
            if free.size == 0:
                free = np.arange(n)
            h_new[free[0]] += diff
            h_new = np.clip(h_new, 0.0, 1.0)

        new_val = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        # acceptance rule
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # temperature
            progress = it / max_iters
            T = T0 * (1 - progress) + Tf * progress
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            h = h_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                # early exit if beat target
                if best_val <= 0.3808:
                    break
        # occasional random perturbation for diversification
        if it % 500 == 0:
            if np.random.rand() < 0.3:
                i = np.random.randint(n)
                j = np.random.randint(n-1)
                if j >= i:
                    j += 1
                delta = min(h[i], 1 - h[j]) * np.random.uniform(0.05, 0.3)
                if delta > 0:
                    h_rand = h.copy()
                    h_rand[i] -= delta
                    h_rand[j] += delta
                    h_rand = np.clip(h_rand, 0, 1)
                    # fix sum if needed
                    diff = target_sum - h_rand.sum()
                    if abs(diff) > 1e-12:
                        free = np.where((h_rand > eps) & (h_rand < 1 - eps))[0]
                        if free.size == 0:
                            free = np.arange(n)
                        h_rand[free[0]] += diff
                        h_rand = np.clip(h_rand, 0, 1)
                    val_rand = np.max(np.correlate(h_rand, 1 - h_rand, mode='full')) * dx
                    if val_rand < cur_val:
                        h = h_rand
                        cur_val = val_rand
                        if cur_val < best_val:
                            best_val = cur_val
                            best_h = h.copy()
    # final repair to ensure sum exact
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        free = np.where((best_h > eps) & (best_h < 1 - eps))[0]
        if free.size > 0:
            best_h[free[0]] += diff
        else:
            best_h[0] += diff
        best_h = np.clip(best_h, 0.0, 1.0)
    # convert to list
    h_out = best_h.tolist()
    return (h_out, float(best_val), int(n))

import numpy as np
import math
import time

def _c5_value(h):
    """Compute C5 for a given h (numpy array)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _partner_sum(h, k):
    """Partner sum vector: for each i, sum of h at positions i±k that are inside the array."""
    n = h.shape[0]
    ps = np.zeros(n, dtype=np.float64)
    if k >= 0:
        i1 = np.arange(0, n - k)
        ps[i1] += h[i1 + k]
        i2 = np.arange(k, n)
        ps[i2] += h[i2 - k]
    else:
        kp = -k
        i1 = np.arange(kp, n)
        ps[i1] += h[i1 - kp]
        i2 = np.arange(0, n - kp)
        ps[i2] += h[i2 + kp]
    return ps

def _fix_sum(vec, target):
    """Adjust a single component to enforce exact sum, staying inside [0,1]."""
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    free = np.where((vec > 1e-9) & (vec < 1 - 1e-9))[0]
    if free.size == 0:
        free = np.arange(vec.shape[0])
    vec[free[0]] += diff
    np.clip(vec, 0.0, 1.0, out=vec)
    return vec
