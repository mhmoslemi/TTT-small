def hill_climb(h, target, max_iters, time_check=False):
    best = h.copy()
    best_score = max_overlap(best)
    cur = h.copy()
    cur_score = best_score
    n = len(h)
    it = 0
    while it < max_iters:
        if time_check and (time.time() - start_time) > budget_s - 0.5:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta_max = min(1.0 - cur[i], cur[j])
        if delta_max <= 0:
            continue
        delta = np.random.rand() * delta_max * 0.5
        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        new_score = max_overlap(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new
                best_score = new_score
        it += 1
    return best, best_score

import time
import numpy as np
from scipy.optimize import differential_evolution

def _project_sum_box(v, target):
    """Project vector v onto the hyperplane sum(v)=target with box 0≤v≤1."""
    x = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def _max_overlap(h):
    """Return C₅ for a discrete step function h."""
    return np.correlate(h, 1 - h, mode='full').max() * (2.0 / len(h))

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function (h), its C₅ bound, and the number of samples."""
    start = time.time()
    np.random.seed(seed)

    # ---- problem parameters -------------------------------------------------
    # number of discretisation points (chosen to be modest yet accurate)
    n = 360                         # 2 / dx = n  →  dx = 2/n
    target_sum = n / 2.0
    dx = 2.0 / n
    x = np.linspace(0.0, 2.0, n, endpoint=False) + dx / 2.0

    # Fourier basis size – a small number of cosines gives a rich yet
    # low‑dimensional search space while automatically satisfying the
    # integral constraint (the constant term 0.5 yields ∫h = 1).
    M = 6
    cos_mat = np.cos(np.outer(np.arange(1, M + 1), np.pi * x))   # shape (M,n)

    # ---- optimisation over Fourier coefficients -----------------------------
    bounds = [(-0.5, 0.5)] * M

    def coeff_objective(a):
        # raw periodic function, may leave [0,1]
        raw = 0.5 + np.dot(a, cos_mat)
        h = _project_sum_box(raw, target_sum)
        # overlap value (the objective to minimise)
        return np.correlate(h, 1 - h, mode='full').max() * dx

    # time‑aware DE configuration
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    # allow up to ~15 000 objective evaluations (enough for good quality)
    maxfun = min(15000, int(remaining * 2000))

    popsize = 8
    result = differential_evolution(

        bounds,

        init='random',
        seed=seed,

    if result.success:

    else:

    # construct the best function from the found coefficients
    raw_h = 0.5 + np.dot(best_a, cos_mat)
    best_h = _project_sum_box(raw_h, target_sum)
    best_score = _max_overlap(best_h)

    # ---- local refinement (pairwise mass transfer) -------------------------
    # remaining time decides how many hill‑climbing steps we can afford
    elapsed = time.time() - start

    # final safety projection (ensures exact sum)
    best_h = _project_sum_box(best_h, target_sum)
    best_score = _max_overlap(best_h)

    # -----------------------------------------------------------------------
    return (best_h, float(best_score), int(n))
