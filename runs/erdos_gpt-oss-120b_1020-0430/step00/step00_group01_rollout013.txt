def project_to_sum_box(y, S, tol=1e-9, max_iter=100):
    low = np.min(y) - 1.0
    high = np.max(y)
    for _ in range(max_iter):
        mid = (low+high)/2
        proj = np.clip(y - mid, 0, 1)
        s = proj.sum()
        if abs(s - S) <= tol:
            return proj
        if s > S:
            # need larger lambda to reduce sum (since increasing lambda reduces clipped values)
            low = mid
        else:
            high = mid
    return proj

import time
import numpy as np

def _project_to_sum_box(y: np.ndarray, S: float, tol: float = 1e-10, max_iter: int = 100) -> np.ndarray:
    """ Euclidean projection of y onto {0≤x≤1, sum(x)=S}. """
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - S) <= tol:
            return x
        if s > S:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate(h: np.ndarray) -> float:
    """ Return C₅ for a given step vector h. """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _gradient(h: np.ndarray):
    """ Sub‑gradient of max‑correlation at the worst shift. """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)          # integer shift that attains the maximum
    tp = np.zeros_like(h)          # h[j+shift]
    tm = np.zeros_like(h)          # h[j-shift]

    if shift >= 0:
        tp[: n - shift] = h[shift:]
        tm[shift:] = h[: n - shift]
    else:
        s = -shift
        tp[s:] = h[: n - s]
        tm[: n - s] = h[s:]

    grad = 1.0 - (tp + tm)
    return grad, shift


def _subgrad_descent(start_h, max_iters, step0, S, start_time, budget):
    """ Simple sub‑gradient descent with projection. """
    h = start_h.copy()
    best_h = h.copy()
    best_val = _evaluate(h)

    for it in range(max_iters):
        if time.time() - start_time > budget - 0.05:
            break
        grad, _ = _gradient(h)
        step = step0 / (1.0 + it * 0.01)
        y = h - step * grad
        h = _project_to_sum_box(y, S)
        val = _evaluate(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val
