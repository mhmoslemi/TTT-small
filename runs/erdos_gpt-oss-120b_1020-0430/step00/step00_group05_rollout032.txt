def optimize(n, time_limit):
    # start with random binary half-filled
    indices = np.arange(n)
    np.random.shuffle(indices)
    ones = indices[:n//2]
    h = np.zeros(n, dtype=int)
    h[ones] = 1
    best_h = h.copy()
    best_score = evaluate(h)
    cur_h = h.copy()
    cur_score = best_score

    # maintain list of idx_ones and idx_zeros
    idx_ones = list(np.where(cur_h==1)[0])
    idx_zeros = list(np.where(cur_h==0)[0])

    start = time.time()
    iter = 0
    while time.time() - start < time_limit:
        # sample swap
        i = random.choice(idx_ones)
        j = random.choice(idx_zeros)
        # do swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # update idx lists participants
        idx_ones.remove(i)
        idx_ones.append(j)
        idx_zeros.remove(j)
        idx_zeros.append(i)

        new_score = evaluate(cur_h)
        # acceptance criteria
        if new_score < cur_score or random.random() < np.exp((cur_score - new_score)/max(temp, 1e-12)):
            # accept
            cur_score = new_score
            # potential best update
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            # revert idx lists
            idx_ones.remove(j)
            idx_ones.append(i)
            idx_zeros.remove(i)
            idx_zeros.append(j)
        # temperature schedule
        temp = max(0.0, temp * 0.9999)
        iter+=1

    return best_h, best_score
```

def project_onto_feasible(x, S):
    # clip
    x = np.clip(x, 0.0, 1.0)
    # If sum close to S, done.
    cur = x.sum()
    if abs(cur - S) < 1e-12:
        return x
    # else do water-filling
    # We'll iterate at most a few times.
    # While diff more than tolerance:
    for _ in range(5):
        diff = S - x.sum()
        if abs(diff) < 1e-12:
            break
        free = (x > 0) & (x < 1)
        n_free = free.sum()
        if n_free == 0:
            # All at bounds, cannot fix; adjust by scaling (rare)
            # distribute diff uniformly (could violate bounds) but ignore
            break
        delta = diff / n_free
        # update free entries
        x[free] += delta
        # clip again
        x = np.clip(x, 0.0, 1.0)
    # final correction: adjust small diff by tweaking first element within bounds
    cur = x.sum()
    if abs(cur - S) > 1e-8:
        # push diff to an index not at bound
        for idx in range(len(x)):
            if x[idx] > 1e-12 and x[idx] < 1 - 1e-12:
                x[idx] += (S - cur)
                x[idx] = min(max(x[idx],0.0),1.0)
                break
    return x
```

def subgradient_optimize(h, max_iter, time_limit):
    n = len(h)
    S = n/2.0
    best_h = h.copy()
    best_score = evaluate(h)
    cur_h = h.copy()
    cur_score = best_score
    start = time.time()
    it = 0
    alpha0 = 0.5  # step size initial
    while time.time() - start < time_limit and it < max_iter:
        # compute correlation vector c
        c_full = np.correlate(cur_h, 1 - cur_h, mode='full')
        # find worst lag
        idx = np.argmax(c_full)
        lag = idx - (n - 1)
        # compute gradient for worst lag
        grad = gradient_for_lag(cur_h, lag)
        # step size
        step = alpha0 / math.sqrt(it + 1)
        # update
        new_h = cur_h - step * grad
        # project
        new_h = project_onto_feasible(new_h, S)
        new_score = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * (2.0/n)
        # accept if better else keep current (or also accept with small probability)

import time, math, random
import numpy as np

def evaluate_c5(vec):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on a uniform grid."""
    n = vec.shape[0]
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return np.max(corr) * (2.0 / n)


def project_feasible(x, target):
    """Project x onto the set {0≤x_i≤1, sum(x)=target} (Euclidean projection)."""
    x = np.clip(x, 0.0, 1.0)
    # water‑filling loop – convergence in a few passes for modest n
    for _ in range(10):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = int(free.sum())
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # tiny correction in case of rounding errors
    cur = x.sum()
    if abs(cur - target) > 1e-9:
        for i in range(len(x)):
            if 0.0 < x[i] < 1.0:
                x[i] += target - cur
                x[i] = min(max(x[i], 0.0), 1.0)
                break
    return x


def gradient_for_lag(h, lag):
    """Sub‑gradient of the overlap for a specific lag."""
    n = h.shape[0]
    g = np.zeros(n)
    if lag >= 0:
        # left part (i = 0 .. n-1-lag)
        i = np.arange(0, n - lag)
        g[i] += 1.0 - h[i + lag]
        # right part (j = lag .. n-1)
        j = np.arange(lag, n)
        g[j] -= h[j - lag]
    else:
        kk = -lag
        # left part (i = kk .. n-1)
        i = np.arange(kk, n)
        g[i] += 1.0 - h[i - kk]
        # right part (j = 0 .. n-kk-1)
        j = np.arange(0, n - kk)
        g[j] -= h[j + kk]
    return g


def worst_lag_and_value(h):
    """Return (lag, max_value) where the overlap attains its maximum."""
    n = h.shape[0]
    full = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(full))
    lag = idx - (n - 1)           # lag in [-(n-1), n-1]
    return lag, full[idx]


def subgradient_optimize(start_h, max_time):
    """Run a plain sub‑gradient descent with occasional random perturbations."""
    n = start_h.shape[0]
    target = n / 2.0
    h = start_h.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    cur_val = best_val
    start = time.time()
    it = 0
    alpha0 = 0.6                     # initial step size
    while time.time() - start < max_time:
        lag, _ = worst_lag_and_value(h)
        g = gradient_for_lag(h, lag)
        step = alpha0 / math.sqrt(it + 1.0)
        cand = h - step * g
        cand = project_feasible(cand, target)
        cand_val = evaluate_c5(cand)

        # accept if improvement; very occasional uphill to escape
        if cand_val < cur_val or random.random() < 0.001:
            h, cur_val = cand, cand_val
            if cand_val < best_val - 1e-12:
                best_h, best_val = cand.copy(), cand_val

        # occasional (every few thousand iters) random jitter
        if it % 5000 == 0 and it > 0:
            jitter = np.random.uniform(-0.05, 0.05, size=n)
            h = project_feasible(h + jitter, target)
            cur_val = evaluate_c5(h)

        it += 1
    return best_h, best_val
