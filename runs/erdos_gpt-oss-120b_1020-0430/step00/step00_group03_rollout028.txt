            max_eps = min(1.0 - cur_h[i], cur_h[j])
            if max_eps <= 0:
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                if max_eps <= 0:
                    continue
                src, dst = i, j
            else:
                src, dst = j, i
            eps = max_eps * random.random() * 0.5
            if eps <= 0:
                continue
            cur_h[dst] += eps
            cur_h[src] -= eps
            new_val, _, _ = _full_corr(cur_h)
            if new_val <= cur_val or math.exp(-(new_val - cur_val) / temp) > random.random():
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = cur_h.copy()
            else:
                cur_h[dst] -= eps
                cur_h[src] += eps

    # ------------------------------------------------------------------ #
    # 4.  Final tiny repair (round‑off) and return
    # ------------------------------------------------------------------ #
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        # distribute the tiny remainder to a random coordinate respecting bounds
        idx = np.random.randint(N)
        best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)
        best_val, _, _ = _full_corr(best_h)

    return best_h, best_val, N
