def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    a_rev = h[::-1]
    b = 1.0 - h
    L = 2*n - 1
    # compute convolution via FFT
    fa = np.fft.rfft(a_rev, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    # result length L, includes possibly small imag errors.
    max_val = np.max(conv.real)  # conv size L
    return max_val * dx

def max_corr_shift(h):
    n = len(h)
    L = 2*n - 1
    a_rev = h[::-1]
    b = 1.0 - h
    conv = np.fft.irfft(np.fft.rfft(a_rev, n=L) * np.fft.rfft(b, n=L), n=L)
    max_val = conv.max()
    shift_idx = conv.argmax() - (n - 1)
    return max_val * (2.0 / n), shift_idx

def attempt_targeted_swap(h, shift):
    n = len(h)
    if shift >= 0:
        rng = np.arange(0, n - shift)
        i_bad = rng[(h[rng] == 1) & (h[rng + shift] == 0)]

        return None, None

    h_new = h.copy()

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    a_rev = h[::-1]
    b = 1.0 - h
    L = 2*n - 1
    # rfft
    fa = np.fft.rfft(a_rev, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)

import time
import random
import numpy as np

def _c5_and_shift(h: np.ndarray):
    """Return (C5 value, shift that attains the maximum) for binary step h."""
    n = h.shape[0]
    a_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    # linear correlation via FFT (real input -> real output)
    fa = np.fft.rfft(a_rev, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    idx = int(np.argmax(conv))
    max_val = conv[idx]
    shift = idx - (n - 1)          # shift index in range [-(n-1), n-1]
    return max_val * (2.0 / n), shift


def _targeted_swap(h: np.ndarray, shift: int):
    """Perform a swap that directly improves the worst shift, if possible.
    Returns a new array or None if no improving pair exists."""
    n = h.shape[0]
    if shift >= 0:
        rng = np.arange(0, n - shift)
        bad = rng[(h[rng] == 1) & (h[rng + shift] == 0)]
        good = rng[(h[rng] == 0) & (h[rng + shift] == 1)]
    else:
        off = -shift
        rng = np.arange(off, n)
        bad = rng[(h[rng] == 1) & (h[rng - off] == 0)]
        good = rng[(h[rng] == 0) & (h[rng - off] == 1)]
    if bad.size == 0 or good.size == 0:
        return None
    i = int(np.random.choice(bad))
    j = int(np.random.choice(good))
    if i == j:
        return None
    new_h = h.copy()
    new_h[i] = 0.0
    new_h[j] = 1.0
    return new_h


def _hill_climb(start_h: np.ndarray, deadline: float):
    """Greedy hill‑climb (targeted swaps, fallback random swaps) until time out."""
    cur_h = start_h.copy()
    cur_c5, cur_shift = _c5_and_shift(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    stagnant = 0
    while time.time() < deadline:
        # try a targeted swap based on the worst shift
        new_h = _targeted_swap(cur_h, cur_shift)
        if new_h is None:
            # fall back to a random 1↔0 swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))
            new_h = cur_h.copy()
            new_h[i] = 0.0
            new_h[j] = 1.0

        new_c5, new_shift = _c5_and_shift(new_h)

        if new_c5 < cur_c5:
            cur_h, cur_c5, cur_shift = new_h, new_c5, new_shift
            stagnant = 0
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
        else:
            stagnant += 1
            if stagnant >= 2000:          # no progress – stop this run
                break
    return best_h, best_c5

    """Search for a binary step function on [0,2] with integral 1 that
    minimizes the Erdős C₅ overlap bound."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s * 0.95          # small safety margin

        best_c5, _ = _c5_and_shift(best_h)
