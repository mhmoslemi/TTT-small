    try:

    except NameError:

        N = 101

        np.random.shuffle(arr)
    # ensure exact sum = N/2
    N = arr.size
    target = N / 2.0
    cur = arr.sum()
    if abs(cur - target) > 1e-12:
        # simple projection onto the hyperplane while respecting bounds
        diff = target - cur
        # distribute the difference uniformly over entries that are not at bounds
        free = np.where((arr > 0) & (arr < 1))[0]
        if free.size == 0:
            # fall back to scaling (should rarely happen)
            arr = np.clip(arr * (target / cur), 0, 1)
        else:
            add = diff / free.size
            arr[free] += add
            # clip again and correct any residual
            np.clip(arr, 0, 1, out=arr)
            residual = target - arr.sum()
            if abs(residual) > 1e-12:
                # push residual onto the first non‑boundary element
                for idx in range(N):
                    max_adj = 1 - arr[idx] if residual > 0 else arr[idx]
                    adj = max(min(residual, max_adj), -max_adj)
                    arr[idx] += adj
                    residual = target - arr.sum()
                    if abs(residual) < 1e-12:
                        break
    return arr

def _gradient_overlap(h, lag):
    """Gradient of the overlap sum_{i} h_i h_{i+lag} w.r.t. h for a given (absolute) lag."""
    N = h.size
    if lag == 0:
        return 2.0 * h
    g = np.zeros(N, dtype=np.float64)
    # forward contribution
    g[: N - lag] += h[lag:]
    # backward contribution
    g[lag:] += h[: N - lag]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with a small C5 value."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    h = _initial_h()
    N = h.size
    dx = 2.0 / N

    # current state
    cur_c5 = _eval(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # algorithmic parameters
    max_step = 0.20          # upper bound for delta moves
    min_step = 0.005         # lower bound as we cool
    big_move_prob = 0.07     # probability to try a large (up‑to‑1) move
    initial_T = 0.02         # starting temperature for SA
    iters_without_improve = 0
    max_no_improve = 2500   # restart after this many non‑improving steps

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.99:
            break

        # Find the shift (lag) with the *minimum* overlap – this gives the worst product.
        overlap = np.correlate(h, h, mode="full")
        idx_min = int(overlap.argmin())
        lag = idx_min - (N - 1)
        lag_abs = abs(lag)

        # Gradient for that lag (helps to raise the overlap).
        grad = _gradient_overlap(h, lag_abs)

        # Candidate indices for increase / decrease
        inc_idx = np.where(h < 1.0 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]

        if inc_idx.size == 0 or dec_idx.size == 0:
            # fallback: random indices
            i_inc = random.randrange(N)
            i_dec = random.randrange(N - 1)
            if i_dec >= i_inc:
                i_dec += 1
        else:
            # choose the most promising pair according to the gradient
            i_inc = inc_idx[int(np.argmax(grad[inc_idx]))]
            i_dec = dec_idx[int(np.argmin(grad[dec_idx]))]

        # Annealing schedule
        elapsed = now - start_time
        progress = elapsed / budget_s
        step = max_step * (1 - progress) + min_step * progress
        T = initial_T * (1 - progress)

        # Decide the magnitude of the move
        max_allowed = min(step, 1.0 - h[i_inc], h[i_dec])
        if random.random() < big_move_prob:
            max_allowed = min(1.0 - h[i_inc], h[i_dec])  # possibly a full swap
        if max_allowed <= 0:
            # very rare – pick any feasible pair
            i_inc = inc_idx[0] if inc_idx.size else i_inc
            i_dec = dec_idx[0] if dec_idx.size else i_dec
            max_allowed = min(step, 1.0 - h[i_inc], h[i_dec])
            if max_allowed <= 0:
                continue

        delta = random.uniform(0.0, max_allowed)

        # Build the neighbor
        h_new = h.copy()
        h_new[i_inc] += delta
        h_new[i_dec] -= delta

        # Evaluate neighbor
        new_c5 = _eval(h_new)

        # Acceptance rule (simulated annealing style)
        improved = new_c5 < best_c5 - 1e-12
        if improved:
