def project(v, S, eps=1e-9):
    lo = np.min(v) - 1 # lower bound for lambda, not important
    hi = np.max(v) + 1
    for _ in range(50):
        lam = (lo+hi)/2
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if abs(s - S) <= eps:
            return x
        if s > S:
            lo = lam
        else:
            hi = lam
    # final clip
    return np.clip(v - (lo+hi)/2, 0, 1)

    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    # set n_points
    n = 200
    dx = 2.0 / n
    S = n / 2  # sum required
    # Initialize h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # If length differs from n, we can interpolate onto n.
        if len(h0) != n:
            # Resample by linear interpolation to n points.
            x_old = np.linspace(0,2,len(h0),endpoint=False)
            x_new = np.linspace(0,2,n,endpoint=False)
            h = np.interp(x_new, x_old, h0)
        else:
            h = h0.copy()
        # project onto sum constraint (should already satisfy)
        h = np.clip(h,0,1)
        h = project_onto_capped_simplex(h, S)
    else:
        # random init
        h = rng.random(n)
        h = project_onto_capped_simplex(h, S)
    best_h = h.copy()
    best_c = compute_c5(h, dx)
    # optimization loops
    start = time.time()
    iter = 0
    while time.time() - start < budget_s:

        corr = np.correlate(h, h, mode='full')

def hill_climb_anneal(h, budget, start_time):
    n = len(h)
    S = n / 2
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    T0 = 0.01
    while time.time() - start_time < budget:
        # choose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # random delta
        max_delta = min(cur_h[i], 1 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # apply move
        cur_h[i] -= delta
        cur_h[j] += delta
        # compute C5
        new_c = compute_c5(cur_h)
        # acceptance
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / (T0 * max(1e-12, 1 - (time.time() - start_time) / budget))):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_c

def _project_to_capped_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x_i<=1, sum(x)=target_sum}."""
    # Clip to [0,1] first
    v = np.clip(v, 0.0, 1.0)
    # If already correct sum, return
    cur = v.sum()
    if abs(cur - target_sum) <= 1e-12:
        return v
    # Use bisection on lambda in x_i = clip(v_i - lam, 0, 1)
    lo, hi = -1.0, 2.0  # bounds guarantee sum(lo) > target, sum(hi) < target
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)

        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)
