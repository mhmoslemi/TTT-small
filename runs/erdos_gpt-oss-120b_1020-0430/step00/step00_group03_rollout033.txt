def generate_piecewise_constant(n, n_edge):
    # n even (but not required)
    # compute p
    total_ones = n / 2.0 # sum requirement (float)
    # n_edge may be integer (points with value 1). p solves:
    # n_edge*1 + (n - n_edge)*p = total_ones
    p = (total_ones - n_edge) / (n - n_edge) if n_edge != n else 0.0 # if n_edge = n, p irrelevant but shouldn't happen because n_edge <= n/2
    # left and right edges
    left_len = n_edge // 2
    right_len = n_edge - left_len
    middle_len = n - n_edge
    # Build array
    h = np.empty(n, dtype=float)
    # fill left ones
    start = 0
    if left_len > 0:
        h[start:start+left_len] = 1.0
    start += left_len
    # fill middle with p
    if middle_len > 0:
        h[start:start+middle_len] = p
    start += middle_len
    # fill right ones
    if right_len > 0:
        h[start:start+right_len] = 1.0
    return h

def piecewise_constant_h(n, n_edge):

    - n_edge points with value 1 (split as left/right blocks)
    - remaining points with a constant value p such that the total integral is 1.
