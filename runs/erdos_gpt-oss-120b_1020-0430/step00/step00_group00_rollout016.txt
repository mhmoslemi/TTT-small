def _compute_c5(arr: np.ndarray) -> float:
    """Return the overlap integral C5 for the given step function."""
    n = arr.shape[0]
    dx = 2.0 / n
    # correlation h * (1-h) over the full range (zero padded outside)
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(corr.max() * dx)


def _candidate_move(h: np.ndarray, max_move: float):

    h_new[j] -= delta
