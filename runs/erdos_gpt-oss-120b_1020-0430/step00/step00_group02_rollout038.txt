import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Maximum truncated overlap for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear (non‑circular) correlation, exactly as evaluation uses
    return np.correlate(h, 1.0 - h, mode="full").max() * dx


def project_box_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto {x : sum(x)=target_sum, 0≤x_i≤1}."""
    x = np.clip(v, 0.0, 1.0)
    s = x.sum()
    if abs(s - target_sum) < 1e-12:
        return x
    # water‑filling style iteration – converges in few steps because bounds are [0,1]
    for _ in range(15):
        mask = (x > 0.0) & (x < 1.0)
        free = mask.sum()
        if free == 0:
            break
        delta = (target_sum - s) / free
        x[mask] += delta
        np.clip(x, 0.0, 1.0, out=x)
        s = x.sum()
        if abs(s - target_sum) < 1e-12:
            break
    return x


def gradient_optimize(h: np.ndarray, budget: float, rng: np.random.Generator):
    """Projected sub‑gradient descent on the active shift."""
    n = h.shape[0]
    target_sum = n / 2.0
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    alpha0 = 0.5  # initial step magnitude
    while time.time() - start < budget:
        # find the shift attaining the current max (unscaled correlation)
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)          # signed shift
        # gradient of the selected term
        g = np.zeros_like(cur_h)
        if k >= 0:
            lim = n - k
            if lim > 0:
                g[:lim] += (1.0 - cur_h[k:k + lim])
                g[k:k + lim] -= cur_h[:lim]
        else:
            m = -k
            lim = n - m
            if lim > 0:
                g[m:] += (1.0 - cur_h[:lim])
                g[:lim] -= cur_h[m:]

        frac = elapsed / budget

def hill_climb_optimize(h: np.ndarray, budget: float, rng: np.random.Generator):
    """Simple pairwise random moves with occasional acceptance."""
    n = h.shape[0]
    target_sum = n / 2.0
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    # a tiny temperature for occasional uphill moves
    T0 = 0.001
    while time.time() - start < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # feasible delta range for transferring mass i←j
        lower = max(-cur_h[i], -(1.0 - cur_h[j]))
        upper = min(1.0 - cur_h[i], cur_h[j])
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper)
        cur_h[i] += delta
        cur_h[j] -= delta
        new_score = compute_c5(cur_h)
        # always accept improvement
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_h = cur_h.copy()
                best_score = new_score
                if best_score <= 0.3808:
                    break
        else:
            # occasional uphill acceptance governed by a tiny temperature schedule
            elapsed = time.time() - start
            frac = elapsed / budget
            T = T0 * (1.0 - frac)
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                cur_score = new_score
            else:
                # revert move
                cur_h[i] -= delta
                cur_h[j] += delta
    return best_h, best_score


def upsample_initial(orig_h: np.ndarray, target_n: int, rng: np.random.Generator):
    """Linear up‑sampling of a feasible profile, followed by a quick feasibility fix."""
    n0 = orig_h.shape[0]
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    xt = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h = np.interp(xt, x0, orig_h)
    target_sum = target_n / 2.0
    # adjust total mass while respecting bounds
    diff = target_sum - h.sum()
    while abs(diff) > 1e-12:
        idx = rng.integers(0, target_n)
        if diff > 0.0:
            if h[idx] >= 1.0:

            h[idx] += d
        else:
            if h[idx] <= 0.0:
                continue
            d = min(-diff, h[idx])
            h[idx] -= d
        diff = target_sum - h.sum()
    np.clip(h, 0.0, 1.0, out=h)

    # we try a modest up‑sampling when time permits
    total_budget = budget_s
    if budget_s > 300 and n0 < 180:
        target_n = min(180, n0 * 4)
        h = upsample_initial(init_h, target_n, rng)
    else:
        h = init_h.copy()
    n = h.shape[0]

    # ---- optimisation phases ------------------------------------------------
    # 70 % of time for gradient descent, remaining for hill climbing
    split = 0.70
    grad_budget = total_budget * split
    hc_budget = total_budget - grad_budget - 5.0   # tiny safety margin

    best_h, best_score = gradient_optimize(h, grad_budget, rng)
    # further refinement with random moves
    best_h, best_score = hill_climb_optimize(best_h, hc_budget, rng)

    # final sanity projection (should already be feasible)
    target_sum = n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = project_box_simplex(best_h, target_sum)

    final_score = compute_c5(best_h)
    return best_h.tolist(), float(final_score), n
