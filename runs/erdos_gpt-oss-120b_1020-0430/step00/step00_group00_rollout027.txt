        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        total = w.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def max_corr_shift(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    return corr[idx], shift

def gradient_max_corr(h, shift):
    n = h.size
    grad = np.ones(n)
    if shift >= 0:
        s = shift
        grad[s:] -= h[:n - s]
        grad[:n - s] -= h[s:]
    else:
        s = -shift
        grad[:n - s] -= h[s:]
        grad[s:] -= h[:n - s]
    return grad

def objective(h):
    maxval, _ = max_corr_shift(h)
    dx = 2.0 / h.size
    return maxval * dx

def objective_jac(h):
    _, shift = max_corr_shift(h)
    grad = gradient_max_corr(h, shift)
    dx = 2.0 / h.size
    return grad * dx

def slsqp_optimize(x0, maxiter=150):
    n = x0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    res = minimize(objective, x0, method='SLSQP', jac=objective_jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
    if not res.success:
        return x0
    # Ensure constraints precisely
    x = np.clip(res.x, 0.0, 1.0)
    # Project to meet sum exactly (tiny numerical error)
    current_sum = x.sum()
    if abs(current_sum - target) > 1e-8:
        x = project_to_simplex_bounded(x, target)
    return x

def random_feasible(n, target):
    # generate random vector and project onto bounded simplex
    v = np.random.rand(n)
    return project_to_simplex_bounded(v, target)

def upsample_vector(v, new_n):
    old_n = v.size
    if old_n == new_n:
        return v.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, v)

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start = time.time()
    np.random.seed(seed)
    # Determine n_points
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        n0 = init.size
        # Choose larger n for more freedom
        n = max(90, n0 * 2)  # ensure at least 90 points
        if n % 2 == 1:
            n += 1
        h0 = upsample_vector(init, n)
    else:
        n = 100
        h0 = np.full(n, 0.5)
    target_sum = n / 2.0
    # Project initial to feasible region (should already satisfy, but ensure)
    h0 = project_to_simplex_bounded(h0, target_sum)
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    # First local optimization from base
    cand = slsqp_optimize(best_h)
    cand_val = compute_c5(cand)
    if cand_val < best_val:
        best_h, best_val = cand, cand_val
    # Random restarts loop
    iter_count = 0
    while time.time() - start < budget_s - 2.0:
        # Generate random start or perturb best
        if np.random.rand() < 0.2 and iter_count > 0:
            # perturb best
            noise = np.random.normal(scale=0.1, size=n)
            x0 = best_h + noise
            x0 = np.clip(x0, 0.0, 1.0)
            x0 = project_to_simplex_bounded(x0, target_sum)
        else:
            x0 = random_feasible(n, target_sum)
        # local optimization
        opt = slsqp_optimize(x0)
        val = compute_c5(opt)
        if val < best_val:
            best_h, best_val = opt, val
        iter_count += 1
    # Final projection to guarantee sum exactly
    best_h = project_to_simplex_bounded(best_h, target_sum)
    best_val = compute_c5(best_h)
    return (best_h.tolist(), float(best_val), int(n))

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def _project_to_simplex_bounded(v, s):
    """Project v onto {0<=x_i<=1, sum x_i = s} (Euclidean projection)."""
    lo = -np.max(v) - 1.0   # enough to make all entries 1 after clipping
    hi = np.max(v) + 1.0    # enough to make all entries 0 after clipping
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        total = w.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _max_corr_shift(h):
    """Return (max_corr_value, shift_index) for the full correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return corr[idx], shift


def _gradient_max_corr(h, shift):
    """Gradient of the selected correlation term."""
    n = h.size
    g = np.ones(n)
    if shift >= 0:
        s = shift
        g[s:] -= h[: n - s]
        g[: n - s] -= h[s:]
    else:
        s = -shift
        g[: n - s] -= h[s:]
        g[s:] -= h[: n - s]
    return g


def _objective(h):
    """C5 value for a given step function."""
    val, _ = _max_corr_shift(h)
    return val * (2.0 / h.size)


def _objective_jac(h):
    """Jacobian of the objective (subgradient of the max)."""
    _, shift = _max_corr_shift(h)
    grad = _gradient_max_corr(h, shift)
    return grad * (2.0 / h.size)


def _slsqp_optimize(x0, maxiter=180):
    n = x0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        _objective,
        x0,
        method="SLSQP",
        jac=_objective_jac,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": maxiter, "ftol": 1e-12, "disp": False},
    )
    if not res.success:
        x = x0
    else:
        x = np.clip(res.x, 0.0, 1.0)
    # final projection to enforce the linear equality exactly
    if abs(x.sum() - target) > 1e-8:
        x = _project_to_simplex_bounded(x, target)
    return x


def _random_feasible(n, target):
    """Generate a random vector in [0,1] with prescribed sum."""
    v = np.random.rand(n)
    return _project_to_simplex_bounded(v, target)


def _upsample(v, new_n):
    """Linear interpolation of a periodic sample onto a finer grid."""
    old_n = v.size
    if old_n == new_n:
        return v.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, v)
