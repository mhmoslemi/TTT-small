    n = len(h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    cur_h = h.copy()
    cur_c5 = best_c5
    start = time.time()
    iteration = 0

    step_size = 0.1
    while time.time() - start < budget_s * 0.95 and iteration < max_iter:

        hi = cur_h[i]
        hj = cur_h[j]
        new_hi = hi - delta
        new_hj = hj + delta
        if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:

        # new vector

import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 overlap bound for a discrete step function h."""
    n = h.shape[0]
    # correlation h with (1-h) in "full" mode gives the unscaled integral sums
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _repair_sum(h: np.ndarray) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.shape[0]
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    # simple greedy redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to increase some entry
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            inc = min(1.0 - h[i], diff)
            h[i] += inc
            diff -= inc
        else:
            # need to decrease some entry
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            dec = min(h[i], -diff)
            h[i] -= dec
            diff += dec
    return h


def _local_search(start_h: np.ndarray, budget: float) -> tuple[np.ndarray, float]:
    """Hill‑climbing / simulated‑annealing search improving C5."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    cur_h = start_h.copy()
    cur_c5 = _c5_value(cur_h)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    it = 0
    last_improve = 0
    # annealing schedule
    temp0 = 0.001
    temp_decay = 0.9995

    while time.time() - start < budget:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]

        # feasible transfer amount from i to j (preserves sum)
        max_delta = min(hi, 1.0 - hj)
        min_delta = -min(hj, 1.0 - hi)
        if max_delta <= min_delta:
            continue

        delta = np.random.rand() * (max_delta - min_delta) + min_delta

        # apply trial move
        cur_h[i] = hi - delta
        cur_h[j] = hj + delta
        new_c5 = _c5_value(cur_h)

        delta_c5 = new_c5 - cur_c5
        temp = temp0 * (temp_decay ** it)

        # restart if stuck too long
        if it - last_improve > 150000:
            # jitter the best solution a bit and continue from there
            cur_h = best_h.copy()
            # add tiny random perturbations respecting bounds and sum
            for _ in range(5):
                a, b = np.random.randint(0, n, size=2)
                if a == b:
                    continue
                eps = (np.random.rand() - 0.5) * 0.02  # up to ±0.01
                ha, hb = cur_h[a], cur_h[b]
                max_d = min(ha, 1.0 - hb)
                min_d = -min(hb, 1.0 - ha)
                d = np.clip(eps, min_d, max_d)
                cur_h[a] = ha - d
                cur_h[b] = hb + d
            cur_c5 = _c5_value(cur_h)
            last_improve = it

        it += 1

    return best_h, best_c5


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function achieving a low C5 bound."""
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # 1. Initialise the candidate vector
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # enforce exact integral constraint
        h0 = _repair_sum(h0)
    else:
        # default: 200 uniform samples (value 0.5 gives the required integral)
        n = 200
        h0 = np.full(n, 0.5, dtype=float)

    # ----------------------------------------------------------------------
    # 2. Allocate time for the optimisation phase
    # ----------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed)
    # keep a small safety margin
    search_time = remaining * 0. nine  # 90 % of the remaining budget

    # ----------------------------------------------------------------------
    # 3. Run the local search
    # ----------------------------------------------------------------------
    best_h, best_c5 = _local_search(h0, search_time)

    # ----------------------------------------------------------------------
    # 4. Final evaluation & return
    # ----------------------------------------------------------------------
    final_c5 = _c5_value(best_h)
    # Ensure the returned list obeys the required format
    h_values = best_h.tolist()
    return h_values, float(final_c5), int(n)
