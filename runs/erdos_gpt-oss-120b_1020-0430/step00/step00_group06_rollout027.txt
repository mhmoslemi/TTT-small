def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    # constants
    target = 0.3808
    n = 64
    dx = 2.0 / n
    best_h = None
    best_bound = float('inf')
    # optionally evaluate initial_h_values if present
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=np.float64)
        # compute bound
        init_bound = np.max(np.correlate(init_arr, 1 - init_arr, mode='full') * dx)
        if init_bound < best_bound:
            best_bound = init_bound
            best_h = init_arr.copy()
    # time for optimization
    time_limit = budget_s - 0.5
    # loop restarts
    while time.time() - start < time_limit:
        # random binary start
        h = np.zeros(n, dtype=np.float64)
        perm = np.random.permutation(n)
        h[perm[:n//2]] = 1.0
        # compute bound
        O = np.correlate(h, 1 - h, mode='full')
        O = O * dx
        cur_bound = np.max(O)
        no_improve = 0
        max_no_improve = 5000  # can tweak
        # inner loop
        while time.time() - start < time_limit and no_improve < max_no_improve:
            O = np.correlate(h, 1 - h, mode='full')
            O = O * dx
            max_idx = np.argmax(O)
            k = max_idx - (n - 1)
            # compute a
            a = np.zeros(n, dtype=np.float64)
            if k >= 0:
                a[:n - k] = 1.0 - h[k:]
                a[k:] -= h[:n - k]
            else:
                m = -k
                a[m:] = 1.0 - h[:n - m]
                a[:n - m] -= h[m:]
            # find inc and dec inds
            # inc: a minimal among positions where h < 1
            inc_mask = (h < 1 - 1e-12)
            if not np.any(inc_mask):
                break
            a_inc = a.copy()
            a_inc[~inc_mask] = np.inf
            i = np.argmin(a_inc)
            # dec: a maximal among positions where h > 0
            dec_mask = (h > 1e-12)
            if not np.any(dec_mask):
                break
            a_dec = a.copy()
            a_dec[~dec_mask] = -np.inf
            j = np.argmax(a_dec)
            delta = min(1.0 - h[i], h[j])
            if delta <= 1e-12:
                # no feasible move
                break
            # Apply move
            h[i] += delta
            h[j] -= delta
            # recompute bound
            new_O = np.correlate(h, 1 - h, mode='full') * dx
            new_bound = np.max(new_O)
            if new_bound + 1e-12 < cur_bound:
                cur_bound = new_bound
                no_improve = 0
                if cur_bound < best_bound - 1e-12:
                    best_bound = cur_bound
                    best_h = h.copy()
                # early exit if bound < target
                if best_bound <= target:
                    return (best_h.tolist(), float(best_bound), n)
            else:
                # revert move
                h[i] -= delta
                h[j] += delta
                no_improve += 1
        # after inner loop maybe continue to next restart
    # after time expiring
    if best_h is None:

import time
import numpy as np

def _compute_c5(h):
    """Return C5 for a step function h given as a NumPy array."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max()), corr

def _worst_shift_coeffs(h, k):
    """Gradient (increase direction) of the worst shift k."""
    n = h.shape[0]
    a = np.zeros(n, dtype=np.float64)
    if k >= 0:
        # term 1: (1 - h[i+k]) for i where i+k in range
        a[: n - k] = 1.0 - h[k:]
        # term 2: -h[i-k] for i >= k
        a[k:] -= h[: n - k]
    else:
        m = -k
        a[m:] = 1.0 - h[: n - m]
        a[: n - m] -= h[m:]
    return a

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 minimizing max overlap.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # target bound – stop early if we beat it
    TARGET = 0.3808

    # Problem size (even for binary representation)
    n = 96                      # 96 points → 48 ones, dx = 2/96
    dx = 2.0 / n

    best_h = None
    best_bound = float("inf")

    # ------------------------------------------------------------------
    # If a parent construction is provided, evaluate it as a baseline.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        bound, _ = _compute_c5(init)
        if bound < best_bound:
            best_bound = bound
            best_h = init.copy()

    # ------------------------------------------------------------------
    # Core greedy local‑search: maximize the minimum autocorrelation.
    # ------------------------------------------------------------------
    TIME_LIMIT = budget_s - 0.5   # keep a margin
    MAX_NO_IMPROVE = 5000        # restart after this many failed attempts

    while time.time() - start_time < TIME_LIMIT:
        # start from a random binary vector with exactly n/2 ones
        h = np.zeros(n, dtype=np.float64)
        perm = np.random.permutation(n)
        h[perm[: n // 2]] = 1.0

        cur_bound, _ = _compute_c5(h)
        no_improve = 0

        while time.time() - start_time < TIME_LIMIT and no_improve < MAX_NO_IMPROVE:
            # locate the shift with maximal overlap
            corr = np.correlate(h, 1.0 - h, mode="full")
            max_idx = int(np.argmax(corr))
            k = max_idx - (n - 1)                     # signed shift

            # compute coefficients a_i = dO(k)/dh_i
            a = _worst_shift_coeffs(h, k)

            # indices eligible for increase (h<1) and decrease (h>0)
            inc_mask = h < 1.0 - 1e-12
            dec_mask = h > 1e-12

            if not inc_mask.any() or not dec_mask.any():
                break

            a_inc = a.copy()
            a_inc[~inc_mask] = np.inf
            a_dec = a.copy()
            a_dec[~dec_mask] = -np.inf

            i = int(np.argmin(a_inc))                 # best place to raise
            j = int(np.argmax(a_dec))                 # best place to lower

            delta = min(1.0 - h[i], h[j])
            if delta <= 1e-12:
                break

            # propose the swap (binary → delta will be 1)
            h[i] += delta
            h[j] -= delta

            new_bound, _ = _compute_c5(h)

            if new_bound + 1e-12 < cur_bound:
                # improvement – keep it
                cur_bound = new_bound
                no_improve = 0
                if cur_bound < best_bound - 1e-12:
                    best_bound = cur_bound
                    best_h = h.copy()
                if best_bound <= TARGET:
                    # success – return immediately
                    return best_h.tolist(), best_bound, int(best_h.shape[0])
            else:
                # revert – not helpful
                h[i] -= delta
                h[j] += delta
                no_improve += 1

        # end of inner loop → restart with a fresh random start

    # ------------------------------------------------------------------
    # Fallback: if we never found a better construction, return the best we have.
    # ------------------------------------------------------------------
    if best_h is None:
        # no construction found – fall back to a simple half‑interval
        best_h = np.zeros(n, dtype=np.float64)
        best_h[: n // 2] = 1.0
        best_bound, _ = _compute_c5(best_h)

    return best_h.tolist(), float(best_bound), int(best_h.shape[0])
