    conv = np.correlate(h, 1 - h, mode='full')
    f = conv * dx
    # Weighted gradient
    # Use exponentials
    # We want w = softmax(alpha * f)
    max_af = np.max(alpha * f)
    exps = np.exp(alpha * f - max_af)
    w = exps / np.sum(exps)  # shape (2N-1)
    # Compute gradient as weighted sum of grad_f_s
    # We'll loop over shifts s ranging -(N-1) to N-1
    N = h.size
    grad = np.zeros(N)
    # Pre allocate arrays for multiplication
    # For each shift s index in array correspond to idx = s + (N-1)
    for idx, s in enumerate(range(-N+1, N)):

        # compute grad contribution for shift s = weight * grad_f_s

def compute_f_and_grad_soft(h, dx, alpha):
    import numpy as np
    N = h.shape[0]
    # Convolution
    conv = np.correlate(h, 1 - h, mode='full')
    f = conv * dx
    # Softmax weights
    af = alpha * f
    max_af = np.max(af)
    exps = np.exp(af - max_af)
    w = exps / np.sum(exps)
    # gradient
    grad = np.zeros(N)
    # loop over shifts
    # we will use enumerate over indices to avoid building range list each time
    # but use Python for loop
    for idx, s in enumerate(range(-N + 1, N)):
        weight = w[idx]
        if weight == 0:
            continue
        if s >= 0:
            L = N - s
            i = slice(0, L)
            ip = slice(s, N)
            # Using local references for speed
            grad[i] += weight * (1 - h[ip])
            grad[ip] -= weight * h[i]
        else:
            p = -s
            L = N - p
            i = slice(p, N)
            ip = slice(0, L)
            grad[i] += weight * (1 - h[ip])
            grad[ip] -= weight * h[i]
    grad *= dx
    return f, grad

import time
import numpy as np

def _project_capped_simplex(v, target, lower=0.0, upper=1.0):
    """Project v onto {x | sum(x)=target, lower<=x<=upper}."""
    v = np.asarray(v, dtype=float)
    # initial lambda bounds
    lam_low = np.min(v - upper) - 1.0
    lam_high = np.max(v - lower) + 1.0
    for _ in range(100):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
        if lam_high - lam_low < 1e-12:
            break
    x = np.clip(v - (lam_low + lam_high) * 0.5, lower, upper)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            np.clip(x, lower, upper, out=x)
    return x


def _f_and_grad(h, dx, alpha):
    """Return f (full correlation) and its smoothed gradient."""
    N = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    f = conv * dx                     # raw overlap values
    # softmax weights for a smooth max
    af = alpha * f
    m = np.max(af)
    exp = np.exp(af - m)
    w = exp / exp.sum()

    grad = np.zeros(N, dtype=float)
    # iterate over all shifts s = -(N-1)...(N-1)
    for idx, s in enumerate(range(-N + 1, N)):
        wk = w[idx]
        if wk == 0.0:
            continue
        if s >= 0:
            L = N - s
            i = slice(0, L)
            ip = slice(s, N)
            grad[i] += wk * (1.0 - h[ip])
            grad[ip] -= wk * h[i]
        else:
            p = -s
            L = N - p
            i = slice(p, N)
            ip = slice(0, L)
            grad[i] += wk * (1.0 - h[ip])
            grad[ip] -= wk * h[i]
    grad *= dx
    return f, grad


def _optimize(N, h_start, time_allow):
    """Projected gradient descent with a smooth max."""
    dx = 2.0 / N
    target = N / 2.0
    h = _project_capped_simplex(h_start, target)  # enforce constraints

    # current best
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1 - h, mode="full")) * dx

    alpha = 1.0          # start softmax temperature
    alpha_max = 200.0
    step0 = 1.0          # base step size
    start = time.time()
    it = 0
    max_iters = 3000

    while True:
        if it >= max_iters:
            break
        if time.time() - start > time_allow:
            break
        it += 1

        f, grad = _f_and_grad(h, dx, alpha)
        cur_val = np.max(f)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # decaying step size
        step = step0 / (1.0 + 0.001 * it)
        h = _project_capped_simplex(h - step * grad, target)

        # anneal softmax
        alpha = min(alpha_max, alpha * 1.02)

    return best_h, best_val
