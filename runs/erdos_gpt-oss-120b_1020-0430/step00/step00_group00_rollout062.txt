    for iter_num in range(total_iters):

        corr_vec = compute_corr_vector(h)

import time
import math
import numpy as np

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] (sampled on an equi‑spaced grid)
    that minimizes   C5 = max_k ∫ h(x)(1‑h(x+k))dx .
    The function returns the best h found, its C5 value, and the number of samples.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # configuration ---------------------------------------------------------
    # ----------------------------------------------------------------------
    n = 200                     # number of samples (even, 2 dx = 2/n)
    dx = 2.0 / n                # grid spacing, integral constraint → sum(h)=n/2
    target_sum = n / 2.0

    # start from the uniform half‑filled function (h_i = 0.5) – satisfies constraints
    h = np.full(n, 0.5, dtype=np.float64)

    # ----------------------------------------------------------------------
    # helper utilities ------------------------------------------------------
    # ----------------------------------------------------------------------
    def corr_vector(v):
        """full correlation of h with (1‑h) – NumPy uses zero padding."""
        return np.correlate(v, 1.0 - v, mode="full")

    # initial statistics
    corr = corr_vector(h)
    max_corr = float(np.max(corr))
    best_h = h.copy()
    best_corr = max_corr

    # bound we need to beat: C5 ≤ 0.3808 → max_corr ≤ C5/dx
    C5_target = 0.3808
    max_corr_target = C5_target / dx

    start_time = time.time()
    # simulated‑annealing parameters
    T_start = 5.0
    T_end = 1e-3
    max_iters = 500_000
    # counters for stagnation detection
    no_improve = 0
    max_no_improve = 25_000

    for it in range(max_iters):
        # time budget check
        if time.time() - start_time > budget_s - 0.5:
            break

        # ----- identify the most offending shift -------------------------
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)            # shift ∈ [-(n‑1), n‑1]

        # if we already satisfy the target, stop early
        if max_corr <= max_corr_target:
            best_h = h.copy()
            best_corr = max_corr
            break

        # ----- choose a position i where the contribution is maximal -----
        if shift >= 0:
            i_start, i_end = 0, n - shift
            i_slice = slice(i_start, i_end)
            j_slice = slice(shift, n)
        else:  # shift < 0
            i_start, i_end = -shift, n
            i_slice = slice(i_start, i_end)
            j_slice = slice(0, n + shift)

        hi = h[i_slice]
        hj = h[j_slice]
        contrib = hi * (1.0 - hj)            # contribution of each i to the max correlation
        local = int(np.argmax(contrib))
        i_idx = i_start + local
        j_idx = i_idx + shift                # partner index for the shift

        # ----- propose moving mass from i → j (or the opposite) ----------
        # we always move from a larger h to a smaller one to cut the product
        if h[i_idx] >= h[j_idx]:
            max_delta = min(h[i_idx], 1.0 - h[j_idx])
            if max_delta <= 0.0:
                # fallback: random pair
                i_idx, j_idx = rng.choice(n, size=2, replace=False)
                max_delta = min(h[i_idx], 1.0 - h[j_idx])
                if max_delta <= 0.0:
                    continue
            delta = max_delta * rng.random()
            if delta <= 0.0:
                continue
            h[i_idx] -= delta
            h[j_idx] += delta
        else:
            max_delta = min(h[j_idx], 1.0 - h[i_idx])
            if max_delta <= 0.0:
                i_idx, j_idx = rng.choice(n, size=2, replace=False)
                max_delta = min(h[i_idx], 1.0 - h[j_idx])
                if max_delta <= 0.0:
                    continue
            delta = max_delta * rng.random()
            if delta <= 0.0:
                continue
            h[i_idx] += delta
            h[j_idx] -= delta

        # ----- evaluate new state -----------------------------------------
        new_corr = corr_vector(h)
        new_max = float(np.max(new_corr))

        # simulated‑annealing acceptance
        progress = it / max_iters
        T = T_start * (T_end / T_start) ** progress
        accept = False
        if new_max <= max_corr:
            accept = True
        else:
            prob = math.exp(-(new_max - max_corr) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            corr = new_corr
            max_corr = new_max
            if new_max < best_corr:
                best_corr = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert the move
            if h[i_idx] < 0.0 or h[i_idx] > 1.0 or h[j_idx] < 0.0 or h[j_idx] > 1.0:
                # safety – restore previous values exactly
                h = best_h.copy()
                corr = corr_vector(h)
                max_corr = float(np.max(corr))
            else:
                # undo the delta
                if h[i_idx] < h[j_idx]:       # we added to i, subtracted from j
                    h[i_idx] -= delta
                    h[j_idx] += delta
                else:
                    h[i_idx] += delta
                    h[j_idx] -= delta
            no_improve += 1

        # ----- stagnation handling ----------------------------------------
        if no_improve > max_no_improve:
            # random perturbation to escape a flat region
            perturb = rng.normal(scale=0.02, size=n)
            h = np.clip(best_h + perturb, 0.0, 1.0)
            # re‑project onto the hyper‑plane sum = n/2
            cur_sum = h.sum()
            correction = (target_sum - cur_sum) / n
            h += correction
            h = np.clip(h, 0.0, 1.0)
            corr = corr_vector(h)
            max_corr = float(np.max(corr))
            no_improve = 0

    # ----------------------------------------------------------------------
    # final polishing (small random tweaks) --------------------------------
    # ----------------------------------------------------------------------
    end_time = time.time()
    remaining = budget_s - (end_time - start_time) - 0.5
    it = 0
    while remaining > 0 and it < 30_000:
        if time.time() - start_time > budget_s - 0.5:
            break
        # pick two indices at random and exchange a tiny amount
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            i, j = j, i
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
        delta = max_delta * rng.random() * 0.1   # very small step
        h[i] -= delta
        h[j] += delta
        new_corr = corr_vector(h)
        new_max = float(np.max(new_corr))
        if new_max <= max_corr:
            corr = new_corr
            max_corr = new_max
            if new_max < best_corr:
                best_corr = new_max
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
        it += 1

    # ----------------------------------------------------------------------
    # final integrity checks & result preparation --------------------------
    # ----------------------------------------------------------------------
    # enforce the exact integral (tiny numerical drift)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        diff = target_sum - cur_sum
        # adjust the first component that can accommodate the correction
        for idx in range(n):
            new_val = best_h[idx] + diff
            if 0.0 <= new_val <= 1.0:
                best_h[idx] = new_val
                break

    final_c5 = best_corr * dx
    return (best_h.tolist(), float(final_c5), n)
