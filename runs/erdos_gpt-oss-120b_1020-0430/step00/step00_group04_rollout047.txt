def project_to_feasible(x, target):
    # Clip to [0,1]
    y = np.clip(x, 0.0, 1.0)
    # Now adjust to meet sum target while staying within bounds.
    # Use iterative water-filling.
    max_iter = 10
    for _ in range(max_iter):
        current_sum = y.sum()
        delta = target - current_sum
        # Identify free indices where 0 < y < 1
        free_mask = (y > 0.0) & (y < 1.0)
        n_free = free_mask.sum()
        if abs(delta) < 1e-12:
            break
        if n_free == 0:
            # No free variables, cannot adjust; distribute to all with small tolerance
            # clamp to target by scaling (rare)
            y += delta / len(y)
            y = np.clip(y, 0.0, 1.0)
            continue
        adjust = delta / n_free
        y[free_mask] += adjust
        # Clip again to bounds (maybe some exceed)
        y = np.clip(y, 0.0, 1.0)
    # Final correction: in case sum slightly off due to rounding, distribute small error uniformly across all entries without violating bounds
    # We'll ensure sum through iterative small correction.
    # Or we can use more robust algorithm, but fine.
    # Finally, due to floating errors we may adjust one entry.
    final_sum = y.sum()
    delta = target - final_sum
    if abs(delta) > 1e-9:
        # adjust first entry not at bound
        for i in range(len(y)):
            if (delta > 0 and y[i] < 1) or (delta < 0 and y[i] > 0):
                y[i] = np.clip(y[i] + delta, 0.0, 1.0)
                break
    return y

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.shape[0]
    dx = 2.0 / N
    # correlation of h with (1-h); linear (zero‑padded) version matches the definition.
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project_to_feasible(x: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of x onto the box [0,1]^n intersected with the hyperplane sum=target.
    Simple iterative water‑filling; adequate for the modest dimensions used here.
    """
    y = np.clip(x, 0.0, 1.0)
    for _ in range(15):
        s = y.sum()
        delta = target - s
        if abs(delta) < 1e-12:
            break
        free = (y > 0.0) & (y < 1.0)
        n_free = np.count_nonzero(free)
        if n_free == 0:          # all variables at a bound – distribute uniformly (rare)
            y += delta / len(y)
            y = np.clip(y, 0.0, 1.0)
            continue
        adjust = delta / n_free
        y[free] += adjust
        y = np.clip(y, 0.0, 1.0)
    # final tiny correction if needed
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        for i in range(len(y)):
            if (diff > 0 and y[i] < 1.0) or (diff < 0 and y[i] > 0.0):
                y[i] = np.clip(y[i] + diff, 0.0, 1.0)
                break
    return y


def _binary_optimize(N: int, time_limit: float, init: np.ndarray | None = None):
    """
    Simulated‑annealing style optimisation on the set of binary vectors with exactly N/2 ones.
    Returns (best_vector_as_float, best_c5).
    """
    target = N // 2
    rng = np.random.default_rng()
    if init is None:
        h = np.zeros(N, dtype=int)
        h[:target] = 1
        rng.shuffle(h)
    else:
        h = np.array(init, dtype=int).copy()
        # enforce exact number of ones
        cur = int(h.sum())
        if cur != target:
            diff = target - cur
            if diff > 0:
                zeros = np.flatnonzero(h == 0)
                flip = rng.choice(zeros, diff, replace=False)
                h[flip] = 1
            else:
                ones = np.flatnonzero(h == 1)
                flip = rng.choice(ones, -diff, replace=False)
                h[flip] = 0

    best_h = h.copy()
    best_val = _compute_c5(h.astype(float))
    cur_val = best_val

    start = time.time()
    it_without_improve = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break

        # linear cooling schedule
        T = max(1e-4, 0.25 * (1.0 - elapsed / time_limit))

        # propose a swap (1 <-> 0)
        ones_idx = np.flatnonzero(h)
        zeros_idx = np.flatnonzero(1 - h)
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h[i], h[j] = 0, 1

        new_val = _compute_c5(h.astype(float))

        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / T)

        if accept:
            cur_val = new_val
            it_without_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1, 0
            it_without_improve += 1

        # occasional random restart if stuck for a while
        if it_without_improve > 5000:
            rng.shuffle(h)
            it_without_improve = 0

    return best_h.astype(float), best_val


def _continuous_optimize(x0: np.ndarray, time_limit: float):
    """
    A short SLSQP refinement allowing fractional values.
    Returns (refined_vector, refined_c5). If SciPy is unavailable, returns the input.
    """
    try:
        import scipy.optimize as opt
    except Exception:
        return x0, _compute_c5(x0)

    N = x0.shape[0]
    target = N / 2.0
    bounds = [(0.0, 1.0) for _ in range(N)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}

    def obj(v):
        return _compute_c5(v)

    # very modest budget – we respect the outer time limit
    opts = {
        'maxiter': 200,
        'ftol': 1e-9,
        'disp': False,
    }
    start = time.time()
    res = opt.minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options=opts)
    # enforce feasibility one last time
    x_opt = _project_to_feasible(res.x, target)
    return x_opt, _compute_c5(x_opt)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a triple (h_values, c5_bound, n_points) where
    h_values is a list of step heights on a uniform grid of [0,2].
    The algorithm first searches a binary configuration by simulated annealing
    and then (if time permits) performs a short continuous refinement.
    """
    np.random.seed(seed)
    start_total = time.time()
    # keep a safety margin for final sanity checks
    total_budget = budget_s * 0.97

    # Determine a working discretisation.
    # Prefer 84 points (twice the default size) – still well below 1k.
    base_n = 42
    if 'initial_h_values' in globals():
        base_n = len(initial_h_values)
    # Choose an even size >= base_n, not larger than 200.
    N = max(base_n, 84)
    if N % 2 == 1:
        N += 1
    if N > 200:
        N = 200 - (200 % 2)

    # Build an initial continuous vector.
    if 'initial_h_values' in globals() and len(initial_h_values) == N:
        init_vec = np.array(initial_h_values, dtype=float)
    elif 'initial_h_values' in globals():
        old = np.array(initial_h_values, dtype=float)
        old_n = len(old)
        old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
        new_x = np.linspace(0.0, 2.0, N, endpoint=False)
        init_vec = np.interp(new_x, old_x, old, left=0.0, right=0.0)
    else:
        init_vec = np.random.rand(N)

    # Project onto the feasible set (box + exact integral)
    init_vec = _project_to_feasible(init_vec, N / 2.0)

    # Convert to a binary start for the combinatorial phase.
    order = np.argsort(-init_vec)  # descending
    bin_start = np.zeros(N, dtype=int)
    bin_start[order[: N // 2]] = 1

    # Allocate most of the time to the binary search.
    time_for_binary = total_budget * 0.85
    best_h, best_val = _binary_optimize(N, time_for_binary, init=bin_start)

    # If there is a bit of time left, attempt a smooth refinement.
    elapsed = time.time() - start_total
    remaining = total_budget - elapsed
    if remaining > 3.0:
        best_h, best_val = _continuous_optimize(best_h, remaining)

    # Final projection to guarantee constraints (mostly a safety net).
    best_h = _project_to_feasible(best_h, N / 2.0)

    # Return as required by the harness.
    return best_h.tolist(), float(best_val), int(N)
