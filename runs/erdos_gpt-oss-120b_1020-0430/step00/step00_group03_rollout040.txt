def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode="full")
    # corr array length 2N-1 indexing from -(N-1) to (N-1). The zero shift corresponds to index N-1.

def compute_c5(h):
    n = len(h)
    L = 2*n - 1
    # find next power of 2 >= L for speed
    L_fft = 1 << (L-1).bit_length()
    H = np.fft.rfft(h, L_fft)
    H_rev = np.fft.rfft((1 - h)[::-1], L_fft)
    corr = np.fft.irfft(H * H_rev, L_fft)[:L]
    max_val = np.max(corr)
    return max_val * (2.0 / n)

def targeted_move(h, lag, rng):
    n = len(h)
    # compute overlapping range:
    if lag >= 0:
        i_range = np.arange(0, n - lag)
        j_range = i_range + lag
    else:
        lag = -lag
        i_range = np.arange(lag, n)
        j_range = i_range - lag
    # find indices where h[i] - h[j] is large positive (i.e., donor has high, j low)
    diff = h[i_range] - h[j_range]
    # choose donor where diff > 0
    donors = np.where(diff > 1e-6)[0]
    if len(donors) == 0:
        return False
    idx = rng.choice(donors)
    i = i_range[idx]
    j = j_range[idx]

import time
import math
import numpy as np
from scipy.signal import fftconvolve

def _project_to_box_and_sum(v, target_sum):
    """Project vector v onto the box [0,1] and the affine hyperplane sum=v_target."""
    # Clip first
    w = np.clip(v, 0.0, 1.0)
    # If already satisfied, return
    cur = w.sum()
    if abs(cur - target_sum) < 1e-12:
        return w
    # Find Lagrange multiplier λ such that sum(clip(v-λ,0,1)) = target_sum
    lo, hi = -2.0, 2.0  # bounds for λ; wide enough
    for _ in range(30):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return w

def _correlate(h):
    """Full cross‑correlation of h with (1‑h) (zero‑padded)."""
    return fftconvolve(h, (1.0 - h)[::-1], mode='full')

def _c5_and_lag(h):
    """Return (C5 value, lag giving the maximum) for vector h."""
    corr = _correlate(h)
    max_idx = int(np.argmax(corr))
    n = h.shape[0]
    lag = max_idx - (n - 1)          # zero shift corresponds to index n‑1
    dx = 2.0 / n
    return float(corr[max_idx] * dx), lag

def _gradient_worst_lag(h, lag):
    """Gradient of the worst‑lag correlation (c_ℓ = Σ h_i(1‑h_{i+ℓ}))."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        L = lag
        # i from 0 .. n‑L‑1 contributes (1‑h_{i+L})
        g[: n - L] += (1.0 - h[L:])
        # i from L .. n‑1 contributes –h_{i‑L}
        g[L:] -= h[: n - L]
    else:
        L = -lag
        # i from L .. n‑1 contributes (1‑h_{i‑L})
        g[L:] += (1.0 - h[: n - L])
        # i from 0 .. n‑L‑1 contributes –h_{i+L}
        g[: n - L] -= h[L:]
    return g

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # initialise h
    try:
        init = np.asarray(initial_h_values, dtype=float)  # provided by the harness
        n = init.size
        h = init.copy()
    except NameError:
        # fall back to a simple linear profile, length < 1000
        n = 200
        x = (np.arange(n) + 0.5) * (2.0 / n)       # mid‑point of each sample
        h = 1.0 - x / 2.0                           # linear from 1 to 0
    # enforce constraints exactly
    target_sum = n / 2.0
    h = _project_to_box_and_sum(h, target_sum)

    # initial evaluation
    curr_c5, curr_lag = _c5_and_lag(h)
    best_h = h.copy()
    best_c5 = curr_c5

    # algorithm parameters
    total_time = budget_s * 0.95               # keep a safety margin
    T_start = 0.08
    T_end = 1e-6
    max_iter = 10_000_000                      # safety cap
    it = 0

    while time.time() - start_time < total_time and it < max_iter:
        elapsed = time.time() - start_time
        # linear temperature decay (could be exponential as well)
        t_frac = elapsed / total_time
        T = T_start * (1.0 - t_frac) + T_end * t_frac
        # step size for mass transfers – shrinks over time
        step_scale = 0.25 * (1.0 - t_frac) + 0.005

        # decide move type
        if rng.random() < 0.6:

            # donors: entries > 1e-8, receivers: entries < 1‑1e-8

            i = rng.choice(donors)
            j = rng.choice(recvs)
            if i == j:

            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:

        else:

            # recompute worst lag periodically

            grad = _gradient_worst_lag(h, curr_lag)

    best_h = _project_to_box_and_sum(best_h, target_sum)

    final_c5, _ = _c5_and_lag(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.shape[0])
