def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # compute autocorrelation
    corr = np.correlate(h, h, mode='full')
    product = corr[len(h)-1:]  # L=0..n-1
    min_product = product.min()
    # sum_h may be slightly off due to numerical; but expecting sum= n/2.
    sum_h = h.sum()
    c5 = (sum_h - min_product) * dx
    return float(c5)

def hill_climb(h, iters, rng):
    n = len(h)
    # ensure binary: round
    binary = np.rint(h).astype(int)  # 0/1
    # but if not binary, maybe keep as is
    # We'll treat binary.

    best = binary.copy()
    best_c5 = compute_c5(best)
    # initial indices of ones and zeros
    ones = np.flatnonzero(best)
    zeros = np.flatnonzero(1 - best)
    for t in range(iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        best[i] = 0
        best[j] = 1
        # compute new c5
        c5 = compute_c5(best)
        if c5 < best_c5:
            best_c5 = c5
            # update indices sets
            ones = np.flatnonzero(best)
            zeros = np.flatnonzero(1 - best)
        else:
            # revert
            best[i] = 1
            best[j] = 0
    return best, best_c5

        # Actually contributions_i for index i is h[i+L] + h[i-L] (if within bounds). For binary, this counts how many overlapping pairs where i participates.

        contributions = np.zeros(n, dtype=int)

            # Equivalent to positive shift of magnitude L but reversed direction.
            contributions[:n-L] += h[L:].astype(int)
            contributions[L:] += h[:n-L].astype(int)
        # Now find i with h[i]==1 and minimal contributions, j with h[j]==0 and maximal contributions
        ones_idx = np.where(h > 0.5)[0]  # binary
        zeros_idx = np.where(h <= 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i_min = ones_idx[np.argmin(contributions[ones_idx])]
        j_max = zeros_idx[np.argmax(contributions[zeros_idx])]

        # compute net gain = contributions[j_max] - contributions[i_min]; if positive, swap
        if contributions[j_max] > contributions[i_min]:

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')

def solve_with_solver(n, solver_name):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n/2]
    for L in range(1, n):
        w = cp.Variable(n - L, boolean=True)
        for i in range(n - L):
            constraints.append(w[i] <= h[i])
            constraints.append(w[i] <= h[i+L])
            constraints.append(w[i] >= h[i] + h[i+L] - 1)
        product_sum = cp.sum(w)

    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=solver_name, verbose=False, max_iters=100000)
    return prob, h

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def softmax_c5(h: np.ndarray, alpha: float = 30.0) -> float:
    """
    Smooth approximation of max overlap.
    Uses log-sum-exp on the full correlation vector.
    """
    conv = np.correlate(h, 1.0 - h, mode="full")
    # log-sum-exp is numerically stable for large alpha
    return float(logsumexp(alpha * conv) / alpha)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape h(x)=min(x,2-x) sampled on n points and normalized."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    tri = np.minimum(x, 2.0 - x)
    target = n / 2.0
    # enforce integral constraint exactly
    tri = tri * (target / tri.sum())
    tri = np.clip(tri, 0.0, 1.0)
    diff = target - tri.sum()
    if abs(diff) > 1e-12:
        tri += diff / n
        tri = np.clip(tri, 0.0, 1.0)
    return tri


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random feasible start (values in [0,1] and sum = n/2)."""
    a = rng.random(n)
    a = a * ((n / 2.0) / a.sum())
    a = np.clip(a, 0.0, 1.0)
    # final adjustment to meet the sum exactly
    diff = (n / 2.0) - a.sum()
    if abs(diff) > 1e-12:
        a += diff / n
        a = np.clip(a, 0.0, 1.0)
    return a


def upsample(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    up = np.interp(new_x, old_x, arr)
    # renormalise to the exact integral
    target = new_n / 2.0
    up = up * (target / up.sum())
    up = np.clip(up, 0.0, 1.0)
    diff = target - up.sum()
    if abs(diff) > 1e-12:
        up += diff / new_n
        up = np.clip(up, 0.0, 1.0)
    return up


def solve_one_start(start: np.ndarray, n: int, rng: np.random.Generator,
                    time_deadline: float) -> tuple:
    """
    Run a two‑stage SLSQP optimisation starting from ``start``.
    Returns (h_opt, c5_opt).  If the deadline is exceeded, returns (None, None).
    """
    target_sum = n / 2.0

    def constraint_eq(h):
        return np.sum(h) - target_sum

    cons = {"type": "eq", "fun": constraint_eq}
    bounds = [(0.0, 1.0)] * n

    # first stage – soft objective with small alpha (easier landscape)
    res1 = minimize(
        lambda h: softmax_c5(h, alpha=5.0),
        start,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 250, "ftol": 1e-9, "disp": False},
    )
    if not res1.success or time.time() > time_deadline:
        return None, None

    # second stage – sharper approximation
    res2 = minimize(
        lambda h: softmax_c5(h, alpha=30.0),
        res1.x,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 250, "ftol": 1e-9, "disp": False},
    )
    if not res2.success or time.time() > time_deadline:
        return None, None

    h_opt = np.clip(res2.x, 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises the step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # keep a safety margin

    best_h = None
    best_c5 = np.inf
    best_n = None

    # Candidate grid sizes (even numbers give a perfectly binary feasible point)
    candidate_ns = [45, 60, 80, 100, 120, 150, 180, 200]

    # If the parent provided an initial construction, reuse it (and its size)
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = parent_h.shape[0]
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)
    except NameError:
        parent_h = None
        parent_n = None

    for n in candidate_ns:
        if time.time() > time_limit:
            break

        target_sum = n / 2.0
        # -----------------------------------------------------------------
        # Build a collection of diverse starting points for this resolution
        # -----------------------------------------------------------------
        starts = []

        # 1. Upsampled parent if available
        if parent_h is not None:
            if parent_n == n:
                starts.append(parent_h.copy())
            else:
                starts.append(upsample(parent_h, n))

        # 2. Triangular shape – a classic baseline
        starts.append(generate_triangular(n))

        # 3. Several random feasible vectors
        for _ in range(5):
            starts.append(random_start(n, rng))

        # 4. Small random perturbations of the triangular shape
        for _ in range(3):
            pert = generate_triangular(n) + rng.uniform(-0.2, 0.2, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            diff = target_sum - pert.sum()
            if abs(diff) > 1e-12:
                pert += diff / n
                pert = np.clip(pert, 0.0, 1.0)
            starts.append(pert)

        # -----------------------------------------------------------------
        # Optimise each start; keep the best solution found for this n
        # -----------------------------------------------------------------
        for s in starts:
            if time.time() > time_limit:
                break
            h_opt, c5_opt = solve_one_start(s, n, rng, time_limit)
            if h_opt is not None and c5_opt < best_c5 - 1e-9:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = n

    # -----------------------------------------------------------------
    # Fallback – simple binary block (guaranteed feasible) if nothing else succeeded
    # -----------------------------------------------------------------
    if best_h is None:
        n = 80
        block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        best_h = block.astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Ensure final vector satisfies the integral constraint up to numeric tolerance
    target = best_n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-9:
        best_h = best_h + (target - cur_sum) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    # Final C5 recomputed (just to be safe)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
