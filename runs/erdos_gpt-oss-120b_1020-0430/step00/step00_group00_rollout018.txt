def neighbor_move(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    s = h[i] + h[j]
    low = max(0.0, s - 1.0)
    high = min(1.0, s)
    # Choose new value for i within [low, high]
    new_i = rng.uniform(low, high)
    new_j = s - new_i
    h_new = h.copy()
    h_new[i] = new_i
    h_new[j] = new_j
    return h_new

def sample_feasible(n, target, rng):
    # initial from exponential distribution
    x = rng.exponential(scale=1.0, size=n)
    # Scale to sum target (ignore upper bound)
    x *= target / x.sum()
    # Now ensure <=1
    while True:
        over = x > 1.0
        if not np.any(over):
            break
        excess = np.sum(x[over] - 1.0)
        x[over] = 1.0
        # redistribute excess to underfilled elements proportionally
        under = ~over
        if not np.any(under):
            # all filled to 1, cannot redistribute; break
            break
        # distribute excess equally among under elements (or proportionally)
        # We'll add excess * (x[under] / sum_under) ; but if sum_under is zero may cause issue
        sum_under = np.sum(x[under])
        if sum_under == 0:
            # equally distribute
            x[under] += excess / np.sum(under)
        else:
            # proportionally
            x[under] += excess * (x[under] / sum_under)
        # after adjusting, some may exceed 1 again, loop.
    # Might still have sum slightly off due to rounding; adjust last element
    diff = target - x.sum()
    # adjust a random non-boundary element if possible
    # find indices not at bound 0 or 1
    free = np.where((x > 0) & (x < 1))[0]
    if free.size > 0:
        i = rng.choice(free)
        x[i] += diff
        # clip
        x[i] = np.clip(x[i], 0, 1)
    else:
        # adjust any element with slack
        # try just clip overall
        x = np.clip(x, 0, 1)
        # ensure sum (maybe off) by scaling small
        x *= target / x.sum()
    # final clip
    x = np.clip(x, 0, 1)
    return x

    # If diff > 0, increase values under 1
    # If diff < 0, decrease values over 0

    if abs(diff) < 1e-12:
        return x
    if diff > 0:
        # Need to increase some entries.
        slack = 1 - x
        total_slack = slack.sum()
        if total_slack == 0:
            # cannot increase, just return original
            return x
        increment = diff * (slack / total_slack)
        x += increment
    else:
        # Need to decrease
        slack = x
        total_slack = slack.sum()
        if total_slack == 0:
            return x
        decrement = (-diff) * (slack / total_slack)
        x -= decrement
    # Clip again
    x = np.clip(x, 0, 1)
    return x

import time, math
import numpy as np
from scipy.optimize import minimize

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Initial feasible vector
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # default: binary vector with floor(target) ones and one fractional entry
        n = 45
        target_sum = n / 2.0
        h0 = np.zeros(n, dtype=float)
        order = rng.permutation(n)
        k = int(math.floor(target_sum))
        h0[order[:k]] = 1.0
        rem = target_sum - k
        if rem > 1e-12:
            h0[order[k]] = rem
    n = h0.size
    dx = 2.0 / n
    target_sum = n / 2.0

    # make sure the start respects bounds and the sum constraint (tiny adjustments)
    cur = np.clip(h0, 0.0, 1.0)
    diff = target_sum - cur.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny discrepancy to a random free coordinate
        free = np.where((cur > 0) & (cur < 1))[0]
        if free.size:
            cur[free[0]] += diff
            cur[free[0]] = np.clip(cur[free[0]], 0.0, 1.0)
        else:
            cur += diff / n
            cur = np.clip(cur, 0.0, 1.0)
        # final tilt if still off
        cur += (target_sum - cur.sum()) / n
        cur = np.clip(cur, 0.0, 1.0)

    # ----------------------------------------------------------------------
    # Helpers
    # ----------------------------------------------------------------------
    def compute_score(h):
        """C5 value for a given h (lower is better)."""
        # min autocorrelation corresponds to the worst shift
        min_overlap = np.min(np.correlate(h, h, mode="full"))
        return dx * (target_sum - min_overlap)

    def neighbour_move(h):
        """Random two‑index redistribution that preserves sum and bounds."""
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        s = h[i] + h[j]
        low = max(0.0, s - 1.0)
        high = min(1.0, s)
        if high - low < 1e-14:
            return h.copy()
        new_i = rng.random() * (high - low) + low
        new_j = s - new_i
        nh = h.copy()
        nh[i] = new_i
        nh[j] = new_j
        return nh

    def random_binary_feasible():
        """Binary‑like feasible vector (floor(target) ones and one fractional entry)."""
        x = np.zeros(n, dtype=float)
        perm = rng.permutation(n)
        k = int(math.floor(target_sum))
        x[perm[:k]] = 1.0
        rem = target_sum - k
        if rem > 1e-12:
            x[perm[k]] = rem
        return x

    # ----------------------------------------------------------------------
    # Simulated annealing
    # ----------------------------------------------------------------------
    start_time = time.time()

    anneal_budget = budget_s * 0.85

    Tmin = 1e-7                   # final temperature
    max_iter = 10_000_000        # effectively unlimited, stopped by time
    cur_h = cur.copy()
    cur_score = compute_score(cur_h)
    best_h = cur_h.copy()

    last_improve = 0

    while True:

            break
        # linear cooling schedule
        T = T0 * max(Tmin / T0, 1.0 - (last_improve + 1) / max_iter)

        new_score = compute_score(new_h)
