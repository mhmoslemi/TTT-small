def compute_c5_bound(h):
    """Exact C5 bound for a discrete step function h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * (2.0 / n)


def smooth_max_and_grad(h, gamma, lam, target_sum):
    """
    Smooth approximation of max_k sum_i h_i(1-h_{i+k}) together with its gradient.
    The smooth max is (1/gamma) log sum exp(gamma * g_k) where
        g_k = sum_i h_i - sum_i h_i h_{i+k}.
    A quadratic penalty lam*(sum(h)-target)^2 is added.
    """
    n = h.shape[0]
    S = np.sum(h)                     # Σ_i h_i
    prod = np.correlate(h, h, mode="full")   # Σ_i h_i h_{i+k} for each lag
    g = S - prod                       # vector of length 2n-1

    # ----- smooth max (log‑sum‑exp) -----
    max_g = np.max(g)
    exp_term = np.exp(gamma * (g - max_g))   # shifted for stability
    weight_sum = np.sum(exp_term)
    weights = exp_term / weight_sum           # softmax weights, sum = 1

    log_sum_exp = max_g + np.log(weight_sum)
    f_smooth = log_sum_exp / gamma

    # ----- gradient of the smooth max -----
    # convolution with the weight vector gives Σ_k w_k h_{i±k}
    t0 = n - 1
    conv1 = np.convolve(weights, h, mode="full")
    conv2 = np.convolve(weights[::-1], h, mode="full")
    sum1 = conv1[t0:t0 + n]
    sum2 = conv2[t0:t0 + n]

    grad = np.ones(n) - (sum1 + sum2)       # derivative of g_k part

    # ----- penalty for the sum constraint -----
    diff = S - target_sum
    f = f_smooth + lam * diff * diff
    grad += 2.0 * lam * diff * np.ones(n)
    return f, grad


def optimize_h(start_h, target_sum, time_deadline):

    Returns (h_opt, c5_value) or None on failure.

    lam = 10.0                   # penalty weight for the sum constraint

def upscale_h(old_h, new_n):

        # ------------------------------------------------------------------
        for s in starts:
            if time.time() - start_time > budget_s * 0.95:
                break

            result = optimize_h(s, target, budget_s - (time.time() - start_time))
            if result is None:
                continue
            h_opt, c5_val = result

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n
                best_target = target

        # early exit if we already beat the target 0.3808
        if best_c5 * (2.0 / best_n) <= 0.3808:   # actually best_c5 already scaled, double‑check
            # best_c5 is already the scaled bound
            break

    # final sanity‑check of the sum constraint
    diff = best_target - np.sum(best_h)
    if abs(diff) > 1e-8:
        best_h += diff / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5_bound(best_h)

    return best_h, final_c5, best_n
