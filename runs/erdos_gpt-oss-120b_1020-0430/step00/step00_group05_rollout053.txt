def solve_milp(n, time_limit=None):
    import cvxpy as cp
    import numpy as np

    h = cp.Variable(n, boolean=True)
    Y = cp.Variable((n, n), boolean=True)
    Mmax = cp.Variable(integer=True)

    constraints = []

    # sum constraint
    constraints.append(cp.sum(h) == n//2)  # n even

    # Y diagonal zero (perhaps optional)
    for i in range(n):
        constraints.append(Y[i, i] == 0)

    # product constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(Y[i, j] <= h[i])
            constraints.append(Y[i, j] <= h[j])
            constraints.append(Y[i, j] >= h[i] + h[j] - 1)

    # mismatches constraints
    for d in range(1, n):
        Mpos = cp.sum([h[i] - Y[i, i + d] for i in range(n - d)])
        constraints.append(Mpos <= Mmax)
        Mneg = cp.sum([h[i + d] - Y[i, i + d] for i in range(n - d)])
        constraints.append(Mneg <= Mmax)

    # Mmax nonnegative
    constraints.append(Mmax >= 0)

    objective = cp.Minimize(Mmax)

    prob = cp.Problem(objective, constraints)

    # Solver options
    # Use CBC: cp.CBC? In cvxpy, you can supply solver='CBC' and maybe solve time limit.
    solver_opts = {}
    if time_limit is not None:
        solver_opts['max_iters'] = int(time_limit) # not sure
        # CBC might accept 'seconds' param?
        # Not guarantee. Keep simple.

    try:
        prob.solve(solver='CBC', verbose=False, **solver_opts)
    except cp.error.SolverError:
        # try GLPK_MI
        prob.solve(solver='GLPK_MI', verbose=False)

    # Check status
    if h.value is None:
        raise RuntimeError('Solver failed')
    h_opt = np.rint(np.array(h.value)).astype(int)  # binary
    Mmax_val = int(round(Mmax.value))
    return h_opt, Mmax_val

def project_onto_box_sum(v, target):
    lower = 0.0
    upper = 1.0
    # Because sum is monotonic decreasing with lambda, we binary search lambda
    # lambda low: large negative to force v - lambda large -> clipped to upper bound -> sum = n*upper = n.
    # lambda high: large positive to produce lower bound -> sum = 0.
    lo = -np.max(v) - 1.0
    hi = np.max(v) - lower + 1.0  # maybe big positive
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, lower, upper)
        s = w.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v - lo, lower, upper)
    # Enforce sum exactly by adjusting small residual if any
    residual = target - w.sum()
    # distribute residual among non-bound entries
    if abs(residual) > 1e-10:
        free = (w > lower + 1e-12) & (w < upper - 1e-12)
        if free.any():
            w[free] += residual / free.sum()
            w = np.clip(w, lower, upper)
    return w
```

def optimize_h(h0, time_allow):
    import time, numpy as np
    start = time.time()
    n = len(h0)
    target = n/2.0
    h = h0.copy()
    best_h = h.copy()
    best_bound = compute_c5(h)[0]
    iteration=0
    no_improve_counter=0
    while time.time() - start < time_allow:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        d = max_idx - (n-1)
        # compute gradient for shift d
        g = np.zeros_like(h)
        if d >= 0:
            # i:0..n-d-1, j = i+d
            i = np.arange(0, n - d)
            j = i + d
        else:
            d_abs = -d
            i = np.arange(d_abs, n)
            j = i - d_abs
        # accumulate gradient
        g[i] += 1 - h[j]
        g[j] += -h[i]
        # line search
        old_bound = max_val * (2.0 / n)  # compute bound
        step = 0.5  # initial step size

            h = h_candidate
            if cand_bound < best_bound - 1e-12:
                best_bound = cand_bound
                best_h = h.copy()
            no_improve_counter = 0

    sum_exp = expf.sum()

    # We'll compute gradient as sum over d of w[d] * g_d.

            i = np.arange(d_abs, n)
            j = i - d_abs
        grad[i] += weight * (1 - h[j])
        grad[j] += weight * (-h[i])

def project_onto_box_sum(v, target):
    # Ensure sum = target, 0<=v<=1 via water-filling.
    lower = 0.0
    upper = 1.0
    # set lambda bounds
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, lower, upper)
        s = w.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v - lo, lower, upper)
    # after binary search there may be small residual due to precision
    residual = target - w.sum()
    if abs(residual) > 1e-8:
        # Distribute residual to non-bound values
        free = (w > lower + 1e-9) & (w < upper - 1e-9)
        if free.any():
            w[free] += residual / free.sum()
            w = np.clip(w, lower, upper)
    return w

import time
import numpy as np
import random
import scipy.optimize as opt

# ---------- basic utilities ----------
def compute_c5(h):
    """Return (C5 bound, raw max correlation) for vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = float(np.max(corr))
    return max_corr * dx, max_corr


def project_onto_box_sum(v, target):
    """
    Project v onto the set {x : 0 <= x_i <= 1, sum x_i = target}.
    Uses a simple binary-search on the Lagrange multiplier.
    """
    lower, upper = 0.0, 1.0
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, lower, upper)
        s = w.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v - lo, lower, upper)
    # tiny residual adjustment
    resid = target - w.sum()
    if abs(resid) > 1e-10:
        free = (w > lower + 1e-9) & (w < upper - 1e-9)
        if free.any():
            w[free] += resid / free.sum()
            w = np.clip(w, lower, upper)
    return w


# ---------- smooth‑max optimiser ----------
def _smooth_max_obj(x, alpha):
    """smooth max of the correlation values (no dx factor)."""
    f = np.correlate(x, 1.0 - x, mode='full')
    exp_arg = alpha * f
    exp_arg = np.clip(exp_arg, -700, 700)   # avoid overflow
    return np.log(np.exp(exp_arg).sum()) / alpha


def _smooth_max_grad(x, alpha):
    """Gradient of the smooth‑max objective."""
    n = len(x)
    f = np.correlate(x, 1.0 - x, mode='full')
    exp_arg = alpha * f
    exp_arg = np.clip(exp_arg, -700, 700)
    expf = np.exp(exp_arg)
    total = expf.sum()
    w = expf / total                     # soft‑max weights

    grad = np.zeros_like(x)
    # shifts run from -(n-1) … (n-1)
    for idx, d in enumerate(range(-(n - 1), n)):
        wd = w[idx]
        if wd == 0.0:
            continue
        if d >= 0:
            i = np.arange(0, n - d)
            j = i + d
        else:
            d_abs = -d
            i = np.arange(d_abs, n)
            j = i - d_abs
        grad[i] += wd * (1.0 - x[j])
        grad[j] += wd * (-x[i])
    return grad


def optimise_h_smooth(h0, target, time_allow):
    """
    Multi‑stage SLSQP optimisation using an increasingly sharp smooth‑max.
    Returns (best_h, best_bound).
    """
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_bound, _ = compute_c5(h)

    # schedule of alphas – smoother to sharper
    alphas = [5.0, 10.0, 20.0]

    for alpha in alphas:
        if time.time() - start > time_allow:
            break
        n = len(h)

        # objective / gradient closures
        def fun(x, a=alpha):
            return _smooth_max_obj(x, a)

        def jac(x, a=alpha):
            return _smooth_max_grad(x, a)

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
        bnds = [(0.0, 1.0)] * n

        try:
            res = opt.minimize(fun, h, method='SLSQP', jac=jac,
                               bounds=bnds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
        except Exception:
            break

        if not res.success:
            # continue anyway – we still have a feasible point
            pass

        h = np.clip(res.x, 0.0, 1.0)
        h = project_onto_box_sum(h, target)

        bound, _ = compute_c5(h)
        if bound < best_bound:
            best_bound = bound
            best_h = h.copy()
    return best_h, best_bound


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    best_h = None
    best_bound = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # Determine candidate problem sizes.
    # Prefer the size of any supplied parent construction.
    # ------------------------------------------------------------------
    candidates = []

    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n0 = len(init_arr)
        target0 = n0 / 2.0
        # Re‑scale to exact integral if necessary
        s = init_arr.sum()
        if s != 0:
            init_arr = init_arr * (target0 / s)
        init_arr = np.clip(init_arr, 0.0, 1.0)
        init_arr = project_onto_box_sum(init_arr, target0)
        candidates.append((n0, init_arr))

    # add a few larger sizes to give the optimizer more flexibility
    extra_ns = [80, 100, 120, 140, 160]
    for n in extra_ns:
        candidates.append((n, None))

    # ------------------------------------------------------------------
    # Process each candidate while time permits.
    # ------------------------------------------------------------------
    for n, hint_h in candidates:
        elapsed = time.time() - start
        remaining = budget_s - elapsed
        if remaining < 15.0:          # safety margin
            break

        target = n / 2.0

        if hint_h is None or len(hint_h) != n:
            # random feasible start
            h0 = np.random.rand(n)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = h0 / h0.sum() * target
            h0 = project_onto_box_sum(h0, target)
        else:
            h0 = hint_h.copy()

        # allocate a chunk of the remaining budget for this candidate
        time_allow = min(remaining * 0.6, 300.0)

        h_opt, bound = optimise_h_smooth(h0, target, time_allow)

        if bound < best_bound:
            best_bound = bound
            best_h = h_opt.copy()
            best_n = n

        if best_bound <= 0.3808:
            break

    # ------------------------------------------------------------------
    # Fallback – unlikely to be needed.
    # ------------------------------------------------------------------
    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_n = n
        best_bound, _ = compute_c5(best_h)

    return best_h.tolist(), float(best_bound), int(best_n)
