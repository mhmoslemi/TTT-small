import time
import numpy as np

def _score_and_corr(h, dx):
    """Return (max_overlap, full correlation array)."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max()), corr

def _project(h, target_sum):
    """Project h to the feasible set: [0,1] and given sum."""
    # Clip to bounds first
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    # Simple uniform correction; the needed adjustment is tiny in practice
    if not np.isclose(cur, target_sum, atol=1e-12):
        diff = target_sum - cur
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        # if still off, renormalise proportionally (rare)
        cur = h.sum()
        if not np.isclose(cur, target_sum, atol=1e-12):
            # scale the interior (values not at bounds)
            mask = (h > 0.0) & (h < 1.0)
            if mask.any():
                scale = (target_sum - h[~mask].sum()) / h[mask].sum()
                h[mask] *= scale
                h = np.clip(h, 0.0, 1.0)
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
