Define project_to_sum:

```
def project_to_sum(v, target):
    # Clip to [0,1] first
    v = np.clip(v, 0, 1)
    # Iterative projection onto hyperplane with bounds
    # while not close:
    for _ in range(10):
        cur_sum = v.sum()
        diff = target - cur_sum
        if abs(diff) < 1e-10:
            break
        # Determine free indices (not at bounds)
        free = np.where((v > 1e-12) & (v < 1-1e-12))[0]
        if free.size == 0:
            # adjust all uniformly (some may go out of bounds)
            v += diff / len(v)
            v = np.clip(v, 0, 1)
        else:
            delta = diff / free.size
            v[free] += delta
            v = np.clip(v, 0, 1)
    return v
```

def sa_fractional(h, iterations, rng):
    n = len(h)
    target_sum = n/2.0
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    for it in range(iterations):
        # temperature schedule may not needed.
        # pick two indices i, j such that i != j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Compute max delta possible
        max_inc = min(1 - h[i], h[j])
        max_dec = min(h[i], 1 - h[j])  # negative delta
        if max_inc + max_dec == 0:
            continue
        # choose delta uniformly in [-max_dec, max_inc]
        delta = rng.uniform(-max_dec, max_inc)
        # apply
        h[i] += delta
        h[j] -= delta
        # (sum remains same)
        # Ensure within bounds (should be)
        new_val = compute_c5(h)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (0.01 + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta
        # maybe occasional random perturbation to escape local minima
        if it % 5000 == 0:
            # random small perturbation to many indices
            pert = rng.normal(scale=0.05, size=n)
            h = np.clip(h + pert, 0, 1)
            # project sum
            cur_sum = h.sum()
            diff = target_sum - cur_sum
            # adjust uniformly across free entries
            free = np.where((h > 0) & (h < 1))[0]
            if free.size > 0:
                h[free] += diff / free.size
            h = np.clip(h, 0, 1)
            cur_val = compute_c5(h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
    return best_h, best_val

import time, math
import numpy as np

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the feasible set {0<=x<=1, sum(x)=target}."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size == 0:
            # no free entries – distribute over all non‑boundary entries
            free = np.where((x > 0.0) & (x < 1.0))[0]
            if free.size == 0:
                break
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # tiny adjustment to guarantee exact sum (might violate bounds by 1e-12 – acceptable)
    x *= target / max(x.sum(), 1e-16)
    return x

def _overlap_value(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)

def _worst_shift(h: np.ndarray) -> int:
    """Return the integer shift (relative index) attaining the maximal overlap."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(conv.argmax())
    return idx - (n - 1)

def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the overlap for a fixed shift (without the dx factor)."""
    n = h.shape[0]
