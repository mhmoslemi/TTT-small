def project_onto_box_simplex(v, lower=0.0, upper=1.0, target=None, tol=1e-9, max_iter=100):
    if target is None:
        target = v.sum()
    # shift v so lower bound 0
    v0 = v - lower
    # scale upper bound to bound = upper - lower
    U = upper - lower
    # binary search lambda for sum clip between 0 and U.
    lo = np.min(v0) - U
    hi = np.max(v0)
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v0 - lam, 0, U)
        s = np.sum(x)
        if abs(s - target) < tol:
            break
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v0 - lam, 0, U) + lower
    # due to numeric, adjust to exactly meet sum
    diff = target - np.sum(x)
    # distribute diff among interior indices (0<x<1) proportionally
    if abs(diff) > 1e-6:
        # adjust by adding diff equally to free indices
        free = (x > lower + 1e-9) & (x < upper - 1e-9)
        cnt = np.sum(free)
        if cnt > 0:
            x[free] += diff / cnt
            # clip again
            x = np.clip(x, lower, upper)
    return x
```

import time
import numpy as np

def _project_box_simplex(v, lower=0.0, upper=1.0, target=None,
                         tol=1e-9, max_iter=60):
    """Projection of v onto {x | lower <= x <= upper, sum(x)=target}."""
    if target is None:
        target = np.sum(v)
    # shift to lower = 0
    v0 = v - lower
    U = upper - lower
    lo = np.min(v0) - U      # ensures all clipped to 1
    hi = np.max(v0)          # ensures all clipped to 0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v0 - lam, 0.0, U)
        s = np.sum(x)
        if abs(s - target) <= tol:
            break
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v0 - lam, 0.0, U) + lower
    # tiny residual adjustment
    diff = target - np.sum(x)
    if abs(diff) > 1e-8:
        free = (x > lower + 1e-9) & (x < upper - 1e-9)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            x[free] += diff / cnt
            x = np.clip(x, lower, upper)
    return x


def _c5_value(h):
    """Exact C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)


def _upsample(old_h, new_n):
    """Linear up‑sampling of an existing h to new_n points and projection."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n * 0.5
    return _project_box_simplex(new_h, 0.0, 1.0, target=target)


def _gradient_step(h, lag_set, step):
    """Take a projected gradient step for the current maximal lags."""
    n = h.shape[0]
    g = np.zeros_like(h)
    for L in lag_set:
        if L == 0:
            g += 2.0 * h
        else:
            L = int(L)
            g[: n - L] += h[L:]
            g[L:] += h[: n - L]
    h_new = h + step * g
    target = n * 0.5
    return _project_box_simplex(h_new, 0.0, 1.0, target=target)


def _optimize(start_h, max_iters, step0, start_time, budget):
    """Projected sub‑gradient descent from start_h."""
    n = start_h.shape[0]
    dx = 2.0 / n
    best_h = start_h.copy()
    best_score = _c5_value(best_h)
    h = start_h.copy()
    for it in range(max_iters):
        if time.time() - start_time > budget * 0.92:
            break

        # correlation of h with itself
        corr = np.correlate(h, h, mode='full') * dx
        f = 1.0 - corr
        max_f = np.max(f)

        # lags attaining the maximum (use absolute to avoid double work)
        eps = 1e-12
        idx = np.where(np.abs(f - max_f) <= eps)[0]
        lags = idx - (n - 1)
        lag_set = set(np.abs(lags))

        step = step0 / np.sqrt(it + 1.0)
        h_candidate = _gradient_step(h, lag_set, step)
        cand_score = _c5_value(h_candidate)

        # accept if improvement; occasional random acceptance for escape
        if cand_score < max_f or np.random.rand() < 0.01:
            h = h_candidate

        if cand_score < best_score:
            best_score = cand_score
            best_h = h_candidate.copy()
    return best_h, best_score


def _random_local_search(h, best_score, start_time, budget, trials=2000):
    """Simple pairwise perturbations preserving the integral."""
    n = h.shape[0]
    target = n * 0.5
    while trials > 0 and (time.time() - start_time) < budget * 0.98:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 0.2
        hi = h[i] + delta
        hj = h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        cand = h.copy()
        cand[i] = hi
        cand[j] = hj
        # sum is preserved, bounds held – no projection needed
        cand_score = _c5_value(cand)
        if cand_score < best_score:
            best_score = cand_score
            h = cand
        trials -= 1
    return h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a low Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --- resolution schedule -------------------------------------------------
    n_low = 120          # cheap resolution for initial exploration
    n_high = 360         # refined resolution for final bound
    target_low = n_low * 0.5
    target_high = n_high * 0.5

    # ---------- initial candidates at low resolution -------------------------
    starts_low = []

    if 'initial_h_values' in globals():
        try:
            starts_low.append(_upsample(initial_h_values, n_low))
        except Exception:
            pass

    for _ in range(4):
        rnd = rng.random(n_low)
        rnd = _project_box_simplex(rnd, 0.0, 1.0, target=target_low)
        starts_low.append(rnd)

    best_h_low = None
    best_score_low = np.inf

    # ---------- low‑resolution optimisation ---------------------------------
    for h0 in starts_low:
        h_opt, score_opt = _optimize(
            h0, max_iters=2000, step0=0.25,
            start_time=start_time, budget=budget_s
        )
        if score_opt < best_score_low:
            best_score_low, best_h_low = score_opt, h_opt

        if time.time() - start_time > budget_s * 0.6:
            break

    # ---------- upgrade to high resolution ----------------------------------
    if best_h_low is None:
        best_h_low = np.full(n_low, 0.5)
        best_h_low = _project_box_simplex(best_h_low, 0.0, 1.0, target=target_low)

    h_high = _upsample(best_h_low, n_high)

    # ---------- high‑resolution optimisation --------------------------------
    # start with the up‑sampled solution and a few random seeds
    starts_high = [h_high]
    for _ in range(3):
        rnd = rng.random(n_high)
        rnd = _project_box_simplex(rnd, 0.0, 1.0, target=target_high)
        starts_high.append(rnd)

    best_h = None
    best_score = np.inf

    for h0 in starts_high:
        h_opt, score_opt = _optimize(
            h0, max_iters=2500, step0=0.20,
            start_time=start_time, budget=budget_s
        )
        if score_opt < best_score:
            best_score, best_h = score_opt, h_opt

        # a quick random local search to polish the result
        h_opt, score_opt = _random_local_search(
            h_opt, best_score, start_time, budget_s, trials=3000
        )
        if score_opt < best_score:
            best_score, best_h = score_opt, h_opt

        if time.time() - start_time > budget_s * 0.95:
            break

    # ---------- fallback ----------------------------------------------------
    if best_h is None:
        best_h = np.full(n_high, 0.5)
        best_h = _project_box_simplex(best_h, 0.0, 1.0, target=target_high)
        best_score = _c5_value(best_h)

    # ------------------------------------------------------------------------
    return (best_h, float(best_score), n_high)
