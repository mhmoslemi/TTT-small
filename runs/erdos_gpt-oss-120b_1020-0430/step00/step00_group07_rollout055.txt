def project_onto_constraints(x, target):
    low = -5.0
    high = 5.0
    # expand bounds if needed
    for _ in range(60):
        mid = (low + high) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(x - (low+high)/2.0, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def subgradient_step(h, dx, target, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)

    g = np.ones_like(h)
    if shift >= 0:
        s = shift
        # j with j+shift < n
        if s < n:
            g[:n-s] -= h[s:]
            g[s:] -= h[:n-s]
    else:
        s = -shift
        if s < n:
            g[s:] -= h[:n-s]
            g[:n-s] -= h[s:]

    h_new = h - alpha * g
    h_new = project_onto_constraints(h_new, target)
    return h_new, idx

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    # Determine n_points
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]

        n = 80
        init = np.full(n, 0.5)  # placeholder, will be projected
    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility of initial vector
    h = init.copy()
    h = project_onto_constraints(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h, dx)

    start = time.time()
    max_time = budget_s
    iteration = 0
    alpha0 = 0.6
    # Subgradient phase: 70% of budget
    subgrad_budget = max_time * 0.7
    while time.time() - start < subgrad_budget:
        iteration += 1
        alpha = alpha0 / np.sqrt(iteration)
        h, _ = subgradient_step(h, dx, target_sum, alpha)
        val = compute_c5(h, dx)

import time
import numpy as np

def _project_onto_constraints(x: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of x onto {0<=y<=1, sum(y)=target}."""
    # Bisection on lambda for y_i = clip(x_i - lam, 0, 1)
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Return the C₅ value for a feasible h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _subgrad_step(h: np.ndarray, dx: float, target: float, alpha: float):
    """One projected sub‑gradient step for the max‑overlap objective."""
    n = h.size
    # worst shift (the index of the maximal correlation)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # signed shift k

    # gradient of the worst term: g_j = 1 - (h_{j+shift}+h_{j-shift})
    g = np.ones_like(h)
    if shift >= 0:
        s = shift
        # j+shift term
        if s < n:
            g[: n - s] -= h[s:]
            # j-shift term
            g[s:] -= h[: n - s]
    else:
        s = -shift
        if s < n:
            g[s:] -= h[: n - s]
            g[: n - s] -= h[s:]

    h_new = h - alpha * g
    h_new = _project_onto_constraints(h_new, target)
    return h_new, idx


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except NameError:
        # fallback: uniform start (will be projected)
        n = 80
        h0 = np.full(n, 0.5, dtype=float)
    else:
        n = h0.size

    dx = 2.0 / n
    target_sum = n / 2.0                       # ∫h = 1  ⇔  Σh = n/2

    h = _project_onto_constraints(h0.copy(), target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h, dx)

    start_time = time.time()
    max_time = float(budget_s)

    # ------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent (≈70 % of budget)
    # ------------------------------------------------------------
    subgrad_deadline = start_time + 0.70 * max_time
    it = 0
    alpha0 = 0.6
    while time.time() < subgrad_deadline:
        it += 1
        alpha = alpha0 / np.sqrt(it)          # diminishing step size
        h, _ = _subgrad_step(h, dx, target_sum, alpha)
        val = _c5_value(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # occasional jitter to escape flat regions
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            h = _project_onto_constraints(h + jitter, target_sum)

    # ------------------------------------------------------------
    # Phase 2 – random pairwise transfers (remaining budget)
    # ------------------------------------------------------------
    while time.time() < start_time + max_time:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small amount from i (source) to j (target)
        allowed = min(h[i], 1.0 - h[j])
        if allowed <= 0.0:
            continue
        delta = rng.random() * allowed * 0.5
        h_candidate = h.copy()
        h_candidate[i] -= delta
        h_candidate[j] += delta
        val = _c5_value(h_candidate, dx)

        if val < best_val:
            best_val = val
            best_h = h_candidate.copy()
            h = h_candidate
            continue

        # occasional uphill acceptance (tiny simulated‑annealing)
        elapsed = time.time() - start_time
        temp = max(0.001 * (1.0 - elapsed / max_time), 1e-9)
        if rng.random() < np.exp(-(val - best_val) / temp):
            h = h_candidate

    final_val = _c5_value(best_h, dx)
    return best_h, float(final_val), int(n)
