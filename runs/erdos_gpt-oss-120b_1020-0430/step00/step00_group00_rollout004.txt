def random_feasible_vector(rng, n, target):
    # produce random distribution obeying 0-1 bounds.
    # Use random Dirichlet mapping but then handle overshoot.
    # Simpler: create n random numbers uniform [0,1], then scale to target.
    # Ensure after scaling any exceed 1, adjust.
    v = rng.random(n)
    total = v.sum()
    v = v * (target / total)
    # Clip >1 . If any clipped, we need to reallocate the excess to others.
    excess = v - 1.0
    excess[excess < 0] = 0
    if excess.sum() > 0:
        # Set those entries to 1, remove from sum, and redistribute remaining onto others proportionally.
        v = np.minimum(v,1)
        remaining = target - v.sum()
        # distribute among indices where v<1
        idx = v < 1
        if idx.any():
            # generate random values for those positions
            v[idx] = rng.random(idx.sum())
            v[idx] = v[idx] * (remaining / v[idx].sum())
            # ensure no overflow again
            v = np.clip(v,0,1)
        else:
            # all at 1, not possible because target less than n.
            pass
    return v

import numpy as np, time
from scipy.optimize import minimize

def compute_c5(x, dx):
    return dx * np.max(np.correlate(x, 1-x, mode='full'))

def random_binary_feasible(rng, n, target):
    # target = n/2
    full = int(np.floor(target))
    frac = target - full
    perm = rng.permutation(n)
    arr = np.zeros(n)
    if full>0:
        arr[perm[:full]] = 1.0
    if frac>0:
        arr[perm[full]] = frac
    return arr

def slsqp_opt(x0, target, dx, maxiter=200):
    n = x0.size
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0.0,1.0)]*n
    res = minimize(lambda x: compute_c5(x,dx), x0,
                   method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter':maxiter,'ftol':1e-12,'disp':False})
    if res.success:
        x_opt = res.x
    else:
        x_opt = x0
    val = compute_c5(x_opt,dx)
    return x_opt, val

import numpy as np

from scipy.optimize import minimize

# globals used by the optimizer callbacks
_dx = None          # will be set to 2/n inside run()

    """Return C5 for a given discretised function x (global _dx)."""
    # correlation of x with (1-x); take maximum and scale by dx
    return _dx * np.max(np.correlate(x, 1.0 - x, mode='full'))

    """Callback for scipy.minimize: the quantity we want to minimise."""
    return _compute_c5(x)

    """Equality constraint: sum(x) == target_sum."""
    return np.sum(x) - _target_sum

    full = int(np.floor(target))

    """Search for a step function h on [0,2] with ∫h=1 that minimises max_k ∫h(1‑h(·+k))."""
    global _dx, _target_sum

    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # initialise
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
    else:
        init = None

    if init is None:
        n = 45
        init = np.full(n, 0.5, dtype=float)
    else:
        n = init.size

    _dx = 2.0 / n
    _target_sum = n / 2.0

    # make sure the starter is feasible
    init = np.clip(init, 0.0, 1.0)
    s = init.sum()
    if abs(s - _target_sum) > 1e-12:
        # simple scaling, then tiny correction if needed
        init = init * (_target_sum / s)
        init = np.clip(init, 0.0, 1.0)
        diff = _target_sum - init.sum()
        if abs(diff) > 1e-12:
            idx = np.where((init > 0) & (init < 1))[0]
            if idx.size:
                init[idx[0]] += diff
            else:
                init[0] += diff
            init = np.clip(init, 0.0, 1.0)

    best = init.copy()
    best_val = _compute_c5(best)

    start_time = time.time()
    # ----------------------------------------------------------------------
    # Phase 1 – lightweight hill‑climbing on the initial guess
    # ----------------------------------------------------------------------
    phase1_limit = budget_s * 0.10               # 10 % of total time
    while time.time() - start_time < phase1_limit:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        max_transfer = min(best[i], 1.0 - best[j])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer * 0.5
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _compute_c5(cand)
        if val < best_val:
            best, best_val = cand, val

    # ----------------------------------------------------------------------
    # Phase 2 – random restarts + SLSQP
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.95:
        # random feasible start (binary with one fractional entry)
        x0 = _random_binary_feasible(rng, n, _target_sum)

        # occasional mixing with the current best to diversify
        if rng.random() < 0.2:
            mix = rng.random()
            x0 = mix * best + (1.0 - mix) * x0
            x0 = np.clip(x0, 0.0, 1.0)
            cur = x0.sum()
            if abs(cur - _target_sum) > 1e-12:
                x0 = x0 * (_target_sum / cur)
                x0 = np.clip(x0, 0.0, 1.0)

        x_opt, val = _slsqp_opt(x0, maxiter=150)
        if val < best_val:
            best, best_val = x_opt, val

            # quick local refinement after a successful improvement
            for _ in range(30):
                i, j = rng.integers(n, size=2)
                if i == j:
                    continue
                max_transfer = min(best[i], 1.0 - best[j])
                if max_transfer <= 0.0:
                    continue
                delta = rng.random() * max_transfer * 0.3
                cand = best.copy()
                cand[i] -= delta
                cand[j] += delta
                v2 = _compute_c5(cand)
                if v2 < best_val:
                    best, best_val = cand, v2

    # ----------------------------------------------------------------------
    # Final polishing round
    # ----------------------------------------------------------------------
    best, best_val = _slsqp_opt(best, maxiter=250)

    # Return the step function (numpy array), its C5 bound, and the number of points
    return best, float(best_val), n
