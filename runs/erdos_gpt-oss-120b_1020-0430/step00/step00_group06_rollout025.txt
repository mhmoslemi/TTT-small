import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------
# projection onto the capped simplex {0<=x_i<=1, sum x = total}
def _project_capped_simplex(v, total, max_iter=60):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ----------------------------------------------------------
# compute f_k = dx * sum_i h_i (1 - h_{i+k})  for all integer shifts
def _compute_f(h, dx):
    n = h.shape[0]
    total = h.sum()
    conv = np.correlate(h, h, mode='full')                # length 2n-1
    pref = np.concatenate(([0.0], np.cumsum(h)))         # length n+1
    k = np.arange(-n + 1, n)                             # shifts
    S = np.empty_like(k, dtype=np.float64)
    pos = k >= 0
    # region sum for k >= 0 : prefix[n - k]
    S[pos] = pref[n - k[pos]]
    # region sum for k < 0 : total - prefix[-k]
    S[~pos] = total - pref[-k[~pos]]
    f = dx * (S - conv)
    return f, conv, pref, total, k


# ----------------------------------------------------------
# gradient of the smooth‑max loss (soft‑max weights on f)
def _gradient(h, w, dx):
    n = h.shape[0]
    g = np.zeros_like(h)

    # loop over all shifts
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        k = idx - (n - 1)          # actual shift value
        if k >= 0:
            end = n - k
            if end > 0:
                g[:end] += weight * (1.0 - h[k:])
            if k < n:
                g[k:]   -= weight * h[:end]
        else:  # k < 0
            s = -k
            # first part: i from s..n-1  ->  (1 - h[i-s])
            g[s:] += weight * (1.0 - h[:n - s])
            # second part: i from 0..n-s-1 -> -h[i+s]
            g[:n - s] -= weight * h[s:]

    return g * dx


# ----------------------------------------------------------
# several deterministic / random initial shapes
def _initial_shape(n, kind, rng):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangular':
        h = 1.0 - np.abs(x - 0.5) * 2.0
        h = np.clip(h, 0.0, 1.0)
    elif kind == 'sine':
        h = np.sin(np.pi * x)
    elif kind == 'random_uniform':
        h = rng.random(n)
    elif kind == 'random_binary':
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    else:                          # fallback uniform 0.5
        h = np.full(n, 0.5)
    return h

    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # candidate sizes (use the provided initial vector if present)
    candidates = []
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=np.float64)

    # additional sizes (all < 1000)
    extra_ns = [80, 96, 120, 150, 180, 200]
    for n in extra_ns:
        candidates.append((n, 'random', None))

    # optimisation parameters
    max_iters = 1800
    step0 = 0.6
    decay = 0.995
    beta = 12.0                     # softness of the max (larger → sharper)

    for n, mode, seed_vec in candidates:
        # respect remaining time
        if time.time() - start_time > 0.93 * budget_s:
            break

        total = n / 2.0
        dx = 2.0 / n

        # several initialisations per size
        init_kinds = ['triangular', 'sine', 'random_uniform', 'random_binary']
        if mode == 'provided' and seed_vec is not None:
            # use the given construction directly (it already satisfies constraints)
            h0 = seed_vec.copy()
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = _project_capped_simplex(h0, total)
            start_kinds = [None]                     # single deterministic start
        else:
            start_kinds = init_kinds

        for kind in start_kinds:
            # time guard
            if time.time() - start_time > 0.93 * budget_s:
                break

            if kind is None:
                h = h0.copy()
            else:
                h = _initial_shape(n, kind, rng)
                h = _project_capped_simplex(h, total)

            # record initial C5
            f, _, _, _, _ = _compute_f(h, dx)
            cur_c5 = float(f.max())
            if cur_c5 < best_c5:
                best_c5, best_h, best_n = cur_c5, h.copy(), n

            # main optimisation loop
            for it in range(max_iters):
                # time guard inside loop
                if time.time() - start_time > 0.98 * budget_s:
                    break

                f, _, _, _, _ = _compute_f(h, dx)
                # smooth max via log‑sum‑exp
                loss = logsumexp(beta * f) / beta
                w = np.exp(beta * (f - loss))
                w /= w.sum()

                grad = _gradient(h, w, dx)
                step = step0 * (decay ** it)
                h_new = h - step * grad
                h_new = _project_capped_simplex(h_new, total)

                f_new, _, _, _, _ = _compute_f(h_new, dx)
                new_c5 = float(f_new.max())

                # accept the new point unconditionally (projection keeps feasibility)
                h = h_new
                if new_c5 < best_c5:

                # early stop if we have already beaten the target bound
                if best_c5 <= 0.3808:
                    break
            # after inner loop, check early exit condition
            if best_c5 <= 0.3808:
                break
        if best_c5 <= 0.3808:

    # safety fallback (should never happen)
    if best_h is None:

    n = best_n
    dx = 2.0 / n
    final_c5 = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx)

    return best_h, final_c5, int(best_n)
