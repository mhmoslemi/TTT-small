import time
import numpy as np
import scipy.optimize as opt

# global for constraint functions
CURRENT_N = None

def compute_c5_normalized(h: np.ndarray) -> float:
    """Return the normalized C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return float(max_corr * dx)


def objective_t(x):
    """Objective for the SLSQP solver – minimise the last variable (t)."""
    return x[-1]


def inequality_constraint(x):
    """Inequality: t - correlation(k) >= 0 for every shift."""
    h = x[:-1]
    t = x[-1]
    # correlation of h with (1-h) across all shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr  # vector, each >= 0


def equality_constraint(x):
    """Equality: sum_i h_i = n/2."""
    h = x[:-1]
    return np.sum(h) - (CURRENT_N / 2.0)


def random_binary_vector(n: int, rng: np.random.RandomState) -> np.ndarray:
    """Return a binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1.0
    return h


def hill_climb_binary(start_h: np.ndarray, rng: np.random.RandomState,
                      deadline: float) -> (np.ndarray, float):
    """
    Simple hill‑climbing on binary vectors that keep exactly half the entries 1.
    Returns the best vector found and its C5 value.
    """
    h = start_h.copy()
    best_h = h.copy()
    best_val = compute_c5_normalized(h)
    n = h.shape[0]

    while time.time() < deadline:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        val = compute_c5_normalized(h_new)
        if val < best_val - 1e-12:
            h = h_new
            best_h = h_new.copy()
            best_val = val
            if best_val <= 0.3808:
                break
    return best_h, best_val


def slsqp_optimize(start_h: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Run a single SLSQP optimisation to minimise the maximal overlap.
    Returns the (possibly fractional) improved h and its C5 value.
    """
    global CURRENT_N
    n = start_h.shape[0]
    CURRENT_N = n

    # initial bound t from the current vector
    init_corr = np.correlate(start_h, 1.0 - start_h, mode="full")
    t0 = float(np.max(init_corr))

    x0 = np.concatenate([start_h, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]  # t has only a lower bound

    cons = [
        {"type": "ineq", "fun": inequality_constraint},
        {"type": "eq", "fun": equality_constraint},
    ]

    # adapt iteration budget to remaining time
    remaining = deadline - time.time()
    maxiter = 200
    if remaining < 20:
        maxiter = int(max(30, remaining * 15))

    try:
        res = opt.minimize(
            objective_t,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return start_h, compute_c5_normalized(start_h)

    if not res.success:
        return start_h, compute_c5_normalized(start_h)

    h_opt = np.clip(res.x[:-1], 0.0, 1.0)
    return h_opt, compute_c5_normalized(h_opt)


def interpolate_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing h to a different number of points."""
    if old_h.size == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_h.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, old_h)
    h_new = np.clip(h_new, 0.0, 1.0)

    target = new_n / 2.0
    cur_sum = h_new.sum()
    if cur_sum > 0:
        h_new *= target / cur_sum
        h_new = np.clip(h_new, 0.0, 1.0)

    # tiny adjustment to hit the exact sum (simple water‑filling)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        idx = np.arange(new_n)
        rng = np.random.RandomState(0)
        rng.shuffle(idx)
        for k in idx:
            if diff == 0:
                break
            add = np.sign(diff) * min(abs(diff), 1.0 - h_new[k] if diff > 0 else h_new[k])
            h_new[k] += add
            diff -= add
    return h_new


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_val = 10.0  # large initial value

    # size of the provided construction (if any)
    n_initial = None
    if "initial_h_values" in globals():
        n_initial = len(initial_h_values)

    # candidate problem sizes (keep them modest for speed)
    candidate_ns = set()
    if n_initial:
        candidate_ns.add(n_initial)
    # add a few even sizes around the initial one
    for mult in (1, 2, 3, 4):
        if n_initial:
            candidate_ns.add(n_initial * mult)
    candidate_ns.update([60, 80, 100, 120, 150, 200])

    # keep only reasonable values (≤200 and ≥10)
    candidate_ns = sorted({n for n in candidate_ns if 10 <= n <= 200})

    # main loop over problem sizes
    for n in candidate_ns:
        if time.time() - start_time > budget_s - 5:
            break

        # build a list of starting vectors for this n
        starts = []

        # interpolate previous best if we have one
        if best_h is not None:
            starts.append(interpolate_h(best_h, n))

        # reuse the supplied construction if size matches
        if n_initial == n and "initial_h_values" in globals():
            starts.append(np.array(initial_h_values, dtype=float))

        # random feasible starts
        if n % 2 == 0:
            # binary start (exact half ones)
            starts.append(random_binary_vector(n, rng))
        else:
            # uniform 0.5 works for odd n (sum = n/2)
            starts.append(np.full(n, 0.5, dtype=float))

        # a second random binary / uniform start for diversity
        if n % 2 == 0:
            starts.append(random_binary_vector(n, rng))
        else:
            starts.append(np.full(n, 0.5, dtype=float))

        # try each starting vector
        for h0 in starts:
            if time.time() - start_time > budget_s - 5:
                break

            # --- binary hill climbing (only when feasible) ---
            if n % 2 == 0:
                # ensure we start from a proper binary vector
                if not np.all(np.isin(h0, [0.0, 1.0])):
                    idx = np.argsort(-h0)
                    bin_h = np.zeros(n, dtype=float)
                    bin_h[idx[: n // 2]] = 1.0
                else:
                    bin_h = h0.copy()
                deadline_hc = start_time + budget_s - 5
                hc_h, hc_val = hill_climb_binary(bin_h, rng, deadline_hc)
                if hc_val < best_val:
                    best_val = hc_val
                    best_h = hc_h.copy()
                    if best_val <= 0.3808:
                        break

            # --- continuous refinement via SLSQP ---
            deadline_opt = start_time + budget_s - 5
            opt_h, opt_val = slsqp_optimize(h0, deadline_opt)
            if opt_val < best_val:
                best_val = opt_val
                best_h = opt_h.copy()
                if best_val <= 0.3808:
                    break

        if best_val <= 0.3808:
            break

    # fallback – use whatever is available
    if best_h is None:
        if n_initial and "initial_h_values" in globals():
            best_h = np.array(initial_h_values, dtype=float)
        else:
            # safe default: uniform 0.5 with 50 points (exactly satisfies the integral)
            best_h = np.full(50, 0.5, dtype=float)

    n_points = best_h.shape[0]
    final_val = compute_c5_normalized(best_h)

    return (best_h.tolist(), float(final_val), int(n_points))
