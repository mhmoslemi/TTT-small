    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(max_iter):
        # choose swap
        ones = np.where(h>0.5)[0]
        zeros = np.where(h<0.5)[0]  # As binary, simple 0 or 1
        if len(ones)==0 or len(zeros)==0: break
        i = random.choice(ones)
        j = random.choice(zeros)
        h_new = h.copy()
        h_new[i]=0
        h_new[j]=1
        val = compute_c5(h_new)
        # Accept if improved or with probability exp(-(val-best_val)/temp)
        if val < best_val - 1e-12 or random.random() < np.exp(-(val - best_val) / temp):

            if val < best_val:
                best_val = val
                best_h = h_new.copy()
        # update temp schedule: temp = initial_temp * (1 - it/max_iter)
    return best_h, best_val

    # Setup
    # ----------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # Number of discretisation points (must be even for a binary construction)
    n = 200
    if n % 2:

    target_sum = n / 2.0          # because integral over [0,2] must be 1, dx = 2/n
    dx = 2.0 / n

    # ----------------------------------------------------------------------
    # Helper utilities
    # ----------------------------------------------------------------------
    def compute_c5(h: np.ndarray) -> float:
        """Maximum overlap integral for a given step function."""
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        return float(c.max())

    def project_onto_simplex(v: np.ndarray) -> np.ndarray:
        """
        Project a vector onto the intersection of the box [0,1]^n
        and the hyperplane sum(v) = target_sum.
        This is done by finding a scalar λ such that sum(clip(v+λ,0,1)) = target_sum.
        """
        lo, hi = -1.0, 1.0
        for _ in range(60):                     # bisection – enough for double precision
            mid = (lo + hi) * 0.5
            proj = np.clip(v + mid, 0.0, 1.0)
            s = proj.sum()
            if s < target_sum:
                lo = mid
            else:
                hi = mid
        proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
        # final tiny correction to enforce exact sum (numerical safety)
        diff = target_sum - proj.sum()
        if abs(diff) > 1e-12:
            # adjust a non‑saturated entry if possible
            free = np.where((proj > 0.0) & (proj < 1.0))[0]
            if free.size:
                proj[free[0]] += diff
                proj = np.clip(proj, 0.0, 1.0)
        return proj

        """
        Sub‑gradient of the correlation term C_shift(h) = Σ h[i]*(1‑h[i+shift])
        with respect to h.  Handles both positive and negative shifts.
        """

        if shift >= 0:
            i = np.arange(0, n - shift)
            j = i + shift

            i = np.arange(-shift, n)
            j = i + shift                         # j < i
        g[i] += 1.0 - h[j]                       # ∂/∂h[i] term
        g[j] -= h[i]                             # ∂/∂h[j] term (from the (1‑h) part)

            src_x = np.linspace(0.0, 2.0, m, endpoint=False) + dx_parent * 0.5
            dst_x = xs
            interp = np.interp(dst_x, src_x, parent)
            parent = np.clip(interp, 0.0, 1.0)
        parent = project_onto_simplex(parent)
        starts.append(parent)

    # 3. A handful of random binary feasible vectors
    for _ in range(8):
        h = np.zeros(n)
        idx = np.random.permutation(n)[: n // 2]
        h[idx] = 1.0
        starts.append(h)

    # ----------------------------------------------------------------------
    # Pick the best among the crude starts
    # ----------------------------------------------------------------------
    best_h = None
    best_val = math.inf
    for h0 in starts:
        val = compute_c5(h0)
        if val < best_val:
            best_val = val
            best_h = h0.copy()

    # ----------------------------------------------------------------------
    # Projected sub‑gradient (sub‑gradient descent) optimisation
    # ----------------------------------------------------------------------
    # parameters for the descent
    lr = 0.5                       # initial step size
    min_lr = 1e-6                  # stop when step size becomes tiny
    max_iter = 200_000            # generous upper bound; real stop is time‑driven

    it = 0
    while True:
        # time check
        if time.time() - start_time > budget_s - 0.5:
            break
        if lr < min_lr or it >= max_iter:
            break

        # correlation and worst shift(s)
        corr = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
        cur_max = corr.max()
        # indices where the max is attained (allow tiny tolerance)
        max_idxs = np.where(np.abs(corr - cur_max) <= 1e-12)[0]
        shifts = max_idxs - (n - 1)

        # compute average sub‑gradient over all worst shifts
        g = np.zeros_like(best_h)
        for s in shifts:
            g += corr_gradient(best_h, int(s))
        g /= float(len(shifts))

        # gradient step and projection back onto the feasible set
        h_new = best_h - lr * g
        h_new = project_onto_simplex(h_new)

        new_val = compute_c5(h_new)

        if new_val < best_val - 1e-12:
            # improvement – keep the move and slightly increase step size
            best_h, best_val = h_new, new_val
            lr = min(lr * 1.1, 2.0)
        else:
            # no progress – shrink step size
            lr *= 0.5

        it += 1

        # early exit if we have already beaten the target record
        if best_val < 0.3808:
            break

    # ----------------------------------------------------------------------
    # Final safety projection and output preparation
    # ----------------------------------------------------------------------
    best_h = project_onto_simplex(best_h)
    final_c5 = compute_c5(best_h)

    # Return as plain Python list for compatibility with the harness
    return (best_h.tolist(), float(final_c5), int(n))
