def project(v, target, tol=1e-9, max_iter=100):
    # Bisection on lambda
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - lam, 0.0, 1.0)

def slsqp_optimize(h0, target, alpha=30):
    n = len(h0)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n

                    bounds=bounds, constraints=cons, options={'maxiter': 1000, 'ftol': 1e-9})
    x = np.clip(res.x, 0.0, 1.0)
    # ensure sum exactly target via projection
    x = project(x, target)
    return x

import time
import numpy as np
from scipy.optimize import minimize

def _project_onto_capped_simplex(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project v onto {0<=x<=1, sum(x)=target} using bisection on the dual variable."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng: np.random.RandomState, n: int, target: float) -> np.ndarray:
    v = rng.rand(n)
    return _project_onto_capped_simplex(v, target)


def _compute_c5(h: np.ndarray, dx: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the correlation for a given shift (the shift that attains the max)."""
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        s = shift
        # term1 = 1 - h[i+s] for i < n-s, else 1
        g[: n - s] = 1.0 - h[s:]
        g[n - s :] = 1.0
        # term2 = -h[i-s] for i >= s
        g[s:] += -h[: n - s]
    else:
        s = -shift
        # term1 = 1 - h[i-s] for i >= s, else 1
        g[s:] = 1.0 - h[: n - s]
        g[: s] = 1.0
        # term2 = -h[i+s] for i < n-s
        g[: n - s] += -h[s:]
    return g


def _gd_optimize(h0: np.ndarray, target: float, max_iter: int = 800, step0: float = 0.5,
                 decay: float = 0.995) -> np.ndarray:
    """Projected (sub)gradient descent on the max‑correlation objective."""
    h = h0.copy()
    step = step0
    n = h.shape[0]
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _gradient_for_shift(h, shift)
        h = h - step * g
        h = _project_onto_capped_simplex(h, target)
        step *= decay
    return h


def _smooth_max(x: np.ndarray, alpha: float = 30.0) -> float:
    """Smooth approximation of max using log‑sum‑exp, shift for stability."""
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _slsqp_optimize(h0: np.ndarray, target: float, alpha: float = 30.0,
                    max_iter: int = 1000) -> np.ndarray:
    """SLSQP on a smooth max objective."""
    n = h0.shape[0]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        return _smooth_max(corr, alpha)

    cons = ({'type': 'eq', 'fun': lambda v: np.sum(v) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return _project_onto_capped_simplex(x, target)
