def compute_corr(h):

def subgradient_step(h, step):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # find argmax
    idx = corr.argmax()

    g = np.empty(n)

    h = best_h.copy()

        lo, hi = -2.0, 2.0

            s = proj.sum()
            if s < target_sum:

                hi = mid

    def max_corr(h):
        nloc = len(h)
        dx = 2.0 / nloc
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return corr.max()

    def subgrad_step(h, step):
        nloc = len(h)
        # compute correlation full, find max index
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr_full)
        lag = idx - (nloc - 1)   # integer lag
        # vectorized gradient calculation:
        # (1 - h[i+lag]) term
        term1 = np.zeros(nloc)
        term2 = np.zeros(nloc)

        grad = (term1 - term2) * (2.0 / nloc)
        v = h - step * grad
        return project_to_sum(v)

        # random uniform [0,1]

import numpy as np
import time
import math
import random

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h on [0,2] (as a list), its C5 bound and length."""
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # Helper: projection onto the box [0,1] and the equality sum(h)=n/2
    # ------------------------------------------------------------------
    def _project_to_sum(v, target):
        # Find λ such that sum(clip(v+λ,0,1)) == target.
        lo, hi = -2.0, 2.0
        for _ in range(60):
            lam = (lo + hi) / 2.0
            proj = np.clip(v + lam, 0.0, 1.0)
            s = proj.sum()
            if s < target:
                lo = lam
            else:
                hi = lam
        return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Helper: maximum overlap value for a given h
    # ------------------------------------------------------------------
    def _max_corr(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return corr.max()

    # ------------------------------------------------------------------
    # Subgradient step for the max‑over‑shift objective
    # ------------------------------------------------------------------
    def _subgrad_step(h, step):
        n = len(h)
        # raw (unscaled) correlation to locate the worst shift
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr_full))
        lag = idx - (n - 1)          # lag ∈ [-(n-1), n-1]

        # build gradient = (1 - h[i+lag]) - h[i-lag]  (scaled by dx)
        if lag >= 0:
            # (1 - h[i+lag]) term
            term1 = np.concatenate([1.0 - h[lag:], np.zeros(lag)])
            # h[i-lag] term
            term2 = np.concatenate([np.zeros(lag), h[:n - lag]])
        else:
            d = -lag
            term1 = np.concatenate([np.ones(d), 1.0 - h[:n - d]])
            term2 = np.concatenate([h[d:], np.zeros(d)])

        grad = (term1 - term2) * (2.0 / n)      # multiply by dx
        v = h - step * grad
        target = n / 2.0
        return _project_to_sum(v, target)

    # ------------------------------------------------------------------
    # Generate a random feasible h (box + sum constraint)
    # ------------------------------------------------------------------
    def _random_feasible(n):
        v = np.random.rand(n)
        target = n / 2.0
        return _project_to_sum(v, target)

        n_points = 60
        base_h = None

    target_sum = n_points / 2.0

    # Prepare a list of starting vectors (base + a few random ones)
    starts = []
    if base_h is not None:
        starts.append(base_h.copy())
    for _ in range(8):
        starts.append(_random_feasible(n_points))

    best_h = None
    best_val = float('inf')
    time_limit = budget_s - 0.05   # small safety margin

    # ------------------------------------------------------------------
    # Main optimization loop (multiple starts, subgradient descent)
    # ------------------------------------------------------------------
    for h_start in starts:
        if time.time() - start_time > time_limit:
            break

        h_curr = h_start.copy()
        cur_val = _max_corr(h_curr)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h_curr.copy()

        # simple subgradient descent
        max_iters = 8000
        for it in range(1, max_iters + 1):
            if time.time() - start_time > time_limit:
                break

            step = 1.0 / math.sqrt(it)   # decreasing step size
            h_curr = _subgrad_step(h_curr, step)

            cur_val = _max_corr(h_curr)
            if cur_val + 1e-12 < best_val:
                best_val = cur_val
                best_h = h_curr.copy()
                # early exit if we beat the target record

    # ------------------------------------------------------------------
    if best_h is None:
        best_h = _random_feasible(n_points)
        best_val = _max_corr(best_h)

    # Final check – enforce feasibility one last time (numerical safety)
    best_h = _project_to_sum(best_h, target_sum)

    # Return as plain Python list, the bound, and the number of points.
    return (best_h.tolist(), float(best_val), n_points)
