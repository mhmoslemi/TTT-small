def compute_c5_fft(h, dx):
    N = len(h)
    # pad to length 2N-1 for linear convolution
    M = 1 << ((2*N - 1).bit_length())  # next power of two for speed
    # but using np.fft.fft with length M, but we can also compute conv directly using np.fft.ifft.

    a = np.fft.rfft(h, n=M)

    conv = np.fft.irfft(a * b, n=M)
    # Need only indices 0..2N-2 (len=2N-1)
    conv = conv[:2*N-1]
    return np.max(conv) * dx

    centre = (N-1)/2
    # shape: linearly decreasing from centre to edges
    # max height = 1 maybe, but then total sum might exceed N/2

    total = raw.sum()

def project_to_bounds_and_sum(v, target):
    v = np.clip(v, 0, 1)
    cur = v.sum()
    if np.isclose(cur, target):
        return v
    # Determine the direction: need to add or subtract
    # Use iterative scaling: while not close, adjust.
    # For speed, we can apply simple algorithm: treat v as variables that can be increased up to 1 if need increase, else decreased down to 0.
    # We'll compute sorted indices based on capacity.
    if cur < target:
        # Need to add amount = target - cur across those with v < 1
        deficit = target - cur
        free = np.where(v < 1)[0]
        # Distribute equally
        addition = deficit / len(free)
        # If addition > (1 - v_i) for some, we need cap and iterate.
        # We'll iterate small number of times.
        for _ in range(100):
            caps = 1 - v[free]
            add = np.minimum(addition, caps)
            total_added = add.sum()
            v[free] += add
            deficit -= total_added
            # Remove saturated
            free = np.where((v < 1) & (v > 0))[0]
            if deficit <= 1e-12:
                break
            addition = deficit / len(free) if len(free) > 0 else 0
        # If still deficit, assign leftover to random free.
    else:
        # Need to remove amount
        excess = cur - target
        free = np.where(v > 0)[0]
        # Subtract equally
        sub = excess / len(free)
        for _ in range(100):
            caps = v[free]  # amount we can subtract
            sub_amount = np.minimum(sub, caps)
            total_sub = sub_amount.sum()
            v[free] -= sub_amount
            excess -= total_sub
            free = np.where(v > 0)[0]
            if excess <= 1e-12:
                break
            sub = excess / len(free) if len(free) > 0 else 0
    return v

    a = np.fft.rfft(h, n=M)
    b = np.fft.rfft(np.flip(1 - h), n=M)
    conv = np.fft.irfft(a * b, n=M)
    conv = conv[:2*N-1]
    return conv.max() * dx

import time
import math
import numpy as np

def _correlation_max(h):
    """Return the integer max of ∫ h(x)(1‑h(x+k))dx without the dx factor."""
    # linear convolution of h with reversed (1‑h)
    return np.max(np.convolve(h, np.flip(1 - h), mode='full'))

def _shifted_view(h, s):
    """Return an array `shifted` such that shifted[i] = h[i‑s] (zero outside)."""
    n = h.shape[0]
    if s >= 0:
        # prepend zeros, drop tail
        if s == 0:
            return h.copy()
        return np.concatenate((np.zeros(s, dtype=h.dtype), h[:n - s]))
    else:
        s = -s
        # append zeros, drop head
        if s == 0:
            return h.copy()
        return np.concatenate((h[s:], np.zeros(s, dtype=h.dtype)))


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h that (hopefully) improves the C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Array of values in [0,1] on a uniform grid of length n_points.
    c5_bound : float
        The achieved C5 bound (max overlap integral).
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine the grid size.
    # If the harness supplies an initial solution we reuse its size,
    # otherwise we use a modest size that still allows many iterations.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=np.float64)
        if init_arr.ndim != 1:
            # fallback to default size
            n = 200
        else:
            n = init_arr.shape[0]
    else:
        n = 200

    # Ensure n is even so that the integral constraint is integral.
    if n % 2:
        n += 1

    dx = 2.0 / n
    half_n = n // 2                                    # we need exactly this many ones

    # ------------------------------------------------------------------
    # Initialise a binary feasible vector h (0/1 values, sum = half_n).
    # Prefer the provided initial_h_values when possible.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=np.float64)
        if init_arr.ndim == 1 and init_arr.shape[0] == n:
            # Clip to [0,1] and round to nearest 0/1.
            h = (init_arr >= 0.5).astype(np.int8)
            # Repair the cardinality if necessary.
            cur = h.sum()
            if cur > half_n:
                idx = np.flatnonzero(h == 1)
                rm = rng.choice(idx, size=cur - half_n, replace=False)
                h[rm] = 0
            elif cur < half_n:
                idx = np.flatnonzero(h == 0)
                add = rng.choice(idx, size=half_n - cur, replace=False)
                h[add] = 1
        else:

            h = np.zeros(n, dtype=np.int8)
    else:
        h = np.zeros(n, dtype=np.int8)

    if h.sum() != half_n:
        # Weighted central sampling to bias ones toward the centre.
        positions = np.arange(n)
        centre = (n - 1) / 2.0
        sigma = n / 6.0
        weights = np.exp(-((positions - centre) / sigma) ** 2)
        weights /= weights.sum()
        ones = rng.choice(n, size=half_n, replace=False, p=weights)
        h.fill(0)

    # and try to swap a mismatching 1 with a beneficial 0.

        shifted = _shifted_view(h, s)

        src = np.flatnonzero((h == 1) & (shifted == 0))

    return best_h.astype(float), float(final_val), n
