def project_onto_capped_simplex(v, target):
    # Project onto [0,1]^n and sum constraint = target
    # Using bisection on lambda
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - lo, 0.0, 1.0)
    # due to numerical issues, adjust small difference
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff uniformly onto non-bound entries
        free = np.where((x > 0) & (x < 1))[0]
        if len(free) > 0:
            x[free[0]] += diff
            x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np

def _evaluate(h):
    """Return C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def _project_onto_capped_simplex(v, target):
    """
    Project vector v onto the set {x : 0 <= x_i <= 1, sum(x)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - lo, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = np.where((x > 0) & (x < 1))[0]
        if free.size:
            x[free[0]] += diff
            x = np.clip(x, 0.0, 1.0)
    return x

    target = n / 2.0

    best_h, best_val = h.copy(), cur_val

        new_val = _evaluate(h_new)
        # deterministic acceptance if improvement,
        # occasional uphill moves (simulated annealing style)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(0.001, 0.01 * (1 - (time.time() - start) / time_budget))):
            h, cur_val = h_new, new_val
            if new_val < best_val:
                best_h, best_val = h_new.copy(), new_val
    return best_h, best_val


def _hill_climb(h0, time_budget, rng):
    """Simple random pairwise mass transfer."""
    n = h0.shape[0]
    target = n / 2.0
    cur = h0.copy()
    cur_val = _evaluate(cur)
    best_h, best_val = cur.copy(), cur_val
    start = time.time()
    max_step, min_step = 0.3, 0.001
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        step = max_step * (1 - frac) + min_step * frac
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        lower = max(-step, -cur[j])      # cannot push j below 0
        upper = min(step, cur[i])        # cannot pull i below 0
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper)
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_val = _evaluate(new)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(0.001, 0.005 * (1 - frac))):
            cur, cur_val = new, new_val
            if new_val < best_val:
                best_h, best_val = new.copy(), new_val
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # starting vector (if the harness supplied one)
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)
        n0 = init.shape[0]
    except NameError:
        init = None
        n0 = 48                     # a modest even size if no parent is present

    # sizes we will try – keep them modest to stay well within the time budget
    candidates = [n0]
    while candidates[-1] * 2 <= 200:
        candidates.append(candidates[-1] * 2)

    best_h, best_val = None, np.inf
    start_total = time.time()

    for n in candidates:
        remaining = budget_s - (time.time() - start_total)
        if remaining <= 5.0:
            break

        # time split: 70 % for gradient descent, 30 % for a final local search
        if n != candidates[-1]:
            alloc = min(remaining * 0.5, 120.0)
        else:
            alloc = remaining - 1.0
        gd_time = max(0.1, alloc * 0.7)
        hc_time = max(0.0, alloc - gd_time)

        # ------------------------------------------------------------------
        # build an initial feasible vector of size n
        # ------------------------------------------------------------------
        if init is not None:
            if init.shape[0] == n:
                h0 = init.copy()
            elif init.shape[0] < n:
                # repeat entries to keep the total mass correct
                factor = n // init.shape[0]
                rem = n % init.shape[0]
                h0 = np.repeat(init, factor)
                if rem:
                    h0 = np.concatenate([h0, init[:rem]])
            else:
                # down‑sample by linear interpolation
                x_old = np.linspace(0.0, 2.0, init.shape[0], endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h0 = np.interp(x_new, x_old, init)
        else:
            h0 = np.full(n, 0.5)

        h0 = _project_onto_capped_simplex(h0, n / 2.0)

        # ------------------------------------------------------------------
        # optimisation stages
        # ------------------------------------------------------------------
        h_opt, val_opt = _gradient_descent(h0, gd_time, rng)

        if hc_time > 0.5:
            h_opt2, val_opt2 = _hill_climb(h_opt, hc_time, rng)
            if val_opt2 < val_opt:
                h_opt, val_opt = h_opt2, val_opt2

        if val_opt < best_val:
            best_h, best_val = h_opt.copy(), val_opt

        if best_val <= 0.3808:            # target already reached
            break

    # final safety projection and evaluation
    if best_h is None:
        n = n0
        best_h = np.full(n, 0.5)
    best_h = _project_onto_capped_simplex(best_h, len(best_h) / 2.0)
    final_c5 = _evaluate(best_h)

    return best_h, final_c5, int(len(best_h))
