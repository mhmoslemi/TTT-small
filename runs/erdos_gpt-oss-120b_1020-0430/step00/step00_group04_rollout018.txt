def hill_climb(n, time_budget):
    # start with random binary with n/2 ones
    h = np.zeros(n, dtype=np.int8)
    h[:n//2] = 1
    np.random.shuffle(h)
    best_h = h.copy()
    best_max = evaluate_max(h)  # integer count
    cur_h = h.copy()
    cur_max = best_max
    start = time.time()
    it_without_improve = 0
    while time.time() - start < time_budget:
        # propose swap
        i1 = np.random.choice(np.where(cur_h==1)[0])
        i0 = np.random.choice(np.where(cur_h==0)[0])
        cur_h[i1], cur_h[i0] = 0, 1
        new_max = evaluate_max(cur_h)
        if new_max < cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = cur_h.copy()
                # early exit if below target threshold
                # but we need compute bound
                if new_max * (2.0/n) <= target_bound:
                    return best_h, new_max * (2.0/n), n
            it_without_improve = 0
        else:
            # revert
            cur_h[i1], cur_h[i0] = 1, 0
            it_without_improve += 1
        # random restart if stuck
        if it_without_improve > 5000:
            # restart random but keep best maybe
            cur_h = np.zeros(n, dtype=np.int8)
            cur_h[:n//2] = 1
            np.random.shuffle(cur_h)
            cur_max = evaluate_max(cur_h)
            it_without_improve = 0
    return best_h, best_max*(2.0/n), n
```

import numpy as np, time, math, random

def evaluate_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def max_mismatch(h):
    # integer count
    return int(np.max(np.correlate(h, 1 - h, mode='full')))

def thue_morse_seq(n):
    seq = np.empty(n, dtype=np.int8)
    for i in range(n):
        seq[i] = bin(i).count('1') & 1  # parity
    return seq

def hill_climb(n, time_limit, target_c5=None):
    rng = np.random.default_rng()
    # start with random balanced binary
    h = np.zeros(n, dtype=np.int8)
    h[:n//2] = 1
    rng.shuffle(h)
    cur_max = max_mismatch(h)
    cur_h = h.copy()
    best_h = cur_h.copy()
    best_max = cur_max
    start = time.time()
    stuck_counter = 0
    while time.time() - start < time_limit:
        # try swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        cur_h[i_one], cur_h[i_zero] = 0, 1
        new_max = max_mismatch(cur_h)
        if new_max <= cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = cur_h.copy()
                if target_c5 is not None and new_max * (2.0/n) <= target_c5:
                    return best_h, best_max * (2.0/n)
            stuck_counter = 0

            stuck_counter += 1

            # random restart
            cur_h = np.zeros(n, dtype=np.int8)
            cur_h[:n//2] = 1
            rng.shuffle(cur_h)
            cur_max = max_mismatch(cur_h)
            stuck_counter = 0

import numpy as np
import time
import math

def _c5_value(h):
    """Return the C5 value for a (float or int) vector h."""
    n = len(h)
    # correlation of h with (1-h) over full overlap (zero padding)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _max_mismatch_int(h):
    """Return the integer maximum count of h[i]*(1-h[i+lag]) for all lags."""
    return int(np.max(np.correlate(h, 1 - h, mode="full")))


def _thue_morse(n):
    """Generate the first n bits of the Thue–Morse sequence (0/1)."""
    seq = np.empty(n, dtype=np.int8)
    for i in range(n):
        # parity of popcount
        seq[i] = bin(i).count("1") & 1
    return seq


def _hill_climb(n, time_limit, target_c5=None, rng=None):
    """Simple hill‑climbing: swaps a random 1 with a random 0,
    keeping the move if it does not increase the max mismatch count."""
    if rng is None:
        rng = np.random.default_rng()
    # start from a random balanced binary vector
    cur = np.zeros(n, dtype=np.int8)
    cur[: n // 2] = 1
    rng.shuffle(cur)

    cur_max = _max_mismatch_int(cur)
    best = cur.copy()
    best_max = cur_max
    start = time.time()
    stuck = 0

    while time.time() - start < time_limit:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)

        # try the swap
        cur[i1], cur[i0] = 0, 1
        new_max = _max_mismatch_int(cur)

        if new_max <= cur_max:
            cur_max = new_max
            stuck = 0
            if new_max < best_max:
                best_max = new_max
                best = cur.copy()
                if target_c5 is not None and new_max * (2.0 / n) <= target_c5:
                    return best.astype(float), best_max * (2.0 / n)
        else:
            # revert
            cur[i1], cur[i0] = 1, 0
            stuck += 1

        if stuck > 5000:
            # random restart
            cur[:] = 0
            cur[: n // 2] = 1
            rng.shuffle(cur)
            cur_max = _max_mismatch_int(cur)
            stuck = 0

    return best.astype(float), best_max * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] giving a small C5 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target_bound = 0.3808  # target to beat

    best_h = None
    best_val = float("inf")
    best_n = None

    # if a parent construction is supplied, evaluate it first
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float).copy()
        val = _c5_value(init)
        best_h, best_val, best_n = init, val, len(init)

    # candidate lengths (even numbers, up to ~256)
    candidates = [
        256, 240, 224, 200, 180, 160, 144, 128,
        112, 100, 96, 80, 72, 64, 56, 48, 44, 42, 40, 32
    ]

    # try a few structured sequences first (Thue‑Morse when applicable)
    for n in candidates:
        if time.time() - start_time > budget_s - 2:
            break
        if (n & (n - 1)) == 0:  # power of two → Thue‑Morse has exact half‑ones
            seq = _thue_morse(n)
            val = _c5_value(seq.astype(float))
            if val < best_val:
                best_h, best_val, best_n = seq.astype(float), val, n
                if val <= target_bound:
                    return best_h, best_val, best_n

    # main search loop over the candidate lengths
    for n in candidates:
        if time.time() - start_time > budget_s - 5:
            break

        # quick random sampling (20 attempts)
        for _ in range(20):
            if time.time() - start_time > budget_s - 5:
                break
            vec = np.zeros(n, dtype=np.int8)
            vec[: n // 2] = 1
            rng.shuffle(vec)
            val = _c5_value(vec.astype(float))
            if val < best_val:
                best_h, best_val, best_n = vec.astype(float), val, n
                if val <= target_bound:
                    return best_h, best_val, best_n

        # allocate remaining budget for hill climbing on this n
        remaining = budget_s - (time.time() - start_time) - 3
        if remaining <= 0:
            break
        # allow at most 30 s per size
        t_limit = min(30.0, remaining)
        h_opt, val_opt = _hill_climb(
