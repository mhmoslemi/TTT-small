import time
import random
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def _c5_and_corr(h):
    """Return C5 value and full correlation array for a step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5, corr


def _worst_shift(corr, n):
    """Index (integer shift) that gives the maximal C5 (worst overlap)."""
    idx = int(np.argmax(corr))                # position in full array
    shift = idx - (n - 1)                     # convert to signed shift
    return shift


def _gradient_move(h, shift, step):
    """
    Produce a feasible move that tries to decrease the maximal C5.
    Returns a new candidate vector (or None if no move possible).
    """
    n = h.size
    if shift >= 0:
        # contributions from right and left neighbours
        v = np.empty(n, dtype=np.float64)
        v[: n - shift] = h[shift:]            # h[i+shift]
        v[shift:] += h[: n - shift]           # h[i-shift]
    else:
        s = -shift
        v = np.empty(n, dtype=np.float64)
        v[s:] = h[: n - s]                    # h[i+shift] (shift negative)
        v[: n - s] += h[s:]                   # h[i-shift]

    # candidates for increase / decrease
    up_idx = np.where(h < 1.0 - 1e-12)[0]
    down_idx = np.where(h > 1e-12)[0]
    if up_idx.size == 0 or down_idx.size == 0:
        return None

    # choose where v is largest (best to increase) and smallest (best to decrease)
    i_up = up_idx[np.argmax(v[up_idx])]
    i_down = down_idx[np.argmin(v[down_idx])]

    if i_up == i_down:
        # fallback: pick any distinct pair
        if up_idx.size > 1:
            i_up = up_idx[0]
            i_down = up_idx[1]
        else:
            return None

    delta = min(step, 1.0 - h[i_up], h[i_down])
    if delta <= 0.0:
        return None

    h_new = h.copy()
    h_new[i_up] += delta
    h_new[i_down] -= delta
    return h_new


def _random_pair_move(h, step):
    """
    Randomly select two indices and transfer a small amount delta while
    respecting bounds. Returns the new vector (or None if infeasible).
    """
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # decide direction: move from i to j
    if h[i] < 1e-12 or h[j] > 1.0 - 1e-12:
        # try opposite direction
        i, j = j, i
    max_dec = min(step, h[i])
    max_inc = min(step, 1.0 - h[j])
    delta = min(max_dec, max_inc)
    if delta <= 0.0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def _project_sum_constraint(h, target):
    """
    Tiny corrective projection to bring sum exactly to target while staying
    inside [0,1]. Assumes the deviation is very small.
    """
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # spread the correction over entries that are not at a bound
    free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
    if free.size == 0:
        # fallback: adjust first component within bounds
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        return h
    # distribute uniformly
    inc = diff / free.size
    for idx in free:
        h[idx] = np.clip(h[idx] + inc, 0.0, 1.0)
    # if still off due to clipping, recurse briefly
    return _project_sum_constraint(h, target)


def hill_climb(initial_h, total_time):
    """
    Simple deterministic / stochastic hill‑climbing focused on the worst shift.
    Returns best vector found and its C5 value.
    """
    h = initial_h.copy()
    n = h.size
    target_sum = n / 2.0
    best_c5, _ = _c5_and_corr(h)
    best_h = h.copy()

    start = time.time()
    step = 0.02                     # basic move magnitude
    it = 0
    stagnation = 0

    while time.time() - start < total_time:
        it += 1
        # recompute worst shift
        cur_c5, corr = _c5_and_corr(h)
        shift = _worst_shift(corr, n)

        # try a gradient‑guided move
        cand = _gradient_move(h, shift, step)
        if cand is not None:
            cand_c5, _ = _c5_and_corr(cand)
        else:
            cand_c5 = np.inf

        # occasional random pair move (1 % chance)
        if random.random() < 0.01:
            rnd = _random_pair_move(h, step)
            if rnd is not None:
                rnd_c5, _ = _c5_and_corr(rnd)
                if rnd_c5 < cand_c5:
                    cand = rnd
                    cand_c5 = rnd_c5

        # accept if improvement
        if cand_c5 < best_c5 - 1e-12:
            h = cand
            best_c5 = cand_c5
            best_h = cand.copy()
            stagnation = 0
        else:
            # simulated‑annealing style occasional acceptance
            elapsed = time.time() - start
            remaining = max(total_time - elapsed, 1e-6)
            temp = 0.001 * (remaining / total_time)          # slowly cools
            prob = np.exp(-(cand_c5 - best_c5) / temp) if cand_c5 < np.inf else 0.0
            if random.random() < prob:
                h = cand
                best_c5 = cand_c5
                best_h = cand.copy()
            stagnation += 1

        # adapt step size
        if stagnation > 2000:
            step *= 0.5
            stagnation = 0
            if step < 1e-5:
                step = 1e-5
        elif it % 500 == 0:
            step = min(step * 1.02, 0.05)

        # safety: keep sum exact (tiny drift may appear)
        h = _project_sum_constraint(h, target_sum)

        # early exit if we already beat the target
        if best_c5 <= 0.3808:
            break

    return best_h, best_c5
