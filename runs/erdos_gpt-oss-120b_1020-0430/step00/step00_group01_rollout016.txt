def adjust_sum_to_target(h, target):
    # Use simple projection onto capped simplex.
    # We'll use iterative water-filling.
    n = len(h)
    # Initialize
    current = h.copy()
    # Clip to [0,1]
    current = np.clip(current, 0, 1)
    cur_sum = current.sum()
    diff = target - cur_sum
    if abs(diff) < 1e-12:
        return current
    # Determine indices where we can adjust:
    # if diff >0 need to increase some values less than 1.
    # if diff <0 need to decrease some values greater than 0.
    if diff > 0:
        # compute capacities:
        caps = 1.0 - current
        # total capacity = sum caps
        # If diff larger than capacity (should not happen), cap.
        add = diff
        # allocate proportionally to capacities:
        # For stability, simple loop: while add > 0 and there are indices with capacity > 0:
        # sort capacities descending to fill largest first.

            sub -= dec_i.sum()
            caps = current
    # final clip

    # if still off due to rounding, adjust a small epsilon on a random index not at bound
    cur_sum = current.sum()

    if abs(diff) > 1e-8:

        for _ in range(100):
            idx = np.random.randint(0, n)
            if 0 <= current[idx] + diff <= 1:
                current[idx] += diff
                break
    return current

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def worst_lag_and_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    # gradient of product sum for this lag
    g = np.zeros_like(h)
    if lag >= 0:
        L = n - lag
        if L > 0:
            g[:L] += h[lag:]      # h_i term
            g[lag:] += h[:L]      # h_{i+lag}
    else:
        p = -lag
        L = n - p
        if L > 0:
            g[p:] += h[:L]
            g[:L] += h[p:]
    return lag, g, corr.max()
```

def project_to_feasible(v, target, max_iter=100):
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()

import time
import numpy as np

# ---------- utility functions ----------

def compute_c5(h):
    """Maximum overlap integral for a step function h on [0,2]"""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def worst_lag_and_gradient(h):
    """Return the lag giving the maximal overlap, the gradient of the product
    sum for that lag, and the raw maximal correlation value."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = int(np.argmax(corr))
    lag = idx_max - (n - 1)            # 0‑shift at index n‑1
    g = np.zeros_like(h)

    if lag >= 0:
        L = n - lag
        if L > 0:
            g[:L] += h[lag:]          # ∂/∂h_i term
            g[lag:] += h[:L]          # ∂/∂h_{i+lag} term
    else:   # lag < 0
        p = -lag
        L = n - p
        if L > 0:
            g[p:] += h[:L]
            g[:L] += h[p:]
    return lag, g, float(corr.max())


def project_to_feasible(v, target, max_iter=80):
    """Euclidean projection of v onto the box‑capped simplex:
       0 ≤ x_i ≤ 1,   Σ x_i = target."""
    lo = v.min() - 1.0          # enough to make all entries hit 1 after clipping
    hi = v.max()                # enough to make all entries hit 0 after clipping
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo + hi) / 2.0, 0, 1)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        free = (proj > 0) & (proj < 1)
        cnt = free.sum()
        if cnt > 0:
            proj[free] += diff / cnt
            proj = np.clip(proj, 0, 1)
    return proj


def random_feasible_h(n):
    """Generate a random feasible h (0 ≤ h ≤ 1, Σh = n/2)."""
    v = np.random.rand(n)
    target = n / 2.0
    return project_to_feasible(v, target)


def propose_move(h, grad, step):
    """Propose a small mass transfer from a low‑gradient index to a high‑gradient
    index while respecting the box constraints."""
    n = h.shape[0]
    # select promising indices
    desc = np.argsort(grad)[::-1]   # high gradient first
    asc = np.argsort(grad)         # low gradient first
    k = max(1, n // 10)
    i_candidates = desc[:k]
    j_candidates = asc[:k]

    for _ in range(10):
        i = np.random.choice(i_candidates)
        j = np.random.choice(j_candidates)
        if i != j:
            break
    else:
        i, j = np.random.choice(n, size=2, replace=False)

    # maximum admissible transfer respecting bounds
    max_inc = min(step, 1.0 - h[i])
    max_dec = min(step, h[j])
    delta = min(max_inc, max_dec)
    if delta <= 0:
        # try opposite direction
        max_inc = min(step, 1.0 - h[j])
        max_dec = min(step, h[i])
        delta = min(max_inc, max_dec)
        if delta <= 0:
            return None
        i, j = j, i
    # randomise magnitude a bit
    delta *= np.random.rand()
    return int(i), int(j), float(delta)


# ---------- core optimisation ----------

def local_search(h_start, deadline, start_time):
    """Hill‑climbing on the worst‑lag overlap with simple simulated‑annealing
    acceptance.  Returns the best vector found before *deadline*."""
    n = h_start.shape[0]
    best_h = h_start.copy()
    best_val = compute_c5(best_h)

    cur_h = h_start.copy()
    cur_val = best_val
    step = 0.2
    it = 0

    while time.time() < deadline:
        lag, grad, _ = worst_lag_and_gradient(cur_h)
        mv = propose_move(cur_h, grad, step)
        if mv is None:
            step *= 0.5
            if step < 1e-6:
                break
            continue
        i, j, delta = mv
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_val = compute_c5(new_h)

        accept = False
        if new_val < cur_val:
            accept = True
        elif np.random.rand() < 0.01:   # occasional uphill step
            accept = True

        if accept:
            old_val = cur_val
            cur_h = new_h
            cur_val = new_val
            if new_val < old_val:
                step = min(step * 1.02, 0.5)

        if cur_val < best_val:
            best_h = cur_h.copy()
            best_val = cur_val
            if best_val < 0.3808:      # early stop if target reached
                break

        if it % 5000 == 0 and it > 0:
            step *= 0.9
        it += 1

    return best_h, best_val

    start = time.time()

    # ----- initialise ----- #

        h0 = random_feasible_h(n)

    best_h, best_val = h0.copy(), compute_c5(h0)

    # ----- optimisation loop (restarts until budget exhausted) ----- #
    total_budget = float(budget_s)
    while time.time() - start < total_budget * 0.96:
        # leave a small safety margin for finalisation
        remaining = total_budget - (time.time() - start) - 0.5
        if remaining <= 0.0:
            break
        deadline = time.time() + remaining

        # random restart with some probability
        if np.random.rand() < 0.25:
            h_start = random_feasible_h(n)
        else:
            h_start = best_h.copy()

        cand_h, cand_val = local_search(h_start, deadline, start)
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val
            if best_val < 0.3808:
                break

    # ----- final feasibility enforcement ----- #
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.shape[0] / 2.0
    if not np.isclose(best_h.sum(), target):
        best_h = project_to_feasible(best_h, target)

    final_val = compute_c5(best_h)

    return (best_h, final_val, int(best_h.shape[0]))
