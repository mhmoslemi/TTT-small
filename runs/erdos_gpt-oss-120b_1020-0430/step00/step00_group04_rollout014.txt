import time, math, random
import numpy as np

def _evaluate(h, dx):
    # max correlation of h with 1-h (zero-padded) multiplied by dx
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _binary_from_continuous(v):
    """Return a binary vector with exactly len(v)//2 ones, choosing the largest entries."""
    n = len(v)
    target = n // 2
    idx = np.argsort(-v)               # descending order
    out = np.zeros(n, dtype=float)
    out[idx[:target]] = 1.0
    return out

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points)
    where h_values is a list of values in [0,1] with sum = n_points/2,
    and c5_bound = max_x ∫ h(x)(1-h(x+x))dx (approximated by correlation).
    """
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Build an initial binary step function
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        # force a binary version (exactly half ones)
        base = _binary_from_continuous(base)
        # increase resolution by repeating the pattern
        rep = 4                      # 4‑fold replication (42→168)
        h = np.repeat(base, rep)
    else:
        # fallback: a plain random balanced binary vector
        n = 160                       # even, moderate size
        target = n // 2
        h = np.zeros(n, dtype=float)
        h[:target] = 1.0
        np.random.shuffle(h)

    n = h.size
    target = n // 2
    # Ensure exact binary balance (just in case)
    if int(h.sum()) != target:
        h = _binary_from_continuous(h)

    dx = 2.0 / n
    cur_score = _evaluate(h, dx)
    best_h = h.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing / greedy search on binary swaps
    # ------------------------------------------------------------------
    max_iters = 2_000_000                # generous upper bound
    T0 = 0.04                              # initial temperature
    Tf = 1e-5                              # final temperature
    iteration = 0

    # Pre‑compute index arrays (updated lazily)
    while True:
        # time / iteration stop
        if time.time() - start_time > budget_s * 0.95:
            break
        if iteration >= max_iters:
            break
        if best_score <= 0.3808:          # target reached → stop early
            break

        # recompute correlation to know the worst shift (used for guiding swaps)
        conv = np.correlate(h, 1.0 - h, mode='full')
        max_val = conv.max()
        max_k = np.argmax(conv) - (n - 1)          # shift index in [-n+1, n-1]

        # locate “bad” ones that directly contribute to the maximal overlap
        if max_k >= 0:
            i_range = np.arange(n - max_k)
            bad = i_range[(h[i_range] == 1) & (h[i_range + max_k] == 0)]
        else:
            shift = -max_k
            i_range = np.arange(shift, n)
            bad = i_range[(h[i_range] == 1) & (h[i_range - shift] == 0)]

        if bad.size:
            i_one = int(random.choice(bad))
        else:
            # no direct bad pair – fall back to a random 1‑position
            i_one = int(np.random.choice(np.flatnonzero(h)))

        i_zero = int(np.random.choice(np.flatnonzero(1.0 - h)))

        # perform the swap
        h[i_one] = 0.0
        h[i_zero] = 1.0

        new_score = _evaluate(h, dx)
        delta = new_score - cur_score

        # linear temperature schedule
        T = T0 + (Tf - T0) * (iteration / max_iters)
        if delta <= 0 or random.random() < math.exp(-delta / max(T, 1e-12)):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # reject – revert the swap
            h[i_one] = 1.0
            h[i_zero] = 0.0

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Final sanitisation (exact binary, correct mass)
    # ------------------------------------------------------------------
    best_h = np.rint(best_h).astype(float)     # force 0/1 values
    cur_sum = int(best_h.sum())
    if cur_sum != target:
        # adjust the smallest discrepancy
        diff = target - cur_sum
        if diff > 0:
            zeros = np.flatnonzero(best_h == 0)
            best_h[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.flatnonzero(best_h == 1)
            best_h[ones[: -diff]] = 0.0

    final_c5 = _evaluate(best_h, dx)

    # ------------------------------------------------------------------
    # 4. Return the solution
    # ------------------------------------------------------------------
    return best_h.tolist(), float(final_c5), n
