import time
import math
import numpy as np

def _c5_val(h: np.ndarray) -> float:
    """Return the C5 value for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _corr_vec(h: np.ndarray) -> np.ndarray:
    """Full cross‑correlation vector used repeatedly."""
    return np.correlate(h, 1.0 - h, mode="full")


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function (as a list), its C5 bound, and the number of samples.
    The optimizer performs a simulated‑annealing style search on the
    discretised function values.
    """
    np.random.seed(seed)
    start = time.time()

    # ----- initialise ---------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.array(init, dtype=np.float64)
    else:
        n = 200
        h = np.full(n, 0.5, dtype=np.float64)

    n = h.shape[0]
    target_sum = n / 2.0
    # enforce the integral constraint (tiny rounding errors possible)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # rescale then clip to keep within bounds
        h = h * (target_sum / cur_sum)
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        if cur_sum > 0:
            h = h * (target_sum / cur_sum)

    # ----- optimisation loop -----
    cur_corr = _corr_vec(h)
    cur_val = cur_corr.max() * (2.0 / n)

    best_h = h.copy()
    best_val = cur_val

    temperature = 0.12          # initial temperature for Metropolis steps
    it = 0
    stagnation = 0

    while time.time() - start < budget_s - 0.25:
        it += 1

        # 10 % of iterations: completely random pair, otherwise target worst shift
        if np.random.rand() < 0.10:
            i, j = np.random.randint(n, size=2)
            shift = None
        else:
            # locate the shift that currently gives the maximal overlap
            idx_max = int(np.argmax(cur_corr))
            shift = idx_max - (n - 1)                # shift in index units
            if shift >= 0:
                i = np.random.randint(0, n - shift)
                j = i + shift
            else:
                s = -shift
                i = np.random.randint(s, n)
                j = i - s

        if i == j:
            continue

        # decide transfer direction – move mass from the larger entry toward the smaller
        if shift is not None:
            # bias direction to reduce the worst‑shift contribution
            if h[i] > h[j]:
                forward = True
            else:
                forward = False
        else:
            forward = np.random.rand() < 0.5

        # maximal admissible transfer respecting bounds
        if forward:
            max_tr = min(h[i], 1.0 - h[j])
        else:
            max_tr = min(h[j], 1.0 - h[i])
        if max_tr <= 0.0:
            continue

        # step size: a fraction of the allowed transfer, scaled by temperature
        frac = min(1.0, temperature * 5.0)          # temperature controls aggressiveness
        eps = max_tr * np.random.rand() * frac
        if eps <= 0.0:
            continue

        # apply the candidate move
        if forward:
            h[i] -= eps
            h[j] += eps
        else:
            h[i] += eps
            h[j] -= eps

        new_corr = _corr_vec(h)
        new_val = new_corr.max() * (2.0 / n)

        # Metropolis acceptance
        if new_val < cur_val or np.exp((cur_val - new_val) / max(temperature, 1e-12)) > np.random.rand():
            cur_corr = new_corr
            cur_val = new_val
            stagnation = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            if forward:
                h[i] += eps
                h[j] -= eps
            else:
                h[i] -= eps
                h[j] += eps
            stagnation += 1

        # cooling schedule
        if it % (n * 7) == 0:
            temperature = max(temperature * 0.96, 1e-6)

        # occasional small random shake if stuck for a long time
        if stagnation > 8000:
            noise = (np.random.rand(n) - 0.5) * 0.012
            h += noise
            h = np.clip(h, 0.0, 1.0)
            # renormalise to exact integral
            cur_sum = h.sum()
            if cur_sum > 0:
                h = h * (target_sum / cur_sum)
            cur_corr = _corr_vec(h)
            cur_val = cur_corr.max() * (2.0 / n)
            stagnation = 0

    # ----- finalise -----
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = best_h * (target_sum / cur_sum)

    final_c5 = _c5_val(best_h)

    return best_h.tolist(), float(final_c5), n
