    best_h = h.copy()
    # temperature schedule
    T_initial = 0.1

    # number iterations max maybe large but limited by time.
    iteration = 0
    while time.time() - start < budget_s - 0.5: # reserved time
        # propose random swap
        # pick i where h[i]==1, j where h[j]==0
        idx_one = rng.choice(np.flatnonzero(h == 1.0))
        idx_zero = rng.choice(np.flatnonzero(h == 0.0))

        h[idx_one] = 0.0
        h[idx_zero] = 1.0
        new_c5 = compute_c5(h)

            if rng.random() < np.exp(-delta / T):

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and enforce Σx = target_sum (simple iterative projection)."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        s = x.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = int(free.sum())
        if cnt == 0:
            # distribute uniformly then clip again
            x += diff / x.size
            x = np.clip(x, 0.0, 1.0)
            continue
        add = diff / cnt
        x[free] += add
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction if needed
    s = x.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        idxs = np.where((x > 0.0) & (x < 1.0))[0]
        if idxs.size:
            i = idxs[0]
            delta = np.clip(diff, -x[i], 1.0 - x[i])
            x[i] += delta
    return x


def gradient_J(h: np.ndarray, shift: int) -> np.ndarray:
    """Derivative of J_shift = Σ h_i h_{i+shift} with respect to h."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[: n - shift] += h[shift:]
            g[shift:] += h[: n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += h[: n - s]
            g[: n - s] += h[s:]
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # initialise h
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():

        n = h.size
        target = n / 2.0
        # ensure feasibility (the provided vector should already be feasible,
        # but we defensively project)
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    else:
        n = 200                     # even size, modest resolution
        target = n / 2.0
        h = rng.random(n)
        h = h * (target / h.sum())
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)

    # current best
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    no_improve = 0
    max_no_improve = 3000
    factor = 0.5               # fraction of allowed transfer per move

    # ----------------------------------------------------------------------
    # main optimisation loop
    # ----------------------------------------------------------------------
    while time.time() - start < budget_s - 0.2:
        # correlation and worst shift
        corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(np.argmax(corr_full))
        max_val = float(corr_full[idx_max])
        shift = idx_max - (n - 1)          # shift corresponding to max overlap

        # gradient of the *positive* overlap J_shift
        g = gradient_J(cur_h, shift)

        # candidate indices for mass transfer
        up_mask = cur_h < 1.0 - 1e-12
        down_mask = cur_h > 1e-12

        if up_mask.any() and down_mask.any():

            no_improve = 0

    # final projection (precaution)

    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
