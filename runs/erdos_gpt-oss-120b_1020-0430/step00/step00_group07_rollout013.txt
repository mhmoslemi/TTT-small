def project_to_capped_simplex(y, target):
    lo = -np.max(y)  # ensures y+lo <=0 -> all zero or lower bound
    hi = 1 - np.min(y)  # ensures y+hi >=1 -> all at upper bound
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0, 1)
        s = x.sum()
        if s > target:
            # need to decrease sum -> increase lambda reduces (or?).
            # x_i = clip(y_i + lambda, 0, 1)
            # As lambda increases, all clipped values shift up, unless they saturate at 1.
            # Actually raising lambda increases y_i+lambda, thus increasing x_i (or saturating at 1).
            # So sum is increasing in lambda.
            # Therefore if sum > target, lambda is too high => need lower lambda.
            hi = mid
        else:
            lo = mid
    return np.clip(y + (lo+hi)/2, 0, 1)

import time
import numpy as np

def _project_capped_simplex(y, target):
    """
    Project vector y onto the set {x | 0<=x_i<=1, sum(x)=target}
    using a simple binary search on the Lagrange multiplier.
    """
    lo = -np.max(y)            # enough to push everything to 0
    hi = 1.0 - np.min(y)       # enough to push everything to 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def _evaluate_c5(h, dx):
    """Return C5 for a given step vector h."""
    # np.correlate with mode='full' gives linear convolution (zero padding)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _random_binary(n, rng):
    """Return a binary (0/1) array of length n with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    arr[ones] = 1.0
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    try:
        n_points = int(len(initial_h_values))
        init_h = np.array(initial_h_values, dtype=float).copy()
        have_initial = True
    except Exception:
        n_points = 80
        init_h = None
        have_initial = False

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ------------------------------------------------------------------
    # Initial feasible solution
    # ------------------------------------------------------------------
    if have_initial:
        h_curr = _project_capped_simplex(init_h, target_sum)
    else:
        h_curr = _random_binary(n_points, rng)

    best_h = h_curr.copy()
    best_val = _evaluate_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Phase 1 : binary simulated annealing
    # ------------------------------------------------------------------
    phase1_time = budget_s * 0.75
    phase1_end = start_time + phase1_time
    T_start = 0.05
    T_min = 1e-6

    # Ensure binary representation for this phase
    h_curr = (h_curr > 0.5).astype(float)

    curr_val = best_val
    no_improve = 0
    RESTART_LIMIT = 20000  # iterations without improvement before a restart

    while time.time() < phase1_end:
        # linear cooling based on elapsed fraction of phase 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / phase1_time)
        T = max(T_min, T_start * (1.0 - frac))

        # pick a 1 and a 0 to swap
        ones = np.flatnonzero(h_curr > 0.5)
        zeros = np.flatnonzero(h_curr < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        cand = h_curr.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_val = _evaluate_c5(cand, dx)

        delta = cand_val - curr_val
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            h_curr = cand
            curr_val = cand_val
            if cand_val < best_val - 1e-12:
                best_h = cand.copy()
                best_val = cand_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional restart to avoid deep stagnation
        if no_improve >= RESTART_LIMIT:
            # keep the best so far and randomize the rest
            h_curr = best_h.copy()
            # apply a few random swaps to diversify
            for _ in range(10):
                ones = np.flatnonzero(h_curr > 0.5)
                zeros = np.flatnonzero(h_curr < 0.5)
                if ones.size == 0 or zeros.size == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h_curr[i] = 0.0
                h_curr[j] = 1.0
            curr_val = _evaluate_c5(h_curr, dx)
            no_improve = 0

    # ------------------------------------------------------------------
    # Phase 2 : continuous refinement (keep sum exactly, stay in [0,1])
    # ------------------------------------------------------------------
    phase2_end = start_time + budget_s - 0.2   # tiny safety margin
    # start from the best binary vector found

    curr_val = best_val

    T_start_c = 0.005

    while time.time() < phase2_end:
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / budget_s)
        T = max(T_min_c, T_start_c * (1.0 - frac))

        i, j = rng.integers(0, n_points, size=2)
        if i == j:

        # Determine admissible delta so that both stay in [0,1] and sum unchanged
        lo = max(-h_curr[i], h_curr[j] - 1.0)
        hi = min(1.0 - h_curr[i], h_curr[j])
        if hi <= lo:

        # Take a random delta inside the feasible interval
        delta = rng.uniform(lo, hi)

        cand = h_curr.copy()
        cand[i] += delta
        cand[j] -= delta

        cand_val = _evaluate_c5(cand, dx)
        delta_obj = cand_val - curr_val
        if delta_obj <= 0.0 or rng.random() < np.exp(-delta_obj / T):
            h_curr = cand
            curr_val = cand_val
            if cand_val < best_val - 1e-12:
                best_h = cand.copy()
                best_val = cand_val

    # ------------------------------------------------------------------
    # Final safety checks and return
    # ------------------------------------------------------------------
    # Enforce constraints exactly before returning
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_capped_simplex(best_h, target_sum)
    best_val = _evaluate_c5(best_h, dx)

    return best_h, best_val, n_points
