def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # compute correlation between h and 1-h

    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_fft(h):
    n = len(h)
    m = n*2
    N = 1 << (2*n-1).bit_length()  # next pow of 2 >= 2*n
    H = np.fft.rfft(np.pad(h, (0, N - n), 'constant'))
    G = np.fft.rfft(np.pad(1 - h, (0, N - n), 'constant'))
    corr = np.fft.irfft(H * np.conj(G), n=N)
    # Use only first 2*n-1 entries (linear correlation)
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)

def propose_move(h, step_max):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    if random.random() < 0.5:
        # move from j to i (increase i, decrease j)
        max_delta = min(step_max, 1.0 - h[i], h[j])
        if max_delta <= 1e-12:
            return h  # no feasible move
        delta = random.random() * max_delta
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
    else:
        max_delta = min(step_max, 1.0 - h[j], h[i])
        if max_delta <= 1e-12:
            return h
        delta = random.random() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    return h_new

import numpy as np
import time
import random

def project_onto_bounds_and_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def max_corr_and_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    max_val = corr[idx]
    shift = idx - (len(h) - 1)
    return float(max_val), int(shift)

def gradient_for_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        s = shift
        # j <= n-1-s
        idx1 = slice(0, n - s)
        g[idx1] += 1.0
        g[idx1] -= h[idx1.start + s: idx1.stop + s]  # but this slice w/ offset?
        # easier using arrays:
        j = np.arange(n - s)
        g[j] -= h[j + s]
        # j >= s
        idx2 = slice(s, n)
        j2 = np.arange(s, n)
        g[j2] -= h[j2 - s]
    else:
        t = -shift
        # j >= t
        j = np.arange(t, n)
        g[j] += 1.0
        g[j] -= h[j - t]
        # j <= n-1 - t
        j2 = np.arange(0, n - t)
        g[j2] -= h[j2 + t]
    return g

def perturb(h, magnitude=0.05):
    n = len(h)
    delta = (np.random.rand(n) - 0.5) * 2 * magnitude
    delta -= delta.mean()
    h2 = h + delta
    return project_onto_bounds_and_sum(h2, target=n/2)

def subgradient_descent(h0, time_limit):
    start = time.time()
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_val, _ = max_corr_and_shift(best_h)
    it = 1

        cur_val, shift = max_corr_and_shift(h)

        g = gradient_for_shift(h, shift)

import numpy as np
import time
import random

def _project(v, target):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _max_corr_shift(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx]), int(idx - (len(h) - 1))


def _grad_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        s = shift
        j = np.arange(n - s)
        g[j] += 1.0
        g[j] -= h[j + s]
        j2 = np.arange(s, n)
        g[j2] -= h[j2 - s]
    else:
        t = -shift
        j = np.arange(t, n)
        g[j] += 1.0
        g[j] -= h[j - t]
        j2 = np.arange(0, n - t)
        g[j2] -= h[j2 + t]
    return g


def _perturb(h, mag=0.05):
    n = len(h)
    delta = (np.random.rand(n) - 0.5) * 2 * mag
    delta -= delta.mean()
    return _project(h + delta, target=n / 2)


def _subgrad_descent(h0, timelimit):
    start = time.time()
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_val, _ = _max_corr_shift(best_h)
    it = 1
    while time.time() - start < timelimit:
        cur_val, shift = _max_corr_shift(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        g = _grad_shift(h, shift)
        maxg = np.abs(g).max() + 1e-12
        eta = 0.3 / np.sqrt(it) / maxg
        h_new = _project(h - eta * g, target=n / 2)
        new_val, _ = _max_corr_shift(h_new)
        if new_val < cur_val:
            h = h_new
        if it % 2000 == 0:
            h = _perturb(h, mag=0.02)
        it += 1
        if it > 50000:
            pass
    return best_h, best_val


def _hill_climb(h0, timelimit):
    start = time.time()
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_val, _ = _max_corr_shift(best_h)
