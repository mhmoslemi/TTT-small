import numpy as np
import time

def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value for a step function defined by h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); the maximum multiplied by dx is the bound.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _transfer_move(h: np.ndarray) -> np.ndarray | None:
    """Move a small amount of mass from a positive entry to a non‑full entry."""
    n = h.shape[0]
    src = np.where(h > 1e-12)[0]
    sink = np.where(h < 1.0 - 1e-12)[0]
    if src.size == 0 or sink.size == 0:
        return None
    i = np.random.choice(src)
    j = np.random.choice(sink)
    if i == j:
        return None
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return None
    delta = np.random.rand() * max_delta * 0.5
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return h2


def _gradient_move(h: np.ndarray) -> np.ndarray | None:
    """Increase the overlap for the worst shift using gradient information."""
    n = h.shape[0]
    # worst lag
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    L = idx - (n - 1)                     # shift index (can be negative)

    # gradient of the overlap Σ h[i] h[i+L] with respect to each entry
    grad = np.zeros_like(h)
    if L >= 0:
        # i contributes h[i+L]; i+L contributes h[i]
        grad[: n - L] += h[L:]          # contribution of h[i] * h[i+L] w.r.t. h[i]
        grad[L:] += h[: n - L]          # contribution w.r.t. h[i+L]
    else:
        L = -L
        grad[L:] += h[: n - L]
        grad[: n - L] += h[L:]

    # choose a source (small gradient, has mass) and a sink (large gradient, room)
    src_candidates = np.where(h > 1e-12)[0]
    sink_candidates = np.where(h < 1.0 - 1e-12)[0]
    if src_candidates.size == 0 or sink_candidates.size == 0:
        return None

    src = src_candidates[np.argmin(grad[src_candidates])]
    sink = sink_candidates[np.argmax(grad[sink_candidates])]
    if src == sink:
        return None

    max_delta = min(h[src], 1.0 - h[sink])
    if max_delta <= 0.0:
        return None
    delta = np.random.rand() * max_delta * 0.5
    h2 = h.copy()
    h2[src] -= delta
    h2[sink] += delta
    return h2


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function giving a small C5 bound."""
    np.random.seed(seed)

    # ------------------------------------------------------------------

        # repeat each entry a few times to gain finer control while keeping the sum constraint
        repeat = 4                         # 70 → 280 points (well under 1000)
        h = np.repeat(base, repeat)
    else:
        # fallback: uniform 0.5 over 200 points
        h = np.full(200, 0.5, dtype=np.float64)

    n = h.shape[0]
    target_sum = n / 2.0
    # correct any tiny drift
    h *= target_sum / h.sum()

    best_h = h.copy()
    best_c5 = _c5_of(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

    while time.time() - start < budget_s * 0.95:
