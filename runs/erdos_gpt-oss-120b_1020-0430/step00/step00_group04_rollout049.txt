import numpy as np, time, math, random

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_sum(h, target):
    # ensure sum exactly target while staying in [0,1]
    # adjust using while loop simple: compute diff, adjust a non-bound element.
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # find indices where can add/subtract
    if diff > 0:
        # need to add diff, increase some elements not at 1
        idx = np.where(h < 1)[0]
        # distribute to first idx proportionally
        for i in idx:
            room = 1.0 - h[i]
            inc = min(room, diff)
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # need to subtract
        idx = np.where(h > 0)[0]
        for i in idx:
            room = h[i]
            dec = min(room, -diff)
            h[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    # after loop diff may still not zero due to rounding, ignore.
    return h

import time
import math
import random
import numpy as np

def _evaluate(h):
    """Compute C5 for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h, target):
    """Adjust h in place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0.0:
        # need to add mass, use slots with room to 1
        room = 1.0 - h
        idx = np.argsort(-room)            # biggest room first
        for i in idx:
            if diff <= 0.0:
                break
            add = min(room[i], diff)
            h[i] += add
            diff -= add
    else:
        # need to remove mass, use slots with positive value
        excess = h
        idx = np.argsort(-excess)           # biggest value first
        for i in idx:
            if diff >= 0.0:
                break
            sub = min(excess[i], -diff)
            h[i] -= sub
            diff += sub
    # tiny numerical residue – spread uniformly if needed
    if abs(diff) > 1e-12:
        # distribute the tiny remainder equally (will stay inside bounds)
        h += diff / h.size
    return h

def _random_start(n, target):
    """Create a random feasible h (values in [0,1] with given sum)."""
    h = np.random.rand(n)               # uniform in [0,1)
    h = _repair_sum(h, target)
    return h

    Search for a step function h on [0,2] (discretised) that yields a low
    Erdős C5 bound.

    Returns (h_values, c5_bound, n_points).

    target_sum = n / 2.0           # because ∫_0^2 h = 1   ⇒  sum(h)·dx = 1

    # make sure the vector satisfies the constraints
    h_current = np.clip(h_current, 0.0, 1.0)
    h_current = _repair_sum(h_current, target_sum)

    best_h = h_current.copy()
    best_val = _evaluate(best_h)

    # ------------------------------------------------------------------
    # simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s * 0.98   # small safety margin

    # annealing schedule parameters
    T0 = 5e-4                     # initial temperature
    max_move_start = 0.4
    min_move = 1e-5

    no_improve = 0
    iter_cnt = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s * 0.98)
        temperature = max(1e-9, T0 * (1.0 - progress))
        max_move = max_move_start * (1.0 - progress) + min_move

        i, j = np.random.choice(n, size=2, replace=False)

            continue
