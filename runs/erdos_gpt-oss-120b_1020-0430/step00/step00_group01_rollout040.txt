    return w

import time
import numpy as np

def _project_to_box_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project v onto the feasible set {0 <= x_i <= 1, sum x_i = target_sum}.
    This is the classic capped‑simplex projection via a bisection on λ.
    """
    lo = np.min(v) - 1.0          # low enough → all entries will be clipped to 1
    hi = np.max(v)                # high enough → all entries will be clipped to 0
    for _ in range(60):           # enough for double precision
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return w


def _c5_of(h: np.ndarray) -> float:
    """Return the overlap constant C₅ for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _gradient_of_max_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of f(shift) = Σ h[i] (1‑h[i+shift]) w.r.t. h,
    where out‑of‑range indices are ignored (zero padding).
    """
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        idx = np.arange(0, n - shift)
        g[idx] += 1.0 - h[idx + shift]
        g[idx + shift] -= h[idx]
    else:
        s = -shift
        idx = np.arange(s, n)
        g[idx] += 1.0 - h[idx - s]
        g[idx - s] -= h[idx]
    return g


def _local_optimize(start_h: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Perform a simple projected gradient descent on the maximal overlap.
    Returns the best vector found and its C₅ value.
    """
    n = start_h.size
    target = n * 0.5
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_of(h)

    start = time.time()
    # a modest number of iterations; we stop early if no progress
    for _ in range(2000):
        if time.time() - start > time_limit:
            break

        # locate the shift attaining the current maximum
        n_corr = 2 * n - 1
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_idx = int(np.argmax(corr))
        cur_val = float(corr[max_idx])
        shift = max_idx - (n - 1)

        # gradient of the active term (negative direction reduces it)
        g = _gradient_of_max_shift(h, shift)
        direction = -g

        # line search with backtracking
        step = 0.5
        improved = False
        while step > 1e-7:
            cand = h + step * direction
            cand = _project_to_box_and_sum(cand, target)
            cand_val = _c5_of(cand)
            if cand_val < cur_val - 1e-12:
                h = cand
                cur_val = cand_val
                improved = True
                break
            step *= 0.5

        if not improved:            # cannot descend – stop this run
            break

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= 0.3808:   # good enough, abort early
                break
    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Find a step function h on [0,2] with average 1/2 that minimizes
    C₅ = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine the working discretisation.
    # If a parent construction exists we start from it (after possible
    # re‑interpolation to a convenient size); otherwise we use a simple
    # linear ramp which already satisfies the constraints exactly.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        # Use the given vector length as baseline; we may also try a
        # higher‑resolution version later.
        base_h = np.asarray(initial_h_values, dtype=float)
        n = base_h.size
        # guarantee feasibility (the supplied vector should already be feasible,
        # but we project just in case of tiny rounding errors)
        target_sum = n * 0.5

    else:

        n = 100

        target_sum = n * 0.5

    # ------------------------------------------------------------------
    # Main optimisation loop: several random restarts and a deterministic
    # gradient descent for each start.  The total wall‑clock time is kept
    # below the allotted budget.
    # ------------------------------------------------------------------
    elapsed = lambda: time.time() - start_time
    # first try the supplied construction (if any)

    # ------------------------------------------------------------------
    final_c5 = _c5_of(best_h)
    return (best_h.tolist(), float(final_c5), int(n))
