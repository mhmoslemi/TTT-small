import time
import numpy as np

def _eval_c5(vec, dx):
    """Return the overlap bound for a given vector."""
    # cross‑correlation of h with (1‑h), zero‑padded
    return np.max(np.correlate(vec, 1.0 - vec, mode="full")) * dx


def _project_sum(v, target):
    """Adjust a single coordinate so that sum(v) == target while staying in [0,1]."""
    diff = target - v.sum()
    # try to fix by moving one entry
    for i in range(v.size):
        new = v[i] + diff
        if 0.0 <= new <= 1.0:
            v[i] = new
            return v
    # fallback: distribute diff over two entries
    for i in range(v.size):
        for j in range(i + 1, v.size):
            # move diff from j to i
            if 0.0 <= v[i] + diff <= 1.0 and 0.0 <= v[j] - diff <= 1.0:
                v[i] += diff
                v[j] -= diff
                return v
    # last resort: scale then correct the remainder
    v = v * (target / v.sum())
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    v[-1] = np.clip(v[-1] + diff, 0.0, 1.0)
    return v


def run(seed=42, budget_s=1100, **kwargs):

    # Initialise h

            h = _project_sum(h, target_sum)
    except NameError:
        # No supplied start – create a random binary vector (half ones)
        N = 70

        ones = rng.choice(N, size=N // 2, replace=False)

    target_bound = 0.3808
    if best_c5 <= target_bound:

    cur_c5 = best_c5

    # ------------------------------------------------------------

    best_c5 = _eval_c5(best_h, dx)

    return best_h, best_c5, N
