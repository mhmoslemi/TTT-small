def _objective_vals(h, dx):
    """Return the max overlap integral C5 for a given h."""
    # correlation of h with (1 - h), full mode, then scale by dx
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _clip_and_project(h, target_sum):
    """
    Clip values to [0,1] and adjust to meet the exact sum constraint.
    The adjustment keeps the vector within bounds.
    """
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    # scale then re‑clip (the scaling keeps the sum exact if no clipping occurs)
    scale = target_sum / cur if cur > 0 else 0.0
    h *= scale
    # after scaling some entries may violate bounds; fix them iteratively
    for _ in range(5):
        over = h > 1.0
        under = h < 0.0
        if not over.any() and not under.any():
            break
        excess = h[over] - 1.0
        deficit = -h[under]
        # redistribute excess to deficit proportionally
        total_excess = excess.sum()
        total_deficit = deficit.sum()
        if total_excess > 0 and total_deficit > 0:
            # move mass from over‑filled to under‑filled slots
            transfer = min(total_excess, total_deficit)
            # proportionally adjust
            h[over] -= transfer * excess / total_excess
            h[under] += transfer * deficit / total_deficit
        h = np.clip(h, 0.0, 1.0)
    # final correction of tiny numerical drift
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            add = diff / free.size
            h[free] += add
            h = np.clip(h, 0.0, 1.0)
    return h

    T_end = 1e-5
    max_iters = 500000   # generous upper bound; loop will break on time‑out

    for it in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.96:          # leave a small safety margin
            break

        # linear temperature decay
        T = T_start + (T_end - T_start) * (it / max_iters)

        # propose a transfer of mass between two distinct indices
        i = random.randrange(n)
        # ensure we can take some mass from i
        if h[i] <= 1e-12:
            continue
        j = random.randrange(n)
        # ensure we can give mass to j
        if h[j] >= 1.0 - 1e-12 or j == i:
            continue

        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        # random fraction of the allowable delta (up to 50 % of max_delta)
        delta = max_delta * random.random() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # evaluate the candidate
        new_val = _objective_vals(h_new, dx)

        # keep the best ever seen
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

        # accept according to Metropolis criterion
        if new_val < cur_val or random.random() < np.exp(-(new_val - cur_val) / T):
            h = h_new
            cur_val = new_val

        # occasional small‑scale refinement: try a tiny jitter on a random index
        if it & 0x3FF == 0:   # every 1024 steps
            idx = random.randrange(n)
            # jitter within remaining slack
            slack_down = h[idx]
            slack_up = 1.0 - h[idx]
            eps = (slack_down + slack_up) * 0.01 * random.random()
            direction = 1.0 if random.random() < 0.5 else -1.0
            delta = direction * eps
            h_jitter = h.copy()
            h_jitter[idx] = np.clip(h_jitter[idx] + delta, 0.0, 1.0)
            # renormalise to keep the total integral exactly 1
            h_jitter = _clip_and_project(h_jitter, target_sum)
            jitter_val = _objective_vals(h_jitter, dx)
            if jitter_val < best_val:
                best_val = jitter_val
                best_h = h_jitter.copy()
            if jitter_val < cur_val or random.random() < np.exp(-(jitter_val - cur_val) / T):
                h = h_jitter
                cur_val = jitter_val

    # final projection to guarantee feasibility
    best_h = _clip_and_project(best_h, target_sum)

    # compute the bound with the official helper if available
    if 'evaluate_erdos_solution' in globals():
        c5_bound = float(evaluate_erdos_solution(best_h))
    else:
        c5_bound = float(_objective_vals(best_h, dx))

    return best_h, c5_bound, n
