def compute_c5_fft(h):
    n = len(h)
    # zero pad to length 2*n
    H = np.fft.rfft(h, n=2*n)
    B = np.fft.rfft(1 - h, n=2*n)
    conv = np.fft.irfft(H * B, n=2*n)  # real
    # conv length 2*n
    # We want full mode results for lag from -(n-1) to (n-1)
    # In linear convolution, result at index i corresponds to lag i - (n-1)
    # conv[0] corresponds to lag -(n-1) ??? let's verify.
    # Using convolution of sequences a and b (both length n) zero-padded to length 2n,
    # the result conv[k] = sum_i a[i] * b[k - i], for k=0..2n-2.
    # For correlation with shift s = k - (n-1). So we need conv[ (n-1) + s ]
    # So max across s corresponds to max across conv (maybe except ignoring first and last zero).
    # So we can just take max of conv[:2*n-1].

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_onto_bounds_and_sum(x, target):
    # Simple projection onto [0,1] and sum constraint via bisection
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    """Maximum overlap integral for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def project_onto_bounds_and_sum(x, target):
    """Projection onto the box [0,1] with a fixed sum (simple bisection)."""
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):

        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
