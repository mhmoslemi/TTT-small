def project_to_feasible(h, target):
    # Ensure 0<=h<=1
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum to target using simple water-filling
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # Determine direction
    if diff > 0:
        # need to decrease sum
        # possible decrease amounts = h_i (since can decrease to 0)
        # sort indices by h descending maybe
        idx = np.argsort(-h)  # largest first
        for i in idx:
            dec = min(diff, h[i])
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    else:
        # need to increase
        idx = np.argsort(h)  # smallest first
        diff = -diff
        for i in idx:
            inc = min(diff, 1 - h[i])
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    return h

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

def _compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clamp to [0,1] and adjust the sum to the exact target."""
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:  # need to decrease
        idx = np.argsort(-h)  # largest first
        for i in idx:
            dec = min(diff, h[i])
            h[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    else:          # need to increase
        need = -diff
        idx = np.argsort(h)   # smallest first
        for i in idx:
            inc = min(need, 1.0 - h[i])
            h[i] += inc
            need -= inc
            if need <= 1e-12:
                break
    return h


def _random_feasible_h(n: int) -> np.ndarray:
    """Generate a random feasible vector (sum = n/2, 0<=h_i<=1)."""
    g = np.random.exponential(1.0, size=n)
    h = g / g.sum() * (n / 2.0)
    h = _project_to_feasible(h, n / 2.0)
    return h


def _v_shape(n: int, invert: bool = False) -> np.ndarray:
    """Triangular (or inverted) shape, scaled to meet the integral constraint."""
    idx = np.arange(n)
    norm = np.abs(idx - (n - 1) / 2.0) / ((n - 1) / 2.0)   # 0 at centre, 1 at edges
    base = (1.0 - norm) if invert else norm
    # scale to exact sum
    target = n / 2.0
    factor = target / base.sum()
    h = base * factor
    h = _project_to_feasible(h, target)
    return h


def _refine_slsqp(h0: np.ndarray, maxiter: int = 300) -> (np.ndarray, float):
    """Local SLSQP refinement of a feasible vector."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        return _compute_c5(x)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
    if res.success:
        h_opt = _project_to_feasible(res.x, target)
        return h_opt, _compute_c5(h_opt)
    # fallback – return the original if optimization fails
    return h0, _compute_c5(h0)

    """

    """
    start_time = time.time()
    np.random.seed(seed)

    # -----------------------------------------------------------------
    # Determine number of discretisation points
    # -----------------------------------------------------------------
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        n = h_init.size
        # make sure the supplied vector satisfies the constraints;
        # if not, project it.
        h_init = _project_to_feasible(h_init, n / 2.0)
    else:
        n = 60            # a modest size that still allows a good bound
        h_init = _random_feasible_h(n)

    target_sum = n / 2.0

    # -----------------------------------------------------------------
    # Assemble a few starting candidates
    # -----------------------------------------------------------------
    candidates = [h_init,

    for _ in range(4):

    best_h = None

    # -----------------------------------------------------------------

    for cand in candidates:

        h_opt, c5_val = _refine_slsqp(cand, maxiter=250)

    # -----------------------------------------------------------------
    # Global optimisation with differential evolution (if time permits)

    best_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(best_c5), int(n)
