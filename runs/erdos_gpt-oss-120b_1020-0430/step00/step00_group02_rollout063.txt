def project_to_feasible(v, target):
    n = v.shape[0]
    lo = -np.max(v) - 1.0  # ensure low bound
    hi = 1.0 - np.min(v) + 1.0

            hi = mid

import time
import numpy as np

def _project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x_i<=1, sum(x)=target} by a simple bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _max_overlap(h: np.ndarray) -> float:
    """Return the C5 value for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = None

    # Choose a modest resolution that balances speed and quality.
    # Multiple resolutions could be tried; we stick to one for simplicity.
    n = 150
    target_sum = n / 2.0
    dx = 2.0 / n

    # Helper to create an initial feasible vector.
    def _make_initial(kind):
        if kind == "cos":
            xs = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = (1.0 + np.cos(np.pi * xs)) * 0.5
            return _project_to_feasible(h0, target_sum)
        if kind == "interp" and 'initial_h_values' in globals():
            ih = globals().get('initial_h_values')
            if ih is not None and len(ih) > 1:
                src_x = np.linspace(0.0, 2.0, len(ih), endpoint=False)
                tgt_x = np.linspace(0.0, 2.0, n, endpoint=False)
                h0 = np.interp(tgt_x, src_x, ih)
                h0 = np.clip(h0, 0.0, 1.0)
                return _project_to_feasible(h0, target_sum)
        # random start
        v = rng.rand(n)
        return _project_to_feasible(v, target_sum)

    # Main optimization loop with many random restarts.
    restart = 0
    while time.time() - start_time < budget_s * 0.90:
        restart += 1
        # Use a variety of initial vectors.
        if restart == 1:
            h = _make_initial("cos")
        elif restart == 2:
            h = _make_initial("interp")
        else:
            h = _make_initial("rand")
        # enforce symmetry to reduce dimensionality.
        h = (h + h[::-1]) * 0.5

        step = 0.12
        max_iter = 3000
        for it in range(max_iter):
            now = time.time()
            if now - start_time > budget_s * 0.90:
                break

            # correlation of h with itself (linear, not circular)
            conv = np.correlate(h, h, mode="full")
            min_idx = int(conv.argmin())
            min_corr = float(conv[min_idx])

            # current C5
            cur_c5 = 1.0 - min_corr * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_n = n

            # gradient for the shift that attains the minimum correlation
            shift = abs(min_idx - (n - 1))
            if shift == 0:
                grad = 2.0 * h
            else:
                grad = np.zeros_like(h)
                grad[: n - shift] += h[shift:]
                grad[shift:] += h[: n - shift]

            # ascent step (increase the minimal correlation)
            h_new = h + step * grad
            h_new = _project_to_feasible(h_new, target_sum)
            # keep symmetry
            h_new = (h_new + h_new[::-1]) * 0.5

            # Evaluate improvement
            conv_new = np.correlate(h_new, h_new, mode="full")
            min_corr_new = float(conv_new.min())
            if min_corr_new > min_corr:
                h = h_new
                step = min(step * 1.04, 1.0)   # slightly accelerate
            else:
                step *= 0.55
                if step < 1e-5:
                    break

        # occasional small random perturbation of the best solution found so far
        if best_h is not None and time.time() - start_time < budget_s * 0.95:
            h_pert = best_h.copy()
            for _ in range(200):
                i, j = rng.choice(n, 2, replace=False)
                delta = rng.uniform(-0.03, 0.03)
                h_pert[i] = np.clip(h_pert[i] + delta, 0.0, 1.0)
                h_pert[j] = np.clip(h_pert[j] - delta, 0.0, 1.0)
            h_pert = _project_to_feasible(h_pert, target_sum)
            h_pert = (h_pert + h_pert[::-1]) * 0.5
            c5_pert = _max_overlap(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                best_n = n

    # Fallback in the unlikely event nothing was found
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = _max_overlap(best_h)
        best_n = n

    # Final safety projection (ensures exact constraints)
    best_h = _project_to_feasible(best_h, best_h.shape[0] / 2.0)

    final_c5 = _max_overlap(best_h)

    return best_h, final_c5, best_n
