def hill_climb(h, dx, max_iter=100000, patience=1000):
    best_h = h.copy()
    best_val = compute_c5(best_h)
    no_improve = 0
    n = len(h)
    ones = np.where(best_h == 1)[0]
    zeros = np.where(best_h == 0)[0]

    for it in range(max_iter):
        # random swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i], h[j] = 0, 1
        val = compute_c5(h)
        if val < best_val - 1e-12: # improvement
            best_val = val
            best_h = h.copy()
            # recompute ones, zeros after acceptance
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_improve = 0
        else:
            # revert
            h[i], h[j] = 1, 0
            no_improve += 1
        if no_improve >= patience:
            break
    return best_h, best_val
```

def run(seed=42, budget_s=1100):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    n_points = 200  # even
    target_sum = n_points/2
    dx = 2.0 / n_points

    def evaluate_arr(arr):
        # returns C5
        corr = np.correlate(arr, 1 - arr, mode='full')
        return np.max(corr) * dx

    # projection onto box and sum
    def project_box_sum(v, target):
        lo = -np.max(v) - 1.0
        hi = np.max(1 - v) + 1.0
        for _ in range(60):
            lam = (lo + hi) / 2.0
            proj = np.clip(v - lam, 0.0, 1.0)
            s = proj.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    def smooth_obj_grad(h, alpha):
        n = h.size
        c = np.correlate(h, 1 - h, mode='full')
        maxc = c.max()
        w = np.exp(alpha * (c - maxc))
        w /= w.sum()
        grad = np.zeros_like(h)
        # range of shifts
        for idx, k in enumerate(range(-(n-1), n)):
            weight = w[idx]
            if weight == 0:
                continue
            if k >= 0:
                grad[:n - k] += weight * (1 - h[k:])
            else:
                kk = -k
                grad[kk:] += weight * (1 - h[:-kk])
        return maxc, grad

    # init triangular shape
    x = (np.arange(n_points) + 0.5) * dx
    h_tri = np.minimum(x, 2 - x)  # triangle peaks at 1
    # scale to sum target (should be already measure 1? Check)

    cur_sum = h_tri.sum()
    # ensure sum = target_sum

    np.clip(h_tri, 0, 1, out=h_tri)
    h_tri = project_box_sum(h_tri, target_sum)

    best_h = h_tri.copy()
    best_val = evaluate_arr(best_h)

    # gradient descent parameters
    total_time = time.time() - start

import numpy as np
import time

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Find a step function h on [0,2] (discretized) that minimizes
        C5 = max_k ∫ h(x) (1 - h(x+k)) dx
    under 0 ≤ h ≤ 1 and ∫ h = 1.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # ----------------------------------------------------------------------
    # helpers
    # ----------------------------------------------------------------------
    def evaluate(h: np.ndarray) -> float:
        """exact C5 for a discretised h (dx accounts for interval length)."""
        n = h.size
        dx = 2.0 / n
        return np.max(np.correlate(h, 1 - h, mode="full")) * dx

    def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
        """
        Project vector v onto the set {0 ≤ x ≤ 1, sum(x)=target}
        using a simple bisection on the shift λ:
            proj_i = clamp(v_i - λ, 0, 1)
        The sum of proj_i is monotone decreasing in λ.
        """
        lo = -np.max(v) - 1.0
        hi = np.max(1.0 - v) + 1.0
        for _ in range(60):
            lam = (lo + hi) * 0.5
            proj = np.clip(v - lam, 0.0, 1.0)
            s = proj.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    def smooth_max_grad(h: np.ndarray, alpha: float):
        """smooth approximation of max_k correlation and its gradient."""
        n = h.size
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = corr.max()
        # softmax weights
        w = np.exp(alpha * (corr - max_corr))
        w /= w.sum()
        grad = np.zeros_like(h)
        # shifts from -(n-1) … (n-1)
        for idx, k in enumerate(range(-(n - 1), n)):
            wk = w[idx]
            if wk == 0.0:
                continue
            if k >= 0:
                grad[: n - k] += wk * (1.0 - h[k:])
            else:
                kk = -k
                grad[kk:] += wk * (1.0 - h[: n - kk])
        return max_corr, grad

    # ----------------------------------------------------------------------
    # initialise problem size
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float, copy=False)
        n_pts = h0.size
    else:
        # fallback: moderate size with a simple triangular start
        n_pts = 200  # even, < 1k as required
        dx = 2.0 / n_pts
        xs = (np.arange(n_pts) + 0.5) * dx
        h0 = np.minimum(xs, 2.0 - xs)          # triangle, height 1
        # enforce integral = 1
        target_sum = n_pts / 2.0
        h0 = h0 * (target_sum / h0.sum())
    target_sum = n_pts / 2.0

    # make sure the starting vector respects constraints
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = evaluate(best_h)

    # ----------------------------------------------------------------------
    # optimisation – smooth gradient descent with occasional random refinements
    # ----------------------------------------------------------------------
    deadline = start_time + budget_s * 0.94   # leave a margin for final steps

    def descent(start_h, max_iter=5000, lr0=0.5, alpha0=30.0):
        """gradient descent on the smoothed max‑correlation objective."""
        nonlocal best_h, best_val
        h = start_h.copy()
        lr = lr0
        alpha = alpha0
        it = 0
        while time.time() < deadline and it < max_iter:
            _, g = smooth_max_grad(h, alpha)
            h = h - lr * g
            h = project_box_sum(h, target_sum)
            val = evaluate(h)
            if val + 1e-12 < best_val:
                best_val = val
                best_h = h.copy()
            # slowly shrink step‑size, slowly increase smoothness
            lr *= 0.995
            alpha *= 1.001
            it += 1
        return h

    # start from the supplied construction
    h_cur = descent(h0, max_iter=8000, lr0=0.8, alpha0=20.0)

    # a few random restarts (perturbed starting points)
    for _ in range(3):
        if time.time() >= deadline:
            break
        noise = np.random.randn(n_pts) * 0.07
        pert = h_cur + noise
        pert = np.clip(pert, 0.0, 1.0)
        pert = project_box_sum(pert, target_sum)
        h_cur = descent(pert, max_iter=4000, lr0=0.6, alpha0=25.0)

    # ----------------------------------------------------------------------
    # final local refinement using tiny mass transfers (preserves sum)
    # ----------------------------------------------------------------------
    def local_refine(h, steps=4000, delta_max=0.06):
        nonlocal best_h, best_val
        cur = h.copy()
        cur_val = evaluate(cur)
        n = cur.size
        donors = np.where(cur > 1e-12)[0]
        receivers = np.where(cur < 1.0 - 1e-12)[0]
        for _ in range(steps):
            if time.time() >= deadline:
                break
            if donors.size == 0 or receivers.size == 0:
                break
            i = np.random.choice(donors)
            j = np.random.choice(receivers)
            delta = delta_max * np.random.rand()
            delta = min(delta, cur[i], 1.0 - cur[j])
            if delta <= 0.0:
                continue
            cur[i] -= delta
            cur[j] += delta
            new_val = evaluate(cur)
            if new_val + 1e-12 < cur_val:
                cur_val = new_val
                donors = np.where(cur > 1e-12)[0]
                receivers = np.where(cur < 1.0 - 1e-12)[0]
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = cur.copy()
            else:
                # revert
                cur[i] += delta
                cur[j] -= delta
        return cur

    best_h = local_refine(best_h, steps=5000, delta_max=0.04)

    # ----------------------------------------------------------------------
    # final fixing of the exact integral (tiny adjustment on a free entry)
    # ----------------------------------------------------------------------
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in range(n_pts):
            if diff > 0 and best_h[idx] < 1.0:
                add = min(diff, 1.0 - best_h[idx])
                best_h[idx] += add
                diff -= add
            elif diff < 0 and best_h[idx] > 0.0:
                sub = min(-diff, best_h[idx])
                best_h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    np.clip(best_h, 0.0, 1.0, out=best_h)

    final_val = evaluate(best_h)
    return best_h, final_val, n_pts
