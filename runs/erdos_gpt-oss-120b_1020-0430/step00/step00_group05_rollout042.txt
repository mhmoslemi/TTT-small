def local_search(N, budget, rng):

    # start with constant 0.5
    h = np.full(N, 0.5)
    # adjust sum: since sum = 0.5*N = N/2 already ok. That's fine.

    cur = compute_c5(h)
    best = cur
    best_h = h.copy()

    start = time.perf_counter()
    while time.perf_counter() - start < budget:
        # propose move
        proposal = propose_swap(h, rng)
        if proposal is None:
            continue
        i, j, delta = proposal
        new = compute_c5(h)
        # acceptance
        elapsed = time.perf_counter() - start
        fraction = elapsed / budget
        T = max(1e-6, 0.005 * (1 - fraction))  # start with 0.005 maybe
        if new <= cur or rng.random() < np.exp(-(new - cur) / T):
            # accept
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            # reject: revert
            # revert changes
            h[i] -= delta
            h[j] += delta
        # possibly occasional random restart if no improvement for many steps
    return best_h, best

import numpy as np
import time

def _project_onto_box_and_sum(v, target):
    """Project v onto the box [0,1] with the linear equality sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(v + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h):
    """Compute the C5 bound for a step vector h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _propose_move(h, rng, max_step):
    """Propose a feasible two‑coordinate move that keeps the sum constant."""
    N = h.shape[0]
    i = rng.integers(N)
    j = rng.integers(N - 1)
    if j >= i:
        j += 1
    hi, hj = h[i], h[j]

    # feasible delta so that 0<=hi+δ<=1 and 0<=hj-δ<=1
    delta_min = max(-hi, -(1.0 - hj))
    delta_max = min(1.0 - hi, hj)
    if delta_max <= delta_min:
        return None
    # restrict the step size to keep changes moderate
    span = delta_max - delta_min
    bound = min(max_step, span * 0.5)
    delta = rng.uniform(-bound, bound)
    # clip into feasible interval
    delta = np.clip(delta, delta_min, delta_max)
    return i, j, delta


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        N = h0.size
        # make sure it satisfies the constraints (tiny numerical drift is ok)
        h0 = _project_onto_box_and_sum(h0, N / 2.0)
    else:
        N = 70                     # a reasonable default
        h0 = np.full(N, 0.5)       # start from the constant function
        h0 = _project_onto_box_and_sum(h0, N / 2.0)

    # basic parameters
    target_sum = N / 2.0
    max_step = 0.1                # maximal absolute change in a move
    start = time.perf_counter()
    best_h = h0.copy()
    best_c5 = _c5_value(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    it = 0
    # Simulated‑annealing loop
    while True:
        now = time.perf_counter()
        if now - start > budget_s * 0.94:   # safety margin
            break
        it += 1

        # temperature decreasing linearly
        t_frac = (now - start) / (budget_s * 0.9)
        T = max(1e-5, 0.02 * (1.0 - t_frac))

        proposal = _propose_move(cur_h, rng, max_step)
        if proposal is None:
            continue
        i, j, delta = proposal

        # apply move
        cur_h[i] += delta
        cur_h[j] -= delta

        # enforce box constraints (they are already respected by construction)
        # but because of floating point we clip and re‑project the sum
        cur_h = _project_onto_box_and_sum(cur_h, target_sum)

        new_c5 = _c5_value(cur_h)

        # accept if improvement or with Metropolis probability
        if new_c5 <= cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert move
            cur_h[i] -= delta
            cur_h[j] += delta
            # keep feasible (tiny drift may appear)
            cur_h = _project_onto_box_and_sum(cur_h, target_sum)

        # occasional restart to avoid stagnation
        if it % 2000 == 0:
            if rng.random() < 0.2:
                # start from a fresh random feasible point
                rand_vals = rng.random(N)
                rand_vals = _project_onto_box_and_sum(rand_vals, target_sum)
                cur_h = rand_vals
                cur_c5 = _c5_value(cur_h)

    # final safety: ensure the returned vector satisfies the constraints exactly
    best_h = _project_onto_box_and_sum(best_h, target_sum)
    final_c5 = _c5_value(best_h)

    return best_h, final_c5, N
