def optimize_h(h0, max_iter, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    n = len(h)
    dx = 2.0/n
    start = time.time()
    iteration = 0
    # define step size schedule: start with max_delta = 0.2, reduce gradually.
    max_delta = 0.5
    while time.time() - start < time_limit and iteration < max_iter:
        # pick random i, j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # determine max amount we can move from j to i
        # ensure h_i <=1 , h_j >=0
        avail_i = 1.0 - h[i]
        avail_j = h[j]
        max_move = min(avail_i, avail_j, max_delta)
        if max_move <= 0:
            continue
        delta = np.random.rand() * max_move
        h[i] += delta
        h[j] -= delta
        val = evaluate(h)
        if val < best_val - 1e-9:
            best_val = val
            best_h = h.copy()
        else:
            # Reject
            h[i] -= delta
            h[j] += delta
        # Optionally occasionally reset to best_h to avoid drift
        if np.random.rand() < 0.001:
            h = best_h.copy()
        iteration += 1
        # decrease max_delta slowly
        if iteration % 1000 == 0:
            max_delta *= 0.9
    return best_h, best_val
```

import numpy as np
import time

# ------------------------------------------------------------
# Helper: evaluation of the C5 bound for a given step function
def evaluate_h(h):
    # use the provided evaluator if available, otherwise compute directly
    if 'evaluate_erdos_solution' in globals():
        return evaluate_erdos_solution(h)
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

# ------------------------------------------------------------
# Projection onto the feasible set:
#   0 <= x_i <= 1   and   sum_i x_i = S
def project_box_sum(y, S):
    # we need λ such that sum_i clip(y_i - λ, 0, 1) = S
    lo, hi = -1e6, 1e6
    for _ in range(30):  # enough for double precision tolerance
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        total = x.sum()
        if total > S:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(y - lam, 0.0, 1.0)

# ------------------------------------------------------------
# Core optimizer for a fixed number of points n.
def optimize_h_n(n, time_limit, init_h=None):
    """
    Projected gradient descent with occasional random resets.
    Returns (best_h, best_val) found within time_limit seconds.
    """
    S_target = n / 2.0                     # required sum of h_i
    best_h = None
    best_val = np.inf

    # initialise h
    if init_h is not None:
        h = np.array(init_h, dtype=float)
        if len(h) != n:
            # linear interpolation to required resolution
            x_old = np.linspace(0, 2, len(h), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, h)
    else:
        # random feasible start
        y = np.random.rand(n)
        h = project_box_sum(y, S_target)

    # ensure feasibility
    h = project_box_sum(h, S_target)
    best_h = h.copy()
    best_val = evaluate_h(h)

    start = time.time()
    lr = 0.5            # learning rate
    step = 0

    while time.time() - start < time_limit:
        # ---- compute shift with maximal overlap (worst shift) ----
        conv = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(conv)
        shift = k_max - (n - 1)   # integer shift index (can be negative)

        # ---- gradient of the overlap for this shift ----
        g = np.ones_like(h)       # derivative of the sum_i h_i term
        if shift >= 0:
            g[:n - shift] -= h[shift:]
            g[shift:]       -= h[:n - shift]
        else:
            d = -shift
            g[d:]          -= h[:n - d]
            g[:n - d]      -= h[d:]

        # ---- gradient step + projection ----
        h_new = h - lr * g
        h_new = project_box_sum(h_new, S_target)

        # ---- evaluate ----
        val = evaluate_h(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()

        # accept the move (projected gradient descent)
        h = h_new
        step += 1

        # ---- schedule adjustments ----
        if step % 2000 == 0:
            # occasional reset to explore new regions
            if np.random.rand() < 0.5:
                y = np.random.rand(n)
                h = project_box_sum(y, S_target)
            else:
                # perturb current best
                noise = (np.random.rand(n) - 0.5) * 0.1
                h = np.clip(best_h + noise, 0.0, 1.0)
                h = project_box_sum(h, S_target)

        if step % 5000 == 0:
            lr *= 0.9               # gradual decay of step size

        if best_val < 0.3808:       # early stop if target reached
            break

    return best_h, best_val
