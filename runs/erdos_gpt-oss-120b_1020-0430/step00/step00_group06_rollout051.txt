def compute_correlation(h, dx):
    g = np.correlate(h, 1 - h, mode='full')
    return g * dx

def project_onto_capped_simplex(v, target, lower=0.0, upper=1.0):
    # binary search for λ
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2.0, lower, upper)
    # due to numeric errors, adjust to exact sum by scaling difference small?
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff among non-boundary entries
        mask = (x > lower + 1e-12) & (x < upper - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, lower, upper)
        else:
            # fallback: add diff equally and re-project
            x += diff / len(x)
            x = np.clip(x, lower, upper)
    return x

import time
import numpy as np

def _project_capped_simplex(v, target, lo=0.0, hi=1.0, max_iter=60):
    """Project v onto the set {x | lo <= x_i <= hi, sum x_i = target}."""
    # Bisection on the Lagrange multiplier λ:
    lam_low = -np.max(v) - 1.0
    lam_high = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    x = np.clip(v - (lam_low + lam_high) * 0.5, lo, hi)
    # tiny correction to enforce the exact sum (numerical safety)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > lo + 1e-12) & (x < hi - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            x[free] += diff / cnt
            x = np.clip(x, lo, hi)
    return x

def _max_overlap(h, dx):
    """Return max_k ∫ h(x)(1−h(x+k))dx using discrete correlation."""
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)

        n = 120                     # any even size < 1000 works well
        target_sum = n / 2.0
        h = rng.random(n)
        h = _project_capped_simplex(h, target_sum)

    dx = 2.0 / n
    best_h = h.copy()
    best_C = _max_overlap(h, dx)

    # ------------------------------------------------------------------
    # optimisation parameters
    step = 0.5                     # initial step size
    lam = 20.0                     # start smoothing parameter
    lam_inc = (200.0 - lam) / 50000.0   # slow increase
    iter_count = 0

    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1

        # correlation & smoothed max weights
        g = np.correlate(h, 1 - h, mode="full") * dx          # length 2n-1
        cur_max = g.max()
        if cur_max < best_C - 1e-12:
            best_C = cur_max
            best_h = h.copy()
            # early exit if very good
            if best_C <= 0.3808:
                break

        # smooth max via log‑sum‑exp
        g_max = g.max()
        exp_vals = np.exp(lam * (g - g_max))
        w = exp_vals / exp_vals.sum()          # length 2n‑1, sum=1

        # convolution terms needed for gradient
        conv_full = np.convolve(h, w, mode="full")          # h * w
        conv_full_rev = np.convolve(h, w[::-1], mode="full")  # h * w_rev
        start = n - 1
        conv_neg = conv_full[start:start + n]              # ∑ w_k h[i‑k]
        conv_pos = conv_full_rev[start:start + n]          # ∑ w_k h[i+k]

        grad = dx * (1.0 - conv_pos - conv_neg)            # sub‑gradient of smoothed max

        # tentative update
        h_new = h - step * grad
        h_new = _project_capped_simplex(h_new, target_sum)

        new_max = _max_overlap(h_new, dx)
        if new_max <= cur_max:
            h = h_new
            step = min(step * 1.02, 2.0)   # slowly increase step when successful
        else:
            step *= 0.5                    # back‑track on failure
            # occasional random perturbation to escape flats
            if iter_count % 250 == 0:
                noise = rng.normal(scale=0.03, size=n)
                h_noise = np.clip(h + noise, 0.0, 1.0)
                h_noise = _project_capped_simplex(h_noise, target_sum)
                if _max_overlap(h_noise, dx) < cur_max:
                    h = h_noise

        # increase smoothing parameter gradually
        if lam < 200.0:
            lam += lam_inc

    # final check (in case the loop ends without update)
    final_C = _max_overlap(best_h, dx)
    return best_h, float(final_C), n
