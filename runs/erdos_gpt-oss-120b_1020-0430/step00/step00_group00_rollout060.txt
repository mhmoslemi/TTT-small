def random_transfer(h):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # decide direction:
    if np.random.rand() < 0.5:
        # i receives from j
        max_delta = min(1 - h[i], h[j])
        if max_delta <= 0: return None
        delta = np.random.rand() * max_delta
        h_new = h.copy()

        h_new[j] -= delta
    else:
        # j receives from i
        max_delta = min(1 - h[j], h[i])

        h_new[i] -= delta

    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()

    start = time.time()
    cur_val = best_val
    while time.time() - start < time_limit:
        # compute correlation and worst shift
        corr_full = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr_full)
        cur_val = corr_full[idx_max] * dx
        # update best
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        shift = idx_max - (n - 1)
        # store old copy
        h_old = h.copy()
        # propose move
        moved = False
        if shift != 0:
            if shift > 0:
                L = n - shift
                diff = h[:L] - h[shift:shift+L]
                i = np.argmax(diff)
                dmax = diff[i]
                if dmax > 1e-12:
                    src = i
                    tgt = i + shift
                    max_delta = min(h[src], 1.0 - h[tgt])
                    # step size limit
                    delta = min(dmax / 2.0, max_delta, 0.02)

assistantfinal**Solution Overview**  
The program receives (or creates) a feasible step‑function `h` on the grid  
`[0,2]`.  It then performs a time‑bounded stochastic local search that repeatedly

* evaluates the overlap value `C₅ = max_k ∑ h[i]·(1‑h[i+k])·dx`,
* identifies the shift `k` that attains the maximum,
* tries to decrease that shift by moving a small amount of mass from a point with
  a large value to the corresponding shifted point with a small value,
* accepts the move if it does not increase the current worst overlap (a
  simulated‑annealing rule is used to escape plateaus).

The algorithm works directly on the feasible set (bounds `[0,1]` and the exact
integral constraint) and finishes as soon as the time budget expires or a
solution with `C₅ ≤ 0.3808` is found.

import numpy as np
import time
import math
import random

def _c5_value(h: np.ndarray) -> float:
    """Return the overlap constant C₅ for the current vector."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)


def _make_feasible(vec: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce the exact integral constraint."""
    n = vec.shape[0]
    target = n / 2.0
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    if abs(cur - target) > 1e-12:
        # simple redistribution of the tiny error
        diff = target - cur
        # pick a random index that can absorb the difference without leaving bounds
        for _ in range(10):
            idx = np.random.randint(n)
            val = np.clip(vec[idx] + diff, 0.0, 1.0)
            if abs(val - vec[idx]) > 1e-14:
                vec[idx] = val
                break
    return vec


def _hill_climb(start_h: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """Local stochastic optimisation for a fixed time budget."""
    h = start_h.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = _c5_value(h)

    start = time.time()
    no_improve = 0                     # counter for stagnation detection

    while time.time() - start < time_limit:
        # current correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_val = float(corr[idx] * (2.0 / n))
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= 0.3808:
                break

        shift = idx - (n - 1)          # lag belonging to the maximal correlation

        old_h = h.copy()
        moved = False

        # Try a directed move that reduces the worst shift
        if shift != 0:
            if shift > 0:
                L = n - shift
                diff = h[:L] - h[shift:shift + L]
                i = int(np.argmax(diff))
                if diff[i] > 1e-12:
                    src, tgt = i, i + shift
                    max_delta = min(h[src], 1.0 - h[tgt])
                    delta = min(diff[i] / 2.0, max_delta, 0.02)
                    if delta > 1e-12:
                        h[src] -= delta
                        h[tgt] += delta
                        moved = True
            else:  # shift < 0
                s = -shift
                L = n - s
                diff = h[s:s + L] - h[:L]
                i = int(np.argmax(diff))
                if diff[i] > 1e-12:
                    src, tgt = i + s, i
                    max_delta = min(h[src], 1.0 - h[tgt])
                    delta = min(diff[i] / 2.0, max_delta, 0.02)
                    if delta > 1e-12:
                        h[src] -= delta
                        h[tgt] += delta
                        moved = True

        # If a directed move was impossible, perform a small random transfer
        if not moved:
            i, j = np.random.choice(n, 2, replace=False)
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta > 1e-12:
                delta = np.random.rand() * max_delta * 0.05
                h[i] -= delta
                h[j] += delta

        # Evaluate the new configuration
        new_val = _c5_value(h)

        # Acceptance rule (monotone + occasional uphill moves)
        if new_val <= cur_val + 1e-12:
            cur_val = new_val
            no_improve = 0
        else:
            # simulated‑annealing style probability
            elapsed = time.time() - start
            fraction = elapsed / time_limit
            temperature = max(1e-3, 0.01 * (1.0 - fraction))
            prob = math.exp(-(new_val - cur_val) / temperature)
            if np.random.rand() < prob:
                cur_val = new_val
                no_improve = 0
            else:
                h = old_h                     # revert

        # Stagnation handling – jitter the best solution occasionally
        if not moved:
            no_improve += 1
        if no_improve > 2000:
            # small random perturbation around the best found vector
            h = best_h.copy()
            perturb = (np.random.rand(n) - 0.5) * 0.02
            h += perturb
            h = _make_feasible(h)
            cur_val = _c5_value(h)
            no_improve = 0

    return best_h, best_val

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
