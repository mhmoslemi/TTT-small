def random_move(h, max_step=0.1):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # if both at extremes, maybe swap direction.
    if h[i] > 0 and h[j] < 1:
        src = i
        dst = j
    elif h[j] > 0 and h[i] < 1:
        src = j
        dst = i
    else:
        # no feasible transfer
        return None, None, None
    delta_max = min(h[src], 1 - h[dst])
    if delta_max <= 0:
        return None, None, None
    delta = np.random.rand() * min(delta_max, max_step)
    return src, dst, delta

def gradient_for_lag(h, k):
    n = h.shape[0]
    g = np.ones_like(h)
    idx1 = np.arange(n) - k
    mask = (idx1 >= 0) & (idx1 < n)
    g[mask] -= h[idx1[mask]]
    idx2 = np.arange(n) + k
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] -= h[idx2[mask2]]
    return g

    lo_lam = -1e6

        lam = (lo_lam + hi_lam) / 2.0
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if s > target_sum:
            lo_lam = lam
        else:
            hi_lam = lam
    x = np.clip(v - lam, lo, hi)

    diff = target_sum - x.sum()

import time
import numpy as np
import random

def _project_onto_simplex_box(v, target_sum, lo=0.0, hi=1.0, iters=60):
    """Euclidean projection of v onto {x : lo<=x<=hi, sum(x)=target_sum}."""
    # binary search for lambda such that sum(clip(v - lam, lo, hi)) = target_sum
    lam_low, lam_high = -1e6, 1e6
    for _ in range(iters):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if s > target_sum:
            lam_low = lam
        else:
            lam_high = lam
    x = np.clip(v - lam, lo, hi)
    # tiny correction to hit the sum more accurately
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        free = (x > lo + 1e-12) & (x < hi - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            np.clip(x, lo, hi, out=x)
    return x


def _compute_c5(h, dx):
    """Return C5 for a given discretised step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _random_move(h, max_step):
    """Transfer a small amount of mass from one index to another."""
    n = h.shape[0]
    for _ in range(15):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
            continue
        delta = min(h[i], 1.0 - h[j], max_step) * np.random.rand()
        if delta <= 0:
            continue
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        return new_h
    return None


def _gradient_of_lag(h, lag):
    """Gradient of the overlap I_k for a given lag."""
    n = h.shape[0]
    g = np.ones_like(h)
    idx = np.arange(n)
    left = idx - lag
    right = idx + lag
    mask = (left >= 0) & (left < n)
    g[mask] -= h[left[mask]]
    mask = (right >= 0) & (right < n)
    g[mask] -= h[right[mask]]
    return g


def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # ----- parameters -------------------------------------------------
    n = 200                     # number of discretisation points (< 1000)
    dx = 2.0 / n
    target_sum = n / 2.0        # ∑ h_i = n/2  (because ∫h = 1)
    max_iter = 10**7           # hard cap; the time guard will stop earlier
    # SA / step schedules
    temp = 0.02
    temp_decay = 0.9999
    min_temp = 1e-6
    step = 0.2
    step_decay = 0.9999

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        old = np.asarray(initial_h_values, dtype=float)
        n_old = old.shape[0]
        # linear interpolation onto the finer grid
        x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
        x_new = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.interp(x_new, x_old, old, left=0.0, right=0.0)
        h = _project_onto_simplex_box(h, target_sum, 0.0, 1.0)
    else:
        # random feasible start
        y = np.random.rand(n)
        h = _project_onto_simplex_box(y, target_sum, 0.0, 1.0)

    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)
    cur_h = h.copy()
    cur_score = best_score

    start_time = time.time()
    end_time = start_time + budget_s - 0.2   # safety margin

    it = 0
    stagnation = 0

    while time.time() < end_time and it < max_iter:
        it += 1

        # ----- decide move type (gradient step every few iters) -----
        if it % 7 == 0:
            # gradient step against the currently worst lag
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            lag = int(np.argmax(corr) - (n - 1))
            g = _gradient_of_lag(cur_h, lag)
            new_h = cur_h - step * g
            new_h = _project_onto_simplex_box(new_h, target_sum, 0.0, 1.0)

            if new_h is None:

        # ----- acceptance (simulated annealing) --------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if temp > min_temp and np.random.rand() < np.exp(-(new_score - cur_score) / temp):

        if accept:

            if new_score < best_score - 1e-12:

                stagnation = 0
            else:
                stagnation += 1
        else:

        # ----- schedule updates ------------------------------------
        temp = max(min_temp, temp * temp_decay)

        # ----- occasional restart if stuck -------------------------
        if stagnation > 8000:

    return best_h.tolist(), float(best_score), n
