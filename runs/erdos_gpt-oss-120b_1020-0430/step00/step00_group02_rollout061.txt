def project_onto_sum_bounds(v, target):
    # v: array
    # Want w such that sum(w)=target, 0<=w<=1, minimal L2 distance.
    # Solve w_i = min(1, max(0, v_i + λ)), find λ where sum(w_i) = target.
    # Use binary search for λ between low=-max(v) and high=1 - min(v).
    # Actually to ensure monotonic increase, lower bound can be -max(v) - large; use -1 maybe.
    lo = -1.0
    hi = 1.0
    # Ensure f(lo) <= target and f(hi) >= target.
    # Compute f(λ)
    def f(lmbda):
        w = np.clip(v + lmbda, 0, 1)
        return w.sum()
    # Expand bounds if needed
    while f(lo) > target:
        lo -= 1.0
    while f(hi) < target:
        hi += 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = f(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) / 2.0
    return np.clip(v + λ, 0, 1)

def slsqp_optimize(h0, max_iter=100):
    n = h0.size
    target = n / 2
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    def obj(x):
        # compute max correlation
        conv = np.convolve(x, (1 - x)[::-1], mode='full')
        return np.max(conv) * (2.0 / n)
    # options: set maxiter maybe.
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    # ensure feasible
    x_opt = np.clip(res.x, 0, 1)
    # adjust sum using projection
    x_opt = project_onto_sum_bounds(x_opt, target)
    return x_opt, obj(x_opt)

def build_sym_h(u):
    return np.concatenate([u, 1 - u[::-1]])

def slsqp_optimize_full(h0):
    n = len(h0)
    target = n / 2
    bounds = [(0,1)]*n
    constraints = [{'type':'eq','fun': lambda x: np.sum(x) - target}]
    def obj(x):
        conv = np.convolve(x, (1 - x)[::-1], mode='full')
        return np.max(conv) * (2.0 / n)
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter':200,'ftol':1e-9,'disp':False})
    h_opt = np.clip(res.x, 0, 1)
    # project onto sum with bounds as final adjust (should be close)
    h_opt = project_onto_sum_bounds(h_opt, target)
    return h_opt, obj(h_opt)

def sa_optimize_sym(u, cur_c5, time_remaining, max_iter=20000):
    start = time.time()
    best_u = u.copy()
    best_c5 = cur_c5
    cur_u = u.copy()
    cur_c5_val = cur_c5
    m = len(u)
    # Precompute some? But we'll compute each candidate's h using build_sym_h.
    def eval_u(u_vec):
        h = build_sym_h(u_vec)
        conv = np.convolve(h, (1 - h)[::-1], mode='full')
        return np.max(conv) * (2.0 / (2*m))
    # Determine temperature schedule param
    for it in range(max_iter):
        if time.time() - start > time_remaining:
            break
        # temperature schedule (linear decreasing)
        t = max(1e-6, (max_iter - it) / max_iter)
        # random indexes
        i = np.random.randint(m)
        j = np.random.randint(m)
        if i == j:
            continue
        # ensure possible move: u[i] > 0, u[j] < 1
        if cur_u[i] <= 1e-12 or cur_u[j] >= 1 - 1e-12:
            continue
        max_delta = min(0.1, cur_u[i], 1 - cur_u[j])
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        new_u = cur_u.copy()
        new_u[i] -= delta
        new_u[j] += delta
        new_c5 = eval_u(new_u)
        if new_c5 < cur_c5_val or np.random.rand() < np.exp(-(new_c5 - cur_c5_val) / (0.01 + t)):
            cur_u = new_u
            cur_c5_val = new_c5
            if new_c5 < best_c5:
                best_u = new_u.copy()
                best_c5 = new_c5
    return best_u, best_c5

def build_h_7(a,b,w1,w2,n):
    w3 = 1 - 2*a*w1 - 2*b*w2
    w0 = 1 - w1 - w2 - w3/2
    if min(w3,w0) < 0:
        return None
    lengths = [w0, w1, w2, w3, w2, w1, w0]
    values = [0, a, b, 1, b, a, 0]
    # build as earlier with counts.
    raw_counts = np.array(lengths) * n / 2.0
    counts = np.floor(raw_counts).astype(int)
    remainder = n - counts.sum()
    fractions = raw_counts - counts
    # allocate leftover to highest fractions
    for idx in np.argsort(-fractions)[:remainder]:
        counts[idx] += 1
    # Build vector
    h_parts = [np.full(counts[i], values[i]) for i in range(len(values))]
    h = np.concatenate(h_parts)
    # adjust if length mismatch
    if len(h) != n:
        # adjust near center
        diff = len(h) - n
        if diff > 0:
            # cut from central region (value 1)
            idx_center = 3  # index of w3
            # reduce counts[idx_center] by diff
            if counts[idx_center] >= diff:
                counts[idx_center] -= diff
                # rebuild parts
                h_parts = [np.full(counts[i], values[i]) for i in range(len(values))]
                h = np.concatenate(h_parts)
            else:
                h = h[:n]  # fallback
        else:
            # too few, pad zeros at ends
            h = np.concatenate([h, np.zeros(-diff)])
    # ensure sum constraint exactly
    target = n/2
    h = project_onto_sum_bounds(h, target)
    return h

import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def _evaluate(h):
    """Return C₅ for a discretised step function h."""
    conv = np.convolve(h, (1.0 - h)[::-1], mode="full")
    return np.max(conv) * (2.0 / len(h))


def _project_onto_sum(v, target):
    """Project v onto the box [0,1] with the linear equality sum(v)=target."""
    # solve for lambda: sum(clamp(v+lam,0,1)) = target
    lo, hi = -1.0, 1.0

    def f(lam):
        return np.clip(v + lam, 0.0, 1.0).sum()

    while f(lo) > target:
        lo -= 1.0
    while f(hi) < target:
        hi += 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if f(mid) < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def _build_h_7(a, b, w1, w2, n):
    """
    Build a symmetric 7‑step function on [0,2] with parameters:
    a ≤ b, widths w1 (for level a) and w2 (for level b).
    The pattern is 0 – a – b – 1 – b – a – 0.
    """
    w3 = 1.0 - 2.0 * a * w1 - 2.0 * b * w2          # width of the top plateau
    if w3 <= 0.0:
        return None
    w0 = 1.0 - w1 - w2 - w3 / 2.0                  # outer zero width
    if w0 < 0.0:
        return None

    lengths = np.array([w0, w1, w2, w3, w2, w1, w0])
    values = np.array([0.0, a, b, 1.0, b, a, 0.0])

    raw = lengths * n / 2.0
    cnt = np.floor(raw).astype(int)
    rem = n - cnt.sum()
    if rem > 0:
        frac = raw - cnt
        for idx in np.argsort(-frac)[:rem]:
            cnt[idx] += 1

    parts = [np.full(c, v, dtype=float) for c, v in zip(cnt, values)]
    h = np.concatenate(parts)
    if h.size > n:
        # trim excess from the central plateau (value 1)
        excess = h.size - n
        mid = 3  # index of the plateau
        if cnt[mid] >= excess:
            cnt[mid] -= excess
            parts[mid] = np.full(cnt[mid], 1.0)
            h = np.concatenate(parts)
        else:
            h = h[:n]
    elif h.size < n:
        h = np.concatenate([h, np.zeros(n - h.size)])

    h = _project_onto_sum(h, n / 2.0)
    return h

    """
    np.random.seed(seed)
    start = time.time()

            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_c5 = _evaluate(cand)

                cur_h, cur_c5 = cand, cand_c5
                if cand_c5 < best_local_c5:

    # ------------------------------------------------------------------
    best_h = _project_onto_sum(np.clip(best_h, 0.0, 1.0), n_points / 2.0)
