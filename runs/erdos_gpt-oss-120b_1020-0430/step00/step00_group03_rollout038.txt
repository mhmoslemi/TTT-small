def hill_climb(v):
   best_score = get_overlap(v)
   best_v = v.copy()
   for iteration in range(max_iter):
       # find worst lag
       corr = np.correlate(v, 1-v, mode='full')
       lag = np.argmax(corr) - (len(v)-1)  # shift
       max_corr = corr[lag + len(v)-1]
       # find set of positions i where v[i]=1 and v[i+lag]==0 (if i+lag in range)
       if lag >= 0:
           indices_i = np.where((v[:n-lag] == 1) & (v[lag:] == 0))[0]
           # positions i where v[i]==0 and v[i+lag]==1 (possible swap target)
           indices_j = np.where((v[:n-lag] == 0) & (v[lag:] == 1))[0]
       else:
           # lag negative -> shift left, treat with sign
           lag_abs = -lag
           indices_i = np.where((v[lag_abs:] == 1) & (v[:n-lag_abs] == 0))[0] # i+lag = i - lag_abs
           indices_j = np.where((v[lag_abs:] == 0) & (v[:n-lag_abs] == 1))[0]
           # shift indices appropriately
           # indices_i correspond to i in range(lag_abs,n)
           # to make list of i positions (original index)
           indices_i = indices_i + lag_abs
           indices_j = indices_j + lag_abs

       # If there are any pairs (i in indices_i, j in indices_j) we can swap 1 at i with 0 at j to reduce mismatch count.
       # Best improvement maybe pick any pair.
       if len(indices_i) > 0 and len(indices_j) > 0:
           i = np.random.choice(indices_i)
           j = np.random.choice(indices_j)
           # swap
           v_new = v.copy()
           v_new[i] = 0
           v_new[j] = 1
           # compute new max
           new_score = get_overlap(v_new)
           if new_score <= best_score:
               v = v_new
               best_score = new_score
               best_v = v_new
               if new_score < best_score: # Actually we already assign
                  pass
               continue # go next iteration
       # If no direct improvement, try random swap
       i1 = np.random.randint(n)
       i2 = np.random.randint(n)
       if v[i1] == v[i2]:
           continue
       # swap
       v_new = v.copy()
       v_new[i1], v_new[i2] = v_new[i2], v_new[i1]
       new_score = get_overlap(v_new)
       if new_score < best_score:
            v = v_new
            best_score = new_score
            best_v = v_new
   return best_v, best_score

    corr = np.correlate(h, 1 - h, mode='full')

def repair(h):
    # ensure within [0,1] and sum = target (n/2)
    n = len(h)
    target = n/2
    # Clip
    np.clip(h, 0, 1, out=h)
    s = h.sum()
    # If sum not equal target, adjust using simple scaling while preserving bounds
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Distribute diff to entries not at bound
    # Compute indices where h not at 0 or 1
    free = np.where((h > 0) & (h < 1))[0]
    if free.size > 0:
        # add diff equally among free entries within bounds, but iterate if saturates
        while abs(diff) > 1e-12 and free.size > 0:
            adjust = diff / free.size
            # Restrict adjust so that each stays within [0,1]
            # compute new values
            new_vals = h[free] + adjust
            # find those that would exceed bounds
            over_low = new_vals < 0
            over_high = new_vals > 1
            if np.any(over_low) or np.any(over_high):
                # Set those to bound and remove from free
                for idx in free[over_low]:
                    diff += -h[idx]  # subtract current value, set to 0
                    h[idx] = 0.0
                for idx in free[over_high]:
                    diff += (1 - h[idx])
                    h[idx] = 1.0
                # recompute free
                free = np.where((h > 0) & (h < 1))[0]
                continue
            else:
                h[free] = new_vals
                diff = target - h.sum()
                break
    # If after free adjustment still not meet, then adjust first element forcibly clip
    if abs(h.sum() - target) > 1e-9:
        # add diff to first element within bounds
        i = 0
        # adjust by available capacity
        cap_up = 1 - h[i]
        cap_down = -h[i]
        adj = np.clip(diff, cap_down, cap_up)
        h[i] += adj
    # final clip
    np.clip(h, 0, 1, out=h)
    return h

def _max_overlap(h, dx):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by correlation."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _repair(h):
    """Project h onto [0,1] and enforce the integral constraint ∑h = n/2."""
    n = h.size
    target = n / 2.0

    # simple redistribution to hit the exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-12:

        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
