import numpy as np
import time, math

def _compute_c5(h, dx):
    # max overlap * dx
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)

def _adjust_sum(arr, target, eps=1e-12):
    """Force sum(arr) == target by adding/subtracting small amounts respecting bounds."""
    cur = arr.sum()
    diff = target - cur
    if abs(diff) < eps:
        return arr
    if diff > 0:
        # need to increase some entries
        caps = 1.0 - arr
        order = np.argsort(-caps)  # largest capacity first
        for i in order:
            if diff <= eps:
                break
            add = min(diff, caps[i])
            if add > eps:
                arr[i] += add
                diff -= add
    else:
        # need to decrease some entries
        caps = arr
        order = np.argsort(-caps)
        for i in order:
            if diff >= -eps:
                break
            sub = min(-diff, caps[i])
            if sub > eps:
                arr[i] -= sub
                diff += sub
    return arr

def _random_interval_vector(N, rng, m_intervals=5):
    """Create a binary vector of length N (dx = 2/N) representing the union of m intervals
    of total measure 1 (i.e. exactly N/2 ones after discretisation)."""
    dx = 2.0 / N
    target_ones = N // 2

    # sample interval lengths (sum to 1) and gaps (sum to 1)
    lengths = rng.exponential(1.0, size=m_intervals)
    lengths = lengths / lengths.sum()
    gaps = rng.exponential(1.0, size=m_intervals + 1)
    gaps = gaps / gaps.sum()

    # build intervals in continuous domain
    intervals = []
    pos = 0.0
    for i in range(m_intervals):
        pos += gaps[i]                     # gap before this interval
        start = pos
        end = start + lengths[i]
        intervals.append((start, end))
        pos = end                          # move to end of interval
    # pos + gaps[-1] should be exactly 2.0 (within floating error)

    # discretise to binary vector
    h = np.zeros(N, dtype=float)
    for (a, b) in intervals:
        i0 = int(np.floor(a / dx))
        i1 = int(np.ceil(b / dx))
        i0 = max(0, min(N, i0))
        i1 = max(0, min(N, i1))
        h[i0:i1] = 1.0

    # enforce exact number of ones
    h = _adjust_sum(h, target_ones)
    # after adjustment values may be fractional; clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    # final rounding to keep the integral exact (tiny rounding errors tolerated)
    cur = h.sum()
    if abs(cur - target_ones) > 1e-9:
        # small correction by adding/subtracting 1 up to a few positions
        deficit = target_ones - cur
        if deficit > 0:
            zeros = np.where(h < 1.0 - 1e-12)[0]
            rng.shuffle(zeros)
            for i in zeros:
                add = min(deficit, 1.0 - h[i])
                h[i] += add
                deficit -= add
                if deficit <= 1e-12:
                    break
        else:
            ones = np.where(h > 1e-12)[0]
            rng.shuffle(ones)
            for i in ones:
                sub = min(-deficit, h[i])
                h[i] -= sub
                deficit += sub
                if deficit >= -1e-12:
                    break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.array(init, dtype=float).copy()
        N = h.shape[0]
        # enforce constraints quickly
        np.clip(h, 0.0, 1.0, out=h)
        target_sum = N / 2.0
        h = _adjust_sum(h, target_sum)
    else:
        N = 200                     # a modest size, divisible by 4
        h = _random_interval_vector(N, rng, m_intervals=5)

    dx = 2.0 / N
    target_sum = N / 2.0
    best_h = h.copy()
    best_val = _compute_c5(best_h, dx)

    # ----- random interval search (exploration) -----------------------
    exploration_time = budget_s * 0.45
    while time.time() - start_time < exploration_time:
        # vary number of intervals a bit
        m = rng.integers(3, 7)          # 3 to 6 intervals
        cand = _random_interval_vector(N, rng, m_intervals=m)
        val = _compute_c5(cand, dx)
        if val < best_val:
            best_val = val
            best_h = cand.copy()
        # early exit if we already beat target comfortably
        if best_val < 0.3808:
            break

    # ----- local optimisation (simulated annealing on binary swaps) --
    # ensure binary start for quick swaps
    # Convert best_h to strict 0/1 while preserving sum
    binary_h = (best_h > 0.5).astype(float)
    # adjust to exact count if rounding changed it
    binary_h = _adjust_sum(binary_h, target_sum)
    np.clip(binary_h, 0.0, 1.0, out=binary_h)
    cur_h = binary_h.copy()
    cur_val = _compute_c5(cur_h, dx)

    # SA parameters
    T0 = 0.02
    T = T0
    decay = 0.9995
    min_T = 1e-6
    max_iters = 10_000_000  # safety upper bound

    # Pre‑compute indices of ones/zeros for fast sampling
    ones_idx = np.flatnonzero(cur_h > 0.5)
    zeros_idx = np.flatnonzero(cur_h < 0.5)

    it = 0
    while time.time() - start_time < budget_s * 0.95 and it < max_iters:
        it += 1
        # pick a random 1 and a random 0 to swap
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # propose swap
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_val = _compute_c5(new_h, dx)

        # accept according to Metropolis rule
        if new_val < cur_val - 1e-12 or rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            # maintain index lists
            ones_idx[ones_idx == i] = j
            zeros_idx[zeros_idx == j] = i

            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                # refresh index arrays (they are already correct)

        # temperature schedule
        T = max(min_T, T * decay)

        # occasional random restart if stuck
        if it % 50000 == 0:
            # re‑shuffle a small fraction of the bits to escape plateaus
            flip_frac = 0.02
            n_flip = int(flip_frac * N)
            if n_flip > 0:
                idx_one = rng.choice(np.flatnonzero(cur_h > 0.5), size=n_flip, replace=False)
                idx_zero = rng.choice(np.flatnonzero(cur_h < 0.5), size=n_flip, replace=False)
                cur_h[idx_one] = 0.0
                cur_h[idx_zero] = 1.0
                cur_val = _compute_c5(cur_h, dx)
                ones_idx = np.flatnonzero(cur_h > 0.5)
                zeros_idx = np.flatnonzero(cur_h < 0.5)

    # final sanity enforcement
    best_h = _adjust_sum(best_h, target_sum)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    final_val = _compute_c5(best_h, dx)

    return best_h.tolist(), final_val, N
