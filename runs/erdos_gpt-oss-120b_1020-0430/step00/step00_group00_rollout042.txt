import numpy as np, time

def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c.max()

def generate_three_level(n, n_end, a):
    # n_end integer number of points at each end (0 <= n_end <= n//2)
    # compute b from constraint
    total = n / 2  # sum needed
    if 2 * n_end == n:
        # entire array ends -> all a must be total/n points. But as edges occupy whole, then b not defined. It's just constant a.
        b = a
        h = np.full(n, a)
        # adjust to sum: scale? We'll skip this case.
    else:
        b = (total - 2 * n_end * a) / (n - 2 * n_end)
        if b < 0 or b > 1:
            return None
        h = np.empty(n)
        h[:n_end] = a
        h[-n_end:] = a
        h[n_end:n-n_end] = b
    # enforce 0-1
    return h

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # full linear correlation (zero padding)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def worst_shift_info(h):
    """Find the shift (lag) that gives the current maximal correlation.
    Returns shift, array of i indices (where i and i+shift are in range),
    and the per‑i contribution hi*(1‑hj)."""
    n = h.shape[0]
    raw = np.correlate(h, 1.0 - h, mode="full")
    idx = int(raw.argmax())
    shift = idx - (n - 1)           # shift = k  (positive → forward)
    if shift >= 0:
        hi = h[: n - shift]
        hj = h[shift:]
        i_offset = 0
    else:
        k = -shift
        hi = h[k:]
        hj = h[: n - k]
        i_offset = k
    contrib = hi * (1.0 - hj)
    return shift, i_offset, contrib, raw[idx]


def transfer_mass(h, i, j, delta):
    """Move delta from h[i] to h[j] (preserving total mass)."""
    h[i] -= delta
    h[j] += delta
    # Clip to bounds – numerical safety
    h[i] = max(0.0, min(1.0, h[i]))
    h[j] = max(0.0, min(1.0, h[j]))
    return h


def random_start(n, rng):
    """Create a random feasible vector (sum = n/2, 0≤h≤1)."""
    # start from Dirichlet‑like draw, then project onto the simplex with bounds
    x = rng.random(n)
    x = x / x.sum() * (n / 2.0)      # enforce the sum
    # clip and re‑balance if needed
    over = x - 1.0
    while np.any(over > 1e-12):
        excess = over[over > 0].sum()
        mask = over > 0
        x[mask] = 1.0
        deficit = (n / 2.0) - x.sum()
        if deficit <= 0:
            break
        # distribute the needed deficit uniformly over non‑saturated entries
        free = (~mask)
        if free.sum() == 0:
            break
        add = deficit / free.sum()
        x[free] += add
        over = x - 1.0
    # final tiny correction to meet the exact sum
    diff = x.sum() - (n / 2.0)
    if abs(diff) > 1e-12:
        # shift the diff from a random non‑boundary index
        idxs = np.where((x > 0) & (x < 1))[0]
        if len(idxs):
            i = rng.choice(idxs)
            x[i] -= diff
            x[i] = max(0.0, min(1.0, x[i]))
    return x


def three_level_shape(n, n_end, a):
    """Return a three‑level step function: a on each end of length n_end,
    constant b in the middle chosen so that the integral equals 1."""
    if n_end * 2 >= n:
        return None
    total = n / 2.0
    b = (total - 2.0 * n_end * a) / (n - 2.0 * n_end)
    if b < 0.0 or b > 1.0:
        return None
    h = np.empty(n, dtype=float)
    h[:n_end] = a
    h[-n_end:] = a
    h[n_end:n - n_end] = b
    return h

            # shift == 0 – try to polarise values (move mass from a medium
            # entry to another medium entry)
            i_candidates = np.where((best_h > 1e-8) & (best_h < 1.0 - 1e-8))[0]

        if i_candidates.size == 0:
            # fall back to a tiny random perturbation
            delta = rng.uniform(-0.02, 0.02, size=n)
            h_try = best_h + delta
            h_try = np.clip(h_try, 0.0, 1.0)
            # re‑normalise the sum exactly
            diff = h_try.sum() - target_sum
            if abs(diff) > 1e-12:
                free = (h_try > 0) & (h_try < 1)
                if free.any():
                    h_try[free] -= diff / free.sum()
                    h_try = np.clip(h_try, 0.0, 1.0)
            c5_new = compute_c5(h_try)
            accept = c5_new < best_c5
        else:
            # pick a random index among the feasible ones
            i_local = rng.choice(i_candidates)
            if shift >= 0:
                i = i_off + i_local
                j = i + shift
            else:
                j = i_off + i_local
                i = j - shift   # because shift is negative
            # maximum admissible transfer
            max_move = min(best_h[i], 1.0 - best_h[j])
            if max_move <= 1e-12:
                continue
            delta = max_move * rng.random() * 0.5   # modest step
            h_try = best_h.copy()
            h_try = transfer_mass(h_try, i, j, delta)

            c5_new = compute_c5(h_try)
            # deterministic improvement test
            accept = c5_new < best_c5

            # occasional uphill acceptance (simulated annealing)
            if not accept:
                elapsed = time.time() - start_time
                temp = init_temp * (1.0 - elapsed / (budget_s * 0.95))
                if temp > 0 and math.exp(-(c5_new - best_c5) / temp) > rng.random():
                    accept = True

        if accept:
            best_h = h_try
            best_c5 = c5_new

        # periodic global perturbation to escape shallow basins
        if it % 5000 == 0:
            # small Gaussian jitter on all coordinates, then re‑project to the simplex
            jitter = rng.normal(scale=0.01, size=n)
            h_try = best_h + jitter
            h_try = np.clip(h_try, 0.0, 1.0)
            diff = h_try.sum() - target_sum
            if abs(diff) > 1e-12:
                free = (h_try > 0) & (h_try < 1)
                if free.any():
                    h_try[free] -= diff / free.sum()
                    h_try = np.clip(h_try, 0.0, 1.0)
            c5_new = compute_c5(h_try)
            if c5_new < best_c5:
                best_h = h_try
                best_c5 = c5_new

    # --------------------------------------------------------
    # Final projection onto the feasible set (tiny numeric fix)
    # --------------------------------------------------------
    diff = best_h.sum() - target_sum
    if abs(diff) > 1e-12:
        # redistribute the tiny residual greedily
        idxs = np.arange(n)
        rng.shuffle(idxs)
        for idx in idxs:
            if diff > 0:
                dec = min(diff, best_h[idx])
                best_h[idx] -= dec
                diff -= dec
            else:
                inc = min(-diff, 1.0 - best_h[idx])
                best_h[idx] += inc
                diff += inc
            if abs(diff) < 1e-12:
                break
        # final safeguard clamp
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    # --------------------------------------------------------
    # Return the solution
    # --------------------------------------------------------
    return best_h.tolist(), float(final_c5), n
