def gradient_move(h):
    n = len(h)
    # compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    s = idx_max - (n - 1)   # shift: can be negative
    # Compute gradient g
    g = np.zeros(n)
    # compute term1 for i: (1 - h[i + s]) where index valid
    if s >= 0:
        # i + s valid for i in 0..n-1-s
        i_range = np.arange(n - s)
        g[i_range] = (1 - h[i_range + s])
        # term2: -h[i - s]; i - s valid for i in s..n-1
        j_range = np.arange(s, n)
        g[j_range] -= h[j_range - s]
    else:
        # s negative: i + s valid for i >= -s
        offset = -s
        i_range = np.arange(offset, n)
        g[i_range] = (1 - h[i_range + s])   # i+s >=0
        # term2: -h[i - s] where i - s valid for i in 0..n-1+ s (since s negative)
        j_range = np.arange(0, n + s)   # because s negative => n+s < n
        g[j_range] -= h[j_range - s]
    # Now find source (positive gradient) and sink (negative)
    pos_idx = np.argmax(g)   # max g (positive)
    neg_idx = np.argmin(g)   # min g (negative)
    # If g[pos_idx] <= 0 or g[neg_idx] >= 0, gradient move not helpful.
    if g[pos_idx] <= 0 or g[neg_idx] >= 0:
        return None, None, None
    # Determine possible delta: moving mass from pos_idx (decrease) to neg_idx (increase)
    delta_max = min(h[pos_idx], 1 - h[neg_idx])
    if delta_max <= 0:
        return None, None, None
    # Choose delta maybe a fraction of delta_max (e.g., 0.5*delta_max)
    delta = min(delta_max, 0.05)  # step size limit

import time
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a step function given by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _gradient(h, shift):
    """
    Sub‑gradient of the overlap N_shift = sum h[i]*(1-h[i+shift]).
    Returns g_i = (1-h[i+shift]) - h[i-shift] (out‑of‑range terms are zero).
    """
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        # term (1-h[i+shift]) for i = 0 .. n-1-shift
        g[: n - shift] = 1.0 - h[shift:]
        # term -h[i-shift] for i = shift .. n-1
        g[shift:] -= h[: n - shift]
    else:
        off = -shift
        # term (1-h[i-shift]) for i = off .. n-1   (i+shift = i-off)
        g[off:] = 1.0 - h[: n - off]
        # term -h[i-shift] for i = 0 .. n-1-off   (i-shift = i+off)
        g[: n - off] -= h[off:]
    return g

def _enforce_sum(h):
    """Adjust a feasible vector so that sum(h) = n/2 exactly (within float tolerance)."""
    n = h.shape[0]
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # find an index that can absorb the tiny difference
    for i in range(n):
        if diff > 0 and h[i] < 1.0 - 1e-12:
            take = min(diff, 1.0 - h[i])
            h[i] += take
            diff -= take
        elif diff < 0 and h[i] > 1e-12:
            give = min(-diff, h[i])
            h[i] -= give
            diff += give
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)
