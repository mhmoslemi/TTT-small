def gradient_move(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    # If k==0, pick second largest shift perhaps.
    if k == 0:
        # choose another shift with large correlation
        sorted_idxs = np.argsort(-corr)  # descending
        for idx in sorted_idxs:
            kk = idx - (n - 1)
            if kk != 0:
                k = kk
                max_idx = idx
                break
    # Determine contributions
    if k > 0:
        # i from 0..n-k-1
        # contributions:
        # c_i = h[i] * (1 - h[i+k])
        contrib = h[:n - k] * (1 - h[k:])
        i = np.argmax(contrib)
        # allowed delta: can decrease h[i] by at most delta_max = h[i]
        # can increase h[i+k] up to 1 - h[i+k]
        delta_max = min(h[i], 1 - h[i + k])
        if delta_max <= 0:
            return None
        delta = delta_max * np.random.rand() * 0.5  # up to half
        h_new = h.copy()
        h_new[i] -= delta
        h_new[i + k] += delta
    else:
        kk = -k
        # i from kk..n-1 such that i-kk >=0
        # contributions: h[i] * (1 - h[i-kk])
        contrib = h[kk:] * (1 - h[:n - kk])
        # index relative to full vector
        rel_i = np.argmax(contrib)
        i = rel_i + kk
        # move from i to i-kk
        delta_max = min(h[i], 1 - h[i - kk])
        if delta_max <= 0:
            return None
        delta = delta_max * np.random.rand() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[i - kk] += delta
    return h_new

import numpy as np, time, random, math

def _project_sum(v, target):
    # project onto [0,1] with given sum via simple bisection on offset
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def _eval(h):
    n = h.shape[0]
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * (2.0 / n), c

def _grad_move(h):
    n = h.shape[0]
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(c))
    k = idx - (n - 1)
    if k == 0:                                  # pick a non‑zero shift
        order = np.argsort(-c)
        for i in order:
            kk = i - (n - 1)
            if kk != 0:
                k = kk
                break
    if k > 0:
        L = n - k
        contrib = h[:L] * (1.0 - h[k:])
        i = int(np.argmax(contrib))
        delta_max = min(h[i], 1.0 - h[i + k])
        if delta_max <= 0:
            return None
        delta = delta_max * np.random.rand() * 0.5
        nh = h.copy()
        nh[i] -= delta
        nh[i + k] += delta
    else:                                         # k < 0
        kk = -k
        L = n - kk
        contrib = h[kk:] * (1.0 - h[:L])
        rel = int(np.argmax(contrib))
        i = rel + kk
        delta_max = min(h[i], 1.0 - h[i - kk])
        if delta_max <= 0:
            return None
        delta = delta_max * np.random.rand() * 0.5
        nh = h.copy()
        nh[i] -= delta
        nh[i - kk] += delta
    return nh

def _random_feasible(n):
    v = np.random.rand(n)
    return _project_sum(v, n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
