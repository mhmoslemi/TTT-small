def project_onto_simplex_box(v, target):
    # project onto [0,1] and sum target.
    # find lambda such that sum(min(1, max(0, v - lambda))) = target
    # binary search
    lo = min(v) - 1.0  # if lambda = lo, v - lambda >= 1 for all => clipped to 1 => sum = n
    hi = max(v)        # if lambda = hi, v - lambda <= 0 => sum = 0
    # ensure sum at lo >= target >= sum at hi (monotonic decreasing)
    # Use tolerance.
    for iter in range(100):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0, 1)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo+hi)/2.0, 0, 1)
    return w

def gradient_for_shift(h, k):

import time
import numpy as np

# ---------- Helper functions ----------

def compute_c5(h):
    """Return the C5 bound for the step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using zero‑padding (mode='full')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_onto_bounds_sum(v, target, max_iter=30):
    """
    Euclidean projection of v onto the set {w | 0 <= w_i <= 1, sum(w) = target}.
    Implements the classic “capped simplex” projection via a binary search on the
    Lagrange multiplier λ solving sum_i clip(v_i-λ,0,1) = target.
    """
    lo = v.min() - 1.0      # forces all entries to 1 -> sum = n
    hi = v.max()            # forces all entries to 0 -> sum = 0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    return w


def gradient_for_shift(h, k):
    """Gradient of the correlation at shift k w.r.t. the vector h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]            # ∂/∂h_i part from (1 - h_{i+k})
        grad[k:]      -= h[: n - k]             # ∂/∂h_i part from -h_{i-k}
    else:
        kp = -k
        grad[kp:]     += 1.0 - h[: n - kp]       # i + k >=0  -> i >= kp
        grad[: n - kp] -= h[kp:]                # i - k = i + kp < n
    return grad


def initialise_h(n, kind='constant', seed=0):
    """Create an initial feasible vector of length n."""
    rng = np.random.RandomState(seed)
    if kind == 'constant':
        h = np.full(n, 0.5, dtype=float)
    elif kind == 'triangle':
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    elif kind == 'sine':
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h = 0.5 * (1.0 + np.cos(np.pi * x))
    elif kind == 'random':
        h = rng.rand(n)
    else:
        raise ValueError('unknown kind')
    target = n / 2.0
    h = project_onto_bounds_sum(h, target)
    return h


def upsample_to_target(src, target_n):
    """Interpolate/upsample src (numpy array) to length target_n."""
    src_n = src.shape[0]
    if src_n == target_n:
        return src.copy()
    # centre positions of the source and target intervals
    src_x = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    tgt_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    h = np.interp(tgt_x, src_x, src)
    # enforce constraints
    h = project_onto_bounds_sum(h, target_n / 2.0)
    return h


def subgradient_descent(start_h, max_seconds, step0=0.9, max_iter=20000):
    """
    Simple sub‑gradient descent for the C5 objective.
    Returns the best vector encountered and its C5 value.
    """
    start_time = time.time()
    n = start_h.shape[0]
    target = n / 2.0

    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    for it in range(max_iter):
        # time budget check
        if time.time() - start_time > max_seconds * 0.95:
            break

        # find the worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx = int(np.argmax(corr))
        k = idx - (n - 1)

        # gradient of the active shift
        grad = gradient_for_shift(cur_h, k)

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)

        # trial update
        trial = cur_h - step * grad
        trial = project_onto_bounds_sum(trial, target)

        val = compute_c5(trial)

        # accept if improvement; otherwise attempt a back‑track with smaller steps
        if val < cur_val:
            cur_h, cur_val = trial, val
            if val < best_val:
                best_h, best_val = trial.copy(), val
        else:
            improved = False
            # a few back‑tracking attempts
            tmp_step = step
            for _ in range(4):
                tmp_step *= 0.5
                trial2 = cur_h - tmp_step * grad
                trial2 = project_onto_bounds_sum(trial2, target)
                val2 = compute_c5(trial2)
                if val2 < cur_val:
                    cur_h, cur_val = trial2, val2
                    if val2 < best_val:
                        best_h, best_val = trial2.copy(), val2
                    improved = True
                    break
            if not improved:
                # keep current point
                pass

    return best_h, best_val


def random_local_search(start_h, max_seconds):
    """
    Simple greedy local search: move a small amount of mass from a higher
    point to a lower point when it improves the bound.
    """
    n = start_h.shape[0]
    target = n / 2.0
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h, best_val = cur_h.copy(), cur_val
    rng = np.random.RandomState()

    start_time = time.time()
    while time.time() - start_time < max_seconds * 0.95:
        i, j = rng.randint(n, size=2)
        if i == j:
            continue
        # decide direction of transfer (from larger to smaller)
        if cur_h[i] <= cur_h[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        max_delta = min(cur_h[src], 1.0 - cur_h[dst])
        if max_delta <= 0.0:
            continue
        # random fraction of the feasible delta
        delta = max_delta * rng.rand() * 0.5
        trial = cur_h.copy()
        trial[src] -= delta
        trial[dst] += delta
        # feasibility is guaranteed by construction
        val = compute_c5(trial)
        if val < cur_val:
            cur_h, cur_val = trial, val
            if val < best_val:
                best_h, best_val = trial.copy(), val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Solve the Erdős C5 minimisation problem within the provided time budget.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    target_n = 200                     # resolution (feel free to adjust)
    target_sum = target_n / 2.0

    # ----- Build initial vector -------------------------------------------------
    # If a parent construction is available, up‑sample it; otherwise try a few
    # simple deterministic starts.
    init_vec = None
    if 'initial_h_values' in globals():
        try:
            init_vec = np.asarray(initial_h_values, dtype=float)
        except Exception:
            init_vec = None

    candidates = []

    if init_vec is not None:
        # up‑sample the parent solution to the target resolution
        up = upsample_to_target(init_vec, target_n)
        candidates.append(up)

    # other deterministic seeds
    candidates.append(initialise_h(target_n, kind='constant', seed=seed))
    candidates.append(initialise_h(target_n, kind='triangle', seed=seed + 1))
    candidates.append(initialise_h(target_n, kind='sine', seed=seed + 2))

    # add a random start
    candidates.append(initialise_h(target_n, kind='random', seed=seed + 3))

    best_h = None
    best_val = float('inf')
    start_time = time.time()
    remaining = budget_s

    # ----- Optimize each candidate ------------------------------------------------
    for cand in candidates:
        # allocate a small time slice to each start (roughly equal)
        time_slice = remaining / max(1, len(candidates))
        # sub‑gradient phase
        sg_h, sg_val = subgradient_descent(cand, time_slice, step0=0.9, max_iter=15000)
        # local search phase (optional, gives fine‑tuning)
        ls_h, ls_val = random_local_search(sg_h, time_slice)
        if ls_val < best_val:
            best_h, best_val = ls_h, ls_val
        # update remaining budget
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= 5.0:               # not enough time for another full pass
            break

    # Fallback – if nothing succeeded, fall back to the constant construction
    if best_h is None:
        best_h = initialise_h(target_n, kind='constant', seed=seed)
        best_val = compute_c5(best_h)

    # ----- Return result ---------------------------------------------------------
    h_values = best_h.tolist()
    c5_bound = float(best_val)
    n_points = target_n
    return (h_values, c5_bound, n_points)
