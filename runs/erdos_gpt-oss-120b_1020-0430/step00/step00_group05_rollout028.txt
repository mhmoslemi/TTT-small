def hill_climb_targeted(h, target_max, time_limit):
    start = time.time()
    cur_max = max_corr_of_h(h)
    while cur_max > target_max and time.time() - start < time_limit:
        # compute correlation array
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = int(corr.max())
        # if already <= target, break
        if max_val <= target_max:
            cur_max = max_val
            break
        # find index of max (random among them)
        max_idxs = np.flatnonzero(corr == max_val)
        idx = int(max_idxs[0])

        # find swap indices

        if len(i_candidates) > 0 and len(j_candidates) > 0:
            # Choose random pair
            i = np.random.choice(i_candidates)
            j = np.random.choice(j_candidates)
            # swap
            h[i], h[j] = h[j], h[i]
            # recompute cur_max
            cur_max = int(np.correlate(h, 1 - h, mode='full').max())

            h[i], h[j] = h[j], h[i]
            # recompute cur_max
            cur_max = int(np.correlate(h, 1 - h, mode='full').max())

import time
import numpy as np

def max_corr_of_h(h: np.ndarray) -> int:
    """Maximum correlation value max_x Σ h[x](1‑h[x+k]) for integer shifts."""
    # use int32 to avoid overflow
    c = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
    return int(c.max())

def mismatched_pairs_for_shift(h: np.ndarray, shift: int):
    """Return indices i (h[i]=1, h[i+shift]=0) and j (h[j]=0, h[j+shift]=1)."""
    n = h.shape[0]
    if shift >= 0:
        lim = n - shift
        i_mask = (h[:lim] == 1) & (h[shift:] == 0)
        j_mask = (h[:lim] == 0) & (h[shift:] == 1)
        i_cand = np.flatnonzero(i_mask)
        j_cand = np.flatnonzero(j_mask)
    else:
        s = -shift
        lim = n - s
        i_mask = (h[s:] == 1) & (h[:lim] == 0)
        j_mask = (h[s:] == 0) & (h[:lim] == 1)
        i_cand = np.flatnonzero(i_mask) + s
        j_cand = np.flatnonzero(j_mask) + s
    return i_cand, j_cand

def hill_climb_targeted(h: np.ndarray, target_max: int,
                        rng: np.random.Generator, time_limit: float):
    """Reduce the maximal correlation using swaps that directly attack the worst shift."""
    start = time.time()
    n = h.shape[0]
    cur_max = max_corr_of_h(h)

    while cur_max > target_max and (time.time() - start) < time_limit:
        corr = np.correlate(h.astype(np.int32),
                            (1 - h).astype(np.int32),
                            mode="full")
        max_val = int(corr.max())
        if max_val <= target_max:
            cur_max = max_val
            break

        # pick one shift attaining the maximum
        idxs = np.flatnonzero(corr == max_val)
        shift = int(idxs[0] - (n - 1))

        i_cand, j_cand = mismatched_pairs_for_shift(h, shift)

        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
        else:
            # fall back to a random 1↔0 swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)

        h[i], h[j] = h[j], h[i]
        cur_max = max_corr_of_h(h)

    return h, cur_max

def search_n(n: int, seed: int, time_budget: float):
    """Search for a good binary vector of length n within the given time budget."""
    rng = np.random.default_rng(seed)
    target_max = int(np.floor(0.3808 * n / 2))      # max correlation we need to beat
    best_h = None
    best_max = n + 1
    start = time.time()

    while (time.time() - start) < time_budget:
        # start from a fresh random half‑filled vector
        perm = rng.permutation(n)
        h = np.zeros(n, dtype=np.int8)
        h[perm[: n // 2]] = 1

        cur_max = max_corr_of_h(h)
        if cur_max <= target_max:
            return h, cur_max * 2.0 / n, n

        # give each local search a tiny slice of the remaining budget
        remaining = time_budget - (time.time() - start)
        sub_budget = min(0.5, remaining * 0.5)
        h_opt, max_opt = hill_climb_targeted(h, target_max, rng, sub_budget)

        if max_opt < best_max:
            best_max = max_opt
            best_h = h_opt.copy()
            if best_max <= target_max:
                break

    if best_h is None:
        # fallback deterministic construction (first half ones)
        best_h = np.zeros(n, dtype=np.int8)
        best_h[: n // 2] = 1
        best_max = max_corr_of_h(best_h)

    return best_h, best_max * 2.0 / n, n
