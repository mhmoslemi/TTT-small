def max_corr(h):
    b = 1 - h
    N = len(h)
    size = 1 << ((2*N-1).bit_length()) # next power of 2 >= 2N-1
    # compute rfft
    Ha = np.fft.rfft(h, n=size)
    Hb_rev = np.fft.rfft(b[::-1], n=size)
    conv = np.fft.irfft(Ha * Hb_rev, n=size)
    # conv length = size, but we need first 2N-1 values:
    corr = conv[:2*N-1]
    # multiply by dx
    return np.max(corr) * (2.0 / N)

import numpy as np, time, math, random

# Global variables will be set inside run()
fft_size = None
N = None

def _max_corr(val):
    """Return max overlap integral for binary/continuous array `val`."""
    # val: 1‑D numpy array of length N, entries in [0,1]
    # compute correlation of val with (1-val) using FFT (linear, zero‑padded)
    b_rev = (1.0 - val)[::-1]
    Ha = np.fft.rfft(val, n=fft_size)
    Hb = np.fft.rfft(b_rev, n=fft_size)
    conv = np.fft.irfft(Ha * Hb, n=fft_size)[:2 * N - 1]
    return np.max(conv) * (2.0 / N)   # multiply by dx = 2/N


def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] (discretised) that yields a small
    C5 = max_k ∫ h(x)(1−h(x+k))dx.

    Returns (h_values, c5_bound, n_points)
    """
    global N, fft_size

    rd = np.random.RandomState(seed)
    start = time.time()

    # ---------------------------------------------------------
    # 1. Choose a discretisation size.
    # ---------------------------------------------------------
    # 200 points give dx=0.01 and a manageable problem size.
    N = 200
    target = 0.3808          # we aim to beat this bound
    half = N // 2            # number of ones needed for a binary representation
    fft_size = 1 << ((2 * N - 1).bit_length())   # next power of two ≥ 2N‑1

    # ---------------------------------------------------------
    # 2. Build an initial candidate.
    # ---------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        # Upsample / downsample to length N with linear interpolation
        x_old = np.linspace(0.0, 2.0, num=len(init), endpoint=False)
        x_new = np.linspace(0.0, 2.0, num=N, endpoint=False)
        h = np.interp(x_new, x_old, init)
        h = np.clip(h, 0.0, 1.0)
        # Convert to a binary pattern keeping the correct total mass.
        h_bin = (h >= 0.5).astype(np.float64)
        cur = int(h_bin.sum())
        # Adjust the number of ones to exactly half.
        while cur > half:
            idx = rd.choice(np.flatnonzero(h_bin == 1))
            h_bin[idx] = 0.0
            cur -= 1
        while cur < half:
            idx = rd.choice(np.flatnonzero(h_bin == 0))
            h_bin[idx] = 1.0
            cur += 1
        h = h_bin
    else:
        # Simple random binary start with exactly half ones.
        h = np.zeros(N, dtype=np.float64)
        h[:half] = 1.0
        rd.shuffle(h)

    # ---------------------------------------------------------
    # 3. Evaluate the start and initialise bookkeeping.
    # ---------------------------------------------------------
    best_h = h.copy()
    best_val = _max_corr(best_h)
    current_h = h.copy()
    current_val = best_val

    # Helper for fast random choice of a 1‑position / 0‑position.
    def _rand_one():
        while True:
            i = rd.randint(N)
            if current_h[i] > 0.5:
                return i
    def _rand_zero():
        while True:
            j = rd.randint(N)
            if current_h[j] < 0.5:
                return j

    # ---------------------------------------------------------
    # 4. Simulated‑annealing style local search.
    # ---------------------------------------------------------
    no_improve = 0          # counter for stagnation
    max_no_improve = 20000 # after this we re‑randomise

    while True:
        # time‑out check
        if time.time() - start > budget_s * 0.95:
            break

        # propose a swap of a 1 and a 0
        i = _rand_one()
        j = _rand_zero()
        current_h[i] = 0.0
        current_h[j] = 1.0

        new_val = _max_corr(current_h)

        # temperature that linearly cools to a small floor
        elapsed = time.time() - start
        T = max(0.001, (budget_s - elapsed) / budget_s)  # in (0,1]

        accept = False
        if new_val <= current_val:
            accept = True
        else:
            # Metropolis acceptance
            if rd.rand() < math.exp(-(new_val - current_val) / T):
                accept = True

        if accept:
            current_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_h = current_h.copy()
                # early exit if we already beat the target
                if best_val <= target:
                    break
        else:
            # revert the swap
            current_h[i] = 1.0
            current_h[j] = 0.0
            no_improve += 1

        # occasional restart when stuck
        if no_improve >= max_no_improve:
            # re‑initialise a fresh random binary vector
            current_h[:] = 0.0
            current_h[:half] = 1.0
            rd.shuffle(current_h)
            current_val = _max_corr(current_h)
            no_improve = 0

    # ---------------------------------------------------------
    # 5. Return the best pattern found.
    # ---------------------------------------------------------
    # Ensure the integral constraint holds exactly (tiny floating errors are harmless)
    # but we enforce: sum(h) == N/2
    # If rounding made sum slightly off, adjust a few entries.
    s = best_h.sum()
    diff = half - int(round(s))
    if diff != 0:
        # flip `diff` zeros to ones or vice‑versa
        if diff > 0:
            zeros = np.flatnonzero(best_h == 0)
            picks = rd.choice(zeros, size=diff, replace=False)
            best_h[picks] = 1.0
        else:
            ones = np.flatnonzero(best_h == 1)
            picks = rd.choice(ones, size=-diff, replace=False)
            best_h[picks] = 0.0

    final_val = _max_corr(best_h)   # final recompute
    return best_h, float(final_val), N
