import time
import numpy as np
import math

def _max_corr_and_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx] * (2.0 / n)
    shift = idx - (n - 1)
    return max_val, shift, corr

def _max_corr(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)

def _project_sum_bounds(v, target):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    free = np.where((v > 0) & (v < 1))[0]
    # simple water‑filling loop
    while free.size and abs(diff) > 1e-12:
        delta = diff / free.size
        add = np.clip(v[free] + delta, 0.0, 1.0) - v[free]
        v[free] += add
        diff = target - v.sum()
        free = np.where((v > 0) & (v < 1))[0]
    # if still off (tiny rounding), dump the remainder on a random free entry
    if abs(diff) > 1e-12:
        i = np.random.randint(0, len(v))
        v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        # binarise (keep close to original shape)
        h = (init > 0.5).astype(float)
        target_ones = n // 2
        cur = int(h.sum())
        if cur != target_ones:
            need = target_ones - cur
            if need > 0:
                zeros = np.where(h == 0)[0]
                pick = rng.choice(zeros, size=need, replace=False)
                h[pick] = 1.0
            else:
                ones = np.where(h == 1)[0]
                pick = rng.choice(ones, size=-need, replace=False)
                h[pick] = 0.0
    else:
        n = 200                       # default size if no parent is given
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        rng.shuffle(h)

    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # stage 1 – binary hill climbing / simulated annealing
    binary_end = start + budget_s * 0.75
    best_h = h.copy()
    best_val, best_shift, _ = _max_corr_and_shift(best_h)
    cur_val, cur_shift, _ = best_val, best_shift, None

    no_improve = 0
    restart_threshold = 15000

    while time.time() < binary_end:
        # propose a swap biased towards the current worst shift
        d = cur_shift
        if d == 0:                     # degenerate – fall back to random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            # indices that actually contribute to the max correlation
            if d > 0:
                i_range = np.arange(0, n - d)
            else:
                i_range = np.arange(-d, n)
            i_cand = i_range[(h[i_range] > 0.5) & (h[i_range + d] < 0.5)]
            zeros = np.where(h == 0)[0]
            # zeros that are safe w.r.t. the same shift
            if d > 0:
                out = zeros + d >= n
                safe = out | (h[zeros + d] > 0.5)
                j_cand = zeros[safe]
            else:
                out = zeros + d < 0
                safe = out | (h[zeros + d] > 0.5)
                j_cand = zeros[safe]

            if i_cand.size and j_cand.size:
                i = rng.choice(i_cand)
                j = rng.choice(j_cand)
            else:
                # fallback random pair
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i = rng.choice(ones)
                j = rng.choice(zeros)

        # perform swap
        h[i], h[j] = h[j], h[i]

        new_val, new_shift, _ = _max_corr_and_shift(h)
        delta = new_val - cur_val
        # temperature proportional to remaining time
        remain = binary_end - time.time()
        T = max(1e-4, remain / (binary_end - start)) * 0.02
        accept = False
        if delta <= 0:
            accept = True
        elif rng.random() < math.exp(-delta / T):
            accept = True

        if accept:
            cur_val, cur_shift = new_val, new_shift
            no_improve = 0
            if cur_val < best_val:
                best_val, best_h = cur_val, h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            no_improve += 1

        if no_improve > restart_threshold:
            # random restart while keeping the best found so far
            h = np.zeros(n, dtype=float)
            h[: n // 2] = 1.0
            rng.shuffle(h)
            cur_val, cur_shift, _ = _max_corr_and_shift(h)
            no_improve = 0

    # ------------------------------------------------------------------
    # stage 2 – continuous mass‑transfer refinement
    cont_end = start + budget_s - 0.2   # tiny safety buffer
    h = best_h.astype(float)          # start from the best binary found
    cur_val = best_val
    step_max = 0.12

    while time.time() < cont_end:
        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # decide direction of transfer
        if h[i] < h[j]:
            donor, rec = j, i
        else:
            donor, rec = i, j
        max_transfer = min(step_max, h[donor], 1.0 - h[rec])
        if max_transfer <= 0:
            continue
        eps = rng.random() * max_transfer
        h[donor] -= eps
        h[rec] += eps

        new_val = _max_corr(h)
        delta = new_val - cur_val
        remain = cont_end - time.time()
        T = max(1e-5, remain / (cont_end - start)) * 0.015
        accept = False
        if delta <= 0:
            accept = True
        elif rng.random() < math.exp(-delta / T):
            accept = True

        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val, best_h = cur_val, h.copy()
        else:
            # revert
            h[donor] += eps
            h[rec] -= eps

        # slowly shrink step size
        step_max *= 0.9999
        if step_max < 1e-4:
            step_max = 1e-4

    # ------------------------------------------------------------------
    # final projection to satisfy exact sum (tiny numerical drift)
    best_h = _project_sum_bounds(best_h, target_sum)

    final_c5 = _max_corr(best_h)

    return (best_h.astype(float), float(final_c5), n)
