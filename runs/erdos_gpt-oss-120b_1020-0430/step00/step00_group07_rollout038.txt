    else:

        # Dirichlet‑like sampling (exponential weights) – automatically respects the sum
        x = np.random.rand(n)
        h = x / x.sum() * (n / 2.0)
    # Clip and repair possible slight violations
    n = h.shape[0]
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        # Scale then adjust a random free coordinate
        if cur > 0:
            h *= (target / cur)
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size == 0:
                free = np.arange(n)
            i = np.random.choice(free)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def _worst_shift(h):
    """
    Return (max_corr, shift) where shift is an integer in
    [-(n-1), …, n-1] giving the position of the worst overlap.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr)
    max_corr = corr[k]
    shift = k - (n - 1)   # 0‑shift is at index n-1
    return max_corr, shift


def _neighbor_score(h, d):
    """
    For a given integer shift d, compute the vector
        s[i] = h[i+d] (if inside) + h[i-d] (if inside)
    This is proportional to the sub‑gradient of the worst linear form.
    """
    n = h.shape[0]
    s = np.zeros(n, dtype=float)
    if d >= 0:
        nd = n - d
        if nd > 0:
            s[:nd] += h[d:]          # i+d term
            s[d:] += h[:nd]          # i-d term
    else:
        dpos = -d
        nd = n - dpos
        if nd > 0:
            s[dpos:] += h[:nd]       # i-d term (since d negative)
            s[:nd] += h[dpos:]       # i+d term
    return s


def _exchange(h, i_up, i_down, delta):
    """
    Perform a mass exchange that keeps the total sum unchanged:
        h[i_up] += delta
        h[i_down] -= delta
    Both indices are assumed feasible for the move.
    """
    h[i_up] += delta
    h[i_down] -= delta
    # Numerical safety – clip to [0,1] (tiny overshoots may appear)
    if h[i_up] > 1.0:
        h[i_up] = 1.0
    if h[i_down] < 0.0:
        h[i_down] = 0.0
    return h


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise the step‑function h for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    # reproducibility
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # initialise
    h = _initial_h()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = _evaluate_h(best_h)

    # annealing‑style schedule for the maximal elementary move
    max_step_start = 0.2          # at the beginning we may move up to 0.2 mass
    max_step_end   = 0.001        # at the end we only fine‑tune
    total_budget = max(budget_s - 2.0, 1.0)   # keep a small safety margin

    # counters for occasional re‑randomisation
    stagnation_counter = 0
    last_improvement = best_val

    iter_cnt = 0
    while True:
        # time check (leave a tiny buffer)
        if time.time() - start_time > budget_s - 0.4:
            break

        # ---- identify the worst shift and its neighbourhood scores ----
        _, d = _worst_shift(h)
        scores = _neighbor_score(h, d)

        # feasible candidates
        inc_mask = h < 1.0 - 1e-12   # can be increased
        dec_mask = h > 1e-12         # can be decreased
        inc_idx = np.where(inc_mask)[0]
        dec_idx = np.where(dec_mask)[0]

        if inc_idx.size == 0 or dec_idx.size == 0:
            # should never happen, but break to avoid infinite loop
            break

        # with a small probability add randomness – helps escape flat plateaux
        if np.random.rand() < 0.02:
            i_up = np.random.choice(inc_idx)
            i_down = np.random.choice(dec_idx)
        else:
            # greedy choice: move from low‑score to high‑score positions
            i_up = inc_idx[np.argmax(scores[inc_idx])]
            i_down = dec_idx[np.argmin(scores[dec_idx])]

        # schedule step size (linear decay)
        elapsed = time.time() - start_time
        frac = min(elapsed / total_budget, 1.0)
        max_step = max_step_start * (1.0 - frac) + max_step_end * frac

        # admissible delta respecting bounds
        delta_up = 1.0 - h[i_up]
        delta_down = h[i_down]
        delta = min(delta_up, delta_down, max_step)

        # inject a little randomness into the magnitude
        delta *= 0.5 + 0.5 * np.random.rand()
        if delta <= 0:
            # no feasible move – pick another pair next iteration
            continue

        # apply the exchange
        h = _exchange(h, i_up, i_down, delta)

        # occasional renormalisation to control floating‑point drift
        if iter_cnt % 1024 == 0:
            cur_sum = h.sum()
            if abs(cur_sum - target_sum) > 1e-9:
                diff = target_sum - cur_sum
                free = np.where((h > 0.0) & (h < 1.0))[0]
                if free.size == 0:
                    free = np.arange(n)
                i = np.random.choice(free)
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)

        # evaluate current candidate
        cur_val = _evaluate_h(h)

        # keep the best seen so far
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            stagnation_counter = 0
            last_improvement = cur_val
        else:
            stagnation_counter += 1

        # if stuck for a long stretch, perform a mild random perturbation
        if stagnation_counter > 5000:
            # random 2‑index exchange with a relatively large delta
            i_up = np.random.choice(inc_idx)
            i_down = np.random.choice(dec_idx)
            delta = min(1.0 - h[i_up], h[i_down], max_step_start * 0.5)
            if delta > 0:
                h = _exchange(h, i_up, i_down, delta)
            stagnation_counter = 0

        iter_cnt += 1

    # final evaluation (ensure the reported bound is accurate)
    final_val = _evaluate_h(best_h)

    # Return as required: h_values (list or numpy array), bound, n_points
    return (best_h.tolist(), float(final_val), int(n))
