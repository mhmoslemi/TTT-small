import numpy as np, time, math

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def max_overlap_and_shift(h):
    corr = compute_corr(h)
    idx = np.argmax(corr)
    max_cnt = corr[idx]
    shift = idx - (len(h) - 1)
    return max_cnt, shift, corr

def mismatched_indices(h, shift):
    n = len(h)
    if shift >= 0:
        hi = h[:n-shift]
        hshift = h[shift:]
        i_one = np.where((hi == 1) & (hshift == 0))[0]
        i_zero = np.where((hi == 0) & (hshift == 1))[0]
        # actual indices in original array
        return i_one, i_zero
    else:
        s = -shift
        hi = h[s:]
        hshift = h[:n-s]
        i_one = np.where((hi == 1) & (hshift == 0))[0] + s
        i_zero = np.where((hi == 0) & (hshift == 1))[0] + s
        return i_one, i_zero

def hill_climb(n, seed, time_limit):
    np.random.seed(seed)
    # start with random binary with half ones
    h = np.zeros(n, dtype=np.int8)
    ones = np.random.choice(n, n//2, replace=False)
    h[ones] = 1
    best_h = h.copy()
    max_cnt, shift, _ = max_overlap_and_shift(h)
    best_cnt = max_cnt
    start = time.time()
    last_improve = start
    restart_interval = 30.0
    while time.time() - start < time_limit:
        # compute max overlap
        max_cnt, shift, corr = max_overlap_and_shift(h)
        if max_cnt < best_cnt:
            best_cnt = max_cnt
            best_h = h.copy()
            last_improve = time.time()
        # if sufficiently good break
        # (optional early exit)
        # find mismatched indices for shift
        i_ones, i_zeros = mismatched_indices(h, shift)
        made_swap = False
        # try a few candidate swaps

import time
import numpy as np

def _project_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=60):
    """Project vector v onto {x: lower <= x <= upper, sum(x)=target}."""
    # Find lambda such that sum(clip(v+lam, lower, upper)) = target.
    lo = -np.max(v) - 1.0  # ensures all clipped to lower
    hi = upper - np.min(v) + 1.0  # ensures all clipped to upper
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, lower, upper))
        if s < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, lower, upper)

def _max_corr_shift(h):
    """Return max correlation count, the shift (as integer) that attains it, and the full correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    max_cnt = int(corr[idx])
    shift = idx - (len(h) - 1)  # positive => right shift
    return max_cnt, shift, corr

def _mismatch_indices(h, shift):
    """Indices (i where h[i]=1, h[i+shift]=0) and (j where h[j]=0, h[j+shift]=1)."""
    n = len(h)
    if shift >= 0:
        a = h[:n - shift]
        b = h[shift:]
        i_one = np.where((a == 1) & (b == 0))[0]          # positions of a (original i)
        i_zero = np.where((a == 0) & (b == 1))[0]         # positions of a (original i)
        return i_one, i_zero
    else:
        s = -shift
        a = h[s:]                 # original i
        b = h[:n - s]             # i+shift
        i_one = np.where((a == 1) & (b == 0))[0] + s
        i_zero = np.where((a == 0) & (b == 1))[0] + s
        return i_one, i_zero

def _hill_climb(n, seed, time_budget):
    rng = np.random.default_rng(seed)
    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        # interpolate to length n if needed
        if init.size != n:
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        init = np.clip(init, 0.0, 1.0)
        # binary conversion (keep structure)
        h = (init >= 0.5).astype(np.int8)
        cur = int(h.sum())
        target = n // 2
        if cur > target:
            excess = cur - target
            ones = np.where(h == 1)[0]
            flip = rng.choice(ones, size=excess, replace=False)
            h[flip] = 0
        elif cur < target:
            deficit = target - cur
            zeros = np.where(h == 0)[0]
            flip = rng.choice(zeros, size=deficit, replace=False)
            h[flip] = 1
    else:
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, size=n // 2, replace=False)] = 1

    best_h = h.copy()
    best_cnt, _, _ = _max_corr_shift(best_h)

    start = time.time()
    last_improve = start
    restart_interval = 30.0  # seconds without improvement before random restart
    dx = 2.0 / n

    while time.time() - start < time_budget * 0.95:
        cur_cnt, cur_shift, _ = _max_corr_shift(h)

        if cur_cnt < best_cnt:
            best_cnt = cur_cnt
            best_h = h.copy()
            last_improve = time.time()
            if best_cnt * dx < 0.3808:
                break

        # try to reduce the current maximal shift
        i_one, i_zero = _mismatch_indices(h, cur_shift)

        improved = False
        for _ in range(5):   # a few attempts per iteration
            if i_one.size and i_zero.size:
                i = int(rng.choice(i_one))
                j = int(rng.choice(i_zero))
            else:
                # fallback: random 1/0 pair
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))

            # perform swap
            h[i], h[j] = 0, 1
            new_cnt, _, _ = _max_corr_shift(h)

            if new_cnt <= cur_cnt:
                improved = True
                break
            else:
                # revert
                h[i], h[j] = 1, 0

        if not improved:
            # occasional uphill move to escape plateaus
            if rng.random() < 0.001:
                # keep the last proposed (even if worse)
                pass

        # restart if stagnated
        if time.time() - last_improve > restart_interval:
            h[:] = 0
            h[rng.choice(n, size=n // 2, replace=False)] = 1
            last_improve = time.time()

    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_cnt = int(final_corr.max())
    best_c = best_cnt * dx
    return best_h.astype(float), best_c, n

    """
    Returns a step function achieving a low C5 bound.

    Output:
        h_values : np.ndarray of shape (n_points,)
        c5_bound : float, the achieved C5 value
        n_points : int, number of samples used
    """
    # Choose a modest even grid size – 100 gives a good trade‑off
    n_points = 100
    h_vals, c5, n = _hill_climb(n_points, seed, budget_s)
    return h_vals, c5, n
