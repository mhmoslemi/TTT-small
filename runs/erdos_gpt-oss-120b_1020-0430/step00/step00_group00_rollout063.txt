def compute_gradient(h, weights):
    # weights is w_k for each shift (same length as O_k vector)
    # We'll compute gradient of smoothmax = sum_k w_k * gradient(O_k)
    n = len(h)
    dx = 2.0/n
    grad = np.zeros_like(h)
    # For each shift k from -(n-1) to (n-1)
    # But correlation length includes negative shifts. For O_k defined via full correlation, shift index mapping: negative k corresponds to h shifted left.
    # We'll process both positive and negative k using similar formula.
    for idx, w in enumerate(weights):
        if w == 0: continue
        k = idx - (n-1)  # zero shift corresponds to index n-1
        # Determine contribution:
        # For i from 0 to n-1 where i+k in [0,n-1]
        # For each i s.t. 0 <= i < n, 0 <= i+k < n
        # For each such i: grad[i] += w*dx*(1 - h[i+k])
        # and grad[i+k] += - w*dx*h[i]
        # we can vectorize using slicing.
        if k >= 0:
            i = slice(0, n - k)
            i_k = slice(k, n)
            grad[i] += w * dx * (1 - h[i_k])
            grad[i_k] -= w * dx * h[i]
        else:
            # k negative => shift left
            k_abs = -k
            i = slice(k_abs, n)
            i_k = slice(0, n - k_abs)
            grad[i] += w * dx * (1 - h[i_k])
            grad[i_k] -= w * dx * h[i]
    # That's gradient of O's.    
    # Additionally gradient of sum constraint is constant (since sum must be n/2). We will handle via projection later.
    return grad

import time

def _project_onto_box_and_sum(v, target):
    """Project vector v onto the box [0,1] with exact sum = target."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h):
    """Return the C5 value for a given step function sample vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _worst_shift_and_grad(h):
    """Find the shift attaining max O_k and return (value, shift, gradient)."""
    n = h.size
    dx = 2.0 / n
    O = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(O.argmax())
    shift = idx - (n - 1)                     # shift = idx - (n-1)
    grad = np.zeros_like(h)

    if shift >= 0:
        i = np.arange(0, n - shift)
        j = i + shift
        grad[i] += dx * (1.0 - h[j])
        grad[j] -= dx * h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        j = i - s
        grad[i] += dx * (1.0 - h[j])
        grad[j] -= dx * h[i]

    return float(O[idx]), shift, grad
