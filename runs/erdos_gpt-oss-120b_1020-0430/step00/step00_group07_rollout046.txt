import numpy as np
import time
import math

def _objective(h, dx):
    """C5 value for a given step function."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _gradient_of_shift(h, k):
    """Sub‑gradient of the correlation term with shift k."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if k >= 0:
        i = np.arange(n - k, dtype=int)
        grad[i] += (1.0 - h[i + k])
        grad[i + k] += -h[i]
    else:
        kp = -k
        i = np.arange(n - kp, dtype=int)
        grad[i] += -h[i + kp]
        grad[i + kp] += (1.0 - h[i])
    return grad

def _project_feasible(y, target_sum):
    """
    Projection of y onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum}.
    Uses a bisection on the Lagrange multiplier λ such that
    x_i = clip(y_i - λ, 0, 1) and sum x_i = target_sum.
    """
    lo = np.min(y - 1.0) - 1.0  # ensures all x_i = 1
    hi = np.max(y) + 1.0        # ensures all x_i = 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(y - lam, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low C5 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        n = 80
        h = np.full(n, 0.5, dtype=np.float64)
    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # ensure feasibility
    if not (0.0 <= h).all() or not (h <= 1.0).all() or abs(h.sum() - target_sum) > 1e-12:
        h = np.clip(h, 0.0, 1.0)
        h = _project_feasible(h, target_sum)

    best_h = h.copy()
    best_val = _objective(best_h, dx)
    cur_val = best_val

    # ------------------------------------------------------------------
    # parameters for the main gradient‑descent phase
    total_time = budget_s * 0.93               # reserve some margin
    step_start = 0.5
    step_end = 0.001

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= total_time:
            break
        # linear schedule for step size
        t = elapsed / total_time
        step = step_start * (1.0 - t) + step_end * t

        # correlation & identify maximal shift(s)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        max_idx = np.flatnonzero(corr == max_c)

        # compute (averaged) sub‑gradient for the active shift(s)
        grad = np.zeros_like(h)
        for idx in max_idx:
            k = idx - (n - 1)               # shift corresponding to correlation index
            grad += _gradient_of_shift(h, k)
        grad /= float(len(max_idx))

        # propose a projected gradient step
        y = h - step * grad
        h_new = _project_feasible(y, target_sum)
        new_val = _objective(h_new, dx)

        # accept if improvement, otherwise keep current
        if new_val <= cur_val:
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
        else:
            # occasional Metropolis acceptance to escape plateaus
            T = 0.02 * (1.0 - t) + 1e-6 * t
            if rng.random() < math.exp((cur_val - new_val) / (T + 1e-12)):
                h = h_new
                cur_val = new_val

        # small random perturbations every so often to diversify search
        if rng.random() < 0.001:
            noise = rng.normal(scale=0.02, size=n)
            h_tmp = _project_feasible(best_h + noise, target_sum)
            val_tmp = _objective(h_tmp, dx)
            if val_tmp < best_val:
                best_val = val_tmp
                best_h = h_tmp.copy()
                h = h_tmp
                cur_val = val_tmp

    # ------------------------------------------------------------------
    # final short random‑swap refinement (hill climbing)
    t_refine = time.time()
    while time.time() - t_refine < (budget_s - (time.time() - start_time)):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction: move mass from i to j or opposite, chosen uniformly
        if rng.random() < 0.5:
            max_transfer = min(best_h[i], 1.0 - best_h[j])
            if max_transfer <= 0.0:
                continue
            eps = rng.random() * max_transfer
            cand = best_h.copy()
            cand[i] -= eps
            cand[j] += eps
        else:
            max_transfer = min(best_h[j], 1.0 - best_h[i])
            if max_transfer <= 0.0:
                continue
            eps = rng.random() * max_transfer
            cand = best_h.copy()
            cand[j] -= eps
            cand[i] += eps

        val = _objective(cand, dx)
        if val < best_val:
            best_val = val
            best_h = cand

    # ------------------------------------------------------------------
    # final safety projection (ensures exact feasibility)
    best_h = _project_feasible(best_h, target_sum)

    # return results
    return best_h, float(best_val), n
