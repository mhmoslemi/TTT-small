def compute_c5(h):
    N = len(h)
    dx = 2.0 / N

def hill_climb(h, max_iter, start_temp=0.1, cooling=0.9999):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    N = len(h)
    temp = start_temp
    for it in range(max_iter):
        # choose i where h[i] > 0, j where h[j] < 1
        i = np.random.randint(N)
        if h[i] <= 1e-12:
            continue
        j = np.random.randint(N)
        if h[j] >= 1 - 1e-12:
            continue
        # delta
        max_delta = min(h[i], 1 - h[j])
        # choose delta random within [0, max_delta]
        delta = np.random.random() * max_delta * 0.5  # maybe up to half capacity
        # propose
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # compute c5
        c_new = compute_c5(h_new)
        # acceptance: if better, accept; else maybe accept with probability
        if c_new < best_c or np.random.rand() < np.exp((best_c - c_new) / temp):
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        # temperature cooling
        temp *= cooling
        if temp < 1e-8:
            temp = 1e-8
    return best_h, best_c
```

def anneal(h, max_iter, time_limit):
    best_h = h.copy()
    best_val = max_overlap(h)
    N = len(h)
    start = time.time()
    iter = 0
    temp = 0.1
    while time.time() - start < time_limit:
        # choose random pair
        i = np.random.randint(N)
        j = np.random.randint(N)
        if i == j:
            continue
        # compute max allowable delta
        max_delta = min(h[i], 1 - h[j])
        if max_delta < 1e-8:
            continue
        # choose delta
        delta = max_delta * np.random.rand()
        if delta < 1e-6:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # compute value
        val = max_overlap(h_new)
        # acceptance
        if val < best_val or np.random.rand() < np.exp((best_val - val) / temp):
            h = h_new
            if val < best_val:
                best_val = val
                best_h = h_new.copy()
        # schedule cooling
        temp = max(1e-6, temp * 0.9999)
        iter += 1
    return best_h, best_val
```

def anneal(h, total_iters, temperature_schedule):
    best_h = h.copy()
    best_val = compute_c5(h)
    N = len(h)
    for it in range(total_iters):
        # compute corr
        corr = np.correlate(h, 1 - h, mode='full')
        current_val = np.max(corr) * dx
        # Choose to do targeted move with some probability
        if np.random.rand() < 0.7:
            # targeted
            s_idx = np.argmax(corr)
            shift = s_idx - (N - 1)
            # compute contributions
            if shift >=0:
                i_idx = np.arange(shift, N)
                j_idx = i_idx - shift
            else:
                i_idx = np.arange(0, N + shift)
                j_idx = i_idx - shift
            contrib = h[i_idx] * (1 - h[j_idx])
            # sort descending indices
            order = np.argsort(-contrib)
            found = False
            for idx in order:
                i = i_idx[idx]
                j = j_idx[idx]
                if h[i] > 0 and h[j] < 1:
                    max_delta = min(h[i], 1 - h[j])
                    if max_delta > 1e-6:
                        delta = max_delta * np.random.rand()
                        h_new = h.copy()
                        h_new[i] -= delta
                        h_new[j] += delta
                        found = True
                        break
            if not found: # fallback random
                i = np.random.randint(N)
                j = np.random.randint(N)
                while i == j or h[i] <= 0 or h[j] >=1:
                    i = np.random.randint(N)
                    j = np.random.randint(N)
                max_delta = min(h[i], 1 - h[j])
                delta = max_delta * np.random.rand()
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
        else:
            # random move
            i = np.random.randint(N)
            j = np.random.randint(N)
            while i == j or h[i] <= 0 or h[j] >= 1:
                i = np.random.randint(N)
                j = np.random.randint(N)

            delta = max_delta * np.random.rand()
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta

        # compute new C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_val = np.max(corr_new) * dx

        # acceptance
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        # Simulated annealing acceptance
        # temperature schedule schedule based on iteration
        temp = max(0.001, 0.1 * (1 - it/total_iters))

def project_to_bounds_sum(x, target):
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x + (lo+hi)/2.0, 0.0, 1.0)
    # final small correction
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        # adjust a non-bound element
        mask = (proj > 0) & (proj < 1)
        if np.any(mask):
            idx = np.where(mask)[0][0]
            proj[idx] += diff
            proj[idx] = np.clip(proj[idx], 0.0, 1.0)
    return proj

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project_to_bounds_sum(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project x onto [0,1]^n while forcing sum(x)=target_sum."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if needed
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 0.0) & (proj < 1.0)
        if mask.any():
            idx = np.flatnonzero(mask)[0]
            proj[idx] = np.clip(proj[idx] + diff, 0.0, 1.0)
    return proj


def _anneal_with_time(start_h: np.ndarray, max_seconds: float) -> tuple:
    """Simulated annealing on a fixed time budget."""
    n = start_h.shape[0]
    dx = 2.0 / n
    target = n / 2.0

    cur_h = start_h.copy()
    cur_val = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    while time.time() - start < max_seconds:
        # correlation and worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        cur_val = np.max(corr) * dx
        s_idx = np.argmax(corr)
        shift = s_idx - (n - 1)

        # indices contributing to the worst shift
        if shift >= 0:
            i_idx = np.arange(shift, n)
            j_idx = i_idx - shift
        else:
            i_idx = np.arange(0, n + shift)
            j_idx = i_idx - shift
        contrib = cur_h[i_idx] * (1.0 - cur_h[j_idx])

        # pick a high‑contributing pair (probability proportional to contribution)
        if contrib.size and contrib.sum() > 0:
            probs = contrib / (contrib.sum() + 1e-15)
            choice = np.random.choice(contrib.size, p=probs)
            i = int(i_idx[choice])
            j = int(j_idx[choice])
        else:
            # fallback random pair
            i, j = np.random.randint(n), np.random.randint(n)
            while i == j or cur_h[i] <= 0.0 or cur_h[j] >= 1.0:
                i, j = np.random.randint(n), np.random.randint(n)

        # ensure we can move some mass
        if cur_h[i] <= 0.0 or cur_h[j] >= 1.0:
            # try a few random feasible pairs
            for _ in range(5):
                i, j = np.random.randint(n), np.random.randint(n)
                if i != j and cur_h[i] > 0.0 and cur_h[j] < 1.0:
                    break
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * np.random.rand()

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_val = np.max(np.correlate(new_h, 1.0 - new_h, mode="full")) * dx

        # temperature that linearly cools to a tiny floor
        elapsed = time.time() - start
        temp = max(1e-4, 0.05 * (1.0 - elapsed / max_seconds))

        if new_val < cur_val or np.random.rand() < np.exp((cur_val - new_val) / (temp + 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    # final projection to guarantee feasibility
    best_h = _project_to_bounds_sum(best_h, target)
    best_val = _compute_c5(best_h)
    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function that gives a low C₅ bound.

    The function returns (h_values, c5_bound, n_points) where
    `h_values` is a list of samples on [0,2], `c5_bound` is the computed
    overlap bound, and `n_points` is the length of the sample vector.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine vector length (use the length of a parent solution if given)
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 45
        # triangular initialization (peak at the centre)
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h0 = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))

    target_sum = n / 2.0
    h0 = _project_to_bounds_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    # Main optimisation loop – keep launching short annealing runs
    while time.time() - start_time < budget_s * 0.96:
        remaining = budget_s - (time.time() - start_time)
        # each inner run gets at most 12 seconds (or the remaining time)
        run_time = min(12.0, remaining * 0.7)

        # start either from the current best (slightly perturbed) or from a fresh random vector
        if np.random.rand() < 0.6:
            start_vec = best_h.copy()
            # small random perturbation to escape flat regions
            pert = (np.random.rand(n) - 0.5) * 0.1
            start_vec += pert
            start_vec = np.clip(start_vec, 0.0, 1.0)
            start_vec = _project_to_bounds_sum(start_vec, target_sum)
        else:
            # pure random 0/1 with one half entry
            start_vec = np.zeros(n, dtype=float)
            ones = np.random.choice(n, n // 2, replace=False)
            start_vec[ones] = 1.0
            remaining_idx = np.setdiff1d(np.arange(n), ones)
            half_idx = np.random.choice(remaining_idx)
            start_vec[half_idx] = 0.5

        h_candidate, val_candidate = _anneal_with_time(start_vec, run_time)

        if val_candidate < best_val:
            best_val = val_candidate
            best_h = h_candidate.copy()

        # early exit if we already beat the target record comfortably
        if best_val < 0.38:
            break

    # final safety projection
    best_h = _project_to_bounds_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), n
