import numpy as np
import time
import math

# The harness provides `initial_h_values` and `evaluate_erdos_solution`
# if they exist.  We use them when available.
def _compute_c5(h, dx):
    """Maximum overlap integral C5 for a given step function h."""
    # np.correlate returns sum_i h[i] * (1-h[i+k]) for each integer shift k
    # Multiply by dx to get the integral value.
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretised) to minimise
    C5 = max_k ∫ h(x)(1‑h(x+k))dx.

    Returns
    -------
    h_values : np.ndarray
        Optimised sample values on the original grid.
    c5_bound : float
        The attained C5 bound.
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Initialise h (use the supplied construction if available)
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: a simple triangular shape (area = 1)
        n_fallback = 200
        dx0 = 2.0 / n_fallback
        x = (np.arange(n_fallback) + 0.5) * dx0
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
        # re‑scale to satisfy the exact integral constraint
        cur = h.sum() * dx0
        if cur != 0:
            h *= 1.0 / cur

    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0  # because ∑h_i * dx must equal 1

    # ensure feasibility (tiny numerical adjustments)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny discrepancy uniformly
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

    # ------------------------------------------------------------
    # Simulated‑annealing / targeted local search
    # ------------------------------------------------------------
    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)
    cur_score = best_score

    start_time = time.time()
    # annealing parameters
    T = 0.08                     # initial temperature
    T_min = 1e-6
    decay = 0.99997
    step = 0.12                  # typical amount of mass moved per operation
    step_min = 1e-4
    step_decay = 0.99995

    # probabilities for different move types
    p_random = 0.55
    p_target = 0.30
    p_grad = 0.15

    iter_count = 0
    no_imp_iters = 0

    while time.time() - start_time < budget_s * 0.97:
        iter_count += 1

        # --------------------------------------------------------
        # Choose a move
        # --------------------------------------------------------
        move_type = rng.random()
        if move_type < p_random:
            # ---- random mass transfer between two positions ----------
            # pick a donor (h>0) and a receiver (h<1)
            donors = np.where(h > 1e-12)[0]
            receivers = np.where(h < 1.0 - 1e-12)[0]
            if donors.size == 0 or receivers.size == 0:
                continue
            a = rng.choice(donors)
            b = rng.choice(receivers)
            if a == b:
                continue
            max_delta = min(step, h[a], 1.0 - h[b])
            if max_delta <= 0:
                continue
            delta = rng.uniform(0, max_delta)
            new_h = h.copy()
            new_h[a] -= delta
            new_h[b] += delta

        elif move_type < p_random + p_target:
            # ---- targeted move on the worst shift --------------------
            O = np.correlate(h, 1.0 - h, mode="full") * dx
            k_index = int(np.argmax(O))
            shift = k_index - (n - 1)          # shift can be negative
            if shift == 0:
                continue  # nothing to improve, fallback to random next loop

            length = n - abs(shift)
            if length <= 0:
                continue

            # compute product over the overlapping region
            if shift > 0:
                prod = h[:length] * h[shift:shift + length]
                i_local = int(np.argmin(prod))
                i = i_local
                j = i_local + shift
            else:
                s = -shift
                prod = h[s:s + length] * h[:length]
                i_local = int(np.argmin(prod))
                i = i_local + s
                j = i_local

            # attempt to raise both values a little
            max_delta = min(step, 1.0 - h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue

            # pick a donor that can spare 2*delta
            donors = np.where(h > 2 * max_delta + 1e-12)[0]
            donors = donors[(donors != i) & (donors != j)]
            if donors.size == 0:

            d = rng.choice(donors)

            delta = max_delta * rng.random()
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] += delta
            new_h[d] -= 2.0 * delta

        else:

            O = np.correlate(h, 1.0 - h, mode="full") * dx
            k_index = int(np.argmax(O))
            shift = k_index - (n - 1)

            # gradient of O_k w.r.t. each component
            grad = np.ones(n, dtype=np.float64)
            if shift > 0:
                grad[: n - shift] -= h[shift: n]
                grad[shift: n] -= h[: n - shift]
            elif shift < 0:

                grad[: n - s] -= h[s: n]
                grad[s: n] -= h[: n - s]

            pos_idx = np.where(grad > 1e-12)[0]   # decreasing these reduces O_k
            neg_idx = np.where(grad < -1e-12)[0]  # increasing these reduces O_k
            if pos_idx.size == 0 or neg_idx.size == 0:

            a = rng.choice(pos_idx)
            b = rng.choice(neg_idx)
            max_delta = min(step, h[a], 1.0 - h[b])
            if max_delta <= 0:

            delta = rng.uniform(0, max_delta)

            new_h[a] -= delta

        new_h = np.clip(new_h, 0.0, 1.0)

            idx = rng.integers(n)
            new_h[idx] = np.clip(new_h[idx] + sdiff, 0.0, 1.0)

        new_score = _compute_c5(new_h, dx)
        delta_score = new_score - cur_score

        # --------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------
        accept = False
        if delta_score < -1e-12:
            accept = True
        else:

        # Cooling schedule
        # --------------------------------------------------------

        best_h += (target_sum - cur_sum) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, c5_bound, n
