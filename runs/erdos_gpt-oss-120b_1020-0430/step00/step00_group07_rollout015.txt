def shift(arr, shift):
    n = arr.shape[0]
    if shift > 0:
        out = np.empty_like(arr)
        out[:n-shift] = arr[shift:]
        out[n-shift:] = 0.0
    elif shift < 0:
        s = -shift
        out = np.empty_like(arr)
        out[s:] = arr[:n-s]
        out[:s] = 0.0
    else:
        out = arr.copy()
    return out
```

def gradient_step(h, L, step, dx, S):
    h_plus = shift(h, L)
    h_minus = shift(h, -L)
    grad = dx * (1.0 - (h_plus + h_minus))
    v = h - step * grad
    # projection
    return project_onto_capped_simplex(v, S)

def project_onto_capped_simplex(v, S):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0.0, 1.0)

            break
        if free.size == 0:
            # can't adjust further; break
            break

        v[free] += delta
        # enforce bounds, find indices that hit bounds
        below = v < 0.0
        above = v > 1.0
        # those that violate need to be clipped and removed from free
        if np.any(below) or np.any(above):
            # clip & adjust diff for each changed entry
            # compute new total after clipping
            v = np.clip(v, 0.0, 1.0)
            total = v.sum()
            diff = S - total
            # recompute free indices (still those not at bounds)
            free = np.where((v > 0.0 + 1e-12) & (v < 1.0 - 1e-12))[0]
        else:
            # no bound violation; adjust diff = 0
            diff = 0.0
    return v

import time, math
import numpy as np

def _project_capped_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto {0<=x<=1, sum(x)=S} using a simple water‑filling scheme."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - S) < 1e-12:
        return v
    diff = S - total
    # indices that are not yet at a bound
    free = np.where((v > 0.0) & (v < 1.0))[0]
    while True:
        if abs(diff) < 1e-12 or free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        # enforce bounds and recompute free set
        v = np.clip(v, 0.0, 1.0)
        total = v.sum()
        diff = S - total
        free = np.where((v > 0.0) & (v < 1.0))[0]
    return v


def _shift(arr: np.ndarray, s: int) -> np.ndarray:
    """Shift arr by s positions, filling with zeros (zero‑padding)."""
    n = arr.shape[0]
    if s > 0:
        out = np.empty_like(arr)
        out[: n - s] = arr[s:]
        out[n - s :] = 0.0
    elif s < 0:
        s = -s
        out = np.empty_like(arr)
        out[s:] = arr[: n - s]
        out[:s] = 0.0
    else:
        out = arr.copy()
    return out


def _objective(h: np.ndarray, dx: float) -> float:
    """C5 value for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _gradient_step(h: np.ndarray, lag: int, step: float,
                   dx: float, S: float) -> np.ndarray:
    """Take one projected gradient step for the worst lag."""
    h_plus = _shift(h, lag)
    h_minus = _shift(h, -lag)
    grad = dx * (1.0 - (h_plus + h_minus))
    v = h - step * grad
    return _project_capped_simplex(v, S)


def _resample(old: np.ndarray, N: int) -> np.ndarray:
    """Linear interpolation of an old vector to length N."""
    old_n = old.shape[0]
    if old_n == N:
        return old.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, N, endpoint=False)
    return np.interp(x_new, x_old, old)

    start = time.time()
    deadline = start + budget_s * 0.94  # keep a safety margin

    N = 80

    S = N / 2.0  # required sum of h_i

    best_val = _objective(best_h, dx)

    return best_h, float(best_val), int(N)
