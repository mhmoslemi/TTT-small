def project_to_sum_and_bounds(v, target):
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - (lo + hi) / 2, 0.0, 1.0)
    # Optional final adjustment:
    # Due to rounding may be slight diff; adjust using adding small epsilon to interior points
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        # distribute diff among non-boundary points
        interior_idx = np.where((p > 1e-12) & (p < 1 - 1e-12))[0]
        if interior_idx.size > 0:
            # distribute equally
            incr = diff / interior_idx.size
            p[interior_idx] += incr
            p = np.clip(p, 0.0, 1.0)
        else:
            # fallback: adjust first element within bounds
            # find any index with capacity
            for i in range(len(p)):
                cap_up = 1.0 - p[i]
                cap_down = p[i] - 0.0
                if diff > 0 and cap_up > 1e-12:
                    delta = min(diff, cap_up)
                    p[i] += delta
                    diff -= delta
                elif diff < 0 and cap_down > 1e-12:
                    delta = min(-diff, cap_down)
                    p[i] -= delta
                    diff += delta
                if abs(diff) < 1e-12:
                    break
    return p

def construct_h(n, r):
    dx = 2.0/n
    x = (np.arange(n) + 0.5) * dx  # midpoints
    if r < 1e-12:
        a=0.5
        h = np.full(n, a)
    else:
        w = 2 - 2*r
        a = 1.0/(2 - r)
        h = np.empty(n)
        # left ramp region
        left_idx = x <= r
        # plateau region
        plateau_idx = (x > r) & (x <= r + w)
        # right ramp region
        right_idx = x > r + w
        # left ramp
        h[left_idx] = a * (x[left_idx] / r)
        # plateau
        h[plateau_idx] = a
        # right ramp: distance to right end: 2 - x
        h[right_idx] = a * ((2 - x[right_idx]) / r)
    # ensure within [0,1]
    h = np.clip(h, 0.0, 1.0)
    # project to sum constraints
    target_sum = n / 2
    h = project_to_sum_and_bounds(h, target_sum)
    return h

        v = np.empty(n)

import time
import numpy as np

# optional optimizer
try:
    from scipy.optimize import minimize_scalar
except Exception:  # pragma: no cover
    minimize_scalar = None


def _project_to_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {p | 0 <= p_i <= 1, sum(p)=target}.
    Uses bisection on the Lagrange multiplier λ in p_i = clip(v_i-λ,0,1).
    """
    lo, hi = -1.0, 1.0  # extreme enough to force all entries to 1 or 0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        interior = np.where((p > 1e-12) & (p < 1.0 - 1e-12))[0]
        if interior.size:
            p[interior] += diff / interior.size
            p = np.clip(p, 0.0, 1.0)
        else:  # fallback to a full projection
            p = _project_to_sum_and_bounds(p, target)
    return p


def _construct_h(n: int, r: float) -> np.ndarray:
    """
    Build a symmetric step‑function on [0,2] with two linear ramps of length r
    and a flat top (height to be chosen) so that the integral equals 1.
    The returned vector has length n, values in [0,1] and sum exactly n/2.
    """
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx  # midpoint sampling
    # shape with maximal height 1 (plateau & ramps)
    if r <= 1e-12:
        v = np.ones(n)            # pure rectangle of height 1
    else:
        w = 2.0 - 2.0 * r          # width of the flat top
        v = np.empty(n)
        left = x <= r
        v[left] = x[left] / r
        plateau = (x > r) & (x <= r + w)
        v[plateau] = 1.0
        right = x > r + w
        v[right] = (2.0 - x[right]) / r
    # Scale so that the discrete integral equals 1  (=> sum = n/2)
    target = n * 0.5
    scale = target / v.sum()
    # scale never exceeds 1 because the continuous construction already
    # satisfies the integral, and the discretisation error is tiny.
    scale = min(scale, 1.0)
    h = v * scale
    # Final projection to guarantee exact feasibility
    h = _project_to_sum_and_bounds(h, target)
    return h


def _c5_value(h: np.ndarray) -> float:
    """
    Compute the Erdős overlap constant for a given step function h.
    """
    # Use the provided helper if it exists (might be slightly more accurate)
    if "evaluate_erdos_solution" in globals():
        try:
            return float(evaluate_erdos_solution(h))
        except Exception:  # pragma: no cover
            pass
    dx = 2.0 / h.size
    # correlation implements ∑ h_i (1‑h_{i+k})
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def run(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """
    Search for a step function that yields a small Erdős C₅ value.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # 1. Choose a modest number of points (fast to evaluate, still accurate)
    n = 600                     # any even number < 1000 works well
    best_h = None
    best_c5 = np.inf

    # 2. If a parent construction is supplied, evaluate it first
    if "initial_h_values" in globals():
        try:
            init_h = np.asarray(initial_h_values, dtype=float)
            if init_h.ndim == 1 and init_h.size > 0:
                # enforce exact constraints (projection)
                init_h = _project_to_sum_and_bounds(
                    np.clip(init_h, 0.0, 1.0), n * 0.5
                )
                val = _c5_value(init_h)
                best_h, best_c5 = init_h, val
        except Exception:  # pragma: no cover
            pass

    # 3. Helper returning C5 for a given ramp length r
    def c5_for_r(r):
        h = _construct_h(n, r)
        return _c5_value(h)

    # 4. Bracket optimisation of the single parameter r ∈ [0,1]
    #    Use SciPy's bounded scalar minimiser if available, otherwise a
    #    simple grid search.
    if minimize_scalar is not None:
        res = minimize_scalar(
            c5_for_r,
            bounds=(0.0, 1.0),
            method="bounded",
            options={"xatol": 5e-4, "maxiter": 80},
        )
        r_opt = float(res.x)
        h_opt = _construct_h(n, r_opt)
        c5_opt = _c5_value(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    else:  # fallback grid search
        r_grid = np.linspace(0.0, 1.0, 301)
        for r in r_grid:
            h = _construct_h(n, r)
            val = _c5_value(h)
            if val < best_c5:
                best_h, best_c5 = h, val

    # 5. Optional short random refinement (uses remaining budget)
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    # only attempt if we still have a few seconds left
    if remaining > 2.0 and best_h is not None:
        # simple hill–climbing on the best vector
        h_cur = best_h.copy()
        c5_cur = best_c5
        n_iter = int(min(30000, remaining * 1000))  # ~1000 iterations per second
        target_sum = n * 0.5
        dx = 2.0 / n
        for _ in range(n_iter):
            # pick two distinct indices and shift a small amount δ
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.uniform(-0.02, 0.02)
            # enforce box constraints locally
            if not (0.0 <= h_cur[i] + delta <= 1.0 and 0.0 <= h_cur[j] - delta <= 1.0):
                continue
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta
            # project only if we violated the sum (should stay exact)
            # tiny numerical drift is fixed by a final projection
            h_new = _project_to_sum_and_bounds(h_new, target_sum)
            val = _c5_value(h_new)
            if val < c5_cur:
                h_cur, c5_cur = h_new, val
                if val < best_c5:
                    best_h, best_c5 = h_new, val
        # final check
        best_h, best_c5 = best_h, best_c5

    # Ensure we return a plain Python list
    return best_h.tolist(), float(best_c5), int(n)
