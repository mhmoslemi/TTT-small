def random_initial(n, total):
    # produce random vector in [0,1] with sum exactly total.
    # Approach: assign floor_total ones, one fractional remainder.
    floor_total = int(np.floor(total))
    frac = total - floor_total
    # Choose positions for ones:
    idx = np.random.choice(n, floor_total, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    if frac > 1e-12:
        # choose another index not already one
        remaining = np.setdiff1d(np.arange(n), idx)
        if remaining.size > 0:
            i_frac = np.random.choice(remaining)
            h[i_frac] = frac
        else:
            # all indices assigned ones? fallback: set first index fraction:
            h[0] = frac
    # Ensure sum correct
    # due to rounding errors, adjust small
    diff = total - h.sum()
    if abs(diff) > 1e-12:
        # add diff to random one index (ensuring within [0,1])
        i = np.random.randint(n)
        h[i] = np.clip(h[i] + diff, 0, 1)
        # if violates bounds, fix by adjusting another random index
        # Not needed for small diff
    return h

import numpy as np
import time
import math
import random

def _eval_c5(h, dx):
    """Return the max overlap integral for step function h."""
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def _random_initial(n, total):
    """Generate a random feasible vector with sum == total."""
    # start with floor(total) ones and possibly one fractional entry
    floor_tot = int(np.floor(total))
    frac = total - floor_tot
    h = np.zeros(n, dtype=np.float64)

    # choose positions for the integer ones
    ones_idx = np.random.choice(n, floor_tot, replace=False)
    h[ones_idx] = 1.0

    if frac > 1e-12:
        # pick a place not already occupied
        avail = np.setdiff1d(np.arange(n), ones_idx, assume_unique=True)
        if avail.size == 0:
            # fall back: adjust first entry (should be rare)
            h[0] = frac
        else:
            h[np.random.choice(avail)] = frac
    # small numerical correction (should be negligible)
    diff = total - h.sum()
    if abs(diff) > 1e-12:
        # add diff to a random coordinate that stays in bounds
        i = np.random.randint(n)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises
    C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # obtain initial data / problem size
    # ------------------------------------------------------------------
    try:
        # provided by the harness
        init_h = np.array(initial_h_values, dtype=np.float64)
        n = init_h.shape[0]
    except Exception:          # pragma: no cover
        # fallback – rarely needed because the platform always supplies it
        n = 95
        init_h = None

    total = n / 2.0               # required sum of h_i
    dx = 2.0 / n                  # integration step

    # ------------------------------------------------------------------
    # initial candidate
    # ------------------------------------------------------------------
    if init_h is not None:
        cur_h = init_h.copy()
        # tiny adjustment if float rounding broke the exact sum
        diff = total - cur_h.sum()
        if abs(diff) > 1e-12:
            cur_h[0] = np.clip(cur_h[0] + diff, 0.0, 1.0)
    else:
        cur_h = _random_initial(n, total)

    best_h = cur_h.copy()
    best_c5 = _eval_c5(best_h, dx)
    cur_c5 = best_c5

    # ------------------------------------------------------------------
    # parameters for the stochastic search
    # ------------------------------------------------------------------
    start = time.time()
    T0 = 0.010                     # initial temperature for simulated annealing
    max_no_improve = 2000         # steps before a restart
    stagnation = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s * 0.97:          # leave a tiny safety margin
            break
        # if we have already reached the target we can stop early
        if best_c5 <= 0.3808:
            break

        # correlation of current vector
        full_corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(np.argmax(full_corr))
        cur_c5 = full_corr[max_idx] * dx

        # remember best
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = cur_h.copy()
            stagnation = 0
        else:
            stagnation += 1

        # shift that gives the current maximum overlap
        shift = max_idx - (n - 1)   # zero shift corresponds to index n-1

        # indices that actually participate in this shift
        if shift >= 0:
            i_start, i_end = 0, n - shift
        else:
            i_start, i_end = -shift, n
        i_idx = np.arange(i_start, i_end, dtype=np.int32)
        j_idx = i_idx + shift

        # contribution of each i to the maximal overlap term
        contrib = cur_h[i_idx] * (1.0 - cur_h[j_idx])

        # pick a "bad" position (large contribution) and a "good" one (small)
        bad_local = int(np.argmax(contrib))
        good_local = int(np.argmin(contrib))
        i_bad = i_idx[bad_local]
        i_good = i_idx[good_local]

        # maximum admissible transfer without violating bounds
        delta_max = min(cur_h[i_bad], 1.0 - cur_h[i_good])
        if delta_max <= 1e-12:
            # fallback to a random feasible pair
            pos = np.where(cur_h > 1e-8)[0]
            neg = np.where(cur_h < 1.0 - 1e-8)[0]
            if pos.size == 0 or neg.size == 0:
                continue
            i_bad = int(np.random.choice(pos))
            i_good = int(np.random.choice(neg))
            delta_max = min(cur_h[i_bad], 1.0 - cur_h[i_good])
            if delta_max <= 1e-12:
                continue

        # random fraction of the allowed amount
        delta = delta_max * np.random.rand()

        # propose a new vector
        new_h = cur_h.copy()
        new_h[i_bad] -= delta
        new_h[i_good] += delta

        # evaluate the proposal
        new_c5 = _eval_c5(new_h, dx)

        accept = False

            cur_h = _random_initial(n, total)
            cur_c5 = _eval_c5(cur_h, dx)
            stagnation = 0

    # final evaluation of the best candidate
    final_c5 = _eval_c5(best_h, dx)

    # Return as required: (h_values, c5_bound, n_points)
    return (best_h, final_c5, n)
