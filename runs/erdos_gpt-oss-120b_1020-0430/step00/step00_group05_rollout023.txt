def project_box_sum(h, target, max_iter=50):
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        h_proj = np.clip(h - lam, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(h - ((lo + hi) / 2.0), 0.0, 1.0)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a step function sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def project_box_sum(x: np.ndarray, target_sum: float, iters: int = 50) -> np.ndarray:
    """
    Project a vector onto the intersection of the box [0,1]^n and the hyperplane sum(x)=target_sum.
    Uses a simple binary‑search on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def golden_balanced(N: int) -> np.ndarray:
    """Balanced 0/1 vector (exactly N/2 ones) built from a Sturmian word with slope 1/phi."""
    phi = (1.0 + math.sqrt(5.0)) * 0.5
    alpha = 1.0 / phi                     # ~0.618...
    idx = np.arange(N, dtype=np.float64)
    frac = (idx * alpha) % 1.0
    h = (frac < 0.5).astype(np.float64)

    target = N // 2
    cur = int(h.sum())
    if cur > target:
        # turn some 1's (those closest to the threshold) into 0
        ones = np.where(h == 1)[0]
        fr = frac[ones]
        order = np.argsort(fr)[::-1]      # largest fractional part first
        to_flip = ones[order[:cur - target]]
        h[to_flip] = 0.0
    elif cur < target:
        zeros = np.where(h == 0)[0]
        fr = frac[zeros]
        order = np.argsort(fr)            # smallest fractional part first
        to_flip = zeros[order[:target - cur]]
        h[to_flip] = 1.0
    return h


def hill_climb_one(start_h: np.ndarray, max_swaps: int, T0: float) -> tuple:
    """
    Simple simulated‑annealing hill‑climb on a binary vector.
    Swaps a random pair of opposite bits; accepts if improvement or
    with probability exp(-(Δ)/T) where T linearly cools to zero.
    Returns the best vector and its C5 value seen during the run.
    """
    h = start_h.copy()
    N = h.shape[0]
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    for it in range(max_swaps):
        T = T0 * (1.0 - it / max_swaps)    # linear cooling
        i, j = np.random.randint(N, size=2)
        if h[i] == h[j]:                 # swapping equal bits does nothing
            continue
        # perform the swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)

        improve = new_c < cur_c
        if improve or (T > 0 and np.random.rand() < math.exp(-(new_c - cur_c) / T)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert the swap
            h[i], h[j] = h[j], h[i]

    return best_h, best_c


def gradient_refine(h: np.ndarray, target_sum: float,
                    steps: int = 1000, lr0: float = 0.2,
                    beta0: float = 5.0, beta_max: float = 30.0) -> np.ndarray:
    """
    A few projected‑gradient steps on a smooth approximation of the max‑overlap.
    Uses a soft‑max with temperature beta that is increased during the run.
    """
    N = h.shape[0]
    for step in range(steps):
        # correlation
        corr = np.correlate(h, 1.0 - h, mode="full")          # length 2N‑1
        maxc = corr.max()
        # annealing schedule for beta
        beta = beta0 + (beta_max - beta0) * (step / steps)
        w = np.exp(beta * (corr - maxc))
        w /= w.sum()                                        # soft‑max weights

        g = 1.0 - h
        # conv1 = w * g  (linear convolution)
        conv1 = np.convolve(w, g)[N - 1:N - 1 + N]
        # conv2 = rev(w) * h
        conv2 = np.convolve(w[::-1], h)[N - 1:N - 1 + N]

        grad = conv1 - conv2
        # simple learning‑rate schedule
        lr = lr0 * (1.0 - step / steps)
        h = h - lr * grad
        h = project_box_sum(h, target_sum)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where h_values is a list of
    samples of the step function on a uniform grid of length n_points.
    The algorithm builds a golden‑ratio Sturmian word, improves it by
    simulated‑annealing swaps and a brief gradient refinement.
    """
    start_time = time.time()
    np.random.seed(seed)

    # ---- parameters -------------------------------------------------
    N = 256                                 # grid size (< 1000 for speed)
    target_sum = N / 2.0

    # ---- initial construction ---------------------------------------
    # primary deterministic start (Sturmian word)
    best_h = golden_balanced(N)
    best_c = compute_c5(best_h)

    # if a parent solution exists, incorporate it (interpolation + projection)
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=np.float64)
        if parent.size != N:
            # linear interpolation onto the N‑point grid
            x_old = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            parent = np.interp(x_new, x_old, parent)
        parent = np.clip(parent, 0.0, 1.0)
        parent = project_box_sum(parent, target_sum)
        pc = compute_c5(parent)
        if pc < best_c:
            best_h, best_c = parent, pc

    # ---- stochastic multi‑start hill climbing -----------------------
    # allocate most of the time to random swaps
    while time.time() - start_time < 0.88 * budget_s:
        # small random perturbation of the deterministic start:
        offset = np.random.rand()                     # random phase for the Sturmian word
        idx = np.arange(N, dtype=np.float64)
        alpha = (math.sqrt(5.0) - 1.0) * 0.5           # = 1/phi
        frac = (idx * alpha + offset) % 1.0
        h0 = (frac < 0.5).astype(np.float64)
        # balance to exact N/2 ones
        cur = int(h0.sum())
        if cur != int(target_sum):
            diff = int(target_sum) - cur
            if diff > 0:
                zeros = np.where(h0 == 0)[0]
                order = np.argsort(frac[zeros])
                h0[zeros[order[:diff]]] = 1.0
            else:
                ones = np.where(h0 == 1)[0]
                order = np.argsort(frac[ones])[::-1]
                h0[ones[order[: -diff]]] = 0.0

        # a modest number of swaps; the exact count adapts to remaining time
        remaining = budget_s - (time.time() - start_time)
        max_swaps = int(min(30000, remaining * 6000))   # ~6000 swaps per second is safe
        if max_swaps < 2000:
            max_swaps = 2000
        h_local, c_local = hill_climb_one(h0, max_swaps, T0=0.02)
        if c_local < best_c:
            best_c, best_h = c_local, h_local.copy()

    # ---- final gradient refinement ----------------------------------
    # use the best binary vector found as a warm start for a few gradient steps
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5.0:
        steps = int(min(2000, time_left * 300))   # a few hundred steps per second
        best_h = gradient_refine(best_h, target_sum,
                                 steps=steps,
                                 lr0=0.15,
                                 beta0=8.0,
                                 beta_max=30.0)
        best_c = compute_c5(best_h)

    # final safety projection (might be slightly off due to numerical noise)
    best_h = project_box_sum(best_h, target_sum)
    best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), N
