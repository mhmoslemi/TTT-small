import numpy as np
import cvxpy as cp

def _compute_c5(h):
    # helper that uses the pre‑imported evaluator when available,
    # otherwise falls back to the direct numpy implementation.
    try:
        return evaluate_erdos_solution(h)
    except Exception:
        n = len(h)
        dx = 2.0 / n
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    dx = 2.0 / n
    half = n // 2

    best_h = None
    best_score = float("inf")

    # ---------- try MILP (binary h) ----------
    try:
        # remaining time for the solver (leave a safety margin)
        remaining = max(0.0, budget_s - (time.time() - start_time) - 2.0)
        # variables
        h = cp.Variable(n, boolean=True)
        w = cp.Variable((n, n))                # continuous product variables
        t = cp.Variable(nonneg=True)

        constr = []

        # integral condition
        constr.append(cp.sum(h) == half)

        # bounds on w
        constr.append(w >= 0)
        constr.append(w <= 1)

        # product envelope
        for i in range(n):
            constr.append(w[i, i] == h[i])
            for j in range(i + 1, n):
                constr.append(w[i, j] <= h[i])
                constr.append(w[i, j] <= h[j])
                constr.append(w[i, j] >= h[i] + h[j] - 1)
