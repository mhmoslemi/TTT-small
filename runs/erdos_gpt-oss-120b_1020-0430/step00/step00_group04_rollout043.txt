import numpy as np, time, math, random
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair(h):
    # ensure bounds and sum = n/2
    n = len(h)
    target = n/2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    # Simple water-filling to adjust sum while staying bounds
    # Iterate scaling and clipping
    for _ in range(100):
        if abs(s - target) < 1e-10:
            break
        delta = (target - s) / n
        h += delta
        # clip again
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
    # final correction if still off due to clipping: redistribute residual proportionally
    if abs(h.sum() - target) > 1e-9:
        # adjust by adding residual to first unfixed entries
        residual = target - h.sum()
        # distribute to entries not at bounds
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            inc = residual / free.sum()
            h[free] += inc
            np.clip(h, 0.0, 1.0, out=h)
    return h

def random_feasible(n, rng):
    # generate random vector in [0,1] with sum=n/2
    # start with uniform distribution then project
    h = rng.random(n)
    h = repair(h)
    return h

def improve(h, rng, time_limit):
    n = len(h)
    best = h.copy()
    best_score = compute_c5(best)
    current = best.copy()
    cur_score = best_score
    start = time.time()
    # annealing schedule
    T0 = 0.5
    iter = 0
    max_iter = 1000000
    while time.time() - start < time_limit:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # propose moving a small amount from j to i
        inc_i = 1.0 - current[i]
        dec_j = current[j]
        max_delta = min(inc_i, dec_j)
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta
        # apply move
        current[i] += delta
        current[j] -= delta
        new_score = compute_c5(current)
        if new_score < best_score:
            best_score = new_score
            best = current.copy()
        # acceptance
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / (T0 * max(1e-8, 1.0 - iter / max_iter))):
            cur_score = new_score
        else:
            # revert
            current[i] -= delta
            current[j] += delta
        iter += 1
        if iter % 5000 == 0:
            # optionally reduce temperature
            pass
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    n = kwargs.get('n_points')
    if n is None:
        # try to infer length from initial_h_values if present
        try:
            n = len(initial_h_values)
        except NameError:
            n = 42
    # initialize
    try:
        h0 = np.array(initial_h_values, dtype=float)
        # ensure feasibility
        h0 = repair(h0)
    except NameError:
        h0 = random_feasible(n, rng)
    # allocate time
    start = time.time()
    remain = budget_s - (time.time() - start)
    best_h, best_score = improve(h0, rng, remain*0.9)
    # final evaluation
    c5 = compute_c5(best_h)
    return best_h, c5, len(best_h)

import numpy as np
import time
import math

# -------------------------------------------------------------
# Helper: compute the objective C5 for a given step function.
# -------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


# -------------------------------------------------------------
# Projection onto the feasible set:
#   0 ≤ h_i ≤ 1   and   sum(h) = n/2
# -------------------------------------------------------------
def project_feasible(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    target = n / 2.0
    h = np.clip(x, 0.0, 1.0)

    # simple water‑filling: repeatedly add a constant then clip
    for _ in range(20):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        delta = (target - s) / n
        h += delta
        np.clip(h, 0.0, 1.0, out=h)

    # final tiny correction if needed
    s = h.sum()
    if abs(s - target) > 1e-10:
        resid = target - s
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            inc = resid / free.sum()
            h[free] += inc
            np.clip(h, 0.0, 1.0, out=h)
        else:
            # fall back: adjust the first entry
            h[0] = np.clip(h[0] + resid, 0.0, 1.0)
    return h

# Generate a random feasible vector.
# -------------------------------------------------------------
def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    # start from uniform random numbers and project
    x = rng.random(n)
    return project_feasible(x)


# -------------------------------------------------------------
# Local search / simulated‑annealing style improvement.
# -------------------------------------------------------------
def improve(h_start: np.ndarray, rng: np.random.Generator, time_limit: float):
    n = h_start.shape[0]
    best = project_feasible(h_start.copy())
    best_score = compute_c5(best)

    cur = best.copy()
    cur_score = best_score

    start = time.time()
    it = 0
    max_iter = 2_000_000  # safety upper bound

    while time.time() - start < time_limit and it < max_iter:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)

        # try moving mass from j to i
        inc_i = 1.0 - cur[i]
        dec_j = cur[j]
        max_delta = min(inc_i, dec_j)

        if max_delta <= 1e-12:          # try the opposite direction
            inc_i = cur[i]
            dec_j = 1.0 - cur[j]
            max_delta = min(inc_i, dec_j)

                it += 1
                continue
            delta = rng.random() * max_delta
            cur[i] -= delta
            cur[j] += delta

            delta = rng.random() * max_delta
            cur[i] += delta
            cur[j] -= delta
