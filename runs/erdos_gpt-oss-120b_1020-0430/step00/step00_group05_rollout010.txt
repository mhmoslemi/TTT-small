def evaluate_c5(h, dx):
    n = len(h)
    g_rev = (1 - h)[::-1]
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g_rev, n=L)
    conv = np.fft.irfft(H * G, n=L)[:2*n-1]
    max_val = np.max(conv)
    return max_val * dx

import time
import numpy as np

# ---------- Helper utilities ----------
def compute_correlation_full(h):
    """
    Linear correlation of h with (1-h).
    Returns integer array of length 2*len(h)-1.
    """
    n = h.shape[0]
    g_rev = (1 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g_rev, n=L)
    conv = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    # rounding fixes tiny fft errors
    return np.rint(conv).astype(int)


def max_mismatch_and_shift(h):
    """Return (max_mismatch, shift_index) where shift = idx-(n-1)."""
    conv = compute_correlation_full(h)
    idx = int(np.argmax(conv))
    return int(conv[idx]), idx - (h.shape[0] - 1)


def mismatching_positions(h, shift):
    """Indices i where h[i]=1 and (i+shift) contributes a mismatch."""
    n = h.shape[0]
    if shift >= 0:
        # overlapping part
        part = np.where((h[: n - shift] == 1) & (h[shift:] == 0))[0]
        # tail part (out of range)
        tail_start = n - shift
        tail = np.where(h[tail_start:] == 1)[0] + tail_start
        return np.concatenate((part, tail))
    else:
        s = -shift
        # overlapping part
        part = np.where((h[s:] == 1) & (h[: n - s] == 0))[0] + s
        # head part (out of range)
        head = np.where(h[:s] == 1)[0]
        return np.concatenate((part, head))


def good_zero_positions(h, shift):
    """Zero positions j that would NOT create a new mismatch for this shift."""
    n = h.shape[0]
    zeros = np.where(h == 0)[0]
    if shift >= 0:
        cond = (zeros < n - shift) & (h[zeros + shift] == 1)
    else:
        s = -shift
        cond = (zeros >= s) & (h[zeros - s] == 1)
    return zeros[cond]


# ---------- Main optimisation routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # target: C5 <= 0.3808   →  max_mismatch <= floor(0.3808 / (2/n))
    n = 256                                 # even, power of two ⇒ fast FFT
    dx = 2.0 / n
    target_max = int(np.floor(0.3808 / dx + 1e-9))

    # ----- initialise a feasible binary vector -----
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values).astype(float)
        # up‑sample / repeat to length n
        rep = int(np.ceil(n / init.size))
        h = np.repeat(init, rep)[:n]
        h = (h >= 0.5).astype(int)
    else:
        h = np.zeros(n, dtype=int)
        h[: n // 2] = 1
        rng.shuffle(h)

    # enforce exact weight n/2
    cur_sum = h.sum()
    target_sum = n // 2
    while cur_sum > target_sum:
        i = rng.choice(np.where(h == 1)[0])
        h[i] = 0
        cur_sum -= 1
    while cur_sum < target_sum:
        i = rng.choice(np.where(h == 0)[0])
        h[i] = 1
        cur_sum += 1

    best_h = h.copy()
    best_max, _ = max_mismatch_and_shift(best_h)
    best_c5 = best_max * dx

    # ----- local search with greedy swaps -----
    cur_h = best_h.copy()
    cur_max, cur_shift = max_mismatch_and_shift(cur_h)

    # parameters controlling the search
    no_improve_limit = 2000
    no_improve = 0

    while time.time() - start < budget_s * 0.96:
        # if already good enough, stop
        if cur_max <= target_max:
            best_h = cur_h.copy()
            best_c5 = cur_max * dx
            break

        # try to fix the worst shift
        mismatches = mismatching_positions(cur_h, cur_shift)
        if mismatches.size == 0:               # extremely unlikely
            i = rng.choice(np.where(cur_h == 1)[0])
        else:
            i = rng.choice(mismatches)

        good_zeros = good_zero_positions(cur_h, cur_shift)
        if good_zeros.size == 0:
            j = rng.choice(np.where(cur_h == 0)[0])
        else:
            j = rng.choice(good_zeros)

        # perform the tentative swap
        cur_h[i], cur_h[j] = 0, 1
        new_max, new_shift = max_mismatch_and_shift(cur_h)

        if new_max <= cur_max:                 # accept if not worse

            no_improve = 0
            if new_max * dx < best_c5:
                best_c5 = new_max * dx
                best_h = cur_h.copy()
        else:

            no_improve += 1

            cur_h = best_h.copy()
            # flip a handful of bits to escape local minima

            no_improve = 0

    # final conversion to float array as required by the harness
    return (best_h.astype(float), float(best_c5), int(n))
