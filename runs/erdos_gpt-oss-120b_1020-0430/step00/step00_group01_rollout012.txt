import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def _project_to_sum_and_bounds(v, target):
    """Project v onto the box [0,1]^n while forcing sum(v)=target."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _bound(h):
    """Return C5 bound for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _gradient_of_worst_shift(h):
    """Gradient (subgradient) of the maximal overlap term."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # corresponds to lag = shift
    g = np.ones(n)
    if shift >= 0:
        if shift < n:
            g[: n - shift] -= h[shift:]
            g[shift:] -= h[: n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] -= h[: n - s]
            g[: n - s] -= h[s:]
    return g, shift, corr[idx]


def _descent_optimize(h0, target, time_limit):
    """Simple projected subgradient descent for the minimax objective."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_val = _bound(h)
    step = 0.5
    no_improve = 0
    max_iter = 4000

    while max_iter and (time.time() < time_limit):
        max_iter -= 1
        g, _, max_corr = _gradient_of_worst_shift(h)
        # evaluate current bound
        cur_val = max_corr * (2.0 / n)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # take a step and project back to feasibility
        h = _project_to_sum_and_bounds(h - step * g, target)

        # adapt step size if stuck
        if no_improve >= 200:
            step *= 0.5
            no_improve = 0
            if step < 1e-5:
                break
    return best_h, best_val


def _refine_with_slsqp(h0, target, time_limit):
    """A quick SLSQP refinement (optional, may be skipped if time low)."""
    n = h0.shape[0]
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    # use a small tolerance to avoid excessive iterations
    opts = {"maxiter": 200, "ftol": 1e-8, "disp": False}
    res = minimize(_bound, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options=opts)
    if res.success and time.time() < time_limit:
        return res.x, _bound(res.x)
    return h0, _bound(h0)


def _random_feasible(n, target):
    """Generate a random point in [0,1]^n with prescribed sum."""
    v = np.random.rand(n)
    return _project_to_sum_and_bounds(v, target)


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.99  # little safety margin

    # keep global best
    best_h = None
    best_val = math.inf

    # ----- use provided initial vector if available -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        h0 = _project_to_sum_and_bounds(h0, n0 / 2.0)
        val0 = _bound(h0)
        best_h, best_val = h0, val0

    # candidate grid sizes (doubling scheme)
    grid_sizes = [47, 94, 188, 376]
    # ensure we don't exceed time before starting
    for n in grid_sizes:
        if time.time() > time_limit:
            break
        target = n / 2.0
        # how many restarts for this size (more for larger n)
        restarts = 12 if n <= 188 else 8

        for r in range(restarts):
            if time.time() > time_limit:
                break

            # initial construction for this restart
            if best_h is not None and best_h.shape[0] == n and r == 0:
                h_start = best_h.copy()
            elif best_h is not None and best_h.shape[0] != n:
                # interpolate previous best onto new grid
                x_old = np.linspace(0.0, 2.0, best_h.shape[0], endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h_start = np.interp(x_new, x_old, best_h)
                # add tiny perturbation to escape exact linearity
                h_start += np.random.randn(n) * 0.02
            else:
                # pure random start
                h_start = _random_feasible(n, target)

            h_start = _project_to_sum_and_bounds(h_start, target)

            h_opt, val_opt = _descent_optimize(
                h_start, target, time_limit

            if time.time() < time_limit:

                    h_opt, target, time_limit
                )

            # keep global best
            if val_opt < best_val:
                best_h, best_val = h_opt.copy(), val_opt

    # final safety: ensure feasibility
    if best_h is not None:
        best_h = _project_to_sum_and_bounds(best_h, best_h.shape[0] / 2.0)
        best_val = _bound(best_h)

    # Return as plain Python list for compatibility
    if best_h is None:
        raise RuntimeError("Failed to produce a solution.")
    return best_h.tolist(), float(best_val), best_h.shape[0]
