def project(v, target):
    # ensures 0<=v<=1 and sum=v = target
    v = np.clip(v, 0, 1)
    max_iter = 10
    for _ in range(max_iter):
        sum_v = v.sum()
        if abs(sum_v - target) < 1e-9:
            break
        # compute correction to distribute to non-bound entries
        free = (v > 0 + 1e-12) & (v < 1 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            # cannot adjust, break
            # fallback to shift minimal slack
            break
        delta = (target - sum_v) / nfree
        v[free] += delta
        v = np.clip(v, 0, 1)
    # final clip
    v = np.clip(v, 0, 1)
    # If still not meeting sum due to rounding, adjust minimal values
    sum_v = v.sum()
    if abs(sum_v - target) > 1e-6:
        # adjust uniformly
        v += (target - sum_v) / len(v)
        v = np.clip(v, 0, 1)
        # final fudge: adjust first few entries
        diff = target - v.sum()
        if abs(diff) > 1e-6:
            # add diff to a random interior element if possible
            idx = np.argmax(v < 1) if diff > 0 else np.argmax(v > 0)
            v[idx] = np.clip(v[idx] + diff, 0, 1)
    return v

import time

def run(seed=42, budget_s=1100):
    start = time.time()
    np.random.seed(seed)
    n = 200  # make even
    # Start with random binary vector with n/2 ones
    h = np.zeros(n, dtype=np.float64)
    ones_idx = np.random.choice(n, n//2, replace=False)
    h[ones_idx] = 1.0

    dx = 2.0 / n

    max_val, max_idx = compute_max_corr(h)
    best_score = max_val * dx
    best_h = h.copy()

    temperature = 0.1  # initial temperature
    cooling = 0.9995

    iteration = 0
    while time.time() - start < budget_s * 0.9: # leave slack
        iteration += 1
        # compute current max shift
        max_val, max_idx = compute_max_corr(h)
        cur_score = max_val * dx
        # early exit if we achieve target
        if cur_score <= 0.3808:
            best_score = cur_score
            best_h = h.copy()
            break
        # target shift lag
        lag = max_idx - (n - 1)
        # overlapping range
        start_i = max(0, -lag)
        end_i = min(n, n - lag)  # exclusive
        # find mismatches
        segment = slice(start_i, end_i)
        # vector of h[segment] and h[segment + lag]
        i_range = np.arange(start_i, end_i)
        j_range = i_range + lag  # same shape
        h_i = h[i_range]
        h_j = h[j_range]
        mism_10 = i_range[(h_i == 1) & (h_j == 0)]
        mism_01 = i_range[(h_i == 0) & (h_j == 1)]

        if len(mism_10) > 0 and len(mism_01) > 0:
            i_swap = np.random.choice(mism_10)
            j_swap = np.random.choice(mism_01)
        else:
            # random swap: pick random ones and zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i_swap = np.random.choice(ones)
            j_swap = np.random.choice(zeros)

        # perform swap
        h[i_swap], h[j_swap] = h[j_swap], h[i_swap]

        # evaluate new score
        new_val, _ = compute_max_corr(h)
        new_score = new_val * dx
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            prob = np.exp(-delta / temperature) if temperature > 1e-12 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # revert swap
            h[i_swap], h[j_swap] = h[j_swap], h[i_swap]

        # cool down
        temperature *= cooling
        if iteration % 1000 == 0:
            # optional: compute score again for best?
            pass

    return best_h, best_score, n

def project_to_sum(v, target):
    v = np.clip(v, 0, 1)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    # Simple water-filling algorithm: adjust values within bounds
    # We'll use iterative approach: adjust free entries equally.
    free = np.ones_like(v, dtype=bool)  # all entries initially can be adjusted
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # number of entries that are not at bounds
        free = (v > 1e-12) & (v < 1 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            # distribute diff equally, but will hit bounds; break
            break
        delta = diff / nfree
        v[free] += delta
        v = np.clip(v, 0, 1)
    # final correction if needed
    s = v.sum()
    if abs(s - target) > 1e-6:
        # adjust first entry a little
        diff = target - s
        # find any index not at bound extreme
        idx = np.argmin(np.abs(v - 0.5))
        v[idx] = np.clip(v[idx] + diff, 0, 1)
    return v

import time
import numpy as np

def _project_to_sum(v, target, tol=1e-12, max_iter=20):
    """Clamp v to [0,1] and adjust to meet the sum constraint."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        if abs(s - target) <= tol:
            break
        free = (v > 0.0) & (v < 1.0)
        nfree = free.sum()
        if nfree == 0:
            break
        delta = (target - s) / nfree
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if still off
    diff = target - v.sum()
    if abs(diff) > 1e-9:
        idx = np.argmin(np.abs(v - 0.5))
        v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
    return v


def _max_corr_and_idx(h):
    """Return max correlation value (without dx) and its index for linear correlation."""
    n = h.size
    b = 1.0 - h
    brev = b[::-1]
    L = 1 << ((2 * n - 2).bit_length())   # next power of two >= 2n-1
    fa = np.fft.rfft(h, L)
    fb = np.fft.rfft(brev, L)
    conv = np.fft.irfft(fa * fb, L)[: 2 * n - 1]
    max_idx = np.argmax(conv)
    return conv[max_idx], max_idx


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with a low Erdős C_5 constant.

    Returns
    -------
    h_values : np.ndarray
        Array of samples (length n_points) in [0,1] satisfying the integral constraint.
    c5_bound : float
        The achieved upper bound C_5 = max_k ∫ h(x)(1‑h(x+k))dx.
    n_points : int
        Number of discretisation points (length of h_values).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    margin = 0.95 * budget_s          # keep a safety margin for finalisation

    # ----------------------------------------------------------------------
    # Initialise the problem size and starting vector
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        # use the provided construction (its length is even in the benchmark)
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
        target = n / 2.0
        h = _project_to_sum(h0, target)
    else:
        # default size – a moderate even number that gives enough freedom
        n = 200
        target = n / 2.0
        # random binary start (exactly n/2 ones)
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    dx = 2.0 / n
    # ----------------------------------------------------------------------
    # Initial evaluation
    # ----------------------------------------------------------------------
    cur_max, cur_idx = _max_corr_and_idx(h)
    cur_score = cur_max * dx
    best_h = h.copy()
    best_score = cur_score

    # ----------------------------------------------------------------------
    # Main local‑search loop
    # ----------------------------------------------------------------------
    iter_count = 0
    # Pre‑computed index lists for fast sampling
    ones_idx = np.flatnonzero(h == 1.0)
    zeros_idx = np.flatnonzero(h == 0.0)

    while time.time() - start_time < margin:
        iter_count += 1

        # recompute because the vector may have changed
        cur_max, cur_idx = _max_corr_and_idx(h)
        cur_score = cur_max * dx

        # record the best seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
            # early exit if we already beat the target bound
            if best_score <= 0.3808:
                break

        # Identify the lag that currently dominates the max correlation
        lag = cur_idx - (n - 1)      # lag in the range [-(n-1), n-1]

        # Determine mismatched positions for this lag (where a 1 aligns with a 0)
        if lag >= 0:
            i_range = np.arange(0, n - lag)
        else:
            i_range = np.arange(-lag, n)
        j_range = i_range + lag

        hi = h[i_range]
        hj = h[j_range]
        mism_10 = i_range[(hi == 1.0) & (hj == 0.0)]

        # Choose a 1 participating in the worst lag (if any), otherwise any 1
        if mism_10.size:
            i_swap = rng.choice(mism_10)
        else:
            i_swap = rng.choice(ones_idx)

        # Choose a 0 somewhere (prefer one that would also help the same lag)
        # We look for an index where h=0 and the shifted partner is 1 (the opposite mismatch)
        opp_mism = i_range[(hi == 0.0) & (hj == 1.0)]
        if opp_mism.size:
            j_swap = rng.choice(opp_mism)
        else:
            j_swap = rng.choice(zeros_idx)

        # Perform the swap
        h[i_swap], h[j_swap] = h[j_swap], h[i_swap]

        # Evaluate the new configuration
        new_max, _ = _max_corr_and_idx(h)
        new_score = new_max * dx

        # Accept only if we do not increase the objective
        if new_score <= cur_score:
            # update the index lists efficiently
            if h[i_swap] == 1.0:
                ones_idx = np.append(ones_idx[ones_idx != i_swap], i_swap)
                zeros_idx = np.append(zeros_idx[zeros_idx != i_swap], i_swap)
            else:
                zeros_idx = np.append(zeros_idx[zeros_idx != i_swap], i_swap)

                ones_idx = np.append(ones_idx[ones_idx != j_swap], j_swap)
                zeros_idx = np.append(zeros_idx[zeros_idx != j_swap], j_swap)
            else:
                zeros_idx = np.append(zeros_idx[zeros_idx != j_swap], j_swap)
                ones_idx = np.append(ones_idx[ones_idx != j_swap], j_swap)

            # keep swapped version for next iteration
            cur_score = new_score
            cur_max = new_max
        else:
            # revert the swap
            h[i_swap], h[j_swap] = h[j_swap], h[i_swap]

        # occasional diversification to avoid stagnation
        if iter_count % 5000 == 0:
            # if no improvement in a while, shuffle a small fraction
            if rng.random() < 0.3:

                k = max(2, n // 20)

                idx_zero = rng.choice(zeros_idx, k, replace=False)
                h[idx_one] = 0.0
                h[idx_zero] = 1.0
                # update the index arrays
                ones_idx = np.setdiff1d(ones_idx, idx_one, assume_unique=True)
                zeros_idx = np.setdiff1d(zeros_idx, idx_zero, assume_unique=True)
                ones_idx = np.concatenate([ones_idx, idx_zero])
                zeros_idx = np.concatenate([zeros_idx, idx_one])

    # ----------------------------------------------------------------------
    # Return the best found solution
    # ----------------------------------------------------------------------
    return best_h, float(best_score), n
