def compute_c5_vec(h):
    N = h.size
    m = 1 << ((2*N-1).bit_length())
    H = np.fft.rfft(h, m)
    Z = np.fft.rfft(1-h, m)
    conv = np.fft.irfft(H*Z, m)[:2*N-1]

    and the full correlation array."""
    H = np.fft.rfft(h, m)
    conv = np.fft.irfft(H * (ones_fft - H), m)[:2 * len(h) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # -------------------------------------------------------------
    # Determine problem size and initial vector
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        N = h0.size
        # enforce binary and exact half-sum if needed
        if not np.all(np.isin(h0, [0.0, 1.0])):
            # round to binary while keeping exactly N/2 ones
            target_ones = N // 2
            h0 = _project_binary(h0, target_ones)
    else:
        N = 200                       # a modest size, fast to evaluate
        target_ones = N // 2
        h0 = np.zeros(N, dtype=np.float64)
        h0[rng.choice(N, size=target_ones, replace=False)] = 1.0

    # ensure exact sum
    target_sum = N // 2
    if int(h0.sum()) != target_sum:
        # repair via binary projection
        h0 = _project_binary(h0, target_sum)

    # -------------------------------------------------------------
    # Pre‑computations for fast convolutions
    # -------------------------------------------------------------
    m = 1 << ((2 * N - 1).bit_length())   # next power of two ≥ 2N‑1
    ones_fft = np.fft.rfft(np.ones(N), m)
    dx = 2.0 / N

    # -------------------------------------------------------------
    # Helper to locate mismatched positions for a given lag
    # -------------------------------------------------------------
    def mismatched_positions(h_vec, lag):
        """Return indices of 1→0 mismatches and 0→1 candidates for a lag."""
        if lag >= 0:
            L = N - lag
            ones_mask = (h_vec[:L] == 1) & (h_vec[lag:] == 0)
            cand_mask = (h_vec[:L] == 0) & (h_vec[lag:] == 1)
            ones_idx = np.where(ones_mask)[0]          # positions i where h[i]=1,h[i+lag]=0
            cand_idx = np.where(cand_mask)[0]          # positions i where h[i]=0,h[i+lag]=1
        else:
            d = -lag
            L = N - d
            ones_mask = (h_vec[d:] == 1) & (h_vec[:L] == 0)
            cand_mask = (h_vec[d:] == 0) & (h_vec[:L] == 1)
            ones_idx = np.where(ones_mask)[0] + d      # shift back to original positions
            cand_idx = np.where(cand_mask)[0] + d
        return ones_idx, cand_idx

    # -------------------------------------------------------------
    # Core optimisation loop (targeted hill‑climbing + restarts)
    # -------------------------------------------------------------
    best_h = h0.copy()
    best_val, _, _ = _max_corr_and_lag(best_h, ones_fft, m, dx)
    best_val *= dx

    cur_h = best_h.copy()
    cur_val, cur_idx, cur_corr = _max_corr_and_lag(cur_h, ones_fft, m, dx)
    cur_val *= dx

    no_improve = 0
    max_no_improve = 5000         # trigger a restart after this many futile attempts
    max_total_iters = 200000      # safety bound
    it = 0

    while time.time() - start < budget_s * 0.92 and it < max_total_iters:
        # Identify the lag that currently dominates
        lag = cur_idx - (N - 1)

        # Find mismatched 1→0 positions for this lag
        ones_mis, cand_mis = mismatched_positions(cur_h, lag)

        if ones_mis.size == 0 or cand_mis.size == 0:
            # no direct improvement possible for this lag – random restart
            cur_h = np.zeros(N, dtype=np.float64)
            cur_h[rng.choice(N, size=target_sum, replace=False)] = 1.0
            cur_val, cur_idx, cur_corr = _max_corr_and_lag(cur_h, ones_fft, m, dx)
            cur_val *= dx
            no_improve = 0
            it += 1
            continue

        # Try a few random focused swaps
        improved = False
        for _ in range(5):
            i_one = rng.choice(ones_mis)
            i_zero = rng.choice(cand_mis)

            # perform the swap
            cur_h[i_one] = 0.0
            cur_h[i_zero] = 1.0

            new_val_raw, new_idx, new_corr = _max_corr_and_lag(cur_h, ones_fft, m, dx)
            new_val = new_val_raw * dx

            if new_val < cur_val - 1e-12:
                # accept
                cur_val = new_val
                cur_idx = new_idx
                cur_corr = new_corr
                improved = True
                if new_val < best_val:
                    best_val = new_val
                    best_h = cur_h.copy()
                no_improve = 0
                break
            else:
                # revert
                cur_h[i_one] = 1.0
                cur_h[i_zero] = 0.0

        if not improved:
            no_improve += 1
            # occasionally perform a non‑focused random swap to shake the state
            if no_improve % 100 == 0:
                # random 1↔0 swap
                ones_idx = np.flatnonzero(cur_h)
                zeros_idx = np.flatnonzero(1 - cur_h)
                i_one = rng.choice(ones_idx)
                i_zero = rng.choice(zeros_idx)
                cur_h[i_one] = 0.0
                cur_h[i_zero] = 1.0
                cur_val_raw, cur_idx, cur_corr = _max_corr_and_lag(cur_h, ones_fft, m, dx)
                cur_val = cur_val_raw * dx

        # If we have stagnated for too long, restart from a fresh random configuration
        if no_improve >= max_no_improve:
            cur_h = np.zeros(N, dtype=np.float64)
            cur_h[rng.choice(N, size=target_sum, replace=False)] = 1.0
            cur_val, cur_idx, cur_corr = _max_corr_and_lag(cur_h, ones_fft, m, dx)
            cur_val *= dx
            no_improve = 0

        it += 1

    # final safety evaluation
    final_val_raw, _, _ = _max_corr_and_lag(best_h, ones_fft, m, dx)
    final_val = final_val_raw * dx
    return best_h, float(final_val), int(N)
