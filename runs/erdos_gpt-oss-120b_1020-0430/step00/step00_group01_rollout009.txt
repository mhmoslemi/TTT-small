def overlap_array(h):
    n = len(h)
    L = 1 << ((2*n-1).bit_length())   # next power of 2 >= 2n-1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1-h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[:2*n-1]
    # due to round-off errors, take real part
    return corr * (2.0 / n)   # dx = 2/n

def compute_b(h, shift):
    n = len(h)
    b = np.zeros(n)
    if shift >= 0:
        # pairs (i, i+shift)
        # For i indices i in [0, n-shift-1], we have pair (i,i+shift)
        # b[i] gets h[i+shift], b[i+shift] gets h[i]
        b[:n-shift] += h[shift:]   # b[i] += h[i+shift]

        b[:n-s] += h[s:]   # b[i] (i<n-s) adds h[i+s]

def b_vector(h, shift):
    n = h.shape[0]
    b = np.zeros_like(h)
    if shift >= 0:
        m = n - shift
        if m > 0:
            b[:m] += h[shift:]
            b[shift:] += h[:m]
    else:
        s = -shift
        m = n - s
        if m > 0:
            b[s:] += h[:m]
            b[:m] += h[s:]
    return b

def optimize_h(h, budget_remaining, initial_step=0.05, min_step=1e-5):
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_val, _ = compute_c5(h)
    step = initial_step
    t0 = time.time()
    it = 0
    # termination based on time or max iterations maybe 1e7
    while True:
        elapsed = time.time() - t0
        if elapsed >= budget_remaining * 0.95:
            break
        # also break if step small
        if step < min_step:
            break
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # compute max shift index
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        # compute benefit vector
        b = b_vector(h, shift)
        # indices capable of increase and decrease
        inc_mask = h < 1.0 - 1e-12
        dec_mask = h > 1e-12
        # if no capacities, break
        if not np.any(inc_mask) or not np.any(dec_mask):
            break
        # choose inc and dec
        inc_candidates = np.where(inc_mask)[0]
        dec_candidates = np.where(dec_mask)[0]
        inc_idx = inc_candidates[np.argmax(b[inc_candidates])]
        dec_idx = dec_candidates[np.argmin(b[dec_candidates])]
        # compute delta
        delta = min(step, 1.0 - h[inc_idx], h[dec_idx])
        if delta <= 0:
            # can't move, maybe adjust step
            step *= 0.5
            continue
        h_new = h.copy()
        h_new[inc_idx] += delta
        h_new[dec_idx] -= delta
        # compute new max overlap
        new_val, _ = compute_c5(h_new)
        if new_val + 1e-12 < best_val:
            # Improvement
            h = h_new
            best_h = h_new.copy()
            best_val = new_val
        else:
            # no improvement, maybe accept small probability?
            # simple: if new_val < best_val + tolerance (maybe ignore)
            # else we could try random move or keep current.
            # We'll revert and maybe reduce step.
            step *= 0.99
        # optionally random acceptance to escape
        # maintain step size
        # maybe periodically random perturbation
        if it % 1000 == 0 and np.random.rand() < 0.1:
            # random move: select random i,j and delta random
            i, j = np.random.choice(n, size=2, replace=False)
            d = min(step, 1 - h[i], h[j])
            if d > 0:
                h_tmp = h.copy()
                h_tmp[i] += d

                if v_tmp < best_val:
                    h = h_tmp
                    best_h = h_tmp.copy()
                    best_val = v_tmp
        it += 1
        # slowly reduce step size

import time
import numpy as np

def _c5_and_corr(h):
    """Return (C5, full correlation array) for step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max()), corr

def _benefit_vector(h, shift):
    """Gradient benefit b_i = h[i+shift] + h[i-shift] (zero where out of range)."""
    n = h.size
    b = np.zeros_like(h)
    if shift >= 0:
        m = n - shift
        if m > 0:
            b[:m] += h[shift:]
            b[shift:] += h[:m]
    else:                                 # shift < 0
        s = -shift
        m = n - s
        if m > 0:
            b[s:] += h[:m]
            b[:m] += h[s:]
    return b

def _local_optimize(h, budget, init_step=0.05, min_step=1e-5):
    """Hill‑climb using pairwise exchanges, respecting sum & bounds."""
    n = h.size
    target_sum = n / 2.0                     # because ∫h = 1  and dx = 2/n
    best_h = h.copy()
    best_val, _ = _c5_and_corr(best_h)
    step = init_step
    start = time.time()
    it = 0
    while True:
        if time.time() - start > budget * 0.95:
            break
        if step < min_step:
            break

        # current worst shift
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        # benefit for each coordinate w.r.t. this shift
        b = _benefit_vector(best_h, shift)

        inc_mask = best_h < 1.0 - 1e-12
        dec_mask = best_h > 1e-12
        if not inc_mask.any() or not dec_mask.any():
            break

        inc_idx = np.argmax(b * inc_mask)       # biggest benefit among increasable
        dec_idx = np.argmin(b + (~dec_mask) * 1e6)  # smallest benefit among decreasable

        delta = min(step, 1.0 - best_h[inc_idx], best_h[dec_idx])
        if delta <= 0:
            step *= 0.5
            continue

        cand = best_h.copy()
        cand[inc_idx] += delta
        cand[dec_idx] -= delta

        cand_val, _ = _c5_and_corr(cand)

        # accept only if strictly better
        if cand_val + 1e-12 < best_val:
            best_h = cand
            best_val = cand_val
        else:
            step *= 0.99  # shrink step when we cannot improve

        # occasional random jitter to escape plateaus
        if it % 1000 == 0 and np.random.rand() < 0.07:
            i, j = np.random.choice(n, size=2, replace=False)
            d = min(step, 1.0 - best_h[i], best_h[j])
            if d > 0:
                tmp = best_h.copy()
                tmp[i] += d
                tmp[j] -= d
                tv, _ = _c5_and_corr(tmp)
                if tv < best_val:
                    best_h, best_val = tmp, tv

        it += 1
        if it % 5000 == 0:
            step = max(min_step, step * 0.9)

    return best_h, best_val
