def optimize_binary(n, max_evals, rng):
    # n even, target sum = n//2
    # initialize with random binary vector
    h = np.zeros(n, dtype=np.int8)
    h[:n//2] = 1
    rng.shuffle(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # compute lists of positions of 1 and 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(max_evals):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0, 1
        new_c5 = compute_c5(h)
        # Accept if improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            # update ones zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # revert swap
            h[i], h[j] = 1, 0
    return best_h, best_c5

import time, math, random
import numpy as np

def compute_c5(h):
    """Return C5 for a step vector h (array of 0/1 or floats)."""
    n = h.shape[0]
    # correlation of h with 1-h, zero padded
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def optimize_binary(n, rng, time_budget, init_vec=None):
    """
    Heuristic optimisation for a binary vector of length n with exactly n/2 ones.
    Returns best vector found and its C5 value.
    """
    target = n // 2                      # required number of ones
    best_h = None
    best_c5 = float('inf')
    start = time.time()

    # ---------- helper to initialise a feasible binary vector ----------
    def random_feasible():
        v = np.zeros(n, dtype=int)
        v[:target] = 1
        rng.shuffle(v)
        return v

    # ---------- start from supplied init_vec if compatible ----------
    starts = []
    if init_vec is not None and len(init_vec) == n:
        v = np.array(init_vec, copy=True)
        # make it binary while preserving the required sum
        v = (v >= 0.5).astype(int)
        cur = int(v.sum())
        if cur < target:
            zero_idx = np.where(v == 0)[0]
            rng.shuffle(zero_idx)
            v[zero_idx[: target - cur]] = 1
        elif cur > target:
            one_idx = np.where(v == 1)[0]
            rng.shuffle(one_idx)
            v[one_idx[: cur - target]] = 0
        starts.append(v)
    # add a few random starts
    for _ in range(3):
        starts.append(random_feasible())

    # ---------- simulated‑annealing on each start ----------
    for start_vec in starts:
        h = start_vec.copy()
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # indices of 1s and 0s (re‑computed each iteration – cheap for our n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        # annealing schedule parameters
        T0 = 0.02
        T_end = 1e-6
        it = 0
        max_iters = 500_000          # upper bound; loop will stop by time limit

        while True:
            elapsed = time.time() - start
            if elapsed > time_budget:
                break
            # propose a swap between a 1 and a 0
            i = rng.choice(ones)
            j = rng.choice(zeros)

            # perform swap
            h[i] = 0
            h[j] = 1
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5

            # linear cooling
            T = T0 + (T_end - T0) * (it / max_iters)
            accept = False
            if delta <= 0:
                accept = True

                # Metropolis criterion

            if accept:
