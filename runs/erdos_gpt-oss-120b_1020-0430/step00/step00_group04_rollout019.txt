def project_to_constraints(v, target_sum):
    # Clip to [0,1]
    v = np.clip(v, 0, 1)
    # Compute sum
    s = np.sum(v)
    # If s == target_sum within tolerance, done.
    # Else need to increase or decrease entries away from bounds.
    # We'll perform simple iterative water-filling.
    # For diff >0: need to increase sum => add to entries with v < 1.
    # For diff <0: need to decrease sum => subtract from entries with v > 0.
    # We'll push equally to those free entries.

    # Use while loop
    diff = target_sum - s
    while abs(diff) > 1e-12:
        if diff > 0:
            # increase
            free = v < 1 - 1e-12
            if not np.any(free):
                break
            inc = diff / np.count_nonzero(free)
            v[free] += inc
            v = np.clip(v, 0, 1)
        else:
            # decrease
            free = v > 1e-12
            if not np.any(free):
                break
            dec = diff / np.count_nonzero(free)
            v[free] += dec   # diff negative
            v = np.clip(v, 0, 1)
        s = np.sum(v)
        diff = target_sum - s
    # Ensure final sum exactly target (maybe minor tolerance)
    return v

import numpy as np, time, math

def project_to_constraints(v, target):
    v = np.clip(v, 0.0, 1.0)
    diff = target - np.sum(v)
    # iterative water-filling
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0 - 1e-12
            if not np.any(free):
                break
            inc = diff / np.count_nonzero(free)
            v[free] += inc
            v = np.clip(v, 0.0, 1.0)
        else:
            free = v > 1e-12
            if not np.any(free):
                break
            dec = (-diff) / np.count_nonzero(free)
            v[free] -= dec
            v = np.clip(v, 0.0, 1.0)
        diff = target - np.sum(v)
    if abs(diff) > 1e-8:
        v += diff / len(v)
        v = np.clip(v, 0.0, 1.0)
    return v

    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

import time, math
import numpy as np

def _project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    """Clamp to [0,1] and adjust the sum to the target using simple water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - np.sum(v)
    # iterative redistribution, max 10 rounds – more than enough for the small errors we see
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:          # need to add mass
            free = v < 1.0 - 1e-12
            if not np.any(free):
                break
            inc = diff / np.count_nonzero(free)
            v[free] += inc
        else:                 # need to remove mass
            free = v > 1e-12
            if not np.any(free):
                break
            dec = (-diff) / np.count_nonzero(free)
            v[free] -= dec
        v = np.clip(v, 0.0, 1.0)
        diff = target - np.sum(v)
    # tiny leftover redistribution
    if abs(diff) > 1e-8:
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    return v


def _c5_of(h: np.ndarray) -> float:
    """Exact C₅ for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    """

    else:

        rand_vec = rng.random(n)
