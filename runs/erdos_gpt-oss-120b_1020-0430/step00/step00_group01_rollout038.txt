def local_search(n, max_iter, time_limit):
    # initialize random binary set
    h = np.zeros(n, dtype=np.float64)
    # choose n/2 random positions set to 1
    ones = np.random.choice(n, n//2, replace=False)
    h[ones] = 1.0
    c = compute_corr_vector(h)
    best_h = h.copy()
    best_val = np.max(c) * (2.0/n)
    start = time.time()
    
    zeros = np.where(h == 0)[0]
    ones_list = np.where(h == 1)[0]
    # Precompute shift array
    shifts = np.arange(-(n-1), n, dtype=np.int32)
    # Main loop
    while time.time() - start < time_limit and max_iter>0:
        # Choose random swap
        i = np.random.choice(zeros)   # zero index to turn to 1
        j = np.random.choice(ones_list)  # one index to turn to 0
        # compute delta_i (+1) and delta_j (-1)
        # Compute vect for i
        ip = i + shifts
        im = i - shifts
        # mask for ip in range
        mask_ip = (ip >= 0) & (ip < n)
        vect_plus_i = np.empty_like(shifts, dtype=np.float64)
        vect_plus_i[mask_ip] = 1.0 - h[ip[mask_ip]]
        vect_plus_i[~mask_ip] = 1.0
        mask_im = (im >= 0) & (im < n)
        vect_minus_i = np.zeros_like(shifts, dtype=np.float64)
        vect_minus_i[mask_im] = h[im[mask_im]]
        delta_i = vect_plus_i - vect_minus_i   # since (b-a)=1

        # Compute vect for j (swap 1->0, factor = -1)
        jp = j + shifts
        jm = j - shifts
        mask_jp = (jp >= 0) & (jp < n)
        vect_plus_j = np.empty_like(shifts, dtype=np.float64)
        vect_plus_j[mask_jp] = 1.0 - h[jp[mask_jp]]
        vect_plus_j[~mask_jp] = 1.0

    # This is a convex projection, can use algorithm:
    # Equivalent to solving min ||x - v||^2 s.t. sum x = target, 0 ≤ x ≤ 1.

def gradient_descent(h, max_iter, target):
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    # compute current value
    best_val = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)
    # gradient descent
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * (2.0 / n)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
        if cur_val <= target:
            break
        lag = max_idx - (n - 1)
        # compute gradient
        i = np.arange(n)
        ip = i + lag
        im = i - lag
        grad = np.ones(n, dtype=np.float64)
        mask_ip = (ip >= 0) & (ip < n)
        if mask_ip.any():
            grad[mask_ip] -= h[ip[mask_ip]]
        mask_im = (im >= 0) & (im < n)
        if mask_im.any():
            grad[mask_im] -= h[im[mask_im]]
        alpha = 0.5 * (0.99 ** it)  # decreasing step size
        h_new = h - alpha * grad
        # project
        h_new = np.clip(h_new, 0.0, 1.0)
        h = project_onto_bounds_and_sum(h_new, target_sum)
    # compute final val
    final_val = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n)
    return best_h, final_val

    factor = b - a
    L = 2*N -1

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------

def _project_onto_bounds_and_sum(v, target_sum, max_iter=60):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target_sum}."""
    lo, hi = -10.0, 10.0  # wide enough for any reasonable v
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def _correlation(h):
    """Full linear correlation of h with (1-h)."""
    return np.correlate(h, 1.0 - h, mode="full")


def _c5_from_corr(corr, n):
    """C5 value from correlation array."""
    return corr.max() * (2.0 / n)


def _compute_delta(idx, a, b, h):
    """
    Exact change of the correlation vector when entry idx changes from a to b.
    Returns a vector of length 2*n-1 (ordered as np.correlate does).
    """
    n = h.shape[0]
    d = b - a
    L = 2 * n - 1
    delta = np.zeros(L, dtype=np.float64)

    # plus side: i appears as first factor, i+s = ip
    ip = np.arange(n, dtype=np.int32)
    s_plus = ip - idx              # shift value for each ip
    delta[s_plus + n - 1] += d * (1.0 - h[ip])

    # minus side: i appears as second factor, i-s = im
    im = np.arange(n, dtype=np.int32)
    s_minus = idx - im
    delta[s_minus + n - 1] -= d * h[im]

    # correct the zero shift (quadratic term)
    delta[n - 1] = d * (1.0 - (a + b))      # = b*(1-b) - a*(1-a)

    return delta


# ------------------------------------------------------------
# Binary hill‑climbing search
# ------------------------------------------------------------

def _binary_hill_climb(n, time_budget, target):
    """Search for a binary (0/1) vector of length n with sum=n/2."""
    start = time.time()
    best_h = None
    best_val = np.inf
    half = n // 2
    while time.time() - start < time_budget:
        # random start (exact half ones)
        h = np.zeros(n, dtype=np.float64)
        ones = np.random.choice(n, half, replace=False)
        h[ones] = 1.0

        corr = _correlation(h)
        cur_val = _c5_from_corr(corr, n)

        # local improvement
        no_imp = 0
        max_no_imp = n * 5
        while time.time() - start < time_budget and no_imp < max_no_imp:
            zeros = np.where(h == 0)[0]
            if zeros.size == 0:
                break
            one_idxs = np.where(h == 1)[0]
            i = np.random.choice(zeros)
            j = np.random.choice(one_idxs)

            # delta for turning i from 0->1 and j from 1->0
            di = _compute_delta(i, 0.0, 1.0, h)
            dj = _compute_delta(j, 1.0, 0.0, h)
            new_corr = corr + di + dj
            new_val = _c5_from_corr(new_corr, n)

            if new_val < cur_val - 1e-12:
                # accept
                h[i] = 1.0
                h[j] = 0.0
                corr = new_corr
                cur_val = new_val
                no_imp = 0
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
                if cur_val <= target:
                    return best_h, best_val
            else:
                no_imp += 1

    if best_h is None:
        best_h = h
        best_val = cur_val
    return best_h, best_val


# ------------------------------------------------------------
# Continuous projected‑gradient descent (optional refinement)
# ------------------------------------------------------------

def _grad_descent(h, max_iter, target):
    """Projected gradient descent that reduces the maximal correlation."""
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = _c5_from_corr(_correlation(h), n)

    for it in range(max_iter):
        corr = _correlation(h)
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * (2.0 / n)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        if cur_val <= target:
            break

        lag = max_idx - (n - 1)  # shift corresponding to max
        i = np.arange(n, dtype=np.int32)
        ip = i + lag
        im = i - lag

        grad = np.ones(n, dtype=np.float64)
        mask_ip = (ip >= 0) & (ip < n)
        grad[mask_ip] -= h[ip[mask_ip]]
        mask_im = (im >= 0) & (im < n)
        grad[mask_im] -= h[im[mask_im]]

        # step size schedule (decreasing)
        alpha = 0.5 * (0.97 ** it)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h = _project_onto_bounds_and_sum(h_new, target_sum)

    return best_h, best_val


# ------------------------------------------------------------
# Main driver
# ------------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] giving a low C5 bound.

    Returns (h_values, c5_bound, n_points).  h_values is a list of floats,
    c5_bound is the achieved bound, and n_points is the number of samples.
    """
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # target bound we would like to beat (taken from the statement)
    TARGET = 0.3808

    # a few grid sizes – all even so that a binary solution of exact weight exists
    grid_sizes = [96, 128, 160, 192, 224, 256, 304]

    best_h = None
    best_val = np.inf
    best_n = None

    # --------------------------------------------------------------------
    # Phase 1 – binary hill climbing on several grids
    # --------------------------------------------------------------------
    for n in grid_sizes:
        if time_left() < 5.0:
            break
        # allocate ~40% of remaining time for this grid
        alloc = min(time_left() * 0.4, 300.0)
        h_bin, val_bin = _binary_hill_climb(n, alloc, TARGET)
        if val_bin < best_val:
            best_val = val_bin
            best_h = h_bin.copy()
            best_n = n
        if best_val <= TARGET:
            # good enough – stop early
            break

    # --------------------------------------------------------------------
    # Phase 2 – optional continuous refinement from the best binary solution
    # --------------------------------------------------------------------
    if best_h is not None and time_left() > 10.0 and best_val > TARGET:
        # start from the current best (as float), run a few gradient steps
        refine_iters = 2000
        h_ref, val_ref = _grad_descent(best_h.astype(np.float64), refine_iters, TARGET)
        if val_ref < best_val:
            best_val = val_ref
            best_h = h_ref.copy()
            # best_n unchanged (same grid)

    # --------------------------------------------------------------------
    # Fallback: if nothing worked, use a smooth cosine‑square profile
    # --------------------------------------------------------------------
    if best_h is None or best_val > 0.5:
        n = 200
        dx = 2.0 / n
        xs = (np.arange(n) + 0.5) * dx
        h_fallback = np.cos(np.pi * xs / 4.0) ** 2
        # project to satisfy the area constraint exactly
        h_fallback = _project_onto_bounds_and_sum(h_fallback, n / 2.0)
        best_h = h_fallback
        best_n = n
        best_val = _c5_from_corr(_correlation(best_h), n)

    return best_h.tolist(), float(best_val), int(best_n)
