def evaluate_h(h):

def worst_shift_and_gradient(h):
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (N - 1)
    # gradient
    g = np.zeros(N)
    start_i = max(0, -shift)
    end_i = min(N - 1, N - 1 - shift)
    # create i array
    i_arr = np.arange(start_i, end_i + 1, dtype=int)
    j_arr = i_arr + shift
    g[i_arr] += 1 - h[j_arr]
    # subtract h[i_arr] from g[j_arr]
    np.subtract.at(g, j_arr, h[i_arr])
    return shift, g

def enforce_bounds_and_sum(h, target):
    # clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    # adjust sum to target
    # simple iterative approach
    for _ in range(10):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly over all elements not at bounds limit
        # compute mask of elements that can move in direction of diff
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            # cannot adjust further, break
            break
        delta = diff / cnt
        # apply delta limited to bounds
        adjust = np.where(mask, delta, 0.0)

        if diff > 0:

def evaluate_h(h):
    N = h.shape[0]
    return np.max(np.correlate(h, 1 - h, mode='full') * (2.0 / N))

def worst_shift_and_gradient(h):
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (N - 1)
    g = np.zeros(N, dtype=np.float64)
    start_i = max(0, -shift)
    end_i = min(N - 1, N - 1 - shift)
    i_arr = np.arange(start_i, end_i + 1, dtype=int)
    j_arr = i_arr + shift
    g[i_arr] += 1.0 - h[j_arr]
    np.subtract.at(g, j_arr, h[i_arr])
    return shift, g

def enforce_bounds_and_sum(h, target):
    # clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    # adjust sum iteratively
    for _ in range(15):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        if diff > 0:
            # need to decrease
            dec = np.minimum(delta, h[mask])
            h[mask] -= dec
        else:
            inc = np.minimum(-delta, 1.0 - h[mask])
            h[mask] += inc
        np.clip(h, 0.0, 1.0, out=h)
    return h
