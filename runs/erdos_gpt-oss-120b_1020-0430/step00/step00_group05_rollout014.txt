def compute_max_corr(h):
    N = h.shape[0]
    a = h.astype(np.float64)
    b = (1 - h).astype(np.float64)
    n = 2 * N - 1
    # compute correlation a with b using FFT
    fa = np.fft.rfft(a, n=n)
    fb = np.fft.rfft(b, n=n)
    corr = np.fft.irfft(fa * np.conj(fb), n=n)
    # corr length = n
    # due to numeric errors, round small epsilon
    # use real part
    return np.max(corr)

def hill_climb(N, time_limit):
    start = time.time()
    # initial random binary
    h = np.zeros(N, dtype=np.float64)
    # set half ones randomly
    indices = np.arange(N)
    np.random.shuffle(indices)
    h[indices[:N//2]] = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    # additional: also keep best_score for early exit
    while time.time() - start < time_limit:
        # choose random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # propose swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_score = compute_c5(h_new)
        # Accept if better or with prob
        if new_score < best_score or np.random.rand() < 0.001:
            h = h_new
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        # Optionally break if very good bound reached (< target)
        if best_score <= target*...:
            break
    return best_h, best_score

import time
import numpy as np

def _project_simplex_box(v, target_sum):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target_sum}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_bound(h):
    """C5 bound for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx, corr.max()


def _gradient_max_corr(h, active_lags):
    """Gradient of max_{lag} sum_i h_i (1-h_{i+lag}) restricted to active lags."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for lag in active_lags:
        if lag >= 0:
            L = n - lag
            grad[:L] += (1.0 - h[lag:])
            grad[lag:] -= h[:L]
        else:
            d = -lag
            L = n - d
            grad[d:] += (1.0 - h[:L])
            grad[:L] -= h[d:]
    grad /= len(active_lags)
    return grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points."""
    np.random.seed(seed)
    start_time = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    else:
        n = 100
        h = np.zeros(n, dtype=np.float64)
        h[np.random.choice(n, n // 2, replace=False)] = 1.0

    target_sum = n / 2.0
    h = _project_simplex_box(h, target_sum)

    best_h = h.copy()
    best_c5, _ = _c5_bound(best_h)

    # ----- optimisation loop -----------------------------------------
    # parameters
    step0 = 0.6
    max_iter = 10_000_000  # safety cap; we break by time

    it = 0
    while time.time() - start_time < budget_s * 0.9 and it < max_iter:
        # correlation and active lags (those attaining the maximum)
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_max = corr.max()
        cur_c5 = cur_max * (2.0 / n)

        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()
            if best_c5 <= 0.3808:
                # good enough – keep searching a little more just in case
                pass

        # lags achieving the maximum (allow tiny tolerance)
        tol = 1e-12
        active_idx = np.where(corr >= cur_max - tol)[0]
        active_lags = active_idx - (n - 1)

        # gradient step
        grad = _gradient_max_corr(h, active_lags)
        if np.linalg.norm(grad) > 0:
            step = step0 / np.sqrt(it + 1.0)
            h_candidate = _project_simplex_box(h - step * grad, target_sum)

            # accept if not worse
