def solve_milp(N, time_limit):
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(y) == N/2, t >= 0]

    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.CBC, cbc_params={"seconds": time_limit})

import time, math, random
import numpy as np
import cvxpy as cp

def _c5_of(h):
    """Return C5 for a step function sampled in h."""
    N = h.shape[0]
    dx = 2.0 / N
    # correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _build_milp(N, target_sum):
    """Construct the MILP that minimizes max_{k}∫h(x)(1‑h(x+k))dx for binary h."""
    y = cp.Variable(N, boolean=True)
    t = cp.Variable()                     # will be the un‑scaled max correlation
    constraints = [cp.sum(y) == target_sum, t >= 0]

    for s in range(1, N):
        n_pairs = N - s
        w = cp.Variable(n_pairs, boolean=True)   # w_i = y_i ∧ y_{i+s}

        # linking constraints w_i <= y_i , w_i <= y_{i+s} , w_i >= y_i + y_{i+s} - 1
        constraints.append(w <= y[:n_pairs])
        constraints.append(w <= y[s:])
        constraints.append(w >= y[:n_pairs] + y[s:] - 1)

        pref_fwd = cp.sum(y[:n_pairs])            # Σ_{i=0}^{N‑1‑s} y_i
        pref_bwd = cp.sum(y[s:])                  # Σ_{i=s}^{N‑1} y_i

        constraints.append(pref_fwd - cp.sum(w) <= t)
        constraints.append(pref_bwd - cp.sum(w) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    return prob, y, t

    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-10:
        # simple projection: keep the largest entries as 1 until the sum is right
        idx = np.argsort(-h0)
        h_bin = np.zeros(N, dtype=float)
        h_bin[idx[:int(target_sum)]] = 1.0
        h0 = h_bin

    best_h = h0.copy()
    best_score = _c5_of(best_h)

    # Return the best found construction
    # ------------------------------------------------------------
    return best_h.tolist(), float(best_score), N
