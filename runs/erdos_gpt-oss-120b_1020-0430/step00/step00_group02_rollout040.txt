import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 value for the step function represented by h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the hyperplane sum(v)=target while keeping 0<=v_i<=1.
    Uses a simple bisection on a common shift lambda.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _upsample(arr: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of a step vector to a larger size and projection onto constraints."""
    n_src = arr.size
    if n_src == n_target:
        return arr.copy()
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    interp = np.interp(xt, xs, arr)
    interp = np.clip(interp, 0.0, 1.0)
    target_sum = n_target * 0.5
    return _project_to_sum_box(interp, target_sum)


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (bounds [0,1] and sum=n/2)."""
    v = rng.random(n)
    target = n * 0.5
    return _project_to_sum_box(v, target)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Build initial feasible vector (possibly up‑sampled)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=np.float64)
    else:
        base = None

    # target number of points – a modest size that still allows many moves
    TARGET_N = 180

    if base is not None:
        h = _upsample(base, TARGET_N)
    else:
        h = _random_feasible(TARGET_N, rng)

    n = h.size
    target_sum = n * 0.5

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing style local search
    # ------------------------------------------------------------------
    cur_h = h.copy()
    cur_cost = _compute_c5(cur_h)

    best_h = cur_h.copy()
    best_cost = cur_cost

    # Annealing schedule parameters
    MAX_TEMP = 0.12
    MIN_TEMP = 1e-7
    # Total move budget – will stop earlier if time runs out
    MAX_MOVES = 500_000

    move = 0
    while move < MAX_MOVES:
        # time guard
        if time.time() - start_time > budget_s * 0.96:
            break

        progress = move / MAX_MOVES
        temperature = MAX_TEMP * (1.0 - progress) + MIN_TEMP * progress

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # attempt to move mass from i to j
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        direction = 1  # i -> j
        if max_transfer <= 0.0:
            # try opposite direction
            max_transfer = min(cur_h[j], 1.0 - cur_h[i])
            if max_transfer <= 0.0:
                continue
            i, j = j, i
            direction = -1  # original i receives mass

        # step size shrinks as we progress
        step_factor = max(0.02, 1.0 - progress)
        delta = max_transfer * rng.random() * step_factor

        # apply move
        cur_h[i] -= delta
        cur_h[j] += delta

        new_cost = _compute_c5(cur_h)
        delta_cost = new_cost - cur_cost

        if delta_cost <= 0.0 or rng.random() < math.exp(-delta_cost / temperature):

                best_cost = new_cost
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta

        move += 1

    # ------------------------------------------------------------------
    # 3. Final polishing (few fine‑grained moves with only improving steps)
    # ------------------------------------------------------------------
    for _ in range(20000):
        if time.time() - start_time > budget_s * 0.98:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0.0:
            continue
        delta = max_transfer * rng.random() * 0.01  # very tiny tweaks
        best_h[i] -= delta
        best_h[j] += delta
        new_c = _compute_c5(best_h)
        if new_c < best_cost:
            best_cost = new_c
        else:
            best_h[i] += delta
            best_h[j] -= delta

    # ------------------------------------------------------------------
    # 4. Return result
    # ------------------------------------------------------------------
    # Ensure feasibility one last time (numerical drift)
    best_h = _project_to_sum_box(best_h, target_sum)

    final_cost = _compute_c5(best_h)
    return best_h, final_cost, n
