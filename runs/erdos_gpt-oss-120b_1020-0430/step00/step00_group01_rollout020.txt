def hill_climb(h, rng, max_iters, start_temp, end_temp):
    current = h.copy()
    cur_c5 = compute_c5(current)
    best = current.copy()
    best_c5 = cur_c5

    for it in range(max_iters):
        # Temperature schedule:
        t = start_temp + (end_temp - start_temp) * (it / max_iters)
        # propose swap:
        # choose random index i1 where h=1, i0 where h=0.
        ones_idx = np.where(current == 1)[0]
        zeros_idx = np.where(current == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0: break
        i1 = rng.choice(ones_idx)
        i0 = rng.choice(zeros_idx)
        candidate = current.copy()
        candidate[i1] = 0
        candidate[i0] = 1
        cand_c5 = compute_c5(candidate)
        delta = cand_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta/t):
            current = candidate
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best = current.copy()
                best_c5 = cur_c5
        # optionally break if time up

    return best, best_c5

import numpy as np
import time

# ---------- helper functions ----------
def compute_c5(h):
    """Return the C5 value for a given step function h (numpy array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def compute_c5_and_lag(h):
    """Return (C5, lag) where lag is the shift giving the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    maxc = corr[idx]
    lag = idx - (n - 1)          # shift corresponding to this index
    return maxc * (2.0 / n), lag


def mismatched_positions(h, lag):
    """Indices where a 1 is paired with a 0 for the given lag."""
    n = h.shape[0]
    if lag >= 0:
        i = np.arange(0, n - lag)
        hi = h[i]
        hj = h[i + lag]
    else:
        lag = -lag
        i = np.arange(lag, n)
        hi = h[i]
        hj = h[i - lag]
    mis10 = i[(hi == 1) & (hj == 0)]
    mis01 = i[(hi == 0) & (hj == 1)]
    return mis10, mis01


def random_binary(n, rng):
    """Create a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1
    return h


# ---------- main optimisation routine ----------

    start_time = time.time()
    def elapsed():
        return time.time() - start_time

    # keep the best solution we have seen (maybe the provided one)

    # ---------- binary search on an even grid ----------
    n = 200                      # even length, half ones → feasible
    max_temp = 0.02
    min_temp = 1e-4
    target = 0.3808

    # start from a random binary configuration
    cur_h = random_binary(n, rng).astype(np.float64)
    cur_c5 = compute_c5(cur_h)

    if cur_c5 < best_c5:

    steps_without_improve = 0
    # main loop – simulated annealing with targeted swaps
    while elapsed() < budget_s * 0.95:
        # compute the worst lag for the current configuration
        c5_cur, lag = compute_c5_and_lag(cur_h)
