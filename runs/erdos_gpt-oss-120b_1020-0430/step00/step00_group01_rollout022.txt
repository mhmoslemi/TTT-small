def random_move(h):
    # pick two distinct indices i,j
    # choose delta positive from range such that resulting values in [0,1]
    i, j = random.sample(range(n),2)
    # compute max possible shift from i to j: delta_max = min(h[i], 1 - h[j])
    max_delta = min(h[i], 1 - h[j])
    if max_delta == 0: return None (can't move)
    # choose delta random in [0, max_delta]
    delta = random.random() * max_delta
    # create new h copy, adjust.
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

import numpy as np, time, random, math, sys

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def simulated_annealing(h0, time_limit):
    start = time.time()
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    n = len(h0)
    # parameters
    total_time = time_limit
    T0 = 0.1 * cur_c5  # initial temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= total_time:
            break
        # compute temperature schedule (cooling)
        T = T0 * max(0.0, 1.0 - elapsed / total_time)
        # random move
        i, j = random.sample(range(n), 2)
        # ensure we can move delta
        hi = cur_h[i]
        hj = cur_h[j]
        max_delta = min(hi, 1 - hj)
        # if integrate move to increase variability
        min_delta = min(cur_h[j], 1 - cur_h[i])
        # Actually delta can be positive from i to j (i loses, j gains) or opposite.
        # Choose direction randomly
        if random.random() < 0.5:
            # move from i to j (i decreases, j increases)
            max_delta = min(cur_h[i], 1 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = random.random() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move from j to i
            max_delta = min(cur_h[j], 1 - cur_h[i])
            if max_delta <= 0:
                continue
            delta = random.random() * max_delta
            new_h = cur_h.copy()

            new_h[i] += delta
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - cur_c5
        if delta_c5 < 0 or random.random() < math.exp(-delta_c5 / (T + 1e-12)):

def project_to_sum(v, target):
    # using algorithm for projection onto [0,1] and sum.
    # Steps: first clamp v to [0,1] giving w.
    # Then if sum(w) == target, done.
    # If sum(w) > target, need to reduce some values (similar to simplex projection).
    # If sum(w) < target, need to increase some values.

    n = len(v)
    # We'll use simple iterative algorithm: shift all entries by lambda, then clamp.
    # Equivalent to solving min ||x - v||^2 s.t. sum x = target, 0 <= x <= 1.
    # This can be solved via bisection on lambda: For given lambda, compute x_i = min(1, max(0, v_i - lambda)). The sum of x_i is decreasing function of lambda. Find lambda such that sum = target.
    # We'll bisection between low = min(v) - 1 (makes all x become 1) and high = max(v) (makes all x become 0).
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def project_to_sum(v, target):
    # Clip to [0, 1] after shifting by lambda.
    # Use bisection as described.
    lo = -10.0
    hi = 10.0
    for _ in range(100):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def optimize_continuous_slsqp(n, time_limit):
    start = time.time()
    best_h = None
    best_c5 = float('inf')
    # define constraints and bounds
    bounds = [(0.0, 1.0) for _ in range(n)]
    # equality constraint: sum h = n/2
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}
    # smoothing alpha
    alpha = 10.0
    def obj(x):
        # Use smooth max approx
        n_local = x.shape[0]
        dx = 2.0 / n_local
        corr = np.correlate(x, 1 - x, mode='full') * dx
        # log-sum-exp approx
        m = np.max(corr)
        # compute stable
        exp_sum = np.exp(alpha * (corr - m)).sum()
        return m + np.log(exp_sum) / alpha
    # multiple restarts within time
    while time.time() - start < time_limit:
        # for each restart allocate some time budget: maybe 30 sec each.
        # We'll simply call minimize; we can set maxiter as high but it will stop early.
        # We need to enforce time bound by breaking if total time exceed.
        x0 = random_feasible(n)
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
        if not res.success:
            # maybe ignore
            pass
        h_opt = res.x
        # Due to numerical errors, ensure bounds
        h_opt = np.clip(h_opt, 0.0, 1.0)
        # adjust sum precisely (project maybe)
        # we can project to sum but it's already near target, but to be safe:
        h_opt = project_to_sum(h_opt, n/2.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5

import numpy as np, random, time, math
from scipy.optimize import minimize

_GLOBAL_TARGET_SUM = 0.0
_GLOBAL_ALPHA = 10.0

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def project_to_sum(v, target):
    # Bisection to find lambda s.t. sum(clipped(v - lambda)) = target
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def random_feasible(n):
    v = np.random.rand(n)  # uniform [0,1)
    target = n / 2.0
    return project_to_sum(v, target)

def _eq_sum_constraint(x):
    return np.sum(x) - _GLOBAL_TARGET_SUM

def _smooth_max_obj(x):
    n = x.shape[0]
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = np.max(corr)
    # Use log-sum-exp for smooth max
    exp_sum = np.exp(_GLOBAL_ALPHA * (corr - m)).sum()
    return m + np.log(exp_sum) / _GLOBAL_ALPHA

def optimize_continuous_slsqp(n, time_allow):
    global _GLOBAL_TARGET_SUM
    start = time.time()
    best_h = None
    best_c5 = float('inf')
    _GLOBAL_TARGET_SUM = n / 2.0
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = {'type': 'eq', 'fun': _eq_sum_constraint}
    while time.time() - start < time_allow:
        x0 = random_feasible(n)
        # set max iteration maybe 150
        res = minimize(_smooth_max_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 150, 'ftol': 1e-9, 'disp': False})
        if not res.success:
            # ignore
            pass
        h = np.clip(res.x, 0.0, 1.0)
        # enforce exact sum via projection (small correction)
        h = project_to_sum(h, n/2.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def simulated_annealing_binary(h0, time_limit):
    start = time.time()
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    n = h0.shape[0]
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # Temperature linearly decreasing to near zero
        T = max(1e-8, 0.5 * (1.0 - elapsed / time_limit))
        i, j = random.sample(range(n), 2)
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(new_h)
        diff = new_c5 - cur_c5
        if diff < 0 or random.random() < math.exp(-diff / (T + 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
    return best_h, best_c5

def optimize_binary_sa(n, time_allow):
    # n must be even for binary exact half ones.
    if n % 2 != 0:
        # fallback to continuous
        return optimize_continuous_slsqp(n, time_allow)
    start = time.time()
    best_h = None
    best_c5 = float('inf')
    while time.time() - start < time_allow:
        # generate random binary with n/2 ones
        h0 = np.zeros(n, dtype=float)
        idx = np.random.choice(n, size=n//2, replace=False)
        h0[idx] = 1.0
        # allocate a portion of remaining time for this restart
        elapsed = time.time() - start
        remaining = time_allow - elapsed
        # ensure at least some time
        if remaining <= 0:
            break
        # allocate, say, min(remaining/5, 10s)

            best_c5 = c5_opt
            best_h = h_opt.copy()
    # If binary attempts never succeed, fallback to continuous
    if best_h is None:
        return optimize_continuous_slsqp(n, time_allow)
    return best_h, best_c5

import time, random, math
import numpy as np
from scipy.optimize import minimize

_GLOBAL_TARGET_SUM = 0.0
_GLOBAL_ALPHA = 10.0  # smoothing factor for max approximation


def compute_c5(h: np.ndarray) -> float:
    """C₅ for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] while fixing sum(v)=target."""
    lo, hi = -10.0, 10.0
    for _ in range(80):
        lam = 0.5 * (lo + hi)
        x = np.clip(v - lam, 0.0, 1.0)
        if x.sum() > target:
            lo = lam
        else:
            hi = lam
    lam = 0.5 * (lo + hi)
    return np.clip(v - lam, 0.0, 1.0)


def random_feasible(n: int) -> np.ndarray:
    """Random vector in [0,1] with sum = n/2."""
    v = np.random.rand(n)
    return project_to_sum(v, n / 2.0)


def _eq_sum_constraint(x: np.ndarray) -> float:
    """Equality constraint used by SLSQP."""
    return np.sum(x) - _GLOBAL_TARGET_SUM


def _smooth_max_obj(x: np.ndarray) -> float:
    """Smooth approximation of max_k ∫ h(x)(1‑h(x+k)) dx."""
    n = x.shape[0]
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    # log‑sum‑exp (stable)
    exp_sum = np.exp(_GLOBAL_ALPHA * (corr - m)).sum()
    return m + np.log(exp_sum) / _GLOBAL_ALPHA


def optimize_continuous_slsqp(n: int, time_allow: float):
    """SLSQP on the smooth max objective, multiple random restarts."""
    global _GLOBAL_TARGET_SUM
    _GLOBAL_TARGET_SUM = n / 2.0
    start = time.time()
    best_h = None
    best_c5 = float("inf")
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = {"type": "eq", "fun": _eq_sum_constraint}
    while time.time() - start < time_allow:
        x0 = random_feasible(n)
        res = minimize(
            _smooth_max_obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 150, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            continue
        h = np.clip(res.x, 0.0, 1.0)
        h = project_to_sum(h, n / 2.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def simulated_annealing_binary(h0: np.ndarray, time_limit: float):
    """Binary SA – swaps keep the number of ones unchanged."""
    start = time.time()
    cur = h0.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    n = len(h0)
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = max(1e-8, 0.5 * (1.0 - elapsed / time_limit))
        i, j = random.sample(range(n), 2)
        if cur[i] == cur[j]:
            continue
        nxt = cur.copy()
        nxt[i], nxt[j] = cur[j], cur[i]
        nxt_c5 = compute_c5(nxt)
        diff = nxt_c5 - cur_c5
        if diff < 0.0 or random.random() < math.exp(-diff / (T + 1e-12)):
            cur, cur_c5 = nxt, nxt_c5
            if cur_c5 < best_c5:
                best, best_c5 = cur.copy(), cur_c5
    return best, best_c5


def optimize_binary_sa(n: int, time_allow: float):
    """Binary SA with several random restarts (fallback to continuous)."""
    if n % 2:
        return optimize_continuous_slsqp(n, time_allow)
    start = time.time()
    best_h = None
    best_c5 = float("inf")
    while time.time() - start < time_allow:
        h0 = np.zeros(n, dtype=float)
        idx = np.random.choice(n, size=n // 2, replace=False)
        h0[idx] = 1.0
        remaining = time_allow - (time.time() - start)
        if remaining <= 0:

        # give each restart a modest slice of the remaining budget
        slice_time = min(10.0, remaining / 5.0 + 1e-6)
        h_opt, c5_opt = simulated_annealing_binary(h0, slice_time)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
    if best_h is None:
        return optimize_continuous_slsqp(n, time_allow)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
