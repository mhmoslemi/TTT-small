def compute_c5_fft(h):
    n = h.shape[0]
    b = 1.0 - h
    revb = b[::-1]
    N = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(revb, N)
    conv = np.fft.irfft(fa * fb, N)
    corr = conv[:2*n-1] # length 2n-1
    dx = 2.0 / n
    max_corr = np.max(corr) * dx
    return max_corr, corr # maybe also shift location.

def upscale_h(h_low, n_target):
    n_low = len(h_low)
    x_low = np.linspace(0,2,n_low,endpoint=False)
    x_high = np.linspace(0,2,n_target,endpoint=False)
    h_high = np.interp(x_high, x_low, h_low, left=h_low[0], right=h_low[-1])
    # Now ensure sum = n_target/2; scale accordingly
    total = np.sum(h_high)
    target = n_target/2
    # Scaling factor
    factor = target / total
    h_high = h_high * factor
    # clip to [0,1]; after scaling some may exceed 1; to fix:
    # if any >1, we can set those to 1 and adjust remaining mass proportionally.
    # implement simple projection:
    excess = np.maximum(h_high - 1, 0)
    if np.any(excess):
        # reduce the excess mass from other entries
        over = excess.sum()
        h_high[h_high > 1] = 1
        # reduce total from below-1 entries proportionally:
        # compute capacity to reduce: sum_{i: h_i<1} (h_i - 0)
        # Actually we need to subtract over mass from entries with h_i < 1.
        below = h_high < 1
        avail = (h_high[below] - 0).sum()  # we can reduce them down to 0
        if avail > 0:
            reduction = over * (h_high[below] / h_high[below].sum())
            h_high[below] -= reduction
        else:
            # fallback: just clip but may not maintain sum
            pass
    # similarly for values <0 (should not happen)
    h_high = np.clip(h_high, 0, 1)
    # final adjust sum exactly: compute diff and distribute to random indices with capacity.
    diff = target - h_high.sum()
    # distribute diff across entries with capacity (if diff>0, add; if diff<0, subtract)
    # We'll just adjust by adding diff/n to each entry ensuring bounds: may be small.
    # Use while loop to adjust.
    while abs(diff) > 1e-12:
        # capacity list
        if diff > 0:
            capacity = 1 - h_high
        else:
            capacity = h_high
        total_cap = capacity.sum()
        if total_cap == 0:

        add = diff * (capacity / total_cap)

        diff = target - h_high.sum()

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _compute_c5(h: np.ndarray):
    """Return (C5, full correlation array) using FFT."""
    n = h.shape[0]
    dx = 2.0 / n
    b = 1.0 - h
    revb = b[::-1]
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(revb, N)
    conv = np.fft.irfft(fa * fb, N)
    corr = conv[: 2 * n - 1]
    return float(np.max(corr) * dx), corr

    """Gradient weights for the given shift (the worst shift)."""

    else:
