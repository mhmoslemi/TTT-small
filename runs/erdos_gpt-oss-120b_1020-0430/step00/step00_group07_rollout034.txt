def run(seed=42, budget_s=1100):
    import time, numpy as np, random
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_score = float('inf')
    # initial h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        # enforce sum constraint exact: total sum must be n/2; we can adjust by sorting and threshold
        target_sum = n // 2  # since n even? assume.
        # convert to binary by selecting largest values
        idx_sorted = np.argsort(-h0)  # descending
        binary_h = np.zeros_like(h0)
        binary_h[idx_sorted[:target_sum]] = 1.0
        h = binary_h
    else:
        n = 80  # default
        target_sum = n // 2
        h = np.zeros(n)
        ones_idx = np.random.choice(n, target_sum, replace=False)
        h[ones_idx] = 1.0

    # compute initial score
    score = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_score = score

    # Precompute set of indices for ones and zeros
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()

    # Simulated annealing parameters
    max_iter = 5000000
    temperature = 1.0
    cooling_rate = 0.999995  # adjust
    iter = 0
    while time.time() - start < budget_s * 0.95:
        # pick random swap
        i = random.choice(ones)
        j = random.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_score = evaluate_erdos_solution(h)
        delta = new_score - score
        accept = False
        if delta < 0:
            accept = True
        else:
            # annealing acceptance
            if random.random() < np.exp(-delta / temperature):
                accept = True
        if accept:
            # update sets
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
        # update temperature
        temperature *= cooling_rate
        iter += 1

    return best_h, best_score, len(best_h)

import time
import random
import numpy as np

# Helper to evaluate a candidate (fallback if the harness does not provide one)
def _eval_h(h):
    """
    Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx using the discretisation described in the
    problem statement.  The harness normally supplies evaluate_erdos_solution,
    but we provide a pure‑numpy fallback.
    """
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate implements linear (zero‑padded) correlation.
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

# The harness may inject a pre‑imported evaluator; we use it if present.
try:
    from __main__ import evaluate_erdos_solution  # type: ignore
    _eval = evaluate_erdos_solution
except Exception:  # pragma: no cover
    _eval = _eval_h


def _initial_feasible(h_raw):
    """
    Turn the supplied initial vector into a feasible one:
    * clip to [0,1]
    * enforce the integral constraint sum(h)*dx == 1   (i.e. sum(h) == n/2)
      by normalising the total mass and, if necessary, rounding to a binary
      vector while preserving the total count.
    """
    h = np.clip(np.asarray(h_raw, dtype=float), 0.0, 1.0)
    n = h.shape[0]
    target_sum = n // 2          # because dx = 2/n and ∫h = 1
    cur_sum = h.sum()
    if cur_sum == 0:                     # degenerate case – populate uniformly
        h[:] = 0.5
        cur_sum = h.sum()
    # Scale to the right total mass
    if not np.isclose(cur_sum, target_sum):
        h *= target_sum / cur_sum
    # Small numerical drift – project onto the simplex of 0/1 values
    # (binary rounding while keeping the exact sum)
    flat = h.ravel()
    # number of entries that must be 1
    need_ones = int(round(target_sum))
    # indices of the largest entries become 1, the rest 0
    order = np.argsort(-flat)
    binary = np.zeros_like(flat)
    binary[order[:need_ones]] = 1.0
    return binary.reshape(h.shape)

    """

    np.random.seed(seed)

    # ------------------------------------------------------------------

        h[ones] = 1.0

    n = h.shape[0]
    target_sum = n // 2

    # ------------------------------------------------------------------
    # 2.  Evaluation helpers
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _eval(best_h)

    # Lists of indices currently holding 1 and 0 – kept up‑to‑date
    ones_idx = list(np.where(best_h == 1)[0])
    zeros_idx = list(np.where(best_h == 0)[0])

    # ------------------------------------------------------------------
    # 3.  Simulated annealing parameters
    # ------------------------------------------------------------------
    start_time = time.time()
    # temperature schedule – start fairly hot, cool slowly
    temperature = 1.0
    cooling = 0.99998
    # maximum number of proposals; the loop also stops on timeout
    max_iters = 10_000_000
    it = 0

    # ------------------------------------------------------------------
    # 4.  Main optimisation loop
    # ------------------------------------------------------------------
    while it < max_iters and (time.time() - start_time) < budget_s * 0.95:
        # propose a swap of a random 1 with a random 0
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)

        # apply swap
        h[i] = 0.0
        h[j] = 1.0

        # evaluate new candidate
        new_score = _eval(h)

        # decide acceptance
        delta = new_score - best_score
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            if random.random() < np.exp(-delta / max(temperature, 1e-12)):
                accept = True

        if accept:
            # commit the swap permanently
            ones_idx.remove(i)
            zeros_idx.append(i)
            zeros_idx.remove(j)
            ones_idx.append(j)

            best_score = new_score
            best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

        # temperature decay
        temperature *= cooling
        it += 1

        # occasional random restart if we are stuck for a while
        if it % 200_000 == 0 and best_score > 0.5:
            # completely reshuffle the bits while keeping the count
            perm = np.random.permutation(n)
            h[:] = 0.0

            zeros_idx = list(np.where(h == 0)[0])
            # evaluate the fresh start
            cur = _eval(h)
            if cur < best_score:
                best_score = cur
                best_h = h.copy()

    # ------------------------------------------------------------------
    # 5.  Return the best solution discovered
    # ------------------------------------------------------------------
    return best_h, best_score, n
