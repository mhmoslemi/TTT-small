import time, math

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for the step function represented by h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def simulated_annealing(start_h: np.ndarray, max_seconds: float,

    """Project h onto the hyperplane sum(h)=n/2 while staying in [0,1]."""
