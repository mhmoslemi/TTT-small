import numpy as np, time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
```

import time, math, random
import numpy as np

def _compute_c5(h, dx):
    """Return the C5 value for the current step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _gradient_for_lag(h, lag):
    """Gradient of the active overlap term w.r.t. h."""
    n = h.shape[0]
    if lag >= 0:
        # h_{i+lag}
        hi_plus = np.concatenate((h[lag:], np.zeros(lag, dtype=h.dtype)))
        # h_{i-lag}
        hi_minus = np.concatenate((np.zeros(lag, dtype=h.dtype), h[:n - lag]))
    else:
        a = -lag
        hi_plus = np.concatenate((np.zeros(a, dtype=h.dtype), h[:n - a]))
        hi_minus = np.concatenate((h[a:], np.zeros(a, dtype=h.dtype)))
    return 1.0 - hi_plus - hi_minus  # gradient of f_k = sum_i h_i(1-h_{i+lag})

def _random_initial(n, target):
    """Generate a random feasible h (0<=h<=1, sum = target)."""
    # Start from exponential variables to avoid many values near 0
    vals = np.random.exponential(scale=1.0, size=n)
    # Clip to 1 before scaling to keep feasibility
    vals = np.minimum(vals, 1.0)
    s = vals.sum()
    if s == 0:
        vals[:] = target / n
        return vals
    # Scale to meet the sum constraint
    vals = vals * (target / s)
    # After scaling some entries may exceed 1; fix them iteratively
    excess = vals - 1.0
    while np.any(excess > 1e-12):
        over = excess > 0
        # remove excess mass from overfull entries
        over_sum = excess[over].sum()
        vals[over] = 1.0
        # distribute the removed mass uniformly to the remaining capacity
        free = 1.0 - vals
        free_sum = free.sum()
        if free_sum <= 0:
            break
        vals += free / free_sum * over_sum
        excess = vals - 1.0
    # Final normalization (tiny numerical adjustments)
    vals *= target / vals.sum()
    return vals
