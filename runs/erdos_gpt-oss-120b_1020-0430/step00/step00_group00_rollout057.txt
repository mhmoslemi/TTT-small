import time, numpy as np

def compute_h(alpha, p, n):
    max_d = (n-1)//2
    d = np.minimum(np.arange(n), np.arange(n)[::-1])
    coeff = (1 - (d / max_d) ** p)
    K = np.sum(coeff)
    beta = (n/2 - n*alpha) / K
    h = alpha + beta * coeff
    # Clip for safety (should already be within [0,1])
    h = np.clip(h, 0, 1)
    # Due to small rounding, adjust sum
    total = np.sum(h)
    target = n/2
    diff = target - total
    if abs(diff) > 1e-8:
        # adjust one element
        # find index where we can add diff while staying within bounds
        if diff > 0:
            # choose i with h_i < 1
            idx = np.argmax(1 - h)  # max slack
            add = min(diff, 1 - h[idx])
            h[idx] += add
        else:
            idx = np.argmin(h)  # smallest value
            sub = min(-diff, h[idx])
            h[idx] -= sub
    return h

def evaluate_h(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def hill_climb(start_h, best_val, budget_remaining, rng):
    n = len(start_h)
    dx = 2 / n
    cur_h = start_h.copy()
    cur_val = best_val
    best_h = cur_h.copy()
    best_val_local = best_val
    start_time = time.time()
    iter = 0
    # total iteration limit to avoid infinite loop: e.g., 5 million
    # We'll maintain cooling schedule:
    max_iter = 2000000
    while time.time() - start_time < budget_remaining and iter < max_iter:
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        hi = cur_h[i]
        hj = cur_h[j]
        # bounds for delta
        lo = max(-hi, hj - 1)
        hi_bound = min(1 - hi, hj)
        if lo >= hi_bound:
            continue
        # choose delta uniformly
        delta = lo + (hi_bound - lo) * rng.random()
        # apply
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # evaluate
        new_val = evaluate_h(new_h, dx)
        # Simulated annealing: temperature schedule
        # Compute progress ratio
        ratio = iter / max_iter
        T = 0.01 * (0.1 ** ratio)  # start 0.01, end ~0.01*0.1^1 =0.001?
        # optionally compute diff
        diff = new_val - cur_val
        accept = False
        if diff < 0:
            accept = True
        else:
            if rng.random() < np.exp(-diff / T):

import time
import numpy as np

# ---------- helper functions ----------
def _evaluate(h, dx):
    """return max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    # correlation of h with (1-h) gives the integral (up to dx factor)
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _make_linear_shape(alpha, p, n):
    """
    Build a symmetric shape:
        h_i = alpha + beta * (1 - (d_i/max_d)**p)
    where d_i = distance of index i to the nearest endpoint.
    The parameters satisfy the integral constraint sum(h)=n/2.
    """
    max_d = (n - 1) // 2
    d = np.minimum(np.arange(n), np.arange(n)[::-1])
    coeff = 1 - (d / max_d) ** p
    K = np.sum(coeff)                     # Σ coeff_i
    target = n / 2.0
    beta = (target - n * alpha) / K       # forces the correct total mass
    if beta < 0:
        return None
    h = alpha + beta * coeff
    # numerical safety
    h = np.clip(h, 0.0, 1.0)
    # tiny mass correction (should be essentially zero)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # adjust a single coordinate that still stays in bounds
        if diff > 0:
            slack = 1 - h
            i = np.argmax(slack)
            add = min(diff, slack[i])
            h[i] += add
        else:
            slack = h
            i = np.argmin(slack)
            sub = min(-diff, slack[i])
            h[i] -= sub
    return h


def _grid_search(n):
    """search over a few families of symmetric shapes."""
    dx = 2.0 / n
    best_h = None
    best_val = np.inf
    # try several exponents p (including fractional ones)
    for p in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0]:
        # grid of alpha in [0,0.5]
        for alpha in np.linspace(0.0, 0.5, 501):
            h = _make_linear_shape(alpha, p, n)
            if h is None:
                continue
            # quick bound: edges must stay ≤1
            if np.max(h) > 1.000001:
                continue
            val = _evaluate(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    return best_h, best_val


def _hill_climb(start_h, start_val, time_budget, rng):
    """simple simulated‑annealing hill‑climb preserving the sum."""
    n = len(start_h)
    dx = 2.0 / n
    target = n / 2.0

    cur_h = start_h.copy()
    cur_val = start_val
    best_h = cur_h.copy()
    best_val = cur_val

    max_iter = 2_000_000      # safety cap
    start_time = time.time()
    for it in range(max_iter):
        if time.time() - start_time > time_budget:
            break

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = cur_h[i], cur_h[j]

        # feasible delta so both stay in [0,1] after hi+=delta, hj-=delta
        lo = max(-hi, hj - 1.0)
        hi_bound = min(1.0 - hi, hj)
        if lo >= hi_bound:
            continue
        delta = lo + (hi_bound - lo) * rng.random()
        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_val = _evaluate(new_h, dx)

        # temperature schedule – from 1e-2 down to 1e-5
        t = 1e-2 * (1e-3) ** (it / max_iter)

        diff = new_val - cur_val
        if diff < 0 or rng.random() < np.exp(-diff / t):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val


def _random_binary_half(n, rng):
    """generate a feasible vector consisting of 22 ones, one 0.5, rest zeros."""
    h = np.zeros(n)
    ones = rng.choice(n, size=22, replace=False)
    h[ones] = 1.0
    remaining = np.setdiff1d(np.arange(n), ones)
    half_idx = rng.choice(remaining)
    h[half_idx] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine the number of samples (use the supplied one if present)
    try:

    if val_grid < best_val:

    # -----------------------------------------------------------------
    # 3)  Some random binary‑half seeds

        val = _evaluate(h_rand, dx)
        if val < best_val:
            best_h, best_val = h_rand.copy(), val

    # -----------------------------------------------------------------
    # 4)  Local optimisation (simulated annealing) starting from the best so far
    remaining = max(0.0, budget_s - (time.time() - start_time) - 0.5)
    best_h, best_val = _hill_climb(best_h, best_val, remaining, rng)

    # -----------------------------------------------------------------
    # 5)  Final feasibility fix (tiny rounding errors)
    best_h = np.clip(best_h, 0.0, 1.0)
    current_sum = np.sum(best_h)
    diff = target - current_sum
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - best_h
            i = np.argmax(slack)
            best_h[i] += min(diff, slack[i])
        else:
            slack = best_h
            i = np.argmin(slack)
            best_h[i] -= min(-diff, slack[i])

    final_val = _evaluate(best_h, dx)

    # -----------------------------------------------------------------
    return (best_h.tolist(), float(final_val), int(n))
