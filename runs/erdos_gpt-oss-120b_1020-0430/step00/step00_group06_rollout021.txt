def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.convolve(h, (1 - h)[::-1], mode='full')) * dx

def gradient_for_shift(h, shift):
    # returns gradient array
    n = h.shape[0]
    if shift >= 0:
        forward = np.empty_like(h)
        forward[:n - shift] = h[shift:]
        forward[n - shift:] = 0.0
        backward = np.empty_like(h)
        backward[shift:] = h[:n - shift]
        backward[:shift] = 0.0
    else:
        s = -shift
        forward = np.empty_like(h)
        forward[s:] = h[:n - s]
        forward[:s] = 0.0
        backward = np.empty_like(h)
        backward[:n - s] = h[s:]
        backward[n - s:] = 0.0
    return 1.0 - forward - backward

def random_h(n, target_sum, rng):
    # Generate random start
    h = np.full(n, 0.5, dtype=float)
    noise = (rng.random(n) - 0.5) * 0.4  # amplitude up to ±0.2
    h += noise
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur != 0:
        h *= target_sum / cur
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # adjust via a single index
        i = rng.integers(0, n)
        max_inc = 1 - h[i]
        max_dec = h[i]
        if diff > 0:
            inc = min(diff, max_inc)
            h[i] += inc
        else:
            dec = min(-diff, max_dec)
            h[i] -= dec
    return h

import time, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """return max overlap integral for the discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return float(np.max(corr) * dx)


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """gradient of I(k)=∫h(x)(1‑h(x+k))dx at the shift giving the current max."""
    n = h.shape[0]
    if shift >= 0:
        forward = np.empty_like(h)
        forward[: n - shift] = h[shift:]
        forward[n - shift:] = 0.0
        backward = np.empty_like(h)
        backward[shift:] = h[: n - shift]
        backward[: shift] = 0.0
    else:
        s = -shift
        forward = np.empty_like(h)
        forward[s:] = h[: n - s]
        forward[: s] = 0.0
        backward = np.empty_like(h)
        backward[: n - s] = h[s:]
        backward[n - s:] = 0.0
    return 1.0 - forward - backward


def random_h(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """generate a random feasible step function (values in [0,1], sum = target)."""
    h = np.full(n, 0.5, dtype=float)
    # add modest noise
    h += (rng.random(n) - 0.5) * 0.4   # values stay in [0.3,0.7] before clipping
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if cur > 0:
        h *= target_sum / cur
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # fix the tiny residual on a random index
        i = rng.integers(0, n)
        if diff > 0:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
        else:
            dec = min(-diff, h[i])
            h[i] -= dec
    return h
