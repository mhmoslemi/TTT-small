def subgradient_max(h):

    # compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)  # shift

def evaluate(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def subgradient_max(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)
    if k == 0:
        return 1 - 2 * h
    abs_k = abs(k)
    grad = np.ones_like(h)
    grad[:n-abs_k] -= h[abs_k:]
    grad[abs_k:] -= h[:n-abs_k]
    return grad

import numpy as np
import time
import math

def _evaluate(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _subgradient_max(h):
    """Subgradient of max_k ∫h(x)(1-h(x+k))dx at the current shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)          # signed shift
    if k == 0:
        return 1.0 - 2.0 * h
    a = abs(k)
    grad = np.ones_like(h)
    # forward term
    grad[: n - a] -= h[a:]
    # backward term
    grad[a:] -= h[: n - a]
    return grad

def _project_bounds_sum(x, target):
    """Project x onto {0≤v≤1, sum v = target} (Euclidean projection)."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        v = np.clip(x - lam, 0.0, 1.0)
        s = v.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def _random_feasible(n, target):
    x = np.random.rand(n)
    return _project_bounds_sum(x, target)
