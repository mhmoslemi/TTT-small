def max_corr_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    offset = idx - (n - 1)  # offset can be negative
    # compute gradient using vectorized operations
    # g_i = (1 - h_{i+offset}) - h_{i-offset}
    # For i+offset out of bounds => treat as 0; similarly for i-offset out of bounds => 0.
    # Using numpy slicing.
    g = np.empty_like(h)
    # Initialize to zero
    g.fill(0.0)
    # Build terms
    # term1: (1 - h_{i+offset}) for i where i+offset in [0,n)
    if offset >= 0:
        # i valid such that i+offset < n => i in [0, n-offset)
        i_range = slice(0, n - offset)
        g[i_range] += 1 - h[i_range + offset]
        # term2: -h_{i-offset} where i - offset >=0 => i in [offset, n)
        i_range2 = slice(offset, n)
        g[i_range2] -= h[i_range2 - offset]
    else:
        # offset negative: let off = -offset >0
        off = -offset
        # term1: (1 - h_{i+offset}) => i+offset = i - off >=0 => i >= off
        i_range = slice(off, n)
        g[i_range] += 1 - h[i_range - off]
        # term2: -h_{i-offset} => i - offset = i + off < n => i < n - off
        i_range2 = slice(0, n - off)
        g[i_range2] -= h[i_range2 + off]
    g *= dx
    return g

from scipy.optimize import minimize

def optimize_h(initial_h, max_iter=200):
    n = len(initial_h)
    dx = 2.0 / n

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def jac(h):
        # gradient of objective: derivative of max term
        # compute correlation and find max
        corr = np.correlate(h, 1 - h, mode='full')
        # handle ties: choose first max
        idx = np.argmax(corr)
        offset = idx - (n - 1)

        # compute gradient as described
        g = np.zeros_like(h)
        if offset >= 0:
            g[:n - offset] += 1 - h[offset:]
            g[offset:] -= h[:n - offset]
        else:
            off = -offset
            g[off:] += 1 - h[:n - off]
            g[:n - off] -= h[off:]
        g *= dx
        return g

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n/2,
             'jac': lambda h: np.ones_like(h)},
            )
    bounds = [(0.0, 1.0)] * n

    res = minimize(obj, initial_h, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    return res.x, obj(res.x)

import time
import numpy as np
import scipy.optimize as opt

# ---------- helper functions -------------------------------------------------

def evaluate_c5(h):
    """Return the C5 value for a step function h (array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def max_corr_gradient(h):
    """Gradient of max_{k} ∫ h(x)*(1-h(x+k))dx w.r.t. h (sub‑gradient)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    off = idx - (n - 1)          # shift amount (in indices)

    g = np.zeros_like(h)
    if off >= 0:
        # terms (1 - h_{i+off}) where i+off is in range
        g[: n - off] += 1.0 - h[off:]
        # terms -h_{i-off} where i-off is in range
        g[off:] -= h[: n - off]
    else:
        off = -off
        g[off:] += 1.0 - h[: n - off]
        g[: n - off] -= h[off:]

    g *= dx
    return g


def project_onto_constraints(v, target, tol=1e-12, max_iter=200):
    """
    Euclidean projection of v onto the set {h | 0<=h_i<=1, sum h_i = target}.
    Uses a simple bisection on the Lagrange multiplier λ:
        h_i = clip(v_i - λ, 0, 1)
    """
    lo = np.min(v) - 1.0   # guarantees h_i = 1 for all i when λ=lo
    hi = np.max(v) + 1.0   # guarantees h_i = 0 for all i when λ=hi
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def upscale_h(old_h, n_new):
    """Linearly interpolate a step function to a finer grid."""
    n_old = old_h.shape[0]
    dx_old = 2.0 / n_old

    x_new = np.arange(n_new) * dx_new + dx_new / 2.0
    h_new = np.interp(x_new, x_old, old_h)
    h_new = project_onto_constraints(h_new, n_new / 2.0)
    return h_new


def optimize_h(start_h, max_iter=200):
    """Run SLSQP from start_h, enforcing box and sum constraints."""
    n = start_h.shape[0]
    target = n / 2.0

                                "disp": False})
