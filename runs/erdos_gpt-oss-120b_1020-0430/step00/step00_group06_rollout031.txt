def compute_delta_corr(h, i, j):
    # returns delta array same shape as corr
    n = len(h)
    size = 2*n - 1
    delta = np.zeros(size, dtype=np.int8) or maybe int16
    # contributions from i removal (h_i = 1)
    for l in range(-n+1, n):
        idx = l + n - 1
        # i as first index:
        ii = i + l
        if 0 <= ii < n:
            delta[idx] -= h[ii]   # h[i+l] original (since before swap)
        # i as second index:
        ii2 = i - l
        if 0 <= ii2 < n:
            delta[idx] -= h[ii2]
    # contributions from j addition (h_j originally 0, becomes 1)
    for l in range(-n+1, n):
        idx = l + n - 1
        jj = j + l
        if 0 <= jj < n:
            delta[idx] += h[jj]  # but h[jj] is original before j becomes 1 (so might be 0 or 1)
        jj2 = j - l
        if 0 <= jj2 < n:
            delta[idx] += h[jj2]
    # d⋅d term:
    delta[n-1] += 2     # l=0
    diff = j - i
    if -n+1 <= diff <= n-1:
        delta[diff + n - 1] += -1
    if -n+1 <= -diff <= n-1:
        delta[-diff + n - 1] += -1
    return delta

def compute_delta_corr(h, i, j, eps):
    n = len(h)
    base = n - 1
    delta = np.zeros(2*n-1, dtype=np.float64)
    # i increase by eps
    # indices for l where i + l in range: l from -i to n-1-i
    for l in range(-i, n - i):
        delta[l + base] += eps * h[i + l]
    for l in range(i - (n-1), i + 1):
        delta[l + base] += eps * h[i - l]
    # j decrease by eps (d_j = -eps)
    for l in range(-j, n - j):
        delta[l + base] -= eps * h[j + l]
    for l in range(j - (n-1), j + 1):
        delta[l + base] -= eps * h[j - l]
    # d⋅d term
    delta[base] += 2 * eps * eps
    diff = j - i
    delta[diff + base] -= eps * eps
    delta[-diff + base] -= eps * eps
    return delta

    while True:
        i = np.random.randint(n)

def project_h(h, target_sum):
    # ensure values within [0,1] and sum = target_sum
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # Compute diff
    diff = target_sum - h.sum()
    # If diff > 0, distribute to elements less than 1, up to capacity
    # If diff < 0, subtract from elements greater than 0.
    # Use simple loop: while |diff| > 1e-12:
    while abs(diff) > 1e-12:
        # indices free to adjust
        if diff > 0:
            free = np.where(h < 1 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        # amount per index
        amount = diff / free.size
        # limit to bounds
        for idx in free:
            cap = 1 - h[idx] if diff > 0 else -h[idx]
            add = np.clip(amount, -cap, cap)  # careful sign
            h[idx] += add
            diff -= add
        # Loop again if still diff.
    return h

def compute_max_I(h, corr, dx, target_sum):
    n = len(h)
    prefix = np.concatenate(([0.0], np.cumsum(h)))
    # lags
    # overlapping sums: array of length 2n-1
    # For lapse vector: use np.arange(-n+1, n)
    lags = np.arange(-n+1, n)
    # compute overlapping sum via vectorized:
    # use condition via where:
    positive = lags >= 0
    overlap = np.empty_like(lags, dtype=np.float64)
    # for positive lags:
    overlap[positive] = prefix[n - lags[positive]]
    # for negative:
    overlap[~positive] = target_sum - prefix[-lags[~positive]]
    I = overlap - corr
    return np.max(I) * dx

import time
import numpy as np


def _project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy sum == target (simple water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # distribute the remaining mass uniformly over the not‑bordered entries
    # repeat a few times to converge (few iterations are enough for tiny diffs)
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size == 0:
            # all at a bound, just distribute over everything without violating bounds
            free = np.arange(h.size)
        add = diff / free.size
        # limit addition so we stay inside [0,1]
        add = np.clip(add, -h[free] , 1 - h[free])
        h[free] += add
        diff = target - h.sum()
    return h


def _max_overlap_from_corr(h: np.ndarray, corr: np.ndarray, dx: float, tot: float) -> float:
    """Compute max_{k} ∫ h(x)(1-h(x+k))dx using correlation and prefix sums."""
    n = h.size
    pref = np.concatenate(([0.0], np.cumsum(h)))  # length n+1
    lags = np.arange(-n + 1, n, dtype=np.int64)
    # overlapping sum of h on the region where both signals overlap
    pos = lags >= 0
    overlap = np.empty_like(lags, dtype=np.float64)
    overlap[pos] = pref[n - lags[pos]]
    overlap[~pos] = tot - pref[-lags[~pos]]
    # I(k) = overlap - corr
    I = overlap - corr
    return I.max() * dx


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_limit = budget_s - 0.5  # safety margin

    # ----- initialise h -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.size
    else:
        n = 200                     # even size gives integer total mass
        target = n / 2.0
        # start from a random binary pattern (exact mass) then smooth a bit
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, int(target), replace=False)
        h[ones] = 1.0
        # add a little random noise to break symmetry
        noise = rng.random(n) * 0.1
        h = _project_to_feasible(h + noise, target)

    # enforce feasibility (sum = n/2, 0<=h<=1)
    target = n / 2.0
    h = _project_to_feasible(h, target)

    dx = 2.0 / n
    # initial full autocorrelation of h with itself
    corr = np.correlate(h, h, mode='full')
    cur_c5 = _max_overlap_from_corr(h, corr, dx, target)

    best_h = h.copy()
    best_c5 = cur_c5
    best_corr = corr.copy()

    # ----- simulated annealing -----------------------------------------
    it = 0
    last_improve = 0
    # temperature schedule parameters
    T0 = 0.05
    Tmin = 1e-5

    while time.time() - start < time_limit:
        it += 1
        # linear cooling based on elapsed fraction
        frac = (time.time() - start) / time_limit
        T = max(Tmin, T0 * (1.0 - frac))

        # pick two distinct indices with room to move mass
        # i will receive mass, j will lose mass
        i = rng.integers(n)
        while h[i] >= 1.0 - 1e-12:
            i = rng.integers(n)
        j = rng.integers(n)
        while j == i or h[j] <= 1e-12:
            j = rng.integers(n)

        eps_max = min(1.0 - h[i], h[j])
        if eps_max <= 0:
            continue
        eps = eps_max * rng.random()
        if eps <= 0:
            continue

        # ----- efficient correlation update (delta) -----------------------
        base = n - 1
        delta = np.zeros(2 * n - 1, dtype=np.float64)

        # i receives +eps
        delta[base - i: base - i + n] += eps * h
        delta[i: i + n] += eps * h[::-1]

        # j loses eps
        delta[base - j: base - j + n] -= eps * h
        delta[j: j + n] -= eps * h[::-1]

        # d⋅d term (quadratic, tiny but kept for exactness)
        delta[base] += 2.0 * eps * eps
        diff = j - i
        delta[diff + base] -= eps * eps
        delta[-diff + base] -= eps * eps

        new_corr = corr + delta

        # new h vector
        new_h = h.copy()
        new_h[i] += eps
        new_h[j] -= eps

        # compute new C5
        new_c5 = _max_overlap_from_corr(new_h, new_corr, dx, target)

        # acceptance (Metropolis)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            h = new_h
            corr = new_corr
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_corr = corr.copy()
                last_improve = it

        # occasional restart if stuck for many iterations
        if it - last_improve > 50000 and time.time() - start < time_limit * 0.9:
            # perturb the best solution slightly to explore new region
            h = best_h.copy()
            # random jitter on a few positions
            for _ in range(10):
                idx = rng.integers(n)
                delta_val = (rng.random() - 0.5) * 0.2
                h[idx] = np.clip(h[idx] + delta_val, 0.0, 1.0)
            h = _project_to_feasible(h, target)
            corr = np.correlate(h, h, mode='full')
            cur_c5 = _max_overlap_from_corr(h, corr, dx, target)
            last_improve = it

    # final verification using the original evaluation (full correlate)
    final_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx

    return best_h, float(final_c5), n
