def max_lag_and_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        L = lag
        # first part: i from 0 to n-L-1
        g[:n - L] += (1 - h[L:])
        # second part: i+L from L to n-1
        g[L:] -= h[:n - L]
    else:
        L = -lag
        g[L:] += (1 - h[:n - L])
        g[:n - L] -= h[L:]
    return lag, g

Test for lag 0: gradient is g = (1 - h) - h = 1 - 2h repeated.

def project_to_bounds_sum(v, target):
    # Clip to [0,1] after shifting by λ. Efficient method using bisection.
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0  # ensure range
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0.0, 1.0)

def optimize_h(n, time_limit):
    dx = 2.0 / n
    target = n / 2.0

    # start with uniform 0.5 vector
    h = np.full(n, 0.5)
    cur_obj = evaluate(h)   # compute

    best_h = h.copy()
    best_obj = cur_obj

    cur_h = h.copy()
    cur_obj = cur_obj

    start = time.time()
    it = 0
    # parameters
    prob_grad = 0.3  # probability for gradient step
    prob_swap = 0.7
    # step sizes
    delta_max = 0.2   # for swap step
    step_alpha = 0.2  # for gradient descent (scale)
    # Temperature T baseline for random acceptance if needed
    while time.time() - start < time_limit:
        it += 1
        # choose update
        if np.random.rand() < prob_grad:
            # gradient descent
            lag, g = max_lag_and_gradient(cur_h)
            # normalize gradient to have reasonable magnitude
            norm = np.linalg.norm(g)
            if norm < 1e-12:
                continue
            # propose step
            step = step_alpha / norm  # scaling
            h_new = cur_h - step * g
            # project back
            h_new = project_to_bounds_sum(h_new, target)
        else:
            # random donor/recipient move
            donors = np.where(cur_h > 1e-12)[0]
            recipients = np.where(cur_h < 1 - 1e-12)[0]
            if len(donors) == 0 or len(recipients) == 0:
                continue
            i = np.random.choice(donors)
            j = np.random.choice(recipients)
            delta_limit = min(delta_max, cur_h[i], 1 - cur_h[j])
            if delta_limit <= 0:
                continue
            delta = np.random.rand() * delta_limit
            h_new = cur_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            # no need for extra projection (already within bounds and sum)
        # evaluate
        obj_new = evaluate(h_new)
        # accept if improvement
        if obj_new < cur_obj:
            cur_h = h_new
            cur_obj = obj_new
            # update best
            if obj_new < best_obj:
                best_h = h_new.copy()
                best_obj = obj_new
        else:
            # maybe accept with small probability to escape
            # probability decreasing as temperature declines
            T = 0.001 * (1 - (time.time() - start) / time_limit)
            if np.random.rand() < np.exp(-(obj_new - cur_obj) / (T + 1e-12)):
                cur_h = h_new
                cur_obj = obj_new
        # optionally random restart after many iterations with no improvement
        if it % 20000 == 0:
            # if no improvement for last 20000 iterations, replace cur_h with best_h + small noise
            # we can maintain a counter for improvements
            pass

import time
import numpy as np

def _evaluate(h):
    """Return C5 for a step function h given by values on an equally spaced grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives the required lags
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr


def _max_lag_and_grad(h):
    """Find lag attaining the maximal correlation and return (lag, gradient)."""
    n = h.shape[0]
    # correlation without the dx factor – gradient does not need the scaling
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)      # signed lag
    g = np.zeros_like(h)

    if lag >= 0:           # shift to the right
        L = lag
        # contribution where h[i] appears as the first factor
        g[: n - L] += (1.0 - h[L:])
        # contribution where h[i] appears inside (1 - h[i+L])
        g[L:] -= h[: n - L]
    else:                  # shift to the left
        L = -lag
        g[L:] += (1.0 - h[: n - L])
        g[: n - L] -= h[L:]

    return lag, g


def _project_bounds_sum(v, target):
    """
    Euclidean projection of vector v onto the box [0,1]^n intersected with
    the hyperplane sum(v)=target.  Solution has form clip(v - λ,0,1) for a suitable λ.
    """
    lo, hi = -10.0, 10.0          # wide enough interval for λ
    for _ in range(60):          # binary search
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _optimize(n, time_limit, init_h=None):
    """
    Heuristic optimisation for a given number of points n.
    Returns (best_h, best_obj).  The time_limit is in seconds.
    """
    dx = 2.0 / n

    if init_h is not None:
        # interpolate to the required size

        h = _project_bounds_sum(h, target_sum)
    else:

    cur_h = h.copy()

    best_h = cur_h.copy()
    best_obj = cur_obj

    start = time.time()
    it = 0
    no_imp = 0

    step_alpha = 0.2         # base step size for gradient moves

    while time.time() - start < time_limit * 0.95:
        it += 1
        if np.random.rand() < prob_grad:
            # gradient descent on the currently worst lag
            _, g = _max_lag_and_grad(cur_h)
            norm = np.linalg.norm(g)
            if norm < 1e-12:
                continue
            step = step_alpha / norm
            h_new = cur_h - step * g
            h_new = _project_bounds_sum(h_new, target_sum)
        else:
            # a random donor–recipient transfer that respects the box constraints
            donors = np.where(cur_h > 1e-12)[0]

            if donors.size == 0 or recips.size == 0:

            i = np.random.choice(donors)
            j = np.random.choice(recips)
            delta = np.random.rand() * min(delta_max, cur_h[i], 1.0 - cur_h[j])
            if delta <= 0:

            h_new = cur_h.copy()

        else:

            cur_obj, _ = _evaluate(cur_h)
            no_imp = 0

    # final projection to guarantee the exact sum
    best_h = _project_bounds_sum(best_h, target_sum)
    best_obj, _ = _evaluate(best_h)
    return best_h, best_obj
