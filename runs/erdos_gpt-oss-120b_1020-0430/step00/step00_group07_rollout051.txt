    h = np.zeros(N)
    h[:N//2] = 1.0
    # random shuffle indices maybe.
    indices = np.arange(N)
    rng.shuffle(indices)
    h = h[indices]  # random permutation
    # apply random transfers to create fractional values
    for _ in range(steps):
        i, j = rng.choice(N, 2, replace=False)
        # direction random via source/target selection
        if rng.rand() < 0.5:
            src, tgt = i, j
        else:
            src, tgt = j, i
        max_transfer = min(h[src], 1.0 - h[tgt])
        if max_transfer <= 0:
            continue
        eps = rng.rand() * max_transfer * 0.5  # up to half
        h[src] -= eps
        h[tgt] += eps
    return h

        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    # ensure initial feasible
    def enforce_feasibility(h):
        # Clip and adjust sum make exact.
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) > 1e-10:
            # Adjust proportionally on non-boundary entries
            # Compute diff = target_sum - s
            diff = target_sum - s

                h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

                delta = diff / free.size
                h[free] += delta
                h = np.clip(h, 0.0, 1.0)

            h = h * (target_sum / s)
            # Ensure within bounds after scaling

    # Clip and adjust sum by scaling, iterations.
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h
    # Determine difference
    diff = target_sum - s
    # Distribute diff across entries that are not at bounds
    free = (h > 1e-12) & (h < 1.0 - 1e-12)
    free_idx = np.where(free)[0]
    if free_idx.size > 0:
        # distribute equally (or proportionally). Using equal.
        delta = diff / free_idx.size
        h[free_idx] += delta
        # Clip again
        h = np.clip(h, 0.0, 1.0)
    else:
        # If all at bounds, adjust one element
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    # After adjust, may still be off due to clipping; apply scaling
    s = h.sum()
    if abs(s - target_sum) > 1e-9:
        # scale whole vector to match sum, then clip again
        h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
    return h

        if max_transfer <= 0:

        # Actually we want allow smaller moves later. Use factor = 0.5 * (1 - frac) + 0.05
        factor = 0.5 * (1 - ((now - start_time) / total_budget)) + 0.05
        eps = max_transfer * rng.random() * factor
        new_h = h.copy()
        new_h[src] -= eps
        new_h[tgt] += eps
        new_score = compute_c5(new_h, dx)
        if new_score < cur_score - 1e-12:
            h = new_h
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_h = new_h.copy()
                best_score = new_score
        else:
            # Metropolis acceptance
            diff = new_score - cur_score
            if T > 0.0 and rng.random() < math.exp(-diff / T):
                h = new_h
                cur_score = new_score
        it += 1
    # return best found in this run
    return best_h, best_score

import time

import numpy as np

def compute_c5(h, dx):
    """Return max overlap integral for step vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def enforce_feasibility(h, target_sum):
    """Clip to [0,1] and adjust to have exact sum target_sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h
    diff = target_sum - s
    free = (h > 1e-12) & (h < 1.0 - 1e-12)

        h[free_idx] += delta
        h = np.clip(h, 0.0, 1.0)

        idx = np.argmin(np.abs(h - 0.5))

    s = h.sum()
    if abs(s - target_sum) > 1e-9:
