def project(v, target_sum):
    low = -np.max(v) - 1.0  # ensures v+low <=0
    high = 1.0 - np.min(v) + 1.0  # ensures v+high >=1
    for _ in range(60):
        mid = (low + high) / 2
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) / 2, 0.0, 1.0)

import time
import numpy as np

def _proj_sum_clip(v, target):
    """Project v onto [0,1]^n with sum == target via a simple bisection."""
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _corr_max(h, dx):
    """Maximum correlation value for current h."""
    c = np.correlate(h, 1.0 - h, mode="full")
    return c.max() * dx, c


def _grad_corr(h, shift):
    """Gradient of correlation w.r.t. h for a given integer shift."""
    n = h.size
    if shift == 0:
        return 1.0 - 2.0 * h
    if shift > 0:
        fwd = np.concatenate([h[shift:], np.zeros(shift)])
        bwd = np.concatenate([np.zeros(shift), h[: n - shift]])
    else:  # shift < 0
        s = -shift
        fwd = np.concatenate([np.zeros(s), h[: n - s]])
        bwd = np.concatenate([h[s:], np.zeros(s)])
    return -(fwd + bwd)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.size
    else:
        n = 200                     # fallback size (even, <1k)
        h = np.full(n, 0.5)
        # small perturbation and enforce sum constraint
        h = np.clip(h + rng.uniform(-0.1, 0.1, size=n), 0.0, 1.0)
        target_sum = n * 0.5
        h = _proj_sum_clip(h, target_sum)

    target_sum = n * 0.5
    dx = 2.0 / n

    # make sure the starting point respects the constraints
    h = np.clip(h, 0.0, 1.0)
    h = _proj_sum_clip(h, target_sum)

    best_h = h.copy()
    best_val, _ = _corr_max(best_h, dx)

    # ------------------------------------------------------------------
    # optimisation loop (gradient descent on the worst shift)
    # ------------------------------------------------------------------
    max_iters = 20000
    eta0 = 0.5

    cur_h = h.copy()
    cur_val = best_val

    for it in range(1, max_iters + 1):
        # time check
        if time.time() - start > budget_s * 0.95:
            break

        # current worst shift
        cur_val, full_corr = _corr_max(cur_h, dx)
        idx = int(np.argmax(full_corr))
        shift = idx - (n - 1)          # map index to signed shift

        # compute gradient of the *correlation* (not of -C5)
        grad = _grad_corr(cur_h, shift)

        # simple schedule for step size
        eta = eta0 * (1.0 - it / max_iters) ** 1.5

        # take a step opposite to the gradient (to reduce the max correlation)
        cand = cur_h - eta * grad
        cand = np.clip(cand, 0.0, 1.0)
        cand = _proj_sum_clip(cand, target_sum)

        cand_val, _ = _corr_max(cand, dx)

        # accept if improvement
        if cand_val < cur_val - 1e-9:
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val - 1e-12:
                best_val, best_h = cand_val, cand.copy()
        else:
            # occasional random jitter to escape flat regions
            if it % 500 == 0:
                jitter = rng.uniform(-0.02, 0.02, size=n)
                tmp = np.clip(cur_h + jitter, 0.0, 1.0)
                tmp = _proj_sum_clip(tmp, target_sum)
                tmp_val, _ = _corr_max(tmp, dx)
                if tmp_val < cur_val:
                    cur_h, cur_val = tmp, tmp_val

    # final evaluation on the best candidate found
    final_val, _ = _corr_max(best_h, dx)

    return (best_h.tolist(), float(final_val), int(n))
