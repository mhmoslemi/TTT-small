import time, math, random
import numpy as np

def _compute_c5(arr):
    """Return the C5 value for a discrete step function."""
    n = arr.size
    conv = np.correlate(arr, 1.0 - arr, mode="full")
    return conv.max() * (2.0 / n)


def _project_onto_box_and_sum(vec, target_sum):
    """Euclidean projection of vec onto {0<=x<=1, sum x = target_sum}."""
    lo, hi = -1.0, 2.0  # wide enough bounds for the binary case
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(vec - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - (lo + hi) * 0.5, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C5 value."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # ------------------------------------------------------------------
    # initialise h ------------------------------------------------------
    if "initial_h_values" in globals():

        n = h0.size
        # force binary (0/1) while keeping the total mass n/2
        bin_h = (h0 > 0.5).astype(float)
        cur = int(bin_h.sum())
        tgt = n // 2
        if cur > tgt:
            idx = np.where(bin_h == 1)[0]
            rem = np.random.choice(idx, cur - tgt, replace=False)
            bin_h[rem] = 0.0
        elif cur < tgt:
            idx = np.where(bin_h == 0)[0]
            add = np.random.choice(idx, tgt - cur, replace=False)
            bin_h[add] = 1.0
        h = bin_h
    else:
        # fall back to a random binary vector, size 200 (even)
        n = 200
        h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    # basic data structures for fast swaps
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # optimisation parameters
    max_iters = 5_000_000                # generous upper bound
    T0 = 0.8                              # initial temperature

    log_ratio = math.log(T_end / T0)

    it = 0

            if random.random() < math.exp(-abs(delta) / T):
