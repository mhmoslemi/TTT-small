    i, j = np.random.choice(n, 2, replace=False)
    max_decrease_i = h[i] # can't go below 0
    max_increase_i = 1 - h[i]
    max_decrease_j = h[j]
    max_increase_j = 1 - h[j]
    # we will move delta from i to j: h_i' = h_i - delta, h_j' = h_j + delta
    # delta must satisfy -max_decrease_i <= delta <= max_increase_j
    # to keep both within bounds.

def _solve_lp(n, target_sum):
    if cp is None:
        raise RuntimeError("cvxpy not available")
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum, t >= 0]
    w = []
    for k in range(n):
        wk = cp.Variable(n - k)
        w.append(wk)
        # link wk with h[i] and h[i+k]
        # using loops
        for i in range(n - k):
            constraints.append(wk[i] <= h[i])
            constraints.append(wk[i] <= h[i + k])
            constraints.append(wk[i] >= h[i] + h[i + k] - 1)
            constraints.append(wk[i] >= 0)
        # overlap constraint
        constraints.append(cp.sum(h[:n - k]) - cp.sum(wk) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)

def _solve_lp(n: int, target_sum: float):
    """Linear‐programming relaxation using McCormick envelopes.
    Returns a feasible h (array) and the lower bound t."""
    if cp is None:  # pragma: no cover
        raise RuntimeError("cvxpy not available")
    h = cp.Variable(n)
    t = cp.Variable()
    cons = [h >= 0, h <= 1, cp.sum(h) == target_sum, t >= 0]

    w = []  # list of vectors w_k for each shift k
    for k in range(n):
        wk = cp.Variable(n - k)
        w.append(wk)
        # McCormick constraints linking wk[i] to h[i] and h[i+k]
        for i in range(n - k):
            cons.append(wk[i] <= h[i])
            cons.append(wk[i] <= h[i + k])
            cons.append(wk[i] >= h[i] + h[i + k] - 1)
            cons.append(wk[i] >= 0)
        # overlap constraint: sum_{i< n‑k} h_i – sum wk_i ≤ t
        cons.append(cp.sum(h[: n - k]) - cp.sum(wk) <= t)

    prob = cp.Problem(cp.Minimize(t), cons)
    # ECOS is a fast interior‑point LP solver available in the environment
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    return np.clip(h.value, 0.0, 1.0), t.value

    n = h.shape[0]

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
