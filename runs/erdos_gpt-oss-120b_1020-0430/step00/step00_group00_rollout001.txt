import time
import numpy as np

def _make_feasible(arr: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the target sum."""
    arr = np.clip(arr, 0.0, 1.0)
    cur = arr.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return arr
    # Simple water‑filling / draining
    if diff > 0:  # need to add mass
        for i in range(len(arr)):
            avail = 1.0 - arr[i]
            if avail <= 0:
                continue
            inc = min(avail, diff)
            arr[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:        # need to remove mass
        for i in range(len(arr)):
            avail = arr[i]
            if avail <= 0:
                continue
            dec = min(avail, -diff)
            arr[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return arr

def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h = 1 that minimizes
    C5 = max_k ∫ h(x)(1 - h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    try:
        # Provided by the harness (list/array of length n_points)
        h0 = np.asarray(initial_h_values, dtype=float)
    except NameError:
        # No parent construction: fall back to a simple feasible pattern
        n_points = 45
        target = n_points / 2.0
        h0 = np.zeros(n_points, dtype=float)
        whole = int(np.floor(target))
        h0[:whole] = 1.0
        frac = target - whole
        if frac > 0:
            h0[whole] = frac
    else:
        n_points = h0.size

    target_sum = n_points / 2.0
    h = _make_feasible(h0.copy(), target_sum)

    dx = 2.0 / n_points

    # ------------------------------------------------------------------
    # Local search (simulated annealing / hill climbing)
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_val = _c5_value(best_h, dx)
    cur_h = best_h.copy()
    cur_val = best_val

    start_time = time.time()
    max_iter = 10_000_000  # generous upper bound
    T0 = 0.02               # initial temperature (scale of objective differences)
    min_T = 1e-8

    # Statistics for occasional restart
    no_improve = 0
    RESTART_AFTER = 200_000  # iterations without improvement

    it = 0
    while True:
        # Time check
        if time.time() - start_time > budget_s * 0.99:
            break

        # Temperature schedule (linear cooling)
        T = max(min_T, T0 * (1.0 - it / max_iter))

        # Propose a move: transfer epsilon from i to j
        i, j = rng.integers(n_points, size=2)
        while i == j:
            j = rng.integers(n_points)
        eps_max = min(cur_h[i], 1.0 - cur_h[j])
        if eps_max <= 0:
            it += 1
            continue

        # Step size adapts with temperature (larger steps early)
        step_factor = max(0.01, T / T0)  # never go below 1% of eps_max
        eps = eps_max * step_factor * rng.random()

        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_val = _c5_value(new_h, dx)

        # Acceptance test
        if new_val < cur_val:
            accept = True
        else:
            # uphill move probability
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Occasionally restart from the best solution with a small jitter
        if no_improve >= RESTART_AFTER:
            cur_h = best_h.copy()
            cur_val = best_val
            # Apply a few random tiny transfers to shake out of flat regions
            for _ in range(5):
                a, b = rng.integers(n_points, size=2)
                if a == b:
                    continue
                eps_m = min(cur_h[a], 1.0 - cur_h[b])
                if eps_m <= 0:
                    continue
                eps_m = eps_m * 0.05 * rng.random()
                cur_h[a] -= eps_m
                cur_h[b] += eps_m
            cur_val = _c5_value(cur_h, dx)
            no_improve = 0

        it += 1

    # Final safety: enforce feasibility (tiny drift may have occurred)
    best_h = _make_feasible(best_h, target_sum)

    # Compute final bound
    final_c5 = _c5_value(best_h, dx)

    return best_h, final_c5, n_points
