def subgrad_step(h, alpha):
    n = len(h)
    dx = 2.0/n
    # compute correlation f
    f = np.correlate(h, 1 - h, mode='full') * dx
    max_val = np.max(f)
    # choose index randomly among those within tolerance
    max_idxs = np.where(np.isclose(f, max_val, atol=1e-12))[0]
    idx = np.random.choice(max_idxs)
    shift = idx - (n - 1)
    # compute gradient
    grad = np.full_like(h, dx)
    i = np.arange(n)
    i_fwd = i + shift
    mask_fwd = (i_fwd >= 0) & (i_fwd < n)
    grad[mask_fwd] -= dx * h[i_fwd[mask_fwd]]
    i_bwd = i - shift
    mask_bwd = (i_bwd >= 0) & (i_bwd < n)
    grad[mask_bwd] -= dx * h[i_bwd[mask_bwd]]
    # update
    h_new = h - alpha * grad
    # project
    h_new = project_onto_box_and_sum(h_new, 0., 1., target_sum = n/2)
    return h_new, max_val

def random_transfer(h, step):

def project_onto_box_and_sum(v, lower, upper, target):
    # v is numpy array
    # Clip first
    v = np.clip(v, lower, upper)
    # Compute current sum
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    # Determine direction: need to increase sum or decrease
    # We'll use iterative water-filling.
    # To avoid infinite loops due to floating point, max 100 iterations.
    for _ in range(100):
        diff = target - s
        # find indices that are not at bound in direction of movement
        if diff > 0:
            mask = v < upper - 1e-12
        else:
            mask = v > lower + 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            # cannot adjust further
            break
        # amount to distribute equally among free variables
        delta = diff / cnt
        # Add delta, but ensure bound
        v[mask] += delta
        # Clip again
        v = np.clip(v, lower, upper)
        s = v.sum()
        if abs(s - target) < 1e-12:
            break
    # After loops, due to rounding may be slightly off; correct by adding small offset to some free variables.
    # Ensure sum exactly target by adjusting a random free variable
    s = v.sum()
    if abs(s - target) > 1e-8:
        # compute difference
        diff = target - s
        # adjust one variable within bounds
        # find index where we can add diff without violation
        # search
        for i in range(len(v)):
            if lower <= v[i] + diff <= upper:
                v[i] += diff
                break
    return v

import time, math, random
import numpy as np

# ---------- utility functions ----------
def evaluate_c5(h):
    """Return the overlap value C5 for a step function h."""
    n = h.size
    dx = 2.0 / n
    # use the same definition as the harness
    if 'evaluate_erdos_solution' in globals():
        # the helper already returns the correct bound
        return float(evaluate_erdos_solution(h))
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_onto_box_and_sum(v, lo, hi, target):
    """Euclidean projection of v onto {lo<=x<=hi, sum(x)=target}."""
    v = np.clip(v, lo, hi)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    # water‑filling style iterative adjustment
    for _ in range(100):
        diff = target - s
        if diff > 0:
            mask = v < hi - 1e-12
        else:
            mask = v > lo + 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        v[mask] += delta
        v = np.clip(v, lo, hi)
        s = v.sum()
        if abs(s - target) < 1e-12:
            break
    # final tiny correction if needed
    if abs(v.sum() - target) > 1e-9:
        diff = target - v.sum()
        for i in range(v.size):
            if lo <= v[i] + diff <= hi:
                v[i] += diff
                break
    return v

def random_feasible(n):
    """Create a random vector satisfying the constraints."""
    v = np.random.rand(n)
    return project_onto_box_and_sum(v, 0.0, 1.0, n / 2.0)

def random_transfer(h, step):
    """Move a small amount of mass from one index to another."""
    n = h.size
    src = np.where(h > 1e-12)[0]
    tgt = np.where(h < 1.0 - 1e-12)[0]
    if src.size == 0 or tgt.size == 0:
        return h.copy()
    i = np.random.choice(src)
    j = np.random.choice(tgt)
    max_amt = min(h[i], 1.0 - h[j])
    amt = min(step, max_amt) * np.random.rand()
    new_h = h.copy()
    new_h[i] -= amt
    new_h[j] += amt
    return new_h

def subgrad_step(h, alpha):
    """One projected sub‑gradient step using the worst lag."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    # pick a worst lag (index in full correlation)
    worst = np.where(np.isclose(corr, max_val, atol=1e-12))[0]
    idx = np.random.choice(worst)
    shift = idx - (n - 1)               # shift in index units
    i = np.arange(n)
    grad = np.full(n, dx)               # constant part
    fwd = i + shift
    mask = (fwd >= 0) & (fwd < n)
    grad[mask] -= dx * h[fwd[mask]]
    bwd = i - shift
    mask = (bwd >= 0) & (bwd < n)
    grad[mask] -= dx * h[bwd[mask]]
    new_h = h - alpha * grad
    # projection back to feasible set
    new_h = project_onto_box_and_sum(new_h, 0.0, 1.0, n / 2.0)
    new_val = evaluate_c5(new_h)
    return new_h, new_val, max_val

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # initialise h
    if 'initial_h_values' in globals():

    best_h = h.copy()
    best_val = evaluate_c5(best_h)

    start = time.time()
    max_iter = 500000                # hard cap, will be cut by time
    # schedules
    step0 = 0.5                      # for sub‑gradient
    rand_step0 = 0.05                # for random transfers
    temp0 = 0.02                     # simulated‑annealing temperature

    for it in range(1, max_iter + 1):
        if time.time() - start > budget_s * 0.95:
            break

        # cooling schedules
        prog = it / max_iter
        alpha = step0 * (1.0 - prog) + 0.01
        rand_step = rand_step0 * (1.0 - prog) + 0.001
        temp = temp0 * (1.0 - prog)

        # choose move type
        if np.random.rand() < 0.2:
            # sub‑gradient move
            new_h, new_val, cur_max = subgrad_step(h, alpha)
        else:
            # random transfer move
            cand = random_transfer(h, rand_step)
            new_val = evaluate_c5(cand)
            new_h = cand
            cur_max = new_val

        # acceptance
        accept = False
        if new_val < best_val - 1e-12:
            accept = True
        elif new_val < best_val + 1e-12:   # tie – keep current to avoid endless loops
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(new_val - best_val) / temp)

                    accept = True
        if accept:

            if new_val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = new_val

    best_val = evaluate_c5(best_h)

    return best_h, float(best_val), int(n)
