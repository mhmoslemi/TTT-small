import numpy as np
import time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def project_onto_capped_simplex(v, target_sum):
    # binary search lambda
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust any residual
    diff = target_sum - w.sum()
    if abs(diff) > 1e-9:
        # try to distribute diff on non-boundary entries
        idxs = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if idxs.size > 0:
            w[idxs[0]] += diff
            w = np.clip(w, 0.0, 1.0)
        else:
            # add diff to an element not at bound if possible
            if diff > 0:
                i = np.argmin(w)
                w[i] = min(1.0, w[i] + diff)
            else:
                i = np.argmax(w)
                w[i] = max(0.0, w[i] + diff)
    return w

def random_feasible_vector(n, rng):
    target = n / 2.0
    base = int(np.floor(target))
    frac = target - base
    h = np.zeros(n)
    if base > 0:
        idx = rng.choice(n, size=base, replace=False)
        h[idx] = 1.0
    if frac > 1e-12:
        # pick an index not already set
        remaining_idx = np.where(h == 0)[0]
        if remaining_idx.size == 0:
            # shouldn't happen; fallback set to first
            i = 0
        else:
            i = rng.choice(remaining_idx)
        h[i] = frac
    return h

def pgd_optimize(h0, time_budget, rng):
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    start = time.time()
    # initial step size
    step_base = 0.5
    no_improve = 0

        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # compute max shift

        idx_max = np.argmax(conv)
        shift = idx_max - (n - 1)
        # compute gradient
        if shift >= 0:
            forward = np.concatenate([h[shift:], np.zeros(shift)])
            backward = np.concatenate([np.zeros(shift), h[:n - shift]])
        else:
            s = -shift
            forward = np.concatenate([np.zeros(s), h[:n - s]])
            backward = np.concatenate([h[s:], np.zeros(s)])
        grad = (1.0 - forward) - backward
        # line search

        improved = False
        for _ in range(20):

            if trial_score < cur_score - 1e-12:

                improved = True
                if trial_score < best_score - 1e-12:

import numpy as np
import time

def _compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection onto 0<=w_i<=1, sum w_i = target."""
    lo = np.min(v) - 1.0      # guarantees all entries become 1
    hi = np.max(v)            # guarantees all entries become 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        free = np.where((w > 1e-12) & (w < 1.0 - 1e-12))[0]
        if free.size:
            w[free[0]] += diff
            w = np.clip(w, 0.0, 1.0)
        else:
            if diff > 0:
                i = np.argmin(w)
                w[i] = min(1.0, w[i] + diff)
            else:
                i = np.argmax(w)
                w[i] = max(0.0, w[i] + diff)
    return w

def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a feasible vector (0<=h<=1, sum = n/2) with mostly 0/1 entries."""
    target = n / 2.0
    base = int(np.floor(target))
    frac = target - base
    h = np.zeros(n)
    if base:
        idx = rng.choice(n, size=base, replace=False)
        h[idx] = 1.0
    if frac > 1e-12:
        free = np.where(h == 0)[0]
        i = rng.choice(free) if free.size else 0
        h[i] = frac
    return h

def _pgd_optimize(h0: np.ndarray, budget: float, rng: np.random.Generator):
    """Projected gradient descent on the max‑overlap objective."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    cur = _compute_c5(h)
    best_h, best = h.copy(), cur
    start = time.time()
    step_base = 0.5
    stagn = 0
    while time.time() - start < budget:
        conv = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(conv)
        shift = idx - (n - 1)

        # forward = h shifted left by `shift`, zero‑padded
        if shift >= 0:
            forward = np.concatenate([h[shift:], np.zeros(shift)])
            backward = np.concatenate([np.zeros(shift), h[: n - shift]])
        else:
            s = -shift
            forward = np.concatenate([np.zeros(s), h[: n - s]])
            backward = np.concatenate([h[s:], np.zeros(s)])

        grad = (1.0 - forward) - backward

        # line search
        step = step_base
        improved = False
        for _ in range(20):
            trial = h - step * grad
            trial = _project_capped_simplex(trial, target)
            val = _compute_c5(trial)
            if val < cur - 1e-12:
                h, cur = trial, val
                improved = True
                if val < best - 1e-12:
                    best, best_h = val, trial.copy()
                step_base = min(step_base * 1.1, 1.0)
                break
            step *= 0.5

        if not improved:
            # random exchange move as fallback
            i, j = rng.choice(n, size=2, replace=False)
            if rng.random() < 0.5:
                up, down = i, j
            else:
                up, down = j, i
            max_up = 1.0 - h[up]
            max_down = h[down]
            delta_max = min(max_up, max_down)
            if delta_max > 0:
                delta = delta_max * rng.random() * 0.2
                trial = h.copy()
                trial[up] += delta
                trial[down] -= delta
                val = _compute_c5(trial)
                if val < cur - 1e-12:
                    h, cur = trial, val
                    improved = True
                    if val < best - 1e-12:
                        best, best_h = val, trial.copy()

        if not improved:
            stagn += 1
            if stagn >= 200:
                break
        else:
            stagn = 0
    return best_h, best

def _local_random_search(h: np.ndarray, budget: float, rng: np.random.Generator):
    """Small‑scale stochastic hill climbing preserving the sum exactly."""
    n = h.shape[0]
    best_h, best = h.copy(), _compute_c5(h)
    start = time.time()
    while time.time() - start < budget:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            up, down = i, j
        else:
            up, down = j, i
        max_up = 1.0 - best_h[up]
        max_down = best_h[down]
        delta_max = min(max_up, max_down)
        if delta_max <= 0:
            continue
        delta = delta_max * rng.random() * 0.05
        cand = best_h.copy()
        cand[up] += delta
        cand[down] -= delta
        val = _compute_c5(cand)
        if val < best - 1e-12:
            best_h, best = cand, val
    return best_h, best

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    init = globals().get("initial_h_values", None)
    if init is not None:
        init_h = np.asarray(init, dtype=float)
    else:
        init_h = None

    # candidate resolutions
    n_list = []
    if init_h is not None:
        n0 = init_h.shape[0]
        n_list.append(n0)
        if n0 * 2 <= 200:
            n_list.append(n0 * 2)
    else:
        n_list = [45, 90]

    best_h, best_score = None, np.inf
    overall_start = time.time()
    def _time_left():
        return budget_s - (time.time() - overall_start)

    for n in n_list:
        starts = []
        # deterministic starts
        if init_h is not None and init_h.shape[0] == n:
            starts.append(init_h.copy())
        elif init_h is not None and init_h.shape[0] * 2 == n:
            starts.append(np.repeat(init_h, 2))
        # a few random starts
        for _ in range(3):
            starts.append(_random_feasible(n, rng))

        for h0 in starts:
            if _time_left() < 5.0:
                break
            # allocate modest chunk for the main PGD phase
            alloc = min(200.0, _time_left() * 0.5)
            h_opt, sc_opt = _pgd_optimize(h0, alloc, rng)

            # brief random refinement if time permits
            if _time_left() > 2.0:
                h_opt, sc_opt = _local_random_search(h_opt, min(30.0, _time_left()), rng)

            if sc_opt < best_score:
                best_h, best_score = h_opt.copy(), sc_opt
            if best_score <= 0.3808:
                break
        if best_score <= 0.3808:
            break

    if best_h is None:
        # ultimate fallback
        n = 45
        best_h = _random_feasible(n, rng)
        best_score = _compute_c5(best_h)

    # final projection to guarantee feasibility
    n = best_h.shape[0]
    best_h = _project_capped_simplex(best_h, n / 2.0)
    best_score = _compute_c5(best_h)

    return (best_h, float(best_score), int(n))
