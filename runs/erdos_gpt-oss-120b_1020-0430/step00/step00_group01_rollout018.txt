def evaluate(h):
    conv = np.correlate(h, 1 - h, mode='full')
    return conv.max() * (2.0 / len(h))

import time
import math
import random


def _raw_max_corr(h: np.ndarray) -> float:
    """Maximum of sum_i h[i]*(1-h[i+k]) over all integer shifts k."""
    n = h.shape[0]
    L = 2 * n
    # linear (non‑circular) correlation via FFT
    f = np.fft.rfft(h, n=L)
    g = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(f * np.conj(g), n=L)
    # conv length = L, the last entry is zero (no overlap)
    return float(conv.max())
