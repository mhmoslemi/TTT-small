def project_onto_bounds_and_sum(v, S, lower=0., upper=1., max_iter=100, tol=1e-9):
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        y = np.clip(v + mid, lower, upper)
        s = np.sum(y)
        if s > S:
            hi = mid
        else:
            lo = mid
        if abs(s - S) < tol:
            break
    return y

    new_h[i_down] -= delta
    return new_h, True

def project_onto_sum_and_bounds(v, S):
    # binary search for λ
    lower = -1e6
    upper = 1e6
    for _ in range(100):
        mid = (lower+upper)/2
        y = np.clip(v + mid, 0, 1)
        s = y.sum()
        if s > S:
            upper = mid
        else:
            lower = mid
        if abs(s - S) < 1e-9:
            break
    return y

function get_worst_shift(h):
    corr = np.correlate(h, h, mode='full')
    # compute min correlation
    min_corr = np.min(corr)
    idx = np.argmin(corr)
    s = idx - (len(h) - 1)

    n = len(h)
    g = np.zeros(n)

        # pairs (i,i+s) for i in 0..n-s-1

        left = slice(0, n - s)

        g[left] += h[right]
        g[right] += h[left]
    else:

import time
import numpy as np
import random

def _project_to_sum_and_bounds(v, target_sum, eps=1e-12, max_iter=60):
    """
    Project vector v onto the box [0,1] with the additional affine constraint sum(v)=target_sum.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + lo, 0.0, 1.0)

def _c5_value(h):
    """compute C5 for a step function sampled in h"""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def _worst_shift_and_min_corr(h):
    """return (shift, min_autocorr) where shift is integer lag (negative..positive)"""
    n = h.shape[0]
    auto = np.correlate(h, h, mode='full')
    idx = int(np.argmin(auto))
    shift = idx - (n - 1)
    return shift, auto[idx]

def _gradient_for_shift(h, shift):
    """gradient of the autocorrelation at a given shift (w.r.t. the vector h)"""
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    if shift >= 0:
        left = slice(0, n - shift)
        right = slice(shift, n)
    else:
        s = -shift
        left = slice(0, n - s)
        right = slice(s, n)
    # derivative contributions from the two ends of each overlapping pair
    g[left] += h[right]
    g[right] += h[left]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. Choose discretisation size
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = globals()['initial_h_values']
        n0 = int(len(init))
        # target size (multiple of initial, but not huge)
        n = max(100, n0 * 3 // 2)               # a modest increase over the parent
        # upscale by simple linear interpolation
        x_src = np.linspace(0, 2, n0, endpoint=False)
        x_tgt = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_tgt, x_src, init)
    else:
        n = 200                                   # default size when no parent is given
        h = np.full(n, 0.5)

    # enforce the integral constraint exactly (sum = n/2)
    target_sum = n / 2.0
    h = _project_to_sum_and_bounds(h, target_sum)

    # ----------------------------------------------------------------------
    # 2. Core optimisation loop (simulated‑annealing / greedy)
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_val = _c5_value(best_h)

    cur_h = h.copy()
    cur_val = best_val

    # parameters for the schedule
    max_iters = 500_000
    max_delta_initial = 0.25
    min_delta = 1e-4
    temp_initial = 0.03

    for it in range(max_iters):
        # time check
        if time.perf_counter() - start_time > budget_s * 0.95:
            break

        # cooling / step‑size schedules
        frac = it / max_iters
        temp = max(temp_initial * (1.0 - frac), 1e-4)
        max_delta = max_delta_initial * (1.0 - frac) + min_delta

        # identify the lag with smallest self‑overlap (the bottleneck)
        shift, _ = _worst_shift_and_min_corr(cur_h)

        # gradient that tells where an increase would improve that lag
        grad = _gradient_for_shift(cur_h, shift)

        # choose a position to increase (large gradient, not already at 1)
        # and a position to decrease (small gradient, not already at 0)
        med = np.median(grad)
        inc_candidates = np.where((grad > med) & (cur_h < 1.0 - 1e-12))[0]
        dec_candidates = np.where((grad < med) & (cur_h > 1e-12))[0]

        if inc_candidates.size and dec_candidates.size:

            i_down = int(np.random.choice(dec_candidates))

        delta_max = min(max_delta, 1.0 - cur_h[i_up], cur_h[i_down])

        new_h[i_up] += delta
        new_h[i_down] -= delta
