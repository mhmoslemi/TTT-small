def project_onto_constraints(v, target_sum):
    lo = -np.max(v)  # low enough to push all to 0
    hi = 1 - np.min(v) # high enough to push all to 1
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v + mid, 0, 1)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0, 1)

def project_box_sum(v, target):
    # Binary search for Lagrange multiplier λ.
    low = -np.max(v) - 1 # enough to push all to 0
    high = 1 - np.min(v) + 1 # enough to push all to 1
    for _ in range(50):
        mid = (low + high) / 2
        h = np.clip(v + mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            low = mid
        else:
            high = mid
    h = np.clip(v + (low+high)/2, 0.0, 1.0)

        rng.shuffle(init_h)

    # Ensure constraints: sum = n/2
    # Project just in case
    target_sum = n / 2
    # use projection function
    def project(h):
        # projection to [0,1] and sum target
        low = -np.max(h) - 1.0
        high = 1.0 - np.min(h) + 1.0
        for _ in range(50):

            s = proj.sum()
            if s < target_sum:

            else:

        # due to numeric issues adjust tiny errors
        # Quick fix: if sum not exact, adjust small distribution to meet.
        diff = target_sum - proj.sum()

            if np.any(mask):

                proj = np.clip(proj, 0.0, 1.0)

import time
import numpy as np

def _project_onto_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Projection of v onto the set {0≤x≤1, sum(x)=target_sum}."""
    # binary search for Lagrange multiplier λ so that sum(clip(v+λ,0,1)) = target_sum
    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(40):
        mid = (low + high) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            low = mid
        else:
            high = mid
    w = np.clip(v + (low + high) * 0.5, 0.0, 1.0)
    # tiny correction for possible rounding error
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        mask = (w > 0.0) & (w < 1.0)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


def _shift_with_zeros(arr: np.ndarray, shift: int) -> np.ndarray:
    """Shift arr by `shift` positions, filling the emptied places with zeros."""
    n = arr.size
    if shift == 0:
        return arr.copy()
    if shift > 0:
        out = np.empty_like(arr)
        out[: n - shift] = arr[shift:]
        out[n - shift :] = 0.0
    else:
        s = -shift
        out = np.empty_like(arr)
        out[s:] = arr[: n - s]
        out[:s] = 0.0
    return out


def _compute_c5(h: np.ndarray, n: int):
    """Return (C5 value, max correlation count, full correlation array)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = corr.max()
    c5 = max_corr * (2.0 / n)
    return c5, max_corr, corr


def _random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (values in [0,1] with sum=n/2)."""
    v = rng.random(n)
    return _project_onto_constraints(v, n / 2.0)


def _interpolate(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a periodic step on [0,2] to a new length."""
    old_n = old.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old)
