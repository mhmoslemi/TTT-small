import time
import numpy as np

def _c5_and_shift(h):
    """Return (C5 value, shift) for the given vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    c5 = float(conv[idx] * dx)
    shift = idx - (n - 1)          # corresponds to k = shift*dx
    return c5, shift

def _project_to_sum_box(v, target):
    """
    Euclidean projection of vector v onto the set {x | 0<=x_i<=1, sum x = target}.
    The solution is x_i = clip(v_i + λ, 0, 1) for a suitable λ found by bisection.
    """
    lo = -np.max(v) - 1.0          # makes all v+λ <= 0  → sum = 0
    hi = 1.0 - np.min(v) + 1.0    # makes all v+λ >= 1  → sum = len(v)
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _initial_vector(n, rng):
    """Random feasible start (values in [0,1], sum = n/2)."""
    v = rng.random(n)
    return _project_to_sum_box(v, n / 2.0)

def _improve(h, time_budget):
    """Simple targeted hill‑climbing / simulated‑annealing."""
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5, best_shift = _c5_and_shift(best_h)

    start = time.time()
    rng = np.random.default_rng()
    # parameters
    step_factor = 0.4            # size of each transfer (as fraction of max possible)
    # main loop
    while time.time() - start < time_budget:
        # recompute worst shift each iteration
        c5, shift = _c5_and_shift(best_h)
        if c5 < best_c5 - 1e-10:          # should not happen, safety
            best_c5, best_shift = c5, shift

        # if already good enough, break early
        if best_c5 < 0.3808:
            break

        # ----------- propose a move ---------------------------------
        if shift == 0:
            # improve by making values more extreme (push towards 0 or 1)
            # pick a high (>0.5) and a low (<0.5) entry
            high_idx = np.where(best_h > 0.5)[0]
            low_idx = np.where(best_h < 0.5)[0]
            if high_idx.size == 0 or low_idx.size == 0:
                # fallback to random pair
                i, j = rng.integers(0, n, size=2)
            else:

                j = int(rng.choice(low_idx))
