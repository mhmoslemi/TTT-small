def eval_c5(h):
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * (2.0/len(h))

def hill_climb(N, max_seconds, start_h=None):

def solve_for_N(N, time_limit):
    import cvxpy as cp, numpy as np, time
    # variables
    h = cp.Variable(N, boolean=True)
    t = cp.Variable(integer=True)
    # constraints list
    constraints = [cp.sum(h) == N // 2]
    # Build pair arrays
    pair_i = []
    pair_j = []
    pair_shift = []
    for s in range(-N+1, N):
        if s >= 0:
            i_vals = np.arange(N - s)
            j_vals = i_vals + s
        else:
            i_vals = np.arange(-s, N)
            j_vals = i_vals + s
        pair_i.append(i_vals)
        pair_j.append(j_vals)
        pair_shift.append(np.full_like(i_vals, s))
    pair_i = np.concatenate(pair_i)
    pair_j = np.concatenate(pair_j)
    pair_shift = np.concatenate(pair_shift)
    M = len(pair_i)   # N**2
    y = cp.Variable(M, boolean=True)
    # product constraints
    constraints += [y <= h[pair_i]]
    constraints += [y <= h[pair_j]]
    constraints += [y >= h[pair_i] + h[pair_j] - 1]
    # shift constraints
    for s in range(-N+1, N):
        idx = (pair_shift == s)
        constraints.append(cp.sum(h[pair_i[idx]] - y[idx]) <= t)
    # objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    # try solve with time limit
    try:
        prob.solve(solver=cp.CBC, warm_start=False, max_seconds=time_limit, verbose=False)
    except Exception as e:
        # fallback
        return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None
    h_vals = np.round(h.value).astype(int)
    t_val = int(round(t.value))
    return h_vals, t_val

def hill_climb(N, h_start, time_limit):
    import numpy as np, time, random
    h = h_start.copy()
    # ensure exactly N/2 ones
    # convert to int
    h = h.astype(np.int8)
    cur_score = eval_c5(h)
    start = time.time()
    no_improve = 0
    best_local_h = h.copy()
    best_local_score = cur_score
    max_no_improve = N*10  # e.g., 1000 for N=70
    while time.time() - start < time_limit:
        # propose random indices i, j with different values
        i = random.randrange(N)
        j = random.randrange(N-1)
        if j >= i: j += 1
        if h[i] == h[j]:
            # try again
            continue
        # swap
        h[i] ^= 1
        h[j] ^= 1
        new_score = eval_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            no_improve = 0
            # update best local
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_h = h.copy()
        else:
            # revert
            h[i] ^= 1
            h[j] ^= 1
            no_improve += 1
        if no_improve >= max_no_improve:
            # restart random
            h = random_binary(N)
            cur_score = eval_c5(h)
            no_improve = 0
    return best_local_h, best_local_score

import numpy as np
import random
import time

# ----------------------------------------------------------------------
def eval_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for binary vector h."""
    N = h.shape[0]
    # full cross‑correlation h with (1‑h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / N)


# ----------------------------------------------------------------------
def random_binary(N: int) -> np.ndarray:
    """Binary vector of length N with exactly N/2 ones."""
    ones = np.random.choice(N, N // 2, replace=False)
    h = np.zeros(N, dtype=np.int8)
    h[ones] = 1
    return h


# ----------------------------------------------------------------------
def hill_climb(N: int, h_start: np.ndarray, time_limit: float):
    """Simple hill‑climbing with random swaps."""
    h = h_start.astype(np.int8).copy()
    best_h = h.copy()
    best_score = eval_c5(h)
    cur_score = best_score
    start = time.time()
    no_improve = 0
    max_no_improve = N * 10

    while time.time() - start < time_limit:
        # pick two distinct indices with opposite values
        i = random.randrange(N)
        j = random.randrange(N - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue

        # swap bits (xor works for 0/1)
        h[i] ^= 1
        h[j] ^= 1

        new_score = eval_c5(h)
        if new_score < cur_score:
            cur_score = new_score
            no_improve = 0
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # revert
            h[i] ^= 1
            h[j] ^= 1
            no_improve += 1

        if no_improve >= max_no_improve:
            # restart from a fresh random vector
            h = random_binary(N)
            cur_score = eval_c5(h)
            no_improve = 0

    return best_h, best_score


# ----------------------------------------------------------------------
def solve_milp(N: int, time_limit: float):
    """Solve the exact integer program for a given length N."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    # binary variables for the function
    h = cp.Variable(N, boolean=True)
    t = cp.Variable(integer=True)

    constraints = [cp.sum(h) == N // 2, t >= 0]

    # build all overlapping index pairs (i, j=i+s)
    pair_i = []
    pair_j = []
    pair_shift = []
    for s in range(-N + 1, N):
        if s >= 0:
            i_vals = np.arange(N - s, dtype=int)
            j_vals = i_vals + s
        else:
            i_vals = np.arange(-s, N, dtype=int)
            j_vals = i_vals + s
        pair_i.append(i_vals)
        pair_j.append(j_vals)
        pair_shift.append(np.full(i_vals.shape, s, dtype=int))

    pair_i = np.concatenate(pair_i)
    pair_j = np.concatenate(pair_j)
    pair_shift = np.concatenate(pair_shift)
    M = pair_i.size

    # auxiliary binary variables y_{i,s} = h_i * h_{i+s}
    y = cp.Variable(M, boolean=True)

    # linearisation of the product
    constraints += [y <= h[pair_i]]
    constraints += [y <= h[pair_j]]
    constraints += [y >= h[pair_i] + h[pair_j] - 1]

    # for each shift enforce the overlap bound
    for s in range(-N + 1, N):
        idx = (pair_shift == s)
        # sum_i (h_i - y_i_s)  <= t
        constraints.append(cp.sum(h[pair_i[idx]] - y[idx]) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # solve with a time limit (CBC accepts max_seconds)
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        return None, None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    h_val = np.rint(h.value).astype(np.int8)
    t_val = int(round(t.value))
    # verify the integral constraints (safety)
    if h_val.sum() != N // 2:
        # repair rounding issues
        diff = N // 2 - h_val.sum()
        if diff > 0:
            zeros = np.where(h_val == 0)[0]
            pick = np.random.choice(zeros, diff, replace=False)
            h_val[pick] = 1
        elif diff < 0:
            ones = np.where(h_val == 1)[0]
            pick = np.random.choice(ones, -diff, replace=False)
            h_val[pick] = 0
    return h_val, t_val
