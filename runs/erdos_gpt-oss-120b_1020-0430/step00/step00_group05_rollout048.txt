import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for the discretised step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def solve_binary_mip(n, time_left):
    """Binary (0/1) optimisation via a mixed‑integer programme."""
    import cvxpy as cp

    # Boolean variables for the step function
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()                     # the maximal overlap to minimise
    constraints = [cp.sum(x) == n / 2]    # area constraint (n is even)

    # For every non‑zero shift create auxiliary bilinear variables y = x_i * x_{i+k}
    for k in range(1, n):
        m = n - k
        y = cp.Variable(m, boolean=True)
        constraints += [
            y <= x[:m],
            y <= x[k:],
            y >= x[:m] + x[k:] - 1,
            t >= n / 2 - cp.sum(y)   # H - Σ y >= t  (H = n/2)
        ]

    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # Try CBC first, fall back to GLPK_MI
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=time_left)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.GLPK_MI, verbose=False, max_seconds=time_left)

    if prob.status not in ("optimal", "optimal_inaccurate"):
        raise RuntimeError("MIP failed to solve")

    h = np.array(x.value, dtype=float)
    return h

def run(seed=42, budget_s=1100):
    """Return a step function that improves the C5 bound."""
    start = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------ #
    # 1. Determine problem size and starting point
    # ------------------------------------------------------------------ #
    if "initial_h_values" in globals():
        h_start = np.array(initial_h_values, dtype=float)
        n = len(h_start)
    else:
        n = 70                     # a modest even size
        h_start = np.zeros(n, dtype=float)
        h_start[: n // 2] = 1.0
        np.random.shuffle(h_start)

    # ------------------------------------------------------------------ #
    # 2. Try a binary MILP solution (fast for n ≤ ~80)
    # ------------------------------------------------------------------ #
    best_h = None
    best_val = float("inf")
    try:
        if n % 2 == 0 and n <= 80:
            time_for_mip = max(0.0, budget_s - (time.time() - start) - 1.0)
            binary_h = solve_binary_mip(n, time_for_mip)
            val = compute_c5(binary_h)
            best_h, best_val = binary_h, val
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------ #
    # 3. If MILP not available, fall back to a random binary start
    # ------------------------------------------------------------------ #
    if best_h is None:
        best_h = h_start.copy()
        best_val = compute_c5(best_h)

    # ------------------------------------------------------------------ #
    # 4. Simulated‑annealing style local optimisation (continuous moves)
    # ------------------------------------------------------------------ #
    cur_h = best_h.copy()
    cur_val = best_val

    T = 0.008                # initial temperature
    cool = 0.99993
    min_T = 1e-7
    no_improve = 0
    max_no_improve = 3000

    while time.time() - start < budget_s - 0.3:
        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move mass from i to j (maintains the sum exactly)
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:
            continue
        eps = np.random.rand() * max_transfer
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        new_val = compute_c5(new_h)
        delta = new_val - cur_val

        accept = False
        if delta <= 0:
            accept = True
        elif T > 0 and np.exp(-delta / T) > np.random.rand():
            accept = True

        if accept:

            if new_val < best_val - 1e-12:

                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # temperature schedule
        T = max(T * cool, min_T)

        # occasional small‑perturbation restart when stalled
        if no_improve > max_no_improve:
            cur_h = best_h.copy()

    target_sum = n / 2

        pass
