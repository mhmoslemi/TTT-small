import time
import numpy as np

# ------------------------------------------------------------
# Helper: evaluate the C5 value for a step function h
def _evaluate(h):
    n = h.shape[0]
    if n == 0:
        return 0.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    return np.max(corr) * dx

# ------------------------------------------------------------
# Projection onto intersection of box [0,1] and hyperplane sum = target
def _project_box_sum(y, target, lower=0.0, upper=1.0):
    # solve x = clip(y - λ, lower, upper)  with sum(x)=target
    lam_low = np.min(y - upper) - 1.0
    lam_high = np.max(y - lower) + 1.0
    for _ in range(60):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(y - lam, lower, upper)
        s = x.sum()
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    return x

# ------------------------------------------------------------
# Gradient of the (unscaled) max correlation w.r.t. h
def _max_corr_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    eps = 1e-12
    idxs = np.where(np.abs(corr - max_val) <= eps)[0]
    grad = np.zeros_like(h)
    offset = n - 1
    for idx in idxs:
        L = idx - offset
        if L >= 0:
            i_start, i_end = 0, n - L
            j_start, j_end = L, n
        else:
            i_start, i_end = -L, n
            j_start, j_end = 0, n + L
        i_slice = slice(i_start, i_end)
        j_slice = slice(j_start, j_end)
        grad[i_slice] += (1.0 - h[j_slice])
        grad[j_slice] -= h[i_slice]
    return grad

# Simple gradient descent with projection
def _grad_descent(h0, max_iter, end_time):

def _random_hill_climb(h0, max_iter, end_time):
    h = h0.copy()

    best_h = h.copy()

        if i == j:
            continue
        # move mass from i to j

        if delta <= 0:

        delta *= np.random.rand()
        h[i] -= delta
