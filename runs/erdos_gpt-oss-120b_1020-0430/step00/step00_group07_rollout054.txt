    while abs(diff) > 1e-8:
        if diff > 0:
            # need to add; pick indices where v < 1
            idx = np.where(v < 1-1e-12)[0]

            capacity = (1 - v[idx]).sum()

import time
import numpy as np
from scipy.optimize import minimize, brentq

# ---------- helper utilities ----------
def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the hypercube [0,1]^n while forcing the sum to be `target`.
    This solves min ||x - v|| s.t. 0<=x_i<=1 and sum(x)=target.
    Uses a bisection on the Lagrange multiplier λ where x_i = clip(v_i - λ, 0, 1).
    """
    def f(lam):
        return np.clip(v - lam, 0.0, 1.0).sum() - target

    lo = -np.max(v) - 1.0          # ensures sum >= target
    hi = np.max(v) + 1.0           # ensures sum <= target
    # adjust bounds if the signs are not correct (numerical safety)
    while f(lo) < 0:
        lo -= 1.0
    while f(hi) > 0:
        hi += 1.0
    lam_opt = brentq(f, lo, hi, maxiter=100)
    proj = np.clip(v - lam_opt, 0.0, 1.0)
    # tiny correction for round‑off
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        mask = (proj > 0.0) & (proj < 1.0)
        if mask.any():
            proj[mask] += diff / mask.sum()
        proj = np.clip(proj, 0.0, 1.0)
    return proj


def random_feasible(n: int, target: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n with prescribed sum `target`."""
    v = np.random.rand(n)
    return project_onto_simplex(v, target)


def c5_score(h: np.ndarray, dx: float) -> float:
    """Compute the C5 value for a step function given by array h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def hill_climb(h: np.ndarray, dx: float, target: float,
               steps: int = 2000) -> (np.ndarray, float):
    """Simple stochastic improvement: transfer mass between two coordinates."""
    n = h.shape[0]
    cur = h.copy()
    cur_score = c5_score(cur, dx)
    for _ in range(steps):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta_max = min(cur[i], 1.0 - cur[j])
        if delta_max <= 0.0:
            continue
        delta = delta_max * np.random.rand()
        nxt = cur.copy()
        nxt[i] -= delta
        nxt[j] += delta
        nxt_score = c5_score(nxt, dx)
        if nxt_score < cur_score:
            cur, cur_score = nxt, nxt_score
    # final projection just in case of tiny drift
    cur = project_onto_simplex(cur, target)
    cur_score = c5_score(cur, dx)
    return cur, cur_score


# ---------- main entry point ----------

    """
    Optimize a step function h on [0,2] (discretized) to minimise the
    Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size and initial vector
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n_points = init.shape[0]
    else:
        n_points = 80
        init = None

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∫h = 1 ⇒ sum(h)=1/dx

    # Prepare a feasible starting point
    if init is not None:
        h0 = project_onto_simplex(np.clip(init, 0.0, 1.0), target_sum)
    else:
        h0 = random_feasible(n_points, target_sum)

    best_h = h0.copy()
    best_score = c5_score(best_h, dx)

    # Equality constraint for the optimizer
    eq_cons = {"type": "eq", "fun": lambda h, t=target_sum: np.sum(h) - t}
    bounds = [(0.0, 1.0)] * n_points

    # Main optimisation loop (budget aware)
    iteration = 0
    while time.time() - start_time < budget_s * 0.90:
        iteration += 1

        # Choose a start vector: perturb the current best or random restart
        if np.random.rand() < 0.7:

        else:

        # Ensure feasibility before giving to optimizer
        start = np.clip(start, 0.0, 1.0)
        start = project_onto_simplex(start, target_sum)

        # Run a short SLSQP local optimisation

                lambda x: c5_score(x, dx),
                start,

                bounds=bounds,
                constraints=[eq_cons],
                options={"maxiter": 80, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            continue

        cand = project_onto_simplex(cand, target_sum)
        cand_score = c5_score(cand, dx)

        if best_score < 0.3808:
            break

    # Final polishing phase using a longer hill‑climb
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        best_h, best_score = hill_climb(best_h, dx, target_sum,
                                        steps=8000)

    # Final verification (use the harness evaluator if present)
    if "evaluate_erdos_solution" in globals():
        final_score = evaluate_erdos_solution(best_h)
    else:
        final_score = c5_score(best_h, dx)

    return best_h, float(final_score), int(n_points)
