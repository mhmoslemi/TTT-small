def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random, sys
    np.random.seed(seed)
    random.seed(seed)
    # Determine n:
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    except Exception:
        n = 70
        h0 = np.full(n, 0.5, dtype=float)
    # Ensure sum constraint:
    # Scale to sum exactly n/2 while respecting bounds.
    # For initial vector may have slight sum error; we can adjust by adding to a random element.
    target_sum = n/2.0
    cur_sum = h0.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-9:
        # adjust first element within [0,1] if possible
        # compute allowed adjustment
        adjust = np.clip(diff, -h0[0], 1 - h0[0])
        h0[0] += adjust
        diff -= adjust
        # if remaining diff, adjust second element etc.
        i = 1
        while abs(diff) > 1e-12 and i < n:
            adjust = np.clip(diff, -h0[i], 1 - h0[i])
            h0[i] += adjust
            diff -= adjust
            i += 1
        # if still diff due to rounding, scale entire vector (might break bounds)
        if abs(diff) > 1e-8:
            # fallback: renormalize h0 after clipping
            h0 = np.clip(h0, 0, 1)
            # adjust proportionally
            sum_h = h0.sum()
            if sum_h > 0:
                h0 *= target_sum / sum_h
            else:
                h0[:n//2] = 1.0
                h0[n//2:] = 0.0
    # Compute initial score
    dx = 2.0 / n
    def compute_c5(vec):
        # use convolution
        corr = np.convolve(vec, (1 - vec)[::-1], mode='full')
        # maximum cross term
        max_val = corr.max()
        return max_val * dx
    current_h = h0.copy()
    current_score = compute_c5(current_h)
    best_h = current_h.copy()
    best_score = current_score

    start = time.time()
    time_limit = budget_s - 0.5  # leave some time

    # Simulated annealing parameters
    T_initial = 0.05
    T_final = 0.001
    # We'll adapt temperature based on elapsed fraction.
    # We'll also track iteration count
    it = 0
    while True:
        if time.time() - start > time_limit:
            break
        # Temperature schedule
        elapsed = time.time() - start
        frac = elapsed / time_limit
        # e.g. exponential cooling
        T = T_initial * ( (T_final / T_initial) ** frac )
        # propose move
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)
        hi = current_h[i]
        hj = current_h[j]
        lb = max(-hi, hj - 1.0)
        ub = min(1.0 - hi, hj)
        if ub <= lb:
            continue
        delta = np.random.uniform(lb, ub)
        new_h = current_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        # compute new score
        new_score = compute_c5(new_h)
        # acceptance
        if new_score < current_score or np.random.rand() < math.exp((current_score - new_score) / T):
            current_h = new_h
            current_score = new_score
            # update best if improved
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
        it += 1
    # final answer
    return (best_h.tolist(), best_score, n)

    # get initial h
    try:

        n = 70
        h0 = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    # project onto constraints
    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple rescale while keeping bounds
        # First try to offset smallest adjustments
        diff = target_sum - cur_sum
        # distribute proportionally among free entries
        free = (h0 > 0) & (h0 < 1)
        if free.any():
            # adjust proportionally
            adjust = diff / free.sum()
            h0[free] += adjust
            # clip again
            h0 = np.clip(h0, 0.0, 1.0)
        # final exact fix
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-12:
            # push into first element if possible
            delta = np.clip(diff, -h0[0], 1.0 - h0[0])
            h0[0] += delta
    # now h0 sum should be close to target_sum

    p_target = 0.85  # probability to do a targeted move

        if time.time() - start > time_limit:
