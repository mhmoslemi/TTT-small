    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx, corr

def project_feasible(h, target_sum):
    # Clip, then adjust sum to target as described
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    max_iter = 10
    for _ in range(max_iter):
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        allowed = np.where(mask)[0]
        add = diff / allowed.size
        # cap add to avoid crossing bounds
        if diff > 0:
            max_add = np.min(1.0 - h[allowed])
            if add > max_add:
                add = max_add
        else:
            max_sub = np.min(h[allowed])
            if -add > max_sub:
                add = -max_sub
        h[allowed] += add
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
    # final minor correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # distribute uniformly and clip
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # final scale (small adjustment) to match sum exactly
        s = h.sum()
        if s > 0:
            h *= target_sum / s
    h = np.clip(h, 0.0, 1.0)
    return h

def initial_solution(rng, n_target):

import time, math
import numpy as np

def _compute_c5(h: np.ndarray):
    """Return (C5, full correlation array) for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx, corr


def _project_feasible(h: np.ndarray, target_sum: float):
    """Clip to [0,1] and enforce sum(h)=target_sum (iterative water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(6):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        factor = target_sum / s
        h = np.clip(h * factor, 0.0, 1.0)
    # tiny residual correction (uniform redistribution)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s > 0:
            h *= target_sum / s
    return h


def _initial_h(rng: np.random.Generator, n: int):
    """Create a feasible starting vector, using the supplied parent if present."""
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        m = parent.size
        if m != n:
            # linear interpolation onto the new grid
            x_old = np.linspace(0.0, 2.0, m, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, parent)
        else:
            h = parent.copy()
    else:
        h = rng.random(n)               # uniform start
    h = _project_feasible(h, n / 2.0)
    return h

        temp = init_temp * (1.0 - frac) + final_temp * frac
