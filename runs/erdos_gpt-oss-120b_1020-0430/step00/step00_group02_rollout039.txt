def propose_move(h):
    i, j = random.sample(range(n), 2)
    # choose delta in [-dmax, dmax], where we can move some mass from j to i or vice versa
    max_inc_i = 1 - h[i]
    max_dec_i = h[i]
    max_inc_j = 1 - h[j]
    max_dec_j = h[j]
    # We allow delta positive meaning add to i and subtract from j
    delta_max = min(max_inc_i, max_dec_j)
    delta_min = -min(max_inc_j, max_dec_i)  # negative range
    if delta_max <= 0 and delta_min >= 0:
        return None
    delta = random.uniform(delta_min, delta_max)  # uniform
    # new values
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def propose_move_pair(h):

        j += 1

def compute_c5(h: np.ndarray) -> float:
    """C5 value for a step function given by samples h."""
    n = h.shape[0]
    # full linear correlation of h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * (2.0 / n)


def compute_c5_and_shift(h: np.ndarray):
    """Return (C5, shift) where shift is the integer lag attaining the maximum."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    # lag 0 corresponds to index n-1
    shift = idx - (n - 1)
    return conv[idx] * (2.0 / n), shift


def random_binary_vector(length: int) -> np.ndarray:
    """Binary vector with exactly length/2 ones (length must be even)."""
    half = length // 2
    v = np.zeros(length, dtype=np.float64)
    ones = np.random.choice(length, half, replace=False)
    v[ones] = 1.0
    return v


def mismatch_pair(h: np.ndarray, shift: int):
    """
    Find a pair (i,j) such that h[i]==1, h[i+shift]==0 and
    h[j]==0, h[j+shift]==1. Swapping i and j reduces the
    mismatch count for the given shift.
    Returns (i,j) or (None,None) if no such pair exists.
    """
    n = h.shape[0]
    idx = np.arange(n)
    ok = (idx + shift >= 0) & (idx + shift < n)
    if not np.any(ok):
        return None, None
    hi = h[ok]
    hj = h[ok + shift]
    mask_a = (hi == 1) & (hj == 0)  # 1 -> 0 mismatches
    mask_b = (hi == 0) & (hj == 1)  # 0 -> 1 mismatches
    if not mask_a.any() or not mask_b.any():
        return None, None
    pos_a = idx[ok][mask_a]
    pos_b = idx[ok][mask_b]
    i = np.random.choice(pos_a)
    j = np.random.choice(pos_b)
    return i, j


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (represented by N samples) that
    minimizes the Erdős overlap constant C5.
    Returns (list_of_h, best_C5, N).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # target number of samples – a power of two works well for speed
    N = 128
    half_N = N // 2

    # initialise – repeat a random half‑pattern twice (helps the shift N/2)
    base = random_binary_vector(half_N)
    h = np.concatenate([base, base])

    cur_c5, cur_shift = compute_c5_and_shift(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # annealing / restart control
    steps_without_improve = 0
    RESTART_AFTER = 50000          # steps with no improvement before random restart
    CHECK_EVERY = 2000             # time‑check interval
    total_budget = budget_s * 0.95  # keep a safety margin

    it = 0
    while True:
        # time‑budget check
        if it % CHECK_EVERY == 0:
            if time.time() - start_time > total_budget:
                break

        # try a targeted swap that directly improves the worst shift
        i, j = mismatch_pair(h, cur_shift)
        if i is None:                     # fall back to a random admissible swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))

        # perform the swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0

        new_c5 = compute_c5(h_new)

        if new_c5 < cur_c5:
            # accept improvement
            h = h_new
            cur_c5 = new_c5
            steps_without_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                # stop early if we have already beaten the target record
                if best_c5 <= 0.3808:
                    break
        else:
            steps_without_improve += 1

        # occasionally recompute the worst shift (cheap compared to correlation)
        if steps_without_improve % 1000 == 0 or new_c5 < cur_c5:
            cur_c5, cur_shift = compute_c5_and_shift(h)

        # restart if stuck for a long time
        if steps_without_improve > RESTART_AFTER:
            base = random_binary_vector(half_N)
            h = np.concatenate([base, base])
            cur_c5, cur_shift = compute_c5_and_shift(h)
            steps_without_improve = 0

        it += 1

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), float(final_c5), N)
