def compute_c5_full(h, dx):

def compute_gradient(h, lag):
    n = len(h)
    g = np.ones_like(h)  # derivative from sum part
    # subtract h[i+lag] and h[i-lag] as appropriate
    if lag >= 0:
        # for each i, we need h[i+lag] if i+lag < n
        # and h[i-lag] if i-lag >=0
        # So:
        # subtract shifted versions
        g[:n-lag] -= h[lag:]  # i from 0 to n-lag-1, i+lag = lag..n-1
        g[lag:] -= h[:n-lag]  # i from lag to n-1, i-lag = 0..n-lag-1
    else:
        # lag negative, treat using symmetry
        # we can flip sign: lag = -L
        L = -lag
        # subtract h[i+lag] where i+lag = i - L (valid if i >= L)
        g[L:] -= h[:n-L]   # i>=L -> i-lag = i+L? Wait careful
        # Actually for negative lag, original formula: g[i] = 1 - h[i+lag] - h[i-lag]
        # Where i+lag means i - L; i-lag means i + L.
        # So subtract h[i - L] if i>=L and subtract h[i+L] if i+L<n.
        g[L:] -= h[:n-L]   # subtract h[i-L]
        g[:n-L] -= h[L:]   # subtract h[i+L]
    # Ensure gradient shape correct
    return g

import time
import numpy as np

# ----- helper functions -----------------------------------------------------

def resample(arr, new_n):
    """Linear interpolation of a 1‑D array onto a new grid of length new_n."""
    old_n = len(arr)
    old_pos = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_pos = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(new_pos, old_pos, arr, left=arr[0], right=arr[-1])


def project_to_sum_box(v, target_sum, max_iter=50):
    """
    Project vector v onto the feasible set:
        0 <= w_i <= 1,  sum w_i = target_sum.
    Uses a simple bisection on a common shift lambda.
    """
    low = -np.max(v) - 1.0            # makes all entries <=0 after shift
    high = 1.0 - np.min(v) + 1.0     # makes all entries >=1 after shift
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            low = mid
        else:
            high = mid
    w = np.clip(v + (low + high) * 0.5, 0.0, 1.0)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-10:
        interior = (w > 0.0) & (w < 1.0)
        if interior.any():
            w[interior] += diff / interior.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback: adjust a single component with maximal slack
            slack = np.minimum(w, 1.0 - w)
            i = np.argmax(slack)
            w[i] = np.clip(w[i] + diff, 0.0, 1.0)
    return w


def compute_corr_and_active(h, dx):
    """Return full correlation, its maximum, and the lags attaining the maximum."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = corr.max()
    # indices of lags where the maximum is attained (tolerance for floating error)
    idx = np.where(np.isclose(corr, max_val, atol=1e-12))[0]
    active_lags = idx - (len(h) - 1)   # conversion from index to lag
    return corr, max_val, active_lags


def compute_subgradient(h, active_lags):
    """Subgradient of the max‑overlap objective (averaged over active lags)."""
    n = len(h)
    grad = np.zeros_like(h)
    for lag in active_lags:
        g = np.ones_like(h)
        if lag >= 0:
            L = lag
            if L < n:
                g[: n - L] -= h[L:]
                g[L:] -= h[: n - L]
        else:
            L = -lag
            if L < n:
                g[L:] -= h[: n - L]
                g[: n - L] -= h[L:]
        grad += g
    grad /= len(active_lags)
    return grad


def perturb(h, prob=0.02):
    """Randomly perturb a fraction of entries and re‑project."""
    n = len(h)
    mask = np.random.rand(n) < prob
    if not mask.any():
        return h.copy()
    delta = np.random.uniform(-0.2, 0.2, size=mask.sum())
    h2 = h.copy()
    h2[mask] += delta
    h2 = np.clip(h2, 0.0, 1.0)
    target = n / 2.0
    return project_to_sum_box(h2, target)
