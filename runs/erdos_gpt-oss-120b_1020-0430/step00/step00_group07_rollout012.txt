import time
import math
import numpy as np

def _max_corr(vec):
    """Return the maximum of the full correlation of vec with (1-vec)."""
    return np.max(np.correlate(vec, 1.0 - vec, mode="full"))


def _project_to_feasible(x, target):
    """Clip to [0,1] and adjust the sum to the target by a simple water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    # iterative redistribution, very fast for the tiny dimension we use
    for _ in range(50):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(x < 1 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            inc = np.minimum(inc, 1 - x[free])
            x[free] += inc
        else:
            free = np.where(x > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            dec = np.maximum(dec, -x[free])
            x[free] += dec
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)


def _random_feasible(n, target, rng):
    """Binary feasible starter (exactly target ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=int(target), replace=False)
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise
    C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size and basic constants
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    except NameError:
        n = 80
        h0 = None

    dx = 2.0 / n
    target_sum = n / 2.0  # because integral over [0,2] must be 1

    # ------------------------------------------------------------------
    # initialise a feasible vector
    # ------------------------------------------------------------------
    if h0 is None:
        h = _random_feasible(n, target_sum, rng)
    else:
        h = _project_to_feasible(h0, target_sum)

    # ------------------------------------------------------------------
    # bookkeeping of the best solution found
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_corr = _max_corr(best_h)
    best_c5 = best_corr * dx

    # ------------------------------------------------------------------
    # annealing / hill‑climbing parameters
    # ------------------------------------------------------------------
    T0 = 0.02                     # initial temperature (in correlation units)
    iter_no_improve = 0
    MAX_NO_IMPROVE = 20000        # after this many steps we restart from a fresh random start

    while True:
        # time check
        if time.time() - start_time > budget_s - 0.1:
            break

        # correlation of current state
        full_corr = np.correlate(h, 1.0 - h, mode="full")
        cur_max = full_corr.max()
        shift_idx = full_corr.argmax()
        s = shift_idx - (n - 1)               # shift causing the worst overlap

        # ------------------------------------------------------------------
        # propose a move that directly attacks the worst shift
        # ------------------------------------------------------------------
        if s >= 0:
            i_low, i_high = 0, n - s - 1
        else:
            t = -s
            i_low, i_high = t, n - 1

        if i_high >= i_low:
            i = rng.integers(i_low, i_high + 1)
            j = i + s
            # move mass from i to j (decrease i, increase j)
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max > 1e-12:
                delta = delta_max * rng.random() * 0.5
                h_candidate = h.copy()
                h_candidate[i] -= delta
                h_candidate[j] += delta
            else:
                # try opposite direction
                delta_max = min(h[j], 1.0 - h[i])
                if delta_max > 1e-12:
                    delta = delta_max * rng.random() * 0.5
                    h_candidate = h.copy()
                    h_candidate[j] -= delta
                    h_candidate[i] += delta
                else:

                    a, b = rng.choice(n, size=2, replace=False)
                    if rng.random() < 0.5:
                        delta_max = min(h[a], 1.0 - h[b])
                        if delta_max <= 1e-12:
                            continue
                        delta = delta_max * rng.random() * 0.5
                        h_candidate = h.copy()
                        h_candidate[a] -= delta
                        h_candidate[b] += delta
                    else:
                        delta_max = min(h[b], 1.0 - h[a])
                        if delta_max <= 1e-12:
                            continue
                        delta = delta_max * rng.random() * 0.5
                        h_candidate = h.copy()
                        h_candidate[b] -= delta
                        h_candidate[a] += delta
        else:

        # accept / reject using a simple simulated‑annealing rule
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        temp = T0 * max(0.0, 1.0 - elapsed / budget_s)   # linear cooling

        if improve:

        else:

                if delta_max > 1e-12:

                    h[a] -= delta
                    h[b] += delta
            iter_no_improve = 0

    # ------------------------------------------------------------------
    # final tiny polishing: a few deterministic sweeps that move mass from
    # the worst shift toward the partner position.
    # ------------------------------------------------------------------
    for _ in range(200):
        full_corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        s = full_corr.argmax() - (n - 1)
        if s == 0:
            break
        if s >= 0:
            i_low, i_high = 0, n - s - 1
        else:
            t = -s
            i_low, i_high = t, n - 1
        if i_high < i_low:
            break
        i = rng.integers(i_low, i_high + 1)
        j = i + s
        delta_max = min(best_h[i], 1.0 - best_h[j])
        if delta_max < 1e-8:
            continue
        delta = delta_max * 0.3
        best_h[i] -= delta
        best_h[j] += delta
        # keep feasibility (sum never changes)
    # final projection to guarantee constraints (tiny numerical drift)
    best_h = _project_to_feasible(best_h, target_sum)

    final_c5 = _max_corr(best_h) * dx
    return best_h, float(final_c5), n
