import time
import random
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_onto_constraints(h: np.ndarray) -> np.ndarray:
    """
    Project vector h onto the feasible set:
        * 0 <= h_i <= 1
        * sum(h) = n/2   (integral = 1)
    The algorithm repeatedly clips to [0,1] and distributes the remaining
    deficit/excess uniformly among the non‑boundary entries.
    """
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly
        delta = diff / n
        h += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction (if needed) by adjusting interior entries
    s = h.sum()
    if abs(s - target) > 1e-9:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            cnt = free.sum()
            h[free] += (target - s) / cnt
            h = np.clip(h, 0.0, 1.0)
    # enforce exact sum by a last tiny redistribution
    s = h.sum()
    if abs(s - target) > 1e-12:
        # adjust the first element within bounds to fix any remaining error
        err = target - s
        for i in range(n):
            if 0.0 <= h[i] + err <= 1.0:
                h[i] += err
                break
    return h


def _upscale_h(h_small: np.ndarray, n_new: int) -> np.ndarray:
    """Linearly interpolate a coarse vector to a finer resolution."""
    n_small = h_small.shape[0]
    xs = np.linspace(0.0, 2.0, n_small, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xs, h_small)


def _optimize_h(initial_h: np.ndarray, max_seconds: float, start_time: float):
    """Stochastic hill‑climb / simulated‑annealing on the full vector."""
    h = initial_h.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_score = _compute_c5(best_h)

    # algorithmic parameters
    T_initial = 2e-2          # temperature for Metropolis acceptance
    step_factor = 0.5         # maximal fraction of the feasible transfer
    last_improve = 0
    iter_counter = 0

    while True:
        now = time.time()
        if now - start_time > max_seconds:
            break
        iter_counter += 1

        # linear cooling
        frac = (now - start_time) / max_seconds
        T = T_initial * (1.0 - frac)

        # ----- random pairwise transfer -----
        i, j = np.random.choice(n, 2, replace=False)
        # random direction
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer > 1e-12:
            eps = max_transfer * step_factor * np.random.rand()
            new_h = h.copy()
            new_h[src] -= eps
            new_h[dst] += eps
            new_score = _compute_c5(new_h)
            delta = new_score - best_score
            if delta < -1e-12 or np.exp(-delta / (T + 1e-12)) > np.random.rand():
                h = new_h
                if new_score < best_score:
                    best_score = new_score
                    best_h = new_h.copy()
                    last_improve = iter_counter

        # ----- directed move based on current worst shift -----
        if np.random.rand() < 0.07:
            corr = np.correlate(h, 1.0 - h, mode="full")
            max_idx = int(np.argmax(corr))
            shift = max_idx - (n - 1)        # signed integer shift

            # gradient of the worst term (sub‑gradient of max)
            grad = np.ones(n)
            if shift >= 0:
                grad[: n - shift] -= h[shift:]
                grad[shift:] -= h[: n - shift]
            else:
                s = -shift
                grad[s:] -= h[: n - s]
                grad[: n - s] -= h[s:]

            pos = np.where(grad > 1e-8)[0]
            neg = np.where(grad < -1e-8)[0]
            if pos.size and neg.size:
                src = pos[np.argmax(grad[pos])]
                dst = neg[np.argmin(grad[neg])]
                max_transfer = min(h[src], 1.0 - h[dst])
                if max_transfer > 1e-12:
                    eps = max_transfer * step_factor * np.random.rand()
                    new_h = h.copy()
                    new_h[src] -= eps
                    new_h[dst] += eps
                    new_score = _compute_c5(new_h)
                    delta = new_score - best_score
                    if delta < -1e-12 or np.exp(-delta / (T + 1e-12)) > np.random.rand():
                        h = new_h
                        if new_score < best_score:
                            best_score = new_score
                            best_h = new_h.copy()
                            last_improve = iter_counter

        # ----- occasional large perturbation if stuck -----
        if iter_counter - last_improve > 15000:
            # replace 10% of entries by fresh random values then re‑project
            mask = np.random.rand(n) < 0.10
            if mask.any():
                new_vals = np.random.rand(mask.sum())
                h[mask] = new_vals
                h = _project_onto_constraints(h)
                last_improve = iter_counter

    return best_h, best_score


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function for the Erdos minimum overlap problem.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,) with values in [0,1]
        c5_bound : float, the obtained upper bound on C5
        n_points : int, the number of discretisation points used
    """
    np.random.seed(seed)
    random.seed(seed)

    # Target resolution – small enough to stay fast yet expressive
    n_points = 180

    # Obtain an initial vector – use the provided parent if it matches,
    # otherwise upscale it, otherwise fall back to the trivial 0.5 constant.
    try:
        parent = initial_h_values          # pre‑imported by the harness
        if len(parent) == n_points:
            h0 = np.array(parent, dtype=float)
        else:
            h0 = _upscale_h(np.array(parent, dtype=float), n_points)
    except Exception:
        # no parent supplied
        h0 = np.full(n_points, 0.5, dtype=float)

    h0 = _project_onto_constraints(h0)

    start = time.time()
    max_time = budget_s * 0.94               # preserve a small safety margin
    best_h, best_score = _optimize_h(h0, max_time, start)

    # final safety projection (should already be feasible)
    best_h = _project_onto_constraints(best_h)
    best_score = _compute_c5(best_h)

    return best_h, best_score, n_points
