def hill_climb(h, max_iter, start_T=0.01, end_T=1e-5):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5

    for it in range(max_iter):
        T = start_T * ((end_T/start_T) ** (it / max_iter)) # geometric cooling
        # choose random i from ones, j from zeros
        ones = np.where(cur_h > 0.5)[0] # since binary
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0: break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_c5
```

def construct_set(n, r):
    dx = 2.0 / n
    h = np.ones(n, dtype=np.float64)  # initially all ones (i.e., A contains whole interval)
    intervals = [(0, n-1)]
    k = 0
    while True:
        new_intervals = []
        # compute gap length in points:
        gap_len = max(1, int(round(((1 - r) * (r/2)**k) / dx)))
        # For each interval
        any_removal = False
        for (a,b) in intervals:
            length = b - a + 1
            if gap_len >= length:
                # cannot remove gap, skip (maybe at very small intervals)
                new_intervals.append((a,b))
                continue
            any_removal = True
            # compute middle index
            mid = (a + b) // 2
            # compute gap start and end indices
            half_gap = gap_len // 2
            # For even gap_len, we remove half_gap left and half_gap right, plus maybe one extra.
            start = mid - half_gap
            end = start + gap_len - 1
            # Set h[start:end+1] = 0
            h[start:end+1] = 0.0
            # new intervals: left part [a, start-1] if size>0, right part [end+1,b] if size>0
            if start > a:
                new_intervals.append((a, start-1))
            if end < b:
                new_intervals.append((end+1, b))
        intervals = new_intervals
        k += 1
        # stop when gap_len < 1 (i.e., 0) or intervals become empty or k > some max
        if gap_len <= 1 or not any_removal or k > 50:
            # we can continue until gap_len becomes zero or smaller than 1. At that point, no further removal.
            break
    # After loop, hi contains ones; some zeros may be fractional leftover due to rounding.
    # However sum of h may not be exactly n/2 due to rounding etc. We'll adjust using scaling:
    current_sum = h.sum()
    target = n/2
    # We'll correct by flipping some bits (set some additional ones to zero or zero to one).
    # Since we need exactly half ones, we can adjust by adding zeros or ones randomly.
    difference = int(round(target - current_sum))
    if difference > 0:
        # need to turn zeros to ones
        zero_idx = np.where(h == 0)[0]
        chosen = np.random.choice(zero_idx, size=difference, replace=False)
        h[chosen] = 1
    elif difference < 0:
        # need to turn ones to zeros
        one_idx = np.where(h == 1)[0]
        chosen = np.random.choice(one_idx, size=-difference, replace=False)
        h[chosen] = 0
    return h

def construct_set(n, r):
    # n must be even
    h = np.ones(n, dtype=np.float64)
    intervals = [(0, n-1)]
    points_per_unit = n / 2.0
    k = 0
    max_stages = 30  # avoid infinite loops
    while k < max_stages:
        L_gap = (1 - r) * (r / 2.0) ** k
        gap_points = int(round(L_gap * points_per_unit))
        if gap_points < 1:
            break
        new_intervals = []
        any_removed = False
        for a, b in intervals:
            length = b - a + 1
            if gap_points >= length:
                new_intervals.append((a, b))
                continue
            any_removed = True
            mid = (a + b) // 2
            half = gap_points // 2
            start = mid - half
            # adjust start to ensure gap fits within interval
            if start < a:
                start = a
            end = start + gap_points - 1
            if end > b:
                end = b
                start = end - gap_points + 1
            # set zeros
            h[start:end+1] = 0.0
            # add left interval
            if start > a:
                new_intervals.append((a, start-1))
            # add right interval
            if end < b:
                new_intervals.append((end+1, b))
        intervals = new_intervals
        if not any_removed:
            break
        k += 1
    # Now adjust sum to target
    target = n // 2
    current = int(round(h.sum()))
    diff = target - current
    if diff > 0:
        zeros = np.where(h == 0)[0]
        if len(zeros) >= diff:
            choose = np.random.choice(zeros, size=diff, replace=False)
            h[choose] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        if len(ones) >= -diff:
            choose = np.random.choice(ones, size=-diff, replace=False)
            h[choose] = 0.0
    return h

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # find ones and zeros
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:

        h[i] = 0.0
        h[j] = 1.0

import time, random, numpy as np

def compute_c5(h: np.ndarray) -> float:
    """max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def construct_set(n: int, r: float) -> np.ndarray:
    """Fractal construction: start with whole interval, repeatedly cut central gaps.
    r∈(0,1) controls the distribution of gap sizes.
    Returns a binary vector with exactly n//2 ones."""
    h = np.ones(n, dtype=np.float64)          # everything starts as 1 (belongs to A)
    intervals = [(0, n - 1)]                  # list of alive intervals (inclusive)
    pts_per_unit = n / 2.0                    # points per unit length (dx = 2/n)

    k = 0
    max_stages = 30
    while k < max_stages:
        # length of the gap to be removed at this stage (continuous length)
        gap_len_cont = (1.0 - r) * (r / 2.0) ** k
        # convert to number of discrete points; at least one point
        gap_pts = int(round(gap_len_cont * pts_per_unit))
        if gap_pts < 1:
            break
        new_intervals = []
        any_removed = False
        for a, b in intervals:
            length = b - a + 1
            if gap_pts >= length:                     # cannot cut a gap larger than the interval
                new_intervals.append((a, b))
                continue
            any_removed = True
            mid = (a + b) // 2
            half = gap_pts // 2
            start = mid - half
            # keep gap inside the interval
            if start < a:
                start = a
            end = start + gap_pts - 1
            if end > b:
                end = b
                start = end - gap_pts + 1
            h[start:end + 1] = 0.0
            # left part
            if start > a:
                new_intervals.append((a, start - 1))
            # right part
            if end < b:
                new_intervals.append((end + 1, b))
        intervals = new_intervals
        if not any_removed:
            break
        k += 1

    # enforce exact half measure (n/2 ones)
    target = n // 2
    current = int(round(h.sum()))
    diff = target - current
    if diff > 0:
        zeros = np.where(h == 0)[0]
        if len(zeros) >= diff:
            choose = np.random.choice(zeros, size=diff, replace=False)
            h[choose] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        if len(ones) >= -diff:
            choose = np.random.choice(ones, size=-diff, replace=False)
            h[choose] = 0.0
    return h


def hill_climb_binary(start_h: np.ndarray, max_iter: int, time_limit: float):
    """Greedy hill climbing on a binary vector, preserving the exact half‑sum."""
    n = start_h.shape[0]
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    for _ in range(max_iter):
        if time.time() - start > time_limit:
            break
        # pick two positions with opposite values
        i = random.randrange(n)
        j = random.randrange(n)
        if h[i] == h[j]:
            continue
        # perform swap
        hi, hj = h[i], h[j]
        h[i], h[j] = hj, hi
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = hi, hj
    return best_h, best_c5


def search_best(n: int, budget: float):
    """Search over the fractal parameter r and polish the best candidate."""
    start = time.time()
    best_h = None
    best_c5 = float("inf")
    best_r = None

    # ----- coarse sweep over r -------------------------------------------------
    coarse = np.linspace(0.02, 0.98, 49)
    for r in coarse:
        if time.time() - start > 0.4 * budget:
            break
        h = construct_set(n, r)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h, best_r = c5, h.copy(), r

    # ----- fine sweep around the best r ----------------------------------------
    if best_r is not None:
        lo, hi = max(0.0, best_r - 0.02), min(0.999, best_r + 0.02)
        fine = np.linspace(lo, hi, 41)
        for r in fine:
            if time.time() - start > 0.6 * budget:
                break
            h = construct_set(n, r)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h, best_r = c5, h.copy(), r

    # ----- local improvement by hill climbing ------------------------------------
    remaining = budget - (time.time() - start)
    if best_h is not None and remaining > 5.0:
        # allocate about half of the remaining time to hill climbing
        climb_time = min(remaining * 0.5, 300.0)
        # rough estimate: 1000 iterations per second
        max_iter = int(climb_time * 1000)
        max_iter = max(2000, max_iter)          # guarantee a few thousand moves
        polished_h, polished_c5 = hill_climb_binary(best_h.copy(), max_iter, climb_time)
        if polished_c5 < best_c5:
            best_c5, best_h = polished_c5, polished_h

    # fallback – use initial_h_values if it is better and satisfies the constraints
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=np.float64)
        if abs(init_h.sum() * (2.0 / init_h.shape[0]) - 1.0) < 1e-9:
            init_c5 = compute_c5(init_h)
            if init_c5 < best_c5:
                best_c5, best_h = init_c5, init_h.copy()

    # final sanity check: enforce the exact half‑sum one more time
    target = n // 2
    cur = int(round(best_h.sum()))
    diff = target - cur
    if diff != 0:
        if diff > 0:
            zeros = np.where(best_h == 0)[0]
            choose = np.random.choice(zeros, size=diff, replace=False)
            best_h[choose] = 1.0
        else:
            ones = np.where(best_h == 1)[0]
            choose = np.random.choice(ones, size=-diff, replace=False)
            best_h[choose] = 0.0
        best_c5 = compute_c5(best_h)

    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Entry point required by the evaluator."""
    random.seed(seed)
    np.random.seed(seed)

    # choose a reasonable discretisation size (power of two gives exact half‑measure)
    n_points = 1024

    h_best, c5_bound = search_best(n_points, budget_s)

    # return Python objects as required
    return h_best.tolist(), float(c5_bound), int(n_points)
