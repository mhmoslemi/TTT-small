import cvxpy as cp
import numpy as np

def solve_lp(n):
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]

    # Build p variables for each shift s
    p_vars = []
    for s in range(1, n):
        m = n - s
        # p_{i,s} shape m
        p = cp.Variable(m)
        p_vars.append(p)
        # indices for h_i and h_{i+s}
        h_i = h[:m]
        h_ip = h[s:]
        constraints += [p <= h_i, p <= h_ip, p >= h_i + h_ip - 1]
        constraints.append(cp.sum(p) >= t)
    # objective
    obj = cp.Maximize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, max_iters=5000)
    # after solving, get h_val = h.value
    return h.value, t.value, prob.status
```

def project(h):
    # clip to 0-1
    h = np.clip(h, 0.0, 1.0)
    # adjust to satisfy sum = target by redistributing excess
    target = h.size / 2
    # iterative water-filling
    for _ in range(10):
        cur_sum = h.sum()
        diff = target - cur_sum
        if abs(diff) < 1e-9:
            break
        # indices not at bounds
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:
            # can't adjust further
            break
        delta = diff / n_free
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h
```

def project_to_constraints(x, target_sum):
    # clip to [0, 1]
    x = np.clip(x, 0.0, 1.0)
    # adjust sum via water-filling
    # compute current sum
    cur = x.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-9:
        return x
    # use while loop
    max_iter = 10
    for _ in range(max_iter):
        free = (x > 0) & (x < 1)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        # clip again and recompute diff
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-9:
            break
    # if still not satisfied due to numeric, adjust by adding small to first few free entries
    if abs(diff) > 1e-8:
        # distribute remaining diff to entries not at bounds in a loop
        for i in range(len(x)):
            if diff == 0:
                break
            if diff > 0 and x[i] < 1:
                add = min(diff, 1 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def solve_binary_milp(n, time_limit=None):
    import cvxpy as cp
    target_sum = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == target_sum]
    # t within feasible range:
    constraints += [t >= 0, t <= target_sum]
    for s in range(1, n):
        m = n - s
        if m <= 0:
            continue
        w = cp.Variable(m, boolean=True)
        h_i = h[:m]
        h_j = h[s:]
        constraints += [w <= h_i, w <= h_j, w >= h_i + h_j - 1]
        constraints.append(cp.sum(w) >= t)
    obj = cp.Maximize(t)
    prob = cp.Problem(obj, constraints)
    # solve
    try:
        prob.solve(solver=cp.CBC, eps=1e-4, verbose=False, max_iters=10000000)
    except Exception as e:
        # fallback
        return None, None, None
    if h.value is None:
        return None, None, None
    h_val = np.round(h.value).astype(float)  # ensure 0/1
    t_val = t.value
    return h_val, t_val, prob.status

def project_to_constraints(x, target):
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    # iterative water-filling.
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        free = (x > 0) & (x < 1)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        # clip again
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
    # final adjust if small diff left: distribute to any entries not at bound
    if abs(diff) > 1e-8:
        # add diff to first free entries
        for i in range(len(x)):
            if diff == 0:
                break
            if diff > 0 and x[i] < 1:
                add = min(diff, 1 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h):
    """Return the C5 value for a step function defined by vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def project_to_constraints(x, target_sum):
    """
    Project a vector onto the feasible set:
      0 <= x_i <= 1   and   sum(x) == target_sum
    Uses a simple water‑filling scheme.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    for _ in range(25):
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - x.sum()
    # tiny leftover – distribute greedily
    if abs(diff) > 1e-9:
        for i in range(len(x)):
            if diff == 0.0:
                break
            if diff > 0.0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0.0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x


def gradient_at_worst_shift(h):
    """
    Compute a sub‑gradient of the objective at the shift giving the
    current maximal overlap.
    Returns (grad, shift, max_corr).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)        # signed shift producing the max
    grad = np.zeros_like(h)
    if k >= 0:
        i = np.arange(n - k)
        grad[i] += 1.0 - h[i + k]
        grad[i + k] -= h[i]
    else:
        kk = -k
        i = np.arange(n - kk)
        grad[i + kk] += 1.0 - h[i]
        grad[i] -= h[i + kk]
    return grad, k, corr[idx]

# Optional MILP binary optimisation (small n, fast)
# ------------------------------------------------------------
def solve_binary_milp(n):
    """Solve the exact binary max‑min problem for a given n using CBC."""
    try:
        import cvxpy as cp
    except Exception:  # cvxpy not available
        return None, None, None
    target = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    cons = [cp.sum(h) == target, t >= 0, t <= target]

    for s in range(1, n):
        m = n - s
        if m == 0:
            continue
        w = cp.Variable(m, boolean=True)
        hi = h[:m]
        hj = h[s:]
        cons += [
            w <= hi,
            w <= hj,
            w >= hi + hj - 1,
            cp.sum(w) >= t,
        ]
    prob = cp.Problem(cp.Maximize(t), cons)
    prob.solve(solver=cp.CBC, verbose=False)
    if h.value is None:
        return None, None, None
    h_val = np.round(h.value).astype(float)  # enforce exact 0/1
    return h_val, float(t.value), prob.status
