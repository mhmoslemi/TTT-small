def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def objective(vec, n, penalty_weight=1e3):
    sum_h = np.sum(vec)
    target = n / 2.0
    penalty = penalty_weight * abs(sum_h - target)
    # maybe also penalize if any element > 1 or <0, but DE keeps within bounds.
    return compute_c5(vec) + penalty

def hill_climb(h, max_moves, start_time, budget):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    while max_moves > 0 and time.time() - start_time < budget:
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        # Move delta from i to j
        max_delta = min(best_h[i], 1 - best_h[j])
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta  # random up to max
        # random direction? also could move opposite direction
        # We'll also try moving from j to i
        if np.random.rand() < 0.5:
            # move from i to j
            new_h = best_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move from j to i
            delta_rev = min(best_h[j], 1 - best_h[i])
            if delta_rev <= 0:
                continue
            delta = np.random.rand() * delta_rev
            new_h = best_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        new_score = compute_c5(new_h)
        if new_score < best_score - 1e-9:  # improvement
            best_h = new_h
            best_score = new_score
        max_moves -= 1
    return best_h, best_score

import time
import numpy as np
from scipy.optimize import differential_evolution

def _c5_of(h: np.ndarray) -> float:
    """Return the C₅ value for a step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1‑h), full linear convolution
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _objective(x: np.ndarray, n: int, penalty: float = 1e4) -> float:
    """Objective for the optimiser: C₅ plus a penalty for breaking the integral."""
    s = np.sum(x)
    target = n * 0.5
    pen = penalty * abs(s - target)
    return _c5_of(x) + pen


def _random_binary(n: int) -> np.ndarray:
    """Create a random binary vector with exactly n/2 ones (if n even)."""
    half = n // 2
    h = np.array([1.0] * half + [0.0] * (n - half), dtype=float)
    np.random.shuffle(h)
    return h


def _hill_climb(start: np.ndarray, moves: int, time_limit: float,
                start_time: float) -> (np.ndarray, float):
    """Simple stochastic local search: transfer mass between two positions."""
    h = start.copy()
    best = h.copy()
    best_val = _c5_of(best)
    n = h.size
    while moves and (time.time() - start_time) < time_limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a random amount from i to j (preserving the total sum)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        val = _c5_of(new_h)
        moves -= 1
        if val < best_val - 1e-12:
            best = new_h
            best_val = val
            h = new_h
    return best, best_val
