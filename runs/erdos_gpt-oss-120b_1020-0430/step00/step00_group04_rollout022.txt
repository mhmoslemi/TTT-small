def project_h(v, target):
    x = np.clip(v, 0.0, 1.0)
    s = x.sum()
    if s == target:
        return x
    # determine direction
    if s < target:
        # need to increase
        slack = 1.0 - x
        # while delta left positive
        delta = target - s
        while delta > 1e-12:
            # indices where slack > 1e-12
            idx = np.where(slack > 1e-12)[0]
            if len(idx) == 0:
                break
            # distribute equally
            inc = delta / len(idx)
            inc = np.minimum(slack[idx], inc)
            x[idx] += inc
            slack[idx] -= inc
            delta -= inc.sum()
    else:
        delta = s - target
        slack = x  # amount can decrease before hitting lower bound
        while delta > 1e-12:
            idx = np.where(slack > 1e-12)[0]
            if len(idx) == 0:
                break
            dec = delta / len(idx)
            dec = np.minimum(slack[idx], dec)
            x[idx] -= dec
            slack[idx] -= dec
            delta -= dec.sum()
    return x

        new_c = compute_c5(h)

        accept = False

            # acceptance probability
            if rng.random() < np.exp(-delta / temp):

        if accept:

                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

    return best_h, best_c

import time
import numpy as np

def compute_c5(h):
    """Return C5 for binary (or float) vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), (1.0 - h).astype(np.float64), mode='full')
    return np.max(corr) * dx, corr

def mismatched_indices(h, shift):
    """Indices i where h[i]==1 and the shifted partner is 0."""
    n = h.shape[0]
    if shift >= 0:
        if shift == n:
            return np.array([], dtype=int)
        mask = (h[:n - shift] == 1) & (h[shift:] == 0)
        return np.where(mask)[0]
    else:
        L = -shift
        if L == n:
            return np.array([], dtype=int)
        mask = (h[L:] == 1) & (h[:n - L] == 0)
        return np.where(mask)[0] + L

def random_balanced_binary(n, rng):
    """Create a binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    h[: n // 2] = 1
    rng.shuffle(h)
    return h

def sturmian_binary(n, alpha):
    """Sturmian (Beatty) construction: half ones with low discrepancy."""
    i = np.arange(n)
    frac = (i * alpha) % 1.0
    order = np.argsort(frac)
    h = np.zeros(n, dtype=np.int8)
    h[order[-n // 2 :]] = 1
    return h

def simulated_annealing_binary(n, rng, budget_s, start_time, init_h):
    """Simple SA that tries to minimise the maximum overlap."""
    dx = 2.0 / n
    # current state
    h = init_h.copy()
    cur_c, cur_corr = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    it = 0
    # for occasional restarts
    last_improve = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.97:
            break
        it += 1

        # shift giving the current worst overlap
        max_idx = np.argmax(cur_corr)
        shift = max_idx - (n - 1)

        # choose a mismatched 1 (if any) otherwise any 1
        mism = mismatched_indices(h, shift)
        if mism.size:
            i = rng.choice(mism)
        else:
            ones = np.where(h == 1)[0]
            i = rng.choice(ones)

        # choose a zero to swap with
        zeros = np.where(h == 0)[0]
        if zeros.size == 0:
            break
        j = rng.choice(zeros)

        # propose swap
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]

        new_c, new_corr = compute_c5(h_new)

        # temperature schedule (linear cooling)
        frac = elapsed / budget_s
        temp = 0.02 * (1.0 - frac) + 0.0005 * frac

        delta = new_c - cur_c
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / temp):
                accept = True

        if accept:
            h = h_new
            cur_c = new_c
            cur_corr = new_corr
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
                last_improve = it

        # restart if no improvement for a long stretch
        if it - last_improve > 20000:
            h = random_balanced_binary(n, rng)
            cur_c, cur_corr = compute_c5(h)
            last_improve = it

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # target length (choose an even number <1000)
    n = 200
    if n % 2:
        n += 1

    # start from a few deterministic seeds
    candidates = []

    # 1. Random binary start
    candidates.append(random_balanced_binary(n, rng))

    # 2. Sturmian sequences with different slopes
    phi = (np.sqrt(5) - 1) / 2.0            # 0.618...
    candidates.append(sturmian_binary(n, phi))
    candidates.append(sturmian_binary(n, np.sqrt(2) - 1.0))

    # 3. If a parent construction is provided, use a binary version of it
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.shape[0] == n and n % 2 == 0:
            # take the n/2 largest entries as ones
            order = np.argsort(-init)
            bin_init = np.zeros(n, dtype=np.int8)
            bin_init[order[: n // 2]] = 1
            candidates.append(bin_init)
        elif init.shape[0] < n:
            # tile the parent to reach length n then enforce balance
            reps = (n + init.shape[0] - 1) // init.shape[0]
            tiled = np.tile(init, reps)[:n]
            order = np.argsort(-tiled)
            bin_tiled = np.zeros(n, dtype=np.int8)
            bin_tiled[order[: n // 2]] = 1
            candidates.append(bin_tiled)

    # Evaluate initial candidates, keep the best as a starting point
    best_h = None
    best_c = np.inf
    for cand in candidates:

            break

    # Return as float array (the harness works with NumPy arrays)
    return (best_h.astype(np.float64), best_c, n)
