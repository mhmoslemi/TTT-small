def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time, math
    np.random.seed(seed)
    start = time.time()
    max_time = budget_s - 5.0  # safety

    # Determine n; if initial_h_values existing, use its length else 84
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        init_vec = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 84
        init_vec = None

    target_sum = n / 2.0
    dx = 2.0 / n

    def compute_c5(vec):
        # compute correlation
        c = np.correlate(vec, 1 - vec, mode='full')
        # We might want to compute max directly
        return c.max() * dx

    # projection function
    def project_to_constraints(v):
        # Ensure 0 <= v <= 1 after shifting lambda
        lo = -1.0
        hi = 2.0
        # binary search for lambda such that sum clip(v - lambda) = target_sum
        for _ in range(60):
            mid = (lo + hi) * 0.5
            w = np.clip(v - mid, 0.0, 1.0)
            s = w.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
        # adjust minor diff
        diff = target_sum - w.sum()
        if abs(diff) > 1e-10:
            # distribute diff among free entries
            free = (w > 0.0) & (w < 1.0)
            if free.any():
                w[free] += diff / free.sum()
                w = np.clip(w, 0.0, 1.0)
        return w

    # create initial candidates
    candidates = []
    # include parent if init_vec provided
    if init_vec is not None:
        if not np.isclose(init_vec.sum(), target_sum, atol=1e-8):
            init_vec = project_to_constraints(init_vec)
        candidates.append(init_vec.copy())
    # binary random half ones
    for _ in range(2):
        v = np.zeros(n)
        ones_idx = np.random.choice(n, size=n//2, replace=False)
        v[ones_idx] = 1.0
        candidates.append(v)
    # random continuous with projection
    for _ in range(5):
        v = np.random.rand(n)
        v = project_to_constraints(v)
        candidates.append(v)

    best_h = None
    best_c5 = math.inf
    for h in candidates:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Simulated annealing/hill climbing
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # parameters
    # max iteration counts adapt based on remaining time
    # Use exponential cooling.
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        # compute schedule
        frac = elapsed / max_time
        # Temperature for acceptance of worse moves
        T = 0.01 * (1 - frac) + 0.0001
        # max delta for swaps
        eps = 0.3 * (1 - frac) + 0.005
        # propose swap
        i, j = np.random.choice(n, size=2, replace=False)
        # Determine allowable delta range
        # we can move mass from i to j
        max_pos = min(eps, cur_h[i], 1.0 - cur_h[j])
        max_neg = min(eps, cur_h[j], 1.0 - cur_h[i])  # moving from j to i
        if max_pos <= 0 and max_neg <= 0:
            # cannot move, skip
            continue
        if np.random.rand() < 0.5 and max_pos > 0:
            delta = np.random.uniform(0, max_pos)
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        elif max_neg > 0:
            delta = np.random.uniform(0, max_neg)
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            continue

        # Evaluate
        new_c5 = compute_c5(new_h)

        # Acceptance
        if new_c5 < cur_c5 or np.random.rand() < math.exp(-(new_c5 - cur_c5) / T):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                # Early stop if enough
                if best_c5 <= 0.3808:
                    # break? maybe immediate return
                    # Continue some iterations to confirm
                    pass
        # occasional random reset to diversify
        if np.random.rand() < 0.001:
            # random new start
            v = np.random.rand(n)
            cur_h = project_to_constraints(v)
            cur_c5 = compute_c5(cur_h)

        it += 1

    # final projection just in case
    best_h = project_to_constraints(best_h)

    # compute final c5
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n)

def subgradient_descent(h0, max_iters, max_time):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    for t in range(max_iters):
        if time.time() - start > max_time:
            break
        f = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(f)
        s = max_idx - (len(h) - 1)  # shift value
        # gradient
        grad = np.zeros_like(h)
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_start < i_end:
            idx_i = np.arange(i_start, i_end)
            idx_j = idx_i + s
            grad[idx_i] += 1 - h[idx_j]
            grad[idx_j] -= h[idx_i]
        # step size
        step = 0.5 / math.sqrt(t+1)  # tune
        h_new = h - step * grad
        h_new = project_to_constraints(h_new)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        # if improves or occasionally accept
        if val <= compute_c5(h) or np.random.rand() < 0.01:
            h = h_new
        # occasional random perturbation
        if t % 1000 == 0:
            # small random noise
            pert = np.random.randn(n) * 0.01
            h = project_to_constraints(h + pert)
    return best_h, best_val

import numpy as np, time, math

def evaluate(h, dx):
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def project(v, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        w = np.clip(v - mid, 0.0, 1.0)
        if w.sum() > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo+hi)*0.5, 0.0, 1.0)
    # adjust residual
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w>0.0) & (w<1.0)
        if free.any():
            w[free] += diff/ free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def random_start(n, target):
    v = np.random.rand(n)
    return project(v, target)

def binary_start(n, target):
    v = np.zeros(n)
    k = int(target)  # expected target = n/2 integer if n even
    idx = np.random.choice(n, k, replace=False)
    v[idx] = 1.0
    return v

def cosine_start(n, target, a):
    xs = np.linspace(0, 2, n, endpoint=False) + 1.0/n  # midpoint shift not crucial
    h = 0.5 + a * np.cos(np.pi * xs / 2)
    h = np.clip(h, 0.0, 1.0)
    return project(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s - 5.0
    # decide n
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
    else:
        n0 = None
    n = 84   # choose even
    target_sum = n/2.0
    dx = 2.0/n
    # Build initial candidate list
    candidates = []
    # interpolated parent if exists
    if n0 is not None:
        h_parent = np.array(initial_h_values, dtype=float)
        if n0 != n:
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_parent = np.interp(x_new, x_old, h_parent)
        h_parent = project(h_parent, target_sum)
        candidates.append(h_parent)
    # constant half
    candidates.append(np.full(n, 0.5))
    # binary random
    for _ in range(3):
        candidates.append(binary_start(n, target_sum))
    # random uniform
    for _ in range(3):
        candidates.append(random_start(n, target_sum))
    # cosine family
    for a in np.linspace(0.1, 0.45, 5):
        candidates.append(cosine_start(n, target_sum, a))
    # evaluate candidates
    best_h = None
    best_c5 = math.inf
    for h in candidates:
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    # optimization loop
    cur_h = best_h.copy()
    cur_c5 = best_c5
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break
        # compute correlation
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)
        # gradient for worst shift
        grad = np.zeros_like(cur_h)
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_start < i_end:
            idx_i = np.arange(i_start, i_end)
            idx_j = idx_i + s
            grad[idx_i] += (1 - cur_h[idx_j])
            grad[idx_j] -= cur_h[idx_i]
        # step size schedule
        step = 0.02 / math.sqrt(it + 1)
        h_new = cur_h - step * grad
        h_new = project(h_new, target_sum)
        new_c5 = evaluate(h_new, dx)
        # acceptance criteria (simple downhill + occasional uphill)
        if new_c5 < cur_c5 or np.random.rand() < 0.001:
            cur_h = h_new
            cur_c5 = new_c5
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        # periodic random perturbation to escape

            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        it += 1
    # final projection for safety
    best_h = project(best_h, target_sum)
    final_c5 = evaluate(best_h, dx)
    return (best_h, final_c5, n)

import numpy as np, time, math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s - 5.0  # safety margin

    # ------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        n_parent = len(initial_h_values)
    else:
        n_parent = None

    # use a moderate resolution (even) – larger n gives more flexibility
    n = 84
    target_sum = n / 2.0          # integral must be 1  →  sum h_i = n/2
    dx = 2.0 / n

    # ------------------------------------------------------------
    # helper: projection on box [0,1] and sum constraint
    # ------------------------------------------------------------
    def project(v):
        lo, hi = -1.0, 2.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            w = np.clip(v - mid, 0.0, 1.0)
            if w.sum() > target_sum:
                lo = mid
            else:
                hi = mid
        w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

        # tiny adjustment for numerical drift
        diff = target_sum - w.sum()
        if abs(diff) > 1e-12:
            free = (w > 0.0) & (w < 1.0)
            if free.any():
                w[free] += diff / free.sum()
                w = np.clip(w, 0.0, 1.0)
        return w

    # ------------------------------------------------------------
    # evaluate C5 for a vector h
    # ------------------------------------------------------------
    def evaluate(h):
        # full cross‑correlation of h with (1‑h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return corr.max() * dx

    # ------------------------------------------------------------
    # generate a few diverse starting vectors
    # ------------------------------------------------------------
    starters = []

    # parent (if any) interpolated to the chosen n
    if n_parent is not None:
        parent = np.asarray(initial_h_values, dtype=float)
        if n_parent != n:
            x_old = np.linspace(0.0, 2.0, n_parent, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            parent = np.interp(x_new, x_old, parent)
        starters.append(project(parent))

    # constant 0.5 (already feasible)
    starters.append(np.full(n, 0.5))

    # binary vectors (exactly half ones)
    for _ in range(3):
        v = np.zeros(n)
        ones = np.random.choice(n, size=n // 2, replace=False)
        v[ones] = 1.0
        starters.append(v)

    # uniform random vectors projected onto constraints
    for _ in range(3):
        v = np.random.rand(n)
        starters.append(project(v))

    # cosine‑shaped families: h = 0.5 + a·cos(πx/2) , a∈[0.1,0.45]
    xs = np.linspace(0.0, 2.0, n, endpoint=False) + dx * 0.5
    for a in np.linspace(0.1, 0.45, 5):
        h = 0.5 + a * np.cos(np.pi * xs / 2.0)
        starters.append(project(h))

    # pick the best among the seeds
    best_h = None
    best_val = math.inf
    for h in starters:
        val = evaluate(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    cur_h = best_h.copy()
    cur_val = best_val

    # ------------------------------------------------------------
    # main simulated‑annealing loop
    # ------------------------------------------------------------
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break
        # cooling schedules (linear in elapsed fraction)
        frac = elapsed / max_time
        temperature = (1.0 - frac) * 0.01 + 0.0001
        scale = (1.0 - frac) * 0.5 + 0.01        # maximal perturbation amplitude

        # propose a global perturbation and re‑project
        proposal = cur_h + scale * (np.random.rand(n) - 0.5) * 2.0
        proposal = project(proposal)
        prop_val = evaluate(proposal)

        # metropolis acceptance
        if prop_val < cur_val or np.random.rand() < math.exp(-(prop_val - cur_val) / temperature):
            cur_h = proposal
            cur_val = prop_val
            if prop_val < best_val:
                best_val = prop_val
                best_h = proposal.copy()
                # early exit if target reached
                if best_val <= 0.3808:
                    break

        # occasional random reset to avoid stagnation
        if np.random.rand() < 0.001:
            cur_h = project(np.random.rand(n))
            cur_val = evaluate(cur_h)

        # every few hundred iterations inject a small Gaussian jitter
        if it % 500 == 0 and it > 0:
            jitter = np.random.randn(n) * 0.02
            cur_h = project(cur_h + jitter)
            cur_val = evaluate(cur_h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()

        it += 1

    # final safety projection and evaluation
    best_h = project(best_h)
    final_val = evaluate(best_h)

    return (best_h, final_val, n)
