def gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)  # shift = k
    # gradient for each i: d/dh_i [sum_j h_j - sum_j h_j * h_{j+shift}]
    # So = 1 - h_{i+shift} - h_{i-shift}
    # but note out-of-range indices => treat missing as 0
    grad = np.ones_like(h)
    # subtract h_{i+shift} where i+shift in bounds
    if shift >= 0:
        # term h_i * h_{i+shift}, derivative w.r.t h_i gives -h_{i+shift}
        # and w.r.t h_{i+shift} gives -h_i (not needed for gradient w.r.t h_i)
        # Already accounted.
        # need to subtract h_{i+shift} from grad[i], for i in [0, n-1-shift]
        grad[:n - shift] -= h[shift:]  # h_{i+shift}
        # Also subtract h_{i-shift} where i-shift >= 0
        # shift positive => i-shift range shift..n-1
        grad[shift:] -= h[:n - shift]  # h_{i-shift}
    else:
        # shift negative: similar but reversed
        s = -shift
        # i+shift = i - s
        grad[s:] -= h[:n - s]  # h_{i+shift}
        grad[:n - s] -= h[s:]  # h_{i-shift}
    # gradient of max_val * dx = grad * dx
    return grad * dx

def random_swap_move(h, rng):

    i, j = rng.choice(n, size=2, replace=False)
    # we will move mass from i to j or vice versa
    # choose delta in range [-max_neg, max_pos] where positive delta means increase i, decrease j.
    # compute max positive? Actually we want keep h_i, h_j within [0,1].
    max_inc_i = 1.0 - h[i]
    max_dec_i = h[i]
    max_inc_j = 1.0 - h[j]
    max_dec_j = h[j]
    # if we increase i by delta and decrease j by delta:
    delta_max = min(max_inc_i, max_dec_j)
    delta_min = -min(max_inc_j, max_dec_i) # negative min means decreasing i, increasing j
    # choose delta uniformly in [delta_min, delta_max]
    if delta_max <= delta_min:
        return h  # cannot move
    delta = rng.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

    from scipy.optimize import minimize

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine initial vector
    try:
        init_h_raw = np.array(initial_h_values, dtype=float)
        init_n_raw = init_h_raw.shape[0]
        # Make even length target
        if init_n_raw % 2 == 1:
            # interpolate to next even length
            n_start = init_n_raw + 1
        else:
            n_start = init_n_raw
    except Exception:
        init_h_raw = None
        n_start = 48  # default

    # Function definitions

    def project_to_feasible(v, target):
        # Solve for lambda: sum(clamp(v + λ, 0, 1)) = target
        lo = -np.max(v) - 2.0
        hi = 1.0 - np.min(v) + 2.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(v + mid, 0.0, 1.0))
            if s > target:
                hi = mid
            else:
                lo = mid
        return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

    def interpolate_h(old_h, new_n):
        old_n = len(old_h)
        # treat domain periodic? Not periodic; just linear interpolation
        old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
        new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
        new_h = np.interp(new_x, old_x, old_h)
        new_h = np.clip(new_h, 0.0, 1.0)
        target = new_n / 2.0
        return project_to_feasible(new_h, target)

    def evaluate(h):
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        return max_val * dx

    def gradient(h):
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)  # can be negative
        g = np.ones_like(h)
        if shift >= 0:
            # subtract h[i+shift] for i in 0..n-1-shift
            g[:n - shift] -= h[shift:]
            # subtract h[i-shift] for i in shift..n-1
            g[shift:] -= h[:n - shift]
        else:
            s = -shift
            # shift negative
            g[s:] -= h[:n - s]
            g[:n - s] -= h[s:]
        return g * dx

    # SLSQP optimization for a given start vector
    def opt_start(start_vec):
        n = start_vec.shape[0]
        bounds = [(0.0, 1.0)] * n
        eq_cons = {'type': 'eq',
                   'fun': lambda h: np.sum(h) - n / 2.0,
                   'jac': lambda h: np.ones_like(h)}
        # Options: limit iterations to keep runtime low
        try:
            res = minimize(fun=lambda h: evaluate(h),
                           x0=start_vec,
                           jac=lambda h: gradient(h),
                           bounds=bounds,
                           constraints=[eq_cons],
                           method='SLSQP',
                           options={'maxiter': 150, 'ftol': 1e-9, 'disp': False})
        except Exception:
            return None, np.inf
        # Ensure feasibility after optimization
        h_opt = project_to_feasible(res.x, n / 2.0)
        val = evaluate(h_opt)
        return h_opt, val

    # Generate random feasible start
    def random_feasible_start(n):
        v = rng.random(n)
        return project_to_feasible(v, n / 2.0)

    def random_binary_start(n):
        # n must be even
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        return h

    # Local random swap refinement
    def refine_local(h, steps=200, step_scale=0.5):
        n = h.shape[0]
        target_sum = n / 2.0
        cur_val = evaluate(h)
        best_h = h.copy()
        best_val = cur_val
        for _ in range(steps):
            # propose a swap move
            i, j = rng.choice(n, size=2, replace=False)
            hi, hj = h[i], h[j]
            # compute allowed delta range
            delta_max = min(1.0 - hi, hj)  # increase hi, decrease hj
            delta_min = -min(hi, 1.0 - hj)  # decrease hi, increase hj
            if delta_max <= delta_min:

            delta = rng.uniform(delta_min, delta_max) * step_scale
            h_new = h.copy()
            h_new[i] = hi + delta
            h_new[j] = hj - delta
            # sum remains same, bounds respected by construction
            val_new = evaluate(h_new)
            if val_new < best_val - 1e-12:
                best_h = h_new
                best_val = val_new
                h = h_new
                # possibly increase step_scale if improvement? Not needed
        return best_h, best_val

    # Main optimization loop over grid sizes
    best_h_global = None
    best_val_global = np.inf
    best_n_global = None

    # Determine list of grid sizes: start with even n_start, then maybe double up to limit
    n_list = []
    n_cur = n_start if n_start % 2 == 0 else n_start + 1
    # ensure at least 48

    # generate sizes up to maybe 384 (2^3 * 48)

        n_list.append(n_cur)
        n_cur *= 2  # double

    # For each n in list
    for n in n_list:
        # time check, break if low budget
        if time.time() - start_time > budget_s * 0.98:
            break
        # Prepare initial start vectors for this n

        start_vectors = []

        # Use interpolated previous best if exists
        if best_h_global is not None and best_h_global.shape[0] == n:
            # same size
            start_vectors.append(best_h_global.copy())
        elif best_h_global is not None:
            # interpolate previous best to current n
            start_vectors.append(interpolate_h(best_h_global, n))

        # Use initial_h if available (only first grid)
        if n == n_list[0]:

                                                             n/2.0))
        # Add a few binary and random starts
        # decide number based on time left and n size
        # allocate about 30 random restarts per n
        num_random = 30
        num_binary = 10 if n % 2 == 0 else 0

        for _ in range(num_binary):
            start_vectors.append(random_binary_start(n))

        # Now iterate over starts and optimize

            start = project_to_feasible(start, n/2.0)
            h_opt, val = opt_start(start)
            if h_opt is None:

            # optional local refinement
            if time.time() - start_time < budget_s * 0.95:
                # perform small local refinement
                h_ref, val_ref = refine_local(h_opt, steps=200, step_scale=0.5)
                if val_ref < val:
                    h_opt, val = h_ref, val_ref
            # update global best
            if val < best_val_global - 1e-12:
                best_val_global = val
                best_h_global = h_opt.copy()
                best_n_global = n
                # optional early stop if below target
                # if val <= 0.3808: break  # maybe we want to keep exploring but can break
        # At end of n loop, perhaps attempt additional refinement on best of this n
        if best_h_global is not None and best_h_global.shape[0] == n:
            # local refinement more iterations
            if time.time() - start_time < budget_s * 0.95:
                h_ref2, val_ref2 = refine_local(best_h_global, steps=500, step_scale=0.3)
                if val_ref2 < best_val_global - 1e-12:
                    best_val_global = val_ref2
                    best_h_global = h_ref2.copy()
                    best_n_global = n

    # final projection to ensure feasible
    if best_h_global is not None:
        best_h_global = project_to_feasible(best_h_global, best_n_global / 2.0)
        best_val_global = evaluate(best_h_global)

    # Return values
    return best_h_global, float(best_val_global), int(best_n_global)

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions (top level) ----------

def project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n while fixing sum(v)=target.
    Uses a simple bisection on the shift λ such that sum(clamp(v+λ,0,1))=target."""
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def interpolate_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of step function defined by old_h onto a new grid of size new_n."""
    old_n = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    new_h = np.clip(new_h, 0.0, 1.0)
    return project_to_feasible(new_h, new_n / 2.0)


def evaluate_h(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx for a discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def gradient_h(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of C5 for the shift attaining the maximum."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # shift can be negative
    g = np.ones_like(h)
    if shift >= 0:
        # subtract h[i+shift] for i where i+shift is valid
        g[: n - shift] -= h[shift:]
        # subtract h[i-shift] for i where i-shift is valid
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        g[s:] -= h[: n - s]
        g[: n - s] -= h[s:]
    return g * dx


def eq_constraint_func(h: np.ndarray) -> float:
    """Equality constraint: sum(h) == len(h)/2."""
    return np.sum(h) - (h.shape[0] / 2.0)


def eq_constraint_jac(h: np.ndarray) -> np.ndarray:
    """Jacobian of the equality constraint."""
    return np.ones_like(h)


def random_feasible_start(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a random vector in [0,1]^n with the required sum."""
    v = rng.random(n)
    return project_to_feasible(v, n / 2.0)


def random_binary_start(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a binary feasible vector (exactly n/2 ones). n must be even."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def refine_local(h: np.ndarray, rng: np.random.Generator,
                 steps: int = 200, step_scale: float = 0.5) -> tuple[np.ndarray, float]:
    """Simple local search: swap a small amount of mass between two coordinates."""
    n = h.shape[0]
    cur_val = evaluate_h(h)
    best_h = h.copy()
    best_val = cur_val
    for _ in range(steps):
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h[i], h[j]
        # feasible delta range preserving bounds
        delta_max = min(1.0 - hi, hj)
        delta_min = -min(hi, 1.0 - hj)
        if delta_max <= delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max) * step_scale
        h_new = h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
        val_new = evaluate_h(h_new)
        if val_new < best_val - 1e-12:
            best_h = h_new
            best_val = val_new
            h = h_new
    return best_h, best_val


def opt_h(start_vec: np.ndarray, maxiter: int = 150) -> tuple[np.ndarray, float]:
    """Run SLSQP from a feasible start vector."""
    n = start_vec.shape[0]
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq',
                    'fun': eq_constraint_func,
                    'jac': eq_constraint_jac}]
    res = minimize(evaluate_h,
                   x0=start_vec,
                   method='SLSQP',
                   jac=gradient_h,
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    # safeguard feasibility
    h_opt = project_to_feasible(res.x, n / 2.0)
    return h_opt, evaluate_h(h_opt)


# ---------- main run function ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] giving a small C5 value."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- obtain initial construction if supplied -----
    try:
        init_h_raw = np.array(initial_h_values, dtype=float)  # provided by harness
        init_n_raw = init_h_raw.shape[0]
    except Exception:
        init_h_raw = None
        init_n_raw = None

    # ----- decide on a starting grid size -----
    if init_n_raw is not None:
        n_start = init_n_raw if init_n_raw % 2 == 0 else init_n_raw + 1
    else:
        n_start = 48
    if n_start < 48:
        n_start = 48

    # build a list of grid sizes (doubling each time, never exceeding 384)
    n_list = []
    n_cur = n_start
    while n_cur <= 384:
        n_list.append(n_cur)
        n_cur *= 2

    best_h_global = None
    best_val_global = math.inf
    best_n_global = None

    # ----- main optimisation loop over grid sizes -----
    for n in n_list:
        # abort if time is almost exhausted
        if time.time() - start_time > budget_s * 0.98:
            break

        # collect starting vectors
        starts = []

        # 1) interpolate previous best if it exists
        if best_h_global is not None:
            if best_h_global.shape[0] == n:
                starts.append(best_h_global.copy())
            else:
                starts.append(interpolate_h(best_h_global, n))

        # 2) use the harness initial vector on the first grid
        if n == n_list[0] and init_h_raw is not None:
            if init_h_raw.shape[0] == n:
                starts.append(project_to_feasible(init_h_raw.copy(), n / 2.0))
            else:
                starts.append(interpolate_h(init_h_raw, n))

        # 3) a handful of binary starts (if n even)
        if n % 2 == 0:
            for _ in range(6):
                starts.append(random_binary_start(rng, n))

        # 4) random continuous starts
        for _ in range(18):
            starts.append(random_feasible_start(rng, n))

        # ----- optimise each start -----
        for start in starts:
            if time.time() - start_time > budget_s * 0.98:
                break
            # ensure exact feasibility before sending to optimizer
            start = project_to_feasible(start, n / 2.0)

            h_opt, val = opt_h(start, maxiter=150)

            # small local refinement
            if time.time() - start_time < budget_s * 0.95:
                h_ref, val_ref = refine_local(h_opt, rng, steps=200, step_scale=0.5)
                if val_ref < val:
                    h_opt, val = h_ref, val_ref

            # update global best
            if val < best_val_global - 1e-12:
                best_val_global = val
                best_h_global = h_opt.copy()
                best_n_global = n

        # after all starts for this n, apply a bit stronger refinement to the current best of this grid
        if best_h_global is not None and best_h_global.shape[0] == n:
            if time.time() - start_time < budget_s * 0.95:
                h_ref2, val_ref2 = refine_local(best_h_global, rng, steps=500, step_scale=0.3)
                if val_ref2 < best_val_global - 1e-12:
                    best_val_global = val_ref2
                    best_h_global = h_ref2.copy()
                    best_n_global = n

    # final safety projection & evaluation
    if best_h_global is not None:
        best_h_global = project_to_feasible(best_h_global, best_n_global / 2.0)
        best_val_global = evaluate_h(best_h_global)

    return best_h_global, float(best_val_global), int(best_n_global)
