def project_onto_bounds_sum(v, target):
    # find λ such that sum(np.clip(v + λ, 0, 1)) = target
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi)/2
        w = np.clip(v + mid, 0, 1)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0, 1)

def _evaluate(h, dx):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated by correlation."""
    # full correlation, then scale by dx
    r = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(r.max())

def _project_to_feasible(v, target):
    """Project vector v onto {0≤x≤1, sum(x)=target} by a simple bisection."""
    # monotone sum of clip(v+λ)
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _random_feasible(n, target, rng):
    """Create a random vector in [0,1]^n with given sum."""
    v = rng.random(n)
    return _project_to_feasible(v, target)
