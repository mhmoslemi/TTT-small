def compute_c5(h):
    dx = 2.0 / len(h)
    # using np.correlate
    c = np.correlate(h, 1 - h, mode='full')

def hill_climb(h, max_iter, time_limit):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # pick random positions
        i = random.randint(0, n-1)
        while cur_h[i] == 0:
            i = random.randint(0, n-1)
        j = random.randint(0, n-1)
        while cur_h[j] == 1:
            j = random.randint(0, n-1)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_val = compute_c5(cur_h)
        if new_val <= cur_val or random.random() < 0.001:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_val

import cvxpy as cp
def solve_milp(n, time_limit):
    h = cp.Variable(n, boolean=True)
    # y variables for each pair (i,j) with i<j
    y = {}
    for i in range(n):
        for j in range(i+1, n):
            y[(i,j)] = cp.Variable(boolean=True)
    # Lambda
    lam = cp.Variable(integer=True)
    constraints = []
    # sum constraint
    constraints.append(cp.sum(h) == n//2)
    # product constraints
    for (i,j), y_ij in y.items():
        constraints.append(y_ij <= h[i])
        constraints.append(y_ij <= h[j])
        constraints.append(y_ij >= h[i] + h[j] - 1)
    # shift constraints
    for k in range(1, n):
        # sum y_{i,i+k}
        terms = [y[(i, i+k)] for i in range(n-k)]
        constraints.append(cp.sum(cp.hstack(terms)) >= lam)
    # lambda bound: lam <= n//2 maybe
    constraints.append(lam <= n//2)
    # objective maximize lam
    obj = cp.Maximize(lam)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CBC, TimeLimit=time_limit)
    # After solving, retrieve solution
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # fallback: maybe not solved
        ...
    h_val = np.round(h.value).astype(int) # get binary
    lam_val = lam.value

    h = cp.Variable(n, boolean=True)
    Y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    constraints = []

    constraints.append(cp.sum(h) == n//2)
    # product constraints
    for i in range(n):
        constraints.append(Y[i,i] == 0)  # optional
        for j in range(i+1, n):
            constraints.append(Y[i,j] <= h[i])
            constraints.append(Y[i,j] <= h[j])
            constraints.append(Y[i,j] >= h[i] + h[j] - 1)
    # shift constraints
    total_ones = n // 2
    for k in range(1, n):
        L = n - k
        prefix_sum = cp.sum(h[:L])
        pair_sum = cp.sum(Y[:L, np.arange(k, n)])   # Y[i,i+k] for i in 0..L-1

import time
import random
import numpy as np

def _compute_c5(h):
    """Return C5 for binary step vector h (list/np.array of 0/1)."""
    n = len(h)
    dx = 2.0 / n
    # correlation h with (1-h); mode='full' gives all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def _hill_climb(start_h, max_iters, time_limit):
    """Simple swap‑based hill climbing.
    Returns best (h, value) found within time_limit seconds."""
    n = len(start_h)
    h = start_h.copy()
    best_h = h.copy()
    best_val = _compute_c5(h)
    cur_val = best_val
    start = time.time()
    # pre‑compute indices of ones/zeros for quicker sampling
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    rng = random.Random()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # pick a random 1 and a random 0 to swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0
        h[j] = 1
        # update index lists
        ones_idx = np.where(ones == i)[0][0]
        zeros_idx = np.where(zeros == j)[0][0]
        ones[ones_idx] = j
        zeros[zeros_idx] = i
        new_val = _compute_c5(h)
        # accept if improvement or very small uphill move
        if new_val <= cur_val or rng.random() < 0.001:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            ones[ones_idx] = i
            zeros[zeros_idx] = j
    return best_h, best_val

def _thue_morse(n):
    """Binary Thue‑Morse sequence of length n (n a power of two)."""
    return np.array([bin(i).count("1") & 1 for i in range(n)], dtype=int)

    Returns (h, t) where t is the maximal overlap count (integer or fractional)."""

    Y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()          # maximal overlap count (continuous)

    # exact half‑measure
    constraints.append(cp.sum(h) == n // 2)

    # force diagonal of Y to zero and lower triangle to zero
    for i in range(n):
        constraints.append(Y[i, i] == 0)
        for j in range(i):
            constraints.append(Y[i, j] == 0)

    # product linearisation for i<j
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(Y[i, j] <= h[i])
            constraints.append(Y[i, j] <= h[j])
            constraints.append(Y[i, j] >= h[i] + h[j] - 1)

    # shift constraints: t >= S_k for both directions of each shift
    total_ones = n // 2
    for k in range(1, n):
        L = n - k                       # length of overlap for shift k
        # prefix sum of h over the first L positions
        prefix_sum = cp.sum(h[:L])
        # sum of products h_i * h_{i+k} for i=0..L-1 (using upper‑triangular Y)
        pair_sum = cp.sum(cp.hstack([Y[i, i + k] for i in range(L)]))
        # suffix sum = total_ones - sum of first k entries of h
        suffix_sum = total_ones - cp.sum(h[:k])

        constraints.append(t >= prefix_sum - pair_sum)
        constraints.append(t >= suffix_sum - pair_sum)

    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # CBC accepts the option `max_seconds`; use it if supported
    solve_kwargs = {"solver": cp.CBC}
    try:
        solve_kwargs["max_seconds"] = time_limit
        prob.solve(**solve_kwargs)
    except TypeError:
        # fall back to a plain call without the time limit
        prob.solve(solver=cp.CBC)

    if h.value is None:
        return None, None

    # round the solution to 0/1 (tolerance to handle tiny numerical errors)
    h_sol = np.rint(np.array(h.value)).astype(int)
    # repair exact half‑measure if rounding introduced a tiny error
    cur_sum = int(h_sol.sum())
    target = n // 2
    if cur_sum != target:
        diff = target - cur_sum
        # flip arbitrary bits to fix the sum
        idx = np.where(h_sol == (0 if diff > 0 else 1))[0]
        if len(idx) >= abs(diff):
            h_sol[idx[:abs(diff)]] = 1 if diff > 0 else 0
    t_val = float(t.value) if t.value is not None else None
    return h_sol, t_val

def run(seed=42, budget_s=1100, **_):
    """Search for a step function yielding a low C5 bound."""
    start_total = time.time()
    rng = random.Random(seed)

    # ------------------------------------------------------------------
    # Determine problem size (use the supplied construction if present)
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=int)
        n0 = len(init_h)
        # Ensure the initial vector respects the exact half‑measure
        if init_h.sum() != n0 // 2:
            # simple repair by flipping random bits
            diff = n0 // 2 - int(init_h.sum())
            idx = np.where(init_h == (0 if diff > 0 else 1))[0]
            if len(idx) >= abs(diff):
                init_h[idx[:abs(diff)]] = 1 if diff > 0 else 0
    else:
        init_h = None
        n0 = 70                     # fallback size (even)
    # ------------------------------------------------------------------
    # Baseline evaluation
    # ------------------------------------------------------------------
    best_h = None
    best_c5 = float('inf')
    best_n = None

    def consider_candidate(h_vec):
        nonlocal best_h, best_c5, best_n
        c5 = _compute_c5(h_vec)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vec.copy()
            best_n = len(h_vec)

    if init_h is not None:
        consider_candidate(init_h)

    # ------------------------------------------------------------------
    # Try a modest MILP formulation (only for the current size)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_total)
    if remaining > 60 and n0 <= 80:          # keep the MIP tiny for reliability
        mip_time = min(remaining * 0.5, 250)   # give it at most ~250 s
        h_mip, t_mip = _solve_mip(n0, mip_time)
        if h_mip is not None and t_mip is not None:
            c5_mip = (2.0 / n0) * t_mip
            if c5_mip < best_c5:
                best_c5 = c5_mip
                best_h = h_mip.copy()
                best_n = n0

    # ------------------------------------------------------------------
    # Exhaustive local optimisation on several starts (including the best so far)
    # ------------------------------------------------------------------
    # allocate the rest of the budget to hill‑climbing
    remaining = budget_s - (time.time() - start_total)
    if remaining <= 0:
        return best_h.tolist(), best_c5, best_n

    # number of independent hill‑climb runs we can afford
    runs = 5
    per_run_time = remaining / runs
    max_iters = 200000

    # include the best candidate found so far as one of the starts
    start_vectors = []
    if best_h is not None:
        start_vectors.append(best_h.copy())
    # generate a few random balanced starts
    while len(start_vectors) < runs:
        n = n0
        h_rand = np.zeros(n, dtype=int)
        ones_idx = rng.sample(range(n), n // 2)
        h_rand[ones_idx] = 1
        start_vectors.append(h_rand)

    for h_start in start_vectors:
        t_start = time.time()
        time_left = per_run_time - (time.time() - t_start)
        if time_left <= 0:
            continue
        h_opt, c5_opt = _hill_climb(h_start, max_iters, time_left)
        consider_candidate(h_opt)

    # ------------------------------------------------------------------
    # Final sanity check – ensure feasibility
    # ------------------------------------------------------------------
    if best_h is None:
        # fall back to a simple alternating block if everything failed
        n = n0
        best_h = np.array([1 if i < n // 2 else 0 for i in range(n)], dtype=int)
        best_c5 = _compute_c5(best_h)
        best_n = n

    return best_h.tolist(), best_c5, best_n
