        ones = np.random.choice(n, size=n//2, replace=False)

            h_new = h.copy()

            new_score = evaluate(h_new)
            delta = new_score - current_score
            # Accept if improvement or with probability exp(-delta/T)
            T = T0 * (1 - step / max_steps) # linear cooling

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * dx

def project_onto_sum(v, target, max_iter=60):
    # Use binary search for lambda.
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust small residual if needed
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        # distribute diff proportionally to free variables (not at bounds)
        mask = (h > 1e-8) & (h < 1 - 1e-8)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    # final clip
    return h

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * dx

def project_onto_sum(v, target, max_iter=60):
    # Find λ such that sum_i clip(v_i - λ,0,1) = target.
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    h = np.clip(v - lam, 0.0, 1.0)
    # Small correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

import time, math, numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * dx


def project_onto_sum(v: np.ndarray, target: float, max_iter: int = 60) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x : 0 ≤ x_i ≤ 1, sum(x)=target}.
    The solution has the form x_i = clip(v_i - λ,0,1) for some λ.
    """
    lo = -np.max(v) - 1.0          # guarantees all entries become 1 after clipping
    hi =  np.max(v) + 1.0          # guarantees all entries become 0 after clipping
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:                      # tiny adjustment if necessary
        mask = (x > 1e-8) & (x < 1.0 - 1e-8)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def greedy_binary_optimize(h_bin: np.ndarray, target: float,
                           budget_s: float, start_time: float):
    """
    Local improvement for a binary vector.
    Repeatedly swaps a zero with a one if it increases the overlap of the
    presently worst shift.
    """
    n = len(h_bin)
    best_h = h_bin.copy()
    best_score = compute_c5(best_h)
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        # autocorrelation of the binary set
        autocorr = np.correlate(best_h, best_h, mode="full")
        c_shifts = autocorr[n:]                # shifts k = 1 .. n-1
        overlap = target - c_shifts
        worst_k = int(np.argmax(overlap)) + 1   # shift amount giving max overlap
        # contribution of each position to this shift
        inc = np.zeros(n)
        dec = np.zeros(n)
        inc[: n - worst_k] += best_h[worst_k:]
        inc[worst_k:]   += best_h[: n - worst_k]
        dec[: n - worst_k] += best_h[worst_k:]
        dec[worst_k:]   += best_h[: n - worst_k]
        zeros = np.where(best_h == 0)[0]
        ones  = np.where(best_h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        a = zeros[np.argmax(inc[zeros])]
        b = ones[np.argmin(dec[ones])]
        delta = inc[a] - dec[b]
        if delta <= 0:
            break
        # perform the swap
        best_h[a] = 1
        best_h[b] = 0
        new_score = compute_c5(best_h)
        if new_score < best_score:
            best_score = new_score
        else:                       # revert – no global improvement
            best_h[a] = 0
            best_h[b] = 1
            break
    return best_h, best_score

            best_h[j] = 1
        attempts += 1
    return best_h, best_score


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function giving a low Erdős C5 bound.
    Returns (list_of_h, c5_value, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # helpers common to the whole search
    # ------------------------------------------------------------------
    init_exists = "initial_h_values" in globals()
    init_vec = np.array(initial_h_values, dtype=float) if init_exists else None

    # candidate discretisations (must be even)
    n_candidates = [50, 80, 120, 160, 200, 240, 300]

    best_h = None
    best_score = float("inf")

    for n in n_candidates:
        if time.time() - start_time > budget_s * 0.92:
            break
        target = n / 2.0

        # ----- build an initial continuous vector --------------------
        if init_exists:
            n0 = init_vec.shape[0]
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            xn = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(xn, x0, init_vec)
        else:
            h0 = np.random.rand(n)

        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_sum(h0, target)

        # quick evaluation of the starting point
        cur_score = compute_c5(h0)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h0.copy()
        if best_score <= 0.3808:
            break

        # ----- optimisation passes -----------------------------------
        restarts = 4
        for _ in range(restarts):
            if time.time() - start_time > budget_s * 0.93:
                break

            # perturb the start (first restart uses the exact start)
            if _ == 0:
                cur = h0.copy()
            else:
                cur = h0 + np.random.normal(scale=0.15, size=n)
                cur = np.clip(cur, 0.0, 1.0)
                cur = project_onto_sum(cur, target)

            # sub‑gradient descent on the max–overlap functional
            max_iters = 15000
            for it in range(1, max_iters + 1):
                if time.time() - start_time > budget_s * 0.95:
                    break

                corr = np.correlate(cur, 1.0 - cur, mode="full")
                idx = int(np.argmax(corr))
                shift = idx - (n - 1)
                k = int(abs(shift))

                grad = np.ones_like(cur)
                if k > 0:
                    grad[: n - k] -= cur[k:]
                    grad[k:]      -= cur[: n - k]
                else:
                    grad = 1.0 - 2.0 * cur

                step = 0.45 / math.sqrt(it)
                cur = project_onto_sum(cur - step * grad, target)

                cur_score = compute_c5(cur)
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = cur.copy()
                if cur_score < best_score:
                    best_score = cur_score

                if best_score <= 0.3808:
                    break

            # ----- binary round‑up + combinatorial refinement ----------
            # produce a binary vector with exactly target ones
            bin_vec = np.zeros(n, dtype=float)
            idx_desc = np.argsort(-cur)                     # largest entries first
            bin_vec[idx_desc[: int(target)]] = 1.0
            bin_score = compute_c5(bin_vec)
            if bin_score < best_score:
                best_score = bin_score
                best_h = bin_vec.copy()

            # greedy swap improvement
            bin_vec, bin_score = greedy_binary_optimize(bin_vec, target, budget_s, start_time)
            if bin_score < best_score:
                best_score = bin_score
                best_h = bin_vec.copy()

            # stochastic swap improvement (extra safety)
            bin_vec, bin_score = binary_random_swap(bin_vec, target, budget_s, start_time)
            if bin_score < best_score:
                best_score = bin_score
                best_h = bin_vec.copy()

            if best_score <= 0.3808:
                break

        if best_score <= 0.3808:
            break

    # fallback – should never happen, but keeps the contract safe
    if best_h is None:
        n = 42
        best_h = np.full(n, 0.5)
        best_h = project_onto_sum(best_h, n / 2.0)
        best_score = compute_c5(best_h)

    return best_h.tolist(), float(best_score), len(best_h)
