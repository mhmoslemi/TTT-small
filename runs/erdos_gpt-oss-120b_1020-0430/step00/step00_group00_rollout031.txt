def gradient_for_shift(h, k):
    n = len(h)
    grad = np.zeros_like(h)
    if k == 0:
        grad = 2 * h
    else:
        grad[k:] += h[:-k]  # for i >= k, term h_{i - k}
        grad[:-k] += h[k:]  # for i + k < n, term h_{i + k}
    return grad

def project_onto_box_and_sum(x, target):
    # clip to [0,1]
    y = np.clip(x, 0.0, 1.0)
    # iterate until sum matches target within tolerance
    for _ in range(10):
        s = np.sum(y)
        if abs(s - target) < 1e-9:
            break
        # compute adjustment needed per element
        delta = (s - target) / np.sum((y > 0) & (y < 1))
        # if no free variables, break
        mask = (y > 0) & (y < 1)
        if not np.any(mask):
            # distribute across all non-zero? fallback
            y = y - (s - target) / len(y)
            y = np.clip(y, 0.0, 1.0)
            continue
        y[mask] -= delta
        # clip again
        y = np.clip(y, 0.0, 1.0)
    # final clipping
    y = np.clip(y, 0.0, 1.0)
    # ensure sum exactly target by adjusting the first free element
    s = np.sum(y)
    if abs(s - target) > 1e-8:
        # adjust small amount on any non-bounded entry
        for i in range(len(y)):
            if y[i] > 0 and y[i] < 1:
                y[i] += target - s
                y[i] = min(max(y[i], 0.0), 1.0)
                break
    return y

            cap = y[i]
            sub = min(cap, -diff)
            y[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    # final safety clip
    return np.clip(y, 0.0, 1.0)


def _upscale(vec: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation from the old grid to a finer grid."""
    n_old = len(vec)
    xs_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xs_new, xs_old, vec)


def _score(h: np.ndarray) -> float:
    """C5 value for a candidate h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _grad_step(h: np.ndarray, lr: float) -> tuple[np.ndarray, float]:
    """One projected‑gradient step aimed at reducing the maximal overlap."""
    n = len(h)
    # find the shift that realizes the current worst overlap
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # shift in range [-(n-1), n-1]

    grad = np.zeros_like(h)

    if shift >= 0:
        i = np.arange(0, n - shift)
        j = i + shift
        grad[i] += (1.0 - h[j])
        grad[j] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        j = i - s
        grad[i] += (1.0 - h[j])
        grad[j] -= h[i]

    # move opposite to the gradient (we want to *decrease* the worst value)
    h_new = h - lr * grad
    h_proj = _project_bounds_sum(h_new, n / 2.0)
    return h_proj, _score(h_proj)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h that (approximately) minimises the Erdős C5 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    budget = float(budget_s)

    # -------------------------------------------------------------------------
    # 1.  Base resolution (the size of the supplied parent if any)
    # -------------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_low = np.array(initial_h_values, dtype=np.float64)
        n_low = h_low.size
    else:
        n_low = 45
        h_low = np.full(n_low, 0.5, dtype=np.float64)

    h_low = _project_bounds_sum(h_low, n_low / 2.0)

    # -------------------------------------------------------------------------
    # 2.  Choose a working resolution.
    #    180 points (four times the base) gives a good trade‑off between speed
    #    and expressive power.
    # -------------------------------------------------------------------------
    n = max(180, (n_low // 5) * 5)          # make sure we have at least 180 points
    factor = max(1, n // n_low)
    n = n_low * factor                     # keep it an integer multiple
    h = _upscale(h_low, n)
    h = _project_bounds_sum(h, n / 2.0)

    best_h = h.copy()
    best_score = _score(best_h)

    # -------------------------------------------------------------------------
    # 3.  Simple projected‑gradient descent with occasional random restarts.
    # -------------------------------------------------------------------------
    lr = 0.08                     # initial step size
    it_without_improve = 0
    max_without = 500
    it = 0

    while time.time() - start < 0.92 * budget:
        h, cur_score = _grad_step(h, lr)
        it += 1

        if cur_score + 1e-9 < best_score:
            best_score = cur_score
            best_h = h.copy()
            it_without_improve = 0
        else:
            it_without_improve += 1

        # slowly decay the learning rate
        lr = max(lr * 0.9995, 5e-4)

        # if we are stuck, jitter around the best solution
        if it_without_improve >= max_without:
            jitter = rng.uniform(-0.06, 0.06, size=best_h.shape)
            h = _project_bounds_sum(best_h + jitter, n / 2.0)
            it_without_improve = 0
            lr = 0.08

    # -------------------------------------------------------------------------
    # 4.  Return the best candidate found.
    # -------------------------------------------------------------------------
    return best_h, best_score, n
