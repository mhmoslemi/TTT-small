def move_generic(h):

import time, math, random
import numpy as np

def _c5_and_corr(h: np.ndarray):
    """Return C5 = 1 - min autocorrelation and the full autocorrelation array."""
    n = h.size
    dx = 2.0 / n
    # autocorrelation of h with itself (linear, not circular)
    corr = np.correlate(h, h, mode="full") * dx
    return 1.0 - corr.min(), corr


def run(seed: int = 42, budget_s: float = 1100, **_):
    """
    Return a step function h (as a list), its C5 bound and the number of points.
    The optimizer starts from the parent construction (if any) and improves it.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # initialise h ---------------------------------------------------------
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        # modest up‑sampling if the parent sample is small
        if h0.size < 120:
            h0 = np.repeat(h0, 2)
        n = h0.size
    else:
        # fallback: binary random set of measure 1
        n = 140
        h0 = np.zeros(n, dtype=np.float64)
        ones = np.random.choice(n, n // 2, replace=False)
        h0[ones] = 1.0

    # enforce exact integral = 1  (sum = n/2)
    tgt = n / 2.0
    diff = tgt - h0.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if 0.0 <= h0[i] + diff <= 1.0:
                h0[i] += diff
                break

    cur_h = h0.copy()
    cur_c5, cur_corr = _c5_and_corr(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ----------------------------------------------------------------------
    # optimisation parameters -----------------------------------------------
    # ----------------------------------------------------------------------
    start = time.time()
    T0 = 0.005                     # initial temperature for SA
    max_time = budget_s
    it = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= max_time - 0.1:

        it += 1

        # ------------------------------------------------------------------
        # 1)   try a move that directly improves the currently worst shift
        # ------------------------------------------------------------------
        min_idx = cur_corr.argmin()
        ell = min_idx - (n - 1)                     # shift index (may be negative)

        overlap = ind[mask]
        outside = ind[~mask]

        # choose i from overlap where we can increase h[i] and partner is >0
        partner = overlap + ell
        cand_i = overlap[(cur_h[overlap] < 1.0 - 1e-12) & (cur_h[partner] > 1e-12)]
        move_made = False

        if cand_i.size and outside.size:

            gains = (1.0 - cur_h[cand_i]) * cur_h[cand_i + ell]
