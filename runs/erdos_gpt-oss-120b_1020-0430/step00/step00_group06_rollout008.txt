import time, math, random
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for step function h."""
    n = h.shape[0]
    # correlation of h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)   # dx = 2/n

def _neighbor_sum(h: np.ndarray, shift: int) -> np.ndarray:
    """For each index i, return h[i+shift] + h[i-shift] with zero padding."""
    n = h.shape[0]
    if shift >= 0:
        forward = np.concatenate([h[shift:], np.zeros(shift)])
        backward = np.concatenate([np.zeros(shift), h[:n - shift]])
    else:
        s = -shift
        forward = np.concatenate([np.zeros(s), h[:n - s]])
        backward = np.concatenate([h[s:], np.zeros(s)])
    return forward + backward
