def project_onto_box_simplex(v, S):
    # v: np array
    # Clip to [0,1] after subtraction of lambda.
    # Binary search for lambda in [low, high] such that sum(min(1, max(0, v - lam))) = S.
    low = np.min(v) - 1  # lower bound maybe v - 1 being negative; but we can use low = min(v) - 1.
    high = np.max(v)
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0, 1)
        cur_sum = x.sum()
        if cur_sum > S:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low+high)/2.0, 0, 1)
    # Ensure exactly sum = S within tolerance, maybe adjust few entries.
    # Could apply final correction: diff = S - sum(x); distribute to non-saturated entries.
    # But the binary search should be enough.

import time

# Helper: projection onto the capped simplex {0≤x_i≤1, sum x_i = S}
def project_vector(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto the set 0≤x≤1 and sum(x)=S."""
    # binary search for lambda such that sum(clip(v-lambda,0,1)) = S
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(60):
        lam = (low + high) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        cur = x.sum()
        if cur > S:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = S - x.sum()
    if abs(diff) > 1e-8:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            np.clip(x, 0.0, 1.0, out=x)
    return x


# ------------------------------------------------------------
# Compute C5 value and the subgradient (scaled by dx)
def compute_corr_and_grad(h: np.ndarray):
    """Return (C5 value, gradient of C5) for the current h."""
    n = h.size
    dx = 2.0 / n
    # correlation h with (1-h) in full mode
    d = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(d))
    c_val = d[idx_max] * dx                     # current C5
    shift = idx_max - (n - 1)                    # integer shift producing the max

    grad = np.zeros_like(h)

    if shift >= 0:
        limit = n - shift
        # contribution where h appears as the first factor
        grad[:limit] += (1.0 - h[shift:])        # = 1 - h_{i+shift}
        # contribution where h appears inside (1‑h) of another term
        grad[shift:] -= h[:limit]                # = -h_{i-shift}
    else:  # shift < 0
        t = -shift
        limit = n - t
        # first‑factor part (indices i ≥ t)
        grad[t:] += (1.0 - h[:limit])            # = 1 - h_{i-t}
        # second‑factor part (indices i ≤ n‑t‑1)
        grad[:limit] -= h[t:]                    # = -h_{i+t}

    grad *= dx                                   # scale by dx
    return c_val, grad

def local_optimize(h0: np.ndarray, max_iter: int = 2000, base_eta: float = 0.5):
    """Return a locally improved h and its C5 value."""
    n = h0.size
    S = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val, _ = compute_corr_and_grad(h)

    for t in range(1, max_iter + 1):
        val, grad = compute_corr_and_grad(h)
        eta = base_eta / math.sqrt(t)
        h = h - eta * grad
        h = project_vector(h, S)

        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val

            h0 = project_vector(h0, n / 2.0)

            # optimise locally
            h_opt, val_opt = local_optimize(h0, max_iter=max_iter, base_eta=0.5)

            if val_opt < best_val:
                best_val = val_opt
                best_h = h_opt.copy()
                if best_val <= 0.3808:
                    # We have already beaten the target; keep going a bit to confirm
                    pass

    # ------------------------------------------------------------------
    # final evaluation (use the official evaluator if available)
    try:
        final_val = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_val = float(compute_corr_and_grad(best_h)[0])

    return (best_h, final_val, int(best_h.size))
