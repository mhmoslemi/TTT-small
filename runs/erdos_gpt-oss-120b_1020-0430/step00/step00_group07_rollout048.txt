import time
import numpy as np

def _evaluate(h):
    """Return C5 for step function h (dx is inferred from length)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _make_triangle(n):
    """Triangular step function on [0,2] (peak = 1, base = 2)."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx          # mid‑points
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    # enforce exact integral = 1
    target_sum = n / 2.0                    # because Σh·dx = 1  ⇒ Σh = n/2
    cur = h.sum()
    if not np.isclose(cur, target_sum):
        h *= target_sum / cur
    return h


def _project_feasible(h, target_sum):
    """Clip to [0,1] and adjust to hit target_sum (preserving bounds)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if np.isclose(cur, target_sum):
        return h
    # simple proportional scaling for interior values, then fix any overshoot
    scale = target_sum / cur
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    # small residual fix by adding/subtracting on a random free index
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # try to distribute diff without breaking bounds
        # (guaranteed to succeed because target_sum lies in [0,n])
        idxs = np.arange(h.size)
        np.random.shuffle(idxs)
        for i in idxs:
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a small C5 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------
    # Determine problem size and an initial feasible vector
    # ------------------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        h0 = np.asarray(init, dtype=np.float64).ravel()
        n = h0.size
        target_sum = n / 2.0
        h0 = _project_feasible(h0, target_sum)
    else:
        n = 80                     # default size when no parent is provided
        h0 = None
        target_sum = n / 2.0

    # ------------------------------------------------------------
    # Build a deterministic starting point (triangular shape)
    # ------------------------------------------------------------
    tri = _make_triangle(n)
    best_h = tri.copy()
    best_score = _evaluate(best_h)

    # if a parent vector exists, possibly replace the start
    if h0 is not None:
        parent_score = _evaluate(h0)
        if parent_score < best_score:
            best_h = h0.copy()
            best_score = parent_score

    # ------------------------------------------------------------
    # Simulated annealing / hill‑climbing
    # ------------------------------------------------------------
    cur_h = best_h.copy()
    cur_score = best_score

    total_budget = max(budget_s - 0.5, 1e-3)   # reserve a little head‑room
    temp_start = 0.02
    temp_end = 5e-6
    max_no_improve = 200_000
    no_improve = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.1:
            break

        # temperature decreasing exponentially
        ratio = min(elapsed / total_budget, 1.0)
        temp = temp_start * ((temp_end / temp_start) ** ratio)

        # propose a transfer of mass between two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]
        if hi <= 0.0 or hj >= 1.0:
            continue
        max_delta = min(hi, 1.0 - hj)
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_score = _evaluate(new_h)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_h = new_h
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
        else:
            no_improve += 1

        # occasional restart from the current best to escape stagnation
        if no_improve >= max_no_improve:
            cur_h = best_h.copy()
            cur_score = best_score
            no_improve = 0

    # final clean‑up: enforce feasibility one last time
    best_h = _project_feasible(best_h, target_sum)

    final_score = _evaluate(best_h)
    return best_h, float(final_score), int(n)
