def correlation_fft(a,b):
    n = len(a)
    b_rev = b[::-1]
    N = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b_rev, N)
    conv = np.fft.irfft(A * B, N)[:2*n-1]
    max_idx = np.argmax(conv)
    max_val = conv[max_idx]
    lag = max_idx - (n - 1)
    return max_val, lag

def gradient_for_lag(h, lag):
    n = len(h)
    # compute shifted forward and backward
    forward = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            forward[:n - lag] = h[lag:]
        # else forward stays zero
    else:
        # lag negative: shift left
        l = -lag
        if l < n:
            forward[l:] = h[:n - l]
    backward = np.zeros_like(h)
    # backward is shift by -lag (i.e., forward with opposite sign)
    lag2 = -lag
    if lag2 >= 0:
        if lag2 < n:
            backward[:n - lag2] = h[lag2:]
    else:
        l = -lag2
        if l < n:
            backward[l:] = h[:n - l]
    # gradient = 1 - forward - backward
    return 1.0 - forward - backward

def correlation_fft(h, one_minus_h):
    # compute full linear correlation of h and (1-h)
    n = len(h)
    a = h
    b = one_minus_h
    # reverse b
    b_rev = b[::-1]
    N = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b_rev, N)
    conv = np.fft.irfft(A * B, N)[:2*n-1]
    max_idx = np.argmax(conv)
    max_val = conv[max_idx]
    # lag = max_idx - (n-1)
    return max_val, max_idx - (n - 1)

def gradient_for_lag(h, lag):
    n = len(h)
    # forward shift (i+lag towards i)
    forward = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            forward[:n - lag] = h[lag:]
    else:
        l = -lag
        if l < n:
            forward[l:] = h[:n - l]
    backward = np.zeros_like(h)
    lag2 = -lag
    if lag2 >= 0:
        if lag2 < n:
            backward[:n - lag2] = h[lag2:]
    else:
        l = -lag2
        if l < n:
            backward[l:] = h[:n - l]
    # gradient of f_lag w.r.t h_i = 1 - (h_{i+lag} + h_{i-lag}) where undefined terms are 0.
    return 1.0 - forward - backward

def correlation_fft(h, one_minus_h):
    """Return (max correlation value, lag) for h with (1‑h) using FFT."""
    n = h.shape[0]
    b_rev = one_minus_h[::-1]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(b_rev, N)
    conv = np.fft.irfft(A * B, N)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx - (n - 1)


def evaluate_c5(h, dx):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    max_val, _ = correlation_fft(h, 1 - h)
    return max_val * dx


def gradient_for_lag(h, lag):
    """Gradient of the active lag’s overlap term (direction of increase)."""
    n = h.shape[0]
    forward = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            forward[: n - lag] = h[lag:]
    else:
        l = -lag
        if l < n:
            forward[l:] = h[: n - l]

    backward = np.zeros_like(h)
    lag2 = -lag
    if lag2 >= 0:
        if lag2 < n:
            backward[: n - lag2] = h[lag2:]
    else:
        l = -lag2
        if l < n:
            backward[l:] = h[: n - l]

    return 1.0 - forward - backward


def project_to_box_sum(v, S):

        grad = gradient_for_lag(h, lag)

        pos = np.where((grad > 1e-8) & (h > 1e-8))[0]
        neg = np.where((grad < -1e-8) & (h < 1 - 1e-8))[0]

        if pos.size and neg.size:
            i = int(np.random.choice(pos))
            j = int(np.random.choice(neg))
            delta = min(step, h[i], 1.0 - h[j])
            if delta > 0:
                delta *= np.random.rand()
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                val_new = evaluate_c5(h_new, dx)
                if val_new < best_val - 1e-12:
                    h = h_new
                    best_h = h_new.copy()
                    best_val = val_new
                    stagnation = 0
                    step = min(step * 1.03, 0.5)
                else:
                    if np.random.rand() < 0.001:
                        h = h_new
                    stagnation += 1
                    step *= 0.999
        else:
            # random pairwise move when gradient gives no direction
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            if np.random.rand() < 0.5:
                max_delta = min(step, h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * np.random.rand()
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
            else:
                max_delta = min(step, h[j], 1.0 - h[i])
                if max_delta <= 0:
                    continue
                delta = max_delta * np.random.rand()
                h_new = h.copy()
                h_new[j] -= delta
                h_new[i] += delta
            val_new = evaluate_c5(h_new, dx)
            if val_new < best_val - 1e-12:
                h = h_new
                best_h = h_new.copy()
                best_val = val_new
                stagnation = 0
                step = min(step * 1.03, 0.5)
            else:
                if np.random.rand() < 0.001:
                    h = h_new
                stagnation += 1
                step *= 0.999

        # occasional perturbed restart if stuck too long
        if stagnation > 4000:
            noise = np.random.randn(n) * (step * 0.5)
            h = project_to_box_sum(np.clip(best_h + noise, 0.0, 1.0), n / 2.0)
            stagnation = 0
            step = max(step, 0.01)

        # slowly shrink step size
        if stagnation % 2000 == 0 and stagnation > 0:
            step = max(step * 0.9, 0.001)

    return best_h, best_val


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    total_budget = budget_s - 0.5          # tiny safety margin

    def time_left():
        return total_budget - (time.time() - start_time)

    # possible parent construction
    try:
        init_h = initial_h_values          # supplied by the harness, may be None
    except NameError:
        init_h = None

    best_h_global = None
    best_val_global = float('inf')
    best_n_global = None

    # grid sizes to explore (small → moderate)
    cand_n = [48, 64, 80, 100, 120, 150, 200]
    if init_h is not None:
        cand_n.append(len(init_h))
    cand_n = sorted(set(cand_n))

    for n in cand_n:
        if time_left() < 8.0:
            break
        dx = 2.0 / n

        # how many independent starts we attempt for this grid size
        restarts = 5
        per_restart = max(1.0, time_left() / (len(cand_n) - cand_n.index(n) + 1) / restarts)

        for r in range(restarts):
            if time_left() < 5.0:
                break

            # ---- choose an initial feasible vector ----
            if init_h is not None and len(init_h) == n and r == 0:
                h0 = project_to_box_sum(np.clip(init_h, 0.0, 1.0), n / 2.0)
                h_start = h0
            elif init_h is not None and r == 0:
                h_up = upscale_h(init_h, n)
                h_start = project_to_box_sum(np.clip(h_up, 0.0, 1.0), n / 2.0)
            elif r == 1:
                # cosine shape: 0.5 + 0.5*cos(pi*x)
                x = (np.arange(n) + 0.5) * dx
                h_start = 0.5 + 0.5 * np.cos(np.pi * x)
                h_start = np.clip(h_start, 0.0, 1.0)
                h_start = project_to_box_sum(h_start, n / 2.0)
            else:
                # random feasible start via projection
                v = np.random.rand(n)
                h_start = project_to_box_sum(v, n / 2.0)

            # ---- local optimisation ----
            deadline = time.time() + min(per_restart, time_left())
            h_opt, val_opt = opt_h_local(h_start, dx, deadline)

            if val_opt < best_val_global:
                best_val_global = val_opt
                best_h_global = h_opt.copy()
                best_n_global = n

            # early stop if we already beat the target record
            if best_val_global <= 0.3808:
                # continue a little longer to maybe improve further
                pass

    # Fallback in the extremely unlikely case we never found a point
    if best_h_global is None:
        n = 100
        dx = 2.0 / n
        h = np.zeros(n)
        h[: n // 2] = 1.0
        best_h_global = h
        best_n_global = n
        best_val_global = evaluate_c5(h, dx)

    # Compute the bound to report (use the harness's evaluator when available)
    try:
        c5_bound = evaluate_erdos_solution(best_h_global)
    except Exception:
        c5_bound = evaluate_c5(best_h_global, 2.0 / best_n_global)

    return (best_h_global, c5_bound, best_n_global)
