    # binary search lambda:

    # because sum at lo is n, at hi is 0

        if s > target:

def compute_c5(h):
    n = len(h)

def anneal(h, rng, time_limit):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    best_h = h.copy()
    best_val = compute_c5(best_h)

    # current state
    cur_h = h.copy()
    cur_val = best_val

    # parameters
    start_time = time.time()
    # total steps guess: compute steps = int( (time_limit - (start_time - t0)) / avg_time_per_iter )
    # but we don't know; we just loop until time exceeds limit.
    T0 = best_val  # initial temperature.
    # define schedule: linear decay over steps: T = T0 * (1 - t / total_steps)
    # we compute current iteration count.

    it = 0
    while True:
        t_now = time.time()
        if t_now - start_time > time_limit:
            break
        # choose indices
        i, j = rng.choice(n, size=2, replace=False)
        # compute allowed delta range
        # max delta positive from j -> i
        delta_max_pos = min(1.0 - cur_h[i], cur_h[j])
        delta_max_neg = min(cur_h[i], 1.0 - cur_h[j])
        if delta_max_pos + delta_max_neg == 0:
            continue
        # choose delta uniformly in [-delta_max_neg, delta_max_pos]
        delta = rng.uniform(-delta_max_neg, delta_max_pos)
        # update copy
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # compute new objective
        new_val = compute_c5(new_h)
        # acceptance
        # linear schedule for temperature
        T = T0 * (1.0 - it / 50000)  # adjust max iterations
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
        it += 1
    return best_h, best_val

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x : sum(x)=target, 0<=x<=1} using binary search on λ."""
    lo = v.min() - 1.0          # all entries become 1 -> sum=n
    hi = v.max()                # all entries become 0 -> sum=0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _compute_c5(h: np.ndarray) -> float:
    """Return the C₅ value for a step function represented by vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return _project_to_simplex_box(v, n / 2.0)


def _anneal(initial_h: np.ndarray, rng: np.random.Generator,
            time_limit: float) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing on the C₅ objective."""
    n = initial_h.shape[0]
    cur_h = initial_h.copy()
    cur_val = _compute_c5(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    # linear cooling from initial value down to ~0
    while True:
        now = time.time()
        if now - start > time_limit:
            break
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)

        # feasible delta range so that both stay in [0,1]
        max_pos = min(1.0 - cur_h[i], cur_h[j])   # move from j to i
        max_neg = min(cur_h[i], 1.0 - cur_h[j])   # move from i to j
        if max_pos + max_neg == 0.0:
            continue
        delta = rng.uniform(-max_neg, max_pos)

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_val = _compute_c5(new_h)

        # temperature decreasing linearly
        frac = (now - start) / time_limit
        T = cur_val * (1.0 - frac) + 1e-12

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
        it += 1
    return best_h, best_val


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h on [0,2] that gives a low C₅ bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values – list of values of h on a uniform grid,
        c5_bound – the achieved C₅,
        n_points – number of grid points.
    """
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # ------------------------------------------------------------------
    # 1. Determine starting resolution & initial vector
    # ------------------------------------------------------------------
    try:
        init_vec = np.asarray(initial_h_values, dtype=float)
        n0 = init_vec.shape[0]
        init_vec = _project_to_simplex_box(init_vec, n0 / 2.0)
        have_initial = True
    except Exception:          # initial_h_values not defined
        n0 = 80
        init_vec = _random_feasible(n0, rng)
        have_initial = False

    best_h = init_vec.copy()
    best_val = _compute_c5(best_h)
    best_n = n0

    # ------------------------------------------------------------------
    # 2. Main optimisation loop – possible refinements on finer grids
    # ------------------------------------------------------------------
    # We allow at most three refinements (doubling the resolution each time)
    max_refinements = 3
    current_n = n0
    current_h = init_vec.copy()
    elapsed = time.time() - start_total
    for refinement in range(max_refinements + 1):
        # allocate a slice of the remaining budget for this stage
        remaining = budget_s - (time.time() - start_total)
        if remaining <= 0:
            break
        # budget per stage: half of the remaining on the first
        stage_budget = remaining * 0.5 if refinement == 0 else remaining * 0.3

        # run several independent annealing attempts (restarts)
        restarts = 4
        per_restart = max(0.0, stage_budget / restarts)
        for _ in range(restarts):
            cur_start = time.time()
            # start from a fresh feasible point
            if have_initial and refinement == 0 and _ == 0:
                h0 = current_h.copy()
            else:
                h0 = _random_feasible(current_n, rng)
            h_opt, val_opt = _anneal(h0, rng, per_restart * 0.9)
            if val_opt < best_val:
                best_h, best_val, best_n = h_opt.copy(), val_opt, current_n

            # small safety margin
            if time.time() - start_total > budget_s * 0.99:
                break

        # ------------------------------------------------------------------
        # 3. Refine grid if time permits
        # ------------------------------------------------------------------
        elapsed = time.time() - start_total
        if refinement < max_refinements and elapsed < budget_s * 0.9:
            # double resolution via linear interpolation
            new_n = current_n * 2
            old_x = np.linspace(0.0, 2.0, current_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
            interp_h = np.interp(new_x, old_x, best_h)
            interp_h = _project_to_simplex_box(interp_h, new_n / 2.0)

            # prepare for next refinement iteration
            current_n = new_n
            current_h = interp_h
            have_initial = True
            # keep current best as starting point for next refinement
            best_h = interp_h.copy()
            best_val = _compute_c5(best_h)
            best_n = new_n

        if time.time() - start_total > budget_s * 0.99:
            break

    # ------------------------------------------------------------------
    # 4. Final projection (tiny numeric drift) and return
    # ------------------------------------------------------------------
    best_h = _project_to_simplex_box(best_h, best_n / 2.0)
    final_val = _compute_c5(best_h)

    return best_h.tolist(), float(final_val), int(best_n)
