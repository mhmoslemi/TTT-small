import numpy as np, time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_capped_simplex(v, target):
    # clip to [0,1] and project onto simplex sum=target
    # using binary search on lambda
    low = -1.0
    high = 1.0
    for _ in range(50):
        mid = (low+high)/2
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            high = mid
        else:
            low = mid
    w = np.clip(v + (low+high)/2, 0.0, 1.0)
    # adjust slight diff
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        # distribute diff uniformly among non-bound entries
        free = (w > 1e-12) & (w < 1 - 1e-12)
        if np.any(free):
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def generate_cosine(n, freq):
    x = np.linspace(0, 2, n, endpoint=False) # points from 0 inclusive
    h = 0.5 + 0.5 * np.cos(np.pi * freq * x)  # cos(pi*freq*x) period = 2/freq
    # This automatically satisfies average 0.5? Since integral of cos over integer number of periods is zero.
    # sum h = 0.5 * n + 0.5 * sum cos = 0.5*n (if sum cos =0). For integer freq, sum cos may be close to zero (if n multiple of period).
    # We'll project to ensure sum exactly n/2.

import numpy as np
import time

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def project_capped_simplex(v, target):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum(x)=target}.
    Uses a simple binary search for the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 1e-12) & (w < 1.0 - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


def compute_c5(h):
    """Return the C5 value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def random_binary(n):
    """Binary vector with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=np.float64)
    h[:half] = 1.0
    np.random.shuffle(h)
    return h


def cosine_shape(n, freq):
    """h(x) = 0.5 + 0.5*cos(pi*freq*x) on [0,2] projected onto constraints."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.5 * np.cos(np.pi * freq * x)
    return project_capped_simplex(h, n / 2.0)


def linear_decreasing(n):
    """h(x) = max(0, 1 - x/2) sampled on midpoints, then projected."""
    dx = 2.0 / n
    # mid‑point sampling
    x = (np.arange(n) + 0.5) * dx
    h = np.clip(1.0 - x / 2.0, 0.0, 1.0)
    return project_capped_simplex(h, n / 2.0)


def hill_climb(start_h, max_iter, temp_start=0.02, temp_end=1e-4):
    """
    Simple simulated‑annealing hill‑climber that always respects the sum constraint.
    Moves consist of transferring a small amount of mass from one index to another.
    """
    h = start_h.astype(np.float64).copy()
    n = h.shape[0]
    dx = 2.0 / n
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = np.max(corr_full) * dx
    best_h = h.copy()
    best_c5 = cur_c5

    for it in range(max_iter):
        # linear temperature schedule
        t = temp_start + (temp_end - temp_start) * (it / max_iter)

        idx_max = np.argmax(corr_full)
        shift = idx_max - (n - 1)

        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            s = -shift
            i_range = np.arange(s, n)
            j_range = i_range - s

        # product that actually contributes to the worst overlap
        prod = h[i_range] * (1.0 - h[j_range])
        local_idx = np.argmax(prod)
        i = i_range[local_idx]
        j = j_range[local_idx]

        # feasible transfer from i to j
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max > 1e-12:
            delta = delta_max * (0.05 + 0.95 * np.random.rand())
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # fallback random transfer
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 0:
                continue
            delta = delta_max * (0.05 + 0.95 * np.random.rand())
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta

        # evaluate the move
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx

        # Metropolis acceptance
        if new_c5 <= cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / max(t, 1e-12)):
            h = h_new
            corr_full = corr_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

        # early exit if perfect (unlikely)
        if best_c5 <= 0.1:
            break

    return best_h, best_c5
