import time
import numpy as np
import cvxpy as cp

def _c5_of_h(h):
    """Return the C5 value for a step vector h."""
    N = h.shape[0]
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def _solve_lp(N):
    """Solve the linear program that minimizes the max overlap for size N."""
    # decision variables
    h = cp.Variable(N)                     # values of the step function
    W = cp.Variable((N, N))                # auxiliary variables for products h_i*h_j
    t = cp.Variable()                      # upper bound on each overlap sum

    constraints = []

    # h bounds and total mass
    constraints += [h >= 0, h <= 1, cp.sum(h) == N / 2, t >= 0]

    # diagonal of W is irrelevant for the objective; fix to 0 to avoid free vars
    for i in range(N):
        constraints.append(W[i, i] == 0)

    # McCormick envelope for the product h_i * h_j  (i < j)
    for i in range(N):
        hi = h[i]
        for j in range(i + 1, N):
            hij = W[i, j]
            hj = h[j]
            constraints += [
                hij <= hi,
                hij <= hj,
                hij >= hi + hj - 1,
                hij >= 0,
            ]

    # upper‑bound constraints for every positive lag (forward and backward)
    for lag in range(1, N):
        forward = 0
        backward = 0
        for i in range(N - lag):
            forward   += h[i] - W[i, i + lag]       # Σ (h_i - h_i h_{i+lag})
            backward  += h[i + lag] - W[i, i + lag] # Σ (h_{i+lag} - h_i h_{i+lag})
        constraints.append(forward  <= t)
        constraints.append(backward <= t)

    # minimise the worst‑case overlap
    prob = cp.Problem(cp.Minimize(t), constraints)

    # solve – ECOS is a reliable open‑source conic solver
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    # If the solver did not succeed, fall back to a simple feasible point
    if prob.status not in ("optimal", "optimal_inaccurate"):
        # uniform 0.5 solution (feasible for any N)
        h_val = np.full(N, 0.5)
        t_val = _c5_of_h(h_val) * N / 2.0  # back‑compute the t that would give this bound
        return h_val, t_val
    return h.value, t.value


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step vector h, its C5 bound, and its length."""
    start_time = time.time()

    # Determine problem size – use the supplied construction if present,
    # otherwise fall back to a modest even size.
    try:
        N = len(initial_h_values)                # type: ignore[name-defined]
    except Exception:
        N = 60

    # Solve the linear programme (takes only a few seconds even for N≈70)
    h_opt, t_opt = _solve_lp(N)

    # Numerical cleanup – enforce bounds and exact mass
    h_opt = np.clip(h_opt, 0.0, 1.0)
    target_mass = N / 2.0
    mass_err = target_mass - np.sum(h_opt)
    if abs(mass_err) > 1e-8:
        # distribute the tiny error to a non‑boundary index
        idx = np.argmin(np.abs(h_opt - 0.5))
        h_opt[idx] = np.clip(h_opt[idx] + mass_err, 0.0, 1.0)

    # Final C5 value (scaled by dx)
    c5_val = _c5_of_h(h_opt)

    # Safety – if we ran out of time, return whatever we have
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # Trim to the required format and exit early
        return (h_opt.tolist(), float(c5_val), N)

    return (h_opt.tolist(), float(c5_val), N)
