        if shift >= 0:
            length = n - shift
            deriv[:length] = 1.0 - cur_h[shift:]
        else:
            s = -shift
            length = n - s
            deriv[s:] = 1.0 - cur_h[:length]

        # indices that can lose mass (h>0) and actually influence the worst lag
        down_idx = np.where((cur_h > 1e-12) & (deriv > 0))[0]
        # indices that can gain mass (h<1)
        up_idx = np.where(cur_h < 1.0 - 1e-12)[0]

        if down_idx.size == 0 or up_idx.size == 0:
            # fallback: random tiny shuffle
            i_up, i_down = np.random.choice(n, 2, replace=False)
            eps = min(1.0 - cur_h[i_up], cur_h[i_down]) * 0.1
            if eps <= 0:
                continue
        else:
            # most damaging contributor: largest derivative -> decrease here
            i_down = down_idx[np.argmax(deriv[down_idx])]
            # place to add mass where it hurts the least: smallest derivative
            i_up = up_idx[np.argmin(deriv[up_idx])]
            eps = min(cur_h[i_down], 1.0 - cur_h[i_up]) * 0.5
            if eps <= 0:
                # as a safety net, use a tiny random step
                eps = min(cur_h[i_down], 1.0 - cur_h[i_up]) * 0.1
                if eps <= 0:
                    continue

        candidate = cur_h.copy()
        candidate[i_up] += eps
        candidate[i_down] -= eps
        np.clip(candidate, 0.0, 1.0, out=candidate)

        # re‑impose the exact integral (tiny numerical drift can appear)
        diff = target_sum - candidate.sum()
        if abs(diff) > 1e-12:
            # adjust a random free entry
            free = np.where((candidate > 1e-12) & (candidate < 1.0 - 1e-12))[0]
            if free.size == 0:
                free = np.arange(n)
            idx = free[np.random.randint(free.size)]
            candidate[idx] = np.clip(candidate[idx] + diff, 0.0, 1.0)

        cand_c = _evaluate_cmax(candidate)

        # accept only if we improve the current worst value (or use a tiny SA fallback)
        if cand_c + 1e-12 < cur_c:
            cur_h = candidate
            cur_c = cand_c
            no_improve = 0
            if cand_c + 1e-12 < best_c:
                best_h = candidate.copy()
                best_c = cand_c
        else:
            # simple simulated‑annealing style acceptance
            elapsed = time.time() - start
            T = max(0.0, (budget_s - elapsed) / budget_s) * 0.001
            if T > 0.0 and np.random.rand() < np.exp(-(cand_c - cur_c) / T):
                cur_h = candidate
                cur_c = cand_c
            else:
                no_improve += 1

        # occasional random perturbation to escape possible stagnation
        if no_improve >= max_no_improve:
            jitter = np.random.uniform(-0.03, 0.03, size=n)
            candidate = best_h + jitter
            candidate = np.clip(candidate, 0.0, 1.0)
            # enforce exact integral again
            diff = target_sum - candidate.sum()
            if abs(diff) > 1e-12:
                idx = np.random.randint(n)
                candidate[idx] = np.clip(candidate[idx] + diff, 0.0, 1.0)
            cur_h = candidate
            cur_c = _evaluate_cmax(cur_h)
            no_improve = 0

        iter_cnt += 1

    # final verification (use the official helper if present)
    final_c = _evaluate_cmax(best_h)

    return best_h, final_c, best_h.size
