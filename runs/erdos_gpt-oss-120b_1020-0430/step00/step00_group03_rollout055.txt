import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Retrieve or build an initial feasible vector h.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        # ensure exact sum (might be off by tiny epsilon)
        target_sum = n / 2.0
        cur_sum = h0.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            # repair by scaling the whole vector and clipping
            h0 = np.clip(h0 * (target_sum / cur_sum), 0.0, 1.0)
            diff = target_sum - h0.sum()
            if abs(diff) > 1e-12:
                # add the remaining mass to the first non‑saturated entry
                idx = np.where((diff > 0) & (h0 < 1.0 - 1e-12) |
                               (diff < 0) & (h0 > 1e-12))[0]
                if idx.size:
                    h0[idx[0]] += diff
    else:
        # no parent: build a simple binary half‑filled vector
        n = 70
        h0 = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        rng.shuffle(h0)
        h0 = h0.astype(float)

    # ------------------------------------------------------------------
    # Helper: compute C5 for a candidate h.
    # ------------------------------------------------------------------
    dx = 2.0 / n
    target_sum = n / 2.0

    def compute_c5(h):
        # correlation between h and (1-h) with zero padding, then max and scale
        c = np.correlate(h, 1.0 - h, mode='full')
        return np.max(c) * dx

    # ------------------------------------------------------------------
    # Initialise the search.
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_score = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    # Simulated‑annealing parameters
    init_temp = 0.01
    final_temp = 1e-6

    # Counters for stagnation handling
    no_improve = 0
    max_no_improve = 5000   # after this many rejections we perturb from the best

    # Main loop – stop a bit before the budget to be safe.
    while time.time() - start < budget_s * 0.95:
        # temperature schedule (linear in log‑scale)
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.95)
        temp = init_temp * (final_temp / init_temp) ** frac

        # --------------------------------------------------------------
        # Propose a move that preserves the sum exactly: transfer mass
        # between two distinct indices.
        # --------------------------------------------------------------
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = cur_h[i], cur_h[j]

        # allowed positive transfer (i gains, j loses)
        max_up = min(1.0 - hi, hj)
        # allowed negative transfer (i loses, j gains)
        max_down = min(hi, 1.0 - hj)

        if max_up + max_down == 0:
            continue

        delta = rng.uniform(-max_down, max_up)

        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_score = compute_c5(new_h)

        # --------------------------------------------------------------
        # Acceptance test (standard SA)
        # --------------------------------------------------------------
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_h = new_h
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Stagnation recovery: perturb from the best known solution.
        # --------------------------------------------------------------
        if no_improve > max_no_improve:
            cur_h = best_h.copy()
            # random Gaussian noise on a fraction of entries
            frac_noise = 0.2
            k = max(1, int(frac_noise * n))
            idxs = rng.choice(n, size=k, replace=False)
            cur_h[idxs] += rng.normal(scale=0.05, size=k)
            np.clip(cur_h, 0.0, 1.0, out=cur_h)

            # Repair the sum (diff is tiny because noise is small)
            diff = target_sum - cur_h.sum()
            if abs(diff) > 1e-12:
                # Find a coordinate that can absorb the whole diff
                cand = np.where((diff > 0) & (cur_h < 1.0 - 1e-12) |
                                (diff < 0) & (cur_h > 1e-12))[0]
                if cand.size:
                    cur_h[cand[0]] += diff
                    cur_h[cand[0]] = np.clip(cur_h[cand[0]], 0.0, 1.0)

            cur_score = compute_c5(cur_h)
            no_improve = 0

    # ------------------------------------------------------------------
    # Final repair to guarantee feasibility.
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # Adjust a single free coordinate if possible
        idxs = np.where((diff > 0) & (best_h < 1.0 - 1e-12) |
                        (diff < 0) & (best_h > 1e-12))[0]
        if idxs.size:
            best_h[idxs[0]] += diff
            best_h[idxs[0]] = np.clip(best_h[idxs[0]], 0.0, 1.0)

    final_score = compute_c5(best_h)

    return best_h, float(final_score), n
