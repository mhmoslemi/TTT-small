import numpy as np, time

def compute_c5_and_shift(h):
    n = len(h)
    g = 1 - h
    conv = np.correlate(h, g, mode='full')
    idx = np.argmax(conv)
    c = conv[idx] * (2.0 / n)   # dx = 2/n
    shift = idx - (n - 1)   # integer shift
    return c, shift

def project_sum_box(y, target):
    lo = -np.max(y)
    hi = 1 - np.min(y)
    # binary search
    for _ in range(60):
        mid = (lo + hi) / 2
        z = np.clip(y + mid, 0, 1)
        s = z.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo+hi)/2, 0, 1)

def gradient_for_shift(h, shift):
    n = len(h)
    if shift >= 0:
        plus = np.concatenate((h[shift:], np.zeros(shift)))
        minus = np.concatenate((np.zeros(shift), h[:n-shift]))
    else:
        k = -shift
        plus = np.concatenate((np.zeros(k), h[:n-k]))
        minus = np.concatenate((h[k:], np.zeros(k)))
    return 1.0 - plus - minus

def subgradient_descent(h0, target_sum, max_iter, time_limit, best_c):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    best_h = h.copy()
    best_val = best_c
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        c, shift = compute_c5_and_shift(h)
        # update best
        if c < best_val:
            best_val = c
            best_h = h.copy()
        # compute gradient at worst shift
        g = gradient_for_shift(h, shift)
        # project gradient to zero-sum
        g = g - g.mean()
        # step schedule - simple diminishing step
        step = 0.5 / np.sqrt(it + 1)   # could tune
        y = h - step * g
        # project onto feasibility
        h = project_sum_box(y, target_sum)
    # final evaluate
    c_final, _ = compute_c5_and_shift(best_h)
    return best_h, c_final

def project_sum_box(y, target):
    lo = -np.max(y)
    hi = 1 - np.min(y)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        z = np.clip(y + mid, 0, 1)
        s = z.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + (lo + hi) / 2.0, 0, 1)
    # Due to rounding errors, adjust
    diff = target - z.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly across non-bound variables
        free = (z > 0 + 1e-12) & (z < 1 - 1e-12)
        if free.any():
            z[free] += diff / free.sum()
            z = np.clip(z, 0, 1)
    return z

def subgradient_descent(h0, target_sum, max_iter, time_limit, global_best):
    h = h0.copy()
    n = len(h)
    best_local_h = h.copy()
    best_local_c = np.inf
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        c, shift = compute_c5_and_shift(h)
        # If this c <= global_best already maybe break? Not necessary.
        # Update best local
        if c < best_local_c:
            best_local_c = c
            best_local_h = h.copy()
        # If c is already <= global_best - maybe no need for further improvements but continue
        # compute gradient
        g = gradient_for_shift(h, shift)
        # subtract mean
        g = g - np.mean(g)
        step = 0.4 / np.sqrt(it + 1)  # maybe 0.4
        y = h - step * g
        # project back
        h = project_sum_box(y, target_sum)
    return best_local_h, best_local_c

def compute_c5_and_shift(h):
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(conv)
    cmax = conv[idx] * (2.0 / n)
    shift = idx - (n - 1)
    return cmax, shift
```

def gradient_for_shift(h, shift):
    n = len(h)
    if shift >= 0:
        plus = np.concatenate((h[shift:], np.zeros(shift)))
        minus = np.concatenate((np.zeros(shift), h[:n - shift]))
    else:
        k = -shift
        plus = np.concatenate((np.zeros(k), h[:n - k]))
        minus = np.concatenate((h[k:], np.zeros(k)))
    return 1.0 - plus - minus

import numpy as np
import time

# ---------- helper functions -----------------------------------------

def compute_c5_and_shift(h):
    """Return (C5 value, integer shift) for a step function h."""
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(conv))
    cmax = float(conv[idx]) * (2.0 / n)      # dx = 2/n
    shift = idx - (n - 1)                    # signed shift in number of steps
    return cmax, shift


def gradient_for_shift(h, shift):
    """Gradient of C(k) w.r.t. h for the given integer shift."""
    n = h.shape[0]
    if shift >= 0:
        plus = np.concatenate((h[shift:], np.zeros(shift, dtype=h.dtype)))
        minus = np.concatenate((np.zeros(shift, dtype=h.dtype), h[:n - shift]))
    else:
        k = -shift
        plus = np.concatenate((np.zeros(k, dtype=h.dtype), h[:n - k]))
        minus = np.concatenate((h[k:], np.zeros(k, dtype=h.dtype)))
    return 1.0 - plus - minus


def project_sum_box(y, target):
    """
    Euclidean projection of y onto the set {z | 0<=z_i<=1, sum(z)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(y)          # ensures y+lo <= 0 -> all clipped to 0
    hi = 1.0 - np.min(y)     # ensures y+hi >= 1 -> all clipped to 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(y + mid, 0.0, 1.0)
        s = z.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for floating‑point drift
    diff = target - z.sum()
    if abs(diff) > 1e-12:
        free = (z > 1e-12) & (z < 1.0 - 1e-12)
        if free.any():
            z[free] += diff / free.sum()
            z = np.clip(z, 0.0, 1.0)
    return z


def subgradient_descent(h0, target_sum, max_iter, time_limit, global_best):
    """
    Simple sub‑gradient descent on the max‑overlap objective.
    Returns the best vector seen during the run and its C5 value.
    """
    h = h0.copy()
    best_h = h.copy()
    best_val = np.inf
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        c, shift = compute_c5_and_shift(h)

        # keep the best local solution
        if c < best_val:
            best_val = c
            best_h = h.copy()

        # gradient of the current worst shift
        g = gradient_for_shift(h, shift)
        # make it tangent to the hyperplane sum(h)=target_sum
        g -= g.mean()
        step = 0.4 / np.sqrt(it + 1.0)   # diminishing step size
        y = h - step * g
        h = project_sum_box(y, target_sum)

    return best_h, best_val


def triangular_initial(n):
    """Triangular shape that satisfies the integral constraint."""
    dx = 2.0 / n
    xs = np.arange(n) * dx
    h = np.where(xs <= 1.0, xs, 2.0 - xs)
    return h


def local_refine(h, target_sum, time_limit):
    """
    Tiny random walk that preserves the sum exactly.
    Moves a small amount of mass between two coordinates.
    """
    best_h = h.copy()
    best_c, _ = compute_c5_and_shift(best_h)
    n = h.shape[0]
    start = time.time()
    while time.time() - start < time_limit:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]

        # feasible delta so that hi+δ∈[0,1] and hj-δ∈[0,1]
        delta_low = max(-hi, hj - 1.0)
        delta_high = min(1.0 - hi, hj)
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)

        new_h = best_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        c_new, _ = compute_c5_and_shift(new_h)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h
    return best_h, best_c


# ---------- main entry point -----------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) for the Erdős C5 problem.
    """
    np.random.seed(seed)
    start_time = time.time()

    # grid size – a moderate size gives a good trade‑off between quality and speed
    n = 500                     # number of samples on [0,2]
    target_sum = n / 2.0        # ∫ h = 1  →  Σ h = n/2

    # collect a few sensible initial vectors
    init_vectors = []

    tri = triangular_initial(n)
    tri = project_sum_box(tri, target_sum)
    init_vectors.append(tri)

    try:
        init = initial_h_values          # provided by the harness, if any
        if init is not None and len(init) > 0:
            old_n = len(init)
            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(new_x, old_x, init)
            interp = np.clip(interp, 0.0, 1.0)
            interp = project_sum_box(interp, target_sum)
            init_vectors.append(interp)
    except NameError:
        pass   # no supplied initial vector

    best_h = None
    best_c = np.inf

    # main optimisation loop – repeated restarts until time runs out
    while time.time() - start_time < budget_s - 3.0:
        # pick a start point (70% random, 30% from stored inits)
        if init_vectors and np.random.rand() < 0.3:
            h0 = init_vectors[np.random.randint(len(init_vectors))].copy()
        else:
            h0 = project_sum_box(np.random.rand(n), target_sum)

        # allocate a chunk of the remaining budget for this descent
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5.0:
            break
        descent_time = min(22.0, remaining * 0.85)

        h_opt, c_opt = subgradient_descent(
            h0,
            target_sum,
            max_iter=20000,
            time_limit=descent_time,
            global_best=best_c,
        )
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            # early stop if we have already beaten the target
            if best_c < 0.3808:
                break

    # if nothing was found (unlikely), fall back to a random feasible vector
    if best_h is None:
        best_h = project_sum_box(np.random.rand(n), target_sum)
        best_c, _ = compute_c5_and_shift(best_h)

    # final tiny refinement using remaining time
    time_left = budget_s - (time.time() - start_time)
    if time_left > 0.5:
        best_h, best_c = local_refine(best_h, target_sum, time_left)

    # one last projection to be safe, then recompute the exact bound
    best_h = project_sum_box(best_h, target_sum)
    final_c, _ = compute_c5_and_shift(best_h)

    return best_h, final_c, n
