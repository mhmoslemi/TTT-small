def simulated_annealing_n(n, time_budget, rng):
    start = time.time()
    # initial binary
    h = np.zeros(n, dtype=np.float64)
    h[:n//2] = 1.0
    rng.shuffle(h)
    # compute initial cost
    current_val = max_overlap(h)
    best_h = h.copy()
    best_val = current_val
    # total iterations counter
    iteration = 0
    while time.time() - start < time_budget:
        iteration += 1
        # probability schedule: simple linear cooling
        t_frac = 1.0 - (time.time() - start) / time_budget
        T = max(1e-4, 0.1 * t_frac)  # adjust scale
        # propose swap
        i1 = rng.randint(n)
        while h[i1] < 0.5:
            i1 = rng.randint(n)
        i0 = rng.randint(n)
        while h[i0] > 0.5:
            i0 = rng.randint(n)

        h[i1], h[i0] = 0.0, 1.0

                best_h = h.copy()

            # revert
            h[i1], h[i0] = 1.0, 0.0

def compute_subgradient(h, lag):
    n = h.size
    if lag >= 0:
        # term1: 1 - h_{i+lag} for i where i+lag < n
        term1 = np.concatenate([1 - h[lag:], np.zeros(lag)])
        # term2: h_{i - lag} for i where i - lag >= 0
        term2 = np.concatenate([np.zeros(lag), h[:n - lag]])

        term1 = np.concatenate([np.zeros(l), 1 - h[:n - l]])
        term2 = np.concatenate([h[l:], np.zeros(l)])
    # gradient = dx * (term1 - term2)
    return (2.0 / n) * (term1 - term2)

def compute_corr_vector(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(np.r_[h, np.zeros(L - n)])
    G = np.fft.rfft(np.r_[1 - h, np.zeros(L - n)])
    corr = np.fft.ifft(H * np.conj(G), n=L)[:2*n-1].real
    return corr
```

import time, math
import numpy as np

def _proj_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project vector x onto the box [0,1]^n with sum == target."""
    lo, hi = -1.0, 1.0                     # enough because 0<=x<=1 initially
    for _ in range(30):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def _max_overlap(h: np.ndarray) -> float:
    """C5 value for a discrete step function h (max over all shifts)."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(np.concatenate([h, np.zeros(L - n)]))
    G = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(L - n)]))
    corr = np.fft.ifft(H * np.conj(G), n=L)[: 2 * n - 1].real
    return corr.max() * (2.0 / n)

def _corr_vector(h: np.ndarray) -> np.ndarray:
    """Full correlation vector (without the final dx scaling)."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(np.concatenate([h, np.zeros(L - n)]))
    G = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(L - n)]))
    corr = np.fft.ifft(H * np.conj(G), n=L)[: 2 * n - 1].real
    return corr

def _grad_of_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of the max correlation term for a given lag."""
    n = h.size
    dx = 2.0 / n
    if lag >= 0:
        term1 = np.concatenate([1.0 - h[lag:], np.zeros(lag)])
        term2 = np.concatenate([np.zeros(lag), h[: n - lag]])
    else:
        l = -lag
        term1 = np.concatenate([np.zeros(l), 1.0 - h[: n - l]])
        term2 = np.concatenate([h[l:], np.zeros(l)])
    return dx * (term1 - term2)

def _subgradient_descent(
    h0: np.ndarray, target_sum: float, budget: float, rng: np.random.RandomState
):
    """Projected sub‑gradient minimisation of max C5."""
    h = h0.copy()
    best_h = h.copy()
    best_val = _max_overlap(h)
    start = time.time()
    it = 0
    while time.time() - start < budget:
        it += 1
        corr = _corr_vector(h)
        idx = int(corr.argmax())
        lag = idx - (h.size - 1)
        g = _grad_of_lag(h, lag)
        step = 0.5 / math.sqrt(it)          # simple diminishing step
        h_new = h - step * g
        h_new = _proj_box_sum(h_new, target_sum)
        val = _max_overlap(h_new)
        # keep the best ever seen
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            h = h_new
            continue
        # occasional stochastic acceptance to escape plateaus
        if rng.rand() < 0.001:
            h = h_new
    return best_h, best_val
