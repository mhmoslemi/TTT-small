def autocorr(h):
    n = len(h)
    # compute linear convolution h * rev(h) (full)
    # Using FFT: pad to length m = 2**next_pow_two(2*n-1)
    m = 1 << ((2*n - 1).bit_length())

    # Extract positions n-1 .. n-1 + (2n-1) = full correlation, but only first (2n-1)
    # Actually conv result length = 2n-1 (starts at index 0)
    return corr[:2*n-1]  # still float

def autocorr_counts(h):
    n = h.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    fft_h = np.fft.rfft(h, n=m)
    conv = np.fft.irfft(fft_h * np.conj(fft_h), n=m)
    return conv[:2*n-1]

    dx = 2.0 / n
    conv = autocorr_counts(h)
    min_overlap = conv.min() * dx

def simulated_annealing(h_start, time_limit, start_time):
    h = h_start.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    n = len(h)
    target_sum = n/2
    # SA parameters
    max_iter = 200000
    T0 = 0.02 # maybe appropriate
    for it in range(max_iter):
        # check time
        if time.time() - start_time >= time_limit:
            break
        # temperature decay
        t_progress = it / max_iter
        T = T0 * (1 - t_progress) + 0.0001  # min temp
        # propose move
        i, j = np.random.randint(0, n, size=2)
        # ensure i != j
        while i == j:
            j = np.random.randint(0, n)
        # compute max delta possible
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:

            if max_delta <= 0:

            # swap roles
            i, j = j, i
        # propose delta: random fraction up to max_delta
        delta = max_delta * (np.random.rand() * 0.5)  # up to half of max

        h_new[i] -= delta
        h_new[j] += delta

            current_c5 = c5_new
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        # occasional random restart maybe when stuck
        # not building.
    return best_h, best_c5

import numpy as np
import time
import random

# ------------------------------------------------------------
# Helper functions (top level, no closures)

def autocorr_counts(h: np.ndarray) -> np.ndarray:
    """Linear autocorrelation (counts) of a non‑negative vector h.
    Returns a length 2*len(h)-1 array where index i corresponds to shift i-(len(h)-1)."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())        # next power of two
    fft_h = np.fft.rfft(h, n=m)
    conv = np.fft.irfft(fft_h * np.conj(fft_h), n=m)
    return conv[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    """C₅ value for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = autocorr_counts(h)
    # numerical noise may produce tiny negative numbers
    min_overlap = np.maximum(conv.min(), 0.0) * dx
    return 1.0 - min_overlap


def project_sum_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the hyperplane sum(x)=target_sum with 0≤x≤1.
    The projection is performed by adding a constant and clipping."""
    lo, hi = -2.0, 2.0
    for _ in range(40):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_move(h: np.ndarray) -> np.ndarray:
    """Perform a single feasible mass‑transfer move on h."""
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    # try to move from i to j
    max_delta = min(h[i], 1.0 - h[j])
    direction = 1  # 1 means i→j
    if max_delta <= 0.0:
        # try opposite direction
        max_delta = min(h[j], 1.0 - h[i])
        direction = -1
        if max_delta <= 0.0:
            return None
        i, j = j, i
    delta = max_delta * (np.random.rand() * 0.5)  # not too aggressive
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def simulated_annealing(start_h: np.ndarray, time_limit: float, start_time: float):
    """Simple SA that respects the constraints."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    h = start_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    max_iter = 200000
    T0 = 0.02

    for it in range(max_iter):
        if time.time() - start_time >= time_limit:
            break
        # temperature linear decay
        T = T0 * (1.0 - it / max_iter) + 1e-5

        cand = random_move(h)
        if cand is None:
            continue
        cand_val = compute_c5(cand)
        dE = cand_val - cur_val
        if dE < 0 or np.random.rand() < np.exp(-dE / T):
            h = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
    return best_h, best_val


def improve_h_binary(h0: np.ndarray, time_limit: float, start_time: float):
    """Greedy hill‑climbing for binary vectors using swaps that raise the
    minimal overlap."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)

    while time.time() - start_time < time_limit:
        conv = autocorr_counts(h)
        overlap = conv * (2.0 / n)
        min_ov = overlap.min()
        cur_val = 1.0 - min_ov

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val < 0.3808:
                break

        # indices achieving the worst shift
        worst_idxs = np.where(overlap == min_ov)[0]
        idx = worst_idxs[np.random.randint(len(worst_idxs))]
        k = idx - (n - 1)

        # collect candidate swap positions for this shift
        if k >= 0:
            inc = np.where((h[:n - k] == 0) & (h[k:] == 1))[0]
            dec = np.where((h[:n - k] == 1) & (h[k:] == 0))[0]
        else:
            kk = -k
            inc = np.where((h[kk:] == 0) & (h[:n - kk] == 1))[0] + kk
            dec = np.where((h[kk:] == 1) & (h[:n - kk] == 0))[0] + kk

        made_move = False
        if inc.size and dec.size:
            i = inc[np.random.randint(inc.size)]
            j = dec[np.random.randint(dec.size)]
            h[i], h[j] = h[j], h[i]          # swap
            new_val = compute_c5(h)
            if new_val <= cur_val:
                made_move = True
            else:
                h[i], h[j] = h[j], h[i]      # revert
        if not made_move:

            i = zeros[np.random.randint(zeros.size)]

            new_val = compute_c5(h)
            if new_val > cur_val:            # keep only if not worse
                h[i], h[j] = h[j], h[i]

    return best_h, best_val
