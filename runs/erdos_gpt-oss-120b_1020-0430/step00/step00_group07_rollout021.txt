import time
import math
import numpy as np

def _proj_box_simplex(v, target_sum):
    """Project v onto the set {x: 0<=x_i<=1, sum x = target_sum}."""
    # binary search for the Lagrange multiplier lambda
    lo, hi = -1e6, 1e6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _random_feasible(n, rng):
    """Generate a random feasible vector (values in [0,1], sum = n/2)."""
    # start from uniform and project
    v = rng.rand(n)
    return _proj_box_simplex(v, n / 2.0)

def _evaluate(h):
    """Evaluate C5 for a given step function."""
    # evaluate_erdos_solution is provided by the harness
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C5 value."""
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise first candidate
    # ------------------------------------------------------------------
    try:
        # use the supplied parent if it exists
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = h0.size
        # ensure feasibility (numeric noise may happen)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _proj_box_simplex(h0, n / 2.0)
    except Exception:          # no parent supplied
        n = 80                 # modest size – fast to evaluate
        h0 = _random_feasible(n, rng)

    # store the best solution seen so far
    best_h = h0.copy()
    best_val = _evaluate(best_h)

    # ------------------------------------------------------------------
    # Simulated annealing / hill‑climbing on a continuous vector
    # ------------------------------------------------------------------
    # SA parameters
    T_start = 1.0
    T_end = 1e-4
    max_iters = 300_000          # generous upper bound; loop stops earlier by time
    it = 0
    cur_h = h0.copy()
    cur_val = best_val

    # pre‑compute target sum
    target_sum = n / 2.0

    while True:
        # time check – leave a small safety margin
        if time.time() - start_time > budget_s - 0.4:
            break

        # temperature schedule (exponential cooling)
        T = T_start * ((T_end / T_start) ** (it / max_iters))

        # propose a move: transfer a random amount from a donor (h_j) to a receiver (h_i)
        i, j = rng.choice(n, 2, replace=False)
        hi, hj = cur_h[i], cur_h[j]

        # compute maximal transferable amount while staying in [0,1]
        max_inc = min(1.0 - hi, hj)
        if max_inc <= 0.0:
            # nothing can be moved between these two indices – try another pair
            it += 1
            continue

        # random step size, biased towards larger moves when temperature is high
        delta = max_inc * rng.rand()
        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        # evaluate
        new_val = _evaluate(new_h)
        delta_val = new_val - cur_val

        # Metropolis acceptance
        if delta_val < 0.0 or rng.rand() < math.exp(-delta_val / max(T, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()

        # occasional restart from the current best with a small random perturbation
        if it % 50000 == 0 and it > 0:
            cur_h = best_h.copy()
            cur_val = best_val
            # apply a few random swaps to shake the state
            for _ in range(5):

                cur_h[b] = hb - delta
            cur_val = _evaluate(cur_h)

    # final safety projection (numeric drift)
    best_h = _proj_box_simplex(best_h, target_sum)
    best_val = _evaluate(best_h)

    # return the result in the required form
    return (best_h, float(best_val), int(n))
