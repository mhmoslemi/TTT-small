import time
import numpy as np

# Optional SciPy import – fall back to simple local search if unavailable
try:
    from scipy.optimize import minimize
    _has_scipy = True
except Exception:  # pragma: no cover
    _has_scipy = False


def _c5_value(h, dx):
    """Exact C5 evaluation as described in the specification."""
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _project_to_feasible(h, target_sum):
    """Clamp to [0,1] and enforce exact sum by a simple water‑filling step."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # distribute the remaining mass over free variables (not at a bound)
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            inc = diff / free.sum()
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
        else:
            # no free variable – adjust a single component safely
            idx = np.argmax(np.abs(diff) * (h < 1.0) + np.abs(diff) * (h > 0.0))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        diff = target_sum - h.sum()
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)

    # --- initialise ---------------------------------------------------------
    try:  # use provided parent construction if present
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
    except Exception:  # pragma: no cover
        n = 80
        h0 = np.zeros(n, dtype=np.float64)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)

    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = _project_to_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _c5_value(best_h, dx)

    start = time.time()
    end_time = start + budget_s * 0.98  # reserve a tiny margin

    # ----------------------------------------------------------------------
    # Simple stochastic hill climbing (quickly explores the neighbourhood)
    # ----------------------------------------------------------------------
    temperature = 0.01
    min_temp = 1e-6
    cooling = 0.9995

    while time.time() < end_time * 0.3:
        i, j = rng.randint(0, n, size=2)
        if i == j:
            continue
        # try moving mass from i to j
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 1e-12:
            # opposite direction possible?
            max_transfer = min(best_h[j], 1.0 - best_h[i])
            if max_transfer <= 1e-12:
                continue
            i, j = j, i
        # step size shrinks with time
        ratio = (time.time() - start) / budget_s
        step_factor = max(0.5 * (1.0 - ratio), 0.01)
        eps = max_transfer * rng.rand() * step_factor
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.clip(cand, 0.0, 1.0)  # safety
        cand = _project_to_feasible(cand, target_sum)
        cand_c5 = _c5_value(cand, dx)

        if cand_c5 < best_c5 or rng.rand() < np.exp(-(cand_c5 - best_c5) / temperature):
            best_h = cand
            best_c5 = cand_c5

        temperature = max(min_temp, temperature * cooling)

    # ----------------------------------------------------------------------
    # Optional SciPy optimisation (smooth max approximation)
    # ----------------------------------------------------------------------
    if _has_scipy and time.time() < end_time:
        def smooth_max_obj(vec):
            corr = np.correlate(vec, 1.0 - vec, mode="full") * dx
            alpha = 30.0
            m = np.max(corr)
            # log‑sum‑exp for a smooth max
            return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

        cons = {"type": "eq", "fun": lambda v: np.sum(v) - target_sum}
        bounds = [(0.0, 1.0)] * n

        # a few random restarts within the remaining budget
        while time.time() < end_time:
            # start from a perturbed version of the current best
            start_vec = best_h + rng.randn(n) * 0.05
            start_vec = _project_to_feasible(start_vec, target_sum)

            opts = {"maxiter": 800, "ftol": 1e-9, "disp": False}
            try:
                res = minimize(
                    smooth_max_obj,
                    start_vec,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options=opts,
                )
            except Exception:  # pragma: no cover
                break

            if not res.success:
                # if the optimiser failed, fall back to the current best
                break

            cand = _project_to_feasible(res.x, target_sum)
            cand_c5 = _c5_value(cand, dx)

            if cand_c5 < best_c5:
                best_h, best_c5 = cand, cand_c5

    # final safety projection (tiny numerical drift possible)
    best_h = _project_to_feasible(best_h, target_sum)
    best_c5 = _c5_value(best_h, dx)

    return best_h.tolist(), best_c5, n
