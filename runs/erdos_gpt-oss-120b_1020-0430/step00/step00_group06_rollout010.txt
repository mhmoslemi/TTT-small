import numpy as np
import time

def compute_max_corr(h, dx):
    """Return the max overlap integral for a step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def get_mismatch_indices(h, k):
    """
    For a given integer shift k, return two index arrays:
    - bad_idx : positions i where h[i]=1 and h[i+k]=0  (contribute to the max)
    - opp_idx : positions i where h[i]=0 and h[i+k]=1  (can be swapped with bad)
    h is binary (0/1) and length n.
    """
    n = h.shape[0]
    if k >= 0:
        main = h[: n - k]
        shifted = h[k:]
        offset = 0
    else:
        kk = -k
        main = h[kk:]
        shifted = h[: n - kk]
        offset = kk
    idx_main = np.arange(offset, offset + main.shape[0])
    bad = (main > 0.5) & (shifted < 0.5)
    opp = (main < 0.5) & (shifted > 0.5)
    return idx_main[bad], idx_main[opp]
