def project_to_feasible(h, target_sum):
    # Clip to box and project onto sum hyperplane.
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum via water-filling:
    # Use iterative approach.
    tol = 1e-12
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < tol:
            break
        # identify free indices (0<h[i]<1)
        free = (h > 0.0 + 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            # all at bounds, cannot adjust
            break
        delta = diff / nfree
        # mark where update would violate bounds
        # For positive delta, max increase allowed = 1 - h[i]; for negative, allowed = -h[i]
        # We'll compute incremental clipped update.
        # For each free index, compute new value.
        update = np.where(free, delta, 0.0)
        # ensure we don't exceed bounds
        h_new = h + update
        # If any exceed bounds, adjust those to bounds and compute remaining diff; but to keep simple, clip after.
        h = np.clip(h_new, 0.0, 1.0)
        # Continue loop.
    # After loop, enforce sum precisely by adjusting first non-bound element.
    diff = target_sum - h.sum()
    if abs(diff) > 1e-8:
        # Find index not at bound to add diff.
        for i in range(len(h)):
            if 0 < h[i] < 1:
                add = min(max(diff, -h[i]), 1 - h[i])
                h[i] += add
                break
    return h

import numpy as np, time, math

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    conv = np.convolve(h, (1 - h)[::-1], mode='full')
    return conv.max() * dx

def gradient_shift(h, s):
    N = len(h)
    g = np.ones(N)
    if s >= 0:
        if s != 0:
            g[:N - s] -= h[s:]
            g[s:] -= h[:N - s]
        else:
            g -= 2 * h
    else:
        s_abs = -s
        g[s_abs:] -= h[:N - s_abs]
        g[:N - s_abs] -= h[s_abs:]
    return g

def project_to_feasible(h, target):
    # clip
    h = np.clip(h, 0.0, 1.0)
    # iterative waterfilling
    for _ in range(1000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            break
        delta = diff / nfree
        # limit delta to avoid crossing bounds
        # compute for each free index potential new value
        # apply delta, then clip
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final adjust for tiny diff
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        # adjust first non-bound element
        for i in range(len(h)):
            if 0 < h[i] < 1:
                add = min(max(diff, -h[i]), 1 - h[i])
                h[i] += add
                break
    return h

def upscale(initial, N):
    N0 = len(initial)
    x0 = (np.arange(N0) + 0.5) * (2.0 / N0)
    x = (np.arange(N) + 0.5) * (2.0 / N)
    h_interp = np.interp(x, x0, initial, left=initial[0], right=initial[-1])
    h_interp = np.clip(h_interp, 0.0, 1.0)
    target = N / 2.0
    # scale to meet sum roughly
    s = h_interp.sum()
    if s > 0:
        h = h_interp * (target / s)
    else:
        h = np.full(N, target / N)
    h = project_to_feasible(h, target)
    return h

    N = len(h_start)

    h = h_start.copy()
    best_h = h.copy()

# ---------- helper functions -------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear (full) correlation h * (1-h) reversed
    conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
    return conv.max() * dx


def gradient_shift(h: np.ndarray, s: int) -> np.ndarray:
    """
    Sub‑gradient of the correlation term for a given shift s.
    s > 0  ->  A_s = Σ h[i] (1-h[i+s])
    s < 0  ->  A_s = Σ h[i] (1-h[i+s])  (i+s is smaller index)
    """
    n = h.shape[0]
    g = np.ones(n)
    if s >= 0:
        if s:
            g[: n - s] -= h[s:]          # -h[i+s] part
            g[s:] -= h[: n - s]          # -h[i-s] part (i-s = i+s with i shifted)
        else:                              # s == 0
            g -= 2.0 * h
    else:
        s = -s
        g[s:] -= h[: n - s]               # -h[i+s] (i+s smaller)
        g[: n - s] -= h[s:]               # -h[i-s]
    return g


def project_to_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the box [0,1]^n while enforcing Σ v_i = target_sum.
    Simple water‑filling style iterative projection.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(200):
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    diff = target_sum - v.sum()
    if abs(diff) > 1e-9:
        for i in range(v.size):
            if 0.0 < v[i] < 1.0:
                add = max(min(diff, 1.0 - v[i]), -v[i])
                v[i] += add
                break
    return v


def upscale(initial: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function to a higher resolution."""
    old_n = initial.shape[0]
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    hi = np.interp(x_new, x_old, initial, left=initial[0], right=initial[-1])
    hi = np.clip(hi, 0.0, 1.0)
    target = new_n / 2.0
    # simple scaling then final projection
    s = hi.sum()
    if s > 0:
        hi = hi * (target / s)
    else:
        hi = np.full(new_n, target / new_n)
    hi = project_to_feasible(hi, target)
    return hi


def solve_once(start_h: np.ndarray, time_limit: float):
    """Projected sub‑gradient descent on the max‑correlation objective."""
    n = start_h.shape[0]
    target = n / 2.0
    h = start_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)

    step = 0.02
    t0 = time.time()
    while time.time() - t0 < time_limit:
        # current correlation
        conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
        cur_val = conv.max()
        idxs = np.where(conv >= cur_val - 1e-12)[0]
        shifts = n - 1 - idxs                       # shift = N-1-index
        grad = np.zeros_like(h)
        for sft in shifts:
            grad += gradient_shift(h, sft)

        direction = -grad
        improved = False
        # try a few step multipliers
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = h + step * factor * direction
            cand = project_to_feasible(cand, target)
            cand_val = compute_c5(cand)
            if cand_val < cur_val - 1e-12:
                h = cand
                cur_val = cand_val
                step = step * min(1.5, 1.0 / factor)
                improved = True
                break
        if not improved:
            step *= 0.5

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
    return best_h, best_val


# ---------- main entry point -------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős C5 step function.
    Returns (h_values, c5_bound, n_points)
    """
    start_time = time.time()
    np.random.seed(seed)

    # obtain the parent construction if present
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        # fallback: simple uniform feasible vector of length 45
        init_h = np.full(45, 0.5)

    n0 = init_h.size
    # target resolution – a moderate up‑scale of the supplied 45‑point solution
    n = max(90, min(240, n0 * 4))          # usually 180 for n0==45
    target_sum = n / 2.0

    # ----- initialise -------------------------------------------------------
    h0 = upscale(init_h, n)
    h0 = project_to_feasible(h0, target_sum)
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ----- time budget handling --------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 0.3          # small safety margin
    if remaining <= 0:
        return best_h, float(best_val), n

    slice_time = 5.0                               # time per optimisation try
    while time.time() - start_time < start_time + remaining - 0.1:
        # random restart with modest probability
        if np.random.rand() < 0.2:
            h_start = np.random.rand(n)
            h_start = project_to_feasible(h_start, target_sum)
        else:
            h_start = best_h.copy()

        # allocate time for this run
        time_left = (start_time + remaining) - time.time()
        t_lim = min(slice_time, time_left)
        h_candidate, val_candidate = solve_once(h_start, t_lim)

        if val_candidate < best_val:
            best_val = val_candidate
            best_h = h_candidate.copy()
            # early exit if we already beat the target record comfortably
            if best_val < 0.3805:
                break

    best_val = compute_c5(best_h)
    return best_h, float(best_val), best_h.shape[0]
