def project_onto_simplex(v, target):
    lo = -1e6 # lower bound for λ (so all entries become 0)
    hi = 1e6 # upper bound for λ (so all entries become 1)
    for _ in range(60):
        mid = (lo+hi)/2
        y = np.clip(v + mid, 0, 1)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0, 1)

    max_idx= np.argmax(corr)
    cur_val = corr[max_idx] * dx

def project_onto_capped_simplex(v, target):
    # returns proj onto [0,1] and sum target
    # Use binary search for lambda

    hi = 1.0 - np.min(v) + 1.0

        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # ensure sum close to target within tolerance; adjust last element might cause small error
    diff = target - proj.sum()
    if abs(diff) > 1e-8:

        mask = (proj > 1e-12) & (proj < 1 - 1e-12)

            # else adjust first element
            proj[0] += diff
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def subgradient_descent(h, target, max_iters, time_limit, start_time):

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper functions (top level, no closures or lambdas)
# ------------------------------------------------------------

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto the hyperplane sum(v)=target with box 0<=v<=1.
    Uses a simple binary search for the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0          # ensures v+lo <= 0
    hi = 1.0 - np.min(v) + 1.0     # ensures v+hi >= 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction to hit the target exactly (numerical safety)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj[0] = np.clip(proj[0] + diff, 0.0, 1.0)
    return proj


def _neighbor_sum_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Return neighbor_sum_i = h_{i-lag}+h_{i+lag} (zero outside bounds)."""
    n = h.size
    if lag == 0:
        return 2.0 * h
    out = np.zeros_like(h)
    if lag > 0:
        out[lag:] += h[:-lag]          # i-lag term
        out[: n - lag] += h[lag:]      # i+lag term
    else:  # lag < 0
        shift = -lag
        out[: n - shift] += h[shift:]  # i-lag == i+shift
        out[shift:] += h[: n - shift]  # i+lag == i-shift
    return out


def _max_overlap_and_lag(h: np.ndarray) -> tuple:
    """Return (max_overlap, lag) for the current h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    max_val = corr[idx] * dx
    lag = idx - (n - 1)
    return max_val, lag


def _subgradient_optimize(h0: np.ndarray,
                          target: float,
                          deadline: float) -> tuple:
    """
    Subgradient descent using the max‑overlap (plus a few top rivals) as
    objective. Returns the best vector and its max‑overlap encountered.
    """
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    it = 0
    K = 5                    # number of top lags to blend
    while time.time() < deadline:
        G = np.correlate(h, 1.0 - h, mode='full')
        cur_val = G.max() * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # choose the K largest entries (they correspond to the worst overlaps)
        K_use = min(K, G.size)
        top_idx = np.argpartition(G, -K_use)[-K_use:]
        direction = np.zeros_like(h)
        for idx in top_idx:
            lag = int(idx - (n - 1))
            neigh = _neighbor_sum_for_lag(h, lag)
            direction += (neigh - 1.0)        # -gradient of G(lag)
        direction /= K_use
        # simple diminishing step size
        step = 0.5 / math.sqrt(it + 1.0)
        h = h + step * direction
        h = _project_capped_simplex(h, target)
        it += 1
        # guard against excessive iteration count
        if it > 40000:
            break
    return best_h, best_val

    """
    Returns (h_values, c5_bound, n_points).
    A fast subgradient‑based search with random restarts is employed.
    """
    np.random.seed(seed)

    # Determine problem size and an initial feasible vector
    try:
        init = np.asarray(initial_h_values, dtype=float)  # type: ignore[name-defined]
        n = init.size

        best_val = np.max(np.correlate(h_best, 1.0 - h_best, mode='full')) * dx
    else:

        best_val = np.max(np.correlate(h_best, 1.0 - h_best, mode='full')) * dx

    start = time.time()

    # Outer loop: random restarts until time runs out
    while time.time() < deadline:
