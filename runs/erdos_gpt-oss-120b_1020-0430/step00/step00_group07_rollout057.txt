    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _eval(h):

def _proj(v, target):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)*0.5, 0.0, 1.0)

def _random_feasible(rng, n, target):
    v = rng.random(n)
    return _proj(v, target)

def _slsqp_optimize(start, target, max_iter=200):
    n = len(start)
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = {'type':'eq', 'fun': lambda x: np.sum(x) - target}
    res = minimize(_eval, start, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'ftol':1e-9, 'disp':False})
    if not res.success:
        # still return result
        return res.x, _eval(res.x)
    return res.x, res.fun

def _hill_climb(start, target, time_deadline):
    n = len(start)
    cur = start.copy()
    cur_score = _eval(cur)
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < time_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new_score = _eval(new)
        # simple greedy acceptance
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if cur_score < best_score:

                best_score = cur_score

            if new2_score < cur_score:
                cur = new2
                cur_score = new2_score
                if cur_score < best_score:
                    best = cur.copy()
                    best_score = cur_score
    return best, best_score

def _subgradient_opt(start, target, time_deadline):
    n = len(start)
    h = start.copy()
    score = _eval(h)
    best = h.copy()
    best_score = score
    it = 0
    rng = np.random.default_rng()
    while time.time() < time_deadline:
        # compute worst shift using np.correlate
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = np.zeros(n)
        if shift >= 0:
            k = shift
            # i in [0, n-k)
            i1 = np.arange(0, n - k)
            grad[i1] += 1
            grad[i1] -= h[i1 + k]
            i2 = np.arange(k, n)
            grad[i2] -= h[i2 - k]
        else:
            k = -shift
            i1 = np.arange(k, n)          # overlapping indices for sum term
            grad[i1] += 1
            grad[i1] -= h[i1 - k]
            i2 = np.arange(0, n - k)
            grad[i2] -= h[i2 + k]
        # compute step size
        step = 0.05 / (1 + it * 0.001)   # decreasing
        new = h - step * grad
        # projection
        new = np.clip(new, 0.0, 1.0)
        new = _proj(new, target)
        new_score = _eval(new)
        if new_score < score:
            h = new
            score = new_score
            if new_score < best_score:
                best = h.copy()
                best_score = new_score
        else:
            # occasional random move to escape
            if rng.random() < 0.001:
                # apply random pair transfer
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    max_eps = min(h[i], 1.0 - h[j])
                    if max_eps > 0:
                        eps = rng.random() * max_eps
                        tmp = h.copy()
                        tmp[i] -= eps
                        tmp[j] += eps
                        tmp_score = _eval(tmp)
                        if tmp_score < score:
                            h = tmp
                            score = tmp_score
                            if tmp_score < best_score:
                                best = h.copy()
                                best_score = tmp_score
        it += 1
    return best, best_score

import time, math
import numpy as np

# optional SciPy
try:
    from scipy.optimize import minimize
    _has_scipy = True
except Exception:  # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
def _objective(h):
    """fallback C5 evaluator, used if evaluate_erdos_solution not present."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _c5(h):
    """wrapper that prefers the provided evaluator."""
    try:
        return evaluate_erdos_solution(h)   # pre‑imported by the harness
    except Exception:                     # pragma: no cover
        return _objective(h)


# ----------------------------------------------------------------------
def _proj_capped_simplex(v, target):
    """Projection onto {0≤x_i≤1 , Σx_i = target}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _random_feasible(rng, n, target):
    """Draw a random point in the capped simplex."""
    v = rng.random(n)
    return _proj_capped_simplex(v, target)


# ----------------------------------------------------------------------
def _slsqp_optimize(start, target, max_iter=200):
    """SLSQP local refinement (if SciPy available)."""
    n = len(start)
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(_c5, start, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    if not res.success:
        return start, _c5(start)
    return res.x, res.fun


# ----------------------------------------------------------------------
def _subgradient_opt(start, target, deadline):
    """Simple sub‑gradient descent on the max‑overlap objective."""
    n = len(start)
    h = start.copy()
    cur = _c5(h)
    best_h, best = h.copy(), cur
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)          # integer shift corresponding to max
        grad = np.zeros_like(h)

        if shift >= 0:
            k = shift
            i1 = np.arange(0, n - k)
            grad[i1] += 1.0
            grad[i1] -= h[i1 + k]
            i2 = np.arange(k, n)
            grad[i2] -= h[i2 - k]
        else:
            k = -shift
            i1 = np.arange(k, n)
            grad[i1] += 1.0
            grad[i1] -= h[i1 - k]
            i2 = np.arange(0, n - k)
            grad[i2] -= h[i2 + k]

        # diminishing step size
        step = 0.05 / (1.0 + it * 1e-3)
        new = h - step * grad
        new = np.clip(new, 0.0, 1.0)
        new = _proj_capped_simplex(new, target)
        new_val = _c5(new)

        if new_val < cur:
            h, cur = new, new_val
            if cur < best:
                best_h, best = h.copy(), cur
        else:
            # tiny chance to try a random pair move to escape
            if rng.random() < 0.001:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    delta = min(h[i], 1.0 - h[j])
                    if delta > 0.0:
                        eps = rng.random() * delta
                        tmp = h.copy()
                        tmp[i] -= eps
                        tmp[j] += eps
                        tmp_val = _c5(tmp)
                        if tmp_val < cur:
                            h, cur = tmp, tmp_val
                            if cur < best:
                                best_h, best = h.copy(), cur
        it += 1
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size and initial vector
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None
    target = n / 2.0                     # Σh_i = n/2  (dx = 2/n)

    # ensure a feasible starting point
    if h0 is not None:
        h0 = np.clip(h0, 0.0, 1.0)
        if abs(h0.sum() - target) > 1e-9:
            h0 = _proj_capped_simplex(h0, target)
        best_h, best_val = h0.copy(), _c5(h0)
    else:
        best_h = _random_feasible(rng, n, target)
        best_val = _c5(best_h)

    start_time = time.time()
    end_time = start_time + budget_s - 0.5   # tiny safety margin

    # primary loop: alternating random starts + sub‑gradient descent
    while time.time() < end_time:
        # pick a start point
        if rng.random() < 0.6 and best_h is not None:
            start_vec = best_h.copy()
        else:
            start_vec = _random_feasible(rng, n, target)

        # occasionally shuffle to explore different orderings
        if rng.random() < 0.25:
            rng.shuffle(start_vec)

        # ---- SLSQP refinement if SciPy available --------------------
        if _has_scipy and time.time() + 1.0 < end_time:
            opt_h, opt_val = _slsqp_optimize(start_vec, target, max_iter=250)
            if opt_val < best_val:
                best_h, best_val = _proj_capped_simplex(opt_h, target), _c5(opt_h)

        # ---- Sub‑gradient descent for a few seconds -----------------
        slice_deadline = min(end_time, time.time() + 9.0)
        sg_h, sg_val = _subgradient_opt(start_vec, target, slice_deadline)
        if sg_val < best_val:
            best_h, best_val = sg_h, sg_val

    # final projection (just in case) and final evaluation
    best_h = _proj_capped_simplex(best_h, target)
    best_val = _c5(best_h)

    return best_h, float(best_val), n
