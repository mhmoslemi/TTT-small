        # pick i_zero = random index where cur_h ==0
        i_one = np.random.choice(np.flatnonzero(cur_h))
        i_zero = np.random.choice(np.flatnonzero(1-cur_h))
        # perform swap
        cur_h[i_one] = 0
        cur_h[i_zero] = 1
        val = max_mismatch(cur_h)

                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i_one] = 1
            cur_h[i_zero] = 0
        # maybe reduce temperature gradually
