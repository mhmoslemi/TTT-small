def project_to_feasible(v, target_sum):
    # Clip to bounds
    u = np.clip(v, 0, 1)
    # While sum not within tolerance, adjust.
    # Use water-filling: compute deficit = target - sum(u)
    # We need to distribute deficit among elements not at bounds, proportionally.
    # This is equivalent to solving projection onto box and linear equality.
    # We'll implement iterative algorithm:
    remaining = deficit = target_sum - np.sum(u)
    if abs(deficit) < 1e-10:
        return u
    # Determine free positions: those not at bound corresponding to sign of deficit.
    # If deficit > 0, we need to increase some entries (those < 1).
    # Else need to decrease some entries (those > 0).
    # We'll adjust uniformly among free entries until either we meet target or some hit bound, then repeat.
    max_iter = 10
    for _ in range(max_iter):
        if abs(deficit) < 1e-12:
            break
        if deficit > 0:
            free = (u < 1)
            n_free = np.count_nonzero(free)
            if n_free == 0:
                # cannot increase, but due to rounding, break
                break
            inc = deficit / n_free
            # ensure not exceeding bounds
            inc_vec = np.where(free, inc, 0.0)
            u = np.clip(u + inc_vec, 0, 1)
        else:
            free = (u > 0)
            n_free = np.count_nonzero(free)
            if n_free == 0:
                break
            dec = deficit / n_free  # deficit negative
            dec_vec = np.where(free, dec, 0.0)
            u = np.clip(u + dec_vec, 0, 1)
        deficit = target_sum - np.sum(u)
    # Return final
    return u

def project_simplex_box(v, target):
    # Clip into [0,1]
    x = np.clip(v, 0, 1)
    # Compute total excess/deficit
    diff = np.sum(x) - target
    if np.abs(diff) < 1e-12:
        return x
    # We'll adjust values that are not at bound in direction to fix diff.
    # For diff > 0 (sum too high): need to decrease some free variables (x > 0).
    # For diff < 0: need to increase some free variables (x < 1).
    # We'll use iterative water-filling with sorting.
    # Approach: treat free variables as a set; compute adjustment = diff / count.
    # But need to ensure after adjusting they stay within bounds; if any hit bound, remove them and repeat.
    # This typical algorithm.
    indices = np.arange(len(v))
    # We'll maintain active mask.
    if diff > 0:
        # need to decrease sum: start from variables > 0
        active = (x > 0)
        while diff > 1e-12 and np.any(active):
            free_idx = np.where(active)[0]
            count = free_idx.size
            adjust = diff / count
            # limit adjust to not go below 0
            max_decrease = x[free_idx]
            # if adjust > max_decrease for any, we will clamp.
            # Determine which variables would go negative
            over = adjust > max_decrease
            if np.any(over):
                # set those to 0, update diff, remove from active
                x[free_idx[over]] = 0.0
                diff -= np.sum(max_decrease[over])
                active[free_idx[over]] = False
                continue
            # otherwise, safe to subtract adjust from all active
            x[free_idx] -= adjust
            diff = 0.0
            break
    else:
        # diff < 0: need to increase sum: deficit = -diff
        deficit = -diff
        active = (x < 1)
        while deficit > 1e-12 and np.any(active):
            free_idx = np.where(active)[0]
            count = free_idx.size
            adjust = deficit / count
            max_increase = 1 - x[free_idx]
            over = adjust > max_increase
            if np.any(over):
                x[free_idx[over]] = 1.0
                deficit -= np.sum(max_increase[over])
                active[free_idx[over]] = False
                continue
            x[free_idx] += adjust
            deficit = 0.0
            break
    # final correction for rounding
    # ensure exact sum by small adjustment distributed to free variables still not at bound
    total = np.sum(x)
    if abs(total - target) > 1e-9:
        # distribute residual using simple approach
        residual = target - total
        # distribute to elements not at bound until residual is near zero
        free = np.where((x > 0) & (x < 1))[0]
        if free.size > 0:
            per = residual / free.size
            x[free] += per
            # clip final to bounds just in case
            x = np.clip(x, 0, 1)
    return x

                break
            inc = (target - s) / nfree
            # don't exceed the bound
            inc_vec = np.where(free, inc, 0.0)
            x = np.clip(x + inc_vec, 0.0, 1.0)
        else:
            # need to remove mass from positions not at the lower bound
            free = x > 0.0
            nfree = np.count_nonzero(free)
            if nfree == 0:
                break
            dec = (s - target) / nfree
            dec_vec = np.where(free, dec, 0.0)
            x = np.clip(x - dec_vec, 0.0, 1.0)
        s = x.sum()
    # final tiny correction (distribute residual uniformly among interior points)
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        interior = (x > 0.0) & (x < 1.0)
        idx = np.where(interior)[0]
        if idx.size:
            x[idx] += residual / idx.size
            x = np.clip(x, 0.0, 1.0)
    return x


def _c5_value(h, dx):
    """Return the C5 value for the discrete function h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx


def _make_triangular(n):
    """Triangular hat on [0,2] with height 1 and integral exactly 1."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    base = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    target = n / 2.0
    return _project_to_feasible(base, target)


def _make_power_family(n, p):
    """Shape (1-|x-1|)^p, lifted by a constant offset so that the integral is 1."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    base = np.maximum(0.0, 1.0 - np.abs(x - 1.0)) ** p
    target = n / 2.0
    lo, hi = 0.0, 1.0
    for _ in range(30):
        mid = (lo + hi) / 2.0
        h = np.clip(base + mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(base + (lo + hi) / 2.0, 0.0, 1.0)
    return _project_to_feasible(h, target)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # -------------------------------------------------------------
    # Choose a working discretisation size (small enough to be fast,
    # large enough to allow a good shape). 256 points works well.
    # -------------------------------------------------------------
    n = 256
    dx = 2.0 / n
    target_sum = n / 2.0

    # -------------------------------------------------------------
    # Build a modest pool of start vectors.
    # -------------------------------------------------------------
    starts = []

    # 1) If a parent solution is supplied, upsample / interpolate it.
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != n:
            # linear interpolation onto our finer grid
            xs_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(xs_new, xs_old, h0)
        starts.append(_project_to_feasible(h0, target_sum))

    # 2) Pure triangular hat (the classic deterministic construction)
    starts.append(_make_triangular(n))

    # 3) Random feasible vectors (shuffle a half‑filled binary pattern)
    bin_vec = np.zeros(n)
    bin_vec[: n // 2] = 1.0
    rng.shuffle(bin_vec)
    starts.append(_project_to_feasible(bin_vec, target_sum))

    # 4) A few members of the power‑family (p = 1, 1.5, 2, 2.5, 3)
    for p in (1.0, 1.5, 2.0, 2.5, 3.0):
        starts.append(_make_power_family(n, p))

    # -------------------------------------------------------------
    # Evaluate the pool and keep the best initial point.
    # -------------------------------------------------------------
    best_h = None
    best_val = np.inf
    for h in starts:
        val = _c5_value(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # -------------------------------------------------------------
    # Simulated‑annealing style local exploration.
    # -------------------------------------------------------------
    curr_h = best_h.copy()
    curr_val = best_val

    max_iters = 300000                     # generous upper bound
    for it in range(max_iters):
        # time check
        if time.time() - start > budget_s * 0.95:
            break

        # simple linear cooling schedule
        frac = it / max_iters
        temperature = 0.02 * (1.0 - frac) + 1e-5
        max_move = 0.5 * (1.0 - frac)      # step size shrinks over time

        # propose a mass transfer between two coordinates
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = curr_h[i], curr_h[j]
        # amount we can move from i to j while staying feasible
        allowed = min(hi, 1.0 - hj, max_move)
        if allowed <= 0:
            continue
        delta = rng.random() * allowed
        new_h = curr_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_val = _c5_value(new_h, dx)
        delta_val = new_val - curr_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / max(temperature, 1e-12)):
            curr_h = new_h
            curr_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                # early exit if we have already beaten the current record
                if best_val <= 0.3808:
                    break

        # every few thousand steps make a small global perturbation to escape
        if it % 5000 == 0 and it > 0:
            pert = rng.normal(scale=0.02, size=n)
            trial = curr_h + pert
            trial = _project_to_feasible(trial, target_sum)
            trial_val = _c5_value(trial, dx)
            if trial_val < curr_val or rng.random() < 0.1:
                curr_h = trial
                curr_val = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = trial.copy()
                    if best_val <= 0.3808:
                        break

    # final safety projection (ensures exact feasibility)
    best_h = _project_to_feasible(best_h, target_sum)
    best_val = _c5_value(best_h, dx)

    return best_h.tolist(), float(best_val), n
