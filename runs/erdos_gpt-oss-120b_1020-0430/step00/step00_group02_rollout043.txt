import numpy as np
import random
import time
import math

def _project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the set {0<=x<=1, sum(x)=target}."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Compute the Erdős C5 value for a step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    # linear (non‑periodic) correlation as described in the statement
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(vals.max())


def _gradient(h: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the worst‑shift term."""
    n = h.size
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            plus = np.concatenate((h[k:], np.zeros(k)))
            minus = np.concatenate((np.zeros(k), h[: n - k]))
        else:
            kp = -k
            plus = np.concatenate((np.zeros(kp), h[: n - kp]))
            minus = np.concatenate((h[kp:], np.zeros(kp)))
        grad += 1.0 - (plus + minus)
    if shifts.size:
        grad /= float(shifts.size)
    return grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound and the number of points."""
    random.seed(seed)
    np.random.seed(seed)

    # ---- initial vector -------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=float)  # provided by the harness
        n = h.size
    except NameError:
        n = 180                      # a reasonable even size if no parent
        h = np.random.rand(n)
        target = n / 2.0
        h = _project_to_sum_box(h, target)

    target = n / 2.0
    if not np.isclose(h.sum(), target, atol=1e-12):
        h = _project_to_sum_box(h, target)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ---- optimisation loop ----------------------------------------------
    start = time.time()
    end_time = start + budget_s * 0.95  # keep a safety margin

    temperature = 0.05
    it = 0
    while time.time() < end_time:
        # correlation and worst shift(s)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        cur_c5 = float(corr.max())
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            if best_c5 <= 0.3808:
                break

        # indices achieving the maximum (allow numerical tolerance)
        idx_max = np.where(np.isclose(corr, cur_c5, atol=1e-12))[0]
        shifts = idx_max - (n - 1)                     # map to integer k

        # gradient step
        g = _gradient(h, shifts.astype(int))
        step = temperature * (1.0 / (1.0 + it * 0.001))
        h_new = h - step * g
        h_new = _project_to_sum_box(h_new, target)

        new_c5 = _c5_value(h_new)

        # simulated‑annealing acceptance
        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / max(temperature, 1e-8)):
            h = h_new

        # cooling schedule and occasional jitter
        it += 1
        if it % 2000 == 0:
            temperature *= 0.95
        if it % 5000 == 0:
            noise = (np.random.rand(n) - 0.5) * 0.01
            h = _project_to_sum_box(h + noise, target)

    # final evaluation (ensure best is returned)
    final_c5 = _c5_value(best_h)
    return best_h.tolist(), float(final_c5), int(len(best_h))
