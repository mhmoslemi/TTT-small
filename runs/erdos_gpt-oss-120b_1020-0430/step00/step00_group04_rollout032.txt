def project_to_capped_simplex(v, target):
    # simple binary search for lambda
    lo = -np.max(v) - 1  # ensure lower bound far enough.
    hi = 1 - np.min(v) + 1  # high bound.
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0, 1)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo+hi)/2.0, 0, 1)

import time
import numpy as np

# Helper functions (top level, no closures)

def _project_to_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """ Euclidean projection of v onto {0<=x<=1, sum(x)=target}. """
    lo = -np.max(v) - 5.0
    hi = 1.0 - np.min(v) + 5.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _shift_array(arr: np.ndarray, s: int) -> np.ndarray:
    """ Return an array where out[i] = arr[i+s] (zero if out of bounds). """
    n = arr.shape[0]
    out = np.empty(n, dtype=arr.dtype)
    if s >= 0:
        out[: n - s] = arr[s:]
        out[n - s :] = 0.0
    else:
        s = -s
        out[s:] = arr[: n - s]
        out[:s] = 0.0
    return out


def _compute_c5(h: np.ndarray) -> float:
    """ C5 value for a given step function sampled on the grid. """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _compute_c5_and_shift(h: np.ndarray):
    """ Return (C5, shift) where shift is the lag attaining the maximum. """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    shift = idx - (n - 1)      # index 0 corresponds to shift = -(n-1)
    return max_val * (2.0 / n), shift


def _random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """ Uniform random point in the capped simplex (sum = n/2, 0<=x<=1). """
    v = rng.random(n)
    return _project_to_capped_simplex(v, n / 2.0)


def _interpolate_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """ Linear interpolation of a step function onto a finer grid. """
    n0 = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    return _project_to_capped_simplex(new_h, new_n / 2.0)


def _local_search(initial_h: np.ndarray, deadline: float, rng: np.random.Generator):
    """ Simple hill‑climbing: move a small amount between two coordinates.
        Keeps the sum exactly constant. """
    h = initial_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _compute_c5(best_h)
    iters_without_improve = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # direction: transfer mass from j to i or opposite
        if rng.random() < 0.5:
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            cand = h.copy()
            cand[i] += delta
            cand[j] -= delta
        else:
            max_delta = min(1.0 - h[j], h[i])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            cand = h.copy()
            cand[j] += delta
            cand[i] -= delta

        val = _compute_c5(cand)
        if val + 1e-12 < best_val:
            best_val = val
            best_h = cand
            h = cand
            iters_without_improve = 0
        else:
            iters_without_improve += 1

        # occasional random shake‑up to escape plateaus
        if iters_without_improve >= 20000:
            # replace current point by a fresh random feasible one
            h = _random_feasible_vector(n, rng)
            iters_without_improve = 0
            # keep the global best unchanged unless the new point is better
        # small random perturbations keep the search alive
        if rng.random() < 0.001:
            pert = rng.normal(scale=0.02, size=n)
            cand = h + pert
            cand = _project_to_capped_simplex(cand, n / 2.0)
            val = _compute_c5(cand)
            if val < best_val:
                best_val, best_h, h = val, cand, cand

    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) where
    - h_values : ndarray of step heights on a uniform grid of [0,2]
    - c5_bound : computed C5 value for the returned function
    - n_points : length of the grid (len(h_values))
    The algorithm starts from the supplied `initial_h_values` (if any)
    and performs a time‑bounded hill‑climbing / refinement, possibly
    increasing the resolution.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + float(budget_s)

    # ------------------------------------------------------------------
    # Initial construction
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
        n0 = h0.shape[0]
        h0 = _project_to_capped_simplex(h0, n0 / 2.0)
    else:
        n0 = 42
        h0 = _random_feasible_vector(n0, rng)

    best_h = h0
    best_val = _compute_c5(best_h)
    best_n = best_h.shape[0]

    # ------------------------------------------------------------------
    # Phase 1 : optimise on the original grid (about half the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.5
    h_opt, val_opt = _local_search(best_h, phase1_deadline, rng)
    if val_opt < best_val:
        best_h, best_val = h_opt, val_opt
        best_n = best_h.shape[0]

    # ------------------------------------------------------------------
    # Phase 2 : try finer grids if time permits
    # ------------------------------------------------------------------
    # successive doublings up to a reasonable size
    current_n = best_n
    while time.time() < deadline - 5.0:
        new_n = current_n * 2
        if new_n > 500:          # keep the problem small enough for speed
            break

        # interpolate current best to the finer grid
        h_fine = _interpolate_h(best_h, new_n)

        # allocate a slice of the remaining time (roughly 30‑40 %)
        remaining = deadline - time.time()
        alloc = min(remaining * 0.4, remaining - 4.0)
        if alloc <= 1.0:
            break
        fine_deadline = time.time() + alloc

        h_opt_fine, val_opt_fine = _local_search(h_fine, fine_deadline, rng)
        if val_opt_fine < best_val:
            best_h, best_val, best_n = h_opt_fine, val_opt_fine, new_n

        current_n = new_n

    # final safety projection (numerical drift)
    best_h = _project_to_capped_simplex(best_h, best_h.shape[0] / 2.0)
    final_c5 = _compute_c5(best_h)

    return best_h, final_c5, best_h.shape[0]
