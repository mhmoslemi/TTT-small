def project_feasible(x, target_sum=None):
    # Alternate projection onto [0,1] box and sum constraint
    if target_sum is None:
        target_sum = x.shape[0] / 2.0
    y = x.copy()
    # clip
    y = np.clip(y, 0.0, 1.0)
    for _ in range(10):
        # adjust sum
        diff = target_sum - np.sum(y)
        if abs(diff) < 1e-12:
            break
        y += diff / y.shape[0]
        y = np.clip(y, 0.0, 1.0)
    # final fix if needed
    diff = target_sum - np.sum(y)
    if abs(diff) > 1e-10:
        # try to fix by adjusting a single entry that stays within bounds
        for i in range(y.shape[0]):
            if 0.0 <= y[i] + diff <= 1.0:
                y[i] += diff
                break
        else:
            # adjust many entries
            # distribute diff equally among all entries, then clip
            y += diff / y.shape[0]
            y = np.clip(y, 0.0, 1.0)
    return y

def evaluate_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    # c includes sum h_i*(1-h_{i-l}) across overlapping region indices.
    # Multiply by dx
    c_scaled = c * dx
    return np.max(c_scaled)

def gradient_smooth(h, alpha=30.0, dx=None):
    n = h.shape[0]
    if dx is None:
        dx = 2.0 / n
    # Correlation vector without scaling (just sum)
    corr = np.correlate(h, 1 - h, mode='full')
    c_scaled = corr * dx
    # Stabilize: subtract max before exponent to avoid overflow
    maxc = np.max(c_scaled)
    exp_terms = np.exp(alpha * (c_scaled - maxc))
    sum_exp = np.sum(exp_terms)
    # weights
    w = exp_terms / sum_exp
    # compute gradient
    grad = np.zeros_like(h)
    # lag range from -(n-1) .. (n-1)
    # We'll iterate over lags using slices
    for idx, lag in enumerate(range(-(n-1), n)):
        weight = w[idx] * dx  # include factor dx for derivative
        if lag >= 0:
            # term (1 - h[t - lag])
            # for t = lag ... n-1, corresponding i = t - lag = 0..n-1-lag
            grad[lag:] += weight * (1 - h[:n - lag])
            # term -h[t + lag] for t = 0..n-1-lag => h[t+lag] = h[lag:];
            grad[:n - lag] -= weight * h[lag:]
        else:
            d = -lag
            grad[:n - d] += weight * (1 - h[d:])
            grad[d:] -= weight * h[:n - d]
    return grad

def gradient_descent(h0, steps=200, step0=0.2, alpha=30.0):

    h = h0.copy()

    return h

import numpy as np

def project_feasible(x, target_sum=None):
    """Project vector onto the box [0,1] and the hyperplane sum = target_sum."""
    n = x.shape[0]
    if target_sum is None:
        target_sum = n / 2.0
    y = np.clip(x, 0.0, 1.0)
    # alternating projections – a few sweeps are enough for this small problem
    for _ in range(12):
        diff = target_sum - y.sum()
        if abs(diff) < 1e-12:
            break
        y += diff / n
        y = np.clip(y, 0.0, 1.0)
    # final tiny fix if needed
    diff = target_sum - y.sum()
    if abs(diff) > 1e-10:
        for i in range(n):
            if 0.0 <= y[i] + diff <= 1.0:
                y[i] += diff
                break
        else:
            y[0] = np.clip(y[0] + diff, 0.0, 1.0)
    return y


def evaluate_h(h):
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx evaluated on the discretisation."""
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx


def gradient_smooth(h, alpha=30.0, dx=None):
    """Gradient of the smooth approximation (log‑sum‑exp) of the max overlap."""
    n = h.shape[0]
    if dx is None:
        dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    maxc = np.max(c)
    # stable weights
    w = np.exp(alpha * (c - maxc))
    w /= w.sum()
    grad = np.zeros_like(h)
    # lag runs from -(n-1) … (n-1)
    for idx, lag in enumerate(range(-(n - 1), n)):
        weight = w[idx] * dx
        if lag >= 0:
            grad[lag:] += weight * (1.0 - h[: n - lag])
            grad[: n - lag] -= weight * h[lag:]
        else:
            d = -lag
            grad[: n - d] += weight * (1.0 - h[d:])
            grad[d:] -= weight * h[: n - d]
    return grad


def gradient_descent(h0, steps=200, step0=0.2, alpha=30.0):
    """Projected gradient descent on the smooth max objective."""
    n = h0.shape[0]
    dx = 2.0 / n
    h = h0.copy()
    for it in range(steps):
        step = step0 * (0.99 ** it)
        g = gradient_smooth(h, alpha=alpha, dx=dx)
        h -= step * g
        h = project_feasible(h)
    return h


def gradient_descent_worst(h0, steps=200, step0=0.2):
    """Projected gradient descent focusing on the currently worst lag."""
    n = h0.shape[0]
    dx = 2.0 / n
    h = h0.copy()
    for it in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        max_idx = int(np.argmax(c))
        lag = max_idx - (n - 1)
        grad = np.zeros_like(h)
        if lag >= 0:
            grad[lag:] += dx * (1.0 - h[: n - lag])
            grad[: n - lag] -= dx * h[lag:]
        else:
            d = -lag
            grad[: n - d] += dx * (1.0 - h[d:])
            grad[d:] -= dx * h[: n - d]
        step = step0 * (0.99 ** it)
        h -= step * grad
        h = project_feasible(h)
    return h


def random_feasible(n, rng):
    """Generate a random vector in [0,1] with sum = n/2."""
    v = rng.random(n)
    target = n / 2.0
    v = v * (target / v.sum())
    v = np.clip(v, 0.0, 1.0)
    return project_feasible(v)
