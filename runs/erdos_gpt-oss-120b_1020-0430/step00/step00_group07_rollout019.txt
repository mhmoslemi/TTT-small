import time
import numpy as np
import cvxpy as cp

def _compute_c5(h, dx):
    """Return the overlap integral for a given step function."""
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(corr.max())

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Solve the Erdős minimum‑overlap problem for C₅.

    Returns
    -------
    h_values : np.ndarray
        The step function values (0/1) on a uniform grid of [0,2].
    c5_bound : float
        The achieved upper bound for C₅.
    n_points : int
        Number of discretisation points (the length of h_values).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        n = int(len(initial_h_values))          # use the provided resolution
    else:
        n = 80                                   # default if not supplied
    dx = 2.0 / n
    half_n = n // 2                             # we need exactly half ones

    # ------------------------------------------------------------
    # 2. Build the MILP model
    # ------------------------------------------------------------
    # binary variables h_i  (0 ≤ h_i ≤ 1, integer)
    h = cp.Variable(n, integer=True)
    M = cp.Variable(integer=True)               # max correlation (as an integer count)

    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == half_n,
        M >= 0,
        M <= half_n   # trivial upper bound
    ]

    # auxiliary variables y_{i,j} = h_i * h_j   (i < j)
    n_pairs = n * (n - 1) // 2
    y = cp.Variable(n_pairs, integer=True)
    constraints += [y >= 0, y <= 1]

    # map each unordered pair (i,j) to an index in y
    pair_index = {}
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            pair_index[(i, j)] = idx
            # linearization constraints for the AND
            constraints.append(y[idx] <= h[i])
            constraints.append(y[idx] <= h[j])
            constraints.append(y[idx] >= h[i] + h[j] - 1)
            idx += 1

    # shift constraints: for every k = 1 … n‑1
    for k in range(1, n):
        # indices i that have a partner i+k inside the interval
        overlapping = range(n - k)

        # indices of y variables that correspond to the pairs (i, i+k)
        y_idx = [pair_index[(i, i + k)] for i in overlapping]

        # forward direction  Σ h_i  – Σ y_{i,i+k} ≤ M
        sum_h_fwd = cp.sum([h[i] for i in overlapping])
        sum_y = cp.sum([y[j] for j in y_idx])
        constraints.append(sum_h_fwd - sum_y <= M)

        # backward direction Σ h_{i+k} – Σ y_{i,i+k} ≤ M
        sum_h_bwd = cp.sum([h[i + k] for i in overlapping])
        constraints.append(sum_h_bwd - sum_y <= M)

    # ------------------------------------------------------------
    # 3. Solve
    # ------------------------------------------------------------
    prob = cp.Problem(cp.Minimize(M), constraints)

    # remaining time for the solver (leave a few seconds for clean‑up)
    time_left = max(0.0, budget_s - (time.time() - start_time) - 4.0)

    # Try GLPK first (widely available). If it fails, fall back to CBC.
    solved = False
    try:
        prob.solve(solver=cp.GLPK_MI, glpk={"tmlim": time_left})
        solved = prob.status in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}
    except Exception:
        solved = False

    if not solved:
        try:
            prob.solve(solver=cp.CBC, timeLimit=int(time_left))
            solved = prob.status in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}
        except Exception:
            solved = False

    # ------------------------------------------------------------
    # 4. Retrieve a feasible solution
    # ------------------------------------------------------------
    if solved and h.value is not None and M.value is not None:
        # Round the binary variables (tiny tolerances may appear)
        h_vals = np.rint(np.array(h.value, dtype=float)).astype(float)
        # Repair any possible sum deviation caused by rounding
        diff = int(half_n - h_vals.sum())
        if diff != 0:
            if diff > 0:
                zero_pos = np.where(h_vals == 0)[0]
                if len(zero_pos) >= diff:
                    h_vals[zero_pos[:diff]] = 1.0
            else:
                one_pos = np.where(h_vals == 1)[0]
                if len(one_pos) >= -diff:
                    h_vals[one_pos[: -diff]] = 0.0
        h_vals = np.clip(h_vals, 0, 1)

        # Compute the true bound (may be a little smaller than M*dx)
        c5 = _compute_c5(h_vals, dx)
    else:
        # Fallback: use the supplied initial vector (if any) or a simple balanced pattern
        if "initial_h_values" in globals():
            h_vals = np.array(initial_h_values, dtype=float)
        else:
            # simple alternating pattern with exact half ones
            h_vals = np.zeros(n, dtype=float)
            h_vals[:half_n] = 1.0
        c5 = _compute_c5(h_vals, dx)

    # ------------------------------------------------------------
    # 5. Return the result
    # ------------------------------------------------------------
    return (h_vals, c5, n)
