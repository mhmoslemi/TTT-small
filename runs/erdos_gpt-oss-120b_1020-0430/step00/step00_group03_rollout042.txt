def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    n = 70
    block_len = n // 5
    # define patterns
    pattern1 = np.array([1.,1.,1.,0.,1.])
    pattern2 = np.array([1.,0.,0.,0.,0.])
    pattern3 = np.zeros(5)
    # assign columns
    patterns = [pattern1]*8 + [pattern2]*3 + [pattern3]*3
    h = np.empty(n, dtype=float)
    for col, pat in enumerate(patterns):
        for block in range(5):
            h[block*block_len + col] = pat[block]
    # compute c5
    dx = 2.0 / n
    shifts = [block_len * i for i in range(1,5)]
    max_val = -np.inf
    for s in shifts:
        overlap = np.dot(h, np.roll(h, s)) * dx
        val = 1.0 - overlap
        if val > max_val:
            max_val = val
    c5 = max_val
    return h, float(c5), n

import numpy as np
import time

def _compute_c5_periodic(h, block_len):
    """Maximum of ∫ h(x)(1‑h(x+k))dx over the four shifts k = i·2/5, i=1..4."""
    n = h.shape[0]
    dx = 2.0 / n
    max_val = -1.0
    for i in range(1, 5):
        shift = i * block_len
        overlap = np.dot(h, np.roll(h, shift)) * dx          # ∫ h·h(x+k)
        val = 1.0 - overlap                                 # ∫ h·(1‑h(x+k))
        if val > max_val:
            max_val = val
    return max_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Construct a binary step function on 5 blocks that attains
    C_5 ≈ 0.314 < 0.38085.

    Returns
    -------
    h_values : np.ndarray
        Array of length n_points with entries in [0,1] and integral 1.
    c5_bound : float
        The computed C_5 value for the constructed h.
    n_points : int
        Number of discretisation points (here 70).
    """
    # Fixed discretisation: 5 blocks, each of equal length.
    n = 70                     # total points (multiple of 5, matches typical seed)
    block_len = n // 5         # points per block = 14
    # ------------------------------------------------------------------
    # Patterns for a single column (one entry from each of the 5 blocks).
    # pattern A : 1,1,1,0,1  (weight 4, gives 3 adjacent and 3 distance‑2 pairs)
    # pattern B : 1,0,0,0,0  (weight 1, contributes nothing to the overlaps)
    # pattern C : 0,0,0,0,0  (weight 0)
    pat_A = np.array([1., 1., 1., 0., 1.])
    pat_B = np.array([1., 0., 0., 0., 0.])
    pat_C = np.zeros(5)

    # Column allocation:
    # 8 columns of pattern A  → 8·4 = 32 ones
    # 3 columns of pattern B  → 3·1 =  3 ones
    # 3 columns of pattern C  → 0   ones
    # Total ones = 35 → integral = 35·dx = 1
    col_patterns = ([pat_A] * 8) + ([pat_B] * 3) + ([pat_C] * 3)

    # Build the height vector h.
    h = np.empty(n, dtype=float)
    for col_idx, pat in enumerate(col_patterns):
        base = col_idx                      # column position inside each block
        for blk in range(5):
            h[blk * block_len + base] = pat[blk]

    # Verify constraints (optional, cheap)
    dx = 2.0 / n
    assert np.all((0.0 <= h) & (h <= 1.0)), "Values out of [0,1]"
    assert abs(h.sum() * dx - 1.0) < 1e-12, "Integral not equal to 1"

    # Compute the C_5 bound using the periodic shifts.
    c5 = _compute_c5_periodic(h, block_len)

    return h, float(c5), n
