import numpy as np, time, random

def _project_to_sum_box(v, target):
    """Projection of v onto {x: 0<=x<=1, sum(x)=target}."""
    lo, hi = -10.0, 10.0
    # expand bounds if needed
    def s(lam):
        return np.clip(v + lam, 0.0, 1.0).sum()
    while s(lo) > target:
        lo -= 10.0
    while s(hi) < target:
        hi += 10.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if s(mid) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _max_overlap(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx for discretised h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / len(h))
