    n = len(h)

    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

        # scale
        h = h * (target_sum / cur_sum)
        h = np.clip(h, 0.0, 1.0)

        deficit = target_sum - h.sum()
        # distribute uniformly small amount
        if abs(deficit) > 1e-12:
            # add to first entries not exceeding 1
            for idx in range(n):
                add = min(1 - h[idx], deficit)
                h[idx] += add
                deficit -= add
                if abs(deficit) < 1e-12:

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h) if 'evaluate_erdos_solution' in globals() else compute_c5(best_h)

    n = len(h)
    dx = 2.0 / n
    # compute FFTs
    H = np.fft.fft(h)
    G = np.fft.fft(1 - h)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time
    random.seed(seed)
    np.random.seed(seed)
    # Determine n
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = len(h)
    else:
        n = 70
        h = np.full(n, 0.5, dtype=np.float64)
    target_sum = n/2.0
    # adjust to exact sum if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-12:
        h *= target_sum/cur
        # clip to [0,1], then distribute any residual
        h = np.clip(h, 0.0, 1.0)
        deficit = target_sum - h.sum()
        if abs(deficit) > 1e-12:
            # distribute to indices that are not at bound
            for idx in range(n):
                add = min(1.0 - h[idx], deficit)
                h[idx] += add
                deficit -= add
                if abs(deficit) < 1e-12:
                    break
    def compute_max_c5(vec):
        n = len(vec)
        dx = 2.0 / n
        H = np.fft.fft(vec)
        G = np.fft.fft(1.0 - vec)
        corr = np.real(np.fft.ifft(H * np.conj(G)))
        return float(np.max(corr) * dx)
    best_h = h.copy()
    best_c5 = compute_max_c5(best_h)
    start = time.time()
    # parameters
    max_iters = 200000
    eps = 0.2
    decay = 0.9995
    for it in range(max_iters):
        if time.time() - start > budget_s * 0.95:
            break
        i, j = random.sample(range(n), 2)
        # adjust a delta from i to j
        # we move delta from j to i (i increase, j decrease)
        lo = max(-best_h[i], best_h[j] - 1.0)
        hi = min(1.0 - best_h[i], best_h[j])
        if lo >= hi:
            continue
        delta = random.uniform(lo, hi) * eps
        if delta == 0.0:
            continue
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        c5_new = compute_max_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h
        # optional annealing may accept worse occasionally
        eps *= decay
    return best_h, float(best_c5), len(best_h)

import random
import time

def _c5_circular_max(h: np.ndarray) -> float:
    """Maximum of the cyclic overlap integral for a given step function.

    Uses the definition C5 = max_k ∫ h(x) (1 - h(x+k)) dx,
    discretised as a circular cross‑correlation.
    """
    n = h.shape[0]
    dx = 2.0 / n
    # circular cross‑correlation via FFT: r[k] = Σ_i h[i] * (1-h)[i+k mod n]
    H = np.fft.fft(h)
    G = np.fft.fft(1.0 - h)
    corr = np.real(np.fft.ifft(H * np.conj(G)))
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Find a step function h that (approximately) minimises the Erdős C5 constant.

    The routine starts from the provided `initial_h_values` (if any) and
    performs a simple stochastic pair‑adjustment search while keeping the
    constraints 0 ≤ h[i] ≤ 1 and Σ h[i] = n/2 exactly.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.size
    else:
        n = 70                          # a modest size when no parent is given
        h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # scale then clip, finally fix any tiny deficit
        h *= target_sum / cur_sum
        h = np.clip(h, 0.0, 1.0)
        deficit = target_sum - h.sum()
        if abs(deficit) > 1e-12:
            for idx in range(n):
                add = min(1.0 - h[idx], deficit)
                h[idx] += add
                deficit -= add
                if abs(deficit) < 1e-12:
                    break

    # ------------------------------------------------------------------
    # helper that computes the objective (max cyclic C5)
    compute = _c5_circular_max

    best_h = h.copy()
    best_c5 = compute(best_h)

    start = time.time()
    # simple stochastic hill climbing with a decaying step size
    eps = 0.2            # initial magnitude of pair‑adjustments
    decay = 0.9995       # exponential decay per iteration
    max_iters = 200000   # hard upper bound; the time budget stops earlier

    for it in range(max_iters):
        # stop before the allotted time expires
        if time.time() - start > budget_s * 0.95:
            break

        # pick two distinct indices
        i, j = random.sample(range(n), 2)

        # propose moving a small amount delta from j to i
        lo = max(-best_h[i], best_h[j] - 1.0)   # keep both in [0,1]
        hi = min(1.0 - best_h[i], best_h[j])
        if lo >= hi:
            continue
        delta = random.uniform(lo, hi) * eps
        if delta == 0.0:
            continue

        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        c5_new = compute(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

        eps *= decay                       # slowly shrink the step size

    # ------------------------------------------------------------------
    return best_h, float(best_c5), int(len(best_h))
