def project(h, S):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum to S using water-filling
    # While tolerance
    for _ in range(10):
        cur = h.sum()
        diff = cur - S
        if abs(diff) < 1e-8:
            break
        # identify free indices not at bounds
        if diff > 0:
            # need to decrease sum
            free = (h > 0)
            if not free.any():
                break
            # total amount we can decrease
            total_decrease = (h[free] - 0).sum()
            if total_decrease <= 0:
                break
            delta = diff / free.sum()
            # ensure we don't go below 0
            # actually we shift down uniformly
            # but to avoid overshoot:
            step = min(delta, (h[free] - 0).max())
            h[free] -= step
        else:
            free = (h < 1)
            if not free.any():
                break
            delta = diff / free.sum()  # diff negative -> step negative (increase)
            step = max(delta, -(1 - h[free]).max())
            h[free] -= step  # subtract negative = increase
        # after adjust, clip again
        h = np.clip(h, 0.0, 1.0)
    return h

    h_new = h - lr * grad
    return project_to_bounds_and_sum(h_new, _target_sum)


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    global _target_sum, _alpha

    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
    else:
        n = 48                     # a modest size, even number makes sum integer
        h0 = rng.random(n)

    target = n / 2.0
    _target_sum = target
    h0 = project_to_bounds_and_sum(h0, target)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # generate a few starting points (initial + random)
    init_points = [best_h]
    for _ in range(4):
        rnd = rng.random(n)
        init_points.append(project_to_bounds_and_sum(rnd, target))

    # smoothing schedule for the max‑approximation
    alpha_schedule = [1.0, 5.0, 15.0, 30.0]

    # ------------------------------------------------------------------
    # SLSQP optimisation on the smoothed objective
    for h_start in init_points:
        if time.time() - start > budget_s * 0.92:
            break
        h = h_start.copy()
        for a in alpha_schedule:
            if time.time() - start > budget_s * 0.92:
                break
            _alpha = a
            cons = {'type': 'eq', 'fun': sum_eq_constraint}
            bounds = [(0.0, 1.0)] * n
            res = minimize(
                smooth_obj,
                h,
                method='SLSQP',
                jac=smooth_grad,
                bounds=bounds,
                constraints=[cons],
                options={'maxiter': 200, 'ftol': 1e-9, 'disp': False},
            )
            h = project_to_bounds_and_sum(res.x, target)
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # ------------------------------------------------------------------
    # fine‑tune with a simple sub‑gradient descent on the true max
    lr = 0.12
    while time.time() - start < budget_s * 0.98:
        best_h = subgradient_step(best_h, lr)
        cur = compute_c5(best_h)
        if cur < best_val:
            best_val = cur
        lr *= 0.9995
        if lr < 1e-6:
            lr = 1e-6

    # final projection (numerical safety) and evaluation
    best_h = project_to_bounds_and_sum(best_h, target)
    best_val = compute_c5(best_h)

    # return as required: values list, bound, number of points
    return (best_h.tolist(), float(best_val), int(n))
