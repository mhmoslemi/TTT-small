import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h on [0,2] with a low C5 bound.

    Returns:
        h_values: np.ndarray of shape (n_points,) with values in [0,1] and sum = n_points/2.
        c5_bound: the evaluated max overlap ∫h(x)(1−h(x+k))dx.
        n_points: number of discrete samples used.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n_points = h0.size
    except NameError:
        n_points = 70
        h0 = np.zeros(n_points, dtype=np.float64)
        half = n_points // 2
        h0[:half] = 1.0           # simple binary start if no parent provided

    target_sum = n_points / 2.0               # because ∫h = 1  ->  sum(h)*dx = 1, dx=2/n
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:      # repair tiny drift if needed
        diff = target_sum - cur_sum
        idx = rng.integers(n_points)
        h0[idx] = np.clip(h0[idx] + diff, 0.0, 1.0)

    dx = 2.0 / n_points

    # ------------------------------------------------------------------
    # 2.  Helper to compute the C5 value for a candidate
    # ------------------------------------------------------------------
    def c5_of(h):
        # full correlation of h with (1‑h) gives the raw sums; multiply by dx for the integral
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    # ------------------------------------------------------------------
    # 3.  Local optimisation (simulated‑annealing style)
    # ------------------------------------------------------------------
    current = h0.copy()
    cur_score = c5_of(current)
    best = current.copy()
    best_score = cur_score

    max_step = 0.5          # maximal absolute change on a single entry per swap
    temperature = 0.015    # initial temperature for Metropolis moves
    decay = 0.99995        # exponential decay per iteration
    it = 0
    last_improve = 0

    # main loop
    while True:
        # time check
        if time.time() - start > budget_s * 0.94:
            break

        # ----- propose a swap of mass between two positions -----------------
        i, j = rng.integers(n_points, size=2)
        while i == j:
            j = rng.integers(n_points)

        # feasible positive direction (increase i, decrease j)
        dpos = min(max_step, 1.0 - current[i], current[j])
        # feasible negative direction (decrease i, increase j)
        dneg = min(max_step, current[i], 1.0 - current[j])

        if dpos <= 0 and dneg <= 0:
            continue

        if dpos > 0 and dneg > 0:
            if rng.random() < 0.5:
                delta = rng.random() * dpos
            else:
                delta = -rng.random() * dneg
        elif dpos > 0:
            delta = rng.random() * dpos
        else:
            delta = -rng.random() * dneg

        new = current.copy()
        new[i] += delta
        new[j] -= delta

        new_score = c5_of(new)

        # ----- acceptance rule ---------------------------------------------
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            if temperature > 1e-12:
                prob = np.exp((cur_score - new_score) / temperature)
                accept = rng.random() < prob
            else:
                accept = False

        if accept:
            current, cur_score = new, new_score
            if new_score < best_score - 1e-12:
                best, best_score = new.copy(), new_score
                last_improve = it

        temperature *= decay
        it += 1

        # ----- occasional larger diversification if stuck ------------------
        if it - last_improve > 60000:
            # pick a small random subset and redraw its values
            subset = rng.choice(n_points, size=10, replace=False)
            vals = rng.random(10)
            vals = np.clip(vals, 0.0, 1.0)

            cur_subset_sum = current[subset].sum()
            new_subset_sum = vals.sum()
            diff = new_subset_sum - cur_subset_sum

            # find a compensating index outside the subset
            remaining = np.setdiff1d(np.arange(n_points), subset)
            k = rng.choice(remaining)
            new_k_val = current[k] - diff
            if 0.0 <= new_k_val <= 1.0:
                cand = current.copy()
                cand[subset] = vals
                cand[k] = new_k_val
                cand_score = c5_of(cand)

                if cand_score < cur_score or (temperature > 1e-12 and rng.random() < np.exp((cur_score - cand_score) / temperature)):
                    current, cur_score = cand, cand_score
                    if cand_score < best_score - 1e-12:
                        best, best_score = cand.copy(), cand_score
                        last_improve = it
            last_improve = it   # reset stagnation counter

    # ------------------------------------------------------------------
    # 4.  Final evaluation using the provided helper (if present)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_erdos_solution(best)
    except Exception:
        final_score = c5_of(best)

    return best, float(final_score), n_points
