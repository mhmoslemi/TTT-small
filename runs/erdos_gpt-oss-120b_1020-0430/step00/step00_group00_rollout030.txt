def run(seed=42,budget_s=1100):
    import numpy as np, time, math
    timer...

    # set N (maybe 200). Could increase to 300 if time allows.
    N = 200

    # Starting vector: uniform 0.5
    h = np.full(N, 0.5)

    # compute best bound
    def max_overlap(h):
        N = len(h)
        dx = 2.0 / N
        L = 1 << ((2*N-1).bit_length())
        H = np.fft.rfft(h, n=L)
        V = np.fft.rfft((1 - h)[::-1], n=L)
        conv = np.fft.irfft(H * V, n=L)[:2*N-1]
        # due to numerical errors, constrain >0
        return np.max(conv) * dx

    best_h = h.copy()
    best_c = max_overlap(h)

    # Simulated annealing parameters
    # initial temperature based on diff between typical bound 0.5 and maybe current
    T0 = 0.05
    # final temperature
    Tf = 0.0005
    # step size initial
    step0 = 0.3
    step_min = 0.001

    start = time.time()

    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break
        # schedule
        frac = elapsed / budget_s
        T = T0 * (1 - frac) + Tf * frac
        step = step0 * (1 - frac) + step_min * frac

        # propose move
        N_local = N
        i, j = np.random.randint(N_local, size=2)
        if i == j:
            continue
        # allowable delta interval
        lower = max(-h[i], h[j] - 1)
        upper = min(1 - h[i], h[j])
        # intersect with step-limited interval [-step, step]
        low = max(lower, -step)
        high = min(upper, step)
        if low >= high:
            continue
        delta = np.random.uniform(low, high)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # compute overlap
        c_new = max_overlap(h_new)

        if c_new < best_c or np.random.rand() < math.exp(-(c_new - best_c) / T):

import numpy as np, time, math, random

def max_overlap_and_idx(h):
    N = h.shape[0]
    dx = 2.0 / N
    L = 1 << ((2*N-1).bit_length())
    H = np.fft.rfft(h, n=L)
    V = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H*V, n=L)[:2*N-1]
    idx = int(np.argmax(conv))
    val = float(conv[idx]) * dx
    return val, idx

def mismatch_pairs(h, shift):
    N = h.shape[0]
    if shift >= 0:
        i = np.arange(N - shift)
        hi = h[i]
        hj = h[i + shift]
    else:
        i = np.arange(-shift, N)
        hi = h[i]
        hj = h[i + shift]
    # boolean arrays
    mask_one = (hi == 1) & (hj == 0)
    mask_zero = (hi == 0) & (hj == 1)
    return i[mask_one], i[mask_zero]

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    np.random.seed(seed)
    random.seed(seed)
    N = 180  # even
    half = N // 2
    # initial binary vector
    h = np.zeros(N, dtype=np.float64)
    h[:half] = 1.0
    np.random.shuffle(h)
    # compute initial overlap and index
    cur_c, idx_max = max_overlap_and_idx(h)
    best_c = cur_c
    best_h = h.copy()
    start = time.time()
    no_improve = 0
    patience = 5000
    while time.time() - start < budget_s - 0.3:
        shift = idx_max - (N - 1)
        # try to fix mismatches for the worst shift
        mism_one, mism_zero = mismatch_pairs(h, shift)
        if mism_one.size and mism_zero.size:
            i = int(np.random.choice(mism_one))
            j = int(np.random.choice(mism_zero))
        else:
            # random 1 and 0 indices
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))
        # propose swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_c, new_idx = max_overlap_and_idx(h_new)
        # acceptance: greedy
        if new_c <= cur_c:
            h = h_new
            cur_c = new_c
            idx_max = new_idx
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # sometimes accept worse move (simulated annealing)
            # temperature decreases linearly
            elapsed = time.time() - start
            T = 0.01 * (1.0 - elapsed / budget_s)
            if T > 0 and np.random.rand() < math.exp(-(new_c - cur_c) / T):
                h = h_new
                cur_c = new_c
                idx_max = new_idx
                no_improve += 1
            else:
                no_improve += 1
        # restart if stuck
        if no_improve >= patience:
            # random restart
            h = np.zeros(N, dtype=np.float64)
            h[:half] = 1.0
            np.random.shuffle(h)
            cur_c, idx_max = max_overlap_and_idx(h)
            no_improve = 0
    # final evaluation using supplied helper if exists
    try:
        final_c = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c = best_c
    return best_h, final_c, N

import numpy as np
import time
import math
import random

def _max_overlap_and_index(h):
    """Return (max_overlap, index_of_max) for the full correlation of h with 1‑h."""
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, n=L)
    V = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * V, n=L)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    val = float(conv[idx]) * dx
    return val, idx


def _mismatch_pairs(h, shift):
    """
    For a given shift (as used by np.correlate full mode),
    return the indices i where h[i]=1, h[i+shift]=0  (type A)
    and indices i where h[i]=0, h[i+shift]=1           (type B).
    """
    n = h.shape[0]
    if shift >= 0:
        i = np.arange(n - shift)
        hi = h[i]
        hj = h[i + shift]
    else:
        i = np.arange(-shift, n)
        hi = h[i]
        hj = h[i + shift]
    mask_a = (hi == 1) & (hj == 0)
    mask_b = (hi == 0) & (hj == 1)
    return i[mask_a], i[mask_b]


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] giving a low C5 value."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------- parameters -------------------
    N = 180                     # number of discretisation points (must be even)
    half = N // 2               # exactly half of the entries are 1
    patience = 5000             # restart if no improvement for this many steps
    # --------------------------------------------------

    # start from a random balanced binary vector
    h = np.zeros(N, dtype=np.float64)
    h[:half] = 1.0
    np.random.shuffle(h)

    cur_c, cur_idx = _max_overlap_and_index(h)
    best_c, best_h = cur_c, h.copy()
    no_improve = 0
    start = time.time()

    while time.time() - start < budget_s - 0.3:
        # current worst lag (the one giving the current maximum overlap)
        shift = cur_idx - (N - 1)

        # try to fix mismatches for that lag
        mis_one, mis_zero = _mismatch_pairs(h, shift)
        if mis_one.size and mis_zero.size:
            i = int(np.random.choice(mis_one))
            j = int(np.random.choice(mis_zero))
        else:
            # fallback: random 1↔0 swap
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))

        # propose the swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0

        new_c, new_idx = _max_overlap_and_index(h_new)

        # greedy acceptance (with a tiny amount of simulated annealing)
        if new_c <= cur_c:
            h, cur_c, cur_idx = h_new, new_c, new_idx
            if new_c < best_c:
                best_c, best_h = new_c, h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            elapsed = time.time() - start
            T = 0.01 * (1.0 - elapsed / budget_s)   # linear cooling
            if T > 0 and np.random.rand() < math.exp(-(new_c - cur_c) / T):
                h, cur_c, cur_idx = h_new, new_c, new_idx
            no_improve += 1

        # restart if stuck
        if no_improve >= patience:
            h = np.zeros(N, dtype=np.float64)

            np.random.shuffle(h)
            cur_c, cur_idx = _max_overlap_and_index(h)
