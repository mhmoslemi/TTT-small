def evaluate(h):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # ignore the extremes where overlap region small? No, include all.

def improve(h, iterations, eps, rng):
    best = h.copy()
    best_val = evaluate(best)
    n = len(h)
    for it in range(iterations):
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(-eps, eps)
        # propose new values, ensure within [0,1]
        hi, hj = best[i], best[j]
        new_hi = hi + delta
        new_hj = hj - delta
        if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:
            continue
        # create candidate
        cand = best.copy()
        cand[i] = new_hi
        cand[j] = new_hj
        # evaluate
        val = evaluate(cand)
        if val < best_val:
            best = cand
            best_val = val
        # optionally accept worse with probability for simulated annealing
    return best, best_val
```

        s = proj.sum()

            lo = mid
        else:
            hi = mid

    # adjust any tiny sum error:
    diff = proj.sum() - S
    if abs(diff) > 1e-10:

            proj = np.clip(proj, 0.0, 1.0)
            # final correction if still off
            proj = proj - (proj.sum() - S) / len(proj)
            proj = np.clip(proj, 0.0, 1.0)

def _full_corr(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1-h) on [0,2] (no wrap)."""
    n = h.size
    size = 2 * n
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * np.conj(G), n=size)[: 2 * n - 1]
    return corr.real


def _project_to_sum_box(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {x∈[0,1]^n , sum x = target_sum}."""

            lo = mid
        else:
