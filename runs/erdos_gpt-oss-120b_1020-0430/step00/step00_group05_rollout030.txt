def simulated_annealing(h, max_iters, rng):
    N = h.size
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_local_h = cur_h.copy()
    best_local_c5 = cur_c5
    T0 = 0.5  # initial temperature
    for t in range(max_iters):
        # Determine temperature schedule
        T = T0 * (1 - t / max_iters)
        # propose swap
        # sample i (1), j (0)
        i = rng.integers(N)
        while cur_h[i] != 1.0:
            i = rng.integers(N)
        j = rng.integers(N)
        while cur_h[j] != 0.0:
            j = rng.integers(N)
        # swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            # accept
            cur_c5 = new_c5
            if cur_c5 < best_local_c5:
                best_local_c5 = cur_c5
                best_local_h = cur_h.copy()
        else:
            # reject: revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0
        # continue
    return best_local_h, best_local_c5

def simulated_annealing(n, rng, max_iter):
    h = random_binary(n)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for it in range(max_iter):
        T = max(1e-4, 0.5 * (1 - it / max_iter))
        # pick swap
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        # to avoid overhead we could store these arrays each iteration (update after swap) but overhead small.
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def local_optimize(h, rng, start_time, budget_s):
    N = h.shape[0]
    best_h = h.copy()
    conv, max_idx, max_val = compute_conv_and_max_shift(h)
    best_max = max_val
    # we have to compute max count not scaled; later scale at return.
    # define max_iter depending on remaining
    while time.time() - start_time < budget_s:
        # find candidate i:
        L = max_idx
        if L == 0:
            # choose random 1 and random 0
            i = rng.choice(np.flatnonzero(h == 1))
        else:
            end = N - L
            # in-range contributions
            mask_in = (h[:end] == 1) & (h[L:] == 0)
            candidates_in = np.flatnonzero(mask_in)
            # out-of-range contributions
            start_out = N - L
            mask_out = h[start_out:] == 1
            candidates_out = np.flatnonzero(mask_out) + start_out
            # combine
            if candidates_in.size + candidates_out.size > 0:
                # pick from combined
                if candidates_out.size > 0 and (candidates_in.size == 0 or rng.random() < 0.5):
                    i = rng.choice(candidates_out)
                else:
                    i = rng.choice(candidates_in)
            else:

        max_idx_new = int(np.argmax(C_new))

        # accept if non-worsening (max_val_new <= max_val)
        if max_val_new <= max_val:

            max_val = max_val_new
            max_idx = max_idx_new
            if max_val_new < best_max:
                best_max = max_val_new
                best_h = h.copy()

import numpy as np
import time
import math

def _max_count(h: np.ndarray) -> int:
    """Maximum value of the discrete overlap count (unscaled)."""
    n = h.shape[0]
    conv = np.convolve(h, np.flip(1 - h), mode="full")
    # positions n-1 .. 2n-2 correspond to non‑negative shifts
    return int(np.max(conv[n - 1 :]))

def _c5_value(h: np.ndarray) -> float:
    """Scaled C₅ for a binary step vector."""
    n = h.shape[0]
    return _max_count(h) * (2.0 / n)

def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones."""
    vec = np.zeros(n, dtype=np.int8)
    idx = rng.choice(n, size=n // 2, replace=False)
    vec[idx] = 1
    return vec

def _optimize(h: np.ndarray, rng: np.random.Generator,
              start_time: float, budget_s: float):
    """
    Local stochastic optimisation for a fixed length.
    Returns (best_vector, best_scaled_c5).
    """
    n = h.shape[0]
    # current state
    cur = h.copy()
    conv = np.convolve(cur, np.flip(1 - cur), mode="full")
    vals = conv[n - 1 :]
    max_idx = int(np.argmax(vals))
    max_val = vals[max_idx]                       # integer count
    best_vec = cur.copy()
    best_val = max_val

    it = 0
    while time.time() - start_time < budget_s:
        # temperature that cools linearly over the whole budget
        frac = (time.time() - start_time) / budget_s
        T = max(0.001, 0.5 * (1.0 - frac))

        # pick a 1‑position, preferably one that contributes to the current maximum
        if max_idx == 0 or rng.random() < 0.2:
            i = rng.choice(np.flatnonzero(cur == 1))
        else:
            L = max_idx
            end = n - L
            # inside‑range offending ones: h[i]=1, h[i+L]=0
            mask_in = np.zeros(n, dtype=bool)
            if end > 0:
                mask_in[:end] = (cur[:end] == 1) & (cur[L:] == 0)
            # out‑of‑range offending ones: i >= n-L and h[i]=1
            start_out = n - L
            mask_out = np.zeros(n, dtype=bool)
            if start_out < n:
                mask_out[start_out:] = (cur[start_out:] == 1)
            cand = np.flatnonzero(mask_in | mask_out)
            if cand.size:
                i = rng.choice(cand)
            else:
                i = rng.choice(np.flatnonzero(cur == 1))

        # a random zero to swap with
        zeros = np.flatnonzero(cur == 0)
        if zeros.size == 0:
            break
        j = rng.choice(zeros)

        # perform swap
        cur[i], cur[j] = 0, 1

        # recompute overlap
        conv_new = np.convolve(cur, np.flip(1 - cur), mode="full")
        vals_new = conv_new[n - 1 :]
        max_idx_new = int(np.argmax(vals_new))
        max_val_new = vals_new[max_idx_new]

        delta = max_val_new - max_val
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            # accept move
            max_val = max_val_new
            max_idx = max_idx_new
            if max_val_new < best_val:
                best_val = max_val_new
                best_vec = cur.copy()
        else:
            # reject: undo swap
            cur[i], cur[j] = 1, 0

        it += 1

    return best_vec, best_val * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function (as a list), its C₅ bound and the number of points."""
    start = time.time()
    rng = np.random.default_rng(seed)

    best_vec = None
    best_c5 = float("inf")
    best_n = None

    # use the provided construction if it exists
    if "initial_h_values" in globals():
        iv = np.array(initial_h_values, dtype=np.int8)
        if iv.ndim == 1 and iv.shape[0] % 2 == 0 and iv.sum() == iv.shape[0] // 2:
            c5 = _c5_value(iv)

    # fallback: a simple balanced vector
    if best_vec is None:
        n0 = 70
        best_vec = _random_binary(n0, rng)
        best_c5 = _c5_value(best_vec)
        best_n = n0

    # main search – we keep trying different sizes and re‑optimising
    sizes = [70, 100, 140, 180, 200]
    sizes = [s for s in sizes if s % 2 == 0]

    while time.time() - start < budget_s:
        # pick a size: try the current best length most of the time
        if rng.random() < 0.6:
            n = best_n
            # start from the current best but perturb a little to escape a possible local minimum
            cur = best_vec.copy()
            ones = np.flatnonzero(cur == 1)
            zeros = np.flatnonzero(cur == 0)
            if ones.size and zeros.size:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                cur[i], cur[j] = 0, 1
        else:

        n = 70
