def gradient_max_shift(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    # c length = 2n - 1; shift index s = i - (n-1)
    max_val = c.max()
    max_indices = np.where(np.abs(c - max_val) < 1e-12)[0]
    # compute gradient for each active shift and sum them
    grad = np.zeros_like(h)
    for idx in max_indices:
        shift = idx - (n - 1)  # shift k: positive shift means h[i] * (1 - h[i+shift]) for i where i+shift in range.
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            shift_abs = -shift
            i_range = np.arange(n - shift_abs) + shift_abs
            j_range = np.arange(n - shift_abs)
        # contributions
        # derivative w.r.t h[i] = (1 - h[j]) * dx
        # derivative w.r.t h[j] = -h[i] * dx
        grad[i_range] += (1 - h[j_range]) * dx
        grad[j_range] += -h[i_range] * dx
    # For multiple active shifts, we may take average
    grad /= len(max_indices)
    return grad

def subgradient(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    max_val = c.max()
    # tolerance for equality:
    tol = 1e-12 * max(1.0, np.abs(max_val))
    active = np.where(np.abs(c - max_val) <= tol)[0]
    grad = np.zeros_like(h)
    for idx in active:
        shift = idx - (n - 1)
        if shift >= 0:
            i = np.arange(n - shift)
            j = i + shift
        else:
            shift_abs = -shift
            i = np.arange(n - shift_abs) + shift_abs
            j = np.arange(n - shift_abs)
        # contributions
        # to i indices:
        grad[i] += (1.0 - h[j]) * dx
        # to j indices:
        grad[j] += -h[i] * dx
    grad /= len(active)  # average
    return grad

import time, math, random
import numpy as np

def _project_onto_box_sum(y: np.ndarray, target: float) -> np.ndarray:
    """Project y onto the set {0<=x_i<=1, sum(x)=target} using a simple bisection."""
    lo = np.min(y) - 2.0      # ensure all x_i = 1
    hi = np.max(y) + 2.0      # ensure all x_i = 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return C5 for the given step function h (discrete version)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def _subgradient_of_max_shift(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the max‑over‑k overlap functional."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vmax = corr.max()
    tol = 1e-12 * max(1.0, abs(vmax))
    active = np.where(np.abs(corr - vmax) <= tol)[0]
    grad = np.zeros_like(h)
    for idx in active:
        shift = idx - (n - 1)               # integer lag
        if shift >= 0:
            i = slice(0, n - shift)
            j = slice(shift, n)
        else:
            s = -shift
            i = slice(s, n)
            j = slice(0, n - s)
        grad[i] += (1.0 - h[j]) * dx
        grad[j] += -h[i] * dx
    grad /= max(1, len(active))
    return grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 120
        h = np.random.rand(n)
        target_sum = n / 2.0
        h = _project_onto_box_sum(h, target_sum)

    n = h.shape[0]
    target_sum = n / 2.0
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # basic utilities
    # ------------------------------------------------------------------
    def c5_of(v):
        return _c5_value(v)

    current_c5 = c5_of(h)
    best_h = h.copy()
    best_c5 = current_c5

    start = time.time()
    elapsed = lambda: time.time() - start

    # ------------------------------------------------------------------
    # simulated annealing + occasional sub‑gradient descent
    # ------------------------------------------------------------------
    max_step = 0.2                     # maximal mass moved in one random step
    T = 0.04                           # initial temperature
    cooling = 0.9996                   # geometric cooling per iteration
    it = 0
    last_improve_it = 0

    while elapsed() < budget_s - 1.0:
        # 1) random mass‑transfer move respecting bounds and sum
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta_max = min(max_step, h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            continue
        delta = np.random.rand() * delta_max

        h[i] -= delta
        h[j] += delta
        new_c5 = c5_of(h)

        accept = False
        if new_c5 < current_c5:
            accept = True
        else:
            if T > 1e-12:
                prob = math.exp(-(new_c5 - current_c5) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                last_improve_it = it
        else:
            # revert move
            h[i] += delta
            h[j] -= delta

        # 2) occasional sub‑gradient step (every 800 iterations)
        if (it & 0x1FF) == 0:          # 512‑step pattern
            grad = _subgradient_of_max_shift(h)
            # simple diminishing step size
            step = 0.12 / (1.0 + it * 0.001)
            y = h - step * grad
            y = _project_onto_box_sum(y, target_sum)
            cand = c5_of(y)
            if cand <= current_c5:
                h = y
                current_c5 = cand
                if cand < best_c5:
                    best_c5 = cand
                    best_h = y.copy()
                    last_improve_it = it

        # 3) re‑heat if stagnation too long
        if it - last_improve_it > 15000:
            # perturb around the best seen so far
            noise = np.random.randn(n) * 0.02
            y = best_h + noise
            y = np.clip(y, 0.0, 1.0)
            y = _project_onto_box_sum(y, target_sum)
            current_c5 = c5_of(y)
            h = y
            last_improve_it = it

        # temperature update
        T *= cooling
        it += 1

    # final verification (project once more to be safe)
    best_h = _project_onto_box_sum(best_h, target_sum)
    final_c5 = c5_of(best_h)

    return (best_h.tolist(), float(final_c5), int(n))
