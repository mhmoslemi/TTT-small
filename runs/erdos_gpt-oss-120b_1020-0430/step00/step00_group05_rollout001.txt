import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function with a low Erdős‑C₅ overlap value.

    Returns
    -------
    h_values : np.ndarray
        The discretised step function (values in [0,1]).
    c5_bound : float
        The attained C₅ bound.
    n_points : int
        Length of the discretisation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise a feasible vector.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)  # provided by the harness
    except NameError:
        # No parent – fall back to a simple binary split.
        n0 = 140
        init = np.concatenate([np.ones(n0 // 2), np.zeros(n0 // 2)]).astype(float)
    n0 = init.size

    # Choose a resolution that is a multiple of the parent size.
    target_n = max(120, n0 * 2)            # at least 120 points, twice the parent
    if target_n % 2:
        target_n += 1

    if target_n != n0:
        # Repeat each entry evenly (the integral constraint is automatically satisfied).
        repeat = target_n // n0
        remainder = target_n % n0
        h = np.repeat(init, repeat)
        if remainder:
            h = np.concatenate([h, init[:remainder]])
    else:
        h = init.copy()

    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0

    # Clip and fix any tiny floating‑point drift.
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h *= target_sum / cur_sum
        h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_score = _c5_value(best_h)

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    step = 0.25                              # maximal transferred mass per step
    no_improve = 0

    while time.time() < deadline:
        # Correlation and the set of worst lags (where the max is attained)
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        worst_idxs = np.where(corr >= max_corr - 1e-12)[0]

        # Accumulate neighbour sums for all worst lags.
        neigh = np.zeros(n, dtype=float)
        idxs = np.arange(n)
        for w in worst_idxs:
            k = w - (n - 1)          # integer shift, can be negative
            left = idxs - k
            right = idxs + k

            mask = (left >= 0) & (left < n)
            neigh[mask] += h[left[mask]]

            mask = (right >= 0) & (right < n)
            neigh[mask] += h[right[mask]]

        # Choose donor (small neighbour sum) and receiver (large neighbour sum)
        donor = int(np.argmin(neigh))
        receiver = int(np.argmax(neigh))
        if donor == receiver:
            donor, receiver = rng.choice(n, size=2, replace=False)

        # Feasible transfer amount respecting box constraints.
        max_transfer = min(step, h[donor], 1.0 - h[receiver])
        if max_transfer <= 0.0:
            # if stuck, try a tiny random swap
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(-step, step)
            new_i, new_j = h[i] + delta, h[j] - delta
            if 0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0:
                old_i, old_j = h[i], h[j]
                h[i], h[j] = new_i, new_j
                new_score = _c5_value(h)
                if new_score < best_score:
                    best_score, best_h = new_score, h.copy()
                    no_improve = 0
                else:
                    h[i], h[j] = old_i, old_j
                    no_improve += 1
            else:
                no_improve += 1
            continue

        delta = rng.uniform(0.0, max_transfer)
        # Apply the move.
        h[donor] -= delta
        h[receiver] += delta

        new_score = _c5_value(h)
        if new_score < best_score - 1e-12:
            best_score, best_h = new_score, h.copy()
            no_improve = 0
            step = max(step * 0.9995, 1e-5)     # gradually tighten steps
        else:
            # revert the move
            h[donor] += delta
            h[receiver] -= delta
            no_improve += 1

        # Occasionally inject a small random perturbation to escape plateaus.
        if no_improve >= 2000:
            noise = rng.uniform(-step, step, size=n)
            trial = h + noise
            trial = np.clip(trial, 0.0, 1.0)

            # Re‑normalise the integral exactly.
            diff = target_sum - trial.sum()
            if abs(diff) > 1e-12:
                # Distribute the deficit/profit on entries that can still move.
                if diff > 0:
                    slack = 1.0 - trial
                    total = slack.sum()
                    if total > 0:
                        inc = np.minimum(slack, diff * slack / total)
                        trial += inc
                else:
                    slack = trial
                    total = slack.sum()
                    if total > 0:
                        dec = np.minimum(slack, -diff * slack / total)
                        trial -= dec

            trial_score = _c5_value(trial)
            if trial_score < best_score:
                h = trial
                best_h, best_score = trial.copy(), trial_score
                no_improve = 0
                step = max(step * 0.99, 1e-5)
            else:
                no_improve = 0  # reset counter even if no improvement

        # Early exit if we already reach the target.
        if best_score <= 0.3808:
            break

    # ------------------------------------------------------------------
    # 3.  Final tidy‑up – guarantee the integral exactly.
    # ------------------------------------------------------------------
    final_sum = best_h.sum()
    if abs(final_sum - target_sum) > 1e-12:
        # Adjust one non‑boundary entry.
        for idx in range(n):
            if 0.0 <= best_h[idx] + (target_sum - final_sum) <= 1.0:
                best_h[idx] += (target_sum - final_sum)
                break

    final_score = _c5_value(best_h)

    return best_h, float(final_score), n
