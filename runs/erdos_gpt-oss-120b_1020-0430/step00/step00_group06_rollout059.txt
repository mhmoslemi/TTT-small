def compute_c5(h, dx):
    # compute overlapping integral
    # use np.correlate in full mode
    O = np.correlate(h, 1 - h, mode='full')

def projection_sum_box(v, target_sum):
    # clip then adjust
    v = np.clip(v, 0.0, 1.0)
    # iterative water-filling
    # tolerance 1e-12
    for _ in range(1000):
        cur_sum = v.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        nfree = free.sum()
        if nfree == 0:
            # cannot adjust - just break
            break
        delta = diff / nfree
        v[free] += delta
        # clip again
        np.clip(v, 0.0, 1.0, out=v)
    # ensure exact sum due to rounding (adjust smallest error)
    # compute residual diff and adjust a single element not at bound
    residual = target_sum - v.sum()
    if abs(residual) > 1e-12:
        # find index not at bound (or any) to adjust
        idx = np.argmax(np.abs(v - 0.5)) # arbitrary
        v[idx] = np.clip(v[idx] + residual, 0.0, 1.0)
        # will change sum again maybe by clipping but small
    return v

def gradient_substep(h, k, alpha):
    n = len(h)
    # compute shifted arrays
    # we treat out-of-range as zero
    # shift +k
    if k >= 0:
        shift_plus = np.concatenate((np.zeros(k), h[:n - k]))
    else:
        shift_plus = np.concatenate((h[-k:], np.zeros(-k)))
    # shift -k
    if k >= 0:
        shift_minus = np.concatenate((h[k:], np.zeros(k)))
    else:
        shift_minus = np.concatenate((np.zeros(-k), h[:n + k]))
    grad = 1.0 - (shift_plus + shift_minus)
    # update
    new_h = h - alpha * grad
    return new_h

import time
import math
import numpy as np

def _project_sum_box(v, target):
    """
    Euclidean projection of v onto the set {x | 0<=x_i<=1, sum(x)=target}.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(200):
        cur = v.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        v[free] += delta
        np.clip(v, 0.0, 1.0, out=v)
    # final tiny adjustment if still off (should be rare)
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        # adjust a single non‑border element if possible
        idx = np.argmax(np.abs(v - 0.5))
        v[idx] = np.clip(v[idx] + residual, 0.0, 1.0)
    return v


def _shift(arr, s):
    """Shift arr by integer s, zero‑padding outside the interval."""
    n = arr.size
    if s > 0:
        return np.concatenate((np.zeros(s), arr[: n - s]))
    if s < 0:
        s = -s
        return np.concatenate((arr[s:], np.zeros(s)))
    return arr.copy()


def _c5_value(h, dx):
    """C₅ value for a given step vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64)
    else:
        # fallback: 100 points, uniform 0.5 (satisfies the integral)
        n = 100
        h = np.full(n, 0.5, dtype=np.float64)

    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0

    # enforce integral and box constraints
    h = _project_sum_box(h, target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h, dx)

    # parameters for the local search
    step0 = 0.03                     # initial maximal transfer
    min_step = 1e-4
    grad_thr = 1e-3                 # ignore near‑zero gradient entries

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.96:
            break

        # current worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(corr.argmax())
        cur_val = corr[idx_max] * dx
        k = idx_max - (n - 1)                 # shift index (can be negative)

        # compute sub‑gradient of the active overlap
        hp = _shift(h, k)
        hm = _shift(h, -k)
        grad = 1.0 - (hp + hm)                # d/dh_i O(k)

        # split indices where we should decrease / increase h_i
        dec_idx = np.where(grad > grad_thr)[0]
        inc_idx = np.where(grad < -grad_thr)[0]

        if dec_idx.size and inc_idx.size:
            # choose a random pair (i from dec, j from inc)
            i = np.random.choice(dec_idx)
            j = np.random.choice(inc_idx)

            # maximal feasible transfer from i to j
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max > 0.0:
                # step size shrinks with time
                step = max(min_step, step0 * (1.0 - elapsed / budget_s))
                delta = min(step, delta_max) * np.random.rand()
                if delta > 0.0:
                    h[i] -= delta
                    h[j] += delta

                    # evaluate
                    new_val = _c5_value(h, dx)
                    if new_val <= cur_val:
                        cur_val = new_val
                        if cur_val < best_val:
                            best_val = cur_val
                            best_h = h.copy()
                    else:
                        # revert the move
                        h[i] += delta
                        h[j] -= delta
        else:
            # no clear gradient direction – perform a small random transfer
            i, j = np.random.choice(n, 2, replace=False)
            hi, hj = h[i], h[j]
            dmin = max(-hi, -(1.0 - hj))
            dmax = min(1.0 - hi, hj)
            if dmax > dmin:
                step = max(min_step, step0 * (1.0 - elapsed / budget_s))
                delta = dmin + (dmax - dmin) * np.random.rand() * step
                delta = np.clip(delta, dmin, dmax)
                if abs(delta) > 0:
                    h[i] += delta
                    h[j] -= delta
                    cur_val = _c5_value(h, dx)
                    if cur_val < best_val:
                        best_val = cur_val
                        best_h = h.copy()

        # occasional global perturbation to avoid stagnation
        if np.random.rand() < 0.001:
            # shuffle a fraction of the mass
            mask = np.random.rand(n) < 0.2
            rand_vals = np.random.rand(mask.sum())
            h[mask] = rand_vals
            h = _project_sum_box(h, target_sum)

    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum_box(best_h, target_sum)
    final_val = _c5_value(best_h, dx)

    # return as plain Python list for compatibility
    return best_h.tolist(), float(final_val), int(n)
