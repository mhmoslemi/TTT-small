import numpy as np
import time

def eval_c5(h):
    dx = 2.0 / h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    # Use a deterministic simple construction independent of random.
    n = 200
    dx = 2.0 / n
    # Center positions of each interval (optional), but we can define h_i = (i+0.5)/n
    i = np.arange(n, dtype=np.float64)
    h = (i + 0.5) / n  # goes from 0.5/n to (n-0.5)/n, average = 0.5
    # Ensure sum constraint exactly: sum_i h_i = n/2
    # It holds analytically; small floating error may exist; fix precisely.
    target = n / 2.0
    current_sum = h.sum()
    diff = target - current_sum
    if abs(diff) > 1e-12:
        h += diff / n
        # clip just in case
        h = np.clip(h, 0.0, 1.0)
    c5 = eval_c5(h)
    return h, float(c5), n

import numpy as np
import time

def _c5_value(h: np.ndarray) -> float:
    """
    Compute C₅ for a given step function vector `h`.
    """
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h) over all integer shifts
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

    The function is h(x) = x/2 on [0,2], discretised on a uniform grid.
