def project_to_bounds_sum(v, target):

    hi = 1.0

    hi = 1 - min(v) # ensure v+λ<=1

def hillclimb(h, best_val, max_iter, start_temp, decay):
    h_cur = h.copy()
    val_cur = best_val
    for i in range(max_iter):
        # pick two distinct random indices
        i1, i2 = rng.integers(0, n, size=2)
        while i2 == i1:
            i2 = rng.integers(0, n)
        # compute allowable delta bounds
        max_up = 1.0 - h_cur[i1]
        max_down = h_cur[i2]  # amount we can subtract from h[i2] (since we will add to i1)
        # choose delta random uniform in [-max_down, max_up]
        delta = (rng.random() * (max_up + max_down)) - max_down

            continue

        h_new[i1] += delta
        h_new[i2] -= delta
        # evaluate
        val_new = compute_c5(h_new)
        # acceptance

def project_to_bounds_sum(v, target):
    # Clip v to [0,1] and then adjust via bisection on lambda so that sum(min(max(v + lam,0),1)) = target
    lo = -np.max(v) - 1e-12  # ensure v+lam >= 0 for smallest lam
    hi = 1 - np.min(v) + 1e-12  # ensure v+lam <= 1 for largest lam
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(v + mid, 0, 1)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo+hi)/2, 0, 1)
    # Because of floating errors, adjust small diff by adding tiny epsilon to first non bound variable.
    diff = target - proj.sum()
    if abs(diff) > 1e-8:
        # adjust the element with room
        for i in range(len(proj)):
            if diff > 0 and proj[i] < 1:
                inc = min(diff, 1 - proj[i])
                proj[i] += inc
                diff -= inc
            elif diff < 0 and proj[i] > 0:
                dec = min(-diff, proj[i])
                proj[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return proj

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_to_bounds_sum(v, target, max_iter=60):
    # Bisection on lambda
    lo = -np.max(v) - 1e-12
    hi = 1.0 - np.min(v) + 1e-12
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo+hi)*0.5, 0.0, 1.0)
    # adjust residual
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        # distribute diff greedily
        for i in range(len(proj)):
            if diff > 0 and proj[i] < 1.0:
                inc = min(diff, 1.0 - proj[i])
                proj[i] += inc
                diff -= inc
            elif diff < 0 and proj[i] > 0.0:
                dec = min(-diff, proj[i])
                proj[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return proj

def random_feasible(n, rng):
    v = rng.random(n)
    target = n / 2.0
    return project_to_bounds_sum(v, target)

def hillclimb(initial_h, max_time, rng, start_temp=0.02, decay=0.9995, step_scale=0.2):
    h_cur = initial_h.copy()
    val_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_val = val_cur
    n = len(h_cur)
    it = 0
    t_start = time.time()
    while True:
        if time.time() - t_start > max_time:
            break
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # allowed delta bound
        max_up = 1.0 - h_cur[i]
        max_down = h_cur[j]
        max_step = min(max_up, max_down, step_scale)
        if max_step <= 0:
            continue
        delta = (rng.random()*2.0 - 1.0) * max_step
        # apply
        h_new = h_cur.copy()
        h_new[i] += delta
        h_new[j] -= delta
        val_new = compute_c5(h_new)
        if val_new < val_cur:
            h_cur, val_cur = h_new, val_new
            if val_new < best_val:
                best_h, best_val = h_new.copy(), val_new
        else:
            T = start_temp * (decay ** it)
            if T > 1e-12:
                prob = math.exp(-(val_new - val_cur) / T)
                if rng.random() < prob:
                    h_cur, val_cur = h_new, val_new
        it += 1
    return best_h, best_val

def upscale_h(old_h, new_n):
    old_n = len(old_h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False) + (2.0/old_n)/2.0
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + (2.0/new_n)/2.0
    new_h = np.interp(x_new, x_old, old_h)
    # ensure constraints
    target = new_n / 2.0
    new_h = project_to_bounds_sum(new_h, target)
    return new_h

import numpy as np
import time
import math

# ----- Helper functions -----------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a given discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def project_to_bounds_sum(v: np.ndarray, target: float, max_iter: int = 60) -> np.ndarray:
    """
    Project vector v onto the box [0,1]^n while enforcing sum(v) = target.
    Uses a bisection on a scalar lambda such that sum(clip(v+lambda,0,1)) = target.
    """
    lo = -np.max(v) - 1e-12          # makes all entries <= 0 after shift
    hi = 1.0 - np.min(v) + 1e-12    # makes all entries >= 1 after shift
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # tiny residual correction (to avoid rounding issues)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        for i in range(len(proj)):
            if diff > 0 and proj[i] < 1.0:
                inc = min(diff, 1.0 - proj[i])
                proj[i] += inc
                diff -= inc
            elif diff < 0 and proj[i] > 0.0:
                dec = min(-diff, proj[i])
                proj[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return proj


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n whose entries sum to n/2."""
    v = rng.random(n)
    return project_to_bounds_sum(v, n / 2.0)


def hillclimb(
    init_h: np.ndarray,
    max_time: float,
    rng: np.random.Generator,
    start_temp: float = 0.02,
    decay: float = 0.9995,
    step_scale: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climber / simulated‑annealing on the C5 objective.
    Pairwise adjustments keep the total integral exact.
    """
    h_cur = init_h.copy()
    val_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_val = val_cur
    n = h_cur.shape[0]

    it = 0
    t0 = time.time()
    while time.time() - t0 < max_time:
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_up = 1.0 - h_cur[i]          # how much we may increase i
        max_down = h_cur[j]              # how much we may decrease j
        max_step = min(max_up, max_down, step_scale)
        if max_step <= 0.0:
            continue

        delta = (rng.random() * 2.0 - 1.0) * max_step
        h_new = h_cur.copy()
        h_new[i] += delta
        h_new[j] -= delta

        val_new = compute_c5(h_new)

        if val_new < val_cur:
            h_cur, val_cur = h_new, val_new
            if val_new < best_val:
                best_h, best_val = h_new.copy(), val_new
        else:
            T = start_temp * (decay ** it)
            if T > 1e-12:
                prob = math.exp(-(val_new - val_cur) / T)
                if rng.random() < prob:
                    h_cur, val_cur = h_new, val_new
        it += 1

    return best_h, best_val


def upscale_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """
    Interpolate an existing step function onto a finer grid.
    Afterwards the sum constraint is re‑imposed.
    """
    old_n = old_h.shape[0]
    # cell centres
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False) + (2.0 / old_n) * 0.5
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + (2.0 / new_n) * 0.5
    new_h = np.interp(x_new, x_old, old_h)
    return project_to_bounds_sum(new_h, new_n / 2.0)
