    h = np.zeros(n, dtype=np.int8)
    h[:n//2] = 1
    np.random.shuffle(h)
    O = compute_overlap(h)
    min_overlap = O.min()
    best_h = h.copy()
    best_min = min_overlap

    # main loop
    max_no_improve = 5000
    no_improve = 0
    while time.time() < time_end and no_improve < max_no_improve:
        # random indices
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = np.random.choice(ones)
        b = np.random.choice(zeros)
        O_new = O.copy()
        # apply swap
        # remove at a
        apply_change(O_new, h, a, -1)
        h[a] = 0
        # add at b
        apply_change(O_new, h, b, +1)
        h[b] = 1

        new_min = O_new.min()
        if new_min >= min_overlap:
            # accept
            O = O_new
            min_overlap = new_min
            no_improve = 0
            if new_min > best_min:

                best_h = h.copy()

                # but will continue anyway
        else:
            # revert
            h[a] = 1
            h[b] = 0
            # no change to O
            no_improve += 1
    return best_h, best_min

    # solve for lambda using bisection on range [min(v)-1, max(v)]
    lo = np.min(v) - 1

        s = w.sum()
        if s > target_sum:

                best_h = h_new.copy()
        # temperature schedule
        iteration += 1
        T *= T_decay
    return best_h, best_c

def gradient_descent_step(h):
    n = len(h)
    # compute correlation
    cor = np.correlate(h, 1 - h, mode='full')
    # find max (worst shift)
    idx_max = np.argmax(cor)  # integer
    k = idx_max - (n - 1)
    # compute gradient
    g = np.zeros(n)
    if k >= 0:
        # compute 1 - h_{i+k}
        g[:n - k] = 1 - h[k:]  # for i=0..n-1-k
        # compute h_{i - k}
        g[k:] -= h[:n - k]  # i>=k, subtract h_{i - k}
    else:
        kk = -k

import numpy as np
import time
import random

def compute_c5(h):
    """Return the C5 value for a step function h given as a 1‑D array."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear (non‑periodic) correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((corr * dx).max())

def project_to_capped_simplex(v, target):
    """
    Project a vector v onto the set {x | 0<=x_i<=1, sum x_i = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def random_h(n):
    """Generate a random feasible step function of length n."""
    v = np.random.rand(n)
    return project_to_capped_simplex(v, n / 2.0)

def gradient_for_shift(h, k):
    """
    Gradient of C(k) = sum_i h_i (1‑h_{i+k}) with respect to h,
    where out‑of‑range indices are ignored (linear correlation).
    Returns an array of size len(h).
    """
    n = h.shape[0]
    g = np.zeros(n, dtype=np.float64)
    if k >= 0:
        L = n - k
        # contribution of the +1 part (only where i+k exists)
        g[:L] += 1.0 - h[k:]
        # contribution of the -h_{i‑k} part
        g[k:] -= h[:L]
    else:
        kk = -k
        L = n - kk
        # when k is negative the roles of i and i+k are swapped
        g[kk:] += 1.0 - h[:L]
        g[:L] -= h[kk:]
    return g

def improve_h(initial_h, time_limit):
    """
    Heuristic improvement of a feasible h using gradient‑guided
    mass transfers between two coordinates.  Stops when the time
    limit is reached.
    Returns the best h found and its C5 value.
    """
    n = initial_h.shape[0]
    target_sum = n / 2.0
    best_h = initial_h.copy()
    best_c = compute_c5(best_h)
    current_h = best_h.copy()
    cur_c = best_c

    step_factor = 0.5          # initial magnitude of a transfer
    no_improve = 0
    it = 0
    while time.time() < time_limit:
        # correlation to locate the worst shift (largest C)
        corr = np.correlate(current_h, 1.0 - current_h, mode="full")
        idx_max = int(np.argmax(corr))
        k = idx_max - (n - 1)               # shift corresponding to the max

        # gradient of that worst shift
        g = gradient_for_shift(current_h, k)

        # indices to move mass from (positive gradient) to (negative gradient)
        i_inc = int(np.argmin(g))            # most negative => raise h_i
        i_dec = int(np.argmax(g))            # most positive => lower h_i

        # if gradient is flat, pick random pair
        if g[i_inc] >= -1e-12 or g[i_dec] <= 1e-12:
            i_inc, i_dec = np.random.choice(n, 2, replace=False)

        # feasible transfer amount respecting bounds
        delta_max = min(1.0 - current_h[i_inc], current_h[i_dec])
        if delta_max <= 0:
            # nothing to move, try a random perturbation
            i_inc, i_dec = np.random.choice(n, 2, replace=False)
            delta_max = min(1.0 - current_h[i_inc], current_h[i_dec])
            if delta_max <= 0:
                continue

        # stochastic step size
        delta = delta_max * step_factor * random.random()
        if delta <= 0:
            continue

        trial = current_h.copy()
        trial[i_inc] += delta
        trial[i_dec] -= delta

        # evaluate
        new_c = compute_c5(trial)

        if new_c < cur_c - 1e-12:
            # accept improvement
            current_h = trial
            cur_c = new_c
            no_improve = 0
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = trial.copy()
        else:
            no_improve += 1

        it += 1
        # anneal step size slowly
        if it % 500 == 0:
            step_factor = max(0.01, step_factor * 0.9)

        # occasional restart if stuck
        if no_improve > 2000:
            current_h = random_h(n)
            cur_c = compute_c5(current_h)
            no_improve = 0
            step_factor = max(0.1, step_factor * 0.9)

    # final projection to guarantee feasibility (tiny numerical drift may appear)
    best_h = project_to_capped_simplex(best_h, target_sum)
    best_c = compute_c5(best_h)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
    h_values is a NumPy array of the step function, c5_bound is the
    achieved C5 value, and n_points is the discretisation size.
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size and initial point
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
        # Ensure feasibility
        h0 = project_to_capped_simplex(h0, n / 2.0)
    else:
        n = 140                     # fallback size (even)
        h0 = random_h(n)

    start_time = time.time()
    # allocate most of the budget to the improvement routine
    end_time = start_time + budget_s - 0.5   # small safety margin
    best_h, best_c = improve_h(h0, end_time)

    return best_h, best_c, n
