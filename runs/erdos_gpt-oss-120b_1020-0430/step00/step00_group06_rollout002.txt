    # ensure v within bounds after projection
    # algorithm: treat unconstrained solution: v' = v - λ, then clip to [0,1].
    # We need λ such that sum(clip(v - λ,0,1)) = s.

        total = proj.sum()
        if total > s:
            lower = λ
        else:
            upper = λ
    return proj

def compute_gradient(h, shift):

    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        # i+shift in range: i up to n-1-shift

        term2[shift:] = h[:n-shift]

        s = -shift
        term1[s:] = 1 - h[:n-s]
        # i - shift = i + s
        term2[:n-s] = h[s:]
    grad = term1 - term2

import time
import numpy as np

def _project_capped_simplex(v, target_sum):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum}.
    Uses binary search on the Lagrange multiplier λ for the formula
    x = clip(v - λ, 0, 1).
    """
    lo = -np.max(v) - 1.0          # makes all entries after clipping equal 1
    hi = 1.0 - np.min(v) + 1.0    # makes all entries after clipping equal 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_max_corr(h):
    """
    Compute subgradient of the maximal overlap term
    g(k) = Σ_i h_i (1 - h_{i+k}) at the shift k attaining the maximum.
    Returns the gradient vector (size len(h)).
    """
    n = h.shape[0]
    # raw correlation (without dx factor) – we only need the argmax
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = np.argmax(corr) - (n - 1)   # lag corresponding to maximal C5
    grad = np.zeros_like(h)

    if k >= 0:
        # part where i+ k is inside the array
        grad[: n - k] += 1.0 - h[k:]
        # part where i - k is inside the array
        grad[k:]   -= h[: n - k]
    else:
        s = -k
        grad[s:]   += 1.0 - h[: n - s]
        grad[: n - s] -= h[s:]

    return grad, k


def _c5_value(h):
    """Evaluate C5 for a given vector h (including the dx factor)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)


def _random_feasible(rng, n, target_sum):
    """Generate a random vector in [0,1]^n with the required sum."""
    v = rng.rand(n)                 # uniform in [0,1)
    return _project_capped_simplex(v, target_sum)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] (discretised) that gives a small
    C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # 1. Initialise problem size and a feasible starting point
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
    else:
        n = 95
        h0 = np.zeros(n, dtype=np.float64)

    target_sum = n / 2.0               # because ∫ h = 1 and length = 2

    # Ensure feasibility (box + sum)
    h = _project_capped_simplex(h0, target_sum)
    cur_c5 = _c5_value(h)

    best_h = h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # 2. Parameters for the (projected) gradient descent / restart loop
    # ------------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s - 0.5   # small safety margin

    # step‑size schedule: start relatively large and decay linearly
    alpha0 = 0.8
    # counters for stagnation detection
    stagnation_counter = 0
    stagnation_limit = 4000      # iterations without improvement before a perturbation

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < end_time:
        # ---- compute gradient at the current most‑violating shift ----
        grad, _ = _gradient_max_corr(h)

        # ---- schedule a decreasing step size ----
        elapsed = time.time() - start_time
        progress = elapsed / budget_s               # in [0,1]
        alpha = max(alpha0 * (1.0 - progress), 1e-4)

        # ---- projected gradient step ----
        h_candidate = h - alpha * grad
        h_candidate = _project_capped_simplex(h_candidate, target_sum)

        cand_c5 = _c5_value(h_candidate)

        # ---- acceptance of the candidate ----
        if cand_c5 <= cur_c5:               # descent step
            h = h_candidate
            cur_c5 = cand_c5
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        # ---- keep track of the globally best vector ----
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = h_candidate.copy()

        # ---- occasional random perturbation to escape plateaus ----
        if stagnation_counter >= stagnation_limit:
            # perturb the best known solution a little and continue from there
            eps = 0.06
            perturbed = best_h + (rng.rand(n) - 0.5) * 2.0 * eps
            perturbed = np.clip(perturbed, 0.0, 1.0)
            h = _project_capped_simplex(perturbed, target_sum)
            cur_c5 = _c5_value(h)
            stagnation_counter = 0

        # ---- occasional full random restart (low probability) ----
        if rng.rand() < 0.0008:   # roughly one restart every ~1250 iterations
            h = _random_feasible(rng, n, target_sum)
            cur_c5 = _c5_value(h)
            stagnation_counter = 0

    # ------------------------------------------------------------------
    # 4. Return the best construction found
    # ------------------------------------------------------------------
    return best_h.tolist(), float(best_c5), n
