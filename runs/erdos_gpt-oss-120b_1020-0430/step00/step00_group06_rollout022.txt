def gradient_max_corr(h):
    N = len(h)
    dx = 2.0/N
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (N - 1)  # shift
    g = np.zeros_like(h)
    if s >= 0:
        # term (1 - h_{i+s}) for i in range 0..N-1-s
        i_range = slice(0, N - s)
        # first contribution (derivative from first factor)
        g[i_range] += 1 - h[i_range + s]  # need to handle addition with offset
        # second contribution (derivative from second factor)
        j_range = slice(s, N)
        g[j_range] -= h[j_range - s]
    else:

        # derivative from second factor (since shift negative)
        # conv(s) = sum_i h_{i+t} (1 - h_i), i from 0..N-1-t

        j_range = slice(t, N)
        g[j_range] += 1 - h[j_range - t]

def gradient_and_max(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argmax(corr)
    s = idx - (N - 1)  # integer shift
    # compute gradient vector g of same length
    g = np.zeros_like(h)
    if s >= 0:
        # i in [0, N-s-1]
        i_end = N - s
        # (1 - h[i+s]) term

        lo = -2.0
        hi = 2.0

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_box_sum(x, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(35):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x + lam, 0.0, 1.0)

def gradient_max_corr(h, dx):
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    s = idx - (N - 1)
    g = np.zeros_like(h)
    if s >= 0:
        # first term for positions 0..N-s-1
        g[:N - s] += (1.0 - h[s:])
        # second term for positions s..N-1
        g[s:] -= h[:N - s]
    else:
        t = -s
        g[:N - t] -= h[t:]
        g[t:] += (1.0 - h[:N - t])
    g *= dx
    max_val = corr[idx] * dx
    return g, max_val

def build_trapezoid(a, N, dx):
    a = max(0.0, min(1.0, a))
    x = (np.arange(N) + 0.5) * dx
    b = 2.0 - 2.0 * a
    half_plateau = b / 2.0
    c = 1.0 / (2.0 - a) if a != 2.0 else 1.0
    d = np.abs(x - 1.0)
    h = np.zeros_like(x)
    mask_plateau = d <= half_plateau
    h[mask_plateau] = c
    mask_slope = (d > half_plateau) & (d <= half_plateau + a)
    if a > 1e-12:
        excess = d[mask_slope] - half_plateau
        h[mask_slope] = c * (1.0 - excess / a)
    # project onto constraints
    target_sum = N / 2
    h = project_box_sum(h, target_sum)
    return h

def build_two_level(L, N, dx):

import numpy as np
import time

def compute_c5(h, dx):
    """Maximum overlap C5 for a step function h."""
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)


def project_box_sum(x, target_sum, lo=-2.0, hi=2.0, iters=35):
    """Project onto the cube [0,1]^n and the hyperplane sum(x)=target_sum."""
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x + lam, 0.0, 1.0)


def gradient_max_corr(h, dx):
    """Sub‑gradient of the maximal correlation for the current h."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # integer shift attaining the max
    g = np.zeros_like(h)

    if shift >= 0:
        g[: n - shift] += (1.0 - h[shift:])
        g[shift:]      -= h[: n - shift]
    else:
        t = -shift
        g[: n - t] -= h[t:]
        g[t:]       += (1.0 - h[: n - t])
    g *= dx
    max_val = corr[idx] * dx
    return g, max_val


def build_trapezoid(a, n, dx):
    """Symmetric trapezoid: slope length a (0≤a≤1), plateau height = 1/(2‑a)."""
    a = max(0.0, min(1.0, a))
    x = (np.arange(n) + 0.5) * dx
    b = 2.0 - 2.0 * a               # plateau length
    half_plateau = b / 2.0
    c = 1.0 / (2.0 - a) if a != 2.0 else 1.0   # plateau height (≤1)

    d = np.abs(x - 1.0)
    h = np.zeros_like(x)

    # plateau
    mask_plateau = d <= half_plateau
    h[mask_plateau] = c

    # slopes
    mask_slope = (d > half_plateau) & (d <= half_plateau + a)
    if a > 1e-12:
        excess = d[mask_slope] - half_plateau
        h[mask_slope] = c * (1.0 - excess / a)

    return project_box_sum(h, n / 2)


def build_two_level(L, n, dx):
    """Two‑level function: height 1 on a central block of length L (0≤L≤1),
    constant height c elsewhere to keep total integral 1."""
    L = max(0.0, min(1.0, L))
    halfL = L / 2.0
    c = (1.0 - L) / (2.0 - L) if L < 2.0 else 0.0
    x = (np.arange(n) + 0.5) * dx
    h = np.full(n, c)
    h[np.abs(x - 1.0) <= halfL] = 1.0
    return project_box_sum(h, n / 2)


def local_optimize(start_h, n, dx, max_iters, deadline):
    """Projected sub‑gradient descent from start_h."""
    target = n / 2
    h = start_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    lr = 0.5

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        g, cur_val = gradient_max_corr(h, dx)
        x = h - lr * g
        h = project_box_sum(x, target)
        new_val = compute_c5(h, dx)

        if new_val < best_val:
            best_val = new_val
            best_h = h.copy()

        # simple step‑size adaptation
        if new_val < cur_val:
            lr = min(lr * 1.02, 1.0)
        else:
            lr = max(lr * 0.6, 1e-4)

    return best_h, best_val
