                j = int(rng.choice(idx_j))

                current[i], current[j] = 0.0, 1.0

            no_improve = 0

    return (best_h.tolist(), float(final_score), n)

import time, math, random
import numpy as np

def _corr_counts(vec):
    """Full cross‑correlation of h with (1‑h). Returns integer counts."""
    return np.correlate(vec, 1 - vec, mode='full')

def _score(vec, dx):
    """C₅ value for a given vector."""
    return np.max(_corr_counts(vec)) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # ------------------------------------------------------------------ #
    # 1. Determine problem size and initialise a feasible step function
    # ------------------------------------------------------------------ #
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.shape[0]
    else:
        n = 70
        init = None

    target_sum = n / 2                # ∫h = 1  →  Σh = n/2
    dx = 2.0 / n

    if init is not None:
        h = np.clip(init, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target_sum) > 1e-8:
            # simple scaling + final correction
            h = h * (target_sum / cur)
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - h.sum()
            h[0] += diff
            h = np.clip(h, 0.0, 1.0)
    else:
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, int(target_sum), replace=False)
        h[ones] = 1.0

    # Turn the starting vector into a strict 0/1 design (required for swaps)
    if not np.all(np.logical_or(h == 0, h == 1)):
        r = np.rint(h).astype(int)
        diff = int(target_sum) - r.sum()
        if diff > 0:
            zeros = np.where(r == 0)[0]
            if zeros.size >= diff:
                idx = rng.choice(zeros, diff, replace=False)
                r[idx] = 1
        elif diff < 0:
            ones = np.where(r == 1)[0]
            if ones.size >= -diff:
                idx = rng.choice(ones, -diff, replace=False)
                r[idx] = 0
        h = r.astype(float)

    # ------------------------------------------------------------------ #
    # 2. Quick check: maybe the supplied construction already meets target
    # ------------------------------------------------------------------ #
    best_h = h.copy()
    best_score = _score(best_h, dx)
    if best_score <= 0.38085:
        return (best_h.tolist(), float(best_score), n)

    # ------------------------------------------------------------------ #
    # 3. Optimisation: greedy repairs on the worst lag + simulated annealing
    # ------------------------------------------------------------------ #
    cur_h = best_h.copy()
    cur_score = best_score

    start = time.time()
    T = 0.02                # initial temperature for SA
    decay = 0.99995
    no_improve = 0
    max_no_improve = 20000

    while time.time() - start < budget_s * 0.95:
        # temperature schedule
        T *= decay

        # current correlation data
        corr = _corr_counts(cur_h)
        max_cnt = int(corr.max())
        lag = int(corr.argmax() - (n - 1))   # lag in [-(n-1), …, n-1]

        # -------------------------------------------------------------- #
        # Greedy fix of the currently worst lag (if possible)
        # -------------------------------------------------------------- #
        improved = False
        if max_cnt > 0:
            if lag >= 0:
                L = lag
                # indices i with (h[i]=1, h[i+L]=0)
                S1 = np.where((cur_h[:n - L] == 1) & (cur_h[L:] == 0))[0]
                # indices i with (h[i]=0, h[i+L]=1)
                S0 = np.where((cur_h[:n - L] == 0) & (cur_h[L:] == 1))[0]
            else:
                L = -lag
                # shift left by L
                S1 = np.where((cur_h[L:] == 1) & (cur_h[:n - L] == 0))[0] + L
                S0 = np.where((cur_h[L:] == 0) & (cur_h[:n - L] == 1))[0] + L

            if S1.size and S0.size:
                i = int(rng.choice(S1))
                j = int(rng.choice(S0))
                cur_h[i], cur_h[j] = 0.0, 1.0
                cur_score = _score(cur_h, dx)
                improved = True
                no_improve = 0
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_h = cur_h.copy()
                    if best_score <= 0.3808:
                        break

        # -------------------------------------------------------------- #
        # If greedy step not possible, use a random swap (with SA)
        # -------------------------------------------------------------- #
        if not improved:
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            cand = cur_h.copy()
            cand[i], cand[j] = 0.0, 1.0
            cand_score = _score(cand, dx)
            delta = cand_score - cur_score
            if delta < -1e-12:
                cur_h = cand
                cur_score = cand_score
                no_improve = 0
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best_h = cand.copy()
                    if best_score <= 0.3808:
                        break
            else:
                # Metropolis acceptance
                if T > 0 and rng.random() < math.exp(-delta / T):
                    cur_h = cand
                    cur_score = cand_score
                else:
                    no_improve += 1

        # -------------------------------------------------------------- #

        # -------------------------------------------------------------- #
        if no_improve >= max_no_improve:
            cur_h = best_h.copy()
            cur_score = best_score
            for _ in range(5):

                cur_h[i], cur_h[j] = 0.0, 1.0
            cur_score = _score(cur_h, dx)
            no_improve = 0

    final_score = best_score
    return (best_h.tolist(), float(final_score), n)
