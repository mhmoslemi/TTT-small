def gradient_max_shift(h):
    n = len(h)
    conv = np.convolve(h, (1 - h)[::-1], mode='full')
    t_max = np.argmax(conv)
    s = t_max - (n - 1)  # shift such that conv[t] = sum_i h[i] * (1 - h[i - s])
    # compute g
    g = np.zeros_like(h)
    # vectorized
    # term1: (1 - h[i - s]) for i where i - s in [0, n-1]
    i = np.arange(n)
    i_minus_s = i - s
    mask1 = (i_minus_s >= 0) & (i_minus_s < n)
    g[mask1] += (1 - h[i_minus_s[mask1]])
    # term2: -h[i + s] for i where i + s in range
    i_plus_s = i + s
    mask2 = (i_plus_s >= 0) & (i_plus_s < n)
    g[mask2] -= h[i_plus_s[mask2]]
    return g, conv[t_max]

def solve_milp(n, solver_list=('CBC','GLPK_MI','SCIP')):
    import cvxpy as cp, time
    x = cp.Variable(n, boolean=True)
    constraints = [cp.sum(x) == n/2]
    t = cp.Variable()
    constraints.append(t >= 0)
    for s in range(1, n):
        length = n - s
        y_s = cp.Variable(length, boolean=True)
        # product constraints
        constraints.append(y_s <= x[s:])
        constraints.append(y_s <= x[:length])
        constraints.append(y_s >= x[s:] + x[:length] - 1)
        constraints.append(t >= (n/2) - cp.sum(y_s))
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in solver_list:
        try:
            prob.solve(solver=solver, warm_start=True, feastol=1e-6, abstol=1e-6, reltol=1e-6, max_iters=5000)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                break
        except Exception:
            continue
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        # fallback maybe no solution
        raise RuntimeError("MILP not solved")
    x_opt = np.round(x.value).astype(int)
    t_opt = t.value
    # compute c5 bound
    dx = 2.0 / n
    c5 = t_opt * dx
    return x_opt, c5

def solve_milp_binary(n, total_budget, start_time):
    import cvxpy as cp
    # set time limit per solve maybe total_budget - (time.time()-start_time) - small.
    # but cvxpy doesn't have direct time limit; we rely on solver speed.
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == n/2, t >= 0]
    for s in range(1, n):

        y = cp.Variable(length, boolean=True)

        constraints.append(y <= x[:length])
        constraints.append(y >= x[s:] + x[:length] - 1)

import time, random, math
import numpy as np

def _c5_val(h, dx):
    """max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    conv = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return np.max(conv) * dx

def _solve_milp_binary(n):
    """Exact binary solution via MILP (if a MILP solver is available)."""
    import cvxpy as cp

    # binary variables
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()                     # the maximal overlap we minimise
    constraints = [cp.sum(x) == n / 2, t >= 0]

    # for every non‑zero shift we linearise the product x_i·x_{i‑s}
    for s in range(1, n):
        L = n - s
        y = cp.Variable(L, boolean=True)               # y_i = x_i·x_{i+s}
        constraints += [
            y <= x[s:],                                 # y_i ≤ x_{i+s}
            y <= x[:L],                                 # y_i ≤ x_i
            y >= x[s:] + x[:L] - 1,                     # y_i ≥ x_{i+s}+x_i-1
            t >= cp.sum(x[s:]) - cp.sum(y),             # shift  +s
            t >= cp.sum(x[:L]) - cp.sum(y)              # shift  -s
        ]

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try a few open‑source MILP solvers
    for solver in ('CBC', 'GLPK_MI', 'SCIP', 'ECOS_BB'):
        try:
            prob.solve(solver=solver, warm_start=True, verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                break
        except Exception:
            continue

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        raise RuntimeError('MILP not solved')

    xv = np.rint(x.value).astype(int)

    # enforce exact half‑weight (round‑off safety)
    target = n // 2
    diff = target - xv.sum()
    if diff != 0:
        zeros = np.where(xv == 0)[0]
        ones = np.where(xv == 1)[0]
        if diff > 0:
            sel = np.random.choice(zeros, size=diff, replace=False)
            xv[sel] = 1
        else:
            sel = np.random.choice(ones, size=-diff, replace=False)
            xv[sel] = 0

    return xv, float(t.value)


def _hill_climb_binary(n, dx, time_limit):
    """Simple stochastic hill‑climbing for a binary step function."""
    # start from a random balanced vector
    h = np.zeros(n, dtype=int)
    h[np.random.choice(n, n // 2, replace=False)] = 1

    best_h = h.copy()
    best_c5 = _c5_val(best_h, dx)
    start = time.time()

    while time.time() - start < time_limit:
        # current worst shift
        conv = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
        max_idx = int(np.argmax(conv))
        s = max_idx - (n - 1)

        # indices that contribute to the worst overlap
        if s >= 0:
            rng = np.arange(s, n)
            mask_one = (h[rng] == 1) & (h[rng - s] == 0)
            mask_zero = (h[rng] == 0) & (h[rng - s] == 1)
            idx_one = rng[mask_one]
            idx_zero = rng[mask_zero]
        else:
            s = -s
            rng = np.arange(0, n - s)
            mask_one = (h[rng] == 1) & (h[rng + s] == 0)
            mask_zero = (h[rng] == 0) & (h[rng + s] == 1)
            idx_one = rng[mask_one]
            idx_zero = rng[mask_zero]

        if idx_one.size and idx_zero.size:
            i = int(np.random.choice(idx_one))
            j = int(np.random.choice(idx_zero))
        else:
            # generic opposite‑value swap
            i = int(np.random.choice(np.flatnonzero(h == 1)))
            j = int(np.random.choice(np.flatnonzero(h == 0)))

        # propose swap
        h[i], h[j] = h[j], h[i]
        new_c5 = _c5_val(h, dx)

        # keep if not worse; otherwise revert
        if new_c5 <= best_c5 + 1e-12:
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

    return best_h, best_c5
