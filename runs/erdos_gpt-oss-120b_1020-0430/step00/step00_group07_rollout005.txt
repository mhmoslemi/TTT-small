def proj_box_simplex(y, target):
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo+hi)/2, 0, 1)
    # due to rounding may be off a tiny amount; adjust by adding small to some entries.
    # We'll enforce sum exactly by adjusting if needed:
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff among non-saturated entries randomly
        # find indices where 0 < x_i < 1
        idx = np.where((x > 1e-12) & (x < 1-1e-12))[0]
        if idx.size > 0:
            # add diff equally
            add = diff / idx.size
            x[idx] += add
            # Clip again
            x = np.clip(x, 0, 1)
        else:
            # all saturated, adjust by adding to ones (or subtract from ones)
            # Not likely
            pass
    return x

def project_box_simplex(y, target):
    lo = -np.max(y) - 1.0   # ensures all y - lo >= 1
    hi = np.max(y) + 1.0    # ensures all y - hi <= 0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
    # due to numeric issues may be slight off
    # adjust
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff to elements not at bounds
        # find indices with 0 < x < 1
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size > 0:
            add = diff / free.size
            x[free] += add
            x = np.clip(x, 0, 1)
        else:
            # all at bounds, adjust using projection onto set with min distance may not be possible
            # fallback: just adjust first element within bounds or ignore
            # but this case unlikely.
            pass
    return x

import numpy as np, time, random, math

def project_box_simplex(y, target):
    # Euclidean projection onto set {x: 0<=x_i<=1, sum x_i = target}
    # binary search on lambda.
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    # correct possible tiny residual difference
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # distribute to non-saturated entries
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        idx = np.where(mask)[0]
        if idx.size > 0:
            add = diff / idx.size
            x[idx] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            # all at bounds, adjust smallest deviation
            # find index where adding diff will not violate bounds
            if diff > 0:
                # need to increase sum, find element <1
                cand = np.where(x < 1.0 - 1e-12)[0]
                if cand.size:
                    x[cand[0]] += diff
            else:
                cand = np.where(x > 1e-12)[0]
                if cand.size:
                    x[cand[0]] += diff
            x = np.clip(x, 0.0, 1.0)
    return x

def objective_val(h, dx):
    # compute C5 bound for step vector h
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def gradient_active_lag(h, L):
    n = h.shape[0]
    i = np.arange(n)
    g = np.zeros(n)
    plus = i + L
    mask = (plus >= 0) & (plus < n)
    g[mask] += 1.0 - h[plus[mask]]
    minus = i - L
    mask2 = (minus >= 0) & (minus < n)
    g[mask2] -= h[minus[mask2]]
    return g

import time, random, math
import numpy as np

def _project_box_simplex(y: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection onto {x | 0<=x_i<=1, sum x_i = target}."""
    lo = -np.max(y) - 1.0          # all entries become 1
    hi = np.max(y) + 1.0           # all entries become 0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)

    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            add = diff / idx.size
            x[idx] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            # all saturated – make a tiny correction on a feasible coordinate
            if diff > 0:
                cand = np.where(x < 1.0 - 1e-12)[0]
                if cand.size:
                    x[cand[0]] += diff
            else:
                cand = np.where(x > 1e-12)[0]
                if cand.size:
                    x[cand[0]] += diff
            x = np.clip(x, 0.0, 1.0)
    return x


def _objective(h: np.ndarray, dx: float) -> float:
    """C₅ value for a step vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _grad_active_lag(h: np.ndarray, L: int) -> np.ndarray:
    """Sub‑gradient of the correlation term for the lag L."""
    n = h.shape[0]
    i = np.arange(n)
    g = np.zeros(n)

    plus = i + L
    mask = (plus >= 0) & (plus < n)
    g[mask] += 1.0 - h[plus[mask]]

    minus = i - L
    mask2 = (minus >= 0) & (minus < n)
    g[mask2] -= h[minus[mask2]]
    return g


def _subgradient_descent(
    h0: np.ndarray,
    target_sum: float,
    dx: float,
    max_iter: int,
    time_deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient minimisation."""
    h = h0.copy()
    best_h = h.copy()
    best_val = _objective(h, dx)
    n = h.shape[0]
    step0 = 0.5

    for it in range(max_iter):
        if time.time() - start_time > time_deadline:
            break

        corr_full = np.correlate(h, 1.0 - h, mode="full")
        cur_val = corr_full.max() * dx
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()

        idx = int(np.argmax(corr_full))
        L = idx - (n - 1)

        g = _grad_active_lag(h, L)
        step = step0 / math.sqrt(it + 1.0)
        y = h - step * g
        h = _project_box_simplex(y, target_sum)

        # tiny random jitter to escape plateaux
        if random.random() < 0.01:
            j = random.randrange(n)
            delta = (random.random() - 0.5) * 0.1
            y = h.copy()
            y[j] = np.clip(y[j] + delta, 0.0, 1.0)
            h = _project_box_simplex(y, target_sum)

    return best_h, best_val


def _hill_climb(
    h_start: np.ndarray,
    target_sum: float,
    dx: float,
    time_deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Simple pair‑wise mass transfer hill climbing."""
    h = h_start.copy()
    best_h = h.copy()
    best_val = _objective(h, dx)
    n = h.shape[0]

    while time.time() - start_time < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose moving a mass delta from i (donor) to j (receiver)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = (np.random.rand() - 0.5) * 0.2  # signed move
        if delta >= 0:
            d = min(delta, max_delta)
            donor, rec = i, j
        else:
            d = min(-delta, min(h[j], 1.0 - h[i]))
            donor, rec = j, i
        if d <= 0:
            continue

        cand = h.copy()
        cand[donor] -= d
        cand[rec] += d
        cand_val = _objective(cand, dx)
        cur_val = _objective(h, dx)

        if cand_val < cur_val - 1e-12:
            h = cand
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_h = cand.copy()
        else:
            # occasional acceptance to diversify
            if random.random() < 0.001:
                h = cand

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function that (approximately) minimises the Erdős C₅ bound."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------ #

    # ------------------------------------------------------------------ #
    if "initial_h_values" in globals():

        h0 = _project_box_simplex(y, target)

    dx = 2.0 / n
    best_h = h0.copy()
    best_val = _objective(best_h, dx)

    # ------------------------------------------------------------ #
    # 2.  Main optimisation loop (sub‑gradient + hill climbing).
    # ------------------------------------------------------------ #
    # keep a small safety margin for the final post‑processing.
    main_deadline = start + budget_s * 0.95

    while time.time() < main_deadline:
