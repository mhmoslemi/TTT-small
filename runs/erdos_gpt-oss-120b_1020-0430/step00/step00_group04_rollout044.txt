def propose_move(h):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    # compute permissible delta range
    max_up = 1.0 - h[i]  # how much we can increase i
    max_down = h[i]      # how much we can decrease i
    # Actually we propose delta to add to i and subtract from j:
    # delta can be positive (increase i, decrease j) or negative (decrease i, increase j)
    # compute max positive delta = min(max_up, h[j]) because j must not go below 0.
    # compute max negative delta = -min(max_down, 1 - h[j]) (i.e., decreasing i and increasing j)
    pos_delta_max = min(max_up, h[j])
    neg_delta_max = -min(max_down, 1.0 - h[j])
    # choose delta uniformly in [neg_delta_max, pos_delta_max]
    if pos_delta_max == neg_delta_max == 0:
        return None
    delta = rng.uniform(neg_delta_max, pos_delta_max)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # sum stays same; bounds satisfied due to delta selection
    return new_h, i, j, delta

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_initial(h):
    # ensure sum = n/2 exactly, adjust small diff
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # distribute diff uniformly while staying in bounds
    # try to add diff/n to all:
    h = h + diff / n
    # clip to [0,1] and reproject, loop a few times
    for _ in range(5):
        # compute sum after clipping
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine number of points and initial vector
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
    except NameError:
        # default length
        n = 60
        # binary start: exactly half 1's
        h0 = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, size=n//2, replace=False)
        h0[ones] = 1.0
    # Ensure feasibility
    h0 = repair_initial(h0)
    cur_h = h0.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    target_sum = n / 2.0
    max_temp = 0.0125   # initial temperature
    # set maximum delta size initially
    max_delta_initial = 0.35
    # counters
    iter_cnt = 0
    no_improve = 0

    # Main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:
            break
        # Temperature schedule (linearly decreasing)
        fraction = elapsed / budget_s
        temp = max_temp * (1.0 - fraction)
        # delta magnitude schedule
        max_delta = max_delta_initial * (1.0 - fraction) + 0.03  # never below small
        # propose a move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # compute allowed delta
        # increase i, decrease j case
        max_up_i = 1.0 - cur_h[i]
        max_down_i = cur_h[i]
        # When delta positive: we add to i, subtract from j
        pos_max = min(max_up_i, cur_h[j], max_delta)
        # When delta negative: subtract from i, add to j
        neg_max = -min(max_down_i, 1.0 - cur_h[j], max_delta)
        if pos_max == 0 and neg_max == 0:
            continue
        delta = rng.uniform(neg_max, pos_max)
        if delta == 0.0:
            continue
        # Apply move
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # Due to floating errors, ensure bounds
        new_h[i] = min(max(new_h[i], 0.0), 1.0)
        new_h[j] = min(max(new_h[j], 0.0), 1.0)
        # compute new objective
        new_val = compute_c5(new_h)
        improve = new_val < cur_val
        if improve or (temp > 0 and rng.random() < math.exp(-(new_val - cur_val) / temp)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        iter_cnt += 1

        # Periodic random restart if stagnation
        if no_improve > 5000:
            # create new random feasible vector
            # random binary with half ones, then jitter
            new_vec = np.zeros(n, dtype=np.float64)
            ones = rng.choice(n, size=n//2, replace=False)
            new_vec[ones] = 1.0
            # add small noise and project to bounds
            noise = (rng.random(n) - 0.5) * 0.02
            new_vec = np.clip(new_vec + noise, 0.0, 1.0)
            new_vec = repair_initial(new_vec)
            cur_h = new_vec
            cur_val = compute_c5(cur_h)
            no_improve = 0

        # Occasionally try a larger reshuffle to explore
        if iter_cnt % 20000 == 0:

def guided_move(h, worst_lag):
    n = len(h)
    offset = abs(worst_lag)
    if offset == 0 or offset >= n:
        # no guided move
        return None
    # identify low pair
    low_vals = h[:n-offset] + h[offset:]  # sum of pair (i,i+offset)
    high_vals = low_vals  # same
    i_low = np.argmin(low_vals)
    i_high = np.argmax(high_vals)
    # compute max delta
    max_up_low = min(1.0 - h[i_low], 1.0 - h[i_low+offset])
    max_down_high = min(h[i_high], h[i_high+offset])
    delta_max = min(max_up_low, max_down_high)
    if delta_max <= 1e-12:
        return None
    delta = delta_max * 0.5  # half the maximum
    new_h = h.copy()
    new_h[i_low] += delta
    new_h[i_low+offset] += delta
    new_h[i_high] -= delta
    new_h[i_high+offset] -= delta
    return new_h
```

import numpy as np, time, math

def _c5_value(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx for the discrete vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); np.correlate uses reversed second argument,
    # which is exactly the desired cross‑correlation with zero padding.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _repair_initial(h: np.ndarray) -> np.ndarray:
    """Project onto the feasible set 0≤h≤1, sum(h)=n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative projection: add the same correction to all entries,
    # then clip again. Converges very quickly for our tiny n.
    for _ in range(5):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _guided_move(h: np.ndarray, lag: int, max_delta: float, rng) -> np.ndarray | None:
    """
    Perform a move that directly tries to improve the worst lag.
    Increase two low‑value positions (distance |lag| apart) and
    decrease two high‑value positions the same distance apart.
    Returns a new feasible vector or None if no sensible move exists.
    """
    n = h.shape[0]
    offset = abs(lag)
    if offset == 0 or offset >= n:
        return None

    # pairwise sums for the given offset
    pair_sum = h[: n - offset] + h[offset :]

    i_low = int(np.argmin(pair_sum))
    i_high = int(np.argmax(pair_sum))
    if i_low == i_high:
        # try a second best candidate
        low_candidates = np.argsort(pair_sum)[:2]
        high_candidates = np.argsort(pair_sum)[-2:]
        i_low = int(low_candidates[0])
        i_high = int(high_candidates[-1])
        if i_low == i_high:
            return None

    # compute maximal feasible delta respecting the box constraints
    up_low = min(1.0 - h[i_low], 1.0 - h[i_low + offset])
    down_high = min(h[i_high], h[i_high + offset])
    delta_max = min(up_low, down_high, max_delta)
    if delta_max <= 1e-12:
        return None

    delta = delta_max * 0.5  # be conservative
    new_h = h.copy()
    new_h[i_low] += delta
    new_h[i_low + offset] += delta
    new_h[i_high] -= delta
    new_h[i_high + offset] -= delta
    return new_h
