   best_score = compute_c5(best_h)
   current_h = best_h.copy()
   current_score = best_score

def perturb(h, rng, max_step):
    i, j = rng.choice(n, size=2, replace=False)
    # compute maximum transferable from i to j without violating bounds
    # we can add epsilon positive or negative; choose sign randomly.
    # Let's decide direction: move from i to j.
    # Choose epsilon uniformly in [-max_step, max_step] but limited by bounds.
    epsilon = rng.uniform(-max_step, max_step)
    # attempt to add epsilon to h[i], subtract from h[j]
    # Actually we want h_i += epsilon, h_j -= epsilon.
    hi_new = h[i] + epsilon
    hj_new = h[j] - epsilon
    # clip ensure within [0,1]
    if hi_new < 0:
        epsilon = -h[i]
        hi_new = 0
        hj_new = h[j] - epsilon
    elif hi_new > 1:
        epsilon = 1 - h[i]
        hi_new = 1
        hj_new = h[j] - epsilon
    # similarly clip j
    if hj_new < 0:
        # need adjust epsilon so that j remains >=0
        # revert; compute max epsilon allowed: epsilon <= h[j] (since we subtract epsilon from j)
        # Actually if epsilon positive, we subtract positive from j, making j smaller; this could go negative if epsilon > h[j].
        # So adjust epsilon = min(epsilon, h[j]) for epsilon>0; for epsilon<0, we add to j which is fine.
        if epsilon > 0:
            epsilon = min(epsilon, h[j])
            hi_new = h[i] + epsilon
            hj_new = h[j] - epsilon
        else:
            # epsilon negative => we add -epsilon to j (increase j) which cannot exceed 1
            max_inc = 1 - h[j]
            epsilon = max(epsilon, -max_inc) # ensure not exceed bound
            hi_new = h[i] + epsilon
            hj_new = h[j] - epsilon
    # after adjust ensure still within bounds
    # Then assign
    h_new = h.copy()
    h_new[i] = hi_new
    h_new[j] = hj_new
    return h_new

def compute_max_corr(h):
    n = len(h)
    M = 1 << (2*n - 1).bit_length()
    H = np.fft.rfft(h, n=M)
    BH = np.fft.rfft(1 - h, n=M)
    corr = np.fft.irfft(H * np.conj(BH), n=M)[:2*n-1]  # real correlation
    return np.max(corr) * (2.0 / n)

def hill_climb_binary(n, rng, max_iters, initial_h):
    h = initial_h.copy()
    # ensure binary
    h = (h > 0.5).astype(float)
    # ensure sum = n/2; if not, adjust.
    # If sum != n/2, we can randomly flip bits to match.
    sum_h = int(np.sum(h))
    target = n//2
    while sum_h > target:
        # turn a 1 into 0
        i = rng.choice(np.where(h>0.5)[0])
        h[i] = 0
        sum_h -= 1
    while sum_h < target:
        i = rng.choice(np.where(h<0.5)[0])
        h[i] = 1
        sum_h += 1

    ones = np.flatnonzero(h==1)
    zeros = np.flatnonzero(h==0)

    # Compute initial C
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score

    for it in range(max_iters):
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        # compute new C
        new_score = compute_c5(h)
        # acceptance
        # temperature schedule
        T = max(0.01 * (1 - it / max_iters), 0.0001)  # adapt
        if new_score < current_score or rng.random() < np.exp((current_score - new_score) / T):
            # accept
            current_score = new_score
            # update indices arrays: swap positions in ones and zeros lists:
            # i becomes 0 now, j becomes 1
            # To update efficiently, we can just recompute ones, zeros each iteration (cost O(n)) but ok.
            # Simpler: recompute after each swap.
            ones = np.flatnonzero(h==1)
            zeros = np.flatnonzero(h==0)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            # no need to update ones zeros (they remain same)
        # early break if best_score below target e.g., 0.3808
        if best_score <= 0.3808:
            break
    return best_h, best_score

def project_to_bounds_and_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    # define tolerance
    if abs(total - target) < 1e-12:
        return v
    if total < target:
        # increase
        excess = target - total
        # indices not at 1
        free = np.where(v < 1.0)[0]
        while excess > 1e-12 and len(free) > 0:
            inc = excess / len(free)
            # but ensure we don't exceed 1
            add = np.minimum(inc, 1.0 - v[free])
            v[free] += add
            excess = target - v.sum()
            # update free
            free = free[v[free] < 1.0 - 1e-12]
        # if still not reached due to rounding, we may need small adjustments
    else:
        # total > target: decrease
        deficit = total - target
        free = np.where(v > 0.0)[0]
        while deficit > 1e-12 and len(free) > 0:
            dec = deficit / len(free)
            sub = np.minimum(dec, v[free])
            v[free] -= sub
            deficit = v.sum() - target
            free = free[v[free] > 1e-12]
    # final clip and adjust minor
    # After loop, some rounding may remain; we can adjust by adding small delta to first index with capacity.
    diff = v.sum() - target
    if abs(diff) > 1e-10:
        # adjust by distributing diff across first free index
        if diff > 0:
            i = np.where(v > 0)[0][0]
            v[i] -= diff
        else:
            i = np.where(v < 1)[0][0]
            v[i] -= diff  # diff negative => add
    v = np.clip(v, 0, 1)
    return v

def compute_c5(h):
    n = len(h)
    M = 1 << (2*n - 1).bit_length() # next power of two >= 2n-1
    H = np.fft.rfft(h, n=M)

def project_to_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if np.isclose(total, target):
        return v
    if total < target:
        # need to add amount
        deficit = target - total
        # Indices where v < 1
        free = np.where(v < 1.0)[0]
        while deficit > 1e-12 and free.size > 0:
            inc = deficit / free.size
            # compute possible increase each
            add = np.minimum(inc, 1.0 - v[free])
            v[free] += add
            deficit = target - v.sum()
            free = free[v[free] < 1.0 - 1e-12]
    else:
        surplus = total - target
        free = np.where(v > 0.0)[0]
        while surplus > 1e-12 and free.size > 0:
            dec = surplus / free.size
            sub = np.minimum(dec, v[free])
            v[free] -= sub
            surplus = v.sum() - target
            free = free[v[free] > 1e-12]
    # final corrections in case of rounding
    diff = v.sum() - target
    if abs(diff) > 1e-9:
        # adjust first element that can be modified
        if diff > 0:
            idx = np.where(v > 0)[0][0]
            v[idx] -= diff
        else:
            idx = np.where(v < 1)[0][0]
            v[idx] -= diff  # diff negative, adds
    # Ensure final bounds
    v = np.clip(v, 0.0, 1.0)
    return v

def simulated_annealing(h, rng, max_steps=20000):

    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    n = len(cur)

def project_to_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if np.isclose(total, target):
        return v
    if total < target:
        # need to increase
        deficit = target - total
        # identify indices where v < 1
        free = np.where(v < 1.0)[0]
        # while deficit > 0 etc.
        while deficit > 1e-12 and free.size > 0:
            inc = deficit / free.size
            # compute actual increase up to bound
            add = np.minimum(inc, 1.0 - v[free])
            v[free] += add
            deficit = target - v.sum()
            free = free[v[free] < 1.0 - 1e-12]
    else:
        surplus = total - target
        free = np.where(v > 0.0)[0]
        while surplus > 1e-12 and free.size > 0:
            dec = surplus / free.size
            sub = np.minimum(dec, v[free])
            v[free] -= sub
            surplus = v.sum() - target
            free = free[v[free] > 1e-12]
    # final correction if still off by tiny amount
    diff = v.sum() - target
    if abs(diff) > 1e-9:
        # adjust first element that can be changed
        if diff > 0:
            idx = np.where(v > 0)[0][0]
            v[idx] -= diff
        else:
            idx = np.where(v < 1)[0][0]
            v[idx] -= diff
    # final clip
    v = np.clip(v, 0.0, 1.0)
    return v

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to obtain exact sum `target`."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if np.isclose(total, target, atol=1e-12):
        return v
    if total < target:
        deficit = target - total
        free = np.where(v < 1.0)[0]
        while deficit > 1e-12 and free.size:
            inc = deficit / free.size
            add = np.minimum(inc, 1.0 - v[free])
            v[free] += add
            deficit = target - v.sum()
            free = free[v[free] < 1.0 - 1e-12]
    else:
        surplus = total - target
        free = np.where(v > 0.0)[0]
        while surplus > 1e-12 and free.size:
            dec = surplus / free.size
            sub = np.minimum(dec, v[free])
            v[free] -= sub
            surplus = v.sum() - target
            free = free[v[free] > 1e-12]
    # final tiny correction
    diff = v.sum() - target

            idx = np.where(v > 0)[0][0]
            v[idx] -= diff
        else:
            idx = np.where(v < 1)[0][0]
            v[idx] -= diff
    return np.clip(v, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Compute C₅ = max_k ∫ h(x)(1‑h(x+k))dx via FFT correlation."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, n=m)
    B = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * np.conj(B), n=m)[: 2 * n - 1]
    return float(np.max(corr.real) * (2.0 / n))

            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        else:
            # move mass from j to i
            delta_max = min(max_transfer, cur[j], 1.0 - cur[i])
            if delta_max <= 0.0:
                continue
            delta = rng.random() * delta_max
            new = cur.copy()
            new[i] += delta
            new[j] -= delta

        new_score = _c5_value(new)

        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        if best_score <= 0.3808:
            break
    return best, best_score


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # discretisation parameters
    # --------------------------------------------------------
    n = 200                     # even → sum must be n/2
    target_sum = n // 2
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx   # mid‑point abscissae

    # --------------------------------------------------------
    # start from the simple V‑shaped construction (c = 0)
    # --------------------------------------------------------
    best_h = np.abs(x - 1.0)
    best_h = _project_to_sum(best_h, target_sum)
    best_score = _c5_value(best_h)

    # --------------------------------------------------------
    # incorporate a possible parent construction
    # --------------------------------------------------------
    if 'initial_h_values' in globals():
        try:
            init = np.asarray(initial_h_values, dtype=float)
            if init.shape[0] != n:
                old_n = init.shape[0]
                x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
                init = np.interp(x, x_old, init)
            init = np.clip(init, 0.0, 1.0)
            init = _project_to_sum(init, target_sum)
            sc = _c5_value(init)
            if sc < best_score:
                best_h = init.copy()
                best_score = sc
        except Exception:
            pass

    # --------------------------------------------------------
    # random search over the 1‑parameter family h(x)=c+(1‑2c)|x‑1|
    # --------------------------------------------------------
    for _ in range(3000):
        c = rng.random() * 0.5                     # 0 ≤ c ≤ 0.5
        h = c + (1.0 - 2.0 * c) * np.abs(x - 1.0)   # V‑shape with centre raised
        h = _project_to_sum(h, target_sum)
        sc = _c5_value(h)
        if sc < best_score:
            best_h = h.copy()
            best_score = sc

    # --------------------------------------------------------
    # main optimisation loop (repeated annealing runs)
    # --------------------------------------------------------
    start_time = time.time()
    while time.time() - start_time < budget_s - 2:
        # choose a starting point
        if rng.random() < 0.6:
            cur = best_h.copy()
            # a few random small mass moves to diversify
            for _ in range(10):
                i, j = rng.choice(n, size=2, replace=False)
                max_move = min(0.05, cur[i], 1.0 - cur[j])
                if max_move > 0:
                    delta = rng.random() * max_move
                    cur[i] -= delta
                    cur[j] += delta
        else:
            cur = np.zeros(n, dtype=float)
            cur[:target_sum] = 1.0
            rng.shuffle(cur)

        # allocate a modest number of annealing steps; respect remaining time
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        steps = int(min(50000, max(2000, remaining * 2500)))  # heuristic
        cand_h, cand_score = _anneal_continuous(cur, rng, steps)

        if cand_score < best_score:
            best_h = cand_h.copy()
            best_score = cand_score

        if best_score <= 0.3808:
            break

    # final projection (safety) and exact evaluation
    best_h = _project_to_sum(best_h, target_sum)
    best_score = _c5_value(best_h)

    return best_h.tolist(), float(best_score), n
