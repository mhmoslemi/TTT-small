def compute_cost(h):
    corr = np.correlate(h, 1 - h, mode='full')
    # multiply by dx
    return corr.max() * (2.0 / len(h))

def propose_move(h, step_scale):
    n = len(h)
    i, j = np.random.randint(0, n, size=2)
    # Ensure i != j:
    while j == i:
        j = np.random.randint(0, n)
    # Choose direction random sign
    if np.random.rand() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    # compute max delta:
    max_delta = min(step_scale, h[src], 1 - h[dst])
    if max_delta <= 0:
        return h  # no move possible
    delta = np.random.rand() * max_delta
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    return new_h

Now, a more sophisticated method: Instead of randomly moving mass, we can use "gradient ascent on overlap". For current worst shift kstar, we could try to increase the overlap for that shift: we have product f_k = sum_i h_i (1 - h_{i+k}). To reduce f_k, we want to increase h_{i+k} for indices where h_i is large, or decrease h_i where h_{i+k} is small. Since sum constraint, we can shift mass from indices where h_{i+k} is large and h_i is small to indices where h_i is large but h_{i+k} is small.

Thus we can propose a move that selects an index i where h_i is large but h_{i+k} is small (so product large). Then select j where h_j is small but h_{j+k} large; then move delta from i to j to reduce product. This is akin to gradient descent.

We can implement such step: For worst shift kstar, compute "benefit" for moving a small mass from index i to index j:

Change in cost for shift kstar approximated by -delta*( (1 - h_{i+k}) - (1 - h_{j+k}) ) + delta*(h_{i-k} - h_{j-k})? Actually gradient we derived earlier: derivative of cost C_k w.r.t h_i = (1 - h_{i+k}) - h_{i - k} (for i such that both terms defined). For direction where we increase h_i (positive delta) cost changes by derivative. So to reduce cost, we want to move mass to indices where derivative is negative (meaning increasing h_i reduces cost). Similarly, decreasing h_i where derivative positive will reduce cost. So we can compute derivative vector for worst shift and perform a water-filling between positive derivative indices (decrease) and negative derivative indices (increase). This is gradient descent with projection.

def gradient_step(h, kstar, step):
    n = len(h)
    # compute derivative g_i:
    # derivative of C_k w.r.t h_i: (1 - h_{i+k}) - (h_{i-k}) [if indices in range], else handle boundary.
    g = np.zeros_like(h)
    if kstar >= 0:
        # i with i + kstar < n
        idx = np.arange(n - kstar)
        g[idx] = (1 - h[idx + kstar])
        # i >= kstar: subtract h[i - kstar]
        idx2 = np.arange(kstar, n)
        g[idx2] -= h[idx2 - kstar]
    else:
        kk = -kstar
        # For negative shift k = -kk: formula symmetrical; we can treat shift positive by swapping.
        # C_{-kk} = sum_i h_i (1 - h_{i-kk}) => derivative = (1 - h_{i-kk}) - h_{i+kk}
        idx = np.arange(kk, n)
        g[idx] = (1 - h[idx - kk])
        idx2 = np.arange(0, n - kk)
        g[idx2] -= h[idx2 + kk]

    # Now we want to move mass from i where g_i > 0 to j where g_j < 0.
    # Determine sets:
    pos_idx = np.where(g > 1e-12)[0]
    neg_idx = np.where(g < -1e-12)[0]
    if len(pos_idx) == 0 or len(neg_idx) == 0:
        # fallback random propose
        return propose_move(h, step)
    # pick random indices
    i = np.random.choice(pos_idx)
    j = np.random.choice(neg_idx)
    # We compute max move from i to j respecting bounds and step size.
    max_move = min(step, h[i], 1 - h[j])
    if max_move <= 0:
        return h.copy()
    delta = np.random.rand() * max_move
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

import numpy as np
import math
import time

def compute_cost(h):
    """Maximum overlap integral for a given step vector h."""
    n = h.shape[0]
    # correlation of h with 1-h (full mode) and scale by dx = 2/n
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def project_to_capped_simplex(v, target_sum):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target_sum}."""
    # binary search on the Lagrange multiplier lambda
    lo = -np.max(v) - 2.0
    hi = np.max(1.0 - v) + 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def propose_random_move(h, max_delta, rng):
    """Transfer a random amount of mass from one coordinate to another."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    # random direction
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    move = min(max_delta, h[src], 1.0 - h[dst])
    if move <= 0.0:
        return h.copy()
    delta = rng.random() * move
    nh = h.copy()
    nh[src] -= delta
    nh[dst] += delta
    return nh


def gradient_move(h, kstar, max_delta, rng):
    """Take a small step guided by the gradient of the worst shift."""
    n = h.shape[0]
    g = np.zeros_like(h)

    if kstar >= 0:
        k = kstar
        # (1 - h[i+k]) part
        g[: n - k] += 1.0 - h[k:]
        # -h[i-k] part
        g[k:] -= h[: n - k]
    else:
        k = -kstar
        # (1 - h[i-k]) part
        g[k:] += 1.0 - h[: n - k]
        # -h[i+k] part
        g[: n - k] -= h[k:]

    pos_idx = np.where(g > 1e-12)[0]
    neg_idx = np.where(g < -1e-12)[0]

    if pos_idx.size and neg_idx.size:
        i = int(rng.choice(pos_idx))
        j = int(rng.choice(neg_idx))
        move = min(max_delta, h[i], 1.0 - h[j])
        if move > 0.0:
            delta = rng.random() * move
            nh = h.copy()
            nh[i] -= delta
            nh[j] += delta
            return nh
    # fall back to a random move if no useful gradient direction
    return propose_random_move(h, max_delta, rng)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose the discretisation size.
    # If a parent construction is provided, honour its length.
    try:

        cur_h = init.copy()
    except NameError:
        n = 200                     # a modest size that still fits the budget
        cur_h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    # Ensure the sum constraint holds exactly (tiny floating errors may appear)
    cur_h = project_to_capped_simplex(cur_h, target_sum)

    best_h = cur_h.copy()
    best_cost = compute_cost(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.95   # a safety margin

    # annealing schedule parameters
    max_delta_start = 0.5
    min_delta = 1e-4
    temp_start = 0.04

    cur_cost = best_cost

    while time.time() - start_time < time_limit:
        elapsed = time.time() - start_time
        frac = elapsed / time_limit

        # adapt step size and temperature
        max_delta = max(min_delta, max_delta_start * (1.0 - frac))
        temperature = max(1e-8, temp_start * (1.0 - frac))

        # locate the worst shift (the one attaining the maximum)
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx_max = int(corr.argmax())
        k_star = idx_max - (n - 1)   # shift corresponding to the maximum

        # propose a move (gradient‑guided with occasional random fallback)
        if rng.random() < 0.9:
            new_h = gradient_move(cur_h, k_star, max_delta, rng)
        else:
            new_h = propose_random_move(cur_h, max_delta, rng)

        new_cost = compute_cost(new_h)
        delta_cost = new_cost - cur_cost

        # Metropolis acceptance
        if delta_cost < 0.0 or rng.random() < math.exp(-delta_cost / temperature):
            cur_h = new_h
            cur_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = new_h.copy()

        # occasional restart from the best found so far to keep momentum
        if rng.random() < 0.001:
            cur_h = best_h.copy()
            cur_cost = best_cost

    # final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_capped_simplex(best_h, target_sum)

    final_cost = compute_cost(best_h)
    return best_h.tolist(), final_cost, n
