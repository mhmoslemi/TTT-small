        import numpy as np
        from scipy.optimize import minimize
        import time

        def run(seed=42, budget_s=1100, **kwargs):
            # If parent is provided, use it. Otherwise, we sample.
            n_points = 42
            dx = 2.0 / n_points
            target_sum = n_points / 2.0

            # Repair function
            def repair_vector(h_vec, target, n):
                h = h_vec.copy().astype(float)
                h_clamp = np.clip(h, 0, 1)
                s_clamp = np.sum(h_clamp)
                if np.isclose(s_clamp, target, atol=1e-5):
                    return h_clamp
                # Adjust by distributing the difference
                delta = (target - s_clamp) / n
                return np.clip(h_clamp + delta, 0, 1)

            # Objective function for optimization
            def objective(h_vec):
                h_clean = repair_vector(h_vec, target_sum, n_points)
                z = 1 - h_clean
                corr = np.correlate(h_clean, z, mode='full')
                max_corr_val = np.max(corr)
                return max_corr_val * dx

            # Setup bounds and constraints for minimizer
            bounds = [(0, 1)] * n_points
            constraints = {'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - target_sum}

            # Timer
            start_time = time.time()
            best_score = float('inf')
            best_sol = None

            # Number of starts requested? We'll do 10.
            if 'initial_h_values' in kwargs and len(kwargs['initial_h_values']) == n_points:
                # Use provided parent
                starts = [kwargs['initial_h_values']]
            else:
                starts = []
                np.random.seed(seed)
                for _ in range(10):
                    # Sample from uniform [0,1] and scale to have target_sum
                    h_rand = np.random.rand(n_points)
                    if np.random.rand() < 0.5:
                        h_rand = 1 - h_rand  # flip
                    h_rand = h_rand / np.sum(h_rand) * target_sum
                    starts.append(h_rand)

            # If provided parent is available and in the starts, we have one parent.
            # Now, run each start
            for i, start in enumerate(starts):
                if time.time() - start_time > budget_s:
                    break

                # Repair the start
                start_clean = repair_vector(start, target_sum, n_points)
                # Minimize the objective
                res = minimize(objective, start_clean, method='SLSQP', bounds=bounds, constraints=constraints, options={'max_eval': 1000, 'disp': False})
                score = res.fun

                if score < best_score:
                    best_score = score
                    best_sol = res.x

            # If we found at least one candidate, repair the best_sol
            best_sol_clean = repair_vector(best_sol, target_sum, n_points)
            # We return the best_sol and the best_score, and n_points.
            return best_sol_clean, best_score, n_points
