def project_to_sum(v, target):
    # clip
    v = np.clip(v, 0.0, 1.0)
    # adjust sum iteratively
    # Use water-filling algorithm
    total = v.sum()
    # tolerance
    if abs(total - target) < 1e-12:
        return v
    # Remove or add mass uniformly across adjustable entries
    # We'll use simple approach: while sum not close
    for _ in range(1000):
        total = v.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass
            free = v < 1.0
            if not free.any():
                break
            inc = diff / free.sum()
            # ensure not exceed bound
            inc_arr = np.minimum(inc, 1.0 - v[free])
            v[free] += inc_arr
        else:
            # need to subtract mass
            free = v > 0.0
            if not free.any():
                break
            dec = (-diff) / free.sum()
            dec_arr = np.minimum(dec, v[free])
            v[free] -= dec_arr
    # final tiny adjustment if still off due to rounding
    total = v.sum()
    diff = target - total
    if abs(diff) > 1e-9:
        # adjust first non-bound variable
        idx = np.where((v > 0) & (v < 1))[0]
        if idx.size == 0:
            # assign to first element (should be okay)
            idx = np.array([0])
        v[idx[0]] = np.clip(v[idx[0]] + diff, 0.0, 1.0)
    return v

def targeted_move(cur_h, step, rng):
    n = len(cur_h)
    corr_full = np.correlate(cur_h, 1 - cur_h, mode='full')
    idx_max = np.argmax(corr_full)
    L = idx_max - (n - 1)
    if L >= 0:
        i_start = 0
        i_end = n - L
        i_inds = np.arange(i_start, i_end)
        contr = cur_h[i_inds] * (1.0 - cur_h[i_inds + L])
    else:
        Lpos = -L
        i_start = Lpos
        i_end = n
        i_inds = np.arange(i_start, i_end)
        contr = cur_h[i_inds] * (1.0 - cur_h[i_inds - Lpos])
    # donor: index with max contr among where cur_h>0
    # Use mask of cur_h>0
    donor_mask = cur_h[i_inds] > 1e-12
    if not np.any(donor_mask):
        return None
    donor_idx = i_inds[donor_mask][np.argmax(contr[donor_mask])]
    # receiver: index where cur_h <1
    recv_mask = cur_h[i_inds] < 1 - 1e-12
    possible_recv = i_inds[recv_mask]
    if possible_recv.size == 0:
        return None
    # select receiver with minimal contr, maybe also ensure not donor
    recv_contr = contr[recv_mask]
    min_idx = np.argmin(recv_contr)
    receiver_idx = possible_recv[min_idx]
    if donor_idx == receiver_idx:
        # pick another random receiver
        other_receivers = possible_recv[possible_recv != donor_idx]
        if other_receivers.size == 0:
            return None
        receiver_idx = rng.choice(other_receivers)
    max_delta = min(step, cur_h[donor_idx], 1.0 - cur_h[receiver_idx])
    if max_delta <= 0:
        return None
    delta = rng.random() * max_delta
    new_h = cur_h.copy()
    new_h[donor_idx] -= delta
    new_h[receiver_idx] += delta
    return new_h

import numpy as np, time, math, sys
from scipy.optimize import minimize

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_to_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    # adjust via simple iterative method
    # Use water-filling: iterate until close
    for _ in range(1000):
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # Determine indices that can be increased/decreased
        if diff > 0:
            free = v < 1.0
            if not free.any():
                break
            inc = diff / free.sum()
            inc_arr = np.minimum(inc, 1 - v[free])
            v[free] += inc_arr
        else:
            free = v > 0.0
            if not free.any():
                break
            dec = (-diff) / free.sum()
            dec_arr = np.minimum(dec, v[free])
            v[free] -= dec_arr
        total = v.sum()
    # final fix for tiny rounding errors
    v = np.clip(v, 0.0, 1.0)
    # adjust again for residual
    total = v.sum()
    if abs(total - target) > 1e-8:
        # distribute the remaining diff to the first element that can absorb it
        diff = target - total
        idxs = np.arange(len(v))
        for i in idxs:
            if diff > 0 and v[i] < 1.0:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
            elif diff < 0 and v[i] > 0.0:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return v

def random_feasible_p(m, rng):
    while True:
        p = rng.random(m)
        p = p / p.sum() * (m / 2.0)
        if np.all(p <= 1.0):
            return p

def local_search(cur_h, dx, time_limit, rng):

    cur_score = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    step_initial = 0.2
    T_initial = 0.01
    step_min = 0.001
    T_min = 1e-6
    iter = 0
    while time.time() - start < time_limit:
        iter += 1
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = max(step_min, step_initial * (1 - frac))
        T = max(T_min, T_initial * (1 - frac)**2)
        # attempt targeted move
        new_h = None
        # compute worst shift
        corr_full = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx_max = np.argmax(corr_full)
        L = idx_max - (n - 1)  # shift
        # define overlapping region and contributions
        if L >= 0:
            i_start = 0
            i_end = n - L
            i_inds = np.arange(i_start, i_end)
            contr = cur_h[i_inds] * (1.0 - cur_h[i_inds + L])
        else:
            Lpos = -L
            i_start = Lpos
            i_end = n
            i_inds = np.arange(i_start, i_end)
            contr = cur_h[i_inds] * (1.0 - cur_h[i_inds - Lpos])
        # donor: largest contribution
        donor_mask = cur_h[i_inds] > 1e-12
        if np.any(donor_mask):
            donor_i = i_inds[donor_mask][np.argmax(contr[donor_mask])]
        else:
            donor_i = None
        # receiver: smallest contribution
        recv_mask = cur_h[i_inds] < 1 - 1e-12
        if np.any(recv_mask):
            recv_i = i_inds[recv_mask][np.argmin(contr[recv_mask])]
        else:
            recv_i = None
        if donor_i is not None and recv_i is not None and donor_i != recv_i:
            max_delta = min(step, cur_h[donor_i], 1.0 - cur_h[recv_i])
            if max_delta > 0:
                delta = rng.random() * max_delta
                new_h = cur_h.copy()
                new_h[donor_i] -= delta
                new_h[recv_i] += delta
        if new_h is None:
            # fallback to random pair move
            donors = np.where(cur_h > 1e-12)[0]
            receivers = np.where(cur_h < 1 - 1e-12)[0]
            if donors.size == 0 or receivers.size == 0:
                continue
            i = rng.choice(donors)
            j = rng.choice(receivers)
            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        new_score = compute_c5(new_h, dx)
        # acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score
        # occasional random restart
        if iter % 5000 == 0:
            if rng.random() < 0.2:
                cur_h = best_h.copy()
                cur_score = best_score
            else:
                # random binary
                cur_h = np.zeros(n)
                cur_h[rng.choice(n, size=n//2, replace=False)] = 1.0
                cur_score = compute_c5(cur_h, dx)
    return best_h, best_score

import time, math
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Core utilities
# -------------------------------------------------
def compute_c5(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral for a discrete step function."""
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto the box [0,1] and the affine hyperplane sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # simple water‑filling style adjustment
    for _ in range(1000):
        diff = target - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:                     # need to add mass
            free = v < 1.0
            if not free.any():
                break
            inc = diff / free.sum()
            inc_arr = np.minimum(inc, 1.0 - v[free])
            v[free] += inc_arr
        else:                            # need to remove mass
            free = v > 0.0
            if not free.any():
                break
            dec = (-diff) / free.sum()
            dec_arr = np.minimum(dec, v[free])
            v[free] -= dec_arr
        total = v.sum()
    # tiny residual fix
    residual = target - v.sum()
    if abs(residual) > 1e-9:
        idxs = np.where((v > 0.0) & (v < 1.0))[0]
        if idxs.size:
            v[idxs[0]] = np.clip(v[idxs[0]] + residual, 0.0, 1.0)
        else:
            # fallback to first element
            v[0] = np.clip(v[0] + residual, 0.0, 1.0)
    return v


def random_feasible_p(m: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random p∈[0,1]^m with Σp = m/2."""
    while True:
        p = rng.random(m)
        p = p / p.sum() * (m / 2.0)
        if np.all(p <= 1.0):
            return p


# -------------------------------------------------
# Coarse block optimisation (few variables)
# -------------------------------------------------
def optimise_blocks(n: int, m: int, rng: np.random.Generator) -> tuple:
    """
    Optimise a step function that is constant on m equal blocks.
    Returns (h_vector, score, n_points).
    """
    block_len = n // m
    dx = 2.0 / n
    target_sum = m / 2.0                     # Σp_i = m/2  (average 0.5)

    cons = ({'type': 'eq', 'fun': lambda p: np.sum(p) - target_sum},)
    bounds = [(0.0, 1.0)] * m

    best_score = np.inf
    best_p = None

    for _ in range(30):                     # restarts
        p0 = random_feasible_p(m, rng)
        res = minimize(
            lambda p: compute_c5(np.repeat(p, block_len), dx),
            p0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200, 'disp': False},
        )
        if not res.success:
            continue
        p_opt = np.clip(res.x, 0.0, 1.0)
        p_opt = project_to_sum(p_opt, target_sum)
        score = compute_c5(np.repeat(p_opt, block_len), dx)
        if score < best_score:
            best_score = score
            best_p = p_opt.copy()

    if best_p is None:                      # fallback – uniform 0.5
        best_p = np.full(m, 0.5)
        best_score = compute_c5(np.repeat(best_p, block_len), dx)

    h = np.repeat(best_p, block_len)
    return h, best_score, n

def local_search(start_h: np.ndarray,
                 dx: float,

    """
    Simulated‑annealing style refinement of a feasible vector.
    Returns (best_h, best_score).
    """

    cur_h = start_h.copy()
    cur_score = compute_c5(cur_h, dx)

    best_h = cur_h.copy()
    best_score = cur_score

    start = time.time()
    it = 0
    step_initial = 0.5
    T_initial = 0.02
    step_min = 0.001
    T_min = 1e-6

        it += 1
        elapsed = time.time() - start
        frac = elapsed / time_limit

        corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(np.argmax(corr_full))
        L = idx_max - (n - 1)                     # shift (may be negative)

        if L >= 0:
            i_start, i_end = 0, n - L
            inds = np.arange(i_start, i_end)
            contrib = cur_h[inds] * (1.0 - cur_h[inds + L])
        else:
            Lp = -L
            i_start, i_end = Lp, n
            inds = np.arange(i_start, i_end)
            contrib = cur_h[inds] * (1.0 - cur_h[inds - Lp])

        donor_mask = cur_h[inds] > 1e-12
        recv_mask = cur_h[inds] < 1.0 - 1e-12
        new_h = None
        if donor_mask.any() and recv_mask.any():
            donor_i = inds[donor_mask][np.argmax(contrib[donor_mask])]
            recv_i = inds[recv_mask][np.argmin(contrib[recv_mask])]

                j = rng.choice(receivers)
                max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    new_h = cur_h.copy()
                    new_h[i] -= delta
                    new_h[j] += delta

        if new_h is None:
            continue

        new_score = compute_c5(new_h, dx)
        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(
                -(new_score - cur_score) / max(T, 1e-12)):
            cur_h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score

        # occasional restart to diversify
        if it % 5000 == 0:
            if rng.random() < 0.2:
                cur_h = best_h.copy()
                cur_score = best_score
            else:
                cur_h = np.zeros(n)
                cur_h[rng.choice(n, size=n // 2, replace=False)] = 1.0
                cur_score = compute_c5(cur_h, dx)

    return best_h, best_score


# -------------------------------------------------
# Main entry point
# -------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine number of sample points
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 80                     # default when no parent is provided

    # if n is not a multiple of a convenient block count, fall back to plain SA
    block_candidates = [10, 5, 4, 2]
    m = None
    for cand in block_candidates:
        if n % cand == 0:
            m = cand
            break
    if m is None:
        # cannot use block model – start from a random feasible vector
        h0 = np.zeros(n)
        h0[rng.choice(n, size=n // 2, replace=False)] = 1.0
        h0 = project_to_sum(h0, n / 2.0)
        dx = 2.0 / n
        best_h, best_score = local_search(h0, dx, budget_s * 0.95, rng)
    else:
        # coarse optimisation on m blocks (takes ~30 % of the budget)
        block_time = budget_s * 0.3
        h0, _, _ = optimise_blocks(n, m, rng)

        # fine‑grained improvement with the remaining budget
        remaining = budget_s - (time.time() - start_time)
        if remaining > 0.1:
            dx = 2.0 / n
            best_h, best_score = local_search(h0, dx, remaining * 0.95, rng)
        else:
            best_h = h0
            dx = 2.0 / n
            best_score = compute_c5(best_h, dx)

    # final safety projection & scoring
    best_h = project_to_sum(best_h, n / 2.0)
    final_score = compute_c5(best_h, 2.0 / n)

    return best_h, float(final_score), n
