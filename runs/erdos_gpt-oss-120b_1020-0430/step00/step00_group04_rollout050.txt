def project(v, target):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        p = np.clip(v + mid, 0, 1)

        if s < target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v + (lo+hi)/2, 0, 1)

def project_to_box_and_sum(v, target, tol=1e-9, max_iter=100):
    lo = -np.max(v) - 1.0  # ensures v+lo <= 0 for all i
    hi = 1.0 - np.min(v) + 1.0  # ensures v+hi >= 1 for all i
    # binary search
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        p = np.clip(v + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # adjust small residual due to tolerance
    residual = target - p.sum()
    if abs(residual) > 1e-12:
        # find i where p not at bound
        free = np.where((p > 1e-12) & (p < 1 - 1e-12))[0]
        if len(free) > 0:
            # distribute residual equally among free indices
            p[free] += residual / len(free)
            p = np.clip(p, 0.0, 1.0)
        else:
            # if all at bounds, adjust the one closest to bounds
            idx = np.argmin(np.abs(p - 0.5))
            p[idx] = np.clip(p[idx] + residual, 0.0, 1.0)
    return p

def _project_to_sum(v: np.ndarray, target: float, tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """Project onto {0≤x_i≤1, sum x = target} by water‑filling."""
    lo = -np.max(v) - 1.0          # all entries become ≤0
    hi = 1.0 - np.min(v) + 1.0     # all entries become ≥1
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        p = np.clip(v + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for accumulated numeric error
    diff = target - p.sum()
    if abs(diff) > tol:
        free = np.where((p > tol) & (p < 1.0 - tol))[0]
        if free.size:
            p[free] += diff / free.size
            p = np.clip(p, 0.0, 1.0)
        else:
            # fallback: adjust a single component (should not happen for target=n/2)
            p[0] = np.clip(p[0] + diff, 0.0, 1.0)
    return p

            grad[kk:]      -= wk * h[: n - kk]
            grad[: n - kk] -= wk * h[kk:]
    return grad * dx

def _objective_fun(alpha, dx, h):
    return _smooth_max(h, alpha, dx)

def _jacobian_fun(alpha, dx, h):
    return _grad_smooth_max(h, alpha, dx)

def _optimize_start(h0: np.ndarray, target: float, alphas, maxiter_phase: int = 80):
    n = h0.size
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    h = h0.copy()
    for a in alphas:
        fun = partial(_objective_fun, a, dx)
        jac = partial(_jacobian_fun, a, dx)
        res = optimize.minimize(fun, h,
                                jac=jac,
                                method="SLSQP",
                                bounds=bounds,
                                constraints=cons,
                                options={"maxiter": maxiter_phase, "ftol": 1e-9, "disp": False})
        h = res.x
        h = _project_to_sum(h, target)     # enforce exact sum after each phase
    return h

def _random_start(n, target):
    v = np.random.rand(n)
    return _project_to_sum(v, target)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # pick a handful of discretisations (more points give more flexibility)
    ns = [42, 84, 126, 168, 200]
    best_h = None
    best_val = np.inf
    best_n = None

    # if a parent solution is supplied, remember it
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        parent_n = parent.size
    except Exception:
        parent = None
        parent_n = None

    # main search loop
    for n in ns:
        if time.time() - start_time > budget_s - 2.0:
            break
        target = n / 2.0                       # because ∫h = 1 ⇒ Σh = n/2
        # several restarts for each n
        restarts = 6 if n <= 84 else 4
        for r in range(restarts):
            if r == 0 and parent is not None and parent_n == n:
                h0 = parent.copy()
            elif r == 1 and parent is not None and parent_n != n:
                # up‑sample / down‑sample the parent linearly
                x_old = np.linspace(0.0, 2.0, parent_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h0 = np.interp(x_new, x_old, parent)
                h0 = _project_to_sum(h0, target)
            else:
                h0 = _random_start(n, target)

            # three smoothing phases – increasingly sharp approximation of max
            alphas = [5.0, 20.0, 60.0]
            h_opt = _optimize_start(h0, target, alphas, maxiter_phase=80)

            # evaluate exact objective
            val = _max_overlap(h_opt)

            # cheap random refinement on the true max (always respects constraints)
            step = 0.07
            while time.time() - start_time < budget_s - 0.5:
                pert = h_opt + np.random.randn(n) * step
                pert = _project_to_sum(pert, target)
                val_pert = _max_overlap(pert)
                if val_pert < val:
                    h_opt, val = pert, val_pert
                    step = min(step * 1.1, 0.15)
                else:
                    step *= 0.97
                if step < 1e-4:
                    break

            if val < best_val:
                best_h, best_val, best_n = h_opt.copy(), val, n

    # fallback – if nothing succeeded, fall back to a trivial feasible vector
    if best_h is None:
        n = 84
        target = n / 2.0
        best_h = np.full(n, 0.5)
        best_val = _max_overlap(best_h)
        best_n = n

    # final clean‑up: exact feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum(best_h, best_n / 2.0)

    return (best_h, best_val, best_n)
