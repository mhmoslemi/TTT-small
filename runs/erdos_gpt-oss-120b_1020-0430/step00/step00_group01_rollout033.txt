def _c5_of_h(h):
    """return the C5 value for a step vector h (length n)"""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def _build_step_h(lengths, heights, n):
    """
    lengths : array of interval lengths, sum == 2
    heights : array of the same size, values in [0,1]
    n       : number of discretisation points
    returns a length‑n vector representing the piecewise constant function.
    """
    pos = (np.arange(n) + 0.5) * (2.0 / n)          # mid‑points of the sub‑intervals
    cum = np.cumsum(lengths)
    idx = np.searchsorted(cum, pos, side='right')
    return heights[idx]

def _adjust_to_sum(h, target):
