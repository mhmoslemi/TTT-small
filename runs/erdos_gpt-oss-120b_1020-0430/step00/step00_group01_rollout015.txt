def compute_c5(h):
    n = h.shape[0]
    dx = 2.0/n
    prod = 1 - h
    # Compute full correlation via FFT
    # Pad both to length 2*n -1
    N = 2*n - 1
    # Use real FFT
    # Compute correlation as convolution of h with reversed prod
    # Actually we want correlate(h, prod, mode='full') = conv(h, prod[::-1])
    # So we can compute convolution via FFT: FFT of length N.
    # Use np.fft.rfft for speed
    H = np.fft.rfft(h, n=N)
    P_rev = np.fft.rfft(prod[::-1], n=N)
    conv = np.fft.irfft(H * P_rev, n=N)
    max_val = np.max(conv)
    return max_val * dx

def propose_move(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # randomly decide direction
    direction = rng.integers(0,2)  # 0: i->j, 1: j->i
    if direction == 0:
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            return None
        delta = rng.random() * max_delta
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        max_delta = min(h[j], 1 - h[i])
        if max_delta <= 0:
            return None
        delta = rng.random() * max_delta
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
    return new_h

def random_binary_solution(n, rng):
    # target sum = n/2
    # For even n: exactly n/2 ones.
    # For odd n: need fractional .5. We'll handle even.
    half = n // 2
    indices = rng.permutation(n)[:half]
    h = np.zeros(n, dtype=float)
    h[indices] = 1.0
    # If n is odd, we need one additional 0.5:
    if n % 2 == 1:
        # pick a random index not already set to 1
        remaining = [i for i in range(n) if h[i]==0.0]
        extra_idx = rng.choice(remaining)
        h[extra_idx] = 0.5
    return h

import time, numpy as np

def compute_c5_fast(h):
    n = h.shape[0]
    dx = 2.0 / n
    # compute correlation via FFT
    N = 2*n - 1
    H = np.fft.rfft(h, n=N)
    P = np.fft.rfft(1 - h[::-1], n=N)   # Actually 1-h reversed
    conv = np.fft.irfft(H * P, n=N)
    max_val = np.max(conv)
    return max_val * dx

def evaluate_h(h):
    # perhaps use evaluate_erdos_solution for consistency

        return evaluate_erdos_solution(h)
    except Exception:
        return compute_c5_fast(h)

def random_binary_solution(n, rng):
    half = n // 2
    h = np.zeros(n, dtype=float)
    indices = rng.choice(n, size=half, replace=False)
    h[indices] = 1.0
    if n % 2 == 1:
        # need half of an extra point: 0.5 added to make sum = n/2
        # already have half ones = floor(n/2) ; we need extra 0.5 to reach n/2 = floor(n/2) + 0.5
        remaining = np.setdiff1d(np.arange(n), indices)
        extra_idx = rng.choice(remaining)
        h[extra_idx] = 0.5
    return h

def propose_move(h, rng, max_step=0.2):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # direction: transfer from i to j if True else transfer from j to i
    if rng.integers(0,2) == 0:
        # transfer from i to j
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            return None
        delta = rng.random() * min(max_delta, max_step)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        return new_h
    else:
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0:
            return None
        delta = rng.random() * min(max_delta, max_step)
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
        return new_h

def hill_climb(initial_h, rng, time_limit, max_iters=None, max_non_improve=5000, max_step=0.2):
    start = time.time()
    n = len(initial_h)
    current_h = initial_h.copy()
    current_c5 = compute_c5_fast(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    non_improve = 0
    it = 0
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        if max_iters is not None and it >= max_iters:
            break
        # propose neighbor
        neighbor = propose_move(current_h, rng, max_step)
        if neighbor is None:
            continue
        new_c5 = compute_c5_fast(neighbor)
        if new_c5 < current_c5 - 1e-12:
            current_h = neighbor
            current_c5 = new_c5
            non_improve = 0
            if new_c5 < best_c5 - 1e-12:
                best_h = neighbor.copy()
                best_c5 = new_c5
        else:
            non_improve += 1
        # restart if stuck
        if non_improve >= max_non_improve:
            # random restart
            current_h = random_binary_solution(n, rng)
            current_c5 = compute_c5_fast(current_h)
            non_improve = 0
        it += 1
    return best_h, best_c5

def propose_gradient_move(h, lag, rng, max_step=0.2):
    n = len(h)
    # compute gradient
    # create arrays of forward and backward indices
    # For each i:
    # forward index i+lag within [0,n-1]
    # backward index i-lag within [0,n-1]
    # g_i = 1 - (fh + bh) where fh = h[i+lag] if valid else 0, bh = h[i-lag] if valid else 0.
    # We'll compute arrays using np.where.
    # Not needed to compute for all i; just find max and min derivative feasible for transfer.
    # We'll compute derivative vector g.
    g = np.ones(n, dtype=float)
    if lag >= 0:
        # forward for i <= n-1-lag
        idx_f = np.arange(n - lag)
        g[idx_f] -= h[idx_f + lag]
        # backward for i >= lag
        idx_b = np.arange(lag, n)
        g[idx_b] -= h[idx_b - lag]
    else:
        lag_abs = -lag
        # forward: i+lag = i - lag_abs; valid for i>=lag_abs
        idx_f = np.arange(lag_abs, n)
        g[idx_f] -= h[idx_f - lag_abs]
        # backward: i-lag = i + lag_abs; valid for i <= n - lag_abs -1
        idx_b = np.arange(n - lag_abs)
        g[idx_b] -= h[idx_b + lag_abs]
    # Now g is derivative (1 - sum of neighbor values counted)
    # Want to move mass from i with high g (positive) to j with low (negative)
    # But we must also consider bounds.
    # Let's find candidates:
    # high indices where h > 0 and g > 0.
    # low indices where h < 1 and g < 0.
    high_idxs = np.where((h > 1e-12) & (g > 0))[0]
    low_idxs = np.where((h < 1 - 1e-12) & (g < 0))[0]
    if len(high_idxs) == 0 or len(low_idxs) == 0:
        return None
    i = rng.choice(high_idxs)
    j = rng.choice(low_idxs)
    # max possible delta from i to j
    max_delta = min(h[i], 1 - h[j])
    if max_delta <= 0:
        return None
    delta = rng.random() * min(max_delta, max_step)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def _corr_and_c5(h):
    """Full correlation of h with (1-h) and corresponding C5."""
    n = h.shape[0]
    N = 2 * n - 1
    # FFT based convolution (correlation = conv with reversed second argument)
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft((1.0 - h)[::-1], n=N)
    conv = np.fft.irfft(H * G, n=N)
    idx = int(np.argmax(conv))
    max_val = conv[idx]
    lag = idx - (n - 1)          # lag corresponding to the max
    c5 = max_val * (2.0 / n)     # dx = 2/n
    return conv, c5, lag


def _fast_c5(h):
    """Fast C5 (without lag) – used only for internal comparisons."""
    _, c5, _ = _corr_and_c5(h)
    return c5

    if n % 2 == 1:
        # need an extra 0.5 to reach exact sum n/2
        rest = np.setdiff1d(np.arange(n), ones)
        h[rng.choice(rest)] = 0.5
    return h


def _clip_and_fix(h):
    """Clip to [0,1] and adjust one entry to enforce exact sum n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # find a place where we can add/subtract the missing amount
    if diff > 0:
        cand = np.where(h < 1 - 1e-12)[0]
    else:
        cand = np.where(h > 1e-12)[0]
    if cand.size == 0:
        return h
    i = cand[0]
    h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


# ------------------------------------------------------------
# Move generators
# ------------------------------------------------------------
def _random_transfer(h, rng, max_step):
    """Transfer a small amount of mass between two random indices."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    if rng.random() < 0.5:
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            return None
        delta = rng.random() * min(max_delta, max_step)
        new = h.copy()
        new[i] -= delta
        new[j] += delta
        return new
    else:
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0:
            return None
        delta = rng.random() * min(max_delta, max_step)
        new = h.copy()
        new[j] -= delta
        new[i] += delta
        return new


def _gradient_transfer(h, lag, rng, max_step):
    """Use the gradient of the worst lag to move mass from a source to a sink."""
    n = h.shape[0]
    # derivative of f_lag = sum h_i - sum h_i h_{i+lag}
    g = np.ones(n, dtype=float)
    if lag >= 0:
        if lag < n:
            fin = np.arange(n - lag)
            g[fin] -= h[fin + lag]
        if lag > 0:
            bin_ = np.arange(lag, n)
            g[bin_] -= h[bin_ - lag]
    else:
        a = -lag
        if a < n:
            fin = np.arange(a, n)
            g[fin] -= h[fin - a]
        if a > 0:
            bin_ = np.arange(n - a)
            g[bin_] -= h[bin_ + a]

    src = np.where((h > 1e-12) & (g > 1e-12))[0]
    dst = np.where((h < 1.0 - 1e-12) & (g < -1e-12))[0]
    if src.size == 0 or dst.size == 0:
        return None
    i = rng.choice(src)
    j = rng.choice(dst)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return None
    delta = rng.random() * min(max_delta, max_step)
    new = h.copy()
    new[i] -= delta
    new[j] += delta
    return new


def _propose(h, lag, rng, max_step, prob_grad):
    """Hybrid move: gradient-based with prob_grad, else random transfer."""
    if rng.random() < prob_grad:
        cand = _gradient_transfer(h, lag, rng, max_step)
        if cand is not None:
            return cand
    return _random_transfer(h, rng, max_step)


# ------------------------------------------------------------
# Local optimisation driver
# ------------------------------------------------------------
def _hill_climb(start_h, rng, time_limit,
                max_step=0.2, prob_grad=0.6,
                max_non_improve=8000):
    """Return the best vector found within the given time slice."""
    n = start_h.shape[0]
    target = n / 2.0

    cur = start_h.copy()
    _, cur_c5, cur_lag = _corr_and_c5(cur)

    best = cur.copy()
    best_val = cur_c5
    non_imp = 0
    start = time.time()

    while True:
        if time.time() - start >= time_limit:
            break

        nxt = _propose(cur, cur_lag, rng, max_step, prob_grad)
        if nxt is None:
            continue

        _, nxt_c5, nxt_lag = _corr_and_c5(nxt)

        if nxt_c5 < cur_c5 - 1e-12:
            cur, cur_c5, cur_lag = nxt, nxt_c5, nxt_lag
            non_imp = 0
            if cur_c5 < best_val - 1e-12:
                best, best_val = cur.copy(), cur_c5
        else:
            non_imp += 1

        if non_imp >= max_non_improve:
            # restart from a fresh binary feasible point
            cur = _random_binary(n, rng)
            _, cur_c5, cur_lag = _corr_and_c5(cur)
            non_imp = 0

    return best, best_val


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ----------------------------------------------------------------
    # Load or create an initial feasible vector
    # ----------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)  # provided by harness
        n0 = init.shape[0]
        init = _clip_and_fix(init)                      # sanity
    except Exception:  # initial_h_values absent or malformed
        n0 = 50
        init = _random_binary(n0, rng)

    # ----------------------------------------------------------------
    # Global best tracking
    # ----------------------------------------------------------------
    best_h = init.copy()
    best_c5 = _fast_c5(best_h)
    best_n = best_h.shape[0]

    # ----------------------------------------------------------------
    # Iterative refinement (local search + optional up‑sampling)
    # ----------------------------------------------------------------
    MAX_N = 200                # do not exceed this resolution
    TARGET = 0.3808

    while time.time() < deadline - 0.3:
        remaining = deadline - time.time()
        # allocate a slice of remaining time to the current optimisation phase
        phase_time = min(remaining * 0.6, 30.0)   # cap to keep loops responsive

        # run local optimisation on the current best vector
        candidate_h, cand_c5 = _hill_climb(best_h, rng, phase_time,
                                          max_step=0.2,
                                          prob_grad=0.6,
                                          max_non_improve=8000)

        # evaluate with the official helper for reliability
        try:
            eval_c5 = evaluate_erdos_solution(candidate_h)
        except Exception:
            eval_c5 = _fast_c5(candidate_h)

        if eval_c5 < best_c5 - 1e-12:
            best_h, best_c5 = candidate_h.copy(), eval_c5

        # early exit if target reached
        if best_c5 <= TARGET:
            break

        # decide whether to increase resolution
        if best_n < MAX_N:
            # simple heuristic: if no improvement in this phase, up‑sample
            if eval_c5 >= best_c5 - 1e-6:
                # double resolution by repeating each entry
                best_h = np.repeat(best_h, 2)
                best_n = best_h.shape[0]
                best_h = _clip_and_fix(best_h)   # keep exact sum
                # recompute C5 after up‑sampling (should be unchanged)
                best_c5 = _fast_c5(best_h)

    # ----------------------------------------------------------------
    # Final preparation of the return values
    # ----------------------------------------------------------------
    best_h = _clip_and_fix(best_h)
    final_c5 = None
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c5 = float(_fast_c5(best_h))

    return (best_h.tolist(), final_c5, int(best_h.shape[0]))
