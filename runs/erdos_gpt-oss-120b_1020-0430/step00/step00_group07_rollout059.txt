import time

import numpy as np

# ------------------------------------------------------------
# Helper: project a vector onto the intersection of the box [0,1]^n
# and the affine hyperplane sum(x)=target.
# The solution has the form  x_i = clip(y_i + λ, 0, 1)  for a suitable λ.
# ------------------------------------------------------------
def _project_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    lo = -np.max(v) - 1.0          # guarantees all entries become 0
    hi = 1.0 - np.min(v)           # guarantees all entries become 1
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= 1e-12:
            return x
        if s > target:
            hi = lam
        else:
            lo = lam
    # final clipping – numerical safety
    lam = (lo + hi) * 0.5
    x = np.clip(v + lam, 0.0, 1.0)
    # tiny residual correction (should be negligible)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 0) & (x < 1)
        n_free = free.sum()
        if n_free > 0:
            x[free] += diff / n_free
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback scaling (keeps feasibility)
            total = x.sum()
            if total > 0:
                x *= target / total
                x = np.clip(x, 0.0, 1.0)
    return x

def _gradient_of_shift(h: np.ndarray, shift: int) -> np.ndarray:
    n = h.shape[0]
    g = np.zeros_like(h)

        l = n - shift
        g[:l] += (1.0 - h[shift:])
        g[shift:] -= h[:l]
    else:
        t = -shift
        l = n - t
        g[t:] += (1.0 - h[:l])
        g[:l] -= h[t:]
    return g

        # default: binary vector with exactly n/2 ones
        n = 80
        h = np.zeros(n, dtype=float)
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    n = h.shape[0]
    target_sum = n / 2.0                     # ∫ h = 1  ⇔  Σ h_i = n/2
    h = np.clip(h, 0.0, 1.0)
    h = _project_box_simplex(h, target_sum)

    dx = 2.0 / n

    # -----------------------------------------------------------------
    # Evaluation helper
    # -----------------------------------------------------------------
    def _c5_of(h_vec: np.ndarray) -> float:
        # Use the provided evaluator if available, fall back to manual.
        try:
            return float(evaluate_erdos_solution(h_vec))
        except Exception:
            return float(np.max(np.correlate(h_vec, 1.0 - h_vec, mode="full")) * dx)

    best_h = h.copy()
    best_val = _c5_of(best_h)
    best_raw = best_val / dx                 # raw max correlation (without dx)

    # Current working point
    cur_h = best_h.copy()
    cur_raw = best_raw
    it_without_improve = 0
    it = 0

    # -----------------------------------------------------------------
    # Main optimisation loop (projected sub‑gradient with occasional restarts)
    # -----------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        # correlation and worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)
        raw_max = corr[idx_max]                 # value before multiplying by dx

        # gradient of the active shift
        grad = _gradient_of_shift(cur_h, shift)

        # Polyak step size (uses best known raw value)
        diff = raw_max - best_raw
        if diff > 1e-12:
            norm_sq = float(np.dot(grad, grad))
            step = diff / (norm_sq + 1e-12)
            # safety bounds
            step = max(step, 1e-8)
            step = min(step, 0.5)
        else:
            step = 0.0

        # gradient step + projection
        nxt_h = cur_h - step * grad
        nxt_h = _project_box_simplex(nxt_h, target_sum)

        # evaluate
        nxt_raw = np.max(np.correlate(nxt_h, 1.0 - nxt_h, mode="full"))
        nxt_val = nxt_raw * dx

        # accept move (always move to the new point to keep the walk alive)
        cur_h = nxt_h
        cur_raw = nxt_raw

        # record global best
        if nxt_val + 1e-12 < best_val:
            best_val = nxt_val
            best_raw = nxt_raw
            best_h = nxt_h.copy()
            it_without_improve = 0
        else:
            it_without_improve += 1

        # occasional restart if stuck
        if it_without_improve >= 5000:
            if np.random.rand() < 0.6:
                # perturb the current best slightly
                cur_h = best_h + np.random.normal(scale=0.07, size=n)
            else:
                # completely new binary pattern
                cur_h = np.zeros(n, dtype=float)
                cur_h[np.random.choice(n, n // 2, replace=False)] = 1.0
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_box_simplex(cur_h, target_sum)
            it_without_improve = 0

        it += 1

    # -----------------------------------------------------------------
    # Final clean‑up and return
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_simplex(best_h, target_sum)
    final_val = _c5_of(best_h)

    return best_h, float(final_val), n
