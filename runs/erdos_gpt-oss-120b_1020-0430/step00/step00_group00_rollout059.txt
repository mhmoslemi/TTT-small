        s = w.sum()

        else:

def project_onto_bounds_sum(v, target):

    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def _project_onto_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {0 <= x_i <= 1, sum x_i = target}."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        if w.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """Compute C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _gradient_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of the worst‑lag correlation w.r.t. h."""
    n = h.shape[0]
    shift = lag
    # h_{i+lag}
    hp = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            hp[: n - shift] = h[shift:]
    else:
        s = -shift
        if s < n:
            hp[s:] = h[: n - s]
    # h_{i-lag}
    hm = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            hm[shift:] = h[: n - shift]
    else:
        s = -shift
        if s < n:
            hm[: n - s] = h[s:]
    return 1.0 - hp - hm

    cur_val = _c5_of(h)
    best_h, best_val = h.copy(), cur_val

    for i in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        lag = int(np.argmax(corr)) - (n - 1)
        grad = _gradient_for_lag(h, lag)
        eta = step0 / np.sqrt(i + 1.0)
        h_new = h - eta * grad
        h_new = _project_onto_bounds_sum(h_new, target_sum)
        val = _c5_of(h_new)

        if val < cur_val - 1e-12:
            h, cur_val = h_new, val
        if val < best_val - 1e-12:
            best_h, best_val = h_new.copy(), val

        # small random perturbation every few hundred steps
        if (i + 1) % 500 == 0:
            pert = np.random.normal(scale=0.02, size=n)
            h_pert = _project_onto_bounds_sum(h + pert, target_sum)
            val_pert = _c5_of(h_pert)
            if val_pert < cur_val:
                h, cur_val = h_pert, val_pert
                if val_pert < best_val:
                    best_h, best_val = h_pert.copy(), val_pert
    return best_h, best_val


def _random_feasible(n: int, target_sum: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n with the prescribed sum."""
    v = np.random.rand(n)
    return _project_onto_bounds_sum(v, target_sum)


def _local_swaps(
    h0: np.ndarray,
    target_sum: float,
    steps: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Simple swap‑based hill climbing (preserves the sum)."""
    n = h0.shape[0]
    h = h0.copy()
    cur_val = _c5_of(h)
    best_h, best_val = h.copy(), cur_val

    for _ in range(steps):
        if time.time() > deadline:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # decide direction of mass transfer
        if random.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        if h[src] <= 1e-9 or h[dst] >= 1.0 - 1e-9:
            continue
        max_delta = min(0.05, h[src], 1.0 - h[dst])
        if max_delta <= 1e-12:
            continue
        delta = random.random() * max_delta
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        val = _c5_of(h_new)
        if val < cur_val:
            h, cur_val = h_new, val
            if val < best_val:
                best_h, best_val = h_new.copy(), val
    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a small C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        init = initial_h_values  # provided by the harness
        h0 = np.asarray(init, dtype=float)
        n = h0.shape[0]
    except NameError:
        n = 45
        h0 = _random_feasible(n, n / 2.0)

    target = n / 2.0
    if abs(h0.sum() - target) > 1e-12:
        h0 = _project_onto_bounds_sum(h0, target)

    best_h, best_val = h0.copy(), _c5_of(h0)

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    start = time.time()
    limit = start + budget_s - 2.0  # a little safety margin

    # first pass from the provided (or random) start point
    h_opt, val_opt = _subgrad_descent(
        best_h, target, max_iter=5000, step0=1.0, deadline=limit
    )
    if val_opt < best_val:
        best_h, best_val = h_opt, val_opt

    # polish with swaps
    h_opt2, val_opt2 = _local_swaps(best_h, target, steps=2000, deadline=limit)
    if val_opt2 < best_val:
        best_h, best_val = h_opt2, val_opt2

    # random restarts while time permits
    while time.time() < limit:
        h_start = _random_feasible(n, target)
        h_mid, val_mid = _subgrad_descent(
            h_start, target, max_iter=3000, step0=1.0, deadline=limit
        )
        if val_mid < best_val:
            best_h, best_val = h_mid, val_mid
            h_mid2, val_mid2 = _local_swaps(best_h, target, steps=1500, deadline=limit)
            if val_mid2 < best_val:
                best_h, best_val = h_mid2, val_mid2
        else:
            # occasional extra polishing of the current best
            if time.time() < limit - 0.5:
                h_best2, val_best2 = _local_swaps(best_h, target, steps=800, deadline=limit)
                if val_best2 < best_val:
                    best_h, best_val = h_best2, val_best2

    final_val = _c5_of(best_h)
    return best_h.tolist(), float(final_val), n
