def project_box_sum(v, target):
    # Clip to [0,1] after shifting by lambda.
    lo = -1e6
    hi = 1e6
    for _ in range(60):
        mid = (lo+hi)/2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    # final projection
    proj = np.clip(v - (lo+hi)/2, 0, 1)
    # due to rounding may not exactly hit target; we can adjust small tolerance:
    # if sum differs > 1e-9, adjust using linear scaling on non-bound values.
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        # distribute diff to elements not at bounds using scaling.
        # elements where 0<proj<1 are interior.
        interior = (proj > 1e-12) & (proj < 1-1e-12)

            # adjust by diff / count interior
            adjust = diff / interior.sum()
            proj[interior] += adjust
            # Clip again (could push out-of-bounds)
            proj = np.clip(proj, 0, 1)
            # Could iterate but tolerable.

def project_box_sum(v, target):
    # Ensure v is numpy array
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Adjust small rounding
    diff = target - proj.sum()
    if abs(diff) > 1e-8:
        # Distribute diff to interior points (non-bound) if any
        mask = (proj > 1e-12) & (proj < 1 - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

import numpy as np
import time
import math

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {h | 0<=h_i<=1, sum h_i = target}."""
    lo, hi = -2.0, 2.0  # wide enough bounds for the Lagrange multiplier
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction in case of rounding error
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_value(h: np.ndarray) -> float:
    """Exact C5 for a given discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))


def _subgradient_optimize(h: np.ndarray, max_iter: int = 200) -> np.ndarray:
    """Sub‑gradient descent on the max‑overlap objective."""
    n = h.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    for it in range(max_iter):
        # locate the worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        s = idx - (n - 1)

        # gradient of the worst term (ignoring the constant part)
        grad = np.zeros_like(h)
        if s >= 0:
            grad[: n - s] -= dx * h[s:]
            grad[s:] -= dx * h[: n - s]
        else:
            t = -s
            grad[t:] -= dx * h[: n - t]
            grad[: n - t] -= dx * h[t:]

        # step size slowly decreasing
        step = 0.5 / (1.0 + it * 0.02)
        h = h - step * grad
        h = _project_box_sum(h, target)
    return h


def _hill_climb(
    h: np.ndarray,
    steps: int = 2000,
    current_val: float | None = None,
) -> tuple[np.ndarray, float]:
    """Simple stochastic local search (mass transfer moves)."""
    n = h.shape[0]
    if current_val is None:
        current_val = _c5_value(h)

    start_temp = 1e-2
    end_temp = 1e-4

    for i in range(steps):
        # pick two distinct indices
        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        if np.random.rand() < 0.5:
            src, dst = i1, i2
        else:
            src, dst = i2, i1

        delta_max = min(h[src], 1.0 - h[dst])
        if delta_max <= 0.0:
            continue

        delta = delta_max * np.random.rand() * 0.5  # move at most half the allowed amount
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta

        val = _c5_value(h_new)
        if val < current_val:
            h, current_val = h_new, val
        else:
            # Metropolis acceptance
            t = start_temp * (1.0 - i / steps) + end_temp * (i / steps)
            if np.random.rand() < math.exp(-(val - current_val) / t):
                h, current_val = h_new, val
    return h, current_val
