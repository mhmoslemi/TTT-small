    O_raw = np.convolve(h, 1 - h, mode='full')
    max_val = O_raw.max()

def shift_array(arr, shift):
    n = len(arr)
    if shift == 0:
        return arr.copy()
    if shift > 0:
        # shifting forward: out[i] = arr[i + shift] if i+shift < n else 0
        out = np.empty_like(arr)
        out[:n - shift] = arr[shift:]
        out[n - shift:] = 0.0
        return out
    else:
        s = -shift
        out = np.empty_like(arr)
        out[s:] = arr[:n - s]
        out[:s] = 0.0
        return out
```

def project_to_feasible(x, target_sum):
    # clip
    x = np.clip(x, 0.0, 1.0)
    # waterfilling to satisfy sum
    # While not within tolerance
    eps = 1e-12
    for _ in range(100):
        s = np.sum(x)
        diff = target_sum - s
        if abs(diff) <= eps:
            break
        if diff > 0:
            # need to increase
            mask = x < 1.0 - eps
        else:
            mask = x > eps
        count = np.count_nonzero(mask)
        if count == 0:
            # cannot adjust
            break
        delta = diff / count
        # add delta to masked entries and clip
        idx = np.where(mask)[0]
        x[idx] += delta
        # clip after addition
        x = np.clip(x, 0.0, 1.0)
    return x
```

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _shift(arr: np.ndarray, k: int) -> np.ndarray:
    """Return arr shifted by k (arr[i+k]), zeros outside the interval."""
    n = arr.shape[0]
    if k == 0:
        return arr.copy()
    if k > 0:
        out = np.empty_like(arr)
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
        return out
    else:  # k < 0
        s = -k
        out = np.empty_like(arr)
        out[s:] = arr[: n - s]
        out[:s] = 0.0
        return out


def _project_sum_and_box(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project x onto the intersection {0≤x≤1, sum(x)=target_sum}."""
    x = np.clip(x, 0.0, 1.0)
    # water‑filling style iteration (fast for the small sizes we use)
    for _ in range(50):
        s = x.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        x[mask] += delta
        # re‑clip after the move
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction (unlikely to be needed)
    s = x.sum()
    if abs(s - target_sum) > 1e-9:
        # distribute the tiny remainder uniformly over free coordinates
        free = (x > 0.0) & (x < 1.0)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            x[free] += (target_sum - s) / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _c5_value(h: np.ndarray) -> float:
    """Return the (scaled) C5 value for a step function h."""
    conv = np.convolve(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / h.size)


def _softmax_gradient(h: np.ndarray, beta: float):
    """Gradient of the smooth max_{k} O(k) with temperature beta."""
    n = h.size
    # raw overlap values (no scaling)
    O = np.convolve(h, 1.0 - h, mode="full")
    m = O.max()
    # stable softmax
    w = np.exp(beta * (O - m))
    w /= w.sum()
    # compute weighted sum of shifted copies
    grad = np.ones(n)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)  # actual shift
        grad -= wk * _shift(h, k)       # (1 - h_{i+k}) part already accounted by the leading 1
        grad -= wk * _shift(h, -k)      # -h_{i-k} part
    return grad, m
