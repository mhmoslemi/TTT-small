    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(vec + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    proj = np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)
    # very small residual correction (rare)
    diff = target_sum - proj.sum()
    if abs(diff) > eps:
        mask = (proj > 0.0) & (proj < 1.0)
        if mask.any():
            adj = diff / mask.sum()
            proj[mask] += adj
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _max_corr(h):
    """Return max_{lag} ∫ h(x)(1-h(x+lag))dx approximated on the grid."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _smooth_max_corr_grad(h, alpha):
    """Gradient of the smooth max_{lag} (no dx factor)."""
    n = h.size
    # correlation without the dx scaling
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(alpha * corr)
    w = np.exp(alpha * corr - max_c)        # stability
    w /= w.sum()                            # softmax weights
    grad = np.zeros_like(h)

    # positive lags (including zero)
    for lag in range(n):
        wk = w[lag + n - 1]
        if wk == 0.0:
            continue
        end = n - lag
        # sum term
        grad[:end] += wk
        # product term (first factor)
        grad[:end] -= wk * h[lag:]
        # product term (second factor)
        grad[lag:] -= wk * h[:end]

    # negative lags
    for lag in range(1, n):
        wk = w[-lag + n - 1]
        if wk == 0.0:
            continue
        start = lag
        length = n - lag
        grad[start:] += wk
        grad[start:] -= wk * h[:length]
        grad[:length] -= wk * h[start:]

    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # Problem size
    try:
        init = np.array(initial_h_values, copy=False)
        n = init.size
    except NameError:
        n = 200                     # default resolution (even)
        init = None
    dx = 2.0 / n
    target_sum = n / 2.0            # because ∫h = 1

    # ------------------------------------------------------------------
    # Initial guess
    if init is not None:
        h = np.clip(init.astype(float), 0.0, 1.0)
        h = _project_onto_constraints(h, target_sum)
    else:
        # start from a smooth shape (sin^2) then project
        xs = (np.arange(n) + 0.5) * dx
        h = np.sin(np.pi * xs / 2.0) ** 2
        h = _project_onto_constraints(h, target_sum)

    best_h = h.copy()
    best_val = _max_corr(h)

    # ------------------------------------------------------------------
    # Gradient‑descent phase (smooth max)
    max_iters = 12000
    step = 0.6
    alpha_start, alpha_end = 1.0, 30.0

    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.70:
            break
        # schedule for the smoothness parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        grad = _smooth_max_corr_grad(h, alpha)
        h_new = h - step * grad
        h_new = _project_onto_constraints(h_new, target_sum)

        cur_val = _max_corr(h_new)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h_new.copy()
        # simple acceptance rule (keep best, otherwise move anyway)
        h = h_new

        # decay step size a little
        if (it + 1) % 2000 == 0:
            step *= 0.5

    # ------------------------------------------------------------------
    # Turn into a binary vector (exactly n/2 ones) – a good discrete start
    count_one = int(round(target_sum))
    idx = np.argpartition(best_h, -count_one)[-count_one:]
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx] = 1.0
    cur_val = _max_corr(h_bin)
    if cur_val < best_val:
        best_val = cur_val
        best_h = h_bin.copy()

    # ------------------------------------------------------------------
    # Simulated‑annealing on the binary representation
    h_bin = best_h.copy()
    cur_val = best_val
    ones = [i for i, v in enumerate(h_bin) if v > 0.5]
    zeros = [i for i, v in enumerate(h_bin) if v <= 0.5]

    # annealing schedule
    T_start, T_end = 0.8, 1e-4
    max_sa_steps = 250000
    for step_sa in range(max_sa_steps):
        if time.time() - start_time > budget_s * 0.95:
            break
        # temperature
        t_frac = step_sa / max_sa_steps
        T = T_start * (T_end / T_start) ** t_frac

        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # propose swap
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_val = _max_corr(h_bin)

        delta = new_val - cur_val
        accept = False
        if delta < 0:
            accept = True
        else:
            if np.random.rand() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_val = new_val
            # update the two lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h_bin.copy()
        else:
            # revert swap
            h_bin[i] = 1.0
            h_bin[j] = 0.0

    # ------------------------------------------------------------------
    # Final result
    final_c5 = best_val
    return best_h.tolist(), float(final_c5), n
