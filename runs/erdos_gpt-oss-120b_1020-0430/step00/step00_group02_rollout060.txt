def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # cross-correlation
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx

def project_onto_bounds_and_sum(x, target):
    # solve projection onto [0,1] and sum constraint using simple bisection
    low, high = -1.0, 1.0
    # ensure that low gives sum >= target
    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    proj = np.clip(x - (low + high) / 2.0, 0.0, 1.0)
    # adjust small numerical error
    # scaling may be needed if slight off; but fine.
    return proj

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(arr, alpha=20.0):
    max_val = np.max(arr)
    shifted = arr - max_val
    exp_sum = np.exp(alpha * shifted).sum()
    return max_val + np.log(exp_sum) / alpha

import time, random, math
import numpy as np
from scipy.optimize import minimize

# Global variable used by the equality constraint
TARGET_SUM_GLOBAL = None


def project_onto_bounds_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector x onto the box [0,1]^n intersected with the hyperplane sum(x)=target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    low, high = -1.0, 1.0  # bounds for lambda; guarantees sum at low > target, at high < target
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k))dx, discretized."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max(arr: np.ndarray, alpha: float = 20.0) -> float:
    """Smooth approximation of max using log‑sum‑exp."""
    m = np.max(arr)
    shifted = arr - m
    return m + np.log(np.exp(alpha * shifted).sum()) / alpha


def objective_smooth(h: np.ndarray, dx: float, alpha: float = 20.0) -> float:
    """Smooth objective used by SLSQP."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, alpha) * dx


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint Σh = TARGET_SUM_GLOBAL."""
    return np.sum(x) - TARGET_SUM_GLOBAL


def make_triangle(n: int) -> np.ndarray:
    """Triangular shape: peak 1 at centre, linear to 0 at edges."""
    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = 1.0 - np.abs(pts - 1.0) / 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_bounds_and_sum(h, n / 2.0)


def make_sin2(n: int) -> np.ndarray:

    dx = 2.0 / n
    pts = np.arange(n) * dx + dx * 0.5
    h = np.sin(np.pi * pts / 2.0) ** 2
    return project_onto_bounds_and_sum(h, n / 2.0)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of the provided initial vector to length n."""
    old_n = init.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    return project_onto_bounds_and_sum(h, n / 2.0)
