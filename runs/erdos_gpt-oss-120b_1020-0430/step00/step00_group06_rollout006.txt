    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    n = globals().get('initial_h_values', None)

        # random permutation
        rng.shuffle(h)
    dx = 2.0 / n

    def eval_h(vec):

        return np.max(np.correlate(vec, 1 - vec, mode='full')) * dx

    best_h = h.copy()
    best_val = eval_h(best_h)

    current_h = h.copy()
    current_val = best_val

    start = time.time()
    # schedule parameters
    p_targeted = 0.7
    T0 = 0.1  # temperature scale
    # 'step_factor' maximum proportion of possible delta we use
    max_step_factor = 0.5

    iteration = 0

    while True:

        if elapsed > budget_s * 0.98:

        remaining = max(0.0, budget_s - elapsed)
        fraction = remaining / budget_s
        T = T0 * fraction
        step_factor = max_step_factor * fraction

        # compute correlation and worst lag
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = int(np.argmax(corr))
        max_val = corr[max_idx] * dx
        # randomly decide move type
        if rng.random() < p_targeted:
            shift = max_idx - (n - 1)

                j_idx = i_idx + shift
                contrib = current_h[i_idx] * (1 - current_h[j_idx])
            else:

                i_idx = np.arange(s, n)

                move_type = 'random'
