def project_to_simplex_box(x):
    # ensure 0<=x<=1 and sum = target.
    # We'll use simple method: clip to [0,1].
    x_clipped = np.clip(x, 0, 1)
    # Adjust to satisfy sum exactly using iterative water-filling:
    # Compute excess or deficit.
    total = x_clipped.sum()
    target = n/2
    diff = target - total
    if abs(diff) < 1e-12:
        return x_clipped
    # The set of free variables are those not at boundaries after clipping.
    mask = (x_clipped > 0) & (x_clipped < 1)
    # If mask empty, just fallback: scale down/up entire vector?
    if not np.any(mask):
        # fallback: scale to target while staying in bounds
        # if sum too small, set all to target/n (which is 0.5) else set to 1 etc.
        # For now, just return x_clipped (should not happen)
        return x_clipped
    # Distribute diff equally among free vars.
    # But ensure we don't push them out of bounds.
    while True:
        free = mask
        n_free = free.sum()
        if n_free == 0:
            break
        add = diff / n_free
        x_clipped[free] += add
        # Clip again
        overflow_up = x_clipped > 1
        overflow_down = x_clipped < 0
        if not overflow_up.any() and not overflow_down.any():
            # satisfied within bounds
            break
        # For any that go out of bounds, fix them
        x_clipped[overflow_up] = 1
        x_clipped[overflow_down] = 0
        # recompute total and diff
        total = x_clipped.sum()
        diff = target - total
        mask = (x_clipped > 0) & (x_clipped < 1)
    return x_clipped

import numpy as np
import time
import math

def _evaluate_h(h: np.ndarray, dx: float) -> float:
    """Return the C5 value for a step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project_box_sum(h: np.ndarray, target_sum: float) -> np.ndarray:
    """
    In‑place projection of h onto the box [0,1]^n intersected with the hyperplane sum(h)=target_sum.
    Uses a simple water‑filling iteration.
    """
    np.clip(h, 0.0, 1.0, out=h)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    mask = (h > 0) & (h < 1)
    # iterate until the sum matches (or mask becomes empty)
    while True:
        n_free = mask.sum()
        if n_free == 0:
            break
        add = diff / n_free
        h[mask] += add
        # re‑clip because the uniform add may have pushed some entries out of bounds
        over = h > 1
        under = h < 0
        if over.any():
            h[over] = 1.0
        if under.any():
            h[under] = 0.0
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1)
    return h


def _gradient_step(h: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the C5 objective at h (the shift that attains the current maximum).
    Returns the gradient vector.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)                     # shift index (negative ⇔ left shift)

    if k >= 0:
        # forward shift (i+k) -> zeros on the left
        h_forward = np.concatenate((np.zeros(k), h[: n - k]))
        # backward shift (i‑k) -> zeros on the right
        h_backward = np.concatenate((h[k:], np.zeros(k)))
    else:

        h_forward = np.concatenate((h[kk:], np.zeros(kk)))   # shift left
        h_backward = np.concatenate((np.zeros(kk), h[: n - kk]))  # shift right
    grad = 1.0 - (h_forward + h_backward)
    return grad
