def project_onto_capped_simplex(v, target, eps=1e-12):
    # v: array
    low = -1.0
    high = 1.0
    # For feasible lambda search: h_i = clip(v_i - lambda, 0, 1)
    # Sum as function of lambda is decreasing continuous.
    for _ in range(60):
        lam = (low + high) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - (low + high)/2.0, 0.0, 1.0)

import numpy as np
import time, random, math

def _project_capped_simplex(v, target, eps=1e-12):
    """Project v onto the box [0,1] with sum == target."""
    # Solve h = clip(v - lam, 0, 1) with sum(h)=target via bisection on lam.
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _c5_value(h):
    """Return C5 for a vector h using the specification."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def run(seed=42, budget_s=1100):
    """Search for a step function giving a low C5 bound."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        # fallback: uniform 0.5 on 70 points
        n = 70
        h = np.full(n, 0.5, dtype=float)
    else:
        n = h.shape[0]

    target_sum = n / 2.0                     # because ∫h = 1 and dx = 2/n
    # repair any slight feasibility violation
    cur_sum = h.sum()
    if not np.isclose(cur_sum, target_sum, atol=1e-10):
        h = _project_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_score = float(_c5_value(best_h))

    start = time.time()
    # ------------------------------------------------------------------
    # simulated‑annealing parameters
    max_iter = 5_000_000                     # hard limit, will stop by time
    T_start = 0.12
    T_end = 1e-4
    delta_start = 0.30
    delta_end = 1e-5

    it = 0
    cur_h = h
    cur_score = best_score

    while time.time() - start < budget_s - 0.2 and it < max_iter:
        it += 1
        # temperature and max delta decay linearly
        frac = it / max_iter
        T = T_start * (1.0 - frac) + T_end * frac
        max_delta = delta_start * (1.0 - frac) + delta_end * frac

        # ---- propose a move: transfer delta between two entries ----------
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)

        hi, hj = cur_h[i], cur_h[j]

        # feasible delta range for hi -> hi+delta, hj -> hj-delta
        delta_low = -min(hi, 1.0 - hj)          # make sure hi stays >=0, hj <=1
        delta_high = min(1.0 - hi, hj)          # keep hi <=1, hj >=0
        if delta_low > delta_high:
            continue

        # restrict magnitude
        delta_low = max(delta_low, -max_delta)
        delta_high = min(delta_high, max_delta)
        if delta_low > delta_high:
            continue

        delta = np.random.uniform(delta_low, delta_high)

        h_new = cur_h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta

        new_score = float(_c5_value(h_new))

        # ----- accept according to Metropolis criterion ------------------
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / max(T, 1e-16)):
            cur_h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                # early stop if we beat the target bound
                if best_score <= 0.3808:
                    break

        # occasional random restart to diversify search
        if it % 4000 == 0:
            # keep best half of entries, randomise the rest and project
            mask = np.random.rand(n) < 0.5
            trial = best_h.copy()
            trial[mask] = np.random.rand(mask.sum())
            trial = _project_capped_simplex(trial, target_sum)
            trial_score = float(_c5_value(trial))
            if trial_score < cur_score:
                cur_h = trial
                cur_score = trial_score

    # final verification / final score compute
    final_score = float(_c5_value(best_h))
    return (best_h.tolist(), final_score, n)
