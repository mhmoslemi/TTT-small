def hill_climb(a, m, max_iters, eps, temp):
    best_a = a.copy()
    best_c5 = compute_c5_from_a(a, m)
    current_a = a.copy()
    current_c5 = best_c5
    for it in range(max_iters):
        # random pair
        i, j = np.random.choice(len(a), size=2, replace=False)
        # compute feasible delta range:
        lower = max(-current_a[i], current_a[j] - 1.0, -eps)
        upper = min(1.0 - current_a[i], current_a[j], eps)
        if lower >= upper:
            continue
        delta = np.random.uniform(lower, upper)
        new_a = current_a.copy()
        new_a[i] += delta
        new_a[j] -= delta
        new_c5 = compute_c5_from_a(new_a, m)
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / max(temp * (1 - it / max_iters), 1e-9)):
            current_a = new_a
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_a = new_a.copy()
                best_c5 = new_c5
        # optionally reduce eps gradually
        eps = max(eps * 0.9995, 0.0001)
        temp *= 0.9995
    return best_a, best_c5

import time, math
import numpy as np
from scipy.optimize import differential_evolution

def _compute_c5_from_heights(heights, reps):
    """Heights: array of length 5, sum = 2.5.
       reps   : repetitions per block (int)."""
    h = np.repeat(heights, reps)
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for a 5‑step construction."""
    start = time.time()
    np.random.seed(seed)

    # ----- configuration -------------------------------------------------
    n_blocks = 5                     # number of steps
    reps_per_block = 120            # points per block (total points = 600)
    n_points = n_blocks * reps_per_block
    target_sum = n_blocks / 2.0     # sum of heights must be 2.5
    # --------------------------------------------------------------------

    # ----- initial guess (optional, using parent if available) ----------
    if 'initial_h_values' in globals():
        try:
            init_h = np.asarray(initial_h_values, dtype=float)
            if init_h.size % n_blocks == 0:
                per = init_h.size // n_blocks
                a0 = np.array([init_h[i*per:(i+1)*per].mean() for i in range(n_blocks)])
                a0 = a0 * (target_sum / a0.sum())
                a0 = np.clip(a0, 0.0, 1.0)
                if a0.sum() > 1e-12:
                    # adjust the last component to hit the exact sum
                    diff = target_sum - a0.sum()
                    a0[-1] = np.clip(a0[-1] + diff, 0.0, 1.0)
            else:
                a0 = np.full(n_blocks, 0.5)
        except Exception:
            a0 = np.full(n_blocks, 0.5)
    else:
        a0 = np.full(n_blocks, 0.5)
    # --------------------------------------------------------------------

    # ----- objective for DE (optimises first 4 heights, 5th is forced) --
    def de_objective(x):
        # x has length 4
        heights = np.empty(n_blocks, dtype=float)
        heights[: n_blocks - 1] = x
        heights[-1] = target_sum - np.sum(x)
        # penalty if any height leaves [0,1]
        if np.any(heights < 0.0) or np.any(heights > 1.0):
            penalty = np.sum(np.clip(-heights, 0.0, None)) + np.sum(np.clip(heights - 1.0, 0.0, None))
            return 1e6 + penalty * 1e6
        return _compute_c5_from_heights(heights, reps_per_block)

    # ----- differential evolution ----------------------------------------
    bounds = [(0.0, 1.0)] * (n_blocks - 1)
    # limit maxiter so that we stay well within the time budget
    maxiter = 200
    de_res = differential_evolution(
        de_objective,
        bounds,
        maxiter=maxiter,
        popsize=15,
        polish=False,
        seed=seed,
        updating='deferred',
        workers=1,
    )
    best_heights = np.empty(n_blocks, dtype=float)
    best_heights[: n_blocks - 1] = de_res.x
    best_heights[-1] = target_sum - np.sum(de_res.x)
    best_heights = np.clip(best_heights, 0.0, 1.0)

    # ----- local refinement (pairwise transfers) -----------------------
    best_c5 = _compute_c5_from_heights(best_heights, reps_per_block)

    # parameters for the stochastic local search
    eps = 0.08                     # maximal transfer per step
    temp = 0.02                    # temperature for occasional uphill moves
    it = 0

    while time.time() - start < budget_s * 0.94:
        # pick two distinct blocks
        i, j = np.random.choice(n_blocks, 2, replace=False)
        # feasible interval for the transfer delta (preserves sum)
        lo = max(-best_heights[i], best_heights[j] - 1.0, -eps)
        hi = min(1.0 - best_heights[i], best_heights[j], eps)
        if lo >= hi:

    h_values = np.repeat(best_heights, reps_per_block)

    final_c5 = _compute_c5_from_heights(best_heights, reps_per_block)
    return h_values, float(final_c5), int(n_points)
