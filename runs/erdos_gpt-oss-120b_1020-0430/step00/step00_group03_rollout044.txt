def propose_pair_move(h, max_step):
    # choose two distinct indices i, j
    i,j = np.random.choice(len(h),2,replace=False)
    # generate delta random in [-max_step, max_step]
    delta = (np.random.rand()*2-1) * max_step
    # attempt to add delta to h[i], subtract delta from h[j], respecting bounds
    # compute feasible delta range
    low_i = -h[i] # can't go below 0
    high_i = 1 - h[i] # can't exceed 1
    low_j = - (1 - h[j]) # because subtracting delta from h[j]: actually new h[j] = h[j] - delta; bound: 0 <= h[j] - delta <= 1 => delta <= h[j]; delta >= h[j] - 1
    # So delta feasible: delta >= max(low_i, h[j] - 1) and delta <= min(high_i, h[j])
    delta_min = max(low_i, h[j] - 1)
    delta_max = min(high_i, h[j])
    if delta_min > delta_max:
        return h.copy() # no feasible move
    # clamp random delta to feasible range
    delta = np.random.uniform(delta_min, delta_max)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # All remain in [0,1]; sum unchanged.
    return new_h

    corr = np.correlate(h, 1 - h, mode='full')
    # find lag with max correlation
    max_idx = np.argmax(corr)  # index in 0..2*n-2
    # Convert to lag = max_idx - (n-1)
    lag = max_idx - (n - 1)
    # Compute gradient
    g = np.zeros_like(h)
    # For i where i+lag in range (i+lag < n):
    if lag >= 0:
        # term (1 - h[i+lag]) for i from 0..n-1-lag
        g[:n-lag] += (1 - h[lag:])
        # term -h[i - lag] for i from lag..n-1 => i-lag index from 0..n-1-lag

import numpy as np
import time

def project_capped_simplex(v, S):
    """Project vector v onto {x: sum x = S, 0 <= x <= 1}."""
    n = v.shape[0]
    # Quick return if already feasible
    if (v >= 0).all() and (v <= 1).all() and abs(v.sum() - S) < 1e-12:
        return v
    # Binary search for lambda
    lo = -np.max(v) - 1.0   # makes v - lo very large => all clipped to 1
    hi =  np.max(v) + 1.0   # makes v - hi very small => all clipped to 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - lo, 0.0, 1.0)
    # tiny correction for numeric drift
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size:
            i = free[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            # all at bounds, adjust the one with most slack
            i = np.argmin(x) if diff < 0 else np.argmax(x)
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def max_overlap_score(h, dx):
    """Exact score as defined in the problem."""
    # correlation of h with (1-h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def propose_pair_move(h, max_step):
    """Return a new feasible vector by moving mass between two positions."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # feasible delta range so that both entries stay in [0,1]
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    if delta_min >= delta_max:
        return h.copy()
    # Respect max_step bound
    bound = max_step
    delta_min = max(delta_min, -bound)
    delta_max = min(delta_max,  bound)
    if delta_min >= delta_max:
        # fallback to original feasible range
        delta_min = max(-h[i], h[j] - 1.0)
        delta_max = min(1.0 - h[i], h[j])
    delta = np.random.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def gradient_step(h, step, target_sum):
    """One projected gradient step using the worst lag."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    g = np.zeros_like(h)
    if lag >= 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:]    -= h[:n - lag]
    else:
        L = -lag
        g[L:]      += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    h_new = h - step * g
    return project_capped_simplex(h_new, target_sum)
