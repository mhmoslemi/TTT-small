def hill_climb(h_initial, time_budget):
    import time, random, numpy as np
    start = time.time()
    h = h_initial.copy()
    N = len(h)
    dx = 2.0 / N
    # compute correlation fast
    def compute_c5(h):
        cor = np.correlate(h, 1 - h, mode='full') * dx
        return cor.max()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # parameters
    T0 = 0.01
    # each iteration choose random pair
    iter = 0
    while time.time() - start < time_budget:
        # pick random distinct indices
        i, j = np.random.randint(0, N, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        # ensure we can move some mass
        # decide direction: move from i to j or j to i randomly
        if random.random() < 0.5:
            # move from i to j
            max_move = min(hi, 1 - hj)
            if max_move == 0:
                continue
            eps = max_move * random.random()

            h[j] += eps

def apply_move(h, cor_raw, i, j, eps):
    N = len(h)
    delta_i = -eps
    # (i update)
    old_hi = h[i]
    new_hi = old_hi - eps
    # loops
    # q loop (i as first)
    for q in range(N):
        if q == i: continue
        L = i - q
        cor_raw[L + N - 1] += delta_i * (1 - h[q])
    # p loop (i as second)
    for p in range(N):
        if p == i: continue
        L = p - i
        cor_raw[L + N - 1] -= h[p] * delta_i
    # L=0 term
    cor_raw[N - 1] += new_hi * (1 - new_hi) - old_hi * (1 - old_hi)
    h[i] = new_hi

    # Now edit j with delta_j = +eps
    delta_j = eps
    old_hj = h[j]
    new_hj = old_hj + eps
    # q loop (j as first)
    for q in range(N):
        if q == j: continue
        L = j - q
        cor_raw[L + N - 1] += delta_j * (1 - h[q])

def optimize_h(h_initial, time_budget):
    import numpy as np, time, math, random
    N = len(h_initial)
    dx = 2.0 / N
    h = np.array(h_initial, dtype=float)
    cor_raw = np.correlate(h, 1 - h, mode='full')
    cur_val = cor_raw.max() * dx
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    # set temperature schedule
    while time.time() - start < time_budget:
        # Determine random move
        # Equivalent to random pair selection
        # choose i with h[i] > 1e-8, j with h[j] < 1 - 1e-8
        up = np.where(h > 1e-12)[0]
        down = np.where(h < 1 - 1e-12)[0]
        if up.size == 0 or down.size == 0:
            break
        i = int(np.random.choice(up))
        j = int(np.random.choice(down))
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        # choose epsilon (between 0 and max_move)
        eps = max_move * np.random.rand()
        # copy for revert
        h_old = h.copy()
        cor_old = cor_raw.copy()
        # apply move (i->j)
        apply_move(h, cor_raw, i, j, eps)
        new_val = cor_raw.max() * dx
        elapsed = time.time() - start
        # accept or reject
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # temperature schedule linear cooling from T0 to near 0
            # set T0 maybe 0.02 (since C5 ~0.38)
            T0 = 0.02
            T = max(0.0001, (1.0 - elapsed / time_budget) * T0)
            if np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur_val = new_val
            else:
                # revert
                h = h_old
                cor_raw = cor_old
    return best_h, best_val

def apply_move(h, cor_raw, i, j, eps):
    N = h.shape[0]
    delta_i = -eps
    old_hi = h[i]
    new_hi = old_hi + delta_i
    # Prepare mask arrays
    idxs = np.arange(N)
    mask_i = idxs != i
    # Update cor for i
    L1 = i - idxs[mask_i]
    cor_raw[L1 + N - 1] += delta_i * (1 - h[mask_i])
    L2 = idxs[mask_i] - i
    cor_raw[L2 + N - 1] -= h[mask_i] * delta_i
    # L=0 update
    cor_raw[N - 1] += new_hi * (1 - new_hi) - old_hi * (1 - old_hi)
    # set h[i]
    h[i] = new_hi

    # Now for j
    delta_j = eps
    old_hj = h[j]
    new_hj = old_hj + delta_j
    mask_j = idxs != j
    L1 = j - idxs[mask_j]
    cor_raw[L1 + N - 1] += delta_j * (1 - h[mask_j])
    L2 = idxs[mask_j] - j

import time
import math
import random
import numpy as np

def _apply_move(h, cor_raw, i, j, eps, idxs):
    """Move eps from h[i] to h[j] and update linear correlation array."""
    N = h.shape[0]
    # ----- update index i (decrease) -----
    delta_i = -eps
    old_hi = h[i]
    new_hi = old_hi + delta_i

    mask_i = idxs != i
    L1 = i - idxs[mask_i]               # i as first factor
    cor_raw[L1 + N - 1] += delta_i * (1.0 - h[mask_i])
    L2 = idxs[mask_i] - i               # i as second factor
    cor_raw[L2 + N - 1] -= h[mask_i] * delta_i
    cor_raw[N - 1] += new_hi * (1.0 - new_hi) - old_hi * (1.0 - old_hi)
    h[i] = new_hi

    # ----- update index j (increase) -----
    delta_j = eps
    old_hj = h[j]
    new_hj = old_hj + delta_j

    mask_j = idxs != j
    L1 = j - idxs[mask_j]               # j as first factor
    cor_raw[L1 + N - 1] += delta_j * (1.0 - h[mask_j])
    L2 = idxs[mask_j] - j               # j as second factor
    cor_raw[L2 + N - 1] -= h[mask_j] * delta_j
    cor_raw[N - 1] += new_hj * (1.0 - new_hj) - old_hj * (1.0 - old_hj)
    h[j] = new_hj


def _optimize_h(h_init, time_budget):
    """Local simulated‑annealing optimizer for a fixed grid size."""
    N = h_init.shape[0]
    dx = 2.0 / N
    h = h_init.copy()
    cor = np.correlate(h, 1.0 - h, mode="full")
    cur_val = cor.max() * dx
    best_h = h.copy()
    best_val = cur_val

    idxs = np.arange(N)
    start = time.time()
    while time.time() - start < time_budget:
        # choose a donor (h>0) and a receiver (h<1)
        donors = np.where(h > 1e-12)[0]
        receivers = np.where(h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = int(np.random.choice(donors))
        j = int(np.random.choice(receivers))
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0.0:
            continue
        eps = max_move * np.random.rand()  # random fraction of the allowable transfer

        # backup for possible rejection
        h_old = h.copy()
        cor_old = cor.copy()

        # perform move
        _apply_move(h, cor, i, j, eps, idxs)
        new_val = cor.max() * dx

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            elapsed = time.time() - start
            frac = elapsed / time_budget
            T0 = 0.02
            T = max(1e-6, (1.0 - frac) * T0)
            if np.random.rand() < math.exp(-(new_val - cur_val) / T):
                cur_val = new_val
            else:
                h = h_old
                cor = cor_old
        # early exit if we beat the target
        if best_val < 0.3808:
            break
    return best_h, best_val
