    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # We'll adjust using loop.
    if diff > 0:
        # need to increase sum
        # iterate over indices where h < 1
        indices = np.where(h < 1)[0]
        for idx in indices:
            cap = 1.0 - h[idx]
            add = min(cap, diff)
            h[idx] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # diff < 0: need to decrease sum
        diff = -diff
        indices = np.where(h > 0)[0]
        for idx in indices:
            cap = h[idx]
            sub = min(cap, diff)
            h[idx] -= sub
            diff -= sub
            if diff <= 1e-12:
                break
    # After loop, may still have tiny diff due to rounding; distribute to first entry
    if abs(diff) > 1e-9:
        # adjust first index; ensure within bounds
        idx = 0
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

import time, math, numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _repair(vec, target):
    """Clip to [0,1] and enforce exact sum == target."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return vec
    # increase
    if diff > 0:
        for i in np.where(vec < 1.0)[0]:
            inc = min(1.0 - vec[i], diff)
            vec[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:                       # decrease
        diff = -diff
        for i in np.where(vec > 0.0)[0]:
            dec = min(vec[i], diff)
            vec[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    # tiny leftover – push to first coordinate safely
    if abs(diff) > 1e-12:
        i = 0
        vec[i] = np.clip(vec[i] + diff, 0.0, 1.0)
    return vec

def _objective(vec):
    """C5 value for a given step function."""
    n = vec.shape[0]
    return np.max(np.correlate(vec, 1.0 - vec, mode="full")) * (2.0 / n)

def _max_shift(vec):
    """Return (shift, max_val) where shift is the integer index offset."""
    n = vec.shape[0]
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    idx = int(corr.argmax())
    shift = idx - (n - 1)
    return shift, corr[idx] * (2.0 / n)

def _directed_move(h, shift, eps):
    """Move a small amount eps from a larger entry to a smaller one
       separated by *shift* (if such a pair exists).  Returns new vector
       or None if no improving pair is found."""
    n = h.shape[0]
    if shift == 0:
        return None
    if shift > 0:
        src_range = np.arange(n - shift)
        diff = h[src_range] - h[src_range + shift]
        cand = np.where(diff > 1e-12)[0]
        if cand.size == 0:
            return None
        i = cand[np.random.randint(cand.size)]
        src, dst = i, i + shift
    else:  # shift < 0
        s = -shift
        src_range = np.arange(s, n)
        diff = h[src_range] - h[src_range - s]
        cand = np.where(diff > 1e-12)[0] + s
        if cand.size == 0:
            return None
        i = cand[np.random.randint(cand.size)]
        src, dst = i, i - s
    max_transfer = min(h[src], 1.0 - h[dst])
    delta = min(eps, max_transfer)
    if delta <= 1e-12:
        return None
    new = h.copy()
    new[src] -= delta
    new[dst] += delta
    return new

    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Determine problem size and initial vector

    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)

        h0 = np.minimum(xs, 2.0 - xs)
        n0 = n
    target_sum = n / 2.0
    h0 = _repair(h0, target_sum)

    # ------------------------------------------------------------------
    # Parameters for the stochastic search
    # ------------------------------------------------------------------
    max_time = budget_s * 0.95            # reserve a bit for finalisation
    start = time.time()
    # annealing schedule
    init_temp = 0.015
    init_eps = 0.07                      # initial transfer magnitude
    # book‑keeping
    best_h = h0.copy()
    best_score = _objective(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    iters_no_imp = 0

    while time.time() - start < max_time:
        elapsed = time.time() - start
        prog = elapsed / max_time
        # linear cooling
        temp = max(1e-6, (1.0 - prog) * init_temp)
        eps = max(1e-4, (1.0 - prog) * init_eps)

        # --------------------------------------------------------------
        # 1) targeted move against the currently worst shift
        # --------------------------------------------------------------
        shift, _ = _max_shift(cur_h)
        if abs(shift) > 0 and rng.random() < 0.65:
            new = _directed_move(cur_h, shift, eps)
            if new is not None:
                new_score = _objective(new)
                if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / temp):
                    cur_h, cur_score = new, new_score
                    if new_score + 1e-12 < best_score:
                        best_h, best_score = new.copy(), new_score
                        iters_no_imp = 0
                    else:
                        iters_no_imp += 1
                    continue   # skip random move this iteration

        # --------------------------------------------------------------
        # 2) generic random two‑point exchange (preserves sum exactly)
        # --------------------------------------------------------------
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # feasible delta interval ensuring bounds after exchange
        lower = max(-cur_h[i], -(1.0 - cur_h[j]))
        upper = min(1.0 - cur_h[i], cur_h[j])
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper) * (eps / max(upper - lower, 1.0))
        # scaled to stay inside the interval
        delta = rng.uniform(lower, upper)
        new = cur_h.copy()
        new[i] += delta
        new[j] -= delta
        new_score = _objective(new)
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / temp):
            cur_h, cur_score = new, new_score
            if new_score + 1e-12 < best_score:
                best_h, best_score = new.copy(), new_score
                iters_no_imp = 0
            else:
                iters_no_imp += 1

        # --------------------------------------------------------------
        # 3) occasional restart from the best found
        # --------------------------------------------------------------
        if iters_no_imp > 25000:
            cur_h = best_h.copy()
            cur_score = best_score
            iters_no_imp = 0

    # ------------------------------------------------------------------
    # Final refinement: keep applying directed moves until no improvement
    # ------------------------------------------------------------------
    for _ in range(5000):

        new_score = _objective(new)
        if new_score + 1e-12 < best_score:
            best_h, best_score = new, new_score
        else:
            break

    # --------------------------------------------------------------
    # Ensure feasibility exactly before returning
    # --------------------------------------------------------------
    best_h = _repair(best_h, target_sum)
    final_score = _objective(best_h)
    return best_h, final_score, n
