def _project_to_simplex_capped(v: np.ndarray, total: float) -> np.ndarray:
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum(x) = total}.
    Uses a bisection on the shift lambda for the formula
        x_i = clip(v_i - lambda, 0, 1)
    """
    lo = np.min(v) - 1.0        # at lo, every entry is clipped to 1 -> sum = len(v)
    hi = np.max(v)              # at hi, every entry is <= 0   -> sum = 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for rounding errors
    diff = total - x.sum()
    if abs(diff) > 1e-12:
        # find an index that can absorb the difference without breaking bounds
        idx = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if idx.size == 0:
            idx = np.arange(len(x))
        i = int(idx[0])
        x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def _c5_value(h: np.ndarray) -> float:

    if _eval is not None:
        # the helper already multiplies by dx
        return float(_eval(h))
    # fallback – identical to the reference implementation
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _worst_shift_and_grad(h: np.ndarray):
    """return the shift giving the maximal overlap and the gradient of that term."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                     # signed shift in samples

    grad = np.zeros_like(h)
    if shift >= 0:
        # overlapping indices: i = 0 .. n-1-shift
        i_end = n - shift
        grad[:i_end] += 1.0 - h[shift:]       # ∂/∂h_i
        grad[shift:] += -h[:i_end]           # ∂/∂h_{i+shift}
    else:
        s = -shift
        i_start = s
        i_end = n
        grad[i_start:] += 1.0 - h[:n - s]    # ∂/∂h_i
        grad[:n - s] += -h[i_start:]         # ∂/∂h_{i-s}
    return shift, grad
