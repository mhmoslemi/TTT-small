# helper: make a vector respect box [0,1] and exact sum target
def _project_to_feasible(x: np.ndarray, target: float) -> np.ndarray:
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    # simple scaling‑and‑clipping loop; converges quickly for our tiny size
    for _ in range(8):
        if abs(s - target) < 1e-12:
            break
        # scale towards target, then re‑clip
        scale = target / s if s != 0 else 0.0
        x = np.clip(x * scale, 0.0, 1.0)
        s = x.sum()
    # tiny residual fix by distributing uniformly on unfixed entries
    diff = target - s
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            incr = diff / free.size
            x[free] += incr
            np.clip(x, 0.0, 1.0, out=x)
    return x
