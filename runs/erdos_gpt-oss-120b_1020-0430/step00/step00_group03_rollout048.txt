import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size and initial vector
    # ----------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, copy=False)
        n = init.size
    except Exception:                      # no initial_h_values provided
        n = 70
        init = np.zeros(n, dtype=np.float64)
        init[rng.choice(n, size=n // 2, replace=False)] = 1.0

    target_sum = n / 2.0                 # because ∫ h = 1  and  dx = 2/n

    # ----------------------------------------------------------------------
    # Helper: enforce 0≤h≤1 and ∑h = target_sum
    # ----------------------------------------------------------------------
    def enforce(v):
        v = np.clip(v, 0.0, 1.0)
        diff = target_sum - v.sum()
        eps = 1e-12
        # simple greedy fix – add/subtract on random admissible indices
        while abs(diff) > eps:
            if diff > 0:                     # need to add mass
                cand = np.where(v < 1 - eps)[0]
                if cand.size == 0:
                    break
                i = rng.choice(cand)
                delta = min(diff, 1.0 - v[i])
                v[i] += delta
                diff -= delta
            else:                           # need to remove mass
                cand = np.where(v > eps)[0]
                if cand.size == 0:
                    break
                i = rng.choice(cand)
                delta = min(-diff, v[i])
                v[i] -= delta
                diff += delta
        return v

    # ----------------------------------------------------------------------
    # Objective: C5 = max_k ∫ h·(1‑h∘τ_k) = max(correlation)·dx
    # ----------------------------------------------------------------------
    def c5_of(v):
        # using the same evaluation as the benchmark harness
        corr = np.correlate(v, 1.0 - v, mode='full')
        return float(corr.max() * (2.0 / n))

    # ----------------------------------------------------------------------
    # Initialise current & best solutions
    # ----------------------------------------------------------------------
    cur = enforce(init.astype(np.float64))
    cur_val = c5_of(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    max_temp = 0.05                     # initial temperature for SA
    stagnant = 0                        # steps without improvement
    last_improve = start

    # ----------------------------------------------------------------------
    # Simulated‑annealing style local search
    # ----------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.96:
            break

        # ------- propose a transfer move (preserves sum) -----------------
        inc_idx = np.where(cur < 1.0 - 1e-12)[0]
        dec_idx = np.where(cur > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break

        i = rng.choice(inc_idx)      # index that will increase
        j = rng.choice(dec_idx)      # index that will decrease (j ≠ i)
        max_delta = min(1.0 - cur[i], cur[j])
        if max_delta <= 0:
            continue

        # occasionally use the full transfer (swap) for binary‑like steps
        if rng.random() < 0.5:
            delta = max_delta
        else:
            delta = rng.random() * max_delta

        cand = cur.copy()
        cand[i] += delta
        cand[j] -= delta

        # ------- evaluate candidate --------------------------------------
        cand_val = c5_of(cand)

        # ------- acceptance rule -----------------------------------------
        # temperature linearly decays to 0
        T = max_temp * (1.0 - elapsed / budget_s)
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            if T > 0 and rng.random() < np.exp(-(cand_val - cur_val) / (T + 1e-12)):
                accept = True

        if accept:
            cur = cand
            cur_val = cand_val
            if cand_val < best_val - 1e-12:          # strict improvement
                best = cand.copy()
                best_val = cand_val
                stagnant = 0
                last_improve = time.time()
            else:
                stagnant += 1
        else:
            stagnant += 1

        # ------- occasional restart to avoid stagnation ------------------
        if stagnant > 8000:
            # restart from current best (or from a fresh random vector)
            if rng.random() < 0.3:
                # fresh random binary start
                cur = np.zeros(n, dtype=np.float64)
                cur[rng.choice(n, size=n // 2, replace=False)] = 1.0
                cur = enforce(cur)
            else:
                cur = best.copy()
            cur_val = c5_of(cur)
            stagnant = 0

    # Final sanity check
    best = enforce(best)
    best_val = c5_of(best)

    return best, best_val, n
