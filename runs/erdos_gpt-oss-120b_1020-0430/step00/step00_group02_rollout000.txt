import time, math, random
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    # ------------------------------------------------------------
    # Helper: compute C5 bound for a binary (0/1) step function h
    # ------------------------------------------------------------
    def c5_of(arr):
        # arr : 1‑D array of 0/1, length n
        # max_{k} ∫ h(x)(1‑h(x+k))dx ≈ max correlation * dx
        corr = np.correlate(arr, 1 - arr, mode='full')
        return float(np.max(corr)) * (2.0 / arr.size)

    # ------------------------------------------------------------
    # Initial parameters
    # ------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    total_time = budget_s * 0.94          # keep a margin for finalisation
    # target length – 45 (initial) scaled by 4 gives 180 points, a good trade‑off
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        scale = 4
        n = init.size * scale
        # repeat the pattern and keep the highest values as 1's
        repeated = np.repeat(init, scale)
        half = n // 2
        order = np.argsort(repeated)[::-1]
        h = np.zeros(n, dtype=np.int8)
        h[order[:half]] = 1
    else:
        n = 180
        half = n // 2
        h = np.zeros(n, dtype=np.int8)
        h[:half] = 1
        np.random.shuffle(h)

    # ------------------------------------------------------------
    # Simulated annealing / hill climbing
    # ------------------------------------------------------------
    best_h = h.copy()
    best_val = c5_of(best_h)
    cur_h = h.copy()
    cur_val = best_val

    # temperature schedule (linear from T_start to T_end)
    T_start = 0.5
    T_end = 1e-5

    # restart counter when stuck
    steps_without_improve = 0
    RESTART_LIMIT = 80000     # iterations with no improvement before a restart

    # fast random index selectors for ones and zeros
    def rand_one():
        while True:
            i = random.randrange(n)
            if cur_h[i] == 1:
                return i

    def rand_zero():
        while True:
            j = random.randrange(n)
            if cur_h[j] == 0:
                return j

    # main optimisation loop
    while True:
        now = time.time()
        if now - start_time >= total_time:
            break

        # cooling
        elapsed = now - start_time
        frac = elapsed / total_time
        T = max(T_start * (1.0 - frac) + T_end, T_end)

        # propose a swap of a 1 and a 0
        i = rand_one()
        j = rand_zero()
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]          # swap
        new_val = c5_of(cur_h)

        delta = new_val - cur_val
        accept = False
        if delta <= 0:
            accept = True
        else:
            # Metropolis criterion
            if random.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                steps_without_improve = 0
                # early stop if we beat the target bound
                if best_val <= 0.3808:
                    break
            else:
                steps_without_improve += 1
        else:
            # revert the swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            steps_without_improve += 1

        # restart if stuck for a while
        if steps_without_improve >= RESTART_LIMIT:
            # random re‑initialisation
            cur_h = np.zeros(n, dtype=np.int8)
            cur_h[:half] = 1
            np.random.shuffle(cur_h)
            cur_val = c5_of(cur_h)
            steps_without_improve = 0

    # ------------------------------------------------------------
    # Return the best discovered step function
    # ------------------------------------------------------------
    h_values = best_h.astype(float)          # required dtype is float in [0,1]
    c5_bound = best_val
    return h_values, c5_bound, n
