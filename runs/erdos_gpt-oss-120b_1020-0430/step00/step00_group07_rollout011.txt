            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def repair_sum(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # if diff > 0 need to increase sum -> add to entries with h<1
    # else need decrease sum -> subtract from entries with h>0
    # allocate diff greedily

        if diff > 0:
            candidates = np.where(h < 1 - 1e-12)[0]
            if len(candidates) == 0:
                break
            i = rng.choice(candidates)
            capacity = 1 - h[i]
            delta = min(capacity, diff)
            h[i] += delta
            diff -= delta

            if len(candidates) == 0:
                break
            i = rng.choice(candidates)
            capacity = h[i]  # amount we can decrease
            delta = min(capacity, -diff)
            h[i] -= delta
            diff += delta
    return h

import time, numpy as np

# ------------------- helper utilities ------------------- #
def evaluate(h):
    """C5 value for a step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def gradient_of_shift(h, k):
    """Gradient of L_k = sum h_i (1-h_{i+k}) w.r.t h, zero‑padded."""
    n = h.size
    grad = np.ones(n)
    if k >= 0:
        if k < n:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
    else:
        kk = -k
        if kk < n:
            grad[kk:] -= h[: n - kk]
            grad[: n - kk] -= h[kk:]
    return grad


def repair_sum(h, target, rng):
    """Clip to [0,1] and adjust a few entries so that sum(h)=target."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    # fix tiny numerical drift by moving mass on free coordinates

        if diff > 0:
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            delta = min(1.0 - h[i], diff)
            h[i] += delta
            diff -= delta
        else:
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            delta = min(h[i], -diff)
            h[i] -= delta
            diff += delta
    return h


def triangular_h(n, target, rng):
    """Initial triangular shape (tent) projected to the feasible set."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.minimum(x, 2.0 - x)
    return repair_sum(h, target, rng)


def sinusoidal_h(n, target, rng):
    """Sinusoidal shape 0.5+0.5*sin(pi x) projected to the feasible set."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = 0.5 + 0.5 * np.sin(np.pi * x)
    return repair_sum(h, target, rng)


# ------------------- local optimizer ------------------- #
def local_optimize(h0, deadline, target, rng):
    """Simple sub‑gradient / random‑move optimizer respecting the constraints."""
    n = h0.size
    cur = h0.copy()
    cur_score = evaluate(cur)
    best_h = cur.copy()
    best_score = cur_score
    it = 0
    step0 = 0.7  # initial step magnitude

    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.6:                               # gradient step
            corr = np.correlate(cur, 1.0 - cur, mode="full")
            max_val = corr.max()
            idxs = np.where(corr == max_val)[0]
            k = rng.choice(idxs) - (n - 1)
            grad = gradient_of_shift(cur, k)
            eta = step0 / np.sqrt(it + 1)
            h_new = cur - eta * grad
            h_new = repair_sum(h_new, target, rng)
        else:                                                # random mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            if rng.random() < 0.5:
                inc_i = 1.0 - cur[i]
                dec_j = cur[j]
                delta = min(inc_i, dec_j)
            else:
                inc_j = 1.0 - cur[j]
                dec_i = cur[i]
                delta = min(inc_j, dec_i)
            if delta <= 0.0:
                continue
            delta *= rng.random()
            h_new = cur.copy()
            if rng.random() < 0.5:
                h_new[i] += delta
                h_new[j] -= delta
            else:
                h_new[i] -= delta
                h_new[j] += delta
            # sum is preserved; only clipping may be needed
            np.clip(h_new, 0.0, 1.0, out=h_new)

        new_score = evaluate(h_new)

        # simple simulated‑annealing acceptance
        frac = (time.time() - (deadline - (budget_s * 0.95))) / (deadline - (deadline - (budget_s * 0.95)))
        T = max(1e-6, 0.3 * (1.0 - frac))
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur, cur_score = h_new, new_score
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score

        it += 1

        # occasional re‑heat from the best found so far
        if it % 50000 == 0 and rng.random() < 0.2:
            cur = best_h + 0.01 * rng.standard_normal(n)
            np.clip(cur, 0.0, 1.0, out=cur)
            cur = repair_sum(cur, target, rng)
            cur_score = evaluate(cur)

    return best_h, best_score


# ------------------- main entry point ------------------- #
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_time = budget_s * 0.95                     # keep a safety margin

    # choose resolution – a moderate size gives flexibility without heavy cost
    n = 200
    target_sum = n / 2.0

    # collect a few diverse starting points
    starts = []

    # 1) optionally use the provided construction (interpolated if sizes differ)
    init_parent = globals().get("initial_h_values")
    if init_parent is not None:
        parent = np.asarray(init_parent, dtype=float)
        if parent.size != n:
            # linear interpolation onto the chosen grid
            xs_src = np.linspace(0, 2, parent.size, endpoint=False) + 0.5 * (2.0 / parent.size)
            xs_tgt = (np.arange(n) + 0.5) * (2.0 / n)
            parent = np.interp(xs_tgt, xs_src, parent)
        starts.append(repair_sum(parent, target_sum, rng))

    # 2) triangular shape
    starts.append(triangular_h(n, target_sum, rng))

    # 3) sinusoidal shape
    starts.append(sinusoidal_h(n, target_sum, rng))

    # 4) a random feasible point
    rand = rng.random(n)
    starts.append(repair_sum(rand, target_sum, rng))

    best_h = None
    best_val = np.inf

    # split remaining time evenly among the start points
    per_start_deadline = start + total_time / len(starts)

    for h0 in starts:
        deadline = min(per_start_deadline, start + total_time)
        h_opt, val = local_optimize(h0, deadline, target_sum, rng)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()
        if time.time() >= start + total_time:
            break
        per_start_deadline = time.time() + total_time / len(starts)  # re‑balance

    # final safety projection and evaluation
    best_h = repair_sum(best_h, target_sum, rng)
    best_val = evaluate(best_h)

    return best_h, float(best_val), n
