def repair_vector(v, target):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # water-filling
    if diff > 0:
        # need to increase
        while diff > 1e-12:
            free = np.where(v < 1 - 1e-12)[0]
            if free.size == 0:
                break
            add = diff / free.size
            inc = np.minimum(add, 1 - v[free])
            v[free] += inc
            diff -= inc.sum()
    else:
        # need to decrease
        while diff < -1e-12:
            free = np.where(v > 1e-12)[0]
            if free.size == 0:
                break
            sub = (-diff) / free.size
            dec = np.minimum(sub, v[free])
            v[free] -= dec
            diff += dec.sum()
    # final clip
    v = np.clip(v, 0.0, 1.0)
    # ensure sum close
    # Maybe adjust tiny residual by adjusting first index:
    res = target - v.sum()
    if abs(res) > 1e-12:
        # Find any index not at bound
        idx = np.where((v > 0) & (v < 1))[0]
        if idx.size == 0:
            idx = np.arange(len(v))
        v[idx[0]] += res
        v = np.clip(v, 0.0, 1.0)

def repair_vector(v, target):
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # water-filling approach
    if diff > 0:
        # need to add
        while diff > 1e-12:
            free = np.where(v < 1 - 1e-12)[0]
            if free.size == 0:
                break
            add = diff / free.size
            inc = np.minimum(add, 1 - v[free])
            v[free] += inc
            diff -= inc.sum()
    else:
        while diff < -1e-12:
            free = np.where(v > 1e-12)[0]
            if free.size == 0:
                break
            sub = (-diff) / free.size
            dec = np.minimum(sub, v[free])
            v[free] -= dec
            diff += dec.sum()
    v = np.clip(v, 0.0, 1.0)
    # final small residual adjust
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        idx = np.argmax(v) if residual > 0 else np.argmin(v)
        v[idx] += residual
        v = np.clip(v, 0.0, 1.0)
    return v

import time
import math
import numpy as np

def _repair_vector(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the exact target sum."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # Water‑filling style adjustment
    if diff > 0.0:          # need to add mass
        while diff > 1e-12:
            free = np.where(v < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = np.minimum(diff / free.size, 1.0 - v[free])
            v[free] += inc
            diff -= inc.sum()
    else:                   # need to remove mass
        while diff < -1e-12:
            free = np.where(v > 1e-12)[0]
            if free.size == 0:
                break
            dec = np.minimum((-diff) / free.size, v[free])
            v[free] -= dec
            diff += dec.sum()
    v = np.clip(v, 0.0, 1.0)
    # tiny residual fix
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        idx = np.argmax(v) if residual > 0 else np.argmin(v)
        v[idx] = np.clip(v[idx] + residual, 0.0, 1.0)
    return v


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum‑overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Initialise the candidate vector
    # --------------------------------------------------------------
    base = globals().get("initial_h_values", None)
    if base is not None:
        base_arr = np.asarray(base, dtype=float)
        n0 = base_arr.size
        up_factor = 2                     # double resolution for more freedom
        n = n0 * up_factor
        h = np.repeat(base_arr, up_factor)
    else:
        n = 84
        h = rng.random(n)

    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = _repair_vector(h, target_sum)

    dx = 2.0 / n

    # --------------------------------------------------------------
    # 2. Helper to evaluate the objective C5
    # --------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        # max_k ∫ h(x)(1‑h(x+k)) dx  = max_k correlate(h,1‑h)*dx
        return np.max(np.correlate(vec, 1.0 - vec, mode="full")) * dx

    current = h.copy()
    cur_score = _score(current)
    best = current.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # 3. Simulated annealing core
    # --------------------------------------------------------------
    max_time = budget_s * 0.94  # reserve a little margin for final polishing
    T_start, T_end = 0.015, 1e-6
    step_max, step_min = 1.0, 0.001

    while time.time() - start < max_time:
        # schedule parameters
        elapsed = time.time() - start
        t_frac = elapsed / max_time
        T = T_start * (1.0 - t_frac) + T_end * t_frac
        step_factor = step_min + (step_max - step_min) * (1.0 - t_frac)

        # propose a transfer of ε from i to j (preserves the integral)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if rng.random() < 0.5:
            max_eps = min(current[i], 1.0 - current[j])
            if max_eps <= 0.0:
                continue
            eps = rng.uniform(0.0, max_eps) * step_factor
            new_vec = current.copy()
            new_vec[i] -= eps
            new_vec[j] += eps
        else:
            max_eps = min(current[j], 1.0 - current[i])
            if max_eps <= 0.0:
                continue
            eps = rng.uniform(0.0, max_eps) * step_factor
            new_vec = current.copy()
            new_vec[j] -= eps
            new_vec[i] += eps

        new_score = _score(new_vec)
        diff = new_score - cur_score
        if diff <= 0.0 or rng.random() < math.exp(-diff / (T + 1e-16)):
            current = new_vec
            cur_score = new_score
            if new_score < best_score:
                best = new_vec.copy()
                best_score = new_score

    # --------------------------------------------------------------
    # 4. Greedy refinement (only accept improvements)
    # --------------------------------------------------------------
    refine_deadline = start + budget_s * 0.99
    while time.time() < refine_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best[i], 1.0 - best[j])
        if max_eps <= 0.0:
            continue
        eps = rng.uniform(0.0, max_eps) * 0.002   # tiny step
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_score = _score(cand)
        if cand_score + 1e-12 < best_score:
            best = cand
            best_score = cand_score

    # --------------------------------------------------------------
    # 5. Final projection onto constraints & final score
    # --------------------------------------------------------------
    best = _repair_vector(best, target_sum)
    final_score = _score(best)

    # Return as plain Python list for compatibility
    return (best.tolist(), float(final_score), int(n))
