def compute_c5(h):
    n = len(h)
    m = 1 << (2*n-1).bit_length()
    # Actually we need m >= 2*n-1
    m = 1 << ((2*n-1).bit_length())
    # compute FFT
    H = np.fft.rfft(h, m)
    G = np.fft.rfft((1 - h)[::-1], m)
    conv = np.fft.irfft(H * G, m)[:2*n-1]
    c5 = np.max(conv) * (2.0 / n) # dx
    return c5

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft((1 - h)[::-1], m)
    conv = np.fft.irfft(H * G, m)[:2*n-1]
    return np.max(conv) * dx
```

            # Identify the lag that currently dominates the max.
            # ------------------------------------------------------------
            _, worst_lag, _, _ = _c5_and_lag(cur, m)

            # Determine the index range that participates in this lag.
            if worst_lag >= 0:
                i_start, i_end = 0, n - worst_lag
            else:
                i_start, i_end = -worst_lag, n
            if i_end <= i_start:
                continue
            idxs = np.arange(i_start, i_end)

            # Product for each overlapping pair; pick the pair with smallest product.
            prod = cur[idxs] * cur[idxs + worst_lag]
            i_min = idxs[int(np.argmin(prod))]

            # Choose a donor index (somewhere with positive mass, not part of the pair).
            mask = (cur > 1e-9) & (np.arange(n) != i_min) & (np.arange(n) != i_min + worst_lag)
            donors = np.where(mask)[0]
            if donors.size == 0:               # fallback – simple swap
                a, b = np.random.randint(n, size=2)
                if a == b:
                    continue
                max_d = min(cur[a], 1.0 - cur[b])
                if max_d <= 0:
                    continue
                delta = np.random.rand() * max_d
                cur[a] -= delta
                cur[b] += delta
                new_score = evaluate(cur)
                # acceptance
                if new_score < cur_score or (np.random.rand() < math.exp(-(new_score - cur_score) / max(T0 * (deadline - time.time()) / budget_s, 1e-12))):
                    cur_score = new_score
                    if new_score < best_score:
                        best_score, best_h = new_score, cur.copy()
                else:
                    cur[a] += delta
                    cur[b] -= delta
                continue

            donor = np.random.choice(donors)

            # Randomly decide which side of the worst pair to reinforce.
            target = i_min if np.random.rand() < 0.5 else i_min + worst_lag

            max_delta = min(cur[donor], max_step, 1.0 - cur[target])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            cur[donor] -= delta
            cur[target] += delta

            new_score = evaluate(cur)

            # ------------------------------------------------------------
            # Accept / reject (simulated annealing)
            # ------------------------------------------------------------
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score, best_h = new_score, cur.copy()
                # break early if we already beat the target bound
                if best_score < 0.3808:
                    c5 = evaluate_erdos_solution(best_h.tolist())
                    return best_h.tolist(), c5, n
            else:
                # temperature decreasing linearly with remaining global time
                temp = T0 * max(0.0, (deadline - time.time()) / budget_s)
                if temp > 0 and np.random.rand() < math.exp(-(new_score - cur_score) / temp):
                    cur_score = new_score
                    if new_score < best_score:
                        best_score, best_h = new_score, cur.copy()
                else:
                    # revert move
                    cur[donor] += delta
                    cur[target] -= delta

        # after this inner run, possibly keep the best found so far
        if cur_score < best_score:
            best_score, best_h = cur_score, cur.copy()

    # ------------------------------------------------------------------
    # 4. Final tidy‑up: enforce exact integral and report result.
    # ------------------------------------------------------------------
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        # Add the tiny missing mass to a non‑saturated entry
        free = np.where(best_h < 1.0 - 1e-12)[0]
        if free.size:
            best_h[free[0]] = np.clip(best_h[free[0]] + diff, 0.0, 1.0)
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    c5 = evaluate_erdos_solution(best_h.tolist())
    return best_h.tolist(), c5, n
