def gradient_for_shift(h, shift, dx):
    n = len(h)
    grad = np.full(n, dx)  # from sum term
    if shift >= 0:
        # overlapping indices: i from 0..n-1-shift
        # grad[i] -= dx * h[i+shift]
        # grad[i+shift] -= dx * h[i]
        grad[:n-shift] -= dx * h[shift:]
        grad[shift:] -= dx * h[:n-shift]
    else:

        # grad[i-s] -= dx * h[i]
        grad[s:] -= dx * h[:n-s]
        grad[:n-s] -= dx * h[s:]
    return grad

def project_to_sum(v, target):
    lo = -1.0
    hi = 1.0
    for _ in range(30):
        mid = (lo+hi)/2
        proj = np.clip(v + mid, 0, 1)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0, 1)

def simulated_annealing(initial_h, max_time):
    n = len(initial_h)
    target = n/2
    # ensure feasible
    h = project_to_sum(initial_h, target)
    best_h = h.copy()

    start = time.time()
    # schedule parameters
    temp_start = 0.2
    temp_end = 1e-4
    # step factor
    max_step = 0.2  # maximal epsilon magnitude

    iter = 0
    while time.time() - start < max_time:
        # compute temperature linearly decreasing
        t_frac = (time.time() - start) / max_time
        temp = temp_start * (1 - t_frac) + temp_end * t_frac

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: project a vector onto the box [0,1] and the sum constraint.
# --------------------------------------------------------------
def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Find λ such that sum(clamp(v+λ,0,1)) = target (monotone, binary search)."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) * 0.5
    return np.clip(v + λ, 0.0, 1.0)


# --------------------------------------------------------------
# Compute C5 for a given step function h (discrete vector).
# --------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


# --------------------------------------------------------------
# Simulated annealing for a fixed size n.
# --------------------------------------------------------------
def _simulated_annealing(start_h: np.ndarray, max_seconds: float) -> tuple:
    """
    Simple SA that moves mass between two coordinates while keeping
    0<=h_i<=1 and the total sum fixed. Returns best (h, C5) found.
    """
    n = start_h.shape[0]
    target_sum = n / 2.0

    # make the start feasible (projection only once)
    h = _project_to_sum(start_h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_value(h)
    cur_c5 = best_c5

    start = time.time()
    # temperature schedule
    T_start, T_end = 0.3, 1e-5
    max_step = 0.5  # maximal epsilon magnitude (scaled later)

    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        t_frac = elapsed / max_seconds
        T = T_start * (1.0 - t_frac) + T_end * t_frac
        step = max_step * (T / T_start)  # shrink step with temperature

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)

        # direction of transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # feasible epsilon range
        eps_max = min(h[src], 1.0 - h[dst])
        if eps_max <= 0.0:
            continue

        eps = (np.random.rand() * 2.0 - 1.0) * step
        eps = np.clip(eps, -eps_max, eps_max)
        if abs(eps) < 1e-12:
            continue

        # propose new configuration
        h_new = h.copy()
        h_new[src] -= eps
        h_new[dst] += eps

        new_c5 = _c5_value(h_new)
        delta = new_c5 - cur_c5

        if delta <= 0.0 or np.random.rand() < math.exp(-delta / T):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

    return best_h, best_c5


# --------------------------------------------------------------
# Main entry point required by the harness.
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where:
      - h_values : 1‑D NumPy array of step values on [0,2]
      - c5_bound : the achieved C5 value (float)
      - n_points : number of discretisation points used
    """
    np.random.seed(seed)
    start_all = time.time()

    # ------------------------------------------------------------------
    # Prepare a few candidate problem sizes.
    # ------------------------------------------------------------------
    candidates = []
    # use the provided construction if it exists
    if "initial_h_values" in globals():
        init = initial_h_values.astype(float).copy()
        candidates.append((init.size, init))
    # add a few larger sizes (still well below 1000 points)
    for sz in (100, 200, 300):
        candidates.append((sz, None))

    # time split (keep a small guard buffer)
    per_candidate = max(0.0, (budget_s - 0.5) / len(candidates))

    best_h_overall = None
    best_c5_overall = float("inf")
    best_n_overall = None

    for n, given_h in candidates:
        # stop if global budget exhausted
        if time.time() - start_all >= budget_s:
            break
        # allocate time for this sub‑run
        time_left = budget_s - (time.time() - start_all)
        sub_budget = min(per_candidate, time_left)

        # ------------------------------------------------------------------
        # Build an initial vector of length n.
        # ------------------------------------------------------------------
        if given_h is not None:
            if given_h.size != n:
                # linear interpolation to required length
                src_x = np.linspace(0.0, 2.0, given_h.size, endpoint=False)
                tgt_x = np.linspace(0.0, 2.0, n, endpoint=False)
                h0 = np.interp(tgt_x, src_x, given_h)
            else:
                h0 = given_h.copy()
        else:
            h0 = np.random.rand(n)

        # ------------------------------------------------------------------
        # Run the optimizer for the allocated slice of the budget.
        # ------------------------------------------------------------------

        if c5_opt < best_c5_overall:

    # Fallback (should never happen, but keep a safe default)
    # ----------------------------------------------------------------------
    if best_h_overall is None:
        best_n_overall = 42
        best_h_overall = np.full(best_n_overall, 0.5)
        best_c5_overall = _c5_value(best_h_overall)

    return (best_h_overall, float(best_c5_overall), int(best_n_overall))
