def project_to_constraints(v, target_sum):
    # project onto [0,1] hyperbox with sum constraint
    # using iterative clipping and water-filling.
    v = np.clip(v, 0.0, 1.0)
    # iterate a few times
    for _ in range(10):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        free = np.where((v > 0) & (v < 1))[0]
        if free.size == 0:
            # Distribute diff equally among all, then clip
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final clip and adjust small error if any
    v = np.clip(v, 0.0, 1.0)
    # adjust sum final residual via projection onto simplex of remaining capacity
    # but we can just accept small error.
    return v

import time
import math
import random
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """Return the maximum overlap count (without dx factor)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max())

def _project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the box [0,1]^n with a
    prescribed sum `target_sum`.  Simple water‑filling iteration.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(20):  # enough for convergence on n≈200
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            # distribute over all entries, then clip again
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    return v

def _refine_fractional(h_start: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Small gradient–descent refinement on the smooth max objective.
    Operates on fractional values while keeping the constraints.
    """
    N = h_start.size
    target = N / 2.0
    h = h_start.copy()
    start = time.time()
    iteration = 0
    while time.time() - start < time_limit:
        iteration += 1
        # correlation
        C = np.correlate(h, 1.0 - h, mode="full")
        # smooth‑max weights (softmax)
        alpha = 5.0 + iteration * 1e-3  # slowly increase focus on the worst lag
        expC = np.exp(alpha * C)
        w = expC / expC.sum()
        # gradient of the smooth max (see analysis)
        g = np.ones_like(h)
        mid = N - 1
        for idx, wk in enumerate(w):
            if wk == 0.0:
                continue
            k = idx - mid
            if k >= 0:
                g[: N - k] -= wk * h[k:]
                g[k:]       -= wk * h[: N - k]
            else:
                kk = -k
                g[kk:]      -= wk * h[: N - kk]
                g[: N - kk] -= wk * h[kk:]
        # step size (decreasing)
        step = 0.02 / (1.0 + iteration * 0.001)
        h_new = h - step * g
        h_new = _project_to_constraints(h_new, target)
        if _max_overlap(h_new) < _max_overlap(h):
            h = h_new
    return h
