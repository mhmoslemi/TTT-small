def project_to_bounds_and_sum(x, target):
    # clip to [0,1], then adjust sum
    x = np.clip(x, 0, 1)
    for _ in range(10):
        diff = target - x.sum()
        if abs(diff) < 1e-10:
            break
        # compute number of free (0< x <1)
        free = (x > 0) & (x < 1)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds but sum mismatch: adjust by distributing to all (allow small violation)
            x += diff / len(x)
            x = np.clip(x, 0, 1)
        else:
            delta = diff / n_free
            x[free] += delta
            # clip again
            x = np.clip(x, 0, 1)
    return x
```

def project_to_bounds_and_sum(v, target, max_iter=100):
    lo = -1.0  # large negative to allow values shift up
    hi = 1.0   # shift down
    # Actually λ can be any real.
    # However we can start with lambda bounds such that sum_i min(max(v_i - λ, 0),1) is monotonic decreasing in λ.
    # Use bisection to find λ where sum equals target.
    # compute sum for a given λ:
    def sum_proj(lam):
        return np.clip(v - lam, 0, 1).sum()
    # Determine bounds
    # For lam low (very negative), v - lam large -> clipped to 1: sum = len(v)
    # For lam high (very positive), v - lam negative -> clipped to 0: sum = 0
    lo, hi = -2.0, 2.0
    # Expand if needed:
    while sum_proj(lo) < target:
        lo -= 1.0
    while sum_proj(hi) > target:
        hi += 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = sum_proj(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(v - (lo+hi)/2, 0, 1)
    return projected
```

import time
import numpy as np

def _compute_c5(h, dx):
    """Maximum overlap integral for a given h."""
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()

def _project_bounds_sum(v, target, max_iter=60):
    """
    Euclidean projection of v onto the set {x : 0<=x_i<=1, sum x_i = target}.
    Uses a bisection on the Lagrange multiplier lambda for the box-constrained simplex.
    """
    def sum_proj(lam):
        return np.clip(v - lam, 0.0, 1.0).sum()

    lo, hi = -2.0, 2.0
    # expand bounds until they bracket the solution
    while sum_proj(lo) < target:
        lo -= 1.0
    while sum_proj(hi) > target:
        hi += 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        s = sum_proj(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(n, target):
    """Generate a random feasible h (box constraints + sum)."""
    v = np.random.rand(n)
    return _project_bounds_sum(v, target)

def _resample(v, n):
    """Linear resampling of a 1‑D array v to length n."""
    if len(v) == n:
        return v.copy()
    old_x = np.arange(len(v), dtype=float)
    new_x = np.linspace(0.0, len(v) - 1.0, n)
    return np.interp(new_x, old_x, v)

def _descend(h0, dx, target, max_iter=4000, start_alpha=2.0):
