def project_capped(v, S):
    # initial clip
    x = np.clip(v, 0, 1)
    # compute sum difference
    diff = S - x.sum()
    # iterative adjust
    # We maintain mask of indices where x not at bound (0 or 1)
    max_iter = 10 * len(v)  # just safe
    for _ in range(max_iter):
        if abs(diff) < 1e-10:
            break
        # Determine free indices
        free = np.where((x > 0) & (x < 1))[0]
        if free.size == 0:
            # cannot adjust further due to all at bound, but diff may not be zero w.r.t rounding.
            # In that case we adjust by distributing to all variables then clip again
            # fallback: adjust all (allow small violation) then clip.
            x += diff / len(v)
            x = np.clip(x, 0, 1)
            diff = S - x.sum()

        # compute any overshoot

        under_zero = x < 0

        diff = S - x.sum()
        # Clip at bounds
        x = np.clip(x, 0, 1)
        # compute new diff
        diff = S - x.sum()
    # Ensure sum close

def project_capped(v, S, tol=1e-9):
    # find tau such that sum_i clamp(v_i + tau, 0, 1) = S
    low = -1 - v.max()
    high = 1 - v.min()
    for _ in range(40):
        tau = (low + high) / 2
        x = np.clip(v + tau, 0, 1)
        s = x.sum()
        if abs(s - S) <= tol:
            return x
        if s < S:
            low = tau
        else:
            high = tau
    return np.clip(v + (low + high) / 2, 0, 1)

def gradient_descent(h0, max_iters, end_time, dx):
    h = h0.copy()
    N = len(h)
    target_sum = N / 2.0
    best_h = h.copy()
    best_val = evaluate_C5(h, dx)
    for it in range(max_iters):
        if time.time() > end_time:
            break
        # compute correlation
        pad = 2*N - 1
        H = np.fft.rfft(h, n=pad)
        G = np.fft.rfft(1 - h, n=pad)
        corr = np.fft.irfft(H * np.conj(G), n=pad)
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx] * dx
        shift = max_idx - (N - 1)
        # compute subgradient
        g = np.zeros(N, dtype=float)
        if shift >= 0:
            L = N - shift
            if L > 0:
                g[:L] += 1
                g[:L] -= h[shift:]
                g[shift:] -= h[:L]
        else:
            s = -shift
            L = N - s
            if L > 0:
                g[s:] += 1
                g[s:] -= h[:L]
                g[:L] -= h[s:]
        # step size
        alpha = 0.25 * (1.0 - it / max_iters)  # start 0.25
        h_new = h - alpha * g
        # project
        h_new = project_capped(h_new, target_sum)
        # compute new val
        H2 = np.fft.rfft(h_new, n=pad)
        G2 = np.fft.rfft(1 - h_new, n=pad)
        corr2 = np.fft.irfft(H2 * np.conj(G2), n=pad)
        new_val = corr2.max() * dx
        # accept
        if new_val <= cur_val:
            h = h_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # optionally allow acceptance? skip.
            pass
    return best_h, best_val

def simulated_annealing(h0, dx, end_time, rng):
    h = h0.copy()
    N = len(h)
    cur_val = evaluate_C5(h, dx)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    total = end_time - start
    if total <= 0:
        return best_h, best_val
    # temperature schedule start based on cur_val
    T0 = max(1e-6, cur_val * 0.8)  # somewhat
    while time.time() < end_time:
        now = time.time()
        elapsed = now - start
        frac = elapsed / total
        T = max(1e-9, T0 * (1 - frac))
        # pick two distinct indices
        i, j = rng.choice(N, size=2, replace=False)
        # calculate bounds for delta
        # delta positive: transfer from j to i
        delta_max = min(1.0 - h[i], h[j])
        delta_min = -min(h[i], 1.0 - h[j])
        if delta_max <= delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max)
        # apply
        h[i] += delta
        h[j] -= delta
        new_val = evaluate_C5(h, dx)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # accept with probability
            if rng.random() < math.exp(-(new_val - cur_val) / T):
                accept = True
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta
    return best_h, best_val

import time, math, numpy as np

def _project_capped(v, total, tol=1e-12, max_iter=60):
    """Project v onto the set {x in [0,1]^n : sum(x)=total}."""
    low = -1.0 - v.max()
    high = 1.0 - v.min()
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - total) <= tol:
            return x
        if s < total:
            low = mid
        else:
            high = mid
    # final clamp
    x = np.clip(v + (low + high) * 0.5, 0.0, 1.0)
    # tiny correction to hit the sum exactly
    diff = total - x.sum()
    if abs(diff) > tol:
        idx = np.where((x > 0.0) & (x < 1.0))[0]
        if idx.size:
            add = diff / idx.size
            x[idx] = np.clip(x[idx] + add, 0.0, 1.0)
    return x

def _evaluate_c5(h):
    """Return the max overlap C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    pad = 2 * n - 1
    H = np.fft.rfft(h, n=pad)
    G = np.fft.rfft(1.0 - h, n=pad)
    corr = np.fft.irfft(H * np.conj(G), n=pad)
    return float(corr.max() * dx)

def _subgradient(h, shift):
    """Subgradient of the active term C_k at given shift."""
    n = h.shape[0]
    g = np.zeros(n, dtype=np.float64)
    if shift >= 0:
        L = n - shift
        if L > 0:
            g[:L] += 1.0
            g[:L] -= h[shift:]
            g[shift:] -= h[:L]
    else:
        s = -shift
        L = n - s
        if L > 0:
            g[s:] += 1.0
            g[s:] -= h[:L]
            g[:L] -= h[s:]
    return g

def _gradient_descent(start_h, max_iters, end_time):
    """Simple projected sub‑gradient descent."""
    h = start_h.copy()
    n = h.shape[0]
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_val = _evaluate_c5(h)

    for it in range(max_iters):
        if time.time() >= end_time:
            break
        pad = 2 * n - 1
        H = np.fft.rfft(h, n=pad)
        G = np.fft.rfft(1.0 - h, n=pad)
        corr = np.fft.irfft(H * np.conj(G), n=pad)
        idx = int(corr.argmax())
        cur_val = float(corr[idx] * dx)
        shift = idx - (n - 1)

        g = _subgradient(h, shift)
        step = 0.25 * (1.0 - it / max_iters)  # decreasing step
        h_new = _project_capped(h - step * g, target)

        # evaluate new point
        H2 = np.fft.rfft(h_new, n=pad)
        G2 = np.fft.rfft(1.0 - h_new, n=pad)
        corr2 = np.fft.irfft(H2 * np.conj(G2), n=pad)
        new_val = float(corr2.max() * dx)

        if new_val <= cur_val:
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
    return best_h, best_val

def _simulated_annealing(start_h, end_time, rng):
    """Metropolis style annealing using pairwise mass transfers."""
    h = start_h.copy()
    n = h.shape[0]
    dx = 2.0 / n
    cur_val = _evaluate_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    total = end_time - start
    if total <= 0.0:
        return best_h, best_val
    T0 = max(1e-6, cur_val * 0.6)

    while True:
        now = time.time()
        if now >= end_time:
            break
        frac = (now - start) / total
        T = max(1e-12, T0 * (1.0 - frac))

        i, j = rng.choice(n, size=2, replace=False)
        # allowable transfer from j to i while staying in [0,1]
        delta_max = min(1.0 - h[i], h[j])
        delta_min = -min(h[i], 1.0 - h[j])
        if delta_max <= delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max)

        h[i] += delta
        h[j] -= delta
        new_val = _evaluate_c5(h)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if rng.random() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    return best_h, best_val

def _make_triangle(n):
    """Return a triangular step vector on n points (peak 1, zero at the ends)."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    # enforce exact integral constraint by projection
    return _project_capped(h, n / 2.0)

def _upsample(v, new_n):
    """Linear interpolation from len(v) to new_n points, then project."""
    old_n = v.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    v_new = np.interp(x_new, x_old, v)
    return _project_capped(v_new, new_n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s

    # target resolution – moderate size for a good trade‑off
    target_n = 300

    # initialise: upsample supplied construction or start from a triangle
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        if h0.shape[0] != target_n:
            h = _upsample(h0, target_n)
        else:
            h = _project_capped(h0, target_n / 2.0)
    else:
        h = _make_triangle(target_n)

    # Phase 1: short gradient descent (≈20 % of time)
    remaining = end_time - time.time()
    gd_end = time.time() + remaining * 0.2
    max_iters = max(2000, int(5000 * 45 / target_n))
    h, best_val = _gradient_descent(h, max_iters, gd_end)

    # Phase 2: simulated annealing for the rest of the budget
    if time.time() < end_time:
        h, best_val = _simulated_annealing(h, end_time, rng)

    # final clean‑up – enforce constraints exactly
    h = np.clip(h, 0.0, 1.0)
    h = _project_capped(h, target_n / 2.0)
    best_val = _evaluate_c5(h)

    return h.tolist(), best_val, target_n
