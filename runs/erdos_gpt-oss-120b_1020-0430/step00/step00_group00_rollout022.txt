def project_onto_box_and_sum(y, target):
    lo = -np.max(y) - 1  (maybe -2)
    hi = 1 - np.min(y) + 1

        lam = (lo+hi)/2
        x = np.clip(y + lam, 0, 1)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(y + (lo+hi)/2, 0, 1)

def compute_gradient(h, k_vals):
    n = len(h)
    grad = np.zeros_like(h)
    for k in k_vals:
        # For each i, compute contribution 1 - (h_i+k if in range else 0) - (h_i-k if in range else 0)
        # Use vectorized:
        # shift forward (i+k)
        forward = np.zeros(n)
        backward = np.zeros(n)
        if k >=0:
            # forward: h[i+k] for i where i+k < n
            forward[:n-k] = h[k:]
            # backward: h[i-k] for i >= k
            backward[k:] = h[:n-k]
        else:
            # k negative: forward shift backward
            kk = -k
            forward[kk:] = h[:n-kk]
            backward[:n-kk] = h[kk:]
        grad += 1 - forward - backward
    grad /= len(k_vals) # average
    return grad

def propose_swap(h, delta):
    # Choose i != j
    i, j = np.random.choice(n, 2, replace=False)
    # Compute max delta feasible given bounds
    max_decrease = min(delta, h[i]) # cannot go below 0
    max_increase = min(delta, 1 - h[j]) # cannot exceed 1
    actual = min(max_decrease, max_increase)

def max_corr_and_lags(h):

    maxc = corr.max()
    idxs = np.where(corr == maxc)[0]

def gradient(h, ks):
    n = len(h)

def _project_box_sum(y, target, tol=1e-12, max_iter=100):
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def _max_overlap(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def _max_corr_info(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    idx = np.where(corr == m)[0]
    ks = idx - (h.size - 1)
    return m, ks

def _gradient(h, ks):
    n = h.size
    g = np.zeros_like(h)
    for k in ks:
        g += 1.0
        if k >= 0:
            nk = n - k
            g[:nk] -= h[k:]
            g[k:] -= h[:nk]
        else:
            kk = -k
            nk = n - kk
            g[kk:] -= h[:nk]
            g[:nk] -= h[kk:]
    return g / len(ks)

    # choose a resolution that balances quality & speed
    n = 150
    target_sum = n / 2.0

    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        if base.size == n:
            h0 = base.copy()
        elif n % base.size == 0:
            h0 = np.repeat(base, n // base.size)
        else:
            x_old = np.linspace(0.0, 2.0, base.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, base)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project_box_sum(h0, target_sum)
    else:
        h0 = rng.random(n)
        h0 = _project_box_sum(h0, target_sum)

    remaining = budget_s * 0.98
    best_h, best_val = _optimize_one_start(h0, remaining, target_sum, rng)

    best_h = _project_box_sum(best_h, target_sum)
    best_val = _max_overlap(best_h)

    return best_h, best_val, n
