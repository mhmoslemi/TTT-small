def local_search(h, max_iter, time_budget):

    N = len(h)
    ones = np.flatnonzero(h==1)
    zeros = np.flatnonzero(h==0)
    best = h.copy()
    best_c = compute_c5(best)
    start = time.time()
    it = 0
    while time.time() - start < time_budget and it < max_iter:
        # choose random swap
        a = random.choice(ones)
        b = random.choice(zeros)
        h_new = h.copy()
        h_new[a] = 0; h_new[b] = 1
        new_c = compute_c5(h_new)
        if new_c < best_c - 1e-12:
            h = h_new
            best = h_new.copy()
            best_c = new_c
            # update ones, zeros lists
            ones = np.flatnonzero(h==1)
            zeros = np.flatnonzero(h==0)
        it += 1
    return best, best_c

def compute_c5(arr):
    N = len(arr)
    # arr maybe float but fine.
    corr = np.correlate(arr, 1 - arr, mode='full')
    max_corr = corr.max()
    return max_corr * (2.0 / N)

def hill_climb(N, init_h, max_iters, time_limit):
    best_h = init_h.copy()
    best_c5 = compute_c5(best_h)
    h = best_h.copy()
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    start = time.time()

        a = np.random.choice(ones)

            # update lists

            zeros = np.flatnonzero(h == 0)
        it += 1
    return best_h, best_c5

def solve_mip(N):
    import cvxpy as cp
    h = cp.Variable(N, boolean=True)
    t = cp.Variable(integer=True)
    w = cp.Variable((N, N), boolean=True)
    constraints = []
    constraints.append(cp.sum(h) == N//2)
    constraints.append(t >= 0)
    for i in range(N):
        for k in range(N):
            j = i + k
            if j < N:
                constraints.append(w[i, k] <= h[i])
                constraints.append(w[i, k] <= h[j])
                constraints.append(w[i, k] >= h[i] + h[j] - 1)
            else:
                constraints.append(w[i, k] == 0)
    for k in range(N):
        constraints.append(cp.sum(w[:, k]) >= N//2 - t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
        except Exception:
            return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None
    h_sol = np.rint(h.value).astype(int)
    return h_sol, int(round(t.value))

def random_feasible(N):
    # sample random values in [0,1] and then project to sum=N/2 using simple scaling and clipping iterative.
    x = np.random.rand(N)
    # scale to have sum target
    total = x.sum()
    x = x * (N/2) / total
    # after scaling, some may exceed 1; clip and reallocate.
    # simple iterative: while any >1, set them to 1, remove their excess, and renorm remaining.
    excess = x > 1
    while np.any(excess):
        x[excess] = 1.0
        remaining = np.where(~excess)[0]
        if len(remaining) == 0:
            break
        surplus = x.sum() - N/2
        # subtract surplus proportionally from remaining
        # Actually we want to reduce sum to target; but we have overshoot: x.sum() > N/2, need to reduce.
        # Compute total mass of remaining
        total_rem = x[remaining].sum()
        if total_rem == 0:
            break
        reduction = surplus * x[remaining] / total_rem
        x[remaining] -= reduction
        excess = x > 1
    # Ensure non-negative, clip
    x = np.clip(x, 0, 1)
    # after this may not sum exactly to N/2 due to rounding. but close.
    # final adjust by adding small to random indices that have capacity.
    diff = N/2 - x.sum()
    # add diff equally to some positions with space
    # For simplicity ignore diff (should be small) or renorm
    if abs(diff) > 1e-8:
        # distribute diff equally
        idx = np.random.choice(N, size=1)
        x[idx] += diff
        x = np.clip(x, 0, 1)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    start_total = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Determine N and default h0
    try:
        init_h_arr = np.array(initial_h_values, dtype=float)
        N = len(init_h_arr)
    except NameError:
        N = 100
        init_h_arr = None

    if N % 2 == 1:
        N += 1
        if init_h_arr is not None:
            init_h_arr = np.concatenate([init_h_arr, [0.0]])

    target_sum = N / 2.0
    dx = 2.0 / N

    # Create basic initial guesses
    init_guesses = []

    # uniform 0.5
    init_guesses.append(np.full(N, 0.5))

    # triangular shape
    x = np.arange(N) * dx + dx/2.0  # sample in middle of intervals
    tri = np.where(x < 1.0, x, 2.0 - x)  # peaks at 1
    # adjust sum to target
    current_sum = tri.sum()
    if current_sum != 0:
        scaling = target_sum / current_sum
        tri = tri * scaling
        # ensure within [0,1]
        tri = np.clip(tri, 0.0, 1.0)
        # re-balance small diff
        diff = target_sum - tri.sum()
        if abs(diff) > 1e-8:
            # distribute evenly to capacities
            # find indices with slack
            cap = 1.0 - tri
            total_cap = cap.sum()
            if total_cap > 0:
                tri += diff * cap / total_cap
                tri = np.clip(tri, 0.0, 1.0)
    init_guesses.append(tri)

    # if initial_h_values provided, also add normalized version
    if init_h_arr is not None:
        # project onto bounds [0,1] and adjust sum
        h = np.clip(init_h_arr, 0.0, 1.0)
        sum_h = h.sum()
        if sum_h != 0:
            h = h * (target_sum / sum_h)
            # reclip if exceeds 1
            h = np.clip(h, 0.0, 1.0)
        init_guesses.append(h)

    best_h = None
    best_c5 = float('inf')

    def compute_c5_vec(vec):

        return np.max(corr) * (2.0 / N)

    def constraint_sum(vec):
        return np.sum(vec) - target_sum

    def grad_obj(vec):
        corr = np.correlate(vec, 1 - vec, mode='full')
        idx = np.argmax(corr)
        d = abs(idx - (N - 1))
        grad = np.ones(N)
        if d > 0:
            grad[:N - d] -= vec[d:]
            grad[d:] -= vec[:N - d]
        # Scale gradient because objective multiplies by 2/N
        return grad * (2.0 / N)

    bounds = [(0.0, 1.0)] * N
    cons = {'type': 'eq', 'fun': constraint_sum}

    # Optimization loop across initial guesses

        if time.time() - start_total > budget_s - 1.0:

import time
import random
import numpy as np
import scipy.optimize as opt

def _compute_c5_vec(v, N):
    """C5 value for a given vector v (length N)."""
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * (2.0 / N)

def _constraint_sum(v, target):
    """Equality constraint: sum(v) == target."""
    return np.sum(v) - target

def _grad_max_corr(v, N):
    """Gradient of max_k Σ h_i (1‑h_{i+k}) w.r.t. v."""
    corr = np.correlate(v, 1.0 - v, mode='full')
    idx = np.argmax(corr)
    d = abs(idx - (N - 1))               # distance of the worst shift
    if d == 0:
        grad = 1.0 - 2.0 * v               # k = 0 case
    else:
        grad = np.ones(N)
        grad[: N - d] -= v[d:]            # subtract h_{i+d}
        grad[d:]       -= v[: N - d]      # subtract h_{i-d}
    grad *= (2.0 / N)                     # scaling because of the dx factor
    return grad

def _prepare_initial_guesses(N, target_sum):
    """Generate a few sensible starting points."""
    guesses = []

    # 1. Uniform 0.5 (exactly satisfies the sum constraint)
    guesses.append(np.full(N, 0.5))

    # 2. Triangular (“tent”) shape, scaled to have the correct integral
    dx = 2.0 / N
    x = np.arange(N) * dx + dx / 2.0          # mid‑point samples
    tri = np.where(x < 1.0, x, 2.0 - x)      # height grows to 1 at the centre
    cur = tri.sum()
    if cur != 0:
        tri = tri * (target_sum / cur)
        tri = np.clip(tri, 0.0, 1.0)
        diff = target_sum - tri.sum()
        if abs(diff) > 1e-12:                # tiny redistribution to hit the sum exactly
            slack = 1.0 - tri
            tot = slack.sum()
            if tot > 0:
                tri += diff * slack / tot
                tri = np.clip(tri, 0.0, 1.0)
    guesses.append(tri)

    # 3. If a global initial_h_values exists, use it (projected onto the box)
    try:
        global initial_h_values
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        s = h0.sum()
        if s != 0:
            h0 = h0 * (target_sum / s)
            h0 = np.clip(h0, 0.0, 1.0)
        guesses.append(h0)
    except NameError:
        pass

    return guesses

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low C₅ bound."""
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size (use the provided initial construction if any)
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        N = len(init_arr)
    except NameError:
        N = 100                     # fall back to a modest size
        init_arr = None

    if N % 2 == 1:                 # must be even to satisfy the integral exactly
        N += 1
        if init_arr is not None:
            init_arr = np.concatenate([init_arr, [0.0]])

    target_sum = N / 2.0            # because ∫h = 1 and the interval length is 2
    bounds = [(0.0, 1.0)] * N
    cons = {'type': 'eq', 'fun': lambda v: _constraint_sum(v, target_sum)}

    # Helper for the optimizer (captures N via closure)
    def obj(v):
        return _compute_c5_vec(v, N)

    def grad(v):
        return _grad_max_corr(v, N)

    best_h = None
    best_val = float('inf')

    for guess in _prepare_initial_guesses(N, target_sum):
