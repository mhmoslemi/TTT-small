import time
import numpy as np

def _c5_bound(h):
    """Return the C5 bound for a step function h (as a numpy array)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def _max_shift_and_corr(h):
    """Return (k_max, max_raw, corr_array) where k_max is the shift achieving
    the maximal raw overlap (without the dx factor)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_raw = corr[max_idx]
    k = max_idx - (n - 1)               # shift in index units (negative allowed)
    return k, max_raw, corr

def _gradient_for_shift(h, k):
    """Gradient of the raw overlap for a fixed shift k (ignoring the max)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        # (1 - h[i+k]) part
        g[: n - k] = 1.0 - h[k:]
        # -h[i-k] part
        g[k:] -= h[: n - k]
    else:
        k_abs = -k
        # (1 - h[i+k]) = (1 - h[i-k_abs]) for i >= k_abs
        g[k_abs:] = 1.0 - h[: n - k_abs]
        # -h[i-k] = -h[i+k_abs] for i < n - k_abs
        g[: n - k_abs] -= h[k_abs:]
    return g

def _repair_sum(h):
    """Adjust h (in‑place) so that sum(h) == n/2 while staying in [0,1]."""
    n = h.shape[0]
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # simple sweep – add/subtract the remaining slack on the first possible entry
    for i in range(n):
        if diff > 0.0:
            cap = 1.0 - h[i]
            add = min(cap, diff)
            h[i] += add
            diff -= add
        else:
            cap = h[i]
            sub = min(cap, -diff)
            h[i] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    # final safety clipping
    np.clip(h, 0.0, 1.0, out=h)
    return h
