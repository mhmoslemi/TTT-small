def project_to_sum_and_bounds(h, target_sum):
    # Clip to bounds first
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    diff = target_sum - current
    # Adjust using water-filling: add diff equally to all, then reclip iteratively
    if abs(diff) > 1e-12:
        # alias to t
        # We'll do iterative approach: while diff not small
        # At each iteration compute adjustable indices
        adjustable = (h > 0) & (h < 1)  # not at bounds
        if adjustable.any():
            # distribute diff equally on adjustable
            inc = diff / adjustable.sum()
            h[adjustable] += inc
            # Clip again
            h = np.clip(h, 0.0, 1.0)
        else:
            # all at bounds, need to force some movement: force to adjust the farthest from bounds
            # We'll just add diff/N to all and clip
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
    # after loop, maybe still off by small epsilon; adjust first element.
    current = h.sum()
    if abs(current - target_sum) > 1e-9:
        h[0] = np.clip(h[0] + (target_sum - current), 0, 1)
    return h

import time
import random

import numpy as np


def _cmax(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_sum_bounds(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto the box [0,1]^n intersect {sum = target_sum}."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(15):
        cur = x.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = int(free.sum())
        if n_free == 0:                # all at a bound → distribute uniformly
            x += diff / x.size
            x = np.clip(x, 0.0, 1.0)
            break
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)

    # tiny correction for rounding errors
    cur = x.sum()
    if abs(cur - target_sum) > 1e-9:
        diff = target_sum - cur
        for i in range(x.size):
            if 0.0 <= x[i] + diff <= 1.0:
                x[i] += diff
                break
    return x


def _grad_smooth_max(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth max (log‑sum‑exp) of the overlap integral."""
    n = h.size
    dx = 2.0 / n
    g = 1.0 - h

    # smooth max of C_k
    C = np.correlate(h, g, mode="full") * dx          # length 2n‑1
    mx = C.max()
    w = np.exp(beta * (C - mx))
    w /= w.sum()                                     # softmax weights

    grad = np.zeros_like(h)
    # contribution of each lag
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        s = idx - (n - 1)            # actual shift
        # part from g[i+s]
        i0 = max(0, -s)
        i1 = min(n, n - s)
        if i0 < i1:
            grad[i0:i1] += wk * g[i0 + s:i1 + s]
        # part from -h[i‑s]
        i0b = max(0, s)
        i1b = min(n, n + s)
        if i0b < i1b:
            grad[i0b:i1b] -= wk * h[i0b - s:i1b - s]

    grad *= dx
    return grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C₅ bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : list of floats  – the best step function found.
        c5_bound : float           – its maximal overlap integral.
        n_points : int             – number of discretisation points used.
    """
    np.random.seed(seed)
    random.seed(seed)
    t0 = time.time()

    # ----------------------------------------------------------------------
    # 1. Initial construction (interpolate the supplied parent if any)
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: linear ramp from 0 to 1 (satisfies the constraints)
        h0 = np.linspace(0.0, 1.0, 90, dtype=float)

    # target resolution – a modest even number (speed vs quality)
    n_target = max(120, len(h0) * 2)
    if n_target % 2:
        n_target += 1

    # linear interpolation onto the finer grid
    x_old = np.linspace(0.0, 2.0, len(h0), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h = np.interp(x_new, x_old, h0)

    # enforce the linear constraints exactly
    target_sum = n_target / 2.0
    h = _project_sum_bounds(h, target_sum)

    # ----------------------------------------------------------------------
    # 2. Optimisation – smooth‑max gradient descent with projection
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_val = _cmax(best_h)
    cur_val = best_val

    beta = 2.0               # start with a gentle smoothing
    beta_max = 30.0
    beta_inc = (beta_max - beta) / 4000.0

    lr = 0.7                 # learning rate
    lr_decay = 0.9996

    iteration = 0
    while time.time() - t0 < budget_s * 0.95:
        iteration += 1

        # gradient of the smooth max
        g = _grad_smooth_max(h, beta)

        # enforce the sum constraint via a Lagrange multiplier (zero‑mean step)
        lam = g.mean()
        h_new = h - lr * (g - lam)

        # project back onto the feasible set
        h_new = _project_sum_bounds(h_new, target_sum)

        val = _cmax(h_new)

        # keep the best ever seen
        if val < best_val:
            best_val = val
            best_h = h_new.copy()

        # accept if we made progress, otherwise occasionally accept a worse move
        if val <= cur_val or random.random() < 0.001:
            h = h_new
            cur_val = val

        # schedule parameters
        beta = min(beta + beta_inc, beta_max)
        lr *= lr_decay

        # occasional random jitter to escape shallow local minima
        if iteration % 500 == 0:
            jitter = (np.random.rand(n_target) - 0.5) * 0.02
            h = _project_sum_bounds(h + jitter, target_sum)

    # ----------------------------------------------------------------------
    # 3. Return the best candidate found
    # ----------------------------------------------------------------------
    final_val = _cmax(best_h)
    return best_h.tolist(), float(final_val), n_target
