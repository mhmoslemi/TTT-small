import numpy as np, time, math

def compute_c5_vec(h):
    """Return max_k ∫ h(x) (1-h(x+k)) dx approximated by discrete correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * (2.0 / h.size)

def binary_from_initial(initial, factor=2):
    """Convert initial array to binary (0/1) and optionally upscale."""
    base = np.array(initial, dtype=float)
    # ensure binary
    bits = (base >= 0.5).astype(int)
    if factor != 1:
        bits = np.repeat(bits, factor)
    n = bits.size
    target = n // 2
    cur = bits.sum()
    # adjust if needed
    if cur > target:
        # turn some 1s to 0
        idx = np.where(bits == 1)[0]
        np.random.shuffle(idx)
        bits[idx[:cur - target]] = 0
    elif cur < target:
        idx = np.where(bits == 0)[0]
        np.random.shuffle(idx)
        bits[idx[:target - cur]] = 1
    return bits.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    total_time = budget_s * 0.98
    start = time.time()
    # initialise binary vector
    if 'initial_h_values' in globals():
        h0 = binary_from_initial(initial_h_values, factor=2)
    else:
        n = 140
        h0 = np.zeros(n, dtype=float)
        half = n // 2
        idx = np.random.choice(n, half, replace=False)
        h0[idx] = 1.0
    n = h0.size
    # make sure feasibility (binary already respects sum)
    # current solution
    curr_h = h0.copy()
    curr_val = compute_c5_vec(curr_h)
    best_h = curr_h.copy()
    best_val = curr_val
    # annealing schedule
    T0 = 0.3
    Tf = 1e-4
    # main loop
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= total_time:
            break
        # temperature
        progress = elapsed / total_time
        T = T0 * (Tf / T0) ** progress
        # choose a random 1 and a random 0 and swap
        ones = np.where(curr_h == 1)[0]
        zeros = np.where(curr_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        # generate candidate by swapping
        cand_h = curr_h.copy()
        cand_h[i] = 0.0
        cand_h[j] = 1.0
        cand_val = compute_c5_vec(cand_h)
        # accept?
        if cand_val < curr_val or np.random.rand() < math.exp(-(cand_val - curr_val) / max(T, 1e-12)):
            curr_h = cand_h
            curr_val = cand_val
        # update global best
        if cand_val < best_val:
            best_h = cand_h.copy()
            best_val = cand_val
        # occasional large perturbation
        if np.random.rand() < 0.001:
            # perform a batch of 3 random swaps on current solution
            batch = curr_h.copy()
            for _ in range(3):
                ones_b = np.where(batch == 1)[0]
                zeros_b = np.where(batch == 0)[0]
                if len(ones_b) == 0 or len(zeros_b) == 0:
                    break
                i_b = ones_b[np.random.randint(len(ones_b))]
                j_b = zeros_b[np.random.randint(len(zeros_b))]
                batch[i_b] = 0.0
                batch[j_b] = 1.0
            batch_val = compute_c5_vec(batch)
            # accept batch if better than current (no SA)
            if batch_val < curr_val:
                curr_h = batch
                curr_val = batch_val
                if batch_val < best_val:
                    best_h = batch.copy()
                    best_val = batch_val
    # final evaluation (ensuring exact sum)
    # make sure exactly half ones
    half = n // 2
    cur_ones = int(best_h.sum())
    if cur_ones != half:
        # adjust by flipping random bits
        diff = half - cur_ones
        if diff > 0:
            zeros = np.where(best_h == 0)[0]
            choose = np.random.choice(zeros, diff, replace=False)
            best_h[choose] = 1.0
        else:
            ones = np.where(best_h == 1)[0]
            choose = np.random.choice(ones, -diff, replace=False)
            best_h[choose] = 0.0
    final_val = compute_c5_vec(best_h)
    return best_h.tolist(), final_val, n

import numpy as np
import time
import math


def _c5_value(h: np.ndarray) -> float:
    """Return C5 for a step function given by the vector h."""
    # full correlation of h with (1‑h) gives the overlap for each integer shift
    corr = np.correlate(h, 1.0 - h, mode='full')
    # dx = 2 / n  (interval length 2, n points)
    return corr.max() * (2.0 / h.size)


def _binary_from_initial(initial, factor: int = 2) -> np.ndarray:
    """
    Produce a binary (0/1) vector from the supplied initial data.
    - values ≥ 0.5 become 1, otherwise 0
    - optionally repeat each entry `factor` times to obtain a finer grid
    - finally adjust the number of ones to exactly n/2.
    """
    base = np.array(initial, dtype=float)
    bits = (base >= 0.5).astype(int)
    if factor != 1:
        bits = np.repeat(bits, factor)
    n = bits.size
    target = n // 2
    cur = bits.sum()
    if cur > target:  # too many ones → turn some to 0
        idx = np.where(bits == 1)[0]
        np.random.shuffle(idx)
        bits[idx[: cur - target]] = 0
    elif cur < target:  # too few ones → turn some to 1
        idx = np.where(bits == 0)[0]
        np.random.shuffle(idx)
        bits[idx[: target - cur]] = 1
    return bits.astype(float)
