def random_feasible(n, target_sum, rng):
    h = np.zeros(n)
    remaining = target_sum
    indices = np.arange(n)
    rng.shuffle(indices)
    for idx in indices:
        max_val = min(1.0, remaining)
        val = rng.random() * max_val
        h[idx] = val
        remaining -= val
        if remaining <= 1e-12:
            break
    # If leftover >0 after loop, assign to first index with room.
    if remaining > 1e-12:
        for idx in indices:
            cap = 1 - h[idx]
            add = min(cap, remaining)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    return h

import time
import math
import numpy as np

def _compute_c5(h, dx):
    """Maximum overlap integral for a given step function h."""
    # linear correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _random_feasible(n, target_sum, rng):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    remain = target_sum
    h = np.zeros(n, dtype=float)
    # random order of indices
    idx = np.arange(n)
    rng.shuffle(idx)
    for i in idx:
        if remain <= 0:
            break
        # we can place at most 1 in this cell
        cap = min(1.0, remain)
        # draw a random fraction of the remaining capacity
        val = rng.random() * cap
        h[i] = val
        remain -= val
    # any leftover (tiny) is dumped into the first non‑full cell
    if remain > 1e-12:
        for i in idx:
            free = 1.0 - h[i]
            if free > 1e-12:
                add = min(free, remain)
                h[i] += add
                remain -= add
                if remain <= 1e-12:
                    break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with integral 1 minimizing C5."""
    rng = np.random.default_rng(seed)

    # length of the discretisation
    if "initial_h_values" in globals():
        n = int(len(initial_h_values))
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 80
        h0 = np.full(n, 0.5, dtype=float)

    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility of the start vector
    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple proportional scaling then clip‑and‑redistribute
        h0 *= target_sum / cur_sum
        h0 = np.clip(h0, 0.0, 1.0)
        # fix any residual mass
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                free = 1.0 - h0[i] if diff > 0 else h0[i]
                step = min(abs(diff), free)
                if step > 0:
                    h0[i] += step if diff > 0 else -step
                    diff += -step if diff > 0 else step
                if abs(diff) < 1e-12:
                    break

    cur_h = h0.copy()
    cur_score = _compute_c5(cur_h, dx)

    best_h = cur_h.copy()
    best_score = cur_score

    start_time = time.time()
    # schedule parameters
    max_step_start = 0.4   # biggest mass we move in a single swap
    max_step_end   = 0.001 # finetuning step size
    temp_start = 0.015    # initial temperature for Metropolis acceptance
    restart_prob = 5e-4   # occasional random restart chance

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s * 0.97:
            break

        # linear schedules
        prog = elapsed / budget_s
        max_step = max_step_start * (1.0 - prog) + max_step_end * prog
        temperature = temp_start * (1.0 - prog)

        # occasional fresh start to diversify search
        if rng.random() < restart_prob:
            cur_h = _random_feasible(n, target_sum, rng)
            cur_score = _compute_c5(cur_h, dx)

        # propose a pairwise transfer that respects the bounds and the sum
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        # direction of transfer: i -> j if True, else j -> i
        forward = rng.random() < 0.5
        if forward:
            max_delta = min(max_step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            max_delta = min(max_step, cur_h[j], 1.0 - cur_h[i])
            if max_delta <= 0.0:

            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        # evaluate new candidate
        new_score = _compute_c5(new_h, dx)
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or rng.random() < math.exp(-delta_score / (temperature + 1e-12)):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                # early termination if we already beat the target bound
                if best_score <= 0.3808:

    # final clean‑up: enforce constraints with high precision
    best_h = np.clip(best_h, 0.0, 1.0)

    if abs(diff) > 1e-12:

        free_dn = np.where(best_h > 1e-12)[0]
        if diff > 0 and free_up.size:
