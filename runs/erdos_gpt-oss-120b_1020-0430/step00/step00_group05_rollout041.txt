import time, numpy as np

def _max_overlap(h):
    """Return C5 for the given step function h using the problem's evaluation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with zero padding, as described in the statement
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

def _project(h, target):
    """Clip to [0,1] and adjust the sum to the exact target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        # scale then fix any tiny violation
        h *= target / cur
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # add the remainder to a random feasible entry
            cand = np.where((h > 0) & (h < 1.0))[0]
            if cand.size == 0:
                cand = np.arange(h.size)
            i = np.random.choice(cand)
            h[i] = min(1.0, max(0.0, h[i] + diff))
    return h

    best_h = h.copy()

        # current worst shift

        idx = int(np.argmax(conv))
        s = idx - (n - 1)                 # shift corresponding to the max

        if s > 0 and np.random.rand() < 0.7:
