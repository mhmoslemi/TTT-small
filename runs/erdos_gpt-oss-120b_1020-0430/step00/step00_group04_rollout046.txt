def project_onto_cap_simplex(v, S):
    """
    Project v onto the intersection of box [0,1]^n and hyperplane sum = S.
    Solve min ||x - v||^2 s.t. 0 ≤ x_i ≤ 1, sum x_i = S.
    Using bisection on lambda for x = clip(v - lambda, 0, 1).
    """
    n = len(v)
    lo = -np.max(v) - 2  # ensuring all x_i = 1

        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0, 1)
    # Adjust due to numeric errors: after clip sum may deviate a tiny bit; fix by adding small correction evenly across nonbound entries
    # But minor; we can accept.
    return x

def project_onto_cap_simplex(v, S):
    n = v.shape[0]
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0, 1)
    # Minor correction for exact sum: compute residual
    res = S - x.sum()
    if abs(res) > 1e-10:
        # distribute to free variables (not at bounds)
        free = (x > 1e-12) & (x < 1-1e-12)
        if free.any():
            x[free] += res / free.sum()
            # Clip again just in case
            x = np.clip(x, 0, 1)
    return x

def gradient_for_shift(h, shift, dx):
    N = h.shape[0]
    i = np.arange(N)
    j = i + shift
    mask = (j >= 0) & (j < N)
    i = i[mask]
    j = j[mask]
    grad = np.zeros_like(h)
    grad[i] += 1 - h[j]
    grad[j] += -h[i]
    return grad * dx

import numpy as np, time

def _project_to_capped_simplex(v, target):
    """Projection onto [0,1]^n and sum(v)=target."""
    # binary search on lambda
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # small linear correction for exact sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _worst_shift_and_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    return shift, corr[idx]

def _gradient_for_shift(h, shift):
    n = h.shape[0]
    dx = 2.0 / n
    i = np.arange(n)
    j = i + shift
    mask = (j >= 0) & (j < n)
    i = i[mask]
    j = j[mask]
    grad = np.zeros_like(h)
    grad[i] += 1.0 - h[j]
    grad[j] -= h[i]
    return grad * dx

    # Determine dimension and initial point

        h_init = np.array(initial_h_values, dtype=float)
        n = h_init.shape[0]
    else:
        n = 42
        h_init = None

    target_sum = n / 2.0

    # Helper to evaluate a candidate
    eval_func = globals().get('evaluate_erdos_solution', None)
    def _eval(h):
        if eval_func is not None:
            return eval_func(h)
        return _compute_c5(h)

    # Prepare the first feasible solution
    if h_init is not None:
        h0 = _project_to_capped_simplex(h_init.copy(), target_sum)
        best_h = h0.copy()
        best_val = _eval(best_h)
    else:
        best_h = _project_to_capped_simplex(rng.random(n), target_sum)
        best_val = _eval(best_h)

    # Optimization parameters
    max_restarts = 25
    max_iters = 3000

    # Main search
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s - 5:
            break

        if restart == 0 and h_init is not None:
            h = best_h.copy()
        else:
            # Random feasible start; occasionally perturb best solution
            if restart > 5 and best_h is not None:
                h = best_h + rng.normal(scale=0.05, size=n)
            else:
                h = rng.random(n)
            h = _project_to_capped_simplex(h, target_sum)

        for it in range(max_iters):

                break

                h = h_swap

                h = h_swap

    # Final projection and evaluation
    best_h = _project_to_capped_simplex(best_h, target_sum)
    final_val = _eval(best_h)
