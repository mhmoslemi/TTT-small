import time, math, random
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    # --------------------------------------------------------------
    # Setup
    # --------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # starting vector (may be provided by the harness)
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        m = len(init_vec)
    except Exception:
        init_vec = None
        m = 0

    # choose number of discretisation points (bigger = more freedom)
    n = max(200, m)            # at least 200 points for finer search
    dx = 2.0 / n
    target_sum = n / 2.0        # because ∫h = 1  ⇒  Σh = n/2

    # --------------------------------------------------------------
    # Helper: projection onto box [0,1] with exact sum = target_sum
    # --------------------------------------------------------------
    def proj_box_sum(v):
        """Project v onto {0≤x_i≤1, Σx_i = target_sum}."""
        low, high = -2.0, 2.0 + np.max(v)      # wide enough interval
        for _ in range(60):                   # bisection
            mid = (low + high) * 0.5
            w = np.clip(v - mid, 0.0, 1.0)
            s = w.sum()
            if s > target_sum:
                low = mid
            else:
                high = mid
        return np.clip(v - (low + high) * 0.5, 0.0, 1.0)

    # --------------------------------------------------------------
    # Build initial feasible point
    # --------------------------------------------------------------
    if init_vec is None:
        # pure random start
        cur = np.random.rand(n)
        cur = proj_box_sum(cur)
    else:
        # up‑sample the provided vector to the chosen resolution
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        cur = np.interp(x_new, x_old, init_vec)
        cur = proj_box_sum(cur)

    # --------------------------------------------------------------
    # Evaluation of the objective C5
    # --------------------------------------------------------------
    def evaluate(h):
        """Return max_k ∫ h(x)*(1-h(x+k)) dx (discrete)."""
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    cur_val = evaluate(cur)
    best_h = cur.copy()
    best_val = cur_val

    # --------------------------------------------------------------
    # Gradient of the active shift (the one attaining the max)
    # --------------------------------------------------------------
    def max_shift_and_value(h):
        """Return (value, shift) of the correlation entry attaining the max."""
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        val = corr[idx] * dx
        shift = idx - (n - 1)          # zero shift corresponds to idx = n-1
        return val, shift

    def grad_for_shift(h, shift):
        """Gradient of the correlation value for a given integer shift."""
        g = np.zeros_like(h)
        if shift >= 0:
            k = shift
            if k < n:
                g[:n - k] += (1.0 - h[k:])
                g[k:]    -= h[:n - k]
        else:
            k = -shift
            if k < n:
                g[k:]    += (1.0 - h[:n - k])
                g[:n - k] -= h[k:]
        return g * dx

    # --------------------------------------------------------------
    # Simulated annealing / projected gradient descent
    # --------------------------------------------------------------
    # Scheduling parameters
    T0 = 0.02               # initial temperature
    lr0 = 0.12               # initial learning rate for gradient steps
    decay_T = 0.99994
    decay_lr = 0.99995
    T = T0
    lr = lr0

    it = 0
    # Main optimisation loop (time‑driven)
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.96:
            break

        # Decide move type: gradient step (60%) or random swap (40%)
        if random.random() < 0.6:
            # ----- projected gradient step towards decreasing the active shift -----
            val, shift = max_shift_and_value(cur)
            g = grad_for_shift(cur, shift)
            cand = proj_box_sum(cur - lr * g)   # step opposite to gradient
        else:
            # ----- random mass‑transfer between two coordinates (preserves sum) -----
            i = random.randrange(n)
            j = random.randrange(n - 1)
            if j >= i:
                j += 1
            if random.random() < 0.5:
                # move from j to i
                max_delta = min(1.0 - cur[i], cur[j])
                if max_delta <= 0.0:
                    continue
                delta = max_delta * random.random()
                cand = cur.copy()
                cand[i] += delta
                cand[j] -= delta
            else:
                # move from i to j
                max_delta = min(1.0 - cur[j], cur[i])
                if max_delta <= 0.0:
                    continue
                delta = max_delta * random.random()
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta

        cand_val = evaluate(cand)
        delta = cand_val - cur_val

        # Metropolis acceptance
        if delta <= 0.0 or random.random() < math.exp(-delta / max(T, 1e-12)):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur.copy()

        # Parameter decay
        it += 1
        lr = max(lr * decay_lr, 5e-4)
        T = max(T * decay_T, 1e-6)

        # Occasionally re‑seed the current point with the best found so far
        if it % 50000 == 0:
            cur = best_h.copy()
            cur_val = best_val

    # --------------------------------------------------------------
    # Final fine‑tuning (pure projected gradient descent)
    # --------------------------------------------------------------
    fin_iters = 20000
    for _ in range(fin_iters):
        if time.time() - start_time > budget_s * 0.99:
            break
        _, shift = max_shift_and_value(best_h)
        g = grad_for_shift(best_h, shift)
        cand = proj_box_sum(best_h - 0.001 * g)
        cand_val = evaluate(cand)
        if cand_val < best_val:
            best_val = cand_val
            best_h = cand

    # Ensure feasibility one last time
    best_h = proj_box_sum(best_h)
    best_val = evaluate(best_h)

    return best_h.tolist(), float(best_val), n
