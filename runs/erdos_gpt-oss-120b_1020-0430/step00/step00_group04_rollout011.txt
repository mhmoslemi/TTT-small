def project_sum_bounds(h, target_sum):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Compute sum
    cur_sum = np.sum(h)
    if abs(cur_sum - target_sum) < 1e-12:
        return h
    # We'll use a simple iterative algorithm by adjusting free entries
    # free = indices where 0 < h < 1
    # The algorithm: while sum mismatch, compute delta = target_sum - cur_sum.
    # Distribute delta equally among free entries, then clip again, repeat.
    # This converges quickly.

            break
        delta_per = (target_sum - cur_sum) / len(free)

def upscale_h(h, new_n):
    n = len(h)
    # positions on [0,2)
    x_old = np.arange(n) / n * 2.0
    x_new = np.arange(new_n) / new_n * 2.0
    # periodic extension: add one extra point at x=2 equal to first point
    h_ext = np.concatenate([h, [h[0]]])
    x_ext = np.concatenate([x_old, [2.0]])
    new_h = np.interp(x_new, x_ext, h_ext)
    # enforce constraints
    return project_sum_bounds(new_h, new_n/2.0)

Alternative algorithm: Use Lagrange multiplier approach: Solve projection onto simplex with upper bound.

Specifically, it's like we need to find h_i in [0,1] such that sum h_i = target.

Algorithm:

1. Clip to [0,1].

2. Let residual = target_sum - sum(clipped). If residual > 0, we need to increase some entries that are not at 1.

3. Let free = indices where clipped < 1 - eps. Distribute residual equally across free. Then clip again; repeat.

Because each iteration reduces the set of saturated indices.

Implementation:

```
def project_sum_bounds(h, target):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if abs(total - target) <= 1e-10:
        return h
    # Use iterative approach
    for _ in range(100):
        diff = target - total
        if abs(diff) <= 1e-12:
            break
        # Determine which variables can be increased or decreased
        if diff > 0:
            # need to increase sum; find indices where h < 1
            idx = np.where(h < 1 - 1e-12)[0]
        else:
            # need to decrease sum; find indices where h > 0
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            # cannot adjust further, break
            break
        delta = diff / len(idx)
        # apply delta limited by bounds
        if diff > 0:
            # increase but not beyond 1
            inc = np.minimum(delta, 1 - h[idx])
            h[idx] += inc
        else:
            # decrease but not below 0
            dec = np.maximum(delta, -h[idx])
            h[idx] += dec
        total = np.sum(h)
    # final clipping (just in case)
    return np.clip(h, 0.0, 1.0)
```

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)


def project_sum_bounds(vec, target):
    """Project vec onto [0,1] box with sum = target (feasible region)."""
    h = np.clip(vec, 0.0, 1.0)
    total = np.sum(h)
    for _ in range(100):
        diff = target - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        new_vals = h[free] + delta
        new_vals = np.clip(new_vals, 0.0, 1.0)
        h[free] = new_vals
        total = np.sum(h)
    return np.clip(h, 0.0, 1.0)


def random_swap_move(h):
    """Swap a small epsilon between two random indices while preserving sum."""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    if np.random.rand() < 0.5:
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            return h.copy()
        eps = np.random.rand() * max_eps
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
    else:
        max_eps = min(h[j], 1 - h[i])
        if max_eps <= 0:
            return h.copy()
        eps = np.random.rand() * max_eps
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
    return new_h


def upscale_h(h, new_n):
    """Linearly interpolate h to a finer grid (periodic) and enforce constraints."""
    n = len(h)
    x_old = np.arange(n) / n * 2.0
    x_new = np.arange(new_n) / new_n * 2.0
    # periodic extension by appending the first point at x=2
    h_ext = np.concatenate([h, [h[0]]])
    x_ext = np.concatenate([x_old, [2.0]])
    h_new = np.interp(x_new, x_ext, h_ext)
    return project_sum_bounds(h_new, new_n / 2.0)

    n = len(start_h)

                    new_h[i + k] += eps

    best_score = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. set up a hierarchy of grid sizes (do not exceed 168 points)
    grid_sizes = [len(best_h)]
    while grid_sizes[-1] * 2 <= 168:
        grid_sizes.append(grid_sizes[-1] * 2)

    cur_h = best_h.copy()

    for idx, n_target in enumerate(grid_sizes):
        # upscale if we moved to a finer grid
        if len(cur_h) != n_target:
            cur_h = upscale_h(cur_h, n_target)

        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= 0:
            break

        # allocate a fair share of the remaining budget to this level
        remaining_levels = len(grid_sizes) - idx
        time_for_this = remaining / remaining_levels
        # guarantee at least a minimal slice
        if time_for_this < 0.2:
            time_for_this = remaining

        h_opt, score_opt = hill_climb(cur_h, time_for_this)

        if score_opt < best_score:
            best_h = h_opt.copy()
            best_score = score_opt

        cur_h = h_opt.copy()

    # ------------------------------------------------------------------
    # 3. final tidy‑up (ensure constraints exactly)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_bounds(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score

    return best_h, best_score, len(best_h)
