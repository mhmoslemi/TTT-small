import time
import numpy as np
from scipy.optimize import minimize

# -----------------------------------------------------------
# Helper functions (all top‑level as required)
# -----------------------------------------------------------

def _circular_corr(h):
    """Circular correlation c[s] = Σ_i h[i]·(1-h[i+s]) (no dx)."""
    H = np.fft.fft(h)
    G = np.fft.fft(1.0 - h)
    c = np.fft.ifft(H * np.conj(G)).real
    return c


def _smooth_obj_grad(h, alpha):
    """Smooth max of the correlation vector and its gradient."""
    n = h.shape[0]
    c = _circular_corr(h)                     # raw correlation (no dx)

    # smooth max via log‑sum‑exp, numerically stable
    a_c = alpha * c
    m = np.max(a_c)
    exp_shift = np.exp(a_c - m)
    sum_exp = np.exp(m) * np.sum(exp_shift)
    smooth_val = (m + np.log(np.sum(exp_shift))) / alpha   # = log(sum_exp)/alpha

    # softmax weights
    w = exp_shift / np.sum(exp_shift)

    # convolution of w with h and w reversed with h
    W = np.fft.fft(w)
    H = np.fft.fft(h)
    conv = np.fft.ifft(W * H).real               # Σ_s w[s]·h[i‑s]
    conv_rev = np.fft.ifft(np.conj(W) * H).real  # Σ_s w[s]·h[i+s]

    grad_smooth = np.ones(n) - conv - conv_rev   # ∂/∂h_i (smooth max)

    dx = 2.0 / n
    return smooth_val * dx, grad_smooth * dx      # objective, gradient


def _project_to_bounds_and_sum(v, target):
    """
    Project vector v onto the box [0,1]^n while forcing Σ x_i = target.
    The solution has the form x = clip(v + λ,0,1) for some λ.
    """
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + lo, 0.0, 1.0)
    # tiny adjustment to hit the target exactly (clipping may leave a few 1e‑12 error)
    diff = np.sum(x) - target
    if abs(diff) > 1e-12:
        # distribute the tiny remainder uniformly over the non‑boundary entries
        free = (x > 0.0) & (x < 1.0)
        if np.any(free):
            x[free] -= diff / np.count_nonzero(free)
            x = np.clip(x, 0.0, 1.0)
    return x


def _random_start(n, target):
    """Generate a random feasible start (bounds [0,1] and Σh = target)."""
    v = np.random.rand(n)
    return _project_to_bounds_and_sum(v, target)


def _optimize_chain(x0, alphas, target, max_iter=120):
    """
    Perform successive SLSQP optimisations for increasing smoothness.
    Each stage uses the result of the previous as start.
    """
    x = np.copy(x0)
    n = x.shape[0]
    bounds = [(0.0, 1.0)] * n
    eq_cons = {
        'type': 'eq',
        'fun': lambda h: np.sum(h) - target,
        'jac': lambda h: np.ones_like(h)
    }

    for alpha in alphas:
        # frozen closures for the current α
        def fun(h, a=alpha):
            return _smooth_obj_grad(h, a)[0]

        def jac(h, a=alpha):
            return _smooth_obj_grad(h, a)[1]

        res = minimize(
            fun,
            x,
            jac=jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[eq_cons],
            options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
        )
        if not res.success:
            # still keep the best we have
            x = res.x
        else:
            x = res.x
    return x
