import time, math, random
import numpy as np

def _max_corr_fft(h, nfft):
    """Return the maximum of correlation h * (1-h) using FFT (linear convolution)."""
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)
    # keep only the linear part
    L = 2 * len(h) - 1
    return np.max(corr[:L])

def run(seed=42, budget_s=1100, **kwargs):
    """Return a binary step function h on [0,2] that (approximately) minimises
    C5 = max_k ∫ h(x)(1-h(x+k))dx.
    The result is a tuple (h_values, c5_bound, n_points)."""
    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # initialise a binary vector with exactly N/2 ones
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        N = len(init)
        # force binary values
        h = (init > 0.5).astype(float)
        # enforce exact half‑filled constraint
        ones = int(h.sum())
        target = N // 2
        while ones > target:
            idx = random.choice(np.flatnonzero(h == 1))
            h[idx] = 0.0
            ones -= 1
        while ones < target:
            idx = random.choice(np.flatnonzero(h == 0))
            h[idx] = 1.0
            ones += 1
    else:
        N = 200                # reasonable size, < 1000
        h = np.zeros(N, dtype=float)
        h[np.random.choice(N, N // 2, replace=False)] = 1.0

    # preparation for the optimiser
    ones_list = list(np.flatnonzero(h == 1))
    zeros_list = list(np.flatnonzero(h == 0))
    L_full = 2 * N - 1
    nfft = 1 << ((L_full - 1).bit_length())   # next power of two ≥ L_full

    cur_obj = _max_corr_fft(h, nfft)          # integer count of max overlap
    best_obj = cur_obj
    best_h = h.copy()

    start_time = time.time()
    max_time = budget_s * 0.95                  # safety margin
    # temperature schedule (simulated‑annealing style)
    T_start, T_end = 1.0, 1e-4

    # ----------------------------------------------------------------------
    # main optimisation loop
    # ----------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break
        # linear cooling
        t_frac = elapsed / max_time
        T = T_start * (1 - t_frac) + T_end * t_frac

        # current worst lag (the one that attains cur_obj)
        corr_full = np.fft.irfft(
            np.fft.rfft(h, n=nfft) *
            np.conj(np.fft.rfft(1.0 - h, n=nfft)),
            n=nfft)[:L_full]
        idx_max = int(np.argmax(corr_full))
        lag = idx_max - (N - 1)                 # lag ∈ [-(N-1), N-1]

        # collect positions of ones that actually contribute to this worst value
        bad_ones = []
        if lag >= 0:
            for i in ones_list:
                j = i + lag
                if j >= N or h[j] == 0.0:
                    bad_ones.append(i)
        else:  # lag < 0
            for i in ones_list:
                j = i + lag
                if j < 0 or h[j] == 0.0:
                    bad_ones.append(i)

        # choose a swap candidate – try to fix the worst lag
        if bad_ones and zeros_list:
            i = random.choice(bad_ones)
            j = random.choice(zeros_list)
        else:
            # fallback: random swap
            i = random.choice(ones_list)
            j = random.choice(zeros_list)

        # perform the swap
        h[i] = 0.0
        h[j] = 1.0
        new_obj = _max_corr_fft(h, nfft)
        delta = new_obj - cur_obj

        # acceptance test
        if delta < 0 or random.random() < math.exp(-delta / (T + 1e-12)):
            # accepted
            cur_obj = new_obj
            # update the index lists
            ones_list.remove(i)
            zeros_list.append(i)
            zeros_list.remove(j)
            ones_list.append(j)

            if new_obj < best_obj:
                best_obj = new_obj
                best_h = h.copy()
        else:
            # reject – revert the swap
            h[i] = 1.0
            h[j] = 0.0

    # ----------------------------------------------------------------------
    # final evaluation using the reference formula (np.correlate)
    # ----------------------------------------------------------------------
    dx = 2.0 / N
    final_corr = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
    c5_bound = float(final_corr.max())

    # return a plain python list for compatibility
    return (best_h.tolist(), c5_bound, N)
