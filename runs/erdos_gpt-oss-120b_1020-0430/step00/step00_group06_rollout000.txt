def project_to_feasible(x, target):
    # Clip and adjust using lambda solving.
    # Use binary search for lambda.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # due to rounding may not exactly meet sum; we can adjust residual by adding small amounts to non-saturated entries.
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        # distribute diff among entries not at bound
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            # uniform distribution
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """
    Compute the overlap integral C5 for a discrete step function `h`.
    """
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def project_to_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project `x` onto the box [0,1]^n intersected with the hyperplane sum = target_sum.
    Uses a simple binary search on the Lagrange multiplier λ for the equality constraint.
    """
    lo, hi = -2.0, 2.0          # bounds that surely contain the correct λ
    for _ in range(60):        # enough for double precision
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

    # Tiny residual fixing (rare)
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def propose_move(h: np.ndarray, scale: float) -> np.ndarray:
    """
    Generate a neighboring feasible vector by moving a small amount of mass
    from one index to another (preserving the sum).  `scale` controls the
    maximal fraction of the admissible transfer.
    """
    n = h.shape[0]
    for _ in range(10):
        i, j = np.random.choice(n, size=2, replace=False)
        # attempt i → j
        delta = min(h[i], 1.0 - h[j])
        if delta > 1e-14:
            eps = np.random.rand() * scale * delta
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            return nh
        # attempt j → i
        delta = min(h[j], 1.0 - h[i])
        if delta > 1e-14:
            eps = np.random.rand() * scale * delta
            nh = h.copy()
            nh[j] -= eps
            nh[i] += eps
            return nh
    # fallback: small global perturbation followed by projection
    target = h.shape[0] / 2.0
    pert = h + np.random.normal(scale=scale * 0.1, size=n)
    return project_to_feasible(pert, target)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to obtain a low upper bound for C5.
    Returns a tuple (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)   # provided by the harness
        n = init.size
    except NameError:
        n = 95
        init = np.random.rand(n)

    target_sum = n / 2.0
    cur_h = project_to_feasible(init, target_sum)
    cur_c5 = compute_c5(cur_h)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Simulated‑annealing style search
    # ------------------------------------------------------------------
    step_scale = 0.5                 # typical size of a transfer
    max_temp = 0.001                 # initial temperature for uphill moves
    no_improve = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.97:          # leave a small margin
            break

        # linear temperature decay
        frac = elapsed / (budget_s * 0.97)
        temperature = max_temp * (1.0 - frac)

        # propose a neighbour
        cand_h = propose_move(cur_h, step_scale)
        cand_c5 = compute_c5(cand_h)
        delta = cand_c5 - cur_c5

        # Metropolis acceptance
        accept = False
        if delta <= 0:
            accept = True
        elif temperature > 0:
            accept = (np.random.rand() < math.exp(-delta / temperature))

        if accept:
            cur_h = cand_h
            cur_c5 = cand_c5
            no_improve = 0
            if cand_c5 < best_c5:
                best_h = cand_h.copy()
                best_c5 = cand_c5
        else:
            no_improve += 1

        # slowly shrink the move size to allow fine tuning
        if int(elapsed) % 30 == 0:               # roughly every half‑minute
            step_scale *= 0.96
            if step_scale < 0.02:
                step_scale = 0.02

        # random restart if stuck too long
        if no_improve > 200_000:
            cur_h = project_to_feasible(np.random.rand(n), target_sum)
            cur_c5 = compute_c5(cur_h)
            no_improve = 0
            step_scale = 0.5

    # ------------------------------------------------------------------
    # Final projection & evaluation
    # ------------------------------------------------------------------
    best_h = project_to_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), int(n)
