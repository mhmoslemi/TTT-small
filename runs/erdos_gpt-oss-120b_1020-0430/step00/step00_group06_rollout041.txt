import time
import numpy as np

# ----- helper functions (top level) -----------------------------------------

def _project_to_constraints(y: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector y onto the set {h : 0 <= h_i <= 1, sum(h) = target}.
    The projection is the Euclidean projection and can be found by a
    simple binary search on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0          # wide enough bounds for the multiplier
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(y + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def _overlap_max(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx for the discrete representation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)


def _gradient_worst_shift(h: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the worst (largest) overlap with respect to h.
    The worst shift is the lag that attains the maximum in _overlap_max.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)                     # signed shift

    grad = np.zeros_like(h)
    if k >= 0:
        # region where i and i+k both inside the array
        grad[: n - k] += 1.0 - h[k:]          # d/ dh_i of h_i (1‑h_{i+k})
        grad[k:]      -= h[: n - k]           # d/ dh_{i+k} of −h_i h_{i+k}
    else:
        p = -k
        grad[p:]      += 1.0 - h[: n - p]
        grad[: n - p] -= h[p:]

    # the true gradient includes the factor dx
    return grad * dx


# ----- main routine ---------------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] (represented by n points) to lower
    the Erdős C5 bound.  Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)

    # ---- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: 95 points with uniform start
        n_fallback = 95
        h = np.full(n_fallback, 0.5, dtype=np.float64)

    n = h.shape[0]
    target_sum = n / 2.0          # because ∫ h = 1  =>  sum(h)*dx = 1  => sum(h)=n/2
    h = _project_to_constraints(h, target_sum)

    cur_val = _overlap_max(h)
    best_h = h.copy()
    best_val = cur_val

    # ---- simulated annealing -------------------------------------------------
    start_time = time.time()
    max_iters = 500_000
    it = 0
    temperature = 0.15
    decay = 0.99997

    while it < max_iters and time.time() - start_time < budget_s * 0.85:
        # propose a move that preserves the sum exactly by transferring mass
        i, j = rng.choice(n, size=2, replace=False)

        # feasible interval for epsilon (mass moved from i to j)
        low = max(-h[i], h[j] - 1.0)          # keep both in [0,1]
        high = min(1.0 - h[i], h[j])
        if low >= high:
            it += 1
            continue

        eps = low + (high - low) * rng.rand()
        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps

        new_val = _overlap_max(h_new)

        # keep the best ever seen
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

        # Metropolis acceptance
        if new_val < cur_val or rng.rand() < np.exp((cur_val - new_val) / temperature):
            h = h_new
            cur_val = new_val

        # temperature schedule
        temperature *= decay
        it += 1

        # occasional gradient‑descent polishing (every 5000 iterations)
        if it % 5000 == 0:
            for _ in range(8):
                grad = _gradient_worst_shift(h)
                # simple step; step size tuned empirically
                eta = 0.7
                h_candidate = _project_to_constraints(h - eta * grad, target_sum)
                cand_val = _overlap_max(h_candidate)
                if cand_val < cur_val:
                    h = h_candidate
                    cur_val = cand_val
                    if cand_val < best_val:
                        best_val = cand_val
                        best_h = h_candidate.copy()

        # early exit if we already beat the target
        if best_val < 0.3808:
            break

    # ---- final deterministic refinement ---------------------------------------
    # start from the best vector discovered
    h = best_h.copy()
    cur_val = best_val

    for _ in range(1200):
        grad = _gradient_worst_shift(h)
        eta = 0.5
        h_new = _project_to_constraints(h - eta * grad, target_sum)
        new_val = _overlap_max(h_new)
        if new_val < cur_val:
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
        else:
            # if no improvement, shrink step size a little
            eta *= 0.95

        # stop early if good enough
        if best_val < 0.3808:
            break

    # ---- final evaluation ----------------------------------------------------
    final_val = _overlap_max(best_h)

    return (best_h.tolist(), float(final_val), n)
