def project_onto_capped_simplex(v, S):
    # Clip initial
    # Use water-filling algorithm known as "projection onto simplex with upper bound 1"
    n = len(v)

    # Initialize low and high
    low = v.min() - 1.0
    high = v.max()
    for _ in range(50):
        theta = (low + high) * 0.5
        x = np.clip(v - theta, 0, 1)
        s = x.sum()
        if s > S:
            low = theta
        else:
            high = theta
    x = np.clip(v - (low + high) * 0.5, 0, 1)
    # due to numeric, adjust sum to S by adding small amount to some interior points
    diff = S - x.sum()
    if abs(diff) > 1e-8:

    # Not necessary if diff small.

def project(v, S):

    # Adjust sum
    while True:

- Because sum is decreasing in theta, we can binary search for theta where sum(x) = S.

import time
import numpy as np

def _project_capped_simplex(v, target, max_iter=60, tol=1e-12):
    """Euclidean projection of v onto the cube [0,1]^n with sum(v)=target."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(max_iter):
        theta = (lo + hi) * 0.5
        x = np.clip(v - theta, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = theta
        else:
            hi = theta
        if hi - lo < tol:
            break
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        interior = (x > 1e-12) & (x < 1 - 1e-12)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_corr_shift(h, s):
    """Gradient of the correlation term for a given shift s (np.correlate convention)."""
    n = len(h)
    grad = np.zeros_like(h)
    if s >= 0:
        i = np.arange(s, n, dtype=int)
        j = i - s
    else:
        t = -s
        i = np.arange(0, n - t, dtype=int)
        j = i + t
    grad[i] += 1.0
    grad[i] -= h[j]
    grad[j] -= h[i]
    return grad


def _c5_value(h, dx):
    """Return C5 bound for the current h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # problem size – small enough for fast evaluation but fine enough for quality
    n = 150
    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- helper to create a feasible vector ---------------------------------
    def random_feasible():
        v = rng.random(n)
        return _project_capped_simplex(v, target_sum)

    # ----- initial point (use provided construction if present) ---------------
    if 'initial_h_values' in globals():
        low = np.asarray(initial_h_values, dtype=float)
        n_low = low.size
        x_low = np.linspace(0.0, 2.0, n_low, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_low, low)
        h_current = _project_capped_simplex(h0, target_sum)
    else:
        h_current = random_feasible()

    best_h = h_current.copy()
    best_val = _c5_value(best_h, dx)

    # --------- main optimisation loop ----------------------------------------
    max_iters = 200000
    step0 = 0.8
    no_improve = 0
    for it in range(1, max_iters + 1):
        # time check
        if time.time() - start_time > budget_s * 0.96:
            break

        # compute correlation and identify the worst shift(s)
        corr = np.correlate(h_current, 1.0 - h_current, mode='full')
        max_corr = corr.max()
        cur_val = max_corr * dx

        # gradient is the average over all shifts attaining the maximum
        max_idxs = np.where(corr >= max_corr - 1e-12)[0]
        grad = np.zeros_like(h_current)
        for idx in max_idxs:
            s = int(idx - (n - 1))
            grad += _grad_corr_shift(h_current, s)
        grad /= max_idxs.size

        # gradient step with diminishing size
        step = step0 / np.sqrt(it)
        h_proposed = h_current - step * grad
        h_proposed = _project_capped_simplex(h_proposed, target_sum)

        # evaluate proposal
        corr_prop = np.correlate(h_proposed, 1.0 - h_proposed, mode='full')
        new_val = corr_prop.max() * dx

        if new_val < cur_val:
            h_current = h_proposed
            cur_val = new_val
            no_improve = 0
        else:
            no_improve += 1
            # occasional random perturbation to escape plateaus
            if no_improve % 30 == 0:
                noise = rng.normal(scale=0.05, size=n)
                h_pert = _project_capped_simplex(h_current + noise, target_sum)
                val_pert = _c5_value(h_pert, dx)
                if val_pert < cur_val:
                    h_current = h_pert
                    cur_val = val_pert
                    no_improve = 0

        # record best seen so far
        if cur_val < best_val:
            best_val = cur_val
            best_h = h_current.copy()
            no_improve = 0

        # long streak without improvement → fresh restart
        if no_improve >= 200:
            h_current = random_feasible()
            no_improve = 0

    # ----- final polishing: a few local random moves -------------------------
    for _ in range(500):
        if time.time() - start_time > budget_s * 0.98:

        # pick two indices and transfer a small amount while keeping feasibility
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(-0.02, 0.02)
        trial = best_h.copy()
        trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
        trial[j] = np.clip(trial[j] - delta, 0.0, 1.0)

        val = _c5_value(trial, dx)
        if val < best_val:

    # Ensure final solution respects constraints (within tiny numerical error)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_capped_simplex(best_h, target_sum)
