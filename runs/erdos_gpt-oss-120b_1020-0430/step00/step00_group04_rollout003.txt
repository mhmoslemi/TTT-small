import numpy as np, time, math, random
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * (2.0 / n)

def project_onto_feasible(v, target):
    v = np.clip(v, 0.0, 1.0)
    # adjust sum
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    # water-filling
    # We'll use simple iterative scaling.
    diff = target - s
    # Distribute diff uniformly among non-saturated entries
    mask = (v > 0) & (v < 1)
    # If all saturated, just scale to target (should not happen)
    if not np.any(mask):
        # just scale all equally (might break bounds)
        v = v + diff / len(v)
        v = np.clip(v, 0.0, 1.0)
        # adjust again
        return project_onto_feasible(v, target)
    # Distribute diff equally
    v[mask] += diff / np.sum(mask)
    # Clip again and recurse
    return project_onto_feasible(v, target)

def random_feasible(n, target, rng):
    # Dirichlet sample
    a = rng.random(n)
    a = a / a.sum() * target
    # fix if any exceed 1
    a = np.clip(a, 0.0, 1.0)
    # if sum deviates, project
    a = project_onto_feasible(a, target)
    return a

def greedy_refine(h0, target, time_limit, start_time):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    # parameters
    max_alpha = 0.5
    min_alpha = 0.01
    # track iterations without improvement
    no_improve = 0
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > time_limit - 0.05:
            break
        it += 1
        # schedule alpha linearly decreasing
        alpha = max_alpha - (max_alpha - min_alpha) * (elapsed / time_limit)
        # compute gradient for max lag
        n = len(h)
        c = np.correlate(h, 1.0-h, mode='full')
        idx = np.argmax(c)
        L = idx - (n-1)
        # compute gradient g
        g = np.zeros(n)
        if L >= 0:
            i = np.arange(n - L)
            g[i] += 1.0 - h[i + L]
            i2 = np.arange(L, n)
            g[i2] -= h[i2 - L]
        else:
            s = -L
            i = np.arange(s, n)
            g[i] += 1.0 - h[i - s]
            i2 = np.arange(0, n - s)
            g[i2] -= h[i2 + s]
        # choose i_pos, i_neg
        pos_candidates = np.where(g > 1e-12)[0]
        neg_candidates = np.where(g < -1e-12)[0]
        if len(pos_candidates) > 0 and len(neg_candidates) > 0:
            i_pos = pos_candidates[np.argmax(g[pos_candidates])]
            i_neg = neg_candidates[np.argmin(g[neg_candidates])]
            # direction: move mass from i_pos (positive gradient) to i_neg (negative)
            max_delta = min(alpha, h[i_pos], 1.0 - h[i_neg])
            delta = max_delta * 0.5
            if delta > 1e-12:
                h[i_pos] -= delta
                h[i_neg] += delta
                # ensure bounds
                # evaluate
                val = compute_c5(h)
                if val <= cur_val + 1e-12:
                    cur_val = val
                    if val < best_val - 1e-12:
                        best_val = val
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    # revert
                    h[i_pos] += delta
                    h[i_neg] -= delta
                    no_improve += 1
        else:
            # random small perturbation
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(alpha, h[i], 1.0 - h[j])
            delta = delta * 0.5
            if delta > 1e-12:
                h[i] -= delta
                h[j] += delta
                val = compute_c5(h)
                if val <= cur_val:
                    cur_val = val
                    if val < best_val:
                        best_val = val
                        best_h = h.copy()
                else:
                    # revert
                    h[i] += delta
                    h[j] -= delta
        # occasional random jump
        if it % 200 == 0 and no_improve > 5:
            # reinitialize small perturbation
            idx1, idx2 = np.random.choice(n, 2, replace=False)
            # random set values from uniform while keep sum
            # simply randomize two entries subject to sum conservation
            total = h[idx1] + h[idx2]
            a = random.random()
            new1 = max(0.0, min(1.0, a*total))
            new2 = total - new1
            h[idx1], h[idx2] = new1, new2
            cur_val = compute_c5(h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            no_improve = 0
    return best_h, best_val

    from scipy.optimize import minimize
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine n_points
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = len(init_arr)
    else:
        n = 42
        init_arr = None
    if n % 2 != 0:
        # ensure even
        n += 1
    target = n / 2.0
    # Helper to compute c5
    def compute_c5_local(h):
        c = np.correlate(h, 1.0 - h, mode='full')
        return np.max(c) * (2.0 / n)
    # Prepare initial solution(s)
    best_h = None
    best_val = float('inf')
    # if provided initial
    if init_arr is not None and len(init_arr) == n:
        h0 = np.clip(init_arr, 0.0, 1.0)
        # adjust sum if needed
        s = h0.sum()
        if abs(s - target) > 1e-9:

            h0 = h0 * (target / s)
            h0 = np.clip(h0, 0.0, 1.0)

            h0 = project_onto_feasible(h0, target)
        val0 = compute_c5_local(h0)
        best_h = h0.copy()
        best_val = val0
    # local search parameters
    # allocate time to SLSQP runs first
    time_spent = time.time() - start_time
    time_left = budget_s - time_spent
    # function for SLSQP run
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target}

    slsqp_runs = 0
    while time_left > 30 and slsqp_runs < 15:

        x0 = random_feasible(n, target, rng)
        # optional: mix with best found
        if best_h is not None and rng.random() < 0.3:
            # perturb best
            perturb = np.copy(best_h)
            # randomly modify a few entries
            k = max(1, n // 20)
            idxs = rng.choice(n, k, replace=False)
            perturb[idxs] = rng.random(k)
            # project
            perturb = project_onto_feasible(perturb, target)
            x0 = perturb
        # run optimizer
        options = {'maxiter': 500, 'ftol': 1e-8}
        try:
            res = minimize(compute_c5_local, x0, method='SLSQP',
                           bounds=bounds, constraints=cons, options=options)
        except Exception:
            break
        h_opt = np.clip(res.x, 0.0, 1.0)
        # ensure sum exactly
        h_opt = project_onto_feasible(h_opt, target)
        val_opt = compute_c5_local(h_opt)
        if val_opt < best_val - 1e-9:
            best_val = val_opt
            best_h = h_opt.copy()
        slsqp_runs += 1
        time_spent = time.time() - start_time
        time_left = budget_s - time_spent
        # early exit if reaching target bound
        if best_val <= 0.3808:
            break
    # After SLSQP, perform greedy refinement if time left
    time_spent = time.time() - start_time
    time_left = budget_s - time_spent
    if time_left > 5 and best_h is not None:
        # refine

        if refined_val < best_val:

    # final repair to ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_feasible(best_h, target)
    # compute final bound
    final_val = compute_c5_local(best_h)

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- helper functions (top level) -------------------

def compute_c5_general(h, n, target=None):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    # correlation of h with (1-h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * (2.0 / n)


def sum_eq_constraint(h, n, target):
    """Equality constraint: sum(h) == target."""
    return np.sum(h) - target


def project_onto_feasible(v, target, eps=1e-12):
    """
    Project vector v onto the set {0 <= x_i <= 1, sum x_i = target}.
    Simple water‑filling iteration – sufficient for the tiny dimensions used.
    """
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= eps:
        return v
    # iterative redistribution
    for _ in range(20):
        diff = target - s
        # indices that can still move in the needed direction
        if diff > 0:
            mask = v < 1.0 - eps
        else:
            mask = v > eps
        if not np.any(mask):
            break
        delta = diff / np.count_nonzero(mask)
        # apply limited by bounds
        add = np.where(mask, delta, 0.0)
        v = np.clip(v + add, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) <= eps:
            break
    return v


def random_feasible(n, target, rng):
    """Generate a random point in the feasible polytope."""
    a = rng.random(n)
    a = a / a.sum() * target
    a = np.clip(a, 0.0, 1.0)
    return project_onto_feasible(a, target)


def greedy_refine(h0, n, target, budget_s, start_time, rng):
    """
    Simple greedy mass‑transfer refinement.
    Moves a small amount of mass from a positively‑gradient index to a negatively‑gradient one.
    """
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5_general(h, n)
    cur_val = best_val

    max_alpha = 0.5
    min_alpha = 0.01
    it_no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now - start_time > budget_s - 0.05:
            break
        iteration += 1
        # linear schedule for step size
        elapsed = now - start_time
        alpha = max_alpha - (max_alpha - min_alpha) * (elapsed / budget_s)

        # correlation & lag of the current worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        L = idx - (n - 1)

        # gradient of the max‑lag overlap w.r.t. each component
        g = np.zeros(n)
        if L >= 0:
            i = np.arange(n - L)
            g[i] += 1.0 - h[i + L]
            i2 = np.arange(L, n)
            g[i2] -= h[i2 - L]
        else:
            s = -L
            i = np.arange(s, n)
            g[i] += 1.0 - h[i - s]
            i2 = np.arange(0, n - s)
            g[i2] -= h[i2 + s]

        # Choose a promising transfer direction
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        moved = False
        if pos.size and neg.size:
            i_pos = pos[np.argmax(g[pos])]
            i_neg = neg[np.argmin(g[neg])]
            delta = min(alpha, h[i_pos], 1.0 - h[i_neg]) * 0.5
            if delta > 0:
                h[i_pos] -= delta
                h[i_neg] += delta
                new_val = compute_c5_general(h, n)
                if new_val <= cur_val + 1e-12:
                    cur_val = new_val
                    moved = True
                    if new_val < best_val - 1e-12:
                        best_val = new_val
                        best_h = h.copy()
                        it_no_improve = 0
                    else:
                        it_no_improve += 1
                else:
                    # revert
                    h[i_pos] += delta
                    h[i_neg] -= delta
                    it_no_improve += 1
        if not moved:
            # random tiny transfer to keep exploration alive
            i, j = rng.choice(n, 2, replace=False)
            delta = min(alpha, h[i], 1.0 - h[j]) * 0.5
            if delta > 0:
                h[i] -= delta
                h[j] += delta
                new_val = compute_c5_general(h, n)
                if new_val <= cur_val:
                    cur_val = new_val
                    if new_val < best_val:
                        best_val = new_val
                        best_h = h.copy()
                    it_no_improve = 0
                else:
                    h[i] += delta
                    h[j] -= delta
                    it_no_improve += 1

        # occasional random reshuffle when stuck
        if iteration % 200 == 0 and it_no_improve > 5:
            a, b = rng.choice(n, 2, replace=False)
            total = h[a] + h[b]
            new_a = rng.random() * total
            new_b = total - new_a
            h[a], h[b] = new_a, new_b
            cur_val = compute_c5_general(h, n)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            it_no_improve = 0

    return best_h, best_val


# ------------------- main entry point -------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h (as a list), its C5 bound, and the number of points.
    The algorithm mixes a few SLSQP local optimisations with a simple greedy
    refinement that respects the constraints sum(h)=1 and 0≤h≤1.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine problem size
    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.size
    else:
        init_arr = None
        n = 42

    # enforce even length (required for exact half‑mass)
    if n % 2 != 0:
        n += 1

    target = n / 2.0  # because ∫_0^2 h = 1 and dx = 2/n

    # ---- start from provided construction if present ----
    best_h = None
    best_val = float("inf")
    if init_arr is not None and init_arr.size == n:
        h0 = np.clip(init_arr, 0.0, 1.0)
        h0 = project_onto_feasible(h0, target)
        val0 = compute_c5_general(h0, n)
        best_h, best_val = h0, val0

    # ---- SLSQP local searches (time‑bounded) ----
    time_spent = time.time() - start
    remaining = budget_s - time_spent
    bounds = [(0.0, 1.0)] * n
    constraints = {"type": "eq", "fun": sum_eq_constraint}
    # each run gets at most ~0.5 s for a quick solve; stop early if budget exhausted
    while remaining > 20.0:
        x0 = random_feasible(n, target, rng)
        # occasionally perturb the current best to diversify
        if best_h is not None and rng.random() < 0.3:
            pert = best_h.copy()
            k = max(1, n // 15)
            idxs = rng.choice(n, k, replace=False)
            pert[idxs] = rng.random(k)
            pert = project_onto_feasible(pert, target)
            x0 = pert
        options = {"maxiter": 500, "ftol": 1e-9}
        try:
            res = minimize(
                compute_c5_general,
                x0,
                args=(n, target),
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options=options,
            )
        except Exception:
            break
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_onto_feasible(h_opt, target)
        val_opt = compute_c5_general(h_opt, n)
        if val_opt < best_val - 1e-12:
            best_h, best_val = h_opt, val_opt
            # early exit if already under the target threshold
            if best_val <= 0.3808:
                break
        time_spent = time.time() - start
        remaining = budget_s - time_spent

    # ---- greedy refinement with mass transfers ----
    if best_h is not None and remaining > 5.0:
        refined_h, refined_val = greedy_refine(
            best_h, n, target, budget_s, start, rng
        )
        if refined_val < best_val:
            best_h, best_val = refined_h, refined_val

    # final safety projection and recompute exact bound
    best_h = project_onto_feasible(np.clip(best_h, 0.0, 1.0), target)
    final_val = compute_c5_general(best_h, n)

    # Return as plain Python list for the harness
    return best_h.tolist(), float(final_val), n
