def project_onto_simplex(v, target, upper=1.0):
    # Projection onto set {x: 0<=x<=upper, sum x = target}
    # Equivalent to projecting onto simplex with upper bound.
    # Use algorithm: first clip to upper, then project onto simplex (unbounded) with lower bound 0.
    v = np.clip(v, 0, upper)
    # Compute projection onto simplex (no upper bound) of sum target
    # Standard algorithm: sort descending.
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]

    else:

        theta = cssv[rho] / (rho + 1)

    w = np.minimum(w, upper)

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h (as a NumPy array), its C5 bound, and the number
    of discretisation points.  The implementation tries a family of simple
    shapes, the parent construction (if supplied) and a few random starts,
    then refines the best candidates with a smooth approximation of the
    max‑overlap objective using SLSQP.
    """
    start_time = time.time()
    np.random.seed(seed)

    # ---- parameters -------------------------------------------------
    # resolution – we upscale a 45‑point parent if it exists, otherwise use 180 points
    base_n = 45
    scale = 4                     # 45 * 4 = 180 points is a good trade‑off
    n_points = base_n * scale
    dx = 2.0 / n_points
    target_sum = n_points / 2.0   # ∫ h = 1  -->  sum(h) = n/2
    alpha = 30.0                  # smoothing parameter for the max

    # ---- helper utilities -------------------------------------------
    def project_to_feasible(v):
        """Project v onto {0 ≤ h ≤ 1, sum(h)=target_sum} (Euclidean)."""
        # start from clipped vector
        v = np.clip(v, 0.0, 1.0)
        lo, hi = -1.0, np.max(v)         # bounds for Lagrange multiplier λ
        for _ in range(60):               # bisection on λ so that sum==target
            lam = (lo + hi) * 0.5
            w = np.clip(v - lam, 0.0, 1.0)
            s = w.sum()
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        return np.clip(v - lam, 0.0, 1.0)

    def compute_c5(h):
        """Exact C5 value for a discrete step function."""
        conv = np.correlate(h, h, mode='full') * dx
        g = 1.0 - conv
        return float(g.max())

    # ---- initial candidates -----------------------------------------
    candidates = []

    # 1) parent construction if provided (repeat to current resolution)
    try:
        parent = np.array(initial_h_values, copy=True)
        if parent.size != base_n:          # fallback: just use it as is
            parent = parent[:base_n]
        # repeat each entry `scale` times – preserves integral exactly
        parent_rep = np.repeat(parent, scale)
        parent_rep = project_to_feasible(parent_rep)
        candidates.append(parent_rep)
    except Exception:
        pass

    # 2) a simple family of symmetric trapezoids (parameter a = ramp length)
    def trapezoid(a):
        """Symmetric trapezoid on [0,2]; a∈[0,1] is ramp length."""
        if a <= 0.0:
            h = np.full(n_points, 0.5)                     # uniform case
        else:
            H = 1.0 / (2.0 - a)                             # height
            x = (np.arange(n_points) + 0.5) * dx
            h = np.empty_like(x)
            left = x < a
            plateau = (x >= a) & (x <= 2.0 - a)
            right = ~ (left | plateau)
            h[left] = H * x[left] / a
            h[plateau] = H
            h[right] = H * (2.0 - x[right]) / a
        return project_to_feasible(h)

    for a in np.linspace(0.0, 1.0, 21):
        candidates.append(trapezoid(a))

    # 3) a few random feasible vectors (Dirichlet‑like)
    for _ in range(8):
        v = np.random.rand(n_points)
        v = project_to_feasible(v)
        candidates.append(v)

    # keep only the best few according to the exact C5 (cheap evaluation)
    evaluated = [(compute_c5(vec), vec) for vec in candidates]
    evaluated.sort(key=lambda kv: kv[0])
    top_candidates = [vec for _, vec in evaluated[:3]]

    # ---- smooth objective and its gradient ---------------------------
    def smooth_obj(h):
        conv = np.correlate(h, h, mode='full') * dx
        g = 1.0 - conv
        return logsumexp(alpha * g) / alpha

    def smooth_grad(h):
        conv = np.correlate(h, h, mode='full') * dx
        g = 1.0 - conv
        # soft‑max weights
        max_ag = np.max(alpha * g)
        expw = np.exp(alpha * g - max_ag)
        w = expw / expw.sum()
        grad = np.zeros_like(h)
        N = h.size
        shifts = np.arange(-(N - 1), N)        # length 2N‑1, same order as corr result
        for idx, s in enumerate(shifts):
            ws = w[idx]
            if ws == 0.0:
                continue
            if s >= 0:
                if s < N:
                    grad[:N - s] -= dx * ws * h[s:]
                    grad[s:]     -= dx * ws * h[:N - s]
            else:
                t = -s
                if t < N:
                    grad[t:]    -= dx * ws * h[:N - t]
                    grad[:N - t]-= dx * ws * h[t:]
        return grad

    # ---- constraints for SLSQP ---------------------------------------
    eq_cons = ({
        'type': 'eq',
        'fun': lambda h: np.sum(h) - target_sum,
        'jac': lambda h: np.ones_like(h)
    })
    bounds = [(0.0, 1.0)] * n_points

    # ---- local refinement with SLSQP ---------------------------------
    best_h = None
    best_c5 = np.inf

    for init in top_candidates:
        if time.time() - start_time > budget_s - 10:
            break
        # ensure feasibility before feeding optimiser
        x0 = project_to_feasible(init)

        res = minimize(
            smooth_obj,
            x0,
            method='SLSQP',
            jac=smooth_grad,
            bounds=bounds,
            constraints=eq_cons,
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        h_opt = project_to_feasible(res.x if res.success else x0)
        c5_val = compute_c5(h_opt)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()

    # ---- fallback: if nothing succeeded, use the best raw candidate ----
    if best_h is None:
        best_c5, best_h = evaluated[0]

    # final safety projection and evaluation
    best_h = project_to_feasible(best_h)
    final_c5 = compute_c5(best_h)

    # Return the result in the required format
    return (best_h, float(final_c5), n_points)
