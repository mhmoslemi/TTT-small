def propose_move(h):
   i, j = rng.choice(N, size=2, replace=False)
   sign = rng.choice([-1, 1])
   if sign == 1:
      # increase i, decrease j
      max_inc_i = 1.0 - h[i]
      max_dec_j = h[j]
      delta_max = min(max_inc_i, max_dec_j)
   else:
      max_inc_j = 1.0 - h[j]
      max_dec_i = h[i]
      delta_max = min(max_inc_j, max_dec_i)
   if delta_max <= 0:
      return None, None, None
   # choose delta
   delta = delta_max * step_factor * rng.random()
   # apply move
   if sign == 1:
       new_i = h[i] + delta
       new_j = h[j] - delta
   else:
       new_i = h[i] - delta
       new_j = h[j] + delta
   return i, j, (new_i, new_j)

def targeted_move(h, L, delta_factor):
    # find i where i+L in range and product low
    possible_i = np.arange(N - L) if L >= 0 else np.arange(-L, N)
    # compute product
    prod = h[possible_i] * h[possible_i + L]
    # choose i where prod minimal
    i = possible_i[np.argmin(prod)]
    # need to increase h_i and h_i+L
    # find j with biggest h to take from
    j_candidates = np.where(h > 1e-9)[0]
    # exclude i,i+L
    j_candidates = j_candidates[(j_candidates != i) & (j_candidates != i + L)]
    if len(j_candidates) == 0:
        return None
    # best j = argmax h
    j = j_candidates[np.argmax(h[j_candidates])]
    # compute delta_max = min( (1-h_i), (1 - h_iL), h_j ) maybe we split delta equally times two
    max_inc_i = 1 - h[i]
    max_inc_il = 1 - h[i+L]

def _c5_and_lag(h: np.ndarray):
    """Return (C5 value, lag giving the maximum, full correlation array)."""

    max_idx = int(np.argmax(corr))
    val = float(corr[max_idx] * dx)
    lag = max_idx - (n - 1)
