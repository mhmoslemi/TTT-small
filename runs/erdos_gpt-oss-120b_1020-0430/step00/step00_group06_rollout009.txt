def _c5_value(h: np.ndarray) -> float:
    """Return the overlap value C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _worst_shift(h: np.ndarray):
    """
    Return the shift (can be negative) that yields the maximal overlap
    together with the value itself (already multiplied by dx).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                 # zero shift corresponds to index n-1
    return shift, float(corr[idx] * (2.0 / n))


def _improve_one_step(h: np.ndarray, shift: int):

    Perform a small mass transfer that is guaranteed to keep the sum of h constant.
    The transfer is chosen so that it *should* reduce the contribution for the given shift.
    Returns (new_h, succeeded_flag).
    """

    if shift >= 0:
        hi = h[: n - shift]
        hi_s = h[shift:]
        src_score = hi * (1 - hi_s)          # large where contribution is large
        tgt_score = (1 - hi) * hi_s          # large where we would like to add mass
        src_loc = int(np.argmax(src_score))
        tgt_loc = int(np.argmax(tgt_score))
        src = src_loc
        tgt = tgt_loc
    else:

        src_loc = int(np.argmax(src_score))
        tgt_loc = int(np.argmax(tgt_score))
        src = src_loc + d
        tgt = tgt_loc

    if src == tgt:

        order = np.argsort(-tgt_score)
        for loc in order:
