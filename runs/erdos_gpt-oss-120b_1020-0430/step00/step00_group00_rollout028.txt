def propose_move(h):
    i, j = random distinct indices.
    max_inc = min(1 - h[i], h[j])  # amount i can increase, j decrease.
    max_dec = min(h[i], 1 - h[j]) # amount i can decrease, j increase.
    if random < 0.5:
        # increase i, decrease j
        d = np.random.rand() * max_inc
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
    else:
        # decrease i, increase j
        d = np.random.rand() * max_dec
        h_new = h.copy()
        h_new[i] -= d
        h_new[j] += d
    return h_new

    i, j = random.sample(range(n), 2)
    hi, hj = h[i], h[j]
    # direction = +/-1 meaning increase i, decrease j
    direction = 1 if np.random.rand() < 0.5 else -1
    if direction == 1:
        # increase i up to 1, decrease j down to 0
        max_inc = min(1 - hi, hj)
        if max_inc <= 0:
            return None
        d = np.random.rand() * max_inc
        new_h = h.copy()
        new_h[i] = hi + d
        new_h[j] = hj - d
    else:
        # decrease i to 0, increase j up to 1
        max_dec = min(hi, 1 - hj)
        if max_dec <= 0:
            return None
        d = np.random.rand() * max_dec
        new_h = h.copy()
        new_h[i] = hi - d
        new_h[j] = hj + d
    return new_h

import time, math, numpy as np

def _max_corr(h: np.ndarray) -> float:
    """Maximum of h*(1-h) correlation (no scaling)."""
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def _c5_value(h: np.ndarray) -> float:
    """Scaled C5 value."""
    n = h.size
    return (2.0 / n) * _max_corr(h)

def _random_binary_feasible(rng, n):
    """Return a binary {0,1} vector of length n with exact sum n/2 (n must be even)."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    h[:half] = 1.0
    rng.shuffle(h)
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose problem size ------------------------------------------------
    # ------------------------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, dtype=float)  # type: ignore
        n = init_arr.size
    except Exception:
        n = 44                       # even size makes the half–sum integral
        init_arr = None

    target_sum = n / 2.0
    dx = 2.0 / n
    target_c5 = 0.3808               # we aim to beat this
    target_corr = target_c5 / dx

    # ------------------------------------------------------------------
    # Initial candidate -------------------------------------------------
    # ------------------------------------------------------------------
    best_h = None
    best_corr = float('inf')

    if init_arr is not None and init_arr.shape[0] == n:
        # use provided construction if it respects the constraints
        if np.all((0.0 <= init_arr) & (init_arr <= 1.0)):
            s = init_arr.sum()
            if abs(s - target_sum) < 1e-8:
                best_h = init_arr.copy()
                best_corr = _max_corr(best_h)

    # fallback: simple binary half‑ones
    if best_h is None:
        best_h = _random_binary_feasible(rng, n)
        best_corr = _max_corr(best_h)

    # ------------------------------------------------------------------
    # Simulated annealing / hill climbing -------------------------------
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_corr = best_corr

    # temperature schedule – linear cooling from T0 to 0
    T0 = max(0.5, cur_corr * 0.1)

    # helper for a single feasible move (pairwise swap of mass)
    def propose_move(vec):
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        hi, hj = vec[i], vec[j]
        if rng.random() < 0.5:
            # increase i, decrease j
            max_delta = min(1.0 - hi, hj)
            if max_delta <= 0.0:
                return None, None, None
            d = rng.random() * max_delta
            new_vec = vec.copy()
            new_vec[i] = hi + d
            new_vec[j] = hj - d
        else:
            # decrease i, increase j
            max_delta = min(hi, 1.0 - hj)
            if max_delta <= 0.0:
                return None, None, None
            d = rng.random() * max_delta
            new_vec = vec.copy()
            new_vec[i] = hi - d
            new_vec[j] = hj + d
        return new_vec, i, j

    # main loop
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break
        # cooling factor based on elapsed fraction
        frac = (now - start_time) / (budget_s * 0.95)
        T = T0 * (1.0 - frac)
