def compute_c5_vec(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def hill_climb(a, max_iter=1000):
    n = len(a)
    cur = a.copy()
    cur_c5 = compute_c5_vec(cur.astype(float))
    # pre compute ones indices for speed
    while True:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        best_c5 = cur_c5
        best_swap = None
        # iterate over candidate swaps
        for i in ones:
            for j in zeros:
                # swap i and j
                cur[i] = 0
                cur[j] = 1
                new_c5 = compute_c5_vec(cur.astype(float))
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    best_swap = (i, j)
                # revert
                cur[i] = 1
                cur[j] = 0
        if best_swap is not None:
            i, j = best_swap
            cur[i] = 0
            cur[j] = 1
            cur_c5 = best_c5
        else:
            break
    return cur.astype(float), cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    seed_ = seed
    np.random.seed(seed_)
    random.seed(seed_)

    # Determine n_points and initial vector
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.shape[0]
    except NameError:
        # default to n=70
        n = 70
        # start from uniform half ones
        init_h = np.concatenate([np.ones(n//2), np.zeros(n - n//2)])

    # Ensure integer n is even (so sum = n/2 integer)
    if n % 2 != 0:
        # If odd, we can still have sum = n/2 (non integer) but we want binary.
        # Instead we treat target sum = int(round(n/2))
        target_sum = int(np.round(n/2))
    else:
        target_sum = n//2

    # define function to project binary 
    def to_binary(h):
        # threshold 0.5 and adjust count
        a = (h >= 0.5).astype(int)
        cur_sum = a.sum()
        if cur_sum > target_sum:
            # need to turn some 1s to 0
            idx = np.where(a == 1)[0]
            # randomly choose cur_sum - target_sum indices to set to 0
            to_zero = np.random.choice(idx, size=cur_sum - target_sum, replace=False)
            a[to_zero] = 0
        elif cur_sum < target_sum:
            idx = np.where(a == 0)[0]
            to_one = np.random.choice(idx, size=target_sum - cur_sum, replace=False)
            a[to_one] = 1
        return a

    # convert initial to binary
    current_best_h = to_binary(init_h)
    best_h = current_best_h.copy().astype(float)
    best_c5 = compute_c5_vec(best_h)

    # Setup loop
    start = time.time()
    # limit number of restarts based on time
    iterations = 0
    while time.time() - start < budget_s * 0.95:  # leave margin 
        # random binary initialization
        # generate random permutation and set first target_sum to 1
        perm = np.random.permutation(n)
        a = np.zeros(n, dtype=int)
        a[perm[:target_sum]] = 1
        # hill climb
        a_opt, c5_opt = hill_climb(a)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = a_opt.astype(float)
            # optional early exit if cutoff achieved
            if best_c5 <= 0.3808:
                break
        iterations += 1
    # final result
    # compute exact c5 bound using compute_c5_vec
    final_c5 = compute_c5_vec(best_h)
    return best_h, final_c5, n

def optimize_continuous(h0, time_budget):
    cur = h0.copy()

    best_c5 = cur_c5
    start = time.time()

        i, j, delta = move

        cur[j] -= delta
        new_c5 = compute_c5(cur)
        delta_c5 = new_c5 - cur_c5

                best = cur.copy()
        else:
            cur[i] -= delta
            cur[j] += delta
        T = max(T * decay, 1e-7)
        iteration += 1
        if iteration % 10000 == 0 and time.time() - start > time_budget * 0.7:

    return best, best_c5

import numpy as np, time, random, math

def compute_c5(h):
    n = h.shape[0]
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def repair(h):
    n = h.shape[0]
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Add diff to one index if possible
        if diff > 0:
            # need to increase sum
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size > 0:
                i = idx[0]
                h[i] += diff
                if h[i] > 1.0:
                    h[i] = 1.0
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size > 0:
                i = idx[0]
                h[i] += diff  # diff negative
                if h[i] < 0.0:
                    h[i] = 0.0
    return np.clip(h, 0.0, 1.0)

def hill_climb_binary(a):
    # a expected binary int array of shape (n,) with sum = n/2
    n = a.shape[0]
    cur = a.copy()
    cur_c5 = compute_c5(cur.astype(float))
    while True:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        best_c5 = cur_c5
        best_swap = None
        # evaluate all swaps
        for i in ones:
            for j in zeros:
                cur[i] = 0
                cur[j] = 1
                new_c5 = compute_c5(cur.astype(float))
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    best_swap = (i, j)
                cur[i] = 1
                cur[j] = 0
        if best_swap:
            i, j = best_swap
            cur[i] = 0
            cur[j] = 1
            cur_c5 = best_c5
        else:
            break
    return cur.astype(float), cur_c5

import numpy as np, time, random, math

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx  approximated on a uniform grid."""
    n = h.shape[0]
    # correlation of h with its complement, zero‑padded outside [0,2]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def repair(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce the exact integral constraint."""
    n = h.shape[0]
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:                     # need to increase the sum
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size:
                i = idx[0]
                h[i] = min(1.0, h[i] + diff)
        else:                            # need to decrease the sum
            idx = np.where(h > 1e-12)[0]
            if idx.size:
                i = idx[0]
                h[i] = max(0.0, h[i] + diff)
    return np.clip(h, 0.0, 1.0)


def random_binary_vector(n: int) -> np.ndarray:
    """Return a binary vector with exactly n/2 ones (n assumed even)."""
    a = np.zeros(n, dtype=int)
    a[: n // 2] = 1
    np.random.shuffle(a)
    return a


def hill_climb_binary(a: np.ndarray):
    """Steepest‑descent improvement for a binary vector."""
    n = a.shape[0]
    cur = a.copy()
    cur_c5 = compute_c5(cur.astype(float))
    while True:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        best_c5 = cur_c5
        best_swap = None
        for i in ones:
            for j in zeros:
                cur[i] = 0
                cur[j] = 1
                new_c5 = compute_c5(cur.astype(float))
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    best_swap = (i, j)
                cur[i] = 1
                cur[j] = 0
        if best_swap is None:
            break
        i, j = best_swap
        cur[i] = 0
        cur[j] = 1
        cur_c5 = best_c5
    return cur.astype(float), cur_c5


# ---------- simulated annealing ----------
def anneal(start_h: np.ndarray, time_limit: float):
    """Continuous annealing using pairwise mass transfers."""
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_c5 = compute_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5

    T = 0.02            # initial temperature
    min_T = 1e-7
    decay = 0.9995
    start = time.time()

    while time.time() - start < time_limit:
        # ----- decide move type -----
        if np.random.rand() < 0.7:               # random pair move
            i = np.random.randint(n)
            if np.random.rand() < 0.5:          # increase i, decrease j
                if cur[i] >= 1.0 - 1e-12:
                    continue
                j_cands = np.where(cur > 1e-12)[0]
                if j_cands.size == 0:
                    continue
                j = np.random.choice(j_cands)
                delta_max = min(1.0 - cur[i], cur[j])
                if delta_max <= 1e-12:
                    continue
                delta = np.random.rand() * delta_max
                hi, hj = cur[i], cur[j]
                cur[i] = hi + delta
                cur[j] = hj - delta
                changed = (i, j, hi, hj, delta, True)   # True ⇒ i increased
            else:                                 # decrease i, increase j
                if cur[i] <= 1e-12:
                    continue
                j_cands = np.where(cur < 1.0 - 1e-12)[0]
                if j_cands.size == 0:
                    continue
                j = np.random.choice(j_cands)
                delta_max = min(cur[i], 1.0 - cur[j])
                if delta_max <= 1e-12:
                    continue
                delta = np.random.rand() * delta_max
                hi, hj = cur[i], cur[j]
                cur[i] = hi - delta
                cur[j] = hj + delta
                changed = (i, j, hi, hj, delta, False)  # False ⇒ i decreased
        else:                                      # targeted move (worst shift)
            corr = np.correlate(cur, 1.0 - cur, mode="full")
            max_idx = np.argmax(corr)
            shift = max_idx - (n - 1)

            if shift >= 0:
                part_len = n - shift
                if part_len == 0:
                    continue
                h_i = cur[:part_len]
                h_j = cur[shift:]
                idx = np.arange(part_len)
                contrib = h_i * (1.0 - h_j)
            else:
                s = -shift
                part_len = n - s
                if part_len == 0:
                    continue
                h_i = cur[s:]
                h_j = cur[:part_len]
                idx = np.arange(s, n)
                contrib = h_i * (1.0 - h_j)

            # pick a high‑contribution index to *decrease* and a low‑contribution
            # index to *increase*
            i_rel = np.argmax(contrib)
            j_rel = np.argmin(contrib)
            i = idx[i_rel]
            j = idx[j_rel]
            if i == j:
                continue

            # we try “decrease i, increase j” first
            if cur[i] > 1e-12 and cur[j] < 1.0 - 1e-12:
                delta_max = min(cur[i], 1.0 - cur[j])
                orientation_inc_i = False
            elif cur[j] > 1e-12 and cur[i] < 1.0 - 1e-12:
                # opposite direction possible
                delta_max = min(cur[j], 1.0 - cur[i])
                orientation_inc_i = True   # means we will actually increase i
            else:
                continue
            if delta_max <= 1e-12:
                continue
            delta = np.random.rand() * delta_max
            hi, hj = cur[i], cur[j]
            if orientation_inc_i:
                cur[i] = hi + delta
                cur[j] = hj - delta
                changed = (i, j, hi, hj, delta, True)
            else:
                cur[i] = hi - delta
                cur[j] = hj + delta
                changed = (i, j, hi, hj, delta, False)

        # ----- evaluate -----
        new_c5 = compute_c5(cur)
        delta_c5 = new_c5 - cur_c5
        accept = False
        if delta_c5 < 0:
            accept = True
        else:
            if np.random.rand() < math.exp(-delta_c5 / max(T, 1e-12)):
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur.copy()
                if best_c5 <= 0.3808:                 # early exit
                    return best_h, best_c5
        else:
            # revert the tentative move
            i, j, hi, hj, delta, inc_i = changed
            cur[i] = hi
            cur[j] = hj

        # temperature schedule
        T = max(T * decay, min_T)

    return best_h, best_c5


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    random.seed(seed)

    # ---- initialise ----
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    except NameError:
        n = 70
        h0 = np.full(n, 0.5)          # satisfies ∫h = 1 automatically

    h0 = repair(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()

    # ---- a few binary hill‑climbs (fast global exploration) ----
    binary_time = budget_s * 0.25
    while time.time() - start < binary_time:
        a_bin = random_binary_vector(n)
        a_opt, c5_opt = hill_climb_binary(a_bin)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = a_opt.copy()
        if best_c5 <= 0.3808:
            break

    # ---- continuous annealing from the best found vector ----
    remaining = budget_s - (time.time() - start)
    if remaining > 0.5:   # leave a tiny margin for final wrap‑up
        best_h, best_c5 = anneal(best_h, remaining)

    # final repair and exact evaluation
    best_h = repair(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), n
