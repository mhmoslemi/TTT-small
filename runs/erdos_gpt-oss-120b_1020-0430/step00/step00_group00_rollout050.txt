def project_to_feasible(y, target, tol=1e-9):
    x = np.clip(y, 0.0, 1.0)
    # water-filling to meet sum constraint
    while True:
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to decrease sum
            mask = x > 0
        else:
            mask = x < 1
        count = mask.sum()
        if count == 0:
            break
        # compute amount to adjust per free variable
        delta = diff / count
        # adjust
        x[mask] -= delta
        # clip again (since some may cross bounds)
        x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np, time, random
from scipy.optimize import minimize

def evaluate_c(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(v, target, tol=1e-9, max_iter=1000):
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            mask = x > 0
        else:
            mask = x < 1
        cnt = mask.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        x[mask] -= delta
        x = np.clip(x, 0.0, 1.0)
    # final adjustment for numeric issues
    # slightly correct total by distributing residual uniformly within bounds
    residual = target - x.sum()
    if abs(residual) > tol:
        idx = np.where((x > 0) & (x < 1))[0]
        if len(idx) > 0:
            x[idx] += residual / len(idx)
            x = np.clip(x, 0.0, 1.0)
    return x

def random_feasible(n, target):
    v = np.random.rand(n)
    v = project_to_feasible(v, target)
    return v

def optimize_slsqp(h0, time_limit):
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_val = evaluate_c(best_h)
    start = time.time()
    # attempt a few restarts
    for restart in range(5):
        if time.time() - start > time_limit:
            break
        if restart == 0:
            x0 = h0.copy()
        else:
            x0 = random_feasible(n, target)
        cons = ({'type':'eq','fun':lambda x: np.sum(x) - target},)
        bounds = [(0.0,1.0)] * n
        res = minimize(
            lambda x: evaluate_c(x),
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter':200, 'ftol':1e-9, 'disp':False}
        )
        if not res.success:
            continue
        h = project_to_feasible(res.x, target)
        val = evaluate_c(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
    return best_h, best_val

def gradient_descent(h0, time_limit):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate_c(h)
    start = time.time()
    tau = 0.5
    tau_min = 0.005
    alpha = 0.05
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        # softmax weights focusing on larger correlation
        exp_vals = np.exp((corr - max_corr) / tau)
        w = exp_vals / np.sum(exp_vals)
        grad = np.zeros_like(h)
        for idx, wL in enumerate(w):
            shift = idx - (n - 1)
            grad += wL
            if shift > 0:
                grad[:n-shift] -= wL * h[shift:]
                grad[shift:] -= wL * h[:n-shift]
            elif shift < 0:
                s = -shift
                grad[s:] -= wL * h[:n-s]
                grad[:n-s] -= wL * h[s:]
            else:  # shift == 0
                grad -= 2.0 * wL * h
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, target)
        val = evaluate_c(h_new)
        if val < best_val:
            best_val, best_h = val, h_new.copy()
        h = h_new
        # schedule
        tau = max(tau_min, tau * 0.9995)
        alpha = max(0.001, alpha * 0.9995)
        if best_val < 0.3808:
            break
    return best_h, best_val

import time, random
import numpy as np
from scipy.optimize import minimize


def evaluate_c(h):
    """Return C₅ for a step vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def project_to_feasible(v, target, tol=1e-9, max_iter=200):
    """Project v onto {0≤x≤1, sum(x)=target} by water‑filling."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            mask = x > 0
        else:
            mask = x < 1
        cnt = mask.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        x[mask] -= delta
        x = np.clip(x, 0.0, 1.0)
    # tiny final tweak if needed
    residual = target - x.sum()
    if abs(residual) > tol:
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += residual / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible(n, target):
    """Generate a random feasible vector."""
    v = np.random.rand(n)
    return project_to_feasible(v, target)


def optimize_slsqp(h_start, time_limit):
    """Run a few SLSQP optimisations, returning the best feasible vector."""
    n = len(h_start)
    target = n / 2.0
    best_h = h_start.copy()
    best_val = evaluate_c(best_h)
    start = time.time()
    for restart in range(5):
        if time.time() - start > time_limit:
            break
        x0 = h_start.copy() if restart == 0 else random_feasible(n, target)
        cons = ({"type": "eq", "fun": lambda x: np.sum(x) - target},)
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            lambda x: evaluate_c(x),
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            continue
        cand = project_to_feasible(res.x, target)
        val = evaluate_c(cand)
        if val < best_val:
            best_val, best_h = val, cand.copy()
    return best_h, best_val


def gradient_descent(h0, time_limit):
    """Soft‑max gradient descent on the max‑correlation objective."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h, best_val = h.copy(), evaluate_c(h)
    start = time.time()
    tau, tau_min = 0.5, 0.005
    alpha = 0.05
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        exp_vals = np.exp((corr - max_corr) / tau)
        w = exp_vals / np.sum(exp_vals)          # soft‑max over shifts
        grad = np.zeros_like(h)
        for idx, wL in enumerate(w):
            shift = idx - (n - 1)
            grad += wL
            if shift > 0:
                grad[: n - shift] -= wL * h[shift:]
                grad[shift:] -= wL * h[: n - shift]
            elif shift < 0:
                s = -shift
                grad[s:] -= wL * h[: n - s]
                grad[: n - s] -= wL * h[s:]
            else:  # shift == 0
                grad -= 2.0 * wL * h
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, target)
        val = evaluate_c(h_new)
        if val < best_val:
            best_val, best_h = val, h_new.copy()
        h = h_new
        tau = max(tau_min, tau * 0.9995)
        alpha = max(0.001, alpha * 0.9995)
        if best_val < 0.3808:      # early stop once we beat the target
            break
    return best_h, best_val


def random_local_search(h_start, cur_best, time_limit):
    """Simple hill‑climbing by transferring mass between two coordinates."""
    n = len(h_start)
    target = n / 2.0
    h = h_start.copy()
    best_h, best_val = h.copy(), cur_best
    start = time.time()
    while time.time() - start < time_limit:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a random amount from i to j (or opposite)
        max_up = 1.0 - h[j]
        max_down = h[i]
        max_delta = min(max_up, max_down)
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        if np.random.rand() < 0.5:
            h_cand = h.copy()
            h_cand[i] -= delta
            h_cand[j] += delta
        else:
            # opposite direction
            max_up = 1.0 - h[i]
            max_down = h[j]
            max_delta = min(max_up, max_down)
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            h_cand = h.copy()
            h_cand[i] += delta
            h_cand[j] -= delta
        val = evaluate_c(h_cand)
        if val < best_val:
            best_val, best_h = val, h_cand.copy()
            h = h_cand
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) respecting the time budget."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Load / build an initial feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5, dtype=float)          # fallback
    n = len(h0)
    target = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_feasible(h0, target)

    best_h, best_val = h0.copy(), evaluate_c(h0)

    start_total = time.time()
    remaining = budget_s - (time.time() - start_total)

    # ------------------------------------------------------------------
    # 2. SLSQP stage (≈30% of remaining time, up to 2 min)
    # ------------------------------------------------------------------
    t_slsqp = min(0.3 * remaining, 120.0)
    if t_slsqp > 5.0:
        h_s, v_s = optimize_slsqp(best_h, t_slsqp)
        if v_s < best_val:
            best_h, best_val = h_s, v_s

    # ------------------------------------------------------------------
    # 3. Gradient‑descent stage (≈50% of remaining time, up to 10 min)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_total)
    t_grad = min(0.5 * remaining, 600.0)
    if t_grad > 5.0:
        h_g, v_g = gradient_descent(best_h, t_grad)
        if v_g < best_val:
            best_h, best_val = h_g, v_g

    # ------------------------------------------------------------------
    # 4. Random local search with the rest of the budget
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_total)
    if remaining > 1.0:
        h_r, v_r = random_local_search(best_h, best_val, remaining)
        if v_r < best_val:
            best_h, best_val = h_r, v_r

    # ------------------------------------------------------------------
    # 5. Final projection & evaluation
    # ------------------------------------------------------------------
    best_h = project_to_feasible(np.clip(best_h, 0.0, 1.0), target)
    final_c5 = evaluate_c(best_h)

    return best_h.tolist(), final_c5, n
