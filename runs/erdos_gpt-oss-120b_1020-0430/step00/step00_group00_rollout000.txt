import numpy as np, time

from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def adjust_sum(x, target):
    # Ensure sum of x equals target while staying [0,1]
    # Works by moving excess/deficit to a variable with slack
    x = np.clip(x, 0.0, 1.0)
    current = x.sum()
    diff = target - current
    if abs(diff) < 1e-12:
        return x
    # Try to distribute diff equally among all non-saturated entries
    # iterative approach
    # Use while loop to handle potential saturation
    while abs(diff) > 1e-12:
        # indices that can receive (if diff>0) or give (if diff<0)
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            # no capacity left; break (should not happen)
            break
        # amount to distribute per variable
        nmask = mask.sum()
        per = diff / nmask
        # compute actual feasible per variable
        for i in np.where(mask)[0]:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    # final clip
    return np.clip(x, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Return max overlap integral for a given step vector."""
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _adjust_sum(x, target):
    """
    Project vector x onto the hyper‑plane sum(x)=target while keeping 0≤x≤1.
    Simple iterative redistribution that respects bounds.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    # early exit
    if abs(diff) < 1e-12:
        return x
    # iterative redistribution
    # keep looping until the residual is negligible
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to add mass – look for entries < 1
            mask = x < 1.0 - 1e-12
        else:
            # need to remove mass – look for entries > 0
            mask = x > 1e-12
        if not np.any(mask):
            break
        candidates = np.where(mask)[0]
        # equal share of the remainder among free variables
        per = diff / len(candidates)
        for i in candidates:
            if diff > 0:
                cap = 1.0 - x[i]
                add = min(per, cap)
                x[i] += add
                diff -= add
            else:
                cap = x[i]
                sub = min(-per, cap)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _hill_climb(h, c5, deadline, rng):
    """Simple random pairwise mass‑exchange hill‑climbing."""
    n = len(h)
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # choose direction uniformly
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:
            continue
        delta = rng.uniform(0.0, max_move)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        new_c5 = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < c5:
            h, c5 = h_new, new_c5
    return h, c5

    Optimize a step function for the Erdős minimum overlap problem.

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    init_vec = globals().get("initial_h_values")

    else:

        h0 = rng.random(n)
        # scale to required integral
        target = n / 2.0

        h0 = _adjust_sum(h0, target)

    target = n / 2.0
    best_h = _adjust_sum(np.clip(h0, 0.0, 1.0), target)
    best_c5 = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Local optimisation loop (SLSQP)
    # ------------------------------------------------------------------
    def obj(x):
        # objective = max overlap * dx
        dx = 2.0 / len(x)
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx

    eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target}
