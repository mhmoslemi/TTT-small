def local_search(h, time_budget, start_time):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(best_h)
    curr_h = h.copy()
    curr_score = best_score
    # parameters
    max_iter = 0
    step_factor = 0.5  # start with 0.5 of max allowed delta
    # we will gradually reduce step factor.
    # while time left:
    while time.time() - start_time < time_budget:
        # choose source and target
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # direction: let's randomly decide donor-target
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        # compute max possible delta
        max_delta = min(curr_h[src], 1.0 - curr_h[dst])
        if max_delta <= 1e-12:
            continue
        # choose delta
        # step_factor maybe dynamic: we can use max_delta * (np.random.rand() ** 2) * step_factor?
        # The exponent 2 biases to small steps.
        delta = max_delta * step_factor * np.random.rand()
        if delta < 1e-10:
            continue
        # apply
        curr_h[src] -= delta
        curr_h[dst] += delta
        # compute new score
        new_score = compute_c5(curr_h)
        # if improved (lower)
        if new_score < curr_score - 1e-12:
            curr_score = new_score
            # update best if needed
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = curr_h.copy()
        else:
            # revert
            curr_h[src] += delta
            curr_h[dst] -= delta
        # optionally adjust step_factor based on acceptance ratio
        # but simple approach: gradually decay step_factor after some iterations.
        max_iter += 1
        if max_iter % 10000 == 0:
            # decay
            step_factor = max(step_factor * 0.99, 0.01)   # keep >0.01
    return best_h, best_score

import time
import numpy as np

def compute_c5(vec: np.ndarray) -> float:
    """Return the objective value for a given step function."""
    n = vec.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h) over all integer shifts
    corr = np.correlate(vec, 1.0 - vec, mode="full") * dx
    return float(corr.max())

def hill_climb(start_vec: np.ndarray, time_budget: float, start_time: float,
               T0: float = 5e-3) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climber.
    Keeps the linear sum constraint by moving mass between two coordinates.
    """
    n = start_vec.shape[0]
    cur = start_vec.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score

    # parameters
    max_delta_factor = 0.5          # max fraction of allowed transfer per move
    checks = 0

    while True:
        now = time.time()
        if now - start_time >= time_budget:
            break

        # linear cooling
        elapsed = now - start_time
        frac = elapsed / time_budget
        T = max(T0 * (1.0 - frac), 1e-12)

        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random direction of transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        max_delta = min(cur[src], 1.0 - cur[dst])
        if max_delta <= 1e-12:
            continue

        # propose a transfer
        delta = max_delta * max_delta_factor * np.random.rand()
        if delta <= 0.0:
            continue

        cur[src] -= delta
        cur[dst] += delta

        new_score = compute_c5(cur)
        dE = new_score - cur_score

        if dE < -1e-12:                # improvement
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = cur.copy()
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-dE / T):
                cur_score = new_score
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best = cur.copy()
            else:
                # revert move
                cur[src] += delta
                cur[dst] -= delta

        checks += 1
        # occasional tiny decay of step size to allow fine tuning
        if checks % 20000 == 0:
            max_delta_factor = max(max_delta_factor * 0.99, 0.01)

    return best, best_score

    # ------------------------------------------------------------------
    low_time = min(budget_s * 0.30, 200.0)          # at most 200 s for the coarse stage
    low_end = start + low_time
    low_best, low_score = hill_climb(init_vec, low_end, start)

    # ------------------------------------------------------------------
    # 3. Upsample to a finer grid (factor 2 → 90 points) and continue
    # ------------------------------------------------------------------
    high_vec = np.repeat(low_best, 2)               # exact sum preservation

    target_sum = high_vec.shape[0] / 2.0
    cur_sum = high_vec.sum()
    if abs(cur_sum - target_sum) > 1e-12:

    # remaining budget
