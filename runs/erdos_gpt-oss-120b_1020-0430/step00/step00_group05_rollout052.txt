import time
import numpy as np

def _project_capped_simplex(v: np.ndarray, s: float, tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0 <= x_i <= 1, sum(x)=s}.
    The solution has the form x_i = clip(v_i - λ, 0, 1) for some λ;
    we find λ by binary search.
    """
    lo = np.min(v - 1)          # λ low → all entries become 1
    hi = np.max(v)              # λ high → all entries become 0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        cur = np.sum(x)
        if abs(cur - s) <= tol:
            return x
        # sum decreases as λ increases
        if cur > s:
            lo = lam
        else:
            hi = lam
    # final clamp
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def _subgradient_step(h: np.ndarray, shift: int, step: float) -> np.ndarray:
    """Perform one sub‑gradient step for the worst shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        # (1 - h[i+shift]) part
        g[: n - shift] += 1.0 - h[shift:]
        # -h[i-shift] part
        g[shift:] -= h[: n - shift]
    else:
        d = -shift
        g[d:] += 1.0 - h[: n - d]
        g[: n - d] -= h[d:]
    return h - step * g


def _c5_value(h: np.ndarray) -> float:
    """Compute C5 for the current h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step‑function h (as list), its C5 bound, and the number of points.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Choose the discretisation size.

    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
    else:
        n = 200                     # fallback resolution
        init = None

    target_sum = n / 2.0               # ∫ h = 1  ⇒  Σh = n/2
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Helper to evaluate the current bound quickly.
    # ------------------------------------------------------------------
    def evaluate(h_vec):
        return _c5_value(h_vec)

    # ------------------------------------------------------------------
    # Initial best solution.
    # ------------------------------------------------------------------
    if init is not None:
        h_best = _project_capped_simplex(init, target_sum)
    else:
        h_best = _project_capped_simplex(rng.random(n), target_sum)

    best_val = evaluate(h_best)

    # ------------------------------------------------------------------
    # Main optimisation loop (sub‑gradient descent with restarts).
    # ------------------------------------------------------------------
    while time.time() - start < 0.94 * budget_s:
        # ----- initialise a new starting point --------------------------------
        if best_val < 0.5 and rng.random() < 0.15:
            # perturb the current best a little
            h = h_best + 0.04 * rng.standard_normal(n)
            h = np.clip(h, 0.0, 1.0)
            h = _project_capped_simplex(h, target_sum)
        else:

            h = _project_capped_simplex(rng.random(n), target_sum)

        # ----- sub‑gradient descent -----------------------------------------
        max_iters = 30000

            # compute correlation and worst shift
