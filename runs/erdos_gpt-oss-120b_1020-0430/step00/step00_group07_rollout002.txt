    # Clip first
    h = np.clip(h, 0.0, 1.0)
    # Short loop

        cur_sum = np.sum(h)
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        # Determine free indices where we can move in needed direction
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
