import numpy as np
import time
import random
import math

def _evaluate(h):
    """Return C5 for a discrete step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)


def _repair_sum(h, target):
    """Adjust a single entry to make the sum exactly target (keeps bounds)."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to add/subtract diff to a coordinate that can accommodate it
    if diff > 0:
        # need to increase some element
        idx = np.where(h + diff <= 1.0 + 1e-12)[0]
        if idx.size:
            h[idx[0]] += diff
        else:
            # fallback: spread tiny amount over all entries
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
    else:
        # need to decrease some element
        idx = np.where(h + diff >= -1e-12)[0]
        if idx.size:
            h[idx[0]] += diff
        else:
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
    return h


def _optimize(initial_h, budget, N):
    """Local search with annealing to lower the max overlap."""
    h = initial_h.copy()
    best_h = h.copy()
    best_val = _evaluate(h)
    cur_val = best_val

    start = time.time()
    iter_cnt = 0
    last_improve = 0
    epsilon = 0.5                # max magnitude of a single transfer
    stagnation_limit = 20000    # steps without improvement before a shake
    p_target = 0.25             # probability of a targeted move
    target_sum = N / 2.0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget:
            break

        # temperature for simulated annealing (linear decay)
        temp = max(0.001, 0.02 * (1.0 - elapsed / budget))

        # decide move type
        if random.random() < p_target:
            # --- targeted move aiming at the worst shift ---
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx_max = np.argmax(corr)
            shift = idx_max - (N - 1)          # integer shift causing max overlap

            # indices that can be increased on both sides of the shift
            if shift >= 0:
                i_base = np.arange(N - shift)
                i_pair = i_base + shift
            else:
                i_base = np.arange(-shift, N)
                i_pair = i_base + shift

            inc_caps = np.minimum(1.0 - h[i_base], 1.0 - h[i_pair])
            good = np.where(inc_caps > 1e-12)[0]
            if good.size == 0:
                # fallback to a random move
                move_type = 'random'
            else:
                # pick a random index from the feasible set
                pick = random.choice(good.tolist())
                i = i_base[pick]
                # possible increase amount (limited by epsilon)
                max_inc = min(inc_caps[pick], epsilon)
                if max_inc <= 0:
                    move_type = 'random'
                else:
                    delta = random.uniform(0.0, max_inc)
                    # choose a donor index to decrease
                    donors = np.where(h > delta + 1e-12)[0]
                    donors = donors[donors != i]
                    if donors.size == 0:
                        move_type = 'random'
                    else:
                        j = int(random.choice(donors.tolist()))
                        # perform the transfer
                        h[i] += delta
                        h[j] -= delta
                        move_type = 'targeted'
        else:
            move_type = 'random'

        if move_type == 'random':
            # --- simple transfer between two positions ---
            i = random.randrange(N)
            j = random.randrange(N - 1)
            if j >= i:
                j += 1

            # allowable delta so both stay in [0,1]
            delta_max = min(1.0 - h[i], h[j])
            delta_min = -min(h[i], 1.0 - h[j])
            lo = max(delta_min, -epsilon)
            hi = min(delta_max, epsilon)
            if lo >= hi:
                continue
            delta = random.uniform(lo, hi)

            h[i] += delta
            h[j] -= delta

        # evaluate new candidate
        new_val = _evaluate(h)

        # accept if better or with annealing probability
        if new_val < cur_val or random.random() < math.exp((cur_val - new_val) / temp):
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
                last_improve = iter_cnt
        else:
            # revert move
            if move_type == 'targeted':
                h[i] -= delta
                h[j] += delta
            else:  # random
                h[i] -= delta
                h[j] += delta

        # decay epsilon periodically
        if iter_cnt % 1000 == 0 and iter_cnt > 0:
            epsilon = max(epsilon * 0.9, 1e-5)

        # restart if stuck for a while
        if iter_cnt - last_improve > stagnation_limit:
            h = best_h.copy()
            # small random perturbations to escape flat regions
            for _ in range(10):
                a = random.randrange(N)
                b = random.randrange(N - 1)
                if b >= a:
                    b += 1
                d = random.uniform(-0.05, 0.05)
                dmax = min(1.0 - h[a], h[b])
                dmin = -min(h[a], 1.0 - h[b])
                d = max(min(d, dmax), dmin)
                if abs(d) > 1e-12:
                    h[a] += d
                    h[b] -= d
            cur_val = _evaluate(h)
            last_improve = iter_cnt

        iter_cnt += 1

    # final repair of the sum (tiny floating drift)
    best_h = _repair_sum(best_h, target_sum)
    best_val = _evaluate(best_h)
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] that minimizes
    C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine discretisation size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        N = h0.size
    else:
        # fallback: use 70 points, half ones half zeros
        N = 70
        h0 = np.zeros(N, dtype=np.float64)
        h0[: N // 2] = 1.0
        np.random.shuffle(h0)

    # Ensure the sum constraint exactly (integral = 1)
    target_sum = N / 2.0
    current_sum = h0.sum()
    if abs(current_sum - target_sum) > 1e-9:
        h0 = _repair_sum(h0, target_sum)

    # Run the optimizer within the allotted time
    # Allow a small safety margin for returning the result
    time_limit = budget_s - 0.2
    best_h, best_val = _optimize(h0, time_limit, N)

    # Final sanity check / repair
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_sum(best_h, target_sum)

    # Compute final bound explicitly
    final_c5 = _evaluate(best_h)

    return best_h, final_c5, N
