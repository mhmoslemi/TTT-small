Pseudo:

def hill_climb(h0, max_iters=20000, patience=500, initial_step=0.1, step_decay=0.99):
    h = h0.copy()
    n = len(h)
    target_sum = n/2
    best_h = h.copy()
    best_val = evaluate(h)
    step = initial_step
    rng = np.random.default_rng(seed)
    no_improve = 0
    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # delta random in [-step, step]
        delta = (rng.random() * 2 - 1) * step
        # ensure within bounds
        hi, hj = h[i], h[j]
        if hi + delta > 1 or hi + delta < 0 or hj - delta > 1 or hj - delta < 0:
            continue
        h[i] = hi + delta
        h[j] = hj - delta
        val = evaluate(h)
        if val < best_val - 1e-7:
            best_val = val
            best_h = h.copy()
            no_improve = 0
        else:
            # revert
            h[i] = hi
            h[j] = hj
            no_improve += 1

        if no_improve > patience:
            step *= step_decay
            no_improve = 0
            if step < 1e-4:
                break

Return best_h, best_val.

import time, math
import numpy as np

def _evaluate_corr(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate implements linear correlation (zero padding outside)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)

def _simulated_annealing(start_h, max_iters, init_temp, rng, deadline):
    """Simple SA that keeps the sum of h fixed by moving mass between two indices."""
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _evaluate_corr(h)
    cur_val = best_val
    temp = init_temp
    # linear cooling
    decay = init_temp / max_iters if max_iters > 0 else 0.0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # decide direction of transfer
        if rng.random() < 0.5:
            # move mass from j to i
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            h[i] += delta
            h[j] -= delta
            direction = 1
        else:
            # move mass from i to j
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            h[i] -= delta
            h[j] += delta
            direction = -1

        new_val = _evaluate_corr(h)

        improve = new_val < cur_val
        accept = improve or (rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert move
            if direction == 1:
                h[i] -= delta
                h[j] += delta
            else:
                h[i] += delta
                h[j] -= delta

        temp = max(temp - decay, 0.0)

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function h on [0,2] (as a NumPy array), its C5 bound,
    and the number of sample points.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95   # safety margin

    # ------------------------------------------------------------------
    # 1. Obtain a starting vector.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: simple symmetric block of ones covering half the interval
        n0 = 45
        h0 = np.zeros(n0)
        half = n0 // 2
        start = (n0 - half) // 2
        h0[start:start + half] = 1.0
        if n0 % 2 == 1:
            h0[n0 // 2] = 0.5

    n0 = h0.shape[0]

    # ------------------------------------------------------------------
    # 2. Helper to build a random feasible start (respecting bounds and sum).
    # ------------------------------------------------------------------
    def _random_start(n):
        """Return a feasible vector of length n with sum n/2, values in [0,1]."""
        half_int = n // 2
        rem = (n / 2) - half_int  # 0 for even n, 0.5 for odd n
        h = np.zeros(n)
        idx = rng.choice(n, size=half_int, replace=False)
        h[idx] = 1.0
        if rem > 0:
            # pick an additional index for the fractional remainder
            free = np.setdiff1d(np.arange(n), idx, assume_unique=True)
            extra = rng.choice(free)
            h[extra] = rem
        return h

    # ------------------------------------------------------------------
    # 3. Core optimisation loop over a few resolutions.
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_val = _evaluate_corr(best_h)
    best_n = n0

    # list of resolutions to explore (starting from the given size)
    resolutions = [n0, 2 * n0, 4 * n0]

    for n in resolutions:
        if time.time() > deadline:
            break

        # build an initial vector for this resolution
        if n == n0:
            base_h = h0.copy()
        else:
            # upsample by simple repetition (preserves sum exactly)
            factor = n // best_n
            base_h = np.repeat(best_h, factor)[:n]
            # if rounding caused a tiny deviation, fix it
            cur_sum = base_h.sum()
            target = n / 2.0
            if abs(cur_sum - target) > 1e-12:
                base_h += (target - cur_sum) / n

        # make sure bounds are respected after repetition
        np.clip(base_h, 0.0, 1.0, out=base_h)

        # evaluate the upsampled base
        cur_val = _evaluate_corr(base_h)
        if cur_val < best_val:
            best_h, best_val, best_n = base_h.copy(), cur_val, n

        # SA parameters tuned per resolution
        max_iters = int(25000 / (n / n0))   # fewer iterations for larger n
        init_temp = 0.12 * (1.0 + (n0 / n))  # moderate temperature

        # a small number of random starts plus the upsampled start
        starts = [base_h] + [_random_start(n) for _ in range(4)]

        for start_vec in starts:

            h_opt, val_opt = _simulated_annealing(

            if val_opt < best_val:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n_final = best_h.shape[0]
    dx = 2.0 / n_final
    # clip
    np.clip(best_h, 0.0, 1.0, out=best_h)
    # enforce exact integral by a simple projection onto the affine box
    target_sum = n_final / 2.0
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # distribute the difference uniformly, then re‑clip
        best_h += diff / n_final
        np.clip(best_h, 0.0, 1.0, out=best_h)
        # if still off because of clipping, do a final proportional rescale
        cur_sum = best_h.sum()
        if cur_sum != 0:
            best_h *= target_sum / cur_sum
        np.clip(best_h, 0.0, 1.0, out=best_h)

    final_val = _evaluate_corr(best_h)

    return best_h, float(final_val), int(n_final)
