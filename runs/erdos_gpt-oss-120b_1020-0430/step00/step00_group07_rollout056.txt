        # compute worst shift correlation vector
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)

        # compute gradient:

    total_zeros = n - total_ones
    # number of gaps = num_intervals + 1 = 6
    gaps = np.full(num_intervals + 1, total_zeros // (num_intervals + 1)) # baseline

    for i in range(remainder):

        arr.extend([1]*block_len)

       # choose two indices i,j distinct
       i, j = random.sample(range(n), 2)
       # compute feasible delta
       lower = max(-curr[i], curr[j] - 1.0)
       upper = min(1.0 - curr[i], curr[j])
       if lower >= upper: continue
       # propose delta scaled by a step factor:
       # for SA we can use uniform over feasible range or limited by step_max.
       delta = random.uniform(lower, upper) * 0.5  # scaling to avoid huge steps? Actually uniform already respects bounds.
       new = curr.copy()
       new[i] += delta
       new[j] -= delta
       # evaluate:
       new_score = eval(new)
       # compute acceptance
       if new_score < curr_score:
           accept = True
       else:
           # temperature schedule:
           elapsed = time.time() - start_time
           fraction = elapsed / time_limit
           # linear cooling
           T = T0 * (1 - fraction)
           if T <= 0: T = 1e-12
           prob = np.exp((curr_score - new_score) / T)  # (negative exponent) but we want acceptance probability e^{-(Δ)/T} where Δ = new - curr >0

       if accept:
           curr = new
           curr_score = new_score
           if new_score < best_score:
               best = new.copy()
               best_score = new_score

def _proj_to_bounds_and_sum(x, target_sum):
    """Euclidean projection of x onto {0<=v_i<=1, sum(v)=target_sum}."""
    # binary search on lambda such that sum(clip(x-lambda,0,1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        v = np.clip(x - lam, 0.0, 1.0)
        s = v.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _score(h, dx):
    """Return C5 for a given step vector h."""
    # np.correlate does the full linear cross‑correlation
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _gradient_of_worst_shift(h):
    """Gradient of the worst (max) correlation term."""
    n = h.size
    # correlation array
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.argmax(corr) - (n - 1)          # shift index (can be negative)
    grad = np.ones_like(h)
    if m >= 0:
        # h[i] interacts with h[i+m] (right shift)
        grad[: n - m] -= h[m:]
        grad[m:]      -= h[: n - m]
    else:
        p = -m
        grad[p:]      -= h[: n - p]
        grad[: n - p] -= h[p:]
    return grad


def _gd_optimize(start, dx, target_sum, max_iter=2000, step0=0.25):
    """Projected gradient descent on the max‑correlation objective."""
    h = _proj_to_bounds_and_sum(start, target_sum)
    best_h, best_val = h.copy(), _score(h, dx)

    for it in range(max_iter):
        grad = _gradient_of_worst_shift(h)
        step = step0 * (0.995 ** it)               # simple geometric decay
        cand = _proj_to_bounds_and_sum(h - step * grad, target_sum)
        val = _score(cand, dx)

        if val < best_val - 1e-12:
            best_h, best_val = cand.copy(), val

        # accept if we improved; otherwise keep the old point (no momentum)
        if val <= _score(h, dx) + 1e-12:
            h = cand
        # early termination if step becomes tiny
        if step * np.linalg.norm(grad) < 1e-8:
            break
    return best_h, best_val


def _sa_optimize(start, dx, target_sum, time_limit):
    """Simple simulated‑annealing using pairwise mass swaps."""
    n = start.size
    cur = start.copy()
    cur_score = _score(cur, dx)
    best_h, best_score = cur.copy(), cur_score
    start_time = time.time()
    T0 = 0.018                                          # initial temperature

    while time.time() - start_time < time_limit:
        i, j = random.sample(range(n), 2)
        # feasible delta keeping both entries in [0,1] and preserving the sum
        lo = max(-cur[i], cur[j] - 1.0)
        hi = min(1.0 - cur[i], cur[j])
        if lo >= hi:
            continue
        delta = random.uniform(lo, hi)                 # full feasible move
        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        new_score = _score(new, dx)

        if new_score < cur_score:
            accept = True
        else:
            elapsed = time.time() - start_time
            frac = elapsed / time_limit
            T = max(T0 * (1.0 - frac), 1e-12)
            prob = np.exp((cur_score - new_score) / T)
            accept = random.random() < prob

        if accept:
            cur, cur_score = new, new_score
            if new_score < best_score:
                best_h, best_score = new.copy(), new_score
    return best_h, best_score


def _make_triangle(n, target_sum):
    """Linear‑ramp (triangle) shape, then projected onto the feasible set."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0) / 1.0)   # peak =1 at centre
    return _proj_to_bounds_and_sum(h, target_sum)


def _make_cosine(n, target_sum):
    """Cosine bump (½+½ cos πx) projected onto the feasible set."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = 0.5 * (1.0 + np.cos(np.pi * (xs - 1.0)))       # 0 at ends, 1 at centre
    return _proj_to_bounds_and_sum(h, target_sum)


def _make_five_intervals(n, target_sum):
    """Binary pattern: 5 equal blocks of ones, gaps distributed as evenly as possible."""
    ones = n // 2                     # must be exactly half the points
    blocks = 5
    block_len = ones // blocks
    zeros = n - ones
    gaps = np.full(blocks + 1, zeros // (blocks + 1), dtype=int)
    rem = zeros % (blocks + 1)
    gaps[:rem] += 1
    seq = []
    for g in gaps[:-1]:
        seq.extend([0] * g)
        seq.extend([1] * block_len)
    seq.extend([0] * gaps[-1])
    return np.array(seq, dtype=float)


def _random_binary(n):
    """Random 0/1 vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # length of the vector (use the supplied one if present)
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        init_vec = np.asarray(initial_h_values, dtype=float)
    else:
        n = 80                     # default fallback
        init_vec = None

    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # Gather a handful of diverse seeds
    # ------------------------------------------------------------------
    seeds = []

    if init_vec is not None:
        seeds.append(init_vec)

    seeds.append(_make_triangle(n, target_sum))
    seeds.append(_make_cosine(n, target_sum))
    seeds.append(_make_five_intervals(n, target_sum))

    # a few random binary starts
    for _ in range(3):
        seeds.append(_random_binary(n))

    start_time = time.time()
    best_h, best_score = None, float('inf')

    # ------------------------------------------------------------------
    # Phase 1 – projected gradient descent from each seed
    # ------------------------------------------------------------------
    for s in seeds:
        if time.time() - start_time > budget_s * 0.55:
            break
        h_opt, sc = _gd_optimize(s, dx, target_sum, max_iter=2500, step0=0.30)
        if sc < best_score:
            best_h, best_score = h_opt.copy(), sc

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing from the current best solution
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time) - 0.25
    if remaining > 0.3 and best_h is not None:
        h_sa, sc_sa = _sa_optimize(best_h, dx, target_sum, remaining)
        if sc_sa < best_score:
            best_h, best_score = h_sa, sc_sa

    # ------------------------------------------------------------------
    # Final polishing: a short gradient descent to settle any remaining noise
    # ------------------------------------------------------------------
    if best_h is not None:
        h_final, sc_final = _gd_optimize(best_h, dx, target_sum,
                                         max_iter=800, step0=0.10)
        if sc_final < best_score:
            best_h, best_score = h_final, sc_final

    # ------------------------------------------------------------------
    # Compute the exact bound using the supplied evaluator (if any)
    # ------------------------------------------------------------------
    try:
        c5_bound = evaluate_erdos_solution(best_h)
    except Exception:                     # fall back to our own computation
        c5_bound = _score(best_h, dx)

    # Return as required: (h_values, c5_bound, n_points)
    return best_h, float(c5_bound), n
