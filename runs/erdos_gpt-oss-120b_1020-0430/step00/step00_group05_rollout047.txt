def shift_array(arr, k):
    n = arr.shape[0]
    if k >= 0:
        out = np.empty_like(arr)
        out[:n-k] = arr[k:]
        out[n-k:] = 0.0
    else:
        k = -k
        out = np.empty_like(arr)
        out[k:] = arr[:n-k]
        out[:k] = 0.0
    return out

import time
import numpy as np

# ---------- helper utilities ----------
def _project_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x : 0<=x_i<=1, sum x_i = target}."""
    # binary search for lambda such that sum clip(v+lam,0,1) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _shift_array(arr: np.ndarray, k: int) -> np.ndarray:
    """Shift arr by k positions, filling with 0 (zero‑padding)."""
    n = arr.size
    if k >= 0:
        if k >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[:n - k] = arr[k:]
        out[n - k :] = 0.0
        return out
    else:
        k = -k
        if k >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[k:] = arr[: n - k]
        out[:k] = 0.0
        return out


def _evaluate_c5(h: np.ndarray) -> float:
    """Return the C5 value for the step function defined by h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _subgradient_step(h: np.ndarray, step: float) -> tuple[np.ndarray, float]:
    """Perform one sub‑gradient step for the max‑overlap objective."""
    n = h.size
    # correlation and worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)               # actual integer lag
    # gradient of the active shift (see analysis)
    plus = _shift_array(h, shift)
    minus = _shift_array(h, -shift)
    grad = 1.0 - plus - minus
    # descent + projection
    y = h - step * grad
    y = np.clip(y, 0.0, 1.0)
    y = _project_sum_bounds(y, n / 2.0)
    return y, float(corr[idx])


def _local_optimize(start_h: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """Run a bounded sub‑gradient descent from start_h."""
    h = start_h.copy()
    best_h = h.copy()
    best_val = _evaluate_c5(h)
    it = 0
    t0 = time.time()
    while time.time() - t0 < time_limit:
        it += 1
        step = 0.5 / np.sqrt(it)
        h, cur_val = _subgradient_step(h, step)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
    return best_h, best_val


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    total_budget = budget_s * 0.95  # a safety margin

    # ----- choose discretisation size -----
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        repeat = 5                     # enlarge while preserving the average automatically
        h0 = np.tile(base, repeat)
    else:
        # fall‑back: 200 points, random start
        n = 200
        h0 = np.random.rand(n)

    n = h0.size
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_sum_bounds(h0, n / 2.0)

    # initial record
    best_h = h0.copy()
    best_val = _evaluate_c5(best_h)

    # ----- iterative search (restarts) -----
    while time.time() - start_time < total_budget:
        remaining = total_budget - (time.time() - start_time)
        # create a fresh start around the current best
        perturb = 0.02 * (np.random.rand(n) - 0.5)
        start_h = np.clip(best_h + perturb, 0.0, 1.0)
        start_h = _project_sum_bounds(start_h, n / 2.0)

        # allocate part of the remaining budget to this local run
        local_budget = min(remaining * 0.9, 8.0)   # at most a few seconds per restart
        cand_h, cand_val = _local_optimize(start_h, local_budget)

        if cand_val < best_val:
            best_val = cand_val
            best_h = cand_h.copy()

    # final evaluation
    final_c5 = _evaluate_c5(best_h)

    return best_h.tolist(), float(final_c5), n
