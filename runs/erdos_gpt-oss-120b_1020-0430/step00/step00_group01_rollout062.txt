def compute_c5(h):
    # using numpy
    r = np.correlate(h, 1.0 - h, mode='full')
    return np.max(r) * (2.0/len(h))

def hill_climb(initial_h, max_iter, step_init=0.5, step_decay=0.9995):
    h = initial_h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    step = step_init

    for it in range(max_iter):
        # random indices
        i, j = np.random.choice(n, 2, replace=False)
        # compute bounds
        max_inc_i = 1.0 - h[i]
        max_dec_i = h[i]
        max_inc_j = 1.0 - h[j]
        max_dec_j = h[j]
        # delta lower bound: need hi+delta ≤1 => delta ≤ max_inc_i ; hi+delta ≥0 => delta ≥ -max_dec_i
        # hj-delta within bounds: delta ≤ max_dec_j ; delta ≥ -max_inc_j
        low = max(-max_dec_i, -max_inc_j)
        high = min(max_inc_i, max_dec_j)
        if low >= high:
            continue
        delta = np.random.uniform(low, high)
        # apply
        h[i] += delta
        h[j] -= delta

        # compute c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            # keep changes
        else:
            # revert
            h[i] -= delta
            h[j] += delta
            # maybe accept with probability (tempered)
        # decay step maybe not needed because step bound derived from h values. However we could scale delta random within low-high as above, which inherently ensures not too large; but we can also scale step (a factor) to limit magnitude: delta *= step_factor maybe.
        step *= step_decay
    return h, best_c5

def project_sum_and_bounds(h):
    n = len(h)
    desired = n/2.0
    # clip first
    h = np.clip(h, 0, 1)
    # simple projection to sum using water-filling approach:
    # compute residual diff
    diff = desired - h.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # allocate diff equally among indices not at bound
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if len(free) == 0:
            # all at bounds but diff remains; we adjust by saturating at bounds sequentially.
            # Here we can just distribute to all indices, then clip.
            h += diff / n
            h = np.clip(h, 0, 1)
        else:
            h[free] += diff / len(free)
            h = np.clip(h, 0, 1)
        diff = desired - h.sum()
    # final small residual
    if abs(diff) > 1e-9:
        # adjust the first free entry if any
        free = np.where((h > 0) & (h < 1))[0]
        if len(free) > 0:
            h[free[0]] += diff
            h = np.clip(h, 0, 1)
    return h

import time
import numpy as np

# --------------------------------------------------------------
# Utility functions
# --------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx using the provided discretisation."""
    n = h.size
    dx = 2.0 / n
    # full cross‑correlation, zero padding outside the interval
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(r.max() * dx)


def project_sum_and_bounds(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy sum_i h_i = n/2 (exactly, up to numerical noise)."""
    n = h.size
    desired = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # iterative water‑filling style projection
    for _ in range(12):
        diff = desired - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt == 0:                     # all at a bound – distribute uniformly then re‑clip
            h += diff / n
        else:
            h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    # final tiny correction (should be negligible)
    diff = desired - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def propose_move(h: np.ndarray):
    """Propose a feasible two‑index move (i,j,delta) that preserves the sum."""
    n = h.size
    free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
    if free.size < 2:
        return None
    i, j = np.random.choice(free, 2, replace=False)
    # delta added to i, subtracted from j
    low = max(-h[i], -(1.0 - h[j]))
    high = min(1.0 - h[i], h[j])
    if low >= high:
        return None
    delta = np.random.uniform(low, high)
    return i, j, delta


def gradient_move(h: np.ndarray):
    """Move guided by the gradient of the currently worst (maximal) correlation."""
    n = h.size
    # correlation vector and the shift that attains the maximum
    r = np.correlate(h, 1.0 - h, mode="full")
    shift = int(r.argmax()) - (n - 1)          # signed shift

    # gradient of r[shift] w.r.t. each entry of h
    g = np.zeros_like(h)
    idx = np.arange(n)
    ip = idx + shift
    valid = (ip >= 0) & (ip < n)
    g[idx[valid]] += 1.0 - h[ip[valid]]
    im = idx - shift
    valid2 = (im >= 0) & (im < n)
    g[idx[valid2]] -= h[im[valid2]]

    # we want to move opposite to the gradient
    inc_idx = np.where((g > 0) & (h > 1e-12))[0]   # decreasing these entries lowers the max
    dec_idx = np.where((g < 0) & (h < 1.0 - 1e-12))[0]  # increasing these entries lowers the max
    if inc_idx.size == 0 or dec_idx.size == 0:
        return None
    i = np.random.choice(inc_idx)
    j = np.random.choice(dec_idx)

    # i will be decreased, j increased → delta (added to i, subtracted from j) is negative
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 1e-12:
        return None
    step = np.random.uniform(0.0, max_delta)
    delta = -step
    return i, j, delta


def hill_climb(h0: np.ndarray, max_seconds: float) -> tuple:
    """
    Simple greedy / simulated‑annealing hill descent.
    Returns the best (h, C5) found within the time budget.
    """
    start = time.time()
    n = h0.size
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # temperature parameters for occasional uphill acceptance
    T0 = 0.02
    unchanged = 0

    while True:
        if time.time() - start > max_seconds:
            break

        # choose move type
        if np.random.rand() < 0.7:
            mv = gradient_move(cur_h)
        else:
            mv = propose_move(cur_h)

        if mv is None:
            continue
        i, j, delta = mv
        # apply move
        cur_h[i] += delta
        cur_h[j] -= delta
        new_c5 = compute_c5(cur_h)

        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            unchanged = 0
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # possibly accept uphill move
            elapsed = time.time() - start
            T = T0 * max(0.0, 1.0 - elapsed / max_seconds)
            if T > 0 and np.random.rand() < np.exp(-(new_c5 - cur_c5) / T):
                cur_c5 = new_c5
                unchanged = 0
            else:
                # revert
                cur_h[i] -= delta
                cur_h[j] += delta
                unchanged += 1

        # occasional random re‑perturbation if stuck
        if unchanged > 20000:
            pert = best_h + 0.02 * (np.random.rand(n) - 0.5)
            cur_h = project_sum_and_bounds(np.clip(pert, 0.0, 1.0))
            cur_c5 = compute_c5(cur_h)
            unchanged = 0

    return best_h, best_c5


def upscale_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid, followed by projection."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    h_new = project_sum_and_bounds(np.clip(h_new, 0.0, 1.0))
    return h_new
