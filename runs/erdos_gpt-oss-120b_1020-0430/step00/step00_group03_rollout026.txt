import time, random, math
import numpy as np

def _project_to_sum_onehalf(h):
    """Clip to [0,1] and adjust the sum to exactly len(h)/2."""
    h = np.clip(h, 0.0, 1.0)
    N = h.shape[0]
    target = N * 0.5
    diff = target - h.sum()
    # simple iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break
            i = int(np.random.choice(cand))
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            i = int(np.random.choice(cand))
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h


def _score_and_corr(h):
    """return (C5 bound, full correlation array)."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / N), corr


def _gradient_for_shift(h, shift):
    """∂/∂h_i of overlap(shift) = 1 - h_{i+shift} - h_{i-shift}."""
    N = h.shape[0]
    g = np.ones(N, dtype=np.float64)
    if shift == 0:
        g -= 2.0 * h
        return g
    d = abs(shift)
    if shift > 0:
        # term h_{i+shift}
        g[:N - d] -= h[d:]
        # term h_{i-shift}
        g[d:] -= h[:N - d]
    else:  # shift < 0
        # now i+shift = i-d, i-shift = i+d
        g[d:] -= h[:N - d]       # h_{i-shift} = h_{i+d}
        g[:N - d] -= h[d:]       # h_{i+shift} = h_{i-d}
    return g


def _make_move(h, i_up, i_down, delta):
    """transfer delta from i_down to i_up (preserves sum)."""
    new_h = h.copy()
    new_h[i_up] += delta
    new_h[i_down] -= delta
    return new_h


def _random_transfer(h, max_frac=0.5):
    """propose a random feasible transfer between two indices."""
    N = h.shape[0]
    i_up, i_down = np.random.choice(N, size=2, replace=False)
    # decide direction: we will increase i_up, decrease i_down
    cap_up = 1.0 - h[i_up]
    cap_down = h[i_down]
    max_transfer = min(cap_up, cap_down)
    if max_transfer <= 0.0:
        return None, None, None
    delta = max_transfer * random.random() * max_frac
    return i_up, i_down, delta


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function that improves the Erdős C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64)
        N = h.shape[0]
        h = _project_to_sum_onehalf(h)
    else:
        N = 70
        h = np.random.rand(N)
        h = _project_to_sum_onehalf(h)

    best_h = h.copy()
    best_score, _ = _score_and_corr(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.98               # safety margin
    max_temp = 0.03
    min_temp = 1e-6
    max_iters_without_improve = 20000
    iters_without_improve = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() - start_time < time_limit:
        # compute current objective and the worst shift
        cur_score, corr = _score_and_corr(h)
        max_idx = int(np.argmax(corr))
        shift = max_idx - (N - 1)               # np.correlate offset

        # temperature (linear cooling)
        elapsed = time.time() - start_time
        progress = elapsed / time_limit
        T = max_temp * (1.0 - progress) + min_temp * progress

        # decide whether to use a gradient guided move or a random one
        use_gradient = random.random() < 0.7

        if use_gradient:
            g = _gradient_for_shift(h, shift)
            i_up = int(np.argmin(g))            # most negative gradient -> increase
            i_down = int(np.argmax(g))          # most positive gradient -> decrease

            cap_up = 1.0 - h[i_up]
            cap_down = h[i_down]
            max_transfer = min(cap_up, cap_down)
            if max_transfer <= 1e-12:

                i_up, i_down, delta = _random_transfer(h, max_frac=0.8)

            iters_without_improve = 0

    # ------------------------------------------------------------------
    # final reporting
    # ------------------------------------------------------------------
    final_score, _ = _score_and_corr(best_h)
    return best_h.tolist(), final_score, N
