def continuous_anneal(h, time_left, rng):
    n = len(h)
    target_sum = n/2
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    while time.time() - start < time_left:
        # temperature schedule based on elapsed fraction
        t_frac = (time.time() - start) / time_left
        T = (1 - t_frac) * initial_T   # linear cooling
        # propose random pair i != j
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # propose delta
        max_delta = min(0.1, cur_h[j]) # we can move from j to i
        # actually we want to add epsilon to i, subtract from j.
        # epsilon limited by bound constraints.
        # Compute epsilon range:
        eps_max = min(1 - cur_h[i], cur_h[j])
        eps_max = max(eps_max, 0)
        if eps_max <= 0:
            continue
        epsilon = (rng.random() - 0.5) * eps_max * 2 # random in [-eps_max, eps_max]
        # but we want to keep sum unchanged: we add epsilon to i and subtract epsilon from j.
        new_h = cur_h.copy()
        new_h[i] += epsilon
        new_h[j] -= epsilon
        # ensure bounds:
        if new_h[i] < 0 or new_h[i] > 1 or new_h[j] < 0 or new_h[j] > 1:
            continue
        new_val = compute_c5(new_h)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T+1e-12)):
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
    return best_h, best_val

def projected_subgradient_descent(h, time_left):
    n = len(h)
    target = n/2
    start = time.time()
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_h=cur_h.copy()
    best_val=cur_val
    # step size schedule
    while time.time() - start < time_left:
        # find shift with max correlation
        cor = np.correlate(cur_h, 1-cur_h, mode='full')
        max_idx = np.argmax(cor)
        # shift offset: index - (n-1)
        shift = max_idx - (n - 1)
        # compute gradient vector
        grad = np.zeros(n)
        # for i such that i+shift in bounds:
        if shift >=0:
            # i from 0 to n-1-shift
            i_range = np.arange(0, n-shift)
            grad[i_range] = (1 - cur_h[i_range+shift]) - cur_h[i_range]
        else:
            # shift negative: i from -shift to n-1
            i_range = np.arange(-shift, n)
            grad[i_range] = (1 - cur_h[i_range+shift]) - cur_h[i_range]
        # Actually gradient formula: derivative of c = sum_i (h_i*(1-h_{i+shift}))
        # derivative w.r.t h_i: (1 - h_{i+shift}) - h_{i-shift}, because term appears from i as first factor and also as second factor when i is i+shift for some other i.
        # Let's compute properly:
        # For a given shift s, correlation = sum_{i max(0,-s)}^{min(n-1, n-1-s)} h[i] * (1 - h[i+s]).
        # derivative wrt h[j]:
        # If j in range for i as first factor: term contributes (1 - h[j+s]).
        # If j-s in range as second factor: term contributes -h[j-s].
        # So grad[j] = (1 - h[j+s]) if j+s in bounds else 0) - (h[j-s] if j-s in bounds else 0)
        # This matches formula: grad = a - b
        # Implement:
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            # first factor
            if 0 <= j+s < n:   # where s=shift
                val += 1 - cur_h[j+s]
            # second factor
            if 0 <= j-s < n:
                val -= cur_h[j-s]
            grad[j] = val
        # But we can vectorize:
        s = shift
        grad = np.zeros(n)
        # first term:
        idx1 = np.arange(max(0, -s), min(n, n - s))
        # idx1 is possible i where i + s in range.
        grad[idx1] += 1 - cur_h[idx1 + s]
        # second term:
        idx2 = np.arange(max(0, s), min(n, n + s))
        # i where i - s in range i = idx2 such that i - s in bounds.
        grad[idx2] -= cur_h[idx2 - s]
        # This yields grad vector.
        # Now step:
        step = 0.5 # maybe adapt
        new_h = cur_h - step * grad
        # project onto bounds and sum
        new_h = project_bounds_sum(new_h, target)
        new_val = compute_c5(new_h)
        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        else:
            # maybe reduce step size
            step *= 0.5
        # Could break when step small
    return best_h, best_val
```

def subgradient_descent(h_start, time_budget):
    n = len(h_start)
    target_sum = n/2.0
    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    # initial step size
    step = 0.5
    while time.time() - start < time_budget:
        # compute correlation to get shift index
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        # compute subgradient
        grad = np.zeros(n)
        s = shift
        # first term: (1 - h[i+s]) for i where i + s in range
        if s >= 0:
            idx_i = np.arange(0, n - s)
            grad[idx_i] += 1 - cur_h[idx_i + s]
        else:
            idx_i = np.arange(-s, n)
            grad[idx_i] += 1 - cur_h[idx_i + s]
        # second term: -h[i - s] for i where i - s in range
        if s >= 0:
            idx_i = np.arange(s, n)
            grad[idx_i] -= cur_h[idx_i - s]
        else:
            idx_i = np.arange(0, n + s)
            grad[idx_i] -= cur_h[idx_i - s]
        # gradient step
        new_h = cur_h - step * grad
        # project onto bounds and sum
        new_h = project_bounds_sum(new_h, target_sum)
        new_val = compute_c5(new_h)
        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
            # possibly increase step slightly?
            step = min(step * 1.1, 1.0)
        else:
            step = step * 0.5
        # if step becomes too small break maybe
        if step < 1e-6:
            break
    return best_h, best_val
```

But need to reconsider gradient definition: We are minimizing max correlation. At current max shift s, the objective is c(s) = Σ h[i] * (1 - h[i+s]) for i where both indices exist. This is linear minus quadratic term.

Derivative: d/d h[j] of c(s) = (1 - h[j+s]) if j + s in bounds else 0 - h[j-s] if j - s in bounds else 0.

Thus we implemented above. Good.

def hill_climb_binary(start_h, rng, max_steps, time_limit):
    n = len(start_h)
    h = start_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    T0 = 0.01
    for step in range(max_steps):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        # temperature decreasing
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + 1e-8
        i = rng.choice(np.flatnonzero(h == 1))
        j = rng.choice(np.flatnonzero(h == 0))
        # swap
        h[i], h[j] = h[j], h[i]
        new_val = compute_c5(h)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if rng.random() < np.exp(-(new_val - cur_val) / T):
                accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        # optionally early exit if best_val low enough.
    return best_h, best_val

import time
import numpy as np

def compute_c5(h):
    """Return max overlap integral for step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_bounds_sum(x, target):
    """
    Project vector x onto the box [0,1]^n while forcing sum(x)=target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def binary_from_continuous(h):
    """Return a binary (0/1) vector with exactly half of the entries equal to 1."""
    n = h.shape[0]
    k = n // 2
    idx = np.argpartition(h, -k)[-k:]          # indices of the k largest entries
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx] = 1.0
    return bin_h


def thue_morse(n):
    """Generate the first n values of the Thue–Morse binary sequence (0/1)."""
    seq = np.arange(n, dtype=np.int64)
    popcnt = np.vectorize(lambda x: bin(x).count("1") % 2)
    return popcnt(seq).astype(float)


def upsample_initial(target_n):
    """Upsample the global `initial_h_values` (if present) to length target_n."""
    init = np.array(initial_h_values, dtype=float)   # may raise NameError
    n0 = init.shape[0]
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_new = np.interp(x_new, x0, init)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_bounds_sum(h_new, target_n / 2.0)
    return h_new


def hill_climb_binary(start_h, rng, max_steps, time_limit):
    """
    Simulated‑annealing style binary hill climbing.
    Swaps a 1 with a 0 while preserving the exact sum.
    """
    n = start_h.shape[0]
    h = start_h.copy()
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    T0 = 0.015                     # initial temperature
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        # linear temperature schedule
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1.0 - frac) + 1e-12

        # choose random positions of opposite colours
        i = rng.choice(np.flatnonzero(h == 1.0))
        j = rng.choice(np.flatnonzero(h == 0.0))

        # propose swap
        h[i], h[j] = h[j], h[i]
        new_val = compute_c5(h)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if rng.random() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

    return best_h, best_val


def continuous_anneal(h_start, time_budget, rng):
    """
    Continuous simulated annealing preserving the sum exactly.
    Randomly transfers a small amount of mass between two coordinates.
    """
    n = h_start.shape[0]
    target_sum = n / 2.0
    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    T0 = 0.04
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (1.0 - frac) + 1e-12

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # decide direction of transfer
        if rng.random() < 0.5:
            # move from j to i
            max_eps = min(1.0 - cur_h[i], cur_h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps
            new_h = cur_h.copy()
            new_h[i] += eps
            new_h[j] -= eps
        else:
            # move from i to j
            max_eps = min(1.0 - cur_h[j], cur_h[i])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps

        new_val = compute_c5(new_h)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val


def subgradient_descent(h_start, time_budget):
    """
    Projected sub‑gradient descent on the worst shift.
    The step size is adapted dynamically.
    """
    n = h_start.shape[0]
    target_sum = n / 2.0
    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    step = 0.6
    while time.time() - start < time_budget:
        # correlation to locate worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)          # integer shift (can be negative)

        # compute sub‑gradient for this shift
        grad = np.zeros(n, dtype=float)
        s = shift
        # contribution from the first factor: (1 - h[i+s]) where i+s in range
        if s >= 0:
            i1 = np.arange(0, n - s)
        else:
            i1 = np.arange(-s, n)
        grad[i1] += 1.0 - cur_h[i1 + s]

        # contribution from the second factor: -h[i-s] where i-s in range
        if s >= 0:
            i2 = np.arange(s, n)
        else:
            i2 = np.arange(0, n + s)
        grad[i2] -= cur_h[i2 - s]

        # take a step and project
        new_h = cur_h - step * grad
        new_h = project_bounds_sum(new_h, target_sum)
        new_val = compute_c5(new_h)

        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
            step = min(step * 1.2, 1.0)           # gently increase step if successful
        else:
            step *= 0.5                            # shrink step on failure
            if step < 1e-6:
                break
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    total_start = time.time()

    # Choose a resolution that is a power of two (convenient for Thue–Morse)
    n_points = 256
    target_sum = n_points / 2.0

    # Build a set of initial candidates
    candidates = []

    # 1. Thue–Morse binary pattern (already satisfies sum constraint)
    candidates.append(thue_morse(n_points))

    # 2. If a parent construction is available, up‑sample it and also use its binary version
    try:
        up = upsample_initial(n_points)
        candidates.append(up)                     # continuous version
        candidates.append(binary_from_continuous(up))
    except NameError:
        pass

    # 3. A random binary start (fallback)
    rand_bin = np.zeros(n_points, dtype=float)
    rand_idx = rng.choice(n_points, n_points // 2, replace=False)
    rand_bin[rand_idx] = 1.0
    candidates.append(rand_bin)

    # Evaluate all candidates and keep the best
    best_h = None
    best_val = float('inf')
    for cand in candidates:
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()

    # ------------------------------------------------------------------
    # Phase 1: binary simulated annealing (multiple restarts)
    elapsed = time.time() - total_start
    remaining = budget_s - elapsed - 10.0   # keep a safety margin
    if remaining > 5.0:
        n_restarts = 3
        per_restart = remaining / (n_restarts + 2)   # reserve time for later phases
        for _ in range(n_restarts):
            # start either from the current best or a fresh random binary vector
            if rng.random() < 0.6:
                start_bin = binary_from_continuous(best_h)
            else:
                idx = rng.choice(n_points, n_points // 2, replace=False)
                start_bin = np.zeros(n_points, dtype=float)
                start_bin[idx] = 1.0

            # Hill‑climb with a generous step budget; it will stop when the time limit hits
            max_steps = 500_000
            bin_h, bin_val = hill_climb_binary(start_bin, rng, max_steps, per_restart)
            if bin_val < best_val:
                best_val = bin_val
                best_h = bin_h.copy()
            if time.time() - total_start > budget_s - 5.0:
                break

    # ------------------------------------------------------------------
    # Phase 2: continuous annealing (preserves exact sum)
    elapsed = time.time() - total_start
    remaining = budget_s - elapsed - 5.0
    if remaining > 3.0:
        cont_h, cont_val = continuous_anneal(best_h, remaining, rng)
        if cont_val < best_val:
            best_val = cont_val
            best_h = cont_h.copy()

    # ------------------------------------------------------------------
    # Phase 3: projected sub‑gradient descent for final polishing
    elapsed = time.time() - total_start
    remaining = budget_s - elapsed - 2.0
    if remaining > 2.0:
        refined_h, refined_val = subgradient_descent(best_h, remaining)
        if refined_val < best_val:
            best_val = refined_val
            best_h = refined_h.copy()

    # Final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_bounds_sum(best_h, target_sum)
    final_val = compute_c5(best_h)

    return best_h, final_val, n_points
