def evaluate(h): returns float C5.

def hill_climb(initial_h, max_iters, time_limit):
   best = copy of h
   best_score = evaluate(best)
   cur = best
   cur_score = best_score
   for iter in range(max_iters):
        if time exceeded: break
        # randomly pick i_one and i_zero
        idx_one = random choice from indices where h==1
        idx_zero = random choice from where h==0
        # swap them
        cur[idx_one] = 0; cur[idx_zero] =1
        new_score = evaluate(cur)
        if new_score < cur_score:
            cur_score = new_score
        else:
            # revert swap
            cur[idx_one] = 1; cur[idx_zero] =0
        # update best if better
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
   return best,best_score

    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    # Determine n_points:
    if 'initial_h_values' in globals():

        init = np.array(initial_h_values, dtype=float)
        # make binary by threshold 0.5? Or choose top n/2 largest.

        h0[idx] = 1.0
    else:

        n = 42
        # random binary half ones
        h0 = np.zeros(n)

        else:
            cur = best_h.copy()
        cur_score = evaluate(cur)
        # set number of inner iterations
        iters = 0
        # approx inner budget: maybe few thousand per outer loop

            iters += 1

import time
import random
import numpy as np
import cvxpy as cp

def _evaluate(h):
    """C5 for a discrete step function."""
    # h may be list or array
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a good upper bound for C5."""
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    target_bound = 0.3808                      # we try to beat this
    best_h = None
    best_score = float("inf")
    best_n = None

    # ----------------------------------------------------------------------
    # Helper: try an integer programming formulation for a given n
    # ----------------------------------------------------------------------
    def try_milp(n):
        half = n // 2
        # binary variables for the step function
        h = cp.Variable(n, boolean=True)
        # auxiliary AND variables w[i,j] = h[i] * h[j]
        w = cp.Variable((n, n), boolean=True)
        T = cp.Variable()                       # max overlap (uncalibrated)

        constraints = [cp.sum(h) == half]

        # linearisation of w = h_i AND h_j
        for i in range(n):
            for j in range(n):
                constraints.append(w[i, j] <= h[i])
                constraints.append(w[i, j] <= h[j])
                constraints.append(w[i, j] >= h[i] + h[j] - 1)

        # overlap constraints for every non‑zero shift
        for s in range(-(n - 1), n):
            if s == 0:
                continue
            start_i = max(0, -s)
            end_i = min(n, n - s)          # exclusive
            expr = 0
            for i in range(start_i, end_i):
                expr += h[i] - w[i, i + s]
            constraints.append(expr <= T)

        constraints.append(T >= 0)

        prob = cp.Problem(cp.Minimize(T), constraints)

        # try a few MILP solvers; ignore those not installed
        for solver in ("GLPK_MI", "CBC", "SCIP", "ECOS_BB"):
            try:
                prob.solve(solver=solver, warm_start=True, verbose=False)
                if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                    return h.value, float(T.value)
            except Exception:
                continue
        return None, None

    # ----------------------------------------------------------------------
    # Try increasing problem sizes that make the target easier to reach
    # ----------------------------------------------------------------------
    candidate_ns = [60, 84, 100, 120, 42]        # larger n → smaller dx → easier bound
    for n in candidate_ns:
        # stop if we run out of time
        if time.time() - start_time > budget_s * 0.85:

        # if we have already beaten the target, return immediately
        if best_score <= target_bound:

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    if best_h is None:

        best_h = h_curr.tolist()
        best_score = cur_score
