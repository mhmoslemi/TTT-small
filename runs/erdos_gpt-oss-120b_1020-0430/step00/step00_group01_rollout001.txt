    for i in range(n):

        if diff > 0:

        else:

        h[i] += add

import time
import numpy as np
import scipy.optimize

def _dx(h):
    """step size for a vector h representing a function on [0,2]"""
    return 2.0 / h.shape[0]

def _c5_value(h):
    """exact C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = _dx(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _enforce_feasibility(h):
    """Clamp to [0,1] and exactly enforce sum_i h_i = n/2."""
    n = h.shape[0]
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    diff = target - s
    # Distribute the remainder sequentially; this converges quickly because diff is tiny.
    i = 0
    while abs(diff) > 1e-12 and i < n:
        if diff > 0.0:
            cap = 1.0 - h[i]
            add = min(diff, cap)
            h[i] += add
            diff -= add
        else:
            cap = h[i]
            sub = min(-diff, cap)
            h[i] -= sub
            diff += sub
        i += 1
        if i == n and abs(diff) > 1e-12:
            i = 0
    return h

def _propose_move(h, max_step, rng):
    """Return a new feasible h obtained by moving mass between two coordinates."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # random direction
    if rng.random() < 0.5:
        # move from i to j
        allowed = min(max_step, h[i], 1.0 - h[j])
        if allowed <= 0.0:
            return h.copy()
        delta = rng.random() * allowed
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        # move from j to i
        allowed = min(max_step, h[j], 1.0 - h[i])
        if allowed <= 0.0:
            return h.copy()
        delta = rng.random() * allowed
        h_new = h.copy()
        h_new[j] -= delta
        h_new[i] += delta
    return h_new

def _random_feasible(n, rng):

    h = rng.random(n)          # uniform in [0,1)
    target = n * 0.5
    # scale to get the correct sum, then clip and adjust
    total = float(np.sum(h))
    if total > 0:
        h *= target / total
    h = _enforce_feasibility(h)
    return h

    # ------------------------------------------------------------------
