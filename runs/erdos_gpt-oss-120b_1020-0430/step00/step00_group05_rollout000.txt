def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    g = 1 - h
    # Reverse g to compute correlation
    g_rev = g[::-1]
    Nfft = 1 << (2 * n - 1).bit_length()  # next power of two >= 2*n
    # Actually to avoid wrap-around, use Nfft = 2*n
    Nfft = 2*n
    H = np.fft.rfft(h, Nfft)
    G = np.fft.rfft(g_rev, Nfft)
    conv = np.fft.irfft(H * G, Nfft)
    # conv length Nfft, actual full linear correlation values are conv[:2*n-1]
    max_val = np.max(conv[:2*n-1])
    return max_val * dx

import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a binary (or [0,1]) vector h."""
    n = h.size
    dx = 2.0 / n
    # full cross‑correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _random_binary(n: int) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def _thue_morse(n: int) -> np.ndarray:
    """Thue‑Morse sequence of length n (n must be a power of two)."""
    seq = np.fromiter((bin(i).count("1") & 1 for i in range(n)), dtype=float)
    # For powers of two the sequence contains exactly n/2 ones.
    return seq


def _local_search(n: int, time_budget: float, start_vec: np.ndarray = None):
    """
    Simple simulated‑annealing / hill‑climbing on binary vectors of length n.
    Returns the best vector found and its C5 value.
    """
    start = time.time()
    if start_vec is not None:
        cur = start_vec.copy()
    else:
        cur = _thue_morse(n) if (n & (n - 1)) == 0 else _random_binary(n)

    cur_val = _c5_value(cur)
    best = cur.copy()
    best_val = cur_val

    it = 0
    T0 = 0.35  # initial temperature

    while time.time() - start < time_budget:
        # pick a 1 and a 0 and swap them
        one_idx = np.flatnonzero(cur > 0.5)
        zero_idx = np.flatnonzero(cur < 0.5)
        if one_idx.size == 0 or zero_idx.size == 0:
            break
        i1 = np.random.choice(one_idx)
        i0 = np.random.choice(zero_idx)

        new = cur.copy()
        new[i1] = 0.0
        new[i0] = 1.0

        new_val = _c5_value(new)
        delta = new_val - cur_val

        # linear temperature decay
        elapsed = time.time() - start
        temp = T0 * max(0.001, 1.0 - elapsed / time_budget)

        if delta <= 0.0 or np.random.rand() < np.exp(-delta / temp):
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best = cur.copy()
                best_val = cur_val

        it += 1
        # occasional random restart if we have stalled
        if it % 25000 == 0 and cur_val > best_val * 1.01:
            cur = _random_binary(n)
            cur_val = _c5_value(cur)

    return best, best_val
