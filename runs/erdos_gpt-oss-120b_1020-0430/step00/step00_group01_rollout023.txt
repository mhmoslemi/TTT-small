def generate_initial_pattern(n, rng):
    # produce Thue-Morse if n power of 2
    # else random balanced binary
    def thue_morse(k):
        # generate sequence length 2**k
        seq = np.array([0], dtype=int)
        for i in range(k):
            seq = np.concatenate([seq, 1 - seq])
        return seq
    if (n & (n-1)) == 0:  # power of 2
        k = int(np.log2(n))
        seq = thue_morse(k)
    else:
        seq = rng.integers(0, 2, size=n, dtype=int)
    # adjust sum to n/2
    target = n // 2
    current = seq.sum()
    if current > target:
        # set some 1's to 0
        ones = np.where(seq == 1)[0]
        to_flip = rng.choice(ones, size=current-target, replace=False)
        seq[to_flip] = 0
    elif current < target:
        zeros = np.where(seq == 0)[0]
        to_flip = rng.choice(zeros, size=target-current, replace=False)
        seq[to_flip] = 1
    # return as float array
    return seq.astype(float)

def simulated_annealing(h0, max_iter, rng, start, budget):
    cur = h0.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    for it in range(max_iter):
        if time.time() - start > budget * 0.99: break
        # choose random swap
        i = rng.choice(np.flatnonzero(cur == 1))
        j = rng.choice(np.flatnonzero(cur == 0))
        cur[i] = 0.0
        cur[j] = 1.0
        new_c5 = compute_c5(cur)
        if new_c5 < cur_c5:
            accept = True
        else:
            T = 0.02 * (1 - it / max_iter) + 1e-6
            accept = rng.random() < math.exp(-(new_c5 - cur_c5) / T)
        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
    return best, best_c5

import time, math, numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """C₅ for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _thue_morse(n: int) -> np.ndarray:
    """Balanced Thue‑Morse of length n (n must be a power of two)."""
    k = int(np.log2(n))
    seq = np.array([0], dtype=int)
    for _ in range(k):
        seq = np.concatenate([seq, 1 - seq])
    return seq.astype(float)


def _balanced_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random 0/1 vector with exactly n/2 ones."""
    h = rng.integers(0, 2, size=n).astype(float)
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.flatnonzero(h == 1)
        flip = rng.choice(idx, size=cur - target, replace=False)
        h[flip] = 0.0
    elif cur < target:
        idx = np.flatnonzero(h == 0)
        flip = rng.choice(idx, size=target - cur, replace=False)
        h[flip] = 1.0
    return h


def _project_constraints(h: np.ndarray) -> np.ndarray:

    n = h.shape[0]
    target = n / 2.0

    for _ in range(5):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        h -= diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the worst‑shift term."""
    n = h.shape[0]
    g = np.ones_like(h)
    if shift > 0:
        g[: n - shift] -= h[shift:]
        g[shift:] -= h[: n - shift]
    # shift==0 gives g = 1‑2h automatically
    return g


def _optimize_continuous(h0: np.ndarray, max_iter: int,
                        start_alpha: float,
                        rng: np.random.Generator,
                        t_start: float,
                        budget: float) -> tuple:
    """Projected sub‑gradient descent with a tiny annealing noise."""
    h = _project_constraints(h0.astype(float, copy=True))
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _compute_c5(h)

    for it in range(max_iter):
        if time.time() - t_start > budget * 0.99:
            break

        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        lag = int(np.argmax(corr) - (n - 1))
        shift = abs(lag)

        # gradient of the active term
        g = _gradient_for_shift(h, shift)

        # simple linear schedule for step size
        alpha = start_alpha * (1.0 - it / max_iter)

        # gradient step + projection
        h -= alpha * g
        np.clip(h, 0.0, 1.0, out=h)
        # enforce the integral constraint
        h = _project_constraints(h)

        # optional small stochastic perturbation every few hundred steps
        if it % max(200, max_iter // 10 + 1) == 0 and rng.random() < 0.1:
            noise = rng.normal(scale=0.02, size=n)
            h += noise
            np.clip(h, 0.0, 1.0, out=h)
            h = _project_constraints(h)

        cur_c5 = _compute_c5(h)
        if cur_c5 < best_c5 - 1e-9:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ----------------------------------------------------------------------
    # 1.  Existing construction (if supplied)
    # ----------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)  # type: ignore
        if init.ndim == 1 and init.shape[0] % 2 == 0:
            init = np.clip(init, 0.0, 1.0)
            init = _project_constraints(init)
            c5 = _compute_c5(init)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, init.copy(), init.shape[0]
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # 2.  Search parameters
    # ----------------------------------------------------------------------
    lengths = [48, 56, 64, 72, 80, 96, 112, 128, 144, 160, 192, 224, 256]
    lengths = [L for L in lengths if L % 2 == 0]

    # ----------------------------------------------------------------------
    # 3.  Main stochastic optimisation loop
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.96:
        n = int(rng.choice(lengths))
        # choose an initial pattern
        if (n & (n - 1)) == 0 and rng.random() < 0.6:
            h0 = _thue_morse(n)
        else:
            h0 = _balanced_random(n, rng)

        # number of iterations scales with problem size
        max_iter = int(6000 * (n / 128.0))
        start_alpha = 0.4

        h_opt, c5_opt = _optimize_continuous(
            h0, max_iter, start_alpha, rng, start_time, budget_s
        )

        if c5_opt < best_c5:
            best_c5, best_h, best_n = c5_opt, h_opt.copy(), n

    # ----------------------------------------------------------------------
    # 4.  Final polishing of the best candidate (if time allows)
    # ----------------------------------------------------------------------
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        # a longer, low‑step‑size run on the current best
        extra_iter = int(15000 * (best_n / 128.0))
        h_refined, c5_refined = _optimize_continuous(
            best_h, extra_iter, 0.15, rng, start_time, budget_s
        )
        if c5_refined < best_c5:
            best_c5, best_h = c5_refined, h_refined.copy()

    # ensure constraints are satisfied (tiny numerical adjustments)
    if best_h is not None:
        best_h = _project_constraints(best_h)
        best_c5 = _compute_c5(best_h)
        best_n = best_h.shape[0]

    # ----------------------------------------------------------------------
    # 5.  Return result
    # ----------------------------------------------------------------------
    return (best_h.tolist(), float(best_c5), int(best_n))
