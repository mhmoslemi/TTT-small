def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h); mode='full' follows the specification
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
