import time
import numpy as np
from scipy.optimize import differential_evolution

def _project_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] while forcing sum(v)=target."""
    lo = -np.max(v) - 1.0          # ensures all entries become 0 after clipping
    hi = 1.0 - np.min(v) + 1.0     # ensures all entries become 1 after clipping
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + hi, 0.0, 1.0)
    # tiny residual correction on non‑boundary entries
    resid = target - x.sum()
    if abs(resid) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(free.sum())
        if cnt:
            x[free] += resid / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _c5_value(h: np.ndarray) -> float:
    """Return the C₅ value for a feasible vector h."""
    n = h.size
    dx = 2.0 / n
    # length for linear (full) autocorrelation via FFT
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, n=N)
    autocorr = np.fft.ifft(H * np.conj(H)).real
    J = autocorr[: 2 * n - 1] * dx           # ∫ h(x)h(x+k)dx for each shift
    return 1.0 - J.min()                     # max_k ∫ h(1‑h_shift)dx


def _propose_move(h: np.ndarray, sigma: float) -> np.ndarray:
    """Return a new feasible vector obtained by transferring mass between two coordinates."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # feasible interval for transferred amount δ (i loses, j gains)
    delta_low = max(- (1.0 - h[i]), -h[j])
    delta_high = min(h[i], 1.0 - h[j])
    if delta_low >= delta_high:
        return h.copy()
    delta = np.random.normal(scale=sigma)
    delta = np.clip(delta, delta_low, delta_high)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function achieving a low Erdős‑C₅ bound.

    Returns
    -------
    h_values : np.ndarray
        The optimized function values on a uniform grid of n points.
    c5_bound : float
        The achieved C₅ value.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size and starting point
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 128                     # a modest size giving good expressive power
        h0 = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    h0 = _project_sum_bounds(h0, target_sum)

    best_h = h0.copy()
    best_val = _c5_value(best_h)

    # ------------------------------------------------------------------
    # Global search with Differential Evolution
    # ------------------------------------------------------------------
    bounds = [(0.0, 1.0)] * n

    def _de_obj(x):
        h = _project_sum_bounds(x, target_sum)
        return _c5_value(h)

    # Rough budget allocation for DE (≈30 % of total time)
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed - 5.0)   # keep a few seconds for final refinement
    de_budget = min(200.0, remaining * 0.3)          # seconds dedicated to DE
    # Estimate number of generations from a cheap evaluation cost guess
    eval_time_est = 4e-4            # ~0.4 ms per evaluation (conservative)
    popsize = 15
    maxgen = int(max(1, de_budget / (popsize * eval_time_est)))
    maxgen = min(maxgen, 5000)      # safeguard against excessively long runs

    result = differential_evolution(
        _de_obj,
        bounds,
        strategy="best1bin",
        maxiter=maxgen,
        popsize=popsize,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=False,
        disp=False,
        updating="deferred",
        workers=1,
    )
    h_de = _project_sum_bounds(result.x, target_sum)
    val_de = _c5_value(h_de)
    if val_de < best_val:
        best_val, best_h = val_de, h_de.copy()

    # ------------------------------------------------------------------
    # Local refinement – simple stochastic hill climbing
    # ------------------------------------------------------------------
    current_h = best_h.copy()
    sigma0 = 0.2

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:
            break
        frac = elapsed / budget_s
        sigma = sigma0 * (0.5 ** frac)          # decays to ≈0.001 near the end
        cand = _propose_move(current_h, sigma)
        cand_val = _c5_value(cand)

        if cand_val < best_val:
            best_val, best_h = cand_val, cand.copy()
            current_h = cand
        else:
            # occasional Metropolis acceptance to escape plateaus
            T = 0.01 * (0.5 ** frac) + 1e-12
            if np.random.rand() < np.exp(-(cand_val - best_val) / T):
                current_h = cand

        # sporadic full random restart with tiny probability
        if np.random.rand() < 0.001:
            current_h = _project_sum_bounds(np.random.rand(n), target_sum)

    best_h = _project_sum_bounds(best_h, target_sum)
    best_val = _c5_value(best_h)

    return best_h, float(best_val), n
