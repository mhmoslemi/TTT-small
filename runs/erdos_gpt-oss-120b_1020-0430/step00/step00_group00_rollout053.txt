def project_onto_simplex_box(x, target):

    hi = 1.0

        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Ensure sum close to target
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        # distribute diff to a non-bound spot
        # find indices not at bounds
        idx = np.where((y > 0) & (y < 1))[0]

            y[idx[0]] += diff
            y = np.clip(y, 0.0, 1.0)

                y[ind] = min(1.0, y[ind] + diff)

                ind = np.argmin(y)
                y[ind] = max(0.0, y[ind] + diff)

import numpy as np
import random
import time
import math

def compute_c5(vec):
    """Maximum overlap integral for a step function sampled on a uniform grid."""
    n = vec.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h); np.correlate does “full” convolution.
    return np.max(np.correlate(vec, 1 - vec, mode="full")) * dx


def project_onto_simplex_box(x, target):
    """Project x onto the feasible set:
       0 ≤ x_i ≤ 1  and  sum(x) = target."""
    lo, hi = -1.0, 1.0                     # safe bounds for the Lagrange multiplier
    for _ in range(50):                    # binary search for the correct shift λ
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        interior = np.where((y > 0) & (y < 1))[0]
        if interior.size:
            y[interior[0]] += diff
        else:
            if diff > 0:
                i = np.argmax(y)
                y[i] = min(1.0, y[i] + diff)
            else:
                i = np.argmin(y)
                y[i] = max(0.0, y[i] + diff)
        y = np.clip(y, 0.0, 1.0)
    return y


def subgradient_opt(start_vec, deadline):
    """Sub‑gradient descent with a simple simulated‑annealing acceptance rule."""
    n = start_vec.shape[0]
    target = n / 2.0
    cur = start_vec.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    it = 0

    while time.time() < deadline:
        conv = np.correlate(cur, 1 - cur, mode="full")
        maxv = conv.max()
        # all shifts attaining the maximum (within tolerance) form the active set
        tol = 1e-12
        shifts = np.where(np.abs(conv - maxv) <= tol)[0] - (n - 1)

        grad = np.zeros_like(cur)
        for k in shifts:
            if k >= 0:
                grad[: n - k] += 1 - cur[k:]
                grad[k:] -= cur[: n - k]
            else:
                kk = -k
                grad[kk:] += 1 - cur[: n - kk]
                grad[: n - kk] -= cur[kk:]

        step = 0.5 / math.sqrt(it + 1)          # diminishing step size
        new = cur - step * grad
        new = project_onto_simplex_box(new, target)
        new_val = compute_c5(new)

        # deterministic improvement
        if new_val < cur_val - 1e-12:
            cur, cur_val = new, new_val
            if new_val < best_val - 1e-12:
                best, best_val = new.copy(), new_val
        else:
            # occasional uphill move (simulated annealing)
            temp = max(0.01, step)
            if random.random() < math.exp(-(new_val - cur_val) / temp):
                cur, cur_val = new, new_val
        it += 1
    return best, best_val

    vec[idx] = 1.0
    if n % 2 == 1:
        # one extra 0.5 to hit the half‑integer target
        remaining = [i for i in range(n) if i not in idx]
        if remaining:
            vec[remaining[0]] = 0.5
    target = n / 2.0
    return project_onto_simplex_box(vec, target)


def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # -----------------------------------------------------------------
    # Initialise from the parent construction if it exists.
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 45
        h0 = np.zeros(n, dtype=float)
        half = n // 2
        h0[:half] = 1.0
        if n % 2 == 1:
            h0[half] = 0.5

    target_sum = n / 2.0
    if abs(h0.sum() - target_sum) > 1e-9:
        h0 = project_onto_simplex_box(h0, target_sum)

    best_vec = h0.copy()
    best_val = compute_c5(best_vec)

    # -----------------------------------------------------------------
    # Main optimisation loop – a few sub‑gradient runs with occasional
    # random restarts, respecting the overall time budget.
    # -----------------------------------------------------------------
    overall_deadline = start_time + budget_s - 5.0   # keep a small safety margin

    while time.time() < overall_deadline:
        remaining = overall_deadline - time.time()
        slice_time = min(15.0, remaining)            # 15 s per sub‑run
        slice_deadline = time.time() + slice_time

        # 30 % of the runs start from a fresh random feasible point.
        if random.random() < 0.3:
            start_vec = generate_random_feasible(n)
        else:
            start_vec = best_vec.copy()

        vec_opt, val_opt = subgradient_opt(start_vec, slice_deadline)

        if val_opt < best_val:
            best_vec, best_val = vec_opt.copy(), val_opt
            if best_val <= 0.3808:          # early exit if the target is reached
                break

    # -----------------------------------------------------------------
    # Final projection to guarantee feasibility and final C₅ evaluation.
    # -----------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1.0)
    best_vec = project_onto_simplex_box(best_vec, target_sum)
    final_val = compute_c5(best_vec)

    return (best_vec.tolist(), float(final_val), n)
