                break

            break

def compute_c5(arr):
    n = arr.shape[0]
    dx = 2.0 / n
    # size for FFT = next power of 2 of (2*n - 1)
    size = 1 << ((2*n - 1).bit_length())
    # FFT of arr and reversed 1-arr
    fa = np.fft.rfft(arr, size)
    fb = np.fft.rfft((1 - arr)[::-1], size)
    corr = np.fft.irfft(fa * fb, size)[:2*n-1]
    # due to floating errors, ensure real and non-negative
    max_corr = np.max(corr.real)
    return max_corr * dx

import numpy as np
import random
import math

def compute_c5(arr: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = arr.shape[0]
    dx = 2.0 / n
    # length for linear convolution = 2*n-1, round up to next power of two
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, size)
    fb = np.fft.rfft((1.0 - arr)[::-1], size)
    conv = np.fft.irfft(fa * fb, size)[: 2 * n - 1]
    return float(np.max(conv) * dx)


def adjust_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clamp to [0,1] and shift mass to hit the required sum."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # simple iterative redistribution
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            # move everything on a single free coordinate
            idx = np.random.randint(v.size)
            cap_up = 1.0 - v[idx]
            cap_dn = v[idx]
            if diff > 0:
                delta = min(diff, cap_up)
                v[idx] += delta
                diff -= delta
            else:
                delta = min(-diff, cap_dn)
                v[idx] -= delta
                diff += delta
            continue
        inc = diff / free.size
        v[free] += inc
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
    return v


def thue_morse(n: int) -> np.ndarray:

    seq = np.empty(n, dtype=float)
    for i in range(n):
        seq[i] = (bin(i).count("1") & 1)  # 0 or 1
    return seq

    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, arr)

            if new_c5 < best_c5 - 1e-9:
                best_c5 = new_c5
                best_h = h.copy()
                last_improve = iteration
                if best_c5 <= 0.3808:      # early exit on target
                    break

        iteration += 1

    # ----- Final repair and evaluation -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n)
