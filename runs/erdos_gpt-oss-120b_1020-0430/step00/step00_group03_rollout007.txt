def improve_solution(h_initial, time_limit):
    start = time.time()
    n = len(h_initial)
    dx = 2.0/n
    h = h_initial.copy()
    cur_max = max_corr_int(h)
    best_h = h.copy()
    best_val = cur_max
    # list of indices for 1 and 0
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    # Use simple hill climbing: if swap improves, keep.
    while time.time() - start < time_limit:
        # pick random 1 and 0
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        # compute new max
        new_max = max_corr_int(h)
        if new_max <= cur_max:
            # accept
            cur_max = new_max
            # update ones, zeros indices
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
            if new_max < best_val:
                best_val = new_max
                best_h = h.copy()
                # maybe print progress (not needed)

def try_milp_solver(n, target_M, time_limit=60.0):
    import cvxpy as cp, numpy as np, time
    # create model
    x = cp.Variable(n, boolean=True)
    total_ones = cp.sum(x)
    constraints = [total_ones == n // 2]

    # Build y variables and mapping
    y_vars = []
    pairs_per_shift = [] # list for each shift k (1-indexed) of indices into y_vars
    for k in range(1, n):
        indices = []
        for i in range(n - k):
            y = cp.Variable(boolean=True)
            y_vars.append(y)
            indices.append(len(y_vars) - 1)
        pairs_per_shift.append(indices)

    # Add linking constraints
    for k in range(1, n):
        for i in range(n - k):
            y_idx = pairs_per_shift[k-1][i]
            y = y_vars[y_idx]
            constraints.append(y <= x[i])
            constraints.append(y <= x[i + k])
            constraints.append(y >= x[i] + x[i + k] - 1)

    # Add shift constraints
    for k in range(1, n):
        prefix_len = n - k
        prefix_sum = cp.sum(x[:prefix_len])
        y_sum = cp.sum(cp.hstack([y_vars[idx] for idx in pairs_per_shift[k-1]]))
        constraints.append(prefix_sum - y_sum <= target_M)
        suffix_sum = total_ones - cp.sum(x[:k])
        constraints.append(suffix_sum - y_sum <= target_M)

    # Objective: maximize sum of y (or minimize 0)
    objective = cp.Maximize(cp.sum(cp.hstack(y_vars)))
    prob = cp.Problem(objective, constraints)

    # Solve with time limit
    # do error handling
    start = time.time()

import time
import numpy as np
import cvxpy as cp

def _max_corr_int(h):
    """Maximum overlap count for binary step function."""
    return int(np.max(np.correlate(h, 1 - h, mode="full")))

def _c5_from_h(h):
    """Return C5 bound (float) from vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h.astype(float), (1 - h).astype(float), mode="full")) * dx)

def _try_milp(n, target_M, time_budget):
    """Attempt to find a binary vector with max overlap ≤ target_M using MILP."""
    # Decision variables
    x = cp.Variable(n, boolean=True)                     # 0/1 values of h
    total_ones = cp.sum(x)
    constraints = [total_ones == n // 2]                # integral = 1

    # auxiliary variables y_{i,k} = x_i * x_{i+k}
    y_vars = []                                          # flat list of all y
    pairs_per_shift = []                                 # indices of y belonging to each k (1‑based)

    for k in range(1, n):
        idxs = []
        for i in range(n - k):
            y = cp.Variable(boolean=True)
            y_vars.append(y)
            idxs.append(len(y_vars) - 1)
        pairs_per_shift.append(idxs)

    # linking constraints y_{i,k} ≤ x_i, y_{i,k} ≤ x_{i+k},
    #               y_{i,k} ≥ x_i + x_{i+k} - 1
    for k in range(1, n):
        for i in range(n - k):
            y = y_vars[pairs_per_shift[k - 1][i]]
            constraints += [
                y <= x[i],
                y <= x[i + k],
                y >= x[i] + x[i + k] - 1,
            ]

    # shift‑wise overlap constraints:
    #   sum_{i=0}^{n-1-k} x_i - sum y_{i,k} ≤ M   (positive shift)
    #   sum_{i=k}^{n-1}   x_i - sum y_{i,k} ≤ M   (negative shift)
    for k in range(1, n):
        prefix_len = n - k
        prefix_sum = cp.sum(x[:prefix_len])
        y_sum = cp.sum(cp.hstack([y_vars[idx] for idx in pairs_per_shift[k - 1]]))
        constraints.append(prefix_sum - y_sum <= target_M)

        suffix_sum = total_ones - cp.sum(x[:k])                # sum of x_i for i ≥ k
        constraints.append(suffix_sum - y_sum <= target_M)

    # maximise total number of coincident pairs (helps the solver)
    objective = cp.Maximize(cp.sum(cp.hstack(y_vars)))
    prob = cp.Problem(objective, constraints)

    # attempt solve (no explicit time limit – rely on the budget we give the call)
    start = time.time()
    try:
        prob.solve(solver="GLPK_MI", verbose=False)
    except Exception:
        try:
            prob.solve(solver="CBC", verbose=False)
        except Exception:
            # last resort
            prob.solve(solver="ECOS_BB", verbose=False, max_iters=500000)

    if prob.status not in {"optimal", "optimal_inaccurate", "feasible"}:
        return None, None

    xv = np.rint(x.value).astype(int)
    M = _max_corr_int(xv)
    if M > target_M:
        # solution violates the bound (possible due to numeric tolerance), reject
        return None, None
    return xv, M


def _local_search_binary(n, target_M, time_budget, rng):
    """Simple simulated‑annealing hill‑climber on binary vectors."""
    # start from a random balanced vector
    h = np.zeros(n, dtype=np.int8)
    h[: n // 2] = 1
    rng.shuffle(h)
    cur_val = _max_corr_int(h)
    best_h = h.copy()
    best_val = cur_val

    # keep explicit lists of positions for fast random swaps
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    start = time.time()
    # linear cooling schedule
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        temp = max(0.01, 5.0 * (1.0 - frac))

        i = rng.choice(ones)
        j = rng.choice(zeros)

        # try swapping
        h[i] = 0
        h[j] = 1
        new_val = _max_corr_int(h)
        delta = new_val - cur_val

        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / temp):
                accept = True

        if accept:
            cur_val = new_val
            # update the index lists
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)

            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val <= target_M:
                    break
        else:
            # revert
            h[i] = 1
            h[j] = 0

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with integral 1 that minimises the
    Erdős overlap constant C₅.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : numpy.ndarray of shape (n_points,)
        c5_bound : float, the achieved C₅
        n_points : int, number of samples used
    """
    rng = np.random.default_rng(seed)

    # Determine the problem size
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
    else:
        n = 70
        init = None

    dx = 2.0 / n
    # target_M is the integer count that corresponds to the 0.3808 bound
    target_M = int(np.floor(0.3808 / dx))

    best_h = None
    best_c5 = np.inf

    # evaluate the provided construction (if any)
    if init is not None:
        val = _c5_from_h(init)
        if val < best_c5:
            best_c5 = val
            best_h = init.copy()

    start = time.time()
    remaining = budget_s

    # ---- MILP phase -------------------------------------------------
    milp_time = min(60.0, remaining * 0.5)          # give it up to half the budget, capped at 60 s
    milp_h, milp_M = _try_milp(n, target_M, milp_time)
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)

    if milp_h is not None:
        cand_c5 = _c5_from_h(milp_h)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = milp_h.astype(float)

    # ---- Local‑search phase -----------------------------------------
    if remaining > 5.0:   # keep a small safety margin
        ls_h, ls_M = _local_search_binary(n, target_M, remaining, rng)
        cand_c5 = _c5_from_h(ls_h)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = ls_h.astype(float)

    # fallback (should never be needed)
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c5 = _c5_from_h(best_h)

    # final sanity checks (enforce constraints)
    best_h = np.clip(best_h, 0.0, 1.0)
    total = best_h.sum() * dx
    # slight numerical drift is corrected
    if abs(total - 1.0) > 1e-9:
        # project back to the required simplex (simple scaling, then renormalise)
        best_h = best_h * (1.0 / total)

    return best_h, best_c5, n
