def slsqp_opt(x0, maxiter=200):
    n = len(x0)

        corr = np.correlate(x, 1 - x, mode='full')

    res = minimize(obj, x0, method='SLSQP', bounds=bounds,

import numpy as np
import time
from scipy.optimize import minimize

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------------
    def eval_h(x):
        """Return the C5 value for a given vector x."""
        n = len(x)
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * (2.0 / n)

    def project_to_sum(x, target):
        """Adjust a feasible vector x so that sum(x)=target while staying in [0,1]."""
        diff = target - np.sum(x)
        if abs(diff) < 1e-12:
            return x
        n = len(x)
        # First try to fix using a single slack entry
        for i in range(n):
            if diff > 0 and x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
        # If still not perfect, distribute over all non‑bound entries
        if abs(diff) > 1e-12:
            free = np.where((x > 0.0) & (x < 1.0))[0]
            if free.size > 0:
                delta = diff / free.size
                for i in free:
                    if delta > 0:
                        inc = min(delta, 1.0 - x[i])
                        x[i] += inc
                        diff -= inc
                    else:
                        dec = min(-delta, x[i])
                        x[i] -= dec
                        diff += dec
        # Final tiny correction if needed
        if abs(diff) > 1e-12:
            for i in range(n):
                if 0.0 <= x[i] + diff <= 1.0:
                    x[i] += diff
                    break
        return x

    def random_feasible(n):
        """Generate a random feasible vector (bounds [0,1], sum=n/2)."""
        x = np.full(n, 0.5)
        # perform many random exchange moves
        for _ in range(200):
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = np.random.uniform(-0.3, 0.3)
            xi, xj = x[i] + delta, x[j] - delta
            if 0.0 <= xi <= 1.0 and 0.0 <= xj <= 1.0:
                x[i], x[j] = xi, xj
        target = n / 2.0
        x = project_to_sum(x, target)
        return x

    def slsqp_opt(x0, maxiter=200):
        """Run a short SLSQP optimisation from x0."""
        n = len(x0)
        target = n / 2.0

        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            return np.max(corr) * (2.0 / n)

        cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options={"maxiter": maxiter, "ftol": 1e-10, "disp": False},
        )
        if not res.success:
            return x0
        x = np.clip(res.x, 0.0, 1.0)
        x = project_to_sum(x, target)
        return x

    def local_refine(x, max_iters=2000):
        """Simple pairwise exchange hill‑climb preserving the sum."""
        n = len(x)
        best = x.copy()
        best_val = eval_h(best)
        target = n / 2.0
        for _ in range(max_iters):
            if time.time() - start_time > budget_s - 0.5:
                break
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # maximum admissible delta for a swap i<-+->j
            max_inc = min(1.0 - best[i], best[j])
            max_dec = min(best[i], 1.0 - best[j])
            if max_inc <= 0 and max_dec <= 0:
                continue
            if np.random.rand() < 0.5 and max_inc > 0:
                delta = np.random.uniform(0.0, max_inc)
                new_i, new_j = best[i] + delta, best[j] - delta
            elif max_dec > 0:
                delta = np.random.uniform(0.0, max_dec)
                new_i, new_j = best[i] - delta, best[j] + delta
            else:
                continue
            cand = best.copy()
            cand[i], cand[j] = new_i, new_j
            val = eval_h(cand)
            if val + 1e-12 < best_val:
                best, best_val = cand, val
        # final projection (tiny rounding errors)
        best = project_to_sum(best, target)
        return best, eval_h(best)

        candidates = [init_vec]
        best_h = init_vec.copy()

    # Main optimisation loop
    # ------------------------------------------------------------------

        else:

        if best_h is not None and len(candidates) < 4 and time.time() - start_time < budget_s - 5.0:

            i, j = np.random.randint(n_points, size=2)
            delta = np.random.uniform(-0.2, 0.2)
            if 0.0 <= pert[i] + delta <= 1.0 and 0.0 <= pert[j] - delta <= 1.0:

    best_h = project_to_sum(best_h, n_points / 2.0)
