import time, numpy as np, cvxpy as cp

def evaluate_h(h):
    # if evaluate_erdos_solution available, use; else compute ourselves
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        # compute C via correlation
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

def solve_milp(n, time_limit):
    a = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(a) == n / 2]
    for ell in range(1, n):
        m = n - ell
        a_i = a[:m]
        a_i_shift = a[ell:]
        b = cp.Variable(m, boolean=True)
        constraints += [b <= a_i,
                        b <= a_i_shift,
                        b >= a_i + a_i_shift - 1]
        Cpos = cp.sum(a_i) - cp.sum(b)
        Cneg = cp.sum(a_i_shift) - cp.sum(b)
        constraints += [Cpos <= t,
                        Cneg <= t]
    # optional: t >= 0
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    # solve with time limit
    try:
        prob.solve(solver=cp.CBC, warm_start=True, time_limit=time_limit)
    except Exception as e:
        # fallback
        prob.solve(solver=cp.CBC, warm_start=True)
    status = prob.status
    if status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE):
        return None, None
    # get solution
    a_val = np.round(a.value).astype(float)  # ensure exact binary
    t_val = float(t.value)
    return a_val, t_val

import numpy as np
import cvxpy as cp

def _evaluate_h(h):
    """Return the C5 bound for a step function h."""
    try:
        # Provided by the harness
        return evaluate_erdos_solution(h)
    except Exception:
        # Fallback: compute correlation directly
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx


def _solve_binary_milp(n, time_limit):
    """
    Solve the binary formulation:
        minimize t
        subject to
            sum a_i = n/2
            for each shift ell (1..n-1):
                Cpos_ell = sum_{i< n-ell} a_i - sum a_i a_{i+ell} <= t
                Cneg_ell = sum_{i< n-ell} a_{i+ell} - sum a_i a_{i+ell} <= t
            a_i ∈ {0,1}
    Returns (a_vals, t_val) or (None, None) on failure.
    """
    a = cp.Variable(n, boolean=True)
    t = cp.Variable(nonneg=True)

    constraints = [cp.sum(a) == n / 2]

    for ell in range(1, n):
        m = n - ell
        a_i = a[:m]               # a[0] … a[m-1]
        a_ip = a[ell:]            # a[ell] … a[n-1]
        # product variables b_i_ell = a_i * a_{i+ell}
        b = cp.Variable(m, boolean=True)

        # linearization of b = a_i * a_{i+ell}
        constraints += [
            b <= a_i,
            b <= a_ip,
            b >= a_i + a_ip - 1,
        ]

        Cpos = cp.sum(a_i) - cp.sum(b)
        Cneg = cp.sum(a_ip) - cp.sum(b)

        constraints += [Cpos <= t, Cneg <= t]

    prob = cp.Problem(cp.Minimize(t), constraints)

    # solver options – CBC accepts a time limit via the “seconds” parameter
    try:
        prob.solve(solver=cp.CBC, seconds=time_limit, verbose=False)
    except Exception:
        # If the solver does not understand the option, fall back without limit
        prob.solve(solver=cp.CBC, verbose=False)

    if prob.status not in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE}:
        return None, None

    a_val = np.rint(a.value).astype(float)   # force exact 0/1
    t_val = float(t.value)
    return a_val, t_val
