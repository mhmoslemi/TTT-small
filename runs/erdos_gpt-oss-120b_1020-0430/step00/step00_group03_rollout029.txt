import numpy as np
import time

def run(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """Optimize a step function h on [0,2] for the Erdős C5 constant.

    Returns
    -------
    h_values : list of float
        Optimized discretised function values (length n_points).
    c5_bound : float
        The achieved upper bound C5 = max_k ∫ h(x)(1-h(x+k))dx.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------ #
    # 1. Initial vector ------------------------------------------------ #
    try:
        # `initial_h_values` is provided by the harness (if any)
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.size
    except Exception:
        # No parent: start from a binary half‑filled vector (size modest)
        n = 200                     # < 1000 as requested
        h = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    # 2. Enforce the linear constraints -------------------------------- #
    target_sum = n / 2.0            # because ∫h = 1 and dx = 2/n
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # scale then clip, finally fix any residual mass
        h *= target_sum / cur_sum
        np.clip(h, 0.0, 1.0, out=h)
        residual = target_sum - h.sum()
        if abs(residual) > 1e-12:
            # distribute the tiny residual greedily
            for i in np.random.permutation(n):
                if residual == 0:
                    break
                if residual > 0:
                    add = min(residual, 1.0 - h[i])
                    h[i] += add
                    residual -= add
                else:
                    sub = min(-residual, h[i])
                    h[i] -= sub
                    residual += sub

    dx = 2.0 / n
    eps = 1e-12

    # ------------------------------------------------------------------ #
    # 3. Helper: full autocorrelation using FFT (linear, no wrap‑around)
    def autocorr_full(v):
        m = 2 * n - 1
        F = np.fft.fft(v, n=m)
        c = np.fft.ifft(F * np.conj(F)).real
        return c

    # 4. Initial objective ------------------------------------------------
    conv = autocorr_full(h)
    min_corr = conv.min() * dx                     # min_k ∫h·h(x+k)
    best_min_corr = min_corr
    best_h = h.copy()
    best_c5 = 1.0 - best_min_corr

    # ------------------------------------------------------------------ #
    # 5. Optimisation loop (simple stochastic ascent on min_corr)
    it = 0
    no_improve = 0
    # temperature schedule for occasional uphill moves
    init_temp = 5e-4
    while time.time() - start_time < budget_s * 0.94:
        # identify the worst (minimum) shift
        idx_min = conv.argmin()
        shift = idx_min - (n - 1)                     # lag in [-n+1, n-1]

        # indexes that contribute to this lag (i and i+shift are both valid)
        if shift >= 0:
            i_start, i_end = 0, n - shift - 1
        else:
            i_start, i_end = -shift, n - 1

        # Candidates where BOTH points can be raised (still < 1)
        hi = h[i_start:i_end + 1]
        hj = h[i_start + shift:i_end + shift + 1]
        cand_mask = (hi < 1.0 - eps) & (hj < 1.0 - eps)
        cand_inds = np.where(cand_mask)[0]

        if cand_inds.size == 0:
            # fallback: simple transfer between two random positions
            i, j = np.random.choice(n, 2, replace=False)
            if h[i] <= eps or h[j] >= 1.0 - eps:
                continue
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= eps:
                continue
            delta = max_delta * np.random.rand()
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # pick a target pair (i, i+shift) to increase
            ii = i_start + np.random.choice(cand_inds)
            jj = ii + shift

            # pick two source indices (different from target) that can lose mass
            src_candidates = np.where(h > eps)[0]
            src_candidates = src_candidates[(src_candidates != ii) & (src_candidates != jj)]
            if src_candidates.size < 2:
                continue
            s1, s2 = np.random.choice(src_candidates, size=2, replace=False)

            # compute admissible delta
            max_inc = min(1.0 - h[ii], 1.0 - h[jj])
            max_dec = min(h[s1], h[s2])
            max_delta = min(max_inc, max_dec)
            if max_delta <= eps:
                continue
            delta = max_delta * np.random.rand()
            if delta <= eps:
                continue

            h_new = h.copy()
            h_new[ii] += delta
            h_new[jj] += delta
            h_new[s1] -= delta
            h_new[s2] -= delta

        # evaluate candidate
        conv_new = autocorr_full(h_new)
        min_corr_new = conv_new.min() * dx

        # acceptance rule (strict ascent with occasional uphill)
        improve = min_corr_new > best_min_corr + 1e-10
        if improve:
            h = h_new
            conv = conv_new
            best_min_corr = min_corr_new
            best_h = h_new.copy()
            best_c5 = 1.0 - best_min_corr
            no_improve = 0
        else:
            # simulated‑annealing style uphill acceptance
            elapsed = time.time() - start_time
            temp = init_temp * (1.0 - elapsed / (budget_s * 0.94))
            if temp < 0:
                temp = 0
            prob = np.exp((min_corr_new - best_min_corr) / (temp + eps))
            if np.random.rand() < prob:
                h = h_new
                conv = conv_new
            else:
                no_improve += 1

        # occasional random perturbation when stuck
        if no_improve > 8000:
            # disturb a few entries slightly
            for _ in range(3):
                a, b = np.random.choice(n, 2, replace=False)
                if h[a] <= eps or h[b] >= 1.0 - eps:
                    continue
                max_delta = min(h[a], 1.0 - h[b])
                delta = max_delta * np.random.rand() * 0.5
                h[a] -= delta
                h[b] += delta
            conv = autocorr_full(h)
            min_corr = conv.min() * dx
            no_improve = 0

        it += 1

    # ------------------------------------------------------------------ #
    # 6. Final evaluation of the best vector found
    final_c5 = 1.0 - best_min_corr
    # Convert to plain Python list for the harness
    h_values = best_h.tolist()
    return (h_values, float(final_c5), n)
