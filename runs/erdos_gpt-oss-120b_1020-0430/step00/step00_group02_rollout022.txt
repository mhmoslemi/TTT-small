def hill_climb(h, steps, temp_start=0.01, temp_end=1e-5):
    n = len(h)
    best_local_h = h.copy()
    best_local_c5 = compute_c5(best_local_h)
    cur_h = h.copy()
    cur_c5 = best_local_c5
    for t in range(steps):
        # temperature schedule
        T = temp_start + (temp_end - temp_start) * (t / steps)

        # pick a random 1-index and 0-index
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(cur_h)
        # accept if better or with prob
        if new_c5 < cur_c5 or np.random.rand() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            cur_c5 = new_c5
            # maybe update best
            if cur_c5 < best_local_c5:
                best_local_c5 = cur_c5
                best_local_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # break early if time close to budget
        if time.time() - start_time > budget_s * 0.99:
            break
    return best_local_h, best_local_c5

import time
import random
import math
import numpy as np

# ---------- fast evaluation (FFT) ----------
def _fast_c5(h: np.ndarray) -> float:
    """Return C5 using FFT based correlation (fast, slight rounding)."""
    n = h.size
    dx = 2.0 / n
    g = 1.0 - h
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(g, L)
    conv = np.fft.irfft(H * G, L)[: 2 * n - 1]
    return float(conv.max() * dx)


# ---------- precise evaluation (np.correlate) ----------
def _precise_c5(h: np.ndarray) -> float:
    """Exact C5 using the same expression the grader uses."""
    n = h.size
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)


# ---------- binary hill climbing + fractional refinement ----------
def _optimize_one(
    h0: np.ndarray,
    max_swaps: int,
    max_frac: int,
    start_time: float,
    budget_s: float,
) -> tuple[np.ndarray, float]:
    """Local optimisation starting from h0.
    Returns the best vector seen and its *fast* C5 value.
    """
    n = h0.size
    cur = h0.astype(float).copy()
    cur_val = _fast_c5(cur)
    best = cur.copy()
    best_val = cur_val

    # ----- binary swaps (simulated annealing) -----
    T_start, T_end = 2e-2, 1e-6
    for step in range(max_swaps):
        if time.time() - start_time > budget_s * 0.99:
            break
        T = T_start + (T_end - T_start) * (step / max_swaps)

        # choose a 1 and a 0
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_val = _fast_c5(cur)
        improve = new_val < cur_val
        if improve or (math.exp(-(new_val - cur_val) / max(T, 1e-12)) > np.random.rand()):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]

    # ----- fractional moves (greedy) -----
    for step in range(max_frac):
        if time.time() - start_time > budget_s * 0.99:
            break
        donors = np.where(cur > 1e-8)[0]
        receivers = np.where(cur < 1.0 - 1e-8)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = np.random.choice(donors)
        j = np.random.choice(receivers)
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        # take a random fraction of the allowed transfer
        delta = max_delta * np.random.rand() * 0.5
        cur[i] -= delta
        cur[j] += delta
        new_val = _fast_c5(cur)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

    return best, best_val

    cand_lengths = sorted(set(cand_lengths))
    random.shuffle(cand_lengths)

    # 3) optimisation loop
    while time.time() - start < budget_s * 0.95:
        for n in cand_lengths:
            if time.time() - start >= budget_s * 0.95:
                break

            # step budgets (larger n → more steps)
            max_swaps = max(5000, n * 20)
            max_frac = max(2000, n * 5)

            # initialise a start vector
            if init is not None and up.size == n:
                h0 = up.copy()                     # may contain fractions; ok
            else:
                h0 = np.zeros(n, dtype=float)
                h0[np.random.choice(n, n // 2, replace=False)] = 1.0

            # optimise locally
            h_opt, fast_val = _optimize_one(h0, max_swaps, max_frac, start, budget_s)

            # precise value for comparison
            precise_val = _precise_c5(h_opt)

            if precise_val < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), float(precise_val), n
                # early stop if we beat the target record
                if best_c5 < 0.3808:
                    break

        # break outer loop if target reached
        if best_c5 < 0.3808:
            break

    # safety fallback (should never be needed)
    if best_h is None:
        n_fallback = 100
        best_h = np.full(n_fallback, 0.5, dtype=float)
        best_c5 = _precise_c5(best_h)
        best_n = n_fallback

    # final precise evaluation (just in case rounding drifted)
    best_c5 = _precise_c5(best_h)
    best_n = int(best_h.shape[0])

    return best_h, best_c5, best_n
