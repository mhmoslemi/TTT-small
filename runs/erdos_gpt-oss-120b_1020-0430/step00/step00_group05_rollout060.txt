import numpy as np
import time
import math

def _evaluate(vec, dx):
    # max correlation of h with (1-h)
    return float(np.correlate(vec, 1 - vec, mode="full").max() * dx)

def run(seed=42, budget_s=1100, **kwargs):
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.95          # leave a small safety margin

    # --------------------------------------------------------------
    # Build an initial binary indicator of the required length.
    # If a parent construction is supplied we reuse its size.
    # --------------------------------------------------------------
    try:
        # parent vector may be fractional; we binarise it
        init = np.array(initial_h_values, dtype=float)
        n = len(init)
        h = (init > 0.5).astype(int)      # simple threshold to 0/1
        target = n // 2                    # we need exactly n/2 ones (n is even in practice)
        diff = target - h.sum()
        if diff != 0:
            zeros = np.flatnonzero(h == 0)
            ones  = np.flatnonzero(h == 1)
            if diff > 0:
                h[np.random.choice(zeros, diff, replace=False)] = 1
            else:
                h[np.random.choice(ones, -diff, replace=False)] = 0
    except Exception:
        # fallback: 70 points, half of them are 1
        n = 70
        target = n // 2
        h = np.zeros(n, dtype=int)
        h[np.random.choice(n, target, replace=False)] = 1

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Evaluation helper (uses the same formula as the harness)
    # ------------------------------------------------------------------
    def evaluate(vec):
        return _evaluate(vec, dx)

    # ------------------------------------------------------------------
    # Keep the best construction found
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = evaluate(best_h)

    # ------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing on the space of binary vectors
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_score = best_score
    ones_idx = np.flatnonzero(cur_h == 1)
    zeros_idx = np.flatnonzero(cur_h == 0)

    max_iters = 5_000_000                # generous upper bound
    T0 = 0.05                             # initial temperature
    no_improve = 0
    restart_limit = 300_000               # restart after many non‑improving steps

    it = 0
    while time.time() - start_time < time_limit and it < max_iters:
        # pick a random 1 and a random 0 and propose swapping them
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        # perform the swap
        cur_h[i] = 0
        cur_h[j] = 1

        new_score = evaluate(cur_h)

        accept = False

        else:

            zeros_idx = np.append(zeros_idx, i)

            zeros_idx = zeros_idx[zeros_idx != j]
            ones_idx = np.append(ones_idx, j)

            # record a new global best
            if new_score < best_score:
                best_score = new_score

                no_improve = 0
            else:
