import time, math, random
import numpy as np

# ---------- Helper functions (must be top level) ----------
def _fft_len(n):
    """Next power of two >= 2*n-1 for linear convolution."""
    return 1 << ((2 * n - 1).bit_length())


def max_corr_fft(h):
    """Maximum correlation of h with (1-h) using FFT (full mode)."""
    n = h.shape[0]
    N = _fft_len(n)
    H = np.fft.rfft(h, n=N)
    B = np.fft.rfft((1.0 - h)[::-1], n=N)
    conv = np.fft.irfft(H * B, n=N)
    return np.max(conv[: 2 * n - 1])


def compute_c5(h):
    """Exact C5 value using np.correlate (as in the evaluator)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return float(corr * dx)


def compute_gradient(h, lag):
    """
    Sub‑gradient of the worst‑case correlation term.
    lag is the integer shift (negative allowed) that attains the maximum.
    """
    n = h.shape[0]
    grad = np.ones(n, dtype=np.float64)
    if lag == 0:
        grad -= 2.0 * h
        return grad
    if lag > 0:
        s = lag
        grad[s:] -= h[:-s]          # term (1 - h_{k-s})
        grad[: n - s] -= h[s:]      # term -h_{k+s}
    else:
        s = -lag
        grad[: n - s] -= h[s:]      # term (1 - h_{k+s})
        grad[s:] -= h[: n - s]      # term -h_{k-s}
    return grad


def project_to_constraints(x, target_sum):
    """
    Projects vector x onto the feasible set:
      0 <= x_i <= 1  and  sum(x) == target_sum.
    Simple alternating projection (box then hyperplane) works well for modest sizes.
    """
    n = x.shape[0]
    x = np.clip(x, 0.0, 1.0)
    for _ in range(15):
        diff = target_sum - x.sum()
        if abs(diff) < 1e-12:
            break
        # distribute the remaining mass uniformly
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    # final tiny fix if needed
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(x < 1.0)
            inc = min(diff, 1.0 - x[idx])
            x[idx] += inc
        else:
            idx = np.argmax(x > 0.0)
            dec = max(diff, -x[idx])
            x[idx] += dec
    return x


def initial_triangular(n, target_sum):
    """Triangular (mountain) shape, scaled to the required total mass."""
    r = np.arange(n, dtype=np.float64) / (n - 1)
    h = 2.0 * np.minimum(r, 1.0 - r)          # 0…1…0
    scale = target_sum / h.sum()
    h *= scale
    return project_to_constraints(h, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function for the Erdős C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    # ---- problem size -------------------------------------------------
    # Prefer a moderate size (fast FFT, enough freedom). 256 is a power of two.
    n = 256
    # If a parent construction exists and is larger, honour its size.
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, copy=False)
        if parent.size > n:
            n = int(parent.size // 2 * 2)      # keep it even
    dx = 2.0 / n
    target_sum = n / 2.0

    # ---- start point --------------------------------------------------
    # Triangular shape is a good deterministic start.
    h = initial_triangular(n, target_sum)

    # Include the parent solution as a candidate if it has the same length.
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, copy=False)
        if parent.size == n:
            pc5 = compute_c5(parent)
            if pc5 < best_c5:
                best_h = parent.copy()
                best_c5 = pc5

    # ---- parameters ----------------------------------------------------
    start_time = time.time()
    # fraction of budget for gradient phase
    grad_budget = budget_s * 0.55
    # fraction for stochastic hill‑climbing phase
    sa_budget = budget_s - grad_budget

    # Gradient descent (sub‑gradient of the worst lag)
    max_grad_iters = 120000
    step0 = 0.03   # initial step size

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ---- gradient phase ------------------------------------------------
    it = 0
    while it < max_grad_iters and (time.time() - start_time) < grad_budget:
        it += 1
        # correlation to locate worst lag
        N = _fft_len(n)
        H = np.fft.rfft(cur_h, n=N)
        B = np.fft.rfft((1.0 - cur_h)[::-1], n=N)
        conv = np.fft.irfft(H * B, n=N)[: 2 * n - 1]
        lag = int(np.argmax(conv) - (n - 1))

        # sub‑gradient
        grad = compute_gradient(cur_h, lag)

        # decreasing step
        step = step0 * (1.0 - it / max_grad_iters)

        # tentative move
        new_h = cur_h - step * grad
        new_h = project_to_constraints(new_h, target_sum)

        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h.copy()

    # ---- stochastic hill‑climbing (simulated annealing) ---------------
    # Parameters
    max_sa_iters = 300000
    T0 = 0.06          # initial temperature
    max_transfer = 0.12   # maximal mass moved in one jump

    it = 0
    while (time.time() - start_time) < budget_s and it < max_sa_iters:
        it += 1
        # linear temperature schedule
        T = T0 * (1.0 - it / max_sa_iters)

        # propose a transfer of mass from i -> j
        # i must have some mass, j must have room
        for _ in range(5):
            i = random.randrange(n)
            if cur_h[i] <= 1e-12:
                continue
            j = random.randrange(n)
            if i == j or cur_h[j] >= 1.0 - 1e-12:
                continue
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0.0:
                continue
            delta = max_transfer * max_move * random.random()
            if delta <= 0.0:
                continue
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = project_to_constraints(cand, target_sum)
            cand_c5 = compute_c5(cand)
            break
        else:
            continue

        d = cand_c5 - cur_c5
        if d < 0.0 or random.random() < math.exp(-d / (T + 1e-12)):
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_c5, best_h = cand_c5, cand.copy()

    # ---- final evaluation (exact) ------------------------------------
    final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), n
