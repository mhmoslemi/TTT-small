import numpy as np, time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def project_onto_constraints(h):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target_sum = n/2.0
    s = h.sum()
    if s == 0:
        # set random half
        idx = np.random.choice(n, int(target_sum), replace=False)
        h[:] = 0
        h[idx] = 1.0
    else:
        # Scale to match sum, then reclip iteratively
        h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
        # If still off, adjust by adding small constant to all, iterating
        # Use simple iterative correction
        for _ in range(5):
            s = h.sum()
            diff = target_sum - s
            if abs(diff) < 1e-12:
                break
            # distribute diff equally where possible
            # add diff/n to each entry within bounds
            inc = diff / n
            h = np.clip(h + inc, 0.0, 1.0)
        # final scaling as fallback
        s = h.sum()
        if abs(s - target_sum) > 1e-8:
            h = h * (target_sum / s)
            h = np.clip(h, 0.0, 1.0)
    return h

def optimize_continuous(start_h, max_iter=2000):
    n = len(start_h)
    dx = 2.0 / n
    bounds = [(0.0, 1.0) for _ in range(n)]

    def obj(h):
        # objective: max overlap
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def cons_eq(h):
        return np.sum(h) * dx - 1.0

    constraints = ({'type': 'eq', 'fun': cons_eq},)
    res = minimize(obj, start_h, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    if not res.success:
        # return original
        return start_h
    h_opt = res.x
    # Clip and project to ensure sum constraint
    h_opt = project_onto_constraints(h_opt)
    return h_opt

def local_continuous_search(h, best_c5, time_limit):
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    while time.time() - start < time_limit:
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = h[i], h[j]
        # compute allowed transfer
        d_max = min(1.0 - hi, hj)
        d_min = -min(hi, 1.0 - hj)
        if d_max <= d_min:
            continue
        d = np.random.uniform(d_min, d_max)
        if abs(d) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] = hi + d
        h_new[j] = hj - d
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-9:
            h = h_new
            best_c5 = c5
    return h, best_c5

def hill_climb_binary(n, time_limit):
    # binary search with swaps
    start_total = time.time()
    target_ones = n // 2
    best_h = None
    best_c5 = float('inf')
    # generate some random starts
    attempts = max(5, int(time_limit // 2))  # limited
    for _ in range(attempts):
        if time.time() - start_total > time_limit:
            break
        # random binary with exactly half ones
        idx = np.random.permutation(n)[:target_ones]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
        c5 = compute_c5(h)
        # hill climbing using swaps
        no_improve = 0
        while time.time() - start_total < time_limit and no_improve < 2000:
            # choose a 1 and a 0 to swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            # perform swap
            h[i], h[j] = 0, 1
            new_c5 = compute_c5(h)
            if new_c5 < c5 - 1e-9:
                c5 = new_c5
                no_improve = 0
            else:
                # revert
                h[i], h[j] = 1, 0
                no_improve += 1
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap integral C5 for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear (zero‑padded) correlation of h with 1‑h
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_onto_constraints(h: np.ndarray) -> np.ndarray:
    """
    Clip to [0,1] and enforce the integral constraint sum(h)*dx = 1
    (i.e. sum(h) = n/2).
    """
    h = np.clip(h, 0.0, 1.0)
    n = h.shape[0]
    target = n / 2.0
    s = h.sum()
    if s == 0.0:
        # degenerate: put exactly target ones
        idx = np.random.choice(n, int(target), replace=False)
        h[:] = 0.0
        h[idx] = 1.0
        return h
    # simple scaling, then re‑clip and repeat a few times
    for _ in range(5):
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    # final tiny correction if needed
    if abs(s - target) > 1e-8:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h


def optimize_continuous(start_h: np.ndarray, max_iter: int = 2000) -> np.ndarray:
    """Run a short SLSQP optimisation starting from start_h."""
    n = start_h.shape[0]
    dx = 2.0 / n
    bounds = [(0.0, 1.0) for _ in range(n)]

    def obj(h):
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    def eq_cons(h):
        return np.sum(h) * dx - 1.0

    cons = ({'type': 'eq', 'fun': eq_cons},)
    res = minimize(
        obj,
        start_h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        return start_h
    return project_onto_constraints(res.x)


def local_continuous_search(
    h: np.ndarray, current_best: float, time_limit: float
) -> tuple[np.ndarray, float]:
    """Pairwise mass‑transfer hill climbing on a feasible h."""
    start = time.time()
    n = h.shape[0]
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = h[i], h[j]
        # maximum feasible transfer from i to j (keeping 0‑1 bounds)
        d_max = min(1.0 - hi, hj)
        d_min = -min(hi, 1.0 - hj)
        if d_max <= d_min:
            continue
        d = np.random.uniform(d_min, d_max)
        if abs(d) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] = hi + d
        h_new[j] = hj - d
        c5 = compute_c5(h_new)
        if c5 < current_best - 1e-12:
            h = h_new
            current_best = c5
    return h, current_best


def hill_climb_binary(n: int, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simple binary hill‑climbing: keep exactly n/2 ones,
    repeatedly swap a 1 and a 0 if it does not increase C5.
    """
    target = n // 2
    best_h = None
    best_c5 = float("inf")
    start_time = time.time()
    # number of independent restarts
    restarts = max(5, int(time_limit // 4))
    for _ in range(restarts):
        # random balanced binary vector
        idx = np.random.permutation(n)[:target]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
        c5 = compute_c5(h)
        stall = 0
        while time.time() - start_time < time_limit and stall < 3000:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h[i], h[j] = 0.0, 1.0
            new_c5 = compute_c5(h)
            if new_c5 < c5 - 1e-12:
                c5 = new_c5
                stall = 0
            else:
                h[i], h[j] = 1.0, 0.0
                stall += 1
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
