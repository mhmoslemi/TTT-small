import time
import numpy as np
import random

def compute_c5(arr):
    """Return the C5 value for a step function given by arr."""
    n = arr.size
    dx = 2.0 / n
    # correlation with zero padding (mode='full')
    return np.max(np.correlate(arr, 1.0 - arr, mode='full')) * dx

def smooth_shift(arr, k):
    """Average neighbouring entries according to shift k (preserves total sum)."""
    n = arr.size
    if k == 0:
        return arr
    new = arr.copy()
    if k > 0:
        i = slice(0, n - k)
        j = slice(k, n)
    else:  # k < 0
        k = -k
        i = slice(k, n)
        j = slice(0, n - k)
    avg = (arr[i] + arr[j]) * 0.5
    new[i] = avg
    new[j] = avg
    return new

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with small C5."""
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. initialise h
    # ------------------------------------------------------------------
    try:
        # initial_h_values is provided by the harness (list/array of length ≤ 1000)
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    except Exception:
        # fallback: balanced binary vector of moderate size
        n = 256
        h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    # enforce exact sum = n/2 and bounds [0,1]
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple projection: scale then clip and redistribute any residual
        scale = target_sum / cur_sum if cur_sum != 0 else 0.0
        h = h * scale
        np.clip(h, 0.0, 1.0, out=h)
        # fix any remaining discrepancy by distributing uniformly over free entries
        deficit = target_sum - h.sum()
        if abs(deficit) > 1e-12:

            if free.size > 0:

                h[free] += inc
                np.clip(h, 0.0, 1.0, out=h)

                np.clip(h, 0.0, 1.0, out=h)
