import numpy as np
import time

def _project_onto_simplex(v, target):
    """Project v onto the hyperplane sum(v)=target with 0<=v<=1."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    # Simple water-filling: adjust interior points iteratively
    for _ in range(10):
        if abs(cur - target) < 1e-12:
            break
        excess = cur - target
        interior = (v > 0.0) & (v < 1.0)
        cnt = interior.sum()
        if cnt == 0:
            break
        delta = excess / cnt
        v[interior] -= delta
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
    # Final correction if needed (rare)
    if abs(v.sum() - target) > 1e-9:
        # distribute remaining deficit uniformly on non‑fixed entries
        deficit = target - v.sum()
        free = (v > 0.0) & (v < 1.0)
        if free.any():
            v[free] += deficit / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _evaluate(h, dx):
    """Return the C5 value for the current step function."""
    # linear (non‑circular) correlation as defined in the statement
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] with values in [0,1] and
    integral equal to 1, minimizing the Erdős C5 constant.

    Returns
    -------
    h_values : list of float
        The optimized discretised function.
    c5_bound : float
        The achieved bound (max overlap integral).
    n_points : int
        Number of discretisation points used.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. initialise h
    # ------------------------------------------------------------------
    try:
        # use the provided construction if present
        h = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        # fall back to a simple feasible start (half ones, half zeros)
        n = 70
        h = np.zeros(n, dtype=np.float64)
        h[: n // 2] = 1.0
        np.random.shuffle(h)

    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # enforce feasibility exactly
    h = _project_onto_simplex(h, target_sum)

    best_h = h.copy()
    best_val = float(_evaluate(best_h, dx))

    # ------------------------------------------------------------------
    # 2. simple stochastic local optimisation
    # ------------------------------------------------------------------
    # swap‑mass moves: transfer a small amount of mass from one index to another
    # while respecting box constraints.
    # The step size is gradually decreased.
    max_time = budget_s * 0.93  # keep a safety margin
    iteration = 0
    while time.time() - start_time < max_time:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # Decide transfer direction
        if np.random.rand() < 0.5:
            # move mass from i to j
            max_move = min(best_h[i], 1.0 - best_h[j])
        else:
            # move mass from j to i
            max_move = min(best_h[j], 1.0 - best_h[i])
        if max_move <= 0.0:

        # sample a transfer amount
        eps = np.random.uniform(0.0, max_move)
        if np.random.rand() < 0.5:
            # i -> j

            new_h[i] -= eps
