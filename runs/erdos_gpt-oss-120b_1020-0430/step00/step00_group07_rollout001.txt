def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_feasible(n, target):
    h = np.random.rand(n)
    # adjust to target sum preserving bounds
    diff = target - h.sum()
    # simple iterative adjustment
    while abs(diff) > 1e-12:
        idx = np.random.randint(n)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc <= 0:
                continue
            h[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec <= 0:
                continue
            h[idx] -= dec
            diff += dec
    return h

def propose_swap(h, max_step):
    n = h.shape[0]
    # pick distinct indices
    i = np.random.randint(n)
    j = np.random.randint(n-1)
    if j >= i:
        j += 1
    lo = max(-h[i], h[j] - 1.0, -max_step)
    hi = min(1.0 - h[i], h[j], max_step)
    if lo >= hi:
        return None, None, None
    delta = np.random.uniform(lo, hi)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, i, j

def run(seed=42, budget_s=1100, **kwargs):

    # decide dimension

def _c5_value(h):
    """Return the C5 value for a step function h (array)."""
    n = h.shape[0]
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _random_feasible(n, target):
    """Generate a random feasible vector of length n with sum=target and 0≤h≤1."""
    h = np.random.rand(n)
    diff = target - h.sum()
    # simple iterative correction keeping the bounds
    while abs(diff) > 1e-12:
        idx = np.random.randint(n)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc <= 0:
                continue
            h[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec <= 0:
                continue
            h[idx] -= dec
            diff += dec
    return h

def _propose_swap(h, max_step):
    """Propose a new vector by moving a small amount from one entry to another."""
    n = h.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

        return None
    delta = np.random.uniform(lo, hi)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new
