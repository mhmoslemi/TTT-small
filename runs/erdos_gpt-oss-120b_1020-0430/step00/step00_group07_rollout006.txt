import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a step function h (as a NumPy array), the achieved C5 bound,
    and the number of sample points.
    """
    # ------------------------------------------------------------------
    # problem size and basic constants
    # ------------------------------------------------------------------
    try:
        # initial_h_values is provided by the harness (if any)
        n_points = len(initial_h_values)
        h_initial = np.array(initial_h_values, dtype=np.float64)
    except Exception:          # no parent solution given
        n_points = 80
        h_initial = None

    dx = 2.0 / n_points
    target_sum = n_points / 2.0          # because ∫h = 1  →  sum(h)*dx = 1

    # ------------------------------------------------------------------
    # utility: projection onto the box [0,1] and the hyperplane sum = target_sum
    # ------------------------------------------------------------------
    def _proj_box_sum(x):
        """ Euclidean projection of x onto {0≤y_i≤1, sum y = target_sum}. """
        low, high = -np.max(x), 1.0 - np.min(x)          # bounds for the Lagrange multiplier λ
        for _ in range(60):                             # binary search for λ
            lam = (low + high) * 0.5
            y = np.clip(x + lam, 0.0, 1.0)
            s = y.sum()
            if s < target_sum:
                low = lam
            else:
                high = lam
        y = np.clip(x + (low + high) * 0.5, 0.0, 1.0)

        # tiny correction for numerical drift
        diff = target_sum - y.sum()
        if abs(diff) > 1e-12:
            mask = (y > 0.0) & (y < 1.0)
            if mask.any():
                y[mask] += diff / mask.sum()
                y = np.clip(y, 0.0, 1.0)
        return y

    # ------------------------------------------------------------------
    # correlation based evaluation of C5
    # ------------------------------------------------------------------
    def _eval_c5(h):
        """Return max_k ∫ h(x)*(1-h(x+k)) dx for the sampled h."""
        f = np.correlate(h, 1.0 - h, mode="full") * dx
        return float(f.max())

    # ------------------------------------------------------------------
    # shift helper (positive shift → values move left, zero padding on right)
    # ------------------------------------------------------------------
    def _shift(arr, shift):
        n = arr.shape[0]
        if shift > 0:
            out = np.empty_like(arr)
            out[:n - shift] = arr[shift:]
            out[n - shift:] = 0.0
            return out
        if shift < 0:
            out = np.empty_like(arr)
            out[-shift:] = arr[:n + shift]
            out[: -shift] = 0.0
            return out
        return arr.copy()

    # ------------------------------------------------------------------
    # gradient of the smooth approximation of the max‑overlap objective
    # ------------------------------------------------------------------
    def _smooth_grad(h, eps):
        """Gradient of ε·log∑exp(f_k/ε) where f_k = ∫h·(1‑h_shift)."""
        N = h.size
        f = np.correlate(h, 1.0 - h, mode="full") * dx          # shape 2N‑1
        m = f.max()
        # softmax weights, numerically stable
        w = np.exp((f - m) / eps)
        w /= w.sum()

        grad = np.zeros(N)
        for idx, wi in enumerate(w):
            if wi == 0.0:
                continue
            shift = idx - (N - 1)
            forward = _shift(h, shift)
            backward = _shift(h, -shift)
            grad += wi * (dx * (1.0 - (forward + backward)))
        return grad

    # ------------------------------------------------------------------
    # start‑point factories
    # ------------------------------------------------------------------
    def _random_start(rng):
        x = rng.random(n_points)                      # uniform in [0,1)
        return _proj_box_sum(x)

    def _triangle_start():
        x = (np.arange(n_points) + 0.5) * dx           # mid‑point of each cell
        h = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)   # triangular tent, height 1 at centre
        return _proj_box_sum(h)

    def _cosine_start():
        x = (np.arange(n_points) + 0.5) * dx
        h = 0.5 * (1.0 + np.cos(np.pi * (x - 1.0)))   # a smooth bump
        h = np.clip(h, 0.0, 1.0)
        return _proj_box_sum(h)

    start_strategies = [_random_start, _triangle_start, _cosine_start]

    # ------------------------------------------------------------------
    # initialise best solution
    # ------------------------------------------------------------------
    best_h = None
    best_val = np.inf
    if h_initial is not None:
        h_initial = _proj_box_sum(h_initial)
        val = _eval_c5(h_initial)
        best_h, best_val = h_initial.copy(), val

    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # main optimisation loop (restarts with projected gradient descent)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 0.3:
        # pick a start point
        if rng.random() < 0.7:               # mostly random
            h = _random_start(rng)
        else:                                # occasionally a structured start
            h = rng.choice(start_strategies)()   # triangle or cosine

        # local descent
        max_local_iters = 1800
        for it in range(max_local_iters):
            # diminishing step and smoothing parameter
            eps = max(0.001, 0.04 * (0.92 ** (it // 250)))
            step = 1.0 / np.sqrt(it + 1.0)

            grad = _smooth_grad(h, eps)
            # projected step
            h_next = _proj_box_sum(h - step * grad)

            # evaluate
            val_next = _eval_c5(h_next)
            if val_next < best_val:
                best_val, best_h = val_next, h_next.copy()
                # early exit if we already beat the target
                if best_val <= 0.3808:
                    return (best_h, float(best_val), n_points)

            # always move forward (even if not improving) – helps escape plateaus
            h = h_next

            if time.time() - start_time >= budget_s - 0.2:
                break

    # ------------------------------------------------------------------
    # fallback (should never happen)
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n_points, 0.5)
        best_val = _eval_c5(best_h)

    return (best_h, float(best_val), n_points)
