def proj_capped_simplex(v, target):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0, 1)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    # After binary search, w ~ feasible but sum may not equal target due to clamp; adjust:
    w = np.clip(v - (lo+hi)/2, 0, 1)
    # may be slightly off due to discretization; adjust by adding diff to entries not at bound
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        # distribute diff among non-bound entries
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
    return w

def _proj_capped_simplex(v, target, max_iter=60, tol=1e-12):
    """Project v onto {0<=x<=1, sum(x)=target}."""
    lo = -np.max(v) - 1.0   # ensures all v - lo > 1
    hi = np.max(v)          # ensures all v - hi <= 0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= tol:
            break
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > eps:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)

    return dx * np.max(np.correlate(h, 1.0 - h, mode='full'))

    while time.time() - start_time < time_budget:
        # generate random start
        x0 = _random_feasible(n_points, target)
        # local SLSQP
        opt_h, opt_val = _slsqp_optimize(x0, target, dx)
        # short refinement (max 0.15s per candidate)
        rem = time_budget - (time.time() - start_time)
        if rem <= 0:
            break
        max_refine = min(0.15, rem * 0.2)
        opt_h, opt_val = _local_refine(opt_h, target, dx, max_refine)
        if opt_val < best_val:
            best_h, best_val = opt_h, opt_val
        # also try improving the current best occasionally
        if np.random.rand() < 0.2:
            rem = time_budget - (time.time() - start_time)
            if rem > 0.2:
                best_h, best_val = _local_refine(best_h, target, dx, 0.2)

    # Final polishing with remaining time
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0.5:
        best_h, best_val = _local_refine(best_h, target, dx, remaining - 0.1)

    # Final projection to guarantee feasibility
    best_h = _proj_capped_simplex(best_h, target)
    best_val = _compute_c5(best_h, dx)

    return (best_h, float(best_val), int(n_points))
