import time, numpy as np, random, math

def _project_box_sum(v, low, high, target, iters=60):
    """Project v onto [low,high]^n with sum(v)=target (Euclidean projection)."""
    lo, hi = -1e6, 1e6
    for _ in range(iters):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, low, high)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, low, high)


def _eval_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k)) dx   (discretised)."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def _grad_min_overlap(h):
    """Gradient of the overlap for the shift giving the *minimum* overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full')
    min_idx = int(np.argmin(corr))
    k = min_idx - (n - 1)
    k = abs(k)
    g = np.zeros_like(h)
    if k == 0:
        g = 2.0 * h
    else:
        g[: n - k] += h[k:]
        g[k:] += h[: n - k]
    g *= dx
    return g


def _optimize(init_h, max_seconds, target_sum):
    """Projected sub‑gradient ascent on the *minimum* self‑overlap."""
    cur = _project_box_sum(init_h.copy(), 0.0, 1.0, target_sum)
    best = cur.copy()
    best_c5 = _eval_c5(best)

    step = 0.5
    start = time.time()
    while time.time() - start < max_seconds:
        # gradient for the worst (minimum) overlap shift
        g = _grad_min_overlap(cur)
        # make direction zero‑sum so the sum constraint holds to first order
        d = g - g.mean()
        improved = False
        for fac in (1.0, 0.5, 0.25, 0.125):
            hnew = _project_box_sum(cur + step * fac * d, 0.0, 1.0, target_sum)
            c5new = _eval_c5(hnew)
            if c5new < best_c5 - 1e-12:
                cur, best, best_c5 = hnew, hnew, c5new
                step = min(step * 1.1, 5.0)
                improved = True
                break
        if not improved:
            step *= 0.5
            if step < 1e-7:                      # stuck → random tweak
                i, j = np.random.randint(len(cur), size=2)
                if cur[i] > 0.0 and cur[j] < 1.0:
                    delta = min(cur[i], 1.0 - cur[j]) * np.random.rand()
                    hnew = cur.copy()
                    hnew[i] -= delta
                    hnew[j] += delta
                    c5new = _eval_c5(hnew)
                    if c5new < best_c5:
                        cur, best, best_c5 = hnew, hnew, c5new
                        step = 0.5
        # early exit if we already beat the target record
        if best_c5 <= 0.3808:

    return best, best_c5

    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
