def repair_to_sum(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    diff = target - current
    if abs(diff) > 1e-12:
        # try to adjust a single element
        if diff > 0:
            idx = np.argmax(h < 1 - 1e-12)  # index of first less-than-1 element

import time
import random
import numpy as np

# ---------- Helper utilities -------------------------------------------------


def evaluate_h(h: np.ndarray) -> float:
    """Return C5 for a step function sampled by h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))


def repair_to_sum(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust a few entries so that sum(h)=len(h)/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Fix any remaining gap by touching single entries (diff is tiny).
    while abs(diff) > 1e-12:
        if diff > 0:
            idxs = np.where(h < 1.0 - 1e-12)[0]
            if idxs.size == 0:
                break
            i = np.random.choice(idxs)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            idxs = np.where(h > 1e-12)[0]
            if idxs.size == 0:
                break
            i = np.random.choice(idxs)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h


def init_binary_vector(n: int) -> np.ndarray:
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def upscale(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of h to a longer grid."""
    old_n = h.shape[0]
    if old_n == new_n:
        return h.copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    h_new = np.interp(x_new, x_old, h)
    return h_new


# ---------- Core optimizer ---------------------------------------------------


def local_search_continuous(start_h: np.ndarray, time_budget: float) -> tuple:
    """
    Continuous hill‑climbing / simulated annealing.
    Returns a (h, value) pair.
    """
    start = time.perf_counter()
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_val = evaluate_h(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    # step size (max transferred mass) decays with time
    step_start = 0.5
    step_min = 1e-3
    eps = 1e-12

    while time.perf_counter() - start < time_budget:
        elapsed = time.perf_counter() - start
        # linear decay of step
        step = max(step_min, step_start * (1.0 - elapsed / time_budget))

        # identify worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # overlapping region for this shift
        if shift >= 0:
            i_start, i_end = 0, n - shift
        else:
            i_start, i_end = -shift, n

        indices = np.arange(i_start, i_end, dtype=int)
        j_indices = indices + shift

        # candidates that can reduce the term h[i]*(1-h[j])
        mask = (cur_h[indices] > eps) & (cur_h[j_indices] < 1.0 - eps)
        cand = indices[mask]

        if cand.size > 0:
            i = int(np.random.choice(cand))
            j = int(i + shift)
        else:
            # generic random pair fallback
            pos_i = np.where(cur_h > eps)[0]
            pos_j = np.where(cur_h < 1.0 - eps)[0]
            if pos_i.size == 0 or pos_j.size == 0:
                break
            i = int(np.random.choice(pos_i))
            j = int(np.random.choice(pos_j))
            if i == j:
                continue

        max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        delta = np.random.rand() * max_delta

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_val = evaluate_h(new_h)

        # deterministic improvement acceptance
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h, new_val
        else:
            # occasional uphill move (tiny temperature)
            temp = 0.001 * (1.0 - elapsed / time_budget)
            if temp > 0 and np.exp((cur_val - new_val) / temp) > np.random.rand():
                cur_h, cur_val = new_h, new_val

    return best_h, best_val


def search_best(n: int, total_time: float) -> tuple:
    """
    Run several restarts of the local search, optionally seeded with the
    provided global `initial_h_values`.
    Returns the best (h, value) found within total_time seconds.
    """
    start = time.perf_counter()
    best_h = None
    best_val = np.inf
    allocated = 0.0

    # try to use the supplied construction if it matches the size
    try:
        init_h = np.array(initial_h_values, dtype=np.float64)
        if init_h.shape[0] != n:
            init_h = upscale(init_h, n)
        init_h = repair_to_sum(init_h)
        t0 = total_time * 0.25
        allocated += t0
        h_opt, val_opt = local_search_continuous(init_h, t0)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt
    except NameError:
        pass  # no initial_h_values supplied

    remaining = total_time - allocated
    if remaining <= 0.0:
        return best_h, best_val

    # allocate time for several random restarts
    n_restart = max(1, int(remaining // 5.0))
    per_restart = remaining / n_restart
    for _ in range(n_restart):
        init_h = init_binary_vector(n)
        h_opt, val_opt = local_search_continuous(init_h, per_restart)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt
            if best_val < 0.3808:
                break

    # final sanity repair
    best_h = repair_to_sum(best_h)
    best_val = evaluate_h(best_h)
    return best_h, best_val


# ---------- Entry point ------------------------------------------------------


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    total_budget = float(budget_s)
    start_all = time.perf_counter()

    # Phase 1: work on the base grid (n=42)
    n_base = 42
    t_phase1 = total_budget * 0.55
    h_best, val_best = search_best(n_base, t_phase1)

    # Phase 2: if time left, try a refined grid (n=84)
    elapsed = time.perf_counter() - start_all
    remaining = total_budget - elapsed
    if remaining > 5.0:
        n_refined = 84
        # upscale current best solution as a warm start
        h_up = upscale(h_best, n_refined)
        h_up = repair_to_sum(h_up)
        # allocate the rest of the budget to the refined search
        h_ref, val_ref = search_best(n_refined, remaining)
        # keep whichever yields the lower bound
        if val_ref < val_best:
            h_best, val_best = h_ref, val_ref
            n_base = n_refined

    # Ensure final vector satisfies constraints exactly
    h_best = repair_to_sum(h_best)
    final_val = evaluate_h(h_best)

    return (h_best, float(final_val), int(len(h_best)))
