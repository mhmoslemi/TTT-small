def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    # choose n_points
    n = 200  # or dynamic limited by time
    dx = 2.0 / n
    target_sum = n / 2

    # init h
    if 'initial_h_values' in globals():
        h_init = initial_h_values
        # resample to n
        x0 = np.linspace(0, 2, len(h_init), endpoint=False)
        x_target = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_target, x0, h_init)
    else:
        h = np.random.rand(n)
    # projection onto capped simplex
    h = project_onto_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_val = evaluate(h, dx)
    # parameters
    max_iters = 10000  # but early stop by time
    init_step = 1.0  # can adjust
    # main loop
    for it in range(1, max_iters+1):
        if time.time() - start > budget_s * 0.95:
            break
        # compute conv and max shifts
        conv = np.correlate(h, 1 - h, mode='full') * dx
        c5 = conv.max()
        if c5 < best_val - 1e-10:  # improvement
            best_val = c5
            best_h = h.copy()
            # optional early exit if c5 < target 0.3808
            if best_val <= 0.3808:
                # can break? maybe continue to refine lower, but good.
                pass
        # Determine shifts achieving max (or near)
        max_idxs = np.where(np.abs(conv - c5) <= 1e-12)[0]
        # compute gradient average
        grad = np.zeros_like(h)
        for idx in max_idxs:
            s = idx - (n - 1)  # shift index
            grad += grad_for_shift(h, s, dx)

        step = init_step / np.sqrt(it)  # diminishing

    # final evaluation
    final_c5 = evaluate(best_h, dx)
    best_h = best_h.astype(np.float64) # ensure
    return best_h, float(final_c5), n

def grad_for_shift(h, s, dx):
    n = len(h)

        grad[s:] -= dx * h[:n-s]
    else:
        t = -s
        grad[t:] -= dx * h[:n-t]
        grad[:n-t] -= dx * h[t:]
    return grad

import numpy as np
import time
import math

def _c5_value(h, dx):
    """Return max overlap C5 for a step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _project_capped_simplex(y, target, lower=0.0, upper=1.0):
    """
    Euclidean projection of y onto {x: sum(x)=target, lower<=x<=upper}.
    Uses binary search on the Lagrange multiplier.
    """
    n = y.shape[0]
    # lambda bounds: low enough => all clipped to upper, high enough => all lower
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - (lo + hi) * 0.5, lower, upper)
    # tiny adjustment to fix possible rounding error
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute the remainder over non‑boundary entries
        mask = (x > lower) & (x < upper)
        if mask.any():
            x[mask] += diff / mask.sum()
    return x


def _grad_for_shift(h, shift, dx):
    """Subgradient of the overlap for a single shift."""
    n = h.shape[0]
    g = np.full(n, dx)
    if shift >= 0:
        s = shift
        if s < n:
            g[:n - s] -= dx * h[s:]
            g[s:] -= dx * h[:n - s]
    else:
        t = -shift
        if t < n:
            g[t:] -= dx * h[:n - t]
            g[:n - t] -= dx * h[t:]
    return g


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound and number of samples."""
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initial construction (use provided values when available)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        n = max(180, 2 * n0)               # a modest up‑sampling for flexibility
        # linear interpolation to the new grid
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, h0)
    else:
        n = 200
        h = np.random.rand(n)

    dx = 2.0 / n
    target_sum = n / 2.0                     # because ∫h = 1  ⇒  Σh = n/2
    h = _project_capped_simplex(h, target_sum)

    # current best
    best_h = h.copy()
    best_val = _c5_value(best_h, dx)

    # ------------------------------------------------------------------
    # Sub‑gradient descent (few thousand steps)
    # ------------------------------------------------------------------
    max_iters = 3000
    init_step = 5.0                          # magnitude of the raw step
    for it in range(1, max_iters + 1):
        if time.time() - start_time > budget_s * 0.85:
            break

        # correlation and the shift(s) attaining the maximum overlap
        conv = np.correlate(h, 1.0 - h, mode="full")
        max_val = conv.max()
        idxs = np.where(np.abs(conv - max_val) <= 1e-12)[0]
        shifts = idxs - (n - 1)

        # averaged sub‑gradient of the active shift(s)
        grad = np.zeros_like(h)
        for s in shifts:

        grad /= len(shifts)

        # diminishing step size
        step = init_step / math.sqrt(it)
        y = h - step * grad

            if best_val <= 0.3808:
