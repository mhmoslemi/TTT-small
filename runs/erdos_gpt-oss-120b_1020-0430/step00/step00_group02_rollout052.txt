import time, math
import numpy as np

def _project_sum_bounds(h, target_sum):
    """Clip h to [0,1] and adjust to exactly meet target_sum."""
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return
    # distribute the needed mass uniformly over non‑saturated entries
    if diff > 0:  # need to increase
        caps = 1.0 - h
        cap_total = caps.sum()
        if cap_total <= 0:
            return
        add = caps * (diff / cap_total)
        h += add
    else:        # need to decrease
        caps = h
        cap_total = caps.sum()
        if cap_total <= 0:
            return
        sub = caps * (-diff / cap_total)
        h -= sub
    np.clip(h, 0.0, 1.0, out=h)
    # final tiny correction
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # adjust first free entry
        for i in range(len(h)):
            if (diff > 0 and h[i] < 1.0) or (diff < 0 and h[i] > 0.0):
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)
                break


def _upsample(h, new_n):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    target = new_n / 2.0
    _project_sum_bounds(h_new, target)
    return h_new


def _evaluate(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def _gradient_of_worst(h, lag):
    """Gradient of the correlation for a given lag (worst shift)."""
    n = len(h)
    grad = np.ones(n)  # derivative of Σ h_i term
    # subtract h_{i-lag}
    idx_src = np.arange(n) - lag
    mask = (idx_src >= 0) & (idx_src < n)
    grad[mask] -= h[idx_src[mask]]
    # subtract h_{i+lag}
    idx_src = np.arange(n) + lag
    mask = (idx_src >= 0) & (idx_src < n)
    grad[mask] -= h[idx_src[mask]]
    return grad

                continue
            donor, receiver = i, j
            max_delta = min(cur_h[donor], 1.0 - cur_h[receiver])
            if max_delta <= 0:
                continue
            delta = max_delta * (rng.random() ** 2) * base_step
            if delta <= 0:
                continue
            new_h = cur_h.copy()
            new_h[donor] -= delta
            new_h[receiver] += delta
            new_score = _evaluate(new_h)
            diff = new_score - cur_score
            if diff < 0 or rng.random() < math.exp(-diff / (temp + 1e-12)):
                cur_h = new_h
                cur_score = new_score
                if new_score < best_score:
                    best_h = new_h.copy()
                    best_score = new_score

        it += 1

    # final projection to exact constraints (tiny drift may appear)
    _project_sum_bounds(best_h, target_sum)
    final_score = _evaluate(best_h)

    return (best_h.tolist(), float(final_score), n)
```
