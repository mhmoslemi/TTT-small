def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.RandomState(seed)
    init_h_raw = globals().get('initial_h_values', None)
    if init_h_raw is not None:
        init_h_raw = np.asarray(init_h_raw)
        n_points = init_h_raw.shape[0]
        # ensure sum = n_points/2 via top half selection
        k = n_points // 2
        idx = np.argpartition(-init_h_raw, k-1)[:k]  # top k indices
        h = np.zeros(n_points, dtype=np.int8)
        h[idx] = 1
    else:
        n_points = 80
        k = n_points // 2
        h = np.zeros(n_points, dtype=np.int8)
        h[:k] = 1
        rng.shuffle(h)
    # functions
    dx = 2.0 / n_points
    def evaluate(h_vec):
        # h_vec is np.ndarray of ints 0/1
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx
    cur_val = evaluate(h)
    best_h = h.copy()
    best_val = cur_val
    start_time = time.time()
    # Limits
    max_no_improve = 2000
    no_improve_counter = 0
    # main loop
    while time.time() - start_time < budget_s - 0.5:
        # pick random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # propose swap
        h[i1] = 0
        h[i0] = 1
        new_val = evaluate(h)
        # if improvement accept, else revert (maybe accept with probability)
        if new_val < cur_val - 1e-12:  # improvement
            cur_val = new_val
            no_improve_counter = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert swap
            h[i1] = 1
            h[i0] = 0
            no_improve_counter += 1
        # if stuck too long, random restart
        if no_improve_counter >= max_no_improve:
            # restart random binary vector
            h[:] = 0
            h[:k] = 1
            rng.shuffle(h)
            cur_val = evaluate(h)
            no_improve_counter = 0
    # after loop, best_h is best solution found
    # ensure best_h sum equals n_points/2 (it does)
    return best_h.astype(float), best_val, n_points

import numpy as np
import time
import math

def _max_overlap(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1‑h(x+k))dx for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with zero padding (mode='full')
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)


def _random_binary(n: int, rng: np.random.RandomState) -> np.ndarray:
    """Create a random binary vector of length n with exactly n/2 ones."""
    k = n // 2
    h = np.zeros(n, dtype=np.int8)
    h[:k] = 1
    rng.shuffle(h)
    return h


def _greedy_local_optimize(h: np.ndarray) -> tuple[np.ndarray, float]:
    """Exhaustive swap‑based greedy improvement (stops when no swap improves)."""
    improved = True
    cur_val = _max_overlap(h)
    while improved:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i1 in ones:
            for i0 in zeros:
                # try swapping i1 (1) with i0 (0)
                h[i1] = 0
                h[i0] = 1
                new_val = _max_overlap(h)
                if new_val < cur_val - 1e-12:
                    cur_val = new_val
                    improved = True
                    break
                # revert
                h[i1] = 1
                h[i0] = 0
            if improved:
                break
    return h, cur_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h, its overlap bound, and the number of sample points.
    The returned h satisfies 0 ≤ h ≤ 1 and ∫₀² h = 1.
    """
    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # 1. Initialise the vector (binary, sum = n/2)
    # ------------------------------------------------------------------
    init_raw = globals().get('initial_h_values')
    if init_raw is not None:
        init_raw = np.asarray(init_raw)
        n_points = int(init_raw.shape[0])
        k = n_points // 2
        # select the k largest entries (ties broken arbitrarily)
        idx = np.argpartition(-init_raw, k - 1)[:k]
        h = np.zeros(n_points, dtype=np.int8)
        h[idx] = 1
    else:
        n_points = 80
        h = _random_binary(n_points, rng)

    cur_val = _max_overlap(h)
    best_h = h.copy()
    best_val = cur_val

    start_time = time.time()
    time_limit = budget_s * 0.95  # leave a small margin for the final pass
    max_fail = 3000               # attempts without improvement before a restart
    fail_cnt = 0

    # ------------------------------------------------------------------
    # 2. Random‑restart hill climbing (swap one 1 with one 0)
    # ------------------------------------------------------------------
    while time.time() - start_time < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)

        # perform the swap
        h[i1] = 0
        h[i0] = 1
        new_val = _max_overlap(h)

        if new_val < cur_val - 1e-12:          # improvement
            cur_val = new_val
            fail_cnt = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert the change
            h[i1] = 1
            h[i0] = 0
            fail_cnt += 1

        if fail_cnt >= max_fail:
            # random restart
            h = _random_binary(n_points, rng)
            cur_val = _max_overlap(h)
            fail_cnt = 0

    # ------------------------------------------------------------------
    # 3. Final deterministic local optimisation starting from the best found
    # ------------------------------------------------------------------
    h_opt, val_opt = _greedy_local_optimize(best_h.copy())
    if val_opt < best_val - 1e-12:

    # Convert to float array for the required return type
    h_values = best_h.astype(float)

    return h_values, float(best_val), int(n_points)
