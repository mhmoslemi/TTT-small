import time
import numpy as np
from scipy.signal import fftconvolve

def _proj_box_sum(y, total, lo=0.0, hi=1.0):
    """Euclidean projection onto {x: lo<=x<=hi, sum(x)=total}."""
    y = np.clip(y, lo, hi)
    # bounds for lambda
    lam_low = np.min(y - hi) - 1.0
    lam_high = np.max(y - lo) + 1.0
    for _ in range(60):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(y - lam, lo, hi)
        s = x.sum()
        if s > total:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(y - (lam_low + lam_high) * 0.5, lo, hi)


def _c5_and_lag(h, dx):
    """Return (max overlap integral C5, lag achieving it) for a step function h."""
    conv = fftconvolve(h, (1 - h)[::-1], mode='full')
    C = conv * dx
    idx = int(np.argmax(C))
    lag = idx - (len(h) - 1)          # lag in sample units, negative allowed
    return float(C[idx]), lag


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------ #
    # 1. Build initial binary vector of length n (>= 140)
    # ------------------------------------------------------------------ #

        init = np.asarray(initial_h_values, dtype=float)

    except Exception:
        init = None
        n0 = 0

    n = max(140, n0)                     # target resolution
    dx = 2.0 / n
    target_sum = n / 2.0

    if init is not None:
        # Upsample (linear) to length n, then binarise by top‑half rule
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
        # take largest target_sum entries as 1
        order = np.argsort(-h)           # descending
        h_bin = np.zeros_like(h)
        h_bin[order[:int(target_sum)]] = 1.0
        h = h_bin
    else:
        # pure random binary start
        h = np.zeros(n, dtype=float)
        h[:int(target_sum)] = 1.0
        np.random.shuffle(h)

    # ensure exact constraints
    h = _proj_box_sum(h, target_sum, 0.0, 1.0)

    # ------------------------------------------------------------------ #
    # 2. Local search (targeted swaps + occasional random swaps)
    # ------------------------------------------------------------------ #
    best_h = h.copy()
    best_val, _ = _c5_and_lag(best_h, dx)
    cur_h = h.copy()
    cur_val = best_val

    # annealing schedule
    T0 = 0.01
    it = 0
    last_improve = 0
    max_no_imp = 3000                         # restart after this many steps

    while time.time() - start < budget_s * 0.95:
        it += 1
        # current worst lag
        cur_val, lag = _c5_and_lag(cur_h, dx)
        offset = lag

        improved = False

        # -----------------------------------------------------------------

        # -----------------------------------------------------------------
        if offset != 0:

                lim = n - offset
                h0 = cur_h[:lim]

    final_val, _ = _c5_and_lag(best_h, dx)
