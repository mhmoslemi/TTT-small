import time
import numpy as np
from scipy.optimize import minimize

def _c5_value(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1−h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate gives the discrete sum; multiply by dx for the integral.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the box-simplex {0≤x≤1, sum x = n/2}."""
    n = v.shape[0]
    target = n / 2.0
    x = np.clip(v, 0.0, 1.0)
    # Simple water‑filling iteration; converges quickly for tiny n.
    for _ in range(1000):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(x < 1.0)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            # Apply but respect the 1‑bound
            delta = np.minimum(inc, 1.0 - x[free])
            x[free] += delta
        else:
            free = np.where(x > 0.0)[0]
            if free.size == 0:
                break
            dec = diff / free.size   # diff is negative
            delta = np.maximum(dec, -x[free])
            x[free] += delta
    # Tiny leftovers – dump them on any non‑boundary coordinate.
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size:
            x[free[0]] = np.clip(x[free[0]] + diff, 0.0, 1.0)
    return x


def _local_slsqp(start: np.ndarray, time_budget: float):
    """A short SLSQP refinement; run only if enough time is left."""
    n = start.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def obj(x):
        return _c5_value(x)
    # Give SLSQP a little time, but never exceed the budget.
    maxiter = 200
    try:
        res = minimize(
            obj,
            start,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-12, "disp": False},
        )
        if res.success:
            return res.x, float(res.fun)
    except Exception:
        pass
    return start, obj(start)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h on [0,2] that respects the constraints
    and (hopefully) yields a low Erdős C5 bound.

    Output: (h_values, c5_bound, n_points)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise ------------------------------------------------------
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        h = np.asarray(init, dtype=float)
    else:
        # fallback: 45 uniform samples that satisfy the integral.
        n_fallback = 45
        h = np.full(n_fallback, 0.5, dtype=float)
    h = _project_to_simplex(h)
    n = h.shape[0]
    target_sum = n / 2.0

    best_h = h.copy()
    best_val = _c5_value(best_h)

    # ------------------------------------------------------------------
    # 2. Stochastic hill climbing ----------------------------------------
    # ------------------------------------------------------------------
    # 80 % of the budget for cheap random moves, 20 % left for a polish.
    search_budget = budget_s * 0.80
    end_search = start_time + search_budget

    # counters for stagnation handling
    no_improve = 0
    max_no_improve = 20000

    while time.time() < end_search:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # maximal admissible transfer from i → j
        delta_max = min(best_h[i], 1.0 - best_h[j])
        if delta_max <= 0.0:
            # try opposite direction
            delta_max = min(best_h[j], 1.0 - best_h[i])
            if delta_max <= 0.0:
                continue
            i, j = j, i

        # gradually shrink step size as time passes
        remaining = end_search - time.time()
        scale = 0.1 + 0.9 * (remaining / search_budget)   # from 1.0 → 0.1
        delta = delta_max * rng.random() * scale
        if delta <= 0.0:
            continue

        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        val_new = _c5_value(h_new)

        if val_new < best_val - 1e-12:
            best_h = h_new
            best_val = val_new
            no_improve = 0
        else:
            no_improve += 1
            # small uphill probability (simulated annealing style)
            temp = 1e-3 * (remaining / search_budget)
            if val_new < best_val + temp * rng.random():
                best_h = h_new

        # if stuck for a while, do a modest random perturbation
        if no_improve >= max_no_improve:
            pert = best_h.copy()
            for _ in range(5):
                a, b = rng.integers(0, n, size=2)
                if a == b:
                    continue
                dm = min(pert[a], 1.0 - pert[b])
                if dm <= 0.0:
                    continue
                d = dm * rng.random() * 0.5
                pert[a] -= d
                pert[b] += d
            val_pert = _c5_value(pert)
            if val_pert < best_val:
                best_h = pert
                best_val = val_pert
            no_improve = 0

    # ------------------------------------------------------------------
    # 3. Local refinement (SLSQP) ---------------------------------------
    # ------------------------------------------------------------------
    remaining_budget = budget_s - (time.time() - start_time)
    if remaining_budget > 5.0:
        refined_h, refined_val = _local_slsqp(best_h, remaining_budget)
        if refined_val < best_val:
            best_h = refined_h
            best_val = refined_val

    # ------------------------------------------------------------------
    # 4. Final clean‑up -------------------------------------------------
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce the exact integral again (tiny numeric drift)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        diff = target_sum - cur_sum
        free = np.where((best_h > 0.0) & (best_h < 1.0))[0]
        if free.size:
            best_h[free[0]] += diff
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _c5_value(best_h)

    return best_h.tolist(), float(final_c5), n
