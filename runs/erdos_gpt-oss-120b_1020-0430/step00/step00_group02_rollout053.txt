def hill_climb(h, max_iter, dx):
    best_h = h.copy()
    best_val = evaluate_h(best_h, dx)
    n = len(h)
    ones_idx = np.where(best_h > 0.5)[0] # binary as 0/1
    zeros_idx = np.where(best_h <= 0.5)[0]

    for it in range(max_iter):
        # pick random swap
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        val = evaluate_h(new_h, dx)

            best_val = val
            # update idx sets

            zeros_idx = np.where(best_h <= 0.5)[0]
        # else maybe keep? decide not.
    return best_h, best_val

def hill_climb_targeted(h, max_iter, dx):
    best_h = h.copy()
    best_val = evaluate_h(best_h, dx)
    n = len(h)
    for it in range(max_iter):
        # compute correlation and max shift
        corr = scipy.signal.correlate(best_h, 1-best_h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        if max_val * dx >= best_val - 1e-12:
            # no improvement else
            pass
        k = max_idx - (n-1)  # shift
        # compute mismatched indices for this shift
        if k >= 0:
            i_range = np.arange(0, n - k)
            i_mismatched = i_range[(best_h[i_range] > 0.5) & (best_h[i_range + k] <= 0.5)]
        else:
            kk = -k
            i_range = np.arange(kk, n)
            # shift is negative: h[i] * (1 - h[i+k]) = 1 if h[i] =1 and h[i+k]=0, with k negative meaning i+k < i
            i_mismatched = i_range[(best_h[i_range] > 0.5) & (best_h[i_range + k] <= 0.5)]
        if i_mismatched.size == 0:
            # no mismatches? unlikely
            # random swap
            ones_idx = np.where(best_h > 0.5)[0]
            zeros_idx = np.where(best_h <= 0.5)[0]
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
        else:
            i = np.random.choice(i_mismatched)
            # choose j a random zero index (that is not i)
            zeros_idx = np.where(best_h <= 0.5)[0]
            # ensure j != i (should already be zero) but ensure we pick any
            j = np.random.choice(zeros_idx)

def local_improve(h, rng, dx, max_iters):
    n = len(h)
    best_h = h.copy()
    best_val = evaluate_h(best_h, dx)
    # Precompute indices of ones and zeros
    for it in range(max_iters):
        # compute correlation and get shift with max
        corr = scipy.signal.correlate(best_h, 1-best_h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        # Early stop if best_val cannot be improved (maybe converge)
        # Propose swap
        k = max_idx - (n-1)
        # find mismatched ones indices
        mismatches = []
        if k >= 0:
            i_range = np.arange(0, n - k)
            mismatches = i_range[(best_h[i_range] > 0.5) & (best_h[i_range + k] <= 0.5)]
        else:
            # k negative
            kk = -k
            i_range = np.arange(kk, n)
            mismatches = i_range[(best_h[i_range] > 0.5) & (best_h[i_range + k] <= 0.5)]
        # if mismatches empty, random swap
        if mismatches.size > 0:
            i = rng.choice(mismatches)
            # choose a zero location j not equal i
            zeros = np.where(best_h <= 0.5)[0]
            # ensure not picking same as i (i is 1)
            # also pick j distinct from i
            j = rng.choice(zeros)
        else:
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h <= 0.5)[0]

            j = rng.choice(zeros)

import numpy as np, time, math
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def generate_balanced_binary(rng, n):
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h

def local_improve(start_h, rng, dx, max_iters=2000):
    n = len(start_h)
    cur_h = start_h.copy()
    cur_val = evaluate_h(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    half = n // 2
    T0 = 0.2
    for it in range(max_iters):
        T = T0 * (1 - it / max_iters)
        # compute correlation for current h
        corr = correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # find mismatched indices for this shift
        if k >= 0:
            i_range = np.arange(0, n - k)
            mism = i_range[(cur_h[i_range] > 0.5) & (cur_h[i_range + k] <= 0.5)]
        else:
            i_range = np.arange(-k, n)
            mism = i_range[(cur_h[i_range] > 0.5) & (cur_h[i_range + k] <= 0.5)]
        if mism.size > 0:
            i = rng.choice(mism)
            zeros = np.where(cur_h <= 0.5)[0]
            j = rng.choice(zeros)
        else:
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h <= 0.5)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # propose swap
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_val = evaluate_h(new_h, dx)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val
    return best_h, best_val

def search_step_patterns(n, dx):
    half = n // 2
    best_val = np.inf
    best_h = None
    for step in range(1, n):
        if math.gcd(step, n) != 1:
            continue
        idx = (np.arange(n) * step) % n
        h = (idx < half).astype(float)
        val = evaluate_h(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_val = np.inf
    best_n = None
    # list of n to try, even numbers up to 200
    n_list = [180, 200, 160, 140, 120, 100, 80, 60, 45]
    for n in n_list:
        # respect time
        if time.time() - start_time > budget_s * 0.95:
            break
        if n % 2 != 0:
            continue
        dx = 2.0 / n
        # 1. deterministic step patterns
        h_pat, val_pat = search_step_patterns(n, dx)
        if val_pat < best_val:
            best_h, best_val, best_n = h_pat, val_pat, n
        # 2. random hill climbing restarts
        max_restarts = 20 if n <= 100 else 10
        for r in range(max_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
            # initial vector: random balanced binary
            h0 = generate_balanced_binary(rng, n)
            # also try upsampled initial_h if available for first restart
            if r == 0 and 'initial_h_values' in globals():
                init = globals()['initial_h_values']
                if isinstance(init, np.ndarray) and init.ndim == 1 and n % len(init) == 0:
                    factor = n // len(init)
                    h_tmp = np.repeat(init, factor)
                    # convert to binary by threshold
                    h_bin = (h_tmp > 0.5).astype(float)
                    cur_ones = int(h_bin.sum())
                    half_needed = n // 2
                    if cur_ones != half_needed:
                        # adjust
                        if cur_ones > half_needed:
                            idx1 = np.where(h_bin == 1)[0]
                            flip = rng.choice(idx1, size=cur_ones - half_needed, replace=False)
                            h_bin[flip] = 0
                        else:
                            idx0 = np.where(h_bin == 0)[0]
                            flip = rng.choice(idx0, size=half_needed - cur_ones, replace=False)
                            h_bin[flip] = 1
                    h0 = h_bin
            # improve
            h_opt, val_opt = local_improve(h0, rng, dx, max_iters=2000)
            if val_opt < best_val:
                best_h, best_val, best_n = h_opt.copy(), val_opt, n
    # final check: recompute bound
    if best_h is None:
        # fallback trivial half ones contiguous
        n = 60
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[:n//2] = 1.0
        best_val = evaluate_h(best_h, 2.0/n)
    final_val = evaluate_h(best_h, 2.0/best_n)
    return best_h, float(final_val), int(best_n)

import time, math
import numpy as np
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Return C5 value for step function h with spacing dx."""
    corr = correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def generate_balanced_binary(rng, n):
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def local_improve(start_h, rng, dx, max_iters=2000):
    """Stochastic hill‑climb with occasional uphill moves."""
    n = len(start_h)
    cur_h = start_h.copy()
    cur_val = evaluate_h(cur_h, dx)
    best_h, best_val = cur_h.copy(), cur_val
    T0 = 0.2
    for it in range(max_iters):
        T = T0 * (1.0 - it / max_iters)
        # correlation of current state to locate worst shift
        corr = correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)          # shift where overlap is maximal
        # indices that contribute to the current maximum
        if k >= 0:
            i_range = np.arange(0, n - k)
            mism = i_range[(cur_h[i_range] > 0.5) & (cur_h[i_range + k] <= 0.5)]
        else:
            i_range = np.arange(-k, n)
            mism = i_range[(cur_h[i_range] > 0.5) & (cur_h[i_range + k] <= 0.5)]
        if mism.size > 0:
            i = rng.choice(mism)                     # a 1 that mismatches
            zeros = np.where(cur_h <= 0.5)[0]
            j = rng.choice(zeros)                    # a 0 elsewhere
        else:
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h <= 0.5)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # propose swapping i (1) and j (0)
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_val = evaluate_h(new_h, dx)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
    return best_h, best_val

def search_step_patterns(n, dx):
    """Exhaustive search over Sturmian‑type binary sequences."""
    half = n // 2
    best_val = np.inf
    best_h = None
    for step in range(1, n):
        if math.gcd(step, n) != 1:
            continue
        idx = (np.arange(n) * step) % n
        h = (idx < half).astype(float)
        val = evaluate_h(h, dx)
        if val < best_val:
            best_val, best_h = val, h.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_val = np.inf
    best_n = None

    # list of even lengths to explore (larger n gives finer resolution)
    n_list = [180, 200, 160, 140, 120, 100, 80, 60, 45]

    for n in n_list:
        if time.time() - start > budget_s * 0.95:
            break
        if n % 2 != 0:
            continue
        dx = 2.0 / n
        # 1) deterministic Sturmian candidates
        h_pat, val_pat = search_step_patterns(n, dx)
        if val_pat < best_val:
            best_h, best_val, best_n = h_pat.copy(), val_pat, n

        # 2) random local‑search restarts
        max_restarts = 20 if n <= 100 else 10
        for r in range(max_restarts):
            if time.time() - start > budget_s * 0.95:
                break
            # start from a random balanced binary vector
            h0 = generate_balanced_binary(rng, n)

            # optionally seed the first restart with the provided initial solution
            if r == 0 and "initial_h_values" in globals():
                init = np.asarray(globals()["initial_h_values"]).ravel()
                if init.size > 0 and n % init.size == 0:
                    factor = n // init.size
                    h_tmp = np.repeat(init, factor)
                    h_bin = (h_tmp > 0.5).astype(float)
                    cur_ones = int(h_bin.sum())
                    need = n // 2
                    if cur_ones != need:
                        if cur_ones > need:
                            idx1 = np.where(h_bin == 1)[0]
                            flip = rng.choice(idx1, size=cur_ones - need, replace=False)
                            h_bin[flip] = 0.0
                        else:
                            idx0 = np.where(h_bin == 0)[0]
                            flip = rng.choice(idx0, size=need - cur_ones, replace=False)
                            h_bin[flip] = 1.0
                    h0 = h_bin

            # improve via stochastic hill climb
            h_opt, val_opt = local_improve(h0, rng, dx, max_iters=2000)
            if val_opt < best_val:
                best_h, best_val, best_n = h_opt.copy(), val_opt, n

    # final polishing on the best candidate
    if best_h is not None:
        # a longer refinement pass
        h_ref, val_ref = local_improve(best_h, rng, 2.0 / best_n, max_iters=5000)
        if val_ref < best_val:
            best_h, best_val = h_ref.copy(), val_ref

    # safety fallback (should never happen)
    if best_h is None:
        n = 60
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_val = evaluate_h(best_h, 2.0 / n)

    final_c5 = evaluate_h(best_h, 2.0 / best_n)
    return best_h, float(final_c5), int(best_n)
