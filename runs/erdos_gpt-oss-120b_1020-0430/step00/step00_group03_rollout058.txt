def max_mismatch(x):
    n = len(x)
    # compute prefix sum array for quick sum of first k
    # (cumulative sums)
    prefix = np.concatenate(([0], np.cumsum(x))) # length n+1, prefix[i] = sum_{0..i-1}
    total = prefix[-1] # should be n/2
    # Overlap ones for each shift L
    overlap = np.array([np.dot(x[:n-L], x[L:]) for L in range(1, n)]) # shape n-1
    # prefix_ones for each L: sum_{0}^{n-1-L} x_i = prefix[n-L]
    prefix_ones = prefix[n - np.arange(1, n)]
    suffix_ones = total - prefix[:n-1]  # for L: suffix sum from L to n-1 = total - sum_{0}^{L-1}
    # That's array length n-1: suffix_ones[L] = total - prefix[L]
    # mismatches positive shift:
    M_pos = prefix_ones - overlap
    M_neg = suffix_ones - overlap
    # M may be negative? Should be >=0. But due to rounding maybe
    M = np.maximum(M_pos, M_neg)
    # also ensure non-negative:
    # M = np.maximum(M, 0)
    return int(np.max(M)), M

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, cvxpy as cp, time, math

    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n
    try:
        n = int(len(initial_h_values))
        init = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 70
        init = None

    # Build and solve MILP
    # If n > 80 maybe too heavy; but we expect n=70
    m = n // 2
    x = cp.Variable(n, boolean=True)
    Y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()

    constraints = [cp.sum(x) == m, t >= 0]

    # ensure Y diagonal equals x (optional)
    for i in range(n):
        constraints.append(Y[i,i] == x[i])

    # linearization constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(Y[i,j] <= x[i])
            constraints.append(Y[i,j] <= x[j])
            constraints.append(Y[i,j] >= x[i] + x[j] - 1)

    for s in range(1, n):
        prefix = cp.sum(x[:n-s])
        suffix = cp.sum(x[s:])
        overlap = cp.sum([Y[i, i+s] for i in range(n - s)])
        constraints.append(t >= prefix - overlap)
        constraints.append(t >= suffix - overlap)

    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        prob.solve(solver=cp.CBC, verbose=False, warm_start=True)
    except Exception:
        prob.solve(solver=cp.GLPK_MI, verbose=False)

    # Extract solution
    x_val = np.rint(np.array(x.value)).astype(int)
    # Fix any rounding errors to respect sum
    diff = m - x_val.sum()
    if diff != 0:
        # adjust by flipping some bits
        zeros = np.where(x_val == 0)[0]
        ones = np.where(x_val == 1)[0]
        if diff > 0:
            # need more ones
            idx = rng.choice(zeros, size=diff, replace=False)
            x_val[idx] = 1
        else:
            # need fewer ones
            idx = rng.choice(ones, size=-diff, replace=False)
            x_val[idx] = 0

    h = x_val.astype(float)

    dx = 2.0 / n
    c = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # if time remains, try local improvement (optional)
    # but we'll just return.

    return (h, float(c), n)

import time
import numpy as np
import cvxpy as cp

def _shift_vec(h, d):
    """Return vector h shifted by d (zero padding outside)."""
    n = h.shape[0]
    out = np.zeros_like(h)
    if d >= 0:
        out[: n - d] = h[d:]
    else:
        out[-d :] = h[: n + d]
    return out


def _c_max(h, dx):
    """Maximum overlap integral for a step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr


def _local_gradient_step(h, dx, step, rng):
    """Perform one gradient‑based transfer step."""
    n = h.shape[0]
    cval, corr = _c_max(h, dx)
    # shift attaining the maximum
    d = int(np.argmax(corr) - (n - 1))
    # subgradient of the active term
    sp = _shift_vec(h, d)
    sm = _shift_vec(h, -d)
    g = 1.0 - sp - sm
    # candidates for decreasing / increasing
    dec = np.where((g > 1e-12) & (h > 1e-12))[0]
    inc = np.where((g < -1e-12) & (h < 1.0 - 1e-12))[0]
    if dec.size == 0 or inc.size == 0:
        return h, cval, False
    i = dec[np.argmax(g[dec])]
    j = inc[np.argmin(g[inc])]
    delta = min(step, h[i], 1.0 - h[j])
    if delta <= 0:
        return h, cval, False
    h[i] -= delta
    h[j] += delta
    return h, cval, True
