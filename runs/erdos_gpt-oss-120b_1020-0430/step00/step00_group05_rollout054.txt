def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine size
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    except NameError:
        n = 120
        h0 = None
    target = n // 2
    # Build initial binary h
    if h0 is not None:
        # Binary from h0
        binary = (h0 > 0.5).astype(int)
        cur_sum = binary.sum()
        diff = target - cur_sum
        if diff > 0:
            zero_idx = np.where(binary == 0)[0]
            # sort by closeness to 0.5 descending
            order = zero_idx[np.argsort(-h0[zero_idx])]
            binary[order[:diff]] = 1
        elif diff < 0:
            one_idx = np.where(binary == 1)[0]
            order = one_idx[np.argsort(h0[one_idx])]
            binary[order[: -diff]] = 0
        h = binary.astype(float)
    else:
        h = np.zeros(n)
        h[:target] = 1.0
        rng.shuffle(h)
    current_val = max_overlap(h)
    best_h = h.copy()
    best_val = current_val
    # parameters
    max_temp = 0.3
    min_temp = 0.001
    stagnation_limit = 20000
    steps_without_improv = 0
    # main loop
    while True:
        now = time.time()
        if now - start > budget_s * 0.95:
            break
        # choose random swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        new_val = max_overlap(h)
        # temperature
        elapsed = now - start
        frac = elapsed / budget_s
        T = max_temp * (1 - frac) + min_temp * frac
        delta = new_val - current_val
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            current_val = new_val
            steps_without_improv = 0 if delta <= 0 else steps_without_improv + 1
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
            steps_without_improv += 1
        if current_val < best_val:
            best_val = current_val
            best_h = h.copy()
        # restart if stagnation
        if steps_without_improv > stagnation_limit:
            # random new state
            h = np.zeros(n)
            h[:target] = 1.0
            rng.shuffle(h)
            current_val = max_overlap(h)
            steps_without_improv = 0
    return best_h, best_val, n

import numpy as np, time, math

def _max_overlap(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---------- determine problem size ----------
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    except Exception:
        n = 70
        h0 = None
    if n % 2 != 0:                     # need even length for exact half measure
        n += 1
        if h0 is not None:
            h0 = np.append(h0, 0.0)

    target = n // 2                    # number of ones required

    # ---------- initial binary vector ----------
    if h0 is not None:
        bin_h = (h0 > 0.5).astype(int)
        cur = int(bin_h.sum())
        diff = target - cur
        if diff != 0:
            # sort by proximity to 0.5 to decide which bits to flip
            closeness = np.abs(h0 - 0.5)
            if diff > 0:
                zero_idx = np.where(bin_h == 0)[0]
                order = zero_idx[np.argsort(closeness[zero_idx])]
                bin_h[order[:diff]] = 1
            else:
                one_idx = np.where(bin_h == 1)[0]
                order = one_idx[np.argsort(-closeness[one_idx])]
                bin_h[order[: -diff]] = 0
        h = bin_h.astype(float)
    else:
        h = np.zeros(n)
        h[:target] = 1.0
        rng.shuffle(h)

    best_h = h.copy()
    best_val = _max_overlap(best_h)
    cur_h = h.copy()
    cur_val = best_val

    # ---------- optimization loop ----------
    max_temp = 0.30
    min_temp = 0.001
    stagnation_limit = max(20000, n * 300)
    stale = 0

    while time.time() - start < budget_s * 0.95:
        # compute current worst shift
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx_max = int(np.argmax(corr))
        t = idx_max - (n - 1)                     # shift (may be negative)
        # try a directed improvement based on the worst shift
        if t >= 0:
            seg_len = n - t
            left = cur_h[:seg_len]
            right = cur_h[t:]
        else:
            shift = -t
            seg_len = n - shift
            left = cur_h[shift:]
            right = cur_h[:seg_len]

        bad = np.where((left == 1) & (right == 0))[0]
        good = np.where((left == 0) & (right == 1))[0]

        if bad.size and good.size:
            i_loc = int(rng.choice(bad))
            j_loc = int(rng.choice(good))
            if t >= 0:
                i_glob, j_glob = i_loc, j_loc
            else:
                i_glob = i_loc + shift
                j_glob = j_loc
            cur_h[i_glob], cur_h[j_glob] = 0.0, 1.0
            new_val = _max_overlap(cur_h)
        else:
            # random swap fallback
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i_glob = int(rng.choice(ones))
            j_glob = int(rng.choice(zeros))
            cur_h[i_glob], cur_h[j_glob] = 0.0, 1.0
            new_val = _max_overlap(cur_h)

        # acceptance (simulated annealing)
        elapsed = time.time() - start
        frac = elapsed / budget_s
        T = max_temp * (1 - frac) + min_temp * frac
        delta = new_val - cur_val
        if delta <= 0 or (T > 0 and rng.random() < math.exp(-delta / T)):
            cur_val = new_val
            if delta > 0:
                stale += 1
            else:
                stale = 0
        else:
            # revert swap
            cur_h[i_glob], cur_h[j_glob] = 1.0, 0.0
            stale += 1

        # keep best solution
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur_h.copy()
            stale = 0
            if best_val <= 0.3808:
                break

        # restart if stuck too long
        if stale > stagnation_limit:
            cur_h[:] = 0.0
            cur_h[:target] = 1.0
            rng.shuffle(cur_h)
            cur_val = _max_overlap(cur_h)
            stale = 0

    final_val = _max_overlap(best_h)
    return best_h, final_val, n
