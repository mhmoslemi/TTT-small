def grad_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    p = np.argmax(corr)
    k = p - (n - 1)
    grad = np.zeros_like(h)
    # For each i where i + k in range, add (1 - h[i + k]) to grad[i]
    # For each i where i - k in range, subtract h[i - k] from grad[i]
    # coder: if k >= 0:
    #   i from 0 .. n-1-k: grad[i] += 1 - h[i + k]
    #   i from k .. n-1: grad[i] -= h[i - k]
    # else (k < 0):
    #   let kk = -k positive
    #   i from kk .. n-1: grad[i] += 1 - h[i - kk] (since i + k = i - kk)
    #   i from 0 .. n-1+ k: grad[i] -= h[i - k] (i - k = i + kk)
    # Let's unify using loops.
    if k >= 0:
        # term from (1 - h[i + k]) part
        i_end = n - k
        grad[:i_end] += 1 - h[k:]
        # term from -h[i - k] part
        grad[k:] -= h[:n - k]
    else:
        kk = -k
        # for (1 - h[i + k]) = (1 - h[i - kk]) for i >= kk
        grad[kk:] += 1 - h[:n - kk]
        # for -h[i - k] = -h[i + kk] for i < n - kk
        grad[:n - kk] -= h[kk:]
    # gradient of max_corr (without dx) is grad. Then gradient of objective = grad * dx
    return grad * (2.0 / n)   # because dx = 2/n

def _project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n while forcing sum(v)=target using a simple water‑filling."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        if w.sum() < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:  # very unlikely, adjust the first entry
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w


def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _max_corr_and_shift(h: np.ndarray):
    """Maximum correlation value (without dx) and the corresponding shift k."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    p = int(np.argmax(corr))
    n = h.shape[0]
    k = p - (n - 1)
    return corr[p], k


def _grad_of_c5(h: np.ndarray) -> np.ndarray:
    """Analytic gradient of C5 w.r.t. h (uses the shift attaining the max)."""
    n = h.shape[0]
    _, k = _max_corr_and_shift(h)
    grad = np.zeros_like(h)
    if k >= 0:
        # contribution from (1 - h[i+k])
        grad[: n - k] += 1.0 - h[k:]
        # contribution from -h[i-k]
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    dx = 2.0 / n
    return grad * dx


def _eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - TARGET


def _eq_constraint_jac(x: np.ndarray) -> np.ndarray:
    return np.ones_like(x)


def _local_optimize(x0: np.ndarray, maxiter: int = 100) -> np.ndarray:
    """Run a few SLSQP steps starting from x0."""
    n = x0.shape[0]
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _eq_constraint, "jac": _eq_constraint_jac}
    res = minimize(
        fun=lambda x: _c5_of(x),
        x0=x0,
        method="SLSQP",
        jac=_grad_of_c5,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": maxiter, "ftol": 1e-10, "disp": False},

    # enforce the integral exactly (possible tiny drift)
    if abs(h.sum() - TARGET) > 1e-8:

    return h

    """Draw a random point in the box [0,1]^n with the prescribed sum."""
    v = np.random.rand(n)
    return _project_to_bounds_and_sum(v, target)

    # ------------------------------------------------------------------ #
    maxiter_per_run = 80
    while time.time() - start < budget_s * 0.95:
        x0 = _random_feasible(n, TARGET)
        cand = _local_optimize(x0, maxiter=maxiter_per_run)
        cand_val = _c5_of(cand)
        if cand_val < best_val:
            best_h, best_val = cand, cand_val

    # Final safety projection (should already be satisfied)
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - TARGET) > 1e-8:
        best_h = _project_to_bounds_and_sum(best_h, TARGET)

    return best_h.tolist(), float(best_val), int(n)
