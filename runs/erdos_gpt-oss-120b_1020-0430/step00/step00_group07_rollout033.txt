            g[:n - k] -= h[k:]

def adjust_sum_to_target(h, target):
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # find indices that are not at bounds
    idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
    if idx.size == 0:
        # fallback all indices
        idx = np.arange(h.shape[0])
    i = idx[0]
    h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    # after fixing one point may still be off by small rounding, iterate
    if abs(h.sum() - target) > 1e-12:
        # distribute remaining diff uniformly among non‑boundary entries
        remaining = target - h.sum()
        nonb = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if nonb.size > 0:
            per = remaining / nonb.size
            h[nonb] += per
            h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible(n, rng):
    x = rng.random(n)
    target = n / 2.0
    s = target / x.sum()
    h = np.clip(x * s, 0.0, 1.0)
    h = adjust_sum_to_target(h, target)
    return h

import numpy as np, time, math

def compute_c5_and_corr(h):
    """Return C5, index of maximal correlation, and the full correlation array."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    c5 = float(max_val * (2.0 / h.shape[0]))
    return c5, idx, corr


def gradient_for_lag(h, lag):
    """Gradient of the worst‑lag loss with respect to h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        k = lag
        if k < n:
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
    else:
        k = -lag
        if k < n:
            g[k:] = 1.0 - h[: n - k]
            g[: n - k] -= h[k:]
    return g


def adjust_sum_to_target(h, target):
    """Shift a single free entry so that sum(h) equals target (exactly)."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # prefer an entry that is not at a bound
    idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
    if idx.size == 0:
        idx = np.arange(h.shape[0])
    i = idx[0]
    h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    # if a tiny residual remains, spread it uniformly among free entries
    if abs(h.sum() - target) > 1e-12:
        rem = target - h.sum()
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            per = rem / free.size
            h[free] += per
            h = np.clip(h, 0.0, 1.0)
