    dx = 2/n

            break

    n = h.size
    target = n/2
    # clip first
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    diff = target - total

    if diff > 0:
        # need increase sum: increase entries where h < 1

import numpy as np, time, math, random

def compute_corr_full(h: np.ndarray) -> np.ndarray:
    """Full cross‑correlation of h with (1‑h)."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for the discrete step function."""
    n = h.size
    dx = 2.0 / n
    return compute_corr_full(h).max() * dx

def random_feasible_vector(n: int) -> np.ndarray:
    """Return a feasible h (0≤h≤1, Σh = n/2). For odd n one entry is 0.5."""
    target = n / 2.0
    h = np.zeros(n, dtype=np.float64)
    if n % 2 == 0:
        ones = n // 2
        h[:ones] = 1.0
    else:
        ones = n // 2
        h[:ones] = 1.0
        h[ones] = 0.5
    np.random.shuffle(h)
    return h

def weighted_choice(weights: np.ndarray) -> int:
    """Return an index sampled proportionally to non‑negative weights."""
    total = weights.sum()
    if total == 0:
        return -1
    probs = weights / total
    return np.random.choice(len(weights), p=probs)

def local_search(h_start: np.ndarray, time_limit: float) -> tuple:
    """Hill‑climbing / simulated‑annealing on the max‑overlap objective."""
    start = time.time()
    n = h_start.size
    dx = 2.0 / n
    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # parameters
    step = 0.2                     # typical move size
    prob_target = 0.7              # try to attack the worst shift
    min_step = 1e-4
    it = 0

    while time.time() - start < time_limit:
        # correlation and worst shift
        corr = compute_corr_full(cur_h)
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)      # shift corresponding to the maximum value

        # decide move type
        if random.random() < prob_target:
            # directed move from a high‑contributor to a low‑contributor
            if k >= 0:
                length = n - k
                contrib = cur_h[:length] * (1.0 - cur_h[k:])
                src = weighted_choice(contrib)
                if src >= 0:
                    i, j = src, src + k
                else:
                    i = j = -1
            else:
                p = -k
                length = n - p
                contrib = cur_h[p:] * (1.0 - cur_h[:length])
                src = weighted_choice(contrib)
                if src >= 0:
                    i, j = src + p, src
                else:
                    i = j = -1
            if i >= 0 and i != j:
                max_eps = min(step, cur_h[i], 1.0 - cur_h[j])
                if max_eps > 0:
                    eps = random.random() * max_eps
                    new_h = cur_h.copy()
                    new_h[i] -= eps
                    new_h[j] += eps
                else:
                    new_h = None
            else:
                new_h = None
        else:
            # random swap of mass between two positions
            i, j = np.random.randint(0, n, 2)
            if i == j:
                new_h = None
            else:
                # feasible epsilon interval
                hi, hj = cur_h[i], cur_h[j]
                max_eps = min(step, hi, 1.0 - hj)
                min_eps = -min(step, hj, 1.0 - hi)
                if max_eps <= min_eps:
                    new_h = None
                else:
                    eps = (random.random() * 2 - 1) * step
                    eps = max(min(eps, max_eps), min_eps)
                    if abs(eps) < 1e-12:
                        new_h = None
                    else:
                        new_h = cur_h.copy()
                        new_h[i] = hi + eps
                        new_h[j] = hj - eps

        if new_h is None:
            it += 1
            continue

        new_val = compute_c5(new_h)

        # simulated‑annealing acceptance
        elapsed = time.time() - start
        T = max(0.001, (time_limit - elapsed) / time_limit * 0.01)
        if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / T):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val

        # step size adaptation
        if it % 5000 == 0 and it > 0:
            if best_val < cur_val:          # progress made
                step = min(0.5, step * 1.02)
            else:
                step = max(min_step, step * 0.96)
        it += 1

        # early exit if we have already reached the target
        if best_val <= 0.3808:
            break

    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    random.seed(seed)
    np.random.seed(seed)

    # Starting vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 100
        h0 = random_feasible_vector(n)

    best_h, best_val = h0.copy(), compute_c5(h0)

    start_total = time.time()
    time_remaining = budget_s

    # Primary search from the supplied (or generated) start
    primary_time = budget_s * 0.6
    h_opt, val_opt = local_search(best_h, primary_time)
    if val_opt < best_val:
        best_h, best_val = h_opt, val_opt

    # Additional random restarts
    time_elapsed = time.time() - start_total
    time_remaining = budget_s - time_elapsed
    if time_remaining > 5.0:
        restarts = 5
        per_restart = max(1.0, time_remaining / restarts)
        n = best_h.size
        for _ in range(restarts):
            h_rand = random_feasible_vector(n)
            h_opt, val_opt = local_search(h_rand, per_restart)
            if val_opt < best_val:
                best_h, best_val = h_opt, val_opt
            if best_val <= 0.3808:
                break

    # final sanity: enforce bounds (tiny numerical drift)
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_h.size / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny remainder to a non‑saturated entry
        free = np.where((best_h > 0.0) & (best_h < 1.0))[0]
        if free.size == 0:
            free = np.arange(best_h.size)
        idx = free[0]
        best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)

    final_val = compute_c5(best_h)
    return best_h.tolist(), final_val, best_h.size
