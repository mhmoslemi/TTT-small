def project_onto_box_sum(v, target):
    lo = -1e6; hi = 1e6
    for _ in range(50):
        mid = (lo+hi)/2
        u = np.clip(v - mid, 0.0, 1.0)

        if s > target:
            lo = mid
        else:
            hi = mid

    diff = target - u.sum()
    if abs(diff) > 1e-12:

        if mask.any():

def gradient_shift(h, s):
    n = len(h)
    term1 = np.ones_like(h)
    term2 = np.zeros_like(h)
    if s >= 0:
        term1[:n - s] = 1 - h[s:]
        term2[s:] = h[:n - s]
    else:
        ss = -s
        term1[ss:] = 1 - h[:n - ss]
        term2[:n - ss] = h[ss:]
    return term1 - term2

import time
import numpy as np

def _compute_c5(h, n):
    """Return the C5 value for a given vector."""
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()


def _gradient_shift(h, shift):
    """Sub‑gradient of the worst‑case overlap for a given integer shift."""
    n = h.shape[0]
    term1 = np.ones_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:
        # 1 - h[i+shift] where i+shift is in range, else 1
        term1[: n - shift] = 1.0 - h[shift:]
        # h[i-shift] where i-shift is in range, else 0
        term2[shift:] = h[: n - shift]
    else:
        s = -shift
        term1[s:] = 1.0 - h[: n - s]
        term2[: n - s] = h[s:]
    return term1 - term2


def _project_box_sum(v, target, lo=0.0, hi=1.0):
    """
    Euclidean projection of v onto {x : lo <= x_i <= hi , sum x_i = target}.
    Uses a simple binary search on the Lagrange multiplier.
    """
    low = -10.0
    high = 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        u = np.clip(v - mid, lo, hi)
        s = u.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, lo, hi)


def _subgradient_optimize(start_h, n, target, time_limit, start_time):
    """
    Perform sub‑gradient descent starting from start_h.
    Stops when time_limit (seconds) is exhausted.
    Returns the best (h, C5) found during this run.
    """
    h = start_h.copy()
    best_h = h.copy()
    best_val = _compute_c5(h, n)
    it = 1
    while True:
        if time.time() - start_time > time_limit:
            break
        # find shift that attains the current maximum
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)

        # sub‑gradient for that shift
        grad = _gradient_shift(h, shift)

        # diminishing step size (classic sub‑gradient rule)
        step = 0.5 / np.sqrt(it)

        # gradient step + projection back to the feasible set
        v = h - step * grad
        v = np.clip(v, 0.0, 1.0)
        h = _project_box_sum(v, target)

        val = _compute_c5(h, n)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        it += 1
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C5 value."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95   # keep a safety margin

    # ------------------------------------------------------------
    # initialise problem size and a feasible starting vector
    # ------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)   # provided by the harness
        n = init.size
        target_sum = n / 2.0
        h0 = np.clip(init, 0.0, 1.0)
        h0 = _project_box_sum(h0, target_sum)
    except NameError:
        # no preset vector – fall back to a random start of moderate size
        n = 120
        target_sum = n / 2.0
        h0 = rng.random(n)
        h0 = _project_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = _compute_c5(best_h, n)

    # ------------------------------------------------------------
    # main search: several independent restarts with sub‑gradient descent
    # ------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline:
        restart_counter += 1

        # first restart preferentially uses the given construction (if any)
        if restart_counter == 1 and 'initial_h_values' in globals():
            start_vec = h0
        else:
            start_vec = rng.random(n)
            start_vec = _project_box_sum(start_vec, target_sum)

        # run a short sub‑gradient optimisation
        local_h, local_val = _subgradient_optimize(
            start_vec, n, target_sum, deadline, time.time()
        )
        if local_val < best_val:
            best_val = local_val
            best_h = local_h.copy()

        # early exit if we already beat the target record
        if best_val < 0.3808:
            break

    # final housekeeping: ensure constraints are exactly met one last time
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h, n)

    return (best_h.tolist(), float(final_c5), n)
