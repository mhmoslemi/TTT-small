def project_to_box_sum(y, S, lo=0., hi=1., tol=1e-9, max_iter=100):
    # binary search λ
    low = -1e6
    high = 1e6
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        x = np.clip(y + lam, lo, hi)
        s = x.sum()
        if np.abs(s - S) < tol:
            return x
        if s > S:
            high = lam
        else:
            low = lam
    return np.clip(y + (low + high)/2.0, lo, hi)

    n = len(h)
    dx = 2.0 / n
    C_raw = np.correlate(h, 1 - h, mode='full')

def smooth_obj_and_grad(h, eps):
    # returns function value = maxC (approx) and gradient
    n = h.shape[0]
    dx = 2.0 / n
    raw = np.correlate(h, 1 - h, mode='full')
    C = raw * dx
    maxC = np.max(C)
    # compute softmax weights
    exp_vals = np.exp((C - maxC) / eps)  # stable
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    # gradient contributions
    grad = np.zeros_like(h)
    # iterate over shifts
    for idx, r in enumerate(range(-n+1, n)):
        if r >= 0:
            grad[:n - r] += w[idx] * (1 - h[r:])
            grad[r:] -= w[idx] * h[:n - r]
        else:
            s = -r
            grad[s:] += w[idx] * (1 - h[:n - s])
            grad[:n - s] -= w[idx] * h[s:]
    grad = grad * dx
    # smooth objective f = maxC + eps * log(sum_exp)
    f = maxC + eps * np.log(sum_exp)
    return f, grad

import numpy as np, time, math

def _project_box_sum(y, S, lo=0.0, hi=1.0, tol=1e-9, max_iter=60):
    lo_lam = lo - np.max(y) - 1.0
    hi_lam = hi - np.min(y) + 1.0
    for _ in range(max_iter):
        lam = (lo_lam + hi_lam) * 0.5
        x = np.clip(y + lam, lo, hi)
        s = x.sum()
        if abs(s - S) <= tol:
            return x
        if s > S:
            hi_lam = lam
        else:
            lo_lam = lam
    return np.clip(y + (lo_lam + hi_lam) * 0.5, lo, hi)

def _random_feasible(n):
    y = np.random.rand(n)
    return _project_box_sum(y, n / 2.0)

def _evaluate(h):
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        dx = 2.0 / h.size
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def _smooth_obj_grad(h, eps):
    n = h.size
    dx = 2.0 / n
    raw = np.correlate(h, 1 - h, mode='full')
    C = raw * dx
    m = C.max()
    expv = np.exp((C - m) / eps)
    w = expv / expv.sum()
    grad = np.zeros_like(h)
    for k, wk in zip(range(-n + 1, n), w):
        if k >= 0:
            grad[: n - k] += wk * (1 - h[k:])
            grad[k:] -= wk * h[: n - k]
        else:
            s = -k
            grad[s:] += wk * (1 - h[: n - s])
            grad[: n - s] -= wk * h[s:]
    grad *= dx
    f = m + eps * np.log(expv.sum())
    return f, grad

def _gd_smooth(h0, deadline):
    h = h0.copy()
    n = h.size
    S = n / 2.0
    eps = 0.02
    best_h = h.copy()
    best_val = _evaluate(h)
    it = 0
    while time.time() < deadline:
        f, g = _smooth_obj_grad(h, eps)
        eta = 0.1 / np.sqrt(it + 1)
        h = h - eta * g
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, S)
        cur = _evaluate(h)
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
        it += 1
        if it % 500 == 0 and eps > 0.004:
            eps *= 0.8
    return best_h, best_val

def _upscale(h, new_n):
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
