def project_onto_simplex_box(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    # v: array
    # return projection onto [lower, upper] and sum target.
    n = len(v)
    lo = np.min(v - upper) - 1.0  # lower bound for lambda (ensuring all y=upper)
    hi = np.max(v - lower) + 1.0   # upper bound for lambda (ensuring all y=lower)
    # Actually need correct sign: y_i = clip(v_i - λ, lower, upper)
    # So if λ is very negative, v_i - λ large => clip to upper => sum = n*upper
    # if λ is very positive, v_i - λ small => clip to lower => sum = n*lower
    # So target between that.
    # We'll binary search in range [min_possible_lambda, max_possible_lambda].
    lo = np.min(v - upper) - 1.0  # ensures v_i - λ >= upper for all i => λ <= min(v_i - upper)
    hi = np.max(v - lower) + 1.0  # ensures v_i - λ <= lower for all i => λ >= max(v_i - lower)
    # Actually we need lo < hi.
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(v - mid, lower, upper)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            # sum too high -> need to increase lambda (so subtract more)
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, lower, upper)

import time
import math
import numpy as np

def _project_onto_box_sum(v: np.ndarray, target: float,
                          lower: float = 0.0, upper: float = 1.0,
                          max_iter: int = 60, tol: float = 1e-9) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x : lower <= x_i <= upper, sum(x)=target}.
    Uses a simple binary search on the Lagrange multiplier λ where
        y_i = clip(v_i - λ, lower, upper).
    """
    lo = -np.max(v) - 10.0   # ensures y_i = upper for all i -> sum = n*upper >= target
    hi = np.max(v) + 10.0    # ensures y_i = lower for all i -> sum = n*lower <= target
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(v - mid, lower, upper)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, lower, upper)


def _evaluate(h: np.ndarray) -> float:
    """Compute C5 for a given vector h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * (2.0 / n)


def _subgradient(h: np.ndarray):
    """
    Return a subgradient of the max-overlap functional and the current maximal value.
    The subgradient corresponds to the average of gradients for all active lags.
    """
    n = h.shape[0]
    b = 1.0 - h
    conv = np.correlate(h, b, mode='full')
    max_val = conv.max()
    # consider all lags whose value is within a tiny tolerance of the max
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(h)
    for idx in active:
        s = idx - (n - 1)          # shift amount
        if s >= 0:
            # contribution of b[i+s] for i = 0..n-s-1
            grad[:n - s] += b[s:]
            # contribution of -h[i-s] for i = s..n-1
            grad[s:] -= h[:n - s]
        else:
            t = -s
            # b[i-s] = b[i-t] for i = t..n-1
            grad[t:] += b[:n - t]
            # -h[i-s] = -h[i+t] for i = 0..n-t-1
            grad[:n - t] -= h[t:]
    if active.size:
        grad /= float(active.size)
    return grad, max_val
