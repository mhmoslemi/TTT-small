def project_onto_box_sum(x, low, high, target):
    # solves min ||y - x||^2 s.t. low <= y <= high, sum y = target
    # using water-filling (Lagrange multiplier).
    # We'll use bisection on lambda

        else:

def project_onto_box_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-9:
            break
        # indices that are not at bounds
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if not np.any(mask):
            # all at bounds, can't adjust further
            break
        adjust = diff / np.count_nonzero(mask)
        x[mask] += adjust
        # clip again
        x = np.clip(x, 0.0, 1.0)
    # final: If still diff, we might need to distribute among all values
    # compute final diff and adjust a single element.
    if abs(x.sum() - target) > 1e-8:
        # adjust a random non-bound element if possible
        # fallback: scale all elements proportionally
        d = target - x.sum()
        # distribute equally among free
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if np.any(mask):
            x[mask] += d / np.count_nonzero(mask)
            x = np.clip(x, 0.0, 1.0)
        else:
            # all at bounds, adjust the first element within bounds
            idx = np.argmax(x)
            x[idx] = np.clip(x[idx] + d, 0.0, 1.0)

import time
import numpy as np

def _c5_value(h):
    """Return C5 value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr

def _worst_shift(corr):
    """From full correlation array return integer shift that gives max value."""
    n = (corr.shape[0] + 1) // 2
    idx = int(corr.argmax())
    return idx - (n - 1)

def _gradient_worst_shift(h, shift):
    """Subgradient of the worst overlap term (vector of size len(h))."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        # plus term (1 - h[i+k]) for i where i+k < n, else 1
        if k < n:
            g[:n - k] += 1.0 - h[k:]
        g[n - k:] += 1.0
        # minus term -h[i-k] for i >= k
        if k < n:
            g[k:] -= h[:n - k]
    else:
        s = -shift
        # plus term (1 - h[i-s]) for i >= s, else 1
        if s < n:
            g[s:] += 1.0 - h[:n - s]
        g[:s] += 1.0
        # minus term -h[i+s] for i < n-s
        if s < n:
            g[:n - s] -= h[s:]
    return g

def _project_box_sum(x, target):
    """Euclidean projection of x onto [0,1]^n with sum == target."""
    low, high = 0.0, 1.0
    x = np.clip(x, low, high)
    # water‑filling via a simple bisection on the Lagrange multiplier
    lo = np.min(x) - high - 1.0   # ensures all clipped to 1
    hi = np.max(x) - low + 1.0    # ensures all clipped to 0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, low, high)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - (lo + hi) * 0.5, low, high)
    # tiny correction for numerical drift
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        free = (y > low + 1e-12) & (y < high - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, low, high)
    return y

def _local_optimize(h0, max_iters, budget, start_time):
    """Gradient‑descent from start h0 respecting the constraints."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h, best_val, _ = _c5_value(h)
    alphas = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    stagnation = 0
    for it in range(max_iters):
        if time.time() - start_time > budget - 0.5:
            break
        cval, corr = _c5_value(h)
        if cval < best_val:
            best_val, best_h = cval, h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 30:
                break
        shift = _worst_shift(corr)
        grad = _gradient_worst_shift(h, shift)
        improved = False
        for a in alphas:
            h_new = _project_box_sum(h - a * grad, target)
            new_c, _ = _c5_value(h_new)
            if new_c < cval - 1e-12:
                h = h_new
                improved = True
                break
        if not improved:
            # small random kick to escape flat regions
            noise = (np.random.rand(n) - 0.5) * 0.02
            h = _project_box_sum(h + noise, target)
    return best_h, best_val

def _interpolate(h, new_n):
    """Linear interpolation of a step vector onto a finer grid."""
    old_n = h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) meeting the time budget."""
    np.random.seed(seed)
    start = time.time()
    # primary resolution
    n_points = 200                     # moderate size, even
    target_sum = n_points / 2.0
    best_h = None
    best_c = float('inf')

    # use the supplied initial_h_values if present
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        h0 = _interpolate(init, n_points)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project_box_sum(h0, target_sum)
        c0, _ = _c5_value(h0)
        best_h, best_c = h0, c0

    # main random‑restart optimisation loop
    restarts = 0
    while time.time() - start < budget_s - 4.0:
        # generate a random start (binary half‑ones or continuous)
        if restarts % 2 == 0:
            h0 = np.random.rand(n_points)
        else:
            h0 = np.zeros(n_points)
            half = int(target_sum)
            h0[:half] = 1.0
            np.random.shuffle(h0)
        h0 = _project_box_sum(h0, target_sum)

        # run a short local optimisation
        h_opt, c_opt = _local_optimize(
            h0,
            max_iters=2500,
            budget=budget_s,
            start_time=start,
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        restarts += 1

    # optional refinement at doubled resolution if time permits
    remaining = budget_s - (time.time() - start)
    if remaining > 250 and n_points < 500:
        finer = n_points * 2
        h_fine = _interpolate(best_h, finer)
        h_fine = _project_box_sum(h_fine, finer / 2.0)
        c_fine, _ = _c5_value(h_fine)
        if c_fine < best_c:
            best_c, best_h = c_fine, h_fine.copy()
        # one extra local search on the finer grid
        h_opt, c_opt = _local_optimize(
            best_h,
            max_iters=2000,
            budget=budget_s,
            start_time=time.time(),
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        n_points = finer

    # final safety projection (enforce constraints exactly)
    best_h = _project_box_sum(best_h, n_points / 2.0)
    best_c, _ = _c5_value(best_h)

    return best_h.tolist(), float(best_c), int(n_points)
