def simulated_annealing(h, time_deadline, T0=0.02, T_end=0.0005):
    n = h.shape[0]
    cur_val = evaluate_h(h)
    best_h = h.copy()
    best_val = cur_val
    target = n / 2.0
    it = 0
    while time.time() < time_deadline:
        it += 1
        # linear schedule
        frac = it / 200000.0
        if frac > 1.0:
            frac = 1.0
        T = T0 * (1 - frac) + T_end * frac
        i, j = np.random.randint(n, size=2)
        while i == j:
            j = np.random.randint(n)
        hi, hj = h[i], h[j]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue
        # step magnitude scaled by T
        max_step = (delta_max - delta_min) * (T / T0)
        if max_step <= 0:
            continue
        delta = np.random.uniform(-max_step/2, max_step/2)
        # clamp
        if delta < delta_min:
            delta = delta_min
        elif delta > delta_max:
            delta = delta_max
        # apply move
        h[i] = hi + delta
        h[j] = hj - delta
        new_val = evaluate_h(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # Metropolis
            if np.exp(-(new_val - cur_val) / max(T, 1e-12)) > np.random.rand():
                cur_val = new_val
            else:
                # revert
                h[i] = hi
                h[j] = hj
        # occasionally re-evaluate best after many steps maybe not needed
    return best_h, best_val

import numpy as np
import random
import time
import math

def _evaluate_h(h: np.ndarray) -> float:
    """Return C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _project_onto_box_and_sum(v: np.ndarray, target: float,
                              lo: float = 0.0, hi: float = 1.0,
                              max_iter: int = 100) -> np.ndarray:
    """Euclidean projection of v onto {x: lo<=x<=hi, sum(x)=target}."""
    v = np.clip(v, lo, hi)
    if abs(v.sum() - target) <= 1e-12:
        return v
    lam_lo, lam_hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lam_lo + lam_hi) / 2.0
        proj = np.clip(v - lam, lo, hi)
        s = proj.sum()
        if s > target:
            lam_lo = lam
        else:
            lam_hi = lam
        if abs(s - target) < 1e-12:
            break
    return np.clip(v - lam, lo, hi)

def _simulated_annealing(h: np.ndarray, deadline: float,
                         T0: float = 0.02, T_end: float = 0.0005):
    """Simple SA that preserves sum(h) and box constraints."""
    n = h.shape[0]
    cur_val = _evaluate_h(h)
    best_h = h.copy()
    best_val = cur_val
    it = 0
    while time.time() < deadline:
        it += 1
        # linear temperature schedule (200k steps ≈ full decay)
        frac = min(1.0, it / 200000.0)
        T = T0 * (1.0 - frac) + T_end * frac

        i, j = np.random.randint(n, size=2)
        while i == j:
            j = np.random.randint(n)

        hi, hj = h[i], h[j]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue

        max_step = (delta_max - delta_min) * (T / T0)
        if max_step <= 0:
            continue
        delta = np.random.uniform(-max_step / 2.0, max_step / 2.0)
        if delta < delta_min:
            delta = delta_min
        elif delta > delta_max:
            delta = delta_max

        h[i] = hi + delta
        h[j] = hj - delta
        new_val = _evaluate_h(h)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # Metropolis acceptance

                cur_val = new_val
            else:
                h[i] = hi
                h[j] = hj
