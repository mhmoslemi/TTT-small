import time, math, numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear (non‑periodic) correlation of h with its complement
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a small C5 bound.

    Returns
    -------
    tuple (h_values, c5_bound, n_points)
        h_values – NumPy array of length n_points with values in [0,1] and
                    sum(h)*dx == 1  (dx = 2/n_points)
        c5_bound – the best C5 value found for the returned h
        n_points – the number of samples used (size of h_values)
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a working resolution.
    # If a parent construction is supplied we reuse its size,
    # otherwise we start with a modest 140‑point grid.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    else:
        n = 140
        h = np.full(n, 0.5, dtype=float)   # uniform start (integral = 1)

    # enforce the integral constraint exactly (sum = n/2)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # adjust the first element to fix the sum while keeping bounds
        delta = target_sum - cur_sum
        h[0] = np.clip(h[0] + delta, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search.
    # We keep the total mass fixed by moving a small amount from one
    # coordinate to another (or swapping a tiny fraction of mass).
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    cur_h = h
    cur_c5 = best_c5

    start = time.time()
    # total number of iterations – we stop earlier when time runs out
    max_iter = 5_000_000

    # temperature schedule (linear cooling)
    T_start = 0.15
    T_end = 1e-5

    # typical step size (max amount transferred in one move)
    max_step = 0.12

    for it in range(max_iter):
        # ------------------------------------------------------------------
        # time check – leave a small margin so the harness does not kill us
        # ------------------------------------------------------------------
        if time.time() - start > budget_s * 0.94:
            break

        # linear interpolation for temperature
        t_frac = it / max_iter
        T = (1.0 - t_frac) * T_start + t_frac * T_end

        # ------------------------------------------------------------------
        # propose a move: transfer a random amount from i -> j
        # ------------------------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # decide direction (i -> j) or (j -> i)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # maximum transferable amount limited by bounds and a global step size
        avail = cur_h[src]
        room = 1.0 - cur_h[dst]
        max_transfer = min(max_step, avail, room)
        if max_transfer <= 0.0:
            continue

        delta = rng.random() * max_transfer
        cur_h[src] -= delta
        cur_h[dst] += delta

        new_c5 = _c5_value(cur_h)

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        improve = new_c5 < cur_c5
        if improve or rng.random() < math.exp((cur_c5 - new_c5) / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # reject – revert the move
            cur_h[src] += delta
            cur_h[dst] -= delta

        # occasional randomisation to escape deep local minima
        if it % 200_000 == 0 and rng.random() < 0.03:
            # re‑initialise a small fraction of entries uniformly
            mask = rng.random(n) < 0.15
            cur_h[mask] = rng.random(mask.sum())
            # renormalise to maintain the exact total mass
            excess = cur_h.sum() - target_sum
            if abs(excess) > 1e-12:
                # distribute the correction uniformly over all entries
                cur_h -= excess / n
                # clamp to bounds again
                np.clip(cur_h, 0.0, 1.0, out=cur_h)

    # final sanity check – enforce the integral constraint (tiny numerical drift)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        correction = target_sum - cur_sum
        best_h[0] = np.clip(best_h[0] + correction, 0.0, 1.0)

    return best_h, float(best_c5), n
