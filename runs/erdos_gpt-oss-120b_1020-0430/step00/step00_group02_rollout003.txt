def project_to_bounds_sum(x, target):
    y = np.clip(x, 0.0, 1.0)
    # Compute deficit/excess
    total = y.sum()
    if abs(total - target) < 1e-12:
        return y
    # Determine free indices that are not at bounds
    # We'll repeatedly adjust equally among free entries.
    # This may converge in few steps because after each iteration some entries become saturated.
    while True:
        total = y.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (y > 0 + 1e-12) & (y < 1 - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            # can't adjust further, break
            break
        adjust = diff / n_free
        # Update free entries
        y[free] += adjust
        # Clip again to avoid going out of bounds
        y = np.clip(y, 0.0, 1.0)
    return y
```

import time
import numpy as np

def _project_to_bounds_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while fixing the sum to target."""
    y = np.clip(x, 0.0, 1.0)
    # water‑filling style projection
    for _ in range(10):  # enough iterations for convergence
        total = y.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            break
        adjust = diff / n_free
        y[free] += adjust
        y = np.clip(y, 0.0, 1.0)
    return y


def _local_search(initial_h: np.ndarray, budget: float) -> tuple:
    """Simulated‑annealing style local optimisation of the overlap bound."""
    n = initial_h.shape[0]
    target_sum = n / 2.0
    cur_h = initial_h.copy()
    cur_val = evaluate_erdos_solution(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    # iteration budget – will be cut by elapsed time
    max_iter = 3_000_000
    max_temp = 5e-3
    min_temp = 1e-6
    max_step_frac = 0.4  # fraction of the admissible transfer per move

    for it in range(1, max_iter + 1):
        if time.time() - start > budget:
            break

        # temperature schedule (linear)
        progress = it / max_iter
        temp = max_temp * (1.0 - progress) + min_temp * progress

        # propose a transfer of mass between two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if np.random.rand() < 0.5:
            # move from i to j
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer <= 0.0:
                continue
            eps = max_transfer * max_step_frac * np.random.rand()
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # move from j to i
            max_transfer = min(cur_h[j], 1.0 - cur_h[i])
            if max_transfer <= 0.0:
                continue
            eps = max_transfer * max_step_frac * np.random.rand()
            new_h = cur_h.copy()
            new_h[j] -= eps
            new_h[i] += eps

        new_val = evaluate_erdos_solution(new_h)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / temp):
            cur_h = new_h
            cur_val = new_val

        # keep the global best
        if new_val < best_val:
            best_h = new_h
            best_val = new_val

    best_h = _project_to_bounds_sum(best_h, target_sum)
    best_val = evaluate_erdos_solution(best_h)
