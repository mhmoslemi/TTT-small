def compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    # correlation of h with (1 - h), mode='full'
    corr = np.correlate(h, 1.0 - h, mode='full')

        rng.shuffle(h0)

    np.clip(h0, 0.0, 1.0, out=h0)

        # simple projection onto hyperplane while staying in [0,1]
        diff = target_sum - cur_sum
        # distribute diff to first element that can absorb it
        for idx in range(n):
            new_val = h0[idx] + diff
            if 0.0 <= new_val <= 1.0:
                h0[idx] = new_val
                break

    current_h = best_h.copy()
    current_score = best_score
    start = time.time()

    # algorithm parameters
    max_delta = 0.5          # maximal amount transferred in one move
    prob_directed = 0.7      # chance to use gradient‑guided move
    while True:
        now = time.time()
        if now - start > limit:
            break
        # correlation of current solution (also gives worst shift)
        corr = np.correlate(current_h, 1.0 - current_h, mode='full')
        cur_idx = int(np.argmax(corr))
        cur_score = float(corr[cur_idx]) * (2.0 / n)

        # temperature for simulated‑annealing acceptance
        elapsed = now - start

import numpy as np, time, math

def compute_c5(h):
    """Maximum overlap integral for a step function h on [0,2]."""
    n = h.shape[0]
    # correlation of h with (1‑h) (zero‑padded outside the interval)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)  # provided by the harness
        n = h0.size
    except Exception:
        # fall back to a simple binary construction
        n = 200
        h0 = np.zeros(n, dtype=np.float64)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)

    target_sum = n / 2.0
    # enforce bounds and the integral constraint
    np.clip(h0, 0.0, 1.0, out=h0)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        for idx in range(n):
            new_val = h0[idx] + diff
            if 0.0 <= new_val <= 1.0:
                h0[idx] = new_val
                break

    best_h = h0.copy()
    best_score = compute_c5(best_h)
    current_h = best_h.copy()
    current_score = best_score

    # ------------------------------------------------------------------
    # optimisation loop (simulated‑annealing + gradient‑guided moves)
    # ------------------------------------------------------------------
    start = time.time()
    limit = budget_s * 0.95          # leave a safety margin
    max_delta = 0.5                  # biggest amount transferred in one move
    prob_directed = 0.7              # probability of using the gradient‑guided step

    while True:
        now = time.time()
        if now - start > limit:
            break

        # correlation of the current solution – also gives the worst shift
        corr = np.correlate(current_h, 1.0 - current_h, mode="full")
        cur_idx = int(np.argmax(corr))
        cur_score = float(corr[cur_idx]) * (2.0 / n)
        shift = cur_idx - (n - 1)               # signed shift attaining the maximum

        # temperature for the Metropolis acceptance test
        elapsed = now - start
        T = max(1e-6, 1.0 - elapsed / limit)

        # --------------------------------------------------------------
        # 1) gradient‑guided transfer (most often used)
        # --------------------------------------------------------------
        if rng.random() < prob_directed:
            g = np.ones_like(current_h)
            if shift == 0:
                g = 1.0 - 2.0 * current_h
            else:
                idx = np.arange(n)
                right = idx + shift
                mask = (right >= 0) & (right < n)
                g[mask] -= current_h[right[mask]]
                left = idx - shift
                mask = (left >= 0) & (left < n)
                g[mask] -= current_h[left[mask]]

            i_max = int(np.argmax(g))   # decreasing this index tends to lower the worst overlap
            i_min = int(np.argmin(g))   # increasing this index tends to lower the worst overlap
            if i_max != i_min:
                delta = min(max_delta, current_h[i_max], 1.0 - current_h[i_min])
                if delta > 1e-12:
                    new_h = current_h.copy()
                    new_h[i_max] -= delta
                    new_h[i_min] += delta
                    new_score = compute_c5(new_h)
                    if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
                        current_h = new_h
                        current_score = new_score
                        if new_score < best_score:
                            best_score = new_score
                            best_h = new_h.copy()
                        continue   # go to next iteration

        # --------------------------------------------------------------
        # 2) random transfer move (fallback)
        # --------------------------------------------------------------
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        inc = min(max_delta, 1.0 - current_h[i], current_h[j])   # max we can add to i
        dec = min(max_delta, current_h[i], 1.0 - current_h[j])   # max we can subtract from i
        if inc + dec <= 0.0:
            continue
        delta = rng.random() * (inc + dec) - dec                # uniform in [-dec, inc]
        if abs(delta) < 1e-12:

        new_h[i] += delta
        new_h[j] -= delta
        new_score = compute_c5(new_h)
