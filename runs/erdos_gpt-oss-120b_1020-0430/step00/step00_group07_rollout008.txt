def project_onto_box_sum(x, S):
    # clip into [0,1] after adding same lambda to each element.
    # We need to find λ so that sum_i clip(x_i + λ, 0, 1) = S.
    # monotonic: as λ increases, sum increases (clipped).
    # Use binary search between low = -max(x) (makes all zeros) and high = 1 - min(x) (makes all ones)
    lo = -np.max(x)  # ensures all x+lambda <=0 => clips to 0
    hi = 1 - np.min(x)  # ensures all x+λ >=1 => clips to 1
    for _ in range(50):
        lam = (lo + hi) / 2
        projected = np.clip(x + lam, 0, 1)
        s = projected.sum()
        if s < S:
            lo = lam
        else:
            hi = lam
    return np.clip(x + (lo+hi)/2, 0, 1)
```

    import numpy as np, time, math
    start = time.time()
    rng = np.random.default_rng(seed)

    # Determine n
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
    else:
        n = 200
        init = None

    # build starting binary vector
    def make_start():
        if init is not None:
            # use init as basis, threshold >0.5, then adjust to half ones
            h = (init > 0.5).astype(float)
            # adjust to exactly half ones
            required = n // 2
            current = int(h.sum())
            if current > required:
                ones_idx = np.where(h == 1)[0]
                rng.shuffle(ones_idx)
                h[ones_idx[:current - required]] = 0
            elif current < required:
                zeros_idx = np.where(h == 0)[0]
                rng.shuffle(zeros_idx)
                h[zeros_idx[:required - current]] = 1
            return h
        else:
            return random_binary_h(n, rng)

    # define evaluation
    dx = 2.0 / n
    def evaluate(h):
        # ensure float
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # projection onto box sum
    target = n / 2
    def project(x):
        # binary? maybe use clip and adjust linearly to hit sum.
        # We'll implement waterfilling:
        lo = -np.max(x)
        hi = 1 - np.min(x)
        for _ in range(50):
            lam = (lo + hi) / 2
            proj = np.clip(x + lam, 0.0, 1.0)
            s = proj.sum()
            if s < target:
                lo = lam
            else:
                hi = lam
        return np.clip(x + (lo + hi) / 2, 0.0, 1.0)

    # keep best solution
    best_h = make_start()
    best_val = evaluate(best_h)
    # Logging (optional)
    # start local search loops
    cur_h = best_h.copy()
    cur_val = best_val

    # parameters
    max_no_improve = 2000
    no_improve = 0

    while time.time() - start < budget_s * 0.95:
        # Simulated annealing style
        # Randomly decide move type
        if rng.random() < 0.7:
            # binary swap move (maintain integer)
            ones_idx = np.where(cur_h == 1)[0]
            zeros_idx = np.where(cur_h == 0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                # cannot swap, restart
                cur_h = make_start()
                cur_val = evaluate(cur_h)
                continue
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            new_h = cur_h.copy()
            new_h[i] = 0.0
            new_h[j] = 1.0
        else:
            # continuous small perturbation move
            i, j = rng.choice(n, size=2, replace=False)
            # Determine feasible delta range
            low = max(-cur_h[i], cur_h[j] - 1.0)
            high = min(1.0 - cur_h[i], cur_h[j])
            if high <= low:
                continue
            delta = rng.uniform(low, high) * 0.5  # reduce magnitude a bit (can tune)
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        new_val = evaluate(new_h)

        # Acceptance criteria
        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        else:
            # maybe accept with probability decreasing with time
            elapsed = time.time() - start
            temp = max(0.001, (budget_s - elapsed) / budget_s)  # decreasing
            if rng.random() < math.exp(-(new_val - cur_val) / (temp + 1e-12)):
                cur_h = new_h
                cur_val = new_val
                no_improve += 1
            else:
                no_improve += 1

            cur_val = evaluate(cur_h)

            no_improve = 0
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
        else:
            elapsed = time.time() - start_time
            temp = max(0.001, (budget_s - elapsed) / budget_s)
            if rng.random() < math.exp(-(new_val - cur_val) / (temp + 1e-12)):
                cur_h, cur_val = new_h, new_val
            no_improve += 1

        # restart if stuck
        if no_improve >= max_no_improve:
            cur_h = _make_start(n, rng)
            cur_val = _evaluate(cur_h)
            no_improve = 0

        # occasional sub‑gradient refinement (few steps)
        if rng.random() < 0.018:
            h_sg = cur_h.copy()
            step0 = 0.9
            for it in range(150):
                corr = np.correlate(h_sg, 1.0 - h_sg, mode="full")
                k = np.argmax(corr) - (n - 1)
                grad = _gradient_for_shift(h_sg, k)
                step = step0 / math.sqrt(it + 1)
                h_sg = _project_onto_box_sum(h_sg - step * grad, target_sum)
                val_sg = _evaluate(h_sg)
                if val_sg < best_val:
                    best_h, best_val = h_sg.copy(), val_sg
                    cur_h, cur_val = h_sg.copy(), val_sg
                    no_improve = 0

    # ----- final dedicated sub‑gradient descent -----------------------------
    h_ref = best_h.copy()
    step0 = 0.8
    for it in range(4200):
        corr = np.correlate(h_ref, 1.0 - h_ref, mode="full")
        k = np.argmax(corr) - (n - 1)
        grad = _gradient_for_shift(h_ref, k)
        step = step0 / math.sqrt(it + 1)
        h_ref = _project_onto_box_sum(h_ref - step * grad, target_sum)
        val = _evaluate(h_ref)
        if val < best_val:
            best_h, best_val = h_ref.copy(), val

    # final sanity projection (eliminate tiny drift)
    best_h = _project_onto_box_sum(best_h, target_sum)
    best_val = _evaluate(best_h)

    return (best_h.tolist(), float(best_val), n)
