                break

    factor = 4   # results in n=base_len*factor = 180

        return np.max(conv) * (2.0 / len(vec))
    best_h = h.copy()

    # Simulated annealing
    start_time = time.time()

    cooling = 0.9995

            break

        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # bounds for epsilon
        hi = cur_h[i]
        hj = cur_h[j]
        low = max(-hi, hj - 1.0)

def compute_score(v: np.ndarray) -> float:
    """Maximum overlap integral for a step function given by vector v."""
    conv = np.correlate(v, 1.0 - v, mode="full")
    return conv.max() * (2.0 / v.size)


def random_feasible(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with prescribed sum."""
    while True:
        x = rng.random(n)
        scaled = x * (target_sum / x.sum())
        if np.all(scaled <= 1.0):
            return scaled


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Starting point
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=np.float64)
    else:
        base_len = 45
        target_base = base_len / 2.0
        base = random_feasible(base_len, target_base, rng)

    # Upscale to gain more degrees of freedom (factor 4 → 180 samples)
    factor = 4
    n = len(base) * factor
    h = np.repeat(base, factor).astype(np.float64)
    target_sum = n / 2.0

    # Correct tiny rounding errors so that the integral constraint holds exactly
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if 0 <= h[0] + diff <= 1.0:
            h[0] += diff
        else:
            h = np.clip(h, 0, 1)
            h *= target_sum / h.sum()

    best_h = h.copy()
    best_score = compute_score(best_h)

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing style local search
    # ------------------------------------------------------------------
    max_time = budget_s - 0.5          # small safety margin
    start_time = time.time()

    cur_h = h.copy()
    cur_score = best_score

    T = 0.02          # initial temperature
    T_min = 1e-6
    cooling = 0.9998
    step = 0.5        # maximal epsilon before clipping
    step_min = 1e-5
    no_improve = 0

    while time.time() - start_time < max_time:
        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        hi, hj = cur_h[i], cur_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if high <= low:
            continue

        eps = rng.uniform(-step, step)
        eps = max(low, min(high, eps))
        if abs(eps) < 1e-14:
            continue

        # apply move in place
        cur_h[i] = hi + eps
        cur_h[j] = hj - eps
        new_score = compute_score(cur_h)

        if new_score < cur_score - 1e-12:
            # deterministic improvement
            cur_score = new_score
            no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # Metropolis acceptance of a worse move
            if T > 0.0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    cur_score = new_score
                    no_improve = 0
                else:
                    # revert
                    cur_h[i] = hi
                    cur_h[j] = hj
                    no_improve += 1
            else:
                cur_h[i] = hi
                cur_h[j] = hj
                no_improve += 1

        # annealing schedule
        T = max(T * cooling, T_min)
        step = max(step * 0.9999, step_min)

        # long stagnation → random restart
        if no_improve > 80000:
            cur_h = random_feasible(n, target_sum, rng)
            cur_score = compute_score(cur_h)
            no_improve = 0

    # ------------------------------------------------------------------
    # 3. Final feasibility correction and result
    # ------------------------------------------------------------------
    total = best_h.sum()
    diff = target_sum - total
    if abs(diff) > 1e-9:
        for idx in range(n):
            if 0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break
        else:
            best_h = np.clip(best_h, 0, 1)
            best_h *= target_sum / best_h.sum()

    final_score = compute_score(best_h)
    return (best_h.tolist(), float(final_score), n)
