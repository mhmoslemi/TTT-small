def solve_minimax_binary(n, time_limit_sec):
    import cvxpy as cp
    # n must be even
    half = n // 2
    # variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)  # max count objective
    cons = [cp.sum(h) == half]
    # For each shift
    for s in range(-(n-1), n):
        # compute overlapping indices
        # For shift s, i ranges such that i in [0, n-1] and i+s in [0, n-1].
        # Equivalent to i from max(0, -s) to min(n-1, n-1-s)
        i_start = max(0, -s)
        i_end = min(n-1, n-1-s)
        length = i_end - i_start + 1
        if length <= 0:
            continue
        # sum_h_i = sum of h[i] over overlapping region
        idx = np.arange(i_start, i_end+1)
        # We'll write sum_h_i - Sum(y_{i,s}) <= t
        # Introduce binary y for each i
        y = cp.Variable(length, boolean=True)
        # constraints linking y to h[i] and h[i+s]
        # Need references to h indices; we can formulate linear constraints:
        # y[i] <= h[idx[i]]
        # y[i] <= h[idx[i] + s]
        # y[i] >= h[idx[i]] + h[idx[i] + s] - 1
        cons.append(y <= h[idx])
        cons.append(y <= h[idx + s])
        cons.append(y >= h[idx] + h[idx + s] - 1)
        # define sum_h_i - sum(y) <= t
        cons.append(cp.sum(h[idx]) - cp.sum(y) <= t)
    # objective min t
    prob = cp.Problem(cp.Minimize(t), cons)
    prob.solve(solver=cp.CBC, time_limit=time_limit_sec) # may not support time limit param; but we can set warm_start or max_iters; but let's use cp.CBC default.

import time, numpy as np

def _max_corr(h: np.ndarray) -> float:
    """Maximum overlap count for a binary step function."""
    # correlation of h with (1-h); result length = 2*len-1
    return float(np.correlate(h, 1 - h, mode="full").max())

def _c5_bound(h: np.ndarray) -> float:
    """C5 = max_corr * dx,  dx = 2/n."""
    n = h.shape[0]
    return _max_corr(h) * (2.0 / n)

# ----------------------------------------------------------------------
# Exact solver for small even n using MILP (cvxpy + CBC)
def _solve_milp_binary(n: int, time_limit: float = 30.0):
    """Return (h, c5) for optimal binary h of length n, or (None, None)."""
    try:
        import cvxpy as cp
    except Exception:                     # cvxpy not available
        return None, None
    if n % 2:
        return None, None                 # binary formulation needs even n
    half = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True, nonneg=True)
    cons = [cp.sum(h) == half]
    # build constraints for each shift s
    for s in range(-(n - 1), n):
        i0 = max(0, -s)
        i1 = min(n - 1, n - 1 - s)
        if i0 > i1:
            continue
        idx = np.arange(i0, i1 + 1, dtype=int)
        L = idx.size
        y = cp.Variable(L, boolean=True)                     # y = h_i * h_{i+s}
        cons += [y <= h[idx],
                 y <= h[idx + s],
                 y >= h[idx] + h[idx + s] - 1]
        cons.append(cp.sum(h[idx]) - cp.sum(y) <= t)        # overlap count ≤ t
    prob = cp.Problem(cp.Minimize(t), cons)
    # CBC solver; ignore time_limit if not supported
    try:
        prob.solve(solver=cp.CBC, verbose=False)
    except Exception:
        try:                           # fallback to ECOS_BB
            prob.solve(solver=cp.ECOS_BB, verbose=False)
        except Exception:
            return None, None
    if prob.status not in {"optimal", "optimal_inaccurate"}:
        return None, None
    h_val = np.rint(np.asarray(h.value)).astype(float)      # force 0/1
    # numerical safety: clip
    h_val = np.clip(h_val, 0, 1)
    return h_val, _c5_bound(h_val)

# Heuristic local search for larger n (binary only)
def _search_binary(n: int, rng: np.random.Generator,
                   time_budget: float,
                   start_h: np.ndarray | None = None,
                   target_max_corr: int | None = None):
    """Hill‑climbing with random restarts; returns (h, c5, max_corr)."""
    half = n // 2
    dx = 2.0 / n

    def random_binary():
        a = np.zeros(n, dtype=float)
        a[:half] = 1.0
        rng.shuffle(a)
        return a

    best_h = random_binary() if start_h is None else start_h.copy()
    best_mc = int(_max_corr(best_h))
    # early target in terms of count (not scaled)
    target_cnt = target_max_corr

    start = time.time()
    nosuccess = 0
    while time.time() - start < time_budget:
        if target_cnt is not None and best_mc <= target_cnt:
            break
        # pick a random 1 and a random 0 to swap
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i], cand[j] = 0.0, 1.0
        cand_mc = int(_max_corr(cand))
        if cand_mc < best_mc:
            best_h, best_mc = cand, cand_mc
            nosuccess = 0
        else:
            nosuccess += 1
        if nosuccess > 200 * n:                # restart after stagnation
            best_h = random_binary()
            best_mc = int(_max_corr(best_h))
            nosuccess = 0
    return best_h, best_mc * dx, best_mc
