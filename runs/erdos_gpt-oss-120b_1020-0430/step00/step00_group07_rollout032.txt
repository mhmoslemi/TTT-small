def gradient_C5(h):
    n = len(h)
    # compute B array via correlation:
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)  # index
    k = idx_max - (n - 1)  # shift
    # compute gradient
    g = np.zeros_like(h)
    if k >= 0:
        L = n - k
        # contributions for j in [0, L-1] (first region)
        g[:L] += 1 - h[k:]   # 1 - h_{j+k}
        # contributions for j in [k, n-1] (second region)
        g[k:] += -h[:n - k]   # -h_{j - k}
    else:

        # prefix region j in [0, n-d-1]
        g[:n - d] += -h[d:]   # -h_{j + d}
    return g, k

        # correlation and the shift that attains the current maximum
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = int(np.argmax(corr))          # index in full array
        k = idx_max - (n - 1)                   # signed shift (positive → right)

        # gradient of the maximal term B_k = Σ h (1‑h shifted)
        g = np.zeros_like(h)
        if k >= 0:
            L = n - k
            g[:L] += 1 - h[k:]                  # +1 - h_{i+k}
            g[k:] += -h[:n - k]                 # -h_{i‑k}
        else:                                   # k < 0
            d = -k
            g[d:] += 1 - h[:n - d]              # +1 - h_{i‑d}
            g[:n - d] += -h[d:]                # -h_{i+d}

        # pick a source (positive gradient – reducing h lowers the worst term)
        src_mask = (g > 1e-12) & (h > 1e-12)
        dst_mask = (g < -1e-12) & (h < 1.0 - 1e-12)

        if src_mask.any() and dst_mask.any():
            src_candidates = np.where(src_mask)[0]
            dst_candidates = np.where(dst_mask)[0]
            src = src_candidates[np.argmax(g[src_candidates])]
            dst = dst_candidates[np.argmin(g[dst_candidates])]
            delta = min(step, h[src], 1.0 - h[dst])
            if delta > 0:
                h[src] -= delta
                h[dst] += delta
        else:
            # fallback: random admissible swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = min(step, h[i], 1.0 - h[j])
                if delta > 0:
                    h[i] -= delta
                    h[j] += delta

        # evaluate
        cur_c5, _ = compute_c5(h)
        if cur_c5 < best_c5 - 1e-10:
            best_c5 = cur_c5
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

        # step size decay
        step = max(min_step, step * decay)

        # occasional random jitter if we have been stagnant too long
        if stagnant > 3000:
            for _ in range(5):
                i, j = np.random.randint(n, size=2)
                delta = min(step, h[i], 1.0 - h[j])
                if delta > 0:
                    h[i] -= delta
                    h[j] += delta
            stagnant = 0

        # periodic restart from best with a tiny perturbation
        if it % 30000 == 0:
            h = best_h.copy()
            noise = (np.random.rand(n) - 0.5) * step * 2.0
            h += noise
            h = np.clip(h, 0.0, 1.0)
            # re‑project to the exact mass
            cur = h.sum()
            diff = target_sum - cur
            if abs(diff) > 1e-12:
                movable = (h > 1e-12) & (h < 1.0 - 1e-12)
                if movable.any():
                    adjust = diff / movable.sum()
                    h[movable] += adjust
                else:
                    for i in range(n):
                        if diff > 0 and h[i] < 1.0:
                            add = min(diff, 1.0 - h[i])
                            h[i] += add
                            diff -= add
                        elif diff < 0 and h[i] > 0.0:
                            sub = min(-diff, h[i])
                            h[i] -= sub
                            diff += sub
                        if abs(diff) < 1e-12:
                            break

    # ------------------------------------------------------------
    # final clean‑up of the best solution
    # ------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur = best_h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-10:
        for i in range(n):
            if diff > 0 and best_h[i] < 1.0:
                add = min(diff, 1.0 - best_h[i])
                best_h[i] += add
                diff -= add
            elif diff < 0 and best_h[i] > 0.0:
                sub = min(-diff, best_h[i])
                best_h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    final_c5, _ = compute_c5(best_h)

    return (best_h.tolist(), float(final_c5), n)
