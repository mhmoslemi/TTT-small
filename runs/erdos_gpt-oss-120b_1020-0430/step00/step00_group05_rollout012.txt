def hill_climb(h, max_iter, best_bound=None):
    n = len(h)
    cur_h = h.copy()
    cur_bound = compute_bound(cur_h)
    if best_bound is None or cur_bound < best_bound: best = cur_h.copy(); best_bound = cur_bound
    ones = np.flatnonzero(cur_h)
    zeros = np.flatnonzero(1 - cur_h)
    for it in range(max_iter):
        i_one = np.random.choice(ones)
        i_zero = np.random.choice(zeros)
        # swap
        cur_h[i_one], cur_h[i_zero] = 0, 1
        # update lists
        ones[ones == i_one] = i_zero
        zeros[zeros == i_zero] = i_one
        # evaluate
        new_bound = compute_bound(cur_h)
        if new_bound <= cur_bound:
            cur_bound = new_bound
            if new_bound < best_bound:
                best_bound = new_bound
                best = cur_h.copy()
            # keep swap
        else:
            # revert swap
            cur_h[i_one], cur_h[i_zero] = 1, 0
            # restore lists
            ones[ones == i_zero] = i_one
            zeros[zeros == i_one] = i_zero
            # optionally accept with prob? not needed
    return best, best_bound

    best_bound = cur_bound

        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        # swap
        cur_h[i_one] = 0
        cur_h[i_zero] = 1
        new_bound = compute_bound(cur_h)[0]
        if new_bound <= cur_bound:

            cur_h[i_one] = 1
            cur_h[i_zero] = 0
        # early exit if bound already low enough
        if best_bound < 0.3808:

    return best_h, best_bound

import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * dx)

def _hill_climb(start_h: np.ndarray, max_iter: int, rng: np.random.Generator) -> tuple:
    """Simple hill‑climbing on binary vectors with fixed number of ones."""
    cur = start_h.copy()
    cur_val = _c5_value(cur)
    best_h = cur.copy()
    best_val = cur_val

    # maintain index lists for fast swaps
    ones = list(np.flatnonzero(cur))
    zeros = list(np.flatnonzero(1 - cur))

    for _ in range(max_iter):
        # propose a swap 1<->0
        i_one = rng.integers(len(ones))
        i_zero = rng.integers(len(zeros))
        pos_one = ones[i_one]
        pos_zero = zeros[i_zero]

        # perform swap
        cur[pos_one] = 0
        cur[pos_zero] = 1
        ones[i_one] = pos_zero
        zeros[i_zero] = pos_one

        new_val = _c5_value(cur)
        if new_val <= cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur.copy()
        else:
            # revert swap
            cur[pos_one] = 1
            cur[pos_zero] = 0
            ones[i_one] = pos_one
            zeros[i_zero] = pos_zero

        # early exit if we already beat the target
        if best_val < 0.3808:
            break

    return best_h, best_val

    start_time = time.time()
    target = 0.3808

    if init is not None:

        if n0 % 2 == 0 and n0 not in n_candidates:
            n_candidates.append(n0)
        if 2 * n0 <= 200 and (2 * n0) not in n_candidates:
            n_candidates.append(2 * n0)

    n_candidates = sorted(set(n_candidates))

    best_h = None
    best_val = np.inf
    best_n = None

    # --- main search loop ------------------------------------------------------------
    for n in n_candidates:
        # stop if budget exhausted
        if time.time() - start_time > budget_s * 0.95:
            break

        # build a few starting vectors for this size
        starts = []

        if init is not None:
            # repeat / truncate the initial pattern to length n
            if init.shape[0] == n:
                h0 = (init > 0.5).astype(np.int8)
                starts.append(h0)
            elif init.shape[0] < n:
                rep = n // init.shape[0]
                rem = n % init.shape[0]
