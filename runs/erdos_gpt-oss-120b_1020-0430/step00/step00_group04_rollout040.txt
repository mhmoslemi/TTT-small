def proj_onto_capped_simplex(v, S):
    n = v.shape[0]
    # If S is 0, return zeros. if S is n, return ones.
    if S <= 0:
        return np.zeros_like(v)
    if S >= n:
        return np.ones_like(v)
    low = -1e5
    high = 1e5
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > S:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low + high) / 2.0, 0, 1)
    # Due to rounding, sum may not be exactly S; adjust residual by distributing among non-saturated entries.
    residual = S - x.sum()
    if abs(residual) > 1e-8:
        # Find indices where 0 < x < 1
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            # Distribute evenly among free entries
            x[free] += residual / free.sum()
            # Clip again to maintain bounds
            x = np.clip(x, 0, 1)
        else:
            # No free entries; adjust all near bound with small epsilon.
            pass
    return x
```

def simulated_annealing(n, time_budget, init_h):
    best_h = init_h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        # choose two indices
        i, j = np.random.choice(n, size=2, replace=False)
        # compute feasible delta range
        low = -min(cur_h[i], 1 - cur_h[j])
        high = min(1 - cur_h[i], cur_h[j])
        if low >= high:
            continue
        delta = np.random.uniform(low, high)
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = compute_c5(new_h)
        # temperature schedule
        temp = max(0.001, (time_budget - (time.time() - start)) / time_budget)  # linear decay
        if new_val < cur_val or np.random.rand() < np.exp((cur_val - new_val) / temp):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
        it += 1
    return best_h, best_val
```

def gradient_descend(h, n, time_budget):
    start = time.time()
    target = n/2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    it = 0
    while time.time() - start < time_budget:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        # compute gradient of g(lag):
        grad = gradient_of_corr(h, lag)
        # step size schedule
        step = 0.5 / math.sqrt(it + 1)
        v = h - step * grad
        h = proj_onto_capped_simplex(v, target)
        val = corr.max() * 2.0 / n
        if val < best_val:
            best_val = val
            best_h = h.copy()
        it += 1
    return best_h, best_val

import time
import math
import numpy as np

# ---------- core utilities ----------
def compute_c5(h):
    """C5 bound for a step function sampled in h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def proj_onto_capped_simplex(v, target_sum, max_iter=60, eps=1e-10):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target_sum}."""
    n = v.shape[0]
    if target_sum <= 0:
        return np.zeros_like(v)
    if target_sum >= n:
        return np.ones_like(v)

    # binary search for lambda such that sum(clip(v-lambda,0,1)) = target_sum
    low = np.min(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
        if high - low < eps:
            break
    x = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    # small residual correction (rare)
    residual = target_sum - x.sum()
    if abs(residual) > 1e-8:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += residual / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def upsample_h(old_h, new_n):
    """Linear interpolation of a step function to a finer grid."""
    old_n = old_h.shape[0]
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, old_h)


def triangular_h(n):
    """Symmetric triangular hat of height 1 on [0,2]."""
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.where(xs <= 1.0, xs, 2.0 - xs)
    return h.astype(float)


def random_initial(n):
    """Random feasible vector (sum = n/2, 0<=h<=1)."""
    v = np.random.rand(n)
    return proj_onto_capped_simplex(v, n / 2.0)


def gradient_of_corr(h, lag):
    """Sub‑gradient of g_lag(h)=∑ h_i(1−h_{i+lag}) w.r.t. h."""
    n = h.shape[0]
    grad = np.ones_like(h)
    if lag >= 0:
        if lag < n:
            grad[: n - lag] -= h[lag:]
            grad[lag:] -= h[: n - lag]
    else:
        k = -lag
        if k < n:
            grad[k:] -= h[: n - k]
            grad[: n - k] -= h[k:]
    return grad


# ---------- optimisation algorithms ----------
def simulated_annealing(n, time_budget, init_h):
    """Simple SA that preserves the sum by swapping mass between two coordinates."""
    start = time.time()
    cur_h = init_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # annealing schedule parameters
    start_temp = 0.02
    end_temp = 0.001

    while True:
        now = time.time()
        if now - start >= time_budget:
            break

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]

        # feasible delta range so both stay in [0,1]
        delta_max_pos = min(1.0 - hi, hj)          # increase i, decrease j
        delta_max_neg = -min(hi, 1.0 - hj)         # decrease i, increase j
        if delta_max_pos <= 0 and delta_max_neg >= 0:
            continue
        delta = np.random.uniform(delta_max_neg, delta_max_pos)

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_val = compute_c5(new_h)

        # linear temperature decay
        elapsed = now - start
        frac = elapsed / time_budget
        temp = start_temp * (1.0 - frac) + end_temp * frac

        if new_val < cur_val or np.random.rand() < math.exp((cur_val - new_val) / max(temp, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()

    return best_h, best_val


def gradient_descent(h, time_budget):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = h.shape[0]
    target = n / 2.0
    start = time.time()
    best_h = h.copy()
    best_val = compute_c5(h)
    it = 0

    while True:
        now = time.time()
        if now - start >= time_budget:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)

        grad = gradient_of_corr(h, lag)

        # step size diminishing like 1/sqrt(it)
        step = 0.5 / math.sqrt(it + 1.0)
        v = h - step * grad
        h = proj_onto_capped_simplex(v, target)

        cur_val = corr.max() * (2.0 / n)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        it += 1

    return best_h, best_val


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound and the number of samples."""
    np.random.seed(seed)
    overall_start = time.time()

    # choose a resolution (larger gives smoother shapes, still cheap)
    n = 168                       # 2 · 84 gives a nice even grid
    target_sum = n / 2.0

    # initialise from parent if available, otherwise from a triangular shape
    if 'initial_h_values' in globals():
        init_h = upsample_h(initial_h_values, n)
    else:
        init_h = triangular_h(n)

    init_h = proj_onto_capped_simplex(init_h, target_sum)

    # divide the budget: 70 % for SA, 30 % for a short gradient refinement
    remaining = budget_s - (time.time() - overall_start)
    time_sa = remaining * 0.70
    time_gd = max(0.0, remaining - time_sa)

    # simulated annealing phase
    h_sa, val_sa = simulated_annealing(n, time_sa, init_h)

    # gradient refinement (optional – fast, helps polishing)
    h_opt, val_opt = gradient_descent(h_sa, time_gd)

    # final safety projection and evaluation
    h_opt = proj_onto_capped_simplex(h_opt, target_sum)
    final_val = compute_c5(h_opt)

    return h_opt, final_val, n
