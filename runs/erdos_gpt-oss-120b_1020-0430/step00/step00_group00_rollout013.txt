def enforce_sum(h):
    target = n/2
    diff = target - h.sum()
    # Add diff to first index that can accommodate
    for idx in range(len(h)):
        if diff > 0 and h[idx] < 1:
            add = min(diff, 1 - h[idx])
            h[idx] += add
            diff -= add
        elif diff < 0 and h[idx] > 0:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    # If still diff > tolerance, but we can't adjust within bounds, we accept approximate.
    return h

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    if 'initial_h_values' in globals():

    target = n/2.0
    cur_sum = h.sum()

    if abs(cur_sum - target) > 1e-12:
        # scale proportionally
        h = h * (target / cur_sum)
        h = np.clip(h, 0, 1)
        # enforce sum
        h = enforce_sum(h)

    dx = 2.0 / n

    def compute_c5(vec):
        return np.max(np.correlate(vec, 1 - vec, mode='full')) * dx

            break

        j = rng.integers(n-1)

        # Could incorporate but not needed

    return best_h.tolist(), float(best_c5), n

import time
import math
import numpy as np

def _compute_c5(h, dx):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    # correlation of h with (1-h), mode='full' gives all shifts with zero padding
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _w_for_lag(h, lag):
    """For a given lag, return w_i = 1 - h_{i+lag} (zero if out of range)."""
    n = h.shape[0]
    w = np.zeros_like(h)
    if lag >= 0:
        if lag < n:
            w[: n - lag] = 1.0 - h[lag:]
    else:
        p = -lag
        if p < n:
            w[p:] = 1.0 - h[: n - p]
    return w

def _enforce_sum_and_bounds(h, target):
    """Clip to [0,1] and adjust a single index to meet the exact sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Adjust the first index that can absorb the difference
    for idx in range(h.shape[0]):
        if diff > 0.0 and h[idx] < 1.0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        elif diff < 0.0 and h[idx] > 0.0:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # 1. Initialise the vector
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        h_cur = init.copy()
    else:
        n = 45                     # default size when no parent is present
        h_cur = np.full(n, 0.5)    # uniform feasible start

    target_sum = n / 2.0            # because ∫h = 1  ->  sum(h) = 1/dx = n/2
    # make sure the starting vector respects the constraints
    h_cur = _enforce_sum_and_bounds(h_cur, target_sum)

    dx = 2.0 / n
    best_h = h_cur.copy()
    best_c5 = _compute_c5(best_h, dx)

    # ------------------------------------------------------------
    # 2. Main optimisation loop (simulated‑annealing + greedy descent)
    # ------------------------------------------------------------
    no_improve = 0
    # temperature schedule (high at start, low at end)
    T_start = 5e-3
    T_end = 1e-6

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s - 0.05:  # leave a tiny margin for the return
            break

        # current full correlation and C5
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        cur_c5 = corr.max() * dx
        lag_idx = corr.argmax()
        lag = lag_idx - (n - 1)               # shift corresponding to the current maximum

        # --------------------------------------------------------
        # 2a) Greedy move that directly attacks the worst lag
        # --------------------------------------------------------
        w = _w_for_lag(h_cur, lag)

        # source: position with large w (i.e. large (1‑h_{i+lag})) and positive mass
        src_candidates = np.where(h_cur > 1e-12)[0]
        tgt_candidates = np.where(h_cur < 1.0 - 1e-12)[0]

        moved = False
        if src_candidates.size and tgt_candidates.size:
            w_src = w[src_candidates]
            i_src = src_candidates[np.argmax(w_src)]

            w_tgt = w[tgt_candidates]
            i_tgt = tgt_candidates[np.argmin(w_tgt)]

            if w[i_src] > w[i_tgt] + 1e-12:
                cap = min(h_cur[i_src], 1.0 - h_cur[i_tgt])
                if cap > 0:
                    # take a moderate fraction of the transferable mass
                    delta = cap * (0.2 + 0.8 * rng.random())
                    h_new = h_cur.copy()
                    h_new[i_src] -= delta
                    h_new[i_tgt] += delta
                    new_c5 = _compute_c5(h_new, dx)
                    if new_c5 < cur_c5 - 1e-12:          # accept only if improvement
                        h_cur = h_new
                        cur_c5 = new_c5
                        moved = True
                        if new_c5 < best_c5:
                            best_h = h_new.copy()
                            best_c5 = new_c5
                            no_improve = 0

        # --------------------------------------------------------
        # 2b) If greedy move failed, try a random pair move (SA style)
        # --------------------------------------------------------
        if not moved:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            cap = min(h_cur[i], 1.0 - h_cur[j])
            if cap > 0:
                delta = cap * rng.random()
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_c5 = _compute_c5(h_new, dx)

                # simulated‑annealing acceptance
                progress = elapsed / budget_s
                temp = T_start * (1.0 - progress) + T_end * progress
                if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
                    h_cur = h_new
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = new_c5
                        no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        # --------------------------------------------------------
        # 2c) Stagnation handling – occasional multi‑step jitter
        # --------------------------------------------------------
        if no_improve >= 5000:
            # apply a few tiny random swaps to shake the solution
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                cap = min(h_cur[i], 1.0 - h_cur[j])
                if cap > 0:
                    delta = cap * rng.random()
                    h_cur[i] -= delta
                    h_cur[j] += delta
            cur_c5 = _compute_c5(h_cur, dx)
            no_improve = 0

        # early exit if we have already beaten the target record
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------
    # 3. Final projection onto the feasible set and final evaluation
    # ------------------------------------------------------------
    best_h = _enforce_sum_and_bounds(best_h, target_sum)
    best_c5 = _compute_c5(best_h, dx)

    return best_h.tolist(), float(best_c5), n
