def adjust_to_target(x, target):
    diff = target - x.sum()
    if abs(diff) < 1e-10:

import time
import numpy as np

def _project_onto_box_sum(v, target, eps=1e-12, max_iter=60):
    """Project vector v onto {0<=x<=1, sum(x)=target}."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction (should be negligible)
    diff = target - x.sum()
    if abs(diff) > eps:
        free = (x > eps) & (x < 1.0 - eps)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _worst_lags(conv):
    """Return integer lags (relative to zero) attaining the maximal correlation."""
    maxv = conv.max()
    idx = np.where(conv >= maxv - 1e-12)[0]
    n = (len(conv) + 1) // 2
    return idx - (n - 1)

def _grad_for_lags(h, lags):
    """Sub‑gradient of the maximal correlation for a set of lags."""
    n = h.shape[0]
    g = np.zeros_like(h)
    for lag in lags:
        if lag >= 0:
            i = np.arange(0, n - lag, dtype=int)
            g[i] += 1.0 - h[i + lag]
            g[i + lag] -= h[i]
        else:
            l = -lag
            i = np.arange(l, n, dtype=int)
            g[i] += 1.0 - h[i - l]
            g[i - l] -= h[i]
    return g

def _local_optimize(h0, target, budget, rng, max_iter=5000):
    """Gradient descent on the max‑correlation objective."""
    n = h0.shape[0]
    dx = 2.0 / n
    best_h = h0.copy()
    best_val = np.inf
    h = h0.copy()
    step0 = 0.5
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > budget:
            break
        conv = np.correlate(h, 1.0 - h, mode='full') * dx
        cur = conv.max()
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
        # gradient w.r.t. the worst lag(s)
        lags = _worst_lags(conv)
        g = _grad_for_lags(h, lags)
        # simple step‑size schedule
        step = step0 / (1.0 + it * 0.001)
        hnew = h - step * g
        # occasional stochastic jitter to escape plateaus
        if (it & 127) == 0 and rng.random() < 0.15:
            hnew += (rng.random(n) - 0.5) * 0.04
        # project back to feasible region
        h = _project_onto_box_sum(hnew, target)
    return best_h, best_val
