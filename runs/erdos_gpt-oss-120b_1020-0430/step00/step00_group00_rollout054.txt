    max_val = np.max(alpha * corr)

def smooth_max_corr(h, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    # compute softmax weights
    a_corr = alpha * corr
    max_a = np.max(a_corr)
    w = np.exp(a_corr - max_a)
    w_sum = w.sum()
    w /= w_sum
    # smooth objective value (weighted average)
    smooth_val = np.dot(w, corr)  # weighted average = smooth approximation of max
    # gradient
    n = len(h)
    vec1 = np.zeros_like(h, dtype=float)
    vec2 = np.zeros_like(h, dtype=float)
    for shift in range(-(n-1), n):
        weight = w[shift + n - 1]
        if weight == 0:
            continue
        if shift >= 0:
            sz = n - shift
            if sz > 0:
                vec1[:sz] += weight * h[shift:]
                vec2[shift:] += weight * h[:sz]
        else:
            s = -shift
            sz = n - s
            if sz > 0:
                vec1[s:] += weight * h[:sz]
                vec2[:sz] += weight * h[s:]
    grad = 1 - vec1 - vec2
    return smooth_val, grad
```

import time
import numpy as np

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, sum(x)=target}."""
    lo, hi = -2.0, 2.0  # wide enough for all possible values
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _smooth_max_corr(h: np.ndarray, alpha: float):
    """Soft‑max approximation of max correlation and its gradient."""
    n = h.shape[0]
    # exact correlation (no dx scaling)
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * corr
    a_max = np.max(a)
    w = np.exp(a - a_max)
    w /= w.sum()                     # soft‑max weights, sum to 1

    # gradient of weighted average of corr
    vec1 = np.zeros_like(h, dtype=float)
    vec2 = np.zeros_like(h, dtype=float)
    for shift in range(-(n - 1), n):
        weight = w[shift + n - 1]
        if weight == 0.0:
            continue
        if shift >= 0:
            sz = n - shift
            if sz:
                vec1[:sz] += weight * h[shift:]          # h[i+shift]
                vec2[shift:] += weight * h[:sz]          # h[i-shift]
        else:
            s = -shift
            sz = n - s
            if sz:
                vec1[s:] += weight * h[:sz]              # h[i+shift] = h[i-s]
                vec2[:sz] += weight * h[s:]              # h[i-shift] = h[i+s]
    grad = 1.0 - vec1 - vec2
    smooth_val = np.dot(w, corr)      # weighted average = smooth max
    return smooth_val, grad


def _max_corr(h: np.ndarray) -> float:
    """Exact max correlation (scaled by dx) for a step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def _optimize_from(h0: np.ndarray, budget_s: float, start_time: float):
    """Projected gradient descent with soft‑max smoothing."""
    n = h0.shape[0]
    target = n / 2.0
    h = np.clip(h0, 0.0, 1.0)
    h = _project_box_sum(h, target)

    best_h = h.copy()
    best_val = _max_corr(h)

    max_iter = 2000
    alpha_start, alpha_end = 0.5, 40.0
    lr_start = 0.6

    for it in range(max_iter):
        if time.time() - start_time > budget_s - 0.05:
            break
        # schedule (exponential for alpha, geometric for lr)
        t = it / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (0.5 ** t)

        _, grad = _smooth_max_corr(h, alpha)
        h_new = h - lr * grad
        h_new = _project_box_sum(h_new, target)

        val = _max_corr(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # Determine number of points and initial vector (if supplied)
    if 'initial_h_values' in globals():
        init_vec = np.array(initial_h_values, dtype=float)
        n = init_vec.size
    else:
        # fallback size – still well below the 1 k limit
        n = 45
        init_vec = None

    target = n / 2.0

    best_h = None
    best_val = np.inf

    # evaluate provided construction first
    if init_vec is not None:
        if not (np.all(init_vec >= 0) and np.all(init_vec <= 1)):
            init_vec = np.clip(init_vec, 0.0, 1.0)
        if abs(init_vec.sum() - target) > 1e-9:
            init_vec = _project_box_sum(init_vec, target)
        val = _max_corr(init_vec)
        best_h, best_val = init_vec.copy(), val

    # use several random starts (including the initial one if any)
    restarts = 12
    for _ in range(restarts):
        if time.time() - start > budget_s - 0.8:
            break
        # random feasible point
        rand_vec = np.random.rand(n)
        rand_vec = _project_box_sum(rand_vec, target)
        h_opt, val = _optimize_from(rand_vec, budget_s, start)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()

    # final safety projection and evaluation
    if best_h is not None:
        best_h = _project_box_sum(np.clip(best_h, 0.0, 1.0), target)
        best_val = _max_corr(best_h)
        return best_h.tolist(), float(best_val), n
    else:
        # fallback – uniform feasible vector (rare case)
        uniform = np.full(n, target / n)
        val = _max_corr(uniform)
        return uniform.tolist(), float(val), n
