def project_onto_simplex_bounds(v, target):
    # v is array
    # we project onto set {h: sum h = target, 0 <= h <= 1}
    # Use iterative water-filling.
    h = np.clip(v, 0, 1)
    sum_h = h.sum()
    # The difference to adjust
    diff = target - sum_h
    # While diff not small:
    iteration = 0
    while abs(diff) > 1e-10 and iteration < 100:
        # Identify free indices (not at bounds)
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:
            # All at bounds, cannot adjust; break
            break
        # Compute adjustment per free variable
        delta = diff / n_free
        # Apply
        h[free] += delta
        # Clip again to stay within bounds
        np.clip(h, 0, 1, out=h)
        sum_h = h.sum()
        diff = target - sum_h
        iteration += 1
    return h

def gradient_for_shift(h, s):

import numpy as np

def evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_bounds_and_sum(v, target_sum):
    h = np.clip(v, 0.0, 1.0)
    # Water-filling iterative
    for _ in range(100):
        sum_h = h.sum()
        diff = target_sum - sum_h
        if abs(diff) <= 1e-12:
            break
        # free indices
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:  # cannot adjust further
            # clip to extremes adjusting some indices based on sign
            # If need to increase but all at 1, impossible; else.
            break
        delta = diff / n_free
        # adjust
        h[free] += delta
        # reclip
        np.clip(h, 0.0, 1.0, out=h)
    # final clip for safety
    np.clip(h, 0.0, 1.0, out=h)
    # final scaling if still off due to rounding
    sum_h = h.sum()
    if abs(sum_h - target_sum) > 1e-8:
        # apply simple scaling for minor error not hitting bounds
        # but ensure remains within bounds
        scale = target_sum / sum_h
        h = np.clip(h * scale, 0.0, 1.0)
        # after scaling might violate sum; redo simple adjust
        # but we ignore tiny error.
    return h

def optimize_h(h0, budget_s, start_time):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    step = 0.1
    iter_count = 0
    while time.time() - start_time < budget_s - 5.0:   # 5s margin
        c5 = evaluate_c5(h)
        # we may already have best, but compute if needed
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # compute correlation and shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)

        g = gradient_for_shift(h, shift)
        # propose update
        proposed = project_to_bounds_and_sum(h - step * g, target)
        new_c5 = evaluate_c5(proposed)
        if new_c5 < c5:
            h = proposed
            # increase step slightly
            step = min(step * 1.02, 1.0)
        else:
            # reject, reduce step
            step = step * 0.5

    delta_max = 0.1
    iter_count = 0
    while time.time() - start_time < budget_s - 2.0:
        # generate random i,j
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        max_delta_i = h[i]  # can decrease at most h[i]
        max_delta_j = 1.0 - h[j]  # can increase at most (1 - h[j])
        max_delta = min(delta_max, max_delta_i, max_delta_j)
        if max_delta <= 1e-12:

        iter_count += 1

def project_to_bounds_and_sum(v, target):
    h = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s

            break

        n_free = free.sum()
        if n_free == 0:

        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    # final small linear scaling if needed
    s = h.sum()
    if abs(s - target) > 1e-8:
        # adjust uniformly then clip
        scale = target / s
        h = np.clip(h * scale, 0.0, 1.0)

import time
import numpy as np
from scipy.interpolate import interp1d

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def evaluate_c5(h: np.ndarray) -> float:
    """Return the discrete C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation (full mode) gives the overlap for all shifts
    return dx * np.max(np.correlate(h, 1.0 - h, mode="full"))


def project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project a vector onto the feasible set:
        0 ≤ h[i] ≤ 1,  sum(h) = target
    Simple water‑filling iteration – fast for the small dimensions we use.
    """
    h = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        cur = h.sum()
        diff = target - cur
        if abs(diff) <= 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        n_free = free.sum()
        if n_free == 0:                                 # all at a bound – cannot move further
            break
        delta = diff / n_free
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    # tiny scaling correction for possible round‑off
    cur = h.sum()
    if abs(cur - target) > 1e-8:
        scale = target / cur
        h = np.clip(h * scale, 0.0, 1.0)
    return h


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of M(shift) = Σ h[i]·(1‑h[i+shift]) w.r.t. h.
    Positive shift: indices that run off the right side see w=1.
    Negative shift: indices that run off the left side see w=1.
    """
    n = h.shape[0]
    g = np.empty_like(h)
    if shift >= 0:
        g[: n - shift] = 1.0 - h[shift:]
        g[n - shift :] = 1.0
    else:
        s = -shift
        g[:s] = 1.0
        g[s:] = 1.0 - h[: n - s]
    return g


def upsample_to_n(h_low: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a low‑resolution step function to size n."""
    m = h_low.shape[0]
    x_low = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    f = interp1d(x_low, h_low, kind="linear", fill_value="extrapolate", assume_sorted=True)
    h_new = f(x_new).astype(float)
    return project_to_bounds_and_sum(h_new, n / 2.0)


# ------------------------------------------------------------
# Local optimisation – targeted hill‑climbing
# ------------------------------------------------------------
def hill_climb_targeted(
    h0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Greedy improvement that repeatedly moves a little mass from a point
    with large gradient (w.r.t. the current worst shift) to a point with
    small gradient.  The sum constraint is kept exactly by transferring
    the same amount.
    """
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    cur_val = best_val
    max_delta = 0.05                     # initial step size

    for it in range(max_iters):
        # time guard – stop a little before the global deadline
        if time.time() > deadline - 0.02:
            break

        # find the shift that attains the current maximum overlap
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)

        # gradient tells which indices increase/decrease M(shift) the most
        g = gradient_for_shift(h, shift)
        i = int(np.argmax(g))          # candidate to decrease
        j = int(np.argmin(g))          # candidate to increase

        # move a small amount from i to j, respecting the box limits
        delta_max = min(max_delta, h[i], 1.0 - h[j])
        if delta_max > 1e-12:
            delta = rng.random() * delta_max
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            new_val = evaluate_c5(h_new)
            if new_val <= cur_val:
                h = h_new
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h_new.copy()

        # occasional random transfer to avoid stagnation
        if (it & 511) == 0:            # every 512 iterations
            i2, j2 = rng.integers(0, n, size=2)
            if i2 != j2:
                delta_max2 = min(0.02, h[i2], 1.0 - h[j2])
                if delta_max2 > 1e-12:
                    delta2 = rng.random() * delta_max2
                    h_tmp = h.copy()
                    h_tmp[i2] -= delta2
                    h_tmp[j2] += delta2
                    val2 = evaluate_c5(h_tmp)
                    if val2 < cur_val:
                        h = h_tmp
                        cur_val = val2
                        if val2 < best_val:
                            best_val = val2
                            best_h = h_tmp.copy()

        # slowly shrink the maximum step size
        if (it & 1023) == 1023:        # every 1024 iterations
            max_delta = max(max_delta * 0.95, 1e-4)

    return best_h, best_val

    """
    Returns a step function h, its C5 bound, and the grid size.
    The implementation performs a two‑stage targeted hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---------- initialise ----------
    if "initial_h_values" in globals():
        low = np.asarray(initial_h_values, dtype=float)

        target_mid = n_mid / 2.0
        h_mid = rng.random(n_mid)
        h_mid = project_to_bounds_and_sum(h_mid, target_mid)

    # ---------- first optimisation phase (coarse) ----------
    remaining = budget_s - (time.time() - start)
    if remaining > 30.0:

    remaining = budget_s - (time.time() - start)
