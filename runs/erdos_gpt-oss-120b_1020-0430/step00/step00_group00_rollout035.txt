def compute_c5_and_shift(h):
    n = len(h)
    dx = 2.0 / n

    shift_mod = shift % n
    return max_val, shift_mod

def adjust_sum(vec):
    target = len(vec) / 2.0
    diff = target - np.sum(vec)
    # small residual correct
    if abs(diff) < 1e-12:
        return vec
    # try up to few attempts
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # pick random index where adjusting keeps within bounds
        idx = random.randint(0, len(vec)-1)
        new_val = vec[idx] + diff
        if new_val < 0.0:
            # cannot decrease below 0; set to 0 and adjust diff accordingly
            diff += vec[idx]  # because we set to 0, we removed vec[idx]; diff to compensate.
            vec[idx] = 0.0
        elif new_val > 1.0:
            diff -= (1.0 - vec[idx])  # we will increase to 1, leaving diff left.
            vec[idx] = 1.0
        else:
            vec[idx] = new_val
            diff = target - np.sum(vec)
    return vec

import time
import random
import math
import numpy as np

def _c5_and_shift(h: np.ndarray):
    """Return (C5 value, shift index giving the worst overlap) using the
    evaluation rule from the harness (np.correlate with mode='full')."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    c5 = float(corr[idx])
    shift = idx - (n - 1)          # signed shift in [-n+1, n-1]
    shift_mod = shift % n          # cyclic version for convenience
    return c5, shift_mod


def _fix_sum(vec: np.ndarray):
    """Adjust a feasible vector so that its integral equals 1
    (i.e. sum = n/2) while keeping 0≤h≤1."""
    n = vec.shape[0]
    target = n / 2.0
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    # try to fix the tiny residual by touching a few random coordinates
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        i = random.randrange(n)
        new_val = vec[i] + diff
        if new_val < 0.0:
            diff += vec[i]            # set to 0, keep leftover diff
            vec[i] = 0.0
        elif new_val > 1.0:
            diff -= (1.0 - vec[i])    # set to 1, keep leftover diff
            vec[i] = 1.0
        else:
            vec[i] = new_val
            diff = target - vec.sum()
    return vec


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    total_time = budget_s

    # ------------------------------------------------------------------
    # initialise a feasible step function
    init = globals().get("initial_h_values")
    if init is not None:
        base = np.asarray(init, dtype=float)
        base_n = base.shape[0]
        # up‑sample a bit to gain extra degrees of freedom (keep ≤300 points)
        factor = min(6, max(2, 300 // base_n))
        h = np.repeat(base, factor)
    else:
        # fall‑back: 120 points, random Dirichlet‑like start, then project
        n = 120
        raw = np.random.rand(n)
        raw = raw / raw.sum() * (n / 2.0)
        h = np.clip(raw, 0.0, 1.0)
        h = _fix_sum(h)

    h = np.clip(h, 0.0, 1.0)
    h = _fix_sum(h)

    # ------------------------------------------------------------------
    # keep the best found solution
    best_h = h.copy()
    best_c, _ = _c5_and_shift(best_h)

    cur_h = h.copy()
    cur_c = best_c

    # parameters for the simulated‑annealing / hill‑climbing loop
    step_start = 0.5   # max fraction of the feasible transfer at the beginning
    step_end = 0.01    # final max fraction
    T0 = 0.001         # temperature scale for occasional uphill moves

    while True:
        elapsed = time.time() - start_time
        if elapsed >= total_time * 0.95:
            break

        # linear decay of the allowed step size
        frac = 1.0 - elapsed / total_time
        step_frac = step_end + (step_start - step_end) * frac

        # current worst shift and its gradient information
        cur_c, cur_shift = _c5_and_shift(cur_h)

        n = cur_h.shape[0]
        if cur_shift == 0:                     # degenerate – pick random pair
            i, j = random.sample(range(n), 2)
        else:
            # gradient of the worst overlap w.r.t each coordinate
            plus = np.roll(cur_h, -cur_shift)
            minus = np.roll(cur_h, cur_shift)
            grad = plus + minus

            inc_mask = cur_h < 1.0 - 1e-12
            dec_mask = cur_h > 1e-12
            inc_idx = np.where(inc_mask)[0]
            dec_idx = np.where(dec_mask)[0]

            if inc_idx.size == 0 or dec_idx.size == 0:

            else:
                i = inc_idx[np.argmax(grad[inc_idx])]   # best place to add mass

        # maximum admissible transfer respecting box constraints

        if new_c < cur_c - 1e-12:

            if new_c < best_c - 1e-12:
