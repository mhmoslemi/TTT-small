import time
import numpy as np

def _max_corr(vec: np.ndarray) -> int:
    """Return the integer max correlation for a binary (or 0‑1) vector."""
    # np.correlate returns float for float inputs; cast to int after max
    return int(np.max(np.correlate(vec, 1 - vec, mode="full")))


def _mismatched_indices(vec: np.ndarray, shift: int) -> np.ndarray:
    """Indices i where vec[i]==1 and vec[i+shift]==0 (within overlap)."""
    n = vec.shape[0]
    if shift >= 0:
        i = np.arange(0, n - shift, dtype=int)
        mask = (vec[i] == 1) & (vec[i + shift] == 0)
    else:
        s = -shift
        i = np.arange(s, n, dtype=int)
        mask = (vec[i] == 1) & (vec[i - s] == 0)
    return i[mask]


def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary vector of length n with exactly n/2 ones."""
    vec = np.zeros(n, dtype=int)
    ones = rng.choice(n, n // 2, replace=False)
    vec[ones] = 1
    return vec


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # length of the discretisation
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
    else:
        n = 70                     # a modest size that works well
        init = None

    dx = 2.0 / n
    target_c5 = 0.38085
    target_corr = int(np.floor(target_c5 * n / 2))

    # initialise best solution
    if init is not None:
        # ensure feasibility of the supplied start (clip and fix sum)
        h = np.clip(init, 0.0, 1.0)
        cur_sum = h.sum()
        # adjust the last entry to get the exact required sum
        needed = n / 2 - cur_sum
        h[-1] = np.clip(h[-1] + needed, 0.0, 1.0)
        # if still not integral we simply fall back to a random binary start
        if not np.isclose(h.sum() * dx, 1.0):
            h = _random_binary(n, rng).astype(float)
    else:
        h = _random_binary(n, rng).astype(float)

    # work with an integer binary version for the core optimisation
    cur_vec = (h > 0.5).astype(int)
    best_vec = cur_vec.copy()
    best_corr = _max_corr(best_vec)

    start_time = time.time()
    time_limit = budget_s * 0.95  # a safety margin

    # parameters for the local search
    max_no_improve = 3000       # restart after this many failed attempts
    outer_restart_limit = 1000  # safety cap; will be limited by time

    for _ in range(outer_restart_limit):
        if time.time() - start_time > time_limit:
            break

        # start from a fresh random binary vector
        cur_vec = _random_binary(n, rng)
        no_improve = 0

        while True:
            if time.time() - start_time > time_limit:
                break

            corr = np.correlate(cur_vec, 1 - cur_vec, mode="full")
            cur_max = int(corr.max())

            if cur_max < best_corr:
                best_corr = cur_max
                best_vec = cur_vec.copy()
                if best_corr <= target_corr:
                    # optimal enough – stop early
                    break

            # pick a worst shift (any of the maxima)
            worst_idxs = np.flatnonzero(corr == cur_max)
            idx = rng.choice(worst_idxs)
            shift = int(idx - (n - 1))

            # try to fix a mismatched pair for this shift
            mism = _mismatched_indices(cur_vec, shift)
            if mism.size:
                i = int(rng.choice(mism))
                j = i + shift
                # perform the 1↔0 swap
                cur_vec[i], cur_vec[j] = cur_vec[j], cur_vec[i]
                new_max = _max_corr(cur_vec)

                if new_max < cur_max:
                    no_improve = 0

                cur_vec[i], cur_vec[j] = cur_vec[j], cur_vec[i]
                no_improve += 1
            else:
                # no direct mismatches for this shift → random 1/0 swap
                ones = np.flatnonzero(cur_vec == 1)
                zeros = np.flatnonzero(cur_vec == 0)
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                cur_vec[i], cur_vec[j] = cur_vec[j], cur_vec[i]
                new_max = _max_corr(cur_vec)

                if new_max < cur_max:
                    no_improve = 0
                    continue
                cur_vec[i], cur_vec[j] = cur_vec[j], cur_vec[i]
                no_improve += 1

            if no_improve >= max_no_improve:
                break

        if best_corr <= target_corr:
            break

    # final answer – convert back to float step‑function values
    h_values = best_vec.astype(float).tolist()
    c5_bound = best_corr * dx
    n_points = n
    return h_values, c5_bound, n_points
