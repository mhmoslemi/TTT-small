def _project_onto_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:

        else:

def _corr_vector(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1‑h)."""
    return np.correlate(h, 1.0 - h, mode="full")


def _max_corr(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral C₅ for a given h."""
    return np.max(_corr_vector(h)) * dx


def _gradient_of_max_corr(h: np.ndarray, dx: float) -> np.ndarray:
    """Sub‑gradient of maxₖ∫h(1−hₖ)dx with respect to h."""
    n = h.shape[0]
    corr = _corr_vector(h)
    max_val = corr.max()
    # indices (in full length 2n‑1) attaining the maximum
    mask = np.isclose(corr, max_val, atol=1e-12)
    idxs = np.nonzero(mask)[0]
    grad = np.zeros_like(h)
    for j in idxs:
        s = j - (n - 1)          # shift corresponding to index j

        else:
            s = -s
            grad[s:] += (1.0 - h[: n - s])
