    # propose transfer delta bounded by possible move within [0,1] for both

    return new_h

def random_transfer(h):

        return None
    delta = max_transfer * np.random.rand()

    new_h[dst] += delta

import time, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_{shift} ∫ h(x)*(1-h(x+shift)) dx approximated by correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def random_transfer(h: np.ndarray) -> np.ndarray | None:
    """Move a random amount of mass from one index to another (preserves sum)."""
    n = h.size
    i = np.random.randint(0, n)
    j = np.random.randint(0, n - 1)
    if j >= i:
        j += 1
    # random direction
    if np.random.rand() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_delta = min(h[src], 1.0 - h[dst])
    if max_delta <= 0.0:
        return None
    # occasionally take the whole transferable amount
    if np.random.rand() < 0.5:
        delta = max_delta
    else:
        delta = max_delta * np.random.rand()
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    return new_h

def targeted_transfer(h: np.ndarray) -> np.ndarray | None:
    """Transfer mass between indices that belong to the currently worst lag."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)          # shift in [-n+1, n-1]
    if shift >= 0:
        # contributions: h[i] * (1 - h[i+shift]), i = 0..n-shift-1
        prod = h[: n - shift] * (1.0 - h[shift:])
        i_local = int(np.argmax(prod))
        src = i_local
        dst = i_local + shift
    else:
        s = -shift
        # contributions: h[i] * (1 - h[i-s]), i = s..n-1
        prod = h[s:] * (1.0 - h[: n - s])
        i_local = int(np.argmax(prod))
        src = i_local + s
        dst = i_local
    max_delta = min(h[src], 1.0 - h[dst])
    if max_delta <= 0.0:
        return None
    if np.random.rand() < 0.5:
        delta = max_delta
    else:
        delta = max_delta * np.random.rand()
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    return new_h

def generate_random_binary(n: int) -> np.ndarray:
    """Create a feasible binary (or half‑binary for odd n) vector with sum=n/2."""
    h = np.zeros(n, dtype=np.float64)
    ones = n // 2
    idx = np.random.choice(n, ones, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        # need an extra 0.5 to reach n/2
        remaining = np.setdiff1d(np.arange(n), idx)
        if remaining.size:
            h[np.random.choice(remaining)] = 0.5
    return h

def _fix_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Tiny correction to enforce the exact integral (sum=h*dx = 1)."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # distribute uniformly, then clip
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    # final tiny tweak if needed
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

    """Search for a step function h on [0,2] with ∫h=1 that minimizes C₅."""
    np.random.seed(seed)

    start = time.time()
    # ------------------------------------------------------------------

    no_improve = 0
    iteration = 0

    # main loop
    while time.time() - start < budget_s * 0.96:

    final_score = compute_c5(best_h)

    return best_h.tolist(), float(final_score), n
