def local_search(h0, rng, start_time, budget_s):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c5 = best_c5

    # parameters
    max_iters = 50000
    # Determine remaining time:
    while time.time() - start_time < budget_s * 0.9 and iter < max_iters:
        # schedule step size:
        # step size decays linearly based on iteration or temperature.
        progress = iter / max_iters
        step = 0.1 * (1 - progress) + 0.001   # from 0.1 -> 0.001
        # choose i with cur_h[i] > 0, j with cur_h[j] < 1
        # select two indices randomly and then adjust epsilon.
        i = rng.integers(n)
        j = rng.integers(n)
        # ensure we can move mass from i to j
        # compute max move
        max_move_i = cur_h[i]  # can take this amount
        max_move_j = 1.0 - cur_h[j]  # can add this
        max_move = min(max_move_i, max_move_j, step)
        if max_move <= 1e-12:
            continue
        epsilon = rng.random() * max_move
        # propose new_h
        new_h = cur_h.copy()
        new_h[i] -= epsilon
        new_h[j] += epsilon
        new_c5 = compute_c5(new_h)
        # acceptance
        # simple hill climbing: accept if better, else maybe accept with prob.
        # We'll implement simulated annealing with temperature schedule.
        T = 0.01 * (1 - progress) + 0.0001  # small
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / max(T,1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            # update best
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
        iter += 1

    return best_h, best_c5

    n = len(h_start)
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    max_iters = 50000
    # compute allowed iterations based on time left
    max_time = deadline - time.time()
    # If max_time low, reduce iterations
    if max_time < 0:
        return best_h, best_c5
    # we estimate each iteration ~0.00001 seconds maybe; but simpler:
    if max_time < 5:
        max_iters = int(max_iters * max_time / 30.0)  # reduce
        max_iters = max(2000, max_iters)
    # else keep as is

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # schedule step and temperature
        progress = it / max_iters
        step = 0.2 * (1 - progress) + 0.001  # max move amplitude
        T = 0.01 * (1 - progress) + 0.0001

        # choose i and j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_move_i = cur_h[i]
        max_move_j = 1.0 - cur_h[j]
        max_move = min(max_move_i, max_move_j, step)
        if max_move <= 0:
            continue
        epsilon = rng.random() * max_move
        # new vector
        new_h = cur_h.copy()
        new_h[i] -= epsilon
        new_h[j] += epsilon
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5 or rng.random() < math.exp((cur_c5 - new_c5) / max(T, 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
    return best_h, best_c5

        conv = np.convolve(h, 1.0 - h, mode='full')
        max_val = conv.max()

        shift = s_max - (n - 1)

        # Use vectorized: i indexes
        i = np.arange(n)

        plus = i + shift
        minus = i - shift
        term_plus = np.where((plus>=0) & (plus<n), h[plus], 0.0)

def project_onto_simplex_bounded(v, target):
    n = v.shape[0]
    # Ensure target within [0,n]
    if target < 0:
        target = 0.0
    elif target > n:
        target = float(n)
    lo = np.min(v) - 1.0 - 1e-8
    hi = np.max(v) + 1.0 + 1e-8
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust any tiny deviation
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # adjust first element not at bound
        for i in range(n):
            if diff > 0 and proj[i] < 1.0:

                proj[i] += inc
                diff -= inc
            elif diff < 0 and proj[i] > 0.0:

                diff += dec

import time, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max overlap C5 for step vector h (dx = 2/n)."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.convolve(h, 1.0 - h, mode='full')
    return float(conv.max() * dx)


def _project_onto_bounded_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x_i<=1, sum x_i = target} (Euclidean projection)."""
    n = v.size
    if target <= 0:
        return np.zeros_like(v)
    if target >= n:
        return np.ones_like(v)

    lo = np.min(v) - 1.0 - 1e-9
    hi = np.max(v) + 1.0 + 1e-9
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for floating error
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # distribute the remaining diff to non‑boundary entries
        idx = np.where((proj > 0.0) & (proj < 1.0))[0]
        if idx.size == 0:
            idx = np.arange(n)
        for i in idx:
            if diff > 0:
                inc = min(diff, 1.0 - proj[i])
                proj[i] += inc
                diff -= inc
            else:
                dec = min(-diff, proj[i])
                proj[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return proj


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (sum = n/2, 0≤h≤1)."""
    target = n / 2.0
    v = rng.random(n)
    return _project_onto_bounded_simplex(v, target)


def _local_search(h_start: np.ndarray, rng: np.random.Generator,
                  start_time: float, deadline: float) -> tuple:
    """Simple stochastic hill‑climbing with occasional uphill moves."""
    n = h_start.shape[0]
    target = n / 2.0
    best_h = h_start.copy()
    best_val = _compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    # number of iterations depends on remaining time
    max_iters = 40000
    if deadline - time.time() < 10.0:
        max_iters = int(max_iters * 0.2)
    max_iters = max(2000, max_iters)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # annealing schedule
        progress = it / max_iters
        step = 0.2 * (1.0 - progress) + 0.001
        T = 0.01 * (1.0 - progress) + 0.0001

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        max_move_i = cur_h[i]
        max_move_j = 1.0 - cur_h[j]
        max_move = min(max_move_i, max_move_j, step)
        if max_move <= 0.0:
            continue

        eps = rng.random() * max_move
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps

        new_val = _compute_c5(new_h)

        if new_val < cur_val or rng.random() < math.exp((cur_val - new_val) / max(T, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val
    # final projection to satisfy equality exactly
    best_h = _project_onto_bounded_simplex(best_h, target)
    best_val = _compute_c5(best_h)
    return best_h, best_val


def _subgradient_descent(h_start: np.ndarray, rng: np.random.Generator,
                         deadline: float) -> tuple:
    """Projected sub‑gradient descent on max‑overlap objective."""
    n = h_start.shape[0]
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_val = _compute_c5(h)

    max_iter = 8000
    for it in range(max_iter):
        if time.time() > deadline:
            break
        conv = np.convolve(h, 1.0 - h, mode='full')
        s_max = int(conv.argmax())
        shift = s_max - (n - 1)                     # shift of the active correlation
        idx = np.arange(n)
        plus = idx + shift
        minus = idx - shift
        term_plus = np.where((plus >= 0) & (plus < n), h[plus], 0.0)
        term_minus = np.where((minus >= 0) & (minus < n), h[minus], 0.0)
        grad = 1.0 - term_plus - term_minus          # sub‑gradient of the active term
        eta = 0.5 / math.sqrt(it + 1.0)              # diminishing step size
        h_new = h - eta * grad
        h_new = _project_onto_bounded_simplex(h_new, target)
        val = _compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    best_h = _project_onto_bounded_simplex(best_h, target)
    best_val = _compute_c5(best_h)
    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
    h_values is a numpy array of step heights,
    c5_bound is the achieved max overlap,
    and n_points is the grid size used.
    """
    start_time = time.time()
    deadline = start_time + budget_s * 0.97   # keep a safety margin

    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # 1) Use the supplied construction if it exists
    # ------------------------------------------------------------------
    init_h = globals().get('initial_h_values', None)
    if init_h is not None:
        init_h = np.asarray(init_h, dtype=np.float64).copy()
        n0 = init_h.shape[0]
        init_h = _project_onto_bounded_simplex(init_h, n0 / 2.0)
        # a few local refinements from the given start
        for _ in range(3):
            if time.time() > deadline:
                break
            opt_h, opt_c5 = _local_search(init_h, rng, start_time, deadline)
            if opt_c5 < best_c5:
                best_h, best_c5, best_n = opt_h, opt_c5, n0
            init_h = opt_h  # continue from the improved point

    # ------------------------------------------------------------------
    # 2) If nothing sensible yet, start from a simple binary pattern
    # ------------------------------------------------------------------
    if best_h is None:
        n0 = 80
        h0 = np.zeros(n0, dtype=np.float64)
        h0[: n0 // 2] = 1.0
        best_h, best_c5, best_n = h0, _compute_c5(h0), n0

    # ------------------------------------------------------------------
    # 3) Explore larger grids by interpolation + local search
    # ------------------------------------------------------------------
    n = max(80, best_n)
    while time.time() < deadline and n <= 376:
        if n != best_n:
            # interpolate the current best to the new grid size
            old_n = best_n
            x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_start = _project_onto_bounded_simplex(h_interp, n / 2.0)
        else:
            h_start = best_h.copy()

        # several random restarts on this grid
        attempts = 4
        for att in range(attempts):
            if time.time() > deadline:
                break
            if att == 0:
                seed_h = h_start
            else:
                seed_h = _random_feasible(n, rng)
            opt_h, opt_c5 = _local_search(seed_h, rng, start_time, deadline)
            if opt_c5 < best_c5:
                best_h, best_c5, best_n = opt_h, opt_c5, n
                # early exit if we already beat the target
                if best_c5 <= 0.3808:
                    break
        if best_c5 <= 0.3808:
            break
        n = min(n * 2, 376)

    # ------------------------------------------------------------------
    # 4) Final polishing with sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline and best_h is not None:
        refined_h, refined_c5 = _subgradient_descent(best_h, rng, deadline)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    # ------------------------------------------------------------------
    # 5) Final clean‑up and return
    # ------------------------------------------------------------------
    if best_h is None:
        raise RuntimeError("Failed to produce a feasible solution.")
    best_h = _project_onto_bounded_simplex(best_h, best_n / 2.0)
    final_c5 = _compute_c5(best_h)

    return best_h, float(final_c5), int(best_n)
