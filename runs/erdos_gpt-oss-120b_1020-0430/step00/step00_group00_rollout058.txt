import time, math, random
import numpy as np
from scipy.optimize import minimize

def _c5_of(h: np.ndarray) -> float:
    """compute C5 = max_k ∫ h(x)(1‑h(x+k))dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _binary_start(n: int) -> np.ndarray:
    """random binary (plus a single 0.5 if n is odd) respecting ∑h = n/2."""
    m = n // 2
    h = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, size=m, replace=False)
    h[ones] = 1.0
    if n % 2:
        # one extra 0.5 to hit the exact half‑mass
        remaining = np.setdiff1d(np.arange(n), ones)
        h[random.choice(remaining.tolist())] = 0.5
    return h


def _local_refine(h0: np.ndarray) -> np.ndarray:
    """few SLSQP steps starting from h0 (keeps constraints)."""
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def obj(x):
        return _c5_of(x)

    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 250, "ftol": 1e-10, "disp": False},
    )
    if not res.success:
        return h0
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction to exact sum (float drift)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        # adjust any coordinate that can absorb the correction
        for i in range(n):
            if 0 <= x[i] + diff <= 1:
                x[i] += diff
                break
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h on [0,2] that (hopefully) improves the
    known upper bound for the Erdős C5 constant.

    Returns (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1. initialise h
    # ------------------------------------------------------------------
    try:
        # initial_h_values is supplied by the harness when a parent exists
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
        target = n / 2.0
        # tiny projection onto the hyperplane in case of rounding error
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            # spread the correction over a random admissible coordinate
            for i in range(n):
                if 0 <= h[i] + diff <= 1:
                    h[i] += diff
                    break
    except NameError:
        # no parent – start from a random binary configuration (n=45)
        n = 45
        h = _binary_start(n)

    best_h = h.copy()
    best_c5 = _c5_of(best_h)

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing style hill‑climber
    # ------------------------------------------------------------------
    start_time = time.time()
    budget = float(budget_s)
    max_delta = 0.3          # maximal single‑coordinate shift
    init_temp = 0.05         # start temperature
    it_without_imp = 0
    max_no_imp = 18000       # restart after many non‑improving steps

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget - 0.2:
            break

        # propose a pairwise adjustment that keeps the total mass exact
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]

        # feasible interval for delta so that 0<=h[i]+δ<=1 and 0<=h[j]‑δ<=1
        lo = max(-hi, hj - 1.0)
        hi_d = min(1.0 - hi, hj)
        if lo >= hi_d:
            continue
        lo = max(lo, -max_delta)
        hi_d = min(hi_d, max_delta)
        if lo >= hi_d:
            continue

        delta = np.random.uniform(lo, hi_d)
        new_h = best_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_c5 = _c5_of(new_h)

        # acceptance rule
        if new_c5 < best_c5 - 1e-12:
            best_h, best_c5 = new_h, new_c5
            it_without_imp = 0
        else:
            # temperature linearly decays to zero
            progress = elapsed / budget
            T = init_temp * (1.0 - progress)
            if T > 0:
                prob = math.exp(-(new_c5 - best_c5) / T)
                if np.random.rand() < prob:
                    best_h, best_c5 = new_h, new_c5
            it_without_imp += 1

        # occasional random restart to escape deep plateaus
        if it_without_imp > max_no_imp:
            best_h = _binary_start(n)
            best_c5 = _c5_of(best_h)

    refined = _local_refine(best_h)
    refined_c5 = _c5_of(refined)

        for i in range(n):

                break
