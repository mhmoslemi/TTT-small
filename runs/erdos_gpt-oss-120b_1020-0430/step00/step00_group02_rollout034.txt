def project_to_feasible(y, target):
    lo = -2.0  # enough negative
    hi = 2.0  # enough positive
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - lo, 0.0, 1.0)
    # After projection, due to numeric tolerance, maybe sum slightly off; we can adjust small difference by adding to some entries below 1 uniformly.
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff across entries not at bound
        free = (x > 0.0 + 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust first entry
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

def improve_h_greedy(h, dx, max_iter, step_scale, rng):
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    conv = np.convolve(h, h[::-1], mode='full')
    min_val = conv.min() * dx
    best_val = 1 - min_val
    for it in range(max_iter):
        # compute convolution J
        conv = np.convolve(h, h[::-1], mode='full')
        min_idx = np.argmin(conv)
        min_J = conv[min_idx] * dx
        cur_val = 1 - min_J
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
        # compute shift
        shift = min_idx - (n - 1)
        s = abs(shift)
        if s == 0:
            # already max correlation? Actually we need for shift 0; maybe continue.
            # Random perturbation:
            i1, i2 = rng.choice(n, size=2, replace=False)
            # adjust random delta
            low = max(-h[i1], h[i2] - 1)
            high = min(1 - h[i1], h[i2])
            if low >= high:
                continue
            delta = rng.uniform(low, high) * step_scale
            h[i1] += delta
            h[i2] -= delta
            continue
        # compute gradient for shift (dJ/dh)
        d = np.zeros_like(h)
        if shift >= 0:
            # shift positive
            d[:n-s] += h[s:]  # h[i+s]
            d[s:] += h[:n-s]  # h[i-s]
        else:

            d[:n-s] += h[s:]  # same
            d[s:] += h[:n-s]
        # indices allowable for increase/decrease

        dec_candidates = np.where(h > 1e-12)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            break

        # keep sum constant automatically
        # optionally small random jitter:
        # maybe occasionally break out
    # final best
    return best_h, best_val

def improve_h_greedy(h, max_iter, step_scale, rng):
    n = len(h)
    target_sum = n/2.0
    best_h = h.copy()
    J = compute_J(h)
    best_val = 1 - J.min()
    for it in range(max_iter):
        J = compute_J(h)
        min_idx = np.argmin(J)
        cur_val = 1 - J[min_idx]
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
        s = min_idx  # shift index (0..n-1)
        # compute derivative vector
        d = np.roll(h, -s) + np.roll(h, s)
        # For s == 0, d = 2*h
        # select inc/dec candidates
        inc_candidates = np.where(h < 1 - 1e-12)[0]
        dec_candidates = np.where(h > 1e-12)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            break
        inc_vals = d[inc_candidates]
        dec_vals = d[dec_candidates]
        i_inc = inc_candidates[np.argmax(inc_vals)]
        i_dec = dec_candidates[np.argmin(dec_vals)]
        # compute possible delta
        max_delta = min(1 - h[i_inc], h[i_dec])
        if max_delta <= 1e-12:
            # random small move instead
            i1, i2 = rng.choice(n, size=2, replace=False)
            low = max(-h[i1], h[i2] - 1)
            high = min(1 - h[i1], h[i2])
            if low >= high:
                continue
            delta = rng.uniform(low, high) * step_scale
            h[i1] += delta
            h[i2] -= delta
            continue
        delta = step_scale * max_delta
        if delta <= 1e-12:
            delta = max_delta * 0.5
        h[i_inc] += delta
        h[i_dec] -= delta
    return best_h, best_val

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper: project a vector onto the box [0,1]^n intersecting the hyperplane sum = target
def _project_to_feasible(y, target):
    lo, hi = -1.0, 1.0
    # binary search for lambda such that sum(clip(y - lam,0,1)) = target
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - lo, 0.0, 1.0)
    # tiny correction for numeric drift
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            adj = diff / free.sum()
            x[free] += adj
            x = np.clip(x, 0.0, 1.0)
        else:
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

# ----------------------------------------------------------------------
# Periodic autocorrelation J(s) = ∫ h(x)h(x+s) dx  (s integer shift on the grid)
def _compute_J(h):
    n = h.shape[0]
    # circular convolution via FFT, real-valued
    H = np.fft.rfft(h)
    corr = np.fft.irfft(H * np.conj(H), n=n)
    dx = 2.0 / n
    return corr * dx  # J array, length n

# ----------------------------------------------------------------------
# Greedy improvement that raises the minimum autocorrelation
def _greedy_improve(h, max_iter, step_scale, rng):
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_val = 1.0 - _compute_J(h).min()  # C5 value
    for it in range(max_iter):
        J = _compute_J(h)
        s = int(np.argmin(J))           # worst shift
        cur = 1.0 - J[s]
        if cur < best_val - 1e-12:
            best_val = cur
            best_h = h.copy()
        # gradient for this shift (circular indices)
        d = np.roll(h, -s) + np.roll(h, s)
        # candidates for increase / decrease
        inc_idx = np.where(h < 1.0 - 1e-12)[0]
        dec_idx = np.where(h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break
        i_inc = inc_idx[np.argmax(d[inc_idx])]
        i_dec = dec_idx[np.argmin(d[dec_idx])]
        max_delta = min(1.0 - h[i_inc], h[i_dec])
        if max_delta <= 1e-12:
            # fallback random move
            i1, i2 = rng.choice(n, 2, replace=False)
            low = max(-h[i1], h[i2] - 1.0)
            high = min(1.0 - h[i1], h[i2])
            if low >= high:
                continue
            delta = rng.uniform(low, high) * step_scale
            h[i1] += delta
            h[i2] -= delta
            continue
        delta = step_scale * max_delta
        if delta <= 1e-12:
            delta = max_delta * 0.5
        h[i_inc] += delta
        h[i_dec] -= delta
    return best_h, best_val

# ----------------------------------------------------------------------
# Optional final local refinement with SLSQP (if time permits)
def _slsqp_refine(h, dx, target_sum, time_left):
    # tiny time guard – skip if not enough budget
    if time_left < 1.0:
        return h, 1.0 - _compute_J(h).min()
    n = h.shape[0]
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    def obj(x):
        J = _compute_J(x)
        return 1.0 - J.min()

    opts = {'maxiter': 200, 'ftol': 1e-9, 'disp': False}
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options=opts)
    if res.success:
        h2 = np.clip(res.x, 0.0, 1.0)
        h2 = _project_to_feasible(h2, target_sum)
        val = 1.0 - _compute_J(h2).min()
        return h2, val
    return h, 1.0 - _compute_J(h).min()

# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = float(budget_s)

    # ------------------------------------------------------------------
    # Determine base size and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        base_n = h0.size
    else:
        base_n = 45
        h0 = np.full(base_n, 0.5, dtype=np.float64)

    # upscale to a larger, even size (factor 4 if reasonable)
    factor = 4
    if base_n * factor > 300:
        factor = 2
    n = base_n * factor
    if factor > 1:
        h0 = np.repeat(h0, factor)
    # ensure feasibility
    target_sum = n / 2.0
    h0 = _project_to_feasible(h0, target_sum)

    # ------------------------------------------------------------------
    # Main multi‑restart greedy search
    best_h = h0.copy()
    best_val = 1.0 - _compute_J(best_h).min()  # current C5
    max_rest = 200  # generous upper bound, will be limited by time
    rest = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget - 0.5 or rest >= max_rest:
            break
        # generate a new random feasible start
        y = rng.random(n)
        h = _project_to_feasible(y, target_sum)
        # improve
        h_opt, val_opt = _greedy_improve(h, max_iter=4000,
                                         step_scale=0.5, rng=rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            # early exit if we already beat target record
            if best_val < 0.3808:
                break
        rest += 1

    # ------------------------------------------------------------------
    # Final refinement with SLSQP if time permits
    remaining = budget - (time.time() - start_time)
    best_h, best_val = _slsqp_refine(best_h,
                                    dx=2.0 / n,
                                    target_sum=target_sum,
                                    time_left=remaining)

    # ------------------------------------------------------------------
    # Ensure constraints exactly
    best_h = _project_to_feasible(best_h, target_sum)
    c5_bound = 1.0 - _compute_J(best_h).min()
    return (best_h, float(c5_bound), int(n))
