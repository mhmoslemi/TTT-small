import numpy as np, time

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    # waterfilling
    # Use simple iterative scheme.
    for _ in range(10):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        nfree = free.sum()
        if nfree == 0:
            # cannot adjust further
            break
        delta = diff / nfree
        h[free] += delta
        # reclip
        h = np.clip(h, 0.0, 1.0)
    # final fix for small diff
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # add diff to a free position if possible
        idxs = np.where((h + diff >= 0) & (h + diff <= 1))[0]
        if idxs.size > 0:
            h[idxs[0]] += diff
        else:
            # fallback adjust first element within bounds
            h[0] = np.clip(h[0] + diff, 0, 1)
    return h
```

def grad_for_lag(h, lag, dx):
    n = len(h)
    grad = np.zeros_like(h)
    if lag >= 0:
        # positive
        n1 = n - lag
        grad[:n1] += (1 - h[lag:]) * dx
        grad[lag:] -= h[:n1] * dx
    else:
        m = -lag
        n1 = n - m
        grad[m:] += (1 - h[:n1]) * dx
        grad[:n1] -= h[m:] * dx
    return grad
```

from scipy.special import logsumexp

TARGET_SUM = None
CURRENT_ALPHA = None
DX = None

def project_to_feasible(v, target):
    # clip
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        total = v.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-8:
        idx = np.where((v + diff >= 0) & (v + diff <= 1))[0]
        if idx.size:
            v[idx[0]] += diff
        else:
            v[0] = np.clip(v[0] + diff, 0.0, 1.0)

    best_val = np.max(np.correlate(best_h, 1 - best_h, mode='full') * DX)
    n = h.shape[0]
    for _ in range(max_iter):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        if max_inc <= 0.0 or max_dec <= 0.0:
            continue
        delta = np.random.rand() * min(max_inc, max_dec)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        val = np.max(np.correlate(h_new, 1 - h_new, mode='full') * DX)
        if val < best_val or np.random.rand() < np.exp(-(val - best_val) / temperature):
            h = h_new
            if val < best_val:
                best_val = val
                best_h = h_new.copy()
        # gradual cooling
        temperature *= 0.9999
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Optimize the Erdős C₅ bound."""
    np.random.seed(seed)
    start_time = time.time()

    # ---- initialise -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n = h0.shape[0]
    except Exception:                     # no parent construction available
        n = 200
        h0 = np.random.rand(n).astype(np.float64)

    global TARGET_SUM, CURRENT_ALPHA, DX
    DX = 2.0 / n
    TARGET_SUM = n / 2.0
    h0 = project_to_feasible(h0, TARGET_SUM)

    best_h = h0.copy()
    best_val = np.max(np.correlate(best_h, 1 - best_h, mode='full') * DX)

    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': sum_eq_constraint},)

    # ---- smooth‑max SLSQP optimisation -------------------------------
    for alpha in (5.0, 20.0, 50.0, 100.0):
        CURRENT_ALPHA = alpha
        res = minimize(
            smooth_max_obj,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 400, 'ftol': 1e-9, 'disp': False},
        )
        h_candidate = project_to_feasible(res.x, TARGET_SUM)
        cand_val = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full') * DX)
        if cand_val < best_val:
            best_val = cand_val
            best_h = h_candidate.copy()
        if time.time() - start_time > budget_s * 0.90:
            break

    # ---- optional local random refinement ---------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 2.0:                     # modest budget for refinement
        iters = int(min(200000, remaining * 1000))
        temp = 0.05
        best_h, best_val = local_random_search(best_h, iters, temp)

    # final projection and exact bound
    best_h = project_to_feasible(best_h, TARGET_SUM)
    final_c5 = float(np.max(np.correlate(best_h, 1 - best_h, mode='full') * DX))

    return best_h, final_c5, n
