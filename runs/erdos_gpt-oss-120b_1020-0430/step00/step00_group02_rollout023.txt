    best_h = np.copy(initial_h); best_score = compute_c5(best_h)
    cur_h = np.copy(initial_h); cur_score = best_score
    for t in range(n_iters):
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        # propose epsilon
        eps = (np.random.rand() - 0.5) * 0.2 # random shift magnitude maybe up to +-0.1?
        # Actually we need to keep sum unchanged, so we set delta = eps, apply to i, -eps to j.
        # Ensure bounds:
        if not (0 <= cur_h[i] + eps <= 1 and 0 <= cur_h[j] - eps <= 1):
            continue
        new_h = cur_h.copy()
        new_h[i] += eps
        new_h[j] -= eps
        new_score = compute_c5(new_h)
        # Accept if better or with probability exp((cur-score - new_score)/temp)
        # Since lower better.
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / (temp0 * (temp_decay**t) + 1e-12)):

            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
    return best_h, best_score

def random_feasible_h(n):

    target = n/2
    h = np.zeros(n)
    # Fill random values but respect bound
    # We'll sample random order and assign random portion of remaining
    indices = np.arange(n)
    np.random.shuffle(indices)
    rem = target
    for i in indices:
        if rem <= 0:
            break
        max_allowed = min(1.0, rem)
        # assign random amount up to max_allowed
        hi = np.random.rand() * max_allowed
        h[i] = hi
        rem -= hi
    # if remaining >0 due to rounding, distribute evenly among non-full entries
    if rem > 1e-12:
        free = np.where(h < 1 - 1e-12)[0]
        if free.size > 0:
            add = rem / free.size
            for i in free:
                inc = min(add, 1 - h[i])
                h[i] += inc
                rem -= inc
            # if still remains due to capacity, loop again
            if rem > 1e-12:
                # fill with tiny amounts (should not happen)
                i = np.argmin(h)
                h[i] += rem
    # final clip and adjust minor sum errors
    h = np.clip(h, 0, 1)
    # adjust sum precisely (maybe small diff)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-8:
        # distribute proportionally among non-bound entries
        slack = ((h > 1e-12) & (h < 1 - 1e-12))
        if slack.any():
            h[slack] += diff / slack.sum()
            h = np.clip(h, 0, 1)
        else:
            h = h * (target / cur)

import time
import numpy as np
from scipy.optimize import minimize

def _c5_value(h: np.ndarray) -> float:
    """Compute C₅ for a discrete step function h on [0,2]."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / h.size)


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    target = n * 0.5
    h = np.zeros(n, dtype=float)
    idx = np.arange(n)
    np.random.shuffle(idx)
    rem = target
    for i in idx:
        if rem <= 0:
            break
        max_allowed = min(1.0, rem)
        hi = np.random.rand() * max_allowed
        h[i] = hi
        rem -= hi
    # distribute any leftover mass uniformly over non‑full slots
    if rem > 1e-12:
        free = np.where(h < 1.0 - 1e-12)[0]
        if free.size:
            add = rem / free.size
            inc = np.minimum(add, 1.0 - h[free])
            h[free] += inc
            rem -= inc.sum()
    # final tiny adjustment to hit the exact sum
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        slack = (h > 1e-12) & (h < 1.0 - 1e-12)
        if slack.any():
            h[slack] += diff / slack.sum()
        else:
            h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    return h

        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        # repair possible tiny violations
        h0 = np.clip(h0, 0.0, 1.0)
        cur = h0.sum()
        if abs(cur - n * 0.5) > 1e-9:
            h0 *= (n * 0.5) / cur
        best_h = h0.copy()
        best_score = _c5_value(best_h)
    else:
        n = 45
        best_h = _random_feasible(n)
        best_score = _c5_value(best_h)

    # ------------------------------------------------------------------
    # Helper for a quick hill‑climb move that keeps the sum invariant.
    # ------------------------------------------------------------------
    def _neighbour(cur_vec, step):
        """Propose a neighbour by moving δ from one index to another."""
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            return None, None
        hi, hj = cur_vec[i], cur_vec[j]
        # admissible range for δ (increase i, decrease j)
        lo = max(-hi, -(hj))            # cannot go below 0
        hi_max = min(1.0 - hi, hj)      # cannot exceed 1 or go below 0 at j
        if hi_max <= lo:
            return None, None
        delta = (np.random.rand() * 2.0 - 1.0) * step
        delta = np.clip(delta, lo, hi_max)
        if abs(delta) < 1e-12:
            return None, None
        new = cur_vec.copy()
        new[i] += delta
        new[j] -= delta
        return new, delta

    # ------------------------------------------------------------------
    # Main optimisation loop (random restarts + hill‑climb + SLSQP).
    # ------------------------------------------------------------------
    step = 0.2
    while time.time() - start_time < 0.95 * budget_s:
        # Decide start point: either the current best or a fresh random.
        if np.random.rand() < 0.3 and best_score < 0.7:
            cur_vec = best_h.copy()
        else:
            cur_vec = _random_feasible(n)

        cur_score = _c5_value(cur_vec)

        # ---- Simple stochastic hill‑climb ---------------------------------
        max_iters = 2000
        for _ in range(max_iters):
            if time.time() - start_time > 0.95 * budget_s:
                break
            nxt, _ = _neighbour(cur_vec, step)
            if nxt is None:
                continue
            nxt_score = _c5_value(nxt)
            if nxt_score < cur_score:
                cur_vec, cur_score = nxt, nxt_score
                if nxt_score < best_score:
                    best_h, best_score = nxt.copy(), nxt_score
        # ---- Local refinement with SLSQP ----------------------------------
        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5},)
        bnds = [(0.0, 1.0) for _ in range(n)]
        res = minimize(_c5_value, cur_vec, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
        if res.success:
            sol = np.clip(res.x, 0.0, 1.0)
            # enforce the equality constraint more strictly
            diff = sol.sum() - n * 0.5
            if abs(diff) > 1e-8:
                slack = (sol > 1e-12) & (sol < 1.0 - 1e-12)
                if slack.any():
                    sol[slack] -= diff / slack.sum()
                else:
                    sol *= (n * 0.5) / sol.sum()
                sol = np.clip(sol, 0.0, 1.0)
            final = _c5_value(sol)
            if final < best_score:
                best_h, best_score = sol, final
        # slowly decay the move size
        step = max(0.01, step * 0.99)

    # ------------------------------------------------------------------
    # Final sanity fix (ensure constraints are met exactly).
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    total = best_h.sum()
    target = n * 0.5
    if abs(total - target) > 1e-9:
        slack = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if slack.any():
            best_h[slack] += (target - total) / slack.sum()
        else:
            best_h *= target / total
        best_h = np.clip(best_h, 0.0, 1.0)

    best_score = _c5_value(best_h)
    return best_h, best_score, n
