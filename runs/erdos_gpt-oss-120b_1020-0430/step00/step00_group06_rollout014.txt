def _project_box_sum(v: np.ndarray, target: float, tol: float = 1e-12) -> np.ndarray:
    """Project v onto {x | 0<=x<=1, sum(x)=target} using a simple water‑filling."""
    # λ such that x_i = clip(v_i - λ, 0, 1) and sum(x)=target
    lo = -max(1.0, np.max(v)) - 2.0   # makes x_i = 1 for all i
    hi = max(1.0, np.max(v)) + 2.0   # makes x_i = 0 for all i

def _c5_and_shift(h: np.ndarray) -> tuple[float, int]:
    """Return (C5, shift) where shift is the index offset that attains the max."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # shift range: -(n-1) … (n-1)
    return float(corr[idx]), shift

        it += 1

    # Final sanity check – enforce constraints tightly
    best_h = _project_box_sum(best_h, target_sum)
    best_c5, _ = _c5_and_shift(best_h)

    return best_h, best_c5, n
