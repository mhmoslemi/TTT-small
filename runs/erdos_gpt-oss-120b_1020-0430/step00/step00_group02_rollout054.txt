       # evaluate

    cur_h = h.copy()

        remaining = time_limit - elapsed

            # compute corr, s_max

            idx_max = np.argmax(corr)
            shift = idx_max - (n - 1)

            g = np.empty(n, dtype=float)

            # term2: h[i-shift] if i-shift in range else 0

import time, math
import numpy as np

def _eval_c5(h: np.ndarray) -> float:
    """Return C5 for a given vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _gradient_step(h: np.ndarray, rng: np.random.Generator) -> tuple:
    """Perform one gradient descent step on the worst shift."""
    n = h.size
    # correlation without dx – we only need the index of the worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)                     # shift corresponding to that index

    i = np.arange(n)
    ip = i + shift
    im = i - shift

    term1 = np.where((ip >= 0) & (ip < n), 1.0 - h[ip], 1.0)
    term2 = np.where((im >= 0) & (im < n), h[im], 0.0)
    g = term1 - term2                         # sub‑gradient of the worst term

    d = -g                                    # descent direction
    d -= d.mean()                             # enforce zero‑sum (preserve integral)

    # maximal feasible step respecting bounds
    pos = d > 1e-12
    neg = d < -1e-12
    alpha_max = np.inf
    if np.any(pos):
        alpha_max = min(alpha_max, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        alpha_max = min(alpha_max, np.min((-h[neg]) / d[neg]))
    if not np.isfinite(alpha_max) or alpha_max <= 0:
        return None, None

    # step size (decays later – callers decide scaling)
    return d, alpha_max


def _random_transfer(h: np.ndarray, rng: np.random.Generator) -> tuple:
    """Propose a random mass transfer between two coordinates."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None, None
    if rng.random() < 0.5:
        donor, receiver = i, j
    else:
        donor, receiver = j, i
    max_delta = min(h[donor], 1.0 - h[receiver])
    if max_delta <= 0.0:
        return None, None
    return donor, receiver, max_delta


def _local_optimize(h_init: np.ndarray, time_limit: float,
                    rng: np.random.Generator) -> tuple:
    """Hill‑climbing / simulated‑annealing with gradient steps."""
    h = h_init.copy()
    best_h = h.copy()
    best_score = _eval_c5(h)
    cur_score = best_score
    start = time.time()
    n = h.size

    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        elapsed = now - start
        remaining = time_limit - elapsed
        # blend gradient moves (70%) with random transfers (30%)
        if rng.random() < 0.7:
            # gradient move
            step = _gradient_step(h, rng)
            if step[0] is None:
                continue
            d, alpha_max = step
            # schedule a decreasing step magnitude
            step_scale = 0.5 * (1.0 - elapsed / time_limit) + 0.001
            alpha = min(step_scale, alpha_max)
            new_h = h + alpha * d
            np.clip(new_h, 0.0, 1.0, out=new_h)  # safety
            new_score = _eval_c5(new_h)
            if new_score < cur_score:
                h = new_h
                cur_score = new_score
                if new_score < best_score:
                    best_h = new_h.copy()
                    best_score = new_score
        else:
            # random transfer move
            tr = _random_transfer(h, rng)
            if tr[0] is None:
                continue
            donor, receiver, max_delta = tr
            step_scale = 0.5 * (1.0 - elapsed / time_limit) + 0.001
            delta = rng.random() * max_delta * step_scale
            new_h = h.copy()
            new_h[donor] -= delta
            new_h[receiver] += delta
            new_score = _eval_c5(new_h)
            if new_score < cur_score:
                h = new_h
                cur_score = new_score
                if new_score < best_score:
                    best_h = new_h.copy()
                    best_score = new_score
    return best_h, best_score


def _upscale(h: np.ndarray, factor: int,
             rng: np.random.Generator) -> np.ndarray:
    """Upscale vector by integer factor with tiny random jitter."""
    h_big = np.repeat(h, factor).astype(float)
    # add a small perturbation to break perfect repetition
    jitter = rng.normal(scale=0.008, size=h_big.shape)
    h_big += jitter
    np.clip(h_big, 0.0, 1.0, out=h_big)
    # renormalise the integral exactly
    target = h_big.size / 2.0
    cur = h_big.sum()
    if cur > 0:
        h_big *= (target / cur)
    else:
        h_big.fill(0.5)
    np.clip(h_big, 0.0, 1.0, out=h_big)
    return h_big


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function for the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_all = time.time()
    total_budget = float(budget_s)

    # ----- initial vector -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        # straightforward fallback: half 1's, half 0's
        n0 = 45
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)], dtype=float)
    n0 = h0.size
    target0 = n0 / 2.0
    cur_sum0 = h0.sum()
    if cur_sum0 != target0:
        h0 = h0 * (target0 / cur_sum0)
    np.clip(h0, 0.0, 1.0, out=h0)

    # best solution trackers
    best_h = h0.copy()
    best_score = _eval_c5(best_h)

    # ----- stage 1 : n = 45 ---------------------------------------------
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    t1 = min(remaining * 0.32, 360.0)            # about one third of the budget
    h1, s1 = _local_optimize(best_h, t1, rng)
    if s1 < best_score:
        best_h, best_score = h1, s1

    # ----- stage 2 : upscale to n = 90 ----------------------------------
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    if remaining > 100.0:
        h90 = _upscale(best_h, 2, rng)          # 45 → 90
        t2 = min(remaining * 0.33, 380.0)
        h2, s2 = _local_optimize(h90, t2, rng)
        if s2 < best_score:
            best_h, best_score = h2, s2

    # ----- stage 3 : upscale to n = 180 ---------------------------------
    elapsed = time.time() - start_all
    remaining = total_budget - elapsed
    if remaining > 80.0:
        h180 = _upscale(best_h, 2, rng)         # 90 → 180 (or 45 → 180 if previous skipped)
        t3 = remaining - 5.0                     # use whatever is left, keep a little margin
        h3, s3 = _local_optimize(h180, t3, rng)
        if s3 < best_score:
            best_h, best_score = h3, s3

    # ----- final tidy‑up --------------------------------------------------
    np.clip(best_h, 0.0, 1.0, out=best_h)
    target_final = best_h.size / 2.0
    cur = best_h.sum()
    if cur > 0:
        best_h *= (target_final / cur)
    else:
        best_h.fill(0.5)
    np.clip(best_h, 0.0, 1.0, out=best_h)

    final_score = _eval_c5(best_h)

    return (best_h, float(final_score), best_h.size)
