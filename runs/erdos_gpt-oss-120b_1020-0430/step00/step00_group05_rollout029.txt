def project_box_sum(v, S):
    # returns y = argmin ||y - v||^2 s.t. 0 <= y_i <= 1, sum y_i = S
    # This is isotonic projection; solve using water-filling.
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(v - mid, 0.0, 1.0)
        s = y.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def evaluate(h):
    r = np.correlate(h, 1 - h, mode='full')
    return r.max() * (2.0 / h.size)

def max_lag_and_corr(h):
    r = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(r)
    L = idx - (h.size - 1)
    max_val = r[idx]
    return L, max_val

def gradient_for_lag(h, L):
    n = h.shape[0]
    g = np.ones(n)
    if L > 0:
        g[:n-L] -= h[L:]
        g[L:] -= h[:n-L]
    elif L < 0:
        pos = -L
        g[pos:] -= h[:n-pos]
        g[:n-pos] -= h[pos:]
    else:  # L == 0
        g -= 2 * h
    return g

def project_box_sum(v, S):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(v - mid, 0.0, 1.0)
        s = y.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_descent_optimize(h0, max_iters, start_time, budget_s):
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = evaluate(h)
    for t in range(1, max_iters+1):
        if time.time() - start_time > budget_s - 1:
            break
        L, _ = max_lag_and_corr(h)
        g = gradient_for_lag(h, L)
        eta = 0.5 / math.sqrt(t)
        x = h - eta * g
        h = project_box_sum(x, target_sum)
        val = evaluate(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val

def simulated_annealing(n, max_iters, start_time, budget_s, initial_a=None):
    m = n // 2
    # Use symmetric representation
    def a_to_h(a):
        return np.concatenate([a, a[::-1]])
    if initial_a is not None:
        a = initial_a.copy()
    else:
        a = np.random.rand(m)
        a = project_box_sum(a, m / 2.0)
    h = a_to_h(a)
    curr_val = evaluate(h)
    best_val = curr_val
    best_a = a.copy()
    # SA schedule
    T0 = 0.02
    T_end = 1e-4
    for it in range(1, max_iters+1):
        if time.time() - start_time > budget_s - 1:
            break
        # Decrease temperature linearly
        T = T0 * (1 - it / max_iters) + T_end * (it / max_iters)
        i, j = np.random.randint(0, m, size=2)
        if i == j:
            continue
        # compute feasible delta
        max_decrease_i = a[i]  # can reduce to 0
        max_increase_j = 1.0 - a[j]  # can increase to 1
        delta_max = min(max_decrease_i, max_increase_j)
        max_decrease_j = a[j]
        max_increase_i = 1.0 - a[i]
        delta_min = -min(max_decrease_j, max_increase_i)
        if delta_max <= 0 and delta_min >= 0:
            continue
        delta = np.random.uniform(delta_min, delta_max)
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure bounds
        # optional: clamp epsilon due to numeric
        # compute new h and value
        h_new = a_to_h(a_new)
        new_val = evaluate(h_new)
        dE = new_val - curr_val
        if dE < 0 or np.random.rand() < math.exp(-dE / (T + 1e-12)):
            a = a_new
            h = h_new
            curr_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
    return a_to_h(best_a), best_val

import numpy as np
import math
import time

def _evaluate(h: np.ndarray) -> float:
    """C5 value for a given step function h."""
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(r.max() * (2.0 / h.size))


def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n with sum(v)=target (water‑filling)."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        y = np.clip(v - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of f_k(h)=sum h_i(1-h_{i+lag}) w.r.t. h."""
    n = h.size
    g = np.ones(n, dtype=float)
    if lag > 0:
        g[: n - lag] -= h[lag:]
        g[lag:] -= h[: n - lag]
    elif lag < 0:
        p = -lag
        g[p:] -= h[: n - p]
        g[: n - p] -= h[p:]
    else:  # lag == 0
        g -= 2.0 * h
    return g


def _subgradient_descent(h0: np.ndarray, max_iters: int, start_time: float, budget: float):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    target = n * 0.5
    best_h, best_val = h.copy(), _evaluate(h)
    for t in range(1, max_iters + 1):
        if time.time() - start_time > budget - 0.5:
            break
        # worst lag
        r = np.correlate(h, 1.0 - h, mode="full")
        idx = int(r.argmax())
        lag = idx - (n - 1)
        # sub‑gradient step
        g = _gradient_for_lag(h, lag)
        eta = 0.5 / math.sqrt(t)
        h = _project_box_sum(h - eta * g, target)
        val = _evaluate(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
    return best_h, best_val


def _simulated_annealing(n: int, iters: int, start_time: float, budget: float, a0: np.ndarray = None):
    """SA on the symmetric representation (a, a[::-1])."""
    m = n // 2
    def _to_h(a):
        return np.concatenate([a, a[::-1]])

    if a0 is None:
        a = np.random.rand(m)
    else:
        a = a0.copy()
    a = _project_box_sum(a, m * 0.5)

    h = _to_h(a)
    cur_val = _evaluate(h)
    best_a, best_val = a.copy(), cur_val

    T0, T1 = 0.02, 1e-4
    for it in range(1, iters + 1):
        if time.time() - start_time > budget - 0.5:
            break
        T = T0 * (1.0 - it / iters) + T1 * (it / iters)

        i, j = np.random.randint(0, m, size=2)
        if i == j:
            continue
        # feasible transfer delta (preserve sum)
        dec_i = a[i]                 # max we can take from i
        inc_j = 1.0 - a[j]           # max we can give to j
        delta_max = min(dec_i, inc_j)
        dec_j = a[j]
        inc_i = 1.0 - a[i]
        delta_min = -min(dec_j, inc_i)

        if delta_max <= 0 and delta_min >= 0:
            continue
        delta = np.random.uniform(delta_min, delta_max)

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        h_new = _to_h(a_new)
        new_val = _evaluate(h_new)
        dE = new_val - cur_val
        if dE < 0.0 or np.random.rand() < math.exp(-dE / (T + 1e-12)):
            a, h, cur_val = a_new, h_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a_new.copy()
    return _to_h(best_a), best_val


def run(seed: int = 42, budget_s: int = 1100, **_):
    """Search for a step function with a small C5 bound."""
    np.random.seed(seed)
    start = time.time()
    best_h, best_val, best_n = None, float("inf"), None

    # work with a modest even size (symmetry simplifies the search)
    n = 140
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    # ---- deterministic base candidates ------------------------------------
    cand_funcs = []

    # uniform 0.5
    cand_funcs.append(lambda: np.full(n, 0.5))

    # linear descending (integral exactly 1)
    cand_funcs.append(lambda: 1.0 - (np.arange(n) + 0.5) / n)

    # symmetric triangular (peak =1 at centre)
    cand_funcs.append(lambda: np.maximum(0.0, 1.0 - np.abs(xs - 1.0)))

    # parabolic shape scaled to area 1
    def parabola():
        raw = 1.0 - (xs - 1.0) ** 2
        raw = np.clip(raw, 0.0, None)
        scale = (n * 0.5) / raw.sum()
        return np.clip(raw * scale, 0.0, 1.0)

    cand_funcs.append(parabola)

    # try an initial construction if available
    try:
        init = np.array(initial_h_values, dtype=float)
        if init.size == n:
            cand_funcs.append(lambda: init.copy())
    except NameError:
        pass

    for fn in cand_funcs:
        h = _project_box_sum(fn(), n * 0.5)
        val = _evaluate(h)
        if val < best_val:
            best_h, best_val, best_n = h.copy(), val, n

    # ---- stochastic search (symmetric representation) --------------------
    # one longer SA run
    h_sa, val_sa = _simulated_annealing(n, 120000, start, budget_s)
    if val_sa < best_val:
        best_h, best_val, best_n = h_sa.copy(), val_sa, n

    # fine‑tune the current best with sub‑gradient descent
    if best_h is not None:
        h_opt, val_opt = _subgradient_descent(best_h, 6000, start, budget_s)
        if val_opt < best_val:
            best_h, best_val, best_n = h_opt.copy(), val_opt, n

    # ---- additional short SA bursts while time permits --------------------
    while time.time() - start < budget_s - 2.0:
        # start from a random symmetric vector
        h_candidate, val_candidate = _simulated_annealing(n, 20000, start, budget_s)
        if val_candidate < best_val:
            best_h, best_val = h_candidate.copy(), val_candidate
            # a quick gradient polish
            h_opt, val_opt = _subgradient_descent(best_h, 2000, start, budget_s)
            if val_opt < best_val:
                best_h, best_val = h_opt.copy(), val_opt

    # final safety projection and evaluation
    if best_h is None:
        best_h = np.full(n, 0.5)
    best_h = _project_box_sum(best_h, n * 0.5)
    best_val = _evaluate(best_h)

    return best_h.tolist(), float(best_val), int(n)
