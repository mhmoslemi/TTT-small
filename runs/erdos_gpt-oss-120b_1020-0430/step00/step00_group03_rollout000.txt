    # Return best solution as list
    return best.tolist(), float(best_score), N

import time, math, random
import numpy as np

def _evaluate_score(h):
    """Return C5 bound for a step function represented by samples h."""
    N = h.shape[0]
    dx = 2.0 / N
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (sampled uniformly) that
    minimizes the Erdős overlap constant C5.

    Returns
    -------
    h_vals : list of float
        Sample values of the best-found function.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of samples (length of h_vals).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise a feasible starting point
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        N = init.shape[0]
        # Make a binary copy (the known good constructions are binary)
        h = (init > 0.5).astype(float)
        target_ones = N // 2
        # Repair to exact half‑ones if needed
        cur = int(h.sum())
        if cur != target_ones:
            ones = np.flatnonzero(h == 1.0)
            zeros = np.flatnonzero(h == 0.0)
            while cur > target_ones:
                i = np.random.choice(ones)
                h[i] = 0.0
                cur -= 1
                ones = np.flatnonzero(h == 1.0)
                zeros = np.flatnonzero(h == 0.0)
            while cur < target_ones:
                i = np.random.choice(zeros)
                h[i] = 1.0
                cur += 1
                ones = np.flatnonzero(h == 1.0)
                zeros = np.flatnonzero(h == 0.0)
    else:
        # No parent: start from scratch with even length
        N = 100                     # modest size, fast to evaluate
        target_ones = N // 2
        h = np.zeros(N, dtype=float)
        h[np.random.choice(N, target_ones, replace=False)] = 1.0

    # ------------------------------------------------------------------
    # Core simulated‑annealing loop (binary swaps)
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _evaluate_score(best_h)
    cur_h = h.copy()
    cur_score = best_score

    # Early exit if the initial solution already meets the target
    TARGET = 0.3808
    start_time = time.time()
    time_limit = budget_s * 0.95

    # Temperature schedule parameters
    T0 = max(best_score * 0.5, 1e-6)

    it = 0
    stagnation_counter = 0
    while time.time() - start_time < time_limit:
        it += 1

        # ----- propose a swap of a 1 and a 0 -------------------------
        ones_idx = np.flatnonzero(cur_h == 1.0)
        zeros_idx = np.flatnonzero(cur_h == 0.0)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i_one = np.random.choice(ones_idx)
        i_zero = np.random.choice(zeros_idx)

        cand = cur_h.copy()
        cand[i_one] = 0.0
        cand[i_zero] = 1.0
        cand_score = _evaluate_score(cand)

        # ----- acceptance test ---------------------------------------
        if cand_score < cur_score:
            accept = True
        else:
            elapsed = time.time() - start_time
            frac = elapsed / time_limit
            T = T0 * (1.0 - frac)
            if T <= 0:
                T = 1e-12
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / T)

        if accept:
            cur_h = cand
            cur_score = cand_score
            stagnation_counter = 0
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                if best_score <= TARGET:
                    break
        else:
            stagnation_counter += 1

        # ----- occasional diversification -----------------------------
        if it % 200000 == 0:
            # if we have not improved for a while, shake the current state
            if stagnation_counter > 50000:
                # random 5‑swap perturbation
                for _ in range(5):
                    ones_idx = np.flatnonzero(cur_h == 1.0)
                    zeros_idx = np.flatnonzero(cur_h == 0.0)
                    if ones_idx.size == 0 or zeros_idx.size == 0:
                        break
                    i1 = np.random.choice(ones_idx)
                    i0 = np.random.choice(zeros_idx)
                    cur_h[i1] = 0.0
                    cur_h[i0] = 1.0
                cur_score = _evaluate_score(cur_h)
                stagnation_counter = 0
            else:
                # occasional jump back to the best found so far
                cur_h = best_h.copy()
                cur_score = best_score

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return best_h.tolist(), float(best_score), N
