def compute_c5(h):
    n = len(h)
    hx = h.astype(np.float64)
    yh = 1 - hx
    # size for FFT: at least 2*n
    m = 1 << (2*n - 1).bit_length()
    # Actually we need convolution length 2*n-1, so pad both to size m
    X = np.fft.rfft(np.concatenate([hx, np.zeros(m - n)]))
    Y = np.fft.rfft(np.concatenate([yh, np.zeros(m - n)]))
    conv = np.fft.irfft(X * Y, n=m)
    # conv length m, but only first 2*n-1 entries correspond to full conv:
    conv = conv[:2*n-1] * (2.0 / n) # dx = 2/n
    return conv.max()

This yields C5.

def gradient_for_shift(h, k):
    n = len(h)
    # shift_plus = h[i+k] if within bounds
    idx = np.arange(n) + k
    shift_plus = np.zeros(n, dtype=np.float64)
    valid = (idx >=0) & (idx < n)
    shift_plus[valid] = h[idx[valid]]
    idx2 = np.arange(n) - k
    shift_minus = np.zeros(n, dtype=np.float64)
    valid2 = (idx2 >=0) & (idx2 < n)
    shift_minus[valid2] = h[idx2[valid2]]
    grad = 1.0 - (shift_plus + shift_minus)
    return grad

import time, math, random
import numpy as np

def _project_onto_box_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while enforcing sum(v)=target_sum.
    Uses a simple bisection on the global shift λ such that sum(clip(v+λ,0,1))=target."""
    low = -np.max(v)          # enough to make all ≤0 after shift
    high = 1.0 - np.min(v)    # enough to make all ≥1 after shift
    for _ in range(50):
        mid = (low + high) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            low = mid
        else:
            high = mid
    w = np.clip(v + (low + high) * 0.5, 0.0, 1.0)
    return w

def _c5_and_corr(h: np.ndarray):
    """Return (C5, full correlation array) for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max()), corr

def _grad_for_shift(h: np.ndarray, k: int):
    """Gradient of the overlap for a fixed shift k."""
    n = h.shape[0]
    idx = np.arange(n) + k
    pos = (idx >= 0) & (idx < n)
    shift_plus = np.zeros_like(h)
    shift_plus[pos] = h[idx[pos]]
    idx2 = np.arange(n) - k
    pos2 = (idx2 >= 0) & (idx2 < n)
    shift_minus = np.zeros_like(h)
    shift_minus[pos2] = h[idx2[pos2]]
    return 1.0 - (shift_plus + shift_minus)

def _random_feasible(n: int, target: float):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    v = np.random.rand(n)
    return _project_onto_box_and_sum(v, target)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise problem size and starting vector
    # ------------------------------------------------------------------
    init = globals().get('initial_h_values', None)
    if init is not None:
        h0 = np.asarray(init, dtype=np.float64)
        n = h0.size
    else:
        n = 200                     # a reasonable default size
        h0 = _random_feasible(n, n / 2.0)

    target_sum = n / 2.0
    h0 = _project_onto_box_and_sum(h0, target_sum)

    # ------------------------------------------------------------------
    # evaluate starting point
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_c5, _ = _c5_and_corr(best_h)
    cur_h = best_h.copy()
    cur_c5, _ = _c5_and_corr(cur_h)

    start_time = time.time()
    no_improve = 0
    eta0 = 0.5                     # initial step size

    while time.time() - start_time < budget_s - 0.4:
        # recompute correlation and the shift attaining the current maximum
        cur_c5, corr = _c5_and_corr(cur_h)
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = cur_h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # pick a shift among those attaining the maximal value (break ties randomly)
        max_val = cur_c5
        ties = np.where(np.abs(corr - max_val) <= 1e-12)[0]
        k = int(ties[np.random.randint(ties.size)] - (n - 1))

        # gradient for this shift
        grad = _grad_for_shift(cur_h, k)

        # line search – backtrack until improvement (or stop after a few reductions)
        eta = eta0
        improved = False
        for _ in range(12):

            cand_c5, _ = _c5_and_corr(cand)
