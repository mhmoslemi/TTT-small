import time, math, random
import numpy as np

def _cmax(h, n):
    """Maximum overlap for vector h (length n)."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)


def _grad_shift(h, n, L):
    """Gradient of the overlap at shift L (integer, can be negative)."""
    grad = np.zeros(n, dtype=float)
    if L >= 0:
        ov = n - L
        if ov > 0:
            grad[:ov] = 1.0
            grad[:ov] -= h[L:]          # -h[i+L]
            grad[L:] -= h[:ov]          # -h[i-L] (i-L out of range -> 0)
    else:
        L0 = -L
        ov = n - L0
        if ov > 0:
            grad[L0:] = 1.0
            grad[L0:] -= h[:ov]         # -h[i+L] (i+L = i-L0)
            grad[:ov] -= h[L0:]         # -h[i-L] (i-L = i+L0)
    return (2.0 / n) * grad


def _repair(vec, target):
    """Clip to [0,1] and enforce exact sum = target (assumes feasibility)."""
    v = np.clip(vec, 0.0, 1.0)
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute the tiny difference to a non‑boundary element if possible
        for idx in range(len(v)):
            if 0.0 < v[idx] < 1.0:
                v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
                break
        else:
            # fallback – adjust the last entry directly
            v[-1] = np.clip(v[-1] + diff, 0.0, 1.0)
    return v


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős C5 overlap constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # ----- initialise -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        # no parent solution – start from a simple half‑filled pattern
        n = 70
        h0 = np.full(n, 0.5, dtype=float)
    else:
        n = h0.shape[0]

    target_sum = n / 2.0
    h = _repair(h0, target_sum)

    cur_score = _cmax(h, n)
    best_h = h.copy()
    best_score = cur_score

    # ----- annealing parameters ---------------------------------------
    start = time.time()
    stop_time = start + budget_s - 0.5          # small safety margin
    T0 = 0.02
    T1 = 1e-4

    # ----- main optimisation loop ------------------------------------
    while time.time() < stop_time:
        # temperature (exponential schedule)
        elapsed = time.time() - start
        prog = elapsed / (budget_s - 0.5)
        T = T0 * (T1 / T0) ** prog

        # current worst shift and its gradient
        full_corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(np.argmax(full_corr))
        L = idx_max - (n - 1)                     # shift value (can be negative)

        grad = _grad_shift(h, n, L)

        # candidate sets based on sign of gradient
        pos_idx = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg_idx = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]

        if pos_idx.size and neg_idx.size:
            # strongest improving pair
            i = pos_idx[np.argmax(grad[pos_idx])]
            j = neg_idx[np.argmin(grad[neg_idx])]
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0:
                # fallback to a random feasible pair
                i = rng.choice(pos_idx)
                j = rng.choice(neg_idx)
                max_transfer = min(h[i], 1.0 - h[j])
            # move a moderate fraction of the available mass
            delta = max_transfer * (0.2 + 0.3 * rng.random())
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # pure random transfer when gradient gives nothing useful
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            # choose direction that respects bounds
            if h[i] > 0 and h[j] < 1:
                src, dst = i, j
            elif h[j] > 0 and h[i] < 1:
                src, dst = j, i
            else:
                continue
            max_transfer = min(h[src], 1.0 - h[dst])
            if max_transfer <= 0:
                continue
            delta = max_transfer * rng.random()
            # occasionally make larger jumps
            if rng.random() < 0.05:
                delta = max_transfer * (0.5 + 0.5 * rng.random())
            new_h = h.copy()
            new_h[src] -= delta
            new_h[dst] += delta

        # evaluate candidate
        new_score = _cmax(new_h, n)
        delta_f = new_score - cur_score

        # Metropolis acceptance
        if delta_f < 0 or rng.random() < math.exp(-delta_f / max(T, 1e-12)):
            h = new_h
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_h = new_h.copy()
                best_score = new_score

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.001:
            noise = rng.normal(scale=0.05, size=n)
            trial = h + noise
            trial = _repair(trial, target_sum)
            trial_score = _cmax(trial, n)
            if trial_score < cur_score or rng.random() < math.exp(-(trial_score - cur_score) / max(T, 1e-12)):
                h = trial
                cur_score = trial_score
                if trial_score < best_score - 1e-12:
                    best_h = trial.copy()
                    best_score = trial_score

    # final safety repair and evaluation
    best_h = _repair(best_h, target_sum)
    final_score = _cmax(best_h, n)

    return best_h.tolist(), float(final_score), n
