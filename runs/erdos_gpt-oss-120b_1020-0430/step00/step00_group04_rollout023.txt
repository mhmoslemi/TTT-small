import time, math, numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small margin for safety
    deadline = start_time + budget_s * 0.95

    # ---------- helpers ----------
    def project_to_sum(v, target):
        """Project v onto the box [0,1] with sum == target (L2 projection)."""
        v = np.clip(v, 0.0, 1.0)
        for _ in range(200):
            s = v.sum()
            diff = target - s
            if abs(diff) < 1e-12:
                break
            free = (v > 0.0) & (v < 1.0)
            nfree = int(free.sum())
            if nfree == 0:
                break
            delta = diff / nfree
            v[free] += delta
            v = np.clip(v, 0.0, 1.0)
        return v

    def evaluate_vec(h):
        """C5 value for a given vector h."""
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def random_feasible():
        """Generate a random feasible vector (bounds + sum)."""
        v = rng.random(n)  # uniform in [0,1)
        return project_to_sum(v, target_sum)

    # ---------- configuration ----------
    # prefer the same length as the provided parent if possible
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        n0 = base.size
    else:
        base = None
        n0 = 0

    # choose a modest size (enough degrees of freedom, cheap evaluation)
    n = max(42, n0)          # at least 42, keep it small
    target_sum = n / 2.0
    dx = 2.0 / n

    # build a starting vector
    if base is not None:
        if n0 != n:
            # linear interpolation to the new size (no SciPy needed)
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False) + (1.0 / n0) * 0.5
            x  = np.linspace(0.0, 2.0, n,  endpoint=False) + (1.0 / n)  * 0.5
            interp = np.interp(x, x0, base, left=0.0, right=0.0)
            h_start = project_to_sum(interp, target_sum)
        else:
            h_start = project_to_sum(base.copy(), target_sum)
    else:
        # triangular shape (area = 1)
        xs = (np.arange(n) + 0.5) * dx
        tri = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
        h_start = project_to_sum(tri, target_sum)

    best_h = h_start.copy()
    best_score = evaluate_vec(best_h)

    # ---------- local search ----------
    # parameters for the stochastic search
    max_no_improve = 4000          # restart after this many failures
    step_start = 0.2               # initial transfer magnitude
    step_min = 1e-3                # never go below this
    step_decay = 0.95              # decay every 2000 steps

    # main time‑bounded loop
    while time.time() < deadline:
        # start from a random feasible point (occasionally from best)
        if rng.random() < 0.2:
            cur = best_h.copy()
        else:
            cur = random_feasible()
        cur_score = evaluate_vec(cur)

        step = step_start
        no_improve = 0
        iter_cnt = 0

        # inner hill‑climbing loop
        while time.time() < deadline and no_improve < max_no_improve:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

            # feasible delta interval respecting bounds
            lo = max(-step, -cur[i], cur[j] - 1.0)
            hi = min(step, 1.0 - cur[i], cur[j])
            if lo >= hi:
                continue

            delta = rng.uniform(lo, hi)
            new = cur.copy()
            new[i] += delta
            new[j] -= delta

            new_score = evaluate_vec(new)

            if new_score < cur_score - 1e-12:
                cur, cur_score = new, new_score
                no_improve = 0
                if new_score < best_score - 1e-12:
                    best_h, best_score = new.copy(), new_score
            else:
                no_improve += 1

            iter_cnt += 1
            # slowly shrink the step size to fine‑tune
            if iter_cnt % 2000 == 0:
                step = max(step_min, step * step_decay)

    # final safeguard: enforce exact sum (tiny correction)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        # find a component with slack in the needed direction
        if diff > 0:
            # need to increase something; pick the one farthest from 1
            idx = np.argmax(1.0 - best_h)
            best_h[idx] = min(1.0, best_h[idx] + diff)
        else:
            # need to decrease; pick the largest entry
            idx = np.argmax(best_h)
            best_h[idx] = max(0.0, best_h[idx] + diff)

    # return as a plain list for the harness
    return best_h.tolist(), float(best_score), n
