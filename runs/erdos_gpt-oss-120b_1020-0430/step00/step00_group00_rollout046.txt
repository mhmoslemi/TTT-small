def random_feasible(n):
    # sample from Dirichlet with equal parameters
    a = np.random.rand(n)
    # Normalize to sum = target
    target = n/2
    a = a / a.sum() * target
    # but may exceed 1, we need to ensure <=1.
    # If any >1, we can cap them at 1 and redistribute excess among remaining.
    # iterative:
    while True:
        over = a > 1
        if not over.any():
            break
        excess = (a[over] - 1).sum()
        a[over] = 1
        # Redistribute excess to underfull entries proportionally
        under = ~over
        if not under.any():
            # impossible? just set a to target equally?
            a = np.full(n, target / n)
            break
        a[under] += excess / under.sum()
    return a

import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    # Dirichlet‑like sampling
    v = np.random.rand(n)
    v = v / v.sum() * target
    # enforce the upper bound 1 via water‑filling
    for _ in range(20):
        over = v > 1.0
        if not np.any(over):
            break
        excess = (v[over] - 1.0).sum()
        v[over] = 1.0
        under = v < 1.0
        if not np.any(under):
            break
        v[under] += excess / under.sum()
    v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    s = v.sum()
    if abs(s - target) > 1e-8:
        diff = target - s
        mask = (v > 0.0) & (v < 1.0)
        cnt = mask.sum()
        if cnt > 0:
            v[mask] += diff / cnt
        else:
            v = np.full(n, target / n)
        v = np.clip(v, 0.0, 1.0)
    return v


def _project_feasible(x: np.ndarray) -> np.ndarray:
    """Project x onto the feasible set: 0≤x≤1 and sum(x)=len(x)/2."""
    n = x.size
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    for _ in range(15):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add diff to entries with room
            cap = 1.0 - x
            mask = cap > 1e-12
            if not np.any(mask):
                break
            total_cap = cap[mask].sum()
            if total_cap <= diff + 1e-12:
                x[mask] = 1.0
                continue
            inc = diff / mask.sum()
            x[mask] += inc
            x = np.clip(x, 0.0, 1.0)

def _local_search(initial_h: np.ndarray, time_limit: float) -> (np.ndarray, float):
    """Simulated‑annealing style hill climbing for a fixed amount of time."""
    start = time.time()
    n = initial_h.size
    cur_h = _project_feasible(initial_h.astype(float, copy=True))
    cur_val = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    it = 0
    T0 = 0.02  # initial temperature scale
    while time.time() - start < time_limit:
        it += 1
        # linear temperature decay
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / time_limit) + 1e-12

        # propose a move that preserves the sum exactly
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)
        # feasible delta range keeping both entries in [0,1]
        dmin = max(-cur_h[i], cur_h[j] - 1.0)
        dmax = min(1.0 - cur_h[i], cur_h[j])
        if dmax <= dmin:
            continue
        delta = np.random.uniform(dmin, dmax)
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = _compute_c5(new_h)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
    # final projection (safety)
    best_h = _project_feasible(best_h)
    best_val = _compute_c5(best_h)
    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start_total = time.time()
    total_budget = budget_s * 0.95  # keep a safety margin

    # ------------------------------------------------------------------
    # Determine initial resolution and vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init_vec = np.asarray(initial_h_values, dtype=float).copy()
        n0 = init_vec.size
    else:
        n0 = 45
        init_vec = _random_feasible(n0)

    # Ensure feasibility of the starting vector
    init_vec = _project_feasible(init_vec)

    # ------------------------------------------------------------------
    # Stage 1: optimise at the original resolution
    # ------------------------------------------------------------------
    stage1_time = total_budget * 0.70
    best_h, best_val = _local_search(init_vec, stage1_time)

    # ------------------------------------------------------------------
    # Stage 2: try a higher resolution if time permits
    # ------------------------------------------------------------------
    elapsed = time.time() - start_total
    remaining = total_budget - elapsed
    if remaining > 15.0:  # need a decent chunk to be worthwhile
        # upsample to a finer grid (choose 60 points, at least double if small)
        n_fine = max(60, n0 * 2)
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_fine, endpoint=False)
        fine_vec = np.interp(x_new, x_old, best_h)
        fine_vec = _project_feasible(fine_vec)
        fine_h, fine_val = _local_search(fine_vec, remaining)
        if fine_val < best_val:
            best_h, best_val = fine_h, fine_val

    # Final safety projection and recompute exact bound
    best_h = _project_feasible(best_h)
    best_val = _compute_c5(best_h)

    return best_h, float(best_val), int(best_h.size)
