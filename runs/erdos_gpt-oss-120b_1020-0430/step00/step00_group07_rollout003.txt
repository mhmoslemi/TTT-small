def propose_random_pair_move(h, max_step):
    n = h.shape[0]
    # choose donor i where h[i] > 0
    donors = np.where(h > 1e-12)[0]
    if donors.size == 0:
        return None
    i = np.random.choice(donors)
    # choose receiver j where h[j] < 1
    receivers = np.where(h < 1 - 1e-12)[0]
    if receivers.size == 0:
        return None
    j = np.random.choice(receivers)
    # ensure i != j
    if i == j:
        # try again but limited attempts
        if donors.size > 1:
            # choose another donor
            i = np.random.choice(donors[donors != j])
        else:
            return None
    delta_max = min(h[i], 1.0 - h[j])
    if delta_max <= 0:
        return None
    # decide delta: sample uniform up to delta_max, also cap by max_step

    max_allowed = min(delta_max, max_step)
    if max_allowed <= 0:

    delta = np.random.rand() * max_allowed

    if delta < 1e-12:
        delta = max_allowed * 0.5
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta

import numpy as np
import time
import math
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _initial_h():
    """Return the provided initial construction if it exists, else None."""
    try:
        return initial_h_values  # type: ignore[name-defined]
    except NameError:
        return None


def compute_c5(h: np.ndarray) -> float:
    """Evaluate the Erdős C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx


def propose_random_pair_move(h: np.ndarray, max_step: float) -> np.ndarray | None:
    """
    Produce a new h by moving a random amount of mass from a donor index
    (where h>0) to a receiver index (where h<1). The total sum is preserved.
    """
    n = h.shape[0]
    donors = np.where(h > 1e-12)[0]
    receivers = np.where(h < 1.0 - 1e-12)[0]
    if donors.size == 0 or receivers.size == 0:
        return None
    i = np.random.choice(donors)
    j = np.random.choice(receivers)
    # ensure i != j
    if i == j:
        # rare, pick another
        if donors.size > 1:
            i = np.random.choice(donors[donors != j])
        else:
            return None
    delta_max = min(h[i], 1.0 - h[j])
    if delta_max <= 0.0:
        return None
    allowed = min(delta_max, max_step)
    if allowed <= 0.0:
        return None
    delta = np.random.rand() * allowed
    if delta < 1e-12:   # avoid zero move
        delta = allowed * 0.5
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def propose_greedy_move(h: np.ndarray, max_step: float = 0.5) -> np.ndarray | None:
    """
    Focus on the current worst shift (the one attaining max C5) and move
    mass from a site contributing heavily to that shift towards a site
    with little or no contribution.
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)          # integer lag producing the max

    # Build the weight vector w_i = (1 - h_{i+shift}) where defined,
    # zero elsewhere.
    w = np.empty(n, dtype=h.dtype)
    if shift >= 0:
        limit = n - shift
        w[:limit] = 1.0 - h[shift:]
        w[limit:] = 0.0
    else:
        off = -shift
        w[off:] = 1.0 - h[: n - off]
        w[:off] = 0.0

    contrib = h * w                     # contribution to the worst shift
    if np.all(contrib == 0):
        # Nothing to improve via this heuristic; fall back.
        return propose_random_pair_move(h, max_step)

    i = int(np.random.choice(np.flatnonzero(contrib == contrib.max())))
    # Receiver: minimal weight among positions that can still receive mass
    mask = h < 1.0 - 1e-12
    if not np.any(mask):
        return None
    w_masked = np.where(mask, w, np.inf)
    j = int(np.argmin(w_masked))

    if i == j or h[i] <= 1e-12:
        # fallback to a random move if degenerate
        return propose_random_pair_move(h, max_step)

    delta_max = min(h[i], 1.0 - h[j])
    if delta_max <= 0.0:
        return None
    delta = delta_max * np.random.rand()
    if delta < 1e-12:
        delta = delta_max * 0.5
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def _project_to_simplex(h: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Simple repair: clip to [0,1] then adjust the total sum by scaling the
    interior (0,1) entries. The routine is deterministic and cheap for the
    small dimensions we use.
    """
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    # Identify free variables (not on a bound)
    free = (h > 0.0) & (h < 1.0)
    if free.any():
        scale = (target_sum - h[~free].sum()) / h[free].sum()
        h[free] *= scale
        h = np.clip(h, 0.0, 1.0)          # final safety clip
    else:
        # All at bounds – fallback: distribute the difference uniformly
        diff = target_sum - cur
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h


def _refine_slsqp(h: np.ndarray, time_left: float) -> np.ndarray:
    """
    One quick SLSQP refinement, respecting the remaining time budget.
    """
    n = h.size
    target = n / 2.0
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    # use a very modest iteration limit; we only call if time permits
    opts = {"maxiter": 250, "ftol": 1e-12, "disp": False}
    try:
        res = minimize(compute_c5, h, method="SLSQP",
                       bounds=bounds, constraints=cons, options=opts)
        if res.success:
            return _project_to_simplex(res.x, target)
        return h
    except Exception:
        return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h with a low Erdős C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values – numpy array of length n_points with entries in [0,1]
        c5_bound – float, the achieved upper bound
        n_points – integer, number of discretization points
    """
    np.random.seed(seed)
    start_total = time.time()
    # --------------------------------------------------------
    # 1. Initialise problem size and a feasible start vector
    # --------------------------------------------------------
    init = _initial_h()
    if init is not None:
        h = np.array(init, dtype=np.float64).copy()
        n = h.shape[0]
    else:
        n = 80
        h = np.zeros(n, dtype=np.float64)
        h[: n // 2] = 1.0
        np.random.shuffle(h)

    target_sum = n / 2.0
    h = _project_to_simplex(h, target_sum)

    # --------------------------------------------------------
    # 2. Local random/greedy search
    # --------------------------------------------------------
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    max_step = 0.5
    it_without_improve = 0
    iteration = 0
    # Simple schedule parameters
    max_iters_est = 500_000

    while time.time() - start_total < budget_s * 0.93:
        iteration += 1

        # Choose move type: greedy every 5th iteration
        if iteration % 5 == 0:
            cand = propose_greedy_move(cur_h, max_step)
            if cand is None:
                cand = propose_random_pair_move(cur_h, max_step)
        else:
            cand = propose_random_pair_move(cur_h, max_step)

        if cand is None:
            continue

        new_c5 = compute_c5(cand)

        # Accept if better (deterministic hill climbing)
        if new_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = cand, new_c5
            it_without_improve = 0
            if new_c5 + 1e-12 < best_c5:
                best_h, best_c5 = cand.copy(), new_c5
                # Early exit if we already beat the target bound
                if best_c5 <= 0.3808:
                    break
        else:
            # Occasional probabilistic acceptance (simulated annealing)
            T0 = 0.02
            T = T0 * max(0.0, 1.0 - iteration / max_iters_est)
            if T > 0.0 and np.random.rand() < math.exp(-(new_c5 - cur_c5) / T):
                cur_h, cur_c5 = cand, new_c5
                it_without_improve = 0
            else:
                it_without_improve += 1

        # Gradually shrink the maximum allowed transfer size
        if iteration % 10000 == 0:
            max_step = max(0.01, max_step * 0.9)

        # Restart if we get stuck for many iterations
        if it_without_improve > 200_000:
            # generate a fresh binary start
            cur_h = np.zeros(n, dtype=np.float64)
            cur_h[: n // 2] = 1.0
            np.random.shuffle(cur_h)
            cur_h = _project_to_simplex(cur_h, target_sum)
            cur_c5 = compute_c5(cur_h)
            it_without_improve = 0

    # --------------------------------------------------------
    # 3. Optional SLSQP refinement if time permits
    # --------------------------------------------------------
    time_left = budget_s - (time.time() - start_total)
    if time_left > 5.0:          # enough slack for a short optimisation

        refined_c5 = compute_c5(refined)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined, refined_c5

    # Ensure constraints are still met before returning
    best_h = _project_to_simplex(best_h, target_sum)

    return best_h, best_c5, n
