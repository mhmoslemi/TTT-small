def simulated_annealing(h, max_time, display=False):

    N = len(h)
    target_sum = N/2
    best_h = h.copy()
    best_val = evaluate(h)
    curr_h = h.copy()
    curr_val = best_val
    start = time.time()
    # parameters
    T0 = 0.02
    # schedule: T = T0 * (1 - progress)
    # step size: delta_max = 0.05 maybe gradually decreasing

    iteration = 0
    while time.time() - start < max_time:
        # progress fraction
        elapsed = time.time() - start
        fraction = elapsed / max_time
        T = T0 * (1 - fraction) + 1e-6
        delta_max = 0.04 * (1 - fraction) + 0.001  # smaller as time passes

        # propose move
        i, j = np.random.randint(N, size=2)
        if i == j:
            continue
        delta = np.random.uniform(-delta_max, delta_max)
        # adjust limited by bounds
        hi_i = 1 - curr_h[i]
        lo_i = -curr_h[i]
        hi_j = curr_h[j]  # we subtract delta from j, ensure >=0
        lo_j = -(1 - curr_h[j])
        # combined: need i new = curr_h[i] + delta (must be between0 &1)
        # and j new = curr_h[j] - delta (must be between0 &1)
        # So delta must satisfy: -curr_h[i] <= delta <= 1 - curr_h[i] AND
        #   -(1 - curr_h[j]) <= delta <= curr_h[j]
        delta_low = max(-curr_h[i], -(1 - curr_h[j]))
        delta_high = min(1 - curr_h[i], curr_h[j])
        # Clip delta
        if delta < delta_low or delta > delta_high:
            # re-sample delta within feasible
            delta = np.random.uniform(delta_low, delta_high)
        # Now apply
        new_h = curr_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # Evaluate
        new_val = evaluate(new_h)
        # Accept if better or probabilistically
        if new_val < curr_val or np.random.rand() < np.exp((curr_val - new_val) / T):
            curr_h = new_h
            curr_val = new_val
            if new_val < best_val:
                best_h = new_h
                best_val = new_val
        iteration += 1

    return best_h, best_val

import numpy as np
import time
import math

def evaluate_c5(h):
    """Maximum overlap integral for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def project_sum_bounds(v, target_sum):
    """
    Project vector v onto the cube [0,1]^n while enforcing sum(v)=target_sum.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - lam, 0.0, 1.0)
    # tiny correction in case of numerical error
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        x += diff / x.size
        x = np.clip(x, 0.0, 1.0)
    return x


def simulated_annealing(h0, max_seconds):
    """
    Simple SA that preserves the integral constraint by moving mass between two points.
    """
    n = h0.shape[0]
    best_h = h0.copy()
    best_val = evaluate_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    start = time.time()
    T0 = 0.02          # initial temperature
    max_delta0 = 0.1   # maximum delta at start

    while True:
        if time.time() - start >= max_seconds:
            break

        elapsed = time.time() - start
        frac = elapsed / max_seconds
        T = T0 * (1 - frac) + 1e-8
        max_delta = max_delta0 * (1 - frac) + 0.001

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # feasible delta so that both entries stay in [0,1]
        delta_low = max(-cur_h[i], -(1 - cur_h[j]))
        delta_high = min(1 - cur_h[i], cur_h[j])
        if delta_low >= delta_high:
            continue

        delta = np.random.uniform(delta_low, delta_high)
        # enforce magnitude limit
        if delta > max_delta:
            delta = max_delta
        elif delta < -max_delta:
            delta = -max_delta

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_val = evaluate_c5(new_h)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < math.exp((cur_val - new_val) / T):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # --------------------------------------------------------------------
    # initialise from the supplied parent if present
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(45, 0.5, dtype=float)

    n0 = h_init.shape[0]
    h_init = project_sum_bounds(h_init, n0 / 2.0)

    best_h = h_init.copy()
    best_val = evaluate_c5(best_h)
    best_n = n0

    start_total = time.time()

    # progressively increase resolution
    size_list = [n0, 90, 150, 240]  # feel free to extend if time permits
    for idx, N in enumerate(size_list):
        # time bookkeeping
        elapsed = time.time() - start_total
        time_left = budget_s - elapsed - 5.0   # small safety margin
        if time_left <= 0:
            break
        stages_rem = len(size_list) - idx
        # allocate roughly equal share of remaining time
        stage_time = max(0.5, (time_left / stages_rem) * 0.9)

        # interpolate current best to the new grid if needed
        if N != best_n:
            old_x = np.linspace(0.0, 2.0, best_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, N, endpoint=False)
            interp = np.interp(new_x, old_x, best_h)
            h_start = project_sum_bounds(interp, N / 2.0)

            h_start = best_h.copy()

        # SA optimisation at this resolution
        h_opt, val_opt = simulated_annealing(h_start, stage_time)

        # final projection to guarantee feasibility
        h_opt = project_sum_bounds(h_opt, N / 2.0)
        val_opt = evaluate_c5(h_opt)

        if val_opt < best_val:
            best_h, best_val, best_n = h_opt.copy(), val_opt, N

    # final tidy‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_bounds(best_h, best_n / 2.0)
    final_c5 = evaluate_c5(best_h)

    return best_h, final_c5, int(best_n)
