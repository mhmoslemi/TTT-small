def max_overlap_fft(h):
    n = h.shape[0]
    m = 2*n - 1
    N = 1 << (m-1).bit_length()
    # compute FFTs
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft((1 - h)[::-1], n=N)
    conv = np.fft.irfft(H * G, n=N)[:m]
    return np.max(conv) * (2.0 / n)

    high = 1 - np.min(v) + 2.0

import time
import random
import math
import numpy as np

def _max_overlap(h):
    """Return (max_overlap, full_correlation) for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.convolve(h, (1.0 - h)[::-1], mode="full")
    return float(conv.max() * dx), conv


def _project_box_sum(v, target):
    """
    Project vector v onto the box [0,1]^n while forcing the sum to equal target.
    Solves for λ in w = clip(v + λ,0,1) with sum(w)=target via binary search.
    """
    lo = -np.max(v) - 2.0   # ensures v+lo <= 0
    hi = 1.0 - np.min(v) + 2.0  # ensures v+hi >= 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 1e-12) & (w < 1.0 - 1e-12)
        if free.any():
            idx = np.where(free)[0]
            w[idx] += diff / len(idx)
            w = np.clip(w, 0.0, 1.0)
        else:
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretised) to minimise the
    overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : list of floats in [0,1] (length n_points)
        c5_bound : float, the achieved max overlap
        n_points : int, the number of samples used
    """
    random.seed(seed)
    np.random.seed(seed)

    # Choose a working resolution (even number for exact sum feasibility)
    n = 200  # 200 samples give dx = 0.01
    target_sum = n / 2.0

    # Initialise h: start from the supplied construction if it exists,
    # otherwise use a random balanced binary vector.
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # If the supplied vector has a different length, resize by interpolation.
        if h0.size != n:
            # simple linear interpolation to the new grid
            xs = np.linspace(0, 2, h0.size, endpoint=False)
            xs_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(xs_new, xs, h0)
        h = _project_box_sum(h0, target_sum)
    else:
        # binary start: exactly half ones
        h = np.zeros(n, dtype=float)
        h[:n // 2] = 1.0
        np.random.shuffle(h)

    # Track the best solution found
    cur_val, _ = _max_overlap(h)
    best_h = h.copy()
    best_val = cur_val

    start_time = time.time()
    eta0 = 0.6          # initial step size
    max_iter = 1000000  # safety cap

    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.99:
            break

        # correlation and the shift attaining the maximum
        cur_val, conv = _max_overlap(h)
        idx = int(np.argmax(conv))
        shift = idx - (n - 1)          # corresponds to integer shift in samples

        # gradient of the worst‑case overlap (sub‑gradient of max)
        g = np.zeros_like(h)
        if shift >= 0:
            s = shift
            g[:n - s] += 1.0 - h[s:]
            g[s:] -= h[:n - s]
        else:
            s = -shift
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]

        # step size that decays towards the end of the time budget
        t = elapsed / budget_s
        eta = eta0 * (1.0 - t) ** 2

        # projected sub‑gradient step
        h_new = _project_box_sum(h - eta * g, target_sum)
        new_val, _ = _max_overlap(h_new)

        if (it + 1) % 500 == 0:
            jitter = np.random.randn(n) * 0.02
            candidate = _project_box_sum(best_h + jitter, target_sum)
            cand_val, _ = _max_overlap(candidate)
            if cand_val < best_val:
                best_val = cand_val
                best_h = candidate.copy()
            # reset the working point to the incumbent best
            h = best_h.copy()
            cur_val = best_val

    # Final local refinement using small swaps (keeps the sum exact)
    refined = best_h.copy()
    refined_val = best_val
    for _ in range(5000):

        if i == j:

        inc_up = 1.0 - refined[i]
        dec_down = refined[j]

            refined = cand
            refined_val = cand_val

    # Final projection to guarantee exact sum
    refined = _project_box_sum(refined, target_sum)
    final_val, _ = _max_overlap(refined)

    return refined.tolist(), final_val, n
