def improvement_step(h):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(c)
    shift = max_idx - (n - 1)  # shift positive when shift to right

    # compute grad as described
    grad = np.zeros(n)
    # vectorized approach:
    # term1 for indices where j+shift in range
    if shift >= 0:
        # term1 for j where j+shift < n
        term1 = np.zeros(n)
        term1[:n - shift] = 1 - h[shift:]
        # term2 for j where j-shift >=0
        term2 = np.zeros(n)
        term2[shift:] = h[:n - shift]
    else:
        # shift negative
        s = -shift
        term1 = np.zeros(n)
        term1[s:] = 1 - h[:n - s]  # because j+shift = j - s
        term2 = np.zeros(n)
        term2[:n - s] = h[s:]

def propose_gradient_move(h):

        term1[:] = 0.0

        term2[:] = 0.0
