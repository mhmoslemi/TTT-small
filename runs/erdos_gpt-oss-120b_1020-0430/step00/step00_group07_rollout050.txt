def project_01_sum(x, target):
    # x is array. We'll apply following:
    # clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # compute current sum
    cur = x.sum()
    # if cur == target: return x
    # else need to adjust.
    # Determine diff
    diff = target - cur
    # We'll adjust using capacity.
    # While |diff| > 1e-12:
    # Find indices where we can add (if diff > 0) or subtract (if diff < 0)
    # Add uniformly across those indices.
    i = np.arange(len(x))
    if diff > 0:
        # indices where x < 1
        idx = np.where(x < 1 - 1e-12)[0]
        if len(idx) == 0:
            # cannot add, but diff >0 due to rounding; just return x
            return x
        # compute total capacity
        cap = (1 - x[idx]).sum()
        # delta per index proportionally: add = diff * ( (1 - x[idx]) / cap )
        add = diff * (1 - x[idx]) / cap
        x[idx] += add
    else:
        # diff < 0: need to subtract
        idx = np.where(x > 1e-12)[0]
        if len(idx) == 0:
            return x
        cap = x[idx].sum()
        sub = (-diff) * x[idx] / cap
        x[idx] -= sub
    # after one pass, rounding errors may remain; adjust small leftover by iterating again
    # but likely sum now close to target.
    # Ensure again clip
    x = np.clip(x, 0.0, 1.0)
    return x

import time, math
import numpy as np

def _project_01_sum(x, target):
    """Project x onto the box [0,1] while enforcing sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    if abs(cur - target) < 1e-12:
        return x
    diff = target - cur
    # at most two passes are enough for this simple problem
    for _ in range(2):
        if diff > 0:
            idx = np.where(x < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            cap = (1.0 - x[idx]).sum()
            if cap <= 0:
                break
            add = diff * (1.0 - x[idx]) / cap
            x[idx] += add
        else:
            idx = np.where(x > 1e-12)[0]
            if idx.size == 0:
                break
            cap = x[idx].sum()
            if cap <= 0:
                break
            sub = (-diff) * x[idx] / cap
            x[idx] -= sub
        x = np.clip(x, 0.0, 1.0)
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
    return x


def _c5_value(h):
    """Return the C5 objective for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _worst_shift(h):
    """Return the integer shift (offset) that gives the maximal overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    return idx - (n - 1)          # convert index to signed offset


def _gradient_worst_shift(h, shift):
    """Sub‑gradient of the worst‑shift correlation w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    i = np.arange(n)

    ip = i + shift
    mask_ip = (ip >= 0) & (ip < n)
    g[i[mask_ip]] += 1.0 - h[ip[mask_ip]]

    im = i - shift
    mask_im = (im >= 0) & (im < n)
    g[i[mask_im]] -= h[im[mask_im]]
    return g


def _gradient_step(h, target_sum, step):
    """Perform one projected gradient step using the worst shift."""
    d = _worst_shift(h)
    g = _gradient_worst_shift(h, d)
    new_h = h - step * g
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = _project_01_sum(new_h, target_sum)
    return new_h


def _random_feasible(n, target_sum, rng):
    """Generate a random feasible h (values in [0,1] with prescribed sum)."""
    x = rng.random(n)
    return _project_01_sum(x, target_sum)


def _local_search(start_h, time_budget, rng):
    """Simple stochastic local search with gradient moves and random swaps."""
    n = start_h.shape[0]
    target = n / 2.0
    best_h = cur_h = start_h.copy()
    best_val = cur_val = _c5_value(cur_h)

    start = time.time()
    max_step, min_step = 0.5, 0.001

    while time.time() - start < time_budget:
        # linearly decreasing step size
        frac = (time.time() - start) / time_budget
        step = max_step * (1.0 - frac) + min_step * frac

        # gradient move
        cand = _gradient_step(cur_h, target, step)
        cand_val = _c5_value(cand)

        # Metropolis acceptance
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / max(step, 0.005)):
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val

        # occasional tiny random transfer to escape flat regions
        if rng.random() < 0.02:
            i, j = rng.integers(n, size=2, replace=False)
            # try moving mass from i to j
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer > 0:
                delta = rng.random() * max_transfer
                cur_h[i] -= delta
                cur_h[j] += delta
                cur_h = _project_01_sum(cur_h, target)
                cur_val = _c5_value(cur_h)
                if cur_val < best_val:
                    best_h, best_val = cur_h.copy(), cur_val
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C5 constant.

    Returns
        (h_values, c5_bound, n_points)
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = rng.random(n)

    target_sum = n / 2.0
    h0 = _project_01_sum(np.clip(h0, 0.0, 1.0), target_sum)

    # ------------------------------------------------------------------
    # optimisation loop – a short refinement of the supplied start and several random restarts
    # ------------------------------------------------------------------
    start_time = time.time()
    time_limit = float(budget_s)

    best_h, best_val = h0.copy(), _c5_value(h0)

    # 1) Refine the supplied construction (if any) for up to ~30 % of the budget
    elapsed = time.time() - start_time
    remaining = time_limit - elapsed
    if remaining > 20.0:
        phase_time = min(remaining * 0.4, 300.0)
        cur_h, cur_val = _local_search(best_h, phase_time, rng)
        if cur_val < best_val:
            best_h, best_val = cur_h, cur_val

    # 2) Random restart phase – many short local searches
    while time.time() - start_time < time_limit * 0.95:
        elapsed = time.time() - start_time
        time_left = time_limit - elapsed
        if time_left < 5.0:
            break
        # allocate a modest chunk for each restart (up to 20 s)
        chunk = min(20.0, time_left * 0.2)
        start_h = _random_feasible(n, target_sum, rng)
        cand_h, cand_val = _local_search(start_h, chunk, rng)
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val

    # final sanity checks (tiny tolerances)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_01_sum(best_h, target_sum)
    best_val = _c5_value(best_h)

    return best_h, best_val, n
