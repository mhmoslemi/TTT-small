    corr = np.correlate(h, 1 - h, mode='full')
    shift_idx = np.argmax(corr)

    # compute gradient: grad[j] = 1 - (h_{j+shift} + h_{j-shift}) where out-of-bounds treat as 0.

def project_onto_capped_simplex(v, target_sum, low=0.0, high=1.0, tol=1e-10, max_iter=100):
    # Solve w = np.clip(v - lam, low, high) s.t. sum w = target_sum.
    # lam is a scalar.
    # Use binary search on lam.
    lo = -np.max(v) - 10.0  # Ensure w_i = high at lo => sum high*n >= target_sum
    hi = np.max(v) + 10.0   # Ensure w_i = low at hi => sum low*n <= target_sum
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, low, high)
        s = w.sum()
        if np.abs(s - target_sum) < tol:
            return w
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    # Return final approximation
    return np.clip(v - (lo+hi)/2, low, high)

def project_onto_capped_simplex(v, target):
    lo = -np.max(v) - 10.0
    hi = np.max(v) + 10.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return w
```

def compute_c5(h):

Now gradient:

```
def grad_worst_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    if shift >= 0:
        sp = np.concatenate([h[shift:], np.zeros(shift)])
        sm = np.concatenate([np.zeros(shift), h[:n - shift]])
    else:
        s = -shift
        sp = np.concatenate([np.zeros(s), h[:n - s]])
        sm = np.concatenate([h[s:], np.zeros(s)])
    grad = 1.0 - (sp + sm)
    return grad, shift
```

import numpy as np, time, math

def project_onto_capped_simplex(v, target):
    lo = -np.max(v) - 10.0
    hi = np.max(v) + 10.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def grad_worst_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    if shift >= 0:
        sp = np.concatenate([h[shift:], np.zeros(shift)])
        sm = np.concatenate([np.zeros(shift), h[:n - shift]])
    else:
        s = -shift
        sp = np.concatenate([np.zeros(s), h[:n - s]])
        sm = np.concatenate([h[s:], np.zeros(s)])
    grad = 1.0 - (sp + sm)
    return grad, shift

def anneal(start_h, start_score, time_limit):
    start = time.time()
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_score = start_score
    best_h = cur_h.copy()
    best_score = cur_score
    T0 = 0.03
    T = T0
    cooling = 0.9999
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, size=2, replace=False)
        max_inc = min(1.0 - cur_h[i], cur_h[j])
        max_dec = min(cur_h[i], 1.0 - cur_h[j])
        if max_inc <= 0 and max_dec <= 0:
            continue
        if max_inc > 0 and (max_dec <= 0 or np.random.rand() < 0.5):
            delta = np.random.rand() * max_inc
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            delta = np.random.rand() * max_dec
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        new_score = compute_c5(new_h)
        diff = new_score - cur_score
        if diff < 0 or np.random.rand() < math.exp(-diff / T):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
        T *= cooling
        if T < 1e-8:
            T = T0
    return best_h, best_score

def gradient_refine(h, target, steps=200):
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    for _ in range(steps):
        grad, _ = grad_worst_shift(cur_h)
        step = 0.5
        improved = False
        for _ in range(10):
            new_h = cur_h - step * grad
            new_h = project_onto_capped_simplex(new_h, target)
            new_score = compute_c5(new_h)
            if new_score < cur_score:
                cur_h = new_h
                cur_score = new_score
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return cur_h, cur_score

import numpy as np, time, math

def _project(v, target):
    lo = -np.max(v) - 10.0
    hi = np.max(v) + 10.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _grad_worst(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    if shift >= 0:
        sp = np.concatenate([h[shift:], np.zeros(shift)])
        sm = np.concatenate([np.zeros(shift), h[: n - shift]])
    else:
        s = -shift
        sp = np.concatenate([np.zeros(s), h[: n - s]])
        sm = np.concatenate([h[s:], np.zeros(s)])
    return 1.0 - (sp + sm), shift

def _anneal(start_h, start_score, limit):
    n = start_h.shape[0]
    cur_h, cur_sc = start_h.copy(), start_score
    best_h, best_sc = cur_h.copy(), cur_sc
    T0, T, decay = 0.08, 0.08, 0.9999
    t0 = time.time()
    while time.time() - t0 < limit:
        i, j = np.random.choice(n, 2, replace=False)
        inc = min(1.0 - cur_h[i], cur_h[j])
        dec = min(cur_h[i], 1.0 - cur_h[j])
        if inc <= 0 and dec <= 0:
            continue
        if inc > 0 and (dec <= 0 or np.random.rand() < 0.5):
            d = np.random.rand() * inc
            nh = cur_h.copy()
            nh[i] += d
            nh[j] -= d
        else:
            d = np.random.rand() * dec
            nh = cur_h.copy()
            nh[i] -= d
            nh[j] += d
        ns = _c5(nh)
        diff = ns - cur_sc
        if diff < 0 or np.random.rand() < math.exp(-diff / T):
            cur_h, cur_sc = nh, ns
            if ns < best_sc:
                best_h, best_sc = nh.copy(), ns
        T = max(T * decay, 1e-9)
    return best_h, best_sc

def _refine(h, target, steps=200):
    cur_h, cur_sc = h.copy(), _c5(h)
    for _ in range(steps):
        g, _ = _grad_worst(cur_h)
        step = 0.5
        improved = False
        for _ in range(10):
            nh = cur_h - step * g
            nh = _project(nh, target)
            ns = _c5(nh)
            if ns < cur_sc:
                cur_h, cur_sc = nh, ns
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return cur_h, cur_sc

def run(seed=42, budget_s=1100, **_):
    np.random.seed(seed)
    start = time.time()
    n = 256                     # even number of samples
    target = n / 2.0            # ∑h = n/2  (dx = 2/n)
    init = globals().get("initial_h_values")
    if init is not None:
        init = np.asarray(init, float)
        x_src = np.linspace(0.0, 2.0, init.shape[0], endpoint=False)
        x_dst = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_dst, x_src, init)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project(h0, target)
    else:
        h0 = _project(np.random.rand(n), target)

    best_h, best_sc = h0.copy(), _c5(h0)

    # multi‑restart simulated annealing
    while True:
        elapsed = time.time() - start
        remain = budget_s - elapsed
        if remain < 5.0:
            break
        if np.random.rand() < 0.3:
            h_start = best_h + (np.random.rand(n) - 0.5) * 0.02
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = _project(h_start, target)
        else:
            h_start = _project(np.random.rand(n), target)
        sc_start = _c5(h_start)
        slice_t = min(remain * 0.6, 8.0)
        cand_h, cand_sc = _anneal(h_start, sc_start, slice_t)
        if cand_sc < best_sc:
            best_h, best_sc = cand_h.copy(), cand_sc

    # final local gradient refinement
    best_h, best_sc = _refine(best_h, target, steps=200)

    return best_h, best_sc, n
