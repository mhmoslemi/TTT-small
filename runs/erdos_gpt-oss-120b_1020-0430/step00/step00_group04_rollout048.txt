import numpy as np
import time
import random
import math

def max_corr(h: np.ndarray) -> float:
    """Maximum value of corr(h, 1‑h) using full linear correlation."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))


def compute_val(h: np.ndarray, dx: float) -> float:
    """C₅ value for a given step function."""
    return max_corr(h) * dx


def repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Clamp to [0,1] and adjust the sum to the required target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple iterative redistribution; diff is tiny after most operations
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return np.clip(h, 0.0, 1.0)

    np.random.seed(seed)
    random.seed(seed)

    n = 200

        val0 = compute_val(h_interp, dx)
        best_h = h_interp
        best_val = val0

    # ------------------------------------------------------------------
    t = np.linspace(0.0, 2.0, n, endpoint=False)
