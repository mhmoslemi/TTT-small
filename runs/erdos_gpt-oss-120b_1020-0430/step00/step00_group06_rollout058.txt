def project_onto_capped_simplex(v, s):
    # clip using bisection
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0, 1)
        cur_sum = proj.sum()
        if cur_sum < s:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0, 1)

def project_onto_capped_simplex(v, target_sum):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # if due to rounding sum not exact, adjust small diff by distributing among non-bound variables
    return w

import numpy as np
import time

# Helper: projection onto the intersection of the hyperplane sum=h_sum and box [0,1]
def _project_onto_capped_simplex(v, h_sum):
    # Find τ such that sum_i clip(v_i+τ,0,1) = h_sum
    lo = -np.max(v) - 1.0          # guarantees all clipped to 0
    hi = 1.0 - np.min(v) + 1.0     # guarantees all clipped to 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < h_sum:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction to hit the sum exactly (avoid rounding drift)
    diff = h_sum - w.sum()
    if abs(diff) > 1e-12:
        # distribute diff among interior points
        interior = (w > 0.0) & (w < 1.0)
        if interior.any():
            w[interior] += diff / interior.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback: adjust the first element within bounds
            idx = np.argmin(np.abs(v))
            w[idx] = np.clip(w[idx] + diff, 0.0, 1.0)
    return w

# compute C5 using the same formula the evaluator uses
def _c5_from_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

# gradient of the minimal self‑correlation (conv) wrt h
def _gradient_min_conv(h, conv, min_indices):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for idx in min_indices:
        shift = idx - (n - 1)          # actual shift value (can be negative)
        s = abs(shift)
        if s == 0:
            grad += 2.0 * h
        else:
            # i+shift part
            grad[:n - s] += h[s:]
            # i-shift part
            grad[s:] += h[:n - s]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function improving the Erdős C5 bound."""

    start_time = time.time()
    np.random.seed(seed)

    # ---- initialise ---------------------------------------------------------

    best_h = None

    # ---- optimisation loop --------------------------------------------------
    for init_h in candidates:

        # keep the best seen

        step = 0.2                     # initial step size
