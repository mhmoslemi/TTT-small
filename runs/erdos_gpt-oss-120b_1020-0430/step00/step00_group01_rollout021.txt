        # sort zeros by original arr descending, choose top diff
        order = np.argsort(-arr[zeros_idx])  # highest values first
        select = zeros_idx[order[:diff]]
        h[select] = 1
    elif diff < 0:
        # need to turn some ones into zeros, choose ones with smallest original value
        ones_idx = np.where(h == 1)[0]
        order = np.argsort(arr[ones_idx])  # smallest values first
        select = ones_idx[order[: -diff]]
        h[select] = 0
    return h

import time
import math
import numpy as np

def _c5_value(h):
    """Return the C5 value for a step function given by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    target_c5 = 0.3808          # we aim to beat this value
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Helper: binary vector with exact half‑ones
    # ------------------------------------------------------------------
    def random_binary(n):
        """Create a binary vector of length n with exactly n/2 ones."""
        half = n // 2
        h = np.zeros(n, dtype=float)
        h[:half] = 1.0
        rng.shuffle(h)
        return h

    # ------------------------------------------------------------------
    # Candidate grid sizes (keep them modest – <1000 points)
    # ------------------------------------------------------------------
    candidate_ns = [200, 300, 400, 500]

    for n in candidate_ns:
        if time.time() - start > budget_s * 0.92:
            break
        if n % 2:
            n += 1                     # need even length for half‑ones

        # initialise a random binary vector
        cur_h = random_binary(n)
        cur_c5 = _c5_value(cur_h)

        best_local_h = cur_h.copy()
        best_local_c5 = cur_c5
        stagnation = 0                # iterations without improvement

        # ------------------------------------------------------------------
        # Main stochastic local search loop for this grid size
        # ------------------------------------------------------------------
        while time.time() - start < budget_s * 0.95:
            # locate the worst shift (largest correlation)
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)           # shift s where max occurs

            # ------------------------------------------------------------------
            # Try a targeted swap that directly reduces the worst shift
            # ------------------------------------------------------------------
            i_swap = j_swap = None
            performed = False

            if shift >= 0:
                left = cur_h[: n - shift]
                right = cur_h[shift:]
                mism = (left == 1) & (right == 0)   # 1 → 0 mismatches
                cand = (left == 0) & (right == 1)   # 0 → 1 compensation
                mism_idx = np.flatnonzero(mism)
                cand_idx = np.flatnonzero(cand)
                if mism_idx.size and cand_idx.size:
                    i_swap = int(rng.choice(mism_idx))
                    j_swap = int(rng.choice(cand_idx))
                    cur_h[i_swap], cur_h[j_swap] = cur_h[j_swap], cur_h[i_swap]
                    performed = True
            else:
                p = -shift
                left = cur_h[p:]               # i index
                right = cur_h[: n - p]         # i‑p index
                mism = (left == 1) & (right == 0)
                cand = (left == 0) & (right == 1)
                mism_idx = np.flatnonzero(mism)
                cand_idx = np.flatnonzero(cand)
                if mism_idx.size and cand_idx.size:
                    i_swap = int(p + rng.choice(mism_idx))
                    j_swap = int(rng.choice(cand_idx))
                    cur_h[i_swap], cur_h[j_swap] = cur_h[j_swap], cur_h[i_swap]
                    performed = True

            if not performed:
                # plain random 1↔0 swap
                ones = np.flatnonzero(cur_h == 1)
                zeros = np.flatnonzero(cur_h == 0)
                if ones.size and zeros.size:
                    i_swap = int(rng.choice(ones))
                    j_swap = int(rng.choice(zeros))
                    cur_h[i_swap], cur_h[j_swap] = cur_h[j_swap], cur_h[i_swap]
                else:
                    break   # degenerate case (should not happen)

            new_c5 = _c5_value(cur_h)

            if new_c5 < cur_c5 - 1e-12:
                # accept improvement
                cur_c5 = new_c5
                if new_c5 < best_local_c5 - 1e-12:
                    best_local_c5 = new_c5
                    best_local_h = cur_h.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                # revert the swap
                cur_h[i_swap], cur_h[j_swap] = cur_h[j_swap], cur_h[i_swap]
                stagnation += 1

            # ------------------------------------------------------------------
            # Escape stagnation – random perturbation
            # ------------------------------------------------------------------
            if stagnation >= 5000:
                # perform a few fresh random swaps
                for _ in range(12):
                    ones = np.flatnonzero(cur_h == 1)
                    zeros = np.flatnonzero(cur_h == 0)
                    if not ones.size or not zeros.size:
                        break
                    a = int(rng.choice(ones))
                    b = int(rng.choice(zeros))
                    cur_h[a], cur_h[b] = cur_h[b], cur_h[a]
                cur_c5 = _c5_value(cur_h)
                stagnation = 0

            # early exit if we already beat the target
            if best_local_c5 <= target_c5:
                break

        # keep the best we have seen for this grid size
        if best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = best_local_h.copy()
            best_n = n
        if best_c5 <= target_c5:
            break

    # ----------------------------------------------------------------------
    # Compare with any provided initial_h_values (if it exists)
    # ----------------------------------------------------------------------
    try:
        init_h = np.asarray(initial_h_values, dtype=float)
        if init_h.ndim == 1 and init_h.size > 0:
            # ensure feasibility (clamp and renormalise if needed)
            h = np.clip(init_h, 0.0, 1.0)
            total = h.sum()
            desired = h.size / 2.0
            if abs(total - desired) > 1e-9:
                # simple scaling while staying in [0,1]; fallback to binary conversion
                # convert to binary via threshold 0.5 and adjust count
                binary = (h >= 0.5).astype(float)
                diff = int(round(desired - binary.sum()))
                if diff > 0:
                    zeros = np.flatnonzero(binary == 0)
                    if zeros.size:
                        take = zeros[:diff]
                        binary[take] = 1.0
                elif diff < 0:
                    ones = np.flatnonzero(binary == 1)
                    if ones.size:
                        take = ones[: -diff]
                        binary[take] = 0.0
                h = binary
            init_c5 = _c5_value(h)
            if init_c5 < best_c5:
                best_c5 = init_c5
                best_h = h.copy()
                best_n = h.size
    except NameError:
        pass

    # ----------------------------------------------------------------------
    # Return result in requested format
    # ----------------------------------------------------------------------
    # Ensure we return a plain python list
    return (best_h.tolist(), float(best_c5), int(best_n))
