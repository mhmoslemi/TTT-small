def simulated_anneal(initial_h, time_limit=...):
    start = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

def project_to_constraints(v):
    # ensure 0<=v<=1 and sum=v_target
    v = np.clip(v, 0.0, 1.0)
    target = v.size/2
    # Use simple iterative adjustment
    for _ in range(100):

        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        free = (v > 0) & (v < 1)
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # cannot adjust further
            break
        delta = diff / free_idx.size
        v[free_idx] += delta
        # clip again
        v = np.clip(v, 0.0, 1.0)
    return v

def project_to_constraints(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    # iterative scaling
    for _ in range(100):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        idx = np.where(free)[0]
        if idx.size == 0:
            # cannot adjust further, break
            break
        delta = diff / idx.size
        v[idx] += delta
        v = np.clip(v, 0.0, 1.0)
    # Ensure sum close to target
    # If still off due to rounding, adjust by small random distribution among free indices
    s = v.sum()
    if abs(s - target_sum) > 1e-9:
        # adjust diff by adding to first free or subtract from first non-zero
        diff = target_sum - s
        # find index not at bound
        for i in range(len(v)):
            if (diff > 0 and v[i] < 1) or (diff < 0 and v[i] > 0):
                adjustment = min(abs(diff), 1 - v[i]) if diff > 0 else min(abs(diff), v[i])
                v[i] += np.sign(diff) * adjustment
                diff -= np.sign(diff) * adjustment
                if abs(diff) < 1e-12:
                    break
    return v
```

def _c5_of(h: np.ndarray) -> float:
    """return C5 = max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    # correlation with zero padding (full mode)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    """Clamp to [0,1] and enforce Σv = target_sum (∑h_i = n/2)."""
    v = np.clip(v, 0.0, 1.0)

        s = v.sum()

                new_h = cur_h.copy()
                new_h[i] -= eps
                new_h[j] += eps
            else:
                max_eps = min(cur_h[j], 1.0 - cur_h[i])
                if max_eps <= 0:
                    continue
                eps = max_eps * random.random() * 0.5
                new_h = cur_h.copy()
                new_h[j] -= eps
                new_h[i] += eps

        # ----- evaluation and acceptance ---------------------------------------
        new_c5 = _c5_of(new_h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < -1e-12:
            accept = True
        elif T > 0.0 and random.random() < math.exp(-delta / (T + 1e-12)):
            accept = True

        if accept:
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()

    # ------------------------------------------------------------------
    # 5. Final projection to guarantee feasibility (tiny numerical drift possible)
    best_h = _project_to_constraints(best_h, target_sum)
    best_c5 = _c5_of(best_h)

    return (best_h.tolist(), float(best_c5), int(n))
