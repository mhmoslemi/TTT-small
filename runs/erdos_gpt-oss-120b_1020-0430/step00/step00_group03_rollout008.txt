import time
import math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.shape[0])

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that sum(h) == target while keeping 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to fix the first element that can absorb the difference
    for i in range(h.size):
        cap = (1.0 - h[i]) if diff > 0 else h[i]
        adj = max(min(diff, cap), -cap)
        h[i] += adj
        diff -= adj
        if abs(diff) < 1e-12:
            break
    return h

    rng = np.random.default_rng(seed)
    start = time.time()

    cur_h = h0.copy()
