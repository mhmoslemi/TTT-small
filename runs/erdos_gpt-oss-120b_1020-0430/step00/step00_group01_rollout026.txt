def compute_c5(h):
    """return the overlap constant C5 for a step function h."""
    dx = _dx(h)
    # max correlation of h with 1-h (linear convolution, zero padding)
    max_corr = np.correlate(h, 1.0 - h, mode="full").max()
    return float(max_corr * dx)


def max_corr_and_shift(h):
    """
    Compute the (unscaled) maximal correlation value and the corresponding shift.
    Returns (max_corr, shift) where shift = idx - (n-1) of the max element.
    """
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(corr.argmax())
    max_corr = float(corr[idx])

    return max_corr, shift

            delta = random.random() * bound

            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # ------------------- gradient descent step -----------------
            g = gradient_for_shift(h, cur_shift)
            h_new = h - grad_step * g
            enforce_constraints(h_new)

        # evaluate the proposal
        new_max_corr, new_shift = max_corr_and_shift(h_new)
        new_val = new_max_corr * _dx(h_new)

        # Metropolis acceptance
        if new_val < cur_val - 1e-12 or random.random() < math.exp(-(new_val - cur_val) / (temp + 1e-12)):
            h = h_new
            cur_val = new_val
            cur_shift = new_shift
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h_new.copy()
                # early exit if we beat the target record
                if best_val <= 0.3808:
                    break
        else:
            no_improve += 1

        # occasional perturbation when stuck
        if no_improve >= stagnation_limit:
            # apply a small burst of random transfers to escape
            for _ in range(8):
                i = random.randrange(n)
                j = random.randrange(n)
                if i == j:
                    continue
                if h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                    continue
                bound = min(0.1, h[i], 1.0 - h[j])
                if bound <= 0.0:
                    continue
                d = random.random() * bound
                h[i] -= d
                h[j] += d
            enforce_constraints(h)
            cur_max_corr, cur_shift = max_corr_and_shift(h)
            cur_val = cur_max_corr * _dx(h)
            no_improve = 0

    # ------------------------------------------------------------------
    # final sanitisation and return
    # ------------------------------------------------------------------
    enforce_constraints(best_h)
    try:
        final_bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_bound = compute_c5(best_h)

    return (best_h, final_bound, int(best_h.size))
