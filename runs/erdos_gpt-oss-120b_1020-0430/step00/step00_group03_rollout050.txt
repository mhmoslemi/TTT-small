def proj_simplex_box(v, target):
    # Clip v to [0,1] first.
    v = np.clip(v, 0, 1)
    # If sum(v) == target within tolerance, return.
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    # Determine if need to increase or decrease sum.
    if s > target:
        # Need to reduce sum: find λ such that sum(max(v - λ, 0)) = target.
        lo = 0.0
        hi = np.max(v)
        for _ in range(60):
            lam = (lo + hi) / 2.0
            reduced = np.maximum(v - lam, 0.0)
            if reduced.sum() > target:
                lo = lam
            else:
                hi = lam

        backward[n-L:] = 0

import time
import numpy as np

def _proj_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {0<=x_i<=1, sum x_i = target}."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    if s > target:
        # need to reduce sum: find λ ≥ 0 with sum(max(v-λ,0)) = target
        lo, hi = 0.0, np.max(v)
        for _ in range(60):
            lam = (lo + hi) * 0.5
            reduced = np.maximum(v - lam, 0.0)
            if reduced.sum() > target:
                lo = lam
            else:
                hi = lam
        v = np.maximum(v - hi, 0.0)
    else:
        # need to increase sum: find λ ≥ 0 with sum(min(v+λ,1)) = target
        lo, hi = 0.0, 1.0 - np.min(v)
        for _ in range(60):
            lam = (lo + hi) * 0.5
            increased = np.minimum(v + lam, 1.0)
            if increased.sum() < target:
                lo = lam
            else:
                hi = lam
        v = np.minimum(v + hi, 1.0)
    return v

def _objective(h: np.ndarray, dx: float) -> float:
    """Return C5 for the vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    dx = 2.0 / n

    best_h = h0.copy()
    best_val = _objective(best_h, dx)

    # ------------------------------------------------------------------

    # Parameters (tuned heuristically)
    init_temp = 0.05          # initial temperature
    final_temp = 0.001        # final temperature

    iters_done = 0
    while True:

        if rng.random() < 0.3:

        else:

        cur_val = _objective(h_cur, dx)

        # allocate a batch of iterations – adapt to remaining time
        remaining = budget_s - (time.time() - start)
        batch = min(80000, int(remaining * 6000), max_total_iters - iters_done)
