def compute_gradient(h, shift):
    n = len(h)
    left = np.zeros_like(h)
    right = np.zeros_like(h)
    if shift > 0:
        left[shift:] = h[:-shift]
        right[:n-shift] = h[shift:]
    elif shift < 0:
        s = -shift
        left[:n-s] = h[s:]
        right[s:] = h[:n-s]
    else:
        left[:] = h
        right[:] = h
    return 1.0 - left - right

def project_to_box_sum(v, target):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        u = np.clip(v + mid, 0.0, 1.0)
        s = u.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    u = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # final fine adjustment if sum not close
    diff = target - u.sum()
    if abs(diff) > 1e-10:
        # adjust among free indices (not at 0 or 1)
        mask = (u > 1e-12) & (u < 1 - 1e-12)
        if mask.any():
            u[mask] += diff / mask.sum()
            u = np.clip(u, 0.0, 1.0)
    return u

def compute_gradient(h, shift):
    n = h.shape[0]
    # left = h_{i - shift}
    left = np.zeros_like(h)
    right = np.zeros_like(h)
    if shift > 0:
        left[shift:] = h[:-shift]
        right[:n-shift] = h[shift:]
    elif shift < 0:
        s = -shift
        left[:n-s] = h[s:]
        right[s:] = h[:n-s]
    else:
        left[:] = h
        right[:] = h
    return 1.0 - left - right

import numpy as np, time

def _project_to_box_sum(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        u = np.clip(v + mid, 0.0, 1.0)
        s = u.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    u = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - u.sum()
    if abs(diff) > 1e-12:
        mask = (u > 1e-12) & (u < 1.0 - 1e-12)
        if mask.any():
            u[mask] += diff / mask.sum()
            u = np.clip(u, 0.0, 1.0)
    return u

def _evaluate_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)

def _corr_info(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    max_corr = corr[idx]
    return max_corr, idx, corr

def _gradient(h, shift):
    n = h.shape[0]
    left = np.zeros_like(h)
    right = np.zeros_like(h)
    if shift > 0:
        left[shift:] = h[:-shift]
        right[:n-shift] = h[shift:]
    elif shift < 0:
        s = -shift
        left[:n-s] = h[s:]
        right[s:] = h[:n-s]
    else:
        left[:] = h
        right[:] = h
    return 1.0 - left - right
