def propose_move(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    hi = h[i]
    hj = h[j]
    # feasible delta such that new hi=hi-delta, hj=hj+delta in [0,1]
    low = max(hi - 1.0, -hj)
    high = min(hi, 1.0 - hj)
    if low >= high:
        return None
    # choose delta uniformly
    delta = rng.uniform(low, high)
    # optionally scale with epsilon
    # but as low-high may be small, delta automatically limited.
    h2 = h.copy()
    h2[i] = hi - delta
    h2[j] = hj + delta
    return h2

def propose_move(h, rng):
    n = len(h)
    # randomly select two indices
    i, j = rng.choice(n, size=2, replace=False)
    hi = h[i]
    hj = h[j]
    # feasible delta
    lo = max(hi - 1.0, -hj)
    hi_ = min(hi, 1.0 - hj)
    if lo >= hi_:
        return None
    delta = rng.uniform(lo, hi_)
    # create new h copy
    h_new = h.copy()
    h_new[i] = hi - delta
    h_new[j] = hj + delta
    return h_new

def adjust_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    asc = np.argsort(- (1 - h)) if diff > 0 else np.argsort(- h)
    # sort by capacity descending
    for idx in asc:
        if diff <= 0:
            cap = h[idx]  # can decrease up to h
            delta = min(cap, -diff)
            h[idx] -= delta
            diff += delta
        else:
            cap = 1 - h[idx]  # can increase
            delta = min(cap, diff)
            h[idx] += delta
            diff -= delta
        if abs(diff) < 1e-12:

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time

    rng = np.random.default_rng(seed)
    # determine n and initial h
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=np.float64)
        n = h_initial.size
    else:
        n = 70  # fallback
        # start binary equally split
        h_initial = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, size=n//2, replace=False)
        h_initial[ones] = 1.0

    dx = 2.0 / n
    target_sum = n / 2.0

    def evaluate(h):
        # returns C5 bound
        # Use np.correlate; however, (1-h) may be float, ensure correct dtype
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    def repair(h):
        # Ensure sum = target and bounds
        # Clip to [0,1]
        h = np.clip(h, 0.0, 1.0)
        # Adjust sum
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-12:
            # Determine indices where we can add or subtract
            if diff > 0:
                # need to increase sum, increase some entries not at 1
                caps = 1.0 - h
                # sort indices by capacity descending
                idx = np.argsort(-caps)
                for i in idx:
                    if caps[i] <= 0:
                        continue
                    delta = min(caps[i], diff)
                    h[i] += delta
                    diff -= delta
                    if diff <= 1e-12:
                        break
            else:
                # need to decrease sum, reduce some entries not at 0
                caps = h
                idx = np.argsort(-caps)
                for i in idx:
                    if caps[i] <= 0:
                        continue
                    delta = min(caps[i], -diff)
                    h[i] -= delta
                    diff += delta
                    if diff >= -1e-12:
                        break
        # final clip again
        h = np.clip(h, 0.0, 1.0)
        return h

    # start from initial, repair just to be safe
    h_current = repair(h_initial.copy())
    best_h = h_current.copy()
    best_score = evaluate(best_h)
    current_score = best_score

    start_time = time.time()
    # budget for evaluation
    time_limit = budget_s * 0.95

    max_iter = 10_000_000  # large, actual break by time
    # parameters
    T0 = 0.02
    cooling_rate = 0.9995
    T = T0
    stagnation = 0
    stagn_limit = 5000
    # We'll also incorporate restarts
    while True:
        if time.time() - start_time > time_limit:
            break
        # propose move
        # Randomly choose two indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        hi = h_current[i]
        hj = h_current[j]
        low = max(hi - 1.0, -hj)
        high = min(hi, 1.0 - hj)
        if low >= high:
            # try again
            continue
        delta = rng.uniform(low, high)
        # create candidate
        h_new = h_current.copy()
        h_new[i] = hi - delta
        h_new[j] = hj + delta
        # Evaluate
        new_score = evaluate(h_new)
        # accept?
        if new_score < current_score:
            accept = True
        else:
            # Probabilistic acceptance
            prob = np.exp(-(new_score - current_score) / max(T, 1e-12))
            accept = rng.random() < prob
        if accept:
            h_current = h_new
            current_score = new_score
            stagnation = 0
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            stagnation += 1

        # cooling
        T = T * cooling_rate
        # restart if stagnant
        if stagnation >= stagn_limit:
            # restart from best with random perturbation
            h_current = best_h.copy()
            # apply few random swaps to randomize
            for _ in range(5):
                i2, j2 = rng.integers(0, n, size=2, endpoint=False)
                if i2 == j2:
                    continue
                hi2 = h_current[i2]
                hj2 = h_current[j2]
                low2 = max(hi2 - 1.0, -hj2)
                high2 = min(hi2, 1.0 - hj2)
                if low2 >= high2:
                    continue
                delta2 = rng.uniform(low2, high2)
                h_current[i2] = hi2 - delta2
                h_current[j2] = hj2 + delta2
            # recalc current score
            current_score = evaluate(h_current)
            stagnation = 0
            # Reset temperature a bit
            T = T0

    # final repair to enforce constraints precisely
    best_h = repair(best_h)
    final_score = evaluate(best_h)
    # return tuple
    return (best_h.tolist(), float(final_score), int(n))

import time
import numpy as np

def _evaluate(h, dx):
    """Return C5 bound for a step vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _repair(h, target_sum):
    """Clip to [0,1] and adjust the sum to target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            caps = 1.0 - h
            order = np.argsort(-caps)               # largest capacity first
            for i in order:
                inc = min(caps[i], diff)
                if inc <= 0:
                    continue
                h[i] += inc
                diff -= inc
                if diff <= 1e-12:
                    break
        else:
            caps = h
            order = np.argsort(-caps)               # largest value first
            for i in order:
                dec = min(caps[i], -diff)
                if dec <= 0:
                    continue
                h[i] -= dec
                diff += dec
                if diff >= -1e-12:
                    break
    return np.clip(h, 0.0, 1.0)

def _gradient(h, L):
    """Gradient of the overlap for a given integer shift L."""
    n = h.size
    g = np.zeros(n, dtype=np.float64)
    if L >= 0:
        for i in range(n - L):
            g[i]     += 1.0 - h[i + L]   # ∂/∂h_i of h_i*(1-h_{i+L})
            g[i + L] -= h[i]            # ∂/∂h_{i+L}
    else:
        Lp = -L
        for i in range(Lp, n):
            j = i - Lp
            g[i]     += 1.0 - h[j]
            g[j]     -= h[i]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values, dtype=np.float64)
        n = h_current.size
    else:
        n = 70
        h_current = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, size=n // 2, replace=False)
        h_current[ones] = 1.0

    dx = 2.0 / n
    target_sum = n / 2.0
    h_current = _repair(h_current, target_sum)

    best_h = h_current.copy()
    best_score = _evaluate(best_h, dx)

    # ------------------------------------------------------------------
    # simulated‑annealing parameters
    T0 = 0.02
    cooling = 0.9995
    T = T0
    stagn = 0
    stagn_limit = 5000
    guided_prob = 0.7
    start = time.time()
    time_limit = budget_s * 0.95

    while time.time() - start < time_limit:
        # current evaluation and worst shift
        corr = np.correlate(h_current, 1.0 - h_current, mode='full')
        idx_max = int(np.argmax(corr))
        cur_score = corr[idx_max] * dx
        if cur_score < best_score:
            best_h = h_current.copy()
            best_score = cur_score
            stagn = 0
        else:
            stagn += 1

        L = idx_max - (n - 1)          # integer shift producing the max

        # propose a move
        if rng.random() < guided_prob:
            # gradient‑guided move
            g = _gradient(h_current, L)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if g[i] > 0 and g[j] < 0:
                hi, hj = h_current[i], h_current[j]
                max_delta = min(hi, 1.0 - hj)
                if max_delta > 1e-12:
                    delta = rng.random() * max_delta * 0.5
                    cand = h_current.copy()
                    cand[i] = hi - delta
                    cand[j] = hj + delta
                    cand_score = _evaluate(cand, dx)
                    if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                        h_current = cand
                        cur_score = cand_score
                        stagn = 0
                        if cand_score < best_score:

                            best_score = cand_score

            low = max(hi - 1.0, -hj)
            high = min(hi, 1.0 - hj)
            if low >= high:
                continue
            delta = rng.uniform(low, high)
            cand = h_current.copy()
            cand[i] = hi - delta
            cand[j] = hj + delta
            cand_score = _evaluate(cand, dx)
            if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                h_current = cand
                cur_score = cand_score
                stagn = 0

                    best_score = cand_score

        if stagn >= stagn_limit:

            # a few small random perturbations to escape local plateau
            for _ in range(4):

                if i == j:

                high = min(hi, 1.0 - hj)

                h_current[i] = hi - delta
                h_current[j] = hj + delta
            h_current = _repair(h_current, target_sum)
            stagn = 0
            T = T0

    # final polishing
    best_h = _repair(best_h, target_sum)
    final_score = _evaluate(best_h, dx)

    return (best_h.tolist(), float(final_score), int(n))
