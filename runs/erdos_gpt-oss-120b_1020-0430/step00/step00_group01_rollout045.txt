```
def adjust_sum_to_target(h, target_sum):
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    # Choose indices sorted by values for efficiency
    if diff > 0:
        # need to increase some entries
        idx = np.argsort(h)  # smallest first
        for i in idx:
            inc = min(diff, 1 - h[i])
            if inc <= 0:
                continue
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # diff < 0, need to decrease some entries
        idx = np.argsort(-h)  # largest first
        for i in idx:
            dec = min(-diff, h[i])
            if dec <= 0:
                continue
            h[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return h
```

import time
import numpy as np
import random
import math

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 bound for a discrete step function `h`."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def adjust_sum_to_target(h: np.ndarray, target: float) -> np.ndarray:
    """
    Adjust `h` in-place so that its sum equals `target`,
    while keeping all entries inside [0,1].
    """
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase some entries; start from the smallest ones
        idx = np.argsort(h)
        for i in idx:
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            inc = min(cap, diff)
            h[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        # need to decrease some entries; start from the largest ones
        idx = np.argsort(-h)
        for i in idx:
            cap = h[i]
            if cap <= 0.0:
                continue
            dec = min(cap, -diff)
            h[i] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):

    random.seed(seed)

    start_time = time.time()
    time_budget = budget_s

    else:
