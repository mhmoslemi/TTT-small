def project_onto_simplex_box(y, S):
    # Input y arbitrary; output x in [0,1] with sum S minimizing squared distance.
    # simple binary search for lambda:
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(y + lam, 0, 1)
        s = x.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    x = np.clip(y + (lo+hi)/2.0, 0, 1)
    # due to rounding may not exactly sum S; adjust small residual by adding to some free indices:
    diff = S - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff to indices not at bounds
        mask = (x > 0) & (x < 1)
        free = np.where(mask)[0]
        if len(free) > 0:
            # distribute equally
            x[free] += diff / len(free)
            x = np.clip(x, 0, 1)
        else:
            # if no free, distribute to any index (e.g., first) and clip
            idx = np.random.randint(len(x))
            x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

import time, math
import numpy as np

# ---------- helper functions ----------
def _project_onto_box_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the set {x : 0<=x<=1, sum(x)=len(v)/2}
    using a simple bisection on the Lagrange multiplier."""
    n = v.size
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        if x.sum() > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v + lam, 0.0, 1.0)

    # tiny correction to hit the target sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust a single entry (should be rare)
            i = np.random.randint(n)
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def _random_feasible(n: int) -> np.ndarray:
    """Generate a random point satisfying the constraints."""
    y = np.random.rand(n)
    return _project_onto_box_simplex(y)

def _c5(h: np.ndarray) -> float:
    """Compute the C5 value for a step function sampled by h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # small safety margin

    # Determine problem size and initial vector
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        n = init.size
        if init.ndim != 1:
            init = init.ravel()
        cur = _project_onto_box_simplex(init)
    except NameError:
        n = 80
        cur = _random_feasible(n)

    best_h = cur.copy()
    best_score = _c5(best_h)

    # Simulated annealing parameters
    T0, T1 = 1.0, 1e-4          # temperature start / end
    step0, step1 = 0.2, 1e-3   # step-size start / end

    iteration = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        prog = elapsed / (budget_s - 0.1)

        # temperature and step size schedules (exponential / linear)
        T = math.exp(math.log(T0) * (1 - prog) + math.log(T1) * prog)
        step = step0 * (1 - prog) + step1 * prog

        # propose a new feasible point
        trial = cur + np.random.randn(n) * step
        trial = _project_onto_box_simplex(trial)

        score = _c5(trial)

        # accept according to Metropolis rule
        if score < best_score:
            best_h = trial.copy()
            best_score = score
        if score < _c5(cur) or np.random.rand() < math.exp(-(score - _c5(cur)) / T):
            cur = trial

        iteration += 1

        # occasional diversification
        if iteration % 5000 == 0:
            if np.random.rand() < 0.3:
                cur = _random_feasible(n)
            else:
                cur = best_h.copy()

    # final projection & evaluation (to guarantee feasibility)
    best_h = _project_onto_box_simplex(best_h)
    best_score = _c5(best_h)

    return (best_h, float(best_score), n)
