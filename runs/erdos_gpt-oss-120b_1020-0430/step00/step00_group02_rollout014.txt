import time
import math
import numpy as np

def compute_c5(arr):
    """return (C5 bound, full correlation array) for a step vector."""
    n = len(arr)
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * dx, corr

def shift_mismatches(arr, s):
    """indices where a swap would lower the correlation for shift s."""
    n = len(arr)
    if s > 0:
        idx = np.arange(n - s)
        cond1 = (arr[idx] > 0.5) & (arr[idx + s] < 0.5)   # 1,0  -> contributes 1
        cond0 = (arr[idx] < 0.5) & (arr[idx + s] > 0.5)   # 0,1  -> contributes 0
        m1 = idx[cond1]
        m0 = idx[cond0]
    elif s < 0:
        sabs = -s
        idx = np.arange(sabs, n)
        cond1 = (arr[idx] > 0.5) & (arr[idx - sabs] < 0.5)
        cond0 = (arr[idx] < 0.5) & (arr[idx - sabs] > 0.5)
        m1 = idx[cond1]
        m0 = idx[cond0]
    else:
        m1 = np.empty(0, dtype=int)
        m0 = np.empty(0, dtype=int)
    return m1, m0

def _initial_binary(n):
    """binary 0‑1 vector of length n with exactly n/2 ones using the golden‑ratio Beatty cut."""
    phi = (math.sqrt(5.0) - 1.0) / 2.0          # ≈0.618
    idx = np.arange(n, dtype=float)
    frac = np.mod(idx * phi, 1.0)
    h = (frac < 0.5).astype(float)

    target = n // 2
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(h == 0)[0]
        order = zeros[np.argsort(-frac[zeros])]     # closest to 0.5 from below
        h[order[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        order = ones[np.argsort(frac[ones])]        # farthest from 0.5
        h[order[: -diff]] = 0.0
    return h

    h_values is a list of values in [0,1] defined on a uniform grid of size n_points.
    """

    start = time.time()

    # ---------- choose the discretisation ----------
    n = 200                     # even → exact half‑mass attainable with binary values
    h = _initial_binary(n)

    # If the parent construction is present and already respects the integral,
    # start from it (it may already be very good).
    if 'initial_h_values' in globals():

            if parent.shape == (n,) and abs(parent.sum() - n / 2) < 1e-9:
                # same size, valid integral – use it as seed
                h = parent.copy()
        except Exception:
            pass

    best_h = h.copy()
    best_score, corr = compute_c5(best_h)

    target = 0.3808                     # we stop once we reach this bound
    dx = 2.0 / n

    # ---------- local improvement loop ----------
    while time.time() - start < budget_s * 0.95:
        # find the worst shift for the current vector
        idx_max = int(np.argmax(corr))
        worst_val = float(corr[idx_max])          # still without dx factor
        s = idx_max - (n - 1)                     # shift corresponding to the max

        if worst_val * dx <= target:

        m1, m0 = shift_mismatches(best_h, s)
