def project_box_sum(v, target):
    # v is float array shape (n,)
    # returns projected vector with 0<=x<=1, sum(x)=target
    from scipy.optimize import brentq
    # define phi
    def phi(l):
        return np.sum(np.clip(v - l, 0, 1)) - target
    # set lo, hi
    lo = -1.0
    while phi(lo) > 0:
        lo -= 1.0
    hi = np.max(v) + 1.0
    while phi(hi) < 0:
        hi += 1.0
    lam = brentq(phi, lo, hi, xtol=1e-12, maxiter=100)
    return np.clip(v - lam, 0, 1)

def random_local_search(h, iters, temperature):
    cur_h = h.copy()
    cur_val = compute_val(cur_h)
    best_local_h = cur_h.copy()
    best_local_val = cur_val
    for t in range(iters):
        # choose i and j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # compute max delta we can move from i to j
        delta_max = min(cur_h[i], 1 - cur_h[j])
        if delta_max <= 0:
            continue
        # choose random fraction of delta_max, could be sqrt or uniform
        delta = delta_max * rng.random()
        # propose move
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # evaluate bound
        new_val = compute_val(new_h)
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            if new_val < best_local_val:
                best_local_h, best_local_val = new_h.copy(), new_val
        else:
            # maybe accept with prob
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    cur_h, cur_val = new_h, new_val

import numpy as np, time, math
from scipy.optimize import brentq

def _project_box_sum(v, target):
    # project onto [0,1] with sum constraint
    def phi(l):
        return np.sum(np.clip(v - l, 0.0, 1.0)) - target
    lo = -1.0
    # ensure phi(lo) >0
    while phi(lo) > 0:
        lo -= 1.0
    hi = np.max(v) + 1.0
    while phi(hi) < 0:
        hi += 1.0
    lam = brentq(phi, lo, hi, xtol=1e-12, maxiter=100)
    return np.clip(v - lam, 0.0, 1.0)

def _correlation(h):
    # raw sum correlation (no dx)
    return np.correlate(h, 1.0 - h, mode='full')

def _max_val(h, dx):
    # compute max integral value
    return np.max(_correlation(h)) * dx

def _gradient_of_shift(h, k):
    n = h.shape[0]
    if k == 0:
        return 1.0 - 2.0 * h
    d = abs(k)
    g = np.ones_like(h)
    # subtract h[i+k]
    g[:n-d] -= h[d:]
    # subtract h[i-k]
    g[d:] -= h[:n-d]
    return g

def _subgradient_descent(start_h, target_sum, dx, max_iters, step0, rng, time_deadline):
    h = start_h.copy()
    best_local_h = h.copy()
    best_local_val = _max_val(h, dx)
    n = h.shape[0]
    for it in range(max_iters):
        c = _correlation(h)
        idx = int(np.argmax(c))
        k = idx - (n-1)
        # update best local
        cur_val = c[idx] * dx
        if cur_val < best_local_val:
            best_local_val = cur_val
            best_local_h = h.copy()
        # gradient for the active shift
        g = _gradient_of_shift(h, k)
        alpha = step0 / math.sqrt(it + 1)
        h_next = h - alpha * g
        h = _project_box_sum(h_next, target_sum)
        if time.time() > time_deadline:
            break
    return best_local_h, best_local_val

def _local_random_swap(h, target_sum, dx, n_swaps, temp0, rng, time_deadline):
    cur = h.copy()
    cur_val = _max_val(cur, dx)
    best_h = cur.copy()
    best_val = cur_val
    temperature = temp0
    n = cur.shape[0]
    for _ in range(n_swaps):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # amount we can move from i to j while staying in [0,1]
        delta_max = min(cur[i], 1.0 - cur[j])
        if delta_max <= 0:
            continue
        delta = delta_max * rng.random()
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_val = _max_val(new, dx)
        if new_val < cur_val - 1e-12:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best_h, best_val = new, new_val
        else:
            # occasional uphill move
            if temperature > 0:
                prob = math.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    cur, cur_val = new, new_val
        temperature *= 0.9995
        if time.time() > time_deadline:
            break
