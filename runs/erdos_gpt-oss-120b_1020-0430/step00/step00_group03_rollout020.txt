function project_to_sum_and_bounds(vec, target_sum):

    # shift method: find alpha such that sum(clamp(vec + alpha, 0, 1)) = target_sum

    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        clipped = np.clip(vec + mid, 0.0, 1.0)
        s = clipped.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) / 2, 0.0, 1.0)

import time
import numpy as np

def _project_to_sum_and_bounds(z, target_sum):
    """Project vector z onto the hyperplane sum(x)=target_sum with 0<=x<=1.
    The projection is the unique solution of min ||x-(z)||^2 subject to the constraints.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(z + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
    return x


def _evaluate(vec, dx):
    """C5 value for a given sample vector."""
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return float(np.max(corr) * dx)


def _grad_min_corr(h, lag):
    """Gradient of the autocorrelation term for the lag that attains the minimum."""
    N = h.shape[0]
    g = np.zeros(N, dtype=np.float64)
    if lag >= 0:
        k = lag
        if k < N:
            g[: N - k] += h[k:]
            g[k:] += h[: N - k]
    else:
        k = -lag
        if k < N:
            g[k:] += h[: N - k]
            g[: N - k] += h[k:]
    return g
