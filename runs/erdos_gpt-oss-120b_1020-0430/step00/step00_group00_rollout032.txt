        # compute allowed delta interval

        hi = min(step, 1 - cur_h[i], cur_h[j])

        delta = np.random.uniform(lo, hi)

        cur_h[i] += delta

                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] -= delta
            cur_h[j] += delta
        iteration += 1

def slsqp_refine(h0, deadline, compute_c5):
    import numpy as np, time
    from scipy.optimize import minimize
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        # compute C5 via compute_c5 defined above maybe
        # but compute_c5 uses dx internally; re-calc
        corr = np.correlate(x, x, mode='full')
        min_corr = np.min(corr[n-1:]) * dx
        return 1.0 - min_corr

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target})
    bounds = [(0.0, 1.0)] * n
    # limit iterations based on remaining time
    remaining = deadline - time.time()
    maxiter = 200 if remaining > 10 else 50
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    if res.success:
        h_opt = res.x
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt
    else:
        return h0, compute_c5(h0)

def project_feasible(v, target, max_iter=100):
    import numpy as np
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0) & (v < 1))[0]
        if free.size == 0:
            # cannot adjust further
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final correction if needed
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        # distribute proportionally to available slack
        # find indices that can increase
        inc_mask = v < 1 - 1e-12
        dec_mask = v > 1e-12
        # try simple flat distribution
        if diff > 0:
            # need increase
            candidates = np.where(inc_mask)[0]
            for idx in candidates:
                add = min(diff, 1 - v[idx])
                v[idx] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            # need decrease
            candidates = np.where(dec_mask)[0]
            for idx in candidates:
                sub = min(-diff, v[idx])
                v[idx] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    return v

import time, random
import numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def project_feasible(v, target, max_iter=100):
    """Project v onto the hyperplane sum(v)=target with 0<=v<=1."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0) & (v < 1))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # tiny fix if needed
    diff = target - v.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            inc = np.where(v < 1)[0]
            for i in inc:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            dec = np.where(v > 0)[0]
            for i in dec:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    return v


def random_feasible_h(n):
    target = n / 2.0
    v = np.random.rand(n)
    v = v * target / v.sum()
    v = project_feasible(v, target)
    return v


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx = 1 - min_k ∫ h(x)h(x+k)dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, h, mode="full")
    min_corr = np.min(corr[n - 1:]) * dx
    return 1.0 - min_corr


# ---------- local improvement ----------
def improve_worst(h0, deadline, compute_c5_func):
    """Targeted improvement: raise the minimum autocorrelation."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5_func(h)
    cur_val = best_val
    step = 0.4
    it = 0
    while time.time() < deadline:
        corr = np.correlate(h, h, mode="full")
        sub = corr[n - 1:]                 # lag 0 .. n-1
        k = int(np.argmin(sub))            # worst lag
        if k == 0:
            # improve a single coordinate
            prod = h * h
            i = int(np.argmin(prod))
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            j = int(np.argmax(h[mask]))
            # map j back to original index
            j = int(np.arange(n)[mask][j])
            inc = min(step, 1.0 - h[i], h[j])
            if inc <= 1e-12:
                step *= 0.9
                continue
            h[i] += inc
            h[j] -= inc
        else:
            # improve a pair (i,i+k)
            prod = h[: n - k] * h[k:]
            i = int(np.argmin(prod))
            i_k = i + k
            mask = np.ones(n, dtype=bool)
            mask[[i, i_k]] = False
            donor_idx = np.where(mask)[0]
            if donor_idx.size == 0:
                break
            j = int(donor_idx[np.argmax(h[donor_idx])])
            max_inc_i = 1.0 - h[i]
            max_inc_ik = 1.0 - h[i_k]
            max_dec_j = h[j]
            delta_max = min(2 * max_inc_i, 2 * max_inc_ik, max_dec_j, step)
            if delta_max <= 1e-12:
                step *= 0.9
                continue
            delta = np.random.uniform(0, delta_max)
            inc = delta / 2.0
            h[i] += inc
            h[i_k] += inc
            h[j] -= delta
        new_val = compute_c5_func(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert the move
            if k == 0:
                h[i] -= inc
                h[j] += inc
            else:
                h[i] -= inc
                h[i_k] -= inc
                h[j] += delta
        it += 1
        if it % 200 == 0:
            step = max(0.01, step * 0.95)
    return best_h, best_val


def improve_random(h0, deadline, compute_c5_func):
    """Simple random pairwise adjustments."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5_func(h)
    cur_val = best_val
    step = 0.5
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        lo = max(-step, -h[i], -(1.0 - h[j]))
        hi = min(step, 1.0 - h[i], h[j])
        if lo >= hi:
            continue
        delta = np.random.uniform(lo, hi)
        h[i] += delta
        h[j] -= delta
        new_val = compute_c5_func(h)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
        it += 1
        if it % 500 == 0:
            step = max(0.01, step * 0.98)
    return best_h, best_val


def slsqp_refine(h0, deadline, compute_c5_func):
    """Tiny SLSQP polishing."""
    n = h0.shape[0]
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, x, mode="full")
        min_corr = np.min(corr[n - 1:]) * dx
        return 1.0 - min_corr

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    maxiter = 200 if remaining > 5 else 50
    try:
        res = minimize(
            obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
        )
    except Exception:
        return h0, compute_c5_func(h0)
    if res.success:
        h_opt = res.x
        return h_opt, compute_c5_func(h_opt)
    return h0, compute_c5_func(h0)
