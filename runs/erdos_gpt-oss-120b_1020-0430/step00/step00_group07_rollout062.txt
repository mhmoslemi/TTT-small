import time
import numpy as np
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Return max_{k} ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(x, target_sum, tol=1e-12, max_iter=20):
    """Clip to [0,1] and adjust to satisfy sum(x)=target_sum."""
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) <= tol:
        return x
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        if diff > 0.0:
            free = x < 1.0 - 1e-14
            if not np.any(free):
                break
            inc = diff / np.count_nonzero(free)
            inc = np.minimum(inc, 1.0 - x[free])
            x[free] += inc
        else:
            free = x > 1e-14
            if not np.any(free):
                break
            dec = (-diff) / np.count_nonzero(free)
            dec = np.minimum(dec, x[free])
            x[free] -= dec
        diff = target_sum - np.sum(x)
    # final tiny adjustment on a single free entry if needed
    if abs(diff) > 1e-9:
        i = np.argmax(x) if diff > 0 else np.argmin(x)
        x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return np.clip(x, 0.0, 1.0)

def shape_from_a(a, n, dx, target_sum):
    """One‑parameter family: linear ramps of length a (0≤a≤1) and flat top.
    Height H = 1/(2−a) ensures total area =1."""
    if a <= 0.0:
        # constant function of height 0.5
        h = np.full(n, 0.5, dtype=np.float64)
        return make_feasible(h, target_sum)
    H = 1.0 / (2.0 - a)
    ne = int(round(a * n / 2.0))
    ne = min(ne, n // 2)
    plateau_len = n - 2 * ne
    h = np.empty(n, dtype=np.float64)
    # left ramp
    if ne:
        i = np.arange(ne)
        x = (i + 0.5) * dx
        h[:ne] = H * x / a
    # plateau
    h[ne: ne + plateau_len] = H
    # right ramp
    if ne:
        i = np.arange(ne)
        x = (n - ne + i + 0.5) * dx
        h[n - ne:] = H * (2.0 - x) / a
    h = np.clip(h, 0.0, 1.0)
    return make_feasible(h, target_sum)

def slsqp_optimize(x0, target_sum, dx, max_iter=300):
    """Local refinement with SLSQP."""
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)
    bounds = [(0.0, 1.0)] * len(x0)

    def obj(x):
        return evaluate_h(x, dx)

    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = make_feasible(res.x, target_sum)
    return x, evaluate_h(x, dx)

def simulated_annealing(start_h, target_sum, dx, time_limit):
    """Simple pairwise SA keeping the sum exactly constant."""
    n = len(start_h)
    h = start_h.copy()
    cur_val = evaluate_h(h, dx)
    best_h, best_val = h.copy(), cur_val

        # feasible delta range (move mass from j to i)
        pos_up = min(step, 1.0 - h[i], h[j])          # increase i, decrease j
        neg_up = min(step, h[i], 1.0 - h[j])          # decrease i, increase j
        if pos_up + neg_up == 0.0:
            continue
        delta = np.random.rand() * (pos_up + neg_up) - neg_up
        # apply move
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # safety clip (should be already feasible)
        h_new = np.clip(h_new, 0.0, 1.0)

        new_val = evaluate_h(h_new, dx)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            h, cur_val = h_new, new_val
            if cur_val < best_val:
                best_h, best_val = h.copy(), cur_val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = make_feasible(np.array(initial_h_values, dtype=np.float64),
                           target_sum=n / 2.0)
    else:
        n = 80
        h0 = None
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # evaluate the provided construction if any
    if h0 is not None:
        val = evaluate_h(h0, dx)
        best_h, best_val = h0, val

    # ------------------------------------------------------------------
    # one‑parameter family search (parameter a ∈ [0,1])
    a_grid = np.linspace(0.0, 1.0, 101)
    for a in a_grid:
        if time.time() - start_time > budget_s * 0.15:
            break
        h = shape_from_a(a, n, dx, target_sum)
        val = evaluate_h(h, dx)
        if val < best_val:
            best_h, best_val = h, val

    # ------------------------------------------------------------------
    # random binary starts + quick SLSQP (helps explore other basins)
    while time.time() - start_time < budget_s * 0.5:
        # binary half‑ones
        h_rand = np.zeros(n, dtype=np.float64)
        h_rand[:n // 2] = 1.0
        np.random.shuffle(h_rand)
        # small perturbation
        h_rand += (np.random.rand(n) - 0.5) * 0.1
        h_rand = make_feasible(h_rand, target_sum)

        h_ref, val_ref = slsqp_optimize(h_rand, target_sum, dx, max_iter=150)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref

        if time.time() - start_time > budget_s * 0.5:
            break

    # ------------------------------------------------------------------
    # final simulated annealing from the current best
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5.0:
        h_sa, val_sa = simulated_annealing(best_h, target_sum, dx, time_left * 0.7)
        if val_sa < best_val:
            best_h, best_val = h_sa, val_sa

    # ------------------------------------------------------------------
    # final polish with a short SLSQP run
    time_left = budget_s - (time.time() - start_time)
    if time_left > 2.0:
        h_final, val_final = slsqp_optimize(best_h, target_sum, dx, max_iter=200)
        if val_final < best_val:
            best_h, best_val = h_final, val_final

    # ensure feasibility one last time
    best_h = make_feasible(best_h, target_sum)
    best_val = evaluate_h(best_h, dx)

    return best_h, float(best_val), n
