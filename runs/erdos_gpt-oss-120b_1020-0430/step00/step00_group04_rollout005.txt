def project(h, target_sum):
    # Clip to box
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    cur = h.sum()
    # if sum already target within tolerance, return.

        else:

        count = np.count_nonzero(mask)

            # cannot adjust further, break
            break
        delta = diff / count

import time
import numpy as np
from scipy.interpolate import interp1d

def _c5_value(h):
    """Return the C5 bound for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_to_feasible(h, target):
    """
    Clip to [0,1] and then adjust to hit the exact sum `target`
    (the hyper‑plane ∑h = target) while respecting the box constraints.
    """
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    # water‑filling style correction
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        if diff > 0:
            inc = np.minimum(delta, 1.0 - h[mask])
            h[mask] += inc
        else:
            dec = np.minimum(-delta, h[mask])
            h[mask] -= dec
        cur = h.sum()
        diff = target - cur
    # tiny final correction (should be negligible)
    if abs(h.sum() - target) > 1e-9:
        # distribute the remaining error over non‑boundary entries
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            h[free] += (target - h.sum()) / free.size
            h = np.clip(h, 0.0, 1.0)
    return h


def _gradient_step(h, shift):
    """
    One gradient step for the shift that currently attains the maximal overlap.
    The gradient of the correlation w.r.t. h is:
        g_i = 1 - h_{i+shift} - h_{i-shift}
    (out‑of‑range indices are treated as zero).
    """
    n = h.shape[0]
    if shift >= 0:
        plus = np.concatenate((h[shift:], np.zeros(shift)))
        minus = np.concatenate((np.zeros(shift), h[: n - shift]))
    else:
        s = -shift
        plus = np.concatenate((np.zeros(s), h[: n - s]))
        minus = np.concatenate((h[s:], np.zeros(s)))
    return 1.0 - plus - minus


def _descend(h, target, max_iters=2000):
    """
    Simple projected gradient descent that repeatedly reduces the
    worst (largest) overlap value.
    """
    n = h.shape[0]
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)          # signed shift causing the max overlap
        g = _gradient_step(h, shift)
        # linearly decreasing step size
        alpha = 0.15 * (1.0 - it / max_iters)
        h = h - alpha * g
        h = _project_to_feasible(h, target)
    return h


def _local_search(h, target, iters, rng):
    """
    Tiny hill‑climbing moves: transfer a small amount `delta` from a
    randomly chosen entry i to another entry j, keeping the sum exact.
    Accept the move only if it improves the bound.
    """
    best_h = h.copy()
    best_val = _c5_value(best_h)
    n = h.shape[0]

    for _ in range(iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount that can be moved from i to j while staying in [0,1]
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _c5_value(cand)
        if val < best_val:
            best_h, best_val = cand, val
    return best_h, best_val


def _optimize(n, time_budget, rng):
    """
    Main optimisation driver: many random restarts, each consisting of a
    gradient descent phase followed by a short local‑search refinement.
    Returns the best vector found and its C5 bound.
    """
    target = n / 2.0
    best_h = None
    best_val = np.inf
    start = time.time()

    restart = 0
    while time.time() - start < time_budget:
        restart += 1

        # ----- initialise -------------------------------------------------
        if restart == 1 and "initial_h_values" in globals():
            init = globals()["initial_h_values"]
            if len(init) != n:
                # linear interpolation to the new grid
                x_old = np.linspace(0.0, 2.0, len(init), endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                f = interp1d(x_old, init, kind="linear", fill_value="extrapolate")
                h0 = f(x_new)
            else:
                h0 = init.copy()
        else:
            # binary seed (exactly n/2 ones)
            h0 = np.zeros(n)
            h0[: n // 2] = 1.0
            rng.shuffle(h0)

        # small random perturbation to escape pure binary basin
        eps = 0.04
        h0 += rng.uniform(-eps, eps, size=n)
        h0 = _project_to_feasible(h0, target)

        # ----- gradient descent -----------------------------------------
        h_mid = _descend(h0, target, max_iters=1500)

        # ----- local refinement -----------------------------------------
        h_fin, val = _local_search(h_mid, target, iters=2500, rng=rng)

        if val < best_val:
            best_val, best_h = val, h_fin.copy()
            # early exit if we already beat the target
            if best_val <= 0.3808:
                break

    # final safety projection
    best_h = _project_to_feasible(best_h, target)
    best_val = _c5_value(best_h)
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """

    """
    rng = np.random.default_rng(seed)

    # choose a modest grid size – 84 points gives a good trade‑off

    # guarantee even size (required by the integral constraint)
    if n % 2:

    # reserve a small margin for final bookkeeping
    remaining = max(0.0, budget_s - 0.2)

    best_h, best_val = _optimize(n, remaining, rng)

    # return the solution
    return best_h, best_val, n
