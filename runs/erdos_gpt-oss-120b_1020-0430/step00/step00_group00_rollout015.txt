        if abs(eps) < 1e-12:
            continue

        h_new[i] += eps
        h_new[j] -= eps
        # ensure no small violation due to numeric
        # evaluate

        grad[valid_i2] -= h[valid_i2 - shift]

        grad[valid_i] = 1 - h[valid_i - s]
        # term -h[j+ s] for j from 0..n-1-s
        valid_i2 = np.arange(0, n - s)
        grad[valid_i2] -= h[valid_i2 + s]
    return grad

def grad_shift(h, shift):
    n = h.shape[0]
    grad = np.zeros(n)
    if shift >= 0:
        # part1: i from 0..n-1-shift: (1 - h[i+shift])
        part = 1 - h[shift:]  # size n-shift
        grad[:n-shift] = part
        # part2: -h[i-shift] for i from shift..n-1
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        # part1: i from s..n-1: (1 - h[i-s])
        grad[s:] = 1 - h[:n-s]
        # part2: -h[i+s] for i from 0..n-s-1
        grad[:n-s] -= h[s:]
    return grad
```

import numpy as np
import time, math
from scipy.optimize import minimize

def _evaluate(h, dx):
    """Return max overlap integral for step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _grad_shift(h, shift):
    """Subgradient of the overlap for a given shift (index in full correlation)."""
    n = h.shape[0]
    g = np.zeros(n)
    if shift >= 0:
        # term (1 - h[i+shift]) for i=0..n-1-shift
        g[: n - shift] = 1.0 - h[shift:]
        # term -h[i-shift] for i=shift..n-1
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        # term (1 - h[i-s]) for i=s..n-1
        g[s:] = 1.0 - h[: n - s]
        # term -h[i+s] for i=0..n-1-s
        g[: n - s] -= h[s:]
    return g

def _project_capped_simplex(v, total, tol=1e-10, max_iter=60):
    """Euclidean projection of v onto {0<=x_i<=1, sum x_i = total}."""
    lo = -np.max(v) - 5.0
    hi = np.max(v) + 5.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    y = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if needed
    diff = total - y.sum()
    if abs(diff) > tol:
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
        else:
            # all at bounds – adjust a single coordinate safely
            idx = np.argmin(y) if diff < 0 else np.argmax(y)
            y[idx] = min(1.0, max(0.0, y[idx] + diff))
    return y

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -------------------------------------------------------
    if 'initial_h_values' in globals():
        h_cur = np.asarray(initial_h_values, dtype=float)
        n = h_cur.size
    else:
        n = 120                     # fallback size when no parent solution
        h_cur = rng.random(n)
    target_sum = n / 2.0
    # ensure feasibility
    h_cur = _project_capped_simplex(h_cur, target_sum)

    dx = 2.0 / n
    best_h = h_cur.copy()
    best_val = _evaluate(best_h, dx)

    # ----- parameters -------------------------------------------------------
    step = 0.5                       # maximal epsilon for pair updates
    decay_factor = 0.9995
    min_step = 1e-4
    no_improve = 0
    iter_cnt = 0

    # ----- main optimisation loop -------------------------------------------
    while time.time() - start < budget_s * 0.95:
        # current objective & shift giving the maximum
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_idx = int(np.argmax(corr))
        cur_val = corr[max_idx] * dx
        if cur_val + 1e-12 < best_val:
            best_val = float(cur_val)
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # decide move type
        if rng.random() < 0.7:          # gradient‑guided move
            shift = max_idx - (n - 1)
            grad = _grad_shift(h_cur, shift)
            i_up = int(np.argmin(grad))   # most negative ‑ increase here
            i_down = int(np.argmax(grad)) # most positive ‑ decrease here
            if i_up != i_down:
                max_eps = min(step, h_cur[i_down], 1.0 - h_cur[i_up])
                if max_eps > 1e-12:

                    h_new = h_cur.copy()
                    h_new[i_up] += eps
                    h_new[i_down] -= eps
                    new_val = _evaluate(h_new, dx)
                    if new_val <= cur_val or rng.random() < math.exp(

        else:                           # random pairwise move
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # direction +epsilon on i, -epsilon on j
                max_eps = min(step, 1.0 - h_cur[i], h_cur[j])
                if max_eps > 1e-12 and rng.random() < 0.5:

                    h_new[i] += eps
                    h_new[j] -= eps

                else:

                    if max_eps > 1e-12:

            no_improve = 0

        iter_cnt += 1

    # ----- final refinement (optional) --------------------------------------
    def _objective(v):
        return _evaluate(v, dx)

    cons = {'type': 'eq', 'fun': lambda v: v.sum() - target_sum}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'disp': False}
        )
        if res.success:
            best_h = _project_capped_simplex(res.x, target_sum)
    except Exception:
        pass

    best_val = _evaluate(best_h, dx)
    # ensure final feasibility
    best_h = _project_capped_simplex(best_h, target_sum)
    best_val = _evaluate(best_h, dx)

    return best_h.tolist(), float(best_val), n
