    # direction random: add to i, subtract from j
    # compute max shift up (increase i) allowed:
    max_up = min(1 - h[i], h[j])

    new_h[i] += delta
    new_h[j] -= delta
    return new_h, i, j, delta

def compute_c5(h):
    n = len(h)
    # Using vectorized loops maybe
    m = n
    best = 0.0
    for s in range(-n+1, n):
        if s >=0:
            cv = h[:m-s] * (1 - h[s:])
        else:
            cv = h[-s:] * (1 - h[:m+s])
        val = cv.sum()
        if val > best:
            best = val
    return best * (2.0 / n)

def targeted_move(h):

        partner = i_inds + s

        i_inds = np.arange(s_abs, n)
        partner = i_inds + s  # negative shift
    # contributions
    contr = h[i_inds] * (1 - h[partner])
    i_best = i_inds[np.argmax(contr)]
    p_best = partner[np.argmax(contr)]
    # move epsilon from i_best to p_best
    eps_max = min(h[i_best], 1.0 - h[p_best])
    if eps_max <= 1e-12:

    epsilon = eps_max * 0.5
    new_h = h.copy()
    new_h[i_best] -= epsilon
    new_h[p_best] += epsilon
    return new_h
