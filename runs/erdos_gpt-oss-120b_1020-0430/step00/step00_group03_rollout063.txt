def _score(arr, dx):
    # evaluation of C_5 using the provided helper if available,
    # otherwise fall back to the reference implementation.
    try:
        return evaluate_erdos_solution(arr)

        return np.max(np.correlate(arr, 1 - arr, mode="full")) * dx

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 70
        h0 = None

    # enforce even n (required for the sum‑to‑n/2 constraint)
    if n % 2:
        n += 1
        if h0 is not None:
            h0 = np.append(h0, 0.0)

    target_sum = n / 2.0                     # integral of h must be 1
    dx = 2.0 / n

    # ---- initial solution ---------------------------------------------
    if h0 is not None and h0.size == n:
        cur = h0.copy()
        # enforce box constraints and exact integral
        cur = np.clip(cur, 0.0, 1.0)
        cur_sum = cur.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            # simple repair: scale then correct the last entry
            cur *= target_sum / cur_sum
            cur[-1] = np.clip(cur[-1], 0.0, 1.0)
    else:
        # random binary start (exactly n/2 ones)
        cur = np.zeros(n, dtype=float)
        cur[:int(target_sum)] = 1.0
        rng.shuffle(cur)

    cur_score = _score(cur, dx)
    best = cur.copy()
    best_score = cur_score

    # ---- annealing parameters -----------------------------------------
    T0 = 0.04                     # initial temperature
    min_T = 1e-5
    restart_patience = 30000      # steps without improvement before restart
    patience = 0

    # pre‑compute useful index sets for fast moves
    def donor_receiver_indices(v):
        donors = np.flatnonzero(v > 1e-12)
        receivers = np.flatnonzero(v < 1.0 - 1e-12)
        return donors, receivers

    donors, receivers = donor_receiver_indices(cur)

    # ---- main optimisation loop ----------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.98:
            break

        # linear cooling schedule
        T = max(min_T, T0 * (1.0 - elapsed / budget_s))

        # pick a donor and a receiver for a small mass transfer
        if donors.size == 0 or receivers.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(receivers)
        if i == j:
            continue

        # maximal amount we may move from i to j while staying in [0,1]
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:

        # propose a transfer proportional to temperature
        delta = max_delta * (rng.random() ** 2) * (T / T0)

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # evaluate the new candidate
        new_score = _score(new, dx)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):

            patience = 0
            donors, receivers = donor_receiver_indices(cur)

                best = cur.copy()
                best_score = cur_score

            patience += 1

        if patience >= restart_patience:

            cur[:int(target_sum)] = 1.0
            rng.shuffle(cur)
