def smooth_max(v, p=10.0):
    # returns smoothed max (log-sum-exp) and weights
    maxv = np.max(v)
    expv = np.exp(p * (v - maxv))
    sumexp = np.sum(expv)
    weights = expv / sumexp
    smooth = maxv + np.log(sumexp) / p
    return smooth, weights

def gradient_full(h, weights):
    n = h.size
    g = np.zeros_like(h)
    L = 2 * n - 1
    # weights length L
    for idx, w in enumerate(weights):
        if w == 0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            if n - shift > 0:
                g[:n - shift] += w * (1 - h[shift:])
                g[shift:] -= w * h[:n - shift]
        else:
            t = -shift
            if n - t > 0:
                g[t:] += w * (1 - h[:n - t])
                g[:n - t] -= w * h[t:]
    return g

import time
import numpy as np

def _proj_box_sum(y, target_sum):
    """Project y onto the set {x | 0<=x<=1, sum(x)=target_sum}."""
    lo = -np.max(y)          # makes all y+lo <= 0
    hi = 1.0 - np.min(y)     # makes all y+hi >= 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def _random_feasible(n, target_sum):
    """Generate a random feasible vector of length n."""
    y = np.random.rand(n)
    return _proj_box_sum(y, target_sum)


def _smooth_weights(v, p):
    """Return log‑sum‑exp smooth maximum and the associated weights."""
    vmax = np.max(v)
    expv = np.exp(p * (v - vmax))
    w = expv / expv.sum()
    # smooth value (not needed for optimisation, kept for reference)
    smooth = vmax + np.log(expv.sum()) / p
    return smooth, w


def _gradient_full(h, weights):
    """Gradient of the smoothed max objective w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    offset = n - 1
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        shift = idx - offset
        if shift >= 0:
            lo = n - shift
            if lo > 0:
                g[:lo] += w * (1.0 - h[shift:])
                g[shift:] -= w * h[:lo]
        else:
            t = -shift
            lo = n - t
            if lo > 0:
                g[t:] += w * (1.0 - h[:lo])
                g[:lo] -= w * h[t:]
    return g


def _c5_of(h):
    """Return C5 for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small Erdős C5 bound.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # choose a modest resolution; even length simplifies the constraints
    n = 100
    target_sum = n / 2.0
    dx = 2.0 / n
    p = 20.0                     # smoothing parameter for the max

    # keep the best solution seen so far
    best_h = None
    best_c5 = float('inf')

    # if a parent construction is supplied, evaluate it
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)

        if c5_0 < best_c5:
            best_c5 = c5_0
            best_h = h0.copy()

    # ------------------------------------------------------------------
    # main optimisation loop (random restarts + projected gradient descent)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.94:
        # start from a fresh random feasible point
        h = _random_feasible(n, target_sum)
        v = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(v) * dx
        if c5 < best_c5:
