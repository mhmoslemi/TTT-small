def adjust_sum(h, target):
    # h is within [0,1]; adjust while preserving bounds.
    # Use simple algorithm:
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # compute based on free variables
    free = (h>0) & (h<1)
    # if no free variables, cannot adjust; we just return h (should not happen)
    # else we distribute diff uniformly across free places.
    # iterate until adjustment small.
    max_iter=1000
    for _ in range(max_iter):
        free = (h>0) & (h<1)
        if not np.any(free):
            break
        cnt = free.sum()
        delta = diff / cnt
        # apply delta but keep bounds
        h[free] += delta
        # clip and recompute diff
        over = h > 1
        under = h < 0
        if np.any(over):
            h[over] = 1
        if np.any(under):
            h[under] = 0
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h
```

def adjust_sum(x, target):
    x = x.copy()
    # compute diff = target - sum
    diff = target - x.sum()
    # iterate for at most n times
    for _ in range(len(x)):
        if abs(diff) < 1e-12:
            break
        # indices that can be increased or decreased
        if diff > 0:
            # increase ones with x < 1
            free = np.where(x < 1 - 1e-12)[0]
            if len(free) == 0:
                break
            delta = diff / len(free)
            # clip delta
            delta = min(delta, 1 - x[free].max())
            # Actually we need to ensure not exceed each individually. 
            # We'll distribute equally but clip per element.
            # Simpler: increase each free element by min(delta, 1 - x[i]), then recompute diff.
            inc = np.minimum(delta, 1 - x[free])
            x[free] += inc
            diff = target - x.sum()

            free = np.where(x > 1e-12)[0]

            x[free] += dec
            diff = target - x.sum()

        # adjust first element within bounds

        for i in range(len(x)):
            avail_up = 1 - x[i]
            avail_down = x[i]

def adjust_sum(x, target):
    # ensure sum exactly target using simple water-filling
    x = np.copy(x)
    # clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # compute current sum
    current = x.sum()
    if abs(current - target) < 1e-12:
        return x
    # if need to increase sum
    if current < target:
        # need to add diff
        diff = target - current
        # while diff > 0 and there are indices that can be increased
        while diff > 1e-12:
            free = np.where(x < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            # distribute equally
            inc = diff / free.size
            # limit to capacity
            capacity = 1.0 - x[free]
            inc_per = np.minimum(inc, capacity)
            # if some capacity smaller than inc, they will fill and we need to recompute diff later
            x[free] += inc_per
            diff = target - x.sum()
    else:
        diff = current - target
        while diff > 1e-12:
            free = np.where(x > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            capacity = x[free]
            dec_per = np.minimum(dec, capacity)
            x[free] -= dec_per
            diff = x.sum() - target
    # final small residual correction
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # adjust first element within bounds
        i = 0
        # move residual as much as possible
        if residual > 0:
            x[i] = min(1.0, x[i] + residual)
        else:
            x[i] = max(0.0, x[i] + residual)
    return x
```

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def feasible_delta_val(a, b):
    # a = h[i], b = h[j]; want delta s.t. a+delta in [0,1], b-delta in [0,1]
    lo = max(-a, b - 1.0)
    hi = min(1.0 - a, b)
    return lo, hi

def neighbor(h, max_step):
    n = len(h)
    for _ in range(10):
        i, j = np.random.choice(n, 2, replace=False)
        a = h[i]
        b = h[j]
        lo, hi = feasible_delta_val(a, b)
        # restrict to max_step
        lo = max(lo, -max_step)
        hi = min(hi, max_step)

            delta = np.random.uniform(lo, hi)
            new_h = h.copy()
            new_h[i] = a + delta
            new_h[j] = b - delta
            # clip just in case numeric
            new_h[i] = min(1.0, max(0.0, new_h[i]))
            new_h[j] = min(1.0, max(0.0, new_h[j]))
            return new_h

def adjust_sum(x, target):
    # Simple water-filling to hit exact target while staying in [0,1]
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    eps = 1e-12
    if abs(cur - target) <= eps:
        return x
    if cur < target:
        # need to increase
        diff = target - cur
        while diff > eps:
            free = np.where(x < 1.0 - eps)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            # capacity left
            cap = 1.0 - x[free]
            add = np.minimum(inc, cap)
            x[free] += add
            diff = target - x.sum()
    else:
        # need to decrease
        diff = cur - target
        while diff > eps:
            free = np.where(x > eps)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            cap = x[free]
            sub = np.minimum(dec, cap)
            x[free] -= sub
            diff = x.sum() - target
    # Small residual correction
    residual = target - x.sum()

        for i in range(len(x)):
            if residual > 0:
                avail = 1.0 - x[i]
                add = min(avail, residual)
                x[i] += add
                residual -= add
            else:
                avail = x[i]
                sub = min(avail, -residual)
                x[i] -= sub
                residual += sub
            if abs(residual) <= eps:
                break
    return x

import numpy as np, time, math, random

def evaluate(h):
    """C5 value for discrete h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx


def feasible_delta_val(a, b):
    """Return feasible interval for delta such that a+δ∈[0,1], b-δ∈[0,1]."""
    low = max(-a, b - 1.0)
    high = min(1.0 - a, b)
    return low, high


def neighbor(h, max_step):
    """Two‑index move that conserves the sum exactly."""
    n = len(h)
    for _ in range(10):
        i, j = np.random.choice(n, 2, replace=False)
        a, b = h[i], h[j]
        lo, hi = feasible_delta_val(a, b)
        lo = max(lo, -max_step)
        hi = min(hi, max_step)
        if lo < hi:
            delta = np.random.uniform(lo, hi)
            new_h = h.copy()
            new_h[i] = a + delta
            new_h[j] = b - delta
            # numeric safety
            new_h[i] = min(1.0, max(0.0, new_h[i]))
            new_h[j] = min(1.0, max(0.0, new_h[j]))
            return new_h
    return h.copy()


def adjust_sum(x, target):
    """Project x onto [0,1]^n with sum exactly target (water‑filling)."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    eps = 1e-12
    if abs(cur - target) <= eps:
        return x
    if cur < target:                         # need to increase
        while True:
            diff = target - x.sum()
            if diff <= eps:
                break
            free = np.where(x < 1.0 - eps)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            cap = 1.0 - x[free]
            add = np.minimum(inc, cap)
            x[free] += add
    else:                                    # need to decrease
        while True:
            diff = x.sum() - target
            if diff <= eps:
                break
            free = np.where(x > eps)[0]
            if free.size == 0:
                break
            dec = diff / free.size
            cap = x[free]
            sub = np.minimum(dec, cap)
            x[free] -= sub
    # tiny residual fix
    residual = target - x.sum()
    if abs(residual) > eps:
        for i in range(len(x)):
            if residual > 0:
                amt = min(1.0 - x[i], residual)
                x[i] += amt
                residual -= amt
            else:
                amt = min(x[i], -residual)
                x[i] -= amt
                residual += amt
            if abs(residual) <= eps:
                break
    return x


def max_corr_and_grad(h):
    """Correlation maximum and its gradient (for the dominant shift)."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    max_corr = corr[idx]
    k = idx - (n - 1)
    grad = np.zeros_like(h)
    p = np.arange(n)

    p_plus = p + k
    mask_plus = (p_plus >= 0) & (p_plus < n)
    grad[mask_plus] += 1.0 - h[p_plus[mask_plus]]

    p_minus = p - k
    mask_minus = (p_minus >= 0) & (p_minus < n)
    grad[mask_minus] -= h[p_minus[mask_minus]]
    return max_corr, grad


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # ---- initialise -------------------------------------------------
    init = None
    n = None
    if 'initial_h_values' in globals():

    else:

        init = np.clip(init, 0.0, 1.0)
        init = adjust_sum(init, n / 2.0)

    target_sum = n / 2.0
    best_h = init.copy()
    best_val = evaluate(best_h)

    curr_h = best_h.copy()
    curr_val = best_val

    # ---- parameters -------------------------------------------------
    max_step_initial = 0.5
    min_step = 0.001
    temp_initial = 0.02
    temp_final = 1e-6
    # ---- main optimisation loop ------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.96:
            break
        progress = elapsed / (budget_s * 0.96)

        step = max_step_initial * ((min_step / max_step_initial) ** progress)
        temp = temp_initial * ((temp_final / temp_initial) ** progress)

        # Choose move type
        if random.random() < 0.25:                     # gradient step
            _, grad = max_corr_and_grad(curr_h)
            new_h = curr_h - step * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = adjust_sum(new_h, target_sum)
        else:                                          # random two‑index swap
            new_h = neighbor(curr_h, step)

        new_val = evaluate(new_h)

        if new_val < curr_val - 1e-12:
            curr_h, curr_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h, new_val
        else:
            # Metropolis acceptance
            if temp > 0 and random.random() < math.exp(-(new_val - curr_val) / temp):
                curr_h, curr_val = new_h, new_val

        if best_val <= 0.3808:
            break

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum(best_h, target_sum)
    final_val = evaluate(best_h)

    return best_h.tolist(), float(final_val), n
