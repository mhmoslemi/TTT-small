def project_to_feasible(h):
    h = np.clip(h,0,1)
    target = len(h)/2
    diff = target - h.sum()
    # tolerance
    while abs(diff) > 1e-10:
        if diff > 0:
            # need to increase some entries below 1
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            # clamp increments
            inc = np.minimum(inc, 1 - h[idx])
            h[idx] += inc
            diff = target - h.sum()
        else:
            # diff <0, need to decrease some entries >0
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            dec = (-diff) / idx.size
            dec = np.minimum(dec, h[idx])
            h[idx] -= dec
            diff = target - h.sum()
    return h

import time
import numpy as np

def _evaluate(h, n):
    """C5 value for a discrete h (dx = 2/n)."""
    rev = (1 - h)[::-1]
    conv = np.convolve(h, rev, mode="full")
    return np.max(conv) * (2.0 / n)


def _lag_of_max(h, n):
    """Lag (integer shift) that attains the maximal overlap."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    return idx - (n - 1)   # zero lag corresponds to index n‑1


def _gradient(h, lag, n):
    """Sub‑gradient of the correlation at the maximal lag."""
    g = np.zeros_like(h)
    if lag >= 0:
        g[: n - lag] += 1 - h[lag:]
        g[lag:] -= h[: n - lag]
    else:
        l = -lag
        g[l:] += 1 - h[: n - l]
        g[: n - l] -= h[l:]
    return g


def _project(h, n):
    """Clip to [0,1] and enforce sum_i h_i = n/2."""
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple iterative redistribution
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        inc = diff / idx.size
        if diff > 0:
            add = np.minimum(inc, 1.0 - h[idx])
        else:
            add = np.maximum(inc, -h[idx])
        h[idx] += add
        diff = target - h.sum()
    # final safety clip
    h = np.clip(h, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Choose a working resolution. 200 points give enough freedom while

    n = 200
