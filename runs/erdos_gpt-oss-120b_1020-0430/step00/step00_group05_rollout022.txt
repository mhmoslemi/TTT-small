def simulated_annealing(h, time_limit):
    start = time.time()
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score
    n = len(h)
    dx = 2.0 / n
    
    # Temperature schedule: start T = initial_temp = 1.0 maybe, end T = 1e-4.
    # Use exponential cooldown based on elapsed fraction.
    while time.time() - start < time_limit:
        # Determine temperature based on elapsed fraction
        frac = (time.time() - start) / time_limit
        T = max(0.001, (1 - frac) * 0.5) # linear decline
        # Choose random pair i!=j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # Determine max delta
        max_increase = 1.0 - cur_h[i]
        max_decrease = cur_h[j]
        # also ensure max_increase >0 and max_decrease>0
        if max_increase <= 0 or max_decrease <=0:
            continue
        # delta magnitude sample: random fraction of min(max_increase, max_decrease)
        delta = (np.random.rand() * 0.1) * min(max_increase, max_decrease) # up to 10%?
        # optionally sample with exponential distribution
        # apply
        cur_h[i] += delta
        cur_h[j] -= delta
        # compute new score
        new_score = compute_c5(cur_h)
        # acceptance
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / (T + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] -= delta
            cur_h[j] += delta

    return best_h, best_score

def compute_c5(vec):

def max_overlap_info(vec):
    n = vec.size
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    O = corr * dx
    max_val = O.max()
    # return max value and list of lag offsets achieving near max
    # compute all lags where O within tolerance of max_val
    # tolerance relative small
    eps = 1e-12
    idxs = np.where(O >= max_val - eps)[0]
    lags = idxs - (n - 1)
    return max_val, lags

def neighbor_sum(vec, lags):
    n = vec.size
    neigh = np.zeros(n)
    for lag in lags:
        if lag >= 0:
            neigh[:n-lag] += vec[lag:]
            neigh[lag:] += vec[:n-lag]
        else:
            pos = -lag
            neigh[pos:] += vec[:n-pos]

import numpy as np
import time
import random

def compute_c5(vec: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1−h(x+k))dx for the discrete vector."""
    n = vec.size
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float(corr.max() * dx)


def max_overlap_info(vec: np.ndarray):
    """Return (max_overlap, array_of_lags_with_max)."""
    n = vec.size
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    O = corr * dx
    max_val = O.max()
    # consider all lags whose value is within a tiny tolerance of the maximum
    eps = 1e-10
    idxs = np.where(O >= max_val - eps)[0]
    lags = idxs - (n - 1)   # convert index to lag (zero lag is n-1)
    return max_val, lags


def neighbor_sum(vec: np.ndarray, lags: np.ndarray) -> np.ndarray:
    """For each index i compute Σ_{lag∈lags} (h[i+lag]+h[i-lag]) where defined."""
    n = vec.size
    neigh = np.zeros(n, dtype=np.float64)
    for lag in lags:
        if lag >= 0:
            neigh[: n - lag] += vec[lag:]
            neigh[lag:] += vec[: n - lag]
        else:
            pos = -lag
            neigh[pos:] += vec[: n - pos]
            neigh[: n - pos] += vec[pos:]
    return neigh


def enforce_constraints(vec: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust the total sum to exactly n/2."""
    vec = np.clip(vec, 0.0, 1.0)
    n = vec.size
    target = n / 2.0
    cur = vec.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute the difference to the first coordinates that can absorb it
        for i in range(n):
            room = (1.0 - vec[i]) if diff > 0 else vec[i]
            shift = np.sign(diff) * min(abs(diff), room)
            vec[i] += shift
            diff -= shift
            if abs(diff) < 1e-12:
                break
    return np.clip(vec, 0.0, 1.0)


def steepest_descent(start_vec: np.ndarray, time_limit: float):
    """
    Simple coordinate‑wise descent using the gradient of the worst lag.
    Returns the best vector found and its C5 value.
    """
    n = start_vec.size
    cur = start_vec.copy()
    best = cur.copy()
    best_score = compute_c5(best)

    start = time.time()
    max_iters = 2_000_000
    it = 0
    while it < max_iters and (time.time() - start) < time_limit:
        max_val, lags = max_overlap_info(cur)
        neigh = neighbor_sum(cur, lags)

        inc = np.where(cur < 1.0 - 1e-12)[0]
        dec = np.where(cur > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break

        i_up = inc[np.argmax(neigh[inc])]
        i_dn = dec[np.argmin(neigh[dec])]

        if i_up == i_dn:
            break

        allowed = min(1.0 - cur[i_up], cur[i_dn])
        # step size slowly shrinks with iterations
        frac = it / max_iters
        step_frac = max(0.02, 0.2 * (1.0 - frac))
        delta = allowed * step_frac
        if delta <= 0:
            break

        cur[i_up] += delta
        cur[i_dn] -= delta

        new_score = compute_c5(cur)
        if new_score + 1e-12 < best_score:
            best_score = new_score
            best = cur.copy()
        if new_score + 1e-12 < max_val:
            # keep the improvement
            pass
        else:
            # revert
            cur[i_up] -= delta
            cur[i_dn] += delta

        it += 1
    return best, best_score


def simulated_annealing(start_vec: np.ndarray, time_limit: float):
    """
    Simple Metropolis search that respects the box and sum constraints.
    """
    n = start_vec.size
    cur = start_vec.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    while (time.time() - start) < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = max(1e-4, 0.05 * (1.0 - frac))

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        inc_room = 1.0 - cur[i]
        dec_room = cur[j]
        if inc_room <= 0 or dec_room <= 0:
            continue

        # random transfer, not larger than half of the feasible amount
        delta = np.random.rand() * min(inc_room, dec_room) * 0.5
        cur[i] += delta
        cur[j] -= delta

        new_score = compute_c5(cur)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if np.random.rand() < np.exp((cur_score - new_score) / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            cur[i] -= delta
            cur[j] += delta
    return best, best_score


def random_binary_vector(n: int) -> np.ndarray:
    """Generate a binary vector of length n with exactly n/2 ones (or nearest)."""
    half = n // 2
    vec = np.concatenate([np.ones(half), np.zeros(n - half)])
    np.random.shuffle(vec)
    # if n is odd, put a 0.5 in the remaining slot
    if n % 2:
        vec[-1] = 0.5
    return vec.astype(np.float64)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize the step function h and return (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h from the provided global if possible
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n_pts = h0.size

    h0 = enforce_constraints(h0)

    total_budget = float(budget_s)
    start_time = time.time()

    # --------------------------------------------------------------
    # Phase 1 – deterministic descent on the supplied start
    time_phase1 = total_budget * 0.4
    best_h, best_score = steepest_descent(h0, time_phase1)

    # --------------------------------------------------------------
    # Phase 2 – a few random binary restarts with brief descent
    time_elapsed = time.time() - start_time
    time_left = total_budget - time_elapsed
    # allocate about half of the remaining budget to this phase
    time_phase2 = time_left * 0.5
    phase2_start = time.time()
    while (time.time() - phase2_start) < time_phase2:
        # generate a fresh binary start and improve it quickly
        rnd_start = random_binary_vector(n_pts)
        # give each restart a tiny time slice (max 5 s)
        slice_time = min(5.0, time_phase2 - (time.time() - phase2_start))
        if slice_time <= 0:
            break
        cand_h, cand_score = steepest_descent(rnd_start, slice_time)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing from the current best
    time_elapsed = time.time() - start_time
    time_left = total_budget - time_elapsed
    if time_left > 0.5:
        best_h, best_score = simulated_annealing(best_h, time_left)

    # final safety projection
    best_h = enforce_constraints(best_h)
    best_score = compute_c5(best_h)

    return best_h.tolist(), float(best_score), int(n_pts)
