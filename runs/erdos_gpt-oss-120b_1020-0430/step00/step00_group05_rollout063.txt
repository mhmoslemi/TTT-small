def project_onto_box_sum(v, target):
    # find lambda s.t. sum(clip(v + lam, 0,1)) = target
    # lambda lower bound: -np.max(v) ensures all <= 0 => sum=0
    # lambda upper bound: 1 - np.min(v) ensures all >=1 => sum=n
    lo = -np.max(v) - 1.0  # maybe a bit extra
    hi = 1 - np.min(v) + 1.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        projected = np.clip(v + lam, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def project_onto_box_sum(v, target):
    # binary search for lambda
    lo, hi = -np.max(v) - 1.0, 1.0 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    proj = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # final adjust to ensure sum exactly due to numerical errors:
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly among non-saturated entries
        free = (proj > 0) & (proj < 1)
        cnt = free.sum()
        if cnt > 0:
            proj[free] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_gradient(h, k):
    n = h.shape[0]
    g = np.ones(n)
    if k >= 0:
        g[:n - k] -= h[k:]
        if k > 0:
            g[k:] -= h[:n - k]
    else:
        m = -k
        g[m:] -= h[:n - m]
        if m > 0:
            g[:n - m] -= h[m:]
    return g

def hill_climb(h, max_iter, dx, target_sum, time_limit):
    best_h = h.copy()
    best_c = compute_cmax(best_h, dx)
    n = len(h)
    for iter in range(max_iter):
        # random swap
        ones = np.where(h > 0.5)[0]  # those equal 1 (allow tolerance)
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        # compute new c
        c = compute_cmax(h, dx)
        if c < best_c:
            best_c = c

            # revert swap

def minimize_max_c(h, n_iter, step0, step_decay, dx, target_sum, start_time, budget):
    n = len(h)
    best_h = h.copy()
    best_c = compute_cmax(h, dx)
    step = step0
    for it in range(n_iter):
        # compute correlation array
        conv = np.correlate(h, 1 - h, mode='full')
        C_arr = conv * dx
        cur_c = C_arr.max()
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # find shift of max
        idx = np.argmax(C_arr)
        k = idx - (n - 1)
        # compute gradient
        g = np.ones(n)
        if k >= 0:
            if k < n:
                g[:n - k] -= h[k:]
            if k > 0:
                g[k:] -= h[:n - k]
        else:
            m = -k
            if m < n:
                g[m:] -= h[:n - m]
            if m > 0:
                g[:n - m] -= h[m:]
        # update
        h_new = h - step * g
        # clip and project onto sum
        h_new = np.clip(h_new, 0.0, 1.0)
        h = project_onto_box_sum(h_new, target_sum)
        # step decay
        step *= step_decay
        # time check
        if time.time() - start_time > budget * 0.9:
            break
    return best_h, best_c

        # Interpolate to n points

    # number of restarts
    for restart in range(3):
        # copy initial
        h = h0.copy()
        # random perturbation maybe.

        iter_limit = 20000
        step0 = 0.5
        step_decay = 0.9995

def hill_climb_binary(n, max_iters, time_limit):
    # start with random balanced binary array.
    b = np.zeros(n, dtype=float)
    b[:n//2] = 1.0
    np.random.shuffle(b)
    dx = 2.0/n
    best_b = b.copy()
    best_c = compute_cmax(b, dx)

    start = time.time()
    for it in range(max_iters):
        # random swap
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        b[i], b[j] = b[j], b[i]
        c = compute_cmax(b, dx)
        if c < best_c:
            best_c = c
            best_b = b.copy()
        else:
            # revert with probability maybe 0.1 / (iteration+1)
            # Accept with small probability to escape local minima
            if np.random.rand() < 0.001:
                # keep bad
                pass
            else:
                b[i], b[j] = b[j], b[i]  # revert
        # time check
        if time.time() - start > time_limit * 0.9:
            break
    return best_b, best_c

import time
import numpy as np

def _project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the set {0<=x<=1, sum(x)=target}."""
    # binary search for lambda such that sum(clip(v+lam,0,1)) == target
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(35):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        idx = np.flatnonzero(free)
        if idx.size:
            proj[idx[0]] += diff
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _cmax_and_shift(h: np.ndarray, dx: float):
    """Return (max C value, shift index) for the current h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (len(h) - 1)


def _gradient_step(h: np.ndarray, k: int):
    """Compute the sub‑gradient of C(k) w.r.t. h."""
    n = h.size
    g = np.ones(n)
    if k >= 0:
        if k < n:
            g[: n - k] -= h[k:]
        if k > 0:
            g[k:] -= h[: n - k]
    else:
        m = -k
        if m < n:
            g[m:] -= h[: n - m]
        if m > 0:
            g[: n - m] -= h[m:]
    return g


def _optimize(h0: np.ndarray, dx: float, budget: float, start: float):
    """Simple sub‑gradient descent with projection."""
    n = h0.size
    target = n * 0.5
    h = _project_onto_box_sum(h0.copy(), target)

    best_h = h.copy()
    best_c, _ = _cmax_and_shift(h, dx)

    step = 0.5
    decay = 0.9995
    iteration = 0
    while True:
        # time guard
        if time.time() - start > budget * 0.94:
            break
        c, shift = _cmax_and_shift(h, dx)

        if c < best_c:
            best_c = c
            best_h = h.copy()

        # gradient of the worst shift
        g = _gradient_step(h, shift)
        # update and project
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        h = _project_onto_box_sum(h, target)

        step *= decay
        iteration += 1
        if iteration > 30000:
            break
    return best_h, best_c
