def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    start = time.time()
    rng = np.random.default_rng(seed)
    # Determine n_points
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, copy=False)  # may be list
        n = init.shape[0]
        orig_h = init.astype(float)
    else:
        n = 80
        # initial random feasible vector:
        # start with uniform 0.5 each.
        orig_h = np.full(n, 0.5)
    # Ensure feasibility
    target_sum = n / 2.0
    def repair(vec):
        # clip to [0,1] then iterate scaling and clipping
        v = np.clip(vec, 0, 1)
        for _ in range(10):
            s = v.sum()
            if abs(s - target_sum) < 1e-12:
                break
            factor = target_sum / s
            v = v * factor
            v = np.clip(v, 0, 1)
        return v

    h = repair(orig_h)
    def evaluate(v):
        corr = np.correlate(v, 1 - v, mode='full')
        return np.max(corr) * (2.0 / n)
    best_h = h.copy()
    best_c = evaluate(best_h)
    current_h = best_h.copy()
    current_c = best_c

    # parameters
    max_iter = 2000000  # will break earlier by time
    temp0 = 0.005
    temp_end = 1e-6
    # temperature schedule: linear in iteration: temp = temp0 * (1 - it/ max_iter) + temp_end * (it/ max_iter)
    it = 0
    while True:
        now = time.time()
        if now - start > budget_s - 0.5:
            break
        # annealing schedule
        frac = it / max_iter
        temp = temp0 * (1 - frac) + temp_end * frac
        # propose perturbation
        sigma = 0.1 * (1 - frac) + 0.005 * frac  # start large, then small
        # random Gaussian but only few indices to keep small changes
        # Choose k indices
        k = max(1, int(0.05 * n))  # about 4 indices
        idx = rng.choice(n, size=k, replace=False)
        delta = np.zeros(n)
        delta[idx] = rng.normal(scale=sigma, size=k)
        cand = current_h + delta
        cand = repair(cand)
        cand_c = evaluate(cand)
        if cand_c < current_c or rng.random() < math.exp(-(cand_c - current_c) / (temp + 1e-12)):
            current_h = cand
            current_c = cand_c
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand.copy()
        it += 1
        if it >= max_iter:
            break

    final_c5 = evaluate(best_h)

    best_h = best_h.astype(float)
    # final check
    if abs(best_h.sum() - target_sum) > 1e-6:
        best_h = repair(best_h)
    return (best_h, final_c5, n)

import numpy as np
import time
import math

def repair(vec, target, max_iter=20):
    """Project vec onto the [0,1] box and enforce sum(vec)=target."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        if abs(s - target) < 1e-12:
            break
        if s == 0.0:
            v = np.full_like(v, target / v.size)
            continue
        factor = target / s
        v = v * factor
        v = np.clip(v, 0.0, 1.0)
    return v

def evaluate(v, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx for the sampled vector v."""
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def binary_from_vector(v, num_ones):
    """Return a binary vector with exactly num_ones ones, preserving the largest entries of v."""
    n = v.shape[0]
    idx = np.argpartition(v, -int(num_ones))[-int(num_ones):]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1. Load or construct an initial feasible vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
    else:
        n = 80
        init = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0               # ∫ h = 1  ⇒  sum = n/2
    dx = 2.0 / n                       # sample width

    # Make the initial vector feasible
    best_h = repair(init, target_sum)
    best_c = evaluate(best_h, dx)

    # Also consider the binary projection of the initial vector
    bin_init = binary_from_vector(best_h, target_sum)
    bin_init_c = evaluate(bin_init, dx)
    if bin_init_c < best_c:
        best_h = bin_init
        best_c = bin_init_c

    # ------------------------------------------------------------------
    # 2. Phase 1 – binary simulated annealing
    # ------------------------------------------------------------------
    current_h = bin_init.copy()
    current_c = bin_init_c

    phase1_frac = 0.68                     # share of the budget for phase 1
    phase1_end = start + phase1_frac * budget_s
    temp_start = 1.0
    temp_end = 1e-4

    while time.time() < phase1_end:
        # linear temperature schedule based on elapsed fraction of phase 1
        t_frac = (time.time() - start) / (phase1_frac * budget_s)
        temp = temp_start * (1.0 - t_frac) + temp_end * t_frac

        ones = np.flatnonzero(current_h == 1)
        zeros = np.flatnonzero(current_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        cand = current_h.copy()
        cand[i1] = 0.0
        cand[i0] = 1.0

        cand_c = evaluate(cand, dx)
        delta = cand_c - current_c
        if delta < 0.0 or rng.random() < math.exp(-delta / (temp + 1e-12)):
            current_h = cand
            current_c = cand_c
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand.copy()

        # occasional random restart to keep diversity
        if rng.random() < 0.001:
            rnd = rng.random(n)
            current_h = binary_from_vector(rnd, target_sum)
            current_c = evaluate(current_h, dx)

    # ------------------------------------------------------------------
    # 3. Phase 2 – continuous refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < start + budget_s - 2.0:
        cur = best_h.copy()
        cur_c = best_c
