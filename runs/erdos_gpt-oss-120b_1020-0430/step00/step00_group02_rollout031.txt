def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimizes
       C5 = max_k ∫ h(x)*(1-h(x+k)) dx.
       Returns (h_values, c5_bound, n_points)."""
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_val = float('inf')

        best_h, best_val, best_n = init, float(val0), n0
    except Exception:
        pass

    # Helper to compute the bound for a given vector and its length
    def bound(vec, length):
        return np.max(np.correlate(vec, 1.0 - vec, mode='full')) * (2.0 / length)

    # Candidate sizes (even numbers, larger gives finer granularity)
    candidate_sizes = [60, 100, 200]   # feel free to adjust within budget
    target = 0.3808

    for N in candidate_sizes:

        if N % 2:

        dx = 2.0 / N
        # how many independent restarts at this size
        restarts = 30 if N <= 100 else 20
        max_no_imp = 5000          # stop hill‑climb after this many failures

        for _ in range(restarts):
            # random binary vector with exactly N/2 ones
            h = np.zeros(N, dtype=np.int8)
            h[: N // 2] = 1
            np.random.shuffle(h)

            cur_val = bound(h, N)
            if cur_val < best_val:
