def project_onto_capped_simplex(v, M):
    # returns projection onto [0,1] and sum = M
    # binary search for lambda
    n = len(v)
    lo = -np.max(v) - 1.0  # guarantee w_i = 1 for small lambda
    hi = np.max(v) + 1.0   # guarantee w_i = 0 for large lambda
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > M:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust for possible tiny sum error due to rounding
    # ensure exactly M maybe by distributing small diff
    diff = M - w.sum()
    if abs(diff) > 1e-10:
        # adjust by adding diff equally to non-saturated entries
        # but maintain bounds
        # simple iteration: while diff not zero:
        # choose indices where w_i in (0,1); add small epsilon
        # If none, it's at bounds; distribution may be impossible; but diff should be small.
        indices = np.where((w > 0) & (w < 1))[0]
        if len(indices) > 0:
            # distribute evenly
            w[indices] += diff / len(indices)
            w = np.clip(w, 0.0, 1.0)
    return w

def project_to_bounds_and_sum(v, M):
    # Clip to [0,1]
    w = np.clip(v, 0.0, 1.0)
    # iterative adjustment
    while True:
        s = w.sum()
        diff = M - s
        if abs(diff) < 1e-12:
            break
        # indices where w is not at bound.
        free = (w > 0.0) & (w < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust, break
            break
        delta = diff / n_free
        w[free] += delta
        # clip again
        w = np.clip(w, 0.0, 1.0)
    return w
```

def _project_to_capped_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum }.
    Simple water-filling iteration – fast for the small dimensions we use.
    """
    w = np.clip(v, 0.0, 1.0)
    # iterative redistribution of the remaining deficit / surplus
    for _ in range(100):          # enough iterations for convergence
        s = w.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (w > 0.0) & (w < 1.0)
        n_free = int(free.sum())
        if n_free == 0:          # should not happen for feasible target_sum
            # distribute the tiny remainder uniformly and clip again
            w += diff / w.size
            w = np.clip(w, 0.0, 1.0)
            break
        delta = diff / n_free
        w[free] += delta
        w = np.clip(w, 0.0, 1.0)
    return w


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute the overlap constant C5 for a step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

        init = np.asarray(initial_h_values, dtype=float)  # provided by the harness
        n = init.size
    except Exception:
        init = None
        n = 80                       # fallback size
    dx = 2.0 / n
    target_sum = n / 2.0            # because ∫h = 1  ->  sum(h)*dx = 1

    # ------------------------------------------------------------------
    # Initial feasible point
    # ------------------------------------------------------------------
    if init is not None and init.shape == (n,):
        h0 = _project_to_capped_simplex(init, target_sum)
        best_h = h0.copy()
        best_score = _c5_value(best_h, dx)
    else:
        # start from a random binary vector (half ones, half zeros)
        best_h = _random_binary_vector(rng, n, int(target_sum))
        best_score = _c5_value(best_h, dx)

    # ------------------------------------------------------------------
    # Helper functions for the stochastic search
    # ------------------------------------------------------------------
    # probability of performing a simple two‑coordinate swap move
    SWAP_PROB = 0.35

    # annealing schedule parameters (geometric decay)
    ALPHA_START = 0.60   # weight of a random binary vector at the beginning
    ALPHA_END = 0.001    # weight near the end
    # number of ones in each random binary vector (always integer)
    ONES_IN_BINARY = int(target_sum)

    start_time = time.time()
    elapsed = 0.0

    # Remember the current state (starts from the current best)
    cur_h = best_h.copy()
    cur_score = best_score
    no_improve = 0  # counter to trigger occasional random restarts

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Annealing factor α decreasing from ALPHA_START to ALPHA_END
        fraction = elapsed / budget_s
        alpha = ALPHA_START * (ALPHA_END / ALPHA_START) ** fraction
        temperature = max(alpha, 1e-9)   # used in Metropolis acceptance

        # ------------------------------------------------------------------
        # Generate a candidate point
        # ------------------------------------------------------------------
        if rng.random() < SWAP_PROB:
            # ----- two‑coordinate swap move (preserves sum, stays in [0,1]) -----
            i, j = rng.choice(n, size=2, replace=False)
            # maximum amount we can increase i while decreasing j
            max_inc = min(1.0 - cur_h[i], cur_h[j])
            # maximum amount we can decrease i while increasing j
            max_dec = min(cur_h[i], 1.0 - cur_h[j])
            if max_inc + max_dec == 0:
                continue
            delta = rng.uniform(-max_dec, max_inc) * alpha  # scaled by α
            cand = cur_h.copy()
            cand[i] += delta
            cand[j] -= delta
        else:
            # ----- convex combination with a fresh random binary vector -----
            b = _random_binary_vector(rng, n, ONES_IN_BINARY)
            cand = cur_h * (1.0 - alpha) + b * alpha

        cand_score = _c5_value(cand, dx)

        # ------------------------------------------------------------------
        # Acceptance (Metropolis)
        # ------------------------------------------------------------------
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / temperature):
            cur_h, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # Occasionally restart the local walk to avoid stagnation
        # ------------------------------------------------------------------
        if no_improve > 3000:
            # start a fresh walk from the current global best
            cur_h = best_h.copy()
            cur_score = best_score
            no_improve = 0

    # ----------------------------------------------------------------------
    # Final refinement: small hill‑climbing steps around the best solution
    # ----------------------------------------------------------------------
    remaining = max(0.0, budget_s - (time.time() - start_time))
    # allocate a tiny slice of the remaining budget for polishing
    polish_iters = int(min(200000, remaining * 1500))  # heuristic: ~1500 evals per sec
    for _ in range(polish_iters):
        i, j = rng.choice(n, size=2, replace=False)
        max_inc = min(1.0 - best_h[i], best_h[j])
        max_dec = min(best_h[i], 1.0 - best_h[j])
        if max_inc + max_dec == 0:
            continue
        # tiny steps for fine tuning
        delta = rng.uniform(-max_dec, max_inc) * 0.02
        cand = best_h.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_score = _c5_value(cand, dx)
        if cand_score < best_score:
            best_h, best_score = cand, cand_score

    # Ensure feasibility one final time (tiny numerical drifts may have appeared)
    best_h = _project_to_capped_simplex(best_h, target_sum)

    # Final C5 evaluation after the final projection
    final_score = _c5_value(best_h, dx)

    return best_h, float(final_score), n
