def project_onto_simplex_with_bounds(v, target_sum, lower=0.0, upper=1.0):
    # Initially clip to bounds
    v = np.clip(v, lower, upper)
    # Compute current sum
    cur_sum = v.sum()
    if abs(cur_sum - target_sum) < 1e-10:
        return v
    # Need to adjust values that are not at bounds.
    # We can perform water-filling: let free = mask of (v > lower) & (v < upper)
    # Then we adjust these values by adding λ uniformly.
    # Use while loop, because some may hit bounds after adjustment.
    v = v.astype(float)
    for _ in range(100):
        diff = cur_sum - target_sum
        if abs(diff) < 1e-12:
            break
        free = (v > lower + 1e-12) & (v < upper - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds but sum not reached; adjust by scaling?

            # compute shift to make sum target with clipping

            import cvxpy as cp
            x = cp.Variable(len(v))
            constraints = [x >= lower, x <= upper, cp.sum(x) == target_sum]
            prob = cp.Problem(cp.Minimize(cp.sum_squares(x - v)), constraints)
            prob.solve(solver=cp.ECOS, warm_start=True)

def compute_c5(h):
    # assume h is np.ndarray
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr) * dx
    return max_val

def gradient_step(h, target_sum, step):
    N = len(h)
    # compute correlation to get worst shift
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = np.argmax(corr)
    k = idx_max - (N - 1)
    grad = np.ones(N, dtype=float)
    if k >= 0:
        grad[k:] -= h[:N - k]
        grad[:N - k] -= h[k:]
    else:
        kk = -k
        grad[:N - kk] -= h[kk:]
        grad[kk:] -= h[:N - kk]
    h_new = h - step * grad
    h_new = project_to_simplex_bounds(h_new, target_sum)
    return h_new

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_to_simplex_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust so that sum(v) == target_sum."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target_sum) <= 1e-12:
        return v
    # iterative water‑filling
    for _ in range(100):
        diff = s - target_sum
        if abs(diff) <= 1e-12:
            break
        free = (v > 1e-12) & (v < 1.0 - 1e-12)
        n_free = int(free.sum())
        if n_free == 0:  # fallback to a tiny QP (rare)
            try:
                import cvxpy as cp
                x = cp.Variable(len(v))
                prob = cp.Problem(
                    cp.Minimize(cp.sum_squares(x - v)),
                    [x >= 0, x <= 1, cp.sum(x) == target_sum],
                )
                prob.solve(solver=cp.ECOS, warm_start=True)
                return np.array(x.value).flatten()
            except Exception:
                # as a last resort, distribute uniformly
                v[:] = target_sum / len(v)
                return v
        delta = diff / n_free
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
    return v


def gradient_step(h: np.ndarray, target_sum: float, step: float) -> np.ndarray:
    """Take one projected gradient step that reduces the worst overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)  # shift corresponding to the max overlap
    grad = np.ones(n, dtype=float)
    if k >= 0:
        grad[k:] -= h[: n - k]
        grad[: n - k] -= h[k:]
    else:
        kk = -k
        grad[: n - kk] -= h[kk:]
        grad[kk:] -= h[: n - kk]
    h_new = h - step * grad
    return project_to_simplex_bounds(h_new, target_sum)
