def hill_climb_binary(n, time_limit, start_h=None):
    if start_h is None:
        h = np.zeros(n)
        h[:n//2] = 1
        np.random.shuffle(h)
    else:
        h = start_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    start_time = time.time()
    it = 0
    while time.time() - start_time < time_limit:
        # pick i with h[i]==1, j with h[j]==0
        ones = np.where(h>0.5)[0]
        zeros = np.where(h<0.5)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_swapped = h.copy()
        h_swapped[i] = 0
        h_swapped[j] = 1
        val = compute_c5(h_swapped, n)
        # Accept if better
        if val < best_val - 1e-12:
            best_val = val
            best_h = h_swapped
            h = h_swapped
        else:
            # possibly accept with probability based on temperature schedule
            # compute temperature
            t = max(0.01, 0.1 * (time_limit - (time.time() - start_time)) / time_limit)
            # accept with probability exp(-(val-best_val)/t)
            if np.random.rand() < np.exp(-(val - best_val) / t):
                h = h_swapped
        it += 1
    return best_h, best_val

def optimize_h(n, time_limit):
    # start with random binary feasible h
    h = np.zeros(n)
    # assign 1s to exactly n//2 indices
    indices = np.random.choice(n, n//2, replace=False)
    h[indices] = 1.0

    best_h = h.copy()
    best_val = compute_c5(h, n)

    start = time.time()
    iters = 0

    # SA schedule
    while time.time() - start < time_limit:
        # choose two distinct indices i, j
        i, j = np.random.choice(n, 2, replace=False)
        # choose direction: we will move delta from i to j (i donor)
        if h[i] > 0 and h[j] < 1:
            max_transfer = min(h[i], 1 - h[j])
            delta = max_transfer * np.random.rand() * 0.5  # step factor
            if delta > 0:
                new_h = h.copy()
                new_h[i] -= delta
                new_h[j] += delta
                # maybe also move other small extra?
                val = compute_c5(new_h, n)
                # acceptance
                if val < best_val:
                    best_val, best_h = val, new_h
                    h = new_h
                else:
                    # SA acceptance probability
                    elapsed = time.time() - start
                    remaining = max(0, time_limit - elapsed)
                    # temperature linear from 0.1 to 0.001
                    T = max(0.001, 0.05 * remaining / time_limit)
                    if np.random.rand() < np.exp((best_val - val) / T):
                        h = new_h
        else:
            # cannot transfer; skip iteration
            pass

        iters += 1

    return best_h, best_val

def simulated_annealing_binary_fractional(n, time_limit, init_h=None):
   if init_h is None: generate random binary half ones.
   cur_h = init_h.copy()
   cur_val = compute_c5(cur_h, n)

   best_h = cur_h.copy()
   best_val = cur_val

   start = time.time()
   while time.time() - start < time_limit:
       # choose two distinct indices
       i, j = np.random.choice(n, 2, replace=False)
       # decide direction: which donor?
       if np.random.rand() < 0.5:
           donor, receiver = i, j
       else:
           donor, receiver = j, i
       if cur_h[donor] <= 0 or cur_h[receiver] >= 1:
           continue
       max_transfer = min(cur_h[donor], 1 - cur_h[receiver])
       if max_transfer <= 0:
           continue
       # choose delta as a random fraction times max_transfer
       # step factor decreases with temperature maybe
       delta = max_transfer * np.random.rand() * 0.5   # up to half of possible
       new_h = cur_h.copy()
       new_h[donor] -= delta
       new_h[receiver] += delta
       new_val = compute_c5(new_h, n)
       delta_val = new_val - cur_val
       # temperature schedule
       elapsed = time.time() - start
       frac = 1 - elapsed / time_limit
       T = 0.05 * frac + 0.001  # start 0.05, end 0.001
       if delta_val <= 0 or np.random.rand() < np.exp(-delta_val / T):
           cur_h, cur_val = new_h, new_val
           if new_val < best_val:
               best_val, best_h = new_val, new_h.copy()
   return best_h, best_val

def adjust_sum(h, target):
    current = h.sum()
    diff = target - current
    if abs(diff) < 1e-12:
        return h
    # find index where we can adjust without violating bounds
    if diff > 0:
        # need to increase
        idx = np.where(h < 1 - 1e-12)[0]
        if len(idx) == 0:
            return h
        i = idx[0]
        h[i] += diff
        h[i] = min(1, max(0, h[i]))
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        i = idx[0]
        h[i] += diff
        h[i] = min(1, max(0, h[i]))
    return h

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def adjust_sum(h, target):
    """Shift a single entry so that sum(h) == target while staying in [0,1]."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1 - 1e-12)[0]
        if idx.size:
            i = idx[0]
            h[i] = min(1.0, h[i] + diff)
    else:
        idx = np.where(h > 1e-12)[0]
        if idx.size:
            i = idx[0]
            h[i] = max(0.0, h[i] + diff)
    return h

def simulated_annealing(start_h, time_limit):
    """Simple SA that preserves the sum constraint by moving mass between two entries."""
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best_h, best_val = cur.copy(), cur_val
    start = time.time()
    T0, Tf = 0.08, 0.0005           # temperature schedule
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        i, j = np.random.choice(n, 2, replace=False)
        # decide direction of transfer
        donor, receiver = (i, j) if np.random.rand() < 0.5 else (j, i)
        if cur[donor] <= 1e-12 or cur[receiver] >= 1 - 1e-12:
            continue
        max_transfer = min(cur[donor], 1 - cur[receiver])
        if max_transfer <= 0:
            continue
        # occasional full swap, otherwise a fractional move
        delta = max_transfer if np.random.rand() < 0.2 else max_transfer * np.random.rand() * 0.5
        new = cur.copy()

        new[receiver] += delta
        new_val = compute_c5(new)
        dE = new_val - cur_val
