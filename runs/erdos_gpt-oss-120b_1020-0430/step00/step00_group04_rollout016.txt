def optimize_subgradient(h0, max_iter, time_limit):
    h = project(h0)
    best_h = h.copy()
    best_c = evaluate(h)
    time_start = time.time()
    t = 0
    while t < max_iter and time.time() - time_start < time_limit:
        # compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        k_idx = np.argmax(corr)  # index of max
        s = k_idx - (len(h)-1)    # shift
        # compute gradient
        N = len(h)
        g = np.zeros(N)
        # mask1: i such that i+s < N
        if s >= 0:
            i_range1 = np.arange(N - s)  # indices i where i+s in range
            g[i_range1] += (1 - h[i_range1 + s])
            # mask2: i where i-s >=0 => i >= s
            i_range2 = np.arange(s, N)
            g[i_range2] -= h[i_range2 - s]
        else:
            s_neg = -s
            # s<0 => shift left, equivalently s negative
            # for mask1: i such that i+s >=0 (i >= -s)
            i_range1 = np.arange(-s, N)  # i >= -s
            g[i_range1] += (1 - h[i_range1 + s])  # i+s within bounds
            # mask2: i such that i - s < N => i < N + s
            i_range2 = np.arange(0, N + s)  # since s negative, N+s < N
            g[i_range2] -= h[i_range2 - s]

        g *= (2.0/len(h))

        # evaluate
        c = evaluate_h(h)
        if c < best_c - 1e-9:
            best_c = c; best_h = h.copy()
        t += 1
    return best_h, best_c

def project_to_constraints(v, target):
    # ISO: returns vector clipped to [0,1] and sum = target
    n = len(v)
    x = np.clip(v, 0.0, 1.0)
    # compute difference
    diff = target - np.sum(x)
    # loop
    for _ in range(100):
        if abs(diff) < 1e-9:
            break
        # indices that are not at bounds in direction of diff
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if not np.any(mask):
            # can't adjust
            break
        delta = diff / np.sum(mask)
        # apply delta, but also obey bounds
        x[mask] += delta
        # clip again any overshoot
        over = x > 1.0
        under = x < 0.0
        # correct overshoot
        if np.any(over):
            # move excess back to diff
            excess = np.sum(x[over] - 1.0)
            x[over] = 1.0
            diff = target - np.sum(x)  # recompute diff intentionally for next loop
        elif np.any(under):
            deficit = np.sum(-x[under])
            x[under] = 0.0
            diff = target - np.sum(x)
        else:
            diff = target - np.sum(x)
    # final clip
    x = np.clip(x, 0.0, 1.0)
    # minor residual due to numerical, adjust by distributing tiny residual to non-bound entries if possible
    # final correction: if sum differs more than tolerance, adjust few entries.
    res = target - np.sum(x)

def optimize_subgradient(h0, max_iter, time_limit):
    start = time.time()
    N = len(h0)
    target = N / 2.0
    h = project_to_constraints(h0, target)
    best_h = h.copy()
    best_c = evaluate_h(h)
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        s = k_idx - (N - 1)
        # compute gradient
        g = np.zeros(N)
        if s >= 0:
            g[:N - s] += 1 - h[s:]
            g[s:] -= h[:N - s]
        else:
            t = -s
            g[t:] += 1 - h[:N - t]
            g[:N - t] -= h[t:]
        # multiply by dx
        g *= (2.0 / N)
        # step size schedule
        alpha = 0.4 / np.sqrt(it + 1)
        # update
        h_new = h - alpha * g
        # projection
        h_new = project_to_constraints(h_new, target)
        # evaluate
        c = evaluate_h(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        # optional accept if improves only else continue with h = h_new
        # we can also use simple hill climbing: if improvement else keep old (or maybe accept)
        # Let's update always to h_new (standard gradient descent)
        h = h_new
    return best_h, best_c

import numpy as np, time

def project_to_constraints(v, target):
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    if target <= 0:
        return np.zeros_like(v)
    if target >= n:
        return np.ones_like(v)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_h(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def subgradient_optimize(h0, max_iter, time_limit):
    start = time.time()
    N = h0.shape[0]
    target = N / 2.0
    h = project_to_constraints(h0, target)
    best_h = h.copy()
    best_c = evaluate_h(h)
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # correlation and index of maximal overlap
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_idx = np.argmax(corr)
        s = k_idx - (N - 1)          # signed shift
        # gradient of the active term
        g = np.zeros(N)
        if s >= 0:
            g[:N - s] += 1.0 - h[s:]
            g[s:] -= h[:N - s]
        else:
            t = -s
            g[t:] += 1.0 - h[:N - t]
            g[:N - t] -= h[t:]
        g *= (2.0 / N)               # multiply by dx
        # step size decreasing like 1/sqrt(iter)
        alpha = 0.5 / np.sqrt(it + 1.0)
        h_new = h - alpha * g
        h_new = project_to_constraints(h_new, target)
        c = evaluate_h(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def interpolate_h(old_h, new_n):
    old_n = old_h.shape[0]
    # sample at cell centres
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False) + (2.0/old_n)/2.0
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + (2.0/new_n)/2.0

    target = new_n / 2.0
    new_h = project_to_constraints(new_h, target)
    return new_h

import time
import numpy as np

def _project_to_constraints(v, target):
    """Projection of v onto {0<=x<=1, sum(x)=target}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    if target <= 0:
        return np.zeros_like(v)
    if target >= n:
        return np.ones_like(v)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate_h(h):
    """Return C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _subgradient_optimize(h0, max_iter, time_limit):
    """Projected sub‑gradient descent on max_k ∑h[i](1−h[i+k])."""
    start = time.time()
    n = h0.shape[0]
    target = n / 2.0
    h = _project_to_constraints(h0, target)
    best_h = h.copy()
    best_c = _evaluate_h(h)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        k_idx = int(np.argmax(corr))
        s = k_idx - (n - 1)                     # signed shift giving the max

        g = np.zeros(n)
        if s >= 0:
            g[: n - s] += 1.0 - h[s:]
            g[s:]      -= h[: n - s]
        else:
            t = -s
            g[t:]      += 1.0 - h[: n - t]
            g[: n - t] -= h[t:]
        g *= 2.0 / n                            # multiply by dx

        alpha = 0.5 / np.sqrt(it + 1.0)          # decreasing step size
        h_new = h - alpha * g
        h_new = _project_to_constraints(h_new, target)

        c = _evaluate_h(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c

    """Linear interpolation of a step vector to a finer grid."""
    old_n = old_h.shape[0]
