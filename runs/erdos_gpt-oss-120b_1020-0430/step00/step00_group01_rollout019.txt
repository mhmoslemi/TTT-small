import numpy as np, time, random, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, h, mode='full') * dx
    min_corr = conv.min()
    c5 = 1 - min_corr
    return c5, min_corr

def simulated_annealing(n, time_limit):
    # start with random binary vector
    half = n // 2
    h = np.zeros(n, dtype=np.float64)
    h[:half] = 1
    np.random.shuffle(h)
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5
    t0 = time.time()
    # schedule
    T0 = 0.1
    T_end = 1e-4
    # We will compute T based on elapsed fraction of time
    while time.time() - t0 < time_limit:
        elapsed = (time.time() - t0) / time_limit
        T = T0 * (1 - elapsed) + T_end * elapsed
        # choose random swap
        ones = np.where(current_h > 0.5)[0]
        zeros = np.where(current_h < 0.5)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        new_c5, _ = compute_c5(current_h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / T):

import numpy as np, math, time, random

def compute_c5(h):
    """Return (C5, min_corr) for a step function h."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, h, mode="full")
    min_corr = conv.min() * dx
    return 1.0 - min_corr, min_corr

def project_onto_bounds_sum(v, target_sum):
    """Projection of v onto {x: 0<=x<=1, sum x = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_min_corr(h, shifts):
    """Gradient of the minimal autocorrelation (average over all minimizing shifts)."""
    n = h.size
    grad = np.zeros_like(h)
    for k in shifts:
        if k >= 0:
            grad[: n - k] += h[k:]
            grad[k:] += h[: n - k]
        else:
            kk = -k
            grad[kk:] += h[: n - kk]
            grad[: n - kk] += h[kk:]
    grad /= len(shifts)
    return grad

def optimize_h(h_init, time_limit):
    """Projected gradient ascent on min correlation until time runs out."""
    h = h_init.copy()
    n = h.size
    target_sum = n / 2.0
    dx = 2.0 / n
    conv = np.correlate(h, h, mode="full")
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        min_val = conv.min()
        idxs = np.where(conv <= min_val + 1e-12)[0]
        shifts = idxs - (n - 1)
        grad = gradient_min_corr(h, shifts)

        step = 0.5 / math.sqrt(it)
        improved = False
        cur_c5 = best_c5

        for _ in range(10):
            h_candidate = project_onto_bounds_sum(h + step * grad, target_sum)
            conv_candidate = np.correlate(h_candidate, h_candidate, mode="full")
            c5_candidate = 1.0 - conv_candidate.min() * dx
            if c5_candidate < cur_c5 - 1e-12:
                h = h_candidate
                conv = conv_candidate
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            # occasional random perturbation to escape flat regions
            if random.random() < 0.3:
                pert = np.random.normal(scale=0.03, size=n)
                h_candidate = project_onto_bounds_sum(h + pert, target_sum)
            else:
                i, j = np.random.randint(0, n, 2)
                delta = np.random.uniform(-0.08, 0.08)
                h_candidate = h.copy()
                h_candidate[i] = np.clip(h_candidate[i] + delta, 0.0, 1.0)
                h_candidate[j] = np.clip(h_candidate[j] - delta, 0.0, 1.0)
                h_candidate = project_onto_bounds_sum(h_candidate, target_sum)

            conv_candidate = np.correlate(h_candidate, h_candidate, mode="full")
            c5_candidate = 1.0 - conv_candidate.min() * dx
            if c5_candidate < best_c5 - 1e-12:
                h = h_candidate
                conv = conv_candidate
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_total = time.time()

    # ------------------------------------------------------------------
    # initialise first grid
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.asarray(init, dtype=np.float64)
        n0 = h0.size
    else:
        n0 = 94
        h0 = np.full(n0, 0.5, dtype=np.float64)

    # project onto feasible set
    h0 = project_onto_bounds_sum(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5, _ = compute_c5(best_h)

    # ------------------------------------------------------------------
    # try a sequence of refinements (original size, then doubled sizes)
    sizes = [n0]
    while True:
        remaining = budget_s - (time.time() - start_total)
        if remaining < 30 or sizes[-1] >= 384:
            break
        sizes.append(sizes[-1] * 2)

    for n in sizes:
        remaining = budget_s - (time.time() - start_total)
        if remaining <= 5:
            break
        # time allocated for this size
        t_alloc = remaining * 0.85

        if n == best_h.size:
            h_start = best_h.copy()
        else:
            # linear interpolation to new grid
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_start = np.interp(x_new, x_old, best_h)
            h_start = project_onto_bounds_sum(h_start, n / 2.0)

        h_opt, c5_opt = optimize_h(h_start, t_alloc)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    # ------------------------------------------------------------------
    return best_h.tolist(), best_c5, best_h.size
