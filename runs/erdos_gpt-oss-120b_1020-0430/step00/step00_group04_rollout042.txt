import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _project_to_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the hypercube [0,1]^n
    and the hyperplane sum(x)=target_sum.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """Compute the Erdős C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _initial_sine2(n: int) -> np.ndarray:
    """
    Start from the smooth function h(x) = sin^2(pi*x/2) on [0,2].
    This automatically satisfies the integral constraint in the limit;
    we then project onto the exact feasible polytope.
    """
    xs = np.linspace(0.0, 2.0, n)
    h = np.sin(np.pi * xs / 2.0) ** 2
    target = n * 0.5
    h = _project_to_feasible(h, target)
    return h


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _optimise_sa(seed: int, n: int, budget: float, start_h: np.ndarray) -> tuple:
    """
    Perform a simple simulated‑annealing search.
    Returns the best (h, C5) pair found within the allotted time.
    """
    rng = np.random.RandomState(seed)
    target_sum = n * 0.5
    cur_h = start_h.copy()
    cur_c5 = _c5_of(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    max_iter = 400_000          # generous upper bound; loop will break on time
    max_step = 0.5              # maximal mass transfer per move
    T0 = 0.015                  # initial temperature

    for it in range(max_iter):
        # time guard
        if time.time() - start > budget * 0.98:
            break

        # cooling schedules
        frac = it / max_iter
        step = max_step * (1.0 - frac) + 1e-6
        T = T0 * (1.0 - frac) + 1e-9

        # pick two distinct indices
        i, j = rng.randint(0, n, size=2)
        while i == j:
            j = rng.randint(0, n)

        # random direction of transfer
        if rng.rand() < 0.5:
            src, tgt = i, j
        else:
            src, tgt = j, i

        # compute admissible transfer magnitude
        eps_max = min(step, cur_h[src], 1.0 - cur_h[tgt])
        if eps_max <= 0.0:
            continue
        eps = eps_max * rng.rand()

        # apply move
        cur_h[src] -= eps
        cur_h[tgt] += eps

        new_c5 = _c5_of(cur_h)
        delta = new_c5 - cur_c5

        if delta < 0.0 or rng.rand() < np.exp(-delta / T):
            # accept
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # reject – revert
            cur_h[src] += eps
            cur_h[tgt] -= eps

    # final projection (tiny numerical drift)
    best_h = _project_to_feasible(best_h, target_sum)
    best_c5 = _c5_of(best_h)
    return best_h, best_c5
