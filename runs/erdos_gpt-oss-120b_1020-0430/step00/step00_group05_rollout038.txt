# helpers (evaluate_erdos_solution and maybe initial_h_values are provided externally)

def _project_box_sum(v, target):
    """Project vector v onto the set {0<=x<=1, sum(x)=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _triangular_h(n):
    """Triangular step function with height 1 and base 2 (area = 1)."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx          # mid‑point positions
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    # enforce exact integral constraint
    target = n / 2.0
    return _project_box_sum(h, target)


def _random_h(n, rng):
    """Random continuous vector satisfying the constraints."""
    v = rng.rand(n)
    target = n / 2.0
    return _project_box_sum(v, target)


def _c5_from_h(h):
    """Compute C5 for a given h using the same routine as the evaluator."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx


def _gradient_step(h, shift, step, dx):
    """Perform one projected gradient step for the worst shift."""
    n = h.shape[0]
    g = np.ones_like(h)                     # dO/dh_i before scaling
    if shift > 0:
        g[: n - shift] -= h[shift:]
        g[shift:]      -= h[: n - shift]
    elif shift < 0:
        s = -shift
        g[s:]          -= h[: n - s]
        g[: n - s]     -= h[s:]
    else:                                   # shift == 0
        g -= 2.0 * h
    # gradient step (dx factor not needed for the derivative itself)
    h_new = h - step * g
    # project back onto feasible set
    target = n / 2.0
    h_new = _project_box_sum(h_new, target)
    return h_new
