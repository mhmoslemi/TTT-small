def project_to_simplex_box(v, target):
    # project onto [0,1] and sum = target
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

def weighted_shift_sum_corr(x, w):

import numpy as np
import time
from math import inf

def project_to_simplex_box(v, target, eps=1e-12):
    # Clip to [0,1] after shifting by lambda
    lo = -1.0
    hi = 1.0
    # Reduce search range: you can compute lower/upper bound based on sum
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def weighted_shift_sum_corr(x, w):
    n = x.shape[0]
    # correlation w (len 2n-1) with x (len n)
    corr = np.correlate(w, x, mode='full')
    offset = n - 1
    return corr[offset : offset + n]

def max_overlap(x, dx):
    return np.max(np.correlate(x, 1 - x, mode='full')) * dx

def smooth_weights(c, alpha):
    # c is array of correlation values (already multiplied by dx)
    maxc = c.max()
    exps = np.exp(alpha * (c - maxc))
    w = exps / exps.sum()
    return w

def gradient_step(x, dx, alpha):
    # compute correlation
    c = np.correlate(x, 1 - x, mode='full') * dx
    w = smooth_weights(c, alpha)
    ws_pos = weighted_shift_sum_corr(x, w)
    ws_neg = weighted_shift_sum_corr(x, w[::-1])
    grad = dx * (1.0 - (ws_pos + ws_neg))
    return grad, c.max()

                    vec = f(x_new)

                vec = project_to_simplex_box(vec, target_sum)

                init_vectors.append(vec)
    except Exception:
        pass

    # add several random starts

        v = project_to_simplex_box(v, target_sum)
        init_vectors.append(v)

    # main optimization loop over initiations
    max_iters_per = 5000
    for init in init_vectors:
        if time.time() - start > budget_s * 0.95:

        x = init.copy()
        alpha = 1.0
        alpha_max = 80.0

import time
import numpy as np

def _project_to_sum_box(v: np.ndarray, target: float, eps: float = 1e-12) -> np.ndarray:
    """Project v onto the set {0<=x_i<=1, sum x_i = target}."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _weighted_shift_sum(x: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute for each index j:  sum_{L} w_L * x_{j+L},
    where w is indexed by shift L = -(n-1)..(n-1) (len(w)=2*n-1).
    """
    n = x.shape[0]
    corr = np.correlate(w, x, mode='full')
    off = n - 1
    return corr[off:off + n]


def _smooth_weights(c_raw: np.ndarray, alpha: float) -> np.ndarray:
    """Softmax over correlation values (unscaled)."""
    m = c_raw.max()
    expv = np.exp(alpha * (c_raw - m))
    return expv / expv.sum()


def _gradient_and_max(x: np.ndarray, dx: float, alpha: float):
    """Return gradient of smoothed max and the current max overlap (scaled by dx)."""
    c_raw = np.correlate(x, 1.0 - x, mode='full')
    w = _smooth_weights(c_raw, alpha)
    ws_pos = _weighted_shift_sum(x, w)
    ws_neg = _weighted_shift_sum(x, w[::-1])
    grad = dx * (1.0 - (ws_pos + ws_neg))
    max_val = c_raw.max() * dx
    return grad, max_val


def _max_overlap(x: np.ndarray, dx: float) -> float:
    """Exact max overlap for a given h."""
    return np.max(np.correlate(x, 1.0 - x, mode='full')) * dx


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # resolution (you can increase it if time permits)
    n = 400
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = None
    best_val = np.inf
    target_goal = 0.3808  # we stop if we beat this

    # ----- build list of starting vectors ---------------------------------

    # parent solution, if supplied

                new_x = np.linspace(0, 2, n, endpoint=False)

                vec = f(new_x)
            else:

            vec = _project_to_sum_box(vec, target_sum)

    except Exception:

    # a few random starts
    for _ in range(8):
