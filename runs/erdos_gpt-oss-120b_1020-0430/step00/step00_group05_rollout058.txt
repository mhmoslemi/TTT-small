def project_onto_simplex_box(v, target):
    # Solve min ||x - v||^2 s.t. 0 <= x <= 1, sum x = target.
    # Use bisection on lambda.
    lo = -1.0
    hi = 1.0
    # Actually we need to find lambda such that sum clip(v + lambda, 0, 1) = target.
    # v may be any real values.
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def project_to_box_sum(v, target):
    # Clip solution to satisfy 0 ≤ x ≤ 1 and sum = target.
    # Use bisection on lambda as described.
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # Ensure sum close to target; adjust small diff by adding to interior values
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # add diff equally to all interior elements (not at bounds)
        interior = (x > 1e-12) & (x < 1-1e-12)
        cnt = interior.sum()
        if cnt > 0:
            x[interior] += diff / cnt
            np.clip(x, 0.0, 1.0, out=x)
        else:
            # fallback: add diff to all and clip
            x += diff / len(x)
            np.clip(x, 0.0, 1.0, out=x)
    return x

import numpy as np, time, math, cvxpy as cp

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def gradient_shift(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        idx = np.arange(0, n - shift)
    else:
        idx = np.arange(-shift, n)
    g[idx] += 1
    g[idx] -= h[idx + shift]
    g[idx + shift] -= h[idx]
    return g

def project_box_sum(v, target):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # adjust residual rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        interior = (x > 1e-12) & (x < 1 - 1e-12)
        cnt = interior.sum()
        if cnt > 0:
            x[interior] += diff / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x

def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)
    start = time.time()
    # parameters
    n = 200  # number of samples (even)
    target_sum = n / 2.0
    # initial guess: triangular shape
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.minimum(xs, 2.0 - xs)  # peak at x=1
    # project to satisfy exact sum constraint
    h = project_box_sum(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    # temperature and step parameters
    temperature = 0.02
    min_temp = 1e-6
    decay_temp = 0.9995
    alpha = 0.05
    min_alpha = 1e-4
    decay_alpha = 0.9995
    it = 0
    no_improve = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        # compute correlation and worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = int(np.argmax(corr))
        max_corr = corr[idx_max]
        cur_c5 = max_corr * (2.0 / n)
        # update best
        if cur_c5 < best_c5 - 1e-9:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # get shift value
        shift = idx_max - (n - 1)

        g = gradient_shift(h, shift)
        # center gradient to keep sum unchanged
        g -= g.mean()

        h_new = project_box_sum(h_new, target_sum)
        # evaluate new c5
        new_corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(new_corr) * (2.0 / n)
        # acceptance
        accept = False
        if new_c5 < cur_c5:
            accept = True

import time, math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value (max correlation * dx) for a step vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst‐lag correlation w.r.t. h (zero‑mean not applied)."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        idx = np.arange(0, n - shift, dtype=int)
    else:  # negative shift
        idx = np.arange(-shift, n, dtype=int)
    # linear part
    g[idx] += 1.0
    # quadratic part (two symmetric contributions)
    g[idx] -= h[idx + shift]
    g[idx + shift] -= h[idx]
    return g


def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n with exact sum = target (using a simple bisection)."""
    lo, hi = -np.max(v) - 1.0, 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = interior.sum()
        if cnt:
            x[interior] += diff / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h minimizing the Erdős C5 bound."""
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    try:
        # use the supplied parent construction if it exists
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        if n % 2:
            n -= 1
            init = init[:n]
        target = n * 0.5
        h = _project_box_sum(init, target)
    except NameError:
        # default: a symmetric triangular profile (peak at x=1)
        n = 200                     # even length → exact sum = n/2 after projection
        target = n * 0.5
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.minimum(xs, 2.0 - xs)       # 0 → 1 → 0 shape
        h = _project_box_sum(h, target)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    temperature = 0.02
    min_temp = 1e-6
    decay_temp = 0.9995

    alpha = 0.05                # nominal step size (will be normalised)
    min_alpha = 1e-4
    decay_alpha = 0.9995

    no_improve = 0
    it = 0
    target_sum = n * 0.5

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break

        # current correlation and worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        cur_c5 = float(corr[idx_max] * (2.0 / n))
        shift = idx_max - (n - 1)                     # signed lag

        # keep the best ever seen
        if cur_c5 < best_c5 - 1e-9:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # gradient of the worst‑lag term
        g = _gradient_for_shift(h, shift)
        g -= g.mean()                                 # keep the sum unchanged

        norm = np.linalg.norm(g)
        if norm > 0:
            step = alpha * g / norm
        else:
            step = np.zeros_like(h)

        # propose a new vector and enforce constraints
        h_prop = _project_box_sum(h - step, target_sum)

        # evaluate proposal
        prop_c5 = _c5_value(h_prop)

        # Metropolis acceptance
        accept = False
        if prop_c5 < cur_c5:
            accept = True
        else:
            if temperature > 0:
                prob = math.exp(-(prop_c5 - cur_c5) / temperature)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            h = h_prop

        # schedule updates
        it += 1
        temperature = max(min_temp, temperature * decay_temp)
        alpha = max(min_alpha, alpha * decay_alpha)

        # occasional random perturbation if we stall
        if no_improve > 2000:
            pert = np.random.randn(n) * 0.02
            h = _project_box_sum(best_h + pert, target_sum)
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best solution found
    # ------------------------------------------------------------------
    final_c5 = _c5_value(best_h)
    return best_h.tolist(), float(final_c5), n
