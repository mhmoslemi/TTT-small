    # constants
    np.random.seed(seed)
    # Determine n
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.shape[0]
    except NameError:
        n = 45
        init_h = None
    dx = 2.0 / n
    target = n / 2.0

    def c5_val(h):
        # compute objective
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def random_feasible():
        # binary pattern with exact sum
        h = np.zeros(n, dtype=float)
        base = int(math.floor(target))
        frac = target - base  # 0 or 0.5 for odd n
        indices = np.arange(n)
        np.random.shuffle(indices)
        h[indices[:base]] = 1.0
        if frac > 0:
            h[indices[base]] = frac
        return h

    # start point
    if init_h is not None:
        cur_h = np.clip(init_h.copy(), 0.0, 1.0)
        # adjust sum precisely:
        s = np.sum(cur_h)
        diff = target - s
        if abs(diff) > 1e-12:

    else:
        cur_h = random_feasible()
        cur_val = c5_val(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    start_time = time.time()

import time
import math
import numpy as np

def _c5_value(h, dx):
    """Return C5 for a given step function h."""
    # Correlation of h with (1-h); np.correlate implements cross‑correlation.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _max_shift_and_corr(h):
    """Return (shift, raw_corr_max) where shift is the lag producing the maximal correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (len(h) - 1)          # map index to signed lag
    return shift, max_val


def _random_feasible(n, target):
    """
    Produce a feasible vector with entries in [0,1] and exact sum == target.
    For odd target we use a single 0.5 entry, otherwise a balanced binary vector.
    """
    base = int(math.floor(target))
    frac = target - base                # 0 or 0.5 for integer n
    h = np.zeros(n, dtype=float)
    idx = np.arange(n)
    np.random.shuffle(idx)
    h[idx[:base]] = 1.0
    if frac > 0:
        h[idx[base]] = frac
    return h


def _local_greedy(h, dx, max_iter=3000, step=0.1):
    """
    A deterministic sweep that moves mass from positions contributing most
    to the worst overlap towards positions that reduce it.
    """
    n = len(h)
    target = n / 2.0
    for _ in range(max_iter):
        shift, _ = _max_shift_and_corr(h)
        if shift == 0:
            # improve zero shift by moving from the smallest to the largest entry
            i_small = int(np.argmin(h))
            i_big = int(np.argmax(h))
            if h[i_small] <= 0 or h[i_big] >= 1:
                break
            delta = min(step, h[i_small], 1.0 - h[i_big])
            if delta <= 0:
                break
            h[i_small] -= delta
            h[i_big] += delta
            continue

        # indices where both i and i+shift lie inside the array
        if shift > 0:
            i_valid = np.arange(n - shift)
        else:
            i_valid = np.arange(-shift, n)
        if i_valid.size == 0:
            break

        hi = h[i_valid]
        hj = h[i_valid + shift]

        # Desired move: decrease hi (where hi>0) and increase hj (where hj<1)
        mask_forward = (hi > 1e-12) & (hj < 1.0 - 1e-12)
        if np.any(mask_forward):
            contrib = hi * (1.0 - hj)
            idx = int(np.argmax(contrib * mask_forward))
            i = i_valid[idx]
            j = i + shift
            delta = min(step, h[i], 1.0 - h[j])
            if delta > 0:
                h[i] -= delta
                h[j] += delta
                continue

        # If forward move impossible, try the opposite direction
        mask_back = (hi < 1.0 - 1e-12) & (hj > 1e-12)
        if np.any(mask_back):
            contrib = hi * (1.0 - hj)
            idx = int(np.argmax(contrib * mask_back))
            i = i_valid[idx]
            j = i + shift
            delta = min(step, h[j], 1.0 - h[i])
            if delta > 0:
                h[j] -= delta
                h[i] += delta
                continue

        # No admissible move for this shift
        break
    # final projection onto the hyperplane (tiny numerical drift)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for k in range(n):
            if 0.0 <= h[k] + diff <= 1.0:
                h[k] += diff
                break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:

    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # Helper to evaluate C5 quickly
    # ------------------------------------------------------------------
    def evaluate(h):
        return _c5_value(h, dx)

    # ------------------------------------------------------------------
    # Initialise the current solution
    # ------------------------------------------------------------------
    if init_h is not None:
        cur_h = np.clip(init_h.copy(), 0.0, 1.0)
        # project onto the exact hyperplane if numerical drift occurs
        diff = target_sum - cur_h.sum()
        if abs(diff) > 1e-12:
            # adjust the first element that can absorb the diff without breaking bounds
            for k in range(n):
                if 0.0 <= cur_h[k] + diff <= 1.0:
                    cur_h[k] += diff
                    break
        cur_val = evaluate(cur_h)
    else:
        cur_h = _random_feasible(n, target_sum)
        cur_val = evaluate(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    # ------------------------------------------------------------------
    # Simulated annealing core
    # ------------------------------------------------------------------
    total_budget = budget_s * 0.95          # leave a margin for polishing
    max_no_improve = 20000
    no_improve = 0
    iteration = 0

    while time.time() - start_time < total_budget:
        # linear temperature schedule (warm → cold)
        elapsed = time.time() - start_time
        progress = elapsed / total_budget
        T = 0.02 * (1.0 - progress) + 1e-5 * progress   # temperature in [0.02,1e-5]

        # propose a transfer of mass from i to j (preserving the integral)
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        max_delta = min(0.5, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 1e-12:
            continue
        delta = np.random.rand() * max_delta

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_val = evaluate(new_h)
        diff = new_val - cur_val
        if diff < 0 or math.exp(-diff / max(T, 1e-12)) > np.random.rand():
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_h = new_val, new_h.copy()
        else:
            no_improve += 1

        iteration += 1
        if no_improve > max_no_improve:
            # restart from a fresh random feasible configuration
            cur_h = _random_feasible(n, target_sum)
            cur_val = evaluate(cur_h)
            no_improve = 0

    # ------------------------------------------------------------------
    # Deterministic polishing using a greedy local descent
    # ------------------------------------------------------------------
    improved = True
    while improved and time.time() - start_time < budget_s * 0.97:

        else:

    # final safeguard: enforce exact sum (tiny numerical error)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for k in range(n):
            if 0.0 <= best_h[k] + diff <= 1.0:
                best_h[k] += diff
                break

    return best_h, float(best_val), n
