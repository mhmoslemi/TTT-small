def subgradient_optimize(n, init_h, max_iters=2000, time_limit=None):
    target = n/2
    h = init_h.copy()
    best_h = h.copy()
    best_c = np.inf
    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break
        c_vec = compute_c_vec(h)
        max_c = c_vec.max()
        if max_c < best_c:
            best_c = max_c
            best_h = h.copy()
        # subgradient for max shift

            lo = lam

            hi = lam
    return np.clip(v - (lo + hi)/2.0, 0.0, 1.0)

def project_capped_simplex(v, target):
    lower, upper = 0.0, 1.0
    # Bisection as described
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, lower, upper)
```

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper: projection onto the capped simplex {0<=x_i<=1, sum x_i = target}
def project_capped_simplex(v, target):
    """Project v onto 0<=x<=1 and sum(x)=target (target in [0, len(v)])"""
    lo = -np.max(v) - 1.0          # enough negative to push all entries to 1
    hi = np.max(v) + 1.0           # enough positive to push all entries to 0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

# ------------------------------------------------------------
def compute_corr_max(h):
    """Maximum of np.correlate(h,1-h,'full')  (unscaled, integer shift)"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max()

def evaluate_c5(h):
    """C5 = max correlation * dx, dx = 2/n"""
    n = h.size
    return compute_corr_max(h) * (2.0 / n)

# ------------------------------------------------------------
def local_optimize(h_start, max_iter=300):
    """SLSQP optimisation of max_{k} ∫h(1-h(·+k))  with equality Σh=n/2."""
    n = h_start.size
    target = n / 2.0

    # initial t is the current max correlation (unscaled)
    t0 = compute_corr_max(h_start)

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        sum_h = h.sum()
        # compute vector c_s = sum_h - dot_s for all non‑negative shifts
        c = np.empty(n, dtype=np.float64)
        # shift 0
        dot0 = np.dot(h, h)
        c[0] = sum_h - dot0
        # positive shifts
        for s in range(1, n):
            dot = np.dot(h[: n - s], h[s :])
            c[s] = sum_h - dot
        return t - c                       # t >= each c_s

    constraints = [
        {"type": "eq", "fun": lambda x: np.sum(x[:-1]) - target},
        {"type": "ineq", "fun": ineq_fun},
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    x0 = np.concatenate([h_start, [t0]])

    res = minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        # fallback: return the starting point unchanged
        return h_start, evaluate_c5(h_start)

    h_opt = res.x[:-1]
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = project_capped_simplex(h_opt, target)
    return h_opt, evaluate_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Initial construction (use provided or a smooth tent‑like shape)
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 200                     # a modest grid size
        x = (np.arange(n) + 0.5) * (2.0 / n)          # mid‑points on [0,2]
        # scaled sine that vanishes at the ends, integral = 1
        h0 = (np.pi / 4.0) * np.sin(np.pi * x / 2.0)
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        cur = h0.sum()
        if abs(cur - target) > 1e-12:
            h0 = h0 * (target / cur)
        h0 = project_capped_simplex(h0, target)

    # ------------------------------------------------------------------
    # 2. Simple 1‑parameter family (tent‑like) search to get a good start
    #    h(x) = a + b*|x-1|,   with 2a + b = 1  (integral condition)
    #    vary a∈[0,0.5]  (b = 1‑2a)  – this includes the pure tent at a=1,b=-1
    n = h0.size
    target = n / 2.0
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # grid over a (centre value). The pure tent corresponds to a=1,b=-1,
    # which is outside the [0,0.5] range because of the integral fix.
    # We'll also test the tent explicitly.
    alphas = np.linspace(0.0, 0.5, 81)
    for a in alphas:
        b = 1.0 - 2.0 * a
        h = a + b * np.abs(xs - 1.0)
        h = np.clip(h, 0.0, 1.0)
        # the integral of this family is exactly 1 (up to discretisation)
        cur = h.sum()
        if abs(cur - target) > 1e-9:
            h = h * (target / cur)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # explicit tent shape (peak 1, zero at edges)
    tent = np.abs(xs - 1.0)
    tent = 1.0 - tent
    tent = np.clip(tent, 0.0, 1.0)
    cur = tent.sum()
    if abs(cur - target) > 1e-9:
        tent = tent * (target / cur)
    c5_tent = evaluate_c5(tent)
    if c5_tent < best_c5:
        best_c5 = c5_tent
        best_h = tent.copy()

    time_left = budget_s - (time.time() - start_time)
    if time_left > 15.0:
        h_opt, c5_opt = local_optimize(best_h, max_iter=400)
        if c5_opt < best_c5:

        rand = project_capped_simplex(rand, target)
        h_opt, c5_opt = local_optimize(rand, max_iter=200)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt

    # ------------------------------------------------------------------
    # 5. Increase resolution if time permits (double grid size once)
    if budget_s - (time.time() - start_time) > 30.0 and n < 400:
        n2 = n * 2
        dx2 = 2.0 / n2
        x_old = (np.arange(n) + 0.5) * dx
        x_new = (np.arange(n2) + 0.5) * dx2
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = np.clip(h_interp, 0.0, 1.0)
        target2 = n2 / 2.0
        h_interp = project_capped_simplex(h_interp, target2)

        h_opt, c5_opt = local_optimize(h_interp, max_iter=300)

            n = n2
            dx = dx2

    # ------------------------------------------------------------------
    # Final sanity projection and evaluation
    final_n = best_h.size
    best_h = project_capped_simplex(best_h, final_n / 2.0)
    final_c5 = evaluate_c5(best_h)

    return best_h, final_c5, final_n
