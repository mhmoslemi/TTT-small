import numpy as np
import scipy.optimize as opt

def compute_overlap(h, dx):
    # full cross-correlation of h with (1-h)
    return np.correlate(h, 1 - h, mode='full') * dx

def objective_t(x):
    return x[-1]

def eq_sum_constraint(x):
    n = len(x) - 1
    target = n / 2.0
    return np.sum(x[:n]) - target

def ineq_constraints(x):
    n = len(x) - 1
    dx = 2.0 / n
    h = x[:n]
    t = x[n]
    O = np.correlate(h, 1 - h, mode='full') * dx
    return t - O

def optimize_at_n(n, h_initial, rng, max_iter=300, restarts=5):
    # interpolate initial to required size
    if len(h_initial) != n:
        x_old = np.linspace(0.0, 2.0, len(h_initial), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, h_initial)
    else:
        h0 = h_initial.copy()
    h0 = np.clip(h0, 0.0, 1.0)
    best_h = None
    best_val = float('inf')
    for r in range(restarts):
        if r == 0:
            h_start = h0.copy()
        else:
            # add random perturbation
            noise = rng.normal(scale=0.2, size=n)
            h_start = np.clip(h0 + noise, 0.0, 1.0)
        t_start = np.max(compute_overlap(h_start, 2.0 / n))
        x0 = np.concatenate([h_start, [t_start]])
        bounds = [(0.0, 1.0)] * (n + 1)
        constraints = [
            {'type': 'eq', 'fun': eq_sum_constraint},
            {'type': 'ineq', 'fun': ineq_constraints}
        ]
        res = opt.minimize(
            objective_t,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
        )
        if not res.success:
            continue
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        val = np.max(compute_overlap(h_opt, 2.0 / n))
        if val < best_val:
            best_val = val
            best_h = h_opt
    if best_h is None:
        best_h = h0
        best_val = np.max(compute_overlap(best_h, 2.0 / n))
    return best_h, best_val

    best_c5 = float('inf')
    best_n = None
    # sequence of grid sizes to explore
    ns = [len(base_h), len(base_h) * 2, len(base_h) * 4]
    prev_h = base_h
    for n in ns:
        if time.time() - start > budget_s * 0.9:

        h_opt, c5_val = optimize_at_n(n, prev_h, rng, max_iter=250, restarts=6)
        if c5_val < best_c5:

        prev_h = h_opt
    if best_h is None:
        best_h = base_h
        best_c5 = np.max(compute_overlap(best_h, 2.0 / len(base_h)))
        best_n = len(base_h)
    # final evaluation using provided helper if available

import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Core helpers – top‑level, no closures or lambdas
# ----------------------------------------------------------------------
def compute_overlap(h, dx):
    """Full correlation of h with (1‑h) multiplied by dx."""
    return np.correlate(h, 1 - h, mode="full") * dx


def objective_t(x):
    """Objective for the epigraph formulation – minimise the auxiliary variable t."""
    return x[-1]


def eq_sum_constraint(x):
    """Equality: Σ h_i = n/2 (i.e. integral of h equals 1)."""
    n = len(x) - 1
    target = n / 2.0
    return np.sum(x[:n]) - target


def ineq_constraints(x):
    """Inequalities: t - O_k ≥ 0 for every shift k."""
    n = len(x) - 1
    dx = 2.0 / n
    h = x[:n]
    t = x[n]
    O = np.correlate(h, 1 - h, mode="full") * dx
    return t - O


def optimize_at_n(n, h_initial, rng, max_iter=250, restarts=6):
    """
    Run a short SLSQP optimisation for a grid of size n,
    starting from h_initial (interpolated if necessary).
    Returns the best feasible h and its C5 value.
    """
    # Interpolate the initial vector to the required size
    if len(h_initial) != n:
        old_x = np.linspace(0.0, 2.0, len(h_initial), endpoint=False)
        new_x = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(new_x, old_x, h_initial)
    else:
        h0 = h_initial.copy()
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = None
    best_val = float("inf")
    dx = 2.0 / n

    for r in range(restarts):
        # first start is the interpolated vector itself, later ones are noisy
        if r == 0:
            h_start = h0.copy()
        else:
            noise = rng.normal(scale=0.15, size=n)
            h_start = np.clip(h0 + noise, 0.0, 1.0)

        t_start = np.max(compute_overlap(h_start, dx))
        x0 = np.concatenate([h_start, [t_start]])

        bounds = [(0.0, 1.0)] * (n + 1)
        constraints = [
            {"type": "eq", "fun": eq_sum_constraint},
            {"type": "ineq", "fun": ineq_constraints},
        ]

        res = opt.minimize(
            objective_t,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            continue

        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        val = np.max(compute_overlap(h_opt, dx))
        if val < best_val:
            best_val = val
            best_h = h_opt

    # fallback in the unlikely case that all runs failed
    if best_h is None:
        best_h = h0
        best_val = np.max(compute_overlap(best_h, dx))

    return best_h, best_val


def hill_climb(h, dx, rng, time_limit):
    """
    Simple pairwise hill‑climbing that preserves the integral.
    Returns an improved h (or the original if no improvement found).
    """
    import time

    n = len(h)
    best_h = h.copy()
    best_val = np.max(compute_overlap(best_h, dx))
    start = time.time()
    step_scale = 0.5  # controls typical move size

    # main loop – stops when time limit is exhausted
    while time.time() - start < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]

        # maximal admissible delta so that both coordinates stay in [0,1]
        max_pos = min(1.0 - hi, hj)   # delta > 0: i increases, j decreases
        max_neg = min(hi, 1.0 - hj)   # delta < 0: i decreases, j increases
        if max_pos == 0 and max_neg == 0:
            continue

        # draw a delta uniformly from the admissible interval, then shrink it
        delta = rng.uniform(-max_neg, max_pos) * step_scale
        if delta == 0.0:
            continue

        new_h = best_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        # evaluate the new candidate
        val = np.max(compute_overlap(new_h, dx))
        if val < best_val - 1e-9:                     # strict improvement
            best_h = new_h
            best_val = val
            # occasionally boost the step size after a good move
            step_scale = min(1.0, step_scale * 1.02)
        else:
            # slowly damp the step size if we are not improving
            step_scale *= 0.999
            if step_scale < 0.001:
                step_scale = 0.001

    return best_h, best_val
