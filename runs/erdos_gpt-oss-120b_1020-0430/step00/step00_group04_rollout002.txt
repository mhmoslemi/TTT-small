def project_onto_constraints(y, target):
    lo = -1.0  # enough low
    hi = 2.0   # enough high; ensures outputs clamped.
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0.0, 1.0)

def project_simplex_box(y, target):
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0.0, 1.0)

import numpy as np, time, math, random

def _project_box_sum(v, target):
    # projection onto [0,1] box and sum=target
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _c5_value(h, dx):
    n = h.shape[0]
    omh = 1.0 - h
    max_val = 0.0
    for k in range(n):
        val = np.dot(h[:n - k], omh[k:]) * dx
        if val > max_val:
            max_val = val
    return max_val

def _smooth_max_and_grad(h, beta, dx):
    n = h.shape[0]
    # correlation R_k
    R = np.empty(n)
    for k in range(n):
        R[k] = np.dot(h[:n - k], h[k:]) * dx
    f = 1.0 - R
    # stable log-sum-exp
    m = np.max(f)
    expb = np.exp(beta * (f - m))
    sum_exp = np.sum(expb)
    smooth = m + np.log(sum_exp) / beta
    w = expb / sum_exp  # normalized weights
    grad = np.zeros_like(h)
    for k in range(n):
        wk = w[k]
        if wk == 0.0:
            continue
        # i where i+k < n  (first term)
        grad[:n - k] -= dx * wk * h[k:]
        # i where i-k >= 0 (second term)
        grad[k:] -= dx * wk * h[:n - k]
    return smooth, grad

import time, math, random
import numpy as np

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {0≤x≤1, sum(x)=target}."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Exact C₅ for a discrete step function h."""
    n = h.shape[0]
    omh = 1.0 - h
    best = 0.0
    for k in range(n):
        val = np.dot(h[: n - k], omh[k:]) * dx
        if val > best:
            best = val
    return best


def _smooth_max_and_grad(h: np.ndarray, beta: float, dx: float):
    """Smooth approximation of maxₖ(1‑ρₖ) and its gradient."""
    n = h.shape[0]
    # autocorrelation ρₖ = Σ h_i h_{i+k} dx
    rho = np.empty(n)
    for k in range(n):
        rho[k] = np.dot(h[: n - k], h[k:]) * dx
    f = 1.0 - rho                     # fₖ = 1‑ρₖ
    m = np.max(f)                     # stability shift
    expb = np.exp(beta * (f - m))
    Z = expb.sum()
    smooth = m + np.log(Z) / beta      # log‑sum‑exp /β
    w = expb / Z                      # normalized weights, Σ w = 1

    grad = np.zeros_like(h)
    for k in range(n):
        wk = w[k]
        if wk == 0.0:
            continue
        # contribution h_{i+k}
        grad[: n - k] -= dx * wk * h[k:]
        # contribution h_{i‑k}
        grad[k:] -= dx * wk * h[: n - k]
    return smooth, grad
