def compute_corr(h):
    return np.correlate(h, h, mode='full')
def evaluate(h):
    corr = compute_corr(h)
    sum_h = h.sum()
    min_corr = corr.min()
    dx = 2.0 / len(h)
    c5 = (sum_h - min_corr) * dx
    return c5, min_corr, corr

def greedy_optimize(h, time_budget):
    start = time.time()
    n = len(h)
    target_sum = n / 2
    best_h = h.copy()
    best_c5, _, _ = evaluate(h)
    h_current = h.copy()
    last_improve_iter = 0
    iter_count = 0
    while time.time() - start < time_budget:
        corr = np.correlate(h_current, h_current, mode='full')
        min_corr = corr.min()
        lag = np.argmin(corr) - (n - 1)
        # compute gradient g
        g = np.zeros(n)
        # vectorized computation: shift arrays
        if lag != 0:
            # shift forward
            shift = lag
            if shift > 0:
                # g[i] = (h[i+shift] if i+shift<n else 0) + (h[i-shift] if i-shift>=0 else 0)
                g[:n-shift] += h_current[shift:]
                g[shift:] += h_current[:n-shift]
            else:
                # negative shift
                shift = -shift
                # same as positive due to symmetry
                g[:n-shift] += h_current[shift:]
                g[shift:] += h_current[:n-shift]
        else:
            g = 2 * h_current
        # define candidates
        up_ids = np.where(h_current < 1 - 1e-12)[0]
        down_ids = np.where(h_current > 1e-12)[0]
        if len(up_ids) == 0 or len(down_ids) == 0:
            break
        # choose i_up with max g among up ids
        i_up = up_ids[np.argmax(g[up_ids])]
        i_down = down_ids[np.argmin(g[down_ids])]
        # ensure distinguish
        if i_up == i_down:
            # choose second best/second worst
            # sort up_ids by g descending
            sorted_up = up_ids[np.argsort(-g[up_ids])]
            for cand in sorted_up:
                if cand != i_down:
                    i_up = cand
                    break
            sorted_down = down_ids[np.argsort(g[down_ids])]
            for cand in sorted_down:
                if cand != i_up:
                    i_down = cand

Algorithm:

- Input v.

Because the problem is convex and the function is monotonic in λ, we can solve via bisection.

def project_to_capped_simplex(v, target):
    # v: array
    # target must be within [0, n] where n = len(v) (since max each is 1)
    # Use bisection for lambda.
    n = len(v)
    # lower bound: lambda_min such that all entries after shift are <=0 => sum=0. If v_i + λ <= 0 for all i => λ <= -max(v)
    # upper bound: λ such that all entries after shift >=1 => v_i + λ >= 1 for all i => λ >= 1 - min(v)
    lam_low = -np.max(v) - 1  # ensure all zero after clip
    lam_high = 1 - np.min(v) + 1  # ensure all one after clip
    # But we can set lam_low = -np.max(v) - 1, lam_high = 1 - np.min(v) + 1
    # Actually we need bounds such that sum at low is 0, sum at high is n.
    # We'll set lam_low = -np.max(v) - 2, lam_high = 1 - np.min(v) + 2.
    for _ in range(100):
        lam_mid = (lam_low + lam_high) / 2.0
        proj = np.clip(v + lam_mid, 0, 1)
        s = proj.sum()
        if abs(s - target) < 1e-12:
            return proj
        if s < target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v + lam_mid, 0, 1)
```

    final_c5 = ((best_h.sum() - np.correlate(best_h, best_h, mode='full').min()) * (2.0 / n))
    # Return result
    return (best_h.tolist(), float(final_c5), n)

import time, random, numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Return C5, min correlation, correlation array."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full')
    min_corr = corr.min()
    c5 = (h.sum() - min_corr) * dx
    return float(c5), float(min_corr), corr

def _proj_capped_simplex(v, target):
    """Project v onto {0<=x<=1, sum x = target}."""
    lo, hi = -np.max(v) - 2.0, 1.0 - np.min(v) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < 1e-12:
            return proj
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + mid, 0.0, 1.0)

def _random_feasible(n, target):
    """Generate a random vector satisfying the constraints."""
    rnd = np.random.rand(n)
    return _proj_capped_simplex(rnd, target)

def _greedy_optimize(h0, time_budget):
    """Simple pairwise transfer greedy improvement."""
    start = time.time()
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5, _, _ = _evaluate_c5(h)
    it_no_imp = 0
    while time.time() - start < time_budget:
        corr = np.correlate(h, h, mode='full')
        min_corr = corr.min()
        lag = int(np.argmin(corr) - (n - 1))
        if lag == 0:
            # choose a non‑zero lag if possible
            idxs = np.argsort(corr)
            for idx in idxs[1:]:
                if corr[idx] > min_corr:
                    lag = int(idx - (n - 1))
                    break
        # gradient for this lag
        g = np.zeros(n)
        if lag > 0:
            k = lag
            g[:n - k] += h[k:]
            g[k:] += h[:n - k]
        elif lag < 0:
            k = -lag
            g[:n - k] += h[k:]
            g[k:] += h[:n - k]
        else:
            g = 2.0 * h
        up_ids = np.where(h < 1.0 - 1e-12)[0]
        down_ids = np.where(h > 1e-12)[0]
        if up_ids.size == 0 or down_ids.size == 0:
            break
        i_up = up_ids[np.argmax(g[up_ids])]
        i_dn = down_ids[np.argmin(g[down_ids])]
        if i_up == i_dn:
            # choose alternatives
            alt_up = up_ids[np.argsort(-g[up_ids])]
            alt_dn = down_ids[np.argsort(g[down_ids])]
            for cand in alt_up:
                if cand != i_dn:
                    i_up = cand
                    break
            for cand in alt_dn:
                if cand != i_up:
                    i_dn = cand
                    break
        eps = min(1.0 - h[i_up], h[i_dn], 0.01)
        if eps <= 0.0:
            break
        h_new = h.copy()
        h_new[i_up] += eps
        h_new[i_dn] -= eps
        new_c5, _, _ = _evaluate_c5(h_new)
        if new_c5 + 1e-12 < best_c5:
            best_c5, best_h = new_c5, h_new.copy()
            h = h_new
            it_no_imp = 0
        else:
            it_no_imp += 1
            if it_no_imp > 5000:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    t0 = time.time()
    # initialise h
    if 'initial_h_values' in globals():

    # greedy phase (≈60% of time)
    t_greedy = budget_s * 0.6
    best_h, best_c5 = _greedy_optimize(h_start, t_greedy)
    # local optimisation with SLSQP (remaining time)
    remaining = budget_s - (time.time() - t0)
    if remaining > 5.0:
        def obj(x):
            corr = np.correlate(x, x, mode='full')
            return ((x.sum() - corr.min()) * (2.0 / n))
        cons = {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}
        bounds = [(0.0, 1.0)] * n
        # number of random restarts limited by remaining time
        max_restarts = max(1, int(remaining // 30))
        for _ in range(max_restarts):
            if time.time() - t0 > budget_s:
                break
            x0 = _random_feasible(n, n / 2.0)
            res = minimize(obj, x0, method='SLSQP', bounds=bounds,

    # final projection and evaluation
    best_h = _proj_capped_simplex(best_h, n / 2.0)
    final_c5, _, _ = _evaluate_c5(best_h)
    return (best_h.tolist(), float(final_c5), int(n))
