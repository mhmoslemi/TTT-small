import numpy as np
import time
import math

def _c5_value(h):
    """Exact same metric that the evaluator uses."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement (zero‑padded)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _neighbor_sum(h, shift):
    """Sum of the two partners of each element for a given shift."""
    n = h.shape[0]
    if shift == 0:
        return 2.0 * h
    s = np.zeros_like(h)
    # forward partner
    if shift < n:
        s[: n - shift] += h[shift:]
    # backward partner
    if shift > 0:
        s[shift:] += h[: n - shift]
    return s


def _improve_once(h, step):
    """Perform one greedy move aimed at reducing the worst lag."""
    n = h.shape[0]
    S = np.sum(h)                     # = n/2  (kept exact)
    # autocorrelation of h with itself (non‑negative lags)
    auto = np.correlate(h, h, mode="full")
    T = auto[n - 1 :]                 # T[l] = Σ_i h_i h_{i+l}
    C = S - T                         # C[l] = Σ_i h_i (1‑h_{i+l})
    worst_shift = int(np.argmax(C))  # shift that attains the current maximum

    # gradient of C w.r.t. h (up to a positive factor)
    s = _neighbor_sum(h, worst_shift)

    # pick indices to transfer mass between
    inc_cands = np.where(h < 1.0 - 1e-12)[0]
    dec_cands = np.where(h > 1e-12)[0]
    if inc_cands.size == 0 or dec_cands.size == 0:
        return h, False

    inc_idx = inc_cands[np.argmax(s[inc_cands])]
    dec_idx = dec_cands[np.argmin(s[dec_cands])]

    if inc_idx == dec_idx:
        return h, False

    delta = min(step, 1.0 - h[inc_idx], h[dec_idx])
    if delta <= 0.0:
        return h, False

    h_new = h.copy()
    h_new[inc_idx] += delta
    h_new[dec_idx] -= delta
    return h_new, True


def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function (as a NumPy array) that gives a low value of C5.
    The returned tuple is (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # Choose a grid size.  We take the size of the supplied initial vector if
    # it exists, otherwise fall back to a modest 200‑point grid.
    # ----------------------------------------------------------------------
    try:
        n = int(initial_h_values.shape[0])   # provided by the harness
    except NameError:
        n = 200
    if n % 2:
        n += 1                  # enforce even length (sum must be n/2)

    dx = 2.0 / n
    target_c5 = 0.3808

    # ----------------------------------------------------------------------
    # Initial construction.
    # If a parent is supplied we start from it (it already satisfies the
    # constraints).  Otherwise we use a symmetric “U‑shape”: high near the
    # boundaries, low in the centre – a reasonable manual guess.
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
        # Numerical drift can break the exact sum; re‑project.
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        if abs(cur_sum - n / 2) > 1e-12:
            h *= (n / 2) / cur_sum
    else:
        # deterministic U‑shaped start: linear rise then fall
        x = (np.arange(n) + 0.5) * dx
        h = 1.0 - np.abs(x - 1.0)          # tent, then flip
        h = 1.0 - h                        # makes edges high, centre low
        # scale to satisfy the integral exactly
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        h *= (n / 2) / cur_sum

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ----------------------------------------------------------------------
    # Parameters for the greedy local search.
    # ----------------------------------------------------------------------
    step = 0.02                       # initial amount to transfer
    min_step = 0.001
    decay = 0.9995                    # geometric decay of step size
    max_iters = 300000                # generous upper bound (time will stop us)

    it = 0
    while time.time() - start < budget_s * 0.95 and it < max_iters:
        # one greedy modification aimed at the current worst lag
        h_new, changed = _improve_once(h, step)
        if not changed:
            # nothing to move – shrink step and continue
            step = max(step * decay, min_step)
            it += 1
            continue

        new_c5 = _c5_value(h_new)

        # accept any improvement; otherwise keep the old vector
        if new_c5 < best_c5 - 1e-9:
            best_h = h_new.copy()
            best_c5 = new_c5

        # keep the new vector if it improved the worst‑lag value,
        # otherwise revert (but still shrink the step a little)
        if new_c5 < _c5_value(h) - 1e-12:
            h = h_new
        else:
            # occasionally allow a non‑improving move to escape plateaus
            if rng.random() < 0.001:
                h = h_new
            # otherwise discard the move

        # decay the step size slowly
        step = max(step * decay, min_step)

        # early stop if we already beat the target
        if best_c5 <= target_c5:
            break

        it += 1

    # final evaluation (in case the loop stopped without an update)
    final_c5 = _c5_value(best_h)

    return (best_h, final_c5, n)
