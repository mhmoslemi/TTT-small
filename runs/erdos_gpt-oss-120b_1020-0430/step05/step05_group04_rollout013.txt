def compute_c5(h):
    """Return C5 for step vector h (list/ndarray)."""
    h = np.asarray(h, dtype=float)
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")

def adjust_sum_exact(vec, rng):
    """Greedy redistribution to enforce Σvec = len(vec)/2 (in‑place)."""
    n = vec.size
    target = n / 2.0
    diff = target - vec.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - vec[i])
        else:
            add = max(diff, -vec[i])
        vec[i] += add
        diff = target - vec.sum()
    return vec


def gradient_refine(h, rng, step=0.02, max_iter=100):
    """Projected (sub)gradient descent on the worst shift."""
    n = h.size

    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


# ---------- binary simulated‑annealing ----------
def binary_sa_search(start_bin, rng, deadline):
    """Hill‑climb with SA on a binary vector (exact half‑ones)."""
    cur = start_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    init_T = max(0.01 * cur_c, 1e-5)
    it = 0
    last_imp = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
                last_imp = it
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it - last_imp > 20000:
            # restart from best
            cur = best.copy()
            cur_c = best_c
            ones = list(np.flatnonzero(cur == 1))
            zeros = list(np.flatnonzero(cur == 0))
            last_imp = it
    return best, best_c


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # ----- decide problem size -----
    TARGET_N = 720
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        # interpolate to TARGET_N points
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, TARGET_N, endpoint=False)
        interp = np.interp(xt, xs, parent)
        # binary version with exact half‑ones
        idx = np.argpartition(interp, -TARGET_N // 2)[-TARGET_N // 2 :]
        start_bin = np.zeros(TARGET_N)
        start_bin[idx] = 1.0
        adjust_sum_exact(start_bin, rng)
    else:
        start_bin = np.zeros(TARGET_N)
        start_bin[rng.choice(TARGET_N, TARGET_N // 2, replace=False)] = 1.0
        adjust_sum_exact(start_bin, rng)

    # ----- binary SA -----
    bin_dead = min(deadline, time.time() + (deadline - time.time()) * 0.5)
    best_bin, best_c = binary_sa_search(start_bin, rng, bin_dead)

    # ----- continuous refinement -----
    cur = best_bin.astype(float)
    cur = gradient_refine(cur, rng, step=0.02, max_iter=120)

    # ----- SLSQP cascade (few betas) -----
    betas = (10, 100, 500, 2000, 5000, 20000)
    for beta in betas:
        if time.time() > deadline:
            break
        cur = slsqp_optimize(cur, beta, rng, max_iter=300)
        cur_c = compute_c5(cur)
        if cur_c < best_c:
            best_c = cur_c
            best_bin = np.rint(cur).astype(float)  # keep a binary copy for safety
            adjust_sum_exact(best_bin, rng)

    # final rounding and validation
    cur = np.clip(cur, 0.0, 1.0)
    adjust_sum_exact(cur, rng)
    final_c = compute_c5(cur)

    return cur.tolist(), float(final_c), int(TARGET_N)
