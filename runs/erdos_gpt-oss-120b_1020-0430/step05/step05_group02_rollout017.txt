def periodic_pattern(n, block_len=3):
    # pattern of one 1 followed by block_len-1 zeros
    pattern = np.zeros(block_len)
    pattern[0] = 1.0
    repeats = n // block_len + 1
    h = np.tile(pattern, repeats)[:n]
    # adjust sum
    target = n / 2.0
    cur = h.sum()
    if cur < target:
        # set extra ones randomly
        zeros = np.flatnonzero(h == 0)
        need = int(target - cur)
        idx = np.random.choice(zeros, need, replace=False)
        h[idx] = 1.0
    elif cur > target:
        ones = np.flatnonzero(h == 1)
        need = int(cur - target)
        idx = np.random.choice(ones, need, replace=False)
        h[idx] = 0.0
    return h

def compute_c5_fft(h):
    n = h.size
    rev = (1.0 - h)[::-1]
    N = 2 * n
    H = np.fft.fft(h, N)
    R = np.fft.fft(rev, N)
    conv = np.fft.ifft(H * R).real
    return float(conv.max() * (2.0 / n))

                g = 1 - 2 * h
            elif k > 0:
                g[:n - k] += 1 - h[k:]
                g[k:] -= h[:n - k]
            else:
                kk = -k
                g[kk:] += 1 - h[:n - kk]
                g[:n - kk] -= h[kk:]
            alpha = 0.02
            h_new = h - alpha * g
            np.clip(h_new, 0.0, 1.0, out=h_new)
            target = n / 2.0
            h_new = adjust_sum_exact(h_new, target, rng)
            new_c = compute_c5_fft(h_new)
            if new_c < cur_c:
                h = h_new
                cur_c = new_c
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
    return best_h, best_c

def binary_sa(h_bin, rng, deadline, max_iter=100000):
    n = h_bin.size
    target = n / 2.0
    def worst(arr):
        c = np.correlate(arr, arr, mode="full")
        return target - c.min()
    cur_w = worst(h_bin)
    best_h = h_bin.copy()
    best_w = cur_w
    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))
    T0 = cur_w * 0.05 + 1e-4
    decay = 0.9999
    it = 0
    while time.time() < deadline and it < max_iter:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_w = worst(h_bin)
        T = T0 * (decay ** it)
        accept = False
        if new_w < cur_w - 1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-(new_w - cur_w) / max(T, 1e-12)):
                accept = True
        if accept:
            cur_w = new_w
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_w < best_w - 1e-12:
                best_w = cur_w
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
        it += 1
    return best_h, best_w

def slsqp_refine(h0, rng, deadline):
    h = np.clip(h0, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
                 2000.0, 5000.0, 10000.0):
        if time.time() > deadline:
            break
        CURRENT_BETA = beta
        max_iter = 400 if beta < 200 else 800
        res = minimize(softmax_obj, h, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": max_iter,
                                "ftol": 1e-12, "disp": False})
        h = np.clip(res.x, 0.0, 1.0)
        h = adjust_sum_exact(h, target, rng)
    c = compute_c5_fft(h)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.96 * budget_s
    best_h = None
    best_c = np.inf
    best_n = None
    n_candidates = [360, 480, 600, 720]
    for idx_n, n in enumerate(n_candidates):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        alloc = remaining / (len(n_candidates) - idx_n)
        n_dead = time.time() + alloc
        target = n / 2.0
        inits = []
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            if parent.size != n:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                interp = np.interp(xt, xs, parent)
            else:
                interp = parent.copy()
            inits.append(interp)
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            inits.append(bin_vec)
        inits.append(random_binary(n, rng))
        inits.append(random_fractional(n, rng))
        inits.append(periodic_pattern(n, 3, rng))
        inits.append(periodic_pattern(n, 4, rng))
        for vec in inits:
            if time.time() > n_dead:
                break
            vec = np.clip(vec, 0.0, 1.0)
            if abs(vec.sum() - target) > 1e-9:
                vec = adjust_sum_exact(vec, target, rng)
            if np.all((vec == 0) | (vec == 1)):
                # binary simulated annealing
                sub_dead = min(n_dead, time.time() + 0.4 * (n_dead - time.time()))
                h_bin, worst = binary_sa(vec.copy(), rng, sub_dead)
                c5 = worst * (2.0 / n)
                if c5 < best_c:
                    best_c, best_h, best_n = c5, h_bin.astype(float).copy(), n
                vec = h_bin.astype(float)
                # optional refinement
                if time.time() < n_dead:
                    h_ref, c_ref = slsqp_refine(vec, rng, n_dead)
                    if c_ref < best_c:
                        best_c, best_h, best_n = c_ref, h_ref.copy(), n
                continue
            max_iters = int(3e5 * 360 / n)
            h_opt, c_opt = hill_climb(vec, rng, n_dead, max_iters=max_iters)
            if c_opt < best_c:
                best_c, best_h, best_n = c_opt, h_opt.copy(), n
            if time.time() < n_dead:
                h_ref, c_ref = slsqp_refine(best_h, rng, n_dead)
                if c_ref < best_c:
                    best_c, best_h, best_n = c_ref, h_ref.copy(), n
    # up‑sampling refinement if time permits
    if (best_h is not None and time.time() < deadline and best_n * 2 <= 720):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = adjust_sum_exact(h_up, n_up / 2.0, rng)
        max_iters = int(3e5 * 360 / n_up)
        h_up, c_up = hill_climb(h_up, rng, deadline, max_iters=max_iters)
        if time.time() < deadline:
            h_up, c_up = slsqp_refine(h_up, rng, deadline)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = adjust_sum_exact(best_h, n / 2.0, rng)
        best_c = compute_c5_fft(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = adjust_sum_exact(best_h, best_n / 2.0, rng)
        best_c = compute_c5_fft(best_h)
    return best_h.tolist(), float(best_c), int(best_n)
