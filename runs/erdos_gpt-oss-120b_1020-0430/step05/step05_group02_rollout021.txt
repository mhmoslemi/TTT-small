def gradient_refine_worst(h, target, rng, max_iter=20, step0=0.02):

   n = h.size; step = step0

   for it in range(max_iter):

       corr = np.correlate(h, 1 - h, mode='full')

       idx = np.argmax(corr) - (n - 1)

       k = abs(idx)

       g = np.zeros_like(h)

       if k == 0:
           g[:] = 1.0 - 2.0 * h
       else:
           upper = n - k
           g[:upper] += 1.0 - h[k:]
           g[k:] -= h[:upper]

       h = h - step * g

       h = np.clip(h, 0.0, 1.0)

       # adjust sum
       diff = target - h.sum()
       if abs(diff) > 1e-12:
           # similar random adjustment
           for _ in range(200):
               i = rng.integers(0, n)
               add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
               h[i] += add
               diff = target - h.sum()
               if abs(diff) < 1e-12:
                   break
       step *= 0.5

   return h

def smooth_max_and_grad(h, beta, dx):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_corr = np.max(corr)
    # subtract max_corr for numerical stability
    exp_vals = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    smooth = max_corr + np.log(sum_exp) / beta
    w = exp_vals / sum_exp  # length 2n-1
    # convolution
    conv_pos = np.convolve(w[::-1], h, mode='full')
    conv_neg = np.convolve(w, h, mode='full')
    idx = n - 1 + np.arange(n)
    sum_neighbors = conv_pos[idx] + conv_neg[idx]
    grad = dx * (1.0 - sum_neighbors)
    return smooth, grad

                    break
        step *= 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.99 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        parent = np.array(parent, dtype=float)

    candidate_ns = [240, 360, 480, 720]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        dx = 2.0 / n
        init_vecs = []

        if parent is not None:
            xs_p = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xs_n = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xs_n, xs_p, parent)
            init_vecs.append(interp)
            binv = np.zeros(n)
            top = np.argpartition(interp, -n // 2)[-n // 2 :]
            binv[top] = 1.0
            init_vecs.append(binv)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        init_vecs.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        adjust_sum_exact(rand_frac, target, rng)
        init_vecs.append(rand_frac)

        # alternating binary
        alt_bin = np.tile([1.0, 0.0], n // 2)
        init_vecs.append(alt_bin)

        # triangular shape
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        tri = 1.0 - np.abs(xs - 1.0)
        tri = np.clip(tri, 0.0, 1.0)
        tri *= target / tri.sum()
        adjust_sum_exact(tri, target, rng)
        init_vecs.append(tri)

        for h0 in init_vecs:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0).copy()
            adjust_sum_exact(h, target, rng)

            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

            betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]
            for beta in betas:
                if time.time() > deadline:
                    break
                if beta < 200:
                    max_iter = 200 if n <= 360 else 120
                elif beta < 2000:
                    max_iter = 120
                else:
                    max_iter = 200
                for _ in range(max_iter):
                    if time.time() > deadline:
                        break
                    smooth, grad = smooth_max_and_grad(h, beta, dx)
                    if np.linalg.norm(grad) < 1e-12:
                        break
                    step_sz = 1.0
                    while step_sz > 1e-8:
                        h_new = h - step_sz * grad
                        h_new = np.clip(h_new, 0.0, 1.0)
                        diff = target - h_new.sum()
                        if abs(diff) > 1e-12:
                            h_new += diff / n
                            h_new = np.clip(h_new, 0.0, 1.0)
                            diff = target - h_new.sum()
                            if abs(diff) > 1e-12:
                                for __ in range(200):
                                    i = rng.integers(0, n)
                                    add = min(diff, 1.0 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
                                    h_new[i] += add
                                    diff = target - h_new.sum()
                                    if abs(diff) < 1e-12:
                                        break
                        smooth_new, _ = smooth_max_and_grad(h_new, beta, dx)
                        if smooth_new <= smooth + 1e-12:
                            break
                        step_sz *= 0.5
                    h = h_new
                c5 = compute_c5(h)
                if c5 < best_c:
                    best_h, best_c, best_n = h.copy(), c5, n

            h = gradient_refine_worst(h, target, rng, max_iter=15, step0=0.02)
            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

    # optional up‑sampling
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 720
        and time.time() + 30 < deadline
    ):
        n_up = best_n * 2
        xs_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xs_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xs_up, xs_old, best_h)
        target_up = n_up / 2.0
        dx_up = 2.0 / n_up
        adjust_sum_exact(h_up, target_up, rng)

        betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]
        for beta in betas:
            if time.time() > deadline:
                break
            max_iter = 150
            for _ in range(max_iter):
                if time.time() > deadline:
                    break
                smooth, grad = smooth_max_and_grad(h_up, beta, dx_up)
                if np.linalg.norm(grad) < 1e-12:
                    break
                step_sz = 1.0
                while step_sz > 1e-8:
                    h_new = h_up - step_sz * grad
                    h_new = np.clip(h_new, 0.0, 1.0)
                    diff = target_up - h_new.sum()
                    if abs(diff) > 1e-12:
                        h_new += diff / n_up
                        h_new = np.clip(h_new, 0.0, 1.0)
                        diff = target_up - h_new.sum()
                        if abs(diff) > 1e-12:
                            for __ in range(200):
                                i = rng.integers(0, n_up)
                                add = min(diff, 1.0 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
                                h_new[i] += add
                                diff = target_up - h_new.sum()
                                if abs(diff) < 1e-12:
                                    break
                    smooth_new, _ = smooth_max_and_grad(h_new, beta, dx_up)
                    if smooth_new <= smooth + 1e-12:
                        break
                    step_sz *= 0.5
                h_up = h_new
            c5_up = compute_c5(h_up)
            if c5_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c5_up, n_up

        h_up = gradient_refine_worst(h_up, target_up, rng, max_iter=15, step0=0.02)
        c5_up = compute_c5(h_up)
        if c5_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c5_up, n_up

    # final random perturbation search if time permits
    while time.time() + 6 < deadline:
        h_tmp = best_h + rng.normal(scale=0.01, size=best_n)
        h_tmp = np.clip(h_tmp, 0.0, 1.0)
        adjust_sum_exact(h_tmp, best_n / 2.0, rng)
        for beta in [5, 20, 80, 200, 500, 1000, 2000]:
            if time.time() > deadline:
                break
            for _ in range(20):
                smooth, grad = smooth_max_and_grad(h_tmp, beta, 2.0 / best_n)
                step_sz = 1.0
                for _ in range(3):
                    h_new = h_tmp - step_sz * grad
                    h_new = np.clip(h_new, 0.0, 1.0)
                    diff = (best_n / 2.0) - h_new.sum()
                    if abs(diff) > 1e-12:
                        h_new += diff / best_n
                        h_new = np.clip(h_new, 0.0, 1.0)
                    smooth_new, _ = smooth_max_and_grad(h_new, beta, 2.0 / best_n)
                    if smooth_new <= smooth:
                        break
                    step_sz *= 0.5
                h_tmp = h_new
        h_tmp = gradient_refine_worst(h_tmp, best_n / 2.0, rng, max_iter=10, step0=0.02)
        c_tmp = compute_c5(h_tmp)
        if c_tmp < best_c:
            best_h, best_c, best_n = h_tmp.copy(), c_tmp, best_n

    if best_h is None:
        best_n = 360
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, best_n / 2.0, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
