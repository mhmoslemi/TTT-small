def gradient_refine_simple(h, rng, step=0.02, max_iter=20):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1 - 2 * h
        else:
            upper = n - k
            g[:upper] += 1 - h[k:]
            g[k:] -= h[:upper]
        # learning rate decays
        lr = step * (0.7 ** it)
        h = h - lr * g
        # projection and sum adjustment
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

            for _ in range(200):

                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break

        sin_vec *= target_sum / sin_vec.sum()
        adjust_sum_exact(sin_vec, rng)
        start_vecs.append(sin_vec)

        # double Gaussian bumps
        gauss = np.zeros(n)
        for _ in range(2):
            center = rng.integers(0, n)
            width = rng.integers(n // 20, n // 5 + 1)
            xs = np.arange(n)
            bump = np.exp(-0.5 * ((xs - center) / width) ** 2)
            gauss += bump
        gauss = np.clip(gauss, 0.0, 1.0)
        gauss *= target_sum / gauss.sum()
        adjust_sum_exact(gauss, rng)
        start_vecs.append(gauss)

        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h = np.clip(h0.copy(), 0.0, 1.0)
            adjust_sum_exact(h, rng)

            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_h, best_c, best_n = h.copy(), cur_c, n

            # surrogate smooth‑max optimization
            betas = [5, 20, 80, 200, 800, 2000, 8000, 20000]
            for beta in betas:
                if time.time() > deadline:
                    break
                steps = 15 + int(4 * np.log10(beta + 1))
                lr0 = 0.5
                for it in range(steps):
                    if time.time() > deadline:
                        break
                    # correlation and overlaps
                    corr = np.correlate(h, 1 - h, mode='full')
                    overlaps = corr * dx

                    # log‑sum‑exp weights
                    max_ov = np.max(overlaps)
                    w = np.exp(beta * (overlaps - max_ov))
                    w_sum = w.sum()
                    w_norm = w / w_sum

                    # split weights
                    w_forward = w_norm[n - 1:]          # shifts >=0, length n
                    w_neg = w_norm[:n - 1]              # shifts <0, length n-1
                    w_neg_forward = w_neg[::-1]        # ordered by positive distance

                    one = 1.0 - h

                    # term1_pos (forward, using reversed one)
                    conv_rev_one = np.convolve(w_forward, one[::-1])
                    term1_pos = conv_rev_one[n - 1::-1]

                    # term2_pos (forward, using h)
                    conv_h = np.convolve(w_forward, h)
                    term2_pos = conv_h[:n]

                    # term1_neg (negative shifts)
                    term1_neg = np.zeros(n)
                    if n > 1:
                        conv_neg_one = np.convolve(w_neg_forward, one)
                        term1_neg[1:] = conv_neg_one[:n - 1]

                    # term2_neg (negative shifts, using reversed h)
                    term2_neg = np.zeros(n)
                    if n > 1:
                        conv_neg_rev_h = np.convolve(w_neg_forward, h[::-1])
                        rev_slice = conv_neg_rev_h[n - 2::-1]
                        term2_neg[:n - 1] = rev_slice[:n - 1]

                    grad = dx * (term1_pos + term1_neg - term2_pos - term2_neg)

                    lr = lr0 * (0.9 ** it)
                    h = h - lr * grad
                    h = np.clip(h, 0.0, 1.0)
                    adjust_sum_exact(h, rng)

                cur_c = compute_c5(h)
                if cur_c < best_c:
                    best_h, best_c, best_n = h.copy(), cur_c, n

            # final gradient refinement targeting worst shift
            h = gradient_refine_simple(h, rng, step=0.02, max_iter=30)
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_h, best_c, best_n = h.copy(), cur_c, n

            # if essentially binary, run a light binary hill climb
            if np.all((h <= 1e-8) | (h >= 1 - 1e-8)):
                h_bin = np.rint(h).astype(float)
                adjust_sum_exact(h_bin, rng)
                h_bin, c_bin = binary_hill_climb_simple(
                    h_bin, rng, max_iter=3000, deadline=deadline
                )
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin

    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
