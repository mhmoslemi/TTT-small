def subgradient_step(h, step, rng):
    grad = np.zeros_like(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    # indices
    max_idxs = np.flatnonzero(np.abs(corr - max_val) < 1e-12)
    n = h.size
    for idx in max_idxs:
        k = idx - (n - 1)
        shift = abs(k)
        if shift == 0:
            # c[0] = sum h_i (1 - h_i) = sum (h_i - h_i^2)
            # derivative: (1 - 2 h_i)
            grad += 1 - 2 * h
        elif k > 0:
            # positive shift
            i_src = slice(0, n - shift)
            i_dst = slice(shift, n)
            grad[i_src] += 1 - h[i_dst]
            grad[i_dst] += -h[i_src]
        else: # k < 0
            i_src = slice(shift, n)
            i_dst = slice(0, n - shift)
            grad[i_src] += 1 - h[i_dst]
            grad[i_dst] += -h[i_src]
    # perform step
    h_new = h - step * grad
    # clip
    np.clip(h_new, 0, 1, out=h_new)
    # adjust sum
    target = n/2
    diff = target - h_new.sum()
    # distribute diff over random coordinates with capacity
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
        h_new[i] += add
        diff = target - h_new.sum()
    return h_new

def adjust_sum_exact(h, target, rng):
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        # compute allowable add
        if diff > 0:
            # can increase up to 1 - h[i]
            add = min(diff, 1 - h[i])
            if add <= 0:
                continue
        else:
            # decrease down to -h[i]
            add = max(diff, -h[i])
            if add >= 0:
                continue
        h[i] += add
        diff = target - h.sum()
    return h

def compute_corr_fft(h):
    """Cross‑correlation of h with (1‑h) using FFT (no scaling)."""
    n = h.size
    size = 2 * n
    fa = np.fft.rfft(h, size)
    fb = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(fa * np.conj(fb), size)
    return corr[: 2 * n - 1]

def compute_c5_from_corr(corr, n):
    """Convert raw correlation to the C5 value."""
    return float(corr.max() * (2.0 / n))

def adjust_sum_exact(h, target, rng):
    """Redistribute a tiny amount to hit the exact target sum."""
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

    bc = CURRENT_BETA * corr
