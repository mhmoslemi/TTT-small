def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # While diff not near zero
    while abs(diff) > 1e-12:
        # Choose indices that can be adjusted in the needed direction
        if diff > 0:
            # need to increase
            candidates = np.where(h < 1.0)[0]
            if len(candidates) == 0:
                break
            i = rng.choice(candidates)
            add = min(diff, 1.0 - h[i])
        else:
            candidates = np.where(h > 0.0)[0]
            if len(candidates) == 0:
                break
            i = rng.choice(candidates)
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
        else:
            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
