def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    # define softmax objective
    def obj(x):
        S = np.fft.irfft(np.abs(np.fft.rfft(x))**2)
        bc = -CURRENT_BETA * S
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    res = minimize(obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def binary_sa_fast(n, rng, time_budget):
    start = time.time()
    # initialize binary array
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    # compute autocorrelation min
    S = np.fft.irfft(np.abs(np.fft.rfft(h))**2)
    minS = S.min()
    best_h = h.copy()
    best_minS = minS
    best_c = 1.0 - 2.0*minS / n

    # init ones and zeros lists and position dicts
    ones_list = list(ones)
    zeros_list = [i for i in range(n) if i not in set(ones)]
    # position dicts
    pos_one = {idx: i for i, idx in enumerate(ones_list)}
    pos_zero = {idx: i for i, idx in enumerate(zeros_list)}

    T0 = 0.02
    T_end = 1e-4

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # pick random indices
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new minS
        S_new = np.fft.irfft(np.abs(np.fft.rfft(h))**2)
        minS_new = S_new.min()
        # delta C5 = 1 - 2*minS_new/n - (1 - 2*minS/n) = -2*(minS_new - minS)/n
        delta = -2.0 * (minS_new - minS) / n
        accept = False
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            accept = True
        if accept:
            minS = minS_new
            # update lists and dicts swap i<->j
            # i was in ones_list at pos_one[i]; replace with j
            idx_i = pos_one[i]
            ones_list[idx_i] = j
            pos_one[j] = idx_i
            del pos_one[i]
            # j was in zeros_list at pos_zero[j]; replace with i
            idx_j = pos_zero[j]
            zeros_list[idx_j] = i
            pos_zero[i] = idx_j
            del pos_zero[j]
            # update best
            if minS_new > best_minS:
                best_minS = minS_new
                best_h = h.copy()
                best_c = 1.0 - 2.0*best_minS / n
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            # lists unchanged
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# global variables used by the SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None

# ----------------------------------------------------------------------
def compute_c5_fast(h: np.ndarray) -> float:
    """Return C5 for a step function given by array h."""
    n = h.size
    # autocorrelation via FFT
    minS = np.fft.irfft(np.abs(np.fft.rfft(h))**2).min()
    return 1.0 - 2.0 * minS / n


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (largest) overlap O_k."""
    n = h.size
    S = np.fft.irfft(np.abs(np.fft.rfft(h))**2)
    k = int(np.argmin(S))  # shift giving min autocorrelation → max overlap
    if k == 0:
        return 1.0 - 2.0 * h
    else:
        idx_plus = (np.arange(n) + k) % n
        idx_minus = (np.arange(n) - k) % n
        grad = np.ones_like(h)
        grad -= h[idx_plus]
        grad -= h[idx_minus]
        return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer a small amount from a positive‑gradient entry to a negative‑gradient entry."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5_fast(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        # re‑enforce the sum constraint
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}

    def obj(x):
        S = np.fft.irfft(np.abs(np.fft.rfft(x))**2)
        bc = -CURRENT_BETA * S
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def binary_sa_fast(n: int, rng: np.random.Generator, time_budget: float):
    """Binary simulated‑annealing (0/1) with exact autocorrelation via FFT."""
    start = time.time()
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0

    S = np.fft.irfft(np.abs(np.fft.rfft(h))**2)
    minS = S.min()
    best_h = h.copy()
    best_minS = minS
    best_c = 1.0 - 2.0 * minS / n

    # maintain mutable lists + position maps for O(1) swaps
    ones_set = set(ones)
    zeros = [i for i in range(n) if i not in ones_set]
    ones_list = list(ones)
    zeros_list = zeros
    pos_one = {idx: i for i, idx in enumerate(ones_list)}
    pos_zero = {idx: i for i, idx in enumerate(zeros_list)}

    T0 = 0.02
    T_end = 1e-4

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        S_new = np.fft.irfft(np.abs(np.fft.rfft(h))**2)
        minS_new = S_new.min()
        delta = -2.0 * (minS_new - minS) / n  # change in C5
        accept = False
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            accept = True
        if accept:
            minS = minS_new
            # update list structures
            idx_i = pos_one[i]
            ones_list[idx_i] = j
            pos_one[j] = idx_i
            del pos_one[i]
            idx_j = pos_zero[j]
            zeros_list[idx_j] = i
            pos_zero[i] = idx_j
            del pos_zero[j]
            if minS_new > best_minS:
                best_minS = minS_new
                best_h = h.copy()
                best_c = 1.0 - 2.0 * best_minS / n

                    return best_h, best_c
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            # lists unchanged
    return best_h, best_c
