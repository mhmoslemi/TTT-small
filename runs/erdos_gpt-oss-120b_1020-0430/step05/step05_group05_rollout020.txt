import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the SLSQP surrogate objective
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Maximum overlap integral C5 for a step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth‑max surrogate for max overlap (log‑sum‑exp)."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=1500):
    """Run SLSQP on the smooth‑max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly (random walk)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def adjust_sum_exact(h, rng):
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.025, max_iter=30):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]        # derivative from h_j term
            g[k:]   -= h[:up]            # derivative from -h_j h_{j+k}
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # re‑enforce sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(150):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def fractional_to_binary(h, rng):
    """Binary vector with exactly n/2 ones – take the largest entries."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    adjust_sum_exact(bin_h, rng)
    return bin_h

def binary_targeted_improve(h, rng, deadline):
    """Greedy swaps that lower the worst overlap (periodic version)."""
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        shift = abs(shift)
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        # pick a 1 with minimal neighbor sum, a 0 with maximal neighbor sum
        i = ones[np.argmin(neigh[ones])]
        j = zeros[np.argmax(neigh[zeros])]
        if neigh[i] - neigh[j] < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        # try a few random swaps as fall‑back
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

def binary_local_search(h, rng, deadline, max_trials=20000):
    """Simple hill‑climbing: random swaps that strictly improve max overlap."""
    n = h.size
    best_c = compute_c5(h)
    ones = set(np.flatnonzero(h == 1))
    zeros = set(np.flatnonzero(h == 0))
    trials = 0
    while time.time() < deadline and trials < max_trials:
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # perform swap
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        trials += 1
        if c < best_c - 1e-12:
            best_c = c
            ones.remove(i); ones.add(j)
            zeros.remove(j); zeros.add(i)
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99
    TARGET_C5 = 0.3808   # stop early if we reach this bound

    # ------------------------------------------------------------------
    # base size
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
    else:
        parent = None
        n0 = 360

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # generate start vectors
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(fractional_to_binary(parent, rng))

    # random binary seeds
    for _ in range(5):
        hb = np.zeros(n0)
        hb[rng.choice(n0, n0 // 2, replace=False)] = 1.0
        starts.append(hb)

    # random fractional seeds
    for _ in range(5):
        hf = rng.random(n0)
        hf *= (n0 / 2) / hf.sum()
        hf = np.clip(hf, 0.0, 1.0)
        adjust_sum_exact(hf, rng)
        starts.append(hf)

    # heavy‑tailed fractional seeds
    for _ in range(2):
        hf = rng.beta(0.5, 0.5, size=n0)
        hf *= (n0 / 2) / hf.sum()
        hf = np.clip(hf, 0.0, 1.0)
        adjust_sum_exact(hf, rng)
        starts.append(hf)

    # ------------------------------------------------------------------
    # schedule of betas for the smooth‑max surrogate
    betas = (0.5, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0,
             500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0)

    # ------------------------------------------------------------------
    # main optimisation loop over all start vectors
    for base in starts:
        if time.time() > deadline or best_c <= TARGET_C5:
            break
        h = np.clip(base, 0.0, 1.0)
        adjust_sum_exact(h, rng)

        # cascade of SLSQP runs with increasing beta
        for beta in betas:
            if time.time() > deadline or best_c <= TARGET_C5:
                break
            max_iter = 1500 if beta < 200 else 2000
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            # add a tiny random perturbation to escape possible plateaus
            if rng.random() < 0.3:
                noise = rng.normal(scale=0.01, size=h.size)
                h = np.clip(h + noise, 0.0, 1.0)
                adjust_sum_exact(h, rng)

            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n0
                if best_c <= TARGET_C5:
                    break

        # projected gradient polishing (more iterations)
        h = gradient_refine(h, rng, step=0.025, max_iter=40)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n0
            if best_c <= TARGET_C5:
                break

        # binary rounding + greedy improvement
        hb = fractional_to_binary(h, rng)
        hb = binary_targeted_improve(hb, rng, deadline)
        c_b = compute_c5(hb)
        if c_b < best_c:
            best_h, best_c, best_n = hb.copy(), c_b, n0
            if best_c <= TARGET_C5:
                break

        # extra binary hill climbing
        hb, c_b2 = binary_local_search(hb, rng, deadline)
        if c_b2 < best_c:
            best_h, best_c, best_n = hb.copy(), c_b2, n0
            if best_c <= TARGET_C5:
                break

    # ------------------------------------------------------------------
    # optional up‑sampling refinement (to 2× resolution)
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        for beta in betas:
            if time.time() > deadline or best_c <= TARGET_C5:
                break
            max_iter = 1500 if beta < 200 else 2000
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=max_iter)

        h_up = gradient_refine(h_up, rng, step=0.025, max_iter=30)
        h_up_bin = fractional_to_binary(h_up, rng)
        h_up_bin = binary_targeted_improve(h_up_bin, rng, deadline)
        c_up = compute_c5(h_up_bin)
        if c_up < best_c:
            best_h, best_c, best_n = h_up_bin, c_up, n_up

    # ------------------------------------------------------------------
    # fallback (should never be needed)
    if best_h is None:
        best_n = n0
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
