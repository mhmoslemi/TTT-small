def binary_sa(h, rng, deadline):
    n = h.size
    h = h.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(h > 0.5))
    zeros = list(np.flatnonzero(h <= 0.5))
    init_T = max(1e-4, cur_c * 0.2)
    last_improve = 0
    i = 0
    max_no_improve = max(20000, n*30)
    while time.time() < deadline:
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        # swap
        h[i_one] = 0.0
        h[i_zero] = 1.0
        new_c = compute_c5(h)
        T = init_T * (0.99995 ** i) + 1e-12
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            # update lists
            ones.remove(i_one)
            ones.append(i_zero)
            zeros.remove(i_zero)
            zeros.append(i_one)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                last_improve = i
        else:
            # revert
            h[i_one] = 1.0
            h[i_zero] = 0.0
        i += 1
        if i - last_improve > max_no_improve:
            break
    return best_h, best_c
```

def compute_corr_fft(h):
    n = h.size
    m = (2*n-1).bit_length()
    size = 1 << m
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)

def gradient_refine(h, rng, max_iter, step_start, deadline):
    n = h.size
    target = n / 2.0
    step = step_start
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr_fft(h)
        idx = np.argmax(corr)
        k = idx - (n - 1)
        k = abs(k)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            up = n - k
            g = np.zeros_like(h)
            g[:up] = 1.0 - h[k:]
            g[k:] -= h[:up]
        # line search: try step and step/2 until improvement
        cur_c = corr.max() * (2.0 / n)
        # try reduce
        for ls in (step, step*0.5, step*0.25):
            new_h = h - ls * g
            new_h = np.clip(new_h, 0.0, 1.0)
            diff = target - new_h.sum()
            if abs(diff) > 1e-12:
                # adjust a random index
                i = rng.integers(0, n)
                add = min(diff, 1.0 - new_h[i]) if diff > 0 else max(diff, -new_h[i])
                new_h[i] += add
            new_c = compute_c5_fft(new_h)
            if new_c < cur_c - 1e-12:
                h = new_h
                step = ls
                break
        else:
            step *= 0.5
        # otherwise no improvement, reduce step
    return h

def binary_sa(h, rng, deadline, max_no_improve_factor=30):
    n = h.size
    h = h.copy()
    cur_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c
    # ones and zeros lists and positions
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    ones = ones.tolist()
    zeros = zeros.tolist()
    pos_one = np.full(n, -1, dtype=int)
    pos_zero = np.full(n, -1, dtype=int)
    for idx, i in enumerate(ones):
        pos_one[i] = idx
    for idx, i in enumerate(zeros):
        pos_zero[i] = idx
    # temperature
    T = max(1e-4, cur_c * 0.2)
    dec = 0.9999
    last_improve = 0
    i = 0
    max_no_improve = max(20000, n * max_no_improve_factor)
    while time.time() < deadline:
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        # perform swap
        h[i_one] = 0.0
        h[i_zero] = 1.0
        new_c = compute_c5_fft(h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            # update lists using pos
            idx_one = pos_one[i_one]
            idx_zero = pos_zero[i_zero]
            # swap entries
            ones[idx_one] = i_zero
            zeros[idx_zero] = i_one
            # update pos arrays
            pos_one[i_zero] = idx_one
            pos_one[i_one] = -1
            pos_zero[i_one] = idx_zero
            pos_zero[i_zero] = -1
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                last_improve = i
        else:
            # revert
            h[i_one] = 1.0
            h[i_zero] = 0.0
        i += 1
        T = max(T * dec, 1e-12)
        if i - last_improve > max_no_improve:
            break
    return best_h, best_c
```

def upsample_and_refine(h, rng, deadline):
    n = h.size
    if n * 2 > 720:
        return h, compute_c5_fft(h)
    h_up = np.repeat(h, 2)
    target = h_up.size / 2.0
    # adjust sum if needed
    diff = target - h_up.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(0, h_up.size)
        add = min(diff, 1.0 - h_up[i]) if diff > 0 else max(diff, -h_up[i])
        h_up[i] += add
    # gradient refine
    h_up = gradient_refine(h_up, rng, max_iter=200, step_start=0.05, deadline=deadline)
    # SA if binary?
    # after gradient, may still be fractional; turn binary if near binary
    if np.all((h_up >= 0) & (h_up <= 1)):
        # we can binarize top half
        idx = np.argpartition(h_up, -h_up.size//2)[-h_up.size//2:]

        h_up = gradient_refine(h_bin, rng, max_iter=150, step_start=0.02, deadline=deadline)
        final_c = compute_c5_fft(h_up)
        return h_up, final_c

        final_c = compute_c5_fft(h_up)

import time
import numpy as np

# ---------- correlation helpers ----------
def compute_corr_fft(h):
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * G.conj(), n=size)
    return corr[: 2 * n - 1]


def compute_c5_fft(h):
    n = h.size
    return float(compute_corr_fft(h).max() * (2.0 / n))


# ---------- gradient refinement ----------
def gradient_refine(h, rng, max_iter, step_start, deadline):
    n = h.size
    target = n / 2.0
    step = step_start
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr_fft(h)
        idx = int(np.argmax(corr))
        k = abs(idx - (n - 1))
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            up = n - k
            g = np.zeros_like(h)
            g[:up] = 1.0 - h[k:]
            g[k:] -= h[:up]
        cur_c = corr.max() * (2.0 / n)
        improved = False
        for ls in (step, step * 0.5, step * 0.25):
            nh = h - ls * g
            nh = np.clip(nh, 0.0, 1.0)
            diff = target - nh.sum()
            if abs(diff) > 1e-12:
                i = rng.integers(0, n)
                add = min(diff, 1.0 - nh[i]) if diff > 0 else max(diff, -nh[i])
                nh[i] += add
            nc = compute_c5_fft(nh)
            if nc < cur_c - 1e-12:
                h = nh
                step = ls
                improved = True
                break
        if not improved:
            step *= 0.5
    return h


# ---------- binary simulated annealing ----------
def binary_sa(h, rng, deadline, max_no_improve_factor=200):
    n = h.size
    h = h.copy()
    cur_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    pos_one = np.full(n, -1, dtype=int)
    pos_zero = np.full(n, -1, dtype=int)
    for i, v in enumerate(ones):
        pos_one[v] = i
    for i, v in enumerate(zeros):
        pos_zero[v] = i
    T = max(1e-4, cur_c * 0.5)
    dec = 0.9999
    last_improve = 0
    it = 0
    max_no_improve = max(20000, n * max_no_improve_factor)
    while time.time() < deadline:
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        h[i_one] = 0.0
        h[i_zero] = 1.0
        new_c = compute_c5_fft(h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            idx_one = pos_one[i_one]
            idx_zero = pos_zero[i_zero]
            ones[idx_one] = i_zero
            zeros[idx_zero] = i_one
            pos_one[i_zero] = idx_one
            pos_one[i_one] = -1
            pos_zero[i_one] = idx_zero
            pos_zero[i_zero] = -1
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                last_improve = it
        else:
            h[i_one] = 1.0
            h[i_zero] = 0.0
        it += 1
        T = max(T * dec, 1e-12)
        if it - last_improve > max_no_improve:
            break
    return best_h, best_c


# ---------- up‑sampling and extra refinement ----------
def upsample_and_refine(h, rng, deadline):
    n = h.size
    if n * 2 > 720:
        return h, compute_c5_fft(h)
    h_up = np.repeat(h, 2)
    target = h_up.size / 2.0
    diff = target - h_up.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(0, h_up.size)
        add = min(diff, 1.0 - h_up[i]) if diff > 0 else max(diff, -h_up[i])
        h_up[i] += add
    h_up = gradient_refine(h_up, rng, max_iter=200, step_start=0.05, deadline=deadline)
    idx = np.argpartition(h_up, -h_up.size // 2)[-h_up.size // 2 :]
    h_bin = np.zeros_like(h_up)
    h_bin[idx] = 1.0
    h_up = h_bin
    h_up, _ = binary_sa(h_up, rng, deadline)
    h_up = gradient_refine(h_up, rng, max_iter=150, step_start=0.02, deadline=deadline)
    return h_up, compute_c5_fft(h_up)


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = globals().get('initial_h_values', None)

    candidate_ns = [720, 600, 480, 360, 240, 180]
    candidate_ns = [n for n in candidate_ns if n % 2 == 0 and n <= 720]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        start_vecs = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp)
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)

        # two random binary starts
        for _ in range(2):
            bin_rand = np.zeros(n)
            bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            start_vecs.append(bin_rand)

        # random fractional start
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        diff = target - rand_frac.sum()
        if abs(diff) > 1e-12:
            i = rng.integers(0, n)
            add = min(diff, 1.0 - rand_frac[i]) if diff > 0 else max(diff, -rand_frac[i])
            rand_frac[i] += add
        start_vecs.append(rand_frac)

        for base_h in start_vecs:
            if time.time() > deadline:
                break
            h = np.array(base_h, dtype=float)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
            h = np.clip(h, 0.0, 1.0)

            if not np.all((h == 0) | (h == 1)):
                h = gradient_refine(h, rng, max_iter=200, step_start=0.08, deadline=deadline)

            if not np.all((h == 0) | (h == 1)):
                idx = np.argpartition(h, -n // 2)[-n // 2 :]
                hb = np.zeros_like(h)
                hb[idx] = 1.0
                h = hb

            h_sa, c_sa = binary_sa(h, rng, deadline, max_no_improve_factor=200)
            h_final = gradient_refine(h_sa, rng, max_iter=100, step_start=0.02, deadline=deadline)
            c_final = compute_c5_fft(h_final)

            if c_final < best_c:
                best_h, best_c, best_n = h_final.copy(), c_final, n

    # optional up‑sampling if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        h_up, c_up = upsample_and_refine(best_h, rng, deadline)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, best_n * 2

    # fallback
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_c = compute_c5_fft(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(0, best_n)
        add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
        best_h[i] += add
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
