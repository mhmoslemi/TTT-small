import time
import numpy as np
import scipy.optimize

# ---------- Global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for discretized h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass to enforce sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth max (log-sum-exp) of the correlation vector."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # (log sum exp)/beta  =  m/beta + log(sum(exp(bc-m)))/beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 600) -> np.ndarray:
    """SLSQP on the smooth max surrogate, keeping exact sum after."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = scipy.optimize.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    # tiny exact correction via random walk
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def hill_climb(h: np.ndarray, rng: np.random.Generator,
               max_iter: int = 200, step: float = 0.01) -> np.ndarray:
    """Simple random mass transfer hill‑climbing."""
    n = h.size
    target = n / 2.0
    cur_c = compute_c5(h)
    for _ in range(max_iter):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or h[i] <= 0.0 or h[j] >= 1.0:
            continue
        max_delta = min(step, h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c_new = compute_c5(h_new)
        if c_new < cur_c - 1e-12:
            h = h_new
            cur_c = c_new
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    def maybe_update(h):
        nonlocal best_h, best_c, best_n
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_h = h.copy()
            best_c = c
            best_n = h.size

    # ----- start from provided parent if any -----
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        maybe_update(parent)

    # main search parameters
    betas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    n_choices = [240, 360, 480, 720]

    # ----- main optimization loop -----
    while time.time() < deadline:
        # pick a size (prefer 360 as core)
        n = rng.choice(n_choices)
        target = n / 2.0

        # generate a random fractional start
        h = rng.random(n)
        h *= target / h.sum()
        h = np.clip(h, 0.0, 1.0)
        adjust_sum_exact(h, rng)

        # occasional mixing with parent if available
        if 'initial_h_values' in globals() and rng.random() < 0.25:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            base = np.interp(xt, xs, parent)
            mix = rng.random()
            h = mix * base + (1 - mix) * h
            adjust_sum_exact(h, rng)

        # cascade of SLSQP on increasingly sharp smooth‑max
        for beta in betas:
            if time.time() >= deadline:
                break
            max_iter = 400 if beta < 200 else 800
            # give extra iterations for the highest betas if time permits
            if beta >= 5000 and deadline - time.time() > 5.0:
                max_iter = 1500
            h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            maybe_update(h)

        # local gradient refinement
        h = gradient_refine(h, rng, step=0.02, max_iter=12)
        maybe_update(h)

        # hill‑climbing fine tune
        h = hill_climb(h, rng, max_iter=200, step=0.01)
        maybe_update(h)

        # occasional extra gradient sweep
        h = gradient_refine(h, rng, step=0.015, max_iter=8)
        maybe_update(h)

        # break if remaining time is too short for another full pass
        if deadline - time.time() < 3.0:
            break

    # ----- final up‑sampling and intensive refinement -----
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        up = np.interp(xt, xs, best_h)
        adjust_sum_exact(up, rng)
        h = up
        for beta in betas:
            if time.time() >= deadline:
                break
            max_iter = 800 if beta < 200 else 1500
            h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
        h = gradient_refine(h, rng, step=0.02, max_iter=12)
        maybe_update(h)

    # final safety tweaks
    if best_h is None:
        # fallback to a simple alternating pattern
        n = 360
        best_h = np.tile([1.0, 0.0], n // 2).astype(float)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
