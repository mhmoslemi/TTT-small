def compute_c5_fft(h):
    n = h.size
    L = 2*n
    H = np.fft.rfft(h, L)  # length L//2+1
    G = np.fft.rfft(1-h, L)
    conv = np.fft.irfft(H * G, L)  # real part
    # full linear correlation (size 2n-1) lies at indices 0..2n-2: conv[0:2*n-1]
    # It corresponds to correlation where index shift k corresponds to conv[n-1 + k]
    # compute max*
    max_corr = np.max(conv[:2*n-1])
    return max_corr * (2.0 / n)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # tolerance
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to increase sum: choose index with h < 1
            candidates = np.where(h < 1.0)[0]
            if candidates.size == 0:
                break
            i = rng.choice(candidates)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            # need to decrease sum: choose index with h > 0
            candidates = np.where(h > 0.0)[0]
            if candidates.size == 0:
                break
            i = rng.choice(candidates)
            add = max(diff, -h[i])

        diff = target - h.sum()
    return h

    n = h.size
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)

    else:
        # shift contributions
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

import time
import numpy as np

# ---------- utility functions ----------
def compute_c5_fft(h: np.ndarray) -> float:
    """C5 for a step function on [0,2] using FFT for speed."""
    n = h.size
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)[: 2 * n - 1]
    return float(corr.max() * (2.0 / n))


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = n/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1.0)[0]
        else:
            cand = np.where(h > 0.0)[0]
        if cand.size == 0:
            break
        i = rng.choice(cand)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), L)[: 2 * n - 1]
    idx = int(np.argmax(corr)) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def hill_climb_deterministic(h0: np.ndarray, rng: np.random.Generator,
                             max_iters: int = 10000, init_step_scale: float = 0.5):
    """Gradient‑directed hill climb with a deterministic step size."""
    h = h0.copy()
    best_c = compute_c5_fft(h)
    step_scale = init_step_scale
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * step_scale
        h_candidate = h.copy()
        h_candidate[i] -= eps
        h_candidate[j] += eps
        c_candidate = compute_c5_fft(h_candidate)
        if c_candidate < best_c:
            h, best_c = h_candidate, c_candidate
        if (it + 1) % decay_interval == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
    return h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512

    # initial vector (binary, exact integral)

    best_c = compute_c5_fft(best_h)

    # ---------- stochastic multi‑restart hill climbing ----------
    while time.time() - start < 0.93 * budget_s:

        time_left = budget_s - (time.time() - start)

        max_iters = int(min(15000, max(500, time_left * 5000)))
        cur_h, cur_c = hill_climb_deterministic(

    # ---------- final refinements ----------

    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
