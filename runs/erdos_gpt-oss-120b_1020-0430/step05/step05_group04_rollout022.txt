def solve_milp(n, M_ub, rng, deadline):
    kmax = n // 2
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, kmax), nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(h) == n/2,
                   M >= 0,
                   M <= M_ub]
    for k in range(1, kmax+1):
        max_i = n - k
        # y constraints - vectorized
        # y_i_k = y[:max_i, k-1]
        hi = h[:max_i]
        hj = h[k:]
        yk = y[:max_i, k-1]
        constraints.append(yk <= hi)
        constraints.append(yk <= hj)
        constraints.append(yk >= hi + hj - 1)
        s_k = cp.sum(yk)
        constraints.append(M >= n/2 - s_k)
    prob = cp.Problem(cp.Minimize(M), constraints)
    remaining = deadline - time.time()
    if remaining <= 0:
        return None
    try:
        prob.solve(solver=cp.CBC, maxSeconds=remaining, verbose=False)
    except Exception:
        return None
    if h.value is None or M.value is None:
        return None
    h_sol = np.rint(h.value).astype(int)  # ensure binary
    # adjust sum to exactly n/2 if rounding gives off by 1 due to numeric
    cur_sum = h_sol.sum()
    if cur_sum != n//2:
        # fix by flipping some bits

            # need to set -diff ones to 0
            ones = np.where(h_sol == 1)[0]

        max_i = n - k
        hi = h[:max_i]
        hj = h[k:]
        yk = y[:max_i, k - 1]
        cons.extend([yk <= hi, yk <= hj, yk >= hi + hj - 1])
        cons.append(M >= n / 2 - cp.sum(yk))
    prob = cp.Problem(cp.Minimize(M), cons)
    rem = deadline - time.time()
    if rem <= 1.0:
        return None
    try:
        prob.solve(solver=cp.CBC, maxSeconds=rem, verbose=False)
    except Exception:
        return None
    if prob.status in (cp.INFEASIBLE, cp.UNBOUNDED):
        return None
    if h.value is None:
        return None
    h_sol = np.rint(h.value).astype(int)
    # repair tiny rounding errors
    cur = h_sol.sum()
    diff = n // 2 - cur
    if diff > 0:
        zeros = np.where(h_sol == 0)[0]
        h_sol[zeros[:diff]] = 1
    elif diff < 0:
        ones = np.where(h_sol == 1)[0]
        h_sol[ones[: -diff]] = 0
    return h_sol, float(M.value)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ---------- heuristic at larger resolution ----------
    n_big = 180
    best_h_big, best_c_big = heuristic_binary(n_big, rng, deadline)

    # ---------- MILP at moderate resolution ----------
    n_mid = 120
    if time.time() < deadline:
        best_h_mid, best_c_mid = heuristic_binary(n_mid, rng, deadline)
        M_ub = best_c_mid * n_mid / 2.0 * 1.005  # small slack
        milp_res = solve_milp(n_mid, M_ub, rng, deadline)
        if milp_res is not None:
            h_mid, M_val = milp_res
            c_mid = compute_c5(h_mid)
            if c_mid < best_c_mid:
                best_h_mid, best_c_mid = h_mid, c_mid

    # ---------- choose the best of the candidates ----------
    candidates = [(best_h_big, best_c_big, n_big)]
    if 'best_h_mid' in locals():
        candidates.append((best_h_mid, best_c_mid, n_mid))
    best_h, best_c, best_n = min(candidates, key=lambda x: x[1])

    # final safety: ensure binary and exact sum
    best_h = np.rint(best_h).astype(float)
    cur = best_h.sum()
    diff = best_n / 2 - cur
    if diff > 0:
        zeros = np.where(best_h == 0)[0]
        best_h[zeros[:int(diff)]] = 1.0
    elif diff < 0:
        ones = np.where(best_h == 1)[0]
        best_h[ones[:int(-diff)]] = 0.0
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
