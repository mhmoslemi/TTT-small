def smooth_max_obj(h, beta):
    n = h.size
    target = n/2
    # compute correlation
    fft_h = np.fft.fft(h)
    corr = np.real(np.fft.ifft(fft_h * np.conj(fft_h)))
    # compute g = target - corr
    g = target - corr
    max_g = np.max(g)  # for numerical stability
    # log-sum-exp
    return (max_g + np.log(np.sum(np.exp(beta * (g - max_g)))))/beta

def gradient_descent_schedule(h0, beta_list, deadline):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for beta in beta_list:
        # determine max_iter proportional to remaining time and size
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # set max_iter, e.g., 2000 but also limited by time
        max_iter = min(2000, int(remaining / 0.0005))  # guess each iter 0.5 ms
        # step size schedule
        step = 0.5  # initial step size
        decay = 0.995
        for it in range(max_iter):
            if time.time() > deadline:
                break
            # compute FFT once per iteration
            fft_h = np.fft.fft(h)
            corr = np.real(np.fft.ifft(fft_h * np.conj(fft_h)))
            g = target - corr
            # compute smooth max for possible line search
            m = np.max(g)
            exp_term = np.exp(beta * (g - m))
            f = (m + np.log(exp_term.sum())) / beta
            # compute weights
            w = exp_term / exp_term.sum()
            # gradient of smooth max
            grad = -2.0 * np.real(np.fft.ifft(np.fft.fft(w) * fft_h))
            # descent step
            h_candidate = proj_box_sum(h - step * grad, target)
            # compute f for candidate
            fft_hc = np.fft.fft(h_candidate)
            corr_c = np.real(np.fft.ifft(fft_hc * np.conj(fft_hc)))
            g_c = target - corr_c
            m_c = np.max(g_c)
            exp_term_c = np.exp(beta * (g_c - m_c))
            f_c = (m_c + np.log(exp_term_c.sum())) / beta
            if f_c <= f:
                h = h_candidate
                # maybe increase step size a bit
                step = step * 1.02
            else:
                step = step * 0.5
            decay = decay  # not used
            step = max(step, 1e-8)
    return h

import time
import numpy as np

def proj_box_sum(y, target, tol=1e-12, max_iter=100):
    """Projection of y onto {0<=x<=1, sum x = target}."""
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def smooth_max(g, beta):
    """log‑sum‑exp of vector g with temperature beta."""
    m = np.max(g)
    e = np.exp(beta * (g - m))
    return (m + np.log(e.sum())) / beta, e / e.sum()


def gradient_descent_schedule(h0, betas, deadline):
    """Refine h0 for a list of β values using projected gradient descent."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for beta in betas:
        step = 0.5
        for _ in range(1500):
            if time.time() > deadline:
                return h
            # correlation and smooth‑max
            fft_h = np.fft.fft(h)
            corr = np.real(np.fft.ifft(fft_h * np.conj(fft_h)))
            g = target - corr
            f, w = smooth_max(g, beta)
            # gradient of smooth‑max
            grad = -2.0 * np.real(np.fft.ifft(np.fft.fft(w) * fft_h))
            # trial step
            h_new = proj_box_sum(h - step * grad, target)
            # evaluate new smooth‑max
            fft_hn = np.fft.fft(h_new)
            corr_n = np.real(np.fft.ifft(fft_hn * np.conj(fft_hn)))
            g_n = target - corr_n
            f_n, _ = smooth_max(g_n, beta)
            if f_n <= f:
                h = h_new
                step = min(step * 1.05, 1.0)
            else:
                step *= 0.5
    return h


def binary_hill_climb(h_bin, rng, deadline, max_iter=2000):
    """Simple hill‑climbing on binary vectors (swap 1↔0)."""
    n = h_bin.size
    cur_c = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c = cur_c
    ones = [i for i in range(n) if h_bin[i] == 1.0]
    zeros = [i for i in range(n) if h_bin[i] == 0.0]
    it = 0
    while it < max_iter and time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c = compute_c5(h_bin)
        if new_c < cur_c - 1e-12:
            cur_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
        it += 1
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    # base dimension
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        base_h = np.array(initial_h_values, dtype=float)
    else:
        n = 360
        base_h = None
    target = n / 2.0

    # β schedule
    betas = [0.1, 0.5, 1.0, 5.0, 20.0, 80.0, 200.0,
             500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    best_h = None
    best_c = np.inf
    best_n = n

    # ----- multiple restarts -----
    restart = 0
    while time.time() < deadline and restart < 8:
        if base_h is not None:
            h0 = base_h + rng.normal(scale=0.2, size=n)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = proj_box_sum(h0, target)
        else:
            h0 = rng.random(n)
            h0 = proj_box_sum(h0, target)

        # continuous refinement
        h_opt = gradient_descent_schedule(h0, betas, deadline)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, n

        # binary rounding + hill climb
        if n % 2 == 0:
            k = int(target)
            idx = np.argpartition(h_opt, -k)[-k:]
            h_bin = np.zeros_like(h_opt)
            h_bin[idx] = 1.0
            c_bin = compute_c5(h_bin)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin.copy(), c_bin, n
            # local improvement on binary
            hb, cb = binary_hill_climb(h_bin.copy(), rng, deadline)
            if cb < best_c:
                best_h, best_c, best_n = hb.copy(), cb, n

        restart += 1

    # ----- optional up‑sampling -----
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = proj_box_sum(h_up, n_up / 2.0)
        h_up = gradient_descent_schedule(h_up, betas, deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # final safety projection
    if best_h is None:
        best_n = n
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_h = proj_box_sum(best_h, target)

    best_h = proj_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
