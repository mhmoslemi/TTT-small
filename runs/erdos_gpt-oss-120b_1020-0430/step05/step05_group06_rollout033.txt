import time

from scipy.optimize import minimize

# ---------- helpers ----------
def compute_c5(h):
    """Exact C5 for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Exact sum correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    beta_corr = CURRENT_BETA * corr
    m = np.max(beta_corr)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(beta_corr - m)))) / CURRENT_BETA

    TARGET = 0.3808  # early‑stop target

    # pre‑computed shift indices
    k_vals = np.arange(-(n - 1), n, dtype=int)

    # ---- simulated annealing ----
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        # linear schedules
        T = 0.01 * (1.0 - frac) + 1e-5 * frac
        max_delta = 1.0 * (1.0 - frac) + 5e-4 * frac

        # pick a pair (i,j) with i!=j
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # admissible delta range (preserve sum)
        up_i = 1.0 - h_cur[i]
        down_j = h_cur[j]
        if up_i <= 0 and down_j <= 0:
            continue
        delta = rng.uniform(-1.0, 1.0) * max_delta
        if delta > up_i:
            delta = up_i
        elif delta < -down_j:
            delta = -down_j
        if abs(delta) < 1e-12:
            continue

        # compute a_i and a_j (vector of length 2n-1)
        idx1_i = i + k_vals
        term1_i = np.where((0 <= idx1_i) & (idx1_i < n), 1.0 - h_cur[idx1_i], 0.0)
        idx2_i = i - k_vals
        term2_i = np.where((0 <= idx2_i) & (idx2_i < n), h_cur[idx2_i], 0.0)
        a_i = term1_i - term2_i

        idx1_j = j + k_vals
        term1_j = np.where((0 <= idx1_j) & (idx1_j < n), 1.0 - h_cur[idx1_j], 0.0)
        idx2_j = j - k_vals
        term2_j = np.where((0 <= idx2_j) & (idx2_j < n), h_cur[idx2_j], 0.0)
        a_j = term1_j - term2_j

        diff_a = a_i - a_j
        corr_new = corr_cur + delta * diff_a
        max_new = corr_new.max()
        dF = max_new - max_cur

        if dF <= 0.0 or rng.random() < np.exp(-dF / max(T, 1e-12)):
            # accept move
            h_cur[i] += delta
            h_cur[j] -= delta
            # clamp tiny numerical overshoot
            if h_cur[i] > 1.0:
                h_cur[i] = 1.0
            if h_cur[i] < 0.0:
                h_cur[i] = 0.0
            if h_cur[j] > 1.0:
                h_cur[j] = 1.0
            if h_cur[j] < 0.0:
                h_cur[j] = 0.0
            corr_cur = corr_new
            max_cur = max_new
            if max_cur < best_max:
                best_max = max_cur
                best_h = h_cur.copy()
                best_c = best_max * (2.0 / n)
                if best_c <= TARGET:
                    break

    # ---- smooth‑max refinement (if time permits) ----
    if time.time() < deadline:
        # ensure feasibility before SLSQP
        adjust_sum_exact(best_h, rng)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() >= deadline:
                break
            max_iter = 500 if beta < 200 else 800
            try:
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            cur_c = compute_c5(best_h)
            if cur_c < best_c:
                best_c = cur_c
                best_max = best_c * n / 2.0
            if best_c <= TARGET:
                break

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
