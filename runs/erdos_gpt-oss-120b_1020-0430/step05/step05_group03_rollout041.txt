def gradient_descent_refine(h, rng, time_limit, start_time):
    cur_c = compute_c5(h)
    step_factor = 0.9
    max_iter = 2000
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        grad = worst_gradient(h)
        # compute max feasible alpha to stay within [0,1]
        pos = grad > 1e-12
        neg = grad < -1e-12
        # For positive grad, we will subtract alpha*grad => decrease h_i.
        # max alpha from h_i >=0: alpha <= h_i / grad_i
        # For negative grad, subtracting negative => increase, bound by 1 - h_i.
        max_alpha = np.inf
        if np.any(pos):
            alphas = h[pos] / grad[pos]
            max_alpha = min(max_alpha, alphas.min())
        if np.any(neg):
            alphas = (1 - h[neg]) / (-grad[neg])
            max_alpha = min(max_alpha, alphas.min())
        if max_alpha == np.inf:
            break
        # Choose alpha = step_factor * max_alpha
        alpha = step_factor * max_alpha
        # Clip alpha to a max step like 0.2 to avoid too big change
        alpha = min(alpha, 0.2)
        # candidate
        h_new = h - alpha * grad
        # enforce sum (mass may change)
        h_new = enforce_sum(h_new, rng)
        new_c = compute_c5(h_new)
        if new_c < cur_c - 1e-12:
            h, cur_c = h_new, new_c
            # maybe increase step_factor
            step_factor = min(step_factor * 1.05, 0.95)
        else:
            # reduce step_factor
            step_factor = max(step_factor * 0.5, 1e-3)
        if step_factor < 1e-6:
            break
    return h, cur_c

def lbfgsb_optimize(h, rng, target, beta, max_iter, time_limit, start_time):
    n = h.size
    lam = (beta * 1e-2)  # penalty weight, tune
    def obj(x):
        # x may be outside [0,1] during L-BFGS internal? Bound enforced.
        corr = np.correlate(x, 1 - x, mode='full')
        # softmax
        mb = np.max(beta * corr)
        sm = (mb + np.log(np.sum(np.exp(beta * corr - mb)))) / beta
        return sm + lam * (np.sum(x) - target) ** 2
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                   options={'maxfun': max_iter, 'maxiter': max_iter,
                            'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, rng)
    c = compute_c5(x)
    return x, c

import time
import numpy as np
from scipy.optimize import minimize

# ----- basic utilities -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop – fast enough for our sizes
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k != 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ----- projected gradient descent with line search --------------------
def gradient_descent_refine(h: np.ndarray, rng: np.random.Generator,
                           start_time: float, deadline: float) -> tuple:
    """Simple projected gradient descent targeting the max‑overlap."""
    cur_c = compute_c5(h)
    step_factor = 0.9
    max_iters = 2000
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        grad = worst_gradient(h)

        # maximal feasible α to stay inside [0,1]
        pos = grad > 1e-12
        neg = grad < -1e-12
        alpha_max = np.inf
        if np.any(pos):
            alpha_max = min(alpha_max, (h[pos] / grad[pos]).min())
        if np.any(neg):
            alpha_max = min(alpha_max, ((1 - h[neg]) / (-grad[neg])).min())
        if not np.isfinite(alpha_max) or alpha_max <= 0:
            break

        alpha = step_factor * alpha_max
        alpha = min(alpha, 0.2)            # avoid too big jumps
        h_new = h - alpha * grad
        h_new = enforce_sum(h_new, rng)
        new_c = compute_c5(h_new)

        if new_c < cur_c - 1e-12:
            h, cur_c = h_new, new_c
            step_factor = min(step_factor * 1.05, 0.95)
        else:
            step_factor = max(step_factor * 0.5, 1e-4)
        if step_factor < 1e-6:
            break
    return h, cur_c


# ----- main driver ----------------------------------------------------
