def enforce_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to add mass
        # find slack = 1 - h
        slack = 1 - h
        total_slack = slack.sum()
        # allocate proportionally
        h += diff * (slack / total_slack)

        total_mass = mass.sum()
        h -= (-diff) * (mass / total_mass)
    # due to numeric errors, clip
    np.clip(h, 0, 1, out=h)

    # loop few times if needed
    return h

def enforce_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        slack = 1 - h
        total_slack = slack.sum()
        if total_slack < 1e-12:
            # cannot add, adjust via random point
            # fallback to random redistribution
            i = np.random.randint(len(h))
            h[i] = min(1, h[i] + diff)
            return h
        h += diff * slack / total_slack
    else:
        mass = h
        total_mass = mass.sum()
        if total_mass < 1e-12:
            i = np.random.randint(len(h))
            h[i] = max(0, h[i] + diff)
            return h
        h += diff * mass / total_mass  # diff negative
    # clip bounds due to numeric errors
    np.clip(h, 0, 1, out=h)
    # recompute diff and repeat if needed (rare)
    # simple loop max 5 times
    for _ in range(5):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1 - h
            total_slack = slack.sum()
            if total_slack == 0:
                break
            h += diff * slack / total_slack
        else:
            mass = h
            total_mass = mass.sum()
            if total_mass == 0:
                break
            h += diff * mass / total_mass
        np.clip(h, 0, 1, out=h)
    return h
```

    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)

    conv = conv[:2*n-1]
    idx = np.argmax(conv)
    max_corr = conv[idx]
    shift = idx - (n - 1)
    k = abs(shift)
    if k == 0:
        grad = 1 - 2 * h
    else:
        grad = np.zeros_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return max_corr, grad

def slsqp_optimize(h0, beta, max_iter=500):
    n = h0.size
    target = n / 2
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax_obj(x):
        corr = np.fft.rfft(x, 2*n) * np.conj(np.fft.rfft(1 - x, 2*n))

        # log-sum-exp
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def enforce_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # add or subtract proportionally
    if diff > 0:
        slack = 1 - h
        total_slack = slack.sum()
        if total_slack > 1e-12:
            h += diff * slack / total_slack
    else:
        mass = h
        total_mass = mass.sum()
        if total_mass > 1e-12:
            h += diff * mass / total_mass   # diff negative
    # clip after adjustment
    np.clip(h, 0.0, 1.0, out=h)
    # small residual fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute remainder uniformly among non-saturated indices
        idx = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if idx.size > 0:
            h[idx] += diff / idx.size
            np.clip(h, 0.0, 1.0, out=h)
    return h
```

import time, numpy as np
from scipy.optimize import minimize, LinearConstraint

def compute_c5_fft(h):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)
    conv = np.fft.irfft(H * np.conj(G), L)[:2 * n - 1]
    return conv.max() * (2.0 / n)

def corr_and_worst_gradient(h):
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)
    conv = np.fft.irfft(H * np.conj(G), L)[:2 * n - 1]
    idx = np.argmax(conv)
    max_corr = conv[idx]
    shift = idx - (n - 1)
    k = abs(shift)
    if k == 0:
        grad = 1 - 2 * h
    else:
        grad = np.zeros_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return max_corr, grad

def enforce_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        slack = 1 - h
        tot = slack.sum()
        if tot > 1e-12:
            h += diff * slack / tot
    else:
        mass = h
        tot = mass.sum()
        if tot > 1e-12:
            h += diff * mass / tot
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if idx.size:
            h[idx] += diff / idx.size
            np.clip(h, 0.0, 1.0, out=h)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target)

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

class SoftmaxObjective:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
        self.L = 2 * n
    def __call__(self, x):
        H = np.fft.rfft(x, self.L)
        G = np.fft.rfft(1 - x, self.L)
        conv = np.fft.irfft(H * np.conj(G), self.L)[:2 * self.n - 1]
        bc = self.beta * conv
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / self.beta

def slsqp_optimize(h0, beta, max_iter=500):
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    lc = LinearConstraint(np.ones((1, n)), [target], [target])
    obj = SoftmaxObjective(beta, n)
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=[lc],
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, target)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target = n / 2.0

    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target)
    else:
        h = random_feasible_vector(n, rng)

    best_h = h.copy()
    best_c = compute_c5_fft(best_h)

    step_frac = 0.5
    last_improve = 0
    it = 0

    # projected gradient descent on worst shift
    while time.time() - start < 0.75 * budget_s:
        max_corr, grad = corr_and_worst_gradient(h)
        c = max_corr * (2.0 / n)
        if c < best_c - 1e-12:
            best_c, best_h = c, h.copy()
            last_improve = it
        d = -grad
        pos = d > 1e-12
        neg = d < -1e-12
        max_step = np.inf
        if np.any(pos):
            max_step = min(max_step, np.min((1 - h[pos]) / d[pos]))
        if np.any(neg):
            max_step = min(max_step, np.min((-h[neg]) / d[neg]))
        if max_step == np.inf or max_step <= 0:
            break
        step = step_frac * max_step
        if step < 1e-12:
            break
        h = h + step * d
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum(h, target)

        # adapt step size
        if c < best_c + 1e-6:
            step_frac = min(step_frac * 1.01, 0.5)
        else:
            step_frac = max(step_frac * 0.99, 0.02)

        it += 1
        if it % 5000 == 0:
            i, j = rng.integers(0, n, size=2)
            eps = min(0.01, h[i], 1 - h[j])
            if eps > 0:
                h[i] -= eps
                h[j] += eps
                h = enforce_sum(h, target)
        if it - last_improve > 20000:
            h = best_h.copy()
            for _ in range(10):
                i, j = rng.integers(0, n, size=2)
                eps = rng.random() * 0.05
                eps = min(eps, h[i], 1 - h[j])
                if eps > 0:
                    h[i] -= eps
                    h[j] += eps
            h = enforce_sum(h, target)
            last_improve = it

    # SLSQP refinement with increasing beta
    for beta in (20., 80., 200., 800., 2000., 8000., 20000.):
        if time.time() - start > 0.90 * budget_s:
            break
        maxit = 400 if beta < 200 else 600
        h_s = slsqp_optimize(best_h, beta, max_iter=maxit)
        c = compute_c5_fft(h_s)
        if c < best_c:
            best_c, best_h = c, h_s.copy()

    # final stochastic tweaks
    while time.time() - start < budget_s * 0.998:
        _, grad = corr_and_worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5_fft(nh)
        if nc < best_c:
            best_c, best_h = nc, nh.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target)
    final_c = compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c), int(n)
