def binary_hill_climb(n, rng, max_no_improve=2000, max_swaps=2000):
    # random binary with n/2 ones
    h = np.zeros(n, dtype=float)
    ones_idx = rng.choice(n, n//2, replace=False)
    h[ones_idx] = 1.0
    zeros_idx = np.setdiff1d(np.arange(n), ones_idx)
    c = np.correlate(h, 1-h, mode='full')
    cur_max = c.max()
    no_improve = 0
    swaps = 0
    while no_improve < max_no_improve and swaps < max_swaps:
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i], h[j] = h[j], h[i]  # i becomes 0, j becomes 1
        c_new = np.correlate(h, 1-h, mode='full')
        new_max = c_new.max()
        if new_max < cur_max - 1e-12:  # improvement
            cur_max = new_max
            c = c_new
            # update lists
            ones_idx = ones_idx[ones_idx != i]
            zeros_idx = zeros_idx[zeros_idx != j]
            # add new ones/zeros
            ones_idx = np.append(ones_idx, j)
            zeros_idx = np.append(zeros_idx, i)
            no_improve = 0
            swaps += 1
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            no_improve += 1
    return h, cur_max
```

import time
import numpy as np

# ------------------------------------------------------------
# helper utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """C5 value for a given step function (array) h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Exactly enforce Σh = n/2 while staying inside [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple deterministic repair; loop limited to avoid infinite loop
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)          # shift giving max overlap
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # exact sum repair
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def binary_hill_climb(n: int, rng: np.random.Generator,
                     deadline: float, max_no_improve: int = 3000,
                     max_swaps: int = 5000) -> tuple:
    """
    Random binary start (weight n/2) then greedy swaps that improve the
    maximal overlap. Returns the binary array and its maximal (uncscaled)
    overlap count.
    """
    # initialise random mask
    mask = np.zeros(n, dtype=bool)
    mask[rng.choice(n, n // 2, replace=False)] = True
    h = mask.astype(float)

    cur_corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = cur_corr.max()
    no_improve = 0
    swaps = 0

    while (no_improve < max_no_improve and swaps < max_swaps
           and time.time() < deadline):
        i = rng.integers(n)
        if not mask[i]:
            continue
        j = rng.integers(n)
        if mask[j]:
            continue
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        mask[i] = False
        mask[j] = True

        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = new_corr.max()

        if new_max < cur_max - 1e-12:
            cur_max = new_max
            cur_corr = new_corr
            no_improve = 0
            swaps += 1
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            mask[i] = True
            mask[j] = False
            no_improve += 1

    return h, cur_max
