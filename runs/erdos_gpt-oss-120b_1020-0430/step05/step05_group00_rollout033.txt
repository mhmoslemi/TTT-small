def compute_c5_fft(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, L)
    H_rev = np.fft.rfft(h[::-1], L)  # reversed
    conv = np.fft.irfft(H * H_rev, L)
    # conv length L. The part indices n to n+n-2 correspond to positive shifts 1..n-1
    start = n
    end = n + n - 1
    # to ensure we have enough length, maybe L >= end+1.
    corr_vals = conv[start:end]  # shape n-1
    min_corr = np.min(corr_vals)
    return 1 - (2.0 / n) * min_corr

def softmax_obj(x):
    # compute correlation via FFT
    n = x.size
    # compute corr for shifts k=1..n-1
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(x, L)
    H_rev = np.fft.rfft(x[::-1], L)
    conv = np.fft.irfft(H * H_rev, L)
    corr = conv[n: n + n - 1]   # length n-1
    v = -corr  # negative corr
    beta = CURRENT_BETA
    # compute softmax = (1/beta) * log sum exp(beta * v)
    max_beta_v = np.max(beta * v)
    sum_exp = np.sum(np.exp(beta * v - max_beta_v))
    f = (max_beta_v + np.log(sum_exp)) / beta
    return f

def softmax_grad(x):
    n = x.size
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(x, L)
    H_rev = np.fft.rfft(x[::-1], L)
    conv = np.fft.irfft(H * H_rev, L)
    corr = conv[n: n + n - 1] # length n-1
    v = -corr
    beta = CURRENT_BETA
    beta_v = beta * v
    max_beta_v = np.max(beta_v)
    exp_shifted = np.exp(beta_v - max_beta_v)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp  # probabilities for shifts 1..n-1
    # gradient = -sum_k p_k * (h shifted left k + h shifted right k)
    grad = np.zeros_like(x)
    # For each shift k, we need to subtract p_k * (h[i - k] + h[i + k])
    # We'll vectorize with loop over k
    for k in range(1, n):
        w = p[k-1]
        # left part: i >= k
        grad[k:] -= w * x[:-k]
        # right part: i < n - k
        grad[:n - k] -= w * x[k:]
    return grad

This gradient loop O(n^2). For n=512, 512*512/2 ~ 130k operations per gradient, ok. SLSQP may call gradient many times; within each gradient call we also compute FFT for corr, but that's negligible. So overall fine.

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using FFT: 1 - (2/n)*min_{k>0} corr_k, where corr_k = sum_i h_i h_{i-k}."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    H_rev = np.fft.rfft(h[::-1], L)
    conv = np.fft.irfft(H * H_rev, L)
    corr = conv[n : n + n - 1]          # shifts 1..n-1
    min_corr = np.min(corr)
    return 1.0 - (2.0 / n) * min_corr


# soft‑max (smooth max) of O_k = n/2 - corr_k  <=> smooth -min_corr
CURRENT_BETA = None
def softmax_obj(x: np.ndarray) -> float:
    n = x.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(x, L)
    H_rev = np.fft.rfft(x[::-1], L)
    conv = np.fft.irfft(H * H_rev, L)
    corr = conv[n : n + n - 1]          # positive shifts
    v = -corr                           # we want min_corr = -max(v)
    beta = CURRENT_BETA
    beta_v = beta * v
    m = np.max(beta_v)
    s = np.sum(np.exp(beta_v - m))
    return (m + np.log(s)) / beta


def softmax_grad(x: np.ndarray) -> np.ndarray:
    n = x.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(x, L)
    H_rev = np.fft.rfft(x[::-1], L)
    conv = np.fft.irfft(H * H_rev, L)
    corr = conv[n : n + n - 1]
    v = -corr
    beta = CURRENT_BETA
    beta_v = beta * v
    m = np.max(beta_v)
    exp_shift = np.exp(beta_v - m)
    p = exp_shift / np.sum(exp_shift)          # probabilities for shifts 1..n-1
    grad = np.zeros_like(x)
    # grad = - Σ_k p_k * (h shifted left k + h shifted right k)
    for k in range(1, n):
        w = p[k - 1]
        grad[k:] -= w * x[:-k]
        grad[: n - k] -= w * x[k:]
    return grad
