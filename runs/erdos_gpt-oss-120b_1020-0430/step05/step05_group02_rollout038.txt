import time
import numpy as np
from scipy.optimize import minimize

# global placeholders for SLSQP objective
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """log‑sum‑exp surrogate for max_k ∫h(1‑h∘T_k)."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute to satisfy Σh = n/2 exactly (preserves bounds)."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 600) -> np.ndarray:
    """SLSQP on the smooth max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return adjust_sum_exact(x, rng)

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Greedy projected gradient on the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # cheap correction
            for _ in range(100):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa(h_bin: np.ndarray, rng: np.random.Generator,
              deadline: float, max_iter: int = 300000):
    """Swap‑based simulated annealing on binary vectors."""
    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst
    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    T0 = cur_worst * 0.04 + 1e-4
    decay = 0.99996
    it = 0
    stagn = 0
    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # update auto‑correlation efficiently
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        c[start_i : start_i + n] -= hi_vec
        h_bin[i] = 0

        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j + 1 :]))
        c[start_j : start_j + n] += hj_vec
        h_bin[j] = 1

        new_min = c.min()
        new_worst = target - new_min

        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_worst, cur_min = new_worst, new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst, best_h = cur_worst, h_bin.copy()
                stagn = 0
            else:
                stagn += 1
        else:
            # revert
            h_bin[j] = 0
            c[start_j : start_j + n] -= hj_vec
            h_bin[i] = 1
            c[start_i : start_i + n] += hi_vec
            stagn += 1

        if stagn > 80000:
            break
        it += 1
    return best_h, best_worst

def local_search(h: np.ndarray, rng: np.random.Generator,
                 deadline: float, max_steps: int = 60000):
    """Simple two‑coordinate random walk with annealing."""
    n = h.size
    cur_c5 = compute_c5(h)
    step = 0.07
    stagn = 0
    while time.time() < deadline and max_steps > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction: move mass from i to j
        delta_max = min(h[i], 1.0 - h[j])
        delta_min = -min(h[j], 1.0 - h[i])
        if delta_max <= delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max) * step
        if abs(delta) < 1e-12:
            continue
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            stagn = 0
        else:
            # revert
            h[i] += delta
            h[j] -= delta
            stagn += 1
        max_steps -= 1
        step = max(step * 0.9997, 1e-4)
        if stagn > 20000:
            break
    return h, cur_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None
    candidate_ns = [360, 720]  # try two resolutions if time permits

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0

        # build a few starting vectors
        starts = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            inter = np.interp(xt, xs, parent)
            starts.append(inter)                     # fractional
            idx = np.argpartition(inter, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            starts.append(bin_vec)                    # binary

        # random binary balanced
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        # random fractional (Dirichlet‑like)
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        starts.append(adjust_sum_exact(rand_frac, rng))

        # alternating binary pattern (helps avoid pathological cases)
        alt_bin = np.tile([1.0, 0.0], n // 2)
        starts.append(alt_bin)

        for base_h in starts:
            if time.time() > deadline:
                break
            h = np.clip(base_h, 0.0, 1.0)
            h = adjust_sum_exact(h, rng)

                sub_dead = min(deadline, time.time() + 0.35 * (deadline - time.time()))
                h_bin, raw_worst = binary_sa(h.copy(), rng, sub_dead)
                c5 = raw_worst * (2.0 / n)
                if c5 < best_c:

                h = h_bin.astype(float)

            # gradient refinement

            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

            # continuation of smoothed‑max SLSQP
            for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000):
                if time.time() > deadline:
                    break
                max_iter = 500 if beta < 500 else 800
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c5 = compute_c5(h)
                if c5 < best_c:
                    best_h, best_c, best_n = h.copy(), c5, n

            # final gradient sweep
            h = gradient_refine(h, rng, step=0.02, max_iter=12)
            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

            # local random walk (post‑processing)
            sub_dead = min(deadline, time.time() + 0.12 * (deadline - time.time()))
            h, c5 = local_search(h, rng, sub_dead, max_steps=30000)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

    # optional up‑sampling of the overall best (if time permits)
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = adjust_sum_exact(h_up, rng)

        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # safety clip and final evaluation
    if best_h is None:
        # fallback: simple balanced binary interval
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = adjust_sum_exact(best_h, rng)
        best_n = n
        best_c = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
