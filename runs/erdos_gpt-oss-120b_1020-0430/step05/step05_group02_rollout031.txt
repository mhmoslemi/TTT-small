def gradient_refine(h, target, rng, init_step=0.1, iters=30, decay=0.5):
    n = h.size
    step = init_step
    for it in range(iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            # compute gradient via broadcasting
            # g = zeros
            g = np.zeros_like(h)
            # for i from 0 to n-1:
            # g_i = -h_{i+k} - h_{i-k} (with periodic)
            # and also +1 from first term for all i
            # we can compute as:
            # g = 1 - roll(h, k) - roll(h, -k)
            g = 1.0 - np.roll(h, k) - np.roll(h, -k)
        # step
        h = h - step * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # project sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust a random index up/down
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        step *= decay
    return h
```

def local_search_swap(h, target, rng, deadline, max_iter=20000):
    n = h.size

            break

        if i == j: continue
        # propose small delta
        delta = (rng.random() - 0.5) * 0.02  # between -0.01 and 0.01
        # adjust i up, j down to preserve sum: add delta to i, subtract delta from j
        if h[i] + delta < 0 or h[i] + delta > 1 or h[j] - delta < 0 or h[j] - delta > 1:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # ensure sum unchanged
        new_c5 = compute_c5(h_new)
        # accept if better
        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = new_c5
        else:
            # maybe accept with prob when own temperature decaying
            pass
    return h, cur_c5

def smooth_max_optimize(h, target, rng, beta_seq, step0=0.1, steps_per_beta=20, decay=0.9):
    n = h.size
    for beta in beta_seq:
        # compute f = corr = np.correlate(h, 1 - h, mode='full')[n-1:]
        for s in range(steps_per_beta):
            # compute correlation vector
            corr_full = np.correlate(h, 1.0 - h, mode='full')
            f = corr_full[n-1:]  # length n
            max_f = np.max(f)
            exp_beta = np.exp(beta * (f - max_f))
            Z = exp_beta.sum()
            w = exp_beta / Z  # length n
            # compute roll_sum1 and roll_sum2 via FFT
            H = np.fft.rfft(h)
            W = np.fft.rfft(w)
            roll1 = np.fft.irfft(H * W, n=n)
            w_rev = w[::-1]  # reverse
            W_rev = np.fft.rfft(w_rev)
            roll2 = np.fft.irfft(H * W_rev, n=n)
            grad = 1.0 - (roll1 + roll2)
            # step
            step = step0 * (decay ** (beta * 0 + s))  # maybe decreasing overall
            h = h - step * grad
            # clip
            h = np.clip(h, 0.0, 1.0)
            # sum adjust
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                # add diff to random index that can accommodate
                for _ in range(100):
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
    return h

def local_search_swap(h, target, rng, deadline, max_iter=5000):
    cur_c5 = compute_c5(h)
    n = h.size
    while max_iter > 0 and time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            max_iter -= 1
            continue
        # propose delta
        delta = (rng.random() - 0.5) * 0.02  # range approx -0.01 to 0.01
        if h[i] + delta < 0 or h[i] + delta > 1 or h[j] - delta < 0 or h[j] - delta > 1:
            max_iter -= 1
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # sum unchanged
        new_c5 = compute_c5(h_new)
        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
        max_iter -= 1
    return h, cur_c5
```

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_sum(h, target, rng):
    """Adjust h so that sum(h) == target, preserving bounds."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # simple random fix-up
    for _ in range(2000):
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def random_fractional_start(n, target, rng):
    h = rng.random(n)
    h *= target / h.sum()
    return project_sum(h, target, rng)

def binary_random_start(n, target, rng):
    h = np.zeros(n)
    ones = rng.choice(n, int(target), replace=False)
    h[ones] = 1.0
    return h

def gradient_refine(h, target, rng, init_step=0.05, iters=40, decay=0.5):
    n = h.size
    step = init_step
    for _ in range(iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = 1.0 - np.roll(h, k) - np.roll(h, -k)
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        h = project_sum(h, target, rng)
        step *= decay
    return h

def smooth_max_optimize(h, target, rng,
                        beta_seq=(10, 50, 200, 1000, 5000),
                        step0=0.02, steps_per_beta=30, decay=0.9):
    n = h.size
    for beta in beta_seq:
        for s in range(steps_per_beta):
            H = np.fft.fft(h)
            G = np.fft.fft(1.0 - h)
            f = np.real(np.fft.ifft(H * np.conj(G)))  # length n
            max_f = np.max(f)
            exp_beta = np.exp(beta * (f - max_f))
            Z = exp_beta.sum()
            w = exp_beta / Z
            W = np.fft.fft(w)
            roll1 = np.real(np.fft.ifft(W * H))
            roll2 = np.real(np.fft.ifft(np.conj(W) * H))
            grad = 1.0 - (roll1 + roll2)
            # normalize gradient to keep step size reasonable
            gmax = np.max(np.abs(grad))
            if gmax > 0:
                grad = grad / gmax
            step = step0 * (decay ** s)
            h = h - step * grad
            h = np.clip(h, 0.0, 1.0)
            h = project_sum(h, target, rng)
    return h

def local_search_swap(h, target, rng, deadline, max_iter=4000):
    cur_c5 = compute_c5(h)
    n = h.size
    while max_iter > 0 and time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            max_iter -= 1
            continue
        delta = (rng.random() - 0.5) * 0.02  # [-0.01,0.01]
        if not (0.0 <= h[i] + delta <= 1.0 and 0.0 <= h[j] - delta <= 1.0):
            max_iter -= 1
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = compute_c5(h_new)
        if new_c5 < cur_c5:
            h = h_new
            cur_c5 = new_c5
        max_iter -= 1
    return h, cur_c5

# Optional SLSQP refinement (similar to previous solution)
CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c5 = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    candidate_ns = [180, 240, 360, 480, 600]
    restarts_per_n = 6

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        for rep in range(restarts_per_n):
            if time.time() > deadline:
                break
            # choose start
            if rep == 0:
                h = random_fractional_start(n, target, rng)
            elif rep == 1:
                h = binary_random_start(n, target, rng)
            elif rep == 2 and parent is not None:
                # interpolate parent to this n
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(xt, xs, parent)
                h = project_sum(h, target, rng)
            elif rep == 3:
                # alternating binary pattern
                h = np.tile([1.0, 0.0], n // 2).astype(float)
                h = project_sum(h, target, rng)
            else:
                # random fractional again
                h = random_fractional_start(n, target, rng)

            # coarse gradient refinement
            h = gradient_refine(h, target, rng, init_step=0.05, iters=40, decay=0.5)

            # smooth‑max gradient descent
            h = smooth_max_optimize(h, target, rng,
                                    beta_seq=(10, 50, 200, 1000, 5000),
                                    step0=0.02, steps_per_beta=30, decay=0.9)

            # local swap search
            h, _ = local_search_swap(h, target, rng, deadline, max_iter=4000)

            # optional SLSQP fine‑tuning (short run)
            if time.time() < deadline:
                for beta in (2000, 5000, 10000):
                    h = slsqp_optimize(h, beta, rng, max_iter=200)

            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h, best_c5, best_n = h.copy(), c5, n

    # Upsample the best solution if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        target_up = n_up / 2.0
        h_up = project_sum(h_up, target_up, rng)

        h_up = gradient_refine(h_up, target_up, rng, init_step=0.05, iters=50, decay=0.5)
        h_up = smooth_max_optimize(h_up, target_up, rng,
                                   beta_seq=(10, 50, 200, 1000, 5000, 10000),
                                   step0=0.02, steps_per_beta=30, decay=0.9)
        h_up, _ = local_search_swap(h_up, target_up, rng, deadline, max_iter=4000)
        for beta in (2000, 5000, 10000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=200)
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_h, best_c5, best_n = h_up, c_up, n_up

    # fallback safe solution
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_sum(best_h, n / 2.0, rng)
        best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, best_n / 2.0, rng)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_n)
