def simulated_annealing_binary(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    # precompute ones and zeros lists
    # We'll maintain arrays of indices of ones and zeros as Python lists (or np.where dynamic each iteration). Since after each swap we need to update, use sets for O(1) updates.
    ones = set(np.flatnonzero(h))
    zeros = set(np.flatnonzero(1 - h))

    # Temperature schedule: start = 0.001 maybe, end = 1e-6. Use exponential cooling based on time fraction.
    init_temp = 0.002
    final_temp = 1e-6

    while time.time() < deadline:
        # compute temp based on elapsed fraction
        t_frac = (time.time() - start) / (deadline - start)
        temp = init_temp * (final_temp / init_temp) ** t_frac  # exponential

        # propose swap
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        # update sets
        ones.remove(i)
        zeros.add(i)
        zeros.remove(j)
        ones.add(j)

        new_c = compute_c5(h)

        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones.add(i)
            zeros.remove(i)
            zeros.add(j)
            ones.remove(j)

        # optional: break early if no improvement after many tries

    return best_h, best_c

def anneal_binary(h0, rng, deadline, start_time):
    n = h0.size
    dx = 2.0/n
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    # initial temperature guess: difference maybe up to 0.1? So set high.
    init_temp = 0.005
    final_temp = 1e-6
    total_time = deadline - start_time
    # Pre compute list of indices of zeros and ones to choose quickly: use Python lists.
    ones = [int(i) for i, val in enumerate(h) if val == 1]
    zeros = [int(i) for i, val in enumerate(h) if val == 0]

    # For faster choice, use rng.integers(len(list)) then index.
    # We'll maintain lists after each swap.
    while True:
        now = time.time()
        if now >= deadline:
            break
        # time fraction elapsed from start to deadline
        elapsed = now - start_time
        t_frac = elapsed / total_time
        temp = init_temp * (final_temp / init_temp) ** t_frac  # exponential cooling

        # random swap
        # Choose i from ones, j from zeros, if any list empty skip
        if not ones or not zeros:
            break
        i = ones[rng.integers(0, len(ones))]
        j = zeros[rng.integers(0, len(zeros))]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0

        # update lists
        # replace i in ones with j; and j in zeros with i
        # Find positions in lists: O(1) if we store positions dictionary, but n small.
        # We'll remove and append.
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)

        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            # revert lists
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
        # optional early exit if best_c already below target and we have time

def anneal_binary(h0, rng, deadline, start_time):
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # init and final temp
    init_temp = 0.01
    final_temp = 1e-6
    total_time = deadline - start_time
    if total_time <= 0:
        return best_h, best_c
    # prepare lists
    ones = [int(i) for i in range(n) if h[i] == 1.0]
    zeros = [int(i) for i in range(n) if h[i] == 0.0]

    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start_time
        t_frac = elapsed / total_time
        temp = init_temp * (final_temp / init_temp) ** t_frac
        # pick random indices
        i_pos = rng.integers(0, len(ones))
        j_pos = rng.integers(0, len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # update lists
        ones[i_pos] = j
        zeros[j_pos] = i
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones[i_pos] = i
            zeros[j_pos] = j
        # optional break if best_c lower than 0.3808 and near deadline? Not needed
    return best_h, best_c

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def compute_c5_binary(h):
    n = h.size
    self_corr = np.correlate(h, h, mode="full")
    min_corr = self_corr.min()
    return (2.0 / n) * (n / 2 - min_corr)

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def fractional_to_binary(h, rng):
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    adjust_sum_exact(bin_h, rng)
    return bin_h

def binary_targeted_improve(h, rng, deadline):
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        neigh_one = neigh[ones]
        i = ones[np.argmin(neigh_one)]
        neigh_zero = neigh[zeros]
        j = zeros[np.argmax(neigh_zero)]
        delta = neigh[i] - neigh[j]
        if delta < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

def make_binary_seed(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def alternating_seed(n):
    return (np.arange(n) % 2).astype(float)

def step_seed(n):
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    return h

def anneal_binary(h0, rng, deadline, start_time):
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5_binary(h)
    best_h = h.copy()
    best_c = cur_c
    init_temp = 0.01
    final_temp = 1e-6
    total_time = max(deadline - start_time, 1e-12)
    ones = [int(i) for i in range(n) if h[i] == 1.0]
    zeros = [int(i) for i in range(n) if h[i] == 0.0]
    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start_time
        t_frac = elapsed / total_time
        temp = init_temp * (final_temp / init_temp) ** t_frac
        i_pos = rng.integers(0, len(ones))
        j_pos = rng.integers(0, len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]
        h[i] = 0.0
        h[j] = 1.0
        ones[i_pos] = j
        zeros[j_pos] = i
        new_c = compute_c5_binary(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones[i_pos] = i
            zeros[j_pos] = j
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
    else:
        parent = None
        n = 360

    best_h = None
    best_c = float("inf")
    best_n = n

    seeds = []
    if parent is not None:
        seeds.append(fractional_to_binary(parent, rng))
    seeds.append(alternating_seed(n))
    seeds.append(step_seed(n))
    for _ in range(4):
        seeds.append(make_binary_seed(n, rng))

    for idx, h0 in enumerate(seeds):
        if time.time() >= deadline:
            break
        adjust_sum_exact(h0, rng)
        c0 = compute_c5(h0)
        if c0 < best_c:
            best_h, best_c, best_n = h0.copy(), c0, n
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        anneal_time = min(15.0, remaining * 0.25)
        seed_deadline = time.time() + anneal_time
        h_ann, c_ann = anneal_binary(h0, rng, seed_deadline, start)
        if c_ann < best_c:
            best_h, best_c, best_n = h_ann.copy(), c_ann, n

    # final long anneal on the current best
    if best_h is not None and time.time() < deadline:
        h_final, c_final = anneal_binary(best_h.copy(), rng, deadline, start)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    # optional up‑sampling refinement
    if (
        best_h is not None
        and time.time() < deadline - 5.0
        and best_n * 2 <= 720
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        h_up_ann, c_up_ann = anneal_binary(h_up, rng, deadline, start)
        if c_up_ann < best_c:
            best_h, best_c, best_n = h_up_ann, c_up_ann, n_up

    if best_h is None:
        best_n = n
        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
