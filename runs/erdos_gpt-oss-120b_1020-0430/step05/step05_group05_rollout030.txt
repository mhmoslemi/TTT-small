def compute_c5_fast(h):
    n = h.size
    dx = 2.0/n
    a = h
    b = 1.0 - h
    m = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, m)
    B = np.fft.rfft(b[::-1], m)
    conv = np.fft.irfft(A * B, m)[:2*n-1]
    return np.max(conv) * dx
```

def simulated_annealing(h0, time_limit, rng):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    n = h.size
    # Maintain list of indices for ones and zeros
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    # Temperature schedule
    T_initial = 0.0015
    T = T_initial
    cooling = 0.99995  # per iteration
    while time.time() - start < time_limit:
        p = rng.choice(ones)
        q = rng.choice(zeros)
        # perform swap
        h_candidate = h.copy()
        h_candidate[p] = 0.0
        h_candidate[q] = 1.0
        c = compute_c5(h_candidate)
        delta = c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            h = h_candidate
            cur_c = c
            # update indices
            ones.remove(p)
            zeros.append(p)
            zeros.remove(q)
            ones.append(q)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        T *= cooling
        if T < 1e-8:
            T = T_initial
    return best_h, best_c

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a binary step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def to_binary(frac_h):
    """Convert a fractional vector to a binary one with exact half ones."""
    n = frac_h.size
    idx = np.argpartition(frac_h, -n // 2)[-n // 2 :]
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx] = 1.0
    return bin_h


def greedy_local_search(h0, time_limit, rng):
    """Greedy hill climbing by swapping a 1 with a 0."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    attempts_without_improve = 0
    # maintain lists of positions
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))

    while time.time() - start < time_limit:
        p = rng.choice(ones)
        q = rng.choice(zeros)
        h_candidate = h.copy()
        h_candidate[p] = 0.0
        h_candidate[q] = 1.0
        c = compute_c5(h_candidate)
        if c < best_c - 1e-9:
            h = h_candidate
            best_h = h_candidate.copy()
            best_c = c
            ones = list(np.flatnonzero(h == 1))
            zeros = list(np.flatnonzero(h == 0))
            attempts_without_improve = 0
        else:
            attempts_without_improve += 1
        if attempts_without_improve > 10000:
            break
    return best_h, best_c


def simulated_annealing(h0, time_limit, rng):
    """Swap‑based simulated annealing."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    n = h.size
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))

    T = 0.0015          # initial temperature
    cooling = 0.99995   # per‑iteration multiplier
    min_T = 1e-8

    while time.time() - start < time_limit:
        p = rng.choice(ones)
        q = rng.choice(zeros)
        h_candidate = h.copy()
        h_candidate[p] = 0.0
        h_candidate[q] = 1.0
        c = compute_c5(h_candidate)
        delta = c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            h = h_candidate
            cur_c = c
            # update lists
            ones.remove(p)
            zeros.append(p)
            zeros.remove(q)
            ones.append(q)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        T = max(T * cooling, min_T)
        if T == min_T:
            T = 0.0015
    return best_h, best_c


def adjust_sum_exact(h, target):
    """Force Σh = target while staying in {0,1}."""
    diff = int(target - h.sum())
    if diff == 0:
        return
    if diff > 0:
        zeros = list(np.flatnonzero(h == 0))
        for i in zeros[:diff]:
            h[i] = 1.0
    else:
        ones = list(np.flatnonzero(h == 1))
        for i in ones[: -diff]:
            h[i] = 0.0
