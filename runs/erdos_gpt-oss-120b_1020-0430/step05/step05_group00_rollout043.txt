def apply_swap(a, C, i, j):
    # a: binary array (np.uint8)
    # C: C_pos array (size n) representing autocorrelation for k>=0.
    # Compute updates
    n = a.shape[0]
    # removal subtractions
    # k from 1 to n-1-i:
    if i < n-1:
        C[1:n-i] -= a[i+1:]  # a[i+1 :] length n-1-i
    # k from 1 to i:
    if i > 0:
        C[1:i+1] -= a[i-1::-1]  # shape i
    # addition addition (using original a before modifications)
    if j < n-1:
        C[1:n-j] += a[j+1:]
    if j > 0:
        C[1:j+1] += a[j-1::-1]
    # flip bits
    a[i] = 0
    a[j] = 1

def project_onto_box_simplex(v, target):
    # project v onto {x: sum x = target, 0 ≤ x ≤ 1}
    # We can use bisection on λ.
    n = v.size

    for _ in range(30):

        x = np.clip(v + lam, 0, 1)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo+hi)/2, 0, 1)

import time

# --------------------------------------------------------------
def project_onto_box_simplex(v: np.ndarray, target: float, max_iter: int = 50) -> np.ndarray:
    """Project v onto {x | sum(x)=target, 0<=x<=1} using bisection on λ."""
    lo = -np.max(v)          # lowest λ to keep >=0
    hi = 1.0 - np.min(v)     # highest λ to keep <=1
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def compute_OC(h: np.ndarray):
    """Return O_k array and C_k array for vector h (step function on [0,2])."""
    n = h.size
    n_fft = 2 * n
    h_fft = np.fft.fft(h, n_fft)
    # autocorrelation C_k = Σ_i h_i h_{i+k}
    c_fft = h_fft * np.conj(h_fft)
    c_full = np.fft.ifft(c_fft).real
    C = c_full[n - 1:n - 1 + n]            # C[0] .. C[n-1]
    # A_k = Σ_{i=0}^{n-1-k} h_i
    cum = np.cumsum(h)
    A = cum[::-1]                          # A[0]=total sum, A[k]=sum up to n-1-k
    O = A - C
    return O, C


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for a step function on [0,2]."""
    O, _ = compute_OC(h)
    return O.max() * (2.0 / h.size)


def worst_gradient(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of O_k (the worst shift) w.r.t. h."""
    n = h.size
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros_like(h)
    g[: n - k] = 1.0 - h[k:]
    g[k:] -= h[: n - k]
    return g
