def mutation_swap(ind, rng, swaps=1):
    n = ind.shape[0]
    for _ in range(swaps):
        ones = np.where(ind == 1)[0]
        zeros = np.where(ind == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        ind[i] = 0
        ind[j] = 1
    return ind

def binary_sa_refine(h, rng, time_limit):
    n = h.size
    target = n//2
    # compute fitness
    c_curr = compute_c5(h)
    start = time.time()
    # schedule: linear decay
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = max(0.01, (0.5 - elapsed / time_limit) * 0.5)  # start 0.5, end small
        # pick random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            c_curr = c_new
        else:
            # revert
            h[i] = 1

import time
import numpy as np

# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # Simple random redistribution; converges quickly because diff is tiny
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


# --------------------- GA utilities ----------------------------
def uniform_crossover(p1: np.ndarray, p2: np.ndarray,
                      rng: np.random.Generator) -> np.ndarray:
    mask = rng.random(p1.size) < 0.5
    child = np.where(mask, p1, p2)
    return child


def mutation_swap(ind: np.ndarray, rng: np.random.Generator,
                  swaps: int = 1) -> np.ndarray:
    n = ind.size
    for _ in range(swaps):
        ones = np.where(ind == 1)[0]
        zeros = np.where(ind == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        ind[i] = 0
        ind[j] = 1
    return ind


def tournament_select(fitness: np.ndarray,
                      rng: np.random.Generator,
                      tour_size: int = 5) -> int:
    """Return index of best individual in a random tournament."""
    candidates = rng.choice(fitness.shape[0], size=tour_size, replace=False)
    best = candidates[np.argmin(fitness[candidates])]
    return int(best)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf

    # ---- initial solution from parent (if present) ----
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size

        c = compute_c5(binary)
        best_h = binary.copy()
        best_c = c
        best_n = n
    else:
        best_n = 512
        n = best_n
        target = n // 2

    # ---- GA parameters ----
    pop_size = 200
    tour_size = 5
    # Initialise population with random balanced binaries
    pop = np.empty((pop_size, n), dtype=np.uint8)
    pop_fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        ones = rng.choice(n, target, replace=False)
        ind = np.zeros(n, dtype=np.uint8)
        ind[ones] = 1
        pop[i] = ind
        pop_fitness[i] = compute_c5(ind)
        if pop_fitness[i] < best_c:
            best_c = pop_fitness[i]
            best_h = ind.copy()
    # Ensure the best individual is in the population
    worst_idx = np.argmax(pop_fitness)
    if best_c < pop_fitness[worst_idx]:
        pop[worst_idx] = best_h.copy()
        pop_fitness[worst_idx] = best_c

    # ---- Genetic Algorithm main loop ----
    ga_time = total_budget * 0.65
    while time.time() - start < ga_time:
        # parent selection
        p1_idx = tournament_select(pop_fitness, rng, tour_size)
        p2_idx = tournament_select(pop_fitness, rng, tour_size)
        child = uniform_crossover(pop[p1_idx], pop[p2_idx], rng)
        child = mutation_swap(child, rng, swaps=1)

        # evaluate child
        c_child = compute_c5(child)

        # replacement if better than worst
        worst_idx = np.argmax(pop_fitness)
        if c_child < pop_fitness[worst_idx]:
            pop[worst_idx] = child
            pop_fitness[worst_idx] = c_child

        # update global best
        if c_child < best_c:
            best_c = c_child
            best_h = child.copy()
            if best_c <= 0.3808:
                break

    # ---- Local refinement (hill climbing) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0.5 and best_h is not None:
        # Convert to float for gradient based hill climb
        h_float = best_h.astype(float)
        # allocate iterations roughly proportional to remaining time
        max_iters = int(min(20000, remaining * 300))
        max_iters = max(max_iters, 500)  # ensure some work
        h_float, c_hc = guided_hill_climb(h_float, rng,
                                           max_iters=max_iters,
                                           step_scale=0.7)
        if c_hc < best_c:
            best_c = c_hc
            best_h = h_float.copy()

        # Additional fine‑tuning
        if time.time() - start < total_budget * 0.95:
            h_float = gradient_refine(best_h.astype(float), rng,
                                      step=0.01, max_iter=5)
            c_ref = compute_c5(h_float)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_float.copy()

    # ---- Final sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
