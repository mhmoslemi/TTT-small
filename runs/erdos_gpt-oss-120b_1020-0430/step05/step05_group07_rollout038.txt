def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def binary_sa(h, rng, deadline):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = np.flatnonzero(cur==1)
    zeros = np.flatnonzero(cur==0)
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c-cur_c)/T):
            cur_c = c
            ones = np.where(cur==1)[0]
            zeros = np.where(cur==0)[0]
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it > 20000:
            break
    return best, best_c

def fractional_sa(h, rng, deadline):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # Transfer from i to j
        max_transfer = min(cur[i], 1.0 - cur[j])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[i] -= delta
        cur[j] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it > 20000:
            break
    return best, best_c

def gradient_multi(h, rng, step=0.05, K=5, max_iter=15):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-K:]  # largest K indices
        shifts = idx - (n-1)
        g = np.zeros_like(h)
        ar = np.arange(n)
        for s in shifts:
            g += (1.0 - h[(ar + s) % n]) - h[(ar - s) % n]
        # for s=0, both terms same => 1-2h, correct
        h = h - step * (0.5 ** it) * g
        np.clip(h, 0.0, 1.0, out=h)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # redistribute diff over random indices
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def slsqp_optimize(h0, beta, rng, max_iter=800):
    n = h0.size
    target = n/2.0
    # Use global or closure? We'll just pass inside functions.
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def cons_eq(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    constraints = {'type':'eq', 'fun':cons_eq}
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def binary_sa(h, rng, deadline):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 20000:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones[ones == i] = j
            zeros[zeros == j] = i
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
    return best, best_c

def fractional_sa(h, rng, deadline):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < 20000:
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(cur[i], 1.0 - cur[j])
        if max_transfer <= 0:
            it += 1

        delta = rng.random() * max_transfer
        cur[i] -= delta
        cur[j] += delta
        c = compute_c5(cur)

def gradient_multi(h, rng, step=0.05, K=5, max_iter=15):
    n = h.size
    target = n / 2.0
    ar = np.arange(n)
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argpartition(corr, -K)[-K:]
        shifts = idx - (n - 1)
        g = np.zeros_like(h)
        for s in shifts:
            g += (1.0 - h[(ar + s) % n]) - h[(ar - s) % n]
        h = h - step * (0.5 ** it) * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def slsqp_optimize(h0, beta, rng, max_iter=800):
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def eq_con(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq_con}
    res = minimize(obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
            h_bin, c_bin = binary_sa(h.copy(), rng, sub_dead)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin.copy(), c_bin, n
            h = h_bin.astype(float)

        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
        h_frac, c_frac = fractional_sa(h.copy(), rng, sub_dead)
        if c_frac < best_c:
            best_h, best_c, best_n = h_frac.copy(), c_frac, n
        h = h_frac

        h = gradient_multi(h, rng, step=0.05, K=5, max_iter=20)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
        for beta in betas:
            if time.time() > deadline:
                break
            max_iter = 800 if beta < 200 else 1200
            if beta >= 20000:
                max_iter = 2000
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        h = gradient_multi(h, rng, step=0.03, K=5, max_iter=15)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # main loop over resolutions
    for n in [base_n, base_n * 2]:
        if time.time() > deadline:
            break
        for h0 in generate_candidates(n):
            if time.time() > deadline:
                break
            process_candidate(h0, n)

    # optional upsampling of the current best
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        diff = n_up / 2 - h_up.sum()
        for _ in range(200):
            i = rng.integers(0, n_up)
            add = min(diff, 1.0 - h_up[i]) if diff > 0 else max(diff, -h_up[i])
            h_up[i] += add
            diff = n_up / 2 - h_up.sum()
            if abs(diff) < 1e-12:
                break
        for _ in range(2):
            if time.time() > deadline:
                break
            h_up = gradient_multi(h_up, rng, step=0.05, K=5, max_iter=20)
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
            h_up, _ = fractional_sa(h_up, rng, sub_dead)
        if time.time() < deadline:
            try:
                h_up = slsqp_optimize(h_up, beta=20000.0, rng=rng, max_iter=1500)
            except Exception:
                pass
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # deep refinement on the current best with remaining budget
    if best_h is not None and time.time() < deadline:
        best_h = deep_refine(best_h, best_n, deadline, rng)
        best_c = compute_c5(best_h)

    # final safety adjustments
    if best_h is None:
        n = base_n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        diff = best_n / 2 - best_h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, best_n)
                add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
                best_h[i] += add
                diff = best_n / 2 - best_h.sum()
                if abs(diff) < 1e-12:
                    break
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
