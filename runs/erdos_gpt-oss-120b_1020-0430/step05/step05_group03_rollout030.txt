        # compute softmax weights

        max_corr = np.max(beta * corr)
        exp_vals = np.exp(beta * corr - max_corr)
        w = exp_vals / np.sum(exp_vals)
        # compute gradient
        grad = np.zeros_like(h)

        offset0 = -(n-1)
        for idx in range(2*n-1):

import time
import numpy as np

# --------------------------------------------------------------
# basic utilities
# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Projection of v onto the set {x | 0 <= x_i <= 1, sum x_i = target}.
    Uses a binary search on the Lagrange multiplier.
    """
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return project_onto_box_simplex(h, n / 2.0)


# --------------------------------------------------------------
# gradients
# --------------------------------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_simplex(h, target)
    return h


# --------------------------------------------------------------
# smooth (soft‑max) descent
# --------------------------------------------------------------
def smooth_descent(h: np.ndarray, beta: float, step: float,
                   max_iter: int, target: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    Projected gradient descent on the soft‑max surrogate
    of max_k correlation.
    """
    n = h.size
    offset0 = -(n - 1)
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")          # size 2n‑1
        # stable soft‑max
        scaled = beta * corr
        m = np.max(scaled)
        w = np.exp(scaled - m)
        w /= w.sum()                                        # weights

        # weighted sum of per‑offset gradients
        grad = np.zeros_like(h)
        for idx, wk in enumerate(w):
            off = offset0 + idx
            L = n - abs(off)
            if off >= 0:
                grad[:L] += wk * (1 - h[off:off + L])
                grad[off:off + L] -= wk * h[:L]
            else:
                d = -off
                grad[d:] += wk * (1 - h[:L])
                grad[:L] -= wk * h[d:]
        # global scaling (dx)
        grad *= (2.0 / n)

        # projected step
        v = h - step * grad
        h = project_onto_box_simplex(v, target)
    return h


# --------------------------------------------------------------
# sub‑gradient descent on the true max
# --------------------------------------------------------------
def subgradient_descent(h: np.ndarray, steps, target: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on C5 = max_k correlation.
    steps is an iterable of step sizes (larger → smaller).
    """
    for s in steps:
        grad = worst_gradient(h)
        v = h - s * grad
        h = project_onto_box_simplex(v, target)
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)

    else:

        h0 = random_feasible_vector(n0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)
    best_n = best_h.size

    # pre‑computed step schedules
    sub_steps = np.geomspace(0.6, 0.001, num=1200)  # for sub‑gradient
    beta_schedule = [
        (5.0, 20, 0.5),
        (20.0, 20, 0.4),
        (80.0, 20, 0.3),
        (200.0, 20, 0.2),
        (800.0, 20, 0.15),
        (3000.0, 20, 0.1),
        (8000.0, 20, 0.07),
    ]

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() - start < 0.92 * budget_s:
        # sometimes change resolution
        if rng.random() < 0.15:
            n = int(rng.choice([256, 384, 448, 512]))
            if n != best_n:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = project_onto_box_simplex(h, n / 2.0)
                best_n = n
            else:
                h = best_h.copy()
        else:
            h = best_h.copy()
            # small random jitter
            if rng.random() < 0.4:
                v = h + rng.normal(scale=0.04, size=h.shape)
                h = project_onto_box_simplex(v, h.size / 2.0)

        target = h.size / 2.0

        # --------------------------------------------------------------
        # sub‑gradient phase (many cheap steps)
        # --------------------------------------------------------------
        h = subgradient_descent(h, sub_steps, target)

        # --------------------------------------------------------------
        # smooth‑max phase (higher‑quality refinement)
        # --------------------------------------------------------------
        for beta, iters, step in beta_schedule:
            if time.time() - start > 0.97 * budget_s:
                break
            h = smooth_descent(h, beta, step, iters, target, rng)

        # --------------------------------------------------------------
        # final hill‑climb & tidy‑up
        # --------------------------------------------------------------
        h, _ = guided_hill_climb(h, rng, max_iters=2500, step_scale=0.45)
        h = gradient_refine(h, rng, step=0.008, max_iter=14)

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ------------------------------------------------------------------
    # tiny stochastic tweaks until time expires
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.998:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.015)
        nc = compute_c5(nh)
        if nc < best_c:
            best_c, best_h = nc, nh.copy()

    # final sanity & exact projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_simplex(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
