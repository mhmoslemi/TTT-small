def project_h(v, target):
   # v is array
   l = np.max(v) - 1.0
   u = np.min(v)
   for _ in range(50):
       mid = (l + u) / 2.0
       x = np.clip(v - mid, 0.0, 1.0)
       s = x.sum()
       if s > target:
           l = mid
       else:
           u = mid
   return np.clip(v - (l+u)/2.0, 0.0, 1.0)

def gradient_optimize(h, target, max_iter, step_start, step_decay, min_step, deadline):
    n = h.size
    step = step_start
    best_c = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        # try step
        improved = False
        while step >= min_step:
            h_new = project_h(h - step * g, target)
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                h = h_new
                best_c = c_new
                improved = True
                break
            else:
                step *= 0.5
        if not improved:
            # can't improve with current step, break maybe
            # reduce step for next iteration
            step = max(step * 0.5, min_step)
        else:
            # after improvement, maybe increase step a bit for next iteration
            step = step_start
    return h

def fractional_sa(h, rng, deadline, max_iters=50000):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = rng.choice(n, size=2, replace=False)
        # random direction: can transfer from i to j
        if cur[i] == 0 or cur[j] == 1:
            it += 1
            continue
        max_transfer = min(cur[i], 1.0 - cur[j])
        delta = rng.random() * max_transfer
        cur[i] -= delta
        cur[j] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        it += 1
    return best, best_c

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def project_h(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1] with sum == target using water‑filling."""
    lo = np.max(v) - 1.0
    hi = np.min(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_optimize(
    h: np.ndarray,
    target: float,
    max_iter: int,
    step_start: float,
    step_decay: float,
    min_step: float,
    deadline: float,
) -> np.ndarray:
    n = h.size
    step = step_start
    best_c = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = abs(int(corr.argmax()) - (n - 1))
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        improved = False
        cur_step = step
        while cur_step >= min_step:
            h_new = project_h(h - cur_step * g, target)
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                h = h_new
                best_c = c_new
                improved = True
                break
            cur_step *= 0.5
        if not improved:
            step = max(step * 0.5, min_step)
        else:
            step = step_start * (step_decay ** it)
    return h

def fractional_sa(
    h: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray, float]:
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] == 0.0 or cur[j] == 1.0:
            it += 1
            continue
        max_transfer = min(cur[i], 1.0 - cur[j])
        delta = rng.random() * max_transfer
        cur[i] -= delta
        cur[j] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            cur[i] += delta
            cur[j] -= delta
        it += 1
    return best, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    base_ns = [240, 360, 480, 720]
    for n in base_ns:
        if time.time() > deadline:
            break
        target = n / 2.0

        start_vecs = []
        if "initial_h_values" in globals():
            xs = np.linspace(0.0, 2.0, len(initial_h_values), endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, initial_h_values)
            start_vecs.append(interp)
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            binv = np.zeros(n)
            binv[idx] = 1.0
            start_vecs.append(binv)

        # random continuous (uniform)
        u = rng.random(n)
        u *= target / u.sum()
        start_vecs.append(u)

        # random continuous (biased)
        b = rng.random(n) ** 2
        b *= target / b.sum()
        start_vecs.append(b)

        # random binary
        binv = np.zeros(n)
        binv[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(binv)

        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h = project_h(np.clip(h0, 0.0, 1.0), target)

            # gradient descent
            h = gradient_optimize(
                h,
                target,
                max_iter=200,
                step_start=0.2,
                step_decay=0.95,
                min_step=1e-5,
                deadline=deadline,
            )
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # fractional SA
            if time.time() < deadline:
                h_sa, c_sa = fractional_sa(h.copy(), rng, deadline)
                if c_sa < best_c:
                    best_h, best_c, best_n = h_sa.copy(), c_sa, n
                h = h_sa

            # final gradient refinement
            h = gradient_optimize(
                h,
                target,
                max_iter=200,
                step_start=0.1,
                step_decay=0.95,
                min_step=1e-6,
                deadline=deadline,
            )
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # up‑sampling loop (single refinement step)
    while best_h is not None and time.time() < deadline and best_n < 1080:
        n_up = min(best_n * 2, 1080)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = project_h(h_up, n_up / 2.0)

        h_up = gradient_optimize(
            h_up,
            n_up / 2.0,
            max_iter=200,
            step_start=0.2,
            step_decay=0.95,
            min_step=1e-5,
            deadline=deadline,
        )
        if time.time() < deadline:
            h_up, c_up = fractional_sa(h_up, rng, deadline)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

        h_up = gradient_optimize(
            h_up,
            n_up / 2.0,
            max_iter=200,
            step_start=0.1,
            step_decay=0.95,
            min_step=1e-6,
            deadline=deadline,
        )
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

        best_n = n_up

    if best_h is None:
        n = 240
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_h(best_h, n / 2.0)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = project_h(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
