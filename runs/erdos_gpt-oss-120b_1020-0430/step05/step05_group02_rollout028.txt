def binary_hill_climb(h_bin, rng, max_iter, no_improve_limit):
    n = len(h_bin)
    target = n//2
    c = np.correlate(h_bin, h_bin, mode='full')
    cur_min = c.min()
    best_h = h_bin.copy()
    best_min = cur_min

    ones = list(np.flatnonzero(h_bin))
    zeros = list(np.flatnonzero(1 - h_bin))
    it = 0
    no_improve = 0
    while it < max_iter and no_improve < no_improve_limit:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # update correlation for i ->0
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i+1:]))
        c[start_i:start_i+n] -= hi_vec
        # update correlation for j ->1
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j+1:]))
        c[start_j:start_j+n] += hj_vec
        # Note: after updating c, we still have h_bin[i]=1 and h_bin[j]=0; need to swap values
        h_bin[i] = 0.0
        h_bin[j] = 1.0

        new_min = c.min()
        if new_min >= cur_min:
            # accept
            cur_min = new_min
            # update ones/zeros lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_min > best_min:
                best_min = cur_min
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert changes
            # revert h_bin
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            # revert correlation: subtract added and add subtracted
            c[start_j:start_j+n] -= hj_vec
            c[start_i:start_i+n] += hi_vec
            no_improve += 1

        it += 1
    return best_h, best_min
```

def binary_optimize(n, rng, time_limit):
    target = n//2
    # start with random binary
    h = np.zeros(n)
    h[rng.choice(n, target, replace=False)] = 1.0
    c = np.correlate(h, h, mode='full')
    cur_min = c.min()
    best_h = h.copy()
    best_min = cur_min

    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))

    # parameters
    start_T = (target - cur_min) * 0.1 + 1e-4
    decay = 0.99997
    T = start_T
    it = 0
    no_improve = 0
    max_iters = 500000
    while time.time() < time_limit and it < max_iters:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # update corr for i ->0
        start_i = n - 1 - i
        hi_vec = np.concatenate((h[i::-1], h[i+1:]))
        c[start_i:start_i+n] -= hi_vec
        h[i] = 0.0

        # update corr for j ->1
        start_j = n - 1 - j
        hj_vec = np.concatenate((h[j::-1], h[j+1:]))
        c[start_j:start_j+n] += hj_vec
        h[j] = 1.0

        new_min = c.min()
        # acceptance criteria
        if new_min > cur_min or (new_min == cur_min and rng.random() < 0.001):
            # accept
            cur_min = new_min
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_min > best_min:
                best_min = cur_min
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h[j] = 0.0
            c[start_j:start_j+n] -= hj_vec
            h[i] = 1.0
            c[start_i:start_i+n] += hi_vec
            no_improve += 1

        if no_improve > 100000:
            break

        it += 1
        T = max(T * decay, 1e-8)
    return best_h, best_min
```

import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def find_improving_swap(h, k_abs, rng):
    n = h.shape[0]
    if k_abs == 0:
        return None
    # reduce to smallest equivalent shift
    if k_abs > n // 2:
        k_abs = n - k_abs
    if k_abs == 0:
        return None
    # masks for positions where swapping improves overlap for this shift
    mask1 = (h[:n - k_abs] == 1) & (h[k_abs:] == 0)
    mask0 = (h[:n - k_abs] == 0) & (h[k_abs:] == 1)
    idx1 = np.flatnonzero(mask1)
    idx0 = np.flatnonzero(mask0)
    if idx1.size == 0 or idx0.size == 0:
        return None
    i = int(rng.choice(idx1))
    j = int(rng.choice(idx0))
    return i, j

def binary_optimize_fixed_target(n, rng, time_limit, target_min_c):
    target = n // 2
    # random binary start
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, target, replace=False)] = 1.0
    c = np.correlate(h, h, mode='full')
    cur_min = int(c.min())
    best_h = h.copy()
    best_min = cur_min

    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))

    while time.time() < time_limit:
        idx = int(np.argmin(c))
        k_abs = abs(idx - (n - 1))
        # try targeted move
        move = None
        if k_abs > 0:
            move = find_improving_swap(h, k_abs, rng)
        if move is None:
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i, j = move

        # update correlation for i -> 0
        start_i = n - 1 - i
        hi_vec = np.concatenate((h[i::-1], h[i+1:]))
        c[start_i:start_i+n] -= hi_vec
        h[i] = 0.0

        # update correlation for j -> 1 (using updated h)
        start_j = n - 1 - j
        hj_vec = np.concatenate((h[j::-1], h[j+1:]))
        c[start_j:start_j+n] += hj_vec
        h[j] = 1.0

        new_min = int(c.min())
        if new_min >= cur_min:
            cur_min = new_min
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_min > best_min:
                best_min = cur_min
                best_h = h.copy()
                if best_min >= target_min_c:
                    return best_h, best_min
        else:
            # revert
            h[j] = 0.0
            c[start_j:start_j+n] -= hj_vec
            h[i] = 1.0
            c[start_i:start_i+n] += hi_vec
        # No explicit stagnation break; loop will exit on time

    return best_h, best_min

    start = time.time()
    deadline = start + 0.99 * budget_s

    # required minimal autocorrelation to beat C5=0.3808
    required_min_c = int(np.ceil(target - 0.3808 * n / 2))

    best_h = None
    best_min = -1

    # try parent if available and size matches

        parent = np.array(initial_h_values, dtype=float)

            elif diff < 0:

                idx = rng.choice(ones, int(-diff), replace=False)

            h_candidate, min_c = binary_optimize_fixed_target(n, rng, slice_end, required_min_c)
            if min_c > best_min:

                if best_min >= required_min_c:

                    final_c5 = compute_c5(best_h)
                    return best_h.tolist(), float(final_c5), int(n)

    while time.time() < deadline and best_min < required_min_c:
        remaining = deadline - time.time()
        slice_time = min(30.0, remaining)
        slice_end = time.time() + slice_time
        h_candidate, min_c = binary_optimize_fixed_target(n, rng, slice_end, required_min_c)

    """Return a swap (i from ones, j from zeros) that improves overlap for shift k,

    n = h.shape[0]
    if k == 0:

    i = int(rng.choice(idx1))
    j = int(rng.choice(idx0))
    return i, j

    Returns the best vector and its min‑correlation."""

                h[idx] = 0.0

        if move is None:
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i, j = move

        # i: 1 → 0
        start_i = n - 1 - i
        hi_vec = np.concatenate((h[i::-1], h[i + 1 :]))
        c[start_i : start_i + n] -= hi_vec
        h[i] = 0.0

        # j: 0 → 1 (uses updated h)
        start_j = n - 1 - j
        hj_vec = np.concatenate((h[j::-1], h[j + 1 :]))
        c[start_j : start_j + n] += hj_vec
        h[j] = 1.0

        new_min = int(c.min())
        if new_min >= cur_min:
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_min > best_min:
                best_min = cur_min
                best_h = h.copy()
                if best_min >= target_min_c:
                    return best_h, best_min
        else:
            # revert
            h[j] = 0.0
            c[start_j : start_j + n] -= hj_vec
            h[i] = 1.0
            c[start_i : start_i + n] += hi_vec

    return best_h, best_min

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.99 * budget_s

    n = 360
    target = n // 2
    # C5 ≤ 0.3808  →  min autocorrelation ≥ ceil(target - 0.3808 * n / 2)
    required_min_c = int(np.ceil(target - 0.3808 * n / 2))

    best_h = None
    best_min = -1

    # If a parent is supplied, start from its binary rounding.
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size == n:
            parent_bin = (parent >= 0.5).astype(float)
            slice_end = min(deadline, time.time() + 30.0)
            cand_h, cand_min = binary_optimize_fixed_target(
                n, rng, slice_end, required_min_c, h0=parent_bin
            )
            if cand_min > best_min:
                best_min, best_h = cand_min, cand_h
            if best_min >= required_min_c:
                return best_h.tolist(), float(compute_c5(best_h)), int(n)

    # Random restarts until time runs out or we meet the target.
    while time.time() < deadline and best_min < required_min_c:
        remaining = deadline - time.time()
        slice_time = min(30.0, remaining)
        slice_end = time.time() + slice_time
        cand_h, cand_min = binary_optimize_fixed_target(
            n, rng, slice_end, required_min_c, h0=None
        )
        if cand_min > best_min:
            best_min, best_h = cand_min, cand_h
        if best_min >= required_min_c:
            break

    if best_h is None:
        # Fallback: simple alternating pattern.
        best_h = np.tile([1.0, 0.0], n // 2).astype(float)
        best_min = int(np.correlate(best_h, best_h, mode="full").min())

    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(n)
