def worst_shifts_grad(h):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    max_val = corr.max()
    # find indices where corr == max_val (floating)
    # we can use np.isclose
    idxs = np.where(np.isclose(corr, max_val, atol=1e-12))[0]
    # convert to k shift: idx - (n-1)
    shifts = idxs - (n-1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k == 0:
            grad += 1 - 2*h
        else:
            kk = abs(k)
            # shift contributions same for +k or -k (both give same)
            # treat k positive:
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]
    return grad

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in‑place so Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_shifts_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient summed over all shifts attaining the current max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    idxs = np.where(corr >= max_val - 1e-12)[0]
    shifts = idxs - (n - 1)
    grad = np.zeros_like(h)
    for k in shifts:
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            kk = abs(k)
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        grad = worst_shifts_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the summed worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        grad = worst_shifts_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_h(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    return enforce_sum(h, rng)

            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)

        if rng.random() < 0.3:
            for _ in range(5):
                i, j = rng.integers(0, n, size=2)
                if h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]

        h, c = guided_hill_climb(h, rng,
                                 max_iters=int(2500 * 256 / n),
                                 step_scale=0.7)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        for beta in (20.0, 200.0, 2000.0, 8000.0):
            if time.time() - start > 0.7 * budget_s:
                break
            h = slsqp_optimize(h, beta, rng,
                               max_iter=500 if beta < 200 else 300)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        h = gradient_refine(h, rng, step=0.015, max_iter=5)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        h, c = guided_hill_climb(h, rng,
                                 max_iters=int(1500 * 256 / n),
                                 step_scale=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ---------- Phase 2: ensure 512‑point refinement ----------
    if best_n < 512 and time.time() - start < 0.75 * budget_s:
        target_n = 512
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)

        h_up, c_up = guided_hill_climb(h_up, rng,
                                       max_iters=4000,
                                       step_scale=0.6)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, target_n

        for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0):
            if time.time() - start > 0.85 * budget_s:
                break
            h_up = slsqp_optimize(h_up, beta, rng,
                                  max_iter=500 if beta < 200 else 300)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, target_n

        h_up = gradient_refine(h_up, rng, step=0.01, max_iter=8)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # ---------- Phase 3: random 512‑point restarts ----------
    if best_n == 512:
        while time.time() - start < 0.95 * budget_s:
            h_rand = random_feasible_h(512, rng)
            h_rand, c_rand = guided_hill_climb(h_rand, rng,
                                               max_iters=2000,
                                               step_scale=0.6)
            for beta in (20.0, 200.0, 2000.0):
                if time.time() - start > 0.95 * budget_s:
                    break
                h_rand = slsqp_optimize(h_rand, beta, rng, max_iter=300)
            h_rand = gradient_refine(h_rand, rng, step=0.015, max_iter=4)
            c_rand = compute_c5(h_rand)
            if c_rand < best_c:
                best_h, best_c, best_n = h_rand.copy(), c_rand, 512

    # ---------- Phase 4: annealing / guided perturbations ----------
    if best_h is not None:
        n = best_h.size
        while time.time() - start < budget_s * 0.999:
            frac = (time.time() - start) / budget_s
            scale = max(0.005, 0.05 * (1.0 - frac))
            T = max(1e-7, 0.0005 * (1.0 - frac))
            if rng.random() < 0.7:
                grad = worst_shifts_gradient(best_h)
                nh = guided_perturb(best_h, grad, rng, scale=scale)
            else:
                i, j = rng.integers(0, n, size=2)
                max_eps = min(best_h[i], 1.0 - best_h[j])
                if max_eps <= 0:
                    continue
                eps = rng.random() * max_eps * scale
                nh = best_h.copy()
                nh[i] -= eps
                nh[j] += eps
            nc = compute_c5(nh)
            delta = nc - best_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                best_h, best_c = nh, nc

    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
