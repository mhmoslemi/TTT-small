def binary_swap_hill_climb(h, max_iters, rng):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        k = np.argmax(corr) - (n - 1)
        # compute s_i = h_{i+k} + h_{i-k}
        s = h[(np.arange(n) + k) % n] + h[(np.arange(n) - k) % n]
        # among zeros, find index i with smallest s
        zeros = np.where(h == 0)[0]
        if zeros.size == 0:
            break
        i_idx = zeros[np.argmin(s[zeros])]
        # among ones, find index j with largest s
        ones = np.where(h == 1)[0]
        if ones.size == 0:
            break
        j_idx = ones[np.argmax(s[ones])]
        # attempt swap i<->j
        h_new = h.copy()
        h_new[i_idx] = 1
        h_new[j_idx] = 0
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
            h = h_new  # accept
        else:
            # maybe accept with low probability for annealing
            # compute temperature schedule
            T = max(0.01, 1.0 - it / max_iters)
            if rng.random() < np.exp((best_c - c_new) / T):
                h = h_new
    return best_h, best_c

def binary_swap_hill_climb(h, max_iters, rng):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    c = best_c
    for it in range(max_iters):
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        k = np.argmax(corr) - (n - 1)
        # compute s = h[i+k] + h[i-k] with zero padding
        s = np.zeros(n, dtype=int)
        if k >= 0:
            s[:n-k] += h[k:]
            s[k:] += h[:n-k]
        else:
            kk = -k
            s[kk:] += h[:n-kk]
            s[:n-kk] += h[kk:]
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        i = zeros[np.argmin(s[zeros])]
        j = ones[np.argmax(s[ones])]
        # swap
        h_new = h.copy()
        h_new[i] = 1
        h_new[j] = 0
        c_new = compute_c5(h_new)
        # Accept if better or with prob
        if c_new < c:
            h = h_new
            c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        else:
            # temperature schedule linearly decreasing
            T = 0.5 * (1 - it / max_iters)  # from 0.5 to 0
            if T > 0 and rng.random() < math.exp((c - c_new) / T):
                h = h_new
                c = c_new
        # early stop if best_c already below target (0.3808)
        if best_c < 0.3808:
            break
    return best_h, best_c

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    c = best_c
    scale = step_scale
    decay = max(1, max_iters // 5)

        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c_new = compute_c5(nh)
        if c_new < best_c:
            best_c, best_h, h = c_new, nh.copy(), nh
            c = c_new
        else:
            h = nh if c_new < c else h
            c = min(c, c_new)
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
        # early exit if best_c < target
        if best_c < 0.3808:
            break
    return best_h, best_c

def gradient_refine(h, rng, step=0.015, max_iter=10):

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    # Using scipy.optimize.minimize
    from scipy.optimize import minimize
    res = minimize(softmax_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time, math, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter,
                                             "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:

            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    c = best_c
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale)
        c_new = compute_c5(nh)
        if c_new < best_c:
            best_c, best_h, h = c_new, nh.copy(), nh
            c = c_new
        else:
            if c_new < c:
                h, c = nh, c_new
        if (it + 1) % decay == 0:

    return best_h, best_c

def binary_swap_hill_climb(h, max_iters, rng):

    no_imp = 0
