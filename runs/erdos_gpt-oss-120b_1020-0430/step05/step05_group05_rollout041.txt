def swap_update(h, i, j, overlap):
    n = len(h)
    # compute arrays of indices for k from 0..n-1
    idx = np.arange(n, dtype=np.int64)
    ipk = (i + idx) % n
    imk = (i - idx) % n
    jpk = (j + idx) % n
    jmk = (j - idx) % n
    delta = h[jpk] + h[jmk] - (h[ipk] + h[imk])
    overlap += delta
    # perform swap
    h[i], h[j] = h[j], h[i]
    return overlap
```

from scipy.optimize import minimize

def compute_corr(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def compute_overlap_counts(h):
    n = h.size
    corr = np.correlate(h, h, mode='full')
    return corr[n-1:n-1+n]

def slsqp_smooth_max(h0, beta, max_iter=800):
    n = h0.size
    target = n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target}
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:

        for _ in range(2000):

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

        overlap = np.correlate(h, h, mode="full")[n - 1 : n - 1 + n]
        new_overlap = overlap + delta
        cur_max = (n / 2 - overlap.min()) * (2 / n)
        new_max = (n / 2 - new_overlap.min()) * (2 / n)
        if new_max < cur_max - 1e-12:
            h[i], h[j] = h[j], h[i]
            continue
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            delta2 = h[(j2 + ks) % n] + h[(j2 - ks) % n] - h[(i2 + ks) % n] - h[(i2 - ks) % n]
            delta2[0] = 0
            new_overlap2 = overlap + delta2
            new_max2 = (n / 2 - new_overlap2.min()) * (2 / n)
            if new_max2 < cur_max - 1e-12:
                h[i2], h[j2] = h[j2], h[i2]
                improved = True
                break
        if not improved:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    # ----- size and optional parent -----
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        parent = np.array(initial_h_values, dtype=float)
    else:
        n = 360
        parent = None

    # ----- build seed pool -----
    seeds = []
    if parent is not None:
        parent = np.clip(parent, 0.0, 1.0)
        parent = _adjust_sum_exact(parent, rng, n)
        seeds.append(parent.copy())
        idx = np.argpartition(parent, -n // 2)[-n // 2 :]
        b = np.zeros(n, dtype=float)
        b[idx] = 1.0
        b = _adjust_sum_exact(b, rng, n)
        seeds.append(b)
    for _ in range(3):
        hb = np.zeros(n, dtype=float)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        seeds.append(hb)
    for _ in range(2):
        hf = rng.random(n)
        hf *= (n / 2) / hf.sum()
        hf = np.clip(hf, 0.0, 1.0)
        hf = _adjust_sum_exact(hf, rng, n)
        seeds.append(hf)

    best_h = None
    best_c = np.inf
    best_min_overlap = -np.inf

    betas = [10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0]

    for base in seeds:
        if time.time() > deadline:
            break
        h = _adjust_sum_exact(base.copy(), rng, n)

        for beta in betas:
            if time.time() > deadline:
                break
            h = _slsqp_smooth_max(h, beta, rng, max_iter=1200)

        # binary rounding + greedy improvement
        idx = np.argpartition(h, -n // 2)[-n // 2 :]
        hb = np.zeros(n, dtype=float)
        hb[idx] = 1.0
        hb = _adjust_sum_exact(hb, rng, n)
        hb = _binary_targeted_improve(hb, rng, deadline)

        c_bin = _compute_corr(hb)
        if c_bin < best_c:
            best_c = c_bin
            best_h = hb.copy()
            best_min_overlap = n / 2 - c_bin * (n / 2)

        c_frac = _compute_corr(h)
        if c_frac < best_c:
            best_c = c_frac
            best_h = h.copy()
            best_min_overlap = n / 2 - c_frac * (n / 2)

    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_h = _adjust_sum_exact(best_h, rng, n)

    # work with a binary copy for annealing
    cur_h = (best_h > 0.5).astype(float)
    overlap = _compute_overlap_counts(cur_h)
    cur_min = overlap.min()
    best_min_overlap = max(best_min_overlap, cur_min)
    best_sol = cur_h.copy()
    ks = np.arange(n, dtype=np.int64)

    # ----- simulated annealing -----
    while time.time() < deadline:
        t_frac = (time.time() - start) / budget_s
        T = (1.0 - t_frac) * 2.0 + 0.001  # temperature from ~2 → ~0

        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        delta = cur_h[(j + ks) % n] + cur_h[(j - ks) % n] - cur_h[(i + ks) % n] - cur_h[(i - ks) % n]
        delta[0] = 0
        new_overlap = overlap + delta
        new_min = new_overlap.min()
        delta_min = new_min - cur_min

        if delta_min >= 0 or rng.random() < math.exp(delta_min / T):
            # accept move
            overlap = new_overlap
            cur_min = new_min
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            if cur_min > best_min_overlap:
                best_min_overlap = cur_min
                best_sol = cur_h.copy()

    final_c = 1.0 - best_min_overlap / (n / 2)
    return best_sol.tolist(), float(final_c), int(n)
