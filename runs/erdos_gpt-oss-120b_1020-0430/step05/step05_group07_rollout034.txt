CURRENT_BETA = 1.0


def compute_c5(h):
    """Maximum overlap C₅ for a discretised step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Redistribute a tiny amount of mass so that Σh = n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random walk redistribution
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1‑h[i+k])
            g[k:] -= h[:upper]            # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # ------------------------------------------------------------
    # discretisation
    n = 360                     # number of sample points (can be doubled later)
    dx = 2.0 / n
    x_grid = np.linspace(0.0, 2.0, n, endpoint=False)

    # Fourier basis (cosine terms only)
    m = 12                      # number of cosine harmonics
    ks = np.arange(1, m + 1)
    cos_mat = np.cos(np.pi * np.outer(x_grid, ks))   # shape (n,m)

    # bounds for the coefficients a_k
    a_bounds = [(-0.5, 0.5) for _ in range(m)]

    best_c = np.inf
    best_h = None
    best_a = None

    # ------------------------------------------------------------
    # objective used by differential evolution (hard penalty on bounds)
    PENALTY = 1e6

    def obj_de(a):
        h = 0.5 + cos_mat @ a
        if np.any(h < 0.0) or np.any(h > 1.0):
            return PENALTY
        # sum is analytically n/2, but floating error may appear
        if abs(h.sum() - n / 2.0) > 1e-8:
            return PENALTY + abs(h.sum() - n / 2.0) * 1e6
        return compute_c5(h)

    # ------------------------------------------------------------
    # main optimisation loop (many DE restarts + local SLSQP)
    while time.time() < deadline - 0.8:
        # ---- differential evolution ---------------------------------
        de_res = opt.differential_evolution(
            obj_de,
            bounds=a_bounds,
            strategy="best1bin",
            maxiter=120,
            popsize=15,
            tol=1e-7,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=int(rng.integers(1 << 30)),
            polish=False,
            init="latinhypercube",
            updating="deferred",
        )
        a_cur = de_res.x

        # ---- cascade of smooth‑max SLSQP refinements ---------------
        global CURRENT_BETA

        def softmax_obj_a(a):
            h = 0.5 + cos_mat @ a
            if np.any(h < 0.0) or np.any(h > 1.0):
                return 1e6 + np.max(np.abs(np.clip(h, 0, 1) - h)) * 1e6
            corr = np.correlate(h, 1.0 - h, mode="full")
            bc = CURRENT_BETA * corr
            mval = np.max(bc)
            # log‑sum‑exp (stable)
            return (mval + np.log(np.sum(np.exp(bc - mval)))) / CURRENT_BETA

        for beta in [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]:
            if time.time() > deadline - 0.5:
                break
            CURRENT_BETA = beta
            slsqp_res = opt.minimize(
                softmax_obj_a,
                a_cur,
                method="SLSQP",
                bounds=a_bounds,
                options={"maxiter": 200, "ftol": 1e-12, "disp": False},
            )
            if slsqp_res.success:
                a_cur = slsqp_res.x

        # ---- evaluate candidate --------------------------------------
        h_cur = 0.5 + cos_mat @ a_cur
        h_cur = np.clip(h_cur, 0.0, 1.0)
        adjust_sum_exact(h_cur, rng)
        c_cur = compute_c5(h_cur)

        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
            best_a = a_cur.copy()

        # ---- cheap stochastic refinement on the vector ------------
        if time.time() < deadline - 0.2:
            cur = h_cur.copy()
            cur_c = c_cur
            # limited number of random transfers
            for _ in range(3000):
                i, j = rng.choice(n, size=2, replace=False)
                max_transfer = min(cur[i], 1.0 - cur[j])
                if max_transfer <= 0.0:
                    continue
                delta = rng.random() * max_transfer
                cur[i] -= delta
                cur[j] += delta
                c = compute_c5(cur)
                if c < cur_c:
                    cur_c = c
                else:
                    cur[i] += delta
                    cur[j] -= delta
                if time.time() > deadline - 0.15:
                    break
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur.copy()
                # best_a not needed any more

    # ------------------------------------------------------------
    # optional up‑sampling + tiny refinement (if time permits)
    if best_h is not None and time.time() < deadline - 0.4 and n * 2 <= 720:
        n_up = n * 2
        x_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_up, x_grid, best_h)
        adjust_sum_exact(h_up, rng)

        # a couple of gradient + random‑transfer cycles
        for _ in range(2):
            if time.time() > deadline - 0.2:
                break
            h_up = gradient_refine_h(h_up, rng, step=0.03, max_iter=18)
            # short random‑transfer hill‑climb
            cur = h_up.copy()
            cur_c = compute_c5(cur)
            for _ in range(2000):
                i, j = rng.choice(n_up, size=2, replace=False)
                max_transfer = min(cur[i], 1.0 - cur[j])
                if max_transfer <= 0.0:
                    continue
                delta = rng.random() * max_transfer
                cur[i] -= delta
                cur[j] += delta
                c = compute_c5(cur)
                if c < cur_c:
                    cur_c = c
                else:
                    cur[i] += delta
                    cur[j] -= delta
                if time.time() > deadline - 0.15:
                    break
            h_up = cur
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            n = n_up

    # ------------------------------------------------------------
    # final safety checks
    if best_h is None:
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
