def continuous_sa(h, rng, time_budget):
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    curr_h = h.copy()
    curr_c = best_c
    # cooling schedule: T0 = 0.02, T_end = 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # ensure move possible
        hi = curr_h[i]
        hj = curr_h[j]
        if hi == 0.0 or hj == 1.0:
            continue
        eps_max = min(hi, 1.0 - hj)
        eps = rng.random() * eps_max * 0.5 # scale 0.5
        new_h = curr_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - curr_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            curr_h, curr_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
                # Early exit if below target
                if best_c <= 0.3808:
                    return best_h, best_c
    return best_h, best_c
```

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = len(h)

        i = rng.integers(n)
        # compute add possible

import time
import numpy as np

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def propose_move(h: np.ndarray, rng: np.random.Generator, scale: float):
    """Return (i, j, eps) such that eps is moved from i to j (preserving sum)."""
    n = h.size
    for _ in range(15):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        if hi <= 0.0 or hj >= 1.0:
            continue
        eps_max = min(hi, 1.0 - hj)
        if eps_max <= 0.0:
            continue
        eps = rng.random() * eps_max * scale
        if eps <= 0.0:
            continue
        return i, j, eps
    return None


def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simulated annealing with mass‑transfer moves."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    curr_c = best_c

    start = time.time()
    T0 = 0.04
    T_end = 5e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # scale shrinks linearly to keep moves big early, small later
        scale = 0.8 * (1.0 - elapsed / time_budget) + 0.02
        mv = propose_move(h, rng, scale)
        if mv is None:
            continue
        i, j, eps = mv
        hi, hj = h[i], h[j]
        h[i] = hi - eps
        h[j] = hj + eps
        new_c = compute_c5(h)
        delta = new_c - curr_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    max_iter: int = 200, step0: float = 0.05) -> np.ndarray:
    """Projected gradient descent focusing on the worst shift."""
    n = h.size
    target = n / 2.0
    step = step0
    c_curr = compute_c5(h)

    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        # gradient of O_k (worst shift) w.r.t. h
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, target, rng)
        c_new = compute_c5(h_new)

        if c_new < c_curr:
            h, c_curr = h_new, c_new
            step = min(step * 1.1, step0)
        else:
            step *= 0.6

        if step < 1e-6:
            break
    return h


def upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid (still on [0,2])."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

    # ----- final tiny stochastic tweaks -------------------------------
    while time.time() - start < budget * 0.99:
        mv = propose_move(best_h, rng, scale=0.05)
        if mv is None:
            break
        i, j, eps = mv
        hi, hj = best_h[i], best_h[j]
        best_h[i] = hi - eps
        best_h[j] = hj + eps
        c_new = compute_c5(best_h)
        if c_new < best_c:
            best_c = c_new
        else:
            best_h[i] = hi
            best_h[j] = hj

    # ----- final sanitisation -----------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
