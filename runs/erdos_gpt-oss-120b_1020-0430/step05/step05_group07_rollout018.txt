def compute_c5_corr(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")

def fractional_sa(h, rng, deadline):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5_corr(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # ensure i is source (has >0), j can receive (has <1). To have possibility of transfer, we need max_transfer>0
        max_transfer = min(cur[i], 1.0 - cur[j])
        # Also try opposite direction randomly:
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
            max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5_corr(cur)
        T = init_T * (0.9997 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 50000:   # no improvement for many iterations, break
            break
    return best, best_c
```

def delta_autocorr(h, a, b):
    n = h.shape[0]
    size = 2*n - 1
    delta = np.zeros(size, dtype=float)
    offset = n - 1
    # a as first
    for k in range(-a, n - a):
        delta[k + offset] -= h[a + k]
    # b as first
    for k in range(-b, n - b):
        delta[k + offset] += h[b + k]
    # a as second
    for k in range(a - (n - 1), a + 1):
        delta[k + offset] -= h[a - k]
    # b as second
    for k in range(b - (n - 1), b + 1):
        delta[k + offset] += h[b - k]
    # conv3
    delta[offset] += 2.0
    delta[a - b + offset] -= 1.0
    delta[b - a + offset] -= 1.0
    return delta
```

def local_search_binary(h0, deadline):
    h = h0.astype(float)
    n = h.size
    const = n / 2.0
    A = np.correlate(h, h, mode='full')
    min_A = A.min()
    O_max = const - min_A
    best_h = h.copy()
    best_O = O_max
    rng = np.random.default_rng()
    offset = n - 1
    while time.time() < deadline:
        # find worst shifts
        worst = np.flatnonzero(A == min_A)
        if worst.size == 0:
            break
        # shuffle order
        idxs = rng.permutation(worst)
        found = False
        for wk in idxs:
            k = wk - offset
            if k >= 0:
                L = n - k
                mask1 = (h[:L] == 1) & (h[k:] == 0)
                mask0 = (h[:L] == 0) & (h[k:] == 1)
                idx1 = np.flatnonzero(mask1)
                idx0 = np.flatnonzero(mask0)
                if idx1.size == 0 or idx0.size == 0:
                    continue
                i = rng.choice(idx1)
                j = rng.choice(idx0)
                a = i
                b = j
            else:
                t = -k
                L = n - t
                mask1 = (h[t:] == 1) & (h[:L] == 0)
                mask0 = (h[t:] == 0) & (h[:L] == 1)
                idx1 = np.flatnonzero(mask1)
                idx0 = np.flatnonzero(mask0)
                if idx1.size == 0 or idx0.size == 0:
                    continue
                i = rng.choice(idx1)
                j = rng.choice(idx0)
                a = i + t
                b = j + t
            # compute delta
            delta = delta_autocorr(h, a, b)
            A_new = A + delta
            min_A_new = A_new.min()
            O_max_new = const - min_A_new
            if O_max_new <= O_max:
                # accept swap
                h[a] = 0.0
                h[b] = 1.0
                A = A_new
                min_A = min_A_new
                O_max = O_max_new
                if O_max_new < best_O:
                    best_O = O_max_new
                    best_h = h.copy()
                found = True
                break
        if not found:
            # no improving swap for any worst shift
            break
    c5 = O_max * (2.0 / n)
    return best_h, c5
```

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for SLSQP constraints (softmax surrogate)
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5_corr(h):
    """C5 from correlation h with (1-h)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k Corr_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP on the smooth max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (simple redistribution)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def adjust_sum_exact(h, rng):
    """Redistribute mass so that Σh = n/2 (maintaining 0≤h≤1)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def delta_autocorr(h, a, b):
    """Incremental change of autocorrelation when swapping 1 at a ->0 and 0 at b ->1."""
    n = h.shape[0]
    size = 2 * n - 1
    delta = np.zeros(size, dtype=float)
    offset = n - 1

    # a as first index (removed)
    for k in range(-a, n - a):
        delta[k + offset] -= h[a + k]
    # b as first index (added)
    for k in range(-b, n - b):
        delta[k + offset] += h[b + k]
    # a as second index (removed)
    for k in range(a - (n - 1), a + 1):
        delta[k + offset] -= h[a - k]
    # b as second index (added)
    for k in range(b - (n - 1), b + 1):
        delta[k + offset] += h[b - k]

    # d*d term
    delta[offset] += 2.0               # k = 0
    delta[a - b + offset] -= 1.0       # k = a-b
    delta[b - a + offset] -= 1.0       # k = b-a
    return delta

def local_search_binary(h0, deadline):
    """Targeted swap search on binary vectors."""
    h = h0.astype(float)
    n = h.size
    const = n / 2.0
    A = np.correlate(h, h, mode="full")
    min_A = A.min()
    O_max = const - min_A
    best_h = h.copy()
    best_O = O_max
    rng = np.random.default_rng()
    offset = n - 1

    while time.time() < deadline:
        worst = np.flatnonzero(A == min_A)
        if worst.size == 0:
            break
        order = rng.permutation(worst)
        improved = False
        for wk in order:
            k = wk - offset
            if k >= 0:
                L = n - k
                mask1 = (h[:L] == 1) & (h[k:] == 0)
                mask0 = (h[:L] == 0) & (h[k:] == 1)
                idx1 = np.flatnonzero(mask1)
                idx0 = np.flatnonzero(mask0)
                if idx1.size == 0 or idx0.size == 0:
                    continue
                i = rng.choice(idx1)
                j = rng.choice(idx0)
                a, b = i, j
            else:
                t = -k
                L = n - t
                mask1 = (h[t:] == 1) & (h[:L] == 0)
                mask0 = (h[t:] == 0) & (h[:L] == 1)
                idx1 = np.flatnonzero(mask1)
                idx0 = np.flatnonzero(mask0)
                if idx1.size == 0 or idx0.size == 0:
                    continue
                i = rng.choice(idx1)
                j = rng.choice(idx0)
                a, b = i + t, j + t
            delta = delta_autocorr(h, a, b)
            A_new = A + delta
            min_A_new = A_new.min()
            O_max_new = const - min_A_new
            if O_max_new <= O_max:
                h[a] = 0.0
                h[b] = 1.0
                A = A_new
                min_A = min_A_new
                O_max = O_max_new
                if O_max_new < best_O:
                    best_O = O_max_new
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    c5 = O_max * (2.0 / n)
    return best_h, c5

def fractional_sa(h0, rng, deadline):
    """Continuous transfer‑based simulated annealing."""
    h = h0.copy()
    n = h.size
    cur_c = compute_c5_corr(h)
    best = h.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-5)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[src] -= delta
        h[dst] += delta
        c = compute_c5_corr(h)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = h.copy()
                last_improve = it
        else:
            h[src] += delta
            h[dst] -= delta
        it += 1
        if it - last_improve > 120000:
            break
    return best, best_c

def binary_from_values(vals):
    """Return binary vector (half ones) choosing largest entries."""
    n = vals.size
    target = n // 2
    idx = np.argpartition(vals, -target)[-target:]
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx] = 1.0
    return bin_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ------------------------------------------------------------------
    # Determine problem size
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
        if n % 2 == 1:  # enforce even length
            n -= 1
            parent = parent[:n]
    else:
        parent = None
        n = 360

    # ------------------------------------------------------------------
    # Build start vectors (binary from parent + random binaries + one random fraction)
    start_vectors = []

    if parent is not None:
        start_vectors.append(binary_from_values(parent))

    for _ in range(3):
        hb = np.zeros(n, dtype=float)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vectors.append(hb)

    # random fractional start (might be useful for SA)
    frac = rng.random(n)
    frac *= (n / 2) / frac.sum()
    frac = np.clip(frac, 0.0, 1.0)
    adjust_sum_exact(frac, rng)
    start_vectors.append(frac)

    best_h = None
    best_c = np.inf
    best_n = n

    # ------------------------------------------------------------------
    # Phase 1: binary directed swap search (limited time per seed)
    for h0 in start_vectors:
        if time.time() > deadline:
            break
        if np.all((h0 == 0) | (h0 == 1)):
            # allocate modest slice of remaining time
            time_left = deadline - time.time()
            per = max(5.0, time_left * 0.2)
            sub_dead = min(deadline, time.time() + per)
            h_bin, c_bin = local_search_binary(h0, sub_dead)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin, c_bin, n

    # ------------------------------------------------------------------
    # Phase 2: fractional SA on the best solution found so far
    if best_h is not None and time.time() < deadline:
        h_sa, c_sa = fractional_sa(best_h, rng, deadline)
        if c_sa < best_c:
            best_h, best_c, best_n = h_sa, c_sa, n

    # ------------------------------------------------------------------
    # Phase 3: optional SLSQP refinement (few beta steps, respecting time)
    if best_h is not None and time.time() < deadline:
        betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]
        for beta in betas:
            if time.time() > deadline:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, rng, max_iter=800)
                c_ref = compute_c5_corr(h_ref)
                if c_ref < best_c:
                    best_h, best_c, best_n = h_ref, c_ref, n
            except Exception:
                continue

    # ------------------------------------------------------------------
    # Ensure feasibility before returning
    if best_h is None:
        # fallback: simple half‑ones pattern
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_n = n
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5_corr(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
