def compute_grad(h, O, beta):
    # O: array of size 2n-1 containing overlaps for each shift
    # compute w via stable exp
    O_max = np.max(O)
    exp_vals = np.exp(beta * (O - O_max))
    w = exp_vals / np.sum(exp_vals)  # shape 2n-1
    n = h.shape[0]
    grad = np.ones(n)  # contribution from 1
    # subtract weighted h[t + k] and h[t - k]
    # we need w indexed by shift k = idx - (n-1)
    for idx in range(2*n-1):
        wk = w[idx]
        if wk == 0:
            continue
        k = idx - (n-1)
        if k >= 0:
            # h[t + k]
            grad[:n - k] -= wk * h[k:]
            # h[t - k]
            grad[k:] -= wk * h[:n - k]
        else:
            off = -k
            grad[off:] -= wk * h[:n - off]
            grad[:n - off] -= wk * h[off:]
    return grad

def optimize_h(h0, n, deadline):
    h = h0.copy()
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    beta_schedule = [(1.0, 150), (5.0, 150), (20.0, 150), (80.0, 150), (200.0, 200), (500.0, 200), (1000.0, 200), (2000.0, 300)]
    for beta, steps in beta_schedule:
        if time.time() > deadline:
            break
        # Adam state
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        # learning rate scaled inversely sqrt(beta)
        lr = 0.2 / max(1.0, np.sqrt(beta))
        for it in range(1, steps+1):
            if time.time() > deadline:
                break
            grad = compute_grad(h, beta)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            step = - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = proj_capped_simplex(h + step, target)
        # after stage evaluate
        c_val = compute_c5(h)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    return best_h, best_c

import time
import numpy as np

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact max overlap for a discrete h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def proj_capped_simplex(y: np.ndarray, target: float) -> np.ndarray:
    """
    Projection of y onto the intersection of the box [0,1]^n
    and the hyperplane sum(x)=target (capped simplex).
    Uses a simple binary search for the Lagrange multiplier.
    """
    low = np.min(y) - 1.0
    high = np.max(y)
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(y - (low + high) / 2.0, 0.0, 1.0)


def random_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random start, projected to feasibility."""
    y = rng.random(n)
    return proj_capped_simplex(y, n / 2.0)


def compute_grad(h: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate:
        f_beta(h) = (1/β) log Σ exp(β·O_k(h)),
    where O_k are the overlaps.
    Uses the symmetry O_{-k}=O_k to obtain a simple formula.
    """
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    O_max = O.max()
    w = np.exp(beta * (O - O_max))
    w /= w.sum()
    conv = np.correlate(w, h, mode="full")
    # convex combination of shifted copies of h
    grad = np.ones(n) - 2.0 * conv[n - 1 : 2 * n - 1]
    return grad


def greedy_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Deterministic greedy construction of a binary vector with n/2 ones."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h.astype(float)


def get_initial_candidates(n: int, rng: np.random.Generator):
    """Collect a few diverse starting vectors for a given size."""
    cand = []
    # parent if available and compatible
    if "initial_h_values" in globals():
        p = np.array(initial_h_values, dtype=float)
        if p.size == n:
            cand.append(p.copy())
    # greedy binary
    cand.append(greedy_binary(n, rng))
    # a couple of random smooth starts
    for _ in range(2):
        cand.append(random_h(n, rng))
    return cand


def optimize_h(h0: np.ndarray, n: int, deadline: float):
    """
    Adam‑style projected gradient descent on the smooth‑max surrogate.
    Returns the best vector found and its exact C5 bound.
    """
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    # schedule: (beta, steps)
    schedule = [
        (1.0, 150),
        (5.0, 150),
        (20.0, 150),
        (80.0, 150),
        (200.0, 200),
        (500.0, 200),
        (1000.0, 200),
        (2000.0, 300),
    ]

    for beta, steps in schedule:
        if time.time() > deadline:
            break
        # Adam state
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        # learning rate scaled with sqrt(beta) (larger beta → smaller steps)
        lr = 0.2 / max(1.0, np.sqrt(beta))
        for it in range(1, steps + 1):
            if time.time() > deadline:
                break
            grad = compute_grad(h, beta)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = -lr * m_hat / (np.sqrt(v_hat) + eps)
            h = proj_capped_simplex(h + step, target)

        c_val = compute_c5(h)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    return best_h, best_c


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # Try a couple of discretisations; the smaller one is quicker.
    for n in (360, 720):
        if time.time() > deadline:
            break
        candidates = get_initial_candidates(n, rng)
        for h0 in candidates:
            if time.time() > deadline:
                break
            h_opt, c_opt = optimize_h(h0, n, deadline)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n

    # Fallback – should never happen
    if best_h is None:
        best_n = 360
        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
        best_c = compute_c5(best_h)

    # Final safety projection and evaluation
    best_h = proj_capped_simplex(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
