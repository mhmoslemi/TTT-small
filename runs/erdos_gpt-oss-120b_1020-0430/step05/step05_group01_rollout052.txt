    cur_c = compute_c5(cur_h)

        i, j = rng.integers(0, n, size=2)
        if i == j:

        if max_eps <= 0:

import time
import numpy as np
from scipy.optimize import minimize

# ----- global vars for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max of overlaps using FFT for speed."""
    n = x.size
    H = np.fft.rfft(x, n * 2)
    G = np.fft.rfft(np.flip(1.0 - x), n * 2)
    conv = np.fft.irfft(H * G, n * 2)
    corr = conv[: 2 * n - 1]
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 using FFT (fast)."""
    n = h.shape[0]
    H = np.fft.rfft(h, n * 2)
    G = np.fft.rfft(np.flip(1.0 - h), n * 2)
    conv = np.fft.irfft(H * G, n * 2)
    corr = conv[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple greedy fix
    for _ in range(20000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    # final patch (rare)
    if abs(h.sum() - target) > 1e-10:
        i = rng.integers(0, n)
        needed = target - h.sum()
        needed = min(needed, 1.0 - h[i]) if needed > 0 else max(needed, -h[i])
        h[i] += needed
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (differences only)."""
    n = h.size
    H = np.fft.rfft(h, n * 2)
    G = np.fft.rfft(np.flip(1.0 - h), n * 2)
    conv = np.fft.irfft(H * G, n * 2)
    corr = conv[: 2 * n - 1]
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = -2.0 * h          # constant term omitted (cancels for sum‑preserving moves)
    else:
        grad[: n - k] -= h[k:]      # -h_{i+k}
        grad[k:] -= h[: n - k]      # -h_{i‑k}
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 5000, step_scale: float = 0.5):
    """Monotone hill‑climb using worst‑shift gradient."""
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    decay = max(1, max_iters // 5)
    scale = step_scale
    for it in range(max_iters):
        grad = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * scale
                cur_h[i] -= eps
                cur_h[j] += eps
        cur_c = compute_c5(cur_h)
        if cur_c < best_c:
            best_c, best_h = cur_c, cur_h.copy()
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 8) -> np.ndarray:
    """Fine‑tune using explicit worst‑shift gradient steps."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
    return h

                h = enforce_sum(h, rng)
        else:
            h = np.zeros(n)
            h[rng.choice(n, n // 2, replace=False)] = 1.0

        if rng.random() < 0.2:
            for _ in range(3):
                i, j = rng.integers(0, n, size=2)
                if h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]

        # first hill‑climb
        it1 = int(2500 * 256 / n)
        h, c = guided_hill_climb(h, rng, max_iters=it1, step_scale=0.7)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # soft‑max SLSQP passes
        for beta in (5.0, 20.0, 100.0, 500.0, 2000.0):
            if time.time() - start > 0.75 * budget_s:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=500)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # gradient tidy‑up
        h = gradient_refine(h, rng, step=0.015, max_iter=5)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # final hill‑climb for this run
        it2 = int(1500 * 256 / n)
        h, c = guided_hill_climb(h, rng, max_iters=it2, step_scale=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ---- up‑sample to full resolution ------------------------------------
    if best_n < 512 and time.time() - start < 0.85 * budget_s:
        target_n = 512
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)

        # deep hill‑climb at high resolution
        h_up, c_up = guided_hill_climb(h_up, rng, max_iters=6000, step_scale=0.6)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, target_n

        # refined soft‑max passes
        for beta in (5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0):
            if time.time() - start > 0.92 * budget_s:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=500)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, target_n

        # final gradient polishing
        h_up = gradient_refine(h_up, rng, step=0.01, max_iter=8)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # ---- simulated annealing on the best candidate ------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 1.0 and best_h is not None:
        sa_h, sa_c = simulated_annealing(best_h, rng, time_limit=remaining * 0.94)
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c

    # ---- final sanity check ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
