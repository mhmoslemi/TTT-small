def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    # while loop
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        attempts += 1
    # If still not exact due to numeric, scale
    if abs(diff) > 1e-9:
        h += diff / n
    # Clip final
    np.clip(h, 0.0, 1.0, out=h)
    # Adjust final sum again if off
    # Not essential
    return h

def slsqp_optimize(h0, beta, max_iter=600):
    n = h0.size
    target = n/2.0
    bounds = [(0.0,1.0)]*n
    cons = {"type":"eq", "fun": lambda x: np.sum(x) - target}

        m = np.max(bc)

                   constraints=cons, options={"maxiter": max_iter, "ftol":1e-12, "disp":False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())

def compute_c5(h):
    n = h.size
    nfft = 1 << ((2*n - 1).bit_length())
    fft_h = np.fft.rfft(h, n=nfft)
    fft_one_minus_h = np.fft.rfft(1.0 - h, n=nfft)
    conv = np.fft.irfft(fft_h * fft_one_minus_h, n=nfft)[:2*n-1]
    return conv.max() * (2.0 / n)

    nfft = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, n=nfft)
    f1h = np.fft.rfft(1.0 - h, n=nfft)
    conv = np.fft.irfft(fh * f1h, n=nfft)[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    attempts = 0
    # simple redistribution loop
    while abs(diff) > 1e-12 and attempts < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        attempts += 1
    # final tiny correction if needed
    if abs(diff) > 1e-9:
        h += diff / n
    np.clip(h, 0.0, 1.0, out=h)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst shift (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed epsilon transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    scale = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h.copy()
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 600):
    """Smooth max‑C5 (soft‑max) SLSQP refinement."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, rng)


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def tleft(): return budget_s - (time.time() - start)

    n = 512
    # initialise from parent if present
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        if h.size != n:
            # resample to required size
            x_old = np.linspace(0.0, 2.0, h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, h)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        # simple half‑ones start
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    best_h, best_c = h.copy(), compute_c5(h)

    # ---- quick gradient hill‑climb ----
    if tleft() > 0.1 * budget_s:
        best_h, best_c = guided_hill_climb(best_h, rng,
                                           max_iters=3000, step_scale=0.8)

    # ---- SLSQP refinement with increasing β ----
    if tleft() > 0.05 * budget_s:
        for beta in (20., 80., 200., 800., 3000., 8000., 20000., 50000.):
            if tleft() < 0.02 * budget_s:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=400)
            best_c = compute_c5(best_h)

    # ---- Simulated annealing on the full grid ----
    T = 0.02                      # initial temperature
    cooling = 0.99995
    max_eps_scale = 0.5
    it = 0
    while tleft() > 0.15 * budget_s:
        i, j = rng.choice(n, size=2, replace=False)
        eps_max = min(best_h[i], 1.0 - best_h[j])
        if eps_max <= 0:
            continue
        eps = rng.random() * eps_max * max_eps_scale
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        delta = nc - best_c
        if delta < 0:
            accept = True
        else:
            # avoid overflow in exp
            accept = (delta / T) < 700 and rng.random() < math.exp(-delta / T)
        if accept:
            best_h, best_c = nh, nc
        T = max(T * cooling, 1e-8)
        it += 1
        if it % 5000 == 0:
            max_eps_scale = min(1.0, max_eps_scale * 1.1)

    # ---- final gradient tidy‑up ----
    if tleft() > 0.05 * budget_s:
        best_h, best_c = guided_hill_climb(best_h, rng,
                                           max_iters=2000, step_scale=0.6)

    # ---- final sanity enforcement ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
