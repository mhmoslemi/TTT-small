def simulated_annealing_binary(h0, rng, deadline):
    n = len(h0)
    dx = 2.0/n
    h = h0.copy()
    cur_c = evaluate(h, dx)
    best_h = h.copy()
    best_c = cur_c
    # Parameters:
    # start temperature based on cur_c (maybe cur_c*0.02)
    T0 = cur_c * 0.02  # or set fixed like 0.01?
    T = T0
    decay = 0.9995 # adjust depending on iterations
    iter = 0
    while time.time() < deadline:
        # select random swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = evaluate(h, dx)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):

            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # cooling
        T *= decay
        iter += 1
        # optionally adjust decay to reach near zero before deadline

    n = h.shape[0]
    # ensure sum exact
    adjust_sum_exact(h, rng)
    # betas list

    zeros = set(np.flatnonzero(1 - h).tolist())

            break

        j = rng.choice(list(zeros))

            zeros.add(i)

import time
import numpy as np
from scipy.optimize import minimize

# global variables for the SLSQP surrogate
CURRENT_TARGET = 0.0
CURRENT_BETA = 1.0


def sum_constraint(x):
    """equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def surrogate_obj(x):
    """smooth approximation of max correlation via log-sum-exp."""
    global CURRENT_BETA
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Force Σh = n/2 while keeping 0 ≤ h ≤ 1."""
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    # up to 5000 attempts – sufficient for n ≤ 720
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def binary_from_fractional(h_frac, rng):
    """Round to a binary vector with exactly n/2 ones (largest entries)."""
    n = h_frac.shape[0]
    idx = np.argpartition(h_frac, -n // 2)[-n // 2 :]
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx] = 1.0
    # sum is already exact, but keep safety
    adjust_sum_exact(h_bin, rng)
    return h_bin


def run_slsqp_cascade(h_start, rng, deadline):
    """Run SLSQP on a smooth‑max surrogate with increasing β."""
    n = h_start.shape[0]
    adjust_sum_exact(h_start, rng)
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = n / 2.0
    h = h_start.copy()
    betas = [1.0, 5.0, 20.0, 80.0, 300.0, 1000.0, 5000.0, 20000.0, 100000.0]
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > deadline:
            break
        CURRENT_BETA = beta
        cons = {"type": "eq", "fun": sum_constraint}
        try:
            res = minimize(
                surrogate_obj,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 2500, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        h = np.clip(res.x, 0.0, 1.0)
        adjust_sum_exact(h, rng)
    return h

    """Binary simulated annealing that keeps exactly n/2 ones."""
    n = h0.shape[0]
    dx = 2.0 / n
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # Initial temperature – a small fraction of the current objective
    T = max(cur_c * 0.05, 0.005)
    decay = 0.9995
    # maintain index sets for speed

    zeros = set(np.flatnonzero(1.0 - h).tolist())
