def compute_c5_fft(b, m):
    # b: 1-D array of floats 0 or 1.
    n = b.shape[0]
    rev_nb = (1.0 - b)[::-1]
    fb = np.fft.rfft(b, n=m)
    fnb = np.fft.rfft(rev_nb, n=m)
    conv = np.fft.irfft(fb * np.conj(fnb), n=m)
    corr = conv[: 2 * n - 1]
    max_corr = np.max(corr)
    return float(max_corr * (2.0 / n))

        new_c = compute_c5_fft(b, m)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_b = b.copy()
                if target_c is not None and best_c <= target_c:
                    break
        else:
            # revert swap
            b[i] = 1.0
            b[j] = 0.0
            ones[idx_i] = i
            zeros[idx_j] = j
            pos_in_ones[i] = idx_i
            pos_in_zeros[i] = -1
            pos_in_ones[j] = -1
            pos_in_zeros[j] = idx_j

        T = max(T * factor, T_min)

    return best_b, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate sizes (larger first)
    n_candidates = [720, 480, 360]
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        pn = parent.size
        if pn % 2 == 0:
            n_candidates.insert(0, pn)
    n_candidates = [n for n in dict.fromkeys(n_candidates) if n % 2 == 0]

    target_bound = 0.3808

    for n in n_candidates:
        if time.time() > deadline:
            break
        m = 1 << ((2 * n - 1).bit_length())

        # initialise binary vector
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            b = np.zeros(n, dtype=float)
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            b[idx] = 1.0
        else:
            b = np.zeros(n, dtype=float)
            b[rng.choice(n, n // 2, replace=False)] = 1.0

        # simulated annealing
        b_opt, c_opt = simulated_annealing_binary(b, rng, deadline, m, target_c=target_bound)

        if c_opt < best_c:
            best_c = c_opt
            best_h = b_opt.copy()
            best_n = n
        if best_c <= target_bound:
            break

        # quick greedy refinement (few random swaps targeting worst shift)
        if time.time() < deadline:
            sweeps = 5
            bh = b_opt.copy()
            for _ in range(sweeps):
                rev_nb = (1.0 - bh)[::-1]
                fb = np.fft.rfft(bh, n=m)
                fnb = np.fft.rfft(rev_nb, n=m)
                conv = np.fft.irfft(fb * np.conj(fnb), n=m)
                corr = conv[: 2 * n - 1]
                max_idx = np.argmax(corr)
                k = max_idx - (n - 1)
                if k >= 0:
                    i_end = n - k
                    i_vals = np.arange(i_end)
                    mask_i = (bh[i_vals] == 1) & (bh[i_vals + k] == 0)
                    S1 = i_vals[mask_i]
                    j_vals = np.arange(n)
                    mask_j = (bh[j_vals] == 0) & ((j_vals + k >= n) | (bh[j_vals + k] == 1))
                    S0 = j_vals[mask_j]
                else:
                    k_abs = -k
                    i_start = k_abs
                    i_vals = np.arange(i_start, n)
                    mask_i = (bh[i_vals] == 1) & (bh[i_vals - k_abs] == 0)
                    S1 = i_vals[mask_i]
                    j_vals = np.arange(n)
                    mask_j = (bh[j_vals] == 0) & ((j_vals - k_abs < 0) | (bh[j_vals - k_abs] == 1))
                    S0 = j_vals[mask_j]
                if S1.size == 0 or S0.size == 0:
                    break
                best_pair = None
                best_pair_c = corr[max_idx]
                trials = min(200, S1.size * S0.size)
                for __ in range(trials):
                    i = rng.choice(S1)
                    j = rng.choice(S0)
                    bh[i] = 0.0
                    bh[j] = 1.0
                    c_new = compute_c5_fft(bh, m)
                    bh[i] = 1.0
                    bh[j] = 0.0
                    if c_new < best_pair_c - 1e-12:
                        best_pair_c = c_new
                        best_pair = (i, j)
                if best_pair is None:
                    break
                i, j = best_pair
                bh[i] = 0.0
                bh[j] = 1.0
            c_refined = compute_c5_fft(bh, m)
            if c_refined < best_c:
                best_c = c_refined
                best_h = bh.copy()
                best_n = n

    # fallback simple construction
    if best_h is None:
        best_n = 360
        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
        m = 1 << ((2 * best_n - 1).bit_length())
        best_c = compute_c5_fft(best_h, m)

    # ensure exact half‑sum (tiny adjustments if rounding occurred)
    target_sum = best_n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        idxs = np.where((diff > 0) & (best_h < 1 - 1e-12))[0] if diff > 0 else np.where((diff < 0) & (best_h > 1e-12))[0]
        for i in idxs:
            if abs(diff) < 1e-12:
                break
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target_sum - best_h.sum()

    final_c = compute_c5_fft(best_h, 1 << ((2 * best_n - 1).bit_length()))
    return best_h.tolist(), float(final_c), int(best_n)
