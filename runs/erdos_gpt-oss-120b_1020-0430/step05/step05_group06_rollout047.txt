    best_h = h.copy()
    best_S_arr = S_arr.copy()

    best_zeros = zeros.copy()

        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        i = ones[i_idx]
        j_idx = rng.integers(len(zeros))
        j = zeros[j_idx]
        # compute deltas
        # To speed, we can use local variables
        nmid = n - 1
        deltas = {}
        # subtract contributions for i
        # loop over ones list, which includes i
        for q in ones:
            if q == i:
                idx = nmid
                deltas[idx] = deltas.get(idx, 0) - 1
            else:
                d = q - i
                idx1 = nmid + d
                idx2 = nmid - d
                deltas[idx1] = deltas.get(idx1, 0) - 1

        # add contributions for j (excluding i)

            idx2 = nmid - d
            deltas[idx1] = deltas.get(idx1, 0) + 1
            deltas[idx2] = deltas.get(idx2, 0) + 1

    from scipy.optimize import minimize

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    def compute_c5(h):
        n = h.size
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * (2.0 / n))

    def adjust_sum_exact(h, rng):
        target = h.size / 2.0
        diff = target - np.sum(h)
        n = h.size
        # simple loop: adjust random entries
        attempts = 0
        while abs(diff) > 1e-12 and attempts < 10000:
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - np.sum(h)
            attempts += 1
        return h

    # baseline
    best_h = None
    best_c = np.inf
    best_n = None

    if 'initial_h_values' in globals():

        if c_parent < best_c:

        nmid = n - 1

        no_improve_counter = 0
        # loop

            # pick random swap
            if not best_ones or not best_zeros:
                break
            i_idx = rng.integers(len(best_ones))
            j_idx = rng.integers(len(best_zeros))
            i = best_ones[i_idx]
            j = best_zeros[j_idx]
            # compute deltas
            deltas = {}
            # subtract contributions for i
            for q in best_ones:
                if q == i:
                    idx = nmid
                    deltas[idx] = deltas.get(idx, 0) - 1
                else:

                    idx1 = nmid + d
                    idx2 = nmid - d
                    deltas[idx1] = deltas.get(idx1, 0) - 1

            # add contributions for j (excluding i)

                if q == i:

                d = q - j
                idx1 = nmid + d
                idx2 = nmid - d

import time
import math
import numpy as np
from scipy.optimize import minimize

# globals used by the smooth‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Exact C5 for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Exact correction so that Σh = n/2 while respecting 0≤h≤1."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def smooth_max_obj(x):
    """β‑scaled smooth max of the overlap correlation."""
    bc = CURRENT_BETA * np.correlate(x, 1.0 - x, mode="full")
    m = np.max(bc)
    # log‑sum‑exp in a stable form
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter, rng):
    """Run SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_max_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def hill_climb_binary_incremental(
    h, s_arr, ones, zeros, target, n, rng, sub_deadline
):
    """
    Deterministic hill‑climb on a binary vector.
    `h` is mutable np.int8, `s_arr` = correlation of h with itself (int32).
    """
    nmid = n - 1
    best_c = (target - int(s_arr.min())) * (2.0 / n)
    best_h = h
    best_s = s_arr
    best_ones = ones
    best_zeros = zeros
    no_improve = 0

    while time.time() < sub_deadline:
        if not best_ones or not best_zeros:
            break
        i_idx = rng.integers(len(best_ones))
        j_idx = rng.integers(len(best_zeros))
        i = best_ones[i_idx]
        j = best_zeros[j_idx]

        # collect deltas in a dict
        deltas = {}

        # remove contributions of i (currently 1)
        for q in best_ones:
            if q == i:
                deltas[nmid] = deltas.get(nmid, 0) - 1
            else:
                d = q - i
                idx1 = nmid + d
                idx2 = nmid - d
                deltas[idx1] = deltas.get(idx1, 0) - 1
                deltas[idx2] = deltas.get(idx2, 0) - 1

        # add contributions of j (will become 1)
        for q in best_ones:
            if q == i:
                continue
            d = q - j
            idx1 = nmid + d
            idx2 = nmid - d
            deltas[idx1] = deltas.get(idx1, 0) + 1
            deltas[idx2] = deltas.get(idx2, 0) + 1
        deltas[nmid] = deltas.get(nmid, 0) + 1  # self term for j

        # apply
        for idx, dlt in deltas.items():
            best_s[idx] += dlt

        new_min = int(best_s.min())
        new_c = (target - new_min) * (2.0 / n)

        if new_c < best_c - 1e-12:
            # accept swap
            best_c = new_c
            best_h[i] = 0
            best_h[j] = 1
            best_ones.pop(i_idx)
            best_zeros.append(i)
            best_zeros.pop(j_idx)
            best_ones.append(j)
            no_improve = 0
        else:
            # revert
            for idx, dlt in deltas.items():
                best_s[idx] -= dlt
            no_improve += 1
            if no_improve >= 2000:
                break

    return best_h, best_s, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99  # a little safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # --- baseline from parent if supplied ---
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        c_parent = compute_c5(parent)
        if c_parent < best_c:
            best_h = parent.copy()
            best_c = c_parent
            best_n = parent.size

    # --- parameters for the stochastic binary phase ---
    final_refine_time = min(30.0, budget_s * 0.03)  # reserve for SLSQP
    max_hill_time = 0.7  # per restart budget

    # Alternate between n=360 and n=720; start with whichever is not current best
    while time.time() < deadline - final_refine_time:
        # choose length
        if best_n is None or best_n == 720:
            n = 360
        else:
            n = 720
        target = n // 2
        L = target

        # random binary start
        perm = rng.permutation(n)
        ones_idx = perm[:L]
        zeros_idx = perm[L:]
        h_bin = np.zeros(n, dtype=np.int8)
        h_bin[ones_idx] = 1

        # correlation of binary with itself
        s_arr = np.correlate(h_bin, h_bin, mode="full").astype(np.int32)

        # hill climb with a time limit
        sub_dead = min(deadline - final_refine_time, time.time() + max_hill_time)
        h_opt, s_opt, c_opt = hill_climb_binary_incremental(
            h_bin,
            s_arr,
            list(ones_idx),
            list(zeros_idx),
            target,
            n,
            rng,
            sub_dead,
        )
        # recompute exact C5 for safety
        c_exact = compute_c5(h_opt.astype(float))
        if c_exact < best_c:
            best_c = c_exact
            best_h = h_opt.astype(float).copy()
            best_n = n

    # --- final smooth‑max SLSQP refinement (if time permits) ---
    if best_h is not None and time.time() < deadline:
        h_float = best_h.copy()
        adjust_sum_exact(h_float, rng)
        # cascade of β values
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() >= deadline:
                break
            try:
                h_float = slsqp_optimize(h_float, beta, max_iter=600, rng=rng)
                c_new = compute_c5(h_float)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_float.copy()
            except Exception:
                continue

    # --- final sanity checks ---
    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = int(best_h.size)

    return best_h.tolist(), float(best_c), int(best_n)
