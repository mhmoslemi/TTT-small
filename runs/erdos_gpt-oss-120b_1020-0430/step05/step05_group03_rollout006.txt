def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    # scale to target sum within bounds
    total = h.sum()
    if total == 0:
        h[:] = 1.0 / 2.0
        total = h.sum()
    h = h * (target / total)
    # clip and adjust diff iteratively
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # redistrib diff using random indices while staying within bounds
    max_iter = 1000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        # choose random index where we can add or subtract
        if diff > 0:
            # need to add
            candidates = np.where(h < 1.0)[0]
            if len(candidates) == 0:
                break
            i = rng.choice(candidates)
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
        else:
            # need to subtract
            candidates = np.where(h > 0.0)[0]
            if len(candidates) == 0:
                break
            i = rng.choice(candidates)
            sub = min(h[i], -diff)
            h[i] -= sub
            diff += sub
        it += 1
    return h

def worst_shift_and_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1) # shift k where that correlation happens
    k = abs(idx)
    # compute gradient g
    g = np.ones_like(h)
    if k == 0:
        g = 1.0 - 2.0 * h
    else:
        # first n-k entries
        g[: n - k] -= h[k:]
        # last n-k entries from index k onward
        g[k:] -= h[: n - k]
    return k, g

def worst_gradient(h):

        g = 1.0 - 2.0 * h
    else:
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    return g

def slsqp_optimize(h0, beta, max_iter=600):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum (since SLSQP may have slight violation)
    # Use simple scaling method
    n = x.size
    target = n / 2.0
    total = x.sum()
    if total == 0:
        x[:] = target / n
    else:
        x = x * (target / total)
    # clip then adjust diff if needed
    x = np.clip(x, 0.0, 1.0)
    # adjust diff
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # add diff to random indices within bounds
        rng = np.random.default_rng()
        while abs(diff) > 1e-12:
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def enforce_sum(h, rng, max_iter=1000):
    n = h.size
    target = n / 2.0
    total = h.sum()
    if total != 0:
        h = h * (target / total)
    else:
        h[:] = target / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        # choose random index where we can adjust in direction of diff
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
            if len(candidates) == 0:
                break
            i = rng.choice(candidates)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            candidates = np.where(h > 0.0)[0]
            if len(candidates) == 0:
                break
            i = rng.choice(candidates)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        it += 1
    return h

import time, math, numpy as np
from scipy.optimize import minimize

def enforce_sum(h, rng, max_iter=1000):
    n = h.size
    target = n / 2.0
    total = h.sum()
    if total > 0:
        h = h * (target / total)
    else:
        h[:] = target / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            cand = np.where(h < 1.0)[0]
            if len(cand) == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            cand = np.where(h > 0.0)[0]
            if len(cand) == 0:
                break
            i = rng.choice(cand)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        it += 1
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = h * ((n / 2.0) / h.sum())
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def worst_shift_and_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    g = np.ones_like(h)
    if k == 0:
        g = 1.0 - 2.0 * h
    else:
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    return k, g

def worst_gradient(h):
    _, g = worst_shift_and_gradient(h)
    return g

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb_time(h, rng, time_limit):
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step_scale = 0.6
    decay_cnt = 0
    while time.time() < time_limit:
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        decay_cnt += 1
        if decay_cnt >= 200:
            step_scale = max(step_scale * 0.8, 1e-4)
            decay_cnt = 0
    return best_h, best_c

def gradient_refine(h, rng, step=0.015, max_iter=12):
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def slsqp_optimize(h0, beta, rng, max_iter=600):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def soft_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(soft_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter,
                                             "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, rng)
    return x

def to_binary(h):
    n = h.size
    m = n // 2
    idx = np.argpartition(h, -m)[-m:]
    bh = np.zeros_like(h, dtype=float)
    bh[idx] = 1.0
    return bh

def binary_simulated_annealing(h_bin, rng, time_limit):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    start = time.time()
    while time.time() < time_limit:
        elapsed = time.time() - start
        T = max(0.01, 1.0 - elapsed / (time_limit - start + 1e-12))
        if rng.random() < 0.7:
            _, grad = worst_shift_and_gradient(cur_h)
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            i = ones[np.argmax(grad[ones])]
            j = zeros[np.argmin(grad[zeros])]
        else:
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if new_c < best_c:
            best_c, best_h = new_c, cur_h.copy()
        if new_c < cur_c or (T > 0 and rng.random() < math.exp(-delta / T)):
            cur_c = new_c
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_c
