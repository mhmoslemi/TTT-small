def full_transfer_step(h, g):
    # h, g arrays
    pos_idx = np.where(g > 0)[0]
    neg_idx = np.where(g < 0)[0]
    if pos_idx.size == 0 or neg_idx.size == 0:
        return h
    # sort pos by descending g
    pos_order = np.argsort(-g[pos_idx])
    pos_idx = pos_idx[pos_order]
    # sort neg by ascending g (most negative first)
    neg_order = np.argsort(g[neg_idx])
    neg_idx = neg_idx[neg_order]
    h_new = h.copy()
    # capacities
    pos_cap = h_new[pos_idx]  # can reduce by at most this
    neg_cap = 1.0 - h_new[neg_idx]  # can increase by at most this
    p = 0
    n_idx = 0
    while p < pos_idx.size and n_idx < neg_idx.size:
        i = pos_idx[p]
        j = neg_idx[n_idx]
        delta = min(pos_cap[p], neg_cap[n_idx])
        if delta <= 0:
            # shouldn't happen
            if pos_cap[p] <= 0:
                p += 1
                continue
            if neg_cap[n_idx] <= 0:
                n_idx += 1
                continue
        h_new[i] -= delta
        h_new[j] += delta
        pos_cap[p] -= delta
        neg_cap[n_idx] -= delta
        if pos_cap[p] <= 1e-12:
            p += 1
        if neg_cap[n_idx] <= 1e-12:
            n_idx += 1
    return h_new
```

def compute_c5_fft(h):
    n = h.size
    a = np.fft.rfft(h, n*2)  # compute length 2n conv
    b = np.fft.rfft(1-h, n*2)
    corr = np.fft.irfft(a * b, n*2)
    # corr length 2n-1 after truncation: take first 2n-1
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global state for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # quick exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def full_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear (full) correlation of h with (1-h) using FFT."""
    n = h.size
    size = 2 * n
    H = np.fft.rfft(h, size)
    B = np.fft.rfft(1.0 - h, size)
    conv = np.fft.irfft(H * B, size)
    return conv[: 2 * n - 1]


def compute_c5_fft(h: np.ndarray) -> float:
    """Discrete C5 for step function h."""
    n = h.size
    return np.max(full_corr_fft(h)) * (2.0 / n)


def full_transfer_step(h: np.ndarray, g: np.ndarray) -> np.ndarray:
    """Transfer mass from positive‑gradient indices to negative‑gradient ones."""
    pos = np.where(g > 0)[0]
    neg = np.where(g < 0)[0]
    if pos.size == 0 or neg.size == 0:
        return h
    # Sort: most positive first, most negative first
    pos_ord = np.argsort(-g[pos])
    neg_ord = np.argsort(g[neg])
    pos = pos[pos_ord]
    neg = neg[neg_ord]

    h_new = h.copy()
    pos_cap = h_new[pos].copy()          # max we can subtract
    neg_cap = (1.0 - h_new[neg]).copy()  # max we can add

    i = j = 0
    while i < pos.size and j < neg.size:
        delta = min(pos_cap[i], neg_cap[j])
        if delta <= 0:
            if pos_cap[i] <= 0:
                i += 1
            if neg_cap[j] <= 0:
                j += 1
            continue
        pi = pos[i]
        nj = neg[j]
        h_new[pi] -= delta
        h_new[nj] += delta
        pos_cap[i] -= delta
        neg_cap[j] -= delta
        if pos_cap[i] <= 1e-14:
            i += 1
        if neg_cap[j] <= 1e-14:
            j += 1
    return h_new

            if np.any(h_up < -1e-12) or np.any(h_up > 1 + 1e-12):
                h_up = np.clip(h_up, 0.0, 1.0)
                h_up = enforce_sum(h_up, rng)

        # polish up‑sampled version
        if remaining() > 5.0:
            for beta in (20., 200., 2000., 8000.):
                if remaining() < 2.0:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=500)
                c_up = compute_c5_fft(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ----- final sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
