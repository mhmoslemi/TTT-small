def binary_local_search(h, rng, max_swaps=2000, max_noimprove=500):
    n = len(h)
    best_raw = raw_corr_counts(h)
    best_max = best_raw.max()
    no_improve = 0
    for s in range(max_swaps):
        raw = raw_corr_counts(h)
        max_idx = np.argmax(raw)
        cur_max = raw[max_idx]
        if cur_max < best_max:
            best_max = cur_max
            best_raw = raw.copy()
            no_improve = 0
        else:
            no_improve += 1
        if cur_max == 0:
            break
        k = max_idx - (n - 1)
        # compute mismatched sets
        # define ranges where i and i+k are in bounds
        if k >= 0:
            i_range = np.arange(n - k)
            i_to = i_range + k
        else:
            i_range = np.arange(-k, n)
            i_to = i_range + k
        # mismatched 1->0
        mask1 = (h[i_range] == 1) & (h[i_to] == 0)
        S1 = i_range[mask1]
        # mismatched 0->1
        mask0 = (h[i_range] == 0) & (h[i_to] == 1)
        S0 = i_range[mask0]
        if S1.size == 0 or S0.size == 0:
            # try a different shift
            # try top few shifts
            # get sorted indices descending
            idxs = np.argsort(-raw)  # descending
            swapped = False
            for alt_idx in idxs[1:10]: # check next 9
                alt_k = alt_idx - (n - 1)
                if alt_k >= 0:
                    ir = np.arange(n - alt_k)
                    it = ir + alt_k
                else:
                    ir = np.arange(-alt_k, n)
                    it = ir + alt_k
                mask1 = (h[ir] == 1) & (h[it] == 0)
                S1 = ir[mask1]
                mask0 = (h[ir] == 0) & (h[it] == 1)
                S0 = ir[mask0]
                if S1.size > 0 and S0.size > 0:
                    i = rng.choice(S1)
                    j = rng.choice(S0)
                    h[i], h[j] = h[j], h[i]
                    swapped = True
                    break
            if not swapped:
                # cannot improve further
                break
        else:
            i = rng.choice(S1)
            j = rng.choice(S0)
            h[i], h[j] = h[j], h[i]
    return h

def binary_search_best(h, rng, deadline, max_swaps=2000):
    n = len(h)
    cur_raw = np.correlate(h, 1 - h, mode='full')
    cur_max = cur_raw.max()
    best_raw = cur_raw.copy()
    best_max = cur_max
    total_swaps = 0
    while total_swaps < max_swaps and time.time() < deadline:
        # find worst shift
        max_idx = np.argmax(cur_raw)
        k = max_idx - (n - 1)
        # compute mismatched sets for shift k
        if k >= 0:
            i_range = np.arange(n - k)
            i_to = i_range + k
        else:
            i_range = np.arange(-k, n)
            i_to = i_range + k
        mask1 = (h[i_range] == 1) & (h[i_to] == 0)
        S1 = i_range[mask1]
        mask0 = (h[i_range] == 0) & (h[i_to] == 1)
        S0 = i_range[mask0]
        if S1.size == 0 or S0.size == 0:
            # no improving swap for this shift
            # attempt alternative shift(s)
            # try next worst shift
            # get sorted indices descending
            sorted_idxs = np.argsort(-cur_raw)
            found = False
            for alt_idx in sorted_idxs[1:10]:  # try up to 9 other shifts
                alt_k = alt_idx - (n - 1)
                if alt_k >= 0:
                    ir = np.arange(n - alt_k)
                    it = ir + alt_k
                else:
                    ir = np.arange(-alt_k, n)
                    it = ir + alt_k
                mask1 = (h[ir] == 1) & (h[it] == 0)
                S1 = ir[mask1]
                mask0 = (h[ir] == 0) & (h[it] == 1)
                S0 = ir[mask0]
                if S1.size > 0 and S0.size > 0:
                    k = alt_k
                    i_range = ir
                    i_to = it
                    found = True
                    break
            if not found:
                break  # cannot improve further
        # attempt swaps
        improved = False
        attempts = 0
        max_attempts = 20
        while attempts < max_attempts and not improved and time.time() < deadline:
            i = rng.choice(S1)
            j = rng.choice(S0)
            # perform tentative swap
            h[i], h[j] = h[j], h[i]
            new_raw = np.correlate(h, 1 - h, mode='full')
            new_max = new_raw.max()
            if new_max <= cur_max:
                # keep swap
                cur_raw = new_raw
                cur_max = new_max
                total_swaps += 1
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_raw = cur_raw.copy()
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]
                attempts += 1
        if not improved:
            # cannot find improving swap for this shift, break
            break
    c = best_max * (2.0 / n)

import time
import numpy as np

# ---------------------------------------------------------
# Core helpers
# ---------------------------------------------------------
def compute_c5(h):
    """Return the C5 value for a (float) vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(vec, rng):
    """In‑place redistribution so that sum(vec) = len(vec)/2 exactly."""
    n = vec.size
    target = n / 2.0
    diff = target - vec.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - vec[i])
        else:
            add = max(diff, -vec[i])
        vec[i] += add
        diff = target - vec.sum()
    return vec

# Binary local search (pair‑swap descent)
# ---------------------------------------------------------
def binary_pair_swap_search(h, rng, deadline, max_swaps=800, max_attempts=15):
    """
    Monotone descent on the worst shift by swapping a mismatched 1↔0.
    h is a binary integer NumPy array (0/1) with sum = n/2.
    Returns (h, C5) after search.
    """
    n = h.size
    # initial raw correlation
    cur_raw = np.correlate(h, 1 - h, mode='full')
    cur_max = cur_raw.max()
    best_raw = cur_raw.copy()
    best_max = cur_max
    swaps_done = 0

    while swaps_done < max_swaps and time.time() < deadline:
        # worst shift
        max_idx = np.argmax(cur_raw)
        k = max_idx - (n - 1)

        # range of indices where both i and i+k are inside [0,n)
        if k >= 0:
            i_range = np.arange(n - k)
            i_to = i_range + k
        else:
            i_range = np.arange(-k, n)
            i_to = i_range + k

        # mismatched 1→0 and 0→1 for this shift
        mask1 = (h[i_range] == 1) & (h[i_to] == 0)
        S1 = i_range[mask1]
        mask0 = (h[i_range] == 0) & (h[i_to] == 1)
        S0 = i_range[mask0]

        if S1.size == 0 or S0.size == 0:
            # try a few alternative shifts
            alt_idxs = np.argsort(-cur_raw)  # descending
            found = False
            for alt_idx in alt_idxs[1:10]:
                alt_k = alt_idx - (n - 1)
                if alt_k >= 0:
                    ir = np.arange(n - alt_k)
                    it = ir + alt_k
                else:
                    ir = np.arange(-alt_k, n)
                    it = ir + alt_k
                m1 = (h[ir] == 1) & (h[it] == 0)
                m0 = (h[ir] == 0) & (h[it] == 1)
                S1 = ir[m1]
                S0 = ir[m0]
                if S1.size and S0.size:
                    k = alt_k
                    i_range = ir
                    i_to = it
                    found = True
                    break
            if not found:
                break  # no improving swap available

        # try a few random pairs until an improvement is found
        improved = False
        attempts = 0
        while attempts < max_attempts and not improved and time.time() < deadline:
            i = rng.choice(S1)
            j = rng.choice(S0)
            # tentative swap
            h[i], h[j] = h[j], h[i]
            new_raw = np.correlate(h, 1 - h, mode='full')
            new_max = new_raw.max()
            if new_max <= cur_max:
                cur_raw = new_raw
                cur_max = new_max
                swaps_done += 1
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_raw = cur_raw.copy()
            else:
                # revert
                h[i], h[j] = h[j], h[i]
                attempts += 1

        if not improved:
            break  # stagnation

    final_c5 = best_max * (2.0 / n)
    return h, float(final_c5)
