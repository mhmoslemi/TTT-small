def continuous_sa(h0, rng, deadline, init_temp_factor=0.02, min_temp=1e-6, cooling_rate=0.9995, max_move=0.2):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.shape[0]
    T = max(init_temp_factor * cur_c, 1e-6)
    # maybe increase max_move for early steps: we can multiply max_move by (T / (init_temp_factor*cur_c)) but ignore.
    while time.time() < deadline and T > min_temp:
        proposal = propose_move(h, rng, max_move)
        if proposal is None:
            continue
        h_new, delta = proposal
        new_c = compute_c5(h_new)
        dC = new_c - cur_c
        if dC < 0 or rng.random() < math.exp(-dC / T):
            h = h_new
            cur_c = new_c
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
        # decay
        T *= cooling_rate
    return best_h, best_c

def binary_sa(h_bin, rng, deadline, max_iter=2000000):
    n = h_bin.size
    target = n // 2
    # auto-correlation
    c = np.correlate(h_bin, h_bin, mode='full')
    cur_min = c.min()
    cur_worst = target - cur_min  # max overlap
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    # temperature schedule
    T0 = cur_worst * 0.05 + 1e-4
    decay = 0.9999  # slower cooling
    T = T0
    it = 0
    # no_improve count to break early maybe.
    no_improve = 0
    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # swap i (1->0) and j (0->1)
        # Update correlation efficiently
        # remove contribution of i
        start_i = n - 1 - i
        # hi_vec (as per earlier)
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i+1:]))
        diff_i = -1.0
        c[start_i:start_i + n] += diff_i * hi_vec
        h_bin[i] = 0.0

        # add contribution of j (after i set to 0)
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j+1:]))
        diff_j = 1.0
        c[start_j:start_j + n] += diff_j * hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        # accept or reject
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            # probabilistic accept
            prob = math.exp(-(new_worst - cur_worst) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            # update ones and zeros lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert changes
            # revert j
            h_bin[j] = 0.0
            c[start_j:start_j + n] -= diff_j * hj_vec
            # revert i
            h_bin[i] = 1.0
            c[start_i:start_i + n] -= diff_i * hi_vec
            no_improve += 1

        # temperature decay
        T *= decay
        # early stop if no improvement for long
        if no_improve > 500000:
            break
        it += 1

    return best_h, best_worst

def adjust_sum_exact(h, target, rng):

    n = h.shape[0]
    for _ in range(2000):

            break
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # ensure within bounds
    np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip and adjust sum
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # redistrib
            for _ in range(200):
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

CURRENT_TARGET = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

import time, math, numpy as np
from scipy.optimize import minimize

# ----- global vars for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on smoothed max."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum repair
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def compute_c5(h):
    """Exact C5 for discretised h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def adjust_sum_exact(h, target, rng):
    """Redistribute to hit target exactly."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa(h_bin, rng, deadline, max_iter=2000000):
    """Simulated‑annealing on binary vectors (swap 0↔1)."""
    n = h_bin.size
    target = n // 2
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    T = cur_worst * 0.05 + 1e-4
    decay = 0.9999
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # 1 -> 0 at i
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        diff_i = -1.0
        c[start_i : start_i + n] += diff_i * hi_vec
        h_bin[i] = 0.0

        # 0 -> 1 at j (h_bin already has i=0)
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j + 1 :]))
        diff_j = 1.0
        c[start_j : start_j + n] += diff_j * hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            prob = math.exp(-(new_worst - cur_worst) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h_bin[j] = 0.0
            c[start_j : start_j + n] -= diff_j * hj_vec
            h_bin[i] = 1.0
            c[start_i : start_i + n] -= diff_i * hi_vec
            no_improve += 1

        T *= decay
        if no_improve > 500000:
            break
        it += 1
    return best_h, best_worst

def propose_move(h, rng, max_move):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    # direction random
    if rng.random() < 0.5:
        max_delta = min(max_move, h[i], 1.0 - h[j])
        if max_delta <= 0:
            return None
        delta = rng.random() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        return h_new
    else:
        max_delta = min(max_move, h[j], 1.0 - h[i])
        if max_delta <= 0:
            return None
        delta = rng.random() * max_delta
        h_new = h.copy()
        h_new[j] -= delta
        h_new[i] += delta
        return h_new

def continuous_sa(h0, rng, deadline, init_temp_factor=0.02, min_temp=1e-6,
                  cooling_rate=0.9999, max_move=0.2):
    """Simulated annealing on continuous vectors (mass‑preserving swaps)."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = max(init_temp_factor * cur_c, 1e-6)
    while time.time() < deadline and T > min_temp:
        h_new = propose_move(h, rng, max_move)
        if h_new is None:
            continue
        new_c = compute_c5(h_new)
        dC = new_c - cur_c
        if dC < 0 or rng.random() < math.exp(-dC / T):
            h = h_new
            cur_c = new_c
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
        T *= cooling_rate
    return best_h, best_c

        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, best_n / 2.0, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
