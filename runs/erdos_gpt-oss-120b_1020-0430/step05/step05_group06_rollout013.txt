def max_corr_fft(h):

import time
import numpy as np

# ---------- Global placeholders for the SLSQP objective ----------
_CURRENT_BETA = None
_CURRENT_N = None


def adjust_sum_exact(h, rng):
    """Exact sum correction to enforce Σh = N/2."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def max_corr_fft(h):
    """Maximum of the full correlation h * (1-h) using FFT."""
    n = h.shape[0]
    rev = np.flip(1.0 - h)
    size = 2 * n - 1
    H = np.fft.rfft(h, size)
    R = np.fft.rfft(rev, size)
    conv = np.fft.irfft(H * R, size)
    return float(conv.max())


def compute_c5(h):
    """Exact C5 value."""
    n = h.shape[0]
    return max_corr_fft(h) * (2.0 / n)


def random_binary(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def greedy_binary(n, rng):
    """Greedy construction adding ones one by one minimizing current max overlap."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # current correlation
    # O is initially all zeros because h is all zeros
    for _ in range(L):
        O_adj = O + 1  # effect of setting a new 1 at position p
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start:start + n] - h
            w = window.max()
            if (best_w is None) or (w < best_w):
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0  # ensure zero at shift 0 (always true for binary)
    return h.astype(float), O


def hill_climb_binary(h, cur_c, rng, step_deadline, max_swaps_per_iter=5):
    """Swap‑based hill climbing limited by a per‑iteration deadline."""
    n = h.shape[0]
    current_c = cur_c
    while time.time() < step_deadline:
        if h.sum() == 0 or h.sum() == n:
            break
        # choose positions to swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_c = max_corr_fft(h) * (2.0 / n)
        if new_c < current_c:
            current_c = new_c
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return h, current_c


def perturb_h(h, rng, swaps=3):
    """Randomly swap a few bits to perturb a binary vector."""
    h2 = h.copy()
    n = h2.size
    for _ in range(swaps):
        ones = np.flatnonzero(h2)
        zeros = np.flatnonzero(1 - h2)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h2[i] = 0.0
        h2[j] = 1.0
    return h2


# ---------- SLSQP helpers ----------
def sum_eq_constraint(x):
    return np.sum(x) - _CURRENT_N / 2.0


def softmax_obj(x):
    # raw correlation (no scaling)
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = _CURRENT_BETA * corr
    m = np.max(bc)
    return (np.log(np.sum(np.exp(bc - m))) + m) / _CURRENT_BETA


# ---------- Main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98  # safety margin

    best_h = None
    best_c = float("inf")
    best_n = None

    # --- start from parent if provided ---
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        c_parent = compute_c5(parent)
        if c_parent < best_c:
            best_h = parent.copy()
            best_c = c_parent
            best_n = parent.size

    # --- work with n = 720 (good trade‑off) ---
    n = 720
    if time.time() >= deadline:
        # fallback if no time left
        if best_h is None:
            best_n = n
            best_h = np.zeros(n, dtype=float)
            best_h[: n // 2] = 1.0
            best_c = compute_c5(best_h)
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        return best_h.tolist(), float(best_c), int(best_n)

    # initial greedy construction
    h_greedy, _ = greedy_binary(n, rng)
    c_greedy = compute_c5(h_greedy)
    if c_greedy < best_c:
        best_h = h_greedy.copy()
        best_c = c_greedy
        best_n = n

    # --- repeated hill‑climbing attempts ---
    # allocate ~2 sec per attempt, stop when little time remains
    while True:
        remaining = deadline - time.time()
        if remaining < 3.0:
            break
        # choose start: 70% random, 30% perturb best
        if best_h is not None and rng.random() < 0.3:
            h0 = perturb_h(best_h, rng, swaps=5)
        else:
            h0 = random_binary(n, rng)
        c0 = compute_c5(h0)
        step_deadline = time.time() + min(2.0, remaining / 2.0)
        h_opt, c_opt = hill_climb_binary(h0, c0, rng, step_deadline)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            best_n = n

    # --- final smooth‑max SLSQP refinement (if time permits) ---
    if time.time() < deadline and best_h is not None and best_h.size == n:
        h_float = best_h.astype(float).copy()
        adjust_sum_exact(h_float, rng)

        _CURRENT_N = n
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        # a few beta values, modest iteration budget
        for beta in (200, 500, 1000, 2000):
            if time.time() >= deadline:
                break
            _CURRENT_BETA = beta
            try:
                res = opt.minimize(
                    softmax_obj,
                    h_float,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 400, "ftol": 1e-12, "disp": False},
                )
                h_float = np.clip(res.x, 0.0, 1.0)
                adjust_sum_exact(h_float, rng)
                c_new = compute_c5(h_float)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_float.copy()
                    best_n = n
            except Exception:
                continue

    # --- final safety checks ---
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
