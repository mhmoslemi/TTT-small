function polish_h(h, rng, deadline):
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    # Determine initial temperature as fraction of cur_c
    init_T = max(1e-4, 0.02 * cur_c)
    T = init_T
    decay = 0.9995
    while time.time() < deadline:
        i, j = rng.choice(len(cur), size=2, replace=False)
        # ensure transfer possible
        max_transfer = min(cur[i], 1 - cur[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random()
        cur[i] -= delta
        cur[j] += delta
        c = compute_c5(cur)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c; best = cur.copy()
        else:
            cur[i] += delta; cur[j] -= delta
        T = max(1e-12, T * decay)
    return best, best_c

import time, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq}
    res = minimize(_softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- gradient refinement ----------
def gradient_refine(h, rng, step=0.025, iters=20):
    n = h.size
    target = n / 2.0
    for it in range(iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = abs(np.argmax(corr) - (n - 1))
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- fractional simulated annealing ----------
def fractional_sa(h, rng, deadline, init_T_factor=0.02, no_improve_limit=20000):
    cur = h.copy()
    cur_c = compute_c5(cur)
    best, best_c = cur.copy(), cur_c
    T = max(1e-12, init_T_factor * cur_c)
    decay = 0.9995
    no_improve = 0
    n = len(cur)
    while time.time() < deadline and no_improve < no_improve_limit:
        i, j = rng.choice(n, size=2, replace=False)
        max_tr = min(cur[i], 1.0 - cur[j])
        if max_tr <= 0:
            continue
        delta = rng.random() * max_tr
        cur[i] -= delta
        cur[j] += delta
        c = compute_c5(cur)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c, best = c, cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            cur[i] += delta
            cur[j] -= delta
            no_improve += 1
        T = max(1e-12, T * decay)
    return best, best_c
