def simulated_annealing(h0, rng, time_budget, target):
    n = h0.size
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    start = time.time()
    T0 = 0.2
    T_end = 1e-5
    max_step_factor = 1.0
    min_step_factor = 0.001
    # track iterations for periodic sum correction
    it = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac
        # pick i != j
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # max epsilon based on box constraints
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            # try opposite direction
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            # swap indices to maintain direction (now move from j to i)
            i, j = j, i
        # epsilon selection
        # step factor scales with temperature
        step_factor = max(min_step_factor, T / T0)
        eps = rng.random() * max_eps * step_factor
        if eps < 1e-12:
            continue
        # apply
        h[i] -= eps
        h[j] += eps
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
        # occasional sum correction to avoid drift
        if (it & 0xFF) == 0:
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                idx = rng.integers(n)
                add = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, -h[idx])
                h[idx] += add
        it += 1
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 30) -> np.ndarray:
    """Fast deterministic refinement using worst‑gradient."""
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        grad = worst_gradient(h)
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            i = rng.integers(n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
    return h


# --------------------------------------------------------------
def simulated_annealing(h0: np.ndarray, rng: np.random.Generator,
                        time_budget: float, target: float):
    """Continuous SA that preserves Σh = target."""
    n = h0.size
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    start = time.time()
    T0, T_end = 0.2, 1e-5
    min_step_factor = 0.001
    it = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            # try opposite direction
            max_eps = min(h[j], 1.0 - h[i])
            if max_eps <= 0.0:
                continue
            i, j = j, i
        step_factor = max(min_step_factor, T / T0)
        eps = rng.random() * max_eps * step_factor
        if eps < 1e-12:
            continue
        h[i] -= eps
        h[j] += eps
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
        else:
            h[i] += eps
            h[j] -= eps
        if (it & 0xFF) == 0:  # occasional sum correction
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                idx = rng.integers(n)
                add = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, -h[idx])
                h[idx] += add
        it += 1
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ----- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n0 = 512
        h0 = np.zeros(n0, dtype=float)
        ones = rng.choice(n0, n0 // 2, replace=False)
        h0[ones] = 1.0
    n = h0.size
    target = n / 2.0
    h0 = enforce_sum(h0, target, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ----- main SA pool --------------------------------------------------
    time_used = time.time() - start
    time_left = total_budget - time_used

    # first SA from the (possibly) good parent
    if time_left > 30:
        t = min(30.0, time_left * 0.5)
        cand_h, cand_c = simulated_annealing(best_h, rng, t, target)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c
        time_left = total_budget - (time.time() - start)

    # many short SA runs from random binary starts
    while time_left > 20:
        t = min(20.0, time_left * 0.5)
        h_rand = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h_rand[idx] = 1.0
        h_rand = enforce_sum(h_rand, target, rng)
        cand_h, cand_c = simulated_annealing(h_rand, rng, t, target)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c
        time_left = total_budget - (time.time() - start)

    # ----- final deterministic refinement ----------------------------------
    time_remaining = total_budget - (time.time() - start)
    if time_remaining > 2.0:
        iters = max(1, int(time_remaining * 200))  # about 0.01 s per iter
        best_h = gradient_refine(best_h, rng, step=0.02, max_iter=iters)

    # optional brief SLSQP polish if still time
    time_remaining = total_budget - (time.time() - start)
    if time_remaining > 5.0:
        for beta in (10., 100., 800., 3000.):
            if time.time() - start > total_budget * 0.99:
                break
            h_opt = slsqp_optimize(best_h, beta, rng, max_iter=300)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # ----- final sanitisation --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
