def gradient_softmax(h, beta):
    n = h.shape[0]
    scaled_corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    max_sc = scaled_corr.max()
    exp_term = np.exp(beta * (scaled_corr - max_sc))
    p = exp_term / exp_term.sum()
    # compute padded h
    pad = np.zeros(n-1)
    h_pad = np.concatenate([pad, h, pad])
    # g1 = np.correlate(h_pad, p, mode='valid')
    g1 = np.correlate(h_pad, p, mode='valid')
    p_rev = p[::-1]
    g2 = np.correlate(h_pad, p_rev, mode='valid')
    g = (2.0 / n) * (1.0 - (g1 + g2))
    return g
```

    n = h.size
    L = 2*n - 1
    # compute scaling factor
    # compute correlation scaled
    corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    D = compute_D_matrix(h)
    best_c = corr.max()
    no_improve = 0
    while time.time() < deadline:
        # find worst shift index
        idx = np.argmax(corr)
        # shift k
        k = idx - (n - 1)
        # compute S1, S0 for this shift (as before)
        i_range = np.arange(n)
        # Check boundaries for i+k: 
        valid = (i_range + k >= 0) & (i_range + k < n)
        cond_i = (h == 1) & valid & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & valid & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        # generate candidate pairs
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 5000
        if total_pairs <= max_samples:
            pairs = [(i,j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_new_c = best_c
        # Evaluate all pairs
        for i,j in pairs:
            delta = D[i] - D[j]
            new_c = np.max(corr + delta)
            if new_c < best_new_c - 1e-12:
                best_new_c = new_c
                best_pair = (i,j)
        if best_pair is None:
            # no improvement, maybe break or try random swap
            no_improve += 1
            if no_improve > 5:
                break
            # maybe do a random swap to escape
            i = rng.choice(S1)
            j = rng.choice(S0)
            best_pair = (i,j)
            delta = D[i] - D[j]
            # accept even if not improvement?
            # To avoid stuck, we could accept if improvement not found after some attempts.
            continue
        else:
            # apply swap
            i,j = best_pair
            h[i] = 0.0
            h[j] = 1.0

            delta = D[i] - D[j]  # using previous D before swap? But after swap D recomputed anyway.

import time
import numpy as np

# -------------------------------------------------
# basic utilities
def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force sum(h) = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def binary_from_fractional(frac: np.ndarray) -> np.ndarray:
    """Top‑half thresholding to obtain exactly n/2 ones."""
    n = frac.size
    idx = np.argpartition(frac, -n // 2)[-n // 2 :]
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec


# -------------------------------------------------
# D‑matrix for fast delta computation (binary case)
def compute_D_matrix(h: np.ndarray) -> np.ndarray:
    """Return D[i, s] = h_{i+s}+h_{i-s} for all shifts s (‑n+1…n‑1)."""
    n = h.shape[0]
    L = 2 * n - 1
    pad = np.zeros(n - 1, dtype=h.dtype)
    h_pad = np.concatenate([pad, h, pad])
    D = np.empty((n, L), dtype=float)
    for i in range(n):
        F = h_pad[i : i + L]                 # forward window
        B = h_pad[i + L - 1 : i - 1 : -1]    # backward (reversed) window
        D[i] = F + B
    return D


# -------------------------------------------------
# smooth‑max gradient (continuous case)
def gradient_softmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/β)·log∑exp(β·corr) w.r.t. h, where corr is scaled overlap."""
    n = h.size
    scaled_corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = scaled_corr.max()
    exp_term = np.exp(beta * (scaled_corr - m))
    p = exp_term / exp_term.sum()
    pad = np.zeros(n - 1, dtype=h.dtype)
    h_pad = np.concatenate([pad, h, pad])
    g1 = np.correlate(h_pad, p, mode="valid")
    p_rev = p[::-1]
    g2 = np.correlate(h_pad, p_rev, mode="valid")
    g = (2.0 / n) * (1.0 - (g1 + g2))
    return g


# -------------------------------------------------
# binary local search using D‑matrix delta trick
def local_search_binary(h: np.ndarray, rng: np.random.Generator,
                        deadline: float, max_swaps: int = 150,
                        max_samples: int = 2000) -> tuple[np.ndarray, float]:
    """Improve binary h by swapping 1↔0 to lower max overlap."""
    n = h.size
    L = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    D = compute_D_matrix(h)
    best_c = corr.max()
    swaps = 0
    while swaps < max_swaps and time.time() < deadline:
        idx = int(np.argmax(corr))
        k = idx - (n - 1)
        i_range = np.arange(n)
        valid = (i_range + k >= 0) & (i_range + k < n)
        cond_i = (h == 1) & valid & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & valid & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]

        best_pair = None
        best_new_c = best_c
        for i, j in pairs:
            delta = D[i] - D[j]            # length L
            new_c = np.max(corr + delta)
            if new_c < best_new_c - 1e-12:
                best_new_c = new_c
                best_pair = (i, j)
        if best_pair is None:
            break

        i, j = best_pair
        delta = D[i] - D[j]
        h[i] = 0.0
        h[j] = 1.0
        corr = corr + delta
        D = compute_D_matrix(h)
        best_c = best_new_c
        swaps += 1
    return h, best_c
