def smooth_max_and_grad(h, beta):
    n = h.size
    g = 1.0 - h
    L = 2*n - 1
    # compute correlation via FFT
    N = 1 << ((L + n - 1).bit_length())
    fft_h = np.fft.rfft(h, N)
    fft_g = np.fft.rfft(g, N)
    corr = np.fft.irfft(fft_h * fft_g, N)
    corr = corr[:L]  # length 2n-1
    # smooth max
    max_corr = corr.max()
    # compute weights
    exp_term = np.exp(beta * (corr - max_corr))
    Z = exp_term.sum()
    w = exp_term / Z  # length L
    # compute conv w * g
    fft_w = np.fft.rfft(w, N)
    conv_w_g = np.fft.irfft(fft_w * fft_g, N)  # length N
    # slice term1
    start = n - 1
    term1 = conv_w_g[start:start + n]
    # compute conv w_rev * h
    w_rev = w[::-1]
    fft_w_rev = np.fft.rfft(w_rev, N)
    fft_h = np.fft.rfft(h, N)
    conv_w_rev_h = np.fft.irfft(fft_w_rev * fft_h, N)
    term2 = conv_w_rev_h[start:start + n]
    grad = term1 - term2
    # objective value f = max_corr + np.log(Z)/beta   (alternatively compute as below)
    f = max_corr + np.log(Z) / beta
    return f, grad

def slsqp_optimize_with_grad(h0, beta, max_iter, rng):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {
        'type':'eq',
        'fun': lambda x: np.sum(x) - target,
        'jac': lambda x: np.ones_like(x)
    }
    def obj(x):
        f, _ = smooth_max_and_grad(x, beta)
        return f
    def obj_jac(x):
        _, g = smooth_max_and_grad(x, beta)
        return g
    res = minimize(obj, h0, method='SLSQP', jac=obj_jac, bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

    cons = {
        "type": "eq",
        "fun": lambda x: np.sum(x) - target,
        "jac": lambda x: np.ones_like(x),
    }
    def obj(x):
        f, _ = smooth_max_and_grad(x, beta)
        return f
    def obj_jac(x):
        _, g = smooth_max_and_grad(x, beta)
        return g
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        jac=obj_jac,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def generate_random_continuous(n, rng):
    h = rng.random(n)
    adjust_sum_exact(h, rng)
    return h

def generate_random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    return h

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    # base dimension
    n = 360
    target = n / 2.0
    best_h = None
    best_c = np.inf
    best_n = n

    # parent construction, if given
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            n = h0.size
            target = n / 2.0
            best_n = n
        best_h = h0.copy()
        best_c = compute_c5(best_h)
        best_n = n

    # build candidate pool
    candidates = []
    if best_h is not None:
        candidates.append(best_h.copy())
    for _ in range(12):
        candidates.append(generate_random_continuous(n, rng))
    for _ in range(12):
        candidates.append(generate_random_binary(n, rng))

    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]

    for h in candidates:
        if time.time() > deadline:
            break
        cur = h.copy()
        adjust_sum_exact(cur, rng)
        for beta in beta_schedule:
            if time.time() > deadline:
                break
            max_iter = 300 if beta < 200 else 500
            cur = slsqp_optimize_with_grad(cur, beta, max_iter, rng)
        c = compute_c5(cur)
        if c < best_c:
            best_c = c
            best_h = cur.copy()
            best_n = n

    # fallback safety
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
