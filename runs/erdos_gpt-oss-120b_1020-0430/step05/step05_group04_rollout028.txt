def compute_delta(h, i, j):
    N = h.shape[0]
    delta = np.zeros(N, dtype=int)
    # i contributions
    # i+k
    if i+1 < N:
        delta[:N-i-1] -= h[i+1:]
    # i-k
    if i > 0:
        delta[1:i+1] -= h[i-1::-1]
    # j contributions
    if j+1 < N:
        delta[:N-j-1] += h[j+1:]
    if j > 0:
        delta[1:j+1] += h[j-1::-1]
    # delta[0] stays 0
    return delta

def binary_sa_optimize(h_start, rng, deadline, max_iter=2000000):
    N = h_start.shape[0]
    h = h_start.astype(np.int8).copy()
    # compute P
    P = np.correlate(h, h, mode='full')[N-1:].astype(np.int32)
    cur_max = N//2 - int(P.min())
    best_h = h.copy()
    best_max = cur_max

    # positions
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    pos_in_ones = np.full(N, -1, dtype=np.int32)
    for idx, val in enumerate(ones):
        pos_in_ones[val] = idx
    pos_in_zeros = np.full(N, -1, dtype=np.int32)
    for idx, val in enumerate(zeros):
        pos_in_zeros[val] = idx

    # temperature
    T = max(cur_max * 0.5, 1e-6)
    decay = 0.99995
    # optionally allow reheating if stagnation
    last_improve_iter = 0
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # choose random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # compute delta
        delta = np.empty(N, dtype=np.int32)
        delta.fill(0)
        if i+1 < N:
            delta[:N-i-1] -= h[i+1:]
        if i > 0:
            delta[1:i+1] -= h[i-1::-1]
        if j+1 < N:
            delta[:N-j-1] += h[j+1:]
        if j > 0:
            delta[1:j+1] += h[j-1::-1]
        # delta[0] remains 0

        # compute new min P
        # To avoid copying P+delta we can compute min by iterating? Using numpy: newP = P + delta; new_min = newP.min()
        newP_min = (P + delta).min()
        new_max = N//2 - int(newP_min)
        d = new_max - cur_max

        accept = False
        if d <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-d / T):
                accept = True
        if accept:
            P += delta
            cur_max = new_max
            # update h
            h[i] = 0
            h[j] = 1
            # update ones/zeros lists and position maps
            # i -> zeros, j -> ones
            idx_i = pos_in_ones[i]
            idx_j = pos_in_zeros[j]
            # swap in ones list: replace i with j
            ones[idx_i] = j
            pos_in_ones[j] = idx_i
            pos_in_ones[i] = -1
            # swap in zeros list: replace j with i
            zeros[idx_j] = i
            pos_in_zeros[i] = idx_j
            pos_in_zeros[j] = -1
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                last_improve_iter = it
        # else reject

        # Update temperature
        T = max(T * decay, 1e-9)

        # If no improvement for many iterations, reheat
        if it - last_improve_iter > 200000:
            T = max(cur_max * 0.5, 1e-6)
            last_improve_iter = it

    return best_h.astype(np.float64), best_max

def greedy_reduce_max(h, P, max_iter, rng):
    N = len(h)
    best_h = h
    best_P = P
    best_max = N//2 - int(P.min())
    for _ in range(max_iter):
        minP = P.min()
        ks = np.where(P == minP)[0]  # shift(s) with min product (worst)
        k = ks[0]  # choose first
        # compute mismatched sets for this shift
        # S1: i where h[i]==1 and i+k<N and h[i+k]==0
        # S0: j where h[j]==0 and j+k<N and h[j+k]==1
        # We can compute arrays
        i_range = np.arange(N-k)
        S1 = i_range[(h[i_range]==1) & (h[i_range+k]==0)]
        j_range = np.arange(k, N)
        # Actually for S0 we need j where h[j]==0 and h[j-k]==1 (since j = i+k, i in S1). Equivalent.
        S0 = np.arange(k, N)[(h[np.arange(k, N)]==0) & (h[np.arange(k, N)-k]==1)]

        if S1.size == 0 or S0.size == 0:
            break
        # if total pairs small, evaluate all
        total_pairs = S1.size * S0.size
        sample_pairs = 200
        if total_pairs <= sample_pairs:
            pairs = [(i,j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(sample_pairs)]
        improved = False
        for i, j in pairs:
            # compute delta for this swap
            delta = np.empty(N, dtype=np.int32)
            delta.fill(0)
            if i+1 < N:
                delta[:N-i-1] -= h[i+1:]
            if i > 0:
                delta[1:i+1] -= h[i-1::-1]
            if j+1 < N:
                delta[:N-j-1] += h[j+1:]
            if j > 0:
                delta[1:j+1] += h[j-1::-1]
            newPmin = (P + delta).min()
            new_max = N//2 - int(newPmin)
            if new_max < best_max:
                # accept
                P += delta
                h[i] = 0; h[j] = 1
                best_max = new_max
                improved = True
                break
        if not improved:
            break
    return h, P, best_max

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact overlap constant for a (float) vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- gradient refinement ----------
def gradient_refine(h, rng, step=0.02, max_iter=40):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # + (1 - h[i+k])
            g[k:] -= h[:upper]               # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- smooth‑max surrogate for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=200):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- binary simulated annealing ----------
def binary_sa_optimize(h_start, rng, deadline):
    """SA on binary vectors using an O(N) incremental overlap update."""
    N = h_start.shape[0]
    h = h_start.astype(np.int8).copy()
    P = np.correlate(h, h, mode="full")[N - 1 :].astype(np.int32)  # product correlation
    cur_max = N // 2 - int(P.min())
    best_h = h.copy()
    best_max = cur_max

    # position maps for O(1) swaps
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    pos_one = np.full(N, -1, dtype=np.int32)
    for idx, v in enumerate(ones):
        pos_one[v] = idx
    pos_zero = np.full(N, -1, dtype=np.int32)
    for idx, v in enumerate(zeros):
        pos_zero[v] = idx

    T = max(cur_max * 0.5, 1e-6)
    decay = 0.99995
    last_improve = 0
    it = 0
    while time.time() < deadline:
        it += 1
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        # delta for all shifts
        delta = np.empty(N, dtype=np.int32)
        delta.fill(0)
        if i + 1 < N:
            delta[: N - i - 1] -= h[i + 1 :]
        if i > 0:
            delta[1 : i + 1] -= h[i - 1 :: -1]
        if j + 1 < N:
            delta[: N - j - 1] += h[j + 1 :]
        if j > 0:
            delta[1 : j + 1] += h[j - 1 :: -1]

        new_minP = (P + delta).min()
        new_max = N // 2 - int(new_minP)
        d = new_max - cur_max

        accept = False
        if d <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-d / T):
                accept = True

        if accept:
            P += delta
            cur_max = new_max
            # update h and the position lists
            h[i] = 0
            h[j] = 1
            idx_i = pos_one[i]
            idx_j = pos_zero[j]
            ones[idx_i] = j
            pos_one[j] = idx_i
            pos_one[i] = -1
            zeros[idx_j] = i
            pos_zero[i] = idx_j
            pos_zero[j] = -1

            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                last_improve = it

        T = max(T * decay, 1e-9)
        if it - last_improve > 200000:
            T = max(cur_max * 0.5, 1e-6)
            last_improve = it
    return best_h.astype(np.float64), best_max

def greedy_reduce_max(h, P, max_iter, rng):
    """Target the current worst shift and try improving swaps."""
    N = h.shape[0]
    cur_max = N // 2 - int(P.min())
    for _ in range(max_iter):
        minP = P.min()
        ks = np.where(P == minP)[0]
        k = ks[0]  # pick one worst shift

        # mismatched positions for this shift
        i_range = np.arange(N - k)
        S1 = i_range[(h[i_range] == 1) & (h[i_range + k] == 0)]
        j_range = np.arange(k, N)
        S0 = j_range[(h[j_range] == 0) & (h[j_range - k] == 1)]

        if S1.size == 0 or S0.size == 0:
            break

        total = int(S1.size) * int(S0.size)
        sample = 200
        if total <= sample:
            pairs = [(int(i), int(j)) for i in S1 for j in S0]
        else:
            pairs = [(int(rng.choice(S1)), int(rng.choice(S0))) for __ in range(sample)]

        improved = False
        for i, j in pairs:
            delta = np.empty(N, dtype=np.int32)
            delta.fill(0)
            if i + 1 < N:
                delta[: N - i - 1] -= h[i + 1 :]
            if i > 0:
                delta[1 : i + 1] -= h[i - 1 :: -1]
            if j + 1 < N:
                delta[: N - j - 1] += h[j + 1 :]
            if j > 0:
                delta[1 : j + 1] += h[j - 1 :: -1]

            new_minP = (P + delta).min()
            new_max = N // 2 - int(new_minP)
            if new_max < cur_max:
                P += delta
                h[i] = 0
                h[j] = 1
                cur_max = new_max
                improved = True
                break
        if not improved:
            break
    return h, P, cur_max
