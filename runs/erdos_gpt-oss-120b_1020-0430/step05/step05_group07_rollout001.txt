import time, math
import numpy as np
from scipy.optimize import minimize

# Global placeholders for constraints used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Maximum overlap for a step function sample vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iters=2000):
    """Redistribute mass so that sum(h) == n/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iters):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k ∫h(x)(1−h(x+k))."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=600):
    """SLSQP minimizing the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def gradient_refine(h, rng, step=0.04, max_iter=15):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]          # derivative from h_i term
            g[k:]  -= h[:up]               # derivative from -h_{i+k} term
        step_sz = step * (0.5 ** it)
        h -= step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # quick redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    """Swap‑based SA for binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < math.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 8000:
            break
    return best_h, best_c

def fractional_sa_hill_climb(h, rng, time_deadline):
    """Transfer‑based SA for continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < math.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 12000:
            break
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # Primary resolution
    n = 360
    candidates = []

    # Parent if provided
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:

            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            parent = np.interp(xt, xs, parent)
        candidates.append(parent.copy())

    # Simple periodic binary blocks
    for p in range(2, n // 4, 2):
        if n % p != 0:
            continue
        block = np.zeros(p)
        block[:p // 2] = 1.0
        h = np.tile(block, n // p)
        candidates.append(h.copy())

    # Random binary and fractional seeds
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    candidates.append(bin_rand)

    rand_frac = rng.random(n)
    target = n / 2.0
    rand_frac *= target / rand_frac.sum()
    adjust_sum_exact(rand_frac, rng)
    candidates.append(rand_frac)

    rand_frac2 = rng.random(n) ** 2
    rand_frac2 *= target / rand_frac2.sum()
    adjust_sum_exact(rand_frac2, rng)
    candidates.append(rand_frac2)

    # Random binary sampling → top‑k
    top_k = 20
    num_samples = 50000
    best_bin_list = []
    for _ in range(num_samples):
        if time.time() > deadline - 5:
            break
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        c = compute_c5(h)
        if len(best_bin_list) < top_k:
            best_bin_list.append((c, h.copy()))
            best_bin_list.sort(key=lambda x: x[0])
        elif c < best_bin_list[-1][0]:
            best_bin_list[-1] = (c, h.copy())
            best_bin_list.sort(key=lambda x: x[0])
    for _, h in best_bin_list:
        candidates.append(h.copy())

    # Process each candidate
    for h0 in candidates:
        if time.time() > deadline:
            break
        h = np.clip(h0, 0.0, 1.0)
        adjust_sum_exact(h, rng)

        h = gradient_refine(h, rng, step=0.04, max_iter=15)
        c = compute_c5(h)
        if c < best_c:

        # SLSQP cascade

            max_iter = 800 if beta >= 5000 else 500
            try:

            except Exception:
                continue
            c = compute_c5(h)
            if c < best_c:

        h = gradient_refine(h, rng, step=0.02, max_iter=10)
        c = compute_c5(h)

        if c_up < best_c:

    # Fallback safe construction

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
