def slsqp_optimize(h0, betas, rng, max_iter_per_beta):
    n = h0.size
    target = n/2.0
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    bounds = [(0.,1.)]*n
    cons = {'type':'eq','fun': sum_eq_constraint}
    h = np.clip(h0, 0.,1.)
    for beta, max_iter in zip(betas, max_iter_per_beta):
        CURRENT_BETA = beta
        # minimize smooth_max_obj
        res = minimize(smooth_max_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        h = np.clip(res.x, 0., 1.)
        # exact sum correction using adjust_sum_exact
        h = adjust_sum_exact(h, rng)
    return h
```

# global placeholders for constraints
CURRENT_TARGET = None
CURRENT_BETA = None

def corr_full(h):
    """Linear correlation of h with (1-h) using FFT (full length)."""
    n = h.size
    N = 2 * n
    fft_h = np.fft.fft(h, N)
    fft_1mh = np.fft.fft(1.0 - h, N)
    conv = np.fft.ifft(fft_h * fft_1mh).real
    return conv[: 2 * n - 1]

def compute_c5(h):
    n = h.size
    corr = corr_full(h)
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def smooth_max_obj(x):
    n = x.size
    corr = corr_full(x)
    # stable log‑sum‑exp
    lse = logsumexp(CURRENT_BETA * corr)
    smooth = lse / CURRENT_BETA
    return smooth * (2.0 / n)

def adjust_sum_exact(h, rng, max_iter=2000):
    """Redistribute a tiny mass to satisfy Σh = n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0

                else:
                    add = max(diff, -h[i])
                h[i] += add

                if abs(diff) < 1e-12:
                    break
    return h

def slsqp_optimize(h0, betas, max_iters, rng):
    """Cascade of SLSQP passes on smooth‑max surrogate."""
    n = h0.size
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    h = np.clip(h0, 0.0, 1.0)
    for beta, itmax in zip(betas, max_iters):
        CURRENT_BETA = beta
        res = minimize(
            smooth_max_obj,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": itmax, "ftol": 1e-12, "disp": False},
        )
        h = np.clip(res.x, 0.0, 1.0)
        h = adjust_sum_exact(h, rng)
    return h

def fractional_sa_hill_climb(h, rng, time_deadline):
    """Transfer‑based SA on continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c, best = c, cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 12000:
            break
    return best, best_c
