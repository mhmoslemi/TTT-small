def worst_shift_grad_descent(h, rng, steps=100, step0=0.1):
    n = h.size
    target = n/2.0
    for it in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0*h
        else:
            # np.roll: np.roll(h, -k) shifts left (i.e., index i becomes i-k)

    decay = 0.99995
    it = 0
    no_improve = 0

        j = rng.choice(zeros)
