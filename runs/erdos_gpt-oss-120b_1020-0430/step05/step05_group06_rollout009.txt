def simulated_anneal(h, c0, rng, deadline):
    n = len(h)
    sum_h = h.sum()
    best_h = h.copy()
    best_c = c0
    cur_h = h.copy()
    cur_c = c0
    # precompute ones and zeros indices
    ones = np.flatnonzero(cur_h)
    zeros = np.flatnonzero(1 - cur_h) # or np.where(cur_h ==0)
    # time schedule
    start = time.time()
    total = deadline - start
    # parameters for temperature
    T0 = (c0) * 0.1 # maybe 0.01
    Tf = 1e-7
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = T0 * (Tf / T0) ** frac
        # pick random swap
        # ensure at least one one and zero
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        # compute new c
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_c = new_c
            # update indices
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
        # maybe refresh ones, zeros if shape changed, but swaps preserve counts.
    return best_h, best_c

def simulated_anneal(h_start, c_start, rng, deadline):
    cur_h = h_start.copy()
    cur_c = c_start
    best_h = cur_h.copy()
    best_c = cur_c
    n = len(cur_h)
    # ensure binary
    cur_h = np.round(cur_h).astype(float)  # in case non-binary
    # Precompute ones and zeros sets
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    start = time.time()
    total = max(deadline - start, 1e-6)
    # temperature schedule
    T0 = max(cur_c * 0.1, 1e-3)
    Tf = 1e-6
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = T0 * (Tf / T0) ** frac
        # pick random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_c = new_c
            # update indices
            # find positions in lists
            # Use np.where? but we can find via array search (slow). Better store indices map?

            idx_i = np.where(ones == i)[0][0]

def softmax_obj(x):
    # compute smooth max of overlap values: overlap for each shift k = h_i * (1 - h_{i+k}) sum
    # compute correlation h with (1 - h)
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    # stable log-sum-exp
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    # method SLSQP
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----- utilities ---------------------------------------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """C5 value for a vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Force Σh = n/2 exactly by tiny adjustments."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth‑max surrogate of the overlap vector."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def greedy_binary(n, rng):
    """Greedy construction of a binary sequence (0/1) with half density."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # correlation of all‑zero vector
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0  # O_0 must stay zero
    return h, O


def simulated_anneal(h_start, c_start, rng, deadline):
    """SA on binary vectors (swap 1↔0) aiming to lower C5."""
    n = len(h_start)
    cur_h = np.round(h_start).astype(float).copy()
    cur_c = c_start
    best_h = cur_h.copy()
    best_c = cur_c

    # maintain O(1) updates of the index lists
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    pos_in_ones = np.full(n, -1, dtype=int)
    pos_in_zeros = np.full(n, -1, dtype=int)
    for idx, i in enumerate(ones):
        pos_in_ones[i] = idx
    for idx, i in enumerate(zeros):
        pos_in_zeros[i] = idx

    start = time.time()
    total = max(deadline - start, 1e-9)
    T0 = max(cur_c * 0.05, 1e-4)
    Tf = 1e-7

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = T0 * (Tf / T0) ** frac
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = (delta <= 0) or (rng.random() < np.exp(-delta / T))

        if accept:
            cur_c = new_c
            # update index structures
            idx_i = pos_in_ones[i]
            idx_j = pos_in_zeros[j]
            ones[idx_i] = j
            zeros[idx_j] = i
            pos_in_ones[i] = -1
            pos_in_zeros[i] = idx_j
            pos_in_ones[j] = idx_i
            pos_in_zeros[j] = -1
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    return best_h, best_c


# ----- main entry --------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # parent construction, if provided
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        c_parent = compute_c5(parent)
        if c_parent < best_c:
            best_h = parent.copy()
            best_c = c_parent
            best_n = parent.size

    # only the 360‑sample version (fast and already good)
    candidate_ns = [360]

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        # ----- greedy binary start -----
        h_bin, _ = greedy_binary(n, rng)
        c_val = compute_c5(h_bin.astype(float))
        if c_val < best_c:
            best_c = c_val
            best_h = h_bin.astype(float).copy()
            best_n = n

        # ----- deterministic hill climbing (simple swaps) -----
        cur_h = h_bin.astype(float).copy()
        cur_c = c_val
        improved = True
        while improved and time.time() < deadline:
            improved = False
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
            if len(ones) == 0 or len(zeros) == 0:
                break
            for _ in range(1000):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                cur_h[i] = 0.0
                cur_h[j] = 1.0
                new_c = compute_c5(cur_h)
                if new_c < cur_c - 1e-12:
                    cur_c = new_c
                    improved = True
                    # update index arrays
                    idx_i = np.where(ones == i)[0][0]
                    idx_j = np.where(zeros == j)[0][0]
                    ones[idx_i] = j
                    zeros[idx_j] = i
                    break
                else:
                    cur_h[i] = 1.0
                    cur_h[j] = 0.0
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
            best_n = n

        # ----- simulated annealing (multiple restarts) -----
        if time.time() < deadline:
            remaining = deadline - time.time()
            restarts = 4
            for r in range(restarts):
                if time.time() >= deadline:
                    break
                # start point
                if best_n == n and best_h is not None and r == 0:
                    start_h = best_h.copy()
                else:
                    start_h = np.zeros(n, dtype=float)
                    start_h[: n // 2] = 1.0
                    rng.shuffle(start_h)
                start_c = compute_c5(start_h)

                now = time.time()
                time_left = deadline - now
                per_restart = time_left / (restarts - r)
                sub_deadline = now + per_restart * 0.99

                h_sa, c_sa = simulated_anneal(start_h, start_c, rng, sub_deadline)
                if c_sa < best_c:
                    best_c = c_sa
                    best_h = h_sa.copy()
                    best_n = n

        # ----- smooth‑max refinement with SLSQP -----
        if best_n == n and best_h is not None and time.time() < deadline:
            adjust_sum_exact(best_h, rng)
            for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000):
                if time.time() >= deadline:
                    break
                try:
                    best_h = slsqp_optimize(best_h, beta, rng, max_iter=400)
                except Exception:
                    continue
                c_new = compute_c5(best_h)
                if c_new < best_c:
                    best_c = c_new

    # ----- fallback (unlikely) -----
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
