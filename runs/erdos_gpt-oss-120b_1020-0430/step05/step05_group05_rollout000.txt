def hill_climb_binary(h, rng, max_iters, deadline):
    n = h.size
    best_c = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute worst shift
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_idx = np.argmax(corr)
        k = abs(max_idx - (n - 1))
        # compute candidate swap
        p = n - k
        swapped = False
        if p > 0:
            # indices up to p-1
            h_prefix = best_h[:p]
            h_suffix = best_h[k:]
            # candidates
            idx_one = np.flatnonzero(h_prefix == 1)
            idx_zero = np.flatnonzero(h_prefix == 0)
            if idx_one.size and idx_zero.size:
                # contributions for each position i in prefix
                contrib = 1.0 - h_suffix  # shape p
                i = idx_one[np.argmax(contrib[idx_one])]
                j = idx_zero[np.argmin(contrib[idx_zero])]
                # swap
                h2 = best_h.copy()
                h2[i] = 0.0
                h2[j] = 1.0
                c2 = compute_c5(h2)
                if c2 < best_c - 1e-12:
                    best_h, best_c = h2, c2
                    swapped = True
                    no_improve = 0
        if not swapped:
            # random swap fallback
            one_idx = rng.choice(np.flatnonzero(best_h == 1))
            zero_idx = rng.choice(np.flatnonzero(best_h == 0))
            h2 = best_h.copy()
            h2[one_idx] = 0.0
            h2[zero_idx] = 1.0
            c2 = compute_c5(h2)
            if c2 < best_c - 1e-12:
                best_h, best_c = h2, c2
                no_improve = 0
            else:
                no_improve += 1
        # optional early restart if no_improve large
        if no_improve >= 2000:
            # random restart? Or break
            break
    return best_h, best_c
```

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2] (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def fractional_to_binary(arr):
    """Return binary vector with exactly half the entries equal to 1."""
    n = arr.size
    idx = np.argpartition(arr, -n // 2)[-n // 2 :]
    bin_arr = np.zeros_like(arr)
    bin_arr[idx] = 1.0
    return bin_arr

def hill_climb_binary(h0, rng, max_iters, deadline):
    """Hill‑climbing on binary vectors (fixed number of ones)."""
    h = h0.copy()
    n = h.size
    best_c = compute_c5(h)
    no_improve = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = abs(max_idx - (n - 1))
        p = n - k
        improved = False
        if p > 0:
            prefix = h[:p]
            suffix = h[k:]
            ones = np.flatnonzero(prefix == 1)
            zeros = np.flatnonzero(prefix == 0)
            if ones.size and zeros.size:
                contrib = 1.0 - suffix
                i = ones[np.argmax(contrib[ones])]
                j = zeros[np.argmin(contrib[zeros])]
                h2 = h.copy()
                h2[i] = 0.0
                h2[j] = 1.0
                c2 = compute_c5(h2)
                if c2 < best_c - 1e-12:
                    h, best_c = h2, c2
                    improved = True
                    no_improve = 0
        if not improved:
            # random swap fallback
            one_idx = rng.choice(np.flatnonzero(h == 1))
            zero_idx = rng.choice(np.flatnonzero(h == 0))
            h2 = h.copy()
            h2[one_idx] = 0.0
            h2[zero_idx] = 1.0
            c2 = compute_c5(h2)
            if c2 < best_c - 1e-12:
                h, best_c = h2, c2
                no_improve = 0
            else:
                no_improve += 1
        if no_improve >= 5000:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # initial size and vector
    parent = globals().get("initial_h_values")
    if parent is not None:
        h0 = np.array(parent, dtype=float)
        n0 = h0.size
    else:
        n0 = 360
        h0 = np.zeros(n0)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0

    # ensure binary and exact sum
    h0 = fractional_to_binary(h0)
    adjust_sum_exact(h0, rng)

    best_h = h0
    best_c = compute_c5(best_h)
    best_n = n0

    # local hill‑climbing on the base size
    h_opt, c_opt = hill_climb_binary(best_h, rng, max_iters=30000, deadline=deadline)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt

    # up‑sample once (to at most 720 points) and refine
    while best_n * 2 <= 720 and time.time() < deadline:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        # linear interpolation of the current best function
        h_up_frac = np.interp(xt, xs, best_h)
        # binary conversion preserving half‑ones
        h_up_bin = fractional_to_binary(h_up_frac)
        adjust_sum_exact(h_up_bin, rng)

        h_up_opt, c_up_opt = hill_climb_binary(
            h_up_bin, rng, max_iters=60000, deadline=deadline
        )
        if c_up_opt < best_c:
            best_h, best_c, best_n = h_up_opt, c_up_opt, n_up
        else:
            # no improvement – stop trying higher resolution
            break

    # random binary restarts for remaining time
    while time.time() < deadline:
        hb = np.zeros(best_n)
        hb[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
        hb_opt, c_hb_opt = hill_climb_binary(hb, rng, max_iters=20000, deadline=deadline)
        if c_hb_opt < best_c:
            best_h, best_c = hb_opt, c_hb_opt

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
