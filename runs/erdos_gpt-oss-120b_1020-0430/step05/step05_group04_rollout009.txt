def autocorr(h):
    n = h.shape[0]

    N = 2 * n

import time
import numpy as np
from scipy.optimize import minimize

# ----- helpers ---------------------------------------------------------

def compute_c5(h):
    """Exact C5 from definition."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng, max_iter=2000):
    """Redistribute a tiny amount so that Σh = n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def autocorr(h):
    """Linear autocorrelation a_k = Σ_i h_i h_{i+k} for k=0..n-1."""
    n = h.size
    h_rev = h[::-1]
    N = 2 * n
    H = np.fft.rfft(h, N)
    H_rev = np.fft.rfft(h_rev, N)
    conv = np.fft.irfft(H * H_rev, N)
    return conv[n - 1:]  # length n


# ----- sub‑gradient descent -------------------------------------------

def subgradient_descent(h0, rng, max_iter, step0, decay):
    """Projected sub‑gradient descent on C5 (via worst shift)."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    adjust_sum_exact(h, rng)
    best_h = h.copy()

    best_c = 1.0 - (2.0 / n) * a.min()

        a = autocorr(h)
        min_a = a.min()
        c = 1.0 - (2.0 / n) * min_a
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        # gradient of the worst shift(s)
        worst = np.where(a == min_a)[0]
        g = np.zeros_like(h)
        for k in worst:
            if k == 0:
                g -= 2.0 * h
            else:
                g[: n - k] -= h[k:]
                g[k:] -= h[: n - k]
        step = step0 * (decay ** it)
        if step < 1e-9:
            break
        h -= step * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            adjust_sum_exact(h, rng)
        # occasional jitter to escape plateaus
        if (it & 511) == 0 and it > 0:
            jitter = rng.normal(scale=step * 0.5, size=n)
            h += jitter
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                adjust_sum_exact(h, rng)
    return best_h, best_c


# ----- smooth‑max + SLSQP ---------------------------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stabilisation
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """SLSQP on soft‑max surrogate for a given temperature β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


# ----- main entry -----------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (even sizes ≤ 960)
    cand_ns = [360, 540, 720, 900]
    cand_ns = [n for n in cand_ns if n % 2 == 0]

    for n in cand_ns:
        if time.time() > deadline:
            break

        # ---- build a few diverse starts --------------------------------
        starts = []

        if "initial_h_values" in globals():
            parent = np.asarray(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            starts.append(interp)

        # binary random starts
        for _ in range(2):
            hb = np.zeros(n)
            hb[rng.choice(n, n // 2, replace=False)] = 1.0
            starts.append(hb)

        # fractional random starts
        for _ in range(2):
            rf = rng.random(n)
            rf *= (n / 2.0) / rf.sum()
            rf = np.clip(rf, 0.0, 1.0)
            adjust_sum_exact(rf, rng)
            starts.append(rf)

        for h0 in starts:
            if time.time() > deadline:
                break

            # ----- initial projection ------------------------------------
            h = h0.copy()
            adjust_sum_exact(h, rng)
            np.clip(h, 0.0, 1.0, out=h)

            # ----- sub‑gradient phase ------------------------------------
            # iteration budget adapts to remaining time
            time_left = deadline - time.time()
            est_iter = int(min(15000, max(3000, time_left * 6)))
            step0 = 0.25
            decay = 0.9995
            h_local, c_local = subgradient_descent(
                h, rng, max_iter=est_iter, step0=step0, decay=decay
            )

            # ----- short SLSQP polishing ---------------------------------
            # a few high‑temperature passes
            for beta in (2000, 5000, 10000):
                if time.time() > deadline:
                    break
                h_local = slsqp_optimize(h_local, beta, rng, max_iter=200)
            c_local = compute_c5(h_local)

            # ----- keep global best ---------------------------------------
            if c_local < best_c:
                best_c = c_local
                best_h = h_local.copy()
                best_n = n

    # ----- optional up‑sampling of the current best --------------------
    if best_h is not None and time.time() < deadline and best_n * 2 <= 960:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        # a light sub‑gradient burst on the finer grid
        h_up, _ = subgradient_descent(h_up, rng, max_iter=4000, step0=0.18, decay=0.9994)
        # final polishing
        for beta in (3000, 8000, 15000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=200)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
            best_n = n_up

    # ----- fallback (should never happen) -------------------------------
    if best_h is None:
        best_n = 360
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
