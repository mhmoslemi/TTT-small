import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def compute_O_and_max(h, half_n):
    # h is binary (0/1) uint8
    fft_h = np.fft.fft(h)
    C = np.real(np.fft.ifft(fft_h * np.conj(fft_h)))
    C = np.rint(C).astype(np.int16)
    O = half_n - C
    Omax = int(O.max())
    return O, Omax

def adjust_sum_exact(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_optimize_frac(h0, beta, target, rng, max_iter=2000):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = adjust_sum_exact(x, target, rng)
    return x

def greedy_reduce(h, idx_plus, idx_minus, half_n, rng):
    O, Omax = compute_O_and_max(h, half_n)
    n = h.size
    while True:
        idx_max = np.where(O == Omax)[0]
        shift = idx_max[0]  # pick any worst shift
        neigh = h[idx_plus[shift]] + h[idx_minus[shift]]
        ones_idx = np.flatnonzero(h == 1)
        zeros_idx = np.flatnonzero(h == 0)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        min_neigh = neigh[ones_idx].min()
        cand_i = ones_idx[neigh[ones_idx] == min_neigh]
        i = rng.choice(cand_i)
        max_neigh = neigh[zeros_idx].max()
        cand_j = zeros_idx[neigh[zeros_idx] == max_neigh]
        j = rng.choice(cand_j)
        diff = (h[idx_plus[j]] + h[idx_minus[j]]) - (h[idx_plus[i]] + h[idx_minus[i]])
        O_new = O + diff
        Omax_new = int(O_new.max())
        if Omax_new < Omax:
            h[i] = 0
            h[j] = 1
            O = O_new
            Omax = Omax_new
        else:
            break
    return h, O, Omax

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # Determine problem size
    if "initial_h_values" in globals():
        parent_vals = np.array(initial_h_values, dtype=float)
        n = parent_vals.size
    else:
        n = 360
    if n % 2 != 0:
        n += 1  # ensure even

    half_n = n // 2
    ks = np.arange(n, dtype=np.int32)

    # Pre‑compute circular index tables to avoid modulo in the hot loop
    idx_plus = ((np.arange(n).reshape(-1, 1) + ks) % n).astype(np.int16)
    idx_minus = ((np.arange(n).reshape(-1, 1) - ks) % n).astype(np.int16)

    best_h = None
    best_c = np.inf
    best_Omax = None

    # Helper to create binary start vectors
    starts = []
    if "initial_h_values" in globals():

        if vals.size == n:
            idx = np.argpartition(vals, -half_n)[-half_n:]
            hb = np.zeros(n, dtype=np.uint8)
            hb[idx] = 1
            starts.append(hb)
    for _ in range(4):
        hb = np.zeros(n, dtype=np.uint8)
        hb[rng.choice(n, half_n, replace=False)] = 1
        starts.append(hb)

    # Main optimization loop over start vectors
    for start_h in starts:
        if time.time() > deadline:
            break
        h = start_h.copy()
        O, Omax = compute_O_and_max(h, half_n)

        while time.time() < deadline:

            # pick random 1 and 0 positions

            while h[i] == 0:

            j = rng.integers(0, n)

            O_new = O + diff
            Omax_new = int(O_new.max())
            delta = Omax_new - Omax
            accept = False
            if delta <= 0:
                accept = True
            else:
                if rng.random() < np.exp(-delta / T):
                    accept = True
            if accept:
                h[i] = 0
                h[j] = 1
                O = O_new
                Omax = Omax_new
                if Omax < best_Omax:
                    best_Omax = Omax
                    best_c = Omax * (2.0 / n)
                    best_h = h.copy()
                if best_Omax <= int(0.3808 * n / 2):
                    break
            if best_Omax <= int(0.3808 * n / 2):
                break
        if best_Omax <= int(0.3808 * n / 2):
            break

    # Greedy deterministic reduction on the best binary solution
    if best_h is not None and time.time() < deadline:
        h_g, O_g, Omax_g = greedy_reduce(best_h.copy(), idx_plus, idx_minus, half_n, rng)
        c_g = Omax_g * (2.0 / n)
        if c_g < best_c:
            best_c, best_h, best_Omax = c_g, h_g, Omax_g

    # Continuous refinement starting from the (possibly) improved binary solution
    if best_h is not None and time.time() < deadline:
        h_frac = best_h.astype(float)
        target = half_n
        betas = (5., 20., 80., 200., 500., 1000., 2000., 5000., 10000., 20000., 50000.)
        for beta in betas:
            if time.time() > deadline:
                break
            max_iter = 2500 if beta >= 20000 else 1800
            h_frac = slsqp_optimize_frac(h_frac, beta, target, rng, max_iter=max_iter)
            c_val = compute_c5(h_frac)
            if c_val < best_c:
                best_c = c_val
                best_h = h_frac.copy()
                best_Omax = int(best_c * n / 2)
            # small projection onto [0,1] and exact sum
            h_frac = np.clip(h_frac, 0.0, 1.0)
            h_frac = adjust_sum_exact(h_frac, target, rng)

    # Fallback in case nothing succeeded
    if best_h is None:
        n = 360
        half_n = n // 2
        best_h = np.zeros(n, dtype=np.uint8)
        best_h[:half_n] = 1
        best_Omax = half_n  # Omax = half_n when all ones together
        best_c = best_Omax * (2.0 / n)

    # Final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = best_Omax * (2.0 / n)
    return best_h.tolist(), float(final_c), int(n)
