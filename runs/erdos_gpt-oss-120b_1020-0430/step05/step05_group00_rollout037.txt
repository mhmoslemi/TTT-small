def binary_sa_fast(n, rng, time_budget):
    start = time.time()
    target = n // 2
    best_h = None
    best_c5 = np.inf

    offset = n - 1
    # Precompute max possible shift indices 0..2n-2

    while time.time() - start < time_budget:
        # random start vector
        h = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, target, replace=False)
        h[ones] = 1
        # Compute initial correlation
        corr = np.correlate(h, h, mode='full')
        # compute c5
        max_corr = corr.max()
        c5 = (target - max_corr) * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # SA iterations within this start
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            # temperature schedule - log or linear; using exponential cooling
            T = 0.05 * (1e-4/0.05) ** (elapsed / time_budget)  # T0*T_end/T0 ^ t_frac
            # choose random 1 and 0 indices

            zeros_idx = np.where(h == 0)[0]

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)

        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def binary_sa_fast(n, rng, time_budget):
    start = time.time()
    target = n // 2
    offset = n - 1

    # random start
    h = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, target, replace=False)
    h[ones] = 1
    corr = np.correlate(h, h, mode='full')
    max_corr = corr.max()
    c5 = (target - max_corr) * (2.0 / n)
    best_h = h.copy()
    best_c5 = c5

    while time.time() - start < time_budget:
        # pick random i with h[i]==1
        i = int(rng.integers(0, n))
        while h[i] == 0:
            i = int(rng.integers(0, n))
        # pick random j with h[j]==0
        j = int(rng.integers(0, n))
        while h[j] == 1:
            j = int(rng.integers(0, n))
        # other ones indices
        other = np.where(h == 1)[0]
        other = other[other != i]

        s_i_to_other = other - i + offset
        s_other_to_i = i - other + offset
        s_j_to_other = other - j + offset
        s_other_to_j = j - other + offset

        # apply delta
        np.add.at(corr, s_i_to_other, -1)
        np.add.at(corr, s_other_to_i, -1)
        np.add.at(corr, s_j_to_other, +1)
        np.add.at(corr, s_other_to_j, +1)

        # swap
        h[i] = 0
        h[j] = 1

        max_corr = corr.max()
        c5_new = (target - max_corr) * (2.0 / n)

        elapsed = time.time() - start
        t_frac = elapsed / time_budget
        T0 = 0.05
        T_end = 1e-4
        T = T0 * (T_end / T0) ** t_frac
        delta = c5_new - c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # revert
            np.add.at(corr, s_i_to_other, +1)
            np.add.at(corr, s_other_to_i, +1)
            np.add.at(corr, s_j_to_other, -1)
            np.add.at(corr, s_other_to_j, -1)

            h[j] = 0

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for the soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

                   max_iter: int = 500) -> np.ndarray:

    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def binary_sa_fast(n: int, rng: np.random.Generator, time_budget: float):
    """Fast simulated‑annealing on binary vectors with exactly n/2 ones."""
    start = time.time()
    target = n // 2
    offset = n - 1

    # --- initialise random state ---------------------------------------
    h = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, target, replace=False)
    h[ones] = 1

    # bookkeeping structures for O(1) sampling
    zeros = np.setdiff1d(np.arange(n, dtype=int), ones)
    ones_list = list(ones)
    zeros_list = list(zeros)
    pos_one = np.full(n, -1, dtype=int)
    pos_zero = np.full(n, -1, dtype=int)
    for idx, v in enumerate(ones_list):
        pos_one[v] = idx
    for idx, v in enumerate(zeros_list):
        pos_zero[v] = idx

    # correlation of h with itself (linear, non‑circular)
    corr = np.correlate(h, h, mode="full")
    max_corr = corr.max()
    c5 = (target - max_corr) * (2.0 / n)
    best_h = h.copy()
    best_c5 = c5

    while time.time() - start < time_budget:
        # pick random 1 and 0
        i = rng.choice(ones_list)
        j = rng.choice(zeros_list)

        # other ones (excluding i)
        arr = np.asarray(ones_list, dtype=int)
        mask = np.arange(arr.size) != pos_one[i]
        other = arr[mask]

        # shift indices
        s_i_o = other - i + offset
        s_o_i = i - other + offset
        s_j_o = other - j + offset
        s_o_j = j - other + offset

        # apply delta to correlation
        np.add.at(corr, s_i_o, -1)
        np.add.at(corr, s_o_i, -1)
        np.add.at(corr, s_j_o, +1)
        np.add.at(corr, s_o_j, +1)

        # perform swap in h
        h[i] = 0
        h[j] = 1

        # update sampling lists in O(1)
        idx_i = pos_one[i]
        idx_j = pos_zero[j]
        ones_list[idx_i] = j
        zeros_list[idx_j] = i
        pos_one[j] = idx_i
        pos_zero[i] = idx_j
        pos_one[i] = -1
        pos_zero[j] = -1

        # new objective
        max_corr = corr.max()
        c5_new = (target - max_corr) * (2.0 / n)

        # temperature schedule (exponential cooling)
        elapsed = time.time() - start
        t_frac = elapsed / time_budget
        T0, T_end = 0.05, 1e-4
        T = T0 * (T_end / T0) ** t_frac

        delta = c5_new - c5
        accept = delta <= 0 or (T > 0 and rng.random() < math.exp(-delta / T))
        if accept:
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # revert correlation
            np.add.at(corr, s_i_o, +1)
            np.add.at(corr, s_o_i, +1)
            np.add.at(corr, s_j_o, -1)
            np.add.at(corr, s_o_j, -1)
            # revert swap
            h[i] = 1
            h[j] = 0
            # revert sampling lists
            ones_list[idx_i] = i
            zeros_list[idx_j] = j
            pos_one[i] = idx_i
            pos_zero[j] = idx_j
            pos_one[j] = -1
            pos_zero[i] = -1

        # occasional random restart to diversify search
        if rng.random() < 1e-5:
            h[:] = 0
            ones = rng.choice(n, target, replace=False)
            h[ones] = 1
            zeros = np.setdiff1d(np.arange(n, dtype=int), ones)
            ones_list = list(ones)
            zeros_list = list(zeros)
            pos_one.fill(-1)
            pos_zero.fill(-1)
            for idx, v in enumerate(ones_list):
                pos_one[v] = idx
            for idx, v in enumerate(zeros_list):
                pos_zero[v] = idx
            corr = np.correlate(h, h, mode="full")
            max_corr = corr.max()
            c5 = (target - max_corr) * (2.0 / n)

    return best_h, best_c5


def refine_solution(h: np.ndarray, rng: np.random.Generator,
                    start: float, total_budget: float,
                    allocated: float):
    """Hill‑climb + SLSQP + gradient refinement within `allocated` seconds."""
    deadline = start + allocated
    # guided hill‑climb
    max_iters = int(min(5000, (deadline - time.time()) * 600))
    if max_iters > 0:
        h, _ = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.7)
    # SLSQP passes with increasing β
    for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
        if time.time() >= deadline * 0.95:
            break
        iters = 800 if beta >= 200 else 500
        h = slsqp_optimize(h, beta, rng, max_iter=iters)
    # final gradient tidy‑up
    if time.time() < deadline * 0.97:
        h = gradient_refine(h, rng, step=0.01, max_iter=10)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # ---- use parent construction if present ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ---- binary simulated annealing (≈45% of time) ----
    time_binary = total_budget * 0.45
    n_sa = 256
    h_bin, c_bin = binary_sa_fast(n_sa, rng, time_binary)
    if c_bin < best_c:
        best_h, best_c, best_n = h_bin.astype(float), c_bin, n_sa

    # ---- remaining time for refinement ----
    remaining = total_budget - (time.time() - start)
    if remaining <= 0:
        best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), rng)
        return best_h.tolist(), float(compute_c5(best_h)), int(best_h.size)

    # split remaining: 70% for 512‑point refinement, 30% for optional 1024‑point
    time_refine_512 = remaining * 0.70
    time_refine_1024 = remaining - time_refine_512

    # ------- 512‑point phase -------
    if best_n < 512:
        # up‑sample via linear interpolation
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, 512, endpoint=False)
        h_512 = np.interp(x_new, x_old, best_h)
    else:
        h_512 = best_h.copy()
    h_512 = np.clip(h_512, 0.0, 1.0)
    h_512 = enforce_sum(h_512, rng)

    h_512 = refine_solution(h_512, rng, start, total_budget, time_refine_512)
    c_512 = compute_c5(h_512)
    if c_512 < best_c:
        best_h, best_c, best_n = h_512, c_512, 512

    # ------- optional 1024‑point phase -------
    if time_refine_1024 > total_budget * 0.08:
        # up‑sample to 1024 points
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, 1024, endpoint=False)
        h_1024 = np.interp(x_new, x_old, best_h)
        h_1024 = np.clip(h_1024, 0.0, 1.0)
        h_1024 = enforce_sum(h_1024, rng)

        h_1024 = refine_solution(h_1024, rng, start, total_budget, time_refine_1024)
        c_1024 = compute_c5(h_1024)
        if c_1024 < best_c:
            best_h, best_c, best_n = h_1024, c_1024, 1024

    # ---- final stochastic tweaking until budget closes ----
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ---- final sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
