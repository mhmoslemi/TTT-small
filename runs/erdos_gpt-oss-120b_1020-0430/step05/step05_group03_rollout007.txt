def simulated_annealing(h_initial, rng, time_limit, start_time):
    h_cur = h_initial.copy()
    cur_c = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c = cur_c
    # parameters
    T0 = 0.02
    T_end = 1e-5
    # we can compute expected total iter as maybe 1e6 if 1 micro per iter? Actually iteration cost is compute_c5 O(n). For n=512, np.correlate may be ~ 0.001 ms? Hard to estimate. We'll just loop until time runs out.
    last_improve_time = start_time
    while True:
        t = time.time()
        elapsed = t - start_time
        if elapsed >= time_limit:
            break
        # linear cooling
        T = T0 * (T_end / T0) ** (elapsed / time_limit)
        # propose move
        if rng.random() < 0.7:
            # fractional transfer
            i = rng.integers(0, h_cur.size)
            j = rng.integers(0, h_cur.size)
            while i == j:
                j = rng.integers(0, h_cur.size)
            # compute max eps we can transfer
            max_eps = min(h_cur[i], 1.0 - h_cur[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.5
            h_new = h_cur.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            # full swap of values
            i = rng.integers(0, h_cur.size)
            j = rng.integers(0, h_cur.size)
            while i == j:
                j = rng.integers(0, h_cur.size)
            # swap values
            h_new = h_cur.copy()
            tmp = h_new[i]
            h_new[i] = h_new[j]
            h_new[j] = tmp
        new_c = compute_c5(h_new)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            h_cur = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
                last_improve_time = t
        # restart if stuck
        if t - last_improve_time > 0.2 * time_limit:
            # generate new random binary vector
            h_cur = random_feasible_vector_binary(h_cur.size, rng)
            cur_c = compute_c5(h_cur)
            last_improve_time = t
    return best_h, best_c

import time
import numpy as np

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with n points."""
    n = h.size
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    # simple redistribution loop
    while abs(diff) > 1e-12 and it < 2000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h


def random_feasible_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


# ---------- gradient based helpers ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- simulated annealing ----------
def simulated_annealing(h_start: np.ndarray, rng: np.random.Generator,
                        time_limit: float, start_time: float):
    """SA that respects the sum constraint (mass is transferred)."""
    h_cur = h_start.copy()
    cur_c = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c = cur_c

    T0 = 0.02
    T_end = 1e-5
    last_improve = start_time

    n = h_cur.size
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= time_limit:
            break
        # linear‑in‑log cooling
        T = T0 * (T_end / T0) ** (elapsed / time_limit)

        # propose a move
        if rng.random() < 0.7:
            # fractional transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            max_eps = min(h_cur[i], 1.0 - h_cur[j])
            if max_eps == 0.0:
                continue
            eps = max_eps * rng.random() * 0.5
            h_new = h_cur.copy()
            h_new[i] -= eps
            h_new[j] += eps
        else:
            # full swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            h_new = h_cur.copy()
            h_new[i], h_new[j] = h_cur[j], h_cur[i]

        new_c = compute_c5(h_new)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            h_cur, cur_c = h_new, new_c
            if new_c < best_c:
                best_c, best_h = new_c, h_new.copy()
                last_improve = now

        # restart if stuck too long
        if now - last_improve > 0.2 * time_limit:
            h_cur = random_feasible_binary(n, rng)
            cur_c = compute_c5(h_cur)
            last_improve = now

    return best_h, best_c


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n0 = 256
        h0 = random_feasible_binary(n0, rng)

    best_h = h0
    best_c = compute_c5(best_h)
    n = best_h.size

    # ---- simulated annealing (major work) ----
    sa_time = budget_s * 0.80
    h_sa, c_sa = simulated_annealing(best_h, rng, sa_time, start)
    if c_sa < best_c:
        best_h, best_c = h_sa, c_sa
        n = best_h.size

    # ---- gradient‑based tidy‑up with remaining budget ----
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        # rough hill‑climb
        hill_iters = int(min(3000, remaining * 200))
        best_h, best_c = guided_hill_climb(best_h, rng,
                                           max_iters=hill_iters,
                                           step_scale=0.5)
        remaining = budget_s - (time.time() - start)

    if remaining > 2.0:
        # fine gradient refinement
        best_h = gradient_refine(best_h, rng,
                                 step=0.01, max_iter=15)
        best_c = compute_c5(best_h)

    # ---- final stochastic nudges until time expires ----
    while time.time() - start < budget_s * 0.998:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)

        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ---- final sanity check ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
