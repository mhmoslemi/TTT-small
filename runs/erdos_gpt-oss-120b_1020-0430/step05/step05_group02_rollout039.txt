def binary_sa_optimize(n, rng, deadline):
    target = n//2
    # start with random binary
    h = np.zeros(n, dtype=np.int8)
    h[rng.choice(n, target, replace=False)] = 1
    # compute c
    c = np.correlate(h, h, mode='full')  # integer
    best_h = h.copy()
    best_mismatch = target - c.min()
    # track ones, zeros as Python lists for fast random choice
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    # set temperature based on best mismatch
    T = max(best_mismatch * 0.5, 1.0)
    decay = 0.99995
    iteration = 0
    # main loop
    while time.time() < deadline:
        iteration += 1
        # optionally adjust temp gradually
        if iteration % 1000 == 0:
            T = max(T * decay, 1e-6)  # ensure not zero
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute hi_vec, start_i
        start_i = n - 1 - i
        hi_vec = np.concatenate((h[i::-1], h[i+1:]))
        c[start_i:start_i+n] -= hi_vec  # remove i contribution
        start_j = n - 1 - j
        hj_vec = np.concatenate((h[j::-1], h[j+1:]))
        c[start_j:start_j+n] += hj_vec  # add j contribution
        # swap bits
        h[i], h[j] = 0, 1
        # update lists
        # remove i from ones, add i to zeros
        ones.remove(i)
        zeros.append(i)
        # remove j from zeros, add j to ones
        zeros.remove(j)
        ones.append(j)
        # compute new mismatch
        cur_mismatch = target - c.min()
        # acceptance
        if cur_mismatch < best_mismatch:
            best_mismatch = cur_mismatch
            best_h = h.copy()
        if cur_mismatch <= best_mismatch:  # accept all improving or equal (key)
            # keep, continue
            pass
        else:
            # Metropolis
            if rng.random() < np.exp(-(cur_mismatch - best_mismatch) / max(T, 1e-12)):
                # accept non-improving
                pass
            else:
                # revert changes
                # revert c: add hi_vec back, subtract hj_vec
                c[start_i:start_i+n] += hi_vec
                c[start_j:start_j+n] -= hj_vec
                # swap bits back
                h[i], h[j] = 1, 0
                # revert lists
                ones.remove(j)
                zeros.append(j)
                zeros.remove(i)
                ones.append(i)
                # no update to best (since not improving)
        # optional early break if best_mismatch == desired bound, e.g., 68, can break
        if best_mismatch <= 68:
            # we may still continue to see if can go lower
            pass
    return best_h, best_mismatch

def binary_sa_optimize_one_run(h, rng, deadline):
    n = h.size
    target = n//2
    c = np.correlate(h, h, mode='full')
    best_h = h.copy()
    best_mis = target - c.min()
    # maintain ones, zeros as lists
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    T = max(best_mis * 0.5, 1.0)
    decay = 0.99995
    it = 0
    last_improve = 0
    while time.time() < deadline:
        it += 1
        if it % 2000 == 0:
            # decay temperature periodically
            T = max(T * decay, 1e-6)
        # choose random swap
        idx_i = rng.integers(len(ones))
        i = ones[idx_i]
        idx_j = rng.integers(len(zeros))
        j = zeros[idx_j]
        # update c as above
        start_i = n - 1 - i
        hi_vec = np.concatenate((h[i::-1], h[i+1:]))
        c[start_i:start_i+n] -= hi_vec
        start_j = n - 1 - j
        hj_vec = np.concatenate((h[j::-1], h[j+1:]))
        c[start_j:start_j+n] += hj_vec
        # swap bits
        h[i] = 0
        h[j] = 1
        cur_mis = target - c.min()
        # decide
        accept = False
        if cur_mis < best_mis:
            accept = True
        else:
            if rng.random() < np.exp(-(cur_mis - best_mis) / max(T, 1e-12)):
                accept = True
        if accept:
            # update best if better
            if cur_mis < best_mis:
                best_mis = cur_mis
                best_h = h.copy()
                last_improve = 0
            else:
                last_improve += 1
            # update ones, zeros lists
            # replace i with j in ones list
            ones[idx_i] = j
            # replace j with i in zeros list
            zeros[idx_j] = i
        else:
            # revert c
            c[start_i:start_i+n] += hi_vec
            c[start_j:start_j+n] -= hj_vec
            # revert bits
            h[i] = 1
            h[j] = 0
            last_improve += 1
        if last_improve > 200000:
            # stagnation; break
            break
        # optional early break if best_mis <= 68
        if best_mis <= 68:
            break
    return best_h, best_mis
```

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # Exact sum correction
    diff = CURRENT_TARGET - x.sum()
    # adjust by random add/subtract
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa_optimize_one_run(h0, rng, deadline):
    n = h0.size
    target = n // 2
    h = h0.copy()
    c = np.correlate(h, h, mode='full')
    best_h = h.copy()
    best_mis = target - int(c.min())
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    # initial temperature
    T = max(best_mis * 0.5, 1.0)
    decay = 0.99995
    it = 0
    last_improve = 0
    while time.time() < deadline:
        it += 1
        # periodic temperature decay
        if it % 2000 == 0:
            T = max(T * decay, 1e-6)
        # choose random swap indices
        idx_i = rng.integers(len(ones))
        i = ones[idx_i]
        idx_j = rng.integers(len(zeros))
        j = zeros[idx_j]
        # update correlation
        start_i = n - 1 - i
        hi_vec = np.concatenate((h[i::-1], h[i+1:]))
        c[start_i:start_i+n] -= hi_vec
        start_j = n - 1 - j
        hj_vec = np.concatenate((h[j::-1], h[j+1:]))
        c[start_j:start_j+n] += hj_vec
        # swap bits
        h[i] = 0
        h[j] = 1
        cur_mis = target - int(c.min())
        # acceptance
        accept = False
        if cur_mis < best_mis:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(cur_mis - best_mis) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            if cur_mis < best_mis:
                best_mis = cur_mis
                best_h = h.copy()
                last_improve = 0
            else:
                last_improve += 1
            # update ones/zeros lists efficiently:
            # replace i with j in ones
            ones[idx_i] = j
            # replace j with i in zeros
            zeros[idx_j] = i
        else:
            # revert changes
            c[start_i:start_i+n] += hi_vec
            c[start_j:start_j+n] -= hj_vec
            h[i] = 1
            h[j] = 0
            last_improve += 1
        if last_improve > 200000:
            break
        # optional early break if best_mis <= 68
        if best_mis <= 68:
            break
    return best_h, best_mis
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx, discretized."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # smooth max via log‑sum‑exp
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- Binary simulated annealing ----------
def binary_sa_optimize_one_run(h0, rng, deadline):
    """Single SA run on a binary vector."""
    n = h0.size
    target = n // 2
    h = h0.copy()
    c = np.correlate(h, h, mode="full")          # int32 autocorrelation
    best_h = h.copy()
    best_mis = target - int(c.min())
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))

    T = max(best_mis * 0.5, 1.0)
    decay = 0.99995
    it = 0
    stagn = 0

    while time.time() < deadline:
        it += 1
        T = max(T * decay, 1e-9)

        # random 1↔0 swap
        idx_i = rng.integers(len(ones))
        i = ones[idx_i]
        idx_j = rng.integers(len(zeros))
        j = zeros[idx_j]

        # update correlation for removal of i
        start_i = n - 1 - i
        hi_vec = np.concatenate((h[i::-1], h[i + 1 :]))
        c[start_i : start_i + n] -= hi_vec

        # update correlation for addition of j
        start_j = n - 1 - j
        hj_vec = np.concatenate((h[j::-1], h[j + 1 :]))
        c[start_j : start_j + n] += hj_vec

        # perform swap
        h[i] = 0
        h[j] = 1

        cur_mis = target - int(c.min())
        accept = False
        if cur_mis < best_mis:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(cur_mis - best_mis) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            if cur_mis < best_mis:
                best_mis = cur_mis
                best_h = h.copy()
                stagn = 0
            else:
                stagn += 1
            # update bookkeeping lists in O(1)
            ones[idx_i] = j
            zeros[idx_j] = i
        else:
            # revert everything
            c[start_i : start_i + n] += hi_vec
            c[start_j : start_j + n] -= hj_vec
            h[i] = 1
            h[j] = 0
            stagn += 1

        if stagn > 200000:
            break
        if best_mis <= 68:               # good enough, stop early
            break

    return best_h, best_mis

# ---------- Main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    # Determine problem size
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
        # ensure even length for half‑ones
        if n % 2 == 1:
            n -= 1
        parent = parent[:n]
    else:
        parent = None
        n = 360                     # default size
    target = n // 2

    best_h = None
    best_c = float("inf")
    best_mis = target

    # Evaluate binary version of the parent (if any)
    if parent is not None:
        idx = np.argpartition(parent, -target)[-target:]
        bin_parent = np.zeros(n, dtype=np.int32)
        bin_parent[idx] = 1
        c_parent = compute_c5(bin_parent.astype(float))
        if c_parent < best_c:
            best_c = c_parent
            best_h = bin_parent.copy()
            best_mis = target - int(np.correlate(bin_parent, bin_parent, mode="full").min())

    # Allocate time for binary SA
    time_left = deadline - time.time()
    if time_left > 30:
        sa_deadline = deadline - 20          # reserve ~20 s for final polishing
    else:
        sa_deadline = deadline

    # Main SA loop with restarts
    while time.time() < sa_deadline:
        # random binary start
        h0 = np.zeros(n, dtype=np.int32)
        h0[rng.choice(n, target, replace=False)] = 1
        h_res, mis = binary_sa_optimize_one_run(h0, rng, sa_deadline)
        if mis < best_mis:
            best_mis = mis
            best_h = h_res.copy()
            best_c = compute_c5(best_h.astype(float))
        if best_mis <= 68:
            break

    # Optional continuous refinement (few SLSQP steps)
    if best_h is not None and time.time() < deadline - 5:
        h_float = best_h.astype(float)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
            if time.time() > deadline - 5:
                break
            try:
                h_opt = slsqp_optimize(h_float, beta, rng, max_iter=600)
            except Exception:
                continue
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                h_float = h_opt
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # Fallback in the unlikely event nothing was found
    if best_h is None:
        n = 180
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_h.size)
