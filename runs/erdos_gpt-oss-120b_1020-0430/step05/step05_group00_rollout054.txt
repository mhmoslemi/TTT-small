def softmax_grad(x, beta):
    N = len(x)
    corr = np.correlate(x, 1-x, mode='full')
    max_c = corr.max()
    expbeta = np.exp(beta*(corr - max_c))
    p = expbeta / expbeta.sum()
    # compute w
    center = N-1
    w = np.empty(N)
    w[0] = p[center]
    for k in range(1,N):
        w[k] = p[center + k] + p[center - k]
    # compute S = sum_{k} w[k] * x[i + k] (with i + k < N)
    S = np.zeros(N)
    for k in range(N):
        if w[k]==0: continue
        S[:N - k] += w[k] * x[k:]
    g = 1.0 - S
    return g

def project_onto_capped_simplex(x, target):
    lo = -1.0
    hi = max(x)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2, 0, 1)

import time
import numpy as np

# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2] with 2·dx scaling."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {0<=x_i<=1, sum x_i = target}.
    Uses binary search on the Lagrange multiplier λ for the water‑filling.
    """
    lo = -1.0
    hi = np.max(v)
    for _ in range(30):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def worst_gradient(x: np.ndarray) -> np.ndarray:
    """
    Gradient of the overlap for the worst (maximal) shift.
    """
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argmax(corr) - (n - 1)          # shift index (may be negative)
    k = abs(idx)
    grad = np.ones(n)
    if k == 0:
        grad = 1.0 - 2.0 * x
    else:
        grad[: n - k] -= x[k:]
        grad[k:]     -= x[: n - k]
    return grad


def softmax_gradient(x: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of the maximal overlap.
    """
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    expb = np.exp(beta * (corr - m))
    p = expb / expb.sum()                    # soft‑max probabilities

    center = n - 1
    w = np.empty(n)
    w[0] = p[center]
    for k in range(1, n):
        w[k] = p[center + k] + p[center - k]

    # S_i = Σ_{k} w_k * x_{i+k} (zero outside)
    S = np.zeros(n)
    for k in range(n):
        wk = w[k]
        if wk != 0.0:
            S[: n - k] += wk * x[k:]
    return 1.0 - S


def subgradient_descent(pool, target, time_limit, start_time):
    """
    Sub‑gradient descent on the worst‑case overlap for a pool of candidates.
    """
    step0 = 0.6
    step_counter = 0
    n = pool[0].size
    best_h = None
    best_c = np.inf

    while time.time() - start_time < time_limit:
        for i in range(len(pool)):
            if time.time() - start_time >= time_limit:
                break
            h = pool[i]
            # current C5
            c_curr = compute_c5(h)
            # gradient of the worst shift
            grad = worst_gradient(h)
            # step size decays with sqrt(iteration)
            step = step0 / np.sqrt(step_counter + 1.0)
            step_counter += 1
            h_new = h - step * grad
            h_new = project_capped_simplex(h_new, target)
            c_new = compute_c5(h_new)

            # accept if not worse (monotone descent)
            if c_new <= c_curr:
                h = h_new
                c_curr = c_new

            # occasional random perturbation to escape plateaus
            if (step_counter % 2000) == 0 and np.random.rand() < 0.1:
                eps = np.random.rand(n) * 0.2 - 0.1
                h_tmp = np.clip(h + eps, 0.0, 1.0)
                h_tmp = project_capped_simplex(h_tmp, target)
                if compute_c5(h_tmp) < c_curr:
                    h = h_tmp
                    c_curr = compute_c5(h)

            pool[i] = h
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
        # end for
    return best_h, best_c, pool


def softmax_descent(x, target, betas, steps_per_beta, time_limit, start_time):
    """Refinement using the smooth soft‑max objective."""
    n = x.size
    for beta in betas:
        if time.time() - start_time >= time_limit:
            break
        for _ in range(steps_per_beta):
            if time.time() - start_time >= time_limit:
                break
            grad = softmax_gradient(x, beta)
            step = 0.5 / (beta + 1.0)        # modest step size
            x_new = x - step * grad
            x_new = project_capped_simplex(x_new, target)
            x = x_new
    return x
