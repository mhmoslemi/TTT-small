def delta_corr_array(h, corr, i_src, j_dst):
    n = h.shape[0]
    ks = np.arange(-(n-1), n)
    delta = np.zeros_like(corr, dtype=float)
    # i_src
    left_idx = i_src + ks
    mask = (left_idx >= 0) & (left_idx < n)
    delta[mask] -= (1 - h[left_idx[mask]])
    right_idx = i_src - ks
    mask = (right_idx >= 0) & (right_idx < n)
    delta[mask] += h[right_idx[mask]]
    # j_dst
    left_idx = j_dst + ks
    mask = (left_idx >= 0) & (left_idx < n)
    delta[mask] += (1 - h[left_idx[mask]])
    right_idx = j_dst - ks
    mask = (right_idx >= 0) & (right_idx < n)
    delta[mask] -= h[right_idx[mask]]
    return delta
```

import time, math, numpy as np
from scipy.optimize import minimize, LinearConstraint

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


# ---------- gradient based helpers ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap at the worst shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


# ---------- binary hill‑climb utilities ----------
def delta_corr_array(h: np.ndarray, corr: np.ndarray, i_src: int, j_dst: int) -> np.ndarray:
    """Delta of full correlation after swapping i_src (1) and j_dst (0)."""
    n = h.size
    ks = np.arange(-(n - 1), n, dtype=int)           # length 2n‑1
    delta = np.zeros_like(corr, dtype=float)

    # i_src contributions
    left_idx = i_src + ks
    mask = (left_idx >= 0) & (left_idx < n)
    delta[mask] -= (1 - h[left_idx[mask]])

    right_idx = i_src - ks
    mask = (right_idx >= 0) & (right_idx < n)
    delta[mask] += h[right_idx[mask]]

    # j_dst contributions
    left_idx = j_dst + ks
    mask = (left_idx >= 0) & (left_idx < n)
    delta[mask] += (1 - h[left_idx[mask]])

    right_idx = j_dst - ks
    mask = (right_idx >= 0) & (right_idx < n)
    delta[mask] -= h[right_idx[mask]]

    return delta


def binary_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 5000, swap_trials: int = 30,
                      time_limit: float = None, start_time: float = None):
    """Swap‑based local search on binary vectors."""
    n = h.shape[0]
    h = np.where(h >= 0.5, 1.0, 0.0)          # ensure binary
    corr = np.correlate(h, 1 - h, mode="full")
    best_max = corr.max()
    best_h = h.copy()
    best_corr = corr.copy()
    it = 0
    while it < max_iters:
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1 - best_h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_new_max = best_max
        best_delta = None
        best_swap = None
        for _ in range(swap_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            delta = delta_corr_array(best_h, best_corr, i, j)
            new_max = np.max(best_corr + delta)
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_delta = delta
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            best_h[i], best_h[j] = best_h[j], best_h[i]
            best_corr = best_corr + best_delta
            best_max = best_new_max
        else:
            break
        it += 1
    return best_h, best_max


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise with parent if present
    if "initial_h_values" in globals():
        best_h = np.array(initial_h_values, dtype=float)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = best_h.size
    else:
        best_n = 128
        best_h = random_feasible_vector(best_n, rng)
        best_c = compute_c5(best_h)

    n_candidates = [128, 192, 256, 384, 512]

    # ----- binary hill‑climb phase -----
    while time.time() - start < 0.75 * budget_s:
        n = int(rng.choice(n_candidates))
        # random binary start
        h_bin = np.zeros(n, dtype=float)
        h_bin[rng.choice(n, n // 2, replace=False)] = 1.0
        # allocate a small time slice for this hill‑climb
        slice_time = min(0.20 * (budget_s - (time.time() - start)), 25.0)
        bh, max_corr = binary_hill_climb(h_bin, rng,
                                         max_iters=8000,
                                         swap_trials=40,
                                         time_limit=slice_time,
                                         start_time=time.time())
        c5_val = max_corr * (2.0 / n)
        if c5_val < best_c - 1e-7:
            best_h, best_c, best_n = bh.copy(), c5_val, n

        # short guided refinement (continuous) on the best so far
        if time.time() - start < 0.85 * budget_s:
            refined_h, refined_c = guided_hill_climb(best_h, rng,
                                                    max_iters=800,
                                                    step_scale=0.7)
            refined_c = compute_c5(refined_h)
            if refined_c < best_c:
                best_h, best_c, best_n = refined_h.copy(), refined_c, refined_h.size

    # ----- high‑resolution refinement (focus on n=512) -----
    if time.time() - start < 0.90 * budget_s:
        target_n = 512
        if best_n != target_n:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
            best_h = np.interp(x_new, x_old, best_h)
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = enforce_sum(best_h, rng)
            best_n = target_n
        # aggressive SLSQP schedule
        for beta in (20., 80., 200., 800., 3200., 12800., 51200., 200000.):
            if time.time() - start > budget_s * 0.95:
                break
            maxit = 1200 if beta >= 2000 else 800
            best_h = slsqp_optimize(best_h, beta, max_iter=maxit)
            c = compute_c5(best_h)
            if c < best_c:
                best_c = c
        # final gradient tidy‑up
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=15)
        best_c = compute_c5(best_h)

    # ----- stochastic tail until time expires -----
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
