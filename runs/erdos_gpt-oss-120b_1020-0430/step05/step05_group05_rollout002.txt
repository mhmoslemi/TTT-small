def sa_half(a, n, deadline, rng):
    best_a = a.copy()
    best_corr = evaluate_h(construct_h_from_half(a))
    cur_a = a.copy()
    cur_corr = best_corr
    m = n//2
    # parameters
    T0 = 0.5  # initial temperature maybe 0.5?
    # but delta_corr can be up to ~1 (max correlation differences)? Let's set T0=0.2 maybe.
    # step size
    step0 = 0.2
    while time.time() < deadline:
        t = (time.time() - start) / (deadline - start)
        T = max(1e-6, T0 * (1 - t))
        step = step0 * (1 - t) + 0.001
        i = rng.integers(0, m)
        j = rng.integers(0, m-1)
        if j >= i:
            j += 1
        max_delta = min(step, 1 - cur_a[i], cur_a[j])  # we can move from j to i
        # Actually we need delta that increases a[i] and decreases a[j] (keep sum same)
        # delta allowed: [-cur_a[i], 1-cur_a[i]] for a[i]; and for a[j], -cur_a[j] .. 1-cur_a[j]
        # Since we want delta for a[i], and a[j] moves opposite, we need delta <= cur_a[j] and delta <= 1 - cur_a[i]
        # Bc a[j] - delta must be >=0, a[i]+delta <=1.
        max_inc = min(step, 1 - cur_a[i], cur_a[j])
        max_dec = min(step, cur_a[i], 1 - cur_a[j])  # negative delta allowed
        # randomly pick sign
        if max_inc + max_dec == 0:
            continue
        # pick delta uniformly in [-max_dec, max_inc]
        delta = rng.random() * (max_inc + max_dec) - max_dec
        # propose
        a2 = cur_a.copy()
        a2[i] += delta
        a2[j] -= delta
        # compute new corr
        h2 = np.concatenate([a2, 1 - a2])
        corr2 = np.max(np.correlate(h2, 1 - h2, mode="full"))
        delta_corr = corr2 - cur_corr
        if delta_corr <= 0 or rng.random() < np.exp(-delta_corr / T):
            cur_a = a2
            cur_corr = corr2
            if cur_corr < best_corr:
                best_corr = cur_corr
                best_a = cur_a.copy()
    return best_a, best_corr

import time
import numpy as np

TARGET_C5 = 0.38085


def _full_h(a):
    """Construct full h from half-vector a (h[i]+h[i+m]=1)."""
    return np.concatenate([a, 1.0 - a])


def _max_corr(h):
    """Maximum raw overlap sum (no dx factor)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))


def _simulated_annealing_half(a0, n, deadline, rng):
    """SA on the half‑vector a (size n//2) respecting a∈[0,1]."""
    m = n // 2
    a = a0.copy()
    h = _full_h(a)
    cur_corr = _max_corr(h)
    best_a = a.copy()
    best_corr = cur_corr

    start = time.time()
    T0 = 0.5            # initial temperature
    step0 = 0.2         # initial step size

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        t = elapsed / total if total > 0 else 1.0
        T = max(1e-7, T0 * (1.0 - t))
        step = max(1e-4, step0 * (1.0 - t))

        i = rng.integers(m)
        j = rng.integers(m - 1)
        if j >= i:
            j += 1

        max_inc = min(step, 1.0 - a[i], a[j])
        max_dec = min(step, a[i], 1.0 - a[j])
        if max_inc + max_dec <= 0.0:
            continue

        delta = rng.random() * (max_inc + max_dec) - max_dec
        a[i] += delta
        a[j] -= delta

        h = _full_h(a)
        new_corr = _max_corr(h)
        d_corr = new_corr - cur_corr

        if d_corr <= 0.0 or rng.random() < np.exp(-d_corr / T):
            cur_corr = new_corr
            if cur_corr < best_corr:
                best_corr = cur_corr
                best_a = a.copy()
                if best_corr * 2.0 / n < TARGET_C5:
                    # good enough – stop early
                    break
        else:
            a[i] -= delta
            a[j] += delta

    return best_a, best_corr


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ----- initial half‑vector -------------------------------------------------
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
        m0 = n0 // 2
        # project onto complementary symmetry: h[i]+h[i+m]=1
        a0 = (parent[:m0] + (1.0 - parent[m0:])) * 0.5
    else:
        n0 = 360
        m0 = n0 // 2
        a0 = rng.random(m0)

    # ----- annealing on base size --------------------------------------------
    base_deadline = start + (deadline - start) * 0.6
    best_a, best_corr = _simulated_annealing_half(a0, n0, base_deadline, rng)
    best_h = _full_h(best_a)
    best_c5 = best_corr * 2.0 / n0

    # If already under the target, finish early
    if best_c5 < TARGET_C5:
        final_c5 = best_c5
        return best_h.tolist(), float(final_c5), int(n0)

    # ----- up‑sampling --------------------------------------------------------
    if time.time() < deadline and n0 * 2 <= 720:
        n_up = n0 * 2
        m_up = n_up // 2
        xs = np.arange(m0)
        xs_up = np.linspace(0, m0 - 1, m_up)
        a_up = np.interp(xs_up, xs, best_a)
        a_up = np.clip(a_up, 0.0, 1.0)

            if abs(diff) < 1e-12:
                break

    return best_h.tolist(), float(final_c5), int(n0)
