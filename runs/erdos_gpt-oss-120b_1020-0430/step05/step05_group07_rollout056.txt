def project_onto_sum_bounds(x, target, eps=1e-12, max_iter=100):
    lo = -1.0 # maybe -inf
    hi = 1.0
    # Actually we need compute λ such that sum(min(max(x - λ, 0), 1)) = target.
    # λ can be from -inf to +inf; we can bound lo = min(x)-1, hi = max(x)
    lo = min(x) - 1.0
    hi = max(x)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= eps:
            return y
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - mid, 0.0, 1.0)

```python
def compute_c5_fft(h):
    n = h.size
    # length for full correlate: 2n-1
    m = 2*n - 1
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1 - h, m)
    corr = np.fft.irfft(H * G, m)
    # corr is length m, corresponds to np.correlate full
    return float(np.max(corr) * (2.0 / n))
```

from scipy.optimize import minimize

def slsqp_optimize(h, beta, rng, max_iter=800):
    n = h.size
    target = n/2.0
    def softmax_obj(x):
        # compute correlation via FFT
        m = 2*n-1
        H = np.fft.rfft(x, m)
        G = np.fft.rfft(1.0 - x, m)
        corr = np.fft.irfft(H*G, m)
        bc = beta * corr
        # log-sum-exp
        max_bc = np.max(bc)
        lse = max_bc + np.log(np.sum(np.exp(bc - max_bc)))
        return lse / beta
    def sum_eq(x):
        return np.sum(x) - target
    cons = {'type': 'eq', 'fun': sum_eq}
    bounds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # project onto sum exactly via water-filling
    x = project_onto_sum_box(x, target)
    return x

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap using FFT (same scaling as the reference)."""
    n = h.size
    m = 2 * n - 1
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * G, m)
    return float(np.max(corr) * (2.0 / n))

def project_onto_sum_box(x: np.ndarray, target: float, eps=1e-12, max_iter=60) -> np.ndarray:
    """Euclidean projection of x onto {0≤y≤1, sum(y)=target}."""
    if 0 <= x.min() and x.max() <= 1 and abs(x.sum() - target) <= eps:
        return x
    lo, hi = -1.0, 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= eps:
            return y
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def thue_morse(n: int) -> np.ndarray:
    a = np.arange(n, dtype=np.int64)
    return np.fromiter((i.bit_count() & 1 for i in a), dtype=np.int8, count=n)

def legendre_binary(p: int) -> np.ndarray:
    r = np.zeros(p, dtype=np.int8)

def slsqp_optimize(h: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 600):
    n = h.size
    target = n / 2.0

    def softmax_obj(x):
        m = 2 * n - 1
        H = np.fft.rfft(x, m)
        G = np.fft.rfft(1.0 - x, m)

# ---------- main routine ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h, best_c, best_n = None, np.inf, None

    def record(h):
        nonlocal best_h, best_c, best_n
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, h.size

    # candidate sizes (increase later if time permits)
    sizes = [180, 240, 360, 480]

    for n in sizes:
        if time.time() > deadline:
            break

        # --- generate starting vectors ---
        starts = []

        if "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            starts.append(interp)
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            starts.append(bin_vec)

        tm = thue_morse(n).astype(float)
        starts.append(project_onto_sum_box(tm, n / 2.0))

        p = largest_prime_leq(n)
        if p >= 3:
            leg = legendre_binary(p).astype(float)
            leg_big = np.tile(leg, (n // p) + 1)[:n]
            starts.append(project_onto_sum_box(leg_big, n / 2.0))

        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        rand_frac = rng.random(n)
        starts.append(project_onto_sum_box(rand_frac, n / 2.0))

        rand_frac2 = rng.random(n) ** 2
        starts.append(project_onto_sum_box(rand_frac2, n / 2.0))

        # a few Dirichlet-like starts
        for _ in range(2):
            a = rng.random(n) + 0.1
            frac = rng.gamma(shape=a, scale=1.0)
            starts.append(project_onto_sum_box(frac, n / 2.0))

        # --- optimise each start ---
        for h0 in starts:
            if time.time() > deadline:
                break
            h = project_onto_sum_box(h0, n / 2.0)

            # gradient descent with early stop
            max_iter = 1500
            best_local = compute_c5(h)
            no_imp = 0
            step0 = 0.05
            for it in range(max_iter):
                if time.time() > deadline:
                    break
                m = 2 * n - 1
                H = np.fft.rfft(h, m)
                G = np.fft.rfft(1.0 - h, m)
                corr = np.fft.irfft(H * G, m)
                idx = np.argmax(corr) - (n - 1)
                k = abs(idx)

                g = np.zeros_like(h)
                if k == 0:
                    g[:] = 1.0 - 2.0 * h
                else:
                    g[: n - k] += 1.0 - h[k:]
                    g[k:] -= h[: n - k]
                    g[n - k :] += 1.0

                step = step0 * (0.995 ** it)
                h = project_onto_sum_box(h - step * g, n / 2.0)

                if (it % 50) == 0:
                    c = compute_c5(h)
                    if c < best_local - 1e-12:
                        best_local, no_imp = c, 0
                        record(h)
                    else:
                        no_imp += 1
                    if no_imp >= 300:
                        break

                # occasional random transfer to escape plateaus
                if (it % 400) == 0:
                    i, j = rng.choice(n, size=2, replace=False)
                    tr = min(h[i], 1.0 - h[j])
                    if tr > 0:
                        delta = tr * rng.random()
                        h[i] -= delta
                        h[j] += delta
                        h = project_onto_sum_box(h, n / 2.0)

            # short SA refinement
            sub_dead = min(deadline, time.time() + min(6.0, (deadline - time.time()) * 0.2))
            h_sa, c_sa = fractional_sa_hill_climb(h.copy(), rng, sub_dead)
            record(h_sa)
            h = h_sa

    # ----- upsample the best candidate if time allows -----
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = project_onto_sum_box(h_up, n_up / 2.0)

        max_iter = 1200
        best_local = compute_c5(h_up)
        no_imp = 0
        for it in range(max_iter):
            if time.time() > deadline:
                break
            m = 2 * n_up - 1
            H = np.fft.rfft(h_up, m)
            G = np.fft.rfft(1.0 - h_up, m)
            corr = np.fft.irfft(H * G, m)
            idx = np.argmax(corr) - (n_up - 1)
            k = abs(idx)

            g = np.zeros_like(h_up)
            if k == 0:
                g[:] = 1.0 - 2.0 * h_up
            else:
                g[: n_up - k] += 1.0 - h_up[k:]
                g[k:] -= h_up[: n_up - k]
                g[n_up - k :] += 1.0

            step = 0.02 * (0.99 ** it)
            h_up = project_onto_sum_box(h_up - step * g, n_up / 2.0)

            if (it % 60) == 0:
                c = compute_c5(h_up)
                if c < best_local - 1e-12:
                    best_local, no_imp = c, 0
                    record(h_up)
                else:
                    no_imp += 1
                if no_imp >= 200:
                    break

        if compute_c5(h_up) < best_c:
            best_h, best_c, best_n = h_up, compute_c5(h_up), n_up

    # ----- final SLSQP polishing if time remains -----
    if best_h is not None and time.time() < deadline:
        for beta in [20.0, 80.0, 200.0, 500.0, 2000.0, 5000.0]:
            if time.time() > deadline:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, rng, max_iter=800)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c = h_ref, c_ref
            except Exception:
                continue

    # ----- fallback -----
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_onto_sum_box(best_h, n / 2.0)
        best_c = compute_c5(best_h)
        best_n = n

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_sum_box(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
