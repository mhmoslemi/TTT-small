import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # random adjustments until exact
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, rng)
    return x


def autocorr(g: np.ndarray) -> np.ndarray:
    """Circular autocorrelation S_k = Σ g_i g_{i+k} (k=0..n-1)."""
    G = np.fft.fft(g)
    corr = np.fft.ifft(G * np.conj(G)).real
    return np.rint(corr).astype(int)


def anneal(g0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simulated‑annealing on binary g to maximize min_{k>0} S_k."""
    n = g0.size
    g = g0.copy()
    S = autocorr(g)
    best_g = g.copy()
    best_min = int(np.min(S[1:]))
    start = time.time()
    T0, T_end = 200.0, 1e-3
    while time.time() - start < time_budget:
        # temperature schedule
        frac = (time.time() - start) / time_budget
        T = T0 * ((T_end / T0) ** frac)
        # pick a random +1 and -1 to swap
        pos = np.where(g == 1)[0]
        neg = np.where(g == -1)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        g_new = g.copy()
        g_new[i] = -1
        g_new[j] = 1
        S_new = autocorr(g_new)
        min_new = int(np.min(S_new[1:]))
        delta = min_new - best_min
        if delta > 0 or rng.random() < np.exp(delta / max(T, 1e-12)):
            g = g_new
            S = S_new
            if min_new > best_min:
                best_min = min_new
                best_g = g.copy()
    return best_g, best_min


def targeted_hill_climb(g0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Improves min S_k by swapping mismatched pairs for the worst shift."""
    n = g0.size
    g = g0.copy()
    S = autocorr(g)
    best_min = int(np.min(S[1:]))
    best_g = g.copy()
    start = time.time()
    while time.time() - start < time_budget:
        # worst non‑zero shift
        k = int(np.argmin(S[1:]) + 1)
        g_shift = np.roll(g, -k)
        # indices where g_i=1, g_{i+k}=-1
        idx_pos = np.where((g == 1) & (g_shift == -1))[0]
        # indices where g_i=-1, g_{i+k}=1
        idx_neg = np.where((g == -1) & (g_shift == 1))[0]
        if idx_pos.size and idx_neg.size:
            i = rng.choice(idx_pos)
            j = rng.choice(idx_neg)
            g_new = g.copy()
            g_new[i] = -1
            g_new[j] = 1
        else:
            # fallback random opposite‑sign swap
            pos = np.where(g == 1)[0]
            neg = np.where(g == -1)[0]
            if pos.size == 0 or neg.size == 0:
                break
            i = rng.choice(pos)
            j = rng.choice(neg)
            g_new = g.copy()
            g_new[i] = -1
            g_new[j] = 1
        S_new = autocorr(g_new)
        min_new = int(np.min(S_new[1:]))
        if min_new > best_min:
            g, S = g_new, S_new
            best_min = min_new
            best_g = g.copy()
    return best_g, best_min

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512

    # ---- initial binary g -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        # threshold at 0.5 and fix exact balance
        diff = np.abs(h0 - 0.5)
        g = np.where(h0 >= 0.5, 1, -1)
        target_ones = n // 2
        cur_ones = np.sum(g == 1)
        if cur_ones != target_ones:
            if cur_ones > target_ones:
                excess = cur_ones - target_ones
                ones_idx = np.where(g == 1)[0]
                flip = ones_idx[np.argsort(diff[ones_idx])[:excess]]
                g[flip] = -1
            else:
                deficit = target_ones - cur_ones
                neg_idx = np.where(g == -1)[0]
                flip = neg_idx[np.argsort(diff[neg_idx])[:deficit]]
                g[flip] = 1
    else:
        g = np.full(n, -1, dtype=int)
        g[rng.choice(n, n // 2, replace=False)] = 1

    # ---- time budgeting helpers -------------------------------------------
    def time_left():
        return budget_s - (time.time() - start)

    # ---- stage 1: annealing -----------------------------------------------
    t_anneal = min(time_left() * 0.5, 600.0)
    g_best, min_best = anneal(g, rng, t_anneal)

    # ---- stage 2: targeted hill‑climb --------------------------------------
    if time_left() > 5.0:
        t_target = min(time_left() * 0.35, 200.0)
        g_best, min_best = targeted_hill_climb(g_best, rng, t_target)

    # ---- stage 3: binary to continuous ------------------------------------
    h_best = (g_best + 1) / 2.0
    h_best = enforce_sum(h_best, rng)
    best_c = compute_c5(h_best)

    # ---- stage 4: soft‑max SLSQP refinements -------------------------------
    if time_left() > 5.0:
        betas = [10.0, 100.0, 1000.0, 5000.0, 20000.0]
        for beta in betas:
            if time_left() < 2.0:
                break
            # allocate a modest iteration budget
            iters = 400
            h_opt = slsqp_optimize(h_best, beta, rng, max_iter=iters)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c, h_best = c_opt, h_opt
                h_best = enforce_sum(h_best, rng)

    # ---- stage 5: tiny random epsilon hill‑climb ---------------------------
    if time_left() > 2.0:
        h = h_best.copy()
        c = best_c
        steps = int(time_left() * 180)  # about 180 trials per second
        for _ in range(steps):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            eps = 0.01 * rng.random()
            if h[i] - eps < 0.0 or h[j] + eps > 1.0:
                continue
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps
            c_new = compute_c5(h_new)
            if c_new < c:
                h, c = h_new, c_new
        h_best, best_c = h, c

    # ---- final sanity check -------------------------------------------------
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = enforce_sum(h_best, rng)
    final_c = compute_c5(h_best)
    best_c = min(best_c, final_c)

    return h_best.tolist(), float(best_c), int(h_best.size)
