def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, L)
    R = np.fft.rfft(rev, L)
    conv = np.fft.irfft(H * R, L)[:2*n-1]
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

class State:
    def __init__(self, h):
        self.h = h
        self.n = h.shape[0]
        self.S = self.h.sum()
        self.c = self._compute_c()
        self.max_idx = int(np.argmax(self.c))
        self.max_c = float(self.c[self.max_idx])

    def _compute_c(self):
        # q = circular correlation of h with h
        n = self.n
        L = 1 << ((2*n - 1).bit_length())
        H = np.fft.rfft(self.h, L)
        # q_k for circular correlation: real part of ifft(H * conj(H))
        q = np.fft.irfft(H * np.conj(H), L)[:n]
        # q_k corresponds to sum_i h_i * h_{i+k} (circular)
        # c_k = S - q_k
        return self.S - q

    def update_swap(self, i, j):
        # i has value 1, j has 0
        n = self.n
        # compute d_k
        indices = np.arange(n)
        d = self.h[(i + indices) % n] - self.h[(j + indices) % n] + self.h[(i - indices) % n] - self.h[(j - indices) % n]
        # update c
        self.c = self.c + d
        # swap values
        self.h[i] = 0.0
        self.h[j] = 1.0
        self.max_c = float(np.max(self.c))
        self.max_idx = int(np.argmax(self.c))

    def compute_c5(self):
        return (2.0 / self.n) * self.max_c

import time, math, numpy as np
from itertools import combinations
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

# ---------- fast C5 computation ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 using FFT (full correlation)."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    H = np.fft.rfft(h, L)
    R = np.fft.rfft(rev, L)
    conv = np.fft.irfft(H * R, L)[: 2 * n - 1]
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def compute_full_corr(x: np.ndarray) -> np.ndarray:
    """Return full correlation array for soft‑max SLSQP objective."""
    n = x.shape[0]
    rev = (1.0 - x)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(x, L)
    R = np.fft.rfft(rev, L)
    return np.fft.irfft(H * R, L)[: 2 * n - 1]

# ---------- gradient utilities ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.shape[0]
    # locate worst shift using FFT correlation
    rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    R = np.fft.rfft(rev, L)
    conv = np.fft.irfft(H * R, L)[: 2 * n - 1]
    idx = np.argmax(conv) - (n - 1)
    k = abs(idx)

    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

# ---------- SLSQP refinement ----------
def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax_obj(x):
        conv = compute_full_corr(x)
        m = np.max(conv)
        # log‑sum‑exp in a numerically stable way
        return (m + np.log(np.sum(np.exp(beta * (conv - m))))) / beta

    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())

# ---------- state for binary annealing ----------
class BinaryState:
    def __init__(self, h: np.ndarray):
        self.h = h.copy()
        self.n = h.size
        self.S = self.h.sum()          # = n/2
        self._compute_c()

    def _compute_c(self):
        n = self.n
        H = np.fft.rfft(self.h, n)
        q = np.fft.irfft(H * np.conj(H), n)   # circular autocorrelation
        self.c = self.S - q
        self.max_idx = int(np.argmax(self.c))
        self.max_c = float(self.c[self.max_idx])

    def compute_c5(self) -> float:
        return (2.0 / self.n) * self.max_c

# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512

    # ----- exhaustive search over periodic binary patterns (p=8 and p=16) -----
    best_h = None
    best_c = float('inf')
    for p in (8, 16):
        reps = n // p
        half = p // 2
        for ones in combinations(range(p), half):
            base = np.zeros(p, dtype=float)
            base[list(ones)] = 1.0
            h = np.tile(base, reps)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    # ensure feasibility
    best_h = enforce_sum(best_h, rng)

    # ----- optional continuous refinement (remaining time) -----
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    # if we still have at least 0.2 * budget, do a few refinement phases
    if remaining > 0.2 * budget_s:
        # short gradient‑directed hill‑climb
        h_tmp, c_tmp = guided_hill_climb(best_h, rng, max_iters=2500, step_scale=0.7)
        if c_tmp < best_c:
            best_c, best_h = c_tmp, h_tmp.copy()

        # SLSQP passes with increasing β
        for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
            if time.time() - start > 0.95 * budget_s:
                break
            maxit = 800 if beta >= 200 else 500
            h_tmp = slsqp_optimize(best_h, beta, max_iter=maxit)
            c_tmp = compute_c5(h_tmp)
            if c_tmp < best_c:
                best_c, best_h = c_tmp, h_tmp.copy()

        # final fine‑grained gradient tidy‑up
        h_tmp = gradient_refine(best_h, rng, step=0.01, max_iter=12)
        c_tmp = compute_c5(h_tmp)
        if c_tmp < best_c:
            best_c, best_h = c_tmp, h_tmp.copy()

    # ----- final sanity check -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
