def hill_climb(h_bin, start_time, deadline):
    best_h = h_bin.copy()
    best_max = np.max(np.correlate(...))
    cur_h = h_bin.copy()
    ones = list(np.flatnonzero(cur_h==1))
    zeros = list(np.flatnonzero(cur_h==0))
    # while time < deadline
    iter = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i]=0
        cur_h[j]=1
        new_max = np.max(np.correlate(cur_h.astype(np.int8), (1 - cur_h).astype(np.int8), mode='full'))
        if new_max < best_max:
            best_max = new_max
            best_h = cur_h.copy()
            # update lists: replace i with j in ones, replace j with i in zeros
            # Since we swapped, now position i is zero and j is one.
            # Update ones list: find index of i and replace with j.
            idx_i = ones.index(i) # O(len(ones)) but small
            ones[idx_i] = j
            idx_j = zeros.index(j)
            zeros[idx_j] = i
        else:
            # revert
            cur_h[i]=1
            cur_h[j]=0
        iter += 1
        # optional time check every 1000 iter
    return best_h, best_max

def float_perturb(h, start_time, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = 0.1
    while time.time() < deadline:
        n = h.size
        i = rng.integers(0,n)
        j = rng.integers(0,n-1)
        if j >= i: j+=1
        # compute max possible d
        max_pos = min(step, 1 - best_h[i], best_h[j])

            best_h = cand

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq","fun":sum_eq_constraint}
    res = minimize(smooth_max_obj, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def adjust_sum_exact(h, rng):
    target = h.size/2.0
    diff = target - np.sum(h)
    n = h.size
    # adjust iteratively
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0,n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def smooth_max_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_max_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

    best_max = int(

            np.correlate(cur_h.astype(np.int8), (1 - cur_h).astype(np.int8), mode="full")
        )
    )
    ones = np.flatnonzero(cur_h == 1).tolist()
    zeros = np.flatnonzero(cur_h == 0).tolist()
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_max = int(
            np.max(
                np.correlate(cur_h.astype(np.int8), (1 - cur_h).astype(np.int8), mode="full")
            )
        )
        if new_max < best_max:
            best_max = new_max
            best_h = cur_h.copy()
            ones[i_idx] = j
            zeros[j_idx] = i
        else:
            cur_h[i] = 1

        pert_h, pert_c = float_perturb(best_h, rng, deadline)
        if pert_c < best_c:
            best_c = pert_c
            best_h = pert_h.copy()

    # Final feasibility fix
    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(n)
