import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to have sum = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from positive‑gradient index to negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Hill climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Simulated‑annealing on binary vectors with exactly n/2 ones."""
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = np.array(ones, dtype=int)
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i = rng.choice(ones_arr)
            j = rng.choice(zeros_arr)
            h[i], h[j] = 0.0, 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[ones_arr == i] = j
                zeros_arr[zeros_arr == j] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i], h[j] = 1.0, 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c


def continuous_sa(h0: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simulated‑annealing on continuous vectors preserving the sum."""
    n = h0.size
    curr_h = h0.copy()
    curr_c = compute_c5(curr_h)
    best_h, best_c = curr_h.copy(), curr_c
    start = time.time()
    T0, T_end = 0.01, 1e-6
    sf0, sf_end = 0.5, 0.01
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac
        sf = sf0 * (sf_end / sf0) ** frac
        i = rng.integers(0, n)
        if curr_h[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if curr_h[j] >= 1.0 - 1e-12:
            continue
        max_eps = min(curr_h[i], 1.0 - curr_h[j])
        eps = rng.random() * max_eps * sf
        if eps <= 0.0:
            continue
        new_h = curr_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < curr_c or rng.random() < np.exp(-(new_c - curr_c) / max(T, 1e-12)):
            curr_h, curr_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # target size for fast searches
    n = 256

    # initialise from parent if possible
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            if init.size > n:
                factor = init.size // n
                init = init[:n * factor].reshape(n, factor).mean(axis=1)
            else:
                x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                init = np.interp(x_new, x_old, init)
        init = np.clip(init, 0.0, 1.0)
        init = enforce_sum(init, rng)
    else:
        init = rng.random(n)
        init = enforce_sum(init, rng)

    best_h, best_c = init.copy(), compute_c5(init)

    # ---- binary simulated annealing (≈15% of budget) ----
    t_bin = total_budget * 0.15

        h_bin, c_bin = binary_sa(n, rng, t_bin)
        if c_bin < best_c:
            best_h, best_c = h_bin.copy(), c_bin

    # ---- continuous simulated annealing (≈55% of budget) ----
    t_cont = total_budget * 0.55
    if t_cont > 0:
        h_cont, c_cont = continuous_sa(best_h, rng, t_cont)
        if c_cont < best_c:
            best_h, best_c = h_cont.copy(), c_cont

    # ---- guided hill‑climb (≈15% of budget) ----
    t_hc = total_budget * 0.15
    if t_hc > 0:
        max_iters = int(min(3000, t_hc * 500))
        h_hc, c_hc = guided_hill_climb(best_h, rng,
                                         max_iters=max_iters,
                                         step_scale=0.7)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc

    # ---- soft‑max SLSQP refinement (remaining time) ----
    while time.time() - start < total_budget * 0.95:
        for beta in (20., 80., 200., 800., 3000., 8000.):
            if time.time() - start > total_budget * 0.96:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=600)
            best_c = compute_c5(best_h)
            if best_c <= 0.3808:
                break

    # ---- optional up‑sampling to 512 points ----
    if time.time() - start < total_budget * 0.98 and best_h.size * 2 <= 512:
        n_up = best_h.size * 2
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        h_up, c_up = guided_hill_climb(h_up, rng,
                                        max_iters=1500,
                                        step_scale=0.7)
        if c_up < best_c:
            best_h, best_c = h_up.copy(), c_up
        for beta in (20., 200., 2000., 8000.):
            if time.time() - start > total_budget:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=600)
            best_c = compute_c5(best_h)

    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
