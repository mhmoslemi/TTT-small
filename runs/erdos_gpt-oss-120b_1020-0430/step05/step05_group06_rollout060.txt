def swap_and_update(h, O, ones, zeros, i, j):
    # h[i]=1, h[j]=0
    n = h.shape[0]
    offset = n - 1
    # Precompute 1 - h
    not_h = 1 - h

    # i as first index removal
    start_i = offset - i
    O[start_i:start_i + n] -= not_h

    # i as second index addition
    # iterate through ones
    # We could use for k in ones:
    #    if k != i: O[offset + i - k] += 1
    for k in ones:
        if k == i:
            continue
        O[offset + i - k] += 1

    # j as first index addition
    start_j = offset - j
    O[start_j:start_j + n] += not_h
    # correct self term
    O[offset] -= not_h[j]

    # j as second index removal
    for k in ones:
        O[offset + k - j] -= 1

    # swap h entries
    h[i] = 0
    h[j] = 1

    # Update ones and zeros lists: replace i with j in ones; replace j with i in zeros
    # find positions
    idx_i = ones.index(i)
    ones[idx_i] = j
    idx_j = zeros.index(j)
    zeros[idx_j] = i

    # return new max O
    return O.max()

def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2*n - 1, dtype=np.int32)
    for _ in range(L):
        # compute candidate max for each zero position p
        # Equivalent to O_adj = O + 1 (since adding a 1 adds 1 to O for each shift where (1 - h_{i+k}) =1)
        # For each zero p:
        #   window = O_adj[p_start : p_start + n] - h
        #   w = max(window)
        # We'll compute best p that minimizes w.
        # This is O(n^2) but okay for n=360.
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        offset = n - 1
        for p in zeros:
            start = offset - p
            # O_adj slice
            window = O_adj[start:start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[offset] = 0
    return h, O

def hill_climb_binary(n, h_start, deadline, rng, target_max):
    h = h_start.copy()
    O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
    offset = n - 1
    O[offset] = 0
    max_O = int(O.max())
    best_h = h.copy()
    best_max = max_O

    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))

    T = 5.0
    T_min = 1e-4
    decay = 0.9995
    no_improve = 0

    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_max = swap_and_update(h, O, ones, zeros, i, j)

        delta = new_max - max_O
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > T_min:
                if rng.random() < np.exp(-delta / T):
                    accept = True

        if accept:
            max_O = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
                if best_max <= target_max:
                    break
        else:
            # revert swap
            swap_and_update(h, O, ones, zeros, j, i)
            no_improve += 1

        T = max(T * decay, T_min)

        if no_improve >= 5000:
            # random restart
            h = np.zeros(n, dtype=np.int8)
            idx = rng.choice(n, size=n//2, replace=False)
            h[idx] = 1
            O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
            O[offset] = 0
            max_O = int(O.max())
            ones = list(idx)
            zeros = list(np.setdiff1d(np.arange(n), idx))
            no_improve = 0
            T = 5.0

    return best_h, best_max

Now swap_and_update definition:

def swap_and_update(h, O, ones, zeros, i, j):
    # assumes h[i] == 1, h[j] == 0
    n = h.shape[0]
    offset = n - 1
    not_h = 1 - h  # array of 0/1

    # i as first index removal
    start_i = offset - i
    O[start_i:start_i + n] -= not_h

    # i as second index addition
    for k in ones:
        if k == i:
            continue
        O[offset + i - k] += 1

    # j as first index addition
    start_j = offset - j
    O[start_j:start_j + n] += not_h
    # correct self term
    O[offset] -= not_h[j]

    # j as second index removal
    for k in ones:
        O[offset + k - j] -= 1

    # swap h entries
    h[i] = 0
    h[j] = 1

    # update ones and zeros lists
    idx_i = ones.index(i)
    ones[idx_i] = j
    idx_j = zeros.index(j)
    zeros[idx_j] = i

    return int(O.max())

import time
import numpy as np
from scipy.optimize import minimize

def compute_O_full(h):
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    hf = h.astype(np.float64)
    nf = (1.0 - hf)
    H = np.fft.rfft(hf, m)
    N = np.fft.rfft(nf, m)
    conv = np.fft.irfft(H * N, m)[: 2 * n - 1]
    return np.rint(conv).astype(np.int32)

def swap_and_update(h, O, ones, zeros, i, j):
    """swap 1 at i with 0 at j, update O and index lists, return new max."""
    n = h.shape[0]
    off = n - 1
    not_h = 1 - h
    # i as first index removal
    s_i = off - i
    O[s_i : s_i + n] -= not_h
    # i as second index addition
    for k in ones:
        if k == i:
            continue
        O[off + i - k] += 1
    # j as first index addition
    s_j = off - j
    O[s_j : s_j + n] += not_h
    O[off] -= not_h[j]
    # j as second index removal
    for k in ones:
        O[off + k - j] -= 1
    # swap values
    h[i] = 0
    h[j] = 1
    # update index lists
    idx_i = ones.index(i)
    ones[idx_i] = j
    idx_j = zeros.index(j)
    zeros[idx_j] = i
    return int(O.max())

def hill_climb_binary(n, h_start, deadline, rng, target_max):
    h = h_start.copy()
    O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
    O[n - 1] = 0
    max_O = int(O.max())
    best_h = h.copy()
    best_max = max_O
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    T, T_min, decay = 5.0, 1e-4, 0.9995
    stuck = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_max = swap_and_update(h, O, ones, zeros, i, j)
        delta = new_max - max_O
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > T_min and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            max_O = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stuck = 0
                if best_max <= target_max:

        if stuck >= 5000:

def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    off = n - 1
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        cand = []
        for p in zeros:
            start = off - p
            w = (O_adj[start : start + n] - h).max()
            if best_w is None or w < best_w:
                best_w = w
                cand = [p]
            elif w == best_w:
                cand.append(p)
        p = rng.choice(cand)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[off] = 0
    return h, O

# helpers for smooth‑max SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def smooth_max_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_max_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    target = n / 2.0
    diff = target - np.sum(x)
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - np.sum(x)
    return x

                h_best = h_float.copy()
    final_c = max_best * (2.0 / n)
    return h_best.astype(float).tolist(), float(final_c), int(n)
