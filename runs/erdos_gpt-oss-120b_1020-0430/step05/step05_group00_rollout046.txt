import time
import numpy as np
from scipy.optimize import minimize

# ------------------ helper functions ------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Gradient‑directed hill climbing."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0.0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ----- soft‑max SLSQP optimizer -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp (stable)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----- binary simulated annealing (0/1 vectors) -----
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        # random binary start
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = np.array(ones, dtype=int)
        zeros_arr = np.setdiff1d(np.arange(n, dtype=int), ones_arr, assume_unique=True)
        # inner SA loop
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i = ones_arr[rng.integers(len(ones_arr))]
            j = zeros_arr[rng.integers(len(zeros_arr))]
            # swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                # update index lists
                idx_i = np.where(ones_arr == i)[0][0]
                idx_j = np.where(zeros_arr == j)[0][0]
                ones_arr[idx_i] = j
                zeros_arr[idx_j] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c

def continuous_optimize(n, rng, time_budget, init_h=None):
    """Run a long hill‑climb + SLSQP, with optional short restarts."""
    start = time.time()
    best_h, best_c = None, np.inf

    # the first (long) run
    h = init_h.copy() if init_h is not None else rng.random(n)
    h = enforce_sum(h, rng)
    # hill‑climb
    max_iters = int(min(30000, (time_budget) * 500))
    h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.7)
    best_h, best_c = h, c

    # SLSQP passes with increasing β
    beta_seq = (20., 80., 200., 800., 3000., 8000., 20000.)
    for beta in beta_seq:
        if time.time() - start > time_budget * 0.96:
            break
        max_iter = 2000 if beta >= 200 else 1000
        h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # final gradient tidy‑up
    if time.time() - start < time_budget * 0.98:
        h = gradient_refine(best_h, rng, step=0.01, max_iter=8)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # short random restarts if time permits
    while time.time() - start < time_budget * 0.995:
        # initialise from current best with small perturbation or fresh random
        if best_h is not None and rng.random() < 0.5:
            h = best_h.copy()
            for _ in range(5):
                i = rng.integers(0, n)
                delta = (rng.random() - 0.5) * 0.2
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = rng.random(n)
            h = enforce_sum(h, rng)

        # brief hill‑climb
        max_iters = int(min(5000, (time_budget - (time.time() - start)) * 200))
        h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.7)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # quick SLSQP
        for beta in (20., 80., 200.):
            if time.time() - start > time_budget * 0.995:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=500)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # initial solution if provided
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ---------- binary simulated annealing (≈30% time) ----------
    binary_time = total_budget * 0.30
    h_bin, c_bin = binary_sa(256, rng, binary_time)
    if c_bin < best_c:
        best_h, best_c, best_n = h_bin, c_bin, 256

    # ---------- continuous refinement (remaining time) ----------
    remaining = total_budget - (time.time() - start)
    if remaining > 0:
        # target size 512 for final answer
        target_n = 512
        # prepare an initial vector at size 512
        if best_h is not None and best_n == target_n:
            init_h = best_h.copy()
        elif best_h is not None:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
            init_h = np.interp(x_new, x_old, best_h)
            init_h = np.clip(init_h, 0.0, 1.0)
            init_h = enforce_sum(init_h, rng)
        else:
            init_h = rng.random(target_n)
            init_h = enforce_sum(init_h, rng)

        h_cont, c_cont = continuous_optimize(target_n, rng, remaining, init_h=init_h)
        if c_cont < best_c:
            best_h, best_c, best_n = h_cont.copy(), c_cont, target_n

    # ---------- final stochastic tweaking ----------
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        # very small transfer
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            eps = rng.random() * max_eps * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # ---------- sanitisation ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
