def guided_hill_climb_eps(h, rng, max_iters=2000):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    dx = 2.0 / n
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        worst_raw = corr[max_idx]
        # second worst:
        # set current max to -inf temporarily and find next max
        corr2 = corr.copy()
        corr2[max_idx] = -np.inf
        second_raw = np.max(corr2)
        # compute gradient for worst shift
        idx_shift = max_idx - (n - 1)
        k = abs(idx_shift)
        # gradient:
        grad = np.ones_like(h)
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        # find i = argmax(grad), j = argmin(grad)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # nothing to move
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        denom = dx * (grad[i] - grad[j])  # positive
        if denom <= 0:
            eps = max_eps * 0.5
        else:
            eps_needed = (worst_raw - second_raw) / denom
            # Clip to feasible range
            eps = min(max_eps, max(0.0, eps_needed * 0.99))
            if eps <= 1e-12:
                eps = max_eps * 0.5
        # update
        h[i] -= eps
        h[j] += eps
        # compute new c
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # occasional small random perturb to escape
        # maybe after every certain steps
        if (it+1) % (max_iters//10 + 1) == 0:
            # random small perturb
            i2, j2 = rng.integers(0, n, size=2)
            max_eps2 = min(h[i2], 1.0 - h[j2])
            if max_eps2 > 0:
                eps2 = rng.random() * max_eps2 * 0.1
                h[i2] -= eps2
                h[j2] += eps2
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=600):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # use log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum precisely (may be tiny diff)
    return enforce_sum(x, np.random.default_rng())
```

import time, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator, tol: float = 1e-12) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > tol:
        if diff > 0:  # need to add mass
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:        # need to remove mass
            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    h = h * (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


# ---------- gradient‑driven hill climb ----------
def guided_hill_climb_eps(h: np.ndarray, rng: np.random.Generator,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """Hill‑climb using worst‑shift gradient, with adaptive step size."""
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iters):
        # raw correlation and two worst values
        corr = np.correlate(h, 1 - h, mode="full")
        worst_idx = int(np.argmax(corr))
        worst_raw = float(corr[worst_idx])
        # second worst (set worst to -inf temporarily)
        corr2 = corr.copy()
        corr2[worst_idx] = -np.inf
        second_raw = float(np.max(corr2))

        # gradient for the worst shift
        shift = worst_idx - (n - 1)
        k = abs(shift)
        grad = np.ones_like(h)
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:]      -= h[: n - k]

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue

        denom = dx * (grad[i] - grad[j])
        if denom > 1e-12:
            eps_needed = (worst_raw - second_raw) / denom
            eps = min(max_eps, max(0.0, eps_needed * 0.99))
            if eps < 1e-12:
                eps = max_eps * 0.5
        else:
            eps = max_eps * 0.5

        # apply move
        h[i] -= eps
        h[j] += eps

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

        # occasional tiny random perturbation to escape plateaus
        if (it + 1) % (max_iters // 10 + 1) == 0:
            i2, j2 = rng.integers(0, n, size=2)
            if i2 != j2:
                max_eps2 = min(h[i2], 1.0 - h[j2])
                if max_eps2 > 0.0:
                    eps2 = rng.random() * max_eps2 * 0.1
                    h[i2] -= eps2
                    h[j2] += eps2
    return best_h, best_c


# ---------- SLSQP refinement ----------
def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    elapsed = lambda: time.time() - start
    n = 512

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0
    else:
        best_h = random_feasible_vector(n, rng)
    best_c = compute_c5(best_h)

    # ----- phase 1: many binary hill‑climbs -----
    while elapsed() < 0.30 * budget_s:
        h_bin = np.zeros(n)
        h_bin[rng.choice(n, n // 2, replace=False)] = 1.0
        h_bin, c_bin = guided_hill_climb_eps(h_bin, rng, max_iters=1200)
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin

    # ----- phase 2: refined hill‑climb on current best -----
    if elapsed() < 0.70 * budget_s:
        h_ref, c_ref = guided_hill_climb_eps(best_h, rng, max_iters=3000)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ----- phase 3: random local tweaks -----
    while elapsed() < 0.90 * budget_s:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.05
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c = nh, c

    # ----- phase 4: SLSQP fine‑tuning -----
    for beta in (200., 800., 3000., 8000., 20000., 80000.):

        maxit = 600 if beta <= 8000 else 400
        best_h = slsqp_optimize(best_h, beta, max_iter=maxit)
        best_c = compute_c5(best_h)

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
