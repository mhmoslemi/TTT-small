def gradient_from_weights_fft(h, w):
    n = h.size
    # compute conv1 = (w_rev) * (1 - h)
    w_rev = w[::-1]
    Lg = 1 << ((len(w_rev) + n - 1).bit_length())
    A = np.fft.fft(w_rev, n=Lg)
    B = np.fft.fft(1.0 - h, n=Lg)
    conv1_full = np.fft.ifft(A * B).real
    conv1 = conv1_full[:n]  # because conv1 length >= n

    A2 = np.fft.fft(w, n=Lg)
    B2 = np.fft.fft(h, n=Lg)
    conv2_full = np.fft.ifft(A2 * B2).real
    conv2 = conv2_full[:n]
    g = conv1 - conv2
    return g

import time
import numpy as np

# ---------- basic helpers ----------
def compute_corr_fft(h):
    """Linear correlation of h with (1-h) using FFT.
    Returns array of length 2*len(h)-1 ordered from lag -(n-1) .. + (n-1)."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())        # power of two >= 2n-1
    H = np.fft.fft(h, n=L)
    G = np.fft.fft(1.0 - h, n=L)
    corr_full = np.fft.ifft(H * np.conj(G)).real
    # negative lags are at the end
    neg = corr_full[-(n - 1):] if n > 1 else np.empty(0)
    pos = corr_full[:n]
    return np.concatenate([neg, pos])


def compute_c5(h):
    """Maximum overlap C5 for current h."""
    n = h.size
    corr = compute_corr_fft(h)
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, target, rng):
    """Redistribute mass to achieve Σh = target (respecting 0‑1 bounds)."""
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                continue
            h[i] += add
        else:
            sub = min(-diff, h[i])
            if sub <= 0:
                continue
            h[i] -= sub
        diff = target - h.sum()
    # final clipping (tiny numerical errors)
    np.clip(h, 0.0, 1.0, out=h)
    return h


def gradient_from_weights_fft(h, w):
    """Gradient of the soft‑max surrogate for given weights w."""
    n = h.size
    # w_rev for the first convolution term
    w_rev = w[::-1]

    Lg = 1 << ((len(w) + n - 1).bit_length())   # >= 3n-2
    # conv1 = w_rev * (1-h)
    A = np.fft.fft(w_rev, n=Lg)
    B = np.fft.fft(1.0 - h, n=Lg)
    conv1_full = np.fft.ifft(A * B).real
    conv1 = conv1_full[:n]

    # conv2 = w * h
    A2 = np.fft.fft(w, n=Lg)
    B2 = np.fft.fft(h, n=Lg)
    conv2_full = np.fft.ifft(A2 * B2).real
    conv2 = conv2_full[:n]

    return conv1 - conv2


# ---------- binary simulated annealing ----------
def binary_sa_hill_climb(h_bin, rng, deadline):
    """Swap‑based SA on binary vectors."""
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_imp = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_imp = it
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it - last_imp > 8000:
            break
    return best, best_c


# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()
    deadline = start + budget_s * 0.96   # small safety margin

    # ----- initialise h -----

    best_h = h.copy()

    # ----- soft‑max gradient annealing -----
    betas = [0.5, 1, 2, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    for beta in betas:

            if c < best_c:

                best_h = h.copy()

    # ----- binary refinement if needed -----
    if best_c > 0.381:   # still above target

    if time.time() < deadline and n * 2 <= 720:

        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = adjust_sum_exact(h_up, n2 / 2.0, rng)

        # short gradient annealing on the finer grid

                    break

    best_h = adjust_sum_exact(best_h, n / 2.0, rng)
