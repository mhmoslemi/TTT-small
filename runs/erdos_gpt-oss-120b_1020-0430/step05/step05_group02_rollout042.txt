def binary_hill_climb(n, rng, time_deadline):
    target = n/2
    # random start
    h = np.zeros(n, dtype=np.float64) # use float for correlation
    idx_ones = rng.choice(n, n//2, replace=False)
    h[idx_ones] = 1.0
    # compute autocorr
    c = np.correlate(h, h, mode='full')
    cur_worst = target - c.min()
    best_h = h.copy()
    best_worst = cur_worst
    ones = set(idx_ones.tolist())
    zeros = set(range(n)) - ones
    # parameters
    T = cur_worst * 0.5 + 1e-6
    decay = 0.999995
    it = 0
    no_improve = 0
    while time.time() < time_deadline and it < max_iters:
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # Apply i->0
        start_i = n - 1 - i
        vi = np.concatenate((h[:i+1][::-1], h[i+1:]))
        c[start_i:start_i+n] -= vi
        h[i] = 0.0
        # Apply j->1 (h now has i=0)
        start_j = n - 1 - j
        vj = np.concatenate((h[:j+1][::-1], h[j+1:]))
        c[start_j:start_j+n] += vj
        h[j] = 1.0
        new_worst = target - c.min()
        delta = new_worst - cur_worst
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True
        if accept:
            cur_worst = new_worst
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
            if cur_worst < best_worst:
                best_worst = cur_worst
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert: j->0, i->1
            # revert j change
            c[start_j:start_j+n] -= vj
            h[j] = 0.0
            # revert i change
            c[start_i:start_i+n] += vi
            h[i] = 1.0
            no_improve += 1
        # update temperature
        T *= decay
        it += 1
        if no_improve > 50000:
            # random restart
            h = np.zeros(n, dtype=np.float64)
            idx_ones = rng.choice(n, n//2, replace=False)
            h[idx_ones] = 1.0
            c = np.correlate(h, h, mode='full')
            cur_worst = target - c.min()
            ones = set(idx_ones.tolist())
            zeros = set(range(n)) - ones
            T = cur_worst * 0.5 + 1e-6
            no_improve = 0
    # compute C5
    final_c = best_worst * (2.0 / n)
    return best_h, final_c

    best_h = None
    best_worst = n  # large
    # Pre-allocate arrays for speed
    ones = np.empty(target, dtype=np.int64)
    zeros = np.empty(target, dtype=np.int64)
    pos_one = np.full(n, -1, dtype=np.int64)
    pos_zero = np.full(n, -1, dtype=np.int64)

    while time.time() < deadline:
        # init random binary
        idx_ones = rng.choice(n, target, replace=False)
        h = np.zeros(n, dtype=np.int8)
        h[idx_ones] = 1
        # initialize ones, zeros, pos arrays

        zeros_list = np.setdiff1d(np.arange(n), idx_ones, assume_unique=True)
        zeros[:] = zeros_list

def apply_change(h, c, idx, delta):

    # assume h[idx] is currently (delta==+1) ? 0 : 1 before change

        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size-1)
        if j >= i:
            j += 1
        # check ability to move
        # delta limited by min(step, 1 - h[i], h[j])
        delta_max = step
        if h[i] > 1 - step:
            delta_max = min(delta_max, 1 - h[i])
        if h[j] < step:
            delta_max = min(delta_max, h[j])
        if delta_max <= 1e-12:
            continue
        # apply move
        h[i] += delta_max
        h[j] -= delta_max
        new_c = compute_c5(h)
        delta_c = new_c - cur_c
        if delta_c <= 0 or rng.random() < np.exp(-delta_c / max(T, 1e-12)):

        T *= decay
        # optional step size reduction
        if T < 1e-6:
            step *= 0.5
            T = cur_c * 0.5 + 1e-6

import time
import numpy as np

# ------------------------------------------------------------
# Utility: compute C5 for a (fractional) step function.
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


# ------------------------------------------------------------
# Gradient refinement on the worst shift (as in the previous
# solution). Small number of iterations is enough for fine‑tuning.
# ------------------------------------------------------------
def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for __ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ------------------------------------------------------------
# Binary hill‑climbing / simulated‑annealing.
# Works on 0/1 vectors with exactly n/2 ones.
# ------------------------------------------------------------
def binary_hill_climb(n: int, rng: np.random.Generator,
                     deadline: float) -> tuple[np.ndarray, float]:
    target = n // 2
    best_worst = n * 2  # large initial value
    best_h = None
    max_total_iters = 5_000_000
    total_iters = 0
    while time.time() < deadline and total_iters < max_total_iters:
        # ----- random initial binary vector -----
        idx_one = rng.choice(n, target, replace=False)
        h = np.zeros(n, dtype=np.int8)
        h[idx_one] = 1
        c = np.correlate(h, h, mode="full")
        cur_worst = target - c.min()

        # index bookkeeping for O(1) swaps
        ones_idx = np.empty(target, dtype=np.int64)
        zeros_idx = np.empty(target, dtype=np.int64)
        pos_one = np.full(n, -1, dtype=np.int64)
        pos_zero = np.full(n, -1, dtype=np.int64)

        ones_idx[:] = idx_one
        pos_one[idx_one] = np.arange(target)

        zeros = np.setdiff1d(np.arange(n), idx_one, assume_unique=True)
        zeros_idx[:] = zeros
        pos_zero[zeros] = np.arange(target)

        # annealing parameters
        T = cur_worst * 0.5 + 1e-8
        decay = 0.999995
        no_improve = 0

        # ----- inner move loop -----
        while time.time() < deadline and total_iters < max_total_iters:
            total_iters += 1
            # pick a 1 and a 0
            i_pos = rng.integers(0, target)
            i = ones_idx[i_pos]
            j_pos = rng.integers(0, target)
            j = zeros_idx[j_pos]

            # ---- apply removal of i (1 -> 0) ----
            start_i = n - 1 - i
            c[start_i:start_i + i + 1] -= h[: i + 1][::-1]
            c[start_i + i + 1:start_i + n] -= h[i + 1 :]
            h[i] = 0

            # ---- apply addition of j (0 -> 1) ----
            h[j] = 1
            start_j = n - 1 - j
            c[start_j:start_j + j + 1] += h[: j + 1][::-1]
            c[start_j + j + 1:start_j + n] += h[j + 1 :]

            new_worst = target - c.min()
            delta = new_worst - cur_worst
            accept = False
            if delta <= 0:
                accept = True
            else:
                if rng.random() < np.exp(-delta / max(T, 1e-12)):
                    accept = True

            if accept:
                # update bookkeeping
                ones_idx[i_pos] = j
                pos_one[j] = i_pos
                pos_one[i] = -1

                zeros_idx[j_pos] = i
                pos_zero[i] = j_pos
                pos_zero[j] = -1

                cur_worst = new_worst
                if cur_worst < best_worst:
                    best_worst = cur_worst
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # ---- revert addition ----
                c[start_j:start_j + j + 1] -= h[: j + 1][::-1]
                c[start_j + j + 1:start_j + n] -= h[j + 1 :]
                h[j] = 0
                # ---- revert removal ----
                h[i] = 1
                c[start_i:start_i + i + 1] += h[: i + 1][::-1]
                c[start_i + i + 1:start_i + n] += h[i + 1 :]
                no_improve += 1

            T *= decay
            if no_improve > 20000:
                break
        # end of one restart
    final_c = best_worst * (2.0 / n)
    if best_h is None:
        # fall back to a simple deterministic pattern
        best_h = np.zeros(n, dtype=float)
        best_h[:target] = 1.0
        final_c = compute_c5(best_h)
    return best_h.astype(float), float(final_c)


# ------------------------------------------------------------
# Fractional simulated annealing (keeps the total sum exact).
# ------------------------------------------------------------
def fractional_annealing(h: np.ndarray, rng: np.random.Generator,
                         deadline: float) -> tuple[np.ndarray, float]:
    n = h.size
    target = n / 2.0
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = cur_c * 0.5 + 1e-8
    decay = 0.9999
    step = 0.02

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # maximum admissible delta for this pair
        delta = step
        delta = min(delta, 1.0 - h[i])
        delta = min(delta, h[j])
        if delta <= 1e-12:
            continue

        h[i] += delta
        h[j] -= delta
        new_c = compute_c5(h)
        d_c = new_c - cur_c
        accept = False
        if d_c <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-d_c / max(T, 1e-12)):
                accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta

        T *= decay
        if T < 1e-6:
            step *= 0.5
            T = cur_c * 0.5 + 1e-8
    return best_h, best_c


# ------------------------------------------------------------
# Main entry point.
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = float('inf')
    best_n = None

    # --------------------------------------------------------
    # Phase 1: binary hill climbing on the largest feasible grid.
    # --------------------------------------------------------
    max_n = 720
    binary_deadline = start_time + 0.8 * budget_s
    if time.time() < binary_deadline:
        h_bin, c_bin = binary_hill_climb(max_n, rng, binary_deadline)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin, c_bin, max_n

    # --------------------------------------------------------
    # Phase 2: fractional annealing / gradient polishing.
    # --------------------------------------------------------
    if best_h is not None and time.time() < deadline:
        # gradient polish first
        best_h = gradient_refine(best_h, rng, step=0.02, max_iter=8)
        # then fractional SA on the remaining time
        h_frac, c_frac = fractional_annealing(best_h, rng, deadline)
        if c_frac < best_c:
            best_h, best_c = h_frac, c_frac
            best_n = max_n
        # final gradient polish
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=12)
        best_c = compute_c5(best_h)

    # --------------------------------------------------------
    # Phase 3: fallback to a smaller grid if needed.
    # --------------------------------------------------------
    if best_h is None or time.time() < deadline:
        n_small = 360
        if time.time() < deadline:
            h_bin_s, c_bin_s = binary_hill_climb(n_small, rng, deadline)
            if c_bin_s < best_c:
                best_h, best_c, best_n = h_bin_s, c_bin_s, n_small
                best_h = gradient_refine(best_h, rng, step=0.02, max_iter=8)
                best_c = compute_c5(best_h)

    # --------------------------------------------------------
    # Phase 4: use the provided parent construction if it exists.
    # --------------------------------------------------------
    if best_h is not None and 'initial_h_values' in globals() and time.time() < deadline:
        parent = np.array(initial_h_values, dtype=float)
        n_parent = parent.size
        xs = np.linspace(0.0, 2.0, n_parent, endpoint=False)
        xt = np.linspace(0.0, 2.0, max_n, endpoint=False)
        h_up = np.interp(xt, xs, parent)
        # scale to exact sum
        target_up = max_n / 2.0
        h_up *= target_up / h_up.sum()
        np.clip(h_up, 0.0, 1.0, out=h_up)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=8)
        h_up, c_up = fractional_annealing(h_up, rng, deadline)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, max_n
            best_h = gradient_refine(best_h, rng, step=0.01, max_iter=12)
            best_c = compute_c5(best_h)

    # --------------------------------------------------------
    # Final safety fixes.
    # --------------------------------------------------------
    if best_h is None:
        # deterministic fallback
        best_n = 360
        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact integral (sum = n/2)
    target_sum = best_n / 2.0
    diff = target_sum - best_h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, best_n)
        add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
        best_h[i] += add
        diff = target_sum - best_h.sum()
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
