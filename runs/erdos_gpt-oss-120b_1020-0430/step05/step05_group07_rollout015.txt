def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97
    n = 720
    block_count = 20
    block_sz = n // block_count
    # Ensure n divisible by block_count; 720/20=36 ok.

    best_h = None
    best_c = np.inf

    # Random block samples
    max_samples = 20000  # or time-limited
    for _ in range(max_samples):
        if time.time() > deadline:
            break
        # Dirichlet via Gamma
        probs = rng.gamma(1.0, size=block_count)
        d = probs / probs.sum() * (block_count / 2.0)
        if (d > 1).any():
            continue
        h = np.repeat(d, block_sz)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Local block hill climbing
    if best_h is not None:
        d = best_h.reshape(block_count, block_sz).mean(axis=1)  # recover densities
        # Zip to ensure sum exactly correct (should be)
        # Local search loop
        while time.time() < deadline:
            # choose two distinct blocks
            i, j = rng.choice(block_count, size=2, replace=False)
            # max transferable amount from i to j without violating bounds
            max_transfer = min(d[i], 1.0 - d[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            d[i] -= delta
            d[j] += delta
            h_new = np.repeat(d, block_sz)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                # optionally keep d
            else:
                # reject
                d[i] += delta
                d[j] -= delta
        # after block hill climbing, finalize
    # Further refinement with gradient and SA
    if best_h is not None and time.time() < deadline:
        # gradient refinement
        best_h = gradient_refine(best_h, rng, step=0.03, max_iter=20)
        best_c = compute_c5(best_h)
        # fractional SA
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
        best_h, best_c = fractional_sa_hill_climb(best_h, rng, sub_dead)
        # final gradient
        best_h = gradient_refine(best_h, rng, step=0.02, max_iter=10)
        best_c = compute_c5(best_h)
    # fallback if none
    if best_h is None:
        # simple binary half-half
        n = 720
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)
        n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure sum constraint exactly
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        # adjust using random redistribution
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target_sum - best_h.sum()
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)

import time, numpy as np
from scipy.optimize import minimize

BLOCK_SIZE = 0
TOTAL_DENSITY = 0.0

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=10):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def fractional_sa_hill_climb(h, rng, deadline):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 12000:
            break
    return best, best_c

def penalized_obj_5(x):
    # x length 4, using global BLOCK_SIZE and TOTAL_DENSITY
    d_last = TOTAL_DENSITY - np.sum(x)
    d = np.concatenate([x, [d_last]])
    if (d < 0).any() or (d > 1).any():
        # large penalty
        return 10.0 + np.sum(np.maximum(-d, 0.0)) + np.sum(np.maximum(d - 1.0, 0.0))
    h = np.repeat(d, BLOCK_SIZE)
    return compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97

    n = 720
    block_count = 5
    block_sz = n // block_count
    global BLOCK_SIZE, TOTAL_DENSITY
    BLOCK_SIZE = block_sz
    TOTAL_DENSITY = block_count / 2.0  # 2.5

    best_h = None
    best_c = np.inf

    # low‑dimensional Nelder‑Mead on 5‑block densities
    max_restarts = 30
    for _ in range(max_restarts):
        if time.time() > deadline:
            break
        # sample initial 4 variables respecting bound for the 5th
        while True:
            x0 = rng.random(4)  # uniform in [0,1)
            d_last = TOTAL_DENSITY - x0.sum()
            if 0.0 <= d_last <= 1.0:
                break
        # minimise
        res = minimize(penalized_obj_5, x0, method='Nelder-Mead',
                       options={'maxiter': 300, 'xatol':1e-8, 'fatol':1e-8, 'disp':False})
        if not res.success:
            # still evaluate result
            pass
        d_opt = np.concatenate([res.x, [TOTAL_DENSITY - np.sum(res.x)]])
        d_opt = np.clip(d_opt, 0.0, 1.0)
        h = np.repeat(d_opt, block_sz)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # if still none (unlikely), fall back to random block search
    if best_h is None:
        # simple binary block pattern
        d = np.zeros(block_count)
        d[:block_count // 2] = 1.0
        h = np.repeat(d, block_sz)
        best_h = h
        best_c = compute_c5(h)

    # local refinements on the full resolution vector
    # gradient steps
    best_h = gradient_refine(best_h, rng, step=0.03, max_iter=20)
    best_c = compute_c5(best_h)

    # fractional simulated annealing limited by remaining time
    if time.time() < deadline:
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
        best_h, best_c = fractional_sa_hill_climb(best_h, rng, sub_dead)

    # final short gradient pass
    best_h = gradient_refine(best_h, rng, step=0.02, max_iter=10)
    best_c = compute_c5(best_h)

    # enforce constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)

import time
import numpy as np
from scipy.optimize import minimize

# globals for the low‑dimensional optimizer
BLOCK_SIZE = 0          # points per block (set in run)
TOTAL_DENSITY = 0.0     # sum of block densities (set in run)

def compute_c5(h):
    """Maximum overlap for a step function sampled on an equispaced grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass so that Σh = n/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=10):
    """Projected gradient descent against the currently worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def fractional_sa_hill_climb(h, rng, deadline):
    """Continuous simulated annealing on the full‑resolution vector."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 12000:
            break
    return best, best_c

def penalized_obj_5(x):
    """Objective for the 5‑block representation (Nelder‑Mead)."""
    d_last = TOTAL_DENSITY - np.sum(x)
    d = np.concatenate([x, [d_last]])
    if (d < 0).any() or (d > 1).any():
        return 10.0 + np.sum(np.maximum(-d, 0.0)) + np.sum(np.maximum(d - 1.0, 0.0))
    h = np.repeat(d, BLOCK_SIZE)
    return compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97

    # high‑resolution grid
    n = 720
    block_count = 5
    block_sz = n // block_count
    global BLOCK_SIZE, TOTAL_DENSITY
    BLOCK_SIZE = block_sz
    TOTAL_DENSITY = block_count / 2.0        # = 2.5

    best_h = None
    best_c = np.inf

    # ------- low‑dimensional Nelder‑Mead on 5 blocks -------
    max_restarts = 30
    for _ in range(max_restarts):
        if time.time() > deadline:
            break
        # initialise 4 free densities, the 5th follows from the sum constraint
        while True:
            x0 = rng.random(4)
            d5 = TOTAL_DENSITY - x0.sum()
            if 0.0 <= d5 <= 1.0:
                break
        res = minimize(penalized_obj_5, x0, method="Nelder-Mead",
                       options={"maxiter": 300, "xatol": 1e-8,
                                "fatol": 1e-8, "disp": False})
        d_opt = np.concatenate([res.x, [TOTAL_DENSITY - np.sum(res.x)]])
        d_opt = np.clip(d_opt, 0.0, 1.0)
        h = np.repeat(d_opt, block_sz)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # fallback – simple binary 5‑block pattern
    if best_h is None:

        d[: block_count // 2] = 1.0
        best_h = np.repeat(d, block_sz)
        best_c = compute_c5(best_h)

    # ------- fine‑grained refinements on the full vector -------
    best_h = gradient_refine(best_h, rng, step=0.03, max_iter=20)
    best_c = compute_c5(best_h)

    if time.time() < deadline:

        sub_dead = min(deadline,
                       time.time() + (deadline - time.time()) * 0.4)
        best_h, best_c = fractional_sa_hill_climb(best_h, rng, sub_dead)

    best_h = gradient_refine(best_h, rng, step=0.02, max_iter=10)
    best_c = compute_c5(best_h)

    # enforce the exact integral constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
