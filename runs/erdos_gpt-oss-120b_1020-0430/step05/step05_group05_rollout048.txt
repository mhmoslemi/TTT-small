def compute_corr_fft(h, dx):
    n = h.size
    n2 = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=n2)
    K = np.fft.rfft(1.0 - h, n=n2)
    conv = np.fft.irfft(H * np.conj(K), n=n2)[:2*n-1]
    return conv * dx

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

import time
import numpy as np

# ---------- helpers ---------------------------------------------------------

def compute_corr_fft(h, dx):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.size
    n2 = 1 << ((2 * n - 1).bit_length())        # next power of two
    H = np.fft.rfft(h, n=n2)
    K = np.fft.rfft(1.0 - h, n=n2)
    conv = np.fft.irfft(H * np.conj(K), n=n2)[: 2 * n - 1]
    return conv * dx


def compute_c5(h):
    """C5 = max_k ∫ h (1-h shifted) ."""
    return np.max(compute_corr_fft(h, 2.0 / h.size))


def adjust_sum_exact(h, rng, max_iter=2000):
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def fractional_to_binary(h, rng):
    """Binary vector with exactly n/2 ones – take the largest entries."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    hb = np.zeros_like(h)
    hb[idx] = 1.0
    adjust_sum_exact(hb, rng)
    return hb


def binary_targeted_improve(h, rng, deadline):
    """Greedy swaps that lower the current worst overlap."""
    n = h.size
    while time.time() < deadline:
        corr = compute_corr_fft(h, 2.0 / n)
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift == 0:
            break
        k = abs(shift)
        ar = np.arange(n)
        neigh = h[(ar + k) % n] + h[(ar - k) % n]

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        i = ones[np.argmin(neigh[ones])]
        j = zeros[np.argmax(neigh[zeros])]

        # try the deterministic swap
        h[i], h[j] = 0.0, 1.0
        new_c = np.max(compute_corr_fft(h, 2.0 / n))
        if new_c < np.max(corr):
            continue  # keep swap

        # revert
        h[i], h[j] = 1.0, 0.0

        # a few random attempts
        improved = False
        for _ in range(5):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            h[i2], h[j2] = 0.0, 1.0
            new_c2 = np.max(compute_corr_fft(h, 2.0 / n))
            if new_c2 < np.max(corr):
                improved = True
                break
            h[i2], h[j2] = 1.0, 0.0
        if not improved:
            break
    return h


def gradient_descent_smooth_max(h, rng, deadline):
    """Projected gradient descent on the smooth‑max surrogate."""
    n = h.size
    dx = 2.0 / n
    target = n / 2.0

    # beta schedule (smooth -> sharp)
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
             50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
    for beta in betas:
        # iterations per beta – more for larger beta
        if beta < 20:
            iters = 10
        elif beta < 200:
            iters = 20
        else:
            iters = 30
        for _ in range(iters):
            if time.time() >= deadline:
                return h
            corr = compute_corr_fft(h, dx)
            m = corr.max()
            w = np.exp(beta * (corr - m))
            w_sum = w.sum()
            if w_sum == 0:
                continue
            w /= w_sum

            G = np.zeros_like(h)
            # vectorised loop over shifts
            for s in range(-n + 1, n):
                ws = w[s + n - 1]
                if ws == 0.0:
                    continue
                if s >= 0:
                    G[: n - s] += ws * (1.0 - h[s:])
                    G[s:] -= ws * h[: n - s]
                else:
                    t = -s
                    G[t:] += ws * (1.0 - h[: n - t])
                    G[: n - t] -= ws * h[t:]

            # simple step‑size schedule
            step = 0.05 / (1.0 + 0.1 * beta)
            h -= step * G

            # clip & exact sum correction
            np.clip(h, 0.0, 1.0, out=h)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                if diff > 0:
                    caps = 1.0 - h
                else:
                    caps = -h
                cap_sum = caps.sum()
                if cap_sum > 0:
                    h += diff * caps / cap_sum
                else:
                    adjust_sum_exact(h, rng)
                np.clip(h, 0.0, 1.0, out=h)
    return h


# ---------- main entry -----------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # base size
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
    else:
        parent = None
        n0 = 360

    best_h = None
    best_c = np.inf
    best_n = n0

    # candidate sizes (base and optionally double)
    candidates = [n0]
    if n0 * 2 <= 720:
        candidates.append(n0 * 2)

    for n in candidates:
        if time.time() >= deadline:
            break
        dx = 2.0 / n

        starts = []

        # parent (interpolated if needed)
        if parent is not None:
            if n == n0:
                starts.append(parent.copy())
            else:
                xs = np.linspace(0.0, 2.0, n0, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                up = np.interp(xt, xs, parent)
                adjust_sum_exact(up, rng)
                starts.append(up)

        # random binary seeds
        for _ in range(3):
            hb = np.zeros(n)
            hb[rng.choice(n, n // 2, replace=False)] = 1.0
            starts.append(hb)

        # random fractional seeds
        for _ in range(3):
            hf = rng.random(n)
            hf *= (n / 2) / hf.sum()
            np.clip(hf, 0.0, 1.0, out=hf)
            adjust_sum_exact(hf, rng)
            starts.append(hf)

        # heavy‑tailed fractional seeds
        for _ in range(2):
            hf = rng.beta(0.5, 0.5, size=n)
            hf *= (n / 2) / hf.sum()
            np.clip(hf, 0.0, 1.0, out=hf)
            adjust_sum_exact(hf, rng)
            starts.append(hf)

        for h0 in starts:
            if time.time() >= deadline:
                break
            h = h0.copy()
            adjust_sum_exact(h, rng)

            # smooth‑max gradient descent
            h = gradient_descent_smooth_max(h, rng, deadline)

            # evaluate fractional solution
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # binary rounding + greedy improvement
            hb = fractional_to_binary(h, rng)
            c_b = compute_c5(hb)
            if c_b < best_c:
                best_h, best_c, best_n = hb.copy(), c_b, n

            hb = binary_targeted_improve(hb, rng, deadline)
            c_b2 = compute_c5(hb)
            if c_b2 < best_c:
                best_h, best_c, best_n = hb.copy(), c_b2, n

    # fallback in the unlikely case nothing was found
    if best_h is None:
        best_n = 360
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final clean‑up
    np.clip(best_h, 0.0, 1.0, out=best_h)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
