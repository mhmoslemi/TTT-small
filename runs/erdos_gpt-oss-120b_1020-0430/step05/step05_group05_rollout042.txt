def anneal_binary(h, rng, deadline):
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # temperature schedule: start T0 = cur_c * 0.05? Or maybe 0.01
    T0 = cur_c * 0.02 + 1e-6
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # update temperature linearly
        t = (time.time() - start) / max(deadline - start, 1e-9)
        T = T0 * (1 - t)
        # pick random 1 and 0
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best_h, best_c

        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            failed = 0
        else:
            # revert
            h[i] = 1
            h[j] = 0
            failed += 1
    return h, cur_c

Thus our B part method also double counts same term when (a+d == b) or (b+d == a) i.e., when d = b-a or d = a-b.

Similarly for a-d = b or b-d = a.

Removing double count: For d where a+d == b (i.e., d = b - a), the term was included in both contributions: a's i part with subtract h[a+d] (h[b]) is zero subtract; b's i-d part adds h[b-d] (h[a]) = 1. However the actual change due to swapping a and b in this term is: originally: h[a]*h[b] = 1*0=0; after swap: 0*1=0; delta =0. So we must subtract the extra added 1. Similarly for d where a-d == b (i.e., d = a - b), the term appears in a's i-d subtract h[a-d] = h[b] = 0 (no effect) and b's i add h[b+d] = h[b+(a-b)] = h[a] = 1 => extra +1, which is also spurious.

Thus we need to handle these overlapping d's by adjusting delta appropriately.

Specifically, for each swap (a,b), there are at most two overlapping shifts: d1 = b - a, and d2 = a - b = -d1. For each direction, we should correct delta: subtract the extra added contribution due to double counting.

Let's analyze contributions for both swap parts thoroughly.

General B delta contributions (without correcting overlaps) is:

delta_B(d) = -h[a+d] * I_{-a <= d <= n-1-a}    (a i part)
           -h[a-d] * I_{a-(n-1) <= d <= a}   (a i-d part)
           +h[b+d] * I_{-b <= d <= n-1-b}    (b i part)
           +h[b-d] * I_{b-(n-1) <= d <= b}   (b i-d part)

But note that for d where both a+d and b-d refer to the same index, we might double count. Let's examine overlapping scenarios:

- a+d refers to b when d = b - a => a+d = b. In that case, also b-d refers to a (since b - (b-a) = a). So both terms appear: -h[a+d] = -h[b] and +h[b-d] = +h[a]. Since h[b]=0 and h[a]=1, the net is +1. That's the spurious extra for that shift.

- Similarly, a-d refers to b when d = a - b => a-d = b. In that case, also b+d refers to a (b + (a-b) = a). Then we get -h[a-d] = -h[b] (0) and +h[b+d] = +h[a] (1), net +1 again.

Thus for d = b-a, delta_B(d) has extra +1, for d = a-b, delta_B(d) has extra +1. In both cases, actual delta_B should be 0 (since term originally zero and after swap zero). So we need to subtract 1 from delta_B at those two d's.

Thus correction delta_B(d) -= 1 for d = b-a and for d = a-b, if those d are within the valid range (i.e., in - (n-1) ... (n-1)). Since we compute delta for all d; we can simply subtract 1 for both.

But careful: If a and b distances lead to same d? Only if a==b, not allowed; else d1 = b-a, d2 = -d1.

Thus correction:

if - (n-1) <= b - a <= n-1:
    delta[ offset + (b - a) ] -= 1
if - (n-1) <= a - b <= n-1:
    delta[ offset + (a - b) ] -= 1

However note that these deltas in B part are added to delta_S via delta_A - delta_B. So we need to adjust delta_S accordingly: we need delta_S(d) = delta_A(d) - (delta_B(d) - correction?). Actually we derived delta_S = delta_A - delta_B, where delta_B is what we computed (including double counting). To correct double counting, we set delta_B_corrected = delta_B - overlap_correction, where overlap_correction = 1 for d=b-a and d=a-b. Thus delta_S = delta_A - delta_B_corrected = delta_A - (delta_B - overlap_correction) = (delta_A - delta_B) + overlap_correction.

Thus we can equivalently compute delta = delta_A - delta_B, then add +1 at d=b-a and d=a-b (if within range). Because overlap correction adds +1 to delta.

def hill_climb_inc(h, rng, deadline, max_iter=None):
    n = h.size
    offset = n - 1
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    # we can track current max value
    # Stop condition: either time out or enough iterations without improvement
    failed = 0
    max_failed = 5000 # parameter
    while time.time() < deadline and (max_iter is None or max_iter > 0):
        # choose random swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = rng.choice(ones)   # will become 0
        b = rng.choice(zeros)  # will become 1
        # compute delta quickly via slices (we need to copy slices of h but we can use h array)
        # allocate delta
        delta = np.zeros(2*n - 1, dtype=int)
        # delta_A
        delta[offset - a : offset + (n - a)] -= 1
        delta[offset - b : offset + (n - b)] += 1
        # add h[a+d] => add h
        delta[offset - a : offset + (n - a)] += h
        # add h[a-d] => add reversed h
        delta[offset + a - (n - 1) : offset + a + 1] += h[::-1]
        # subtract h[b+d]
        delta[offset - b : offset + (n - b)] -= h
        # subtract h[b-d]
        delta[offset + b - (n - 1) : offset + b + 1] -= h[::-1]
        # overlap correction
        d1 = b - a
        if - (n - 1) <= d1 <= n - 1:
            delta[offset + d1] += 1
        d2 = a - b
        if - (n - 1) <= d2 <= n - 1:
            delta[offset + d2] += 1
        # apply delta
        new_corr = corr + delta
        new_max = new_corr.max()
        # decide acceptance
        if new_max <= cur_max:
            # accept
            corr = new_corr
            cur_max = new_max
            # swap values in h
            h[a] = 0
            h[b] = 1
            failed = 0
        else:
            # reject
            failed += 1
        # iteration limit
        if max_iter is not None:
            max_iter -= 1
        # early exit
        if failed >= max_failed:
            break
    return h, float(cur_max * (2.0 / n))

def hill_climb_inc(h, rng, deadline, max_failed=5000, max_iter=None):
    n = h.size
    offset = n - 1
    # compute initial correlation using np.correlate (int)
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    failed = 0
    delta = np.zeros(2*n - 1, dtype=np.int32)
    while time.time() < deadline and (max_iter is None or max_iter > 0):
        # choose random swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = rng.choice(ones)   # will become 0
        b = rng.choice(zeros)  # will become 1
        # compute delta
        delta.fill(0)
        # delta_A
        delta[offset - a : offset + (n - a)] -= 1
        delta[offset - b : offset + (n - b)] += 1
        # add h[a+d]: +h slice
        delta[offset - a : offset + (n - a)] += h
        # add h[a-d]: +h_rev slice
        delta[offset + a - (n - 1) : offset + a + 1] += h[::-1]
        # subtract h[b+d]: -h slice
        delta[offset - b : offset + (n - b)] -= h
        # subtract h[b-d]: -h_rev slice
        delta[offset + b - (n - 1) : offset + b + 1] -= h[::-1]
        # correct for double count at d=0: subtract 2
        delta[offset] -= 2
        # correct overlapping when b-d = a (i.e., d = b - a)
        d_overlap = b - a
        if - (n - 1) <= d_overlap <= n - 1:
            delta[offset + d_overlap] += 1
        # apply delta
        new_corr = corr + delta
        new_max = new_corr.max()
        if new_max <= cur_max:
            corr = new_corr
            cur_max = new_max
            # perform swap
            h[a] = 0
            h[b] = 1
            failed = 0
        else:
            failed += 1
        if max_iter is not None:
            max_iter -= 1
        if failed >= max_failed:
            break
    final_c = cur_max * (2.0 / n)
    return h, final_c

def simulated_anneal(h, rng, deadline, start_temp_factor=0.02):
    n = h.size
    offset = n - 1
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    cur_c = cur_max * (2.0 / n)
    start_c = cur_c
    T0 = start_c * start_temp_factor + 1e-12
    start_t = time.time()
    while time.time() < deadline:
        t_frac = (time.time() - start_t) / max(deadline - start_t, 1e-9)
        T = T0 * max(0.0, 1 - t_frac)
        # pick swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones)==0 or len(zeros)==0:
            break
        a = rng.choice(ones)
        b = rng.choice(zeros)
        # compute delta as before (use delta array)
        delta = np.zeros(2*n - 1, dtype=np.int32)
        # delta_A
        delta[offset - a : offset + (n - a)] -= 1
        delta[offset - b : offset + (n - b)] += 1
        # plus/minus slices
        delta[offset - a : offset + (n - a)] += h
        delta[offset + a - (n - 1) : offset + a + 1] += h[::-1]
        delta[offset - b : offset + (n - b)] -= h
        delta[offset + b - (n - 1) : offset + b + 1] -= h[::-1]
        delta[offset] -= 2
        d_overlap = b - a
        if - (n - 1) <= d_overlap <= n - 1:
            delta[offset + d_overlap] += 1
        new_corr = corr + delta
        new_max = new_corr.max()
        if new_max <= cur_max:
            corr = new_corr
            cur_max = new_max
            h[a] = 0
            h[b] = 1
            cur_c = cur_max * (2.0 / n)
        else:
            delta_c = (new_max - cur_max) * (2.0 / n)
            if T > 0 and rng.random() < np.exp(-delta_c / T):

                cur_max = new_max
                h[a] = 0
                h[b] = 1
                cur_c = cur_max * (2.0 / n)
        # optional early stop if cur_c already less than target maybe
    return h, cur_c

    delta = np.zeros(2 * n - 1, dtype=np.int32)
    while time.time() < deadline:
        t = (time.time() - start_t) / max(deadline - start_t, 1e-12)
        T = T0 * max(0.0, 1.0 - t)
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if not ones.size or not zeros.size:
            break
        a = rng.choice(ones)
        b = rng.choice(zeros)
        delta.fill(0)
        delta[off - a : off + (n - a)] -= 1
        delta[off - b : off + (n - b)] += 1
        delta[off - a : off + (n - a)] += h
        delta[off + a - (n - 1) : off + a + 1] += h[::-1]
        delta[off - b : off + (n - b)] -= h
        delta[off + b - (n - 1) : off + b + 1] -= h[::-1]
        delta[off] -= 2
        d_ov = b - a
        if - (n - 1) <= d_ov <= n - 1:
            delta[off + d_ov] += 1
        new_corr = corr + delta
        new_max = new_corr.max()
        if new_max <= cur_max:
            corr = new_corr
            cur_max = new_max
            h[a] = 0
            h[b] = 1
            cur_c = cur_max * (2.0 / n)
        else:
            dC = (new_max - cur_max) * (2.0 / n)
            if T > 0 and rng.random() < np.exp(-dC / T):
                corr = new_corr
                cur_max = new_max
                h[a] = 0
                h[b] = 1
                cur_c = cur_max * (2.0 / n)
    return h, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # base size
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=np.float64)
        base_n = parent.size
    else:
        parent = None
        base_n = 360

    best_h, best_c, best_n = None, np.inf, None

    # ---- Phase 1: base resolution ----
    trials_base = 40
    for t in range(trials_base):
        if time.time() >= deadline:
            break
        if t == 0 and parent is not None:
            hb = np.zeros(base_n, dtype=np.int8)
            idx = np.argpartition(parent, -base_n // 2)[-base_n // 2 :]
            hb[idx] = 1
        else:
            hb = np.zeros(base_n, dtype=np.int8)
            hb[rng.choice(base_n, base_n // 2, replace=False)] = 1
        per_dead = time.time() + min(deadline - time.time(), 3.0)
        h_opt, c_opt = hill_climb_inc(hb.copy(), rng, per_dead)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, base_n

    # ---- Phase 2: upscale if time ----
    target_n = min(720, base_n * 2)
    if target_n > base_n and time.time() < deadline:
        xs = np.linspace(0.0, 2.0, base_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, target_n, endpoint=False)
        base_for_interp = best_h if best_h is not None else np.zeros(base_n, dtype=np.int8)
        interp = np.interp(xt, xs, base_for_interp)
        hb_up = np.zeros(target_n, dtype=np.int8)
        hb_up[np.argpartition(interp, -target_n // 2)[-target_n // 2 :]] = 1

        trials_up = 70
        for t in range(trials_up):
            if time.time() >= deadline:
                break
            if t == 0:
                hb = hb_up.copy()
            else:
                hb = np.zeros(target_n, dtype=np.int8)
                hb[rng.choice(target_n, target_n // 2, replace=False)] = 1
            per_dead = time.time() + min(deadline - time.time(), 3.0)
            h_opt, c_opt = hill_climb_inc(hb.copy(), rng, per_dead)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, target_n

    # ---- Final annealing refinement ----
    if best_h is not None and time.time() < deadline:
        h_final, c_final = simulated_anneal(best_h.copy(), rng, deadline)
        if c_final < best_c:
            best_h, best_c, best_n = h_final.copy(), c_final, best_h.size

    if best_h is None:
        best_n = base_n
        best_h = np.zeros(best_n, dtype=np.int8)
        best_h[: best_n // 2] = 1
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0, 1).astype(int)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
