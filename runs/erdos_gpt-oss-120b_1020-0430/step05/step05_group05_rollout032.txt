def worst_shift_grad_full(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)  # could be negative.
    # compute g = 1 - h_{i+shift} - h_{i-shift}
    # using roll
    h_plus = np.roll(h, -shift)
    h_minus = np.roll(h, shift)
    g = 1.0 - h_plus - h_minus
    return g, shift
```

def gradient_descent_half(x0, rng, time_deadline):

    iter = 0
    while time.time() < time_deadline:
        # compute gradient
        g = worst_shift_grad_half(x)
        # step size

        # update
        x = x - step * g
        # project
        x = np.clip(x, 0.0, 1.0)
        # occasional random perturb
        if iter % 200 == 0 and iter > 0:
            eps = 0.05 / np.sqrt(iter // 200 + 1)

        iter += 1
        # maybe break if iter > max_iter

            break
    return x

def sym_full_from_half(x):
    rev = 1 - x[::-1]
    return np.concatenate([x, rev])

def compute_c5_full(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def worst_shift_gradient_full(h):
    n = h.size
    # compute overlap O_k for each shift
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    # gradient = -(h[i+shift] + h[i-shift])  # ignoring constant term
    h_plus = np.roll(h, -shift)
    h_minus = np.roll(h, shift)
    return -(h_plus + h_minus)

def worst_shift_gradient_half(x):
    h = sym_full_from_half(x)
    g_full = worst_shift_gradient_full(h)
    n = h.size
    m = x.shape[0]
    # reverse first m entries of g_full
    rev_g = g_full[::-1][:m]
    g_half = g_full[:m] - rev_g
    return g_half

def gradient_descent_half(x0, rng, deadline):
    x = x0.copy()
    it = 0
    while time.time() < deadline:
        g = worst_shift_gradient_half(x)
        step = 0.3 / np.sqrt(it + 1)
        x -= step * g
        x = np.clip(x, 0.0, 1.0)

            x = np.clip(x, 0.0, 1.0)
        it += 1
        if it >= 8000:  # safety cap
            break
    return x

def binary_from_half(x):

def binary_targeted_improve(h, rng, deadline):
    n = h.size
    ar = np.arange(n)
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        shift = abs(shift)
        if shift == 0:
            break
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        # pick i among ones with minimal neigh
        i_vals = neigh[ones]
        i = ones[np.argmin(i_vals)]
        # pick j among zeros with maximal neigh
        j_vals = neigh[zeros]
        j = zeros[np.argmax(j_vals)]
        delta = neigh[i] - neigh[j]  # negative improves
        if delta < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        # try random swaps
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

import time
import numpy as np

TARGET_C5 = 0.3808  # early stop target


def sym_full_from_half(x):
    """Build full symmetric h from half vector x."""
    rev = 1.0 - x[::-1]
    return np.concatenate([x, rev])


def compute_c5_full(h):
    """Maximum overlap integral for full h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def worst_shift_gradient_full(h):
    """Gradient of the worst overlap w.r.t. h (full vector)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)  # may be negative
    # gradient dO/dh_i = -(h_{i+shift} + h_{i-shift})
    h_plus = np.roll(h, -shift)
    h_minus = np.roll(h, shift)
    return -(h_plus + h_minus)


def worst_shift_gradient_half(x):
    """Gradient of the worst overlap w.r.t. half vector x."""
    h = sym_full_from_half(x)
    g_full = worst_shift_gradient_full(h)
    m = x.shape[0]
    rev_g = g_full[::-1][:m]          # g_full[n-1-i] for i=0..m-1
    return g_full[:m] - rev_g


def gradient_descent_half(x0, rng, deadline, max_iter=20000):
    """Projected sub‑gradient descent on the worst shift."""
    x = x0.copy()
    it = 0
    while time.time() < deadline and it < max_iter:
        g = worst_shift_gradient_half(x)
        step = 0.30 / np.sqrt(it + 1)      # decreasing step
        x -= step * g
        x = np.clip(x, 0.0, 1.0)
        if it % 200 == 0 and it > 0:
            eps = 0.05 / np.sqrt(it // 200 + 1)
            x += rng.uniform(-eps, eps, size=x.shape)
            x = np.clip(x, 0.0, 1.0)
        it += 1
    return x


def binary_from_half(x):
    """Simple threshold to obtain a binary half vector."""
    return (x >= 0.5).astype(float)


def binary_targeted_improve(h, rng, deadline):
    """Greedy swaps on a binary full vector that lower the worst overlap."""
    n = h.size
    ar = np.arange(n)
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        shift = abs(shift)
        if shift == 0:
            break
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(neigh[ones])]
        j = zeros[np.argmax(neigh[zeros])]
        delta = neigh[i] - neigh[j]            # <0 improves the worst shift
        if delta < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        # try a few random alternatives
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # small safety margin

    # ----- base size -----
    if "initial_h_values" in globals():
        parent_full = np.array(initial_h_values, dtype=float)
        n = parent_full.size
        if n % 2 == 1:            # enforce even length
            n -= 1
            parent_full = parent_full[:n]
    else:
        parent_full = None
        n = 360
    m = n // 2

    best_h = None
    best_c = np.inf
    best_n = n

    # ----- generate half‑vector seeds -----
    seeds = []
    if parent_full is not None:
        # symmetrize parent to obtain a half seed
        half_parent = (parent_full[:m] + (1.0 - parent_full[-1:-m-1:-1])) * 0.5
        seeds.append(half_parent)
        seeds.append(binary_from_half(half_parent))

    for _ in range(5):
        seeds.append(rng.random(m))
    for _ in range(2):
        seeds.append(rng.beta(0.5, 0.5, size=m))

    # ----- main optimisation on half vectors -----
    for seed_half in seeds:
        if time.time() >= deadline or best_c <= TARGET_C5:
            break
        x0 = np.clip(seed_half, 0.0, 1.0)
        sub_deadline = deadline
        x_opt = gradient_descent_half(x0, rng, sub_deadline)
        h_opt = sym_full_from_half(x_opt)
        c_opt = compute_c5_full(h_opt)
        if c_opt < best_c:
            best_c, best_h, best_n = c_opt, h_opt.copy(), n

        # binary rounding + greedy improvement
        x_bin = binary_from_half(x_opt)
        h_bin = sym_full_from_half(x_bin)
        # allocate a modest slice of remaining time for the greedy phase
        greedy_deadline = min(deadline, time.time() + 8.0)
        h_bin = binary_targeted_improve(h_bin, rng, greedy_deadline)
        c_bin = compute_c5_full(h_bin)
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n

    # ----- optional up‑sampling refinement -----
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 720
        and best_c > TARGET_C5
    ):
        n2 = best_n * 2
        m2 = n2 // 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        # force symmetry at the finer level
        half_up = (h_up[:m2] + (1.0 - h_up[-1:-m2-1:-1])) * 0.5
        sub_deadline = deadline
        x_up_opt = gradient_descent_half(half_up, rng, sub_deadline, max_iter=15000)
        h_up_opt = sym_full_from_half(x_up_opt)
        c_up_opt = compute_c5_full(h_up_opt)
        if c_up_opt < best_c:
            best_c, best_h, best_n = c_up_opt, h_up_opt.copy(), n2

        # binary rounding + greedy on the refined grid
        x_up_bin = binary_from_half(x_up_opt)
        h_up_bin = sym_full_from_half(x_up_bin)
        greedy_deadline = min(deadline, time.time() + 8.0)
        h_up_bin = binary_targeted_improve(h_up_bin, rng, greedy_deadline)
        c_up_bin = compute_c5_full(h_up_bin)
        if c_up_bin < best_c:
            best_c, best_h, best_n = c_up_bin, h_up_bin.copy(), n2

    # ----- fallback (should never trigger) -----
    if best_h is None:
        best_n = n
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c = compute_c5_full(best_h)

    # final touch‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5_full(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
