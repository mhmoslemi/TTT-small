  best_h = None; best_c = np.inf; best_n = None

  # Prepare candidate n list
  candidate_ns = [360, 540, 720, 180]

  for n in candidate_ns:
    if time.time() > deadline: break
    target = n/2.0

    # Build start vectors list
    start_vectors = []

    if parent is not None:
        # interpolate parent to length n
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        start_vectors.append(interp)
        # binary derived
        idx = np.argpartition(interp, -n//2)[-n//2:]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        start_vectors.append(bin_vec)

    # random binary
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n//2, replace=False)] = 1.0
    start_vectors.append(bin_rand)

    # random fractional
    rand_frac = rng.random(n)
    rand_frac *= target / rand_frac.sum()
    adjust_sum_exact(rand_frac, rng)
    start_vectors.append(rand_frac)

    # alternating pattern if n even
    if n % 2 == 0:
        alt_bin = np.tile([1.0,0.0], n//2)
        start_vectors.append(alt_bin)

    for base_h in start_vectors:
        if time.time() > deadline: break
        h = np.clip(base_h, 0.0, 1.0)
        adjust_sum_exact(h, rng)

        # compute and update
        c5 = compute_c5(h)
        if c5 < best_c:
            best_h, best_c, best_n = h.copy(), c5, n

        # gradient refinement
        h = gradient_refine(h, rng, step=0.02, max_iter=6)
        c5 = compute_c5(h)
        if c5 < best_c:
            best_h, best_c, best_n = h.copy(), c5, n

        # SLSQP cascade
        for beta in (5.0,20.0,80.0,200.0,500.0,1000.0,2000.0):
            if time.time() > deadline: break
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=400)
            except Exception:
                continue
            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

        # final gradient refine
        h = gradient_refine(h, rng, step=0.02, max_iter=8)
        c5 = compute_c5(h)
        if c5 < best_c:
            best_h, best_c, best_n = h.copy(), c5, n

  if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
    n_up = best_n * 2
    xs = np.linspace(0.0, 2.0, best_n, endpoint=False)

    adjust_sum_exact(h_up, rng)

    h_up = gradient_refine(h_up, rng, step=0.02, max_iter=6)

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth max via log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def fractional_sa(h, rng, deadline, max_iter=200000, cooling=0.99997):
    """Simulated‑annealing on fractional vectors preserving the sum."""
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = max(cur_c * 0.05, 1e-6)
    n = h.size
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        low = max(-h[i], -(1 - h[j]))
        high = min(1 - h[i], h[j])
        if low >= high:
            continue
        delta = low + rng.random() * (high - low)
        h[i] += delta
        h[j] -= delta
        new_c = compute_c5(h)
        dC = new_c - cur_c
        if dC < 0 or rng.random() < np.exp(-dC / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
        T *= cooling
    return best_h, best_c
