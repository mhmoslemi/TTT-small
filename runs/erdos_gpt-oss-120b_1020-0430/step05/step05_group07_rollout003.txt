import time
import numpy as np

def compute_c5(h):
    """Maximum overlap C5 for step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def gradient_softmax(h, beta):
    """Gradient of softmax‑approximation of max overlap."""
    n = h.size
    sum_h = h.sum()
    # circular autocorrelation via FFT
    fh = np.fft.fft(h)
    autoc = np.fft.ifft(fh * np.conj(fh)).real
    L = sum_h - autoc                     # L_k for all shifts
    # softmax weights (stable)
    bL = beta * L
    m = np.max(bL)
    w = np.exp(bL - m)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    # convolution w * h (w is symmetric, so orientation irrelevant)
    conv = np.fft.ifft(np.fft.fft(w) * np.fft.fft(h)).real
    return 1.0 - 2.0 * conv

def project_sum_bounds(h, target):
    """Project h onto [0,1] and enforce Σh = target (simple iterative)."""
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        h -= diff / h.size
        h = np.clip(h, 0.0, 1.0)
        # a few refinement passes
        for _ in range(5):
            diff = h.sum() - target
            if abs(diff) < 1e-12:
                break
            mask = (h > 0.0) & (h < 1.0)
            cnt = mask.sum()
            if cnt == 0:
                h -= diff / h.size
            else:
                h[mask] -= diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa(h_bin, rng, deadline):
    """Swap‑based simulated annealing for binary vectors."""
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it % 5000 == 0 and time.time() >= deadline:
            break
    return best, best_c
