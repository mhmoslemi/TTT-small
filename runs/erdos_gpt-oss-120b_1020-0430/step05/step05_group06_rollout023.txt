def local_fractional_search(h, current_c, rng, deadline, step=0.01):
    n = h.size
    best_h = h.copy()
    best_c = current_c
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Ensure we can move
        max_delta = min(step, best_h[i], 1.0 - best_h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * rng.random()
        # update
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# Global variables used by the smooth‑max surrogate
CURRENT_TARGET = None   # sum constraint target
CURRENT_BETA = None     # smoothing parameter
CURRENT_REPEAT = 1      # repetition factor for pattern optimisation


def compute_c5(h):
    """Exact C5 evaluation (max overlap) for a discretised step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Exact correction of a vector so that Σh = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


def sum_eq_constraint(x):
    """Equality constraint for the optimiser."""
    return np.sum(x) - CURRENT_TARGET


def smooth_max_obj(x):
    """Smooth‑max surrogate of the max overlap (unscaled)."""
    if CURRENT_REPEAT == 1:
        h = x
    else:
        h = np.tile(x, CURRENT_REPEAT)
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(x0, beta, rng, max_iter=400):
    """SLSQP optimiser for a given β (smooth‑max)."""
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_max_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def greedy_binary(n, rng):
    """Greedy construction of a binary vector with sum n/2."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # correlation of all‑zero vector
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h


def local_fractional_search(h, cur_c, rng, deadline, step=0.01):
    """Simple random pair‑wise mass transfer that preserves the sum."""
    n = h.size
    best_h = h.copy()
    best_c = cur_c
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # amount we can move from i to j without violating bounds
        max_delta = min(step, best_h[i], 1.0 - best_h[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * rng.random()
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

                    best_n = n

    # ---------- full‑vector random SLSQP restarts ----------
    for restart in range(5):
        if time.time() >= deadline:
            break
        h = rng.random(n)
        h = adjust_sum_exact(h, rng)

        CURRENT_TARGET = n / 2.0
        CURRENT_REPEAT = 1

        for beta in betas:
            if time.time() >= deadline:
                break
            if beta < 200.0:
                max_iter = 200
            elif beta < 2000.0:
                max_iter = 400
            else:
                max_iter = 600
            h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            c_val = compute_c5(h)
            if c_val < best_c:
                best_c = c_val
                best_h = h.copy()
                best_n = n

        if time.time() < deadline and best_h is not None:
            h_ref, c_ref = local_fractional_search(best_h, best_c, rng, deadline)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()
                best_n = n

    # ---------- final safety checks ----------
    if best_h is None:
        # fallback deterministic construction
        best_n = n
        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
