def compute_c5_fft(h):
    n = h.shape[0]
    L = 2 * n
    # rfft returns L//2+1 complex values
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)

def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # water-filling loop
    # cast to float
    while abs(diff) > 1e-12:
        if diff > 0:
            free = h < 1.0 - 1e-12
            if not np.any(free):
                break
            slack = np.sum(1.0 - h[free])
            if diff <= slack + 1e-12:
                add = diff / np.count_nonzero(free)
                h[free] += add
                diff = target - h.sum()
                break
            else:
                # saturate all free to 1
                h[free] = 1.0
                diff = target - h.sum()
        else:
            free = h > 1e-12
            if not np.any(free):
                break
            slack = np.sum(h[free])
            if -diff <= slack + 1e-12:
                sub = -diff / np.count_nonzero(free)
                h[free] -= sub
                diff = target - h.sum()
                break
            else:
                h[free] = 0.0
                diff = target - h.sum()
    # after loop correct any tiny rounding errors
    # ensure within bounds
    h = np.clip(h, 0.0, 1.0)
    # final diff may be small; adjust by adding to first element
    if abs(target - h.sum()) > 1e-10:
        # adjust first element within bounds
        i0 = 0
        # ensure we can adjust
        delta = target - h.sum()
        if delta > 0:
            h[i0] = min(1.0, h[i0] + delta)
        else:
            h[i0] = max(0.0, h[i0] + delta)
    return h

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 using direct correlation (used for occasional checks)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_fft(h: np.ndarray) -> float:
    """Fast C5 via FFT (main evaluation)."""
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to obtain exact sum = target (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # iterative water‑filling
    while abs(diff) > 1e-12:
        if diff > 0:
            free = h < 1.0 - 1e-12
            if not np.any(free):
                break
            slack = np.sum(1.0 - h[free])
            if diff <= slack + 1e-12:
                add = diff / np.count_nonzero(free)
                h[free] += add
                diff = target - h.sum()
                break
            h[free] = 1.0
            diff = target - h.sum()
        else:
            free = h > 1e-12
            if not np.any(free):
                break
            slack = np.sum(h[free])
            if -diff <= slack + 1e-12:
                sub = -diff / np.count_nonzero(free)
                h[free] -= sub
                diff = target - h.sum()
                break
            h[free] = 0.0
            diff = target - h.sum()
    # Tiny residual fix
    if abs(target - h.sum()) > 1e-10:
        delta = target - h.sum()
        i0 = 0
        if delta > 0:
            h[i0] = min(1.0, h[i0] + delta)
        else:
            h[i0] = max(0.0, h[i0] + delta)
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP optimization for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

    return h

            h_init = best_h.copy()
        h256, f256 = differential_evolution_optimize_with_init(
            256, rng, t256, h_init)
        if f256 < best_f:
            best_h, best_f, best_n = h256, f256, 256

    # ---- up‑sample to full resolution if time permits ----
    if best_n != 512 and (time.time() - start) < total_budget - 10.0:
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, 512, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_to_feasible(h_up, 256.0)
        best_h, best_f, best_n = h_up, compute_c5_fft(h_up), 512

    # ---- refinement phase (SLSQP + gradient + hill‑climb) ----
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0 and best_h is not None:
        h = best_h.copy()
        betas = [20., 80., 200., 800., 3000., 8000., 20000.]
        per_beta = remaining / len(betas)
        for beta in betas:
            if time.time() - start > total_budget * 0.99:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=800)
            f = compute_c5_fft(h)
            if f < best_f:
                best_h, best_f, best_n = h.copy(), f, h.size
        # gradient tidy‑up
        if time.time() - start < total_budget * 0.995:
            h = gradient_refine(h, rng, step=0.015, max_iter=20)
            f = compute_c5_fft(h)
            if f < best_f:
                best_h, best_f, best_n = h.copy(), f, h.size
        # final stochastic hill‑climb
        while time.time() - start < total_budget * 0.999:
            grad = worst_gradient(h)
            nh = guided_perturb(h, grad, rng, scale=0.02)
            nh = project_to_feasible(nh, nh.size / 2.0)
            nf = compute_c5_fft(nh)
            if nf < best_f:
                h, best_h, best_f = nh, nh.copy(), nf

    # ---- fallback simple pattern if nothing found ----
    if best_h is None:
        n = 512
        best_h = np.zeros(n, dtype=float)
        best_h[::2] = 1.0
        best_h = project_to_feasible(best_h, n / 2.0)
        best_f = compute_c5_fft(best_h)
        best_n = n

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, best_h.size / 2.0)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
