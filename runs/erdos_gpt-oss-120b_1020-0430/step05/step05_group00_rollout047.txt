def project_box_sum(v, target):
    # v: np.ndarray
    # target: sum constraint
    # returns projection onto [0,1] and sum=target
    # Solve for lambda such that sum(clip(v - lam,0,1)) = target.
    # lambda is bounded.
    lo = np.min(v) - 1.0  # when subtract lo, all v - lo >=1 => clip=1 => sum=n
    hi = np.max(v)        # when subtract hi, v - hi <=0 => clip=0 => sum=0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # correct small residual
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # adjust free entries
        free = (proj > 0) & (proj < 1)
        if free.any():
            adjust = diff / free.sum()
            proj[free] += adjust
            proj = np.clip(proj, 0.0, 1.0)
        else:
            # all at bounds, adjust smallest deviation
            idx = np.argmin(np.abs(v - (lo + hi) / 2.0))
            proj[idx] = np.clip(proj[idx] + diff, 0.0, 1.0)
    return proj

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto [0,1]^n with sum(v)=target."""
    lo = np.min(v) - 1.0          # sum will be > target
    hi = np.max(v)                # sum will be < target
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            adj = diff / free.sum()
            proj[free] += adj
            proj = np.clip(proj, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - (lo + hi) * 0.5))
            proj[i] = np.clip(proj[i] + diff, 0.0, 1.0)
    return proj


# ---------- SLSQP with soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP refinement for a fixed β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # resolution
    n = 512
    target = n / 2.0

    # ---- initial solution ----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        if h.size != n:
            # down‑sample / interpolate to n
            x_old = np.linspace(0.0, 2.0, h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, h)
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target)
    else:
        h = rng.random(n)
        h = project_box_sum(h, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- sub‑gradient descent (most of the budget) ----
    it = 0
    last_improve = 0
    eta = 3.0  # base step size
    while time.time() - start < total * 0.78:
        # correlation and worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)

        # gradient of the worst shift
        grad = np.ones(n)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[kk:] -= h[: n - kk]
            grad[: n - kk] -= h[kk:]

        # step size
        step = eta / np.sqrt(it + 1.0)
        h_new = h - step * grad
        h = project_box_sum(h_new, target)

        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            last_improve = it

        # occasional random restart if stagnated
        if it - last_improve > 5000:
            # perturb the current best modestly
            h = best_h + rng.uniform(-0.05, 0.05, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, target)
            last_improve = it
        it += 1

    # ---- SLSQP refinement (soft‑max) ----
    betas = (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0, 20000.0)
    for beta in betas:
        if time.time() - start > total * 0.95:
            break
        h = slsqp_optimize(best_h, beta, max_iter=800)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if best_c <= 0.3808:
            break

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
