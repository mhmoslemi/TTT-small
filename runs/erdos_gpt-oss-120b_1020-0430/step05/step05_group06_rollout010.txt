def smooth_max_and_grad(h, beta):
    corr = np.correlate(h, 1 - h, mode='full')
    # shape 2n-1
    a = beta * corr
    max_a = np.max(a)
    exp_a = np.exp(a - max_a)
    sum_exp = np.sum(exp_a)
    smooth = (max_a + np.log(sum_exp)) / beta
    w = exp_a / sum_exp  # weights sum to 1
    # compute gradient
    n = h.size
    w_arr = w  # length 2n-1
    conv1 = np.convolve(w_arr, h, mode='full')
    term1 = conv1[n - 1 : n - 1 + n]  # indices n-1 .. n-1 + n-1
    w_rev = w_arr[::-1]
    conv2 = np.convolve(w_rev, h, mode='full')
    term2 = conv2[n - 1 : n - 1 + n]
    grad = 1.0 - term1 - term2
    return smooth, grad

    # Candidate n values

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact adjustment so that sum(h) == len(h)/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Optimize smooth‑max surrogate with SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def greedy_binary(n, rng):
    """Construct a binary vector with low max overlap."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if (best_w is None) or (w < best_w):
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h, O

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # use provided parent if present
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        c_parent = compute_c5(parent)
        best_h = parent.copy()
        best_c = c_parent
        best_n = parent.size

    # search over two discretisations
    for n in (360, 720):
        if time.time() > deadline:
            break

        candidates = []

        if best_n == n and best_h is not None:
            candidates.append(best_h.copy())

        if n == 360:
            # greedy binary
            h_bin, _ = greedy_binary(n, rng)
            candidates.append(h_bin.astype(float))
            # a few random binary starts
            for _ in range(5):
                perm = rng.permutation(n)
                h_rand = np.zeros(n, dtype=float)
                h_rand[perm[: n // 2]] = 1.0
                candidates.append(h_rand)

        else:  # n == 720
            if best_n == 360 and best_h is not None:
                candidates.append(np.repeat(best_h, 2).copy())
            for _ in range(3):
                perm = rng.permutation(n)
                h_rand = np.zeros(n, dtype=float)
                h_rand[perm[: n // 2]] = 1.0
                candidates.append(h_rand)

        for h0 in candidates:
            if time.time() > deadline:
                break
            adjust_sum_exact(h0, rng)

            c0 = compute_c5(h0)
            if c0 < best_c:
                best_c, best_h, best_n = c0, h0.copy(), n

            # smooth‑max cascade
            betas = [10, 50, 200, 1000, 5000, 20000, 50000, 100000]
            for beta in betas:
                if time.time() > deadline:
                    break
                max_iter = 400 if beta < 2000 else 800
                try:
                    h_opt = slsqp_optimize(h0, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                adjust_sum_exact(h_opt, rng)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_c, best_h, best_n = c_opt, h_opt.copy(), n
                h0 = h_opt  # warm start next beta

            # directed local search against the current worst shift
            while time.time() < deadline:
                corr = np.correlate(h0, 1.0 - h0, mode="full")
                idx_max = int(np.argmax(corr))
                k_max = idx_max - (n - 1)
                lo_i = max(0, -k_max)
                hi_i = min(n, n - k_max)
                if hi_i - lo_i <= 0:
                    break
                i = rng.integers(lo_i, hi_i)
                j = i + k_max
                if h0[i] <= 0 or h0[j] >= 1:
                    continue
                max_eps = min(h0[i], 1.0 - h0[j])
                eps = max_eps * rng.random()
                if eps < 1e-8:
                    continue
                h_new = h0.copy()
                h_new[i] -= eps
                h_new[j] += eps
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    best_c, best_h, best_n = c_new, h_new.copy(), n
                    h0 = h_new
                # else keep h0 unchanged

            # generic random pair transfers
            while time.time() < deadline:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                if h0[i] <= 0 or h0[j] >= 1:
                    continue
                max_eps = min(h0[i], 1.0 - h0[j])
                eps = max_eps * rng.random()
                if eps < 1e-8:
                    continue
                h_new = h0.copy()
                h_new[i] -= eps
                h_new[j] += eps
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    best_c, best_h, best_n = c_new, h_new.copy(), n
                    h0 = h_new

    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
