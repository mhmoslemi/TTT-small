def binary_top(arr):
    n = arr.size
    idx = np.argpartition(arr, -n//2)[-n//2:]
    out = np.zeros(n, dtype=np.float64)
    out[idx] = 1.0

def compute_O(h):
    n = h.size
    # L
    prefix = np.concatenate(([0], np.cumsum(h)))
    L = prefix[n - np.arange(n)]  # using broadcasting
    # P using correlation
    P = np.correlate(h, h, mode='full')[n-1:]
    O = L - P
    return O

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def delta_O_for_swap(h, i, j):
    n = h.size
    deltaL = np.zeros(n, dtype=int)
    deltaL[:n-j] += 1
    deltaL[:n-i] -= 1

    deltaP = np.zeros(n, dtype=int)

    # i forward
    len_fi = n - i
    if len_fi:
        deltaP[:len_fi] -= h[i:]
    # i backward
    len_bi = i + 1
    if len_bi:
        deltaP[:len_bi] -= h[:i+1][::-1]

    # j forward
    len_fj = n - j
    if len_fj:
        deltaP[:len_fj] += h[j:]
    # j backward
    len_bj = j + 1
    if len_bj:
        deltaP[:len_bj] += h[:j+1][::-1]

    delta = deltaL - deltaP
    return delta

def greedy_local_search(rng, h, O):
    n = h.size
    while True:
        s = int(np.argmax(O))
        if s == 0:
            break
        # compute neighbor counts
        arr = np.zeros(n, dtype=int)
        arr[:n-s] += h[s:]
        arr[s:] += h[:n-s]

        ones_idx = np.flatnonzero(h)
        zeros_idx = np.flatnonzero(1 - h)

        if ones_idx.size == 0 or zeros_idx.size == 0:
            break

        min_val = arr[ones_idx].min()
        max_val = arr[zeros_idx].max()

        i_candidates = ones_idx[arr[ones_idx] == min_val]
        j_candidates = zeros_idx[arr[zeros_idx] == max_val]

        i = rng.choice(i_candidates)
        j = rng.choice(j_candidates)

        delta = delta_O_for_swap(h, i, j)
        new_O = O + delta
        if new_O.max() < O.max():
            O = new_O
            h[i] = 0.0
            h[j] = 1.0
        else:
            # no improving swap found, break
            break
    return h, O

def simulated_annealing(rng, h, O, n, deadline):
    # temperature schedule
    cur_max = int(O.max())
    cur_c = cur_max * (2.0 / n)
    best_h = h.copy()
    best_O = O.copy()
    best_c = cur_c

    temp = 0.5 * cur_max  # initial temperature (~half current max)
    cooling = 0.9995
    # precompute ones and zeros arrays (list for random choice)
    while time.time() < deadline:
        # choose random swap
        ones_idx = np.flatnonzero(h)
        zeros_idx = np.flatnonzero(1 - h)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        delta = delta_O_for_swap(h, i, j)
        new_O = O + delta
        new_max = new_O.max()
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / (temp + 1e-12)):
            # accept
            O = new_O
            cur_max = new_max
            h[i] = 0.0
            h[j] = 1.0
            # update best
            if cur_max * (2.0 / n) < best_c:
                best_c = cur_max * (2.0 / n)
                best_h = h.copy()
                best_O = O.copy()
        temp *= cooling
        if temp < 1e-6:
            temp = 1e-6
    return best_h, best_O, best_c

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_local(h0, beta, rng, max_iter=800):
    """SLSQP on smooth‑max surrogate (fractional h)."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def binary_top(arr):
    n = arr.size
    idx = np.argpartition(arr, -n // 2)[-n // 2 :]
    out = np.zeros(n, dtype=np.int8)
    out[idx] = 1
    return out

def compute_O(h):
    """O_s = L_s - P_s for all shifts s."""
    n = h.size
    pref = np.concatenate(([0], np.cumsum(h, dtype=np.int64)))
    L = pref[n - np.arange(n)]
    P = np.correlate(h, h, mode="full")[n - 1 :]
    return L - P

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def delta_O_for_swap(h, i, j):
    """Exact O_s change when swapping a 1 at i with a 0 at j."""
    n = h.size
    deltaL = np.zeros(n, dtype=np.int32)
    deltaL[: n - j] += 1
    deltaL[: n - i] -= 1

    deltaP = np.zeros(n, dtype=np.int32)

    # i forward (s=0..n-i-1)
    deltaP[: n - i] -= h[i:]

    # i backward (s=1..i)
    if i > 0:
        deltaP[1 : i + 1] -= h[i - 1 :: -1]

    # j forward
    deltaP[: n - j] += h[j:]

    # j backward (s=1..j)
    if j > 0:
        deltaP[1 : j + 1] += h[j - 1 :: -1]

    # correct double‑count of the (i,j) pair for s=|i-j|>0
    s_pair = abs(j - i)
    if s_pair > 0:
        deltaP[s_pair] -= 1

    # square term s=0 must stay zero
    deltaP[0] += 1

    return deltaL - deltaP

def greedy_local_search(rng, h, O):
    n = h.size
    while True:
        s = int(np.argmax(O))
        if s == 0:
            break
        arr = np.zeros(n, dtype=np.int32)
        arr[: n - s] += h[s:]
        arr[s:] += h[: n - s]

        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break

        min_val = arr[ones].min()
        max_val = arr[zeros].max()
        i_cand = ones[arr[ones] == min_val]
        j_cand = zeros[arr[zeros] == max_val]
        i = rng.choice(i_cand)
        j = rng.choice(j_cand)

        delta = delta_O_for_swap(h, i, j)
        new_O = O + delta
        if new_O.max() < O.max():
            O = new_O
            h[i] = 0
            h[j] = 1
        else:
            break
    return h, O

def simulated_annealing(rng, h, O, n, deadline, target_max):
    cur_h = h.copy()
    cur_O = O.copy()
    cur_max = int(cur_O.max())
    best_h, best_O, best_max = cur_h.copy(), cur_O.copy(), cur_max
    temp = max(1.0, cur_max / 2.0)
    cooling = 0.9999
    steps = 0
    while time.time() < deadline and steps < 250000:
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        delta = delta_O_for_swap(cur_h, i, j)
        new_O = cur_O + delta
        new_max = int(new_O.max())

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h[i] = 0
            cur_h[j] = 1
            cur_O = new_O
            cur_max = new_max
            if cur_max < best_max:
                best_h, best_O, best_max = cur_h.copy(), cur_O.copy(), cur_max
                if best_max <= target_max:
                    break
        temp = max(temp * cooling, 0.5)
        steps += 1
    return best_h, best_O, best_max * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    parent_h = None
    if "initial_h_values" in globals():
        parent_h = np.array(initial_h_values, dtype=np.float64)

    candidate_ns = [360, 400, 440, 480, 500, 560, 640, 720]
    best_c = np.inf
    best_h = None
    best_n = None

    for n in candidate_ns:
        if time.time() > deadline:
            break

        target_max = int(np.floor(0.3808 * n / 2.0))

        if parent_h is not None and n == parent_h.size:
            h0 = binary_top(parent_h)
        else:

                hf = np.interp(xt, xs, parent_h)
            else:
                hf = rng.random(n)
            h0 = binary_top(hf)

        O = compute_O(h0)
        # greedy refinement
        h_g, O_g = greedy_local_search(rng, h0, O)
        c_g = O_g.max() * (2.0 / n)
        if c_g < best_c:
            best_c, best_h, best_n = c_g, h_g.copy(), n
        if O_g.max() <= target_max:
            final_c = compute_c5(best_h)
            return best_h.tolist(), float(final_c), int(best_n)

        # simulated annealing
        h_a, O_a, c_a = simulated_annealing(rng, h_g, O_g, n, deadline, target_max)
        if c_a < best_c:
            best_c, best_h, best_n = c_a, h_a.copy(), n
        if O_a.max() <= target_max:
            final_c = compute_c5(best_h)
            return best_h.tolist(), float(final_c), int(best_n)

        # fractional SLSQP polish
        h_frac = h_a.astype(np.float64)
        for beta in (50, 200, 1000):

            h_frac = slsqp_optimize_local(h_frac, beta, rng, max_iter=800)
            c_frac = compute_c5(h_frac)
            if c_frac < best_c:

            if c_frac <= 0.3808:
                return best_h.tolist(), float(c_frac), int(best_n)

    # fallback simple split
    if best_h is None:
        best_n = 360
        best_h = np.zeros(best_n, dtype=np.int8)
        best_h[: best_n // 2] = 1
        best_c = compute_c5(best_h)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
