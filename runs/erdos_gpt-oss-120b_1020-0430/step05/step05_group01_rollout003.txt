def subgrad_descent(h, rng, time_limit):
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    target_sum = h.size / 2.0
    step0 = 0.05
    it=0
    while time.time() - start < time_limit:
        it += 1
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = np.max(corr) * (2.0 / h.size)
        # optional early tracking for best
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # compute worst shift
        idx = np.argmax(corr) - (h.size - 1)
        k = abs(idx)
        # compute gradient
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            # (1 - h[i+k]) for i where i+k in range; rest gradient 1
            grad = np.ones_like(h)
            grad[:h.size - k] -= h[k:]
            grad[k:] -= h[:h.size - k]
        # step size decays with sqrt(it)
        step = step0 / np.sqrt(it)
        # take step opposite gradient (since we want reduce max)
        h_new = h - step * grad
        # clip
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # enforce sum
        diff = target_sum - h_new.sum()
        # adjust diff:
        if abs(diff) > 1e-12:
            # allocate diff to random indices that can move
            # iterate limited times
            for _ in range(5 * h.size):
                if abs(diff) < 1e-12: break
                i = rng.integers(0, h.size)
                if diff > 0 and h_new[i] < 1.0:
                    add = min(diff, 1.0 - h_new[i])
                elif diff < 0 and h_new[i] > 0.0:
                    add = max(diff, -h_new[i])
                else:
                    continue
                h_new[i] += add
                diff = target_sum - h_new.sum()
        h = h_new
    return best_h, best_c
```

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # ensure diff small by moving mass
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

import time
import numpy as np
from scipy.optimize import minimize

# Global variables used by the SLSQP callbacks
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer a small ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))

        return h.copy()

        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh
