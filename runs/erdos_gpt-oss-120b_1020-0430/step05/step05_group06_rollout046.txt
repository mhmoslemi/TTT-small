def adam_optimize(h0, betas, steps_per_beta, S, rng, deadline):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    for beta in betas:
        alpha = 0.1 / beta # initial step size
        for _ in range(steps_per_beta):
            if time.time() > deadline: break
            t += 1
            grad = grad_smoothmax_corr(h, beta)
            # update Adam
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad ** 2)
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            h_new = h - alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
            # project
            h = project_onto_box_and_sum(h_new, S)
    return h

def hill_climb_swaps(h_bin, max_swaps, rng, deadline):
    best_h = h_bin.copy()
    best_max = np.max(np.correlate(best_h, 1 - best_h, mode='full'))
    cur_h = h_bin.copy()
    n = len(h_bin)
    for _ in range(max_swaps):
        if time.time() > deadline:
            break
        # pick random i (1) and j (0)
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        cur_max = np.max(np.correlate(cur_h, 1 - cur_h, mode='full'))
        if cur_max < best_max:
            best_max = cur_max
            best_h = cur_h.copy()
            # update ones and zeros quickly:
            # but recomputing ones/zeros each time is heavy; we can maintain but fine.
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_max

def continuous_refine(h0, betas, steps_per_beta, target_sum, rng, deadline):
    h = project_to_feasible(h0, target_sum) # ensure initial feasible

    return h

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def grad_smoothmax_corr(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    # compute weights safely
    scaled = beta * corr
    max_s = np.max(scaled)
    exp_scaled = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_scaled)
    w = exp_scaled / sum_exp
    # term2 and term3
    rc = np.correlate(w, h, mode="full")
    term2 = rc[n-1: n-1 + n]
    rc_rev = np.correlate(w[::-1], h, mode="full")
    term3 = rc_rev[n-1: n-1 + n]
    grad = 1.0 - term2 - term3
    return grad

def project_onto_feasible(v, S):
    # binary search for lambda
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def continuous_optimize(h0, betas, steps_per_beta, target_sum, rng, deadline):
    h = project_onto_feasible(h0, target_sum)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    for beta in betas:
        # adapt step size: maybe 0.5 / beta
        alpha = 0.5 / max(beta, 1e-12)
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return h
            t += 1
            grad = grad_smoothmax_corr(h, beta)
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_new = h - step
            h = project_onto_feasible(h_new, target_sum)
    return h

def hill_climb_swaps(h_bin, max_swaps, rng, deadline):
    best_h = h_bin.copy()
    best_max = np.max(np.correlate(best_h, 1 - best_h, mode="full"))
    cur_h = h_bin.copy()
    n = len(h_bin)
    # Precompute arrays of ones and zeros each iteration? Could recompute each iteration.
    for _ in range(max_swaps):
        if time.time() > deadline:
            break
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        cur_max = np.max(np.correlate(cur_h, 1 - cur_h, mode="full"))
        if cur_max < best_max:
            best_max = cur_max
            best_h = cur_h.copy()
            # update ones and zeros? We'll recompute next loop.
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_max

import time
import numpy as np

# ----- basic helpers -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


# ----- gradient of smooth‑max ---------------------------------------
def _grad_smoothmax_corr(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/beta) log‑sum‑exp(beta * corr_k) w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / np.sum(exp_scaled)            # weights, sum = 1

    # term2 = Σ_k w_k * h_{j+k}
    rc = np.correlate(w, h, mode="full")
    term2 = rc[n - 1: n - 1 + n]

    # term3 = Σ_k w_k * h_{j-k}
    rc_rev = np.correlate(w[::-1], h, mode="full")
    term3 = rc_rev[n - 1: n - 1 + n]

    return 1.0 - term2 - term3


# ----- projection onto {0 ≤ x ≤ 1, Σx = S} -------------------------
def _project_onto_feasible(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection onto the capped simplex."""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ----- continuous optimisation (Adam + projection) -----------------
def _continuous_optimize(
    h0: np.ndarray,
    betas,
    steps_per_beta: int,
    target_sum: float,
    rng,
    deadline: float,
) -> np.ndarray:
    """Adam on the smooth‑max surrogate, projecting onto the feasible set."""
    h = _project_onto_feasible(h0, target_sum)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    for beta in betas:
        alpha = 1.0 / max(beta, 1e-12)
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return h
            t += 1
            grad = _grad_smoothmax_corr(h, beta)
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            m_hat = m / (1.0 - 0.9 ** t)
            v_hat = v / (1.0 - 0.999 ** t)
            step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
            h = _project_onto_feasible(h - step, target_sum)
    return h


# ----- simple hill‑climb on binary vectors -------------------------
def _hill_climb_swaps(h_bin: np.ndarray, max_swaps: int, rng, deadline: float):
    """Swap 1↔0 to reduce max correlation."""
    best_h = h_bin.copy()
    best_max = np.max(np.correlate(best_h, 1 - best_h, mode="full"))
    cur_h = h_bin.copy()
    n = h_bin.size
    for _ in range(max_swaps):
        if time.time() > deadline:
            break
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        cur_max = np.max(np.correlate(cur_h, 1 - cur_h, mode="full"))
        if cur_max < best_max:
            best_max = cur_max
            best_h = cur_h.copy()
            # keep the good swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_max


# ----- local fine‑tuning (mass transfer) ----------------------------
def _local_fine_tune(h: np.ndarray, init_c: float, rng, deadline: float, max_iters: int = 2000):
    """Move a small amount of mass between two coordinates if it helps."""
    best_h = h.copy()
    best_c = init_c
    n = h.size
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
    return best_h, best_c
