def project_onto_capped_simplex(v, target, max_iter=50, tol=1e-12):
    lo = -np.max(v) - 1  # lower bound on λ (ensures all v+λ <=0)
    hi = 1 - np.min(v)   # upper bound (ensures all v+λ >=1)
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        w = np.clip(v + mid, 0, 1)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            hi = mid
        else:
            lo = mid
    return w

def worst_shift_gradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = idx
    # compute gradient vector
    g = np.zeros_like(h)
    # term from 1 - h[i+k]
    if k >= 0:
        # i from 0 to n-1-k
        g[:n-k] += 1.0 - h[k:]
    else:
        # k negative
        # i from -k to n-1 (since i+k >=0)
        g[-k:] += 1.0 - h[:n+k]  # because k negative => n+k < n
    # term -h[i-k]
    if k > 0:
        g[k:] -= h[:n-k]
    elif k < 0:
        g[:n+k] -= h[-k:]  # k negative => -k positive
    # edges beyond range already zero
    return g, k, corr.max() * (2.0 / n)

def exchange_refine(h, max_iter=500):

    target = n/2.0

        corr = np.correlate(h, 1.0 - h, mode='full')

        g[: n - m] -= h[m:]

    return g, k, corr.max() * (2.0 / n)

def project_onto_capped_simplex(v, target, tol=1e-12, max_iter=50):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            hi = mid
        else:
            lo = mid
    return w

def projected_gradient_descent(h, target, rng, max_iter=200, step_init=0.05):

def exchange_refine(h, target, max_iter=500):
    n = len(h)
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = idx
        # compute w_i = h_{i+k} if i+k valid else 0
        indices = np.arange(n) + k
        w = np.where((indices >= 0) & (indices < n), h[indices], 0.0)
        # candidates
        dec_mask = h > 1e-12
        inc_mask = h < 1 - 1e-12
        if not np.any(dec_mask) or not np.any(inc_mask):
            break
        dec_idxs = np.where(dec_mask)[0]
        inc_idxs = np.where(inc_mask)[0]
        i = dec_idxs[np.argmax(w[dec_idxs])]
        j = inc_idxs[np.argmin(w[inc_idxs])]
        if w[i] <= w[j] + 1e-12:
            break
        delta = min(h[i], 1 - h[j])
        # optional limit delta to e.g., 0.1 to avoid big jumps causing oscillations
        # delta = min(delta, 0.1)
        h[i] -= delta
        h[j] += delta
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_onto_capped_simplex(v: np.ndarray, target: float, tol: float = 1e-12, max_iter: int = 50) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + mid, 0.0, 1.0)

def gradient_worst_shift(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = idx
    g = np.zeros_like(h)
    if k >= 0:
        g[: n - k] += 1.0 - h[k:]
        g[k:] -= h[: n - k]
    else:
        m = -k
        g[m:] += 1.0 - h[: n - m]
        g[: n - m] -= h[m:]
    cur_c5 = corr.max() * (2.0 / n)
    return g, k, cur_c5

def projected_gradient_descent(h: np.ndarray, target: float,
                               max_iter: int = 200, step_init: float = 0.05) -> np.ndarray:
    step = step_init
    for _ in range(max_iter):
        g, _k, cur_c5 = gradient_worst_shift(h)
        v = h - step * g
        h_new = project_onto_capped_simplex(v, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            step = min(step * 1.1, 0.2)
        else:
            step *= 0.5
        if step < 1e-6:
            break
    return h

def exchange_refine(h: np.ndarray, target: float, max_iter: int = 500) -> np.ndarray:
    n = h.size
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        inds = np.arange(n) + k
        w = np.where((inds >= 0) & (inds < n), h[inds], 0.0)

        dec_mask = h > 1e-12
        inc_mask = h < 1.0 - 1e-12
        if not np.any(dec_mask) or not np.any(inc_mask):
            break

        dec_idxs = np.where(dec_mask)[0]
        inc_idxs = np.where(inc_mask)[0]

        i = dec_idxs[np.argmax(w[dec_idxs])]
        j = inc_idxs[np.argmin(w[inc_idxs])]

        if w[i] <= w[j] + 1e-12:
            break

        delta = min(h[i], 1.0 - h[j])
        h[i] -= delta
        h[j] += delta
    return h

def refine_h(h: np.ndarray, deadline: float) -> np.ndarray:
    target = h.size / 2.0
    while time.time() < deadline:
        old_c5 = compute_c5(h)
        h = projected_gradient_descent(h, target, max_iter=150, step_init=0.07)
        h = exchange_refine(h, target, max_iter=300)
        new_c5 = compute_c5(h)
        if new_c5 >= old_c5 - 1e- nine:
            break
        if time.time() > deadline - 0.5:
            break
    return h

def random_perturb(h: np.ndarray, target: float, rng: np.random.Generator,
                   magnitude: float = 0.05) -> np.ndarray:
    v = h + rng.normal(scale=magnitude, size=h.size)
    return project_onto_capped_simplex(v, target)

def final_refine(h: np.ndarray, deadline: float) -> np.ndarray:
    target = h.size / 2.0
    step = 0.1
    for _ in range(500):
        if time.time() > deadline - 0.5:
            break
        g, _k, cur_c5 = gradient_worst_shift(h)
        v = h - step * g
        h_new = project_onto_capped_simplex(v, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            step = min(step * 1.05, 0.2)
        else:
            step *= 0.7
        if step < 1e-6:
            break
    h = exchange_refine(h, target, max_iter=500)
    return h

    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
