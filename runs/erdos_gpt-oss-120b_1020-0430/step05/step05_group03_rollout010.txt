def softmax_obj_and_grad(h, beta):
    b = 1 - h
    corr = np.correlate(h, b, mode='full')
    max_corr = np.max(corr)
    # compute weights
    exps = np.exp(beta * (corr - max_corr))
    sum_exps = np.sum(exps)
    w = exps / sum_exps
    # compute conv using np.convolve
    conv1 = np.convolve(w, b, mode='full')
    G1 = conv1[(len(w)-1) : (len(w)-1) + len(h)]
    w_rev = w[::-1]
    conv2 = np.convolve(w_rev, h, mode='full')

    return obj, grad

import time
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---------- core utilities ----------
    def compute_c5(h: np.ndarray) -> float:
        """Exact discrete C5 for a step function on [0,2]."""
        dx = 2.0 / h.size
        return np.max(np.correlate(h, 1 - h, mode="full")) * dx

    def project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
        """Project onto [0,1]^n intersect {sum = target} by bisection."""
        lo, hi = -1e6, 1e6
        for _ in range(60):
            mid = (lo + hi) * 0.5
            p = np.clip(x + mid, 0.0, 1.0)
            s = p.sum()
            if s < target:
                lo = mid
            else:
                hi = mid
        return np.clip(x + hi, 0.0, 1.0)

    def worst_gradient(h: np.ndarray) -> np.ndarray:
        """Gradient of the worst shift overlap (the shift giving max C5)."""
        n = h.shape[0]
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.ones_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        return grad

    def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                          max_iters: int = 500, step_scale: float = 0.5):
        """Hill‑climb using gradient‑directed transfers."""
        n = h.shape[0]
        best_h, best_c = h.copy(), compute_c5(h)
        step = step_scale
        for it in range(max_iters):
            grad = worst_gradient(h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * step
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            c = compute_c5(nh)
            if c < best_c:
                best_c, best_h, h = c, nh.copy(), nh
            if (it + 1) % (max_iters // 5 + 1) == 0:
                step = max(step * 0.7, 1e-4)
        return best_h, best_c

    def softmax_obj_and_grad(h: np.ndarray, beta: float):
        """Soft‑max of correlation and its gradient."""
        n = h.shape[0]
        b = 1.0 - h
        corr = np.correlate(h, b, mode="full")
        max_corr = np.max(corr)
        exps = np.exp(beta * (corr - max_corr))
        sum_exps = np.sum(exps)
        w = exps / sum_exps
        off = len(w) - 1
        conv1 = np.convolve(w, b, mode="full")
        G1 = conv1[off:off + n]
        w_rev = w[::-1]
        conv2 = np.convolve(w_rev, h, mode="full")
        G2 = conv2[off:off + n]
        grad = G1 - G2
        obj = max_corr + np.log(sum_exps) / beta
        return obj, grad

    def gradient_descent_h(h: np.ndarray, beta: float, max_iter: int,
                          rng: np.random.Generator):
        """Projected gradient descent on the soft‑max objective."""
        n = h.shape[0]
        target = n / 2.0
        step = 1.0
        obj, grad = softmax_obj_and_grad(h, beta)
        for _ in range(max_iter):
            h_new = project_box_sum(h - step * grad, target)
            obj_new, grad_new = softmax_obj_and_grad(h_new, beta)
            if obj_new < obj:
                h, obj, grad = h_new, obj_new, grad_new
                step *= 1.1
            else:
                step *= 0.5
                if step < 1e-8:
                    break
        return h, obj

    # ---------- main search ----------
    n = 512
    target = n / 2.0
    best_h, best_c = None, np.inf

    # start from parent if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, target)
        c0 = compute_c5(h0)
        best_h, best_c = h0.copy(), c0

    betas = [5., 20., 80., 200., 800., 2000., 8000., 20000.]
    max_iters_beta = {5: 800, 20: 800, 80: 800, 200: 400,
                      800: 300, 2000: 300, 8000: 200, 20000: 200}
    restart = 0
    while time.time() - start < budget_s * 0.85:
        # pick a starting point
        if restart == 0 and best_h is not None:
            h = best_h.copy()
        else:
            if rng.random() < 0.5:
                h = np.zeros(n)
                h[rng.choice(n, n // 2, replace=False)] = 1.0
            else:
                h = project_box_sum(rng.random(n), target)

        # continuation on increasing beta
        for beta in betas:
            if time.time() - start >= budget_s * 0.85:
                break
            h, _ = gradient_descent_h(h, beta, max_iters_beta[beta], rng)

        # fine local search
        if time.time() - start < budget_s * 0.85:
            h, c = guided_hill_climb(h, rng, max_iters=400, step_scale=0.6)
        else:
            c = compute_c5(h)

        if c < best_c:
            best_c, best_h = c, h.copy()

        # random epsilon transfers on the global best
        if time.time() - start < budget_s * 0.85:
            for _ in range(30):
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                max_eps = min(best_h[i], 1.0 - best_h[j])
                if max_eps <= 0.0:
                    continue
                eps = max_eps * rng.random() * 0.05
                nh = best_h.copy()
                nh[i] -= eps
                nh[j] += eps
                nc = compute_c5(nh)
                if nc < best_c:
                    best_c, best_h = nc, nh.copy()
        restart += 1

    # final refinement on the best candidate
    if best_h is not None and time.time() - start < budget_s * 0.95:
        best_h, _ = gradient_descent_h(best_h, 20000., 400, rng)
        best_h, c = guided_hill_climb(best_h, rng, max_iters=600, step_scale=0.4)
        if c < best_c:
            best_c, best_h = c, best_h.copy()

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
