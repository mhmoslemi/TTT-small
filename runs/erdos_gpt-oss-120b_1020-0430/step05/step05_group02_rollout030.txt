def compute_c5_fft(h):
    n = h.shape[0]
    m = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft((1-h)[::-1], m)

    max_val = conv.max()
    return float(max_val * (2.0 / n))

def sa_optimize(h, rng, time_deadline, max_steps=None):
    n = len(h)
    target_sum = n/2.0
    step_scale = 0.1  # max delta per move (maybe start larger)
    cur_c5 = compute_c5_fft(h)
    best_h = h.copy()
    best_c5 = cur_c5
    it = 0
    # initialize temperature based on cur_c5: T0 = cur_c5 * 0.05 + 1e-4
    T = cur_c5*0.05 + 1e-4
    while time.time() < time_deadline and (max_steps is None or it < max_steps):
        # propose move: pick i, j distinct
        i, j = rng.choice(n, size=2, replace=False)
        # delta between -step_scale..+step_scale random
        delta = (rng.random() - 0.5) * 2 * step_scale
        # Adjust h[i] by delta, h[j] by -delta to keep sum.
        # Ensure bounds.
        hi = h[i]
        hj = h[j]
        # compute feasible delta range:
        max_delta = min(step_scale, 1.0 - hi, hj)  # ensure hi+delta <=1, hj-delta >=0
        min_delta = max(-step_scale, -hi, -(1.0 - hj)) # ensure hi+delta >=0, hj-delta <=1
        # Actually we can compute feasible delta = clip(delta, min_delta, max_delta)
        delta = np.clip(delta, min_delta, max_delta)
        if abs(delta) < 1e-12:
            it += 1
            continue
        h[i] = hi + delta
        h[j] = hj - delta
        new_c5 = compute_c5_fft(h)
        d = new_c5 - cur_c5
        if d < -1e-12 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
        # cooling
        T *= 0.99995
        # gradually reduce step_scale
        if it % 1000 == 0 and step_scale > 0.001:
            step_scale *= 0.99
        it += 1
    return best_h, best_c5

def create_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def create_fractional_random(n, rng):
    target = n/2.0
    h = rng.random(n)
    h *= target / h.sum()
    # adjust exactly due to floating point
    adjust_sum_exact(h, rng)
    return h

def binary_from_parent(parent_h, n, rng):
    # interpolate to new n
    xs = np.linspace(0, 2, parent_h.size, endpoint=False)
    xt = np.linspace(0, 2, n, endpoint=False)
    h_interp = np.interp(xt, xs, parent_h)
    # produce binary version: top half indices
    idx = np.argpartition(h_interp, -n//2)[-n//2:]
    bin_h = np.zeros(n)
    bin_h[idx] = 1.0
    return h_interp, bin_h

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft((1.0 - h)[::-1], m)
    conv = np.fft.irfft(H * G, m)[:2*n-1]
    return float(conv.max() * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.shape[0] / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# SLSQP helpers (softmax surrogate)
CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def sa_optimize(h0, rng, deadline, max_steps=None):
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    T = cur_c5 * 0.05 + 1e-4
    step_scale = 0.2
    it = 0
    while time.time() < deadline and (max_steps is None or it < max_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi = cur_h[i]
        hj = cur_h[j]
        # feasible delta range
        max_inc = min(step_scale, 1.0 - hi, hj)
        min_inc = max(-step_scale, -hi, -(1.0 - hj))
        if max_inc <= min_inc:
            it += 1
            continue
        delta = rng.random() * (max_inc - min_inc) + min_inc
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta
        new_c5 = compute_c5(cur_h)
        d = new_c5 - cur_c5
        if d < -1e-12 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            cur_h[i] = hi
            cur_h[j] = hj
        T *= 0.99995
        if it % 2000 == 0 and step_scale > 0.001:
            step_scale *= 0.99
        it += 1
    return best_h, best_c5

def binary_from_parent(parent_h, n, rng):
    xs = np.linspace(0.0, 2.0, parent_h.shape[0], endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    h_interp = np.interp(xt, xs, parent_h)
    idx = np.argpartition(h_interp, -n//2)[-n//2:]
    bin_h = np.zeros(n)
    bin_h[idx] = 1.0
    return h_interp, bin_h

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def random_fractional(n, rng):
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    adjust_sum_exact(h, rng)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.97 * budget_s  # reserve a bit
    best_h = None
    best_c5 = np.inf
    best_n = None

    parent_h = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    # candidate n values (even)
    candidate_ns = [360, 480, 720]

    for n in candidate_ns:
        if time.time() >= deadline:
            break
        target = n / 2.0

        # build initial vectors
        init_vecs = []

        if parent_h is not None:
            if parent_h.shape[0] == n:
                # use parent directly
                init_vecs.append(parent_h.copy())
                # binary version of parent
                idx = np.argpartition(parent_h, -n//2)[-n//2:]
                bin_parent = np.zeros(n)
                bin_parent[idx] = 1.0
                init_vecs.append(bin_parent)
            else:
                interp, bin_interp = binary_from_parent(parent_h, n, rng)
                init_vecs.append(interp)
                init_vecs.append(bin_interp)

        # random binary
        init_vecs.append(random_binary(n, rng))
        # random fractional
        init_vecs.append(random_fractional(n, rng))
        # alternating pattern
        alt = np.tile([1.0, 0.0], n // 2)
        init_vecs.append(alt)

        # Process each seed
        for base_h in init_vecs:
            if time.time() >= deadline:
                break
            # Ensure feasibility
            h = np.clip(base_h, 0.0, 1.0)
            # if not binary, enforce exact sum
            if not np.all((h == 0) | (h == 1)):
                adjust_sum_exact(h, rng)

            # allocate time slice for this seed
            remaining = deadline - time.time()
            # leave at least 30 seconds for final refinements
            if remaining < 30:
                break
            # give each seed up to 0.12 of remaining time, but not more than 45 seconds
            slice_time = min(45.0, remaining * 0.12)
            seed_deadline = time.time() + slice_time

            # simulated annealing
            h_sa, c5_sa = sa_optimize(h, rng, seed_deadline)
            if c5_sa < best_c5:
                best_h, best_c5, best_n = h_sa.copy(), c5_sa, n

            # short gradient refinement
            h_ref = gradient_refine(h_sa.copy(), rng, step=0.02, max_iter=6)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_h, best_c5, best_n = h_ref.copy(), c5_ref, n

            # brief SLSQP cascade
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
                if time.time() >= deadline:
                    break
                try:
                    h_opt = slsqp_optimize(h_ref.copy(), beta, rng, max_iter=200)
                except Exception:
                    continue
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
                h_ref = h_opt  # continue chain

    # If we have a good solution, try upsampling and further refinement
    if best_h is not None and time.time() < deadline:
        if best_n * 2 <= 720:
            n_up = best_n * 2
            xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            adjust_sum_exact(h_up, rng)

            # allocate remaining time for upsampled SA
            remaining = deadline - time.time()
            if remaining > 20:
                seed_deadline = time.time() + remaining * 0.6
                h_up, c5_up = sa_optimize(h_up, rng, seed_deadline)
                # gradient + SLSQP
                h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
                for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0):
                    if time.time() >= deadline:
                        break
                    try:
                        h_up = slsqp_optimize(h_up, beta, rng, max_iter=200)
                    except Exception:
                        continue
                h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
                c5_up = compute_c5(h_up)
                if c5_up < best_c5:
                    best_h, best_c5, best_n = h_up, c5_up, n_up

    # final safety
    if best_h is None:
        # fallback alternating pattern
        n = 180
        best_n = n
        best_h = np.tile([1.0, 0.0], n // 2).astype(float)
        best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_n)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h):
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft((1.0 - h)[::-1], m)
    conv = np.fft.irfft(H * G, m)[: 2 * n - 1]
    return float(conv.max() * (2.0 / n))


def adjust_sum_exact(h, rng):
    target = h.shape[0] / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h


def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


# ---------- Simulated annealing (continuous) ----------
def sa_optimize(h0, rng, deadline, max_steps=None):
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    T = cur_c5 * 0.05 + 1e-4
    step_scale = 0.2
    it = 0
    while time.time() < deadline and (max_steps is None or it < max_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi = cur_h[i]
        hj = cur_h[j]
        max_inc = min(step_scale, 1.0 - hi, hj)
        min_inc = max(-step_scale, -hi, -(1.0 - hj))
        if max_inc <= min_inc:
            it += 1
            continue
        delta = rng.random() * (max_inc - min_inc) + min_inc
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta
        new_c5 = compute_c5(cur_h)
        d = new_c5 - cur_c5
        if d < -1e-12 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            cur_h[i] = hi
            cur_h[j] = hj
        T *= 0.99995
        if it % 2000 == 0 and step_scale > 0.001:
            step_scale *= 0.99
        it += 1
    return best_h, best_c5


# ---------- helpers for initial vectors ----------
def binary_from_parent(parent_h, n, rng):
    xs = np.linspace(0.0, 2.0, parent_h.shape[0], endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    h_interp = np.interp(xt, xs, parent_h)
    idx = np.argpartition(h_interp, -n // 2)[-n // 2 :]
    bin_h = np.zeros(n)
    bin_h[idx] = 1.0
    return h_interp, bin_h


def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def random_fractional(n, rng):
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    adjust_sum_exact(h, rng)
    return h


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.97 * budget_s  # keep a safety margin
    best_h = None
    best_c5 = np.inf
    best_n = None

    parent_h = np.array(initial_h_values, dtype=float) if "initial_h_values" in globals() else None
    candidate_ns = [360, 480, 720]  # even values for balanced binary options

    for n in candidate_ns:
        if time.time() >= deadline:
            break
        init_vecs = []

        if parent_h is not None:
            if parent_h.shape[0] == n:
                init_vecs.append(parent_h.copy())
                idx = np.argpartition(parent_h, -n // 2)[-n // 2 :]
                bin_parent = np.zeros(n)
                bin_parent[idx] = 1.0
                init_vecs.append(bin_parent)
            else:
                interp, bin_interp = binary_from_parent(parent_h, n, rng)
                init_vecs.append(interp)
                init_vecs.append(bin_interp)

        init_vecs.append(random_binary(n, rng))
        init_vecs.append(random_fractional(n, rng))
        init_vecs.append(np.tile([1.0, 0.0], n // 2))

        for base_h in init_vecs:
            if time.time() >= deadline:
                break
            h = np.clip(base_h, 0.0, 1.0)
            if not np.all((h == 0) | (h == 1)):
                adjust_sum_exact(h, rng)

            remaining = deadline - time.time()
            if remaining < 30.0:
                break
            slice_time = min(60.0, remaining * 0.12)
            seed_deadline = time.time() + slice_time

            # ----- simulated annealing -----
            h_sa, c5_sa = sa_optimize(h, rng, seed_deadline)
            if c5_sa < best_c5:
                best_h, best_c5, best_n = h_sa.copy(), c5_sa, n

            # ----- gradient refinement -----
            h_ref = gradient_refine(h_sa.copy(), rng, step=0.02, max_iter=12)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_h, best_c5, best_n = h_ref.copy(), c5_ref, n

            # ----- SLSQP cascade with light gradient passes -----
            for beta, max_it in ((5.0, 200), (20.0, 200), (80.0, 300),
                                 (200.0, 400), (500.0, 500), (1000.0, 600)):
                if time.time() >= deadline:
                    break
                try:
                    h_opt = slsqp_optimize(h_ref.copy(), beta, rng, max_iter=max_it)
                except Exception:
                    continue
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
                # brief gradient after each SLSQP
                h_opt = gradient_refine(h_opt, rng, step=0.02, max_iter=4)
                c5_opt2 = compute_c5(h_opt)
                if c5_opt2 < best_c5:
                    best_h, best_c5, best_n = h_opt.copy(), c5_opt2, n
                h_ref = h_opt

    # ----- up‑sampling for extra refinement -----
    if best_h is not None and time.time() < deadline:
        if best_n * 2 <= 720:
            n_up = best_n * 2
            xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            adjust_sum_exact(h_up, rng)

            remaining = deadline - time.time()
            if remaining > 20.0:
                seed_deadline = time.time() + remaining * 0.6
                h_up, c5_up = sa_optimize(h_up, rng, seed_deadline)
                h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
                for beta, max_it in ((5.0, 300), (20.0, 300), (80.0, 400),
                                     (200.0, 500), (500.0, 600), (1000.0, 800)):
                    if time.time() >= deadline:
                        break
                    try:
                        h_up = slsqp_optimize(h_up, beta, rng, max_iter=max_it)
                    except Exception:
                        continue
                h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
                c5_up = compute_c5(h_up)
                if c5_up < best_c5:
                    best_h, best_c5, best_n = h_up, c5_up, n_up

    # ----- fallback if nothing succeeded -----
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.tile([1.0, 0.0], n // 2).astype(float)
        best_c5 = compute_c5(best_h)

    # ----- final safety adjustments -----
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_n)
