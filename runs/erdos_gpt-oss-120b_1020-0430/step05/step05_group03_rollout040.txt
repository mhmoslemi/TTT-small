def lbfgsb_obj(x):
    beta = _lbfgsb_beta
    target = _lbfgsb_target
    M = _lbfgsb_M
    r = np.correlate(x, x, mode='full')
    # compute logsumexp
    # use np.max for stability
    max_r = np.max(-beta * r)
    # compute sumexp = sum exp(-beta r - max_r)
    sumexp = np.sum(np.exp(-beta * r - max_r))
    f = (max_r + np.log(sumexp)) / beta
    # penalty
    diff = np.sum(x) - target
    f += M * diff * diff
    return f

def lbfgsb_grad(x):
    beta = _lbfgsb_beta
    target = _lbfgsb_target
    M = _lbfgsb_M
    n = x.size
    r = np.correlate(x, x, mode='full')
    # w = exp(-beta * r) / sumexp
    exp_term = np.exp(-beta * r)
    sumexp = np.sum(exp_term)
    w = exp_term / sumexp
    # w_pos is w for shifts >=0: indexes from n-1 to end
    w_pos = w[(n-1):]   # length n
    conv = np.convolve(w_pos, x, mode='full')[:n]
    grad_f = -2.0 * conv
    diff = np.sum(x) - target
    grad = grad_f + 2.0 * M * diff * np.ones_like(x)
    return grad

import time, numpy as np
from scipy.optimize import minimize

# ----------- globals for L‑BFGS ----------- #
_lbfgsb_beta = 1.0
_lbfgsb_target = 1.0
_lbfgsb_M = 1.0

# ----------- utilities ----------- #
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a step function on [0,2] with n samples."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project h onto {0≤h≤1, Σh = n/2} by simple redistribution."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # distribute the deficit/excess uniformly among free coordinates
    free = (h > 0.0) & (h < 1.0)
    while abs(diff) > 1e-12 and free.any():
        delta = diff / free.sum()
        # clip the update to stay inside [0,1]
        upd = np.clip(delta, -h[free], 1.0 - h[free])
        h[free] += upd
        diff = target - h.sum()
        free = (h > 0.0) & (h < 1.0)
    # if still off by a tiny amount adjust a random coordinate
    if abs(diff) > 1e-12:
        i = rng.integers(0, n)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift attaining max C5)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ----------- L‑BFGS objective & gradient (smooth min of autocorrelation) ----------- #
def _lbfgsb_obj(x: np.ndarray) -> float:
    beta = _lbfgsb_beta
    target = _lbfgsb_target
    M = _lbfgsb_M
    r = np.correlate(x, x, mode='full')
    neg_beta_r = -beta * r
    max_val = np.max(neg_beta_r)
    sumexp = np.exp(neg_beta_r - max_val).sum()
    f = (max_val + np.log(sumexp)) / beta
    diff = x.sum() - target
    f += M * diff * diff
    return f


def _lbfgsb_grad(x: np.ndarray) -> np.ndarray:
    beta = _lbfgsb_beta
    target = _lbfgsb_target
    M = _lbfgsb_M
    n = x.size
    r = np.correlate(x, x, mode='full')
    neg_beta_r = -beta * r
    max_val = np.max(neg_beta_r)
    exp_shifted = np.exp(neg_beta_r - max_val)
    sumexp = exp_shifted.sum()
    w = exp_shifted / sumexp               # w_k = exp(-β r_k) / Σexp
    w_pos = w[(n - 1):]                    # shifts k ≥ 0
    conv = np.convolve(w_pos, x, mode='full')[:n]
    grad_f = -2.0 * conv
    diff = x.sum() - target
    grad = grad_f + 2.0 * M * diff * np.ones_like(x)
    return grad


# ----------- binary swap hill‑climb (uses exact autocorrelation) ----------- #
def _binary_swap_hill_climb(h: np.ndarray, R: np.ndarray,
                            max_iter: int, rng: np.random.Generator,
                            deadline: float) -> tuple:
    """Greedy swaps that strictly increase the minimal autocorrelation."""
    n = h.size
    s = np.arange(-n + 1, n)
    best_Rmin = np.min(R)
    it = 0
    while it < max_iter and time.time() < deadline:
        idx_min = np.argmin(R)
        shift = idx_min - (n - 1)                # integer shift of worst overlap
        # contribution of each index to R at this shift
        contrib = np.zeros(n, dtype=int)
        if shift >= 0:
            contrib[shift:] += h[:n - shift]
            contrib[:n - shift] += h[shift:]
        else:
            t = -shift
            contrib[:n - t] += h[t:]
            contrib[t:] += h[:n - t]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(contrib[ones])]
        j = zeros[np.argmax(contrib[zeros])]
        if i == j:
            break
        # compute delta to R caused by swapping i↔j
        left_i = i - s
        mask = (left_i >= 0) & (left_i < n)
        di = np.zeros_like(s, dtype=int)
        di[mask] += h[left_i[mask]]
        right_i = i + s
        mask = (right_i >= 0) & (right_i < n)
        di[mask] += h[right_i[mask]]
        left_j = j - s
        mask = (left_j >= 0) & (left_j < n)
        dj = np.zeros_like(s, dtype=int)
        dj[mask] += h[left_j[mask]]
        right_j = j + s
        mask = (right_j >= 0) & (right_j < n)
        dj[mask] += h[right_j[mask]]
        R_new = R + (dj - di)
        new_min = np.min(R_new)
        if new_min > best_Rmin + 1e-12:
            h[i] = 0
            h[j] = 1
            R = R_new
            best_Rmin = new_min
        else:
            # occasional random swap to escape plateaus
            i_rand = rng.choice(ones)
            j_rand = rng.choice(zeros)
            left_i = i_rand - s
            mask = (left_i >= 0) & (left_i < n)
            di = np.zeros_like(s, dtype=int)
            di[mask] += h[left_i[mask]]
            right_i = i_rand + s
            mask = (right_i >= 0) & (right_i < n)
            di[mask] += h[right_i[mask]]
            left_j = j_rand - s
            mask = (left_j >= 0) & (left_j < n)
            dj = np.zeros_like(s, dtype=int)
            dj[mask] += h[left_j[mask]]
            right_j = j_rand + s
            mask = (right_j >= 0) & (right_j < n)
            dj[mask] += h[right_j[mask]]
            R_new = R + (dj - di)
            new_min = np.min(R_new)
            if new_min > best_Rmin + 1e-12:
                h[i_rand] = 0
                h[j_rand] = 1
                R = R_new
                best_Rmin = new_min
        it += 1
    return h, R, best_Rmin

            best_h, best_c = h_new, c_new

    # --- Phase B: binary hill‑climb via greedy swaps ---------------
    # turn best_h into a balanced binary vector
    binary_h = (best_h >= 0.5).astype(float)
    target_ones = n // 2
    cur_ones = int(binary_h.sum())
    if cur_ones > target_ones:
        idx = np.where(binary_h == 1)[0]
        to_off = cur_ones - target_ones
        off = rng.choice(idx, size=to_off, replace=False)
        binary_h[off] = 0
    elif cur_ones < target_ones:
        idx = np.where(binary_h == 0)[0]
        to_on = target_ones - cur_ones
        on = rng.choice(idx, size=to_on, replace=False)
        binary_h[on] = 1
    R = np.correlate(binary_h, binary_h, mode='full')
    c_bin = (2.0 / n) * (1.0 - np.min(R))
    if c_bin < best_c:
        best_h, best_c = binary_h.copy(), c_bin
    # greedy swaps
    deadline = start + 0.85 * budget_s
    binary_h, R, Rmin = _binary_swap_hill_climb(binary_h, R,
                                                max_iter=50000,
                                                rng=rng,
                                                deadline=deadline)
    c_bin2 = (2.0 / n) * (1.0 - Rmin)
    if c_bin2 < best_c:
        best_h, best_c = binary_h.copy(), c_bin2

    # --- Phase C: final tiny stochastic guided perturbations -------
    while time.time() - start < 0.99 * budget_s:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
