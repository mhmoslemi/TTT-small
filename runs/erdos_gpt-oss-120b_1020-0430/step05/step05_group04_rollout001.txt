import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_worst_shift(h):
    n = h.size
    corr = np.correlate(h, 1.0-h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n-1)  # shift value (negative or positive)
    # compute gradient g_i
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0*h
        return g, k
    if k > 0:
        # for i from 0 to n-k-1: +1 - h[i+k]
        g[:n-k] += 1.0 - h[k:]
        # for i from k to n-1: -h[i-k]
        g[k:] -= h[:n-k]
    else:  # k < 0
        k_abs = -k
        # shift left: same as above with roles swapped
        # For i from k_abs to n-1: +1 - h[i - k_abs]
        g[k_abs:] += 1.0 - h[:n-k_abs]
        # For i from 0 to n-k_abs-1: -h[i + k_abs]
        g[:n-k_abs] -= h[k_abs:]
    return g, k

def pair_descent(h, rng, max_iter=2000, step_max=0.05):
    n = h.size
    best_c = compute_c5(h)
    for it in range(max_iter):
        g, k = gradient_worst_shift(h)
        # find indices with positive and negative gradient
        pos_idx = np.where(g > 1e-12)[0]
        neg_idx = np.where(g < -1e-12)[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            break
        i = pos_idx[rng.integers(0, pos_idx.size)]
        j = neg_idx[rng.integers(0, neg_idx.size)]
        # allowable delta: limited by step_max, g_i, -g_j, and bound constraints
        delta = min(step_max, g[i], -g[j], h[i], 1.0 - h[j])
        if delta <= 0:
            continue
        h[i] -= delta
        h[j] += delta
        # ensure bounds (should hold)
        # compute C5
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
        else:
            # revert if not improved? Maybe accept with small prob.
            # For simplicity, revert if not improve.
            h[i] += delta
            h[j] -= delta
        # optional occasional random perturb
        if it % 200 == 0:
            # small random jitter preserving sum (swap small amounts)
            idx1 = rng.integers(0, n)
            idx2 = rng.integers(0, n-1)
            if idx2 >= idx1:
                idx2 += 1
            # propose epsilon
            eps = (rng.random()*2-1)*step_max*0.2
            # clip to maintain bounds
            eps = max(-h[idx1], min(eps, 1.0 - h[idx1]))
            eps = max(-(1.0 - h[idx2]), min(eps, h[idx2]))
            if abs(eps) > 1e-12:
                h[idx1] += eps
                h[idx2] -= eps
                # maybe accept if improves
                c_new = compute_c5(h)
                if c_new < best_c:
                    best_c = c_new
                else:
                    # revert
                    h[idx1] -= eps
                    h[idx2] += eps
    return h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # ensure sum exactly (post ajuste)
    diff = CURRENT_TARGET - x.sum()
    # same adjust_sum_exact style: modify random indices to fix sum
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=30):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        g, k = gradient_worst_shift(h)
        # scale step
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # fix sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98  # a bit headroom
    
    # Determine target n
    target_n = 720
    if target_n % 2 == 1:
        target_n += 1
    # Build initial h
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        # interpolate to target_n
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h0 = np.interp(xt, xs, parent)
    else:
        # random start
        h0 = rng.random(target_n)
        # scale to correct sum
        target = target_n/2.0
        h0 *= target/h0.sum()
    # ensure bounds and sum exactly
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = adjust_sum_exact(h0, rng)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    best_n = target_n
    
    # Phase 1: gradient refine
    h = gradient_refine(best_h.copy(), rng, step=0.02, max_iter=30)
    c = compute_c5(h)
    if c < best_c:
        best_h, best_c = h, c
    
    # Phase 2: SLSQP cascade
    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]
    for beta in betas:
        if time.time() > deadline:
            break
        h = slsqp_optimize(best_h.copy(), beta, rng, max_iter=400 if beta < 200 else 600)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c
    
    # Phase 3: pair descent
    if time.time() < deadline:
        h, c = pair_descent(best_h.copy(), rng, max_iter=6000, step_max=0.03)
        if c < best_c:
            best_h, best_c = h, c
    
    # Phase 4: final gradient refine
    if time.time() < deadline:
        h = gradient_refine(best_h.copy(), rng, step=0.015, max_iter=30)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c
    
    # Optional upsample if still time
    if best_n * 2 <= 1440 and time.time() < deadline:
        n_up = best_n * 2
        # nearest scale: repeat each entry twice
        h_up = np.repeat(best_h, 2)
        h_up = adjust_sum_exact(h_up, rng)
        # run some refinements
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=30)
        for beta in [2000,5000,10000]:
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=500)
        h_up, c_up = pair_descent(h_up, rng, max_iter=4000, step_max=0.02)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up
    
    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    
    return best_h.tolist(), float(final_c), int(best_n)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- globals for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral (discretized)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Redistribute to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth max (log‑sum‑exp) of the overlap vector."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def gradient_worst_shift(h: np.ndarray):
    """Gradient of the currently worst shift (sub‑gradient of max)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    k = idx - (n - 1)  # integer shift
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
        return g, k
    if k > 0:
        # forward part
        g[: n - k] += 1.0 - h[k:]
        g[k:] -= h[: n - k]
    else:  # k < 0
        kk = -k
        g[kk:] += 1.0 - h[: n - kk]
        g[: n - kk] -= h[kk:]
    return g, k


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.02, max_iter: int = 30) -> np.ndarray:
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        g, _ = gradient_worst_shift(h)
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def pair_descent(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iter: int = 4000,
    step_max: float = 0.03,
) -> tuple[np.ndarray, float]:
    """Move mass from positive‑gradient to negative‑gradient positions."""
    n = h.size
    best_c = compute_c5(h)
    for it in range(max_iter):
        g, _ = gradient_worst_shift(h)
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[rng.integers(0, pos.size)]
        j = neg[rng.integers(0, neg.size)]
        delta = min(step_max, g[i], -g[j], h[i], 1.0 - h[j])
        if delta <= 0:
            continue
        h[i] -= delta
        h[j] += delta
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        if it % 200 == 0:
            # small random jitter preserving total mass
            a = rng.integers(0, n)
            b = rng.integers(0, n - 1)
            if b >= a:
                b += 1
            eps = (rng.random() * 2 - 1) * step_max * 0.2
            eps = max(-h[a], min(eps, 1.0 - h[a]))
            eps = max(-(1.0 - h[b]), min(eps, h[b]))
            if abs(eps) > 1e-12:
                h[a] += eps
                h[b] -= eps
                c2 = compute_c5(h)
                if c2 < best_c:
                    best_c = c2
                else:
                    h[a] -= eps
                    h[b] += eps
    return h, best_c


def binary_round(h: np.ndarray) -> np.ndarray:
    """Round to a binary vector with exactly n/2 ones."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h
