def compute_c5_fft(h):
    n = h.shape[0]
    size = 2*n
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    conv = np.fft.irfft(H * G, size)[:2*n-1]
    return conv.max() * (2.0 / n)
```

def fractional_sa(h, rng, deadline, max_attempts=200000):
    cur = h.copy()
    cur_c = compute_c5_fft(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline and it < max_attempts:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            it += 1
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5_fft(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 20000:
            break
    return best, best_c
```

def smooth_gradient_refine(h, rng, beta=5000, step=0.02, max_iter=20):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        # extract relevant part indices - (n-1) offset
        corr_vals = corr
        weights = np.exp(beta * corr_vals - np.max(beta * corr_vals))
        w = weights / weights.sum()
        # compute gradient = sum_k w_k * grad_corr_k
        # grad_corr_k[i] = 1 - h[i+k] - h[i-k] (with bounds check)
        g = np.zeros_like(h)
        for k in range(-n+1, n):
            wk = w[k + n - 1]
            if wk == 0:
                continue
            # For i where i+k in range
            if k >= 0:
                g[:n - k] += wk * (1 - h[k:])
                g[k:]    -= wk * h[:n - k]
            else:
                kp = -k
                g[kp:]   += wk * (1 - h[:n - kp])
                g[:kp]   -= wk * h[kp:]
        # update
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5_fft(h):
    """Maximum overlap C5 using FFT (linear correlation)."""
    n = h.shape[0]
    size = 2 * n
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    conv = np.fft.irfft(H * G, size)[: 2 * n - 1]
    return float(conv.max() * (2.0 / n))


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max via log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return float((m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA)


def slsqp_optimize(h0, beta, rng, max_iter=600):
    """SLSQP optimization of the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 20000:
            break
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.93  # margin for final steps

    best_h = None
    best_c = np.inf
    best_n = None

    # --------------------------------------------------------------------
    # Base discretisation (n = 360)
    # --------------------------------------------------------------------
    n = 360
    start_vecs = []

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp)
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)
        else:
            start_vecs.append(parent)
            bin_vec = (parent >= 0.5).astype(float)
            bin_vec = adjust_sum_exact(bin_vec, rng)
            start_vecs.append(bin_vec)

    # random binary
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    start_vecs.append(bin_rand)

    # random fractional uniform
    rand_frac = rng.random(n)
    rand_frac *= (n / 2) / rand_frac.sum()
    rand_frac = np.clip(rand_frac, 0.0, 1.0)
    rand_frac = adjust_sum_exact(rand_frac, rng)
    start_vecs.append(rand_frac)

    # random fractional biased
    rand_frac2 = rng.random(n) ** 2
    rand_frac2 *= (n / 2) / rand_frac2.sum()
    rand_frac2 = np.clip(rand_frac2, 0.0, 1.0)
    rand_frac2 = adjust_sum_exact(rand_frac2, rng)
    start_vecs.append(rand_frac2)

    # simple periodic pattern
    pattern = np.tile([1, 0, 0, 1, 0, 0], n // 6 + 1)[:n].astype(float)
    pattern = adjust_sum_exact(pattern, rng)
    start_vecs.append(pattern)

    betas = [10, 50, 200, 800, 3200, 12800, 51200]

    for h0 in start_vecs:
        if time.time() > deadline:
            break
        h = np.clip(h0, 0.0, 1.0)
        h = adjust_sum_exact(h, rng)

        # binary SA if applicable
        if np.all((h == 0) | (h == 1)):
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
            h_bin, c_bin = binary_sa_hill_climb(h.copy(), rng, sub_dead)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin.copy(), c_bin, n
            h = h_bin.astype(float)

        # fractional SA
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
        h_frac, c_frac = fractional_sa_hill_climb(h.copy(), rng, sub_dead)
        if c_frac < best_c:
            best_h, best_c, best_n = h_frac.copy(), c_frac, n
        h = h_frac

        # gradient refinement
        h = gradient_refine(h, rng, step=0.03, max_iter=12)
        c = compute_c5_fft(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # SLSQP cascade
        for beta in betas:
            if time.time() > deadline:
                break
            max_iter = 600 if beta < 800 else 900 if beta < 12800 else 1200
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            c = compute_c5_fft(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # final gradient step
        h = gradient_refine(h, rng, step=0.02, max_iter=15)
        c = compute_c5_fft(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # --------------------------------------------------------------------
    # Upsampling stages (720 → 1440)
    # --------------------------------------------------------------------
    up_factors = [2, 2]  # 360→720→1440
    cur_h = best_h if best_h is not None else np.zeros(n)
    cur_n = best_n if best_n is not None else n

    for factor in up_factors:
        if time.time() > deadline:
            break
        new_n = cur_n * factor
        if new_n > 1440:
            break
        xs = np.linspace(0.0, 2.0, cur_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, new_n, endpoint=False)
        h_up = np.interp(xt, xs, cur_h)
        h_up = adjust_sum_exact(h_up, rng)

        # gradient + SA
        h_up = gradient_refine(h_up, rng, step=0.03, max_iter=15)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        h_up, _ = fractional_sa_hill_climb(h_up, rng, sub_dead)

        # SLSQP cascade again
        for beta in betas:
            if time.time() > deadline:
                break
            max_iter = 600 if beta < 800 else 900 if beta < 12800 else 1200
            try:
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=max_iter)
            except Exception:
                continue

        # extra fine‑tuned SLSQP passes (higher beta)
        extra_betas = [102400, 204800]
        for beta in extra_betas:
            if time.time() > deadline:
                break
            try:
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=1200)
            except Exception:
                pass

        # final gradient
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        c_up = compute_c5_fft(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, new_n

        cur_h, cur_n = h_up, new_n

    # --------------------------------------------------------------------
    # Final adjustment and return
    # --------------------------------------------------------------------
    if best_h is None:
        best_n = 180
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        best_h = adjust_sum_exact(best_h, rng)
        best_c = compute_c5_fft(best_h)
    else:
        best_h = adjust_sum_exact(best_h, rng)
        best_c = compute_c5_fft(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h.tolist(), float(best_c), int(best_n)
