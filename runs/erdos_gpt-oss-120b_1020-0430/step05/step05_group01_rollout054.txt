def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / h.size))


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # random adjustments
    for _ in range(n * 10):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # deterministic sweep if still off
    if abs(diff) > 1e-12:
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift k that attains the maximum overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer a small ε from a max‑gradient index to a min‑gradient index."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def batch_transfer(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, frac: float):
    """Move ε = frac * max_eps from positive‑gradient to negative‑gradient indices."""
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return None
    P = grad[pos].sum()
    N = -grad[neg].sum()
    max_eps = min(P, N)
    eps = max_eps * frac
    if eps <= 0.0:
        return None
    delta_pos = eps * (grad[pos] / P) if P > 0 else np.zeros_like(pos, dtype=float)
    delta_neg = eps * ((-grad[neg]) / N) if N > 0 else np.zeros_like(neg, dtype=float)
    # respect bounds by scaling down if needed
    cap_pos = h[pos]            # can subtract at most this
    cap_neg = 1.0 - h[neg]      # can add at most this
    ratio_pos = np.where(delta_pos > 0, cap_pos / delta_pos, np.inf)
    ratio_neg = np.where(delta_neg > 0, cap_neg / delta_neg, np.inf)
    scale = min(1.0, ratio_pos.min(), ratio_neg.min())
    delta_pos *= scale
    delta_neg *= scale
    nh = h.copy()
    nh[pos] -= delta_pos
    nh[neg] += delta_neg
    # tiny numerical drift correction
    diff = (h.size / 2.0) - nh.sum()
    if abs(diff) > 1e-12:
        nh = enforce_sum(nh, rng)
    return nh


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.01, max_iter: int = 8) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple correction
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# soft‑max SLSQP utilities (optional final polishing)
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, rng)
    return x

# Main routine
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initialise h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        n = 512
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # parameters for batch transfer
    fracs = (0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005)

    # main search loop
    while True:
        now = time.time()
        if now - start > budget_s * 0.95:
            break

        # occasional full restart to diversify
        if rng.random() < 0.02:
            h = np.zeros(n)
            h[rng.choice(n, n // 2, replace=False)] = 1.0

        grad = worst_gradient(h)
        improved = False

        for f in fracs:
            nh = batch_transfer(h, grad, rng, f)
            if nh is None:
                continue
            c = compute_c5(nh)
            if c + 1e-12 < best_c:
                best_c, best_h, h = c, nh.copy(), nh
                improved = True
                break

        if not improved:
            # small random guided step to escape flat regions
            h = guided_perturb(h, grad, rng, scale=0.02)
            h = enforce_sum(h, rng)

    # final polishing: a few SLSQP passes (if time permits)
    for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0):
        if time.time() - start > budget_s * 0.99:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=300)
        best_c = compute_c5(best_h)

    # final gradient polish
    best_h = gradient_refine(best_h, rng, step=0.005, max_iter=12)
    best_c = compute_c5(best_h)

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
