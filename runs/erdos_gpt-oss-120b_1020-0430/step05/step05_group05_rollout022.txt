import time
import numpy as np

def _corr_counts(h):
    """Full correlation of h with (1-h); returns integer counts."""
    return np.correlate(h, 1.0 - h, mode="full")

def _shift_and_max(h):
    """Return (shift, max_count) where shift = argmax - (n-1)."""
    corr = _corr_counts(h)
    max_cnt = int(corr.max())
    shift = int(np.argmax(corr) - (h.size - 1))
    return shift, max_cnt

def _bad_and_zero(h, shift):
    """Indices i where (h[i]=1, h[i+shift]=0) and zeros where (h[i]=0, h[i+shift]=1)."""
    n = h.size
    if shift >= 0:
        i = np.arange(0, n - shift)
        bad = i[(h[i] == 1) & (h[i + shift] == 0)]
        zero = i[(h[i] == 0) & (h[i + shift] == 1)]
    else:
        s = -shift
        i = np.arange(s, n)
        bad = i[(h[i] == 1) & (h[i - s] == 0)]
        zero = i[(h[i] == 0) & (h[i - s] == 1)]
    return bad, zero

        best_h = h.copy()
    else:
        h = best_h.copy()

    best_max_cnt = max(best_max_cnt, 0)
    stagnation = 0
    iteration = 0
    target_cnt = int(np.floor(0.3808 * half + 1e-12))  # ≤68 for n=360

    while time.time() < deadline:
        shift, max_cnt = _shift_and_max(h)

        # record best
        if max_cnt < best_max_cnt:
            best_max_cnt = max_cnt
            best_h = h.copy()
        if max_cnt <= target_cnt:
            best_max_cnt = max_cnt
            best_h = h.copy()
            break

        # try a targeted swap that does not increase the worst count
        bad, zero = _bad_and_zero(h, shift)
        if bad.size and zero.size:
            i = int(rng.choice(bad))
            j = int(rng.choice(zero))
            h[i], h[j] = h[j], h[i]
            new_cnt = int(_corr_counts(h).max())
            if new_cnt <= max_cnt:
                max_cnt = new_cnt
                stagnation = 0
                if new_cnt < best_max_cnt:
                    best_max_cnt = new_cnt
                    best_h = h.copy()
                continue
            # revert if not good
            h[i], h[j] = h[j], h[i]

        # random swap with simulated‑annealing acceptance
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i], h[j] = h[j], h[i]
        new_cnt = int(_corr_counts(h).max())
        delta = new_cnt - max_cnt
        # exponential temperature decay
        T = max(1e-5, 0.02 * np.exp(-iteration / 20000.0))
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            max_cnt = new_cnt
            stagnation = 0
            if new_cnt < best_max_cnt:
                best_max_cnt = new_cnt
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            stagnation += 1

        iteration += 1
        # restart if stuck too long
        if stagnation >= 20000:
            h = np.zeros(n, dtype=float)
            h[rng.choice(n, half, replace=False)] = 1.0
            stagnation = 0

    # final values
    final_c = best_max_cnt * dx
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h.tolist(), float(final_c), int(n)
