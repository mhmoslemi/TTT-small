def smooth_max_and_grad(x, beta):
    N = x.shape[0]
    corr = np.correlate(x, 1.0 - x, mode='full')
    maxc = corr.max()
    w_raw = np.exp(beta * (corr - maxc))
    sumw = w_raw.sum()
    # compute conv1 and conv2
    conv1 = np.convolve(w_raw, x, mode='full')[N-1:N-1+N]
    conv2 = np.convolve(w_raw[::-1], x, mode='full')[N-1:N-1+N]
    grad = 1.0 - (conv1 + conv2) / sumw
    smooth_max = maxc + np.log(sumw) / beta
    return smooth_max, grad, corr
```

import time, numpy as np

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def smooth_max_and_grad(x, beta):
    N = x.shape[0]
    corr = np.correlate(x, 1.0 - x, mode='full')
    maxc = corr.max()
    w_raw = np.exp(beta * (corr - maxc))
    sumw = w_raw.sum()
    conv1 = np.convolve(w_raw, x, mode='full')[N-1:N-1+N]
    conv2 = np.convolve(w_raw[::-1], x, mode='full')[N-1:N-1+N]
    grad = 1.0 - (conv1 + conv2) / sumw
    smooth_max = maxc + np.log(sumw) / beta
    return smooth_max, grad, corr

def binary_from_continuous(x):
    N = x.size
    k = N // 2
    idx = np.argpartition(-x, k)[:k]  # top k indices
    h_bin = np.zeros(N, dtype=np.int8)
    h_bin[idx] = 1
    return h_bin

def hill_climb_binary(h_bin, rng, deadline, max_iters=5000):
    N = h_bin.size
    best = h_bin.copy()
    best_corr = np.max(np.correlate(best.astype(np.int32), (1 - best).astype(np.int32), mode='full'))
    cur = h_bin.copy()
    # maintain lists of indices
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    it = 0
    while time.time() < deadline and it < max_iters and len(ones) > 0 and len(zeros) > 0:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur[i] = 0
        cur[j] = 1
        new_corr = np.max(np.correlate(cur.astype(np.int32), (1 - cur).astype(np.int32), mode='full'))
        if new_corr < best_corr:
            best_corr = new_corr
            best = cur.copy()
            # update lists: replace i with j in ones, j with i in zeros
            ones[ones == i] = j
            zeros[zeros == j] = i
        else:
            # revert
            cur[i] = 1
            cur[j] = 0
        it += 1
    return best, best_corr

def continuous_local_search(x, rng, deadline, max_iters=2000, max_delta=0.03):
    N = x.size
    best = x.copy()
    best_c = compute_c5(best)
    cur = x.copy()
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        # find i with cur[i] > max_delta
        i_candidates = np.where(cur > max_delta)[0]
        if i_candidates.size == 0:
            break
        i = rng.choice(i_candidates)
        # find j where cur[j] < 1 - max_delta
        j_candidates = np.where(cur < 1 - max_delta)[0]
        if j_candidates.size == 0:
            break
        j = rng.choice(j_candidates)
        delta = rng.uniform(0, max_delta)
        delta = min(delta, cur[i], 1 - cur[j])
        if delta <= 0:
            continue
        cur[i] -= delta
        cur[j] += delta
        # adjust sum already preserved
        # clip not needed
        c_new = compute_c5(cur)
        if c_new < best_c:
            best_c = c_new
            best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
    return best, best_c

import time
import numpy as np

# ----------------------------------------------------------------------
def adjust_sum_exact(h, rng):
    """Exactly enforce Σh = len(h)/2 by moving mass between coordinates."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

# ----------------------------------------------------------------------
def smooth_max_and_grad(x, beta):
    """Smooth‑max of the overlap integrand and its gradient w.r.t. x."""
    N = x.shape[0]
    corr = np.correlate(x, 1.0 - x, mode="full")
    maxc = corr.max()
    w = np.exp(beta * (corr - maxc))          # raw soft‑max weights, ≥0, max weight = 1
    sumw = w.sum()
    # conv of w with x gives Σ wℓ x_{j-ℓ}; conv of reversed w with x gives Σ wℓ x_{j+ℓ}
    conv1 = np.convolve(w, x, mode="full")[N - 1 : N - 1 + N]
    conv2 = np.convolve(w[::-1], x, mode="full")[N - 1 : N - 1 + N]
    grad = 1.0 - (conv1 + conv2) / sumw
    smooth = maxc + np.log(sumw) / beta
    return smooth, grad, corr

# ----------------------------------------------------------------------
def binary_from_continuous(x):
    """Return a 0/1 vector with exactly N/2 ones (ties broken arbitrarily)."""
    N = x.size
    k = N // 2
    idx = np.argpartition(-x, k)[:k]
    h = np.zeros(N, dtype=np.int8)
    h[idx] = 1
    return h

# ----------------------------------------------------------------------
def hill_climb_binary(h_bin, rng, deadline, max_iters=20000):
    """Swap 1↔0 to lower the exact max overlap; stops on time or iteration limit."""
    N = h_bin.size
    cur = h_bin.copy()
    best = cur.copy()
    best_corr = np.max(
        np.correlate(best.astype(np.int32), (1 - best).astype(np.int32), mode="full")
    )
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    it = 0
    while time.time() < deadline and it < max_iters and len(ones) and len(zeros):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        cur[i] = 0
        cur[j] = 1
        new_corr = np.max(
            np.correlate(cur.astype(np.int32), (1 - cur).astype(np.int32), mode="full")
        )
        if new_corr < best_corr:
            best_corr = new_corr
            best = cur.copy()
            # update index lists
            ones[ones == i] = j
            zeros[zeros == j] = i
        else:
            # revert
            cur[i] = 1
            cur[j] = 0
        it += 1
    return best, best_corr

# ----------------------------------------------------------------------
def continuous_local_search(x, rng, deadline, max_iters=20000, max_delta=0.03):
    """Transfer a small amount of mass between two coordinates if it improves C5."""
    N = x.size
    cur = x.copy()
    best = cur.copy()
    best_c = compute_c5(best)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        # choose donor i with enough mass
        donors = np.where(cur > max_delta)[0]
        if donors.size == 0:
            break
        i = rng.choice(donors)
        recipients = np.where(cur < 1 - max_delta)[0]
        if recipients.size == 0:
            break
        j = rng.choice(recipients)
        delta = rng.uniform(0, max_delta)
        delta = min(delta, cur[i], 1.0 - cur[j])
        if delta <= 0:
            continue
        cur[i] -= delta
        cur[j] += delta
        c_new = compute_c5(cur)
        if c_new < best_c:
            best_c = c_new
            best = cur.copy()
        else:
            cur[i] += delta
            cur[j] -= delta
    return best, best_c

    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_x = None
    best_c = np.inf

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    candidate_ns = [720, 360]                     # larger grids first

        if time.time() > deadline:

        # ----------------------------------------------------------------

        start_vecs = []
        if best_n == N and best_x is not None:

        for _ in range(num_patterns):
            if time.time() > deadline:
                break
            typ = rng.integers(0, 3)
            i = np.arange(N)
            if typ == 0:  # single sine wave

                phase = rng.uniform(0, 2 * np.pi)
                x = 0.5 * (1 + np.sin(2 * np.pi * i / period + phase))
            elif typ == 1:  # sum of two sines
                p1 = rng.uniform(N / 10.0, N / 2.0)
                p2 = rng.uniform(N / 10.0, N / 2.0)

                x = (x - x.min()) / (x.max() - x.min())
            else:  # piecewise constant with few blocks

                bnds = np.sort(rng.choice(N - 1, size=blocks - 1, replace=False)) + 1
                vals = rng.random(blocks)
                x = np.empty(N, dtype=float)
                start = 0
                for b, v in zip(np.append(bnds, N), vals):

            x = x0.copy()
