def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    # while |diff| > tolerance: adjust random entries within bounds.
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(h.size)
        if diff > 0:
            # need to increase h[i]
            incr = min(diff, 1.0 - h[i])
        else:
            incr = max(diff, -h[i])
        if incr == 0:
            continue
        h[i] += incr
        diff = target - h.sum()
    return h
```

    best_h = h.copy()

    max_steps = 200000  # maybe more depending

            break
        progress = step / max_steps
        T = max(1e-12, (cur_c * 0.01) * (1 - progress))
        # pick i, j
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:

        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            continue
        # step size factor decays
        step_factor = max(0.1 * (1 - progress), 0.001)
        delta = rng.random() * max_delta * step_factor
        # apply
        h[i] -= delta
        h[j] += delta
        new_c = compute_c5(h)
        delta_c = new_c - cur_c
        accept = delta_c < 0 or rng.random() < np.exp(-delta_c / T)
        if accept:
            cur_c = new_c

                best_c = cur_c
                best_h = h.copy()
        else:

            h[i] += delta
            h[j] -= delta

    adjust_sum_exact(best_h, rng)
    best_c = compute_c5(best_h)
    return best_h, best_c

def compute_c5(h):
    """Return C5 for vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def max_corr(h):
    """Maximum correlation value (without dx scaling)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def adjust_sum_exact(h, rng):
    """Adjust h in-place so that sum(h) == len(h)/2, keeping 0<=h<=1."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, betas, rng, deadline):
    """Run SLSQP on the smooth‑max surrogate for a list of betas."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    for beta in betas:
        if time.time() >= deadline:
            break
        CURRENT_BETA = beta
        try:
            res = minimize(
                softmax_obj,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 800, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        h = np.clip(res.x, 0.0, 1.0)
        adjust_sum_exact(h, rng)
    return h

def random_binary(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def to_binary(h):
    """Convert a real vector to a binary one by keeping the largest n/2 entries."""
    n = h.size
    idx = np.argsort(h)[-(n // 2) :]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b

def sa_binary(start_h, n, rng, deadline):
    """Simulated annealing on binary vectors (swap 1↔0)."""
    h = start_h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = float(np.max(corr))
    cur_c = cur_max * (2.0 / n)

    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1.0 - h))
    best_h = h.copy()
    best_c = cur_c

    max_steps = 300000
    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        progress = step / max_steps
        T = max(1e-12, (cur_c * 0.01) * (1.0 - progress))

        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # swap
        h[i], h[j] = 0.0, 1.0
        corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(np.max(corr))
        new_c = new_max * (2.0 / n)
        delta_c = new_c - cur_c
        accept = delta_c < 0.0 or rng.random() < np.exp(-delta_c / T)
        if accept:
            cur_c = new_c
            cur_max = new_max
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0  # revert

    return best_h, best_c

def compute_gradient_for_k(h, k):
    """Gradient of f_k = sum_i h_i(1-h_{i+k}) w.r.t. h."""
    n = h.size
    grad = np.ones(n, dtype=float)
    if k >= 0:
        # term h_{i+k}
        right = np.concatenate((np.zeros(k), h[: n - k]))
        left = np.concatenate((h[k:], np.zeros(k)))
    else:
        k = -k
        right = np.concatenate((h[k:], np.zeros(k)))
        left = np.concatenate((np.zeros(k), h[: n - k]))
    grad -= right
    grad -= left
    return grad

def sa_continuous(start_h, n, rng, deadline):
    """Simple simulated annealing on continuous vectors (transfer Δ between two entries)."""
    h = start_h.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    max_steps = 300000
    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        progress = step / max_steps
        T = max(1e-12, (cur_c * 0.01) * (1.0 - progress))
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            continue
        step_factor = max(0.1 * (1.0 - progress), 0.001)
        delta = rng.random() * max_delta * step_factor
        h[i] -= delta
        h[j] += delta
        new_c = compute_c5(h)
        delta_c = new_c - cur_c
        accept = delta_c < 0.0 or rng.random() < np.exp(-delta_c / T)
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
    adjust_sum_exact(best_h, rng)
    best_c = compute_c5(best_h)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # initial vector
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n = 360
        init_h = np.full(n, 0.5, dtype=float)
    n = init_h.size
    adjust_sum_exact(init_h, rng)

    best_h = init_h.copy()
    best_c = compute_c5(best_h)

    # ----- binary simulated annealing -----
    if time.time() < deadline:
        start_bin = to_binary(best_h) if best_c > 0.38 else random_binary(n, rng)
        bin_h, bin_c = sa_binary(start_bin, n, rng, deadline)
        if bin_c < best_c:
            best_h, best_c = bin_h, bin_c

    # ----- continuous simulated annealing (multiple restarts) -----
    for _ in range(3):
        if time.time() >= deadline:
            break
        if time.time() - start_time < budget_s * 0.1:
            start_cont = random_continuous_init(n, rng)
        else:
            start_cont = best_h.copy()
        cont_h, cont_c = sa_continuous(start_cont, n, rng, deadline)
        if cont_c < best_c:
            best_h, best_c = cont_h, cont_c

    # ----- smooth‑max SLSQP refinement -----
    if time.time() < deadline:
        betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]
        refined = slsqp_optimize(best_h.copy(), betas, rng, deadline)
        refined_c = compute_c5(refined)
        if refined_c < best_c:
            best_h, best_c = refined, refined_c

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
