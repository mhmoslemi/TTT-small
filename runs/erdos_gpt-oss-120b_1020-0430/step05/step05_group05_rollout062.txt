def local_optimize_binary(x):
    n = len(x)
    while True:
        O = np.correlate(x, x, mode='full')[n-1:]  # O_k for k=0..n-1
        cur_min = O.min()
        # compute shift matrices
        ones = np.flatnonzero(x)
        zeros = np.flatnonzero(1-x)
        # precompute shift vectors for zeros
        shift_zeros = np.empty((len(zeros), n), dtype=int)
        for idx, j in enumerate(zeros):
            shift_zeros[idx] = np.roll(x, -j)
        best_gain = 0
        best_pair = None
        # iterate ones
        for i in ones:
            shift_i = np.roll(x, -i)  # vector
            base = O - shift_i
            # compute new min for each zero with vectorized across all zeros: 
            # new_arrays = base[None, :] + shift_zeros  # shape (nz, n)
            # new_mins = new_arrays.min(axis=1)
            # But may allocate large memory: nz=180, n=360 -> 64k elements, fine.
            new_mins = (base[None, :] + shift_zeros).min(axis=1)
            # compute gains
            gains = new_mins - cur_min
            # find max
            max_idx = np.argmax(gains)
            if gains[max_idx] > best_gain:
                best_gain = gains[max_idx]
                best_pair = (i, zeros[max_idx])
        if best_gain > 0:
            i, j = best_pair
            x[i] = 0
            x[j] = 1
            continue
        else:
            break
    return x

    while True:
        O = np.fft.ifft(np.fft.fft(x) * np.fft.fft(x)).real
        O = np.rint(O).astype(np.int64)
        cur_min = O.min()

import time
import numpy as np
from scipy.optimize import minimize

# ----- globals used by the SLSQP surrogate -----
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function h on [0,2] discretized."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint: Σx = target."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth max of overlaps using log‑sum‑exp with parameter β."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 1800) -> np.ndarray:
    """Run SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (simple random walk)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.02, max_iter: int = 20) -> np.ndarray:
    """Simple projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:up]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(150):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ----- choose discretisation size -----
    # 720 points give a finer representation and still run fast enough.
    n = 720
    target_sum = n / 2.0

    best_h = None
    best_c = np.inf

    # ----- build initial seeds -----
    seeds = []

    # parent (if supplied) – upsample by repeating each entry.
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            # repeat to reach n points
            repeat = n // parent.size
            parent_up = np.repeat(parent, repeat)
        else:
            parent_up = parent.copy()
        parent_up = np.clip(parent_up, 0.0, 1.0)
        parent_up = adjust_sum_exact(parent_up, rng)
        seeds.append(parent_up)

    # a few random fractional seeds
    for _ in range(12):
        h = rng.random(n)
        h *= target_sum / h.sum()
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum_exact(h, rng)
        seeds.append(h)

    # ----- optimisation loop over seeds -----
    betas = (5., 20., 80., 200., 500., 1000., 2000., 5000., 10000., 20000., 50000., 100000.)

    for base_h in seeds:
        if time.time() > deadline - 5.0:
            break
        h = base_h.copy()
        # cascade of smooth‑max SLSQP runs
        for beta in betas:
            if time.time() > deadline:
                break
            max_iter = 1500 if beta < 200 else 2500
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            if best_c <= 0.3808:
                break
        # gradient polishing
        h = gradient_refine(h, rng, step=0.015, max_iter=30)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # ----- final high‑beta polishing on the best candidate -----
    if best_h is not None and time.time() < deadline - 2.0:
        for beta in (200000., 500000.):
            if time.time() > deadline:
                break
            try:
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=4000)
            except Exception:
                pass
            c = compute_c5(best_h)
            if c < best_c:
                best_c = c
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=40)

    # ----- final clean‑up -----
    if best_h is None:
        # fallback: simple half‑interval
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
