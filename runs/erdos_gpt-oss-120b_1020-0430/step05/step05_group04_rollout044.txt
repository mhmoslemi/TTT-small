import time
import numpy as np
from scipy.optimize import minimize

# ----- globals for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

# ----- core utilities -----
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute entries so that sum(h) == n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=30):
    """Projected sub‑gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:upper]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, deadline):
    """Simulated‑annealing hill climb on binary vectors (swap 0↔1)."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c

def greedy_improve(h, rng, max_iter=30, deadline=None):
    """Local search: swaps that reduce the current worst shift."""
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

# ----- main entry -----
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    # sizes to explore (larger first)
    candidate_ns = [720, 600, 480, 360, 240, 180]
    candidate_ns = [n for n in candidate_ns if n % 2 == 0]

    TARGET = 0.3808

    for n in candidate_ns:
        if time.time() > deadline:
            break

        # number of random starts (fewer for large n)
        max_restarts = 5 if n >= 600 else 8

        for _ in range(max_restarts):
            if time.time() > deadline:
                break

            # ---- initial vector ----
            if parent is not None and parent.size == n:
                h = parent.copy()
            elif parent is not None:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(xt, xs, parent)
                h = np.clip(h, 0.0, 1.0)
                adjust_sum_exact(h, rng)
            else:
                h = rng.random(n)
                adjust_sum_exact(h, rng)

            # ---- continuous refinement ----
            grad_iters = 80 if n >= 600 else 120
            h = gradient_refine(h, rng, step=0.02, max_iter=grad_iters)

            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n
                if best_c <= TARGET:
                    best_h = np.clip(best_h, 0.0, 1.0)
                    adjust_sum_exact(best_h, rng)
                    return best_h.tolist(), float(best_c), int(best_n)

            # ---- binary threshold & SA ----
            idx = np.argpartition(h, -n // 2)[-n // 2:]
            h_bin = np.zeros(n, dtype=float)
            h_bin[idx] = 1.0
            # allocate a slice of remaining time
            rem = deadline - time.time()
            sub_dead = time.time() + min(rem * 0.2, 30.0)
            if sub_dead > deadline:
                sub_dead = deadline
            h_bin_opt, c_bin = binary_sa_hill_climb(h_bin.copy(), rng, sub_dead)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin_opt.copy(), c_bin, n
                if best_c <= TARGET:
                    best_h = np.clip(best_h, 0.0, 1.0)
                    adjust_sum_exact(best_h, rng)
                    return best_h.tolist(), float(best_c), int(best_n)

            # greedy polish
            if np.all((h_bin_opt == 0) | (h_bin_opt == 1)):
                h_bin_opt = greedy_improve(h_bin_opt, rng, max_iter=30, deadline=deadline)
                c2 = compute_c5(h_bin_opt)
                if c2 < best_c:
                    best_h, best_c, best_n = h_bin_opt.copy(), c2, n
                    if best_c <= TARGET:
                        best_h = np.clip(best_h, 0.0, 1.0)
                        adjust_sum_exact(best_h, rng)
                        return best_h.tolist(), float(best_c), int(best_n)

            # ---- small‑n SLSQP cascade (optional) ----
            if n <= 360:
                betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
                         2000.0, 5000.0, 10000.0, 20000.0)
                for beta in betas:
                    if time.time() > deadline:
                        break
                    try:
                        h = slsqp_optimize(h, beta, rng, max_iter=200)
                    except Exception:
                        continue
                    c = compute_c5(h)
                    if c < best_c:
                        best_h, best_c, best_n = h.copy(), c, n
                        if best_c <= TARGET:
                            best_h = np.clip(best_h, 0.0, 1.0)
                            adjust_sum_exact(best_h, rng)
                            return best_h.tolist(), float(best_c), int(best_n)

    # ----- final up‑sampling & polish -----
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 720
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        adjust_sum_exact(h_up, rng)

        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=100)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up
            if best_c <= TARGET:
                best_h = np.clip(best_h, 0.0, 1.0)
                adjust_sum_exact(best_h, rng)
                return best_h.tolist(), float(best_c), int(best_n)

        # binary version of up‑sampled vector
        idx = np.argpartition(h_up, -n_up // 2)[-n_up // 2:]
        h_bin = np.zeros(n_up, dtype=float)
        h_bin[idx] = 1.0
        rem = deadline - time.time()
        sub_dead = time.time() + min(rem * 0.2, 30.0)
        if sub_dead > deadline:
            sub_dead = deadline
        h_bin_opt, c_bin = binary_sa_hill_climb(h_bin.copy(), rng, sub_dead)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin_opt.copy(), c_bin, n_up
            if best_c <= TARGET:
                best_h = np.clip(best_h, 0.0, 1.0)
                adjust_sum_exact(best_h, rng)
                return best_h.tolist(), float(best_c), int(best_n)

    # ----- fallback -----
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_n = n
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
