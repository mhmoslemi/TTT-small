from scipy.special import logsumexp

def smooth_max_obj_and_grad(h, beta):
    O = np.correlate(h, 1-h, mode='full')
    scaled_O = beta * O
    lse = logsumexp(scaled_O)
    obj = lse / beta
    # compute weights
    w = np.exp(scaled_O - lse)  # shape 2n-1
    # compute gradient
    n = h.shape[0]
    grad = np.zeros_like(h)
    one_minus_h = 1 - h
    # vectorized loop:
    for idx in range(2*n-1):
        s = idx - (n-1)
        w_s = w[idx]
        if s >= 0:
            if s < n:
                grad[:n-s] += w_s * one_minus_h[s:]
                grad[s:] += w_s * (-h[:n-s])
        else:
            d = -s
            if d < n:
                grad[d:] += w_s * one_minus_h[:n-d]
                grad[:n-d] += w_s * (-h[d:])
    return obj, grad
```

def smooth_max_grad(h, beta):
    O = np.correlate(h, 1-h, mode='full')
    scaled_O = beta*O

def slsqp_optimize_grad(h0, beta, rng, max_iter=400):
    n = h0.size
    target = n/2.0
    bounds = [(0.0, 1.0)]*n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x)-target,
            'jac': lambda x: np.ones_like(x)}
    res = minimize(

         h0,

         jac=lambda x: smooth_max_grad(x, beta),
         bounds=bounds,
         constraints=cons,

    )
    x = np.clip(res.x,0,1)
    adjust_sum_exact(x, rng)

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def smooth_max_obj(h, beta):
    O = np.correlate(h, 1.0 - h, mode="full")
    return logsumexp(beta * O) / beta

def smooth_max_grad(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * O
    lse = logsumexp(scaled)
    w = np.exp(scaled - lse)            # weights, sum to 1
    grad = np.zeros_like(h)
    one_minus_h = 1.0 - h
    # loop over all shifts
    for idx, s in enumerate(range(-(n - 1), n)):
        w_s = w[idx]
        if s >= 0:
            if s < n:
                grad[: n - s] += w_s * one_minus_h[s:]
                grad[s:]      += w_s * (-h[: n - s])
        else:
            d = -s
            if d < n:
                grad[d:]      += w_s * one_minus_h[: n - d]
                grad[: n - d] += w_s * (-h[d:])
    return grad

def slsqp_optimize_grad(h0, beta, rng, max_iter=300):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {
        "type": "eq",
        "fun": lambda x: np.sum(x) - target,
        "jac": lambda x: np.ones_like(x),
    }
    res = minimize(
        lambda x: smooth_max_obj(x, beta),
        h0,
        method="SLSQP",
        jac=lambda x: smooth_max_grad(x, beta),
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    base_n = 360                     # keep <1000 samples
    best_h = None
    best_c = np.inf
    best_n = None

    # use provided parent if available, upsample to base_n if needed
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != base_n:
            x_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            x_target = np.linspace(0.0, 2.0, base_n, endpoint=False)
            parent = np.interp(x_target, x_parent, parent)
        parent = np.clip(parent, 0.0, 1.0)
        adjust_sum_exact(parent, rng)
        c_parent = compute_c5(parent)
        best_h, best_c, best_n = parent, c_parent, base_n

    # beta schedule for smooth‑max annealing
    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]

    # random‑restart loop
    while time.time() < deadline:
        # ---- generate a feasible start vector ----
        mode = rng.integers(0, 3)
        if mode == 0:
            # random binary with exact half ones
            h0 = np.zeros(base_n)
            ones = rng.choice(base_n, size=base_n // 2, replace=False)

        elif mode == 1:
