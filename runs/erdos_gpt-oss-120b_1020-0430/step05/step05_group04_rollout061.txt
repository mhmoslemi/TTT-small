def optimize_n(n, rng, deadline):
    best_h = None
    best_c = np.inf
    n_targetsum = n/2

    # Starting vectors: if initial_h_values and n equals its size, include parent; else L interpolation.
    start_vecs = []

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size == n:
            start_vecs.append(parent.copy())
        # interpolated parent if sizes differ
        if parent.size != n:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp)

    # Random binary
    for _ in range(2):
        h_bin = np.zeros(n)
        h_bin[rng.choice(n, n//2, replace=False)] = 1.0
        start_vecs.append(h_bin.astype(float))

    # Random fractional
    for _ in range(2):
        h_rand = rng.random(n)
        h_rand *= (n/2) / h_rand.sum()
        adjust_sum_exact(h_rand, rng)
        start_vecs.append(h_rand)

    for h0 in start_vecs:
        if time.time() > deadline:
            break
        # Ensure sum correct and clip
        h = np.clip(h0,0,1)
        adjust_sum_exact(h, rng)

        # gradient
        h = gradient_refine(h, rng, step=0.02, decay=0.85, max_iter=50)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # SLSQP cascade
        betas = [1,5,20,80,200,500,2000,5000,20000,50000]
        for beta in betas:
            if time.time() > deadline:
                break
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=800 if beta>=200 else 500)
            except Exception:
                continue
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

        # final gradient
        h = gradient_refine(h, rng, step=0.015, decay=0.85, max_iter=70)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # binary conversion
        h_bin = (h >= 0.5).astype(float)
        # ensure sum correct: adjust by picking top values
        if h_bin.sum() != n/2:
            # sort indices by h descending
            idx = np.argsort(-h)
            h_bin[:] = 0
            h_bin[idx[:n//2]] = 1.0
        adjust_sum_exact(h_bin, rng)

        # binary SA
        # allocate a fraction of remaining time
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        h_bin_opt, c_bin = binary_sa_hill_climb(h_bin.copy(), rng, sub_dead)
        if c_bin < best_c:
            best_h, best_c = h_bin_opt.copy(), c_bin

        # greedy improvement
        if np.all((h_bin_opt == 0) | (h_bin_opt == 1)):
            h_bin_opt = greedy_improve(h_bin_opt, rng, max_iter=200, deadline=deadline)
            c_g = compute_c5(h_bin_opt)
            if c_g < best_c:
                best_h, best_c = h_bin_opt.copy(), c_g

    return best_h, best_c

def gradient_refine(h, rng, step0=0.02, decay=0.85, max_iter=50):
    n = h.size
    target = n/2.0
    step = step0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        # compute gradient g based on worst shift
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            # correlation h_i*(1-h_{i+k}) sum.
            # derivative of each term w.r.t h_i: 1 - h_{i+k}
            # derivative w.r.t h_{i+k}: -h_i
            # So for i in [0, n-1-k]: add 1 - h_{i+k}
            # for i+k in [k, n-1]: derivative -h_{i}
            # We'll accumulate.
            if k > 0:
                # forward shift
                g[:n - k] += 1.0 - h[k:]
                g[k:] -= h[:n - k]
            else:
                # negative shift, k < 0
                kk = -k
                g[kk:] += 1.0 - h[:n - kk]
                g[:n - kk] -= h[kk:]
        # Update
        h = h - step * g
        # Clip
        h = np.clip(h, 0.0, 1.0)
        # Adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # Simple redistribution: adjust a random index by diff if possible, else many attempts
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])

    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    # compute weights
    max_corr = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_corr))

    # compute gradient

    # loop over shifts s = - (n-1) ... (n-1)
    for s in range(-n+1, n):
        w_s = w[s + n - 1]
        # forward part: for j where j+s in range
        if s >= 0:

            j_end = n - s
            g[:j_end] += w_s * (1.0 - h[s:])  # (1 - h_{j+s})

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute values to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step0=0.02, decay=0.85, max_iter=50):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    step = step0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        elif k > 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk:]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        adjust_sum_exact(h, rng)
        step *= decay
    return h

def smooth_max_gradient(h, beta):
    """Gradient of (1/β)·log‑sum‑exp(β·corr) with respect to h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)          # length 2n‑1
    g = np.zeros_like(h)
    w0 = w[n - 1]
    g += w0 * (1.0 - 2.0 * h)                # diagonal terms
    for i in range(n):
        hi = h[i]
        for j in range(i + 1, n):
            s = j - i
            w_pos = w[s + n - 1]             # shift  +s
            w_neg = w[-s + n - 1]            # shift  -s
            hj = h[j]
            g[i] += w_pos * (1.0 - hj) - w_neg * hj
            g[j] += w_neg * (1.0 - hi) - w_pos * hi
    return g

def smooth_max_optimize(h, rng, beta_sched, step0, decay, it_per_beta, deadline):

                return h
            g = smooth_max_gradient(h, beta)
            h = h - step * g
            h = np.clip(h, 0.0, 1.0)
            adjust_sum_exact(h, rng)
            step *= decay
    return h

def binary_sa_hill_climb(h_bin, rng, deadline):
    """Simulated‑annealing hill climb on binary vectors (swap 0↔1)."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9998 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 15000:
            break
    return best_h, best_c

def greedy_improve(h, rng, max_iter=200, deadline=None):
    """Local search: swap 1↔0 that reduces the worst shift."""
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        best_pair = None
        best_c = max_c
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:

                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h
