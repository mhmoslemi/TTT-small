def hill_climb_binary(h, start, deadline, rng):
    n = h.size
    idx = np.arange(n)
    # initial correlation via FFT
    H = np.fft.fft(h)
    c = np.fft.ifft(H * np.conj(H)).real
    # Round to nearest integer to avoid small errors
    c = np.rint(c)
    current_min = c.min()
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    init_temp = (c.max() - c.min()) * 0.5
    if init_temp < 1e-12:
        init_temp = 0.5
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta
        ip = (idx + i) % n
        im = (idx - i) % n
        jp = (idx + j) % n
        jm = (idx - j) % n
        delta = -h[ip] - h[im] + h[jp] + h[jm]
        delta[0] = 0.0
        if n % 2 == 0:
            delta[n // 2] *= 0.5
        cand_c = c + delta
        cand_min = cand_c.min()
        accept = False
        if cand_min > current_min + 1e-12:
            accept = True
        else:
            # temperature schedule
            elapsed = time.time() - start
            total = deadline - start
            frac = elapsed / total if total > 0 else 1.0
            temp = init_temp * (1.0 - frac)
            if temp > 1e-12 and rng.random() < np.exp((cand_min - current_min) / temp):
                accept = True
        if accept:
            # apply update
            h[i] = 0.0
            h[j] = 1.0
            c = cand_c
            current_min = cand_min
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
    # after loop, compute final C via compute_c5
    final_c = compute_c5(h)
    return h, final_c

        j = rng.choice(zeros)

        else:

            total = deadline - start

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_binary(h, rng):
    """Force Σh = n/2 while staying in {0,1}."""
    n = h.size
    target = n // 2
    cur = int(round(h.sum()))
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(h == 0)
        if zeros.size:
            chosen = rng.choice(zeros, diff, replace=False)
            h[chosen] = 1.0
    elif diff < 0:
        ones = np.flatnonzero(h == 1)
        if ones.size:
            chosen = rng.choice(ones, -diff, replace=False)
            h[chosen] = 0.0
    # safety loop
    while int(round(h.sum())) != target:
        i = rng.integers(0, n)
        if h[i] == 1 and int(round(h.sum())) > target:
            h[i] = 0.0
        elif h[i] == 0 and int(round(h.sum())) < target:
            h[i] = 1.0
    return h


def binary_from_fractional(frac_h, rng):
    """Take the n/2 largest entries of frac_h (ties broken randomly)."""
    n = frac_h.size
    k = n // 2
    # tiny random perturbation to break exact ties
    pert = rng.random(n) * 1e-12
    idx = np.argpartition(frac_h + pert, -k)[-k:]
    bin_h = np.zeros_like(frac_h)
    bin_h[idx] = 1.0
    adjust_sum_binary(bin_h, rng)
    return bin_h


def hill_climb_binary(h, start, deadline, rng):
    """Simulated‑annealing hill climbing on binary h to maximise min autocorrelation."""
    n = h.size
    idx = np.arange(n)

    # initial cyclic autocorrelation via FFT (c[k] = Σ h[i] h[i+k])
    H = np.fft.fft(h)
    c = np.fft.ifft(H * np.conj(H)).real
    c = np.rint(c)                # should be integer for binary h
    cur_min = c.min()

    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))

    init_temp = (c.max() - c.min()) * 0.5
    if init_temp <= 0.0:
        init_temp = 0.5

    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # vectorised change of autocorrelation caused by swapping i↔j
        ip = (idx + i) % n
        im = (idx - i) % n
        jp = (idx + j) % n
        jm = (idx - j) % n
        delta = -h[ip] - h[im] + h[jp] + h[jm]
        delta[0] = 0.0                     # c[0] must stay constant
        cand_c = c + delta
        cand_c[0] = c[0]

        if n % 2 == 0:
            k = n // 2
            h_new = h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            cand_c[k] = np.dot(h_new, np.roll(h_new, k))

        cand_min = cand_c.min()

        accept = False
        if cand_min > cur_min + 1e-12:
            accept = True
        else:
            elapsed = time.time() - start
            total = deadline - start
            frac = elapsed / total if total > 0 else 1.0
            temp = init_temp * (1.0 - frac)
            if temp > 1e-12 and rng.random() < np.exp((cand_min - cur_min) / temp):
                accept = True

        if accept:
            h[i] = 0.0
            h[j] = 1.0
            c = cand_c
            cur_min = cand_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)

    return h, compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.95   # a little safety margin

    # base size
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
    else:
        parent = None
        n = 360

    best_h = None
    best_c = np.inf

    # ---- generate starting binary seeds ---------------------------------
    seeds = []

    if parent is not None:
        # binary version of the parent (top half entries)
        seeds.append(binary_from_fractional(parent, rng))
        # also keep the parent itself (might be already binary)
        seeds.append(parent.copy())

    for _ in range(3):
        hb = np.zeros(n, dtype=float)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        seeds.append(hb)

    # ---- hill‑climbing on each seed ------------------------------------
    for h0 in seeds:
        if time.time() >= deadline:
            break
        h = h0.copy()
        # ensure binary and exact sum
        if not np.all(np.isin(h, [0.0, 1.0])):
            h = binary_from_fractional(h, rng)
        adjust_sum_binary(h, rng)

        h_opt, c_opt = hill_climb_binary(h, start, deadline, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # ---- optional up‑sampling refinement ---------------------------------
    if best_h is not None and time.time() < deadline and n * 2 <= 720:
        n_up = n * 2
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_frac = np.interp(xt, xs, best_h)          # linear interpolation
        h_up = binary_from_fractional(h_frac, rng)
        adjust_sum_binary(h_up, rng)

        h_up_opt, c_up_opt = hill_climb_binary(h_up, start, deadline, rng)
        if c_up_opt < best_c:
            best_h, best_c, n = h_up_opt.copy(), c_up_opt, n_up

    # ---- final clean‑up ---------------------------------------------------
    if best_h is None:
        # fallback simple half‑interval pattern
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        adjust_sum_binary(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_binary(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
