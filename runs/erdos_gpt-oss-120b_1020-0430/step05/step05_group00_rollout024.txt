    def subgradient_optimize(h0, rng, time_budget, max_iter=100000, step_start=0.6, step_end=0.01):
        start = time.time()
        n = h0.size
        h = np.clip(h0, 0.0, 1.0)
        h = enforce_sum(h, rng)
        best_h = h.copy()
        best_c = compute_c5_fast(h)
        # steps schedule predetermined maybe based on iteration count
        for it in range(max_iter):
            if time.time() - start > time_budget:
                break
            # correlation
            corr = full_corr(h)
            idx = np.argmax(corr)
            O_max = corr[idx] * (2.0 / n)
            if O_max < best_c:
                best_c = O_max
                best_h = h.copy()
            # gradient
            k = abs(idx - (n - 1))
            if k == 0:
                grad = 1.0 - 2.0 * h
            else:
                grad = np.ones_like(h)
                grad[: n - k] -= h[k:]
                grad[k:] -= h[: n - k]
            # step schedule
            step = step_start * ((step_end / step_start) ** (it / max_iter))
            # update
            h = h - step * grad
            # project
            # clip to [0,1]
            np.clip(h, 0.0, 1.0, out=h)
            # enforce sum
            diff = (n / 2.0) - h.sum()
            # We'll differentially adjust with random indices but faster:
            # while diff != 0: adjust mass to/from random index within bounds.
            # Use while loop with limit.
            ctr = 0
            while abs(diff) > 1e-12 and ctr < 1000:
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                if add != 0:
                    h[i] += add
                    diff = (n / 2.0) - h.sum()
                ctr += 1
            # occasionally random perturb for escape
            if (it % 500 == 0) and it > 0:
                # small random noise, project again
                noise = rng.normal(scale=0.01, size=n)
                h += noise
                np.clip(h, 0.0, 1.0, out=h)
                h = enforce_sum(h, rng)
        return best_h, best_c

def subgradient_optimize(h0, rng, time_budget, max_iter=200000, step_start=0.6, step_end=0.01):
    start = time.time()
    h = np.clip(h0, 0.0, 1.0)
    h = enforce_sum(h, rng)
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    for it in range(max_iter):
        if time.time() - start > time_budget:
            break
        corr = full_corr(h)
        idx = np.argmax(corr)
        O_max = corr[idx] * (2.0 / n)
        if O_max < best_c:
            best_c = O_max
            best_h = h.copy()
        k = abs(idx - (n - 1))
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        step = step_start * ((step_end / step_start) ** (it / max_iter))
        h = h - step * grad
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum(h, rng)
        # occasional random jitter
        if (it & 1023) == 0 and it > 0:
            # add small random noise
            noise = rng.normal(scale=0.005, size=n)
            h += noise
            np.clip(h, 0.0, 1.0, out=h)
            h = enforce_sum(h, rng)
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def constraint(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': constraint}
    from scipy.optimize import minimize
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly via random adjustments
    diff = target - x.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        if add != 0:
            x[i] += add
            diff = target - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def full_corr(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (length 2*n-1)."""
    n = h.size
    N = 2 * n
    h_pad = np.concatenate([h, np.zeros(n, dtype=h.dtype)])
    o_pad = np.concatenate([1 - h, np.zeros(n, dtype=h.dtype)])
    H = np.fft.rfft(h_pad)
    O = np.fft.rfft(o_pad)
    corr = np.fft.irfft(H * O.conj(), n=N)
    return corr[: 2 * n - 1]


def compute_c5_fast(h: np.ndarray) -> float:
    """Exact C5 for discrete h (max correlation * 2/n)."""
    n = h.size
    corr = full_corr(h)
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project onto {x∈[0,1]^n, sum x = n/2} by random mass transfers."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h

    h0: np.ndarray,
    rng: np.random.Generator,
    time_budget: float,
    max_iter: int = 200000,
    step_start: float = 0.6,
    step_end: float = 0.01,
) -> tuple[np.ndarray, float]:

    start = time.time()
    h = np.clip(h0, 0.0, 1.0)
    h = enforce_sum(h, rng)
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fast(h)

        if time.time() - start > time_budget:
            break
        corr = full_corr(h)
        idx = np.argmax(corr)
        O_max = corr[idx] * (2.0 / n)
        if O_max < best_c:
            best_c = O_max
            best_h = h.copy()
        k = abs(idx - (n - 1))
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:

            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        step = step_start * ((step_end / step_start) ** (it / max_iter))
        h -= step * grad
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum(h, rng)
        # occasional jitter to escape plateaus
        if (it & 1023) == 0 and it > 0:
            noise = rng.normal(scale=0.005, size=n)
            h += noise
            np.clip(h, 0.0, 1.0, out=h)
            h = enforce_sum(h, rng)
    return best_h, best_c


# ----------------------------------------------------------------------
def slsqp_optimize(
    h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500
) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def cons(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    constraints = {"type": "eq", "fun": cons}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        if add != 0.0:
            x[i] += add
            diff = target - x.sum()
    return x


# ----------------------------------------------------------------------
def binary_sa_optimize(
    h_bin: np.ndarray, rng: np.random.Generator, time_budget: float
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing swap search on binary vectors with exact sum."""
    start = time.time()
    n = h_bin.size
    ones_list = list(np.where(h_bin > 0.5)[0])
    zeros_list = list(np.where(h_bin < 0.5)[0])
    best_h = h_bin.copy()
    best_c = compute_c5_fast(best_h)
    c_curr = best_c
    T0, T_end = 0.02, 1e-4

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i_idx = rng.integers(len(ones_list))
        j_idx = rng.integers(len(zeros_list))
        i, j = ones_list[i_idx], zeros_list[j_idx]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c_new = compute_c5_fast(h_bin)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            ones_list[i_idx] = j
            zeros_list[j_idx] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return best_h, best_c


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n0 = 256
        h0 = np.zeros(n0, dtype=float)
        ones = rng.choice(n0, n0 // 2, replace=False)
        h0[ones] = 1.0

    best_h = h0.copy()
    best_c = compute_c5_fast(best_h)
    best_n = best_h.size

    # ---- stage 1: subgradient on current resolution --------------------
    time_left = total_budget - (time.time() - start)
    if time_left > 0.1 * total_budget:
        t_stage = time_left * 0.3
        h_sub, c_sub = subgradient_optimize(best_h, rng, t_stage)
        if c_sub < best_c:
            best_h, best_c, best_n = h_sub, c_sub, h_sub.size

    # ---- stage 2: upsample and refine ----------------------------------
    time_left = total_budget - (time.time() - start)
    if time_left > 30 and best_n * 2 <= 1024:
        n_up = best_n * 2
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        t_stage = time_left * 0.3
        h_sub2, c_sub2 = subgradient_optimize(h_up, rng, t_stage)
        if c_sub2 < best_c:
            best_h, best_c, best_n = h_sub2, c_sub2, n_up

    # ---- stage 3: binary simulated annealing ----------------------------
    time_left = total_budget - (time.time() - start)
    if time_left > 20 and best_n % 2 == 0:
        n = best_n
        # convert to binary by taking the largest n/2 entries
        idx = np.argpartition(-best_h, n // 2)[: n // 2]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        t_stage = time_left * 0.7
        h_bin_best, c_bin_best = binary_sa_optimize(h_bin, rng, t_stage)
        if c_bin_best < best_c:
            best_h, best_c = h_bin_best, c_bin_best

    # ---- stage 4: final SLSQP polishing --------------------------------
    time_left = total_budget - (time.time() - start)
    if time_left > 5:
        h = best_h.copy()
        for beta in (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0):
            if time.time() - start > total_budget * 0.99:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=600)
            c = compute_c5_fast(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

    # ---- final sanitisation ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fast(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
