def project_onto_capped_simplex(v, s):

        else:

import time
import numpy as np

def _project_capped_simplex(v, s):
    """Project v onto {0<=x<=1, sum(x)=s} using bisection on the dual variable."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        tau = (lo + hi) * 0.5
        w = np.clip(v - tau, 0.0, 1.0)
        if w.sum() > s:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h, size_fft, dx):
    """Return C5 for a step function h using FFT‑based autocorrelation."""
    n = h.size
    H = np.fft.rfft(h, n=size_fft)
    Hr = np.fft.rfft(h[::-1], n=size_fft)
    conv = np.fft.irfft(H * Hr, n=size_fft)[: 2 * n - 1]
    C = conv[n - 1 : n - 1 + n]                # linear autocorrelation
    target = n / 2.0
    A = target - C                             # overlap for each shift
    return float(A.max() * dx)


def _subgradient_optimize(h0, target, deadline, rng):
    """Sub‑gradient descent on max overlap with projection onto the feasible set."""
    n = h0.size
    dx = 2.0 / n
    size_fft = 1 << ((2 * n - 1).bit_length())
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5_value(h, size_fft, dx)

    step0 = 0.5
    it = 0
    while time.time() < deadline:
        # autocorrelation via FFT
        H = np.fft.rfft(h, n=size_fft)
        Hr = np.fft.rfft(h[::-1], n=size_fft)
        conv = np.fft.irfft(H * Hr, n=size_fft)[: 2 * n - 1]
        C = conv[n - 1 : n - 1 + n]
