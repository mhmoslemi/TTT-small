def clone_and_mutate(h, rng, max_transfer=0.2, n_transfers=1):
    nh = h.copy()
    n = h.size
    for _ in range(n_transfers):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # compute max transfer from i to j
        max_eps = min(nh[i], max_transfer, 1.0 - nh[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps
        nh[i] -= eps
        nh[j] += eps
    return nh
```

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- evaluation & basic ops -------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to satisfy sum(h) = n/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    eps = 1e-12
    while abs(diff) > eps:
        i = rng.integers(0, h.size)
        if diff > 0:
            max_adj = min(diff, 1.0 - h[i])
        else:
            max_adj = max(diff, -h[i])
        if max_adj == 0.0:
            continue
        h[i] += max_adj
        diff = target - h.sum()
    return h


def random_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h (continuous)."""
    h = rng.random(n)
    return enforce_sum(h, rng)


# ------------------- gradient & local moves -------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (max) overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = n / 2.0 - h.sum()
    return h


# ------------------- soft‑max SLSQP -------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    eps = 1e-12
    while abs(diff) > eps:
        i = rng.integers(0, n)
        if diff > 0:
            max_adj = min(diff, 1.0 - x[i])
        else:
            max_adj = max(diff, -x[i])
        if max_adj == 0.0:
            continue
        x[i] += max_adj
        diff = CURRENT_TARGET - x.sum()
    return x


# ------------------- population & utilities -------------------
def clone_and_mutate(h: np.ndarray, rng: np.random.Generator,
                     max_transfer: float = 0.2, n_transfers: int = 1) -> np.ndarray:
    nh = h.copy()
    n = h.size
    for _ in range(n_transfers):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(nh[i], max_transfer, 1.0 - nh[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps
        nh[i] -= eps
        nh[j] += eps
    return nh


def convex_combo(h1: np.ndarray, h2: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.random()
    return alpha * h1 + (1.0 - alpha) * h2


def tournament(pop, rng, k: int = 3):
    idx = rng.integers(0, len(pop), size=k)
    best = min(idx, key=lambda i: pop[i][0])
    return pop[best]
