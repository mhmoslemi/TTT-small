def project_to_constraints(v, target):
    # v is np.ndarray
    # returns x satisfying sum(x)=target, 0<=x<=1
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # maybe due to rounding, adjust diff iteratively (rare)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust by adding diff to some non-bound entries
        # find indices not at bounds
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size > 0:
            # distribute evenly
            add = diff / free.size
            x[free] += add
            # clip again
            x = np.clip(x, 0.0, 1.0)
    return x
```

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = abs(np.argmax(corr) - (n - 1))
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -np.max(v) - 1.0, 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v + lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            add = diff / free.size
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x

def subgradient_optimize(h: np.ndarray, target: float, max_iters: int,
                         step0: float, time_limit: float) -> np.ndarray:
    step = step0
    it = 0
    while it < max_iters and time.time() < time_limit:
        grad = worst_gradient(h)
        h = h - step * grad
        h = project_to_constraints(h, target)
        step *= 0.995
        it += 1
    return h

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.03) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()

    best_c = np.inf
    T0, Tend = 0.02, 1e-4

        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones, assume_unique=True)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (Tend / T0) ** (elapsed / time_budget)
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i, j = ones_arr[i_pos], zeros_arr[j_pos]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):

                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
