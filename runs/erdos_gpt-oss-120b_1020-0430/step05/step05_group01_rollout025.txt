def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project h onto [0,1]^n with sum_i h_i = n/2 (water-filling)."""
    n = h.size
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        v = np.clip(h - mid, 0.0, 1.0)
        s = v.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    w = np.exp(bc - m)
    return (m + np.log(np.sum(w))) / CURRENT_BETA

                   max_iter: int = 500) -> np.ndarray:

    n = h0.size
    CURRENT_TARGET = n / 2.0

        softmax_obj,
        h0,

        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )

    if abs(CURRENT_TARGET - x.sum()) > 1e-12:

    return x

                    best_h, best_c, best_n = h_up.copy(), c_up, target_n

            # final gradient polishing
            h_up = gradient_refine(h_up, rng, step=0.005, max_iter=12)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # ---------- stochastic tweaks until time expires --------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        n = best_h.size
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        idx = np.argmax(corr)
        k = abs(idx - (n - 1))
        if k == 0:
            grad = 1.0 - 2.0 * best_h
        else:
            grad = np.ones_like(best_h)
            grad[: n - k] -= best_h[k:]
            grad[k:] -= best_h[: n - k]
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        delta = min(best_h[i_src], 1.0 - best_h[i_dst]) * rng.random() * 0.02
        if delta > 1e-12:
            nh = best_h.copy()
            nh[i_src] -= delta
            nh[i_dst] += delta
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
