def subgrad_descent(h0, rng, max_iter, step0):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(max_iter):
        # correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        lag = idx - (n-1)
        # gradient
        g = np.zeros_like(h)
        if lag >= 0:
            L = lag
            i = np.arange(0, n-L)
            g[i] += 1.0 - h[i+L]
            i = np.arange(L, n)
            g[i] -= h[i-L]
        else:
            L = -lag
            i = np.arange(L, n)
            g[i] += 1.0 - h[i-L]
            i = np.arange(0, n-L)
            g[i] -= h[i+L]
        step = step0 / math.sqrt(t+1)
        h = h - step * g
        # clip & adjust sum
        np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # optional early break if no improvement for many iters
    return best_h, best_c

def subgrad_descent(h0, rng, max_iter, step0, deadline):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        # gradient
        g = np.zeros_like(h)
        if lag >= 0:
            L = lag
            i = np.arange(0, n - L)
            g[i] += 1.0 - h[i + L]
            i = np.arange(L, n)
            g[i] -= h[i - L]
        else:
            L = -lag
            i = np.arange(L, n)
            g[i] += 1.0 - h[i - L]
            i = np.arange(0, n - L)
            g[i] -= h[i + L]
        step = step0 / math.sqrt(t + 1)
        h = h - step * g
        np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

    rand_frac *= target / rand_frac.sum()
    rand_frac = np.clip(rand_frac, 0.0, 1.0)
    adjust_sum_exact(rand_frac, rng)
    vecs.append(rand_frac)
    alphas = [(np.sqrt(5) - 1) / 2, np.sqrt(2) - 1, 0.33333333, rng.random()]
    for alpha in alphas:
        i = np.arange(n)
        frac = (i * alpha) % 1.0
        h = (frac < 0.5).astype(float)
        adjust_sum_exact(h, rng)
        vecs.append(h)
    return vecs

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95

    best_h = None
    best_c = np.inf
    best_n = None

    candidate_ns = [360, 540, 720]
    if 'initial_h_values' in globals():
        pn = np.array(initial_h_values).size
        if pn not in candidate_ns:
            candidate_ns.append(pn)
    candidate_ns = [n for n in candidate_ns if n % 2 == 0 and n <= 720]
    candidate_ns.sort()

    for n in candidate_ns:
        if time.time() > deadline:
            break
        start_vecs = generate_initial_vectors(n, rng)
        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h0 = h0.copy()
            adjust_sum_exact(h0, rng)
            np.clip(h0, 0.0, 1.0, out=h0)
            # subgradient descent
            max_iter = int(5000 * (n / 360))
            step0 = 1.0
            sub_dead = deadline
            h_opt, c_opt = subgrad_descent(h0, rng, max_iter, step0, sub_dead)
            if c_opt < best_c:
                best_c, best_h, best_n = c_opt, h_opt.copy(), n
                if best_c <= 0.3808:
                    adjust_sum_exact(best_h, rng)
                    return best_h.tolist(), float(compute_c5(best_h)), int(best_n)
            # binary refinement
            if np.all((np.abs(h_opt) < 1e-8) | (np.abs(h_opt - 1) < 1e-8)):
                h_bin = h_opt.copy()
            else:
                h_bin = np.zeros_like(h_opt)
                idx = np.argpartition(h_opt, -n // 2)[-n // 2 :]
                h_bin[idx] = 1.0
                adjust_sum_exact(h_bin, rng)
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
            h_bin2, c_bin2 = binary_hill_climb(h_bin, rng, sub_dead)
            if c_bin2 < best_c:
                best_c, best_h, best_n = c_bin2, h_bin2.copy(), n
                if best_c <= 0.3808:
                    adjust_sum_exact(best_h, rng)
                    return best_h.tolist(), float(compute_c5(best_h)), int(best_n)

    # optional up‑sampling
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        adjust_sum_exact(h_up, rng)
        max_iter = int(6000 * (n_up / 360))
        step0 = 1.0
        h_up_opt, c_up_opt = subgrad_descent(h_up, rng, max_iter, step0, deadline)
        if c_up_opt < best_c:
            best_c, best_h, best_n = c_up_opt, h_up_opt.copy(), n_up
        # binary hill climb on up‑sampled
        h_bin = np.zeros_like(best_h)
        idx = np.argpartition(best_h, -best_n // 2)[-best_n // 2 :]
        h_bin[idx] = 1.0
        adjust_sum_exact(h_bin, rng)
        h_bin2, c_bin2 = binary_hill_climb(h_bin, rng, deadline)
        if c_bin2 < best_c:
            best_c, best_h, best_n = c_bin2, h_bin2.copy(), best_n

    # fallback
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    np.clip(best_h, 0.0, 1.0, out=best_h)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
