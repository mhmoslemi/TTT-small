def project(v, target):
    lo, hi = -1.0, 1.0
    # Ensure lo large enough (so sum > target) and hi small enough (sum < target)
    # Actually we need monotonic decreasing sum in mu.
    # We'll expand bounds if necessary.
    for _ in range(100):
        mu = (lo + hi)/2
        x = np.clip(v - mu, 0, 1)
        s = x.sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    x = np.clip(v - (lo+hi)/2, 0, 1)
    # adjust for possible small error:
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # because of clipping at bounds, maybe cannot achieve exact sum; but target between feasible bounds so should be able.
        # fallback to random adjust.
        # Could just add diff to a not-at-bound index.
        idx = np.where((x > 0) & (x < 1))[0]
        if idx.size > 0:
            # distribute diff equally across these indices
            x[idx] += diff / idx.size
            x = np.clip(x, 0, 1)
        else:
            # all at bounds, adjust random index not at bound 0/1.
            i = np.random.randint(0, len(x))
            x[i] = np.clip(x[i] + diff, 0, 1)
    return x

def project_sum_box(v, target):

    lo, hi = -1.0, 2.0  # initial bounds maybe

    # Since v - mu, as mu decreases, x increases up to 1; as mu increases, x decreases to 0.
    for _ in range(50):
        mu = (lo + hi) / 2
        x = np.clip(v - mu, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    x = np.clip(v - (lo + hi)/2, 0.0, 1.0)
    # Possibly adjust due to rounding
    diff = target - x.sum()

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_sum_box(v, target):
    # projection of v onto {0<=x<=1, sum(x)=target}
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mu = (lo + hi) * 0.5
        x = np.clip(v - mu, 0.0, 1.0)
        if x.sum() > target:
            lo = mu
        else:
            hi = mu
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # adjust a single element if all are at bounds
            idx = np.argmax(x) if diff > 0 else np.argmin(x)
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def gradient_refine_proj(h, target, rng, step0=0.02, decay=0.5, max_iter=30):
    cur = h.copy()
    for it in range(max_iter):
        n = cur.size
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        idx = np.argmax(corr) - (n - 1)  # shift (may be negative)
        g = np.zeros_like(cur)
        if idx == 0:
            g[:] = 1.0 - 2.0 * cur
        elif idx > 0:                # positive shift
            shift = idx
            up = n - shift
            g[:up] += 1.0 - cur[shift:]
            g[shift:] -= cur[:up]
        else:                        # negative shift
            shift = -idx
            up = n - shift
            g[shift:] += 1.0 - cur[:up]
            g[:up] -= cur[shift:]
        step = step0 * (decay ** it)
        nxt = cur - step * g
        cur = project_sum_box(nxt, target)
    return cur

def binary_hill_climb(h_bin, target, deadline, rng):
    h = h_bin.copy()
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        if idx == 0:
            break
        if idx > 0:
            shift = idx
            i_rng = np.arange(0, n - shift)
            hi = h[i_rng]
            hk = h[i_rng + shift]
        else:
            shift = -idx
            i_rng = np.arange(shift, n)
            hi = h[i_rng]
            hk = h[i_rng - shift]
        mask_A = (hi == 1) & (hk == 0)
        mask_B = (hi == 0) & (hk == 1)
        A = i_rng[mask_A]
        B = i_rng[mask_B]
        if A.size and B.size:
            i = rng.choice(A)
            j = rng.choice(B)
            h[i] = 0.0
            h[j] = 1.0
            continue
        # fallback random swap
        i = rng.integers(n)
        if h[i] == 1.0:
            zeros = np.flatnonzero(h == 0.0)
            if zeros.size == 0:
                break
            j = rng.choice(zeros)
        else:
            ones = np.flatnonzero(h == 1.0)
            if ones.size == 0:
                break
            j = rng.choice(ones)
        h[i], h[j] = h[j], h[i]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.99 * budget_s

    parent = globals().get("initial_h_values", None)
    best_h = None
    best_c = np.inf
    best_n = None

    candidate_ns = [360, 480, 540]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        starts = []

        # parent (interpolated / binary) if available
        if parent is not None:
            p = np.asarray(parent, dtype=float)
            if len(p) == n:
                starts.append(p)
                idx = np.argpartition(p, -n // 2)[-n // 2 :]
                b = np.zeros(n)
                b[idx] = 1.0
                starts.append(b)
            else:
                xs = np.linspace(0.0, 2.0, len(p), endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                ip = np.interp(xt, xs, p)
                starts.append(ip)
                idx = np.argpartition(ip, -n // 2)[-n // 2 :]
                b = np.zeros(n)
                b[idx] = 1.0
                starts.append(b)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        # random fractional
        rand = rng.random(n)
        rand = project_sum_box(rand * (target / rand.sum()), target)
        starts.append(rand)

        # alternating pattern
        alt = np.tile([1.0, 0.0], n // 2)
        starts.append(alt)

        for h0 in starts:
            if time.time() > deadline:
                break
            h = project_sum_box(np.asarray(h0, dtype=float), target)

            if np.all((h == 0) | (h == 1)):
                sub_dead = min(deadline, time.time() + 0.15 * (deadline - time.time()))
                h = binary_hill_climb(h, target, sub_dead, rng).astype(float)

    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:

        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = project_sum_box(h_up, target_up)

        if np.all((h_up == 0) | (h_up == 1)):
            sub_dead = min(deadline, time.time() + 0.2 * (deadline - time.time()))
            h_up = binary_hill_climb(h_up, target_up, sub_dead, rng).astype(float)
        else:
            h_up = gradient_refine_proj(h_up, target_up, rng,
                                        step0=0.02, decay=0.5, max_iter=40)

        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # fallback
    if best_h is None:

        best_h = np.tile([1.0, 0.0], best_n // 2).astype(float)
        best_h = project_sum_box(best_h, best_n / 2.0)
        best_c = compute_c5(best_h)

    best_h = project_sum_box(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
