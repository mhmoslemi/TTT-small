def swap_reduce(h, max_iter, deadline):
    h = h.copy()
    n = h.size
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        O = np.correlate(h, 1 - h, mode='full')
        O_max = O.max()
        idx_max = O.argmax()
        # early stop if O_max == 0? Not likely
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)  # or np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        k = idx_max - (n - 1)  # shift, can be negative
        # compute ti and tj
        # For each i in ones, sum h[i-k] + h[i+k] where valid
        i_arr = ones
        # compute coordinates for i - k and i + k
        i_m = i_arr - k
        i_p = i_arr + k
        # Use boolean mask to check bounds
        valid_m = (i_m >= 0) & (i_m < n)
        valid_p = (i_p >= 0) & (i_p < n)
        ti = np.zeros_like(i_arr, dtype=int)
        ti[valid_m] = h[i_m[valid_m]]
        ti[valid_p] += h[i_p[valid_p]]
        # For zeros
        j_arr = zeros
        j_m = j_arr - k
        j_p = j_arr + k
        valid_mj = (j_m >= 0) & (j_m < n)
        valid_pj = (j_p >= 0) & (j_p < n)
        tj = np.zeros_like(j_arr, dtype=int)
        tj[valid_mj] = h[j_m[valid_mj]]
        tj[valid_pj] += h[j_p[valid_pj]]
        # compute Δ matrix
        # shape (len(ones), len(zeros))
        # Using broadcasting
        Δ = ti[:, None].astype(int) - tj[None, :].astype(int)
        if k == 0:
            Δ -= 2
        else:
            # Build dict for zero index position
            zero_pos = {val: idx for idx, val in enumerate(j_arr)}
            # For each i
            for ii, i in enumerate(i_arr):
                j_target = i + k
                jt = zero_pos.get(j_target)
                if jt is not None:
                    Δ[ii, jt] += 1  # subtract -1
                j_target2 = i - k
                jt2 = zero_pos.get(j_target2)
                if jt2 is not None:
                    Δ[ii, jt2] += 1
        # Find best swap (most negative Δ)
        best_idx = np.unravel_index(np.argmin(Δ), Δ.shape)
        best_delta = Δ[best_idx]
        if best_delta >= 0:
            # no improving swap for this max shift
            break
        # Perform swap
        i_swap = ones[best_idx[0]]
        j_swap = zeros[best_idx[1]]
        h[i_swap] = 0
        h[j_swap] = 1
        # continue loop
    return h

def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # correlation of all-zero vector
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0  # enforce O_0 = 0
    return h

def swap_reduce(h, deadline):
    h = h.copy()
    n = h.size
    while time.time() < deadline:
        O = np.correlate(h, 1 - h, mode='full')
        O_max = O.max()
        idx_max = O.argmax()
        k = idx_max - (n - 1)
        # get ones and zeros
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)  # alternative: np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        # compute ti for each one
        i_arr = ones
        i_m = i_arr - k
        i_p = i_arr + k
        ti = np.zeros_like(i_arr, dtype=int)
        # mask valid
        valid = (i_m >= 0) & (i_m < n)
        ti[valid] = h[i_m[valid]]
        valid = (i_p >= 0) & (i_p < n)
        ti[valid] += h[i_p[valid]]
        # compute tj for zeros
        j_arr = zeros
        j_m = j_arr - k
        j_p = j_arr + k
        tj = np.zeros_like(j_arr, dtype=int)
        valid = (j_m >= 0) & (j_m < n)
        tj[valid] = h[j_m[valid]]
        valid = (j_p >= 0) & (j_p < n)
        tj[valid] += h[j_p[valid]]
        # compute Δ matrix
        # use broadcasting int for speed
        Δ = ti[:, None] - tj[None, :]
        if k == 0:
            Δ -= 2
        else:
            zero_pos = {val: idx for idx, val in enumerate(j_arr)}
            # loop over ones
            for ii, i in enumerate(i_arr):
                jt = zero_pos.get(i + k)
                if jt is not None:
                    Δ[ii, jt] += 1
                jt = zero_pos.get(i - k)
                if jt is not None:
                    Δ[ii, jt] += 1
        # Find minimal Δ
        min_idx = np.unravel_index(np.argmin(Δ), Δ.shape)
        min_delta = Δ[min_idx]
        if min_delta >= 0:
            # no improving swap
            break
        # Perform swap
        i_swap = ones[min_idx[0]]
        j_swap = zeros[min_idx[1]]
        h[i_swap] = 0
        h[j_swap] = 1
    return h

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    # Loop until diff within tolerance
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final clip
    np.clip(h, 0.0, 1.0, out=h)
    return h

    sumexp = np.sum(np.exp(factor))
    return (m + np.log(sumexp)) / _CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global _CURRENT_TARGET, _CURRENT_BETA
    n = h0.size
    _CURRENT_TARGET = n / 2.0
    _CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _smoothmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

# ---------- binary helpers ----------
def binarize_h(h):
    n = h.size
    half = n // 2
    idx = np.argpartition(-h, half - 1)[:half]
    b = np.zeros(n, dtype=np.int8)
    b[idx] = 1
    return b

def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h

def swap_reduce(h, deadline):
    """Greedy reduction of max overlap by single 1↔0 swaps."""
    h = h.copy()
    n = h.size
    while time.time() < deadline:
        O = np.correlate(h, 1 - h, mode="full")
        O_max = O.max()
        idx_max = O.argmax()
        k = idx_max - (n - 1)
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_arr = ones
        i_m = i_arr - k
        i_p = i_arr + k
        ti = np.zeros_like(i_arr, dtype=int)
        valid = (i_m >= 0) & (i_m < n)
        ti[valid] = h[i_m[valid]]
        valid = (i_p >= 0) & (i_p < n)
        ti[valid] += h[i_p[valid]]

        j_arr = zeros
        j_m = j_arr - k
        j_p = j_arr + k
        tj = np.zeros_like(j_arr, dtype=int)
        valid = (j_m >= 0) & (j_m < n)
        tj[valid] = h[j_m[valid]]
        valid = (j_p >= 0) & (j_p < n)
        tj[valid] += h[j_p[valid]]

        Δ = ti[:, None] - tj[None, :]
        if k == 0:
            Δ -= 2
        else:
            zero_pos = {val: idx for idx, val in enumerate(j_arr)}
            for ii, i in enumerate(i_arr):
                jt = zero_pos.get(i + k)
                if jt is not None:
                    Δ[ii, jt] += 1
                jt = zero_pos.get(i - k)
                if jt is not None:
                    Δ[ii, jt] += 1

        min_idx = np.unravel_index(np.argmin(Δ), Δ.shape)
        min_delta = Δ[min_idx]
        if min_delta >= 0:
            break
        i_swap = ones[min_idx[0]]
        j_swap = zeros[min_idx[1]]
        h[i_swap] = 0
        h[j_swap] = 1
    return h

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # Determine problem size
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
    else:
        n = 360
    if n % 2 != 0:
        n += 1  # make even

    target_sum = n / 2.0
    best_c = np.inf
    best_h = None          # final (float) solution
    best_bin = None        # best binary version found

    # ---- parent (if any) ----
    if "initial_h_values" in globals():
        p = np.clip(parent, 0.0, 1.0)
        adjust_sum_exact(p, rng)
        c_parent = compute_c5(p)
        if c_parent < best_c:
            best_c = c_parent
            best_bin = binarize_h(p)
            best_h = p.copy()
            # binary refinement later

    # ---- greedy binary ----
    if time.time() < deadline:
        gre = greedy_binary(n, rng)
        c_gre = compute_c5(gre)
        if c_gre < best_c:
            best_c = c_gre
            best_bin = gre.copy()
            best_h = gre.astype(float)

    # ---- random restarts ----
    max_random = 20
    for _ in range(max_random):
        if time.time() >= deadline:
            break
        idx = rng.choice(n, size=n // 2, replace=False)
        rand_bin = np.zeros(n, dtype=np.int8)
        rand_bin[idx] = 1
        c_rand = compute_c5(rand_bin)
        if c_rand < best_c:
            best_c = c_rand
            best_bin = rand_bin.copy()
            best_h = rand_bin.astype(float)

    # ---- local swap reduction on the best binary ----
    if best_bin is not None and time.time() < deadline:
        refined_bin = swap_reduce(best_bin, deadline)
        c_ref = compute_c5(refined_bin)
        if c_ref < best_c:
            best_c = c_ref
            best_bin = refined_bin.copy()
            best_h = refined_bin.astype(float)

    # ---- continuous SLSQP refinement ----
    if best_h is not None and time.time() < deadline:
        h_float = best_h.astype(float)
        adjust_sum_exact(h_float, rng)
        schedule = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0)
        for beta in schedule:
            if time.time() >= deadline:
                break
            try:
                h_float = slsqp_optimize(h_float, beta, rng,
                                         max_iter=600 if beta >= 200 else 400)
            except Exception:
                continue
            c_new = compute_c5(h_float)
            if c_new < best_c:
                best_c = c_new
                best_h = h_float.copy()

    # ---- final safety ----
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
