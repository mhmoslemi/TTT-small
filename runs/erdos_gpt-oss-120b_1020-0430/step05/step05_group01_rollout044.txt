import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random walk correction
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    step_scale: float = 0.6,
):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    step: float = 0.015,
    max_iter: int = 10,
) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # ----- initial solution -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        n = 512
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    current_h = best_h.copy()
    current_c = best_c

    # ----- simulated annealing phase ----------------------------------------
    anneal_frac = 0.78
    anneal_end = start + total * anneal_frac
    step_max = 0.05          # max transfer per move
    temp_start = 0.001       # initial temperature

    while time.time() < anneal_end:
        elapsed = time.time() - start
        t_frac = elapsed / (total * anneal_frac)
        temp = temp_start * (1 - t_frac)  # linear cooling

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(current_h[i], 1.0 - current_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * min(step_max, max_eps)
        if eps <= 0.0:
            continue

        new_h = current_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - current_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / max(temp, 1e-12))):
            current_h = new_h
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()

    # ----- hill‑climbing refinement -----------------------------------------
    remaining = start + total - time.time()
    hill_time = remaining * 0.7
    hill_end = time.time() + hill_time
    max_iters = int(2000 * 256 / n)  # ~1000 for n=512
    h_hc = best_h.copy()
    it = 0
    while it < max_iters and time.time() < hill_end:
        sub_iters = min(200, max_iters - it)
        h_hc, c_hc = guided_hill_climb(h_hc, rng, max_iters=sub_iters, step_scale=0.6)
        if c_hc < best_c:
            best_c = c_hc
            best_h = h_hc.copy()
        it += sub_iters

    # ----- gradient polishing -----------------------------------------------
    h_ref = best_h.copy()
    h_ref = gradient_refine(h_ref, rng, step=0.01, max_iter=8)
    c_ref = compute_c5(h_ref)
    if c_ref < best_c:
        best_c = c_ref
        best_h = h_ref.copy()

    # ----- final stochastic tweaks -------------------------------------------
    while time.time() < start + total:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h = nh.copy()
            best_c = nc

    # ----- enforce constraints and return ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
