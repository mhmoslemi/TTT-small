    # v: np array
    # return projected array onto [0,1] and sum=target.
    # lambda search
    lo = -np.max(v) - 1  # lower bound for lambda, ensuring all v_i+lambda <=0

        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid

import time
import numpy as np

GLOBAL_RNG = None  # will be set in run()


def compute_c5_and_corr(h):
    """Return (C5 max value, full correlation array) for sample vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr) * (2.0 / n)
    return max_val, corr


def worst_shift_gradient(h, corr):
    """Gradient of the worst shift (the shift attaining max correlation)."""
    n = h.size
    idx = int(np.argmax(corr))
    s = idx - (n - 1)  # shift: positive => h shifted forward
    g = np.zeros_like(h)
    if s >= 0:
        # corr_s = sum_{i=0}^{n-1-s} h[i+s] * (1 - h[i])
        # ∂/∂h[i] = -h[i+s]; ∂/∂h[i+s] = 1 - h[i]
        g[: n - s] -= h[s:]
        g[s:] += 1.0 - h[: n - s]
    else:
        t = -s
        # corr_{-t} = sum_{i=0}^{n-1-t} h[i] * (1 - h[i+t])
        # ∂/∂h[i] = 1 - h[i+t]; ∂/∂h[i+t] = -h[i]
        g[: n - t] += 1.0 - h[t:]
        g[t:] -= h[: n - t]
    return g, s


def project_capped_simplex(v, target):
    """Project vector v onto {0<=h_i<=1, sum h_i = target}."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(v + mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def gradient_descent_refine(h, target, deadline, max_iter=4000):
    """Projected gradient descent on the worst shift."""
    n = h.size
    best_h = h.copy()
    best_c, _ = compute_c5_and_corr(best_h)
    step0 = 0.5
    it = 0
    while time.time() < deadline and it < max_iter:
        _, corr = compute_c5_and_corr(h)
        g, _ = worst_shift_gradient(h, corr)
        step = step0 / np.sqrt(it + 1.0)
        improved = False
        for factor in (1.0, 0.5, 2.0, 0.25, 4.0):
            alpha = step * factor
            v = h - alpha * g
            h_cand = project_capped_simplex(v, target)
            c_cand, _ = compute_c5_and_corr(h_cand)
            if c_cand < best_c - 1e-12:
                best_c = c_cand
                best_h = h_cand
                h = h_cand
                improved = True
                break
        if not improved:
            step0 *= 0.7
        it += 1
    return best_h, best_c


def fractional_local_search(h, target, deadline, max_iters=20000):
    """Hill climbing by moving mass between two coordinates."""
    n = h.size
    cur = h.copy()
    cur_c, _ = compute_c5_and_corr(cur)
    best = cur.copy()
    best_c = cur_c
    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = GLOBAL_RNG.choice(n, size=2, replace=False)
        if GLOBAL_RNG.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0.0:
            it += 1
            continue
        delta = GLOBAL_RNG.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c_new, _ = compute_c5_and_corr(cur)
        if c_new < cur_c:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best = cur.copy()
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
    return best, best_c


def binary_sa_refine(h_bin, target, deadline, max_iters=20000):
    """Swap‑based simulated annealing on a binary vector."""
    n = h_bin.size
    cur = h_bin.copy()
    cur_c, _ = compute_c5_and_corr(cur)
    best = cur.copy()
    best_c = cur_c
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    it = 0
    while time.time() < deadline and it < max_iters:
        i = GLOBAL_RNG.choice(ones)
        j = GLOBAL_RNG.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        c_new, _ = compute_c5_and_corr(cur)
        if c_new < cur_c:
            cur_c = c_new
            ones = np.flatnonzero(cur == 1)
            zeros = np.flatnonzero(cur == 0)
            if c_new < best_c:
                best_c = c_new
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
    return best, best_c


def refine_for_n(h_start, n_target, deadline):
    """Upsample h_start to n_target and run local refinements."""
    n0 = h_start.size
    if n_target != n0:
        xs = np.linspace(0.0, 2.0, n0, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h = np.interp(xt, xs, h_start)
    else:
        h = h_start.copy()
    target = n_target / 2.0
    h = project_capped_simplex(h, target)

    now = time.time()
    if now >= deadline:
        return h, compute_c5_and_corr(h)[0]

    # gradient descent (≈50% of remaining time)
    t_grad = now + (deadline - now) * 0.5
    h, c = gradient_descent_refine(h, target, t_grad)

    now = time.time()
    if now >= deadline:
        return h, c

    # fractional local search (≈60% of the still remaining time)
    t_frac = now + (deadline - now) * 0.6
    h_frac, c_frac = fractional_local_search(h, target, t_frac)
    if c_frac < c:
        h, c = h_frac, c_frac

    now = time.time()
    if now >= deadline:
        return h, c

    # binary refinement if size is moderate
    if n_target <= 800:
        # round to binary by taking the largest entries
        k = int(target)
        idx = np.argpartition(h, -k)[-k:]
        h_bin = np.zeros(n_target, dtype=float)
        h_bin[idx] = 1.0
        t_bin = deadline
        h_bin_best, c_bin = binary_sa_refine(h_bin, target, t_bin)
        if c_bin < c:
            h, c = h_bin_best.astype(float), c_bin

    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    global GLOBAL_RNG
    GLOBAL_RNG = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98

    # starting vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 360
        h0 = np.zeros(n0, dtype=float)
        h0[GLOBAL_RNG.choice(n0, n0 // 2, replace=False)] = 1.0
    h0 = project_capped_simplex(h0, h0.size / 2.0)

    best_h = None
    best_c = np.inf
    best_n = None

    n_sequence = [360, 540, 720, 1080]

    for n in n_sequence:
        if time.time() >= deadline:
            break
        # allocate a portion of remaining time for this n
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        h_ref, c_ref = refine_for_n(h0, n, sub_deadline)
        if c_ref < best_c:
            best_h, best_c, best_n = h_ref.copy(), c_ref, n
        h0 = best_h.copy()  # feed the best forward

    # final polishing on the best found vector with any remaining time
    if best_h is not None and time.time() < deadline:
        final_deadline = deadline
        best_h, best_c = gradient_descent_refine(best_h, best_n / 2.0, final_deadline)

    # fallback in the unlikely event nothing succeeded
    if best_h is None:
        best_n = 360
        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
        best_h = project_capped_simplex(best_h, best_n / 2.0)
        best_c = compute_c5_and_corr(best_h)[0]

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_capped_simplex(best_h, best_n / 2.0)
    final_c = compute_c5_and_corr(best_h)[0]

    return best_h.tolist(), float(final_c), int(best_n)
