    corr = np.correlate(h, 1-h, mode='full') * (2.0 / n)
    phi = corr  # shape L
    max_phi = np.max(phi)  # for stability -> maybe use log-sum-exp? But we already have weights.
    # compute weights
    # use np.exp(beta*(phi - max_phi)) for stability
    exp_beta = np.exp(beta * (phi - max_phi))
    Z = np.sum(exp_beta)
    w = exp_beta / Z
    F = np.sum(w * phi)
    coeff = w * (1.0 + beta * (phi - F))
    # compute gradient g = (2/n) * sum_s coeff_s * gradient_of_phi_s

        if coeff_s == 0:

def slsqp_optimize(h0, beta, rng, max_iter=300):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0,1.0)] * n
    def obj(x):
        f, _ = smooth_max_and_grad(x, beta)
        return f
    def grad(x):
        _, g = smooth_max_and_grad(x, beta)
        return g
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones_like(x)}
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons, jac=grad,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0,n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()


def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upp = n - k
            g[:upp] += 1.0 - h[k:]
            g[k:] -= h[:upp]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for __ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        step *= 0.5
    return h


def binary_sa(h_bin, rng, deadline, max_iter=200000):
    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst
    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))
    T0 = cur_worst * 0.05 + 1e-4
    decay = 0.99995
    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # i: 1 -> 0
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        c[start_i : start_i + n] -= hi_vec
        h_bin[i] = 0.0

        # j: 0 -> 1 (using updated h_bin)
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j + 1 :]))
        c[start_j : start_j + n] += hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min
        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12)):
                accept = True
        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h_bin[j] = 0.0
            c[start_j : start_j + n] -= hj_vec
            h_bin[i] = 1.0
            c[start_i : start_i + n] += hi_vec
            no_improve += 1
        if no_improve > 50000:
            break
        it += 1
    return best_h, best_worst

def smooth_max_and_grad(h, beta):
    n = h.size
    phi = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)   # shape 2n-1
    max_phi = np.max(phi)
    exp_beta = np.exp(beta * (phi - max_phi))
    Z = exp_beta.sum()
    w = exp_beta / Z
    F = np.dot(w, phi)
    coeff = w * (1.0 + beta * (phi - F))

    g = np.zeros_like(h)
    # shifts from -(n-1) .. (n-1)
    for idx, s in enumerate(range(-(n - 1), n)):
        cs = coeff[idx]
        if cs == 0.0:
            continue
        if s >= 0:
            if s < n:
                end = n - s
                g[:end] += cs * (1.0 - h[s:])
                g[s:] -= cs * h[:end]
        else:
            t = -s
            if t < n:
                end = n - t
                g[t:] += cs * (1.0 - h[:end])
                g[:end] -= cs * h[t:]
    g *= (2.0 / n)
    return F, g


def slsqp_optimize(h0, beta, rng, max_iter=300):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        f, _ = smooth_max_and_grad(x, beta)
        return f

    def grad(x):
        _, g = smooth_max_and_grad(x, beta)
        return g

    cons = {
        "type": "eq",
        "fun": lambda x: np.sum(x) - target,
        "jac": lambda x: np.ones_like(x),
    }

    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        jac=grad,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    parent = np.array(initial_h_values, dtype=float) if "initial_h_values" in globals() else None

    best_h = None
    best_c = np.inf
    best_n = None

    base_n = parent.size if parent is not None else 360
    candidates = [base_n]
    if base_n * 2 <= 720:
        candidates.append(base_n * 2)
    if base_n % 2 == 0 and base_n // 2 >= 180:
        candidates.append(base_n // 2)

    candidates = sorted(set(candidates))

    for n in candidates:
        if time.time() > deadline:
            break
        target = n / 2.0
        start_vectors = []

        # parent based vectors
        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vectors.append(interp)                       # fractional
            idx = np.argpartition(interp, -int(target))[-int(target) :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vectors.append(bin_vec)                      # binary

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, int(target), replace=False)] = 1.0
        start_vectors.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        start_vectors.append(rand_frac)

        # alternating binary (if even)
        if n % 2 == 0:
            alt_bin = np.tile([1.0, 0.0], n // 2)
            start_vectors.append(alt_bin)

        for h0 in start_vectors:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # binary handling
            if np.all((h == 0) | (h == 1)):
                remaining = deadline - time.time()
                sub_dead = time.time() + min(remaining * 0.3, 30.0)
                h_bin, raw_worst = binary_sa(h.copy(), rng, sub_dead)
                c5 = raw_worst * (2.0 / n)
                if c5 < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c5, n
                h = h_bin.astype(float)
            else:
                c5 = compute_c5(h)
                if c5 < best_c:
                    best_h, best_c, best_n = h.copy(), c5, n

            # quick gradient refinement
            h = gradient_refine(h, rng, step=0.02, max_iter=8)
            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

            # smoothed‑max cascade
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
                if time.time() > deadline:
                    break
                max_iter = 200 if beta < 200 else 400
                if beta >= 2000:
                    max_iter = 600
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c5 = compute_c5(h)
                if c5 < best_c:
                    best_h, best_c, best_n = h.copy(), c5, n

            # final gradient refinement
            h = gradient_refine(h, rng, step=0.02, max_iter=12)
            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

        # optional up‑sampling of the best from this n
        if best_h is not None and best_n == n and time.time() < deadline and n * 2 <= 720:
            n_up = n * 2
            xs = np.linspace(0.0, 2.0, n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            adjust_sum_exact(h_up, rng)
            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
                if time.time() > deadline:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # fallback
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
