def delta_a_swap(h, i, j):
    n = h.shape[0]
    conv = np.zeros(n, dtype=int)
    ks = np.arange(i+1)
    conv[ks] -= h[i - ks]

    if diff < n:

    return delta

import time

def compute_c5(h):
    """Exact overlap integral for step function h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def delta_a_swap(h, i, j):
    """Change in the self‑correlation a(k)=∑h[t]h[t+k] caused by swapping
    h[i]=1 → 0 and h[j]=0 → 1 (i!=j). Returns the delta array for k≥0."""
    n = h.shape[0]
    conv = np.zeros(n, dtype=int)               # conv_h_d part
    ks = np.arange(i + 1)
    conv[ks] -= h[i - ks]                       # -h[i‑k]
    ks = np.arange(j + 1)
    conv[ks] += h[j - ks]                       # +h[j‑k]
    delta = 2 * conv                             # 2*conv_h_d
    delta[0] += 2                                # conv_d_d at 0
    diff = j - i
    if diff < 0:
        diff = -diff
    if diff < n:
        delta[diff] -= 2                         # conv_d_d at |j‑i|
    return delta

def greedy_local_opt(h, rng, deadline):

    minimum overlap (i.e. lowers C₅). Allows a few random jumps when stuck."""

def sa_optimize(h, rng, deadline, start_time):
    """Simple simulated annealing on the binary space."""
    n = h.shape[0]
    m = n // 2
    a = np.correlate(h, h, mode="full")[n - 1:]
    cur_c = (m - a[1:].min()) * (2.0 / n)
    best_h = h.copy()
    best_c = cur_c
    T0 = 0.02
    while time.time() < deadline:
        i = rng.choice(np.flatnonzero(h == 1))
        j = rng.choice(np.flatnonzero(h == 0))
        delta = delta_a_swap(h, i, j)
        a_new = a + delta
        new_c = (m - a_new[1:].min()) * (2.0 / n)
        rem = deadline - time.time()
        tot = deadline - start_time
        T = T0 * (rem / tot + 1e-12)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            h[i] = 0
            h[j] = 1
            a = a_new
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

    h = np.zeros(n, dtype=int)
