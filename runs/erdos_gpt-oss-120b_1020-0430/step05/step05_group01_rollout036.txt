def softmax_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)  # relative
    sum_exp_bc = np.sum(exp_bc)
    # compute factor = exp_bc + (max mask)
    max_mask = (bc == max_bc)
    factor = exp_bc.copy()
    factor[max_mask] += 1.0
    # gradient unscaled
    grad = np.zeros(n, dtype=float)
    # loop over shift s
    L = 2 * n - 1
    for s in range(n):
        pos_idx = n - 1 + s
        factor_pos = factor[pos_idx]
        if s < n:
            # positive shift contributions
            grad[:n - s] += factor_pos * (1.0 - h[s:])
            grad[s:] -= factor_pos * h[:n - s]
        if s > 0:
            neg_idx = n - 1 - s
            factor_neg = factor[neg_idx]
            grad[s:] += factor_neg * (1.0 - h[:n - s])
            grad[:n - s] -= factor_neg * h[s:]
    # divide by sum_exp_bc
    grad /= sum_exp_bc
    # objective value
    obj = (max_bc + np.log(sum_exp_bc)) / beta
    return obj, grad

import time
import numpy as np

# ----------------- basic utilities -----------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def apply_constraints(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Symmetrize, clip to [0,1] and enforce Σh = target_sum."""
    # symmetry about the centre
    h = (h + h[::-1]) * 0.5
    # bounds
    h = np.clip(h, 0.0, 1.0)
    # adjust sum if necessary
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            free = h < 1.0
            cap = np.sum(1.0 - h[free])
            if cap > 0:
                h[free] += diff * (1.0 - h[free]) / cap
        else:
            free = h > 0.0
            cap = np.sum(h[free])
            if cap > 0:
                h[free] += diff * h[free] / cap
    return np.clip(h, 0.0, 1.0)


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        nh = apply_constraints(nh, target)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# -------------- soft‑max surrogate -----------------
def softmax_obj_and_grad(h: np.ndarray, beta: float):
    """Smooth approximation of max_k O_k together with its gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)            # relative exponentials
    sum_exp_bc = np.sum(exp_bc)
    max_mask = bc == max_bc
    factor = exp_bc.copy()
    factor[max_mask] += 1.0                 # account for the explicit max term
    grad = np.zeros(n, dtype=float)

    for s in range(n):
        pos = n - 1 + s
        fpos = factor[pos]
        if s < n:
            grad[: n - s] += fpos * (1.0 - h[s:])
            grad[s:]     -= fpos * h[: n - s]
        if s > 0:
            neg = n - 1 - s
            fneg = factor[neg]
            grad[s:]     += fneg * (1.0 - h[: n - s])
            grad[: n - s] -= fneg * h[s:]

    grad /= sum_exp_bc
    obj = (max_bc + np.log(sum_exp_bc)) / beta
    return obj, grad


def gradient_descent_refine(h: np.ndarray, beta: float,
                            lr: float, steps: int, target_sum: float):
    """Projected gradient descent on the soft‑max surrogate."""
    for _ in range(steps):
        _, grad = softmax_obj_and_grad(h, beta)
        h = h - lr * grad
        h = apply_constraints(h, target_sum)
        lr *= 0.9995
    return h


# -------------- random feasible start ---------------
def random_feasible(rng: np.random.Generator, n: int = 512):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    h = apply_constraints(h, n / 2.0)
    return h
