import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # (1 - h[i+k])
            g[k:] -= h[:upper]               # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time
import numpy as np

# ------------------------------------------------------------
def compute_c5(h):
    """Exact C5 for a discretized step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Redistribute a tiny amount so that Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:up]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ------------------------------------------------------------
def adam_optimize(h0, rng, beta_schedule, deadline):
    """
    Adam on the log‑sum‑exp surrogate of max_k ∫h(1‑h(x+k)).
    No explicit equality constraint – the sum is repaired after each step.
    """
    h = h0.copy()
    n = h.size
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # rough per‑iteration budget
        per_iter = 0.015
        remaining = deadline - time.time()
        max_iters = max(1, int(remaining / per_iter))
        iters = min(300, max_iters)

        # Adam state
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.2 / np.sqrt(beta)

        for t in range(1, iters + 1):
            if time.time() > deadline:
                break
            # correlation (no dx – irrelevant for optimisation)
            c = np.correlate(h, 1.0 - h, mode="full")
            c_scaled = c * beta
            c_max = np.max(c_scaled)
            exp_c = np.exp(c_scaled - c_max)
            w = exp_c / exp_c.sum()                     # soft‑max weights

            # gradient of the surrogate
            g = np.zeros_like(h)
            nval = n
            hloc = h
            wloc = w
            # shifts k ≥ 0  (including k=0)
            for k in range(0, nval):
                pk = wloc[k + nval - 1]
                if pk == 0.0:
                    continue
                if k == 0:
                    g += pk * (1.0 - 2.0 * hloc)
                else:
                    end = nval - k
                    g[:end] += pk * (1.0 - hloc[k:])
                    g[k:]   -= pk * hloc[:end]
            # shifts k < 0  (k = –p, p>0)
            for p in range(1, nval):
                pk = wloc[-p + nval - 1]
                if pk == 0.0:
                    continue
                # add (1‑h[i‑p])  for i≥p
                g[p:] += pk * (1.0 - hloc[:nval - p])
                # subtract h[i+p]  for i≤n‑p‑1
                g[:nval - p] -= pk * hloc[p:]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto box and exact sum repair
            h = np.clip(h, 0.0, 1.0)
            adjust_sum_exact(h, rng)
    return h


# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # target discretisation – 720 points (dx≈0.00278)
    n = 720
    target = n // 2

    # -----------------------------------------------------------------
    #   build a handful of diverse starting vectors
    starters = []

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        starters.append(interp)                       # fractional parent

        # binary version – top‑half entries set to 1
        idx = np.argpartition(interp, -target)[-target:]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        starters.append(bin_vec)

    # random binary
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, target, replace=False)] = 1.0
    starters.append(bin_rand)

    # random fractional, then exact‑sum repair
    rand_frac = rng.random(n)
    rand_frac *= target / rand_frac.sum()
    adjust_sum_exact(rand_frac, rng)
    starters.append(rand_frac)

    # -----------------------------------------------------------------
    best_h = None
    best_c = np.inf

    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]

    for h0 in starters:
        if time.time() > deadline:
            break
        h = h0.copy()
        adjust_sum_exact(h, rng)
        h = np.clip(h, 0.0, 1.0)

        # Adam optimisation on the soft‑max surrogate
        h_opt = adam_optimize(h, rng, beta_schedule, deadline)

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

        # modest projected‑gradient polishing on the worst shift
        if time.time() < deadline:
            h_ref = gradient_refine(h_opt.copy(), rng, step=0.02, max_iter=20)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()

    # -----------------------------------------------------------------
    #   optional up‑sampling (duplicate each entry) and a brief re‑optimisation
    if best_h is not None and time.time() < deadline and n * 2 <= 1440:
        n_up = n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)

        h_up = adam_optimize(h_up, rng, beta_schedule, deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up.copy()
            n = n_up
            target = n // 2

    # -----------------------------------------------------------------
    if best_h is None:
        best_h = np.zeros(n)
        best_h[:target] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final safety – clip and exact‑sum repair
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
