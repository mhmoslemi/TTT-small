def best_pairwise_move(h, grad, current_c, M, fractions):
    n = h.shape[0]
    idx_sorted = np.argsort(grad)
    pos = idx_sorted[-M:]  # highest
    neg = idx_sorted[:M]   # lowest
    best = (None, None, None, current_c)  # (i,j,eps, new_c)
    for i in pos:
        hi = h[i]
        if hi <= 0:
            continue
        for j in neg:
            if i == j:
                continue
            max_eps = min(hi, 1.0 - h[j])
            if max_eps <= 0:
                continue
            for f in fractions:
                eps = max_eps * f
                if eps <= 0:
                    continue
                nh = h.copy()
                nh[i] -= eps
                nh[j] += eps
                # optional clip just in case
                # nh = np.clip(nh, 0.0, 1.0)
                # compute C5
                c = compute_c5(nh)
                if c < best[3] - 1e-12:
                    best = (i, j, eps, c)
    if best[0] is None:
        return None
    else:
        i, j, eps, c = best
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        # ensure bounds
        nh = np.clip(nh, 0.0, 1.0)
        return nh, c

def local_search_pairwise(h, rng, time_limit, M=5, fractions=[0.5,1.0]):
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    # optional improvement threshold
    while time.time() - start < time_limit:
        grad = worst_gradient(best_h)
        move = best_pairwise_move(best_h, grad, best_c, M, fractions)
        if move is None:
            # random perturb to escape

import time
import numpy as np
from scipy.optimize import minimize

# ----------------- core utilities -----------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with uniform grid."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project onto {0≤h[i]≤1, Σh = n/2} by simple redistribution."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with prescribed sum."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the shift attaining the current C5."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def best_pairwise_move(h: np.ndarray, grad: np.ndarray, cur_c: float,
                       M: int, fractions: list) -> tuple:
    """Search best epsilon‑transfer between a high‑gradient and low‑gradient index."""
    n = h.size
    order = np.argsort(grad)
    low = order[:M]
    high = order[-M:]
    best_c = cur_c
    best_move = None
    for i in high:
        hi = h[i]
        if hi <= 0.0:
            continue
        for j in low:
            if i == j:
                continue
            max_eps = min(hi, 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            for f in fractions:
                eps = max_eps * f
                if eps <= 0.0:
                    continue
                nh = h.copy()
                nh[i] -= eps
                nh[j] += eps
                c = compute_c5(nh)
                if c < best_c - 1e-12:
                    best_c = c
                    best_move = (i, j, eps, nh)
    if best_move is None:
        return None
    return best_move[3], best_c


def local_search_pairwise(h0: np.ndarray, rng: np.random.Generator,
                          time_limit: float, M: int = 5,
                          fractions: list = (0.5, 1.0)) -> tuple:
    """Iterative pairwise transfer descent using the worst‑shift gradient."""
    start = time.time()
    h = h0.copy()
    cur_c = compute_c5(h)
    while time.time() - start < time_limit:
        grad = worst_gradient(h)
        mv = best_pairwise_move(h, grad, cur_c, M, fractions)
        if mv is None:
            # small random perturbation to escape stagnation
            i = rng.integers(0, h.size)
            j = rng.integers(0, h.size)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = max_eps * rng.random() * 0.1
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            c = compute_c5(nh)
            if c < cur_c - 1e-12:
                h, cur_c = nh, c
        else:
            h, cur_c = mv
    return h, cur_c


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 500) -> np.ndarray:
    """Smooth‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())

                h0[i] -= eps
                h0[j] += eps
                h0 = np.clip(h0, 0.0, 1.0)
        else:
            h0 = best_h.copy()

        h_opt, c_opt = local_search_pairwise(h0, rng, time_limit=0.48,
                                             M=7, fractions=(0.5, 1.0))
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # ---- Optional SLSQP refinement ----
    for beta in (20., 80., 200., 800., 3000.):
        if time.time() - start > 0.90 * budget:
            break
        best_h = slsqp_optimize(best_h, beta, max_iter=500)
        best_c = compute_c5(best_h)

    # ---- Phase 3: final stochastic tweaks ----
    while time.time() < start + budget * 0.999:
        i = rng.integers(0, target_n)
        j = rng.integers(0, target_n)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random() * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = np.clip(nh, 0.0, 1.0)
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c = nh.copy(), c

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
