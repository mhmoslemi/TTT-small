        shift = idx - (n-1)
        shift = abs(shift)
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)

def gradient_descent_worst(h, rng, max_iter, deadline):
    n = h.size
    target = n/2
    while max_iter > 0 and time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n-1)
        k = abs(idx)
        if k == 0:
            break
        # compute gradient
        g = np.zeros_like(h)
        up = n - k
        g[:up] += 1.0 - h[k:]
        g[k:] -= h[:up]
        # step size schedule:
        step = 0.01  # choose small
        h_new = h - step * g
        # project to [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)
        # adjust sum
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            # adjust diff by moving random entries
            for _ in range(200):
                i = rng.integers(n)
                add = min(diff, 1.0 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
                h_new[i] += add
                diff = target - h_new.sum()
                if abs(diff) < 1e-12:
                    break
        h = h_new
        max_iter -= 1
    return h

        target = n / 2.0
        diff = target - h.sum()
        for _ in range(2000):

                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])

        return h

        bc = CURRENT_BETA * corr
        m = np.max(bc)
        # use log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

            h0,
            method='SLSQP',
            bounds=bounds,

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for vector h on [0,2]."""
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1] (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int) -> np.ndarray:
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def gradient_descent_worst(h: np.ndarray, rng: np.random.Generator, max_iter: int, deadline: float) -> np.ndarray:
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    it = 0
    while it < max_iter and time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            break
        up = n - k
        g = np.zeros_like(h)
        g[:up] += 1.0 - h[k:]
        g[k:] -= h[:up]
        step = 0.02 * (0.8 ** it)
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        for _ in range(200):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h_new[i])
            else:
                add = max(diff, -h_new[i])
            h_new[i] += add
            diff = target - h_new.sum()
        h = h_new
        it += 1
    return h


def fractional_to_binary(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Take the largest n/2 entries as 1, the rest 0."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    adjust_sum_exact(bin_h, rng)
    return bin_h


def binary_targeted_improve(h: np.ndarray, rng: np.random.Generator, deadline: float) -> np.ndarray:
    """Greedy swaps that lower the worst overlap."""
    n = h.size
    current_c = compute_c5(h)
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        shift = abs(idx - (n - 1))
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        neigh_one = neigh[ones]
        min_val = neigh_one.min()
        cand_i = ones[neigh_one == min_val]
        i = rng.choice(cand_i)
        neigh_zero = neigh[zeros]
        max_val = neigh_zero.max()
        cand_j = zeros[neigh_zero == max_val]
        j = rng.choice(cand_j)

        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        if new_c + 1e-12 < current_c:
            current_c = new_c
            continue

        h[i] = 1.0
        h[j] = 0.0
        improved = False
        for _ in range(10):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            h[i2] = 0.0
            h[j2] = 1.0
            new_c2 = compute_c5(h)

                current_c = new_c2
                improved = True
                break
            h[i2] = 1.0
            h[j2] = 0.0
        if not improved:
            break
    return h


def generate_sawtooth(n: int) -> np.ndarray:
    """Symmetric trapezoid with height 2/3, integral 1."""
    a = 2.0 / 3.0
    i = np.arange(n)
    quarter = n // 4
    three_quarter = 3 * n // 4
    hf = np.empty(n, dtype=float)
    mask1 = i < quarter
    mask2 = (i >= quarter) & (i < three_quarter)
    mask3 = i >= three_quarter
    hf[mask1] = a * (4.0 * i[mask1] / n)
    hf[mask2] = a
    hf[mask3] = a * (4.0 * (n - i[mask3]) / n)
    return hf
