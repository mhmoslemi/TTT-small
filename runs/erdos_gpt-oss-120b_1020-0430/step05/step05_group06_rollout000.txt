def anneal_binary(h, n, rng, deadline):
    best = h.copy()
    best_c = compute_c5(h)
    cur = h.copy()
    cur_c = best_c
    iteration = 0
    # initialize temperature based on initial c maybe 0.02
    T0 = 0.01
    # compute schedule params:
    total_iters_est = 200000 # estimate
    while time.time() < deadline and iteration < total_iters_est:
        # pick swap
        ones = np.flatnonzero(cur == 1)
        zeros = np.flatnonzero(cur == 0)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur[i], cur[j] = 0, 1
        new_c = compute_c5(cur)
        delta = new_c - cur_c
        # acceptance criteria:
        if delta < 0 or rng.random() < np.exp(-delta / max(T0 * (0.999**iteration), 1e-8)):
            cur_c = new_c
            # update best if needed
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            # revert swap
            cur[i], cur[j] = 1, 0
        iteration += 1
    return best, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- evaluation helpers ----------
def compute_c5(h):
    """C5 for a (float) vector using full correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def compute_c5_bin(h):
    """C5 for a binary (0/1) vector using self‑correlation."""
    n = h.size
    sc = np.correlate(h, h, mode="full")          # length 2n‑1
    # exclude the central element (k = 0) which equals n/2
    min_self = np.min(np.concatenate((sc[:n - 1], sc[n:])))
    return 1.0 - (2.0 / n) * float(min_self)

def adjust_sum_exact(h, rng):
    """Exact correction to enforce Σh = n/2 while keeping 0≤h≤1."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

# ---------- SLSQP smooth‑max optimizer ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # smooth max of corr = h * (1‑h) via log‑sum‑exp
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

# ---------- greedy binary construction ----------
def greedy_binary(n, rng):
    """Construct a binary vector (0/1) with exactly n/2 ones."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # overlap of all‑zero vector
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0  # O_0 = 0 (not used)
    return h

# ---------- simulated annealing on binary vectors ----------
def anneal_binary(h_start, n, rng, deadline):
    """Simulated annealing (swap 1↔0) on a binary vector."""
    cur = h_start.astype(np.int8).copy()
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    cur_c = compute_c5_bin(cur)
    best = cur.copy()
    best_c = cur_c

    iter_cnt = 0
    T0 = 0.03          # initial temperature
    decay = 0.9995     # exponential cooling factor
    max_iters = 400000  # safety cap

    while time.time() < deadline and iter_cnt < max_iters:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = int(ones[i_idx])
        j = int(zeros[j_idx])

        # perform swap
        cur[i] = 0
        cur[j] = 1

        new_c = compute_c5_bin(cur)
        delta = new_c - cur_c
        T = T0 * (decay ** iter_cnt)

        accept = False
        if delta < 0:
            accept = True
        else:
            # exp(‑Δ/T) may underflow, guard the denominator
            if T > 1e-12 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_c = new_c
            # update index lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            # revert swap
            cur[i] = 1
            cur[j] = 0

        iter_cnt += 1

    return best, best_c
