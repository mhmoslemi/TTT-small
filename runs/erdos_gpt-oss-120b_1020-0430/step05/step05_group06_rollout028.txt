    if w is None:

def compute_grad_from_w(h, w):
    n = h.shape[0]
    g = np.zeros_like(h)
    # k >= 0
    for k in range(n):
        wk = w[n-1 + k]
        if wk == 0:
            continue
        nk = n - k
        if nk > 0:
            g[:nk] += wk
            g[:nk] -= wk * h[k:]
        if k > 0:
            g[k:] -= wk * h[:nk]
    # k < 0: k = -kp
    for kp in range(1, n):
        wk = w[n-1 - kp]
        if wk == 0:
            continue
        nk = n - kp
        # +1 for j >= kp
        g[kp:] += wk
        g[kp:] -= wk * h[:nk]
        # -h[j + kp] for j<nk
        g[:nk] -= wk * h[kp:]
    return g

def optimize_from(h0, deadline):
    n = h0.size
    target = n/2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step0 = 0.5
    step = step0
    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000]
    for beta in beta_schedule:
        max_iter = 400
        for it in range(max_iter):
            if time.time() > deadline:
                return best_h, best_c5
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = corr.max()
            c5_val = max_corr / target
            # compute smooth max values
            lse = logsumexp(beta * corr)
            w = np.exp(beta * corr - lse)
            g = compute_grad_from_w(h, w)
            norm = np.linalg.norm(g)
            if norm == 0:
                break
            # normalized step
            h_new = h - (step / (norm + 1e-12)) * g
            # project
            h_new = project_capped(h_new, target)
            # evaluate new c5
            corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            max_corr_new = corr_new.max()

                best_h = h_new.copy()

                h = h_new
                # increase step moderately
                step = min(step * 1.2, 5.0)
            else:
                # reduce step
                step = max(step * 0.5, 1e-6)
            # optional break if step small and no improvement for a while
            # not needed
    return best_h, best_c5

def compute_grad_from_w(h, w):
    n = h.shape[0]
    g = np.zeros_like(h)
    # k >= 0
    for k in range(n):
        wk = w[n - 1 + k]
        if wk == 0.0:
            continue
        nk = n - k
        # +1 term for j in [0, nk)
        g[:nk] += wk
        # subtract h[j+k] for j in [0, nk)
        if nk > 0:
            g[:nk] -= wk * h[k:]
        # subtract h[j - k] for j in [k, n)
        g[k:] -= wk * h[:nk]
    # k < 0, i.e., kp = 1..n-1
    for kp in range(1, n):
        wk = w[n - 1 - kp]
        if wk == 0.0:
            continue
        nk = n - kp
        # +1 term for j in [kp, n)
        g[kp:] += wk
        # subtract h[j - kp] for j in [kp, n)
        g[kp:] -= wk * h[:nk]
        # subtract h[j + kp] for j in [0, nk)
        g[:nk] -= wk * h[kp:]
    return g

import time
import numpy as np
from scipy.special import logsumexp

# ---------- utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def project_capped(v, target, iters=60):
    """Projection onto [0,1]^n with sum = target."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad_from_w(h, w):
    """Gradient of smooth max w.r.t. h given softmax weights w."""
    n = h.shape[0]
    g = np.zeros_like(h)
    # k >= 0
    for k in range(n):
        wk = w[n - 1 + k]
        if wk == 0.0:
            continue
        nk = n - k
        g[:nk] += wk                     # +1 term
        g[:nk] -= wk * h[k:]              # -h_{j+k}
        g[k:]  -= wk * h[:nk]             # -h_{j-k}
    # k < 0
    for kp in range(1, n):
        wk = w[n - 1 - kp]
        if wk == 0.0:
            continue
        nk = n - kp
        g[kp:] += wk                     # +1 term
        g[kp:] -= wk * h[:nk]             # -h_{j-kp}
        g[:nk] -= wk * h[kp:]             # -h_{j+kp}
    return g

def optimize_from(h_start, deadline):
    n = h_start.size
    target = n / 2.0
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.5                       # base step length in Euclidean norm
    beta_sched = [5, 20, 80, 200, 500, 1000, 2000]
    for beta in beta_sched:
        max_iter = 400
        for _ in range(max_iter):
            if time.time() > deadline:
                return best_h, best_c5
            corr = np.correlate(h, 1.0 - h, mode="full")
            max_corr = corr.max()
            c5_val = max_corr / target
            lse = logsumexp(beta * corr)
            w = np.exp(beta * corr - lse)          # softmax weights, sum to 1
            g = compute_grad_from_w(h, w)
            gn = np.linalg.norm(g)
            if gn == 0.0:
                break
            # take a normalized step
            h_new = h - (step / (gn + 1e-12)) * g
            h_new = project_capped(h_new, target)
            corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
            c5_new = corr_new.max() / target
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
            if c5_new < c5_val - 1e-12:
                h = h_new
                step = min(step * 1.2, 5.0)
            else:
                step = max(step * 0.5, 1e-6)
            if step < 1e-8:
                break
    return best_h, best_c5

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    n = 360
    target = n / 2.0

    best_h = None
    best_c5 = np.inf

    # use provided construction if any
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_capped(h0, target)
        c0 = compute_c5(h0)
        best_h, best_c5 = h0, c0

    # auxiliary deterministic start (smooth sinusoid)
    def sinusoid_start():
        x = np.arange(n)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * x / 24.0)
        h = np.clip(h, 0.0, 1.0)
        return project_capped(h, target)

    start_points = []
    if best_h is not None:
        start_points.append(best_h.copy())
    start_points.append(sinusoid_start())
    for _ in range(8):
        h_rand = rng.random(n)
        h_rand = project_capped(h_rand, target)
        start_points.append(h_rand)

    for h0 in start_points:
        if time.time() > deadline:
            break
        h_opt, c_opt = optimize_from(h0, deadline)
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt.copy()

    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_h = project_capped(best_h, target)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_capped(best_h, target)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(n)
