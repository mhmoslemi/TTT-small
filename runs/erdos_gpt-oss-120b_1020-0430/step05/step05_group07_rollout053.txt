def proj(x, target):
    low = -1.0
    high = 2.0
    for _ in range(60):
        mid = (low+high)/2
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(x - (low+high)/2, 0.0, 1.0)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def subgradient_optimize(h0, max_iters, deadline):
    n = h0.size
    target = n/2
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha0 = 0.5
    eps = 1e-12
    it = 0
    while it < max_iters and time.time() < deadline:
        # compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        max_val = corr.max()
        idxs = np.where(corr >= max_val - eps)[0]
        # gradient average
        g = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n-1)
            # compute grad
            if k == 0:
                g += 1 - 2*h
            elif k > 0:
                g += 1
                g[:n-k] -= h[k:]
                g[k:] -= h[:n-k]
            else:
                kk = -k
                g += 1
                g[kk:] -= h[:n-kk]
                g[:n-kk] -= h[kk:]
        g /= len(idxs)
        alpha = alpha0 / np.sqrt(it+1)
        h = proj(h - alpha * g, target)
        it += 1
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # occasional noise
        if it % 2000 == 0:
            noise = np.random.normal(scale=0.1, size=n)  # we will use rng
            h = proj(h + noise, target)
    return best_h, best_c

def binarize(h):
    n = h.size
    idx = np.argpartition(h, -n//2)[-n//2:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def binary_sa(h_bin, rng, time_limit):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    # maintain list of ones and zeros indices for quick swapping
    ones = list(np.flatnonzero(cur))
    zeros = list(np.flatnonzero(1-cur))
    it = 0
    T0 = max(0.01*cur_c, 1e-4)
    while time.time() < time_limit:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur)
        T = T0 * (0.9995 ** it) + 1e-12
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c)/T):
            cur_c = new_c
            # update lists
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            # revert
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it % 5000 == 0 and best_c < cur_c * 0.999:  # optional break
            pass
    return best, best_c

def grad_for_shift(h, k):
    n = h.size
    g = np.ones(n)
    if k == 0:
        g -= 2 * h
    elif k > 0:
        g[:n-k] -= h[k:]
        g[k:] -= h[:n-k]
    else:
        kk = -k
        g[kk:] -= h[:n-kk]
        g[:n-kk] -= h[kk:]
    return g

def subgradient_optimize(h0, max_iters, deadline, rng,
                         epsilon=1e-12, alpha0=0.5, noise_scale=0.05, noise_every=2000):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        idxs = np.where(corr >= max_val - epsilon)[0]
        # compute average gradient
        g = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            g += grad_for_shift(h, k)
        g /= len(idxs)
        alpha = alpha0 / np.sqrt(it + 1)
        h = proj(h - alpha * g, target)
        # occasional noise
        if noise_every and (it + 1) % noise_every == 0:
            h = proj(h + rng.normal(scale=noise_scale, size=n), target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def proj(x, target):
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

        h = proj(h - alpha * g, target)
        if noise_every and (it + 1) % noise_every == 0:
            h = proj(h + rng.normal(scale=noise_scale, size=n), target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

def binarize(h):
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2:]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def binary_sa(h_bin, rng, deadline):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best, best_c = cur.copy(), cur_c
    ones = list(np.flatnonzero(cur))
    zeros = list(np.flatnonzero(1 - cur))
    it = 0
    T0 = max(0.01 * cur_c, 1e-4)
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0.0
        cur[j] = 1.0
        c = compute_c5(cur)
        T = T0 * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c, best = c, cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
        it += 1
    return best, best_c

def softmax_optimize(h0, beta, max_iters, deadline, rng):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        maxc = corr.max()
        expb = np.exp(beta * (corr - maxc))
        w = expb / expb.sum()
        g = np.zeros_like(h)
        for idx, wi in enumerate(w):
            if wi == 0:
                continue
            k = idx - (n - 1)
            g += wi * grad_for_shift(h, k)
        alpha = 0.5 / np.sqrt(it + 1)
        h = proj(h - alpha * g, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    n_vals = [240, 360, 480, 600, 720]
    prev_h = None
    prev_n = None

    for n in n_vals:
        if time.time() >= deadline:
            break
        if prev_h is not None:
            xs = np.linspace(0, 2, prev_n, endpoint=False)
            xt = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(xt, xs, prev_h)
        elif 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0, 2, parent.size, endpoint=False)
            xt = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(xt, xs, parent)
        else:
            h0 = np.zeros(n)
            h0[rng.choice(n, n // 2, replace=False)] = 1.0
        h0 = proj(h0, n / 2.0)

        time_left = deadline - time.time()
        max_iters = int(min(20000, max(2000, time_left * 30)))
        h_opt, c_opt = subgradient_optimize(h0, max_iters, deadline, rng)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, n

        h_bin = binarize(h_opt)
        time_left = deadline - time.time()
        if time_left > 5.0:
            sa_dead = time.time() + min(0.2 * time_left, 20.0)
            h_sa, c_sa = binary_sa(h_bin, rng, sa_dead)
            if c_sa < best_c:
                best_h, best_c, best_n = h_sa.copy(), c_sa, n
        else:
            h_sa, c_sa = h_opt, c_opt

        time_left = deadline - time.time()
        if time_left > 1.0:
            start_h = h_sa if c_sa < c_opt else h_opt
            extra_iters = int(min(5000, max(500, time_left * 30)))
            h_pol, c_pol = subgradient_optimize(start_h, extra_iters, deadline, rng)
            if c_pol < best_c:
                best_h, best_c, best_n = h_pol.copy(), c_pol, n

        # final softmax refinement
        time_left = deadline - time.time()
        if time_left > 2.0 and best_h is not None:
            beta = 5000.0
            iters = int(min(400, max(100, time_left * 20)))
            best_h = softmax_optimize(best_h, beta, iters, deadline, rng)
            best_c = compute_c5(best_h)

        prev_h, prev_n = best_h.copy(), best_n

    if best_h is None:
        n = 240
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_n = n
    best_h = proj(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
