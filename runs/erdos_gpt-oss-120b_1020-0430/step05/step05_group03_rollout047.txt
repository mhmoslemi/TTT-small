import time, numpy as np, math
from scipy.optimize import differential_evolution, LinearConstraint, minimize

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.shape[0]
    target = n / 2
    diff = target - h.sum()
    # while loop as earlier but may be risky infinite. Use tolerance.
    max_iter = 0
    while abs(diff) > 1e-12 and max_iter < 10000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0:
            max_iter += 1
            continue
        h[i] += add
        diff = target - h.sum()
        max_iter += 1
    return h

def worst_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k

def max_step_alpha(h, grad):
    pos = grad > 1e-15
    neg = grad < -1e-15
    # avoid division by zero
    alphas = []
    # For positive grad: h_i - alpha*grad_i >= 0 => alpha <= h_i / grad_i
    if np.any(pos):
        alphas.append(np.min(h[pos] / grad[pos]))
    # For negative grad: h_i - alpha*grad_i <= 1 => alpha <= (1 - h_i)/(-grad_i)
    if np.any(neg):
        alphas.append(np.min((1.0 - h[neg]) / (-grad[neg])))
    if not alphas:
        return 0.0
    return min(alphas)

def gradient_descent_worst(h, rng, time_limit):
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    while time.time() - start < time_limit:
        grad, k = worst_gradient(h)
        alpha_max = max_step_alpha(h, grad)
        if alpha_max < 1e-12:
            break
        # use a factor slightly below max to ensure we stay inside
        alpha = alpha_max * 0.999
        h_new = h - alpha * grad
        # clip and enforce sum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new + 1e-12 < best_c:
            best_c = c_new
            best_h = h_new.copy()
        # accept if improved else maybe still move? if not improved, break?
        # We'll still update h to h_new (to keep moving)
        h = h_new
    return best_h, best_c

def block_de_optimize(n, B, rng, max_iter, time_limit):
    # B blocks, each length L = n // B (assume divisible)
    L = n // B
    target_sum = (n / 2) / L  # sum of block values needed
    bounds = [(0.0, 1.0)] * B
    linear_constraint = LinearConstraint(np.ones((1, B)), [target_sum], [target_sum])
    def objective(p):
        h = np.repeat(p, L)
        return compute_c5(h)
    start = time.time()
    result = differential_evolution(objective,
                                    bounds,
                                    constraints=(linear_constraint,),
                                    maxiter=max_iter,
                                    polish=False,
                                    updating='deferred',
                                    seed=int(rng.random() * 2**32))
    h = np.repeat(result.x, L)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)
    return h

def slsqp_refine(h, rng, beta_values, max_iter_per):
    h_cur = h.copy()
    n = h_cur.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq", "fun":lambda x: np.sum(x) - target}
    for beta in beta_values:
        def softmax_obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            bc = beta * corr
            m = np.max(bc)
            # log-sum-exp trick
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta
        res = minimize(softmax_obj, h_cur, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter_per,
                                                 'ftol':1e-12,'disp':False})
        h_cur = np.clip(res.x, 0.0, 1.0)
        h_cur = enforce_sum(h_cur, rng)
    return h_cur

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Initialize h

        h0 = enforce_sum(h0, rng)
        best_h = h0.copy()
        best_c = compute_c5(best_h)
    else:
        n0 = 256
        h0 = rng.random(n0)
        h0 *= (n0/2) / h0.sum()
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0.copy()
        best_c = compute_c5(best_h)
    # Phase 1: gradient descent on worst shift (coarse)
    time_left = budget_s - (time.time() - start)
    phase1_time = min(0.25 * budget_s, time_left * 0.5)
    h_gd, c_gd = gradient_descent_worst(best_h.copy(), rng, phase1_time)
    if c_gd < best_c:
        best_h, best_c = h_gd, c_gd
    # Phase 2: block DE search with various B
    time_left = budget_s - (time.time() - start)
    block_B_list = [32, 64, 128]  # B must divide n
    n = 512
    # if best_h size not n, upsample
    if best_h.size != n:
        x_old = np.linspace(0.0,2.0,best_h.size,endpoint=False)
        x_new = np.linspace(0.0,2.0,n,endpoint=False)
        best_h = np.interp(x_new, x_old, best_h)
        best_h = np.clip(best_h,0.0,1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
    for B in block_B_list:
        if time.time() - start > budget_s * 0.6:
            break
        # allocate some time for this block DE
        block_time = min(0.12 * budget_s, budget_s - (time.time() - start) - 0.1*budget_s)
        # set max_iter based on time estimate (few tens)
        max_iter = 30
        h_candidate = block_de_optimize(n, B, rng, max_iter, block_time)
        # local refinement with gradient descent
        h_refined, c_refined = gradient_descent_worst(h_candidate.copy(), rng, min(0.05*budget_s, budget_s - (time.time() - start)))
        if c_refined < best_c:
            best_h, best_c = h_refined.copy(), c_refined
    # Phase 3: SLSQP refinement with high beta
    time_left = budget_s - (time.time() - start)
    if time_left > 0.1 * budget_s:
        beta_vals = [20.,80.,200.,800.,3000.,8000.,20000.,50000.]
        max_iter = 800
        h_slsqp = slsqp_refine(best_h.copy(), rng, beta_vals, max_iter)
        c_slsqp = compute_c5(h_slsqp)
        if c_slsqp < best_c:
            best_h, best_c = h_slsqp.copy(), c_slsqp
    # Phase 4: final stochastic hill climbing (guided perturbations)
    while time.time() - start < budget_s * 0.99:
        # use guided perturb but with tiny step
        grad, _ = worst_gradient(best_h)
        # choose top few pos and neg indices
        pos_inds = np.where(grad > np.percentile(grad, 80))[0]
        neg_inds = np.where(grad < np.percentile(grad, 20))[0]
        if pos_inds.size == 0 or neg_inds.size == 0:
            i = rng.integers(0, best_h.size)
            j = rng.integers(0, best_h.size)
        else:
            i = int(rng.choice(pos_inds))
            j = int(rng.choice(neg_inds))
        max_eps = min(best_h[i], 1.0 - best_h[j])
        eps = max_eps * rng.random() * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = enforce_sum(nh, rng)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc
    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np
from scipy.optimize import differential_evolution, LinearConstraint, minimize

# ------------------- utilities ------------------- #
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def worst_gradient(h: np.ndarray):
    """Gradient of C(k) for the currently worst shift k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def max_step_alpha(h: np.ndarray, grad: np.ndarray) -> float:
    """Largest α≥0 such that h-α·grad stays in [0,1]."""
    pos = grad > 1e-15
    neg = grad < -1e-15
    alphas = []
    if np.any(pos):
        alphas.append(np.min(h[pos] / grad[pos]))
    if np.any(neg):
        alphas.append(np.min((1.0 - h[neg]) / (-grad[neg])))
    if not alphas:
        return 0.0
    return min(alphas)


def gradient_descent_worst(h: np.ndarray, rng: np.random.Generator, time_limit: float):
    """Projected gradient descent on the worst‑shift overlap."""
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    while time.time() - start < time_limit:
        grad, _ = worst_gradient(h)
        alpha_max = max_step_alpha(h, grad)
        if alpha_max < 1e-12:
            break
        alpha = alpha_max * 0.999
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new + 1e-12 < best_c:
            best_c, best_h = c_new, h_new.copy()
        h = h_new
    return best_h, best_c


def block_de_optimize(n: int, B: int, rng: np.random.Generator, max_iter: int):
    """DE on a piecewise‑constant representation with B blocks."""
    L = n // B
    target_sum = (n / 2.0) / L
    bounds = [(0.0, 1.0)] * B
    lin_con = LinearConstraint(np.ones((1, B)), [target_sum], [target_sum])

    def obj(p):
        h = np.repeat(p, L)
        return compute_c5(h)

    result = differential_evolution(
        obj,
        bounds,
        constraints=(lin_con,),
        maxiter=max_iter,
        popsize=12,
        polish=False,
        updating="deferred",
        seed=int(rng.random() * 2**32),
    )
    h = np.repeat(result.x, L)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)
    return h


def slsqp_refine(h: np.ndarray, rng: np.random.Generator,
                 beta_vals, max_iter: int):
    """SLSQP on the smooth max (log‑sum‑exp) approximation."""
    n = h.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    cur = h.copy()
    for beta in beta_vals:
        def softmax_obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            bc = beta * corr
            m = np.max(bc)
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta

        res = minimize(
            softmax_obj,
            cur,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        cur = np.clip(res.x, 0.0, 1.0)
        cur = enforce_sum(cur, rng)
    return cur
