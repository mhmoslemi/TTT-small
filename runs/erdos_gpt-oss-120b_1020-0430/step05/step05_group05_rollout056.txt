def solve_binary_milp(n, time_limit):
    h = cp.Variable(n, boolean=True)
    T = cp.Variable()
    constraints = [cp.sum(h) == n//2, T >= 0]

    # Precompute prefix sums for h: but cp can't have dynamic sum over prefix; we can compute via cp.sum(h[:n-s]) easily using slicing.
    for s in range(1, n):
        # length = n - s
        length = n - s
        if length == 0:
            continue
        # product variables
        y = cp.Variable(length, boolean=True)
        # indexing
        h_i = h[:length]
        h_i_s = h[s:s+length]
        constraints += [y <= h_i, y <= h_i_s, y >= h_i + h_i_s - 1]
        C_s = cp.sum(h_i) - cp.sum(y)
        constraints.append(C_s <= T)
    prob = cp.Problem(cp.Minimize(T), constraints)
    try:
        prob.solve(solver=cp.CBC, seconds=time_limit, warm_start=False, verbose=False)
    except Exception:
        # fallback maybe
        return None, None
    if prob.status not in ["optimal", "optimal_inaccurate", "feasible"]:
        return None, None
    h_sol = np.round(h.value).astype(float)  # ensure binary
    T_val = T.value
    return h_sol, T_val
```

def binary_targeted_improve(h, rng, deadline):
    n = len(h)
    while time.time() < deadline:
        corr = np.correlate(h, 1-h, mode='full')
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        # compute overlap region
        k = abs(shift)
        if k == 0:
            break
        # Determine indices in overlapping region (i from 0 to n-1-k) where h_i contributes with (1 - h_{i+k})
        # neighbor measure: for each i, define weight = (1 - h[i+k]) + (1 - h[i-k])? Not exactly.
        # We'll just choose a swap that reduces C_s:
        # Find a 1 inside region where (1 - h_{i+k}) is 1 (i.e., h[i+k] == 0) (so contributes 1)
        # and a 0 outside region where (1 - h_{j+k}) is 0 maybe.
        # Simpler: find index i where h[i]==1 and i < n-k, and h[i+k]==0 (so product adds 1).
        # Swapping this i to 0 reduces C_s by 1.
        # Then find index j where h[j]==0 and j >= n-k (outside region) (so swapping in 1 doesn't affect C_s)
        # If such i,j exist, perform swap.
        # If not, break.
        region = np.arange(n - k)

        if zeros_out.size == 0:
            break
        j = rng.choice(zeros_out) + (n - k)
        # Perform swap
        h[i] = 0

def adjust_sum_exact(h, rng):
    n = len(h)
    target = n // 2
    diff = target - h.sum()
    while diff != 0:
        i = rng.integers(0,n)
        if diff > 0:
            if h[i] == 0:
                h[i] = 1
                diff -= 1
        else:
            if h[i] == 1:
                h[i] = 0
                diff += 1
    return h

import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = len(h)
    target = n // 2
    diff = target - int(h.sum())
    for _ in range(10000):
        if diff == 0:
            break
        i = rng.integers(0, n)
        if diff > 0 and h[i] == 0:
            h[i] = 1
            diff -= 1
        elif diff < 0 and h[i] == 1:
            h[i] = 0
            diff += 1
    return h

def binary_targeted_improve(h, rng, deadline):
    n = len(h)
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        k = abs(shift)
        if k == 0:
            break
        length = n - k
        region = np.arange(length)
        outside = np.arange(length, n)
        inside_one = region[(h[region] == 1) & (h[region + k] == 0)]
        if inside_one.size == 0:
            break
        i = rng.choice(inside_one)
        zero_out = outside[h[outside] == 0]
        if zero_out.size == 0:
            break
        j = rng.choice(zero_out)
        h[i] = 0
        h[j] = 1
    return h

def solve_milp_binary(n, time_limit, rng):
    h = cp.Variable(n, boolean=True)
    T = cp.Variable()
    constraints = [cp.sum(h) == n // 2, T >= 0]
    for s in range(1, n):
        length = n - s
        if length == 0:
            continue
        y = cp.Variable(length, boolean=True)
        h_i = h[:length]
        h_i_s = h[s:s + length]
        constraints += [y <= h_i,
                        y <= h_i_s,
                        y >= h_i + h_i_s - 1]
        cs_pos = cp.sum(h_i) - cp.sum(y)
        cs_neg = cp.sum(h_i_s) - cp.sum(y)
        constraints += [cs_pos <= T, cs_neg <= T]
    prob = cp.Problem(cp.Minimize(T), constraints)
    try:
        prob.solve(solver=cp.CBC, seconds=time_limit, warm_start=False, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS_BB, max_iters=int(time_limit * 1000),
                       warm_start=False, verbose=False)
        except Exception:
            return None, None
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE):
        if h.value is None:
            return None, None
    h_val = np.round(h.value).astype(int)
    T_val = T.value
    return h_val, T_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = float("inf")
    best_n = None

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
        parent_bin = (parent >= 0.5).astype(int)
        parent_bin = adjust_sum_exact(parent_bin, rng)
        c0 = compute_c5(parent_bin)
        best_h, best_c, best_n = parent_bin, c0, n0

    for n in (180, 240, 360):
        if time.time() > deadline:
            break
        rem = deadline - time.time()
        tlim = max(30.0, min(300.0, rem * 0.6))
        h_bin, _ = solve_milp_binary(n, tlim, rng)
        if h_bin is None:
            continue
        h_bin = adjust_sum_exact(h_bin, rng)
        c = compute_c5(h_bin)
        if c < best_c:
            best_h, best_c, best_n = h_bin, c, n
        if time.time() < deadline:
            h_imp = binary_targeted_improve(h_bin.copy(), rng, deadline)
            c_imp = compute_c5(h_imp)
            if c_imp < best_c:
                best_h, best_c, best_n = h_imp, c_imp, n

    if best_h is None:
        n = 200
        h = np.zeros(n, dtype=int)
        h[: n // 2] = 1
        best_h, best_c, best_n = h, compute_c5(h), n

    best_h = np.clip(best_h, 0, 1)
    best_h = adjust_sum_exact(best_h.astype(int), rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
