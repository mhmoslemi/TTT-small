def hill_climb_swaps(h_bin, max_iter, rng, deadline):
    n = h_bin.size
    ones = np.flatnonzero(h_bin)
    zeros = np.flatnonzero(1 - h_bin) # indices where h=0
    cur_h = h_bin.copy()
    cur_corr = np.max(np.correlate(cur_h.astype(float), 1 - cur_h.astype(float), mode='full'))
    best_h = cur_h.copy()
    best_corr = cur_corr
    it = 0
    while it < max_iter and time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0: break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        # update ones/zeros lists
        # simple approach: modify arrays
        # We can remove i from ones and add j; remove j from zeros and add i
        # Use naive list for speed maybe.
        # But we can recompute ones/zeros each iteration? That's O(n). But max_iter 5000, acceptable.
        # Simpler: just recompute after swap
        cur_corr = np.max(np.correlate(cur_h.astype(float), 1 - cur_h.astype(float), mode='full'))
        if cur_corr < best_corr:
            best_corr = cur_corr
            best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        # recompute ones and zeros
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        it += 1
    return best_h, best_corr

def simulated_annealing_swaps(h_bin, rng, deadline, init_temp=1.0, final_temp=0.001):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_corr = np.max(np.correlate(cur_h.astype(float), 1 - cur_h.astype(float), mode='full'))
    best_h = cur_h.copy()
    best_corr = cur_corr
    start_time = time.time()
    # Determine total steps based on available time
    # we can set steps approx to 20000 or more.
    max_steps = 50000
    # we will break if time runs out
    for step in range(max_steps):
        now = time.time()
        if now >= deadline:
            break
        # linear schedule
        progress = (now - start_time) / (deadline - start_time)
        temp = init_temp * (1 - progress) + final_temp * progress
        # if temp very small, may as well break
        # propose swap
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_corr = np.max(np.correlate(cur_h.astype(float), 1 - cur_h.astype(float), mode='full'))
        delta = new_corr - cur_corr
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            cur_corr = new_corr
            if cur_corr < best_corr:
                best_corr = cur_corr
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1

def simulated_annealing_swaps(h_bin, rng, deadline, init_temp=1.0, final_temp=0.001):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_corr = np.max(np.correlate(cur_h.astype(float), 1 - cur_h.astype(float), mode='full'))
    best_h = cur_h.copy()
    best_corr = cur_corr
    ones = list(np.flatnonzero(cur_h))
    zeros = list(np.flatnonzero(1 - cur_h))
    start_time = time.time()
    # maximum steps
    max_steps = 200000  # large
    while time.time() < deadline and max_steps > 0:
        # compute temperature based on elapsed
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total if total > 0 else 1.0
        temp = init_temp * (1.0 - progress) + final_temp * progress
        if len(ones) == 0 or len(zeros) == 0:
            break
        pos_i = rng.integers(len(ones))
        pos_j = rng.integers(len(zeros))
        i = ones[pos_i]
        j = zeros[pos_j]
        # swap in cur_h
        cur_h[i] = 0
        cur_h[j] = 1
        new_corr = np.max(np.correlate(cur_h.astype(float), 1 - cur_h.astype(float), mode='full'))
        delta = new_corr - cur_corr
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            cur_corr = new_corr
            # update lists
            ones[pos_i] = j
            zeros[pos_j] = i
            if cur_corr < best_corr:
                best_corr = cur_corr
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        max_steps -= 1
    return best_h, best_corr

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k))dx  (discrete version)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> None:
    """Modify h in‑place so that Σh = n/2 exactly, respecting 0≤h≤1."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

# ---------- greedy binary ----------
def greedy_binary(n: int, rng: np.random.Generator):
    """Construct a binary vector with Σh = n/2 by greedy minimisation of the max overlap."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)   # correlation of current vector
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start:start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h, O

# ---------- simulated annealing on binary ----------
def simulated_annealing_swaps(h_bin: np.ndarray, rng: np.random.Generator,
                             deadline: float, init_temp: float = 1.0,
                             final_temp: float = 0.001) -> tuple:
    """Swap 1↔0 moves with Metropolis acceptance."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_corr = np.max(np.correlate(cur_h.astype(float), 1 - cur_h.astype(float), mode="full"))
    best_h = cur_h.copy()
    best_corr = cur_corr
    ones = list(np.flatnonzero(cur_h))
    zeros = list(np.flatnonzero(1 - cur_h))
    start = time.time()
    while time.time() < deadline:
        if not ones or not zeros:
            break
        # temperature linear in elapsed fraction
        elapsed = time.time() - start
        total = deadline - start
        prog = elapsed / total if total > 0 else 1.0
        temp = init_temp * (1.0 - prog) + final_temp * prog

        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]

        cur_h[i] = 0
        cur_h[j] = 1
        new_corr = np.max(np.correlate(cur_h.astype(float), 1 - cur_h.astype(float), mode="full"))
        delta = new_corr - cur_corr
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / temp):
                accept = True
        if accept:
            cur_corr = new_corr
            ones[i_pos] = j
            zeros[j_pos] = i
            if cur_corr < best_corr:
                best_corr = cur_corr
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_corr

# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    """SLSQP minimisation of the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

# ---------- random feasible start ----------
def random_feasible_h(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    target = n / 2.0
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    adjust_sum_exact(h, rng)
    return h

# ---------- optimisation for a fixed n ----------
def optimise_n(n: int, rng: np.random.Generator,
               deadline: float, incumbent_h: np.ndarray = None) -> tuple:
    """Return best (h, C5) for this n using a mixture of heuristics."""
    best_h = None
    best_c = np.inf

    # use incumbent if it matches size
    if incumbent_h is not None and incumbent_h.size == n:
        c = compute_c5(incumbent_h)
        best_h, best_c = incumbent_h.copy(), c

    # greedy binary start
    if time.time() < deadline:
        h_bin, _ = greedy_binary(n, rng)
        h_f = h_bin.astype(float)
        c = compute_c5(h_f)
        if c < best_c:
            best_h, best_c = h_f.copy(), c

    # simulated annealing on a binary version of the current best
    if time.time() < deadline and best_h is not None:
        h_bin_start = (best_h > 0.5).astype(np.int8)
        sa_h, corr = simulated_annealing_swaps(h_bin_start, rng, deadline)
        c = corr * (2.0 / n)
        if c < best_c:
            best_h, best_c = sa_h.astype(float).copy(), c

    # smooth‑max SLSQP cascade on the current best
    if time.time() < deadline and best_h is not None:
        adjust_sum_exact(best_h, rng)
        for beta in (10, 50, 200, 800, 2000, 8000):
            if time.time() >= deadline:
                break
            try:
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=800)
            except Exception:
                continue
            c = compute_c5(best_h)
            if c < best_c:
                best_c = c

    # random‑restart SLSQP for any remaining time
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 0.6:
            break
        h0 = random_feasible_h(n, rng)
        h = h0
        for beta in (5, 20, 80, 200, 800):
            if time.time() >= deadline:
                break
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=400)
            except Exception:
                break
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
    return best_h, best_c

# ---------- main entry ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # parent construction, if supplied

    for n in (360, 720):
        if time.time() >= deadline:
            break
        # give each n about half of the remaining budget
        sub_deadline = time.time() + (deadline - time.time()) * 0.5
        h_n, c_n = optimise_n(n, rng, sub_deadline,
                               incumbent_h=best_h if best_n == n else None)
        if c_n < best_c:

    if best_h is None:

        best_h[: n // 2] = 1.0

        best_c = compute_c5(best_h)
    else:

        adjust_sum_exact(best_h, rng)
