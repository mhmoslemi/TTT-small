def binary_local_search(h, deadline):
    n = h.shape[0]
    half = n // 2
    # compute overlap array
    while True:
        # compute conv_hh
        conv = np.correlate(h, h, mode='full')
        center = n - 1
        # overlap for k=1..n-1
        overlaps = half - conv[center+1: center+n]  # integer
        cur_max = overlaps.max()
        c5 = cur_max * (2.0 / n)
        if time.time() > deadline:
            break
        # find worst shift
        k_star = int(overlaps.argmax()) + 1
        # compute candidates
        h_roll = np.roll(h, -k_star)  # h_{i+k}
        i_set = np.flatnonzero((h == 1) & (h_roll == 0))
        j_set = np.flatnonzero((h == 0) & (h_roll == 1))
        if i_set.size == 0 or j_set.size == 0:
            # no improvement possible via this k, break
            break
        # sample pairs
        max_trials = min(200, i_set.size * j_set.size)
        # use rng to sample indices
        best_new_max = cur_max
        best_pair = None
        for _ in range(max_trials):
            i = i_set[rng.integers(i_set.size)]
            j = j_set[rng.integers(j_set.size)]
            # compute delta for overlaps (size n-1)
            # delta_k = (h[(j+k)%n] + h[(j-k)%n]) - (h[(i+k)%n] + h[(i-k)%n])
            # We'll compute delta for k=1..n-1
            ks = np.arange(1, n, dtype=int)
            delta = (h.take((j + ks) % n) + h.take((j - ks) % n)) - (h.take((i + ks) % n) + h.take((i - ks) % n))
            new_max = np.max(overlaps - delta)
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_pair = (i, j, delta)
                if new_max < cur_max - 1e-6:
                    # good enough, break early?
                    pass
        if best_pair is None:
            break
        # apply best swap
        i, j, delta = best_pair
        h[i] = 0
        h[j] = 1
        # update overlaps: overlaps = overlaps - delta
        overlaps = overlaps - delta
        # continue loop
    return h

import time
import numpy as np
from scipy.optimize import minimize

# globals for the smooth‑max SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    # smooth maximum of the overlap vector
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (swap mass between random indices)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def binary_local_search(h_bin, rng, deadline, max_iters=200, max_trials=50):
    """
    h_bin : 0/1 integer array (exactly n/2 ones)
    Returns a (possibly) improved binary vector.
    """
    n = h_bin.shape[0]
    half = n // 2
    # initial overlaps for k=1..n-1
    conv = np.correlate(h_bin, h_bin, mode="full")
    center = n - 1
    overlaps = half - conv[center + 1: center + n]          # integer array, length n-1
    it = 0
    while it < max_iters and time.time() < deadline:
        cur_max = overlaps.max()
        if cur_max == 0:                     # perfect, cannot improve
            break
        k_star = int(overlaps.argmax()) + 1   # shift value (1..n-1)
        # candidates that would improve this shift
        h_roll = np.roll(h_bin, -k_star)      # h_{i+k}
        i_set = np.flatnonzero((h_bin == 1) & (h_roll == 0))
        j_set = np.flatnonzero((h_bin == 0) & (h_roll == 1))
        if i_set.size == 0 or j_set.size == 0:
            # no improving swap for this shift
            break
        best_new_max = cur_max
        best_pair = None
        ks = np.arange(1, n, dtype=int)      # shifts used for delta
        # random trials
        trials = min(max_trials, i_set.size * j_set.size)
        for _ in range(trials):
            i = i_set[rng.integers(i_set.size)]
            j = j_set[rng.integers(j_set.size)]
            delta = (h_bin.take((j + ks) % n) + h_bin.take((j - ks) % n) -
                     h_bin.take((i + ks) % n) - h_bin.take((i - ks) % n))
            new_max = np.max(overlaps - delta)
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_pair = (i, j, delta)
                # early exit if we already beat current max by a comfortable margin
                if best_new_max < cur_max - 1e-6:
                    break
        if best_pair is None:
            break
        i, j, delta = best_pair
        h_bin[i] = 0
        h_bin[j] = 1
        overlaps -= delta                      # exact update
        it += 1
    return h_bin


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (larger first – more freedom)
    candidate_ns = [720, 540, 360, 240, 180]

    parent_exists = 'initial_h_values' in globals()

        half = n // 2

            interp = np.interp(xt, xs, parent)
            idx = np.argpartition(interp, -half)[-half:]
            bin_vec = np.zeros(n, dtype=int)
            bin_vec[idx] = 1
            starts.append(bin_vec.copy())
            # a slightly perturbed version of the above

            for _ in range(4):

                if ones.size == 0 or zeros.size == 0:

    # final safety adjustments
