def project_box_simplex(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            hi = mid
        else:
            lo = mid

def subgradient_proj(h, steps, step0, decay, rng, target):
    n = h.size
    for t in range(steps):
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step = step0 * (decay ** t)
        y = h - step * g
        # project
        h = project_box_simplex(y, target)
    return h

def fractional_sa(h, rng, deadline, max_iter=100000, init_temp=0.01, decay=0.9999):
   # h currently within [0,1] sum target
   n = h.size
   target = n/2
   cur_c = compute_c5(h)
   best_h = h.copy()
   best_c = cur_c
   T = init_temp
   it = 0
   while time.time() < deadline and it < max_iter:
       # propose a move: pick i,j, delta
       i, j = rng.integers(0, n, size=2)
       if i == j:
           continue
       # delta limited by bounds and sum
       # we will transfer delta from i to j, respecting 0<=h_i - delta <=1 and 0<=h_j+delta<=1
       max_delta_i = h[i]  # can remove at most this (to 0)
       max_delta_j = 1.0 - h[j]  # can add at most to 1
       max_delta = min(max_delta_i, max_delta_j)
       if max_delta <= 0:
           it+=1
           continue
       # random delta scaled
       delta = rng.random() * max_delta * 0.5  # up to half of max possible
       # new h copy? Instead compute new C5 efficiently? Simpler recalc.
       h[i] -= delta
       h[j] += delta
       new_c = compute_c5(h)
       dE = new_c - cur_c
       accept = False
       if dE < -1e-12:
           accept = True
       else:
           if rng.random() < np.exp(-dE / max(T, 1e-12)):
               accept = True
       if accept:
           cur_c = new_c
           if cur_c < best_c:
               best_c = cur_c
               best_h = h.copy()
       else:
           # revert
           h[i] += delta
           h[j] -= delta
       T *= decay
       it += 1
   return best_h, best_c

def project_box_simplex(v, target):
    # project onto [0,1] with sum target
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - lo, 0.0, 1.0)
    # adjust numerical diff if needed
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff over entries not at bounds
        # find indices where 0 < x < 1
        free = np.where((x > 1e-12) & (x < 1-1e-12))[0]
        if free.size > 0:
            # split equally
            add = diff / free.size
            x[free] += add
        else:
            # fallback: adjust first entry
            x[0] += diff
        # clip again
        x = np.clip(x, 0.0, 1.0)
    return x

def subgradient_proj(h, steps, step0, decay, rng, target):
    for t in range(steps):
        g = worst_shift_gradient(h)
        step = step0 * (decay ** t)
        y = h - step * g

def fractional_sa(h, rng, deadline, max_iter=5000, init_temp=0.02, decay=0.9999):
    n = h.shape[0]
    target = n / 2.0
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = init_temp
    it = 0
    while time.time() < deadline and it < max_iter:
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if i == j:
            continue
        # compute max delta we can transfer
        max_delta_i = h[i]  # can reduce to 0
        max_delta_j = 1.0 - h[j]  # can increase to 1
        max_delta = min(max_delta_i, max_delta_j)
        if max_delta <= 0.0:
            it += 1
            continue
        delta = rng.random() * max_delta * 0.5
        h[i] -= delta
        h[j] += delta
        new_c = compute_c5(h)
        dE = new_c - cur_c
        accept = False
        if dE < -1e-12:
            accept = True
        else:
            # temperature may be zero for large iterations; ensure no division by zero
            if T > 1e-12 and rng.random() < np.exp(-dE / T):
                accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        T *= decay
        it += 1
    return best_h, best_c

def binary_sa(h_bin, rng, deadline, max_iter=200000):
    """Simulated‑annealing on binary vectors (swap 0↔1)."""
    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode="full")  # auto‑correlation
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    T0 = cur_worst * 0.05 + 1e-4
    decay = 0.99995
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # contribution of i (1→0)
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        diff_i = -1.0
        c[start_i : start_i + n] += diff_i * hi_vec
        h_bin[i] = 0.0

        # contribution of j (0→1) – uses updated h_bin
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j + 1 :]))
        diff_j = 1.0
        c[start_j : start_j + n] += diff_j * hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h_bin[j] = 0.0
            c[start_j : start_j + n] -= diff_j * hj_vec
            h_bin[i] = 1.0
            c[start_i : start_i + n] -= diff_i * hi_vec
            no_improve += 1

        if no_improve > 50000:
            break
        it += 1

    return best_h, best_worst

import time

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_box_simplex(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - lo, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            add = diff / free.size
            x[free] += add
        else:
            x[0] += diff
        x = np.clip(x, 0.0, 1.0)
    return x

def worst_shift_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    s = idx
    g = np.zeros_like(h)
    if s >= 0:
        k = s
        up = n - k
        if up:
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
    else:
        k = -s
        up = n - k
        if up:
            g[k:] += 1.0 - h[:up]
            g[:up] -= h[k:]
    return g

def subgradient_proj(h, steps, step0, decay, rng, target):
    for t in range(steps):
        g = worst_shift_gradient(h)
        step = step0 * (decay ** t)
        y = h - step * g
        h = project_box_simplex(y, target)
    return h

def fractional_sa(h, rng, deadline, max_iter=5000, init_temp=0.02, decay=0.9999):
    n = h.shape[0]
    target = n / 2.0
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = init_temp
    it = 0
    while time.time() < deadline and it < max_iter:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if i == j:
            continue
        max_di = h[i]
        max_aj = 1.0 - h[j]
        max_delta = min(max_di, max_aj)
        if max_delta <= 0.0:
            it += 1
            continue
        delta = rng.random() * max_delta * 0.5
        h[i] -= delta
        h[j] += delta
        new_c = compute_c5(h)
        dE = new_c - cur_c
        accept = False
        if dE < -1e-12:
            accept = True
        elif T > 1e-12 and rng.random() < np.exp(-dE / T):
            accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
        T *= decay
        it += 1
    return best_h, best_c

def binary_sa(h_bin, rng, deadline, max_iter=200000):
    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst
    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))
    T0 = cur_worst * 0.05 + 1e-4
    decay = 0.99995
    it = 0
    no_imp = 0
    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        c[start_i : start_i + n] -= hi_vec
        h_bin[i] = 0.0
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j + 1 :]))
        c[start_j : start_j + n] += hj_vec
        h_bin[j] = 1.0
        new_min = c.min()
        new_worst = target - new_min
        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        elif T > 1e-12 and rng.random() < np.exp(-(new_worst - cur_worst) / T):
            accept = True
        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:

            c[start_j : start_j + n] -= hj_vec
            h_bin[i] = 1.0
            c[start_i : start_i + n] += hi_vec
            no_imp += 1
        if no_imp > 50000:
            break
        it += 1
    return best_h, best_worst
