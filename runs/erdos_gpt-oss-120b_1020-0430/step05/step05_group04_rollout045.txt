def compute_c5(h):
    n = h.size
    # zero pad to length 2n
    N = 2 * n
    # compute conv via FFT
    fh = np.fft.rfft(h, N)
    f1mh = np.fft.rfft(1.0 - h, N)
    conv = np.fft.irfft(fh * f1mh)[:2*n-1]
    # maximum of conv
    return float(conv.max() * (2.0 / n))

def simulated_annealing_binary(h, rng, deadline, max_iter=50000):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # List of indices of ones and zeros
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    # Temperature schedule: start from initial fraction of cur_c (maybe 0.01*cur_c?) or cur_c? We'll use cur_c * 0.02.
    init_T = cur_c * 0.02 + 1e-6
    # cooling factor
    decay = 0.9999
    T = init_T
    it = 0
    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
        # update temperature
        T = T * decay
        if T < 1e-12:
            T = 1e-12
        it += 1
        # occasional random restart? maybe not needed.
    return best_h, best_c

import time, math
import numpy as np

def _corr_array(h: np.ndarray) -> np.ndarray:
    n = h.size
    N = 2 * n
    return np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(1.0 - h, N))[: 2 * n - 1]

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    return float(_corr_array(h).max() * (2.0 / n))

def _adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> None:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

def _local_greedy_binary(h: np.ndarray, rng: np.random.Generator,
                         deadline: float, max_iter: int = 500,
                         max_samples: int = 400) -> tuple[np.ndarray, float]:
    n = h.size
    best_c = compute_c5(h)
    it = 0
    while time.time() < deadline and it < max_iter and best_c > 0.3805:
        it += 1
        conv = _corr_array(h)
        conv[n - 1] = -np.inf                      # ignore zero shift
        # try shifts in decreasing order of current overlap
        order = np.argsort(-conv)
        improved = False
        for idx in order:
            k = idx - (n - 1)
            if k == 0:
                continue
            if k > 0:
                rng_i = np.arange(n - k)
                i1 = rng_i[(h[rng_i] == 1) & (h[rng_i + k] == 0)]
                i0 = rng_i[(h[rng_i] == 0) & (h[rng_i + k] == 1)]
            else:
                kabs = -k
                rng_i = np.arange(kabs, n)
                i1 = rng_i[(h[rng_i] == 1) & (h[rng_i + k] == 0)]
                i0 = rng_i[(h[rng_i] == 0) & (h[rng_i + k] == 1)]
            if i1.size == 0 or i0.size == 0:
                continue
            total = int(i1.size) * int(i0.size)
            if total <= max_samples:
                pairs = [(i, j) for i in i1 for j in i0]
            else:
                pairs = [(i1[rng.integers(i1.size)], i0[rng.integers(i0.size)])
                         for _ in range(max_samples)]
            best_pair = None
            best_c_new = best_c
            for i, j in pairs:
                h[i], h[j] = h[j], h[i]
                cnew = compute_c5(h)
                if cnew < best_c_new - 1e-12:
                    best_c_new, best_pair = cnew, (i, j)
                h[i], h[j] = h[j], h[i]
            if best_pair is not None:
                i, j = best_pair
                h[i], h[j] = h[j], h[i]
                best_c = best_c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c

def _sa_binary(h: np.ndarray, rng: np.random.Generator,
               deadline: float, max_iter: int = 5_000_000) -> tuple[np.ndarray, float]:
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    T = max(1e-6, cur_c * 0.02)
    decay = 0.9999
    it = 0
    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i], cur[j] = 0.0, 1.0
        cnew = compute_c5(cur)
        delta = cnew - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_c = cnew
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if cnew < best_c:
                best_c, best = cnew, cur.copy()
        else:
            cur[i], cur[j] = 1.0, 0.0
        T = max(T * decay, 1e-12)
        it += 1
    return best, best_c

def _initial_binary(parent: np.ndarray, n: int, rng: np.random.Generator,
                    perturb: bool = False) -> np.ndarray:
    xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    interp = np.interp(xt, xs, parent)
    idx = np.argpartition(interp, -n // 2)[-n // 2:]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    if perturb:
        swaps = max(1, n // 200)
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        for _ in range(swaps):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = 0.0, 1.0
            ones[ones == i] = j
            zeros[zeros == j] = i
    return h
