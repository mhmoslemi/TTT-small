    n = h.size
    best_c = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    it = 0

        g = worst_gradient(h)
        # choose i with max g where h[i] > 0
        candidates_i = np.where(h > 1e-12)[0]
        if len(candidates_i) == 0: break

        candidates_j = np.where(h < 1 - 1e-12)[0]

            no_improve += 1
        it += 1
    return best_h, best_c

def project_sum_box(y, target):
    # find λ via bisection
    lo = -1.0 - np.max(y)  # sufficiently low
    hi = 1.0 - np.min(y)   # sufficiently high
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h = np.clip(y + mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def greedy_swap_optimize(h, rng, max_no_improve=500, max_iters=20000):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    no_improve = 0
    it = 0
    # Precompute indices of ones and zeros each iteration? We'll compute each iteration.
    while it < max_iters and no_improve < max_no_improve:
        # gradient
        g = worst_gradient(h)
        # choose i among ones with max gradient
        ones = np.where(h > 0.5)[0]  # binary tolerance
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.argmax(g[ones])]
        j = zeros[np.argmin(g[zeros])]
        if i == j:
            break
        # swap
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
            no_improve += 1
        it += 1
    return best_h, best_c

def annealing_optimize(h, rng, steps, start_temp=0.01, end_temp=0.0001):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    for step in range(steps):
        # linear temperature schedule
        t = start_temp + (end_temp - start_temp) * (step / steps)
        # pick random i,j
        ones_idx = np.where(cur_h > 0.5)[0]
        zeros_idx = np.where(cur_h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # propose swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / t):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_c

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def project_sum_box(y: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0 - y.max(), 1.0 - y.min()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        h = np.clip(y + mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def greedy_swap_optimize(h0: np.ndarray, rng: np.random.Generator,
                         max_no_improve: int = 500, max_iters: int = 20000):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    no_imp = 0
    it = 0
    while it < max_iters and no_imp < max_no_improve:
        g = worst_gradient(h)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmax(g[ones])]
        j = zeros[np.argmin(g[zeros])]
        if i == j:
            break
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c, best_h, no_imp = c, h.copy(), 0
        else:
            h[i], h[j] = 1.0, 0.0
            no_imp += 1
        it += 1
    return best_h, best_c

def annealing_optimize(h0: np.ndarray, rng: np.random.Generator,
                       steps: int, start_temp: float = 0.01,
                       end_temp: float = 0.0001):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    for s in range(steps):
        t = start_temp + (end_temp - start_temp) * (s / steps)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / t):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c
