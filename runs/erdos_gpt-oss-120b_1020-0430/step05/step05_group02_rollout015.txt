def binary_sa_slice(h, rng, deadline, max_iter=200000):
    # h: binary int8 array
    n = h.shape[0]
    target = n // 2
    c = np.correlate(h, h, mode='full')
    c = c.astype(np.int32)
    c_center = n - 1
    cur_min = c.min()
    cur_worst = target - cur_min
    best_worst = cur_worst
    best_h = h.copy()
    # maintain position arrays
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)  # since h is 0/1
    pos_one = np.full(n, -1, dtype=np.int32)
    pos_zero = np.full(n, -1, dtype=np.int32)
    pos_one[ones] = np.arange(len(ones))
    pos_zero[zeros] = np.arange(len(zeros))

    T0 = cur_worst * 0.05 + 1e-6
    decay = 0.99997
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        prev_worst = cur_worst
        # choose random i/j
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # apply swap (i 1->0, j 0->1)
        # removal of i
        c[c_center] -= 1
        if i+1 < n:
            c[c_center+1 : c_center + (n - i)] -= h[i+1:]
        if i > 0:
            c[c_center - i : c_center] -= h[:i]
        h[i] = 0
        # addition of j
        c[c_center] += 1
        if j+1 < n:
            c[c_center+1 : c_center + (n - j)] += h[j+1:]
        if j > 0:
            c[c_center - j : c_center] += h[:j]
        h[j] = 1

        # compute new worst
        new_min = c.min()
        cur_worst = target - new_min
        delta = cur_worst - prev_worst

        T = T0 * (decay ** it)
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True

        if accept:
            # update index arrays
            idx_one_i = pos_one[i]
            idx_zero_j = pos_zero[j]
            # swap entries
            ones[idx_one_i] = j
            zeros[idx_zero_j] = i
            pos_one[j] = idx_one_i
            pos_zero[i] = idx_zero_j
            pos_one[i] = -1
            pos_zero[j] = -1

            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # reject, revert swap by swapping back
            # removal of j (currently 1)
            c[c_center] -= 1
            if j+1 < n:
                c[c_center+1 : c_center + (n - j)] -= h[j+1:]
            if j > 0:
                c[c_center - j : c_center] -= h[:j]
            h[j] = 0
            # addition of i
            c[c_center] += 1
            if i+1 < n:
                c[c_center+1 : c_center + (n - i)] += h[i+1:]
            if i > 0:
                c[c_center - i : c_center] += h[:i]
            h[i] = 1
            # no index updates needed since arrays unchanged
            no_improve += 1

        it += 1
        if no_improve > 50000:
            break

    # after loop, best_h is binary with best worst
    return best_h, best_worst

def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx  (discrete version)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Adjust h so that sum(h) = n/2 exactly (used for fractional vectors)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

                no_improve += 1
        else:
            # ---- revert swap (j→0, i→1) ----
            # remove j
            c[c_center] -= 1
            if j + 1 < n:
                c[c_center + 1 : c_center + (n - j)] -= h[j + 1 :]
            if j > 0:
                c[c_center - j : c_center] -= h[:j]
            h[j] = 0
            # add i
            c[c_center] += 1
            if i + 1 < n:
                c[c_center + 1 : c_center + (n - i)] += h[i + 1 :]
            if i > 0:
                c[c_center - i : c_center] += h[:i]
            h[i] = 1
            cur_worst = prev_worst
            no_improve += 1

        it += 1
        if no_improve > 50000:
            break

    return best_h, best_worst

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    candidate_ns = [720, 540, 360]  # try higher resolution first

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n // 2
        # ----- generate starting vectors -----
        start_vectors = []

        # parent (if available)
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            parent_interp = np.interp(xt, xs, parent)
            # binary version: take top target entries
            idx = np.argpartition(parent_interp, -target)[-target:]
            bin_parent = np.zeros(n, dtype=np.int8)
            bin_parent[idx] = 1
            start_vectors.append(bin_parent)
            # fractional version
            start_vectors.append(parent_interp.copy())

        # random binary
        rand_bin = np.zeros(n, dtype=np.int8)
        rand_bin[rng.choice(n, target, replace=False)] = 1
        start_vectors.append(rand_bin)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        start_vectors.append(rand_frac)

        # alternating binary pattern
        alt_bin = np.tile([1, 0], n // 2).astype(np.int8)
        start_vectors.append(alt_bin)

        # ------ iterate over starts ------
        for base_h in start_vectors:
            if time.time() > deadline:
                break
            h = base_h.copy()

            # ensure exact sum
            if np.issubdtype(h.dtype, np.integer):
                cur = h.sum()
                diff = target - cur
                if diff > 0:
                    zeros = np.flatnonzero(h == 0)
                    h[rng.choice(zeros, diff, replace=False)] = 1
                elif diff < 0:
                    ones = np.flatnonzero(h == 1)
                    h[rng.choice(ones, -diff, replace=False)] = 0
            else:
                adjust_sum_exact(h, rng)

            if np.issubdtype(h.dtype, np.integer):
                # ---- binary simulated annealing ----
                sub_dead = deadline  # share global deadline
                bin_best, worst = binary_sa_slice(h, rng, sub_dead, max_iter=150000)
                c5 = worst * (2.0 / n)
                if c5 < best_c:
                    best_c, best_h, best_n = c5, bin_best.copy(), n
                # optional continuous refinement on the binary result
                if time.time() < deadline:
                    h_float = bin_best.astype(float)
                    h_float = gradient_refine(h_float, rng, step=0.02, max_iter=12)
                    c5_ref = compute_c5(h_float)
                    if c5_ref < best_c:
                        best_c, best_h, best_n = c5_ref, h_float.copy(), n
            else:
                # fractional start – simple local refinement
                if time.time() < deadline:
                    h = gradient_refine(h, rng, step=0.02, max_iter=12)
                c5 = compute_c5(h)
                if c5 < best_c:
                    best_c, best_h, best_n = c5, h.copy(), n

        # after exhausting starts for this n, try a few extra random SA runs if time permits
        extra_runs = 5
        for _ in range(extra_runs):
            if time.time() > deadline:
                break
            # fresh random binary
            h = np.zeros(n, dtype=np.int8)
            h[rng.choice(n, target, replace=False)] = 1
            bin_best, worst = binary_sa_slice(h, rng, deadline, max_iter=120000)
            c5 = worst * (2.0 / n)
            if c5 < best_c:
                best_c, best_h, best_n = c5, bin_best.copy(), n
            if time.time() < deadline:
                h_float = bin_best.astype(float)
                h_float = gradient_refine(h_float, rng, step=0.02, max_iter=12)
                c5_ref = compute_c5(h_float)
                if c5_ref < best_c:
                    best_c, best_h, best_n = c5_ref, h_float.copy(), n

    # ----- final safety check -----
    if best_h is None:
        # fallback simple alternating pattern at n=360
        best_n = 360
        best_h = np.tile([1, 0], best_n // 2).astype(float)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    # make sum exact one (∫ h = 1)
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        i = rng.integers(0, best_n)
        best_h[i] += diff
        best_h[i] = min(max(best_h[i], 0.0), 1.0)

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
