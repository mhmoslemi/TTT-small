def run(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99
    best_h = None
    best_c = np.inf
    best_n = None
    
    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None
    # store best candidate per n maybe for later interpolation
    best_per_n = {}
    
    n_list = [720, 540, 360, 240, 180]
    for n in n_list:
        if time.time() > deadline: break
        # generate start vectors
        start_vecs = []
        if parent is not None and parent.size != n:
            xs = np.linspace(0.0,2.0,parent.size,endpoint=False)
            xt = np.linspace(0.0,2.0,n,endpoint=False)
            interp = np.interp(xt,xs,parent)
            start_vecs.append(interp)
            # binary rounding of interp (top half)
            idx = np.argpartition(interp, -n//2)[-n//2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)
        if best_h is not None and best_n != n:
            xs = np.linspace(0.0,2.0,best_n,endpoint=False)
            xt = np.linspace(0.0,2.0,n,endpoint=False)
            interp = np.interp(xt,xs,best_h)
            start_vecs.append(interp)
        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n,n//2,replace=False)] = 1.0
        start_vecs.append(bin_rand)
        # random fractional uniform
        rand_frac = rng.random(n)
        target = n/2.0
        rand_frac *= target / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        start_vecs.append(rand_frac)
        # random fractional squared
        rand_frac2 = rng.random(n)**2
        rand_frac2 *= target / rand_frac2.sum()
        adjust_sum_exact(rand_frac2, rng)
        start_vecs.append(rand_frac2)
        # alternating pattern
        alt = np.zeros(n)
        alt[::2][:int(n/2)] = 1.0  # fill half ones in even indices
        # might need adjust if n is odd, but n even
        start_vecs.append(alt)
        
        for h0 in start_vecs:
            if time.time() > deadline: break
            h = np.clip(h0,0.0,1.0)
            adjust_sum_exact(h, rng)
            # compute remaining for this candidate
            # allocate small chunk per candidate maybe 0.05 * (deadline - start_time)
            remaining = deadline - time.time()
            # allocate sub_deadline = time.time() + min(remaining*0.5, 30) for SA etc.
            # We'll call optimize_candidate with sub_deadline = time.time() + remaining*0.8 (some)
            sub_deadline = min(deadline, time.time() + remaining*0.8)
            h_opt, c_opt = optimize_candidate(h, n, rng, sub_deadline)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n
                best_per_n[n] = (h_opt.copy(), c_opt)
    # final upsampling if time left, try to go to 720 if not already
    # but maybe already done.
    # If best_n != 720 and time left, try upsampling to 720.
    if best_h is not None and time.time() < deadline and best_n < 720:
        n_up = 720
        xs = np.linspace(0.0,2.0,best_n,endpoint=False)
        xt = np.linspace(0.0,2.0,n_up,endpoint=False)
        h_up = np.interp(xt,xs,best_h)
        adjust_sum_exact(h_up, rng)
        sub_deadline = min(deadline, time.time() + (deadline - time.time())*0.9)
        h_up_opt, c_up_opt = optimize_candidate(h_up, n_up, rng, sub_deadline)
        if c_up_opt < best_c:
            best_h, best_c, best_n = h_up_opt.copy(), c_up_opt, n_up
    
    # final check
    best_h = np.clip(best_h,0.0,1.0)
    adjust_sum_exact(best_h,rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

def optimize_candidate(h, n, rng, deadline):
    # returns improved h and its C5
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    
    # if binary, do binary SA
    if np.all((cur_h == 0) | (cur_h == 1)):
        sub_dead = min(deadline, time.time() + (deadline - time.time())*0.4)
        cur_h, cur_c = binary_sa_hill_climb(cur_h.copy(), rng, sub_dead)
        if cur_c < best_c:
            best_h, best_c = cur_h.copy(), cur_c
    
    # fractional SA
    sub_dead = min(deadline, time.time() + (deadline - time.time())*0.5)
    cur_h, cur_c = fractional_sa_hill_climb(cur_h.copy(), rng, sub_dead)
    if cur_c < best_c:
        best_h, best_c = cur_h.copy(), cur_c
    
    # gradient refine - a few passes
    cur_h = gradient_refine(cur_h, rng, step=0.03, max_iter=12)
    cur_c = compute_c5(cur_h)
    if cur_c < best_c:
        best_h, best_c = cur_h.copy(), cur_c
    
    # SLSQP cascade
    betas = [5.0,20.0,80.0,200.0,500.0,1000.0,2000.0,5000.0,10000.0,20000.0]
    for beta in betas:
        if time.time() > deadline:
            break
        # determine iterations
        max_iter = 700 if beta < 200 else 1100
        if beta >= 5000:
            max_iter = 1500
        try:
            cur_h = slsqp_optimize(cur_h, beta, rng, max_iter=max_iter)
        except Exception:
            continue
        cur_c = compute_c5(cur_h)
        if cur_c < best_c:
            best_h, best_c = cur_h.copy(), cur_c
    
    # final gradient refine with small step
    cur_h = gradient_refine(cur_h, rng, step=0.01, max_iter=20)
    cur_c = compute_c5(cur_h)
    if cur_c < best_c:
        best_h, best_c = cur_h.copy(), cur_c
    
    return best_h, best_c

import time, math, numpy as np
from scipy.optimize import minimize

# Global placeholders for the smooth‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Maximum overlap for a step function sampled on a uniform grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Redistribute mass so that Σh = n/2 exactly while staying in [0,1]."""
    n = h.size
    target = n * 0.5
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth max of the overlap using log‑sum‑exp, scaled by CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n * 0.5
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng, max_iter=2000)
    return x

def gradient_refine(h, rng, step=0.03, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n * 0.5
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # derivative from h[i] term
            g[k:]      -= h[:upper]         # derivative from -h[i+k] term
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, deadline):
    """Simulated annealing on a binary vector (swap 1↔0)."""
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    init_T = max(1e-4, 0.01 * cur_c)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i], cur[j] = 0, 1
        c = compute_c5(cur)
        T = init_T * (0.9999 ** it) + 1e-12
        if c < cur_c or rng.random() < math.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c, best = c, cur.copy()
                last_improve = it
        else:
            cur[i], cur[j] = 1, 0
        it += 1
        if it - last_improve > 12000:
            break
    return best, best_c

def fractional_sa_hill_climb(h, rng, deadline):
    """Simulated annealing on a continuous vector (transfer mass)."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(1e-4, 0.01 * cur_c)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9999 ** it) + 1e-12
        if c < cur_c or rng.random() < math.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c, best = c, cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 20000:
            break
    return best, best_c

def optimize_candidate(h, n, rng, deadline):
    """Run a full refinement pipeline on h using the remaining time."""
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c

    # binary SA if applicable
    if np.all((cur_h == 0) | (cur_h == 1)):

    # fractional SA

    # gradient refinement

    cur_c = compute_c5(cur_h)
    if cur_c < best_c:

    # SLSQP cascade with beta schedule
    betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]
    for beta in betas:
        if time.time() > deadline:

        # iteration budget depends on problem size
