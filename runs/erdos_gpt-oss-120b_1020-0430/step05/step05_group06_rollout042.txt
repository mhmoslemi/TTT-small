def smooth_max_and_grad(h, beta):
    O = np.correlate(h, 1 - h, mode='full')
    max_O = np.max(O)  # maybe for logging
    exp_beta_O = np.exp(beta * O - np.max(beta * O))  # for numerical stability
    sum_exp = np.sum(exp_beta_O)
    w = exp_beta_O / sum_exp  # weights sum to 1
    # compute gradient
    n = h.size
    g = np.ones(n)  # start from 1 sum of w (which is 1)
    # w indexed by p = 0..2n-2, shift k = p - (n-1)
    for p, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = p - (n - 1)
        if k >= 0:
            # i from k to n-1: i - k >=0
            # g[i - k] -= wk * h[i]
            # g[i + k] -= wk * h[i] for i + k < n

import time
import numpy as np

def compute_c5(h):
    """Exact max overlap integral (scaled)."""
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(O) * (2.0 / n))

def project_bounds_sum(v, target):
    """Euclidean projection of v onto [0,1]^n with sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):  # enough for double precision
        lam = (lo + hi) / 2.0
        adj = np.clip(v - lam, 0.0, 1.0)
        s = adj.sum()
        if s < target:
            hi = lam
        else:
            lo = lam
    return np.clip(v - lo, 0.0, 1.0)

def smooth_max_and_grad(h, beta):
    """Return smooth max value and its gradient w.r.t. h."""
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    Omax = np.max(O)
    # stable weights for the log‑sum‑exp
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()               # shape 2n‑1, sums to 1

    # convolution to get Σ_k w_k h_{m±k}
    conv_full = np.convolve(h, w, mode="full")
    conv_rev = np.convolve(h, w[::-1], mode="full")
    conv2 = conv_full[n - 1:n - 1 + n]          # Σ_k w_k h_{m‑k}
    conv1 = conv_rev[n - 1:n - 1 + n]          # Σ_k w_k h_{m+k}
    grad = np.ones(n) - conv1 - conv2          # ∂/∂h (smooth max)
    smooth = Omax + (1.0 / beta) * np.log(exp_term.sum())
    return smooth, grad, O

            step_try = step
            improved = False
            for _ in range(12):
                h_new = h - step_try * grad
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_bounds_sum(h_new, target)
                smooth_new, _, _ = smooth_max_and_grad(h_new, beta)
                if smooth_new <= smooth + 1e-14:
                    h = h_new
                    improved = True
                    break
                step_try *= 0.5
            if not improved:
                no_improve += 1
                if no_improve >= 30:
                    break
                continue
            no_improve = 0
            # occasional exact C5 check
            if it % 100 == 0:
                c_now = compute_c5(h)
                if c_now < best_c:
                    best_c = c_now
                    best_h = h.copy()
        step *= 0.85  # shrink step as β grows
    # final polishing with very large β
    final_beta = 50000.0
    for _ in range(300):
        if time.time() > deadline:
            break
        smooth, grad, _ = smooth_max_and_grad(h, final_beta)
        h_new = h - 0.05 * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_bounds_sum(h_new, target)
        smooth_new, _, _ = smooth_max_and_grad(h_new, final_beta)
        if smooth_new <= smooth:
            h = h_new
        else:
            break
    c_final = compute_c5(h)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.95

    best_h = None
    best_c = np.inf
    best_n = None

    # use provided parent if available
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        c_par = compute_c5(parent)
        if c_par < best_c:
            best_h = parent.copy()
            best_c = c_par
            best_n = parent.size

    # candidate sample sizes (keep ≤ 1000)
    candidate_ns = [360, 720]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        dx = 2.0 / n

        # build a few diverse starters
        starters = []

        # 1) symmetric triangle (continuous optimum shape)
        xs = np.arange(n) * dx + dx / 2.0
        tri = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
        tri = project_bounds_sum(tri, target)
        starters.append(tri)

        # 2) uniform 0.5 (guaranteed feasibility)
        uni = np.full(n, 0.5)
        uni = project_bounds_sum(uni, target)
        starters.append(uni)

        # 3) a few random points
        for _ in range(3):
            rnd = rng.random(n)
            rnd = project_bounds_sum(rnd, target)
            starters.append(rnd)

        # if we already have a parent of same size, include it
        if best_n == n and best_h is not None:
            starters.append(best_h.copy())

        for h0 in starters:
            if time.time() > deadline:
                break
            h_opt, c_opt = optimize_h(h0, deadline)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n

    # fallback if nothing found
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.full(n, 0.5)
        best_h = project_bounds_sum(best_h, n / 2.0)
        best_c = compute_c5(best_h)

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_bounds_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
