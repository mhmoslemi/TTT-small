def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    # maximize corr means minimize -corr; but we want to minimize max, so we use negative of corr?
    # Actually we want to minimize max(corr). So objective = max(corr). But log-sum-exp approximates max by (1/β)log( sum exp(β*corr) )
    # So we can directly compute that.
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def gradient_refine_multi(h, rng, step=0.03, max_iter=30, eps=1e-6):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        # pick shifts where corr within eps of max
        ks = np.where(corr >= max_val - eps)[0] - (n-1)
        # compute gradient sum
        g = np.zeros_like(h)
        for k in ks:
            if k == 0:
                g += 1.0 - 2.0 * h
            else:
                upper = n - abs(k)
                if k > 0:
                    # i from 0..upper-1: index i; i+k <= n-1
                    g[:upper] += 1.0 - h[k:]
                    g[k:] -= h[:upper]
                else:
                    # k < 0: shift left, we can treat similarly
                    kk = -k
                    # i+ k >=0 => i >= kk
                    g[kk:] += 1.0 - h[:n-kk]
                    g[:n-kk] -= h[kk:]
        # step size decay
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip and adjust sum
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff as before
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.95 (some margin)
    best_h, best_c, best_n = None, np.inf, None

    # Determine candidate n values
    candidate_ns = [720, 600, 480, 360, 240, 180]
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        # include parent's size if not already in candidate
        if parent.size not in candidate_ns:
            candidate_ns = [parent.size] + candidate_ns
    else:
        parent = None

    for n in candidate_ns:
        if time.time() > deadline:
            break
        # generate initial vectors list
        init_vecs = []

        if parent is not None:
            # Interpolate parent to n points
            xs = np.linspace(0, 2, parent.size, endpoint=False)
            xt = np.linspace(0, 2, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            init_vecs.append(interp)

            # Binary version from parent: take top n/2 entries
            idx = np.argpartition(interp, -n//2)[-n//2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            init_vecs.append(bin_vec)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= (n/2) / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        init_vecs.append(rand_frac)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n//2, replace=False)] = 1.0
        init_vecs.append(bin_rand)

        # maybe also use "low-discrepancy" start: half ones interleaved? 
        # generate pattern: [1,0,1,0,...] length n
        pattern = np.tile([1,0], n//2)
        init_vecs.append(pattern.astype(float))

        # Process each start
        for h0 in init_vecs:
            if time.time() > deadline:
                break
            h = np.array(h0, copy=True)
            # ensure constraints
            h = np.clip(h, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # SLSQP smoothing sequence
            betas = [10, 50, 200, 800, 3000, 10000]
            for beta in betas:
                if time.time() > deadline:
                    break
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=500)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # Multi-gradient refinement
            h = gradient_refine_multi(h, rng, step=0.04, max_iter=25, eps=1e-7)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # If now binary (almost), perform SA
            if np.all((h <= 1e-12) | (h >= 1-1e-12)):
                # round
                h_bin = np.rint(h).astype(float)
                adjust_sum_exact(h_bin, rng)
                # binary SA upto limited steps
                sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
                h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, sub_dead)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin

            # final gradient refine again
            h = gradient_refine_multi(h, rng, step=0.03, max_iter=15, eps=1e-6)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # Upsample to higher resolution (2x) if still time and n<720
        if n*2 <= 720 and time.time() < deadline:
            n_up = n*2
            # interpolate coarse to up grid
            xs = np.linspace(0, 2, n, endpoint=False)
            xt = np.linspace(0, 2, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h if best_n==n else h)
            adjust_sum_exact(h_up, rng)
            # optimize at higher resolution
            for beta in [200, 1000, 5000, 20000]:
                if time.time() > deadline:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
            h_up = gradient_refine_multi(h_up, rng, step=0.04, max_iter=30, eps=1e-7)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # fallback if none
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[:n//2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    # final clip & sum adjust
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)

import time, math, numpy as np
from scipy.optimize import minimize

# global placeholders for constraints used by the soft‑max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass to enforce Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random walk to correct the sum
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny numerical drift)
    adjust_sum_exact(x, rng)
    return x

def gradient_refine_multi(h, rng, step=0.04, max_iter=30, eps=1e-7):
    """Gradient descent against all near‑worst shifts."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        ks = np.where(corr >= max_val - eps)[0] - (n - 1)
        ks = ks.astype(int)
        if ks.size == 0:
            break
        g = np.zeros_like(h)
        for k in ks:
            if k == 0:
                g += 1.0 - 2.0 * h
            elif k > 0:
                upper = n - k
                g[:upper] += 1.0 - h[k:]
                g[k:] -= h[:upper]
            else:  # k < 0
                kk = -k
                upper = n - kk
                g[kk:] += 1.0 - h[:upper]
                g[:upper] -= h[kk:]
        g /= ks.size
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    """Simulated‑annealing hill‑climb on a binary vector."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it, last_improve = 0, 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < math.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c, best_h, last_improve = c, cur_h.copy(), it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c
