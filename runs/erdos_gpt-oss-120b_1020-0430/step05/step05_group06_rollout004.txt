def adjust_sum_exact_fast(h, target_sum):
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12: return h
    n = h.size
    if diff > 0:
        # need to add diff
        # compute available capacity
        cap = 1.0 - h
        total_cap = cap.sum()
        if total_cap < diff - 1e-12:
            # cannot fill, saturate
            h += cap
            return h
        # fill proportionally: add diff * cap / total_cap
        h += diff * cap / total_cap
    else:
        # need to remove -diff
        diff = -diff
        cap = h
        total_cap = cap.sum()
        if total_cap < diff - 1e-12:
            h -= cap
            return h
        h -= diff * cap / total_cap
    # clip for safety
    np.clip(h, 0.0, 1.0, out=h)
    # Due to rounding might not be exact, so we can repeat small correction with tolerance.
    # Maybe iterate a couple times.
    # final adjust small diff via simple loop
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute tiny diff to one index
        if diff > 0:
            i = np.argmax(1-h)
            add = min(diff, 1.0-h[i])
            h[i] += add
        else:
            i = np.argmax(h)
            sub = min(-diff, h[i])
            h[i] -= sub
    return h

def compute_c5(h):

def adjust_sum_exact(h, rng=None):
    """Scale h∈[0,1] so that sum(h)=len(h)/2 (exact up to 1e-12)."""
    target = h.size / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        total = cap.sum()
        if total > 0:
            h += diff * cap / total
    else:
        diff = -diff
        cap = h
        total = cap.sum()
        if total > 0:
            h -= diff * cap / total
    np.clip(h, 0.0, 1.0, out=h)
    # tiny correction if rounding left a residue
    diff = target - h.sum()
    if abs(diff) > 1e-12:

        if O_new < best_max:
            best_max = O_new
            best_h = cur_h.copy()
            # update index lists efficiently
            ones[ones == i] = j
            zeros[zeros == j] = i
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_max

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # start from provided parent if it exists
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        adjust_sum_exact(parent, rng)
        c_parent = compute_c5(parent)
        best_h = parent.copy()
        best_c = c_parent
        best_n = parent.size

    candidate_ns = [360, 720]
    # ensure the parent size is explored
    if best_n and best_n not in candidate_ns:
        candidate_ns.append(best_n)

    def random_binary_h(n):
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        return h

    # main search
    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        betas = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]
        base_eta = 0.5
        min_eta = 1e-4
        restarts = 5 if n <= 360 else 3

        for r in range(restarts):
            if time.time() > deadline:
                break
            if best_h is not None and best_n == n and r == 0:
                h = best_h.copy()
            else:
                h = random_binary_h(n)
            adjust_sum_exact(h, rng)
            cur_corr = np.correlate(h, 1.0 - h, mode="full")
            cur_maxc = cur_corr.max()
            cur_c = cur_maxc * (2.0 / n)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                best_n = n

            for beta in betas:
                if time.time() > deadline:
                    break
                eta = max(min_eta, base_eta / np.sqrt(beta))
                steps = 200 if n <= 360 else 300
                for _ in range(steps):
                    if time.time() > deadline:
                        break
                    h_new, maxc_new = gradient_step_and_c(h, beta, eta, rng)
                    c_new = maxc_new * (2.0 / n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()
                        best_n = n
                    if maxc_new <= cur_maxc + 1e-12:
                        h = h_new
                        cur_maxc = maxc_new
                    # else keep old h

        # post‑processing: binary projection + hill‑climb
        if time.time() > deadline:
            break
        if best_h is not None and best_n == n:
            hb = binary_projection(best_h)
            c_bin = compute_c5(hb)
            if c_bin < best_c:
                best_c = c_bin
                best_h = hb.copy()
                best_n = n
            if time.time() < deadline:
                max_O = int(round(c_bin * n / 2.0))
                hb_hc, max_O_hc = hill_climb(hb.copy(), max_O, n, rng, deadline)
                c_hc = max_O_hc * (2.0 / n)
                if c_hc < best_c:
                    best_c = c_hc
                    best_h = hb_hc.astype(float).copy()
                    best_n = n

    # safety fallback
    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)

    adjust_sum_exact(best_h, rng)
    best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
