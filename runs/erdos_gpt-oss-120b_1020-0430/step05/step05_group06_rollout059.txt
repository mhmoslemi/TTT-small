def shift_array(arr, shift, n):
    # shift arr of length n to produce contribution array of length 2n-1 for correlation.
    # shift is integer. Output length 2n-1.
    # For each lag ℓ, output arr[ℓ+shift] if 0 <= ℓ+shift < n else 0
    out = np.zeros(2*n-1, dtype=arr.dtype)
    # compute range of ℓ where index is valid
    lo = -shift
    hi = n-1 - shift
    # Ensure lo <= hi
    lo = max(lo, -(n-1))
    hi = min(hi, n-1)
    if lo <= hi:
        l_idx = lo + (n-1)  # start idx in out
        r_idx = hi + (n-1)  # inclusive end
        arr_start = lo + shift  # arr index start which equals 0 if lo = -shift
        out[l_idx:r_idx+1] = arr[arr_start:arr_start+(r_idx-l_idx+1)]
    return out

This yields a vector of length 2n-1.

def shift_contrib(arr, shift, n):
    out = np.zeros(2*n-1, dtype=int)
    i_start = max(0, -shift)
    i_end = min(n, n - shift)
    # i range: [i_start, i_end) inclusive start exclusive end
    # Corresponding out indices:
    out_start = i_start + shift + (n-1)
    out_end = i_end - 1 + shift + (n-1)
    out[out_start:out_end+1] = arr[i_start:i_end]
    return out

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact sum correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    # simple iterative correction; few iterations are enough
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def smoothmax_and_grad(h, beta):
    """
    Returns smooth max value of correlation h*(1-h) and its gradient.
    Uses log-sum-exp stability and analytic gradient.
    """
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    m = O.max()
    # stable weights
    exp_term = np.exp(beta * (O - m))
    Z = exp_term.sum()
    w = exp_term / Z  # shape (2n-1,)

    g = 1.0 - h  # complement
    # gradient part 1: w ⋆ g, take lags 0..n-1
    corr1 = np.correlate(w, g, mode="full")
    grad1 = corr1[n - 1 : n - 1 + n]

    # gradient part 2: w_rev ⋆ h
    w_rev = w[::-1]
    corr2 = np.correlate(w_rev, h, mode="full")
    grad2 = corr2[n - 1 : n - 1 + n]

    grad = (grad1 - grad2) / beta
    val = (m + np.log(Z)) / beta
    return val, grad

def slsqp_smoothmax(start_h, beta, rng, max_iter):
    """Constrained SLSQP optimization of smooth max surrogate."""
    n = start_h.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    # equality constraint with Jacobian
    cons = {
        "type": "eq",
        "fun": lambda x: x.sum() - target,
        "jac": lambda x: np.ones_like(x),
    }

    def fun(x):
        val, grad = smoothmax_and_grad(x, beta)
        return val, grad

    res = minimize(
        fun,
        start_h,
        method="SLSQP",
        jac=True,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 360
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)
    n = h0.size
    adjust_sum_exact(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # beta schedule (logarithmic growth)
    beta_vals = [
        0.5,
        2,
        5,
        10,
        20,
        40,
        80,
        160,
        320,
        640,
        1280,
        2560,
        5120,
        10240,
    ]
    # iteration budget per beta (higher beta needs more)
    iter_map = {b: 500 if b < 20 else 800 if b < 640 else 1200 for b in beta_vals}
    noise = 0.05  # initial perturbation amplitude

    for beta in beta_vals:
        if time.time() > deadline:
            break
        # perturb current best to escape possible plateaus
        h_start = best_h + rng.uniform(-noise, noise, n)
        np.clip(h_start, 0.0, 1.0, out=h_start)
        adjust_sum_exact(h_start, rng)

        max_iter = iter_map[beta]
        try:

        except Exception:
            continue

        c_val = compute_c5(h_opt)
        if c_val < best_c:
            best_c = c_val
            best_h = h_opt.copy()
        # decay noise for later stages
        noise *= 0.8

    # final safety: ensure feasibility and recompute bound
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
