def project_onto_sum_and_bounds(v, target, max_iter=50):
    lo = -np.max(v)  # ensures v + lo <= 0 => all clipped to 0 => sum = 0
    hi = 1 - np.min(v)  # ensures v + hi >=1 => all clipped to 1 => sum = n
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2, 0, 1)

def subgradient_descent(h0, n_iter, step0, rng):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(1, n_iter+1):
        # compute correlation
        corr = np.correlate(h, 1.0 - h, mode="full")
        # compute C5
        c_val = float(np.max(corr) * (2.0 / n))
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        # find shift k (index) that attains max
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)   # shift can be negative
        # compute gradient for this shift
        # g_j = 1 - (h_{j+k} + h_{j-k}) where defined
        g = np.ones(n)
        if k != 0:
            # positive k: interval [0, n - k -1] have h_{j+k}
            # indices for j where j+k < n
            j_valid = np.arange(n - k)
            g[j_valid] -= h[j_valid + k]
            # indices where j-k >=0, i.e., j from k to n-1
            j_valid2 = np.arange(k, n)
            g[j_valid2] -= h[j_valid2 - k]
        else:
            # k = 0: both terms coincide
            g -= 2 * h
        # step size
        step = step0 / np.sqrt(t)
        # gradient descent
        h_new = h - step * g
        # project onto constraints
        h = project_onto_sum_and_bounds(h_new, target)
        # ensure within [0,1]
        # done inside projection
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=20):
    n = h.size
    target = n / 2.0

The best_h we store after possibly improvement is current h which has sum constraint satisfied.

Thus best_h final has sum constraint satisfied.

Now after run, we do final safety projection.

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for discretized h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_onto_sum_and_bounds(v, target, iters=60):
    """
    Euclidean projection of v onto the set {x | sum(x)=target, 0<=x<=1}.
    Uses a simple bisection on the shift λ for x = clip(v+λ,0,1).
    """
    lo = -np.max(v)          # ensures v+lo ≤ 0 → sum=0
    hi = 1.0 - np.min(v)     # ensures v+hi ≥ 1 → sum=n
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def subgradient_descent(h0, n_iter, step0, rng):
    """
    Projected sub‑gradient descent on max_k ∫h(x)(1‑h(x+k)).
    Returns the best h found and its C5 value.
    """
    n = h0.size
    target = n * 0.5
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    for t in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c_val = float(np.max(corr) * (2.0 / n))
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()

        k = np.argmax(corr) - (n - 1)               # shift attaining the max
        k_abs = abs(k)

        # gradient of the worst shift
        g = np.ones(n)
        if k_abs == 0:
            g -= 2.0 * h
        else:
            g[: n - k_abs] -= h[k_abs:]
            g[k_abs:] -= h[: n - k_abs]

        step = step0 / np.sqrt(t)
        h = project_onto_sum_and_bounds(h - step * g, target)

        # occasional tiny random perturbation to escape flat regions
        if t % 500 == 0:
            noise = rng.normal(scale=0.005, size=n)
            h = project_onto_sum_and_bounds(h + noise, target)

    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=20):
    """Deterministic refinement against the current worst shift."""
    n = h.size
    target = n * 0.5
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = np.argmax(corr) - (n - 1)
        k_abs = abs(k)

        g = np.ones(n)
        if k_abs == 0:
            g -= 2.0 * h
        else:
            g[: n - k_abs] -= h[k_abs:]
            g[k_abs:] -= h[: n - k_abs]

        step_sz = step * (0.5 ** it)
        h = project_onto_sum_and_bounds(h - step_sz * g, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    # Try several even sizes, larger first
    for n in (720, 600, 480, 360):
        if n % 2:
            continue
        if time.time() > deadline:
            break

        target = n * 0.5
        starts = []

        # Interpolated parent if available
        if 'initial_h_values' in globals():
            parent = np.asarray(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            starts.append(interp)

            # Binary version of the parent
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_parent = np.zeros(n)
            bin_parent[idx] = 1.0
            starts.append(bin_parent)

        # Random fractional start
        rand_frac = rng.random(n)
        rand_frac = rand_frac * (target / rand_frac.sum())
        starts.append(rand_frac)

        # Random binary start
        bin_vec = np.zeros(n)
        bin_vec[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_vec)

        for h0 in starts:
            if time.time() > deadline:
                break

            h0 = project_onto_sum_and_bounds(h0, target)

            # Sub‑gradient descent
            n_iter = 5000 if n >= 600 else 4000
            step0 = 0.8
            h_opt, c_opt = subgradient_descent(h0, n_iter, step0, rng)

            if c_opt < best_c:
                best_c, best_h, best_n = c_opt, h_opt.copy(), n

            # Light deterministic refinement
            if time.time() < deadline - 0.5:
                h_ref = gradient_refine(h_opt, rng, step=0.02, max_iter=15)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h, best_n = c_ref, h_ref.copy(), n

    # Fallback simple construction
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_onto_sum_and_bounds(best_h, n * 0.5)
        best_c = compute_c5(best_h)

    # Final safety projection and recompute
    best_h = project_onto_sum_and_bounds(best_h, best_n * 0.5)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
