import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def compute_c5(h):
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx  (dx = 2/n)"""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass to enforce Σh = n/2 exactly."""
    n = h.size
    target = n * 0.5
    diff = target - h.sum()
    # simple random walks; bounded iterations
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:
            continue
        h[i] += add
        diff = target - h.sum()
    # final clip to avoid tiny violations
    h[:] = np.clip(h, 0.0, 1.0)
    # tiny final correction (could introduce tiny negativity but harmless)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        h[0] += residual
    return h

# globals for SLSQP soft‑min objective
CUR_N = None
CUR_TARGET = None
CUR_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CUR_TARGET

def _softmin_obj(x):
    """Smooth approximation of -min_k corr_k, where corr_k = Σ h_i h_{i+k}."""
    # correlation for non‑negative shifts
    corr = np.correlate(x, x, mode="full")
    corr = corr[CUR_N - 1 : ]           # length CUR_N
    bc = -CUR_BETA * corr
    m = np.max(bc)
    # - (m + log Σ exp(bc-m)) / β
    return - (m + np.log(np.sum(np.exp(bc - m)))) / CUR_BETA

def slsqp_softmin(h0, beta, rng, max_iter=300):
    """Run SLSQP on the smooth‑min surrogate."""
    global CUR_N, CUR_TARGET, CUR_BETA
    n = h0.size
    CUR_N = n
    CUR_TARGET = n * 0.5
    CUR_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    # SLSQP may occasionally raise; we catch and return current point
    try:
        res = minimize(
            _softmin_obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = np.clip(h0, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift (max overlap)."""
    n = h.size
    target = n * 0.5
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

    start_time = time.time()

            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            h_interp = np.interp(xt, xs, parent)
            h_interp = np.clip(h_interp, 0.0, 1.0)
            adjust_sum_exact(h_interp, rng)
