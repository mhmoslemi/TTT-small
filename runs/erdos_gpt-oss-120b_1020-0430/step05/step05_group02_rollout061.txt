def binary_sa(h_bin, rng, deadline, max_iter=800000):
    n = h_bin.size
    target = n/2

    # compute auto-correlation
    c = np.correlate(h_bin, h_bin, mode='full')
    cur_min = c.min()
    cur_worst = target - cur_min

    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = np.flatnonzero(h_bin == 1.0).tolist()
    zeros = np.flatnonzero(h_bin == 0.0).tolist()

    # temperature schedule
    T0 = cur_worst * 0.3 + 1e-3
    decay = 0.99997

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        i = rng.choice(ones)
        j = rng.choice(zeros)

def adjust_sum_exact(h, rng):

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

        bounds=bounds,

        x[i] += add
        diff = CURRENT_TARGET - x.sum()

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(
    h: np.ndarray, rng: np.random.Generator, step: float = 0.02, max_iter: int = 12
) -> np.ndarray:
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ---------- SLSQP on smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(
    h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500
) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


# ---------- binary simulated annealing ----------
def binary_sa(
    h_bin: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iter: int = 500000,
) -> tuple[np.ndarray, float]:
    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst
    ones = np.flatnonzero(h_bin == 1.0).tolist()
    zeros = np.flatnonzero(h_bin == 0.0).tolist()
    T0 = max(cur_worst, 1.0) * 0.3 + 1e-3
    decay = 0.99997
    it = 0
    while it < max_iter and time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        c[start_i : start_i + n] -= hi_vec
        h_bin[i] = 0.0

        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j + 1 :]))
        c[start_j : start_j + n] += hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min
        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_worst - cur_worst) / T):
                accept = True
        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
        else:
            c[start_j : start_j + n] -= hj_vec
            h_bin[j] = 0.0
            c[start_i : start_i + n] += hi_vec
            h_bin[i] = 1.0
        it += 1
    return best_h, best_worst

def upsample_optimize(
    h: np.ndarray,
    n_target: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    n0 = h.size
    xs = np.linspace(0.0, 2.0, n0, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_up = np.interp(xt, xs, h)
    adjust_sum_exact(h_up, rng)

    h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)

    betas = (
        5.0,
        20.0,
        80.0,
        200.0,
        500.0,
        1000.0,
        2000.0,
        5000.0,
        10000.0,
        20000.0,
    )
    for beta in betas:
        if time.time() >= deadline:
            break
        max_iter = 800 if beta >= 2000 else 600
        h_up = slsqp_optimize(h_up, beta, rng, max_iter=max_iter)
    h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
    return h_up, compute_c5(h_up)
