def gradient_descent_h(h0, rng, budget_deadline):
    h = h0.copy()
    n = len(h)
    target = n/2
    # parameters
    beta_schedule = [(5.0, 50), (20.0, 50), (80.0, 50), (200.0, 50), (1000.0, 50), (2000.0, 50)]
    # lr schedule maybe decreasing
    for beta, steps in beta_schedule:
        for step in range(steps):
            if time.time() > budget_deadline:
                break
            # compute corr
            corr = np.correlate(h, 1 - h, mode='full')
            # compute weights
            max_c = corr.max()
            exp_beta = np.exp(beta * (corr - max_c))
            w = exp_beta / exp_beta.sum()
            # compute convolution terms
            # conv_w_h = np.convolve(w, h, mode='full')
            # but need conv of w with h (size n)
            conv_w_h = np.convolve(w, h, mode='full')
            h_rev = h[::-1]
            conv_rev = np.convolve(w, h_rev, mode='full')
            idx = np.arange(n)
            term_sum = conv_w_h[(n-1) + idx]
            term_rev = conv_rev[(n-1) - idx]
            grad = 1.0 - term_rev - term_sum
            # learning rate depends on beta perhaps: lr = 0.05 / np.sqrt(beta) or so
            lr = 0.05 / np.sqrt(beta)
            h = h - lr * grad
            # clip
            np.clip(h, 0.0, 1.0, out=h)
            # adjust sum exactly
            # use iterative approach like previous
            diff = target - h.sum()
            # For fast converge, we can add diff / n to all, then clip and repeat.
            for _ in range(10):
                if abs(diff) < 1e-12:
                    break
                add = diff / n
                h += add
                # clip and compute new diff
                np.clip(h, 0.0, 1.0, out=h)
                diff = target - h.sum()
            # optionally add small random noise to avoid stagnation
            if step % 20 == 0:
                # small jitter
                h += rng.normal(scale=0.001, size=n)
                np.clip(h, 0.0, 1.0, out=h)
                # readjust sum after jitter

                for _ in range(5):

                    h += diff / n
                    np.clip(h, 0.0, 1.0, out=h)
                    diff = target - h.sum()
    # final adjust and compute C5
    adjust_sum_exact(h, rng)
    c5 = compute_c5(h)

import time

# ----- basic utilities -------------------------------------------------
def compute_c5(h):
    """Exact C5 for a sampled step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Exact sum correction to enforce Σh = n/2 (n even)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # random redistribution, fast convergence in practice
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # tiny residual correction (should be negligible)
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


# ----- gradient descent on smooth‑max ---------------------------------

                    np.clip(h, 0.0, 1.0, out=h)
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
            # update best if improved
            c5 = float(max_corr * (2.0 / n))
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
        # after each β round, enforce exact sum once more
        if time.time() > deadline:
            break
        adjust_sum_exact(h, rng)
    adjust_sum_exact(best_h, rng)
    return best_h, compute_c5(best_h)


# ----- SLSQP refinement (smooth‑max surrogate) ------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (np.log(np.sum(np.exp(bc - m))) + m) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """Constrained SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


# ----- greedy binary construction --------------------------------------
def greedy_binary(n, rng):
    """Greedy binary construction minimizing the worst overlap."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h


# ----- simple binary hill climbing --------------------------------------
def hill_climb_binary(h_bin, rng, deadline, max_iter=2000):
    """Swap‑based hill climbing on a binary vector."""
    best = h_bin.copy()
    n = best.size
    O_best = np.max(np.correlate(best.astype(np.int32), (1 - best).astype(np.int32), mode="full"))
    cur = best.copy()
    O_cur = O_best
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        O_new = np.max(np.correlate(cur.astype(np.int32), (1 - cur).astype(np.int32), mode="full"))
        if O_new < O_cur:
            O_cur = O_new
            best = cur.copy()
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
        else:
            cur[i] = 1
            cur[j] = 0
    return best


# ----- main entry point ------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    n = 360
    best_h = None
    best_c5 = np.inf

    def consider(h_candidate):
        nonlocal best_h, best_c5
        if time.time() > deadline:
            return
        h = np.clip(h_candidate, 0.0, 1.0)
        adjust_sum_exact(h, rng)
        h_opt, c5 = gradient_descent_h(h, rng, deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # 1) parent construction (if provided)
    if "initial_h_values" in globals():
        consider(np.array(initial_h_values, dtype=float))

    # 2) greedy binary + hill‑climb
    if time.time() < deadline:
        h_bin = greedy_binary(n, rng)
        if time.time() < deadline:
            h_bin = hill_climb_binary(h_bin, rng, deadline, max_iter=3000)
        consider(h_bin.astype(float))

    # 3) random binary starters
    for _ in range(5):
        if time.time() > deadline:
            break
        h_bin = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=n // 2, replace=False)
        h_bin[ones] = 1.0
        consider(h_bin)

    # 4) random continuous starters
    for _ in range(3):
        if time.time() > deadline:
            break
        h_rand = rng.random(n)
        consider(h_rand)

    # ----- final SLSQP polishing (if time permits) -----
    if best_h is not None and time.time() < deadline:
        h_ref = best_h.copy()
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
            if time.time() > deadline:
                break
            max_iter = 300 if beta < 200 else 600
            try:
                h_ref = slsqp_optimize(h_ref, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            c5 = compute_c5(h_ref)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()

    # ----- fallback -------------------------------------------------------
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # final safety
    np.clip(best_h, 0.0, 1.0, out=best_h)
    adjust_sum_exact(best_h, rng)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(n)
