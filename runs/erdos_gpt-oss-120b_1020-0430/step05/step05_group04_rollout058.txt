def c5_from_corr(corr, n):
    """C5 = (total_ones - min_corr) * (2 / n)  for binary h."""
    total = n // 2
    return (total - int(corr.min())) * (2.0 / n)


def apply_swap(i, j, h, ones, zeros, one_pos, zero_pos, corr, n):
    """Swap a 1 at i with a 0 at j, updating h, the index lists and the correlation."""
    # ---- remove i (was 1) ----
    for p in ones:
        if p == i:
            corr[n - 1] -= 1               # self‑pair
        else:
            corr[i - p + n - 1] -= 1
            corr[p - i + n - 1] -= 1

    # update list / dict for i -> zero
    pos_i = one_pos[i]
    last_one = ones[-1]
    ones[pos_i] = last_one
    one_pos[last_one] = pos_i
    ones.pop()
    del one_pos[i]

    zeros.append(i)
    zero_pos[i] = len(zeros) - 1

    # ---- add j (becomes 1) ----
    # remove j from zeros first
    pos_j = zero_pos[j]
    last_zero = zeros[-1]

            h_i = h[rng_idx]
            h_i_k = h[rng_idx + k]
            A = rng_idx[(h_i == 1) & (h_i_k == 0)]
            B = rng_idx[(h_i == 0) & (h_i_k == 1)]
        else:
            A = B = np.empty(0, dtype=int)

        if A.size and B.size:
            i = int(rng.choice(A))
            j = int(rng.choice(B))
            apply_swap(i, j, h, ones, zeros, one_pos, zero_pos, corr, n)
            cur_c = c5_from_corr(corr, n)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            no_improve = 0
            continue

        # -------- random improving swap (hill‑climb) ----------
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        apply_swap(i, j, h, ones, zeros, one_pos, zero_pos, corr, n)
        new_c = c5_from_corr(corr, n)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            no_improve = 0
            continue
        # revert non‑improving move
        apply_swap(i, j, h, ones, zeros, one_pos, zero_pos, corr, n)
        no_improve += 1

        # -------- restart after stagnation ----------
        if no_improve > 5000:
            h = np.zeros(n, dtype=np.int8)
            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1
            corr = np.correlate(h, h, mode='full')
            ones = list(np.flatnonzero(h))
            zeros = list(np.flatnonzero(1 - h))
            one_pos = {i: idx for idx, i in enumerate(ones)}
            zero_pos = {i: idx for idx, i in enumerate(zeros)}
            cur_c = c5_from_corr(corr, n)
            no_improve = 0

    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    parent = globals().get('initial_h_values')
    best_h, best_c, best_n = None, float('inf'), None

    # try progressively finer discretisations
    ns = [720, 480, 360]
    for idx_n, n in enumerate(ns):
        if time.time() >= deadline:
            break
        # ---- construct initial binary vector for this n ----
        if parent is not None:
            xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            top = np.argpartition(interp, -n // 2)[-n // 2:]
            init_h = np.zeros(n, dtype=np.int8)
            init_h[top] = 1
        else:
            init_h = np.zeros(n, dtype=np.int8)
            idx = rng.choice(n, n // 2, replace=False)
            init_h[idx] = 1

        # allocate a fair share of remaining time
        remaining = len(ns) - idx_n
        sub_dead = time.time() + (deadline - time.time()) / remaining
        h_n, c_n = search_n(n, init_h, sub_dead, rng)

        if c_n < best_c:
            best_c, best_h, best_n = c_n, h_n, n

    if best_h is None:
        # fallback simple construction
        n = 360
        best_h = np.zeros(n, dtype=np.int8)
        best_h[: n // 2] = 1
        best_c = c5_from_corr(np.correlate(best_h, best_h, mode='full'), n)
        best_n = n

    # final safety: ensure correct type and sum
    best_h = best_h.astype(float)
    return best_h.tolist(), float(best_c), int(best_n)
