import time

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 using aperiodic correlation."""
    # h is a 1‑D float array, values in [0,1]
    dx = 2.0 / h.size
    # np.correlate in 'full' mode gives the needed correlation values
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple greedy correction
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (max) overlap shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

                    rng: np.random.Generator,
                    step: float = 0.015,
                    max_iter: int = 12) -> np.ndarray:
    """Fine‑tune h with explicit worst‑shift gradient steps."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        # re‑enforce the sum constraint
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def binary_sa(n: int,
             rng: np.random.Generator,
             time_limit: float,
             init_h: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb on binary vectors (exactly n/2 ones).
    Returns the best vector found and its C5 value.
    """
    start = time.time()
    # initialise binary vector
    if init_h is not None and init_h.shape[0] == n:
        h = (init_h > 0.5).astype(float)
        # guarantee exactly n/2 ones
        cur_ones = int(h.sum())
        target = n // 2
        if cur_ones > target:
            # turn some ones to zero
            ones_idx = np.where(h == 1)[0]
            remove = rng.choice(ones_idx, cur_ones - target, replace=False)
            h[remove] = 0.0
        elif cur_ones < target:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, target - cur_ones, replace=False)
            h[add] = 1.0
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    # temperature schedule
    temp = 0.08                     # initial temperature
    decay = 0.99999                 # per‑iteration multiplicative decay

    while time.time() - start < time_limit:
        # pick a random 1 and a random 0 (swap them)
        i = rng.integers(n)
        while h[i] < 0.5:          # ensure i is a 1
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:          # ensure j is a 0
            j = rng.integers(n)

        # perform swap
        h[i] = 0.0
        h[j] = 1.0

        new_c = compute_c5(h)
        diff = new_c - cur_c
        accept = False
        if diff <= 0.0:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-diff / temp) if temp > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

        temp *= decay
    return best_h, best_c


def final_hill_climb(h: np.ndarray,
                     rng: np.random.Generator,
                     iters: int = 6000,
                     scale: float = 0.02) -> tuple[np.ndarray, float]:
    """Small guided perturbation hill‑climb on a continuous h."""
    cur_c = compute_c5(h)
    for _ in range(iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        if nc < cur_c:
            h, cur_c = nh, nc
    return h, cur_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # 1. start from the parent construction if it exists
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, h0.size

    # ------------------------------------------------------------------
    # Helper to update the global best
    def update_best(h, c):
        nonlocal best_h, best_c, best_n
        if c < best_c - 1e-12:
            best_h, best_c, best_n = h.copy(), c, h.size

    # ------------------------------------------------------------------
    # 2. low‑resolution binary SA (n = 128)
    remaining = budget_s - (time.time() - start)
    if remaining > 120.0:
        t_alloc = min(150.0, remaining * 0.18)
        h128, c128 = binary_sa(128, rng, t_alloc,
                               init_h=best_h if best_n == 128 else None)
        update_best(h128, c128)
        # small gradient polish
        h128 = gradient_refine(best_h, rng, step=0.008, max_iter=8)
        update_best(h128, compute_c5(h128))
        remaining = budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # 3. medium‑resolution binary SA (n = 256)
    if remaining > 130.0:
        t_alloc = min(320.0, remaining * 0.30)
        h256, c256 = binary_sa(256, rng, t_alloc,
                               init_h=best_h if best_n == 256 else None)
        update_best(h256, c256)
        # gradient polish
        h256 = gradient_refine(best_h, rng, step=0.007, max_iter=10)
        update_best(h256, compute_c5(h256))
        remaining = budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # 4. high‑resolution binary SA (n = 512)
    if remaining > 80.0:
        t_alloc = max(0.0, remaining - 10.0)  # keep a small safety margin
        h512, c512 = binary_sa(512, rng, t_alloc,
                               init_h=best_h if best_n == 512 else None)
        update_best(h512, c512)
        # gradient polish
        h512 = gradient_refine(best_h, rng, step=0.006, max_iter=12)
        update_best(h512, compute_c5(h512))
        remaining = budget_s - (time.time() - start)

    # ------------------------------------------------------------------
    # 5. final fine‑tuned hill‑climb on the current best
    if best_h is not None and (time.time() - start) < budget_s * 0.98:
        h_final, c_final = final_hill_climb(best_h, rng,
                                            iters=8000, scale=0.015)
        update_best(h_final, c_final)

    # ------------------------------------------------------------------
    # 6. final sanity enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
