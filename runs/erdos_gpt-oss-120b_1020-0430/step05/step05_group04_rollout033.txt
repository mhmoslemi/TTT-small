def greedy_improve(h, rng, deadline, max_pairs=300):
    N = h.shape[0]
    while time.time() < deadline:
        # compute corr and max shift
        corr = np.correlate(h, 1-h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / N)
        k = max_idx - (N - 1)
        # find mismatched sets
        if k >= 0:
            s_len = N - k
            h_i = h[:s_len]
            h_ik = h[k:]
            S1 = np.flatnonzero((h_i == 1) & (h_ik == 0))
            S0 = np.flatnonzero((h_i == 0) & (h_ik == 1))
        else:
            k2 = -k
            s_len = N - k2
            h_i = h[k2:]
            h_ik = h[:s_len]
            S1 = np.flatnonzero((h_i == 1) & (h_ik == 0))
            S0 = np.flatnonzero((h_i == 0) & (h_ik == 1))
            # convert indices to absolute positions
            S1 = S1 + k2
            S0 = S0 + k2
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        # sample pairs
        if total_pairs <= max_pairs:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_pairs)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            # swap
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h)
            if c < best_c - 1e-12:
                best_c = c
                best_pair = (i, j)
            # revert
            h[i], h[j] = h[j], h[i]
        if best_pair is None:
            break
        # apply best swap
        i, j = best_pair
        h[i], h[j] = h[j], h[i]
        # continue loop
    return h
```

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum adjustment
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

            g[:upper] += 1.0 - h[k:]        # + (1 - h[i+k])

        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust sum via random adjustments
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global vars for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- core utilities ----------
def compute_c5(h):
    """Return C5 for vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def compute_c5_with_corr(h):
    """Return (C5, correlation array)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n)), corr

def adjust_sum_exact(h, rng, max_iter=2000):
    """Adjust h in‑place so that Σh = len(h)/2 exactly."""
    n = h.shape[0]
    target = n / 2.0
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
    return h

# ---------- SLSQP surrogate ----------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    adjust_sum_exact(x, rng)
    return x

# ---------- gradient refinement ----------
def gradient_refine(h, rng, step=0.02, max_iter=10):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # (1 - h[i+k])
            g[k:] -= h[:upper]                # -h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- binary simulated annealing ----------
def binary_sa(h, rng, deadline):
    """SA on binary h (0/1) to reduce C5. Returns best h and its C5."""
    N = h.shape[0]
    cur_c, corr = compute_c5_with_corr(h)
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0
        h[j] = 1
        new_c, new_corr = compute_c5_with_corr(h)
        T = T0 * (0.9995 ** it) + 1e-12
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            corr = new_corr
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                last_improve = it
        else:
            # revert
            h[i] = 1
            h[j] = 0
        it += 1
        if it - last_improve > 20000:
            break
    return best_h, best_c

# ---------- greedy local improvement ----------
def greedy_improve(h, rng, deadline, max_pairs=300):
    """Iterative swap improvement for binary h until no gain or time out."""
    N = h.shape[0]
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / N)
        k = max_idx - (N - 1)
        if k >= 0:
            slen = N - k
            hi = h[:slen]
            hik = h[k:]
            S1 = np.flatnonzero((hi == 1) & (hik == 0))
            S0 = np.flatnonzero((hi == 0) & (hik == 1))
        else:
            kp = -k
            slen = N - kp
            hi = h[kp:]
            hik = h[:slen]
            S1 = np.flatnonzero((hi == 1) & (hik == 0)) + kp
            S0 = np.flatnonzero((hi == 0) & (hik == 1)) + kp
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        if total_pairs <= max_pairs:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_pairs)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h)
            if c < best_c - 1e-12:
                best_c = c
                best_pair = (i, j)
            h[i], h[j] = h[j], h[i]  # revert
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = h[j], h[i]  # apply
    return h

# ---------- main entry ----------
