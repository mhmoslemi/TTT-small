import time, numpy as np, scipy.optimize as opt

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_corr_fft(h):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    # zero pad
    fh = np.fft.rfft(np.concatenate([h, np.zeros(L - n, dtype=h.dtype)]))
    f1mh = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(L - n, dtype=h.dtype)]))
    corr = np.fft.irfft(fh * f1mh, n=L)[:2*n-1]
    return corr

def compute_c5_fft(h):
    corr = compute_corr_fft(h)
    dx = 2.0 / h.size
    return float(np.max(corr) * dx)

def adjust_sum_exact(h, rng, max_iter=2000, eps=1e-12):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    val = (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    return val

def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def gradient_refine(h, rng, step=0.03, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = compute_corr_fft(h)
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        elif k > 0:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        else:  # k < 0
            kk = -k
            upper = n - kk
            g[kk:] += 1.0 - h[:upper]
            g[:upper] -= h[kk:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # re‑project sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution
            for _ in range(200):

                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:

    return h

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5_fft(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 8000:
            break
    return best_h, best_c

        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 12000:
            break
    return best, best_c

def upsample_vector(h, new_n):
    old_n = h.size
    xs = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xt = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_up = np.interp(xt, xs, h)
    return h_up

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99  # margin
    best_h = None
    best_c = np.inf
    best_n = None

    parent = None
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)

    # candidate sizes to explore
    size_list = [360, 480, 720]

    for n in size_list:
        if time.time() > deadline:
            break

        # generate starting vectors
        starts = []

        # from parent if exists
        if parent is not None:
            # interpolate parent to current n
            h_interp = upsample_vector(parent, n)
            starts.append(h_interp)

            # binary version (top half)
            idx = np.argpartition(h_interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            starts.append(bin_vec)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        # random fractional uniform
        rand_frac = rng.random(n)
        target = n / 2.0
        rand_frac *= target / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)
        starts.append(rand_frac)

        # random fractional biased (square)
        rand_frac2 = rng.random(n) ** 2
        rand_frac2 *= target / rand_frac2.sum()
        rand_frac2 = np.clip(rand_frac2, 0.0, 1.0)
        adjust_sum_exact(rand_frac2, rng)
        starts.append(rand_frac2)

        # clustered binary pattern (alternating blocks)
        # create block sizes randomly summing to n/2 ones
        ones_left = n // 2
        blocks = []
        while ones_left > 0:
            size = rng.integers(1, max(2, ones_left // 2 + 1))
            blocks.append(size)
            ones_left -= size
        # interleave zeros of random lengths
        zeros_left = n - n // 2
        zero_blocks = []
        while zeros_left > 0:
            size = rng.integers(1, max(2, zeros_left // 2 + 1))
            zero_blocks.append(size)
            zeros_left -= size
        # build pattern
        pattern = []
        for a, b in zip(blocks, zero_blocks):
            pattern.extend([1] * a)
            pattern.extend([0] * b)
        # possibly extra remainder
        extra = blocks[len(zero_blocks):] if len(blocks) > len(zero_blocks) else zero_blocks[len(blocks):]
        for val in extra:
            pattern.append(1 if len(pattern) % 2 == 0 else 0)
        # truncate/pad
        pat_arr = np.array(pattern[:n], dtype=float)
        if pat_arr.size < n:
            pat_arr = np.pad(pat_arr, (0, n - pat_arr.size), constant_values=0.0)
        starts.append(pat_arr)

        # iterate over starts
        for h0 in starts:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # binary SA if binary
            if np.all((h == 0) | (h == 1)):
                sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
                h_bin, c_bin = binary_sa_hill_climb(h.copy(), rng, sub_dead)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin.astype(float)

            # fractional SA
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
            h_frac, c_frac = fractional_sa_hill_climb(h.copy(), rng, sub_dead)
            if c_frac < best_c:
                best_h, best_c, best_n = h_frac.copy(), c_frac, n
            h = h_frac

            # gradient refinement
            h = gradient_refine(h, rng, step=0.03, max_iter=12)
            c = compute_c5_fft(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # SLSQP cascade
            betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]
            for beta in betas:
                if time.time() > deadline:
                    break
                max_iter = 800 if beta >= 5000 else 600
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c = compute_c5_fft(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final gradient step
            h = gradient_refine(h, rng, step=0.02, max_iter=15)
            c = compute_c5_fft(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # Upsample if we have time and best_n < 720
    if best_h is not None and time.time() < deadline and best_n < 720:
        n_up = 720
        h_up = upsample_vector(best_h, n_up)
        adjust_sum_exact(h_up, rng)
        # series of refinements
        cycles = 2
        for _ in range(cycles):
            if time.time() > deadline:
                break
            h_up = gradient_refine(h_up, rng, step=0.03, max_iter=18)
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
            h_up, _ = fractional_sa_hill_climb(h_up, rng, sub_dead)
        # final SLSQP pass
        if time.time() < deadline:
            try:
                h_up = slsqp_optimize(h_up, beta=20000.0, rng=rng, max_iter=1000)
            except Exception:
                pass
        c_up = compute_c5_fft(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # fallback
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5_fft(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_n)

import time
import numpy as np
import scipy.optimize as opt

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    pad = np.zeros(L - n, dtype=h.dtype)
    fh = np.fft.rfft(np.concatenate([h, pad]))
    f1mh = np.fft.rfft(np.concatenate([1.0 - h, pad]))
    corr = np.fft.irfft(fh * f1mh, n=L)[: 2 * n - 1]
    return corr


def compute_c5_fft(h: np.ndarray) -> float:
    corr = compute_corr_fft(h)
    dx = 2.0 / h.size
    return float(np.max(corr) * dx)


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000, eps: float = 1e-12):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        i = rng.integers(n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 600) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.03, max_iter: int = 12) -> np.ndarray:
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = compute_corr_fft(h)
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        elif k > 0:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        else:
            kk = -k
            upper = n - kk
            g[kk:] += 1.0 - h[:upper]
            g[:upper] -= h[kk:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def binary_sa_hill_climb(h_bin: np.ndarray, rng: np.random.Generator, time_deadline: float):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5_fft(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 8000:
            break
    return best_h, best_c


def fractional_sa_hill_climb(h: np.ndarray, rng: np.random.Generator, time_deadline: float):

                last_improve = it
        else:

            cur[dst] -= delta
