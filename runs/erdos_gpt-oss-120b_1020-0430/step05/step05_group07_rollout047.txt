import time, numpy as np

def compute_c5(h):
    """Maximum overlap C5 for a step function h on [0,2]."""
    n = h.size
    L = 2 * n - 1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * G, n=L)
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, target, rng):
    """Redistribute mass so that sum(h) == target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent on the worst shift."""
