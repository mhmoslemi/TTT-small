from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=20):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]    # + (1 - h[i+k])
            g[k:] -= h[:upper]         # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple projection of sum by adjusting random entries
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def greedy_improve(h, rng, max_iter=50, deadline=None):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # find mismatched pairs for this shift
        i_range = np.arange(n)
        # positions where h[i]=1 and h[i+k]=0
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        # positions where h[i]=0 and h[i+k]=1
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        # sample pairs
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = float(np.max(corr) * (2.0 / n))
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

def simulated_annealing_binary(h0, rng, time_limit, init_temp_factor=0.1):
    # h0 binary (0/1) float array
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = cur_h.size
    init_T = max(init_temp_factor * cur_c, 1e-5)
    it = 0
    while time.time() < time_limit:
        # pick two different indices
        i, j = rng.integers(0, n, size=2)
        if cur_h[i] == cur_h[j]:
            continue
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(cur_h)
        # temperature schedule
        T = init_T * (0.99995 ** it) + 1e-12
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        it += 1
    return best_h, best_c

def lbsfgs_optimize(h0, rng, deadline):
    n = h0.size
    target = n / 2.0
    lam = 1000.0
    beta = 20000.0
    def obj(x):
        # compute overlap array
        corr = np.correlate(x, 1.0 - x, mode='full')
        O = corr * (2.0 / n)
        # smooth max
        maxb = np.max(beta * O)
        sm = (maxb + np.log(np.sum(np.exp(beta * O - maxb)))) / beta
        pen = lam * (np.sum(x) - target) ** 2
        return sm + pen
    bounds = [(0.0, 1.0)] * n
    maxiter = 300
    # ensure we respect deadline
    remaining = deadline - time.time()
    if remaining < 1.0:
        return h0
    res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def slsqp_optimize(beta, h, rng, max_iter=400):
    # like previous implementation but adapt for larger beta and simple bound constraints
    n = h.size
    target = n / 2.0
    CURRENT_BETA = beta
    CURRENT_TARGET = target
    def sum_eq_constraint(x):
        return np.sum(x) - CURRENT_TARGET
    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=30):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def greedy_improve(h, rng, max_iter=50, deadline=None):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = float(np.max(corr) * (2.0 / n))
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

def simulated_annealing_binary(h0, rng, time_limit):
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = cur_h.size
    init_T = max(0.5 * cur_c, 1e-5)
    it = 0
    while time.time() < time_limit:
        i, j = rng.integers(0, n, size=2)
        if cur_h[i] == cur_h[j]:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(cur_h)
        T = init_T * (0.99995 ** it) + 1e-12
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        it += 1
    return best_h, best_c

def lbsfgs_optimize(h0, rng, deadline):
    n = h0.size
    target = n / 2.0
    lam = 1000.0
    beta = 20000.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        O = corr * (2.0 / n)
        maxb = np.max(beta * O)
        sm = (maxb + np.log(np.sum(np.exp(beta * O - maxb)))) / beta
        pen = lam * (np.sum(x) - target) ** 2
        return sm + pen
    bounds = [(0.0, 1.0)] * n
    if deadline - time.time() < 1.0:
        return h0
    res = minimize(
        obj, h0, method="L-BFGS-B", bounds=bounds,
        options={"maxiter": 300, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def slsqp_optimize(beta, h, rng, max_iter=400):
    n = h.size
    target = n / 2.0
    CURRENT_BETA = beta
    CURRENT_TARGET = target
    def sum_eq(x):
        return np.sum(x) - CURRENT_TARGET
    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    cons = {"type": "eq", "fun": sum_eq}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj, h, method="SLSQP", bounds=bounds,
        constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = float("inf")
    best_n = None

    n_candidates = [720, 540, 480, 360, 240, 180]
    n_candidates = [n for n in n_candidates if n % 2 == 0]

    if "initial_h_values" in globals():
        parent_n = len(initial_h_values)
        if parent_n not in n_candidates:
            n_candidates = [parent_n] + n_candidates

    for n in n_candidates:
        if time.time() > deadline:
            break
        while True:
            now = time.time()
            if now > deadline:
                break
            remaining = deadline - now
            if remaining < 5.0:
                break

            # random binary start
            h0 = np.zeros(n)
            h0[rng.choice(n, n // 2, replace=False)] = 1.0

            # simulated annealing
            sa_time = min(remaining * 0.2, 12.0)
            h_sa, c_sa = simulated_annealing_binary(
                h0, rng, time.time() + sa_time
            )
            if c_sa < best_c:
                best_h, best_c, best_n = h_sa.copy(), c_sa, n

            # gradient refinement
            h_gr = gradient_refine(h_sa.copy(), rng, step=0.02, max_iter=30)
            c_gr = compute_c5(h_gr)
            if c_gr < best_c:
                best_h, best_c, best_n = h_gr.copy(), c_gr, n

            # SLSQP cascade
            betas = (200, 1000, 5000, 20000, 50000)
            h_cur = h_gr
            for beta in betas:
                if time.time() > deadline:
                    break
                h_cur = slsqp_optimize(beta, h_cur, rng, max_iter=400)
                c_cur = compute_c5(h_cur)
                if c_cur < best_c:
                    best_h, best_c, best_n = h_cur.copy(), c_cur, n

            # final gradient refinement
            h_final = gradient_refine(h_cur.copy(), rng, step=0.02, max_iter=30)
            c_final = compute_c5(h_final)
            if c_final < best_c:
                best_h, best_c, best_n = h_final.copy(), c_final, n

            # binary greedy improve if applicable
            if np.all((np.abs(h_final) < 1e-12) | (np.abs(h_final - 1) < 1e-12)):
                h_bin = np.rint(h_final).astype(float)
                adjust_sum_exact(h_bin, rng)
                h_bin = greedy_improve(h_bin, rng, max_iter=50, deadline=deadline)
                c_bin = compute_c5(h_bin)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n

            # optional L-BFGS-B refinement
            if time.time() < deadline - 10.0:
                h_lb = lbsfgs_optimize(h_final.copy(), rng, deadline)
                c_lb = compute_c5(h_lb)
                if c_lb < best_c:
                    best_h, best_c, best_n = h_lb.copy(), c_lb, n

            if time.time() > deadline - 2.0:
                break

    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
