import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> None:
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

def grad_step(h: np.ndarray, beta: float, step: float,
              rng: np.random.Generator) -> np.ndarray:

    # smoothed max weights
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exp_term = np.exp(beta * (corr - max_c))
    w = exp_term / exp_term.sum()

    grad = np.zeros_like(h)

    # k >= 0
    for kk in range(n):
        wk = w[kk + n - 1]
        if wk == 0.0:
            continue
        i_end = n - kk
        if i_end <= 0:
            continue
        i_slice = slice(0, i_end)
        j_slice = slice(kk, n)
        grad[i_slice] += wk * (1.0 - h[j_slice])
        grad[j_slice] -= wk * h[i_slice]

        wk = w[n - 1 - kk]

        grad[i_slice] += wk * (1.0 - h[j_slice])
        grad[j_slice] -= wk * h[i_slice]
