def project_onto_simplex_box(y, target, lower=0.0, upper=1.0):
    # find λ such that sum_i clip(y_i - λ, lower, upper) = target.
    # We'll binary search λ in [low, high].
    low = np.min(y) - upper
    high = np.max(y) - lower
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, lower, upper)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(y - (low + high) / 2.0, lower, upper)
    # ensure sum close
    # adjust slight error by distributing diff among non-bound entries
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust equally among free variables
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            # clip again to ensure bounds
            x = np.clip(x, lower, upper)
    return x

The binary search is monotonic: as λ increases, y_i - λ decreases, so after clipping, sum decreases. So yes.

Now compute weighted gradient:

def weighted_gradient(h, beta):
    n = h.size
    # correlation vector
    corr = np.correlate(h, 1 - h, mode='full')
    # compute weights
    # avoid overflow by subtracting max
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    w = exps / np.sum(exps)
    # compute gradient
    g = np.zeros_like(h)
    # shifts range from -(n-1) to (n-1)
    # index in w: shift + (n-1)
    for shift in range(-(n-1), n):
        wk = w[shift + n - 1]
        if wk == 0:
            continue
        if shift > 0:
            # positive shift: h[i] * (1 - h[i+shift])
            # gradient: (1 - h[i+shift]) for i in [0, n-shift-1]; -h[i-shift] for i in [shift, n-1]
            g[:n-shift] += wk * (1.0 - h[shift:])
            g[shift:]   -= wk * h[:n-shift]
        elif shift < 0:
            k = -shift
            # shift negative: h[i] * (1 - h[i - k]), i >= k
            # gradient: (1 - h[i - k]) for i in [k, n-1]; -h[i + k] for i in [0, n - k -1]
            g[k:]       += wk * (1.0 - h[:n-k])
            g[:n-k]    -= wk * h[k:]
        else:
            # shift = 0: h[i] * (1 - h[i]) => gradient = 1 - 2h[i]
            g += wk * (1.0 - 2.0 * h)
    return g

Complexity: O(n * (2n-1)) ~ O(n^2), which is okay.

def worst_gradient_full(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        k = shift
        if k < n:
            grad[:n - k] += 1.0 - h[k:]
            grad[k:] -= h[:n - k]
    else:
        k = -shift
        grad[k:] += 1.0 - h[:n - k]
        grad[:n - k] -= h[k:]
    return grad

def project_onto_simplex_box(y, target, lower=0.0, upper=1.0):
    # Using bisection for lambda
    low = np.min(y) - upper
    high = np.max(y) - lower
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, lower, upper)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(y - (low + high) / 2.0, lower, upper)
    # adjust any residual diff
    diff = target - x.sum()

        # distribute to non-bound entries
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            # add equally
            x[free] += diff / free.sum()
            x = np.clip(x, lower, upper)
    return x

import time
import numpy as np

# ----- basic utilities -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def project_onto_simplex_box(y: np.ndarray, target: float,
                            lower: float = 0.0, upper: float = 1.0) -> np.ndarray:
    """Projection of y onto {x | lower ≤ x ≤ upper, sum(x)=target}."""
    low = np.min(y) - upper
    high = np.max(y) - lower
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, lower, upper)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(y - (low + high) / 2.0, lower, upper)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, lower, upper)
    return x


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.random(n)
    return project_onto_simplex_box(y, n / 2.0)


def triangular_h(n: int) -> np.ndarray:
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    return np.where(xs <= 1.0, xs, 2.0 - xs)


# ----- gradient of the worst shift -------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    shift = idx
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        if k < n:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
    else:
        k = -shift
        g[k:] += 1.0 - h[: n - k]
        g[: n - k] -= h[k:]
    return g


# ----- perturbations ---------------------------------------------------
def guided_perturb_max(h: np.ndarray, grad: np.ndarray) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    nh = h.copy()
    nh[i] -= max_eps
    nh[j] += max_eps
    return nh

                      max_iters: int = 2000,

        else:
            # occasional tiny random move to escape plateaus
            if rng.random() < 0.02:
                i = rng.integers(0, cur.size)
                j = rng.integers(0, cur.size)
                if i != j:
                    max_eps = min(cur[i], 1.0 - cur[j])
                    if max_eps > 0.0:
                        eps = rng.random() * max_eps * 0.05
                        tmp = cur.copy()
                        tmp[i] -= eps
                        tmp[j] += eps
                        c2 = compute_c5(tmp)
                        if c2 < best_c:
                            best_c, best_h, cur = c2, tmp.copy(), tmp
    return best_h, best_c


# ----- smoothed objective & its gradient -------------------------------
def softmax_c5(h: np.ndarray, beta: float) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sumexp = np.sum(np.exp(beta * (corr - m)))
    return m + np.log(sumexp) / beta


def weighted_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    g = np.zeros_like(h)
    for shift in range(-(n - 1), n):
        wk = w[shift + n - 1]
        if wk == 0.0:
            continue
        if shift > 0:
            k = shift
            g[: n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[: n - k]
        elif shift < 0:
            k = -shift
            g[k:] += wk * (1.0 - h[: n - k])
            g[: n - k] -= wk * h[k:]
        else:  # shift == 0
            g += wk * (1.0 - 2.0 * h)
    return g


# ----- main driver -----------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # target dimension (use parent size if present)
    final_n = 512
    if "initial_h_values" in globals():
        iv = np.array(initial_h_values, dtype=float)
        final_n = iv.size
        iv = np.clip(iv, 0.0, 1.0)
        iv = project_onto_simplex_box(iv, final_n / 2.0)
        candidates = [iv]
    else:
        candidates = []

    # a few deterministic starters
    candidates.append(triangular_h(final_n))
    bin_vec = np.zeros(final_n)
    bin_vec[rng.choice(final_n, final_n // 2, replace=False)] = 1.0
    candidates.append(bin_vec)
    candidates.append(random_feasible_vector(final_n, rng))

    # keep the best of the starters
    best_h = None
    best_c = np.inf
    target = final_n / 2.0
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- smoothed gradient descent with increasing β -----------------
    beta_schedule = [5., 20., 50., 100., 300., 1000., 3000., 8000., 20000.]
    cur = best_h.copy()
    for beta in beta_schedule:
        if time.time() - start > 0.9 * budget_s:
            break
        obj = softmax_c5(cur, beta)
        it = 0
        while time.time() - start < 0.9 * budget_s and it < 2000:
            grad = weighted_gradient(cur, beta)
            # backtracking line search
            alpha = 1.0
            improved = False
            for _ in range(10):
                y = cur - alpha * grad
                nxt = project_onto_simplex_box(y, target)
                obj_nxt = softmax_c5(nxt, beta)
                if obj_nxt < obj - 1e-12:
                    cur, obj = nxt, obj_nxt
                    improved = True
                    break
                alpha *= 0.5
            if not improved:
                # tiny random move as fallback
                i, j = rng.integers(0, final_n, size=2)
                if i != j:
                    max_eps = min(cur[i], 1.0 - cur[j])
                    if max_eps > 0.0:
                        eps = rng.random() * max_eps * 0.1
                        tmp = cur.copy()
                        tmp[i] -= eps
                        tmp[j] += eps
                        obj_tmp = softmax_c5(tmp, beta)
                        if obj_tmp < obj:
                            cur, obj = tmp, obj_tmp
            if it % 20 == 0:
                c5 = compute_c5(cur)
                if c5 < best_c:
                    best_c, best_h = c5, cur.copy()
            if np.linalg.norm(grad) < 1e-8:
                break
            it += 1

    # ----- final hill‑climb refinement ---------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 0.0:
        hc_h, hc_c = guided_hill_climb(cur, rng,
                                       max_iters=4000,
                                       time_budget=remaining * 0.5)
        if hc_c < best_c:
            best_c, best_h = hc_c, hc_h.copy()
            cur = hc_h

    # ----- stochastic tiny tweaks until time runs out ------------------
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb_max(best_h, grad)
        nc = compute_c5(nh)
        if nc < best_c:
            best_c, best_h = nc, nh.copy()

    # final sanity projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_simplex_box(best_h, final_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(final_n)
