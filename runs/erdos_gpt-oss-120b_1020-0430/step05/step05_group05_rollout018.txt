import time
import numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

# ---------- SLSQP with smooth‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """log‑sum‑exp surrogate of the max overlap."""
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # use log‑sum‑exp trick for stability
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=1500):
    """Run SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )

            hf *= (n / 2) / hf.sum()
            hf = np.clip(hf, 0.0, 1.0)
            adjust_sum_exact(hf, rng)
            starts.append(hf)

        return starts

    # ---------- optimisation schedule ----------
    betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
             2000.0, 5000.0, 10000.0, 20000.0, 50000.0]

    def optimise_one(start_h, n):
        if time.time() >= deadline:
            return
        h = np.clip(start_h, 0.0, 1.0)
        adjust_sum_exact(h, rng)

        # SLSQP cascade
        for beta in betas:
            if time.time() >= deadline:
                break
            # allocate more iterations for larger beta
            if beta < 500:
                max_iter = 1200
            elif beta < 5000:
                max_iter = 1800
            else:
                max_iter = 2500
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            try_update(h, n)

        # gradient polishing
        h = gradient_refine(h, rng, step=0.025, max_iter=80)
        try_update(h, n)

        # binary greedy improvement
        hb = fractional_to_binary(h, rng)
        hb = binary_targeted_improve(hb, rng, deadline)
        try_update(hb, n)

    # ---------- first pass at base size ----------
    starts_base = generate_starts(n_base)
    for s in starts_base:
        if time.time() >= deadline:
            break
        optimise_one(s, n_base)

    # ---------- up‑sampling loop ----------
    current_n = n_base
    while (time.time() < deadline) and (current_n * 2 <= 720):
        # up‑sample best known solution
        xs = np.linspace(0.0, 2.0, current_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, current_n * 2, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        current_n *= 2
        starts_up = generate_starts(current_n)
        # replace the first start with the interpolated one (most promising)
        starts_up[0] = h_up
        for s in starts_up:
            if time.time() >= deadline:
                break
            optimise_one(s, current_n)

    # ---------- final fallback (guarantee a result) ----------
    if best_h is None:
        best_n = n_base
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final tidy‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
