def binary_sa_opt(h, rng, deadline):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    indices_one = np.flatnonzero(cur)
    indices_zero = np.flatnonzero(1-cur)
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        if len(indices_one)==0 or len(indices_zero)==0:
            break
        i = rng.choice(indices_one)
        j = rng.choice(indices_zero)
        cur[i] = 0
        cur[j] = 1
        c = compute_c5(cur)
        # cooling schedule
        T = T0 * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            # update lists
            indices_one = np.setdiff1d(indices_one, [i])
            indices_one = np.append(indices_one, j)
            indices_zero = np.setdiff1d(indices_zero, [j])
            indices_zero = np.append(indices_zero, i)
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            # revert
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it % 5000 == 0 and best_c < cur_c:  # early stop maybe
            pass
    return best, best_c
```

import time, numpy as np, cvxpy as cp, math

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, target, rng, max_iter=2000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def fractional_sa_opt(h, rng, deadline):

def compute_c5(h: np.ndarray) -> float:

    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Redistribute mass to meet Σh = target while staying in [0,1]."""

    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.03, max_iter: int = 12) -> np.ndarray:

                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ------------------------------------------------------------------ #
    # Fallback simple construction
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = adjust_sum_exact(best_h, n / 2.0, rng)
        best_c = compute_c5(best_h)

    # Final sanity adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
