import time
import numpy as np

def compute_c5(h):
    """Return the max overlap integral C5 for a step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def max_mismatch(h):
    """Maximum mismatch count over all non‑negative shifts for binary h."""
    n = h.size
    # autocorrelation of h
    ac = np.correlate(h, h, mode="full")
    ac = ac[n - 1 :]  # shift k = 0..n-1
    target = int(h.sum())
    mis = target - ac
    max_m = int(mis.max())
    k = int(mis.argmax())
    return max_m, k, ac

def improve_h(h, rng, max_no_improve=2000, max_time=None):
    """Greedy swap improvement targeting the worst shift."""
    n = h.size
    target = int(h.sum())
    no_improve = 0
    start = time.time()
    while True:
        if max_time is not None and (time.time() - start) > max_time:
            break
        max_m, k, _ = max_mismatch(h)
        if max_m == 0:
            break
        # ones that disagree with shift k
        ones = np.flatnonzero(h)
        j1 = ones + k
        bad_mask = (j1 >= n) | (h[j1] == 0)
        BadOnes = ones[bad_mask]
        # zeros that would become matches for shift k
        zeros = np.flatnonzero(h == 0)
        j0 = zeros + k
        good_mask = (j0 < n) & (h[j0] == 1)
        GoodZeros = zeros[good_mask]
        if BadOnes.size == 0 or GoodZeros.size == 0:
            break
        improved = False
        for _ in range(5):
            i = BadOnes[rng.integers(BadOnes.size)]
            j = GoodZeros[rng.integers(GoodZeros.size)]
            h[i] = 0
            h[j] = 1
            new_max_m, _, _ = max_mismatch(h)
            if new_max_m < max_m:
                max_m = new_max_m
                no_improve = 0
                improved = True
                break
            h[i] = 1
            h[j] = 0
        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    return h, max_m

def binary_search_n(n, rng, deadline):
    """Search for a good binary vector of length n before deadline."""
    best_h = None
    best_c = np.inf
    # possible warm‑start from parent if available
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        idx = np.argpartition(interp, -n // 2)[-n // 2 :]
        h0 = np.zeros(n, dtype=np.int8)
        h0[idx] = 1
        h0, max_m = improve_h(h0, rng, max_time=min(30.0, deadline - time.time()))
        c0 = max_m * (2.0 / n)
        if c0 < best_c:
            best_h, best_c = h0.astype(float), c0

    # main random‑restart loop
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 0.5:
            break
        per_restart = max(0.5, remaining * 0.2)
        # random binary start
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, n // 2, replace=False)] = 1
        h, max_m = improve_h(h, rng, max_time=per_restart)
        c = max_m * (2.0 / n)
        if c < best_c:
            best_h, best_c = h.astype(float), c
        if best_c < 0.3808:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None

    best_n = None

    # try decreasing sizes; larger n gives finer resolution

        if best_c < 0.3808:

    # final polish: small random perturbations of the best solution
    if best_h is not None and time.time() < deadline:

        for _ in range(4):
