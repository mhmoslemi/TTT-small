    diff = target - v.sum()
    for _ in range(100):

        k_worst = min(5, ov.size - 1)
        worst_idx = np.argpartition(ov, k_worst)[:k_worst]
        improved = False
        for idx in worst_idx:
            shift = idx - center
            if shift == 0:
                continue
            k = abs(shift)
            neigh = np.zeros(n, dtype=int)
            if k < n:
                neigh[: n - k] += h[k:]
                neigh[k :] += h[: n - k]
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i_one = ones[np.argmin(neigh[ones])]
            i_zero = zeros[np.argmax(neigh[zeros])]
            if neigh[i_one] < neigh[i_zero]:
                h[i_one] = 0
                h[i_zero] = 1
                improved = True
                break
        if not improved:
            break
    return h


def _run_opt_on_n(n, init_h, best_h, best_c, best_n, deadline):
    """Optimize for a fixed n, updating the global best."""
    target = n / 2.0
    dx = 2.0 / n
    L = 1 << ((3 * n - 2).bit_length())          # FFT length
    betas = [0.5, 2, 5, 10, 20, 50, 100, 200,
             500, 1000, 2000, 5000]
    iters_per_beta = {
        0.5: 200, 2: 150, 5: 120, 10: 100, 20: 80,
        50: 60, 100: 50, 200: 40, 500: 30,
        1000: 20, 2000: 15, 5000: 10
    }
    restarts = 5
    for r in range(restarts):
        if time.time() > deadline:
            break
        if r == 0 and init_h is not None:
            h = np.array(init_h, copy=True)
            if h.size != n:
                xs = np.linspace(0.0, 2.0, init_h.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(xt, xs, init_h)
            h = _project_to_sum_box(h, target)
        else:
            h = np.random.random(n)
            h = h * (target / h.sum())
            h = _project_to_sum_box(h, target)

        for beta in betas:
            if time.time() > deadline:
                break
            iters = iters_per_beta.get(beta, 30)
            for it in range(iters):
                if time.time() > deadline:
                    break
                H = np.fft.fft(h, L)
                overlap = np.fft.ifft(H * np.fft.fft(h[::-1], L)).real[: 2 * n - 1]
                C = dx * (target - overlap)

                maxC = C.max()
                expb = np.exp(beta * (C - maxC))
                p = expb / expb.sum()
                p_rev = p[::-1]

                G = np.fft.ifft(H * np.fft.fft(p_rev, L)).real[:n]
                grad = -2.0 * dx * G
                step = 0.5 / (1.0 + it * 0.1)
                h = h - step * grad
                h = _project_to_sum_box(h, target)

            c_val = _compute_c5(h)
            if c_val < best_c:
                best_h[:] = h
                best_c = c_val
                best_n[0] = n

        # binary rounding + greedy improve
        hb = np.zeros_like(h)
        idx = np.argpartition(h, -n // 2)[-n // 2 :]
        hb[idx] = 1.0
        hb = _project_to_sum_box(hb, target)
        hb = _binary_greedy_improve(hb, deadline)
        c_bin = _compute_c5(hb)
        if c_bin < best_c:
            best_h[:] = hb
            best_c = c_bin
            best_n[0] = n

    return best_h, best_c, best_n[0]


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with an improved C5 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    # base size
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
    else:
        parent = None
        n0 = 360

    # allocate mutable best containers
    best_h = np.zeros(n0, dtype=float)
    best_c = np.inf
    best_n = [n0]

    # optimise at base resolution
    _run_opt_on_n(
        n0,
        init_h=parent if parent is not None else None,
        best_h=best_h,
        best_c=best_c,
        best_n=best_n,
        deadline=deadline,
    )
    # possible up‑sampling
    if time.time() < deadline and best_n[0] * 2 <= 720:
        n_up = best_n[0] * 2
        xs = np.linspace(0.0, 2.0, best_n[0], endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = _project_to_sum_box(h_up, n_up / 2.0)
        best_h = np.zeros(n_up, dtype=float)
        best_c = np.inf
        best_n = [n_up]
        _run_opt_on_n(
            n_up,
            init_h=h_up,
            best_h=best_h,
            best_c=best_c,
            best_n=best_n,
            deadline=deadline,
        )

    # fallback in the unlikely case nothing was found
    if best_h.size == 0:
        n = n0
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_h = _project_to_sum_box(best_h, n / 2.0)
        best_c = _compute_c5(best_h)
        best_n[0] = n

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum_box(best_h, best_n[0] / 2.0)
    final_c = _compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n[0])
