def binary_targeted_improve(h, rng, deadline):
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        # compute neighbor sum with linear boundary
        h_plus = np.concatenate((h[shift:], np.zeros(shift)))  # zeros at end
        h_minus = np.concatenate((np.zeros(shift), h[:n-shift]))
        neigh = h_plus + h_minus
        # find indices of ones and zeros
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        # among ones, choose those with minimal neigh value
        neigh_one = neigh[ones]
        min_val = neigh_one.min()
        cand_i = ones[neigh_one == min_val]
        i = rng.choice(cand_i)
        # among zeros, choose those with maximal neigh value
        neigh_zero = neigh[zeros]
        max_val = neigh_zero.max()
        cand_j = zeros[neigh_zero == max_val]
        j = rng.choice(cand_j)
        # compute delta improvement estimate
        # delta = neigh[i] - neigh[j] (if negative improves)
        delta = neigh[i] - neigh[j]
        if delta < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        # try random swaps
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            d = neigh[i2] - neigh[j2]
            if d < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discretised step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def binary_targeted_improve(h: np.ndarray, rng: np.random.Generator, deadline: float) -> np.ndarray:
