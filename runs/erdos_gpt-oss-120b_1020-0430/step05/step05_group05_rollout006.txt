def slsqp_optimize_sym(x0, beta, rng, max_iter=800):
    n_half = x0.size
    n = n_half*2
    bound = [(0.0,1.0)]*n_half
    # objective closure:
    def obj(x):
        # x is half vector
        h = np.concatenate([x, 1 - x[::-1]])
        # correlation via FFT
        L = 2*n
        H = np.fft.rfft(h, n=L)
        G = np.fft.rfft(1 - h, n=L)
        corr = np.fft.irfft(H * np.conj(G), n=L)[:2*n-1]
        bc = beta * corr
        m = bc.max()
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    # SLSQP
    res = minimize(obj, x0, method='SLSQP', bounds=bound,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # No sum correction needed
    return x
