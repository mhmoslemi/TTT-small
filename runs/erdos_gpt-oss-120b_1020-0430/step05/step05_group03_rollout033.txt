def upscale_binary(x, new_n):
    old_n = len(x)
    factor = new_n / old_n
    if abs(round(factor) - factor) < 1e-9:
        r = int(round(factor))
        return np.repeat(x, r)
    else:
        # Use linear interpolation to get floats, then convert to binary with same sum
        # Interpolate as float
        t_old = np.linspace(0, 2, old_n, endpoint=False)
        t_new = np.linspace(0, 2, new_n, endpoint=False)
        x_float = np.interp(t_new, t_old, x.astype(float))
        # Convert to binary by selecting top new_n/2 values
        thresh = np.partition(x_float, -new_n//2)[-new_n//2]
        # Use >= thresh but may get more than needed; adjust:
        binary = (x_float > thresh).astype(np.int8)
        # adjust to exact count
        while binary.sum() > new_n//2:
            # turn off smallest true
            idx = np.where(binary == 1)[0]
            # among them find one with smallest x_float
            i = idx[np.argmin(x_float[idx])]
            binary[i] = 0
        while binary.sum() < new_n//2:
            idx = np.where(binary == 0)[0]
            i = idx[np.argmax(x_float[idx])]
            binary[i] = 1
        return binary

import time
import numpy as np

def compute_autocorr(x: np.ndarray) -> np.ndarray:
    """Circular autocorrelation of a binary vector using FFT."""
    n = x.shape[0]
    fx = np.fft.rfft(x, n=n)
    corr = np.fft.irfft(fx * np.conj(fx), n=n)
    # Round to nearest integer to avoid floating errors
    return np.rint(corr).astype(np.int32)


def upscale_binary(x: np.ndarray, new_n: int) -> np.ndarray:
    """Upscale a binary vector to a larger size while keeping exactly half ones."""
    old_n = x.shape[0]
    if new_n == old_n:
        return x.copy()
    factor = new_n / old_n
    if abs(round(factor) - factor) < 1e-12:  # integer factor
        r = int(round(factor))
        return np.repeat(x, r)
    # otherwise linear interpolation followed by a projection to binary
    t_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    t_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    xf = np.interp(t_new, t_old, x.astype(float))
    # keep exactly new_n//2 ones (new_n is even in our usage)
    target = new_n // 2
    # indices of the largest values become ones
    idx = np.argpartition(xf, -target)[-target:]
    bin_vec = np.zeros(new_n, dtype=np.int8)
    bin_vec[idx] = 1
    # adjust if ties cause count mismatch
    cur = bin_vec.sum()
    while cur > target:
        i = np.where(bin_vec == 1)[0][0]
        bin_vec[i] = 0
        cur -= 1
    while cur < target:
        i = np.where(bin_vec == 0)[0][0]
        bin_vec[i] = 1
        cur += 1
    return bin_vec


def anneal_binary(
    n: int,
    rng: np.random.Generator,
    max_seconds: float,
    init_x: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """Binary simulated annealing with greedy worst‑shift moves."""
    start = time.time()
    target = n // 2
    # initialise binary vector
    if init_x is None:
        x = np.zeros(n, dtype=np.int8)
        x[rng.choice(n, target, replace=False)] = 1
    else:
        x = np.array(init_x, dtype=np.int8)
        # enforce exact half‑ones
        cur = x.sum()
        if cur > target:
            idx = np.where(x == 1)[0]
            x[rng.choice(idx, cur - target, replace=False)] = 0
        elif cur < target:
            idx = np.where(x == 0)[0]
            x[rng.choice(idx, target - cur, replace=False)] = 1

    S = compute_autocorr(x)
    min_S = int(S.min())
    best_x = x.copy()
    best_c = (target - min_S) * (2.0 / n)

    ks = np.arange(n, dtype=np.int32)
    T = target * 1.0
    decay = 0.9993
    greedy_fails = 0

    while time.time() - start < max_seconds * 0.99:
        # ---- greedy improvement on the worst shift ----
        k_w = int(np.argmin(S))
        x_shift = np.roll(x, -k_w)
        bad = np.where((x == 1) & (x_shift == 0))[0]
        good = np.where((x == 0) & (x_shift == 1))[0]
        if bad.size and good.size:
            i = int(rng.choice(bad))
            j = int(rng.choice(good))
            # delta for autocorrelation
            a_i = x.take((i + ks) % n)
            b_i = x.take((i - ks) % n)
            a_j = x.take((j + ks) % n)
            b_j = x.take((j - ks) % n)
            delta = - (a_i + b_i) + (a_j + b_j)
            delta[0] += 2
            k_ij = (i - j) % n
            k_ji = (j - i) % n
            delta[k_ij] -= 1
            delta[k_ji] -= 1
            S = S + delta
            # perform the swap
            x[i] = 0
            x[j] = 1
            min_S = int(S.min())
            c = (target - min_S) * (2.0 / n)
            if c < best_c:
                best_c = c
                best_x = x.copy()
            greedy_fails = 0
            continue
        else:
            greedy_fails += 1

        # ---- random swap with simulated‑annealing acceptance ----
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        a_i = x.take((i + ks) % n)
        b_i = x.take((i - ks) % n)
        a_j = x.take((j + ks) % n)
        b_j = x.take((j - ks) % n)
        delta = - (a_i + b_i) + (a_j + b_j)
        delta[0] += 2
        k_ij = (i - j) % n
        k_ji = (j - i) % n
        delta[k_ij] -= 1
        delta[k_ji] -= 1
        S_new = S + delta
        min_S_new = int(S_new.min())
        accept = (min_S_new >= min_S) or (
            rng.random() < np.exp((min_S_new - min_S) / (T + 1e-12))
        )
        if accept:
            x[i] = 0
            x[j] = 1
            S = S_new
            min_S = min_S_new
            c = (target - min_S) * (2.0 / n)
            if c < best_c:
                best_c = c
                best_x = x.copy()
        T *= decay
        # occasional temperature reset if stuck
        if greedy_fails > 2000:
            T = target * 1.0
            greedy_fails = 0

    return best_x, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # Determine available initial binary vector (if any)
    init_bin_512 = None
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        if n0 % 2 == 0:
            target0 = n0 // 2
            idx_top = np.argpartition(h0, -target0)[-target0:]
            bin0 = np.zeros(n0, dtype=np.int8)
            bin0[idx_top] = 1
            init_bin_512 = bin0

    # sequence of resolutions to explore
    resolutions = [128, 256, 384, 512]
    best_overall_x = None
    best_overall_c = float("inf")
    best_overall_n = None

    for idx, n in enumerate(resolutions):
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining <= 5.0:
            break
        # allocate equal share of remaining time for each unresolved resolution
        time_for_this = remaining / (len(resolutions) - idx)

        # initialise starting vector
        if best_overall_x is not None:
            init = upscale_binary(best_overall_x, n)
        elif n == 512 and init_bin_512 is not None:
            init = init_bin_512.copy()
        else:
            init = None

        x_opt, c_opt = anneal_binary(n, rng, time_for_this, init_x=init)

        if c_opt < best_overall_c:
            best_overall_c = c_opt
            best_overall_x = x_opt.copy()
            best_overall_n = n

    # final fine‑tuning (short random walk) on the best resolution found
    if best_overall_x is not None:
        extra = budget_s - (time.time() - start_total)
        if extra > 2.0:
            x_refined, c_refined = anneal_binary(
                best_overall_n, rng, extra, init_x=best_overall_x
            )
            if c_refined < best_overall_c:
                best_overall_c = c_refined
                best_overall_x = x_refined.copy()

    # ensure validity
    best_overall_x = np.clip(best_overall_x, 0, 1).astype(float)
    final_c = (best_overall_n // 2 - int(compute_autocorr(best_overall_x.astype(np.int8)).min())) * (2.0 / best_overall_n)

    return best_overall_x.tolist(), float(final_c), int(best_overall_n)
