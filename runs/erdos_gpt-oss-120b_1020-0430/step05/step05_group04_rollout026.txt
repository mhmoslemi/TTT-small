   n = len(h)
   A = h
   B = 1 - h
   corr = np.convolve(A.astype(np.int32), B.astype(np.int32), mode='full')
   best_max = corr.max()
   best_h = A.copy()
   it = 0
   while time.time() < time_deadline and it < max_total_iter:
        # compute worst shift k (index)
        idx = corr.argmax()
        k = idx - (n - 1)
        # compute mismatched sets S1, S0 for shift k
        # S1: i where A[i]==1, and i+k in range and B[i+k]==1 (i.e., A[i+k]==0)
        # Using masks:
        i_range = np.arange(n)
        if k >= 0:
            cond_i = (A == 1) & (i_range + k < n) & (A[i_range + k] == 0)
            cond_j = (A == 0) & (i_range + k < n) & (A[i_range + k] == 1)
        else:
            kk = -k
            cond_i = (A == 1) & (i_range - kk >= 0) & (A[i_range - kk] == 0)
            cond_j = (A == 0) & (i_range - kk >= 0) & (A[i_range - kk] == 1)
        S1 = np.flatnonzero(cond_i)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            # no mismatched for this shift; cannot improve using this shift, maybe random
            # choose random swap
            i = np.where(A == 1)[0][rng.integers(A.sum())]
            j = np.where(A == 0)[0][rng.integers(n - A.sum())]
        else:
            # sample candidate pairs
            pairs = []
            total_pairs = S1.size * S0.size
            if total_pairs <= sample_per_iter:
                # enumerate all
                for i in S1:
                    for j in S0:
                        pairs.append((i, j))
            else:
                # random sampling
                for _ in range(sample_per_iter):

                    j = rng.choice(S0)
                    pairs.append((i, j))
        # evaluate each pair
        best_pair = None

        # prefetch A and B to local variables for speed

            delta[n-1] -= 2

            new_max = (corr_arr + delta).max()

                best_pair = (i, j)
        if best_pair is not None:

            # perform swap
            A[i] = 0
            A[j] = 1
            B[i] = 1
            B[j] = 0
            # update corr
            # Then we can recompute corr = corr + delta for that best swap (which we already computed).
            # But need to recompute delta for best_pair again or store delta
            # Compute delta for best_pair and update corr.
            delta_best = np.zeros_like(corr_arr)
            delta_best[i:i+n] -= B_arr

            best_h = A.copy()
        else:

            # Could try random swaps to escape local minima
            # We'll break after some attempts without improvement
            break
        it += 1
   return best_h, best_max * (2.0 / n)

def anneal_binary(h, time_deadline, start_temp_factor=0.01, cooling=0.999):
    n = len(h)
    A = h
    B = 1 - A
    corr = np.convolve(A.astype(np.int32), B.astype(np.int32), mode='full')
    cur_max = corr.max()
    # initial temperature proportional to cur_max
    T = max(start_temp_factor * cur_max, 1e-6)
    # Precompute list of indices for ones and zeros
    while time.time() < time_deadline:
        # choose random i in ones, j in zeros
        ones = np.flatnonzero(A)
        zeros = np.flatnonzero(A == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta
        delta = np.zeros_like(corr)
        delta[i:i+n] -= B
        delta[j:j+n] += B
        delta[i:i+n] += A
        delta[j:j+n] -= A
        delta[n-1] -= 2
        delta[(j - i) + (n-1)] += 1
        delta[(i - j) + (n-1)] += 1
        new_max = (corr + delta).max()
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            # accept
            A[i] = 0
            A[j] = 1
            B[i] = 1
            B[j] = 0
            corr = corr + delta
            cur_max = new_max
        T *= cooling
    return A.copy(), cur_max

def targeted_anneal(h, time_deadline, start_temp_factor=0.01, cooling=0.999):
    n = len(h)
    A = h
    B = 1 - A
    corr = np.convolve(A.astype(np.int32), B.astype(np.int32), mode='full')
    cur_max = corr.max()
    T = max(start_temp_factor * cur_max, 1e-6)
    while time.time() < time_deadline:
        idx = corr.argmax()
        k = idx - (n - 1)
        # compute mismatched sets S1, S0 as before
        i_range = np.arange(n)
        if k >= 0:
            cond_i = (A == 1) & (i_range + k < n) & (A[i_range + k] == 0)
            cond_j = (A == 0) & (i_range + k < n) & (A[i_range + k] == 1)
        else:
            kk = -k
            cond_i = (A == 1) & (i_range - kk >= 0) & (A[i_range - kk] == 0)
            cond_j = (A == 0) & (i_range - kk >= 0) & (A[i_range - kk] == 1)
        S1 = np.flatnonzero(cond_i)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            # fallback random swap
            i = rng.choice(np.flatnonzero(A == 1))
            j = rng.choice(np.flatnonzero(A == 0))
        else:
            i = rng.choice(S1)
            j = rng.choice(S0)
        # compute delta
        delta = np.zeros_like(corr)
        delta[i:i+n] -= B
        delta[j:j+n] += B
        delta[i:i+n] += A
        delta[j:j+n] -= A
        delta[n-1] -= 2
        delta[(j - i) + (n-1)] += 1
        delta[(i - j) + (n-1)] += 1
        new_max = (corr + delta).max()
        # acceptance
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            A[i] = 0
            A[j] = 1
            B[i] = 1
            B[j] = 0
            corr = corr + delta
            cur_max = new_max
        T *= cooling
    return A.copy(), cur_max

def targeted_anneal(h0, time_deadline, start_temp_factor=0.02, cooling=0.9995):
    n = h0.size
    A = h0.copy()
    B = 1 - A
    corr = np.convolve(A.astype(np.int32), B.astype(np.int32), mode='full')
    cur_max = int(corr.max())
    T = max(start_temp_factor * cur_max, 1e-6)
    i_range = np.arange(n)
    while time.time() < time_deadline:
        idx = corr.argmax()
        k = idx - (n - 1)
        # compute mismatched sets
        if k >= 0:
            cond_i = (A == 1) & (i_range + k < n) & (A[i_range + k] == 0)
            cond_j = (A == 0) & (i_range + k < n) & (A[i_range + k] == 1)
        else:
            kk = -k
            cond_i = (A == 1) & (i_range - kk >= 0) & (A[i_range - kk] == 0)
            cond_j = (A == 0) & (i_range - kk >= 0) & (A[i_range - kk] == 1)
        S1 = np.flatnonzero(cond_i)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            # fallback random swap
            i = rng.choice(np.flatnonzero(A == 1))
            j = rng.choice(np.flatnonzero(A == 0))
        else:
            i = rng.choice(S1)
            j = rng.choice(S0)
        # compute delta
        delta = np.zeros_like(corr)
        # conv(delta_A, B)
        delta[i:i+n] -= B
        delta[j:j+n] += B
        # conv(A, delta_B)
        delta[i:i+n] += A
        delta[j:j+n] -= A
        # conv(delta_A, delta_B)
        delta[2*i] -= 1
        delta[2*j] -= 1
        delta[i + j] += 2
        new_max = int((corr + delta).max())
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            # accept
            A[i] = 0
            A[j] = 1
            B[i] = 1
            B[j] = 0
            corr = corr + delta
            cur_max = new_max
        T *= cooling
    return A, cur_max

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99
    n = 720

    # utility to compute C5
    def c5_val(max_corr):
        return max_corr * (2.0 / n)

    best_h = None
    best_max = None
    best_c = np.inf

    def update_best(h, max_corr):
        nonlocal best_h, best_max, best_c
        c = c5_val(max_corr)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_max = max_corr

    # generate candidate initial vectors
    candidates = []

    # parent if exists
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size == n:
            # binary from ranking
            idx = np.argpartition(parent, -n//2)[-n//2:]
            hparent = np.zeros(n, dtype=np.int8)
            hparent[idx] = 1
            candidates.append(hparent)
        else:
            # interpolate
            xs = np.linspace(0, 2, parent.size, endpoint=False)
            xt = np.linspace(0, 2, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            idx = np.argpartition(interp, -n//2)[-n//2:]
            hinterp = np.zeros(n, dtype=np.int8)
            hinterp[idx] = 1
            candidates.append(hinterp)
    # add some deterministic patterns e.g., clustered first half
    h_cluster = np.zeros(n, dtype=np.int8)
    h_cluster[:n//2] = 1
    candidates.append(h_cluster)

    # main loop over candidates and random restarts
    while time.time() < deadline:
        # Determine how much time left
        remaining = deadline - time.time()
        # if very little time, break
        if remaining < 0.5:
            break
        # pick a candidate: if any leftover in candidates list, pop; else random
        if candidates:
            h0 = candidates.pop()
        else:
            h0 = np.zeros(n, dtype=np.int8)
            h0[rng.choice(n, n//2, replace=False)] = 1
        # allocate a quota per run, e.g., 0.3 sec or half of remaining
        run_time = min(0.3, remaining * 0.5)
        h_opt, max_corr = targeted_anneal(h0, time.time() + run_time,
                                          start_temp_factor=0.02,
                                          cooling=0.9995)
        update_best(h_opt, max_corr)
        # Also add a short perturbation of best for another run
        if best_h is not None and remaining > 1.0:
            # make a copy and swap a few random pairs to diversify
            h_pert = best_h.copy()
            # swap a few random positions (e.g., 5 swaps)
            for _ in range(5):

        best_max = compute_c5_int(best_h)

import time
import numpy as np

def compute_c5_int(h: np.ndarray) -> int:
    """Maximum overlap for binary step function."""
    B = 1 - h
    corr = np.convolve(h.astype(np.int32), B.astype(np.int32), mode="full")
    return int(corr.max())

def targeted_anneal(h0: np.ndarray, deadline: float, rng: np.random.Generator,
                    start_temp_factor: float = 0.02, cooling: float = 0.9995):
    """Hill‑climb / simulated annealing focusing on the worst shift."""
    n = h0.size
    A = h0.copy()
    B = 1 - A
    corr = np.convolve(A.astype(np.int32), B.astype(np.int32), mode="full")
    cur_max = int(corr.max())
    T = max(start_temp_factor * cur_max, 1e-6)
    i_range = np.arange(n)
    while time.time() < deadline:
        idx = corr.argmax()
        k = idx - (n - 1)
        if k >= 0:
            cond_i = (A == 1) & (i_range + k < n) & (A[i_range + k] == 0)
            cond_j = (A == 0) & (i_range + k < n) & (A[i_range + k] == 1)
        else:
            kk = -k
            cond_i = (A == 1) & (i_range - kk >= 0) & (A[i_range - kk] == 0)
            cond_j = (A == 0) & (i_range - kk >= 0) & (A[i_range - kk] == 1)
        S1 = np.flatnonzero(cond_i)
        S0 = np.flatnonzero(cond_j)
        if S1.size and S0.size:
            i = rng.choice(S1)
            j = rng.choice(S0)
        else:
            i = rng.choice(np.flatnonzero(A == 1))
            j = rng.choice(np.flatnonzero(A == 0))

        delta = np.zeros_like(corr)
        # conv(delta_A, B)
        delta[i : i + n] -= B
        delta[j : j + n] += B
        # conv(A, delta_B)
        delta[i : i + n] += A
        delta[j : j + n] -= A
        # conv(delta_A, delta_B)
        delta[2 * i] -= 1
        delta[2 * j] -= 1
        delta[i + j] += 2

        new_max = int((corr + delta).max())
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            A[i] = 0
            A[j] = 1
            B[i] = 1
            B[j] = 0
            corr = corr + delta
            cur_max = new_max
        T *= cooling
    return A, cur_max

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99
    n = 720  # even, provides ample degrees of freedom

    def c5_from_max(m):
        return m * (2.0 / n)

    best_h = None
    best_max = None
    best_c = np.inf

    def update_best(h, m):
        nonlocal best_h, best_max, best_c
        c = c5_from_max(m)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_max = m

    # initial candidates
    candidates = []

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size == n:
            idx = np.argpartition(parent, -n // 2)[-n // 2 :]
            hpar = np.zeros(n, dtype=np.int8)
            hpar[idx] = 1
            candidates.append(hpar)
        else:
            xs = np.linspace(0, 2, parent.size, endpoint=False)
            xt = np.linspace(0, 2, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            hinterp = np.zeros(n, dtype=np.int8)
            hinterp[idx] = 1
            candidates.append(hinterp)

    # a simple clustered start
    h_cluster = np.zeros(n, dtype=np.int8)
    h_cluster[: n // 2] = 1
    candidates.append(h_cluster)

    # main search loop
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 0.6:
            break

        # obtain a start vector
        if candidates:
            h0 = candidates.pop()
        else:
            h0 = np.zeros(n, dtype=np.int8)
            h0[rng.choice(n, n // 2, replace=False)] = 1

        # short anneal
        run_time = min(0.3, remaining * 0.5)
        h_opt, m_opt = targeted_anneal(
            h0,
            time.time() + run_time,
            rng,
            start_temp_factor=0.03,
            cooling=0.9995,
        )
        update_best(h_opt, m_opt)

        # perturb best and anneal again if time permits
        if best_h is not None and deadline - time.time() > 0.5:
            h_pert = best_h.copy()
            for _ in range(5):
                i = rng.choice(np.flatnonzero(h_pert == 1))
                j = rng.choice(np.flatnonzero(h_pert == 0))
                h_pert[i] = 0
                h_pert[j] = 1
            run_time2 = min(0.3, deadline - time.time())
            h_opt2, m_opt2 = targeted_anneal(
                h_pert,
                time.time() + run_time2,
                rng,
                start_temp_factor=0.015,
                cooling=0.9997,
            )
            update_best(h_opt2, m_opt2)

    # final prolonged anneal on the current best
    if best_h is not None and time.time() < deadline:
        h_final, m_final = targeted_anneal(
            best_h,
            deadline,
            rng,
            start_temp_factor=0.005,
            cooling=0.9999,
        )
        update_best(h_final, m_final)

    # fallback
    if best_h is None:
        best_h = np.zeros(n, dtype=np.int8)
        best_h[: n // 2] = 1
        best_max = compute_c5_int(best_h)
        best_c = c5_from_max(best_max)

    return best_h.astype(float).tolist(), float(best_c), int(n)
