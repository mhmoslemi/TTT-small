import time
import numpy as np

def compute_c5(h):
    """Return the overlap constant for a binary (or fractional) step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def random_one(h, rng):
    """Return a random index where h == 1."""
    n = h.shape[0]
    while True:
        i = rng.integers(0, n)
        if h[i] == 1:
            return i

def random_zero(h, rng):
    """Return a random index where h == 0."""
    n = h.shape[0]
    while True:
        i = rng.integers(0, n)
        if h[i] == 0:
            return i

def gradient_refine(h, rng, max_iter=30, step=0.05):
    """Sub‑gradient descent on the exact max‑overlap objective."""
    n = h.shape[0]
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]          # ∂/∂h_i for i < up
            g[k:] -= h[:up]               # ∂/∂h_i for i ≥ k
        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)

        # exact sum correction (few random adjustments)
        diff = target - h.sum()
        for _ in range(200):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ----- initialise binary sequence (n = 720) -----
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=np.int8)
        n = parent.size * 2
        h = np.repeat(parent, 2)                     # binary, exactly n/2 ones
    else:
        n = 720
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, n // 2, replace=False)] = 1

    # ----- pre‑compute correlation and reversed linear term -----
    corr = np.correlate(h, 1 - h, mode="full").astype(np.int32)   # length 2n‑1
    f_rev = (1 - 2 * h)[::-1].astype(np.int8)                    # length n
    size = corr.size

    best_corr = corr.max()
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- buffers for incremental updates -----
    delta = np.empty_like(corr)

    total_time = deadline - start
    T_start, T_end = 2.0, 0.01

    while time.time() < deadline:
        # ----- choose swap, biasing towards the current worst shift -----
        cur_max_idx = np.argmax(corr)
        k = abs(cur_max_idx - (n - 1))

        if k != 0 and rng.random() < 0.7:
            # indices where the worst shift sees two ones together
            overlap = np.where(
                (h[: n - k] == 1) & (h[k:] == 1)
            )[0]
            if overlap.size:
                i = int(rng.choice(overlap))
            else:
                i = random_one(h, rng)
        else:
            i = random_one(h, rng)

        j = random_zero(h, rng)          # j is guaranteed different from i

        # ----- compute delta correlation for this swap -----
        delta.fill(0)
        delta[i : i + n] -= f_rev
        delta[j : j + n] += f_rev

        delta[n - 1] -= 2
        lag = j - i
        idx1 = lag + n - 1
        idx2 = -lag + n - 1
        delta[idx1] += 1
        delta[idx2] += 1

        new_corr = corr + delta
        new_max = new_corr.max()
        cur_max = corr.max()
        dE = new_max - cur_max

        # ----- temperature schedule -----
        elapsed = time.time() - start
        T = T_end + (T_start - T_end) * max(0.0, 1.0 - elapsed / total_time)

        # ----- Metropolis acceptance -----
        accept = False
        if dE <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            # apply swap
            h[i] = 0
            h[j] = 1
            # update reversed linear term
            i_rev = n - 1 - i
            j_rev = n - 1 - j
            f_rev[i_rev] = -f_rev[i_rev]
            f_rev[j_rev] = -f_rev[j_rev]
            # commit new correlation
            corr = new_corr

            # track best solution
            if new_max < best_corr:

                best_h = h.copy()
                best_c = compute_c5(best_h)
                if best_c <= 0.3808:          # target reached, stop early

    if c_after < best_c:

    # final clean‑up (ensure feasibility)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)
