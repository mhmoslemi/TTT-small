import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0,n)
        max_add = (1.0 - h[i]) if diff > 0 else h[i]
        add = min(abs(diff), max_add) * (1 if diff > 0 else -1)
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n-1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n-k] -= h[k:]
        grad[k:] -= h[: n-k]
    return grad

def gradient_refine(h, rng, step=0.015, max_iter=5):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n-1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0*h
        else:
            g[: n-k] += 1.0 - h[k:]
            g[k:] -= h[: n-k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple fix
            for _ in range(200):
                i = rng.integers(0,n)
                max_add = (1.0 - h[i]) if diff > 0 else h[i]
                add = min(abs(diff), max_add) * (1 if diff > 0 else -1)
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def random_move(h, rng, scale=0.5):
    n = h.size
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i == j: return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0: return h
    eps = rng.random() * max_eps * scale
    h[i] -= eps
    h[j] += eps
    return h

def local_opt(h, rng, moves, grad_iter):
    h = h.copy()
    for _ in range(moves):
        h = random_move(h, rng, scale=0.5)
    h = gradient_refine(h, rng, step=0.015, max_iter=grad_iter)
    return h

def crossover(p1, p2, rng):
    n = p1.size
    mask = rng.random(n) < 0.5
    child = np.where(mask, p1, p2)
    child = enforce_sum(child, rng)
    return child

def downsample(h, N):
    n_old = h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, N, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # Prepare initial candidate if present
    init_h512 = None
    if 'initial_h_values' in globals():
        init_h512 = np.array(initial_h_values, dtype=float)
        init_h512 = np.clip(init_h512, 0.0, 1.0)
        init_h512 = enforce_sum(init_h512, rng)

    for N in (256, 384, 512):
        if time.time() - start > budget_s * 0.95:
            break
        popsize = 12
        population = []
        # fill population
        for i in range(popsize):
            if N == 512 and init_h512 is not None and i == 0:
                h = init_h512.copy()
            elif N < 512 and init_h512 is not None and i == 0:
                h = downsample(init_h512, N)
                h = enforce_sum(h, rng)
            else:
                h = rng.random(N)
                h = enforce_sum(h, rng)
            population.append(h)

        # evaluate
        costs = [compute_c5(h) for h in population]
        # keep best per N
        idx_best = int(np.argmin(costs))
        best_n_h = population[idx_best]
        best_n_c = costs[idx_best]

        # evolution loop
        while time.time() - start < budget_s * 0.94:
            # parent selection
            p_idx = rng.integers(popsize)
            p = population[p_idx]
            # random offspring
            child = local_opt(p, rng, moves=rng.integers(10,30), grad_iter=3)
            child_c = compute_c5(child)
            # replace worst if better
            worst_idx = int(np.argmax(costs))
            if child_c < costs[worst_idx]:
                population[worst_idx] = child
                costs[worst_idx] = child_c
                if child_c < best_n_c:
                    best_n_c = child_c
                    best_n_h = child.copy()
            # occasional crossover
            if rng.random() < 0.15:
                i1, i2 = rng.choice(popsize, size=2, replace=False)
                child2 = crossover(population[i1], population[i2], rng)
                child2 = local_opt(child2, rng, moves=rng.integers(5,20), grad_iter=2)
                child2_c = compute_c5(child2)
                worst_idx = int(np.argmax(costs))
                if child2_c < costs[worst_idx]:
                    population[worst_idx] = child2
                    costs[worst_idx] = child2_c
                    if child2_c < best_n_c:
                        best_n_c = child2_c
                        best_n_h = child2.copy()
            # time check exit
            if time.time() - start > budget_s * 0.94:
                break

        # after evolution, compare with global best
        if best_n_c < best_c:
            best_c = best_n_c
            best_h = best_n_h.copy()
            best_n = N

    # final polish on global best
    if best_h is not None and time.time() - start < budget_s * 0.99:

        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        avail = 1.0 - h
        total_avail = avail.sum()
        if total_avail > 0:
            add = np.minimum(avail, diff * avail / total_avail)
            h += add
            diff = target - h.sum()
    else:
        avail = h
        total_avail = avail.sum()

            sub = np.minimum(avail, (-diff) * avail / total_avail)

        if max_eps <= 0: continue
        eps = min(abs(diff), max_eps) * (1 if diff > 0 else -1)

        diff = target - h.sum()

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap for step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project h onto the simplex {0≤h≤1, sum = n/2}."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        avail = 1.0 - h
        total = avail.sum()
        if total > 0:
            add = np.minimum(avail, diff * avail / total)
            h = h + add
            diff = target - h.sum()
    else:
        avail = h
        total = avail.sum()
        if total > 0:
            sub = np.minimum(avail, (-diff) * avail / total)
            h = h - sub
            diff = target - h.sum()
    # tiny residual fix
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        max_eps = (1.0 - h[i]) if diff > 0 else h[i]
        if max_eps <= 0.0:
            continue
        eps = min(abs(diff), max_eps) * (1.0 if diff > 0 else -1.0)
        h[i] += eps
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 5) -> np.ndarray:
    """Few steps of gradient descent with sum‑projection."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # project onto the simplex (quick correction)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple proportional correction
            if diff > 0:
                upd = np.minimum(1.0 - h, diff * (1.0 - h) / np.maximum(1e-12, (1.0 - h).sum()))
                h += upd
            else:
                upd = np.minimum(h, -diff * h / np.maximum(1e-12, h.sum()))
                h -= upd
            # residual fix
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h = enforce_sum(h, rng)
    return h


def random_move(h: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer a random amount from i to j while respecting bounds."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    h[i] -= eps
    h[j] += eps
    return h


def local_opt(h: np.ndarray, rng: np.random.Generator,
              moves: int, grad_iter: int) -> np.ndarray:
    """Random moves followed by a short gradient polishing."""
    h = h.copy()
    for _ in range(moves):
        h = random_move(h, rng, scale=0.7)
    h = gradient_refine(h, rng, step=0.015, max_iter=grad_iter)
    return h


def crossover(p1: np.ndarray, p2: np.ndarray,
              rng: np.random.Generator) -> np.ndarray:
    """Uniform crossover with post‑projection."""
    mask = rng.random(p1.size) < 0.5
    child = np.where(mask, p1, p2)
    child = enforce_sum(child, rng)
    return child


def downsample(h: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation from current resolution to n_new points."""
    n_old = h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # initial candidate at full resolution, if supplied
    init_h512 = None

    # search order: highest resolution first
    for N in (512, 384, 256):
        if time.time() - start > budget_s * 0.97:
            break
        popsize = 12

            elif N < 512 and init_h512 is not None and i == 0:

        costs = [compute_c5(h) for h in population]
        idx_best = int(np.argmin(costs))
        best_n_h = population[idx_best].copy()
        best_n_c = costs[idx_best]

        # evolutionary loop
        while time.time() - start < budget_s * 0.96:
            # parent selection
            p_idx = rng.integers(popsize)
            parent = population[p_idx]
            # mutation + short gradient polish
            moves = rng.integers(max(5, N // 30), max(10, N // 15) + 1)
            child = local_opt(parent, rng, moves=moves, grad_iter=3)
            child_c = compute_c5(child)

            # replace worst if improvement
            worst_idx = int(np.argmax(costs))
            if child_c < costs[worst_idx]:
                population[worst_idx] = child
                costs[worst_idx] = child_c
                if child_c < best_n_c:
                    best_n_c = child_c
                    best_n_h = child.copy()

            # occasional crossover
            if rng.random() < 0.15:
                i1, i2 = rng.choice(popsize, size=2, replace=False)
                ch = crossover(population[i1], population[i2], rng)
                ch = local_opt(ch, rng, moves=rng.integers(5, 15), grad_iter=2)

                    costs[worst_idx] = ch_c
                    if ch_c < best_n_c:

                        best_n_h = ch.copy()

            # quick time break

                break

        # keep global best
        if best_n_c < best_c:

            best_h = best_n_h.copy()
            best_n = N

            best_h = gradient_refine(best_h, rng, step=0.01, max_iter=4)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_h.size)
