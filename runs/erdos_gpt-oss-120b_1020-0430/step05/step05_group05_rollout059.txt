def get_candidate_pairs(h, k_shift, rng, max_trials=5):
    n = h.shape[0]
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    if k_shift >= 0:
        mask_i = ones + k_shift < n
        i_cand = ones[mask_i]
        neigh_i = h[i_cand + k_shift]
        bad_i = i_cand[neigh_i == 0]

        mask_j = zeros + k_shift < n
        j_cand = zeros[mask_j]
        neigh_j = h[j_cand + k_shift]
        good_j = j_cand[neigh_j == 1]
    else:
        k_abs = -k_shift
        mask_i = ones - k_abs >= 0
        i_cand = ones[mask_i]
        neigh_i = h[i_cand - k_abs]
        bad_i = i_cand[neigh_i == 0]

        mask_j = zeros - k_abs >= 0

    # sample up to max_trials pairs
    trials = []

        j = rng.choice(good_j)

import time
import numpy as np

def _correlation(h):
    # linear aperiodic correlation h * (1-h)
    return np.correlate(h.astype(np.int16), (1 - h).astype(np.int16), mode="full")

def _apply_swap(O, h, i, j):
    """
    Return correlation after swapping a 1 at i with a 0 at j.
    O  : current correlation (int16 array of length 2*n-1)
    h  : current binary vector (uint8, length n)
    i,j: indices (i has h[i]=1, j has h[j]=0)
    """
    n = h.shape[0]
    off = n - 1
    f = 1 - h                     # uint8, works with int16 arithmetic
    O2 = O.copy()
    # source i contribution
    O2[off - i : off + (n - i)] -= f
    # source j contribution
    O2[off - j : off + (n - j)] += f
    # extra terms for the bilinear part
    O2[off + (i - j)] += 1
    O2[off + (j - i)] += 1
    O2[off] -= 2                   # k = 0 term
    # target contributions using reversed h
    h_rev = h[::-1]
    O2[off + (i - n + 1) : off + i + 1] += h_rev
    O2[off + (j - n + 1) : off + j + 1] -= h_rev
    return O2

def _sym_swap_correlation(O, h, i, j, half):
    """
    Symmetric swap: swap i<->j (i,j in first half) and also their antipodes.
    Returns the new correlation without mutating h.
    """
    # first swap i,j
    O1 = _apply_swap(O, h, i, j)
    h_tmp = h.copy()
    h_tmp[i], h_tmp[j] = h_tmp[j], h_tmp[i]   # perform the first swap on a copy
    ii, jj = i + half, j + half
    O2 = _apply_swap(O1, h_tmp, ii, jj)       # second swap on the updated copy
    return O2
