    return np.max(corr) * (2.0 / h.size)

    b_corr = beta * corr
    maxb = np.max(b_corr)

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def project_onto_bounds_sum(y, target):
    x = np.clip(y, 0.0, 1.0)
    # Water-filling loop
    for _ in range(10):
        s = x.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(x > 0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            x[idx] -= delta
            # clip under zero
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.where(x < 1)[0]
            if idx.size == 0:
                break
            delta = -diff / idx.size
            x[idx] += delta
            x = np.clip(x, 0.0, 1.0)
    # final adjust any residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1)[0]
            if idx.size > 0:
                inc = min(diff, 1 - x[idx[0]])
                x[idx[0]] += inc
        else:
            idx = np.where(x > 0)[0]
            if idx.size > 0:
                dec = min(-diff, x[idx[0]])
                x[idx[0]] -= dec
    return x

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        if diff > 0:
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        attempts += 1
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator, max_iters: int = 1000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def project_onto_bounds_sum(y: np.ndarray, target: float) -> np.ndarray:
    x = np.clip(y, 0.0, 1.0)
    for _ in range(10):
        s = x.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(x > 0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            x[idx] -= delta
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.where(x < 1)[0]
            if idx.size == 0:
                break
            delta = -diff / idx.size
            x[idx] += delta
            x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1)[0]
            if idx.size:
                inc = min(diff, 1 - x[idx[0]])
                x[idx[0]] += inc
        else:
            idx = np.where(x > 0)[0]
            if idx.size:
                dec = min(-diff, x[idx[0]])
                x[idx[0]] -= dec
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.015, max_iter: int = 5) -> np.ndarray:
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] = -h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, rng)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr[len(x) - 1 :]
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    if abs(CURRENT_TARGET - x.sum()) > 1e-12:
        x = enforce_sum(x, rng)
    return x
