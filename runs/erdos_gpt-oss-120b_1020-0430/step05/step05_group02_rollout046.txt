import time

# ---------- helper functions ----------
def compute_c5(h):
    """max_k ∫ h(x)(1-h(x+k))dx approximated on a uniform grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass so that sum(h) == n/2 exactly."""
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- gradient refinement ----------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(xt, xs, parent)
    else:
        h0 = rng.random(n)
        h0 *= target_sum / h0.sum()
        adjust_sum_exact(h0, rng)

    best_h = None
    best_c = np.inf

    restart = 0
    # main multi‑restart loop
    while time.time() < deadline - 10 and restart < 12:
        restart += 1

        if restart == 1:
            # deterministic rounding of the interpolated parent
            idx = np.argpartition(h0, -int(target_sum))[-int(target_sum):]
            h_bin = np.zeros(n)
            h_bin[idx] = 1.0
        else:
            # fresh random binary vector
            h_bin = np.zeros(n)
            h_bin[rng.choice(n, int(target_sum), replace=False)] = 1.0

        # binary SA with a per‑restart time budget
        sub_dead = min(deadline, time.time() + min(30.0,
                        (deadline - time.time()) / max(1, 12 - restart)))
        h_bin, worst = binary_sa_guided(h_bin, rng, sub_dead)

        c5 = worst * (2.0 / n)
        if c5 < best_c:
            best_h = h_bin.astype(float)
            best_c = c5

        # gradient refinement on the current best (fractional)
        h_frac = best_h.copy()
        h_frac = gradient_refine(h_frac, rng, step=0.02, max_iter=12)
        c5 = compute_c5(h_frac)
        if c5 < best_c:
            best_h = h_frac.copy()
            best_c = c5

        # cascade of SLSQP refinements
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0,
                     1000.0, 2000.0, 5000.0, 10000.0, 20000.0):
            if time.time() > deadline:
                break
            try:
                h_frac = slsqp_optimize(best_h, beta, rng, max_iter=800)
            except Exception:
                continue
            c5 = compute_c5(h_frac)
            if c5 < best_c:
                best_h = h_frac.copy()
                best_c = c5

    # final safety adjustments
    if best_h is None:
        # fallback simple alternating pattern
        best_h = np.tile([1.0, 0.0], n // 2).astype(float)
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
