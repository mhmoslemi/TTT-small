            # update ones and zeros arrays
            # swap indices in ones and zeros

def binary_sa_initial(h, rng):

def binary_sa_main(h, rng, deadline):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5

    # Build lists and pos arrays
    ones = list(np.flatnonzero(cur))
    zeros = list(np.flatnonzero(1 - cur))
    pos_one = np.full(n, -1, dtype=int)
    pos_zero = np.full(n, -1, dtype=int)
    for idx, i in enumerate(ones):
        pos_one[i] = idx
    for idx, i in enumerate(zeros):
        pos_zero[i] = idx

    # temperature schedule
    T = cur_c5 * 0.5 + 1e-6
    decay = 0.99995
    # iteration counter for scheduling
    it = 0
    # maybe also track last improvement iteration
    last_improve = 0

    while time.time() < deadline:
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # apply swap
        cur[i] = 0.0
        cur[j] = 1.0
        # compute new C5
        new_c5 = compute_c5(cur)
        dv = new_c5 - cur_c5
        accept = False
        if dv < 0:
            accept = True
        else:
            if rng.random() < np.exp(-dv / max(T, 1e-12)):
                accept = True
        if accept:
            cur_c5 = new_c5
            # update lists/pos
            pos_i = pos_one[i]
            pos_j = pos_zero[j]
            # swap in lists
            ones[pos_i] = j
            zeros[pos_j] = i
            # update pos arrays
            pos_one[i] = -1
            pos_one[j] = pos_i
            pos_zero[j] = -1
            pos_zero[i] = pos_j

            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best = cur.copy()
                last_improve = it
        else:
            # revert
            cur[i] = 1.0
            cur[j] = 0.0
        # temperature cooling
        T *= decay
        it += 1
        # optional early restart if stuck
        if it - last_improve > 200000:
            # break to restart
            break

    return best, best_c5

def binary_sa(h0, rng, deadline):

    h = h0.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # Build lists and pos arrays
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    pos_one = np.full(n, -1, dtype=int)
    pos_zero = np.full(n, -1, dtype=int)
    for idx, i in enumerate(ones):
        pos_one[i] = idx
    for idx, i in enumerate(zeros):
        pos_zero[i] = idx

    T = cur_c * 0.5 + 1e-6
    decay = 0.99995
    it = 0
    last_improve = 0
    while time.time() < deadline:
        # choose random indices
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h[j] = 1.0

            accept = True
        else:
            if rng.random() < np.exp(-dv / max(T, 1e-12)):
                accept = True
        if accept:
            cur_c = new_c
            # update lists and positions
            pos_i = pos_one[i]
            pos_j = pos_zero[j]
            ones[pos_i] = j

            pos_one[i] = -1

            h[j] = 0.0

# global placeholders for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** _)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

                diff = target - h.sum()

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.97 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    candidate_ns = [360, 420, 480, 540, 600, 720]

    for n in candidate_ns:
        if time.time() >= deadline:
            break
        target = n // 2
        # up to 10 restarts per n, respecting time
        for restart in range(10):
            if time.time() >= deadline:
                break
            if restart == 0 and 'initial_h_values' in globals():
                parent = np.array(initial_h_values, dtype=float)
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                interp = np.interp(xt, xs, parent)
                idx = np.argpartition(interp, -target)[-target:]
                h0 = np.zeros(n)
                h0[idx] = 1.0
            elif restart == 1:
                step = rng.integers(2, n)
                start_idx = rng.integers(0, n)
                idx = [(start_idx + i * step) % n for i in range(target)]
                h0 = np.zeros(n)
                h0[idx] = 1.0
            else:
                h0 = np.zeros(n)
                h0[rng.choice(n, target, replace=False)] = 1.0

            # binary simulated annealing
            sub_dead = time.time() + min((deadline - time.time()) * 0.6, 30.0)
            h_bin, c_bin = binary_sa(h0.astype(float), rng, sub_dead)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin.copy(), c_bin, n
                if best_c < 0.3808:
                    break

            # local SLSQP refinement (fractional)
            h_f = h_bin.astype(float)
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
                if time.time() >= deadline:
                    break
                try:
                    h_f = slsqp_optimize(h_f, beta, rng, max_iter=300)
                except Exception:
                    continue
                adjust_sum_exact(h_f, rng)
                c_f = compute_c5(h_f)
                if c_f < best_c:
                    best_h, best_c, best_n = h_f.copy(), c_f, n
                    if best_c < 0.3808:
                        break

            # gradient sweep
            if time.time() < deadline:
                h_f = gradient_refine(h_f, rng, step=0.02, max_iter=6)
                c_f = compute_c5(h_f)
                if c_f < best_c:
                    best_h, best_c, best_n = h_f.copy(), c_f, n
                    if best_c < 0.3808:
                        break
        if best_c < 0.3808:
            break

    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        final_c = compute_c5(best_h)
        return best_h.tolist(), float(final_c), int(best_n)

    # fallback simple construction
    n = 360
    h = np.zeros(n)
    h[: n // 2] = 1.0
    final_c = compute_c5(h)
    return h.tolist(), float(final_c), int(n)
