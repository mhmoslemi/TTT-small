def gradient_refine(h, rng, step=0.03, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)  # shift index k (may be negative)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            # Since correlation uses zero padding, for shift k > 0:
            # compute contributions for overlapping region of length n-k
            # O(k) = sum_{i=0}^{n-k-1} h[i]*(1 - h[i+k])
            # gradient w.r.t h[i] for i in [0, n-k-1]: partial derivative = (1 - h[i+k])
            # gradient w.r.t h[i+k] for i in same range: partial derivative = -h[i]
            # Thus:
            g[:n-k] += 1.0 - h[k:]   # derivative from term (1 - h[i+k])
            g[k:] -= h[:n-k]          # derivative from term -h[i]
        # step size schedule
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # Adjust sum exactly to target
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution (as prior)
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
```

import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # ----- helpers -------------------------------------------------
    def compute_c5(h):
        n = h.size
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(corr) * (2.0 / n))

    def gradient_refine(h, rng, step=0.03, max_iter=12):
        n = h.size
        target = n / 2.0
        for it in range(max_iter):
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = np.argmax(corr) - (n - 1)
            k = abs(idx)
            g = np.zeros_like(h)
            if k == 0:
                g = 1.0 - 2.0 * h
            else:
                g[: n - k] += 1.0 - h[k:]
                g[k:] -= h[: n - k]
            step_sz = step * (0.5 ** it)
            h = h - step_sz * g
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                for _ in range(200):
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
        return h

    def fractional_sa(h, rng, deadline):
        cur = h.copy()
        cur_c = compute_c5(cur)
        best = cur.copy()
        best_c = cur_c
        it = 0
        while time.time() < deadline:
            i, j = rng.choice(len(cur), size=2, replace=False)
            max_transfer = min(cur[i], 1.0 - cur[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            cur[i] -= delta
            cur[j] += delta
            c = compute_c5(cur)
            T = max(1e-4, 0.01 * cur_c) * (0.9995 ** it) + 1e-12
            if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
                cur_c = c
                if c < best_c:
                    best_c, best = c, cur.copy()
            else:
                cur[i] += delta
                cur[j] -= delta
            it += 1
        return best, best_c

    # ----- MILP solver for binary h -------------------------------
    def solve_milp(n, time_limit):
        n2 = n / 2.0
        h = cp.Variable(n, boolean=True)
        tvar = cp.Variable()
        # offsets for w slices
        offsets = [0]
        for k in range(1, n + 1):
            offsets.append(offsets[-1] + (n - (k - 1)))
        w = cp.Variable(offsets[-1], boolean=True)
        cons = [cp.sum(h) == n2, tvar >= 0, tvar <= 1]
        for s in range(n):
            start = offsets[s]
            end = offsets[s + 1]
            if start == end:
                continue
            i_idx = np.arange(0, n - s)
            w_slice = w[start:end]
            # linking w = h_i * h_{i+s}
            cons.append(w_slice <= h[i_idx])
            cons.append(w_slice <= h[i_idx + s])
            cons.append(w_slice >= h[i_idx] + h[i_idx + s] - 1)
            # tail sum = sum_{i=n-s}^{n-1} h_i (empty if s=0)
            tail = cp.sum(h[n - s : n]) if s > 0 else 0
            # Corr_s <= t * n/2  => sum w + tail + n2*t >= n2
            cons.append(cp.sum(w_slice) + tail + n2 * tvar >= n2)
        prob = cp.Problem(cp.Minimize(tvar), cons)
        try:
            prob.solve(solver=cp.CBC, cbc_params={'maxSeconds': time_limit, 'logLevel': 0})
        except Exception:
            return None, None, None
        if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE):
            return None, None, None
        h_val = np.rint(h.value).astype(float)
        t_val = tvar.value
        return h_val, t_val, prob.status

    # ----- main search ---------------------------------------------
    best_h = None
    best_c = np.inf
    best_n = None

    # try a few even sizes, larger sizes give finer approximations
    for n in (150, 138, 130, 120, 110):
        if time.time() > deadline:
            break
        if n % 2:
            continue
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate at most 30 s for each MILP, but respect remaining budget
        milp_time = min(30.0, remaining * 0.6)
        h_bin, t_val, status = solve_milp(n, milp_time)
        if h_bin is None:
            continue
        c_val = compute_c5(h_bin)
        if c_val < best_c:
            best_h, best_c, best_n = h_bin.copy(), c_val, n

        # optional fractional SA on the binary start
        if time.time() < deadline:
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
            h_frac, c_frac = fractional_sa(h_bin.copy(), rng, sub_dead)
            if c_frac < best_c:
                best_h, best_c, best_n = h_frac.copy(), c_frac, n

        # final gradient refinement
        if best_h is not None and time.time() < deadline:
            h_ref = gradient_refine(best_h.copy(), rng, step=0.02, max_iter=15)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c, best_n = h_ref, c_ref, n

    # ----- fallback simple construction --------------------------------
    if best_h is None:
        n = 120
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)

    # ----- final clean‑up --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for i in range(best_n):
            add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
            best_h[i] += add
            diff = target_sum - best_h.sum()
            if abs(diff) < 1e-12:
                break
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
