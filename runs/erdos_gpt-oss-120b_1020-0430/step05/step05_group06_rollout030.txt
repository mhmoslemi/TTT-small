def simulated_annealing(h, best_c, rng, deadline):
    n = h.size
    cur_h = h.copy()
    cur_c = best_c
    best_h = cur_h.copy()
    best_c = cur_c
    # parameters
    init_temp = 0.01 # maybe 0.005
    final_temp = 1e-5
    max_iter = 200000 # will break earlier if time out
    total_time = deadline - time.time()
    # we assume time for each iteration ~ small
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # temperature schedule linear or exponential
        t = max(final_temp, init_temp * ((deadline - time.time()) / total_time))
        # choose i, j distinct
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # decide direction of delta: transfer from i to j
        # compute max allowable delta
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            # try opposite direction
            max_delta = min(cur_h[j], 1.0 - cur_h[i])
            if max_delta <= 0:
                continue
            # swap i and j meaning transfer from j to i
            i, j = j, i
        # sample delta, maybe random fraction of max_delta: 
        # delta = rng.random() * max_delta   # uniform
        delta = max_delta * rng.random()
        # ensure delta > small threshold
        if delta < 1e-8:
            continue
        # propose new vector by copy
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # compute new C5
        new_c = compute_c5(new_h)
        # acceptance
        if new_c < cur_c:
            accept = True
        else:
            # probability
            accept = rng.random() < np.exp(-(new_c - cur_c) / t)
        if accept:
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c

def simulated_annealing(h, c, rng, deadline):
    n = h.size
    factor = 2.0 / n
    cur_h = h.copy()
    cur_corr = np.correlate(cur_h, 1 - cur_h, mode="full")
    cur_c = cur_corr.max() * factor
    best_h = cur_h.copy()
    best_c = cur_c
    total_time = deadline - time.time()
    init_temp = 0.001
    final_temp = 1e-7
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute temp
        remain = deadline - now
        temp = final_temp + (init_temp - final_temp) * (remain / total_time)
        # choose i
        if rng.random() < 0.5:
            # target
            lag = int(np.argmax(cur_corr) - (n - 1))
            # get indices and contributions
            if lag >= 0:
                idx = np.arange(n - lag)
                contrib = cur_h[idx] * (1 - cur_h[idx + lag])
            else:
                k = -lag
                idx = np.arange(k, n)
                contrib = cur_h[idx] * (1 - cur_h[idx - k])
            i = idx[np.argmax(contrib)]
        else:
            i = rng.integers(0, n)
        # pick j != i
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # determine direction: try moving from i to j
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            # try opposite
            max_delta = min(cur_h[j], 1.0 - cur_h[i])
            if max_delta <= 0:
                continue
            # swap i and j roles
            i, j = j, i
        delta = max_delta * rng.random()**0.5  # bias to larger
        # apply move
        cur_h[i] -= delta
        cur_h[j] += delta
        new_corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        new_c = new_corr.max() * factor
        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / temp):
            cur_corr = new_corr
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        c0 = compute_c5(h0)
        best_h = h0.copy()
        best_c = c0
        best_n = h0.size

    # candidate ns
    candidate_ns = [360, 720]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        # use greedy_binary to get binary solution
        h_bin, O = greedy_binary(n, rng)

        if time.time() < deadline:

def hill_climb(h_bin, max_O, n, rng, deadline):
    """Simple hill‑climb on binary vectors (swap 1↔0)."""
    best_h = h_bin.copy()
    best_max = max_O
    cur_h = h_bin.copy()
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        O_new = np.max(
            np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode="full")
        )
        if O_new < best_max:
            best_max = O_new
            best_h = cur_h.copy()
            # update index lists
            ones[i_idx] = j
            zeros[j_idx] = i
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_max

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a given discretisation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Enforce Σh = n/2 by moving mass between coordinates."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, scaled back
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 400):
    """Smooth‑max SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


# ---------- greedy binary construction ----------
def greedy_binary(n: int, rng: np.random.Generator):
    """Fast greedy construction of a binary (0/1) candidate."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)          # correlation of all‑zero vector
    for _ in range(L):
        O_adj = O + 1                                 # effect of adding a new 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h   # new correlation after setting p=1
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0                                 # enforce O_0 = 0
    return h.astype(float), O


# ---------- binary hill climbing ----------
def hill_climb(h_bin: np.ndarray, max_O: int, n: int, rng: np.random.Generator, deadline: float):
    """Swap‑based hill climbing on the binary vector."""
    best_h = h_bin.copy()
    best_max = max_O
    cur_h = h_bin.copy()
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        cur_h[i] = 0
        cur_h[j] = 1
        O_new = np.max(
            np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode="full")
        )
        if O_new < best_max:
            best_max = O_new
            best_h = cur_h.copy()
            ones[ones == i] = j
            zeros[zeros == j] = i
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_max


# ---------- simulated annealing ----------
def simulated_annealing(h: np.ndarray, rng: np.random.Generator, deadline: float):
    """In‑place SA that respects the sum constraint."""
    n = h.size
    factor = 2.0 / n
    cur_h = h.copy()
    cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
    cur_c = cur_corr.max() * factor
    best_h = cur_h.copy()
    best_c = cur_c

    total_time = max(deadline - time.time(), 1e-12)
    init_temp = 0.001
    final_temp = 1e-7

    while True:
        now = time.time()
        if now >= deadline:
            break
        remain = deadline - now
        temp = final_temp + (init_temp - final_temp) * (remain / total_time)

        # 50 % targeted move, 50 % random pair
        if rng.random() < 0.5:
            lag = int(np.argmax(cur_corr) - (n - 1))
            if lag >= 0:
                idx = np.arange(n - lag)
                contrib = cur_h[idx] * (1.0 - cur_h[idx + lag])
            else:
                k = -lag
                idx = np.arange(k, n)
                contrib = cur_h[idx] * (1.0 - cur_h[idx - k])
            i = idx[np.argmax(contrib)]
        else:
            i = rng.integers(0, n)

        # pick a different j
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # try moving mass from i→j
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            # opposite direction
            max_delta = min(cur_h[j], 1.0 - cur_h[i])
            if max_delta <= 0:
                continue
            i, j = j, i

        # bias towards larger steps (square‑root)
        delta = max_delta * rng.random() ** 0.5
        cur_h[i] -= delta
        cur_h[j] += delta

        new_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        new_c = new_corr.max() * factor

        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / temp):
            cur_corr = new_corr
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta

    return best_h, best_c


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    start = time.time()
    deadline = start + budget_s * 0.99
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = np.inf
    best_n = None

    # parent solution, if supplied

            if c_hc < best_c:
                best_h = h_hc.astype(float).copy()
                best_c = c_hc
                best_n = n
                best_max_O = max_O_hc

        if (
            time.time() < deadline
            and best_h is not None
            and best_n == n
        ):

        best_n = n
