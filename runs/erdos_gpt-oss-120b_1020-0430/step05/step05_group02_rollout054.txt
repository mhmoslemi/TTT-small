def gradient_descent(h, rng, n_iter=200, step0=0.5):
    n = h.size
    target = n/2
    for it in range(n_iter):
        # compute f_s for all s
        f = np.array([np.dot(h, np.roll(h, -s)) for s in range(n)])  # O(n^2)
        # find worst shift (minimum f)
        s = np.argmin(f)
        # gradient = h shifted forward + h shifted backward
        g = np.roll(h, -s) + np.roll(h, s)
        # step size schedule
        step = step0 * (0.95 ** it)
        h_new = h - step * g
        # clip
        h_new = np.clip(h_new, 0.0, 1.0)
        # adjust sum exactly
        diff = target - h_new.sum()
        # adjust by adding diff to random index(s)
        # use loop up to 2000 attempts
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
            h_new[i] += add
            diff = target - h_new.sum()
        h = h_new
    return h
```

                best_c, best_h, best_n = c, h.copy(), n

            # linear-programming refinement
            if time.time() < deadline:
                h_lp, _ = _lp_refine(h, max_iter=6, rng=rng)
                c_lp = _compute_c5(h_lp)
                if c_lp < best_c:
                    best_c, best_h, best_n = c_lp, h_lp.copy(), n
                h = h_lp

            # final gradient cleanup
            if time.time() < deadline:
                h = _gradient_descent(h, rng, n_iter=50, step0=0.2)
                c = _compute_c5(h)
                if c < best_c:
                    best_c, best_h, best_n = c, h.copy(), n

    if best_h is None:
        # fallback simple alternating pattern
        n = 360
        best_n = n
        best_h = np.tile([1.0, 0.0], n // 2).astype(float)
        best_h = _adjust_sum_exact(best_h, rng)
        best_c = _compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _adjust_sum_exact(best_h, rng)
    final_c = _compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
