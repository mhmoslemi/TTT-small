def enforce_sum(h, target=None):
    if target is None:
        target = h.size/2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        slack = 1 - h
        total_slack = slack.sum()
        if total_slack < 1e-12:
            # fallback random add
            i = np.argmax(slack)
            h[i] = min(1, h[i] + diff)
        else:
            h += diff * slack / total_slack
            # due to numeric may exceed 1 slightly; clip
            h = np.clip(h, 0, 1)
    else:
        excess = h
        total_excess = excess.sum()
        # but we need subtract -diff from h; find weights proportional to h
        h -= (-diff) * excess / total_excess
        h = np.clip(h, 0, 1)
    # after this, sum may be slightly off; if still off, adjust via small random.
    # compute new diff and if still > tolerance, recursively call (few times).
    new_diff = target - h.sum()
    if abs(new_diff) > 1e-9:
        # fix with simple loop limiting iterations
        for _ in range(10):
            if new_diff > 0:
                i = np.argmax(1 - h)
                add = min(new_diff, 1 - h[i])
                h[i] += add
            else:
                i = np.argmax(h)
                sub = min(-new_diff, h[i])
                h[i] -= sub
            new_diff = target - h.sum()
            if abs(new_diff) < 1e-12:
                break
    return h
```

import time, numpy as np

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with spacing dx=2/n."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple proportional redistribution, iterated a few times for safety
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            add = diff / mask.sum()
            h[mask] += add
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            sub = (-diff) / mask.sum()
            h[mask] -= sub
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h)


def worst_shift_index(h: np.ndarray) -> int:
    """Return the absolute shift (0..n-1) giving the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    return abs(idx)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    k = worst_shift_index(h)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

    else:
        h = random_feasible_vector(n, rng)

    best_h = h.copy()
    best_c = compute_c5(h)

    # ----- core optimisation loop -----
    while time.time() < deadline:

        k = worst_shift_index(h)
        grad = np.ones_like(h)
        if k == 0:
