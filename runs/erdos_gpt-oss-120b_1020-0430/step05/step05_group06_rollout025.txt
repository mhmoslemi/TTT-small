def project_onto_box_sum(x, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=100):
    # Project x onto [lo,hi] and sum target
    # First clip to [lo,hi] to get x_clipped,
    # then find lambda such that sum(min(hi, max(lo, x - lambda))) = target
    lam_low = x.min() - hi
    lam_high = x.max() - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        y = np.clip(x - lam, lo, hi)
        s = y.sum()
        if abs(s - target) < tol:
            return y
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(x - lam, lo, hi)
```

Check monotonic: As lambda increases, x - lambda decreases, after clipping, sum decreases, yes.

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def autocorr_full(h):
    n = h.size
    fft_h = np.fft.rfft(h, n=2 * n)
    ac = np.fft.irfft(fft_h * np.conj(fft_h), n=2 * n)
    return ac[: 2 * n - 1]

def c5_from_min_corr(min_corr, n):
    return 1.0 - (2.0 / n) * min_corr

def project_onto_box_sum(x, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=80):
    lo_lam = x.min() - hi
    hi_lam = x.max() - lo
    for _ in range(max_iter):
        lam = (lo_lam + hi_lam) * 0.5
        y = np.clip(x - lam, lo, hi)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            lo_lam = lam
        else:
            hi_lam = lam
    return np.clip(x - lam, lo, hi)

def ascend_min_corr(h, target, step):
    n = h.size
    ac = autocorr_full(h)
    k_min = int(np.argmin(ac) - (n - 1))
    d = np.zeros_like(h)
    if k_min >= 0:
        d[: n - k_min] += h[k_min:]
        d[k_min:] += h[: n - k_min]
    else:
        km = -k_min
        d[km:] += h[: n - km]
        d[: n - km] += h[km:]
    h_new = h + step * d
    h_proj = project_onto_box_sum(h_new, target)
    new_ac = autocorr_full(h_proj)
    return h_proj, new_ac.min()

                continue
            jitter = (rng.random(n) - 0.5) * step * 0.2
            h_j = project_onto_box_sum(h + jitter, target)
            min_corr_j = autocorr_full(h_j).min()
            c_j = c5_from_min_corr(min_corr_j, n)
            if c_j < best_c - 1e-12:
                best_c, best_h, h = c_j, h_j.copy(), h_j
                improved = True
        if not improved:
            step *= 0.5
    return best_h, best_c

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ----- greedy binary construction -----
def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w, best_ps = w, [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h, O

# ----- smooth‑max SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99
    best_h, best_c, best_n = None, np.inf, None
    ns = [360, 720]
    for n in ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        candidates = []
        if "initial_h_values" in globals():
            p = np.array(initial_h_values, dtype=float)
            if p.size == n:
                candidates.append(p.copy())
        h_bin, _ = greedy_binary(n, rng)
        candidates.append(h_bin.astype(float))
        for _ in range(2):
            rand = rng.random(n)
            candidates.append(project_onto_box_sum(rand, target))
        for h0 in candidates:
            if time.time() > deadline:
                break
            h0 = project_onto_box_sum(h0, target)
            refined_h, refined_c = refine_h(h0, target, rng, deadline)
            if refined_c < best_c:
                best_c, best_h, best_n = refined_c, refined_h.copy(), n
            if time.time() < deadline:
                try_h = refined_h.copy()
                adjust_sum_exact(try_h, rng)
                for beta in (100, 500, 2000, 5000):
                    if time.time() > deadline:
                        break
                    try:
                        try_h = slsqp_optimize(try_h, beta, rng, max_iter=300)
                    except Exception:
                        continue
                    c2 = compute_c5(try_h)
                    if c2 < best_c:
                        best_c, best_h, best_n = c2, try_h.copy(), n
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
