def grad_smooth_max(h, beta):
    n = h.size
    # compute r
    r = np.correlate(h, 1 - h, mode='full')
    # compute weights
    max_r = np.max(r)
    w = np.exp(beta * (r - max_r))
    w_sum = np.sum(w)
    w = w / w_sum
    # Compute gradient
    g = np.zeros_like(h)
    # shift positive
    for k in range(1, n):
        wk = w[n-1 + k]   # index for shift +k
        if wk != 0:
            # (1 - h[i+k]) contribution (i from 0..n-1-k)
            g[:n-k] += wk * (1 - h[k:])   # careful: (1 - h[i+k]) = (1 - h[i+k]), i ranges 0..n-1-k => h[i+k] = h[k:]
            # -h[i-k] contribution (i from k..n-1)
            g[k:] -= wk * h[:n-k]  # h[i-k] = h[i - k] = h[:n-k]
    # shift zero
    wk0 = w[n-1]
    if wk0 != 0:
        # dr0/dh_i = (1 - 2h_i)
        g += wk0 * (1 - 2*h)
    # shift negative
    for k in range(1, n):
        wk = w[n-1 - k]  # negative shift -k
        if wk != 0:
            # (1 - h[i - k]) contribution for i>=k
            g[k:] += wk * (1 - h[:n-k])
            # -h[i + k] contribution for i <= n-1 - k
            g[:n-k] -= wk * h[k:]
    return g

The projection onto [0,1] and sum constraint is akin to: find h' = argmin ||h' - (h - step*g)||^2 s.t. 0 <= h' <= 1, sum h' = n/2. This is a convex projection onto intersection of box and hyperplane.

Implementation:

def project_onto_simplex_sum(box_sum, lo=0.0, hi=1.0):
    # Input: vector v (maybe after gradient step). Want projection onto {x in [lo,hi]^{n} : sum(x) = box_sum}.
    # Algorithm: we can treat as projection onto hyperplane with box constraints using iterative adjustments (like Duchi et al.).
    # Approach: start with v, clip to [lo,hi]; compute excess = sum(v_clip) - box_sum.
    # If excess > 0: need to reduce sum, subtract equally from non-lo elements, but ensure they stay >= lo.
    # If excess < 0: need to increase sum, add equally to non-hi elements, but ensure <= hi.
    # Use while loop.
    v = np.copy(v)
    v = np.clip(v, lo, hi)
    total = v.sum()
    excess = total - box_sum
    if abs(excess) < 1e-12:
        return v
    # Determine free indices
    free = np.where((v > lo + 1e-12) & (v < hi - 1e-12))[0]
    # If no free, then the target sum is infeasible (should not happen as box_sum = n/2 within feasible range)
    # We'll perform iterative water-filling.
    while abs(excess) > 1e-12 and free.size > 0:
        delta = excess / free.size
        # adjust free elements
        v[free] -= delta
        # clip again
        v = np.clip(v, lo, hi)
        total = v.sum()
        excess = total - box_sum
        free = np.where((v > lo + 1e-12) & (v < hi - 1e-12))[0]
    # If still excess, adjust by distributing to all elements within bounds (might be at extremes)
    # Return v
    return v

    corr = np.correlate(h, 1-h, mode='full')
    k_idx = np.argmax(corr)
    # shift value = k_idx - (n-1). So k = shift.
    k = k_idx - (n - 1)
    # Compute gradient of S_k w.r.t h

def smooth_max_and_grad(h, beta):
    n = h.shape[0]
    r = np.correlate(h, 1.0 - h, mode='full')
    max_r = np.max(r)
    # stabilization
    scaled = beta * (r - max_r)
    exp_scaled = np.exp(scaled)
    sum_exp = np.sum(exp_scaled)
    smooth_val = max_r + np.log(sum_exp) / beta
    w = exp_scaled / sum_exp
    idx0 = n - 1
    w_pos = w[idx0:]  # length n
    # w_neg_shifted: length n, first element zero
    w_neg_shifted = np.empty(n, dtype=w.dtype)
    w_neg_shifted[0] = 0.0
    w_neg_shifted[1:] = w[:idx0]  # w[:n-1]
    # compute terms
    g_pos = np.correlate(w_pos, 1.0 - h, mode='full')[:n]
    conv_h = np.convolve(w_pos, h, mode='full')[:n]
    convA = np.convolve(w_neg_shifted, 1.0 - h, mode='full')[:n]
    corr_h_neg = np.correlate(w_neg_shifted, h, mode='full')[:n]
    g = g_pos - conv_h + convA - corr_h_neg
    return smooth_val, g

        if free.size > 0:

    return x

import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, max_iter=100):
    lo_lambda = np.min(v - hi) - 1.0
    hi_lambda = np.max(v - lo) + 1.0
    for _ in range(max_iter):
        lam = (lo_lambda + hi_lambda) * 0.5
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if s > target:
            lo_lambda = lam
        else:
            hi_lambda = lam
        if hi_lambda - lo_lambda < 1e-12:
            break
    lam = (lo_lambda + hi_lambda) * 0.5
    x = np.clip(v - lam, lo, hi)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = np.where((x > lo + 1e-12) & (x < hi - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
    return x

def smooth_max_and_grad(h, beta):
    n = h.shape[0]
    r = np.correlate(h, 1.0 - h, mode='full')
    max_r = np.max(r)
    scaled = beta * (r - max_r)
    exp_s = np.exp(scaled)
    sum_exp = np.sum(exp_s)
    smooth_val = max_r + np.log(sum_exp) / beta
    w = exp_s / sum_exp
    idx0 = n - 1
    w_pos = w[idx0:]  # length n
    w_neg_shifted = np.empty(n, dtype=w.dtype)
    w_neg_shifted[0] = 0.0
    w_neg_shifted[1:] = w[:idx0]  # n-1 elements
    # gradient components
    g_pos = np.correlate(w_pos, 1.0 - h, mode='full')[:n]
    conv_h = np.convolve(w_pos, h, mode='full')[:n]
    convA = np.convolve(w_neg_shifted, 1.0 - h, mode='full')[:n]
    corr_h_neg = np.correlate(w_neg_shifted, h, mode='full')[:n]
    g = g_pos - conv_h + convA - corr_h_neg
    return smooth_val, g

def adam_optimize(h0, beta, max_iter, lr):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, max_iter + 1):
        _, grad = smooth_max_and_grad(h, beta)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = h - step
        h = project_onto_box_sum(h, target)
    return h

def upsample_h(h_old, n_new):
    n_old = h_old.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
