import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# Helpers ----------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    o = corr[len(x) - 1 :]  # O_k for k>=0
    bc = CURRENT_BETA * o
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def greedy_binary_improve(h: np.ndarray, rng: np.random.Generator,
                         max_iters: int = 5000, target: float = 0.3808):
    """Swap‑based hill climb on binary vectors."""
    n = h.size
    best_c = compute_c5(h)
    for _ in range(max_iters):
        if best_c <= target:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        O = corr[n - 1 :]  # k>=0
        k = int(np.argmax(O))
        if k == 0:
            break
        # mismatched positions for this shift
        hi = h[: n - k]
        hip = h[k:]
        mis_one = np.where((hi == 1) & (hip == 0))[0]
        mis_zero = np.where((hi == 0) & (hip == 1))[0]
        if mis_one.size == 0 or mis_zero.size == 0:
            # cannot improve this shift
            # try a random swap to escape
            i, j = rng.choice(n, 2, replace=False)
            h[i], h[j] = h[j], h[i]
            new_c = compute_c5(h)
            if new_c < best_c:
                best_c = new_c
            else:
                h[i], h[j] = h[j], h[i]
            continue
        i = mis_one[rng.integers(mis_one.size)]
        j = mis_zero[rng.integers(mis_zero.size)]
        h[i], h[j] = h[j], h[i]  # perform the improving swap
        new_c = compute_c5(h)
        if new_c < best_c:
            best_c = new_c
        else:
            h[i], h[j] = h[j], h[i]  # revert
    return h, best_c

        h0 = np.zeros(n0, dtype=float)
        ones = rng.choice(n0, n0 // 2, replace=False)
        h0[ones] = 1.0

    # aim for power‑of‑two size >= 512
    n_target = 1 << (int(np.ceil(np.log2(n0))) )
    if n_target < 512:
        n_target = 512
    if n_target > 1024:      # stay modest for time limits
        n_target = 1024
    if n_target != n0:
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h = np.interp(x_new, x_old, h0)
    else:
        h = h0.copy()
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- continuous refinement with soft‑max SLSQP ----------
    betas = [5., 20., 80., 200., 800., 3000., 8000.]
    for beta in betas:
        if time.time() - start > total_budget * 0.80:
            break
        h = slsqp_optimize(h, beta, rng, max_iter=600)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
        if best_c <= TARGET:
            break

    # ---- binary rounding --------------------------------------
    n = best_h.size
    binary_h = np.zeros_like(best_h)
    # pick top n/2 entries
    idx = np.argpartition(best_h, -n // 2)[-n // 2 :]
    binary_h[idx] = 1.0
    binary_h = enforce_sum(binary_h, rng)   # safety
    binary_c = compute_c5(binary_h)
    if binary_c < best_c:
        best_h, best_c = binary_h.copy(), binary_c

    # ---- greedy binary improvement ---------------------------
    if best_h.ndim == 1 and np.all(np.logical_or(best_h == 0.0, best_h == 1.0)):
        time_left = total_budget - (time.time() - start)
        max_swaps = int(min(5000, time_left * 30))  # cheap estimate
        best_h, best_c = greedy_binary_improve(best_h, rng,
                                                max_iters=max_swaps,
                                                target=TARGET)

    # ---- final sanity check -----------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
