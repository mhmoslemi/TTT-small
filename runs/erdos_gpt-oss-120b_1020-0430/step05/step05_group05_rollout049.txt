def binary_omax(h):
    n = h.size
    prefix = np.cumsum(h)
    h_float = h.astype(float)
    rev = h_float[::-1]
    h_pad = np.concatenate([h_float, np.zeros(n)])
    rev_pad = np.concatenate([rev, np.zeros(n)])
    conv = np.fft.ifft(np.fft.fft(h_pad) * np.fft.fft(rev_pad)).real
    P = conv[n-1:n-1+n]  # length n
    s = np.arange(1, n)
    O_counts = prefix[n - s] - P[s]
    return O_counts.max()

import time
import numpy as np
from scipy.optimize import minimize

# global placeholders for SLSQP callbacks
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_correlation via log-sum-exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=1200):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def adjust_sum_exact(h, rng):
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=30):
    """Projected gradient descent against current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]       # term for i in [0, up)
            g[k:] -= h[:up]            # term for i in [k, n)
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(150):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def fractional_to_binary(h, rng):
    """Exact n/2‑ones binary vector – keep the largest entries."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    adjust_sum_exact(bin_h, rng)
    return bin_h

def binary_targeted_improve(h, rng, deadline):
    """Greedy swaps that lower the worst overlap (uses np.correlate)."""
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        neigh_one = neigh[ones]
        min_val = neigh_one.min()
        cand_i = ones[neigh_one == min_val]
        i = rng.choice(cand_i)

        neigh_zero = neigh[zeros]
        max_val = neigh_zero.max()
        cand_j = zeros[neigh_zero == max_val]
        j = rng.choice(cand_j)

        delta = neigh[i] - neigh[j]          # <0 improves worst shift
        if delta < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue

        # random attempts as backup
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            d = neigh[i2] - neigh[j2]
            if d < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.985  # safety margin

    # ---- base size ----
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
    else:
        parent = None
        n = 360

    best_h = None
    best_c = np.inf
    best_n = None

    # ---- generate start vectors ----
    starts = []

    if parent is not None:
        starts.append(parent.copy())
        starts.append(fractional_to_binary(parent, rng))

    # random binary seeds
    for _ in range(5):
        hb = np.zeros(n)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(hb)

    # random fractional seeds
    for _ in range(5):
        hf = rng.random(n)
        hf *= (n / 2) / hf.sum()
        adjust_sum_exact(hf, rng)
        starts.append(hf)

    # heavy‑tailed fractional seeds
    for _ in range(3):
        hf = rng.beta(0.5, 0.5, size=n)
        hf *= (n / 2) / hf.sum()
        adjust_sum_exact(hf, rng)
        starts.append(hf)

    # ---- main optimisation loop (n) ----
    betas_main = (5., 20., 80., 200., 500., 1000., 2000., 5000., 10000., 20000., 50000.)
    for base in starts:
        if time.time() > deadline:
            break
        h = np.clip(base, 0.0, 1.0)
        adjust_sum_exact(h, rng)

        # cascade of SLSQP with increasing beta
        for beta in betas_main:
            if time.time() > deadline:
                break
            max_iter = 1400 if beta < 200 else 2000
            h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # gradient polishing
        h = gradient_refine(h, rng, step=0.025, max_iter=30)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # binary greedy improvement
        hb = fractional_to_binary(h, rng)
        hb = binary_targeted_improve(hb, rng, deadline)
        c_b = compute_c5(hb)
        if c_b < best_c:
            best_h, best_c, best_n = hb.copy(), c_b, n

        # a final fractional gradient pass after binary step
        hb_frac = hb.astype(float)
        hb_frac = gradient_refine(hb_frac, rng, step=0.02, max_iter=20)
        c_hb_frac = compute_c5(hb_frac)
        if c_hb_frac < best_c:
            best_h, best_c, best_n = hb_frac.copy(), c_hb_frac, n

    # ---- optional up‑sampling refinement (n*2) ----
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 720
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        # SLSQP cascade on upsampled vector
        for beta in betas_main:
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=2500)

        # stronger gradient refinement
        h_up = gradient_refine(h_up, rng, step=0.025, max_iter=40)

        # binary improvement on upsampled version
        hb_up = fractional_to_binary(h_up, rng)
        hb_up = binary_targeted_improve(hb_up, rng, deadline)
        c_up = compute_c5(hb_up)
        if c_up < best_c:
            best_h, best_c, best_n = hb_up.copy(), c_up, n_up

        # also try a fractional gradient pass on upsampled after binary step
        hb_up_frac = hb_up.astype(float)
        hb_up_frac = gradient_refine(hb_up_frac, rng, step=0.02, max_iter=20)

        if c_up_frac < best_c:

    if time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        for _ in range(5):
            if time.time() > deadline:

            hb = np.zeros(n_up)
            hb[rng.choice(n_up, n_up // 2, replace=False)] = 1.0

                best_h, best_c, best_n = hb.copy(), c, n_up

    # ---- final clean‑up ----
    if best_h is None:
        # fallback: simple block of ones
        best_n = n
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
