def hill_climb_simulated_annealing(h_init, rng, deadline):
    cur_h = h_init.copy()
    O_cur = np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode='full')
    cur_max = int(O_cur.max())
    best_h = cur_h.copy()
    best_max = cur_max
    start = time.time()
    while time.time() < deadline:
        # temperature based on remaining fraction
        frac = (deadline - time.time()) / (deadline - start)
        T = max(0.001, frac * 5.0)  # linear cooling
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        # decide single or double swap
        if len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.2:
            i_sel = rng.choice(ones, size=2, replace=False)
            j_sel = rng.choice(zeros, size=2, replace=False)
            cur_h[i_sel] = 0
            cur_h[j_sel] = 1
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur_h[i] = 0
            cur_h[j] = 1
        O_new = np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode='full')
        new_max = int(O_new.max())
        delta = new_max - cur_max
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_max = new_max
            O_cur = O_new
            if cur_max < best_max:
                best_max = cur_max
                best_h = cur_h.copy()
        else:
            # revert swap
            if 'i_sel' in locals():
                cur_h[i_sel] = 1
                cur_h[j_sel] = 0
                del i_sel, j_sel
            else:
                cur_h[i] = 1
                cur_h[j] = 0
    return best_h, best_max

import numpy as np
from scipy.optimize import minimize
import time

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    for _ in range(L):
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode='full')
        O_adj = O + 1
        best_w = None
        best_ps = []
        h_int = h.astype(np.int32)
        zeros = np.flatnonzero(h == 0)
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start:start+n] - h_int
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
    return h

def hill_climb_simulated_annealing(h_init, rng, deadline):
    cur_h = h_init.copy()
    O_cur = np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode='full')
    cur_max = int(O_cur.max())
    best_h = cur_h.copy()
    best_max = cur_max
    start = time.time()
    while time.time() < deadline:
        # cooling
        frac = max(0.0, (deadline - time.time()) / (deadline - start))
        T = max(0.001, frac * 5.0)
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        do_double = len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.2
        if do_double:
            i_sel = rng.choice(ones, size=2, replace=False)
            j_sel = rng.choice(zeros, size=2, replace=False)
            cur_h[i_sel] = 0
            cur_h[j_sel] = 1
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur_h[i] = 0
            cur_h[j] = 1
        O_new = np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode='full')
        new_max = int(O_new.max())
        delta = new_max - cur_max
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_max = new_max
            O_cur = O_new
            if cur_max < best_max:
                best_max = cur_max
                best_h = cur_h.copy()
        else:
            # revert
            if do_double:
                cur_h[i_sel] = 1
                cur_h[j_sel] = 0
            else:
                cur_h[i] = 1
                cur_h[j] = 0
    return best_h, best_max

def run(seed=42, budget_s=1100, **kwargs):

            if c_parent < best_c:
                best_h = parent.copy()
                best_c = c_parent
                best_n = parent.size
    ns = [720, 360]
    for n in ns:
        if time.time() > deadline:
            break
        # start binary solution
        if best_n == n and best_h is not None and np.all((best_h == 0) | (best_h == 1)):
            h_bin = best_h.astype(np.int8)
        else:
            h_bin = greedy_binary(n, rng)
        O = np.correlate(h_bin.astype(np.int32), (1 - h_bin).astype(np.int32), mode='full')
        max_O = int(O.max())
        c_val = max_O * (2.0 / n)
        if c_val < best_c:
            best_c = c_val
            best_h = h_bin.astype(float).copy()
            best_n = n
            best_max_O = max_O
        # simulated annealing hill climbing
        if time.time() < deadline:
            h_sa, max_O_sa = hill_climb_simulated_annealing(h_bin.copy(), rng, deadline)
            c_sa = max_O_sa * (2.0 / n)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa.astype(float).copy()
                best_n = n
                best_max_O = max_O_sa
        # smooth-max SLSQP refinement
        if time.time() < deadline and best_h is not None and best_n == n:
            h_float = best_h.copy()
            adjust_sum_exact(h_float, rng)
            for beta in (2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0,

import numpy as np
from scipy.optimize import minimize
import time

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """C₅ for a sampled step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Exact projection onto Σh = n/2 respecting 0≤h≤1."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth‑max surrogate for max_k ∫h(x)(1−h(x+k))."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def greedy_binary(n, rng):
    """Greedy construction of a binary vector with |S|=n/2."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    for _ in range(L):
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O_adj = O + 1
        best_w = None
        best_ps = []
        h_int = h.astype(np.int32)
        zeros = np.flatnonzero(h == 0)
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h_int
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
    return h


def hill_climb_simulated_annealing(h_init, rng, deadline):
    """
    Simulated‑annealing hill climbing on binary vectors.
    Swaps a random 1↔0 (or a double swap) and accepts with a temperature schedule.
    """
    cur_h = h_init.copy()
    O_cur = np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode="full")
    cur_max = int(O_cur.max())
    best_h = cur_h.copy()
    best_max = cur_max
    start = time.time()
    while time.time() < deadline:
        frac = max(0.0, (deadline - time.time()) / (deadline - start))
        T = max(0.001, frac * 5.0)  # linear cooling
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        do_double = len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.2
        if do_double:
            i_sel = rng.choice(ones, size=2, replace=False)
            j_sel = rng.choice(zeros, size=2, replace=False)
            cur_h[i_sel] = 0
            cur_h[j_sel] = 1
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur_h[i] = 0
            cur_h[j] = 1
        O_new = np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode="full")
        new_max = int(O_new.max())
        delta = new_max - cur_max
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = cur_h.copy()
        else:
            if do_double:
                cur_h[i_sel] = 1
                cur_h[j_sel] = 0
            else:
                cur_h[i] = 1
                cur_h[j] = 0
    return best_h, best_max


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.96  # safety margin
    best_h = None
    best_c = np.inf
    best_n = None

    # parent solution, if supplied
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.ndim == 1:
            c_parent = compute_c5(parent)
            if c_parent < best_c:
                best_h = parent.copy()
                best_c = c_parent
                best_n = parent.size

    # try two discretisations, larger first
    for n in (720, 360):
        if time.time() > deadline:
            break

        # binary starting point
        if best_n == n and best_h is not None and np.all((best_h == 0) | (best_h == 1)):
            h_bin = best_h.astype(np.int8)
        else:
            h_bin = greedy_binary(n, rng)

        O = np.correlate(
            h_bin.astype(np.int32), (1 - h_bin).astype(np.int32), mode="full"
        )
        max_O = int(O.max())
        c_val = max_O * (2.0 / n)
        if c_val < best_c:
            best_c = c_val
            best_h = h_bin.astype(float).copy()
            best_n = n
            best_max_O = max_O

        # simulated‑annealing hill climbing
        if time.time() < deadline:
            h_sa, max_O_sa = hill_climb_simulated_annealing(
                h_bin.copy(), rng, deadline
            )
            c_sa = max_O_sa * (2.0 / n)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa.astype(float).copy()
                best_n = n
                best_max_O = max_O_sa

        # smooth‑max SLSQP refinement
        if (
            time.time() < deadline
            and best_h is not None
            and best_n == n
        ):
            h_float = best_h.copy()
            adjust_sum_exact(h_float, rng)
            for beta in (
                2.0,
                5.0,
                10.0,
                20.0,
                50.0,
                100.0,
                200.0,
                500.0,
                1000.0,
                2000.0,
                5000.0,
                10000.0,
            ):
                if time.time() > deadline:
                    break
                max_iter = 300 if beta < 200 else 600
                if beta >= 2000:
                    max_iter = 800
                try:
                    h_float = slsqp_optimize(h_float, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c_new = compute_c5(h_float)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_float.copy()
                    best_n = n
                    best_max_O = int(round(c_new * n / 2.0))

    # fallback (should never happen)
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
