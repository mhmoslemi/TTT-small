    a = h
    b = 1 - h
    b_rev = b[::-1]
    size = 1 << (2*n-1).bit_length() # next power of two >= 2n
    fft_a = np.fft.rfft(a, size)
    fft_b = np.fft.rfft(b_rev, size)
    conv = np.fft.irfft(fft_a * fft_b, size)
    # conv length = size, but valid region is from 0 to 2n-2 (index)
    corr = conv[:2*n-1]
    return corr

def slp_optimize(h0, max_iter=30, eps=1e-8, delta0=0.5, keep_top=5):

                a[i] += h_curr[i + k]
                a[i + k] += h_curr[i]

                a[i] += h_curr[i + k2]

def slp_optimize(h0, max_iter=20, keep_top=5, delta0=0.5):
    import cvxpy as cp
    n = h0.size
    target = n/2.0
    h_curr = h0.copy()
    delta = delta0
    for it in range(max_iter):
        # compute full correlation h with h
        corr = np.correlate(h_curr, h_curr, mode='full')
        min_val = np.min(corr)
        # get indices of keep_top smallest values
        idxs = np.argpartition(corr, keep_top)[:keep_top]
        h = cp.Variable(n)
        m = cp.Variable()
        constraints = [cp.sum(h) == target, h >= 0, h <= 1]
        if delta is not None:
            # l1 trust region
            u = cp.Variable(n)
            v = cp.Variable(n)
            constraints += [u >= h - h_curr, v >= h_curr - h, u >= 0, v >= 0, cp.sum(u + v) <= delta]
        for idx in idxs:
            k = idx - (n - 1)
            a = np.zeros(n)
            # compute gradient contributions
            if k >= 0:
                i = np.arange(0, n - k)
                a[i] += h_curr[i + k]
                a[i + k] += h_curr[i]
            else:
                k2 = -k
                i = np.arange(0, n - k2)
                a[i + k2] += h_curr[i]
                a[i] += h_curr[i + k2]
            c = corr[idx] - a.dot(h_curr)
            constraints.append(a @ h + c >= m)
        prob = cp.Problem(cp.Maximize(m), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1000)
        except Exception:
            break
        if h.value is None:
            break
        h_new = np.clip(h.value, 0.0, 1.0)
        # adjust sum to exact using simple redistribution
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            # distribute diff uniformly to non-bound positions
            mask = (h_new > 1e-12) & (h_new < 1 - 1e-12)
            if np.any(mask):
                add = diff / mask.sum()
                h_new[mask] += add
                h_new = np.clip(h_new, 0.0, 1.0)
            else:
                # fallback: adjust first element
                h_new[0] = np.clip(h_new[0] + diff, 0.0, 1.0)
        new_c5 = compute_c5_corr(h_new)
        old_c5 = compute_c5_corr(h_curr)
        if new_c5 < old_c5 - 1e-7:
            h_curr = h_new
            delta = max(delta * 0.9, 1e-8)
        else:
            delta = delta * 0.5
            if delta < 1e-8:
                break
    return h_curr

def gradient_refine(h, steps=10, step_size=0.02):
    n = h.size
    target = n/2.0
    for it in range(steps):
        # correlation h with h
        corr = np.correlate(h, h, mode='full')
        # find shift with minimum overlap (worst)
        idx = np.argmin(corr)
        k = idx - (n - 1)
        # compute gradient of overlap (to increase overlap)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 2 * h  # d overlap = 2h
        else:
            # for i where both i and i+k in range
            if k > 0:
                g[:n-k] += h[k:]
                g[k:] += h[:n-k]
            else:
                k2 = -k
                g[:n-k2] += h[k2:]
                g[k2:] += h[:n-k2]
        # we want to increase overlap, i.e., move h in direction of g
        h = h + step_size * g
        # project onto bounds and exact sum
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff uniformly among non-bound elements
            mask = (h > 1e-12) & (h < 1 - 1e-12)
            if np.any(mask):
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)
            else:
                h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

import time, numpy as np, scipy.signal as signal

def compute_c5_corr(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx, using discretization."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass to satisfy Σh = n/2 exactly (keeps 0≤h≤1)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def slp_optimize(h0, max_iter=15, keep_top=5, delta0=0.5):
    """Sequential linear programming to raise the minimum overlap."""
    import cvxpy as cp
    n = h0.size
    target = n / 2.0
    h_curr = h0.copy()
    delta = delta0
    for _ in range(max_iter):
        corr = np.correlate(h_curr, h_curr, mode="full")
        idxs = np.argpartition(corr, keep_top)[:keep_top]
        h = cp.Variable(n)
        m = cp.Variable()
        cons = [cp.sum(h) == target, h >= 0, h <= 1]
        if delta is not None:
            u = cp.Variable(n)
            v = cp.Variable(n)
            cons += [
                u >= h - h_curr,
                v >= h_curr - h,
                u >= 0,
                v >= 0,
                cp.sum(u + v) <= delta,
            ]
        for idx in idxs:
            k = idx - (n - 1)
            a = np.zeros(n)
            if k >= 0:
                i = np.arange(0, n - k)
                a[i] += h_curr[i + k]
                a[i + k] += h_curr[i]
            else:
                k2 = -k
                i = np.arange(0, n - k2)
                a[i + k2] += h_curr[i]
                a[i] += h_curr[i + k2]
            c = corr[idx] - a.dot(h_curr)
            cons.append(a @ h + c >= m)
        prob = cp.Problem(cp.Maximize(m), cons)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1000)
        except Exception:
            break
        if h.value is None:
            break
        h_new = np.clip(h.value, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            mask = (h_new > 1e-12) & (h_new < 1 - 1e-12)
            if mask.any():
                h_new[mask] += diff / mask.sum()
                h_new = np.clip(h_new, 0.0, 1.0)
            else:
                h_new[0] = np.clip(h_new[0] + diff, 0.0, 1.0)
        new_c5 = compute_c5_corr(h_new)
        old_c5 = compute_c5_corr(h_curr)
        if new_c5 < old_c5 - 1e-7:
            h_curr = h_new
            delta = max(delta * 0.9, 1e-8)
        else:
            delta = delta * 0.5
            if delta < 1e-8:
                break
    return h_curr

def gradient_refine(h, steps=12, step=0.02):
    """Increase the worst overlap by projected gradient steps."""
    n = h.size
    target = n / 2.0
    for _ in range(steps):
        corr = np.correlate(h, h, mode="full")
        idx = np.argmin(corr)
        k = idx - (n - 1)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 2 * h
        elif k > 0:
            g[: n - k] += h[k:]
            g[k:] += h[: n - k]
        else:
            k2 = -k
            g[: n - k2] += h[k2:]
            g[k2:] += h[: n - k2]
        h = h + step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            mask = (h > 1e-12) & (h < 1 - 1e-12)
            if mask.any():
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)
            else:
                h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def interpolate_h(parent, n):
    xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(xt, xs, parent)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if "initial_h_values" in globals() else None
    n_candidates = [240, 360, 480, 720]

    for n in n_candidates:
        if time.time() > deadline:
            break
        starts = []

        # from parent
        if parent is not None:
            starts.append(interpolate_h(parent, n))

        # Chebyshev window (smooth positive shape)
        w = signal.chebwin(n, at=120)
        w = w.astype(float)
        w = w * ((n / 2.0) / w.sum())
        starts.append(w)

        # random binary
        bin_h = np.zeros(n)
        bin_h[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_h)

        # random fractional
        rand = rng.random(n)
        rand = rand * ((n / 2.0) / rand.sum())
        starts.append(rand)

        for h0 in starts:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # sequential linear programming
            h = slp_optimize(h, max_iter=12, keep_top=4, delta0=0.4)

            # gradient polishing
            h = gradient_refine(h, steps=10, step=0.03)

            c = compute_c5_corr(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # optional up‑sampling of the best solution
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        h_up = interpolate_h(best_h, n_up)
        adjust_sum_exact(h_up, rng)
        h_up = slp_optimize(h_up, max_iter=10, keep_top=3, delta0=0.3)
        h_up = gradient_refine(h_up, steps=8, step=0.025)
        c_up = compute_c5_corr(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    if best_h is None:
        n = 240
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5_corr(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5_corr(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
