import time
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

# ----------- global helpers for smooth‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

# ----------- core utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=20):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]            # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ----------- MIP feasibility for a given bound ----------
def solve_mip_for_m(n, m_bound, time_limit, solver):
    """Return (feasible, h) for bound m_bound = max(A_k - p_k)."""
    h = cp.Variable(n, boolean=True)
    constraints = [cp.sum(h) == n // 2]
    for k in range(1, n):
        nk = n - k
        w = cp.Variable(nk, boolean=True)
        constraints.append(w <= h[:nk])
        constraints.append(w <= h[k:])
        constraints.append(w >= h[:nk] + h[k:] - 1)
        constraints.append(cp.sum(h[:nk]) - cp.sum(w) <= m_bound)
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=solver, glpk={'tm_lim': int(time_limit * 1000)},
                    warm_start=True, verbose=False)
    except Exception:
        return False, None
    status = str(prob.status).lower()
    if status.startswith("optimal") or status.startswith("feas"):
        hv = np.rint(h.value).astype(int)
        return True, hv
    return False, None

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.98

    best_h = None
    best_c = float("inf")
    best_n = None

    # Choose a solver for MIP
    if "GLPK_MI" in cp.installed_solvers():
        mip_solver = cp.GLPK_MI
    elif "CBC" in cp.installed_solvers():
        mip_solver = cp.CBC
    else:
        mip_solver = None

    # ----------- attempt exact MIP for small n ----------
    if mip_solver is not None:
        for n in (180, 240, 300):
            if time.time() > deadline:
                break
            # quick random upper bound M0
            M0 = None
            for _ in range(20):
                h_rand = np.zeros(n)
                h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
                pref = np.cumsum(h_rand)
                conv = np.correlate(h_rand, h_rand, mode="full")
                diffs = []
                for k in range(1, n):
                    A = pref[n - k - 1] if n - k > 0 else 0
                    p = conv[n - 1 + k]
                    diffs.append(A - p)
                M = max(diffs) if diffs else 0
                if M0 is None or M < M0:
                    M0 = M
            if M0 is None:
                M0 = n // 2
            lo, hi = 0, M0
            feasible_h = None
            feasible_m = None
            while lo <= hi and time.time() < deadline:
                mid = (lo + hi) // 2
                remaining = deadline - time.time()
                tlim = min(60.0, remaining * 0.9)
                feasible, h_sol = solve_mip_for_m(n, mid, tlim, mip_solver)
                if feasible:
                    feasible_h, feasible_m = h_sol, mid
                    hi = mid - 1
                else:
                    lo = mid + 1
            if feasible_h is not None:
                c5 = compute_c5(feasible_h)
                if c5 < best_c:
                    best_h, best_c, best_n = feasible_h.astype(float), c5, n

    # ----------- heuristic fallback / refinement ----------
    if best_h is None or best_c > 0.381:
        # decide dimension
        if 'initial_h_values' in globals():
            n = 360
        else:
            n = 240
        if n % 2:
            n += 1
        # initial vectors
        start_vecs = []
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            if parent.size != n:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                start_vecs.append(np.interp(xt, xs, parent))
            else:
                start_vecs.append(parent.copy())
        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)
        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= (n / 2) / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        start_vecs.append(rand_frac)

        for base_h in start_vecs:
            if time.time() > deadline:
                break
            h = base_h.copy()
            adjust_sum_exact(h, rng)
            h = np.clip(h, 0.0, 1.0)

            h = gradient_refine(h, rng, step=0.02, max_iter=30)

                h = slsqp_optimize(h, beta, rng, max_iter=400)
