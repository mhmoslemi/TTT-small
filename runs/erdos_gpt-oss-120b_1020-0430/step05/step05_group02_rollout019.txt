def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # parent if exists
    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    # We'll focus on n=180
    n = 180
    # generate starting vectors
    start_vectors = []
    # Use parent interpolated if available
    if parent is not None:
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        start_vectors.append(interp.copy())
        # binary threshold version
        idx = np.argpartition(interp, -n//2)[-n//2:]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        start_vectors.append(bin_vec)

    # random binary
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n//2, replace=False)] = 1.0
    start_vectors.append(bin_rand)

    # random fractional
    rand_frac = rng.random(n)
    target = n/2.0
    rand_frac *= target / rand_frac.sum()
    adjust_sum_exact(rand_frac, rng)
    start_vectors.append(rand_frac)

    # alternating pattern
    alt_bin = np.tile([1.0, 0.0], n//2)
    start_vectors.append(alt_bin)

    # For each start vector:
    for h0 in start_vectors:
        if time.time() > deadline:
            break
        # copy and ensure feasible
        h = np.clip(h0, 0.0, 1.0)
        adjust_sum_exact(h, rng)
        # gradient refine low iterations
        h = gradient_refine(h, rng, step=0.03, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # Simulated annealing
        # allocate up to a fraction of remaining time, e.g., 0.2 * remaining.
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # decide max iterations as min(200k, int(remaining*5000)) (since each iteration maybe microseconds).
        max_iters = int(min(200000, remaining * 4000))  # estimate 2500 iterations per second? We'll adjust.
        if max_iters <= 0:
            continue
        h_sa, c_sa = simulated_annealing_continuous(h.copy(), rng, deadline, max_iters)
        if c_sa < best_c:
            best_h, best_c, best_n = h_sa.copy(), c_sa, n

        # final gradient refinement more iterations
        h = gradient_refine(h, rng, step=0.02, max_iter=20)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # Upsample if best found
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        # gradient refine up
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=30)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # final safety
    if best_h is None:
        # fallback simple pattern n=180
        n = 180
        best_h = np.zeros(n)
        best_h[: n//2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)

    target = n / 2.0
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # initial temperature
    T0 = cur_c * 0.05 + 1e-4  # small scaling
    # schedule: T = T0 * (0.99995 ** it)
    # step max also decays with temperature
    # We'll use step = max_step * (T / T0)
    max_step_initial = 0.1  # max change per move
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        # compute temperature
        T = T0 * (0.99995 ** it)
        # step size
        max_step = max_step_initial * (T / T0)
        # pick i source, j target
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute allowed delta
        # we will move from i to j, i decreases.
        delta_max = min(max_step, h[i], 1.0 - h[j])
        if delta_max <= 0:
            continue
        delta = rng.random() * delta_max
        # apply move
        h[i] -= delta
        h[j] += delta
        new_c = compute_c5(h)
        delta_c = new_c - cur_c
        accept = False

                    accept = True
        if accept:

            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        # occasionally reheat or random perturb? maybe not needed.
    return best_h, best_c

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def hill_climb_min_corr(h, rng, deadline, max_iters, init_step=0.2, decay=0.9997):
    n = h.size
    step = init_step
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    it = 0
    while it < max_iters and time.time() < deadline:

        corr_full = np.correlate(h, h, mode="full")
        corr_k = corr_full[n-1:]  # k = 0..n-1
        k_star = np.argmin(corr_k)
        # shifted array for this k
        arr = np.concatenate((h[k_star:], np.zeros(k_star)))
        # source (small arr, h>0)
        src_mask = h > 0
        if not np.any(src_mask):

        arr_src = np.where(src_mask, arr, np.inf)
        i = np.argmin(arr_src)
        # target (large arr, h<1)
