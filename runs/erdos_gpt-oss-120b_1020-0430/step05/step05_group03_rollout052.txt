import time

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # Simple random redistribution; converges very fast for our size.
    while abs(diff) > 1e-12:
        if diff > 0:
            idxs = np.where(h < 1)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            idxs = np.where(h > 0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of C5 for the shift giving the current maximum."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

            g = 1.0 - conv[(n - 1):(n - 1 + n)]     # expected gradient

            # --- take a step and project back ---
            nh = h - step * g
            nh = np.clip(nh, 0.0, 1.0)
            nh = enforce_sum(nh, rng)

            # --- evaluate ---
            c = compute_c5(nh)
            if c < best_c:
                best_c = c
                best_h = nh.copy()
            # keep the iterates moving towards lower smooth objective
            h = nh

            # occasional random jitter to escape plateaus
            if (it + 1) % 200 == 0 and rng.random() < 0.3:
                i, j = rng.integers(0, n, size=2)
                eps = min(0.05, h[i], 1.0 - h[j])
                if eps > 0:
                    nh2 = h.copy()
                    nh2[i] -= eps
                    nh2[j] += eps
                    nh2 = enforce_sum(nh2, rng)
                    c2 = compute_c5(nh2)
                    if c2 < best_c:
                        best_c = c2
                        best_h = nh2.copy()
                    h = nh2

            if time.time() - start > 0.90 * budget_s:
                break

    # --------------------------------------------------------------
    # Final local refinement with the proven hill‑climb machinery
    # --------------------------------------------------------------
    # keep a separate copy for the refinement phases
    h_ref = best_h.copy()
    h_ref, c_ref = guided_hill_climb(h_ref, rng,
                                     max_iters=2500, step_scale=0.6)
    if c_ref < best_c:
        best_c, best_h = c_ref, h_ref.copy()

    # fine‑grained gradient tidy‑up
    h_ref = gradient_refine(h_ref, rng, step=0.01, max_iter=10)
    c_ref = compute_c5(h_ref)
    if c_ref < best_c:
        best_c, best_h = c_ref, h_ref.copy()

    # --------------------------------------------------------------
    # Last stochastic tweaks until the budget expires
    # --------------------------------------------------------------
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_c, best_h = nc, nh.copy()

    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
