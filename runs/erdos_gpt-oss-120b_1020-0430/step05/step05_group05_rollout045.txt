import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap C5 for a binary (or fractional) vector h on [0,2]."""
    n = h.shape[0]
    H = np.fft.fft(h)
    G = np.fft.fft(1.0 - h)
    corr = np.fft.ifft(H * np.conj(G)).real
    return float(np.max(corr) * (2.0 / n))


def anneal_sym(h: np.ndarray, rng: np.random.Generator, deadline: float) -> tuple:
    """Simulated annealing on a vector with antipodal symmetry h[i+n/2]=1-h[i]."""
    n = h.shape[0]
    m = n // 2
    cur_c = compute_c5(h)
    # temperature based on the underlying integer correlation value
    max_corr = cur_c * n / 2.0
    T = max_corr * 4.0 + 5.0
    cooling = 0.9995

    while time.time() < deadline:
        i = rng.integers(0, m)
        # flip the antipodal pair
        new_h = h.copy()
        new_h[i] = 1.0 - new_h[i]
        new_h[i + m] = 1.0 - new_h[i + m]
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            h = new_h
            cur_c = new_c
        T *= cooling
        if T < 1e-6:
            break
        # periodic checkpoint
        if time.time() > deadline - 2.0:
            break
    return h, cur_c


def greedy_improve(h: np.ndarray, rng: np.random.Generator, deadline: float) -> tuple:
    """Greedy swaps (respecting symmetry) that lower the current worst shift."""
    n = h.shape[0]
    m = n // 2
    cur_c = compute_c5(h)

    while time.time() < deadline:
        # circular correlation via FFT
        H = np.fft.fft(h)
        G = np.fft.fft(1.0 - h)
        corr = np.fft.ifft(H * np.conj(G)).real
        k = int(np.argmax(corr))               # shift in [0, n-1]
        # minimal absolute shift (due to periodicity)
        shift = k if k <= n // 2 else n - k
        if shift == 0:
            break

        idx = np.arange(m)
        plus = h[(idx + shift) % n]
        minus = h[(idx - shift) % n]
        neigh = plus + minus                     # sum of the two neighbours for each i < m

        ones = np.where(h[:m] == 1.0)[0]
        zeros = np.where(h[:m] == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        min_neigh = np.min(neigh[ones])
        max_neigh = np.max(neigh[zeros])

        cand_i = ones[neigh[ones] == min_neigh]
        cand_j = zeros[neigh[zeros] == max_neigh]
        i = cand_i[rng.integers(0, cand_i.size)]
        j = cand_j[rng.integers(0, cand_j.size)]

        new_h = h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_h[i + m] = 1.0
        new_h[j + m] = 0.0

        new_c = compute_c5(new_h)
        if new_c < cur_c - 1e-12:
            h = new_h
            cur_c = new_c
        else:
            break
    return h, cur_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # work with n = 720 (<= 1000 as required)
    n = 720
    m = n // 2

    best_h = None
    best_c = np.inf

    # Number of independent random starts (adjusted dynamically)
    max_starts = 6
    start_idx = 0

    while time.time() < deadline and start_idx < max_starts:
        # ----- generate a random symmetric binary seed -----
        b = rng.integers(0, 2, size=m, dtype=np.int8)
        s = b.sum()
        if s == 0:
            b[rng.integers(0, m)] = 1
        elif s == m:
            b[rng.integers(0, m)] = 0

        h = np.empty(n, dtype=np.float64)
        h[:m] = b
        h[m:] = 1 - b

        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

        # ----- annealing phase -----
        h, c = anneal_sym(h, rng, deadline)
        if c < best_c:
            best_c, best_h = c, h.copy()

        # ----- greedy polishing -----
        h, c = greedy_improve(h, rng, deadline)
        if c < best_c:
            best_c, best_h = c, h.copy()

        start_idx += 1

    # ----- fallback in the unlikely event no solution was found -----
    if best_h is None:
        best_h = np.zeros(n, dtype=np.float64)
        best_h[:m] = 1.0
        best_h[m:] = 0.0
        best_c = compute_c5(best_h)

    # final sanity check (enforce constraints exactly)
    best_h = np.clip(best_h, 0.0, 1.0)
    # ensure antipodal symmetry; tiny numerical drift is corrected
    best_h[m:] = 1.0 - best_h[:m]
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
