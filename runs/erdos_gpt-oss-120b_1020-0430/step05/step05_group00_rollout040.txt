def gradient_refine(h, rng, step=0.01, max_iter=20):
    n = h.shape[0]
    target = n/2.0
    for it in range(max_iter):
        # compute worst shift gradient via worst_gradient
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] -= h[k:]
            g[k:]       -= h[: n - k]
        # update
        h = h - (step * (0.5 ** it)) * g
        # clip and enforce sum
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ----- problem size -----
    n = 1024                     # < 1k points, gives finer resolution
    target = n / 2.0
    zero_idx = n - 1
    N = 2 * n - 1                # length of full correlation

    # ----- initial h -----
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        if base.size != n:
            x_old = np.linspace(0.0, 2.0, base.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, base)
        else:
            h = base.copy()
    else:
        h = rng.random(n)

    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)

    # ----- initial correlation (h with h) -----
    P = np.correlate(h, h, mode="full")
    curr_c5 = (target - P.min()) * (2.0 / n)
    best_h = h.copy()
    best_c5 = curr_c5

    # ----- simulated annealing on the continuous variables -----
    time_sa = total_budget * 0.80
    T0 = 0.05
    Tmin = 1e-8
    scale_factor = 5.0
    min_scale = 1e-5
    dP = np.empty(N, dtype=np.float64)

    while time.time() - start < time_sa:
        elapsed = time.time() - start
        frac = elapsed / time_sa
        T = T0 * (Tmin / T0) ** frac   # exponential cooling

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move mass from i to j
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * max(T * scale_factor, min_scale)
        if delta <= 0.0:
            continue
        di = -delta
        dj = delta

        # update correlation incrementally
        dP.fill(0.0)
        rev_h = h[::-1]                                 # view, no copy

        # i‑contributions
        dP[i:i + n] += di * rev_h
        dP[zero_idx - i: zero_idx - i + n] += di * h
        # j‑contributions
        dP[j:j + n] += dj * rev_h
        dP[zero_idx - j: zero_idx - j + n] += dj * h

        # quadratic terms
        dP[zero_idx] += di * di + dj * dj
        if i != j:
            dP[zero_idx + (j - i)] += di * dj
            dP[zero_idx + (i - j)] += di * dj

        new_minP = (P + dP).min()
        new_c5 = (target - new_minP) * (2.0 / n)

        # Metropolis acceptance
        if new_c5 < curr_c5:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_c5 - curr_c5) / max(T, 1e-12))

        if accept:
            # apply the move
            h[i] += di
            h[j] += dj
            P += dP
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

    # ----- final gradient‑descent polishing -----
    # simple projected gradient steps using the worst shift
    while time.time() - start < total_budget * 0.98:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(best_h)
        if k == 0:
            g[:] = 1.0 - 2.0 * best_h
        else:
            g[: n - k] -= best_h[k:]
            g[k:]       -= best_h[: n - k]

        step = 0.02
        new_h = best_h - step * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, rng)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        else:
            # if no improvement, reduce step size a bit
            step *= 0.9
        if time.time() - start > total_budget * 0.99:
            break

    # ----- final sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
