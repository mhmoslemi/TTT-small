def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    # while loop for a few iterations
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=20):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)  # shift value
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:

            # for i in [0, upper): term (1 - h[i+k]) for gradient

            g[k:] -= h[:upper]

        h = h - step_sz * g

        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # redistribute
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time
import numpy as np
from scipy.optimize import minimize
from itertools import combinations

# globals used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap for step function samples h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass to satisfy Σh = n/2 exactly (maintaining bounds)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max via log‑sum‑exp, scaled by CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP on the smooth max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                   step: float = 0.02, max_iter: int = 30) -> np.ndarray:
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)           # shift value (could be negative)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]                # derivative w.r.t. h_i for i<up
            g[k:] -= h[:up]                      # derivative w.r.t. h_{i+k}
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def fractional_sa_hill_climb(h: np.ndarray, rng: np.random.Generator,
                             time_deadline: float):
    """Transfer‑based simulated annealing on continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 15000:
            break
    return best, best_c


def random_local_search(h: np.ndarray, rng: np.random.Generator,
                        time_deadline: float):
    """Very simple stochastic local search for a short final push."""
    n = h.size
    best = h.copy()
    best_c = compute_c5(best)
    target = n / 2.0
    while time.time() < time_deadline:
        i = rng.integers(0, n)
        delta = (rng.random() - 0.5) * 0.04  # small perturbation
        new = best.copy()
        new[i] = np.clip(new[i] + delta, 0.0, 1.0)
        diff = target - new.sum()
        if abs(diff) > 1e-12:
            j = rng.integers(0, n)
            add = np.clip(diff, -new[j], 1.0 - new[j])
            new[j] += add
        c = compute_c5(new)
        if c < best_c:
            best, best_c = new, c
    return best, best_c

            candidates.append(h)

    # c) Beatty‑type (irrational rotation) binary patterns
    alphas = [np.sqrt(2), np.sqrt(3), np.pi, np.e, (np.sqrt(5) - 1) / 2, 0.6180339887498949]
    target_ones = n // 2

        h = (arr < 0.5).astype(float)

            idx = np.argsort(np.abs(arr - 0.5))[:excess]
            h[idx] = 0.0

                except Exception:

                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_c, best_h = c_opt, h_opt.copy()

        # final gradient refinement

        c = compute_c5(h)
        if c < best_c:
