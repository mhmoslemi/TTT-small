import numpy as np
from scipy.optimize import minimize

# ---------- global placeholders ----------
CURRENT_TARGET = None
CURRENT_BETA = None
CURRENT_N = None

# ---------- basic utilities ----------
def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


# ---------- smooth‑max surrogate ----------
def smooth_max_obj(h):
    """Smooth max of overlap = target - correlation."""
    Q = np.correlate(h, h, mode="full")
    O = CURRENT_TARGET - Q
    m = np.max(O)
    exp_terms = np.exp(CURRENT_BETA * (O - m))
    return m + (1.0 / CURRENT_BETA) * np.log(np.sum(exp_terms))


def smooth_max_jac(h):
    Q = np.correlate(h, h, mode="full")
    O = CURRENT_TARGET - Q
    m = np.max(O)
    exp_terms = np.exp(CURRENT_BETA * (O - m))
    w = exp_terms / np.sum(exp_terms)  # length 2n‑1
    conv_full = np.convolve(w, h, mode="full")
    start = CURRENT_N - 1
    grad_same = conv_full[start : start + CURRENT_N]
    return -2.0 * grad_same

    CURRENT_N = n
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_max_obj,
        h0,
        method="SLSQP",
        jac=smooth_max_jac,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


# ---------- local refinements ----------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1‑h[i+k])
            g[k:] -= h[:upper]            # – h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def binary_sa_hill_climb(h_bin, rng, time_deadline):
    """Simulated‑annealing hill climb on binary vectors (swap 0↔1)."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c


def greedy_improve(h, rng, max_iter=50, deadline=None):
    """Local search: try swaps that reduce the current worst shift."""
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        # positions where a 1 mismatches a 0 for this shift
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        # positions where a 0 mismatches a 1
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = compute_c5(h)
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h


def upsample_h(h, new_n):
    """Linear interpolation up‑sampling."""
    old_n = h.size
    xs = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xt = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(xt, xs, h)
    return np.clip(h_new, 0.0, 1.0)


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = globals().get("initial_h_values")
    if parent is not None:
        parent = np.array(parent, dtype=float)

    candidate_ns = [720, 480, 360, 240, 180]
    candidate_ns = [n for n in candidate_ns if n % 2 == 0]

    beta_schedule = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        starts = []

        # parent‑based starts
        if parent is not None:
            if n != parent.size:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                interp = np.interp(xt, xs, parent)
                starts.append(interp)
                idx = np.argpartition(interp, -n // 2)[-n // 2 :]
                bin_vec = np.zeros(n)
                bin_vec[idx] = 1.0
                starts.append(bin_vec)
            else:
                starts.append(parent.copy())
                if not np.all((parent == 0) | (parent == 1)):
                    idx = np.argpartition(parent, -n // 2)[-n // 2 :]
                    bin_vec = np.zeros(n)
                    bin_vec[idx] = 1.0
                    starts.append(bin_vec)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        # random fractional (uniform)
        rand_frac = rng.random(n)
        target = n / 2.0
        rand_frac *= target / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)
        starts.append(rand_frac)

        # a couple of beta‑distributed starts
        for _ in range(2):
            a = rng.random() * 3 + 0.5
            b = rng.random() * 3 + 0.5
            rand_frac2 = rng.beta(a, b, n)
            rand_frac2 *= target / rand_frac2.sum()
            rand_frac2 = np.clip(rand_frac2, 0.0, 1.0)
            adjust_sum_exact(rand_frac2, rng)
            starts.append(rand_frac2)

        for h0 in starts:
            if time.time() > deadline:
                break
            h = h0.copy()
            h = np.clip(h, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # binary simulated annealing (if binary)
            if np.all((h == 0) | (h == 1)):
                sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
                h_bin, c_bin = binary_sa_hill_climb(h.copy(), rng, sub_dead)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin.astype(float)

            # early gradient refinement
            h = gradient_refine(h, rng, step=0.02, max_iter=30)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # smooth‑max SLSQP cascade
            for beta in beta_schedule:
                if time.time() > deadline:
                    break
                max_iter = 300 if beta < 1000 else 500
                try:
                    h = slsqp_optimize_smooth(h, beta, rng, max_iter=max_iter)
                except Exception:
                    # fallback to cheap gradient step
                    h = gradient_refine(h, rng, step=0.01, max_iter=5)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final gradient passes
            h = gradient_refine(h, rng, step=0.01, max_iter=40)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # if essentially binary, try greedy local improvement
            if np.all((np.abs(h - 0) < 1e-9) | (np.abs(h - 1) < 1e-9)):
                h_bin = np.rint(h).astype(float)
                adjust_sum_exact(h_bin, rng)
                h_bin = greedy_improve(h_bin, rng, max_iter=80, deadline=deadline)
                c = compute_c5(h_bin)
                if c < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c, n
                h = h_bin

    # ----- optional up‑sampling of best solution -----
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        h_up = upsample_h(best_h, n_up)
        adjust_sum_exact(h_up, rng)

        # a short refinement after up‑sampling
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=30)
        for beta in beta_schedule[-5:]:  # strongest betas only
            if time.time() > deadline:
                break
            h_up = slsqp_optimize_smooth(h_up, beta, rng, max_iter=500)
        h_up = gradient_refine(h_up, rng, step=0.01, max_iter=40)

        if np.all((np.abs(h_up - 0) < 1e-9) | (np.abs(h_up - 1) < 1e-9)):
            h_up = greedy_improve(h_up, rng, max_iter=80, deadline=deadline)

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ----- fallback (should never trigger) -----
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
