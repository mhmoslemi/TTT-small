def adjust_sum_binary(h, rng, target):
    cur = int(np.sum(h))
    diff = target - cur
    if diff > 0:
        # need to flip diff zeros to ones
        zero_idx = np.flatnonzero(h == 0)
        chosen = rng.choice(zero_idx, size=diff, replace=False)
        h[chosen] = 1
    elif diff < 0:
        # flip -diff ones to zeros
        one_idx = np.flatnonzero(h == 1)
        chosen = rng.choice(one_idx, size=-diff, replace=False)
        h[chosen] = 0
    return h

def hill_climb_targeted(h, max_swaps, rng, deadline):
    cur_h = h.copy()
    best_h = cur_h.copy()
    best_c = compute_c5(cur_h)
    n = cur_h.size
    for it in range(max_swaps):
        if time.time() > deadline:
            break
        # decide mode
        if rng.random() < 0.2:
            # targeted
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            idx = np.argmax(corr)
            k = idx - (n - 1)
            if k != 0:
                if k > 0:
                    i_start = 0
                    i_end = n - k
                else:
                    i_start = -k
                    i_end = n
                i_range = np.arange(i_start, i_end)
                mismatched = i_range[(cur_h[i_range] == 1) & (cur_h[i_range + k] == 0)]
                if mismatched.size > 0:
                    i_sel = rng.choice(mismatched)
                    # choose j zero not equal to i_sel or i_sel+k maybe
                    zero_idx = np.flatnonzero(cur_h == 0)
                    # if zero_idx empty, continue with random
                    if zero_idx.size > 0:
                        j_sel = rng.choice(zero_idx)
                        # apply swap
                        cur_h[i_sel] = 0
                        cur_h[j_sel] = 1
                        new_c = compute_c5(cur_h)
                        if new_c < best_c:
                            best_c = new_c
                            best_h = cur_h.copy()
                        else:
                            # revert
                            cur_h[i_sel] = 1
                            cur_h[j_sel] = 0
                        continue
        # random swap fallback
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:

        i = rng.choice(ones)

import time
import numpy as np
from scipy.optimize import minimize

# ----- basic utilities -------------------------------------------------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Adjust a float vector so that sum(h) == len(h)/2 exactly."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def adjust_sum_binary(h, rng, target):
    """Flip bits of a binary vector to reach the required sum."""
    cur = int(np.sum(h))
    diff = target - cur
    if diff > 0:
        zeros = np.flatnonzero(h == 0)
        choose = rng.choice(zeros, size=diff, replace=False)
        h[choose] = 1
    elif diff < 0:
        ones = np.flatnonzero(h == 1)
        choose = rng.choice(ones, size=-diff, replace=False)
        h[choose] = 0
    return h

# ----- greedy binary constructor ---------------------------------------
def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h, O

# ----- hill climbing with occasional targeted moves --------------------
def hill_climb_targeted(h0, max_swaps, rng, deadline):
    cur_h = h0.copy()
    best_h = cur_h.copy()
    best_c = compute_c5(cur_h)
    n = cur_h.size
    for _ in range(max_swaps):
        if time.time() > deadline:
            break
        # 20% chance to attempt a targeted move
        if rng.random() < 0.2:
            corr = np.correlate(cur_h, 1 - cur_h, mode="full")
            idx = int(np.argmax(corr))
            k = idx - (n - 1)
            if k != 0:
                if k > 0:
                    i_start, i_end = 0, n - k
                else:
                    i_start, i_end = -k, n
                i_range = np.arange(i_start, i_end)
                mismatched = i_range[(cur_h[i_range] == 1) & (cur_h[i_range + k] == 0)]
                if mismatched.size:
                    i_sel = rng.choice(mismatched)
                    zero_idx = np.flatnonzero(cur_h == 0)
                    if zero_idx.size:
                        j_sel = rng.choice(zero_idx)
                        cur_h[i_sel] = 0
                        cur_h[j_sel] = 1
                        new_c = compute_c5(cur_h)
                        if new_c < best_c:
                            best_c = new_c
                            best_h = cur_h.copy()
                        else:
                            cur_h[i_sel] = 1
                            cur_h[j_sel] = 0
                        continue
        # random swap fallback
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        if new_c < best_c:
            best_c = new_c
            best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_c

# ----- smooth‑max SLSQP refinement --------------------------------------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

# ----- main driver ------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.985

    best_h = None
    best_c = np.inf
    best_n = None

    def record(candidate, cval, nval):
        nonlocal best_h, best_c, best_n
        if cval < best_c:
            best_h = candidate.copy()
            best_c = cval
            best_n = nval

    # list of problem sizes to explore
    size_list = [540, 720, 1080]

    for n in size_list:
        if time.time() > deadline:
            break
        target = n // 2

        # ----- collect diverse initial seeds -----
        seeds = []

        # parent if size matches
        if "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
            if parent.size == n:
                seeds.append(parent)

        # greedy binary
        try:
            hb, _ = greedy_binary(n, rng)
            seeds.append(hb.astype(float))
        except Exception:
            pass

        # irrational‑rotation based binaries
        alphas = [
            (np.sqrt(5) - 1) / 2,
            np.sqrt(2) - 1,
            np.sqrt(3) - 1,
            np.sqrt(2) / 2,
            np.sqrt(3) / 3,
        ]
        for a in alphas:
            fr = ((np.arange(n) * a) % 1)
            hbin = (fr < 0.5).astype(np.int8)
            if np.sum(hbin) != target:
                hbin = adjust_sum_binary(hbin, rng, target)
            seeds.append(hbin.astype(float))

        # a few pure random binaries
        for _ in range(3):
            perm = rng.permutation(n)
            hbin = np.zeros(n, dtype=np.int8)
            hbin[perm[:target]] = 1
            seeds.append(hbin.astype(float))

        # ----- process each seed -----
        for seed_h in seeds:
            if time.time() > deadline:
                break
            seed_h = adjust_sum_exact(seed_h.copy(), rng)
            c0 = compute_c5(seed_h)
            record(seed_h, c0, n)

            # allocate swaps proportionally to remaining time
            remain = deadline - time.time()
            max_swaps = int(min(30000, max(0, remain * 500)))
            if max_swaps <= 0:
                continue

            h_hc, c_hc = hill_climb_targeted(seed_h, max_swaps, rng, deadline)
            record(h_hc, c_hc, n)

        # ----- SLSQP refinement on the current best for this n -----
        if best_h is not None and best_n == n:
            h_float = np.clip(best_h, 0.0, 1.0)
            h_float = adjust_sum_exact(h_float, rng)
            betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
            for beta in betas:
                if time.time() > deadline:
                    break
                try:
                    h_float = slsqp_optimize(h_float, beta, rng, max_iter=800)
                except Exception:
                    continue
                c_new = compute_c5(h_float)
                record(h_float, c_new, n)

    # ----- fallback if nothing was found -----
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_h = adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
