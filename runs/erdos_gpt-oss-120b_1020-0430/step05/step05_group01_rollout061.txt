    hi = 1e6

        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, lower, upper)
        s = h.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
        if abs(s - target_sum) < 1e-12:

    return h

def softmax_obj_and_grad(h, beta):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr length 2n-1
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    f = (max_bc + np.log(sum_exp)) / beta

    w_full = exp_bc / sum_exp
    w_pos = w_full[n-1:]  # k=0..n-1
    w_neg = w_full[:n-1][::-1]  # k=1..n-1

import time, math
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Projection of v onto the set {0≤x≤1, sum x = target_sum}."""
    lo, hi = -1e9, 1e9
    for _ in range(60):
        lam = (lo + hi) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
        if abs(s - target_sum) < 1e-12:
            break
    return np.clip(v - lam, 0.0, 1.0)


def correlation_full(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1‑h) using FFT (zero‑padded)."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    conv = np.fft.irfft(H * G, m)[: 2 * n - 1]
    return conv


def softmax_f(h: np.ndarray, beta: float) -> float:
    """Soft‑max of the overlap values (no gradient)."""
    corr = correlation_full(h)
    bc = beta * corr
    max_bc = np.max(bc)
    sum_exp = np.sum(np.exp(bc - max_bc))
    return (max_bc + np.log(sum_exp)) / beta


def softmax_obj_and_grad(h: np.ndarray, beta: float):
    """Soft‑max objective and its exact gradient."""
    n = h.shape[0]
    corr = correlation_full(h)                     # length 2n‑1
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    f = (max_bc + np.log(sum_exp)) / beta

    w_full = exp_bc / sum_exp                      # weights for each shift
    w_pos = w_full[n - 1 :]                        # k = 0 … n‑1
    w_neg = w_full[: n - 1][::-1]                  # k = 1 … n‑1 (for negative shifts)

    grad = np.zeros_like(h)

    # positive shifts (including k=0)
    for k, wk in enumerate(w_pos):
        if wk == 0.0:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[: n - k] += wk * (1.0 - h[k:])     # (1‑h[i+k])
            grad[k:] -= wk * h[: n - k]            # –h[i‑k]

    # negative shifts (k = 1 … n‑1)
    for k, wk in enumerate(w_neg, start=1):
        grad[k:] += wk * (1.0 - h[: n - k])        # (1‑h[j‑k]) for j≥k
        grad[: n - k] -= wk * h[k:]                # –h[j+k] for j+k<n

    return f, grad


# ---------- hill‑climb helpers ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        nh = np.clip(nh, 0.0, 1.0)
        if abs(nh.sum() - target_sum) > 1e-12:
            nh = project_to_simplex(nh, target_sum)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
        h = nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- main optimisation ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = float("inf")
    best_n = None

    # initial construction (if supplied)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_simplex(h0, h0.size / 2.0)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    # candidate resolutions (coarse → fine)
    cand_ns = [256, 384, 512]

    # ---------- multi‑resolution search ----------
    for n in cand_ns:
        # time check
        if time.time() - start > 0.80 * budget_s:
            break

        # initialise a fresh h for this resolution
        if best_h is not None:
            if best_n == n:
                h = best_h.copy()
            else:
                # linear interpolation from best_n → n
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
        else:
            # random binary start
            h = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1.0

        h = np.clip(h, 0.0, 1.0)
        h = project_to_simplex(h, n / 2.0)

        # ----- projected gradient descent with soft‑max -----
        betas = [5, 20, 50, 100, 200, 500, 2000, 8000, 20000]
        alpha0 = 1.0
        target_sum = n / 2.0
        for beta in betas:
            max_iter = 2000
            for it in range(max_iter):
                if time.time() - start > budget_s * 0.95:
                    break
                f, grad = softmax_obj_and_grad(h, beta)
                alpha = alpha0 / (math.sqrt(beta) + 1.0)
                # simple back‑tracking line search
                for _ in range(10):
                    v = h - alpha * grad
                    h_new = project_to_simplex(v, target_sum)
                    f_new = softmax_f(h_new, beta)
                    if f_new <= f:
                        break
                    alpha *= 0.5
                else:
                    h_new = h  # no improvement
                h = h_new
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n
            alpha0 *= 0.7

        # ----- local hill climb for polishing -----
        h, c = guided_hill_climb(h, rng, max_iters=3000, step_scale=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ---------- final high‑resolution refinement ----------
    if best_h is not None and best_h.size == 512 and time.time() - start < budget_s * 0.98:
        h = best_h.copy()
        h, c = guided_hill_climb(h, rng, max_iters=6000, step_scale=0.4)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ---------- annealed stochastic tweaks ----------
    temp = 1e-4
    temp_decay = 0.9995
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nh = np.clip(nh, 0.0, 1.0)
        nh = project_to_simplex(nh, best_h.size / 2.0)
        nc = compute_c5(nh)
        if nc < best_c or rng.random() < math.exp(-(nc - best_c) / max(temp, 1e-12)):
            best_h, best_c = nh.copy(), nc
        temp = max(temp * temp_decay, 1e-12)

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_simplex(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
