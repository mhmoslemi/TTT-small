    n = len(h)
    while True:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        max_corr = corr[idx]
        # stop if max_corr == 0? Not possible if n>0.
        # Determine mismatched indices S1 and S0.
        if k >= 0:
            # S1: i in [0, n-k-1] where h[i]=1 and h[i+k]==0
            i_range = np.arange(n - k)
            S1 = i_range[(h[i_range] == 1) & (h[i_range + k] == 0)]
            # S0: j in [0, n-k-1] where h[j]=0 and h[j+k]==1
            S0 = i_range[(h[i_range] == 0) & (h[i_range + k] == 1)]
        else:
            kk = -k
            i_range = np.arange(kk, n)
            # S1: i in [kk, n-1] where h[i]=1 and h[i-kk]==0
            S1 = i_range[(h[i_range] == 1) & (h[i_range - kk] == 0)]
            # S0: j in [kk, n-1] where h[j]=0 and h[j-kk]==1
            S0 = i_range[(h[i_range] == 0) & (h[i_range - kk] == 1)]

        if S1.size == 0 or S0.size == 0:
            break

        # Choose a pair to swap. We could pick any; maybe random
        i = S1[0]
        j = S0[0]
        h[i] = 0
        h[j] = 1
        # Continue loop
    return h

def compute_c5(h):
    """Return C5 for a (possibly fractional) vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass so that sum(h) == len(h)/2 (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def reduce_max_corr_binary(h):
    """
    Deterministic local improvement for binary vectors.
    Repeatedly swaps a 1/0 pair that decreases the current worst shift
    (the one with maximal correlation) until no such swap exists.
    """
    n = h.shape[0]
    while True:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        # all shifts attaining the maximum
        ks = np.where(corr == max_corr)[0] - (n - 1)
        improved = False
        for k in ks:
            if k >= 0:
                upper = n - k
                if upper <= 0:
                    continue
                rng = np.arange(upper)
                s1 = (h[rng] == 1) & (h[rng + k] == 0)
                s0 = (h[rng] == 0) & (h[rng + k] == 1)
                if s1.any() and s0.any():
                    i = rng[s1][0]
                    j = rng[s0][0]
                    h[i] = 0
                    h[j] = 1
                    improved = True
                    break

                kk = -k
                if kk >= n:
                    continue
                rng = np.arange(kk, n)
                s1 = (h[rng] == 1) & (h[rng - kk] == 0)
                s0 = (h[rng] == 0) & (h[rng - kk] == 1)
                if s1.any() and s0.any():
                    i = rng[s1][0]
                    j = rng[s0][0]
                    h[i] = 0
                    h[j] = 1
                    improved = True
                    break
        if not improved:
            break
    return h

def random_binary(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    h[rng.choice(n, n // 2, replace=False)] = 1
    return h

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate sizes
    cand_ns = [360, 480, 600, 720]
    if "initial_h_values" in globals():
        parent_n = len(initial_h_values)
        if parent_n % 2 == 0:
            cand_ns.append(parent_n)
        half = parent_n // 2
        if half % 2 == 0 and half >= 100:
            cand_ns.append(half)
    cand_ns = [n for n in cand_ns if n % 2 == 0 and 100 <= n <= 720]
    cand_ns = list(set(cand_ns))

    # main stochastic loop
    while time.time() < deadline:
        n = int(rng.choice(cand_ns))
        # ----- initialise binary vector -----
        if "initial_h_values" in globals() and n == len(initial_h_values):
            parent = np.array(initial_h_values, dtype=float)
            idx = np.argpartition(parent, -n // 2)[-n // 2 :]
            h = np.zeros(n, dtype=np.int8)
            h[idx] = 1
        else:
            h = random_binary(n, rng)

        # ----- deterministic reduction -----
        h = reduce_max_corr_binary(h)

        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_h, best_c, best_n = h.copy(), c, n

        # ----- random perturbations + reduction -----
        for _ in range(5):
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(h == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h2 = h.copy()
            h2[i] = 0
            h2[j] = 1
            h2 = reduce_max_corr_binary(h2)
            c2 = compute_c5(h2)
            if c2 < c - 1e-12:
                h, c = h2, c2
                if c2 < best_c - 1e-12:
                    best_h, best_c, best_n = h2.copy(), c2, n

        if time.time() > deadline - 0.5:
            break

    # fallback (should never trigger)
    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=np.int8)
        best_h[: n // 2] = 1
        best_c = compute_c5(best_h)
        best_n = n

    # final safety handling
    best_h = best_h.astype(float)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
