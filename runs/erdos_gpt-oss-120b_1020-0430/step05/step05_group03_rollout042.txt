def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # Use iterative redistribution as before
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def gradient_refine(h, rng, step=0.015, max_iter=12):
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 600) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, rng)

def binary_hill_climb(h: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000):
    n = h.size
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    best_max = int(np.max(corr))
    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        cur_max = int(np.max(corr))
        if cur_max < best_max:
            best_max, best_h = cur_max, h.copy()
            if best_max <= 97:
                break
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        if shift >= 0:
            zeros = (h[:n-shift] == 0)
            ones_shift = (h[shift:] == 1)
            I0 = np.where(zeros & ones_shift)[0]
            ones = (h[:n-shift] == 1)
            zeros_shift = (h[shift:] == 0)
            J0 = np.where(ones & zeros_shift)[0]
        else:
            l = -shift
            zeros_shift = (h[l:] == 0)
            ones = (h[:n-l] == 1)
            I0 = np.where(zeros_shift & ones)[0] + l
            ones_shift = (h[l:] == 1)
            zeros = (h[:n-l] == 0)
            J0 = np.where(ones_shift & zeros)[0] + l
        if I0.size and J0.size:
            i = rng.choice(I0)
            j = rng.choice(J0)
            h[i], h[j] = 1, 0
        else:
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = rng.choice(zeros_idx)
            j = rng.choice(ones_idx)
            h[i], h[j] = 1, 0
    return best_h, best_max

    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    dx = 2.0 / n
    # initialise binary vector

        idx = np.argsort(h0)[- (n // 2):]
        h_bin = np.zeros(n, dtype=np.uint8)
        h_bin[idx] = 1
    else:
        h_bin = None

    best_h = None
    best_c5 = 10.0
    # ---------- binary hill‑climb phase ----------
    while time.time() - start < 0.6 * budget_s:
        if best_h is not None and rng.random() < 0.3:
            h = best_h.copy()
            for _ in range(5):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if ones.size == 0 or zeros.size == 0:
                    break
                i = rng.choice(zeros)
                j = rng.choice(ones)
                h[i], h[j] = 1, 0
        elif h_bin is not None and rng.random() < 0.4:

        else:

        c5 = max_corr * dx
        if c5 < best_c5:

            c = compute_c5(h_cont)

        # final tiny stochastic tweaks
        while time.time() - start < budget_s * 0.999:
            grad = worst_gradient(best_h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))

                break

    best_h = enforce_sum(best_h, rng)
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(best_h.size)
