def project_cap_simplex(v, target):
    n = len(v)
    low = -np.max(np.abs(v)) - 1.0  # ensure sum=n
    high = np.max(v)  # sum=0
    for _ in range(50):
        tau = (low + high) / 2.0
        w = np.clip(v - tau, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = tau
        else:
            high = tau
    return np.clip(v - tau, 0.0, 1.0)

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a sampled step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_cap_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Projection of v onto the set {x : 0 <= x_i <= 1, sum x_i = target}
    using a simple bisection on the offset tau such that
    x_i = clip(v_i - tau, 0, 1) and sum x_i = target.
    """
    low = -np.max(np.abs(v)) - 1.0
    high = np.max(v) + 1.0
    for _ in range(50):
        tau = (low + high) * 0.5
        w = np.clip(v - tau, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = tau
        else:
            high = tau
    return np.clip(v - tau, 0.0, 1.0)

        vecs.append(interp)

        # binary version of the interpolated parent
        idx = np.argpartition(interp, -n // 2)[-n // 2 :]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        vecs.append(bin_vec)

    # random fractional (uniform)
    r = rng.random(n)
    vecs.append(r)

    # random fractional (biased toward 0)
    r2 = rng.random(n) ** 2
    vecs.append(r2)

    # deterministic half‑ones
    half = np.zeros(n)
    half[: n // 2] = 1.0
    vecs.append(half)

    # project all to the feasible capsimplex
    proj = []
    for v in vecs:
        v = np.clip(v, 0.0, 1.0)
        proj.append(project_cap_simplex(v, target))
    return proj


def gd_optimize(
    h0: np.ndarray,
    target: float,
    rng: np.random.Generator,
    max_iters: int,
    deadline: float,
    step0: float = 0.5,
):
    """
    Gradient descent on the worst shift with projection onto the capped simplex.
    Simple backtracking line search and occasional random transfer perturbations.
    """
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    step = step0
    n = h.size

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # find worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        # gradient of O_k
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]

        # try a step with backtracking
        improved = False
        for _ in range(12):
            h_new = project_cap_simplex(h - step * g, target)
            c_new = compute_c5(h_new)
            if c_new <= cur_c:
                improved = True
                break
            step *= 0.5
        if improved:
            h = h_new
            cur_c = c_new
            # modest step increase
            step = min(step * 1.07, 1.0)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            continue

        # if no improvement, perform a small random transfer
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer > 0.0:
            delta = max_transfer * rng.random() * 0.08
            h[i] -= delta
            h[j] += delta
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        # slightly increase step to try escaping again
        step = min(step * 1.13, 1.0)

    return best_h, best_c


def fractional_sa_optimize(h: np.ndarray, rng: np.random.Generator, deadline: float):
    """Simple transfer‑based simulated annealing on a feasible vector."""
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_imp = 0
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        max_tr = min(cur[i], 1.0 - cur[j])
        if max_tr <= 0:
            continue
        delta = max_tr * rng.random()
        cur[i] -= delta
        cur[j] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_imp = it
        else:
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it - last_imp > 20000:
            break
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.985

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate sample sizes (even numbers)
    n_vals = [180, 240, 300, 360, 420, 480, 540, 600, 660, 720]

    for n in n_vals:
        if time.time() >= deadline:
            break
        target = n / 2.0
        init_vecs = get_initial_vectors(n, rng)

        # first quick pass
        best_h_n = None
        best_c_n = np.inf
        for h0 in init_vecs:
            if time.time() >= deadline:
                break
            h_opt, c_opt = gd_optimize(
                h0, target, rng, max_iters=150, deadline=deadline, step0=0.5
            )
            if c_opt < best_c_n:
                best_c_n = c_opt
                best_h_n = h_opt.copy()
        # deeper refinement if we found something
        if best_h_n is not None and time.time() < deadline:
            h_ref, c_ref = gd_optimize(
                best_h_n, target, rng, max_iters=600, deadline=deadline, step0=0.5
            )
            if c_ref < best_c_n:
                best_c_n = c_ref
                best_h_n = h_ref.copy()

        if best_c_n < best_c:
            best_c = best_c_n
            best_h = best_h_n.copy()
            best_n = n

    # optional up‑sampling cycles
    while (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 1080
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = project_cap_simplex(h_up, n_up / 2.0)
        h_up_opt, c_up_opt = gd_optimize(
            h_up, n_up / 2.0, rng, max_iters=300, deadline=deadline, step0=0.5
        )
        if c_up_opt < best_c:
            best_c = c_up_opt
            best_h = h_up_opt.copy()
            best_n = n_up
        else:
            break

    # final SA polishing if time permits
    if best_h is not None and time.time() < deadline:
        h_sa, c_sa = fractional_sa_optimize(best_h, rng, deadline)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
            best_n = best_h.size

    # fallback in the unlikely event of no solution
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_cap_simplex(best_h, n / 2.0)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
