import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global state for SLSQP soft‑max wrapper
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def symmetrize(h: np.ndarray) -> np.ndarray:
    """Enforce h(x)+h(2‑x)=1 (h + complement of reversed). Guarantees sum = n/2."""
    return 0.5 * (h + 1.0 - h[::-1])

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth max for correlation objective."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # (m + log sum exp(bc - m)) / beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # quick redistribution to fix the sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def optimize_h(h0: np.ndarray, rng: np.random.Generator,
               start_time: float, total_budget: float):
    """Full optimisation pipeline for a single start vector."""
    n = h0.size
    h = symmetrize(np.clip(h0, 0.0, 1.0))
    best_h = h.copy()
    best_c = compute_c5(h)

    # ----- soft‑max SLSQP schedule -----
    betas = (20., 80., 200., 800., 3000., 8000., 20000.)
    for beta in betas:
        if time.time() - start_time > total_budget * 0.95:
            break
        max_iter = 800 if beta >= 200 else 500
        h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
        h = symmetrize(np.clip(h, 0.0, 1.0))
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if c <= 0.3808:
                return best_h, best_c

    # ----- guided hill‑climb (descent) -----
    remaining = total_budget - (time.time() - start_time)
    if remaining > 2.0:  # need a few seconds to be worthwhile
        # estimate safe iteration count (≈300 per second)
        max_iters = int(min(15000, remaining * 300))
        h_hc, c_hc = guided_hill_climb(best_h, rng,
                                        max_iters=max_iters, step_scale=0.7)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
            if c_hc <= 0.3808:
                return best_h, best_c

    return best_h, best_c

# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    n = 1024  # target resolution (<1k samples is preferred, 1024 still fine)
    target_sum = n // 2

    best_h = None
    best_c = np.inf

    # ----- collect initial seeds -----
    init_seeds = []

    if 'initial_h_values' in globals():
        h_small = np.array(initial_h_values, dtype=float)
        x_small = np.linspace(0.0, 2.0, h_small.size, endpoint=False)
        x_big = np.linspace(0.0, 2.0, n, endpoint=False)
        h_up = np.interp(x_big, x_small, h_small)
        h_up = symmetrize(np.clip(h_up, 0.0, 1.0))
        init_seeds.append(h_up)

    # one random symmetric start
    h_rand = symmetrize(rng.random(n))
    init_seeds.append(h_rand)

    # ----- optimise each seed while we have time -----
    for h0 in init_seeds:
        if time.time() - start > total_budget * 0.99:
            break
        h_opt, c_opt = optimize_h(h0, rng, start, total_budget)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
            if best_c <= 0.3808:
                break

    # ----- final fine‑tuning with tiny perturbations -----
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ----- sanitise output -----
    if best_h is not None:
        best_h = symmetrize(np.clip(best_h, 0.0, 1.0))
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_h.size)
