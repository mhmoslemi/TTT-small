def softmax_obj_grad(h, beta):

    max_corr = np.max(corr)
    # compute weights
    exps = np.exp(beta * (corr - max_corr))
    sum_exps = exps.sum()

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1] (simple iterative)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- gradient of worst shift ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ---------- guided perturbations ----------
def guided_perturb(h: np.ndarray, grad: np.ndarray,
                  rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h, best_c = h.copy(), compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- soft‑max objective for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction if sum deviates
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- projected gradient descent (continuation) ----------
def softmax_obj_grad(h: np.ndarray, beta: float):
    """Return soft‑max value and its gradient w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    w = exps / exps.sum()
    conv1 = np.convolve(w[::-1], 1.0 - h, mode="full")
    conv2 = np.convolve(w, h, mode="full")
    start = n - 1
    grad = conv1[start:start + n] - conv2[start:start + n]
    obj = (max_corr + np.log(exps.sum())) / beta
    return obj, grad


def project_onto_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto [0,1]^n with prescribed sum (water‑filling)."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) * 0.5
    x = np.clip(v + lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute residual on non‑boundary entries
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(x - 0.5))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def pgd_optimize(h0: np.ndarray, betas, iters, alphas,
                 target_sum: float, start: float, time_limit: float):
    """Projected GD with increasing β."""
    h = h0.copy()
    for beta, its, alpha in zip(betas, iters, alphas):
        for _ in range(its):
            if time.time() - start > time_limit:
                return h
            _, grad = softmax_obj_grad(h, beta)
            step = alpha / (beta if beta > 0 else 1.0)
            h = h - step * grad
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_sum(h, target_sum)
    return h
