def run(seed=42, budget_s=1100,...):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99
    n = 360  # could adjust based on initial_h = len(initial_h_values) if present

    # get initial h: if parent present else uniform random.
    best_h = None
    best_c = np.inf

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            n = parent.size  # adopt.
        c_parent = compute_c5(parent)
        best_h = parent.copy()
        best_c = c_parent

    # Define function to evaluate candidate and update best
    def try_candidate(h):
        nonlocal best_h, best_c
        # clip and adjust sum
        h = np.clip(h, 0.0, 1.0)
        adjust_sum_exact(h, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # First perform some random restarts and smooth-max optimization
    num_restarts = 10
    # Determine allowed restarts based on time.
    for r in range(num_restarts):
        if time.time() > deadline:
            break
        # create random perturbation of best_h (or parent if best_h None)
        if best_h is not None:
            base = best_h
        else:
            base = np.full(n, 0.5)
        # generate noise scaled decreasing with r maybe
        noise_scale = 0.2 * (0.7 ** r)  # decreasing
        h0 = base + rng.uniform(-noise_scale, noise_scale, size=n)
        h0 = np.clip(h0, 0.0, 1.0)
        adjust_sum_exact(h0, rng)
        # smooth max cascade
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() > deadline:
                break
            # allocate iteration count based on beta; increasing for higher betas
            max_iter = 400 if beta < 200 else 800
            try:
                h_opt = slsqp_smooth_max(h0, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            h0 = h_opt  # continue with optimized h for next beta
        # after cascade, update best
        try_candidate(h0)
        # local hill climb using random swaps (pairwise) for a while
        # we implement small moves: pick two indices i,j, exchange delta
        # using random delta up to min(0.05, h[i], 1-h[j]), etc. Evaluate.
        # iterate until time runs out or iteration limit
        it_limit = 2000
        for it in range(it_limit):
            if time.time() > deadline:
                break
            # pick two indices
            i, j = rng.choice(n, size=2, replace=False)
            # if h[i] > h[j], move delta from i to j
            if h0[i] <= h0[j]:
                continue
            max_delta = min(h0[i], 1.0 - h0[j], 0.02)  # small step
            if max_delta <= 1e-9:
                continue
            delta = rng.uniform(0.0, max_delta)
            h_new = h0.copy()
            h_new[i] -= delta
            h_new[j] += delta
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-9:  # improvement
                h0 = h_new
                best_c = c_new
                best_h = h_new.copy()
            # else perhaps accept if improves over previous local (optional)
        # end of restart loop

    # After loops, final adjust and compute final C
    if best_h is None:
        # fallback: simple half 1s
        best_h = np.zeros(n, dtype=float)
        best_h[: n//2] = 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Enforce Σh = n/2 exactly by moving mass between coordinates."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def greedy_binary(n, rng):
    """Binary construction with greedy minimisation of the max overlap."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1

def slsqp_smooth_max(h0, beta, rng, max_iter=500):
    """SLSQP optimisation of the smooth‑max surrogate for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        expv = np.exp(beta * (corr - m))
        return (m + np.log(np.sum(expv))) / beta

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        expv = np.exp(beta * (corr - m))
        w = expv / np.sum(expv)
        f = 1.0 - x
        w_rev = w[::-1]
        term1 = np.convolve(w_rev, f, mode="full")[n - 1 : n - 1 + n]
        term2 = np.convolve(w, x, mode="full")[n - 1 : n - 1 + n]
        return term1 - term2

    res = minimize(
        obj,
        h0,
        jac=grad,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

        if time.time() > deadline:
            break
        if beta < 100:
            max_iter = 200
        elif beta < 2000:
            max_iter = 400
        else:
            max_iter = 800
        try:
            h0 = slsqp_smooth_max(best_h, beta, rng, max_iter=max_iter)
        except Exception:
            continue
        c_now = compute_c5(h0)
        if c_now < best_c:
            best_c = c_now
            best_h = h0.copy()

    # ------------------------------------------------------------------
    # a few random restarts to escape possible local minima
    for _ in range(3):
        if time.time() > deadline:
            break
        h_rand = rng.uniform(0.0, 1.0, size=n)
        adjust_sum_exact(h_rand, rng)
        for beta in beta_list:
            if time.time() > deadline:
                break
            if beta < 100:
                max_iter = 200
            elif beta < 2000:
                max_iter = 400
            else:
                max_iter = 800
            try:
                h_rand = slsqp_smooth_max(h_rand, beta, rng, max_iter=max_iter)
            except Exception:
                break
        c_rand = compute_c5(h_rand)
        if c_rand < best_c:
            best_c = c_rand
            best_h = h_rand.copy()

    # ------------------------------------------------------------------
    # final local refinement
    if time.time() < deadline:
        best_h = local_refine(best_h, max_iters=20000, rng=rng, deadline=deadline)

    # ------------------------------------------------------------------
    # final safety adjustment and result
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
