import time
import numpy as np

def compute_O(h):
    """Linear autocorrelation based overlap O(k)=sum h_i(1-h_{i+k})."""
    n = h.size
    target = h.sum()
    L = 1 << ((3 * n - 2).bit_length())          # length for convolutions
    H = np.fft.rfft(h, n=L)
    autocorr = np.fft.irfft(H * np.conj(H), n=L)[: 2 * n - 1]
    O = target - autocorr
    O[n - 1] = 0.0                                 # lag zero must be zero
    return O

def compute_c5(h):
    """C5 value from a step function."""
    n = h.size
    O = compute_O(h)
    return float(O.max() * (2.0 / n))

def project_onto_bounds_and_sum(x, target, max_iter=20):
    """Project x onto [0,1]^n while keeping sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        add = diff / idx.size
        x[idx] += add
        x = np.clip(x, 0.0, 1.0)
    return np.clip(x, 0.0, 1.0)

def softmax_and_grad(h, beta):
    """Smooth‑max of O and its gradient."""
    n = h.size
    target = h.sum()
    L = 1 << ((3 * n - 2).bit_length())
    H = np.fft.rfft(h, n=L)

    # overlap O
    autocorr = np.fft.irfft(H * np.conj(H), n=L)[: 2 * n - 1]
    O = target - autocorr
    O[n - 1] = 0.0

    # smooth max
    betaO = beta * O
    m = betaO.max()
    exp_terms = np.exp(betaO - m)
    sumexp = exp_terms.sum()
    S = (m + np.log(sumexp)) / beta
    w = exp_terms / sumexp                                   # weights, sum w = 1

    # convolution terms for gradient
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(W_rev * H, n=L)
    term1 = conv_rev[n - 1 : n - 1 + n]

    W = np.fft.rfft(w, n=L)
    conv = np.fft.irfft(W * H, n=L)
    term2 = conv[n - 1 : n - 1 + n]

    grad = 1.0 - term1 - term2
    return S, O, grad

def gradient_step(h, beta, alpha, target):
    """One projected gradient descent step on smooth‑max."""
    S, O, grad = softmax_and_grad(h, beta)
    # make direction zero‑sum to keep the linear constraint
    d = grad - grad.mean()
    h_new = h - alpha * d
    h_proj = project_onto_bounds_and_sum(h_new, target)
    return h_proj, S, O

def greedy_binary(n, rng):
    """Greedy construction of a binary vector with n/2 ones."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h, O
