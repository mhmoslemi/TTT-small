def project_onto_bounds_and_sum(v, S):
    # clip to [0,1]
    v_clipped = np.clip(v, 0, 1)
    # compute surplus: sum(v_clipped) - S
    surplus = v_clipped.sum() - S
    if abs(surplus) < 1e-12:
        return v_clipped
    # Let free = indices where 0 < v_clipped < 1
    # We need to add -surplus distributed among free without violating bounds.
    # We'll iteratively adjust:
    i = np.arange(len(v))
    free = (v_clipped > 0) & (v_clipped < 1)
    # While surplus not zero and free exists:
    while True:
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # cannot adjust, return v_clipped (should be close)
            break
        # distribute equally
        delta = -surplus / free_idx.size
        # update
        v_clipped[free_idx] += delta
        # clip again
        v_clipped = np.clip(v_clipped, 0, 1)
        surplus = v_clipped.sum() - S
        if abs(surplus) < 1e-12:
            break
        # recompute free
        free = (v_clipped > 0) & (v_clipped < 1)
    return v_clipped

def compute_c5_fft(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx using FFT (full correlation)."""
    n = h.shape[0]
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    Y = np.empty_like(H)
    Y[0] = L                     # FFT of constant 1 on length L
    Y[1:] = -H[1:]               # FFT(1 - h) = FFT(1) - FFT(h)
    conv = np.fft.irfft(H * np.conj(Y), n=L)
    max_val = np.max(conv[: 2 * n - 1])
    return float(max_val * (2.0 / n))


def project_onto_bounds_and_sum(v, target_sum):
    """Clip to [0,1] and enforce Σv = target_sum by water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    diff = v.sum() - target_sum
    if abs(diff) < 1e-12:
        return v
    # simple iterative water‑filling (fast for n ≤ 720)
    for _ in range(10):
        free = (v > 0) & (v < 1)
        free_cnt = free.sum()
        if free_cnt == 0:
            break
        delta = diff / free_cnt
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
        diff = v.sum() - target_sum
        if abs(diff) < 1e-12:
            break
    return v


def worst_shifts_and_grad(h, corr, eps=1e-12):
    """
    Given h and its full correlation `corr` (length 2n‑1),
    return the subgradient of the current worst shift(s) and the maximal value.
    """
    n = h.shape[0]
    max_val = np.max(corr)
    idxs = np.where(corr >= max_val - eps)[0]
    ks = idxs - (n - 1)          # shift values
    g = np.zeros_like(h)
    for k in ks:
        if k >= 0:
            g[: n - k] += 1.0
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        else:
            kp = -k
            g[kp:] += 1.0
            g[kp:] -= h[: n - kp]
            g[: n - kp] -= h[kp:]
    if len(ks) > 1:
        g /= len(ks)
    return g, max_val


def subgrad_optimize(h0, rng, time_deadline,
                     max_iters=50000, step0=0.6, decay=0.9995):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    step = step0
    it = 0
    L = 2 * n
    while time.time() < time_deadline and it < max_iters:
        # correlation via FFT (single forward FFT)
        H = np.fft.rfft(h, n=L)
        Y = np.empty_like(H)
        Y[0] = L
        Y[1:] = -H[1:]
        conv = np.fft.irfft(H * np.conj(Y), n=L)
        corr = conv[: 2 * n - 1]

        g, cur_max = worst_shifts_and_grad(h, corr)
        # gradient step
        h_new = h - step * g
        h_new = project_onto_bounds_and_sum(h_new, target)

        cur_c = cur_max * (2.0 / n)    # current max overlap
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h_new.copy()
        h = h_new
        step *= decay
        it += 1
    return best_h, best_c


def binarize_h(h):
    """Return a binary vector with exactly n/2 ones (top values become 1)."""
    n = h.shape[0]
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx] = 1.0
    return bin_h


def binary_hill_climb(h_bin, rng, time_deadline, max_no_improve=5000):
    """Simple hill‑climb on binary vectors (pairwise swaps)."""
    n = h_bin.shape[0]
    cur_h = h_bin.copy()
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    no_improve = 0
    while time.time() < time_deadline and no_improve < max_no_improve:
        i = rng.integers(n)
        while cur_h[i] != 1:
            i = rng.integers(n)
        j = rng.integers(n)
        while cur_h[j] != 0:
            j = rng.integers(n)
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5_fft(cur_h)
        if c < cur_c - 1e-12:
            cur_c = c
            no_improve = 0
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_improve += 1
    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.98  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Candidate sample sizes (even)
    # ------------------------------------------------------------------
    candidate_ns = [720, 600, 480, 360, 240, 180]
    candidate_ns = [n for n in candidate_ns if n % 2 == 0]

    # ------------------------------------------------------------------
    # Helper to build start vectors for a given n
    # ------------------------------------------------------------------
    def build_start_vectors(n):
        vectors = []

        # parent (if available) via repeat or interpolation
        if "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
            if n % parent.size == 0:
                repeat = n // parent.size
                vectors.append(np.repeat(parent, repeat))
            # interpolation version
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            vectors.append(interp)

        # random binary
        bin_vec = np.zeros(n, dtype=float)
        bin_vec[rng.choice(n, n // 2, replace=False)] = 1.0
        vectors.append(bin_vec)

        # random fractional (scaled to sum = n/2)
        rand_frac = rng.random(n)
        rand_frac *= (n / 2.0) / rand_frac.sum()
        rand_frac = project_onto_bounds_and_sum(rand_frac, n / 2.0)
        vectors.append(rand_frac)

        return vectors

    # ------------------------------------------------------------------
    # Main search over sample sizes
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() >= deadline:
            break

        start_vecs = build_start_vectors(n)

        for base_h in start_vecs:
            if time.time() >= deadline:
                break

            # ensure feasibility
            h0 = project_onto_bounds_and_sum(base_h, n / 2.0)

            # ----------- sub‑gradient descent (fractional) ---------------
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
            h_frac, c_frac = subgrad_optimize(h0, rng, sub_dead,
                                              max_iters=60000,
                                              step0=0.7,
                                              decay=0.9994)
            if c_frac < best_c:
                best_h, best_c, best_n = h_frac.copy(), c_frac, n

            # ----------- binary refinement --------------------------------
            h_bin = binarize_h(h_frac)
            bin_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
            h_bin_opt, c_bin = binary_hill_climb(h_bin, rng, bin_dead)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin_opt.copy(), c_bin, n

            # ----------- occasional extra SLSQP sweep (if time permits) ----
            # Re‑use the earlier SLSQP cascade if we still have ample time.
            if time.time() < deadline - 5.0:
                # small cascade: only a few betas
                from scipy.optimize import minimize

                def softmax_obj(x, beta):
                    H = np.fft.rfft(x, n=2 * n)
                    Y = np.empty_like(H)
                    Y[0] = 2 * n
                    Y[1:] = -H[1:]
                    conv = np.fft.irfft(H * np.conj(Y), n=2 * n)
                    corr = conv[: 2 * n - 1]
                    bc = beta * corr
                    m = np.max(bc)
                    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

                def sum_eq(x):
                    return np.sum(x) - n / 2.0

                # start from the best fractional (or binary) we have
                start = best_h if best_n == n else h_frac
                bounds = [(0.0, 1.0)] * n
                cons = {"type": "eq", "fun": sum_eq}
                for beta in (20, 200, 2000):
                    if time.time() >= deadline:
                        break
                    try:
                        res = minimize(
                            softmax_obj,
                            start,
                            args=(beta,),
                            method="SLSQP",
                            bounds=bounds,
                            constraints=cons,
                            options={"maxiter": 300, "ftol": 1e-12, "disp": False},
                        )
                        x = np.clip(res.x, 0.0, 1.0)
                        x = project_onto_bounds_and_sum(x, n / 2.0)
                        c_now = compute_c5_fft(x)
                        if c_now < best_c:
                            best_h, best_c, best_n = x.copy(), c_now, n
                        start = x
                    except Exception:
                        pass

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 180
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_h = project_onto_bounds_and_sum(best_h, n / 2.0)
        best_c = compute_c5_fft(best_h)
        best_n = n

    # final safety projection
    best_h = project_onto_bounds_and_sum(best_h, best_n / 2.0)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
