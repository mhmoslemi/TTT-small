def ccp_optimize_at_n(n, h0, time_budget, rng):
    start = time.time()
    # upsample / downsample h0 to length n
    if h0.size != n:
        x_old = np.linspace(0., 2., h0.size, endpoint=False)
        x_new = np.linspace(0., 2., n, endpoint=False)
        h0 = np.interp(x_new, x_old, h0)
    h = np.clip(h0, 0., 1.)
    h = enforce_sum(h, rng)
    # CVXPY setup
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    old_h_param = cp.Parameter(n)
    old_norm = cp.Parameter()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    # build constraints for each shift except possibly k=0
    for k in range(n):
        expr = cp.sum(old_h_param * cp.roll(h_var, -k)) + cp.sum(h_var * cp.roll(old_h_param, -k)) - old_norm
        constraints.append(expr >= n/2 - t_var)
    prob = cp.Problem(cp.Maximize(t_var), constraints)
    best_h_local = h.copy()
    best_c_local = compute_c5(h)
    # Iterations
    while time.time() - start < time_budget:
        old_h_param.value = h
        old_norm.value = np.dot(h, h)
        # Solve with warm_start
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        except cp.error.SolverError:
            break
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            break
        h_new = np.clip(h_var.value, 0., 1.)
        # enforce sum exactly
        diff = n/2 - h_new.sum()
        if abs(diff) > 1e-10:
            # small adjustment by distributing diff to random entries respecting bounds
            # call enforce_sum using the same routine
            h_new = enforce_sum(h_new, rng)
        c5_new = compute_c5(h_new)
        if c5_new < best_c_local:
            best_c_local = c5_new
            best_h_local = h_new.copy()
        # check convergence
        if np.max(np.abs(h_new - h)) < 1e-8:
            break
        h = h_new
    return best_h_local, best_c_local
```

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.5):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        # optionally decay scale
        if (it + 1) % (max_iters // 5 + 1) == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c
```

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, cvxpy as cp
    rng = np.random.default_rng(seed)
    start = time.time()
    # initial
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 256
        h0 = np.zeros(n0)
        h0[rng.choice(n0, n0//2, replace=False)] = 1.0
    h0 = np.clip(h0,0,1)
    h0 = enforce_sum(h0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    # schedule
    schedule = [128, 192, 256, 384, 512]
    for n in schedule:
        if time.time() - start > budget_s * 0.9:
            break
        remaining = budget_s - (time.time() - start)
        # allocate at most 0.2 of remaining to this stage
        stage_budget = min(remaining * 0.2, remaining - 5)
        if stage_budget < 1.0:
            break
        h_stage, c_stage = ccp_optimize_at_n(n, best_h, stage_budget, rng)
        if c_stage < best_c:
            best_h, best_c = h_stage.copy(), c_stage
        # short hill-climb refinement
        if time.time() - start < budget_s * 0.93:
            h_stage2, c_stage2 = guided_hill_climb(best_h, rng, max_iters= int(1500 * 256 / n), step_scale=0.6)
            if c_stage2 < best_c:
                best_h, best_c = h_stage2.copy(), c_stage2
    # final stochastic hill climbing until time up
    while time.time() - start < budget_s * 0.99:
        # random swap of small epsilon
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc
    # final sanity
    best_h = np.clip(best_h, 0, 1)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:

        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if add == 0.0:
                continue
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


# ----------------------------------------------------------------------
# Gradient‑based hill‑climb utilities
# ----------------------------------------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1000, step_scale: float = 0.5):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay_interval = max(1, max_iters // 5)

    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay_interval == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                if add == 0.0:
                    continue
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 512
        h = np.zeros(n0)
        h[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 1: soft‑max SLSQP passes (rough global improvement)
    # ------------------------------------------------------------------
    for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0):
        if time.time() - start > budget_s * 0.4:
            break
        h = slsqp_optimize(best_h, beta, rng, max_iter=500)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ------------------------------------------------------------------
    # Phase 2: multiple guided hill‑climb restarts (exploration)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.8:
        # start from current best with a small random disturbance
        h0 = best_h.copy()
        # random swaps
        for _ in range(8):
            i, j = rng.integers(0, h0.size, size=2)
            if h0[i] != h0[j]:
                h0[i], h0[j] = h0[j], h0[i]
        # small epsilon transfers
        for _ in range(4):
            i, j = rng.integers(0, h0.size, size=2)
            eps = rng.random() * 0.04
            max_eps = min(h0[i], 1.0 - h0[j])
            eps = min(eps, max_eps)
            if eps > 0.0:
                h0[i] -= eps
                h0[j] += eps
        h0 = enforce_sum(h0, rng)

        max_iters = int(2000 * 256 / h0.size)
        h_clim, c_clim = guided_hill_climb(h0, rng, max_iters=max_iters,
                                           step_scale=0.6)
        if c_clim < best_c:
            best_c, best_h = c_clim, h_clim.copy()

    # ------------------------------------------------------------------
    # Phase 3: long final hill‑climb (intensive polishing)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 1.0:
        max_iters = int(min(30000, remaining * 30))
        h_final, c_final = guided_hill_climb(best_h, rng,
                                             max_iters=max_iters,
                                             step_scale=0.5)
        if c_final < best_c:
            best_c, best_h = c_final, h_final.copy()

    # ------------------------------------------------------------------
    # Phase 4: simulated‑annealing style random ε‑transfers
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.99:
        T = max(1e-4, (budget_s - (time.time() - start)) / budget_s)
        i, j = rng.integers(0, best_h.size, size=2)
        eps = rng.random() * 0.02
        max_eps = min(best_h[i], 1.0 - best_h[j])
        eps = min(eps, max_eps)
        if eps == 0.0:
            continue
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        c_new = compute_c5(nh)
        if c_new < best_c or rng.random() < math.exp(-(c_new - best_c) / T):
            best_h, best_c = nh, c_new

    # ------------------------------------------------------------------
    # Final clean‑up and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
