def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # If diff is zero within tolerance, return
    while abs(diff) > 1e-12:
        # choose random index
        i = rng.integers(0, n)
        # compute allowable addition/subtraction
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
```

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad
```

        return h.copy()

    nh = h.copy()

def binary_swap_local_search(h, rng, max_iters=2000):
    # ensure binary 0/1 floats
    n = h.size
    ones = np.flatnonzero(h > 0.5).tolist()
    zeros = np.flatnonzero(h < 0.5).tolist()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            # update lists
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return h, best_c
```

                                    maxiter=max_iter,
                                    polish=False,
                                    updating='deferred',

import time, math, numpy as np
from scipy.optimize import differential_evolution, LinearConstraint

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with spacing 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def deterministic_hill_climb(h: np.ndarray, rng: np.random.Generator,
                             max_iters: int = 1000, min_eps: float = 1e-12):
    """Greedy max‑epsilon moves along the worst‑shift subgradient."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for _ in range(max_iters):
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] - grad[j] < 1e-12:
            break
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps < min_eps:
            break
        improved = False
        for eps in (max_eps, max_eps * 0.5):
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            c = compute_c5(nh)
            if c < best_c - 1e-12:
                best_h, best_c = nh, c
                improved = True
                break
        if not improved:
            break
    return best_h, best_c


def binary_swap_local_search(h: np.ndarray, rng: np.random.Generator,
                             max_iters: int = 2000):
    """Swap 1↔0 entries to lower C5 (assumes h is binary up to tolerance)."""
    n = h.size
    ones = np.flatnonzero(h > 0.5).tolist()
    zeros = np.flatnonzero(h < 0.5).tolist()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            h[i], h[j] = 1.0, 0.0
    return h, best_c


def de_optimize(n: int, rng: np.random.Generator,
                max_iter: int = 30) -> np.ndarray:
    """Differential evolution for a small n with the exact C5 objective."""
    bounds = [(0.0, 1.0)] * n
    constraint = LinearConstraint(np.ones((1, n)), [n / 2.0], [n / 2.0])
    result = differential_evolution(compute_c5,
                                    bounds,
                                    constraints=(constraint,),
                                    maxiter=max_iter,
                                    polish=False,
                                    updating='deferred',
                                    seed=int(rng.random() * 2**32))
    return enforce_sum(result.x, rng)
