def compute_c5(h):
    n = h.shape[0]
    if n < 256:
        corr = np.correlate(h, 1 - h, mode='full')
    else:
        # compute full convolution using FFT
        N = 2*n - 1
        # Next power of 2 for speed
        Nfft = 1 << (N - 1).bit_length()
        H = np.fft.rfft(h, n=Nfft)
        G = np.fft.rfft(1 - h, n=Nfft)
        conv = np.fft.irfft(H * G, n=Nfft)[:N]
        corr = conv
    return np.max(corr) * (2.0 / n)

def enforce_sum(h, rng):
    target = h.size / 2.0
    # scale
    s = target / h.sum()
    if s != 1.0:
        h = np.clip(h * s, 0.0, 1.0)
    # adjust residual
    diff = target - h.sum()
    tol = 1e-12
    while abs(diff) > 1e-12:
        i = rng.integers(h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, max_iter=800):
    n = h0.size
    target = n/2
    bounds = [(0.,1.)]*n
    cons = {'type':'eq', 'fun': lambda x: x.sum() - target}
    # objective fast
    def obj(x):
        # compute correlation full via FFT
        h = x
        # compute corr using compute_corr (maybe use external)
        corr = compute_corr(h)
        bc = beta * corr
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp': False})
    x = np.clip(res.x, 0., 1.)
    return enforce_sum(x, np.random.default_rng())

def compute_corr(h):
    n = h.size
    N = 2*n - 1
    # choose next power of 2 for speed
    Nfft = 1 << (N - 1).bit_length()
    H = np.fft.rfft(h, n=Nfft)
    G = np.fft.rfft(1 - h, n=Nfft)
    conv = np.fft.irfft(H * G, n=Nfft)[:N]
    return conv

import time, math, numpy as np
from scipy.optimize import minimize, LinearConstraint

# --------------------------------------------------------------
# correlation utilities (FFT based for speed)
def compute_corr(h: np.ndarray) -> np.ndarray:
    n = h.size
    N = 2 * n - 1
    Nfft = 1 << (N - 1).bit_length()
    H = np.fft.rfft(h, n=Nfft)
    G = np.fft.rfft(1 - h, n=Nfft)
    conv = np.fft.irfft(H * G, n=Nfft)[:N]
    return conv


def compute_c5(h: np.ndarray) -> float:
    """exact discrete C5 for a step function on [0,2]"""
    n = h.size
    corr = compute_corr(h)
    return corr.max() * (2.0 / n)


# --------------------------------------------------------------
# feasibility helpers
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    target = h.size / 2.0
    # scaling step
    s = target / h.sum()
    if s != 1.0:
        h = np.clip(h * s, 0.0, 1.0)
    # redistribution of the tiny residual
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    h = enforce_sum(h, rng)
    return h

# gradient of the worst (max) shift
def worst_gradient(h: np.ndarray) -> np.ndarray:

    else:   # k < 0

        grad[kk:] -= h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# soft‑max SLSQP refinement
def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 800) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def obj(x):
        corr = compute_corr(x)
        bc = beta * corr
        m = bc.max()
        # log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())

        # a quick SLSQP polish on the binary output
        if time.time() - start < 0.95 * budget_s:
            h = slsqp_optimize(best_h, 200, max_iter=600)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

    # ------------------------------------------------------------------
    # 4) final stochastic fine‑tuning
    while time.time() - start < 0.99 * budget_s:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c = nh.copy(), c

    # ------------------------------------------------------------------
    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
