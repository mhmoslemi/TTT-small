def lp_refine(h, max_iter, time_budget, start_time, rng):
    n = h.size
    best_h = h.copy()
    best_t = np.min(np.correlate(best_h, best_h, mode='full')[n-1:])
    for it in range(max_iter):
        if time.time() - start_time > time_budget:
            break
        h_old = best_h.copy()
        # autocorrelation
        corr_full = np.correlate(h_old, h_old, mode='full')
        c0 = corr_full[n-1:]  # length n
        # build problem
        x = cp.Variable(n)
        t = cp.Variable()
        constraints = [x >= 0, x <= 1, cp.sum(x) == n/2]
        for k in range(n):
            if k == 0:
                coeff = 2 * h_old
            else:
                coeff = np.zeros(n)
                coeff[:n - k] = h_old[k:]   # h_old_{i+k}
                coeff[k:] += h_old[:n - k]  # h_old_{i - k}
            constraints.append(coeff @ x >= t + c0[k])
        prob = cp.Problem(cp.Maximize(t), constraints)
        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=500, abstol=1e-9, reltol=1e-9)
        except Exception:
            break
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            break
        new_h = np.clip(x.value, 0.0, 1.0)
        new_h = enforce_sum(new_h, rng)
        new_corr_full = np.correlate(new_h, new_h, mode='full')
        new_c0 = new_corr_full[n-1:]
        new_t = np.min(new_c0)
        if new_t > best_t + 1e-9:
            best_h = new_h
            best_t = new_t
        else:
            # no improvement, stop
            break
    return best_h
```

import time
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize, LinearConstraint

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if abs(add) < 1e-15:
            break
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


# ---------- gradient / hill‑climb ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- LP refinement ----------
def lp_refine(h: np.ndarray, max_iter: int, time_budget: float,
              start_time: float, rng: np.random.Generator) -> np.ndarray:
    """Sequential linear‑programming refinement to raise min autocorrelation."""
    n = h.size
    best_h = h.copy()
    best_t = np.min(np.correlate(best_h, best_h, mode='full')[n - 1 :])
    for it in range(max_iter):
        if time.time() - start_time > time_budget:
            break
        h_old = best_h.copy()
        corr_full = np.correlate(h_old, h_old, mode='full')
        c0 = corr_full[n - 1 :]               # length n, autocorrelation values
        # CVXPY variables
        x = cp.Variable(n)
        t = cp.Variable()
        cons = [x >= 0, x <= 1, cp.sum(x) == n / 2]
        for k in range(n):
            if k == 0:
                coeff = 2 * h_old
            else:
                coeff = np.zeros(n)
                coeff[: n - k] = h_old[k:]          # h_old_{i+k}
                coeff[k:] += h_old[: n - k]          # h_old_{i-k}
            cons.append(coeff @ x >= t + c0[k])
        prob = cp.Problem(cp.Maximize(t), cons)
        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=500,
                       abstol=1e-9, reltol=1e-9)
        except Exception:
            break
        if prob.status not in ("optimal", "optimal_inaccurate"):
            break
        new_h = np.clip(x.value, 0.0, 1.0)
        new_h = enforce_sum(new_h, rng)
        new_t = np.min(np.correlate(new_h, new_h, mode='full')[n - 1 :])
        if new_t > best_t + 1e-9:
            best_h = new_h
            best_t = new_t
        else:
            break
    return best_h


# ---------- binary handling ----------
def round_to_binary(h: np.ndarray) -> np.ndarray:
    """Select the n/2 largest entries and set them to 1."""
    n = h.size
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[: n // 2]] = 1.0
    return b


def binary_swap_hill_climb(h: np.ndarray, rng: np.random.Generator,
                           max_iters: int = 2000) -> np.ndarray:
    """Swap 1↔0 to increase the minimal autocorrelation."""
    n = h.size
    O = np.correlate(h, h, mode='full')[n - 1 :]  # autocorrelation (sum of products)
    best_min = O.min()
    best_h = h.copy()
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    if len(ones) == 0 or len(zeros) == 0:
        return h
    for _ in range(max_iters):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # build contribution vectors
        a = np.zeros(n)
        a[: n - i] = h[i:]
        b = np.zeros(n)
        b[: i + 1] = h[i::-1]
        c = np.zeros(n)
        c[: n - j] = h[j:]
        d = np.zeros(n)
        d[: j + 1] = h[j::-1]
        delta = - (a + b) + (c + d)
        new_O = O + delta
        new_min = new_O.min()
        if new_min > best_min + 1e-9:
            # accept swap
            h[i] = 0
            h[j] = 1
            O = new_O
            best_min = new_min
            best_h = h.copy()
            # update index arrays
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
    return best_h
