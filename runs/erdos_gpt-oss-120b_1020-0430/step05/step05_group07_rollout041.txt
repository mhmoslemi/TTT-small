def hillclimb_binary(h_start, n, M, idx, deadline, rng):
    import numpy as np, time, math
    h = h_start.copy()
    # compute A with FFT
    # Use float FFT for speed, then round
    H = np.fft.fft(h.astype(np.float64))
    A = np.rint(np.fft.ifft(H * np.conj(H)).real).astype(np.int64)
    # Ensure A[0] = M
    A[0] = M
    min_A = int(np.min(A[1:]))  # exclude k=0
    cur_c = (M - min_A) * (2.0 / n)
    best_h = h.copy()
    best_c = cur_c
    best_min_A = min_A

    T0 = cur_c * 0.5 + 1e-6  # avoid zero
    last_improve_it = 0
    it = 0
    # loop
    while time.time() < deadline:
        # temperature
        T = max(T0 * (0.9995 ** it), 1e-12)
        # get ones and zeros indices
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta using idx
        # compute h[(idx + j) % n] etc.
        # Since idx is np.arange(n), adding j and taking mod n yields vector
        h_jp = h[(idx + j) % n]
        h_jm = h[(idx - j) % n]
        h_ip = h[(idx + i) % n]
        h_im = h[(idx - i) % n]
        delta = h_jp + h_jm - h_ip - h_im
        new_A = A + delta
        new_min_A = int(np.min(new_A[1:]))  # exclude k=0
        if new_min_A > min_A or rng.random() < math.exp((new_min_A - min_A) / T):
            # accept
            h[i] = 0
            h[j] = 1
            A = new_A
            min_A = new_min_A
            cur_c = (M - min_A) * (2.0 / n)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                best_min_A = min_A
                last_improve_it = it
        it += 1
        if it - last_improve_it > 400000:
            break

    return best_h, best_c

import time
import math
import numpy as np

def _compute_c5(h):
    """C5 for a given vector h (float or int)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _binary_sa(h_start, n, M, idx, deadline, rng):
    """
    Simulated‑annealing on a binary vector.
    Maximises the minimum cyclic autocorrelation A_k = Σ h_i h_{i+k},
    which is equivalent to minimising C5 = (M - min A_k) * 2/n.
    """
    h = h_start.astype(np.uint8).copy()
    # initial autocorrelation via FFT (fast and exact for binary)
    H = np.fft.fft(h.astype(np.float64))
    A = np.rint(np.fft.ifft(H * np.conj(H)).real).astype(np.int64)
    # enforce exact A[0] = M (sum of h)
    A[0] = M
    min_A = int(np.min(A[1:]))                     # ignore k=0 (always M for binary)
    cur_c = (M - min_A) * (2.0 / n)

    best_h = h.copy()
    best_c = cur_c
    best_min_A = min_A

    T0 = max(cur_c * 0.5, 1e-12)
    last_improve = 0
    it = 0
    # main SA loop
    while time.time() < deadline:
        # temperature schedule
        T = max(T0 * (0.9995 ** it), 1e-12)

        # pick a random 1 and a random 0
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # delta autocorrelation for the swap i<->j
        # h[(idx + j) % n]  gives vector of h_{j+k}
        h_jp = h[(idx + j) % n]
        h_jm = h[(idx - j) % n]
        h_ip = h[(idx + i) % n]
        h_im = h[(idx - i) % n]
        delta = h_jp + h_jm - h_ip - h_im          # integer vector, length n
        new_A = A + delta
        new_min_A = int(np.min(new_A[1:]))

        # Metropolis acceptance
        if new_min_A > min_A or rng.random() < math.exp((new_min_A - min_A) / T):
            # accept the swap
            h[i] = 0
            h[j] = 1
            A = new_A
            min_A = new_min_A
            cur_c = (M - min_A) * (2.0 / n)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                best_min_A = min_A
                last_improve = it

        it += 1
        if it - last_improve > 400000:  # stagnation cutoff
            break

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # sizes to try (larger sizes give finer approximations)
    sizes = [240, 360, 480, 720]

    for n in sizes:
        if time.time() > deadline:
            break
        M = n // 2
        idx = np.arange(n, dtype=np.int64)

        # collect initial binary candidates
        candidates = []

        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            top = np.argpartition(interp, -M)[-M:]
            bin_parent = np.zeros(n, dtype=np.uint8)
            bin_parent[top] = 1
            candidates.append(bin_parent)

        # a few random binary starts
        for _ in range(3):
            h = np.zeros(n, dtype=np.uint8)
            h[rng.choice(n, M, replace=False)] = 1
            candidates.append(h)

        # biased random start (weights squared)
        w = rng.random(n) ** 2
        top = np.argpartition(w, -M)[-M:]
        h = np.zeros(n, dtype=np.uint8)
        h[top] = 1
        candidates.append(h)

        for h0 in candidates:
            if time.time() > deadline:
                break
            # run binary simulated annealing
            h_opt, c_opt = _binary_sa(h0, n, M, idx, deadline, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n
                # early exit if already under target
                if best_c < 0.3808:
                    break

        if best_c < 0.3808:
            break

    # fallback in the unlikely event no candidate was produced
    if best_h is None:
        n = 240
        M = n // 2
        h = np.zeros(n, dtype=np.uint8)
        h[:M] = 1
        best_h = h
        best_n = n
        best_c = _compute_c5(best_h)

    # final verification using the same correlation routine as the evaluator
    final_c = _compute_c5(best_h.astype(float))
    return best_h.astype(float).tolist(), float(final_c), int(best_n)
