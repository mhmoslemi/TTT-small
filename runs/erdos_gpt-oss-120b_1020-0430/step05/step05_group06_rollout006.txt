import time
import numpy as np
from scipy.optimize import minimize

# ----- helpers ---------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a step function given by samples h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Deterministically tweak h so that sum(h) = n/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    # final correction (tiny numerical error)
    total = np.sum(h)
    if abs(total - target) > 1e-9:
        h[0] += target - total
    np.clip(h, 0.0, 1.0, out=h)
    return h

def anneal(start_h: np.ndarray, moves: int,
           T_start: float = 0.02, T_end: float = 1e-6,
           rng: np.random.Generator = None,
           deadline: float = None):
    """Binary simulated annealing (swap 1↔0). Returns best found."""
    h = start_h.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))

    check_interval = 1000
    for m in range(moves):
        t = T_start * (T_end / T_start) ** (m / moves)

        # pick a random 1 and a random 0
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = ones[idx_i]
        j = zeros[idx_j]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        ones[idx_i] = j
        zeros[idx_j] = i

        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / t):
