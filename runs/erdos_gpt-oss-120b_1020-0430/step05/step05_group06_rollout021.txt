def simulated_annealing(h_initial, O_initial, rng, deadline):
    h = h_initial.copy()
    v = 1 - h
    n = h.size
    O = O_initial.copy()

    # temperature schedule based on iteration count
    iter = 0
    while time.time() < deadline:

        ones = np.flatnonzero(h)  # could be heavy each iteration - we can maintain list
        zeros = np.flatnonzero(v)
        if len(ones) == 0 or len(zeros) == 0:

        # We'll just set T = 0.5 initially and reduce gradually.

        # We can compute O_new using method: copy O_temp = O.copy(); apply delta updates; then compute O_new_max.
        # That's O(n) extra memory; fine.

        # We'll compute O_new via copy:
        O_new = O.copy()
        # compute v_rev and h_old
        v_rev = v[::-1]
        h_old = h
        # i contributions
        O_new[i:i+n] += -1 * v_rev
        O_new[(n-1-i):(n-1-i)+n] += +1 * h_old
        # j contributions
        O_new[j:j+n] += +1 * v_rev
        O_new[(n-1-j):(n-1-j)+n] += -1 * h_old
        new_max = O_new.max()
        # acceptance
        if new_max <= current_max:
            accept = True
        else:
            # temperature
            # compute fraction of time elapsed
            elapsed = time.time() - start
            frac = elapsed / (deadline - start)
            # temperature decreasing: T = 1.0 * (1 - frac)
            T = max(1e-3, 1.0 * (1.0 - frac))
            prob = math.exp(-(new_max - current_max) / T)
            accept = rng.random() < prob

            h[i] = 0
            h[j] = 1
            v[i] = 1
            v[j] = 0
            O = O_new
            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        # else discard O_new (garbage)
        iter += 1
    return best_h, best_max

def simulated_annealing(h, O, rng, start, deadline):
    n = h.size
    v = 1 - h
    v_rev = v[::-1]  # view
    current_max = O.max()
    best_h = h.copy()
    best_max = current_max
    # for performance, predefine slices? Not needed.
    while True:
        now = time.time()
        if now >= deadline:
            break
        # random pick
        # choose random 1 index
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(v)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap evaluation
        O_new = O.copy()
        # compute updates
        O_new[i:i+n] -= v_rev
        O_new[j:j+n] += v_rev
        O_new[(n-1-i):(n-1-i)+n] += h
        O_new[(n-1-j):(n-1-j)+n] -= h
        O_new[j + n - 1 - i] += 1
        O_new[i + n - 1 - j] += 1
        O_new[n-1] = 0
        new_max = O_new.max()
        # acceptance
        if new_max <= current_max:
            accept = True
        else:
            # compute temperature
            frac = (now - start) / (deadline - start)
            T = max(1e-3, 5.0 * (1.0 - frac))  # start at 5
            prob = math.exp(-(new_max - current_max) / T)
            accept = rng.random() < prob
        if accept:
            # commit swap
            h[i] = 0
            h[j] = 1
            v[i] = 1
            v[j] = 0
            # recompute v_rev view? Actually v_rev = v[::-1] updates automatically due to view.
            # O becomes O_new
            O = O_new
            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        # else continue loop
    return best_h, best_max

def run(...):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None, best_c = inf, best_n = None

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        c_parent = compute_c5(parent)
        if c_parent < best_c:
            best_h = parent.copy()
            best_c = c_parent
            best_n = parent.size

    candidate_ns = [720, 360]  # larger first
    for n in candidate_ns:
        if time.time() >= deadline:
            break
        # Greedy binary initial
        h_bin, O_bin = greedy_binary(n, rng)
        # O_bin is correlation array; O_max = O_bin.max()
        # run SA
        h_sa, max_O_sa = simulated_annealing(h_bin.astype(np.int8), O_bin.copy(), rng, start, deadline)
        c_sa = max_O_sa * (2.0 / n)
        if c_sa < best_c:
            best_h = h_sa.astype(float).copy()
            best_c = c_sa
            best_n = n
        # Optionally, refine with SLSQP
        if time.time() < deadline:
            h_float = best_h.copy()
            adjust_sum_exact(h_float, rng)
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
                if time.time() > deadline:
                    break
                try:
                    h_float = slsqp_optimize(h_float, beta, rng, max_iter=800)
                except Exception:
                    continue
                c_new = compute_c5(h_float)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_float.copy()
                    best_n = n
    # final adjustments
    if best_h is None:
        # fallback: simple binary half
        n = 360
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[:n//2] = 1.0
        best_c = compute_c5(best_h)
    else:
        # ensure feasibility
        best_h = np.clip(best_h, 0.0, 1.0)
        if best_h.ndim == 0:
            best_h = np.array([best_h])
        # enforce exact sum
        if abs(np.sum(best_h) - best_n/2) > 1e-9:
            adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

    target = n/2.0
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    h = h0.copy()

    for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
        if time.time() >= deadline:
            break
        CURRENT_BETA = beta
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        try:

            continue
        h = np.clip(res.x, 0.0, 1.0)

def simulated_annealing(h, O, rng, start, deadline):
    n = h.size
    # O is int64; h is int8 array
    # compute initial ones and zeros
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    pos_ones = {idx: i for i, idx in enumerate(ones)}
    pos_zeros = {idx: i for i, idx in enumerate(zeros)}
    current_max = O.max()
    best_h = h.copy()
    best_max = current_max
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute O_new
        O_new = O.copy()
        v_rev = (1 - h)[::-1]
        O_new[i:i+n] -= v_rev
        O_new[j:j+n] += v_rev
        O_new[(n-1-i):(n-1-i)+n] += h
        O_new[(n-1-j):(n-1-j)+n] -= h
        O_new[j + n - 1 - i] += 1
        O_new[i + n - 1 - j] += 1
        O_new[n-1] = 0
        new_max = O_new.max()
        if new_max <= current_max:
            accept = True
        else:
            frac = (time.time() - start) / (deadline - start)
            T = max(1e-4, 5.0 * (1.0 - frac))
            prob = math.exp(-(new_max - current_max) / T)
            accept = rng.random() < prob
        if accept:
            # update ones & zeros lists.
            # Update ones list: replace i with j
            pos_i = pos_ones[i]
            ones[pos_i] = j
            pos_ones[j] = pos_i
            del pos_ones[i]
            # Update zeros list: replace j with i
            pos_j = pos_zeros[j]
            zeros[pos_j] = i
            pos_zeros[i] = pos_j
            del pos_zeros[j]
            # Update h
            h[i] = 0
            h[j] = 1
            # Update O, current_max, best
            O = O_new
            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
    return best_h, best_max

import time, math
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Exact sum correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_cascade(h0, rng, start, deadline):
    """Smooth‑max SLSQP cascade."""
    n = h0.size
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = n / 2.0
    h = h0.copy()
    for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
        if time.time() >= deadline:
            break
        CURRENT_BETA = beta
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        max_iter = 800 if beta < 200 else 1500
        try:
            res = minimize(
                softmax_obj,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        h = np.clip(res.x, 0.0, 1.0)
        adjust_sum_exact(h, rng)
    return h


def greedy_binary(n, rng):
    """Greedy construction of a binary vector with n/2 ones."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    for _ in range(L):
        O_adj = np.correlate(h, 1 - h, mode="full") + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
    O = np.correlate(h, 1 - h, mode="full")
    O[n - 1] = 0
    return h, O.astype(np.int64)


def simulated_annealing(h, O, rng, start, deadline):
    """SA on binary vectors (swap 1↔0) with incremental O update."""
    n = h.size
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    pos_ones = {idx: i for i, idx in enumerate(ones)}
    pos_zeros = {idx: i for i, idx in enumerate(zeros)}
    cur_max = O.max()
    best_h = h.copy()
    best_max = cur_max

    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        O_new = O.copy()
        v_rev = (1 - h)[::-1]
        O_new[i : i + n] -= v_rev
        O_new[j : j + n] += v_rev
        O_new[(n - 1 - i) : (n - 1 - i) + n] += h
        O_new[(n - 1 - j) : (n - 1 - j) + n] -= h
        O_new[j + n - 1 - i] += 1
        O_new[i + n - 1 - j] += 1
        O_new[n - 1] = 0
        new_max = O_new.max()

        if new_max <= cur_max:
            accept = True
        else:
            frac = (time.time() - start) / (deadline - start)
            T = max(1e-4, 5.0 * (1.0 - frac))
            prob = math.exp(-(new_max - cur_max) / T)
            accept = rng.random() < prob

        if accept:
            # update index structures
            pos_i = pos_ones[i]
            ones[pos_i] = j
            pos_ones[j] = pos_i
            del pos_ones[i]

            pos_j = pos_zeros[j]
            zeros[pos_j] = i
            pos_zeros[i] = pos_j
            del pos_zeros[j]

            h[i] = 0
            h[j] = 1
            O = O_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    return best_h, best_max


def random_float_start(n, rng):
    h = rng.random(n)
    h *= (n / 2) / h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    adjust_sum_exact(h, rng)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = float("inf")
    best_n = None

    # baseline from parent if present
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        c_parent = compute_c5(parent)
        if c_parent < best_c:
            best_h = parent.copy()
            best_c = c_parent
            best_n = parent.size

    candidate_ns = [720, 360]

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        # Greedy binary start
        h_bin, O_bin = greedy_binary(n, rng)
        c_bin = compute_c5(h_bin.astype(float))
        if c_bin < best_c:
            best_h = h_bin.astype(float).copy()
            best_c = c_bin
            best_n = n

        # Simulated annealing on greedy start
        if time.time() < deadline:
            h_sa, max_O_sa = simulated_annealing(
                h_bin.copy(), O_bin.copy(), rng, start, deadline
            )
            c_sa = max_O_sa * (2.0 / n)
            if c_sa < best_c:
                best_h = h_sa.astype(float).copy()
                best_c = c_sa
                best_n = n

        # Short SA from a random binary start (if time permits)
        if time.time() + 5 < deadline:
            # random binary with exact half ones
            h_rand = np.zeros(n, dtype=np.int8)
            h_rand[: n // 2] = 1
            rng.shuffle(h_rand)
            O_rand = np.correlate(h_rand, 1 - h_rand, mode="full").astype(np.int64)
            O_rand[n - 1] = 0
            sub_dead = time.time() + min(30.0, deadline - time.time())
            h_sa2, max_O_sa2 = simulated_annealing(
                h_rand.copy(), O_rand.copy(), rng, start, sub_dead
            )
            c_sa2 = max_O_sa2 * (2.0 / n)
            if c_sa2 < best_c:
                best_h = h_sa2.astype(float).copy()
                best_c = c_sa2
                best_n = n

        # SLSQP cascade on the current best for this n
        if best_h is not None and best_n == n and time.time() < deadline:
            h_float = best_h.copy()
            adjust_sum_exact(h_float, rng)
            h_opt = slsqp_cascade(h_float, rng, start, deadline)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h = h_opt.copy()
                best_c = c_opt
                best_n = n

        # Additional random‑float restarts for SLSQP (if time remains)
        for _ in range(2):
            if time.time() >= deadline:
                break
            h0 = random_float_start(n, rng)
            h_opt = slsqp_cascade(h0, rng, start, deadline)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h = h_opt.copy()
                best_c = c_opt
                best_n = n

    # final safety handling
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
