def sym_binary_targeted_improve(half, rng, deadline):
    n_half = half.size
    n_full = 2 * n_half
    while time.time() < deadline:
        full = sym_concat(half)
        corr = np.correlate(full, 1 - full, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n_full - 1)
        shift = abs(shift)
        if shift == 0:
            break
        ar = np.arange(n_full)
        neigh_full = full[(ar + shift) % n_full] + full[(ar - shift) % n_full]
        # compute neighbor for each half index as sum of the two positions
        neigh_half = neigh_full[:n_half] + neigh_full[-n_half:][::-1]
        # ones and zeros in half
        ones = np.flatnonzero(half == 1)
        zeros = np.flatnonzero(half == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(neigh_half[ones])]
        j = zeros[np.argmax(neigh_half[zeros])]
        # compute new half after swap
        half_swap = half.copy()
        half_swap[i], half_swap[j] = half_swap[j], half_swap[i]
        new_c = compute_c5_half(half_swap)
        cur_c = compute_c5_half(half)
        if new_c < cur_c - 1e-12:
            half = half_swap
            continue
        # try random swaps
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            half_swap = half.copy()
            half_swap[i2], half_swap[j2] = half_swap[j2], half_swap[i2]
            new_c = compute_c5_half(half_swap)
            if new_c < cur_c - 1e-12:
                half = half_swap
                improved = True
                break
        if not improved:
            break
    return half

    n = half0.size
    bounds = [(0.0, 1.0)] * n

        full = sym_concat(x)

import time
import math
import numpy as np
from scipy.optimize import minimize

def sym_concat(half: np.ndarray) -> np.ndarray:
    """Construct full symmetric vector from half."""
    return np.concatenate([half, 1.0 - half[::-1]])

def compute_c5_full(full: np.ndarray) -> float:
    """Maximum overlap for a full vector."""
    n = full.size
    corr = np.correlate(full, 1.0 - full, mode="full")
    return float(np.max(corr) * (2.0 / n))

def compute_c5_half(half: np.ndarray) -> float:
    """Maximum overlap for a half vector (full is built internally)."""
    return compute_c5_full(sym_concat(half))

def slsqp_optimize_half(half0: np.ndarray, beta: float, rng, max_iter: int = 1200) -> np.ndarray:
    """Smooth‑max SLSQP on the symmetric half vector."""
    n = half0.size
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        full = sym_concat(x)
        corr = np.correlate(full, 1.0 - full, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # stable log‑sum‑exp
        return (m + math.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(
        obj,
        half0,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return x

def binary_from_half(half: np.ndarray, rng) -> np.ndarray:
    """Turn a half vector into a binary half (0/1)."""
    bin_half = (half > 0.5).astype(float)
    # ensure at least one 0 and one 1
    if bin_half.sum() == 0:
        bin_half[rng.integers(0, half.size)] = 1.0
    elif bin_half.sum() == half.size:
        bin_half[rng.integers(0, half.size)] = 0.0
    return bin_half

def sym_binary_targeted_improve(half: np.ndarray, rng, deadline: float) -> np.ndarray:
    """Greedy swaps on binary half preserving symmetry."""
    n_half = half.size
    n_full = 2 * n_half
    while time.time() < deadline:
        full = sym_concat(half)
        corr = np.correlate(full, 1.0 - full, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n_full - 1)
        shift = abs(shift)
        if shift == 0:
            break
        ar = np.arange(n_full)
        neigh_full = full[(ar + shift) % n_full] + full[(ar - shift) % n_full]
        # combine opposite positions
        neigh_half = neigh_full[:n_half] + neigh_full[-n_half:][::-1]
        ones = np.flatnonzero(half == 1.0)
        zeros = np.flatnonzero(half == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(neigh_half[ones])]
        j = zeros[np.argmax(neigh_half[zeros])]
        # try deterministic swap
        cand = half.copy()
        cand[i], cand[j] = cand[j], cand[i]
        new_c = compute_c5_half(cand)
        cur_c = compute_c5_half(half)
        if new_c < cur_c - 1e-12:
            half = cand
            continue
        # random attempts
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            cand = half.copy()
            cand[i2], cand[j2] = cand[j2], cand[i2]
            if compute_c5_half(cand) < cur_c - 1e-12:
                half = cand
                improved = True
                break
        if not improved:
            break
    return half

def simulated_annealing_binary_half(half: np.ndarray, rng, deadline: float) -> np.ndarray:
    """Simple annealing on binary half."""
    n_half = half.size
    cur = half.copy()
    cur_c = compute_c5_half(cur)
    best = cur.copy()
    best_c = cur_c
    total = deadline - time.time()
    start_temp = 0.01
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        frac = remaining / total
        T = start_temp * frac
        ones = np.flatnonzero(cur == 1.0)
        zeros = np.flatnonzero(cur == 0.0)
        if ones.size == 0 or zeros.size == 0:

            cur, cur_c = cand, new_c
            if cur_c < best_c:
                best, best_c = cur.copy(), cur_c
        if best_c < 0.3795:
            break
    return best

def symmetrize_full(full: np.ndarray) -> np.ndarray:
    """Average a full vector with its symmetric complement."""
    n = full.size
    half = n // 2
    left = full[:half]
    right = full[-half:][::-1]
    return ((left + (1.0 - right)) / 2.0).clip(0.0, 1.0)
