import time, numpy as np, math
from scipy.optimize import minimize

# Global placeholders for constraint target and beta for SLSQP.
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    # while diff not zero, adjust random indices.
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1-x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12: break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def worst_gradient(h):
    n=h.size
    corr=np.correlate(h, 1-h, mode='full')
    idx=np.argmax(corr) - (n-1)
    k=abs(idx)
    grad=np.ones_like(h)
    if k==0:
        grad[:]=1.0-2.0*h
    else:
        grad[:n-k] -= h[k:]
        grad[k:] -= h[:n-k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it+1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.015, max_iter=10):
    n=h.size
    for it in range(max_iter):
        corr=np.correlate(h, 1-h, mode='full')
        idx=np.argmax(corr) - (n-1)
        k=abs(idx)
        g=np.zeros_like(h)
        if k==0:
            g[:]=1.0-2.0*h
        else:
            g[:n-k] += 1.0 - h[k:]
            g[k:] -= h[:n-k]
        h = h - (step * (0.5**it)) * g
        h = np.clip(h, 0.0, 1.0)
        target=n/2.0
        diff=target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12: break
                i=rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def binary_sa(n, rng, time_budget):
    start=time.time()
    best_h=None
    best_c=np.inf
    T0=0.02
    T_end=1e-4
    while time.time()-start < time_budget:
        h=np.zeros(n, dtype=float)
        ones=rng.choice(n, n//2, replace=False)
        h[ones]=1.0
        c_curr=compute_c5(h)
        # maintain arrays of positions
        ones_arr=ones.copy()
        zeros_arr=np.setdiff1d(np.arange(n), ones_arr)
        elapsed=time.time()-start
        # inner loop
        while time.time()-start < time_budget:
            elapsed=time.time()-start
            T=T0 * (T_end/T0) ** (elapsed / time_budget)
            # pick random swap
            i_idx=rng.integers(len(ones_arr))
            j_idx=rng.integers(len(zeros_arr))
            i=ones_arr[i_idx]
            j=zeros_arr[j_idx]
            # swap
            h[i]=0.0
            h[j]=1.0
            c_new=compute_c5(h)
            delta=c_new - c_curr
            if delta <= 0 or rng.random() < math.exp(-delta/max(T,1e-12)):
                # accept
                c_curr=c_new
                # update arrays
                ones_arr[i_idx]=j
                zeros_arr[j_idx]=i
                if c_new < best_c:
                    best_c=c_new
                    best_h=h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                # revert
                h[i]=1.0
                h[j]=0.0
    if best_h is None:
        best_h=np.zeros(n, dtype=float)
    return best_h, best_c

import time, math
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project onto the simplex {0<=h_i<=1, sum h_i = n/2} by greedy redistribution."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ----- SLSQP utilities -------------------------------------------------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Smooth‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ----- Gradient / hill‑climb utilities ---------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a high‑gradient index to a low‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed mass transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


# ----- Binary simulated annealing --------------------------------------
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """SA on binary vectors with exactly n/2 ones."""
    start = time.time()
    best_h = None
    best_c = np.inf
    T0 = 0.02
    T_end = 1e-4
    idx_all = np.arange(n, dtype=int)

    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)

        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(idx_all, ones_arr)

        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)

            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]

            # perform swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                # revert swap
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c


# ----- Main driver ------------------------------------------------------
