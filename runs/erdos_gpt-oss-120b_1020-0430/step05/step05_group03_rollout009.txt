def solve_binary_milp(n, time_limit):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    # w[i,k] as a 2D variable: shape (n, n); boolean
    w = cp.Variable((n, n), boolean=True)
    # constraints: w[i,k] <= h[i]; w[i,k] <= h[(i+k)%n]; w[i,k] >= h[i] + h[(i+k)%n] - 1
    constraints = [cp.sum(h) == n/2]
    # We'll build list via loops
    for i in range(n):
        for k in range(n):
            j = (i + k) % n
            constraints.append(w[i, k] <= h[i])
            constraints.append(w[i, k] <= h[j])
            constraints.append(w[i, k] >= h[i] + h[j] - 1)
    # define a_k for each shift k: sum_i w[i,k]
    a = cp.sum(w, axis=0)  # shape (n,)
    t = cp.Variable()
    constraints += [a >= t]  # vector inequality
    # objective maximize t
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    if h.value is None:
        return None, None
    h_sol = np.round(h.value).astype(float)  # binary
    # compute C5 using compute_c5
    c5 = compute_c5(h_sol)
    return h_sol, c5
```

def solve_binary_milp(n, time_limit):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n/2]

    # zero out invalid w positions where i + k >= n
    for i in range(n):
        for k in range(n):
            if i + k >= n:
                constraints.append(w[i, k] == 0)
            else:
                j = i + k
                constraints.append(w[i, k] <= h[i])
                constraints.append(w[i, k] <= h[j])
                constraints.append(w[i, k] >= h[i] + h[j] - 1)

    # constraints for each shift k
    for k in range(n):
        # s_k = sum_{i=0}^{n-k-1} h[i]
        s_k = cp.sum(h[:n - k])
        a_k = cp.sum(w[:n - k, k])
        constraints.append(t + a_k >= s_k)

    # t must be >= 0
    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    if h.value is None or prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None
    h_sol = np.rint(h.value).astype(float)  # binary 0 or 1
    # ensure sum equals n/2 (may be off by 1 due to rounding)
    # but we can enforce sum by adjusting few entries if needed
    current_sum = h_sol.sum()
    target = n // 2
    if current_sum > target:
        # turn some ones to zero
        ones = np.where(h_sol == 1)[0]
        rng = np.random.default_rng()
        to_flip = rng.choice(ones, int(current_sum - target), replace=False)
        h_sol[to_flip] = 0
    elif current_sum < target:
        zeros = np.where(h_sol == 0)[0]
        rng = np.random.default_rng()
        to_flip = rng.choice(zeros, int(target - current_sum), replace=False)
        h_sol[to_flip] = 1
    return h_sol, float(t.value) * (2.0 / n)

import time, math, numpy as np

# ----------- basic utilities ---------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # shift index (may be negative)
    k = abs(idx)
    grad = np.ones_like(h)
    if idx == 0:                     # shift zero
        grad[:] = 1.0 - 2.0 * h
    else:
        if idx > 0:                  # positive shift
            grad[: n - k] -= h[k:]
            grad[k:]       -= h[: n - k]
        else:                        # negative shift
            grad[: n - k] -= h[: n - k]          # same formula works (symmetry)
            grad[k:]       -= h[k:]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ----------- MILP for binary step functions ------------------
def solve_binary_milp(n: int, time_limit: float, rng: np.random.Generator):
    """Exact binary optimization: minimize max_k ∑ h_i(1‑h_{i+k})."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)          # w[i,k] = h_i·h_{i+k} when i+k<n
    t = cp.Variable()                               # worst‑case count

    constraints = [cp.sum(h) == n / 2]

    # linear (non‑cyclic) overlap constraints
    for i in range(n):
        for k in range(n):
            if i + k >= n:
                constraints.append(w[i, k] == 0)
            else:
                j = i + k
                constraints.append(w[i, k] <= h[i])
                constraints.append(w[i, k] <= h[j])
                constraints.append(w[i, k] >= h[i] + h[j] - 1)

    for k in range(n):
        # s_k = sum_{i=0}^{n-k-1} h_i
        s_k = cp.sum(h[: n - k])
        a_k = cp.sum(w[: n - k, k])
        # t >= s_k - a_k  →  t + a_k ≥ s_k
        constraints.append(t + a_k >= s_k)

    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        return None, None

    if h.value is None or prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    h_sol = np.rint(h.value).astype(float)   # binary 0/1
    # tiny correction if rounding disturbed the sum
    cur = h_sol.sum()
    target = n // 2
    if cur != target:
        diff = int(target - cur)
        if diff > 0:
            zeros = np.where(h_sol == 0)[0]
            flip = rng.choice(zeros, diff, replace=False)
            h_sol[flip] = 1.0
        else:
            ones = np.where(h_sol == 1)[0]
            flip = rng.choice(ones, -diff, replace=False)
            h_sol[flip] = 0.0

    t_val = float(t.value)
    c5 = t_val * (2.0 / n)
    return h_sol, c5


# ----------- main driver --------------------------------------

    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = float('inf')

    # ---- 1) try the exact binary MILP (n=128) -----------------

    # ---- 3) local refinement (guided hill‑climb) ---------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        # allocate most of the remaining time to hill climbing
        iters = int(min(8000, remaining * 15))   # rough 0.07 s per iteration
        step_scale = 0.65

                                         max_iters=iters,

    # ---- 4) short simulated‑annealing style perturbations -----
    remaining = budget_s - (time.time() - start)
    if remaining > 2.0:
        # temperature schedule: linear decay
        T0 = (best_c * 0.02) + 1e-4
        end_time = start + budget_s - 0.01
        while time.time() < end_time:
            elapsed = time.time() - start
            frac = elapsed / budget_s
            T = T0 * (1.0 - frac)
            grad = worst_gradient(best_h)
            nh = guided_perturb(best_h, grad, rng, scale=0.15)
            nc = compute_c5(nh)
            if nc < best_c or rng.random() < math.exp(-(nc - best_c) / max(T, 1e-12)):
                best_h, best_c = nh, nc

    # ---- final sanity check ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
