def softmax_minimize_step(h, beta, step):
    # compute w normalized
    corr = np.correlate(h, h, mode='full')
    # compute weights for minimizing -corr
    w = np.exp(-beta * corr)
    w_sum = w.sum()
    w = w / w_sum

    n = h.size
    neighbor = np.zeros_like(h)

    # d=0
    w0 = w[n-1]
    neighbor += 2 * w0 * h

    # d>0
    for d in range(1, n):
        wp = w[n-1 + d]
        if wp != 0:
            neighbor[:n-d] += wp * h[d:]
        wm = w[n-1 - d]
        if wm != 0:
            neighbor[d:] += wm * h[:n-d]

    # gradient descent: h = h + step * neighbor
    h_new = h + step * neighbor
    # clip to [0,1]
    np.clip(h_new, 0.0, 1.0, out=h_new)
    # enforce sum
    enforce_sum(h_new, rng)

    return h_new

def sa_optimize(h, rng, time_limit):
    n = h.size
    w = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full')
    O = w - corr   # shape 2n-1
    max_O = O.max()
    best_h = h.copy()
    best_c5 = max_O * dx

    start = time.time()
    T0 = 0.5
    T_end = 0.001
    eps_scale_initial = 0.5

    while time.time() - start < time_limit:
        # progress
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T0 * (1 - progress) + T_end * progress
        eps_scale = eps_scale_initial * (1 - progress) + 0.01 * progress

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        hi = h[i]
        hj = h[j]
        max_eps = min(hi, 1.0 - hj)
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * eps_scale
        # copy O slices
        s_i1 = slice(n-1-i, 2*n-1-i)
        s_i2 = slice(i, i+n)
        s_j1 = slice(n-1-j, 2*n-1-j)
        s_j2 = slice(j, j+n)
        O_i1_old = O[s_i1].copy()
        O_i2_old = O[s_i2].copy()
        O_j1_old = O[s_j1].copy()
        O_j2_old = O[s_j2].copy()

        # removal from i
        O[s_i1] += eps * h
        O[s_i2] += eps * h[::-1]
        h[i] = hi - eps

        # addition to j
        O[s_j1] -= eps * h
        O[s_j2] -= eps * h[::-1]
        h[j] = hj + eps

        new_max_O = O.max()
        delta = new_max_O - max_O
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            max_O = new_max_O
            c5 = max_O * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # revert O
            O[s_i1] = O_i1_old
            O[s_i2] = O_i2_old
            O[s_j1] = O_j1_old
            O[s_j2] = O_j2_old
            h[i] = hi
            h[j] = hj

    return best_h, best_c5

def refine_softmax(h, rng, time_limit):
    # runs SLSQP with decreasing beta
    n = h.size
    current_target = n / 2.0
    bounds = [(0.0,1.0)] * n

    # define constraints for sum equality using lambda that captures target
    def sum_eq(x):
        return np.sum(x) - current_target

    # start time
    start = time.time()
    # schedule of beta values
    for beta in (10.0, 50.0, 200.0, 800.0, 3200.0, 12800.0):
        if time.time() - start > time_limit:
            break
        # Softmax objective: max O = max (w - corr) = w - min_corr, but we approximate using softmax of -corr.
        def softmax_obj(x):
            corr = np.correlate(x, x, mode='full')
            # compute -corr (as we want minimize max O)
            bc = -beta * corr
            m = np.max(bc)
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta + (np.sum(x) - current_target)**2 * 1e6

        cons = {"type": "eq", "fun": sum_eq}
        res = minimize(softmax_obj, h, method="SLSQP", bounds=bounds, constraints=cons, options={"maxiter": 200, "ftol": 1e-12, "disp": False})
        h = np.clip(res.x, 0.0, 1.0)
        # optionally enforce sum again
        diff = current_target - h.sum()
        if abs(diff) > 1e-12:
            # random redistribution
            for _ in range(1000):
                idx = rng.integers(0, n)
                add = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, -h[idx])
                h[idx] += add
                diff = current_target - h.sum()
                if abs(diff) < 1e-12:
                    break

    return h

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:

        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])

import time
import numpy as np

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def sa_optimize(h: np.ndarray, rng: np.random.Generator,
                time_limit: float):
    """Simulated‑annealing style optimisation on max O_k."""
    n = h.size
    w = h.sum()                     # = n/2
    dx = 2.0 / n
    O = w - np.correlate(h, h, mode="full")   # O_k = w - corr
    max_O = O.max()
    best_h = h.copy()
    best_c5 = max_O * dx

    start = time.time()
    T0, T_end = 0.5, 0.001
    eps_scale_initial = 0.5

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        prog = elapsed / time_limit
        T = T0 * (1 - prog) + T_end * prog
        eps_scale = eps_scale_initial * (1 - prog) + 0.01 * prog

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        max_eps = min(hi, 1.0 - hj)
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * eps_scale
        if eps <= 0.0:
            continue

        # prepare delta for O
        delta = np.zeros_like(O)

        s_i1 = slice(n - 1 - i, 2 * n - 1 - i)
        s_i2 = slice(i, i + n)
        s_j1 = slice(n - 1 - j, 2 * n - 1 - j)
        s_j2 = slice(j, j + n)

        eps_h = eps * h
        delta[s_i1] += eps_h
        delta[s_i2] += eps_h[::-1]

        # temporary h after removing eps from i
        h[i] = hi - eps

        eps_h2 = eps * h
        delta[s_j1] -= eps_h2
        delta[s_j2] -= eps_h2[::-1]

        O_new = O + delta
        new_max_O = O_new.max()
        d_val = new_max_O - max_O

        accept = False
        if d_val <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-d_val / T):
                accept = True

        if accept:
            O = O_new
            max_O = new_max_O
            h[j] = hj + eps
            c5 = max_O * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # revert the temporary change of h[i]
            h[i] = hi

    return best_h, best_c5


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # ----- initial function -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        n = 256
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    # ----- simulated annealing (major part of budget) -----------------------
    time_sa = budget_s * 0.85
    best_h, best_c5 = sa_optimize(h, rng, time_sa)

    # ----- final polishing with guided hill climb ---------------------------
    time_left = budget_s - (time.time() - start_total)
    if time_left > 0.1:
        max_iters = int(2000 * 256 / best_h.size)
        best_h, c = guided_hill_climb(best_h, rng,
                                      max_iters=max_iters,
                                      step_scale=0.6)
        if c < best_c5:
            best_c5 = c
            # best_h already updated

    # ----- enforce constraints and final evaluation -------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.size)
