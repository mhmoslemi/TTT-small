import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- globals for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- core utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Redistribute values so that sum(h) == n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on a smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=30):

    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # d/ dh_i for i < n-k
            g[k:] -= h[:upper]            # d/ dh_{i} for i >= k
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, deadline):
    """Simulated‑annealing hill‑climb on a binary vector."""
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones_idx = np.flatnonzero(cur_h == 1)
    zeros_idx = np.flatnonzero(cur_h == 0)
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i_pos = rng.integers(0, ones_idx.size)
        j_pos = rng.integers(0, zeros_idx.size)
        i = ones_idx[i_pos]
        j = zeros_idx[j_pos]
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < math.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            # update index arrays in O(1)
            ones_idx[i_pos] = j
            zeros_idx[j_pos] = i
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c

def greedy_improve(h, rng, max_iter=50, deadline=None):
    """Local swaps that reduce the current worst shift."""
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate sample counts
    n_candidates = [720, 480, 360, 240, 180]
    if 'initial_h_values' in globals():
        parent_n = len(initial_h_values)
        if parent_n not in n_candidates:
            n_candidates = [parent_n] + n_candidates

    for n in n_candidates:
        if time.time() > deadline:
            break
        if n % 2 != 0:
            continue
        target = n // 2

        # estimate how many fractional restarts we can afford
        remaining = deadline - time.time()
        est_per_restart = 30 if n >= 720 else 20 if n >= 480 else 15
        max_restarts = max(1, int(remaining // est_per_restart))
        max_restarts = min(max_restarts, 6)

        best_frac_h = None
        best_frac_c = np.inf

        for _ in range(max_restarts):
            if time.time() > deadline:
                break
            # initialise h
            if 'initial_h_values' in globals() and n == parent_n:
                h = np.array(initial_h_values, dtype=float)
                adjust_sum_exact(h, rng)
            else:
                h = rng.random(n)
                h *= target / h.sum()
                np.clip(h, 0.0, 1.0, out=h)
                adjust_sum_exact(h, rng)

            # smooth‑max cascade
            betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]
            for beta in betas:
                if time.time() > deadline:
                    break
                max_iter = 500 if beta < 200 else 800
                if beta >= 2000:
                    max_iter = 1200
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)

            # gradient refinement
            h = gradient_refine(h, rng, step=0.02, max_iter=30)

            c = compute_c5(h)
            if c < best_frac_c:
                best_frac_c = c
                best_frac_h = h.copy()

        if best_frac_h is None:
            continue

        # ----- binary conversion -----
        bin_h = np.zeros(n, dtype=float)
        idx = np.argpartition(best_frac_h, -target)[-target:]
        bin_h[idx] = 1.0

        # ----- binary SA -----
        if time.time() < deadline:
            bin_h, bin_c = binary_sa_hill_climb(bin_h, rng, deadline)

        # ----- greedy local improvement -----
        if time.time() < deadline:
            bin_h = greedy_improve(bin_h, rng, max_iter=200, deadline=deadline)
            bin_c = compute_c5(bin_h)

        # ----- optional final fractional polishing -----
        if time.time() < deadline:
            h_float = bin_h.copy()
            for beta in [5, 20, 80, 200]:
                if time.time() > deadline:
                    break
                h_float = slsqp_optimize(h_float, beta, rng, max_iter=400)
            h_float = gradient_refine(h_float, rng, step=0.02, max_iter=15)
            c2 = compute_c5(h_float)
            if c2 < bin_c:
                bin_h = h_float
                bin_c = c2

        if bin_c < best_c:
            best_c = bin_c
            best_h = bin_h.copy()
            best_n = n

    # fallback in case nothing was found
    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
