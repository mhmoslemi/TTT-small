def compute_corr_fft(h, L=None):
    n = h.size
    if L is None:
        L = 1 << (2*n-1).bit_length()
    # Compute FFT of h padded
    H = np.fft.rfft(np.concatenate([h, np.zeros(L-n)]))

import time
import numpy as np

# cache for FFT length per n
_CORR_FFT_L = {}

def compute_corr_fft(h):
    """Linear correlation of h with (1-h) using FFT (full convolution)."""
    n = h.shape[0]
    L = _CORR_FFT_L.get(n)
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
        _CORR_FFT_L[n] = L
    # FFT of h padded to L
    H = np.fft.rfft(np.concatenate([h, np.zeros(L - n, dtype=h.dtype)]))
    # reversed (1-h) padded
    rev = (1.0 - h)[::-1]
    B = np.fft.rfft(np.concatenate([rev, np.zeros(L - n, dtype=h.dtype)]))
    conv = np.fft.irfft(H * B, n=L)
    return conv[: 2 * n - 1]

def compute_c5(h):
    """Exact C5 value using np.correlate (as the evaluator does)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98

    # candidate sample sizes (must be even, <=720)
    candidates = [360, 500, 600]
    best_h = None
    best_c = np.inf
    best_n = None

    for n in candidates:
        if time.time() > deadline:
            break

        # ---- initialise binary h ----
        if "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
            n_parent = parent.size
            xs = np.linspace(0.0, 2.0, n_parent, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            h = np.zeros(n, dtype=float)
            h[idx] = 1.0
        else:
            h = np.zeros(n, dtype=float)
            h[rng.choice(n, n // 2, replace=False)] = 1.0

        # ensure exact sum (binary already does it, but guard rounding)
        target_ones = n // 2
        cur_ones = int(h.sum())
        if cur_ones > target_ones:
            excess = cur_ones - target_ones
            ones_idx = np.flatnonzero(h > 0.5)
            h[ones_idx[:excess]] = 0.0
        elif cur_ones < target_ones:
            deficit = target_ones - cur_ones
            zero_idx = np.flatnonzero(h <= 0.5)
