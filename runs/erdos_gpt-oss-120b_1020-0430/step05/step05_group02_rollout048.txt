def greedy_improve_binary(h, rng):
    n = h.size
    target = n // 2  # integer
    while True:
        c = np.correlate(h, h, mode='full')
        cur_min = c.min()
        worst = target - cur_min
        if worst == 0:
            break
        idx_min = np.argmin(c) - (n - 1)
        k = abs(idx_min)
        if k == 0:
            break
        # Find candidates
        left_len = n - k
        a = h[:left_len]
        b = h[k:]
        i_idxs = np.where((a == 1) & (b == 0))[0]
        j_idxs = np.where((a == 0) & (b == 1))[0]
        if i_idxs.size == 0 or j_idxs.size == 0:
            # Cannot improve further for this k
            break
        i = rng.choice(i_idxs)
        j = rng.choice(j_idxs)
        # swap
        h[i], h[j] = h[j], h[i]
    return h

import time, math
import numpy as np
from scipy.optimize import minimize

# global vars for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, target=None):
    n = h.size
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    h[:] = np.clip(h, 0.0, 1.0)
    return h

# ----- SLSQP on smooth max -----
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa(h_bin, rng, deadline, max_iter=400000):
    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    T0 = cur_worst * 0.1 + 1e-4
    decay = 0.99999
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # 1 -> 0 at i
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        c[start_i : start_i + n] -= hi_vec
        h_bin[i] = 0.0

        # 0 -> 1 at j
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j + 1 :]))
        c[start_j : start_j + n] += hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            if rng.random() < math.exp(-(new_worst - cur_worst) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h_bin[j] = 0.0
            c[start_j : start_j + n] -= hj_vec
            h_bin[i] = 1.0
            c[start_i : start_i + n] += hi_vec
            no_improve += 1

        if no_improve > 80000:
            break
        it += 1
    return best_h, best_worst

def greedy_improve_binary(h, rng):
    n = h.size
    target = n // 2
    while True:
        c = np.correlate(h, h, mode='full')
        cur_min = c.min()
        if target - cur_min == 0:
            break
        idx_min = np.argmin(c) - (n - 1)
        k = abs(idx_min)
        if k == 0:
            break
        left_len = n - k
        a = h[:left_len]
        b = h[k:]
        i_idxs = np.where((a == 1) & (b == 0))[0]
        j_idxs = np.where((a == 0) & (b == 1))[0]
        if i_idxs.size == 0 or j_idxs.size == 0:
            break
        i = rng.choice(i_idxs)
        j = rng.choice(j_idxs)
        h[i], h[j] = h[j], h[i]
    return h
