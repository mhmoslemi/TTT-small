def compute_c5_fft(h):
    n = h.size
    L = 2*n - 1
    nfft = 1 << (L-1).bit_length()
    # compute FFT
    fft_h = np.fft.rfft(h, n=nfft)
    fft_1mh = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(fft_h * np.conj(fft_1mh), n=nfft)
    # take first L entries
    corr = corr[:L]
    # compute max correlation * dx
    return np.max(corr) * (2.0 / n)

def worst_gradient_fft(h):
    n = h.size
    L = 2*n - 1
    nfft = 1 << (L-1).bit_length()
    fft_h = np.fft.rfft(h, n=nfft)
    fft_1mh = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(fft_h * np.conj(fft_1mh), n=nfft)[:L]
    idx = np.argmax(corr)
    k = abs(idx - (n - 1))
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        # We will compute contributions
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_hill_climb_fft(h, rng, max_iters=2000, step_scale=0.7):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient_fft(h)
        # find i (argmax grad), j (argmin grad)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # no move possible
            continue
        # epsilon maximum such that h[i] - eps >= 0 and h[j] + eps <= 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            # can't move
            continue
        eps = rng.random() * max_eps * scale
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5_fft(h_new)
        if c_new < best_c:
            best_c, best_h, h = c_new, h_new.copy(), h_new
        # optionally accept if improves? but we set h to h_new only if improved; else keep old.
        # We could also accept with probability but skip.
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # We'll adjust by moving mass from indices where h > 0 to indices where h < 1
    # Use loop up to 10000 tries
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        attempts += 1
        # Choose donor and receiver
        if diff > 0:
            # need to add mass: choose donor with h > 0 (we will subtract from donor)
            # Actually diff>0 means sum too small, need to add mass to some indices.
            # We'll increase a random index j that is not at 1.
            j = rng.integers(n)
            add = min(diff, 1.0 - h[j])
            if add <= 0:
                continue
            h[j] += add
        else:
            # sum too large, need to subtract somewhere
            i = rng.integers(n)
            sub = min(-diff, h[i])
            if sub <= 0:
                continue
            h[i] -= sub
        diff = target - h.sum()
    # After loop, if diff still not zero due to precision, do a deterministic correction:
    if abs(diff) > 1e-12:
        # distribute diff uniformly (small)
        h += diff / n
        # clip and re-adjust iteratively
        h = np.clip(h, 0.0, 1.0)
        # final adjustment
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust a random index
            idx = rng.integers(n)
            h[idx] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global vars for soft‑max SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5_fft(h: np.ndarray) -> float:
    """Discrete C5 using FFT (full correlation)."""
    n = h.size
    L = 2 * n - 1
    nfft = 1 << (L - 1).bit_length()
    fh = np.fft.rfft(h, n=nfft)
    f1 = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(fh * np.conj(f1), n=nfft)[:L]
    return np.max(corr) * (2.0 / n)


def worst_gradient_fft(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    L = 2 * n - 1
    nfft = 1 << (L - 1).bit_length()
    fh = np.fft.rfft(h, n=nfft)
    f1 = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(fh * np.conj(f1), n=nfft)[:L]
    idx = np.argmax(corr)
    k = abs(idx - (n - 1))
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random adjustments
    for _ in range(10000):
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            i = rng.integers(n)
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                continue
            h[i] += add
        else:
            i = rng.integers(n)
            sub = min(-diff, h[i])
            if sub <= 0:
                continue
            h[i] -= sub
        diff = target - h.sum()
    # tiny deterministic fix if needed
    if abs(diff) > 1e-12:
        i = rng.integers(n)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        # final clip
        h = np.clip(h, 0.0, 1.0)
    return h


def guided_hill_climb_fft(h: np.ndarray, rng: np.random.Generator,
                          max_iters: int = 2000, step_scale: float = 0.7):
    """Hill‑climb moving mass from max‑grad to min‑grad positions."""
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient_fft(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * scale
                nh = h.copy()
                nh[i] -= eps
                nh[j] += eps
                c = compute_c5_fft(nh)
                if c < best_c:
                    best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def binary_sa_fft(n: int, rng: np.random.Generator, time_budget: float):
    """Simulated‑annealing on binary vectors with exactly n/2 ones."""
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    c_curr = compute_c5_fft(h)
    ones_list = list(ones)
    zeros_set = set(range(n)) - set(ones)
    zeros_list = list(zeros_set)

    best_h = h.copy()
    best_c = c_curr
    T0, T_end = 0.05, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i_idx = rng.integers(len(ones_list))
        j_idx = rng.integers(len(zeros_list))
        i = ones_list[i_idx]
        j = zeros_list[j_idx]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5_fft(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            ones_list[i_idx] = j
            zeros_list[j_idx] = i
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
