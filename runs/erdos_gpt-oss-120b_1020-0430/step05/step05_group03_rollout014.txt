import time, numpy as np

def conv_full_fft(a, b):
    """Full linear convolution via FFT."""
    n = a.size + b.size - 1
    L = 1 << (n - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = np.fft.irfft(A * B, n=L)
    return C[:n]

def enforce_sum(h, rng):
    """Force sum(h) = len(h)/2 while staying in [0,1]."""
    target = h.size * 0.5
    diff = target - h.sum()
    # simple redistribution loop
    it = 0
    while abs(diff) > 1e-12 and it < 1000:
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:  # tiny uniform correction
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        # final enforce if needed
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            return enforce_sum(h, rng)
    return h

def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with sum = n/2."""
    h = rng.random(n)
    h *= (n * 0.5) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)

def gradient_f(w, h):
    """Gradient of soft‑max approximated max C_k."""
    n = h.size
    start = n - 1
    conv1 = conv_full_fft(w, h)
    s1 = conv1[start:start + n]
    conv2 = conv_full_fft(w[::-1], h)
    s2 = conv2[start:start + n]
    return 1.0 - s1 - s2

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.2):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512

    # initialise
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        h = random_feasible_vector(n, rng)

    best_h = h.copy()
    best_c = np.max(conv_full_fft(h, 1 - h)) * (2.0 / n)

    # schedules
    beta_start, beta_end = 0.1, 8000.0
    step_start, step_end = 0.4, 5e-5

    # main soft‑max gradient descent
    while time.time() - start < budget_s * 0.94:
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.94)
        if frac > 1.0:
            break
        beta = beta_start * (beta_end / beta_start) ** frac
        step = step_start * (step_end / step_start) ** frac

        # occasional random restart
        if rng.random() < 0.001:
            h = random_feasible_vector(n, rng)

        # correlation and soft‑max weights
        C_raw = conv_full_fft(h, 1 - h)          # size 2n‑1
        shift = C_raw - C_raw.max()              # stabilize exp
        w = np.exp(beta * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient and update
        grad = gradient_f(w, h)
        h = h - step * grad
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum(h, rng)

        # track best
        cur_c = C_raw.max() * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

    # final local hill‑climb using worst‑gradient transfers
    h = best_h.copy()
    for _ in range(3000):
        g = worst_gradient(h)
        nh = guided_perturb(h, g, rng, scale=0.05)
        c_new = np.max(conv_full_fft(nh, 1 - nh)) * (2.0 / n)
        if c_new < best_c:
            best_c = c_new
            best_h = nh.copy()
            h = nh
        else:
            h = nh  # keep exploring

    # sanitise result
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = np.max(conv_full_fft(best_h, 1 - best_h)) * (2.0 / n)
