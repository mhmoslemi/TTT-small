```python
    return float(np.max(corr) * (2.0 / n))
```

```python
def slsqp_optimize(h0, beta, target_sum, rng, max_iter):
    n = h0.size
    target = target_sum
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # stable log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum exactly, to compensate any small error
    adjust_sum_exact(x, rng)
    return x
```

```python
def simulated_annealing_bin(h_bin, steps, rng, deadline):
    n = h_bin.size
    best = h_bin.copy()
    best_c = compute_c5(best.astype(float))
    cur = h_bin.copy()
    cur_c = best_c
    # compute temperature schedule
    T_start = 1.0
    T_end = 1e-4
    for s in range(steps):
        if time.time() > deadline:
            break
        # linear schedule
        T = T_start * ((steps - s) / steps) + T_end * (s / steps)
        # choose random swap
        ones = np.flatnonzero(cur == 1)
        zeros = np.flatnonzero(cur == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur.astype(float))
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            # revert
            cur[i] = 1
            cur[j] = 0
    return best, best_c
```

```python
def hill_climb_bin(h_bin, max_no_improve, rng, deadline):
    n = h_bin.size
    best = h_bin.copy()
    best_c = compute_c5(best.astype(float))
    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        # random pair
        ones = np.flatnonzero(best == 1)
        zeros = np.flatnonzero(best == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # test swap
        best[i] = 0
        best[j] = 1
        new_c = compute_c5(best.astype(float))
        if new_c < best_c:
            best_c = new_c
            no_improve = 0
        else:
            # revert
            best[i] = 1
            best[j] = 0
            no_improve += 1
    return best, best_c
```

```python
def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    # Since diff is small, we adjust by iterating over indices.
    # In worst case, diff may be up to 1 due to rounding.
    # We'll loop max 2000 times.
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        # compute possible add
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h
```

```
def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # start as zero vector correlation
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0  # enforce O_0 = 0
    return h
```

```python
def pairwise_smoothing(h, max_iters, rng, deadline):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    eps = 0.05  # step size
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        # choose i with h[i] > eps
        idx_i_candidates = np.where(cur_h > eps)[0]
        idx_j_candidates = np.where(cur_h < 1 - eps)[0]
        if idx_i_candidates.size == 0 or idx_j_candidates.size == 0:
            break
        i = rng.choice(idx_i_candidates)
        j = rng.choice(idx_j_candidates)
        cur_h[i] -= eps
        cur_h[j] += eps
        # ensure within [0,1]
        # compute c
        new_c = compute_c5(cur_h)
        if new_c < cur_c:
            cur_c = new_c
        else:
            # revert
            cur_h[i] += eps
            cur_h[j] -= eps
    return cur_h, cur_c
```

```python
import time
import math
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start:start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode='full')
        O[n - 1] = 0
    return h

def simulated_annealing_bin(h_bin, steps, rng, deadline):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur.astype(float))
    best = cur.copy()
    best_c = cur_c
    T_start = 1.0
    T_end = 1e-4
    for s in range(steps):
        if time.time() > deadline:
            break
        T = T_start * ((steps - s) / steps) + T_end * (s / steps)
        ones = np.flatnonzero(cur == 1)
        zeros = np.flatnonzero(cur == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur.astype(float))
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    return best, best_c

def hill_climb_bin(h_bin, max_no_improve, rng, deadline):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur.astype(float))
    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        ones = np.flatnonzero(cur == 1)
        zeros = np.flatnonzero(cur == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur.astype(float))
        if new_c < cur_c:
            cur_c = new_c
            no_improve = 0
        else:
            cur[i] = 1
            cur[j] = 0
            no_improve += 1
    return cur, cur_c

def slsqp_optimize(h0, beta, target_sum, rng, max_iter):
    n = h0.size
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def pairwise_smoothing(h, max_iters, rng, deadline):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    eps = 0.05
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        idx_i = np.where(cur > eps)[0]
        idx_j = np.where(cur < 1.0 - eps)[0]
        if idx_i.size == 0 or idx_j.size == 0:
            break
        i = rng.choice(idx_i)
        j = rng.choice(idx_j)
        cur[i] -= eps
        cur[j] += eps
        new_c = compute_c5(cur)
        if new_c < cur_c:
            cur_c = new_c
        else:
            cur[i] += eps
            cur[j] -= eps
    return cur, cur_c

    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = float('inf')
    best_n = None

    # candidate lengths

    # parent info if present

    parent_n = None
    if 'initial_h_values' in globals():

    for n in candidate_ns:
        if time.time() > deadline:

        target_sum = n // 2
        # build initial binary candidates
        bin_candidates = []

        # greedy binary
        try:
            h_greedy = greedy_binary(n, rng)
            bin_candidates.append(h_greedy)
        except Exception:
            pass

        # random binary (two variants)
        for _ in range(2):
            h_rand = np.zeros(n, dtype=np.int8)
            h_rand[:target_sum] = 1
            rng.shuffle(h_rand)
            bin_candidates.append(h_rand)

        # parent if size matches
        if parent is not None and parent_n == n:
            bin_candidates.append(np.round(parent).astype(np.int8))

        # parent doubled if possible
        if parent is not None and 2 * parent_n == n:
            h_double = np.concatenate([np.round(parent).astype(np.int8), np.round(parent).astype(np.int8)])
            bin_candidates.append(h_double)

        # keep a list of best binary after SA for this n
        best_bin_h = None
        best_bin_c = float('inf')

        for h_bin in bin_candidates:
            if time.time() > deadline:
                break
            # Determine SA steps based on n and remaining time
            remaining = deadline - time.time()
            # rough cost per step: 0.001 sec
            max_steps = int(remaining / 0.0015) if remaining > 0 else 0
            # baseline steps
            if n <= 360:
                base_steps = 50000
            elif n <= 540:
                base_steps = 30000
            elif n <= 720:
                base_steps = 20000
            else:
                base_steps = 15000
            steps = min(base_steps, max_steps)
            if steps < 1000:
                steps = min(base_steps, max_steps)  # possibly zero but keep small

            # Simulated annealing
            h_sa, c_sa = simulated_annealing_bin(h_bin, steps, rng, deadline)

            # Hill climbing refinement
            h_hc, c_hc = hill_climb_bin(h_sa, max_no_improve=3000, rng=rng, deadline=deadline)

            if c_hc < best_bin_c:
                best_bin_c = c_hc
                best_bin_h = h_hc

        if best_bin_h is None:
            continue

        # Update global best with binary result
        if best_bin_c < best_c:
            best_c = best_bin_c
            best_h = best_bin_h.astype(float).copy()
            best_n = n

        # Continuous refinement (SLSQP) if time permits and this n is current best
        if time.time() < deadline and best_n == n:
            h_float = best_h.copy()
            adjust_sum_exact(h_float, rng)
            betas = [20, 80, 200, 800, 3000, 10000]
            for beta in betas:
                if time.time() > deadline:
                    break
                h_float = slsqp_optimize(h_float, beta, target_sum, rng, max_iter=500)
                c_new = compute_c5(h_float)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_float.copy()
                    best_n = n

            # Pairwise smoothing
            if time.time() < deadline:
                h_smooth, c_smooth = pairwise_smoothing(best_h, max_iters=2000, rng=rng, deadline=deadline)
                if c_smooth < best_c:
                    best_c = c_smooth
                    best_h = h_smooth.copy()
                    best_n = n

    # fallback if nothing found
    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[:n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)

    # final adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
```

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h

def simulated_annealing_bin(h_bin, steps, rng, deadline):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur.astype(float))
    best = cur.copy()
    best_c = cur_c
    T_start = 1.0
    T_end = 1e-4
    for s in range(steps):
        if time.time() > deadline:
            break
        T = T_start * ((steps - s) / steps) + T_end * (s / steps)
        ones = np.flatnonzero(cur == 1)
        zeros = np.flatnonzero(cur == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur.astype(float))
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    return best, best_c

def hill_climb_bin(h_bin, max_no_improve, rng, deadline):
    cur = h_bin.copy()
    cur_c = compute_c5(cur.astype(float))
    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        ones = np.flatnonzero(cur == 1)
        zeros = np.flatnonzero(cur == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur.astype(float))
        if new_c < cur_c:
            cur_c = new_c
            no_improve = 0
        else:
            cur[i] = 1
            cur[j] = 0
            no_improve += 1
    return cur, cur_c

def reduce_max_O(h, target_O, rng, deadline):
    cur = h.copy()
    n = cur.size
    cur_O = np.max(np.correlate(cur, 1 - cur, mode="full"))
    while time.time() < deadline and cur_O > target_O:
        O_full = np.correlate(cur, 1 - cur, mode="full")
        max_val = O_full.max()
        if max_val <= target_O:
            break
        max_lags = np.where(O_full == max_val)[0] - (n - 1)
        lag = int(rng.choice(max_lags))
        if lag >= 0:
            i_cand = np.where((cur[: n - lag] == 1) & (cur[lag:] == 0))[0]
        else:
            d = -lag
            i_cand = np.where((cur[d:] == 1) & (cur[: n - d] == 0))[0] + d
        if i_cand.size == 0:
            i = int(rng.choice(np.flatnonzero(cur == 1)))
        else:
            i = int(rng.choice(i_cand))
        zeros = np.flatnonzero(cur == 0)
        if zeros.size == 0:
            break
        j = int(rng.choice(zeros))
        cur[i] = 0
        cur[j] = 1
        new_O = np.max(np.correlate(cur, 1 - cur, mode="full"))
        if new_O < cur_O:
            cur_O = new_O
        else:
            cur[i] = 1
            cur[j] = 0
    return cur, compute_c5(cur.astype(float))

class SumEqConstraint:
    def __init__(self, target):
        self.target = target
    def __call__(self, x):
        return np.sum(x) - self.target

def smooth_max_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize(h0, beta, target_sum, rng, max_iter):
    cons = {"type": "eq", "fun": SumEqConstraint(target_sum)}
    bounds = [(0.0, 1.0)] * h0.size
    res = minimize(
        smooth_max_obj,
        h0,
        args=(beta,),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def pairwise_smoothing(h, max_iters, rng, deadline):
    cur = h.copy()
    cur_c = compute_c5(cur)
    eps = 0.05
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        idx_i = np.where(cur > eps)[0]
        idx_j = np.where(cur < 1.0 - eps)[0]
        if idx_i.size == 0 or idx_j.size == 0:
            break
        i = int(rng.choice(idx_i))
        j = int(rng.choice(idx_j))
        cur[i] -= eps
        cur[j] += eps
        new_c = compute_c5(cur)
        if new_c < cur_c:
            cur_c = new_c
        else:
            cur[i] += eps
            cur[j] -= eps
    return cur, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = float("inf")
    best_n = None

    candidate_ns = [360, 540, 720, 900]

    parent = None
    parent_n = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        parent_n = parent.size

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target_sum = n // 2
        target_O = int(math.floor(0.3808 * n / 2))

        bin_candidates = []

        try:
            bin_candidates.append(greedy_binary(n, rng))
        except Exception:
            pass

        for _ in range(2):
            h_rand = np.zeros(n, dtype=np.int8)
            h_rand[:target_sum] = 1
            rng.shuffle(h_rand)
            bin_candidates.append(h_rand)

        if parent is not None and parent_n == n:
            bin_candidates.append(np.round(parent).astype(np.int8))

        if parent is not None and 2 * parent_n == n:
            bin_candidates.append(
                np.concatenate([np.round(parent).astype(np.int8), np.round(parent).astype(np.int8)])
            )

        best_bin_h = None
        best_bin_c = float("inf")

        for h_bin in bin_candidates:
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            max_steps = int(remaining / 0.0015) if remaining > 0 else 0
            if n <= 360:
                base_steps = 80000
            elif n <= 540:
                base_steps = 40000
            elif n <= 720:
                base_steps = 25000
            else:
                base_steps = 18000
            steps = min(base_steps, max_steps)
            if steps < 1000:
                steps = min(base_steps, max_steps)

            h_sa, c_sa = simulated_annealing_bin(h_bin, steps, rng, deadline)
            h_hc, c_hc = hill_climb_bin(h_sa, max_no_improve=8000, rng=rng, deadline=deadline)

            # targeted reduction of the maximal overlap if still above target
            cur_max_O = int(round(c_hc * n / 2))
            if cur_max_O > target_O:
                h_adj, c_adj = reduce_max_O(h_hc, target_O, rng, deadline)
                if c_adj < c_hc:
                    h_hc, c_hc = h_adj, c_adj

            if c_hc < best_bin_c:
                best_bin_c = c_hc
                best_bin_h = h_hc

        if best_bin_h is None:
            continue

        if best_bin_c < best_c:
            best_c = best_bin_c
            best_h = best_bin_h.astype(float).copy()
            best_n = n

        if time.time() < deadline and best_n == n:
            h_float = best_h.copy()
            adjust_sum_exact(h_float, rng)
            betas = [20, 80, 200, 800, 3000, 10000]
            for beta in betas:
                if time.time() > deadline:
                    break
                h_float = slsqp_optimize(h_float, beta, target_sum, rng, max_iter=500)
                c_new = compute_c5(h_float)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_float.copy()
                    best_n = n
            if time.time() < deadline:
                h_smooth, c_smooth = pairwise_smoothing(best_h, max_iters=2500, rng=rng, deadline=deadline)
                if c_smooth < best_c:
                    best_c = c_smooth
                    best_h = h_smooth.copy()
                    best_n = n

    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
