def binary_sa_fixed(N, target_ones, start_pattern, rng, time_budget):
    start = time.time()
    best_h = start_pattern.copy()
    best_c = compute_c5(best_h)
    h = best_h.copy()
    curr_c = best_c
    # indices:
    ones = list(np.flatnonzero(h == 1))
    zeros = [i for i in range(N) if h[i] == 0]

    T0 = 0.02
    T_end = 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)  # exponential cooling
        # random swap
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - curr_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            curr_c = new_c
            # update lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                # early exit check
                if best_c <= 0.3805:
                    break
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def upsample_and_refine(h, new_n, rng, time_budget):
    # interpolation
    old_n = h.size
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    # enforce sum
    target_sum = new_n / 2.0
    diff = target_sum - h_new.sum()
    # adjust diff by random adding
    while abs(diff) > 1e-12:
        i = rng.integers(new_n)
        add = min(diff, 1.0 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
        h_new[i] += add
        diff = target_sum - h_new.sum()
    # hill climb
    # determine iterations based on time budget
    max_iters = int(min(3000, time_budget * 500))  # approx
    h_new, c_new = guided_hill_climb(h_new, rng, max_iters=max_iters, step_scale=0.7)

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# -------------------- SLSQP (soft‑max) --------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approx of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# -------------------- gradient helpers --------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1500, step_scale: float = 0.7):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = n / 2.0 - h.sum()
    return h


# -------------------- binary simulated annealing --------------------
def binary_sa_fixed(N: int, target_ones: int, start_pattern: np.ndarray,
                    rng: np.random.Generator, time_budget: float):
    start = time.time()
    h = start_pattern.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    curr_c = best_c
    ones = list(np.flatnonzero(h == 1))
    zeros = [i for i in range(N) if h[i] == 0]
    T0, T_end = 0.02, 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - curr_c
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            curr_c = new_c
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
                if best_c <= 0.3805:
                    break
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# -------------------- up‑sampling + refinement --------------------
def upsample_and_refine(h: np.ndarray, new_n: int, rng: np.random.Generator,
                       time_budget: float):
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, rng)
    # simple hill‑climb; iteration count scaled to remaining time
    max_iters = int(min(3000, time_budget * 500))
    h_new, _ = guided_hill_climb(h_new, rng, max_iters=max_iters, step_scale=0.7)
    return h_new


# -------------------- main run --------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # initial solution (if provided)
    best_h = None
    best_c = np.inf
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c = h0, c0

    # ---------------- binary SA on 512 points ----------------
    time_bin = total * 0.30
    N_bin = 512
    target_ones = N_bin // 2
    # start pattern: round initial solution to binary by taking top values
    if best_h is not None and best_h.size == N_bin:
        idx = np.argpartition(best_h, -target_ones)[-target_ones:]
        start_bin = np.zeros(N_bin, dtype=float)
        start_bin[idx] = 1.0
    else:
        start_bin = np.zeros(N_bin, dtype=float)
        start_bin[rng.choice(N_bin, target_ones, replace=False)] = 1.0
    bin_h, bin_c = binary_sa_fixed(N_bin, target_ones, start_bin, rng, time_bin)
    if bin_c < best_c:
        best_h, best_c = bin_h.copy(), bin_c

    # ---------------- continuous refinement (SLSQP) ----------------
    elapsed = time.time() - start
    time_cont = total * 0.50 - elapsed
    if time_cont > 5 and best_h is not None:
        h = best_h.copy()
        beta_seq = (5., 20., 60., 200., 800., 3000., 10000.)
        for beta in beta_seq:
            if time.time() - start > total * 0.95:
                break
            iters = 1200 if beta >= 800 else 800
            h = slsqp_optimize(h, beta, rng, max_iter=iters)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

        # gradient tidy‑up
        if time.time() - start < total * 0.97:
            h = gradient_refine(h, rng, step=0.015, max_iter=12)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

    # ---------------- optional up‑sampling ----------------
    elapsed = time.time() - start
    time_up = total - elapsed - 5
    if time_up > 10 and best_h is not None and best_h.size * 2 <= 1024:
        n_up = best_h.size * 2
        h_up = upsample_and_refine(best_h, n_up, rng, time_up)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up.copy(), c_up

    # ---------------- final tweaks ----------------
    while time.time() - start < total * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
