import time
import numpy as np
from scipy.optimize import minimize

CURRENT_BETA = 1.0  # will be overwritten during optimization


def compute_c5_full(h_full):
    """Exact C5 for a full-periodic h (array length n)."""
    n = h_full.shape[0]
    corr = np.correlate(h_full, 1.0 - h_full, mode="full")
    return float(np.max(corr) * (2.0 / n))


def compute_c5_half(h_half):
    """C5 for a symmetric h defined by its half."""
    h_full = np.concatenate([h_half, 1.0 - h_half])
    return compute_c5_full(h_full)


def smooth_max_obj_half(h_half):
    """Smooth max (log-sum-exp) of overlaps for current beta."""
    global CURRENT_BETA
    h_full = np.concatenate([h_half, 1.0 - h_half])
    corr = np.correlate(h_full, 1.0 - h_full, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp safely
    val = (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    return val


def slsqp_optimize_half(h0, beta, max_iter=600):
    """Run SLSQP on the smooth‑max surrogate for a given beta."""
    global CURRENT_BETA
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * h0.size
    res = minimize(
        smooth_max_obj_half,
        h0,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    return np.clip(res.x, 0.0, 1.0)


def gradient_refine_half(h_half, rng, step=0.03, max_iter=30):
    """Projected gradient descent against the current worst shift."""
    M = h_half.size
    for it in range(max_iter):
        h_full = np.concatenate([h_half, 1.0 - h_half])
        n = h_full.size
        corr = np.correlate(h_full, 1.0 - h_full, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        k = abs(shift)

        if k == 0:
            g_full = 1.0 - 2.0 * h_full
        else:
            i = np.arange(n)
            g_full = 1.0 - h_full[(i + k) % n] - h_full[(i - k) % n]

        # gradient for half variables (accounting for symmetry)
        g_half = g_full[:M] - g_full[M:]

        step_sz = step * (0.5 ** it)
        h_half = h_half - step_sz * g_half
        h_half = np.clip(h_half, 0.0, 1.0)
    return h_half


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98  # safety margin

    # base discretisation
    n_base = 360
    M_base = n_base // 2

    starts = []

    # parent half (symmetrised if possible)
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size == n_base:
            half_parent = (parent[:M_base] + (1.0 - parent[M_base:])) * 0.5
            starts.append(np.clip(half_parent, 0.0, 1.0))

    # random uniform halves
    for _ in range(2):
        starts.append(rng.random(M_base))

    # random binary halves
    for _ in range(2):
        starts.append(rng.integers(0, 2, size=M_base).astype(float))

    # heavy‑tailed (beta) halves
    for _ in range(2):
        starts.append(rng.beta(0.5, 0.5, size=M_base))

    best_half = None
    best_c = np.inf
    best_n = n_base
    betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    for h0 in starts:
        if time.time() > deadline:

        h = np.clip(h0, 0.0, 1.0)

        # homotopy over betas
        for beta in betas:

            max_iter = 500 if beta < 200 else 800

            c = compute_c5_half(h)
            if c < best_c:

        # gradient polishing

        for _ in range(2):

    final_c = compute_c5_full(h_full)

    return h_full.tolist(), float(final_c), int(n_final)
