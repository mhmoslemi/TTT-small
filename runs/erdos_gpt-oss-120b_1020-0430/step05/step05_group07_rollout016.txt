def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # compute max overlap via smooth max of -sum_i h[i]*h[i+k]? Wait objective is to minimize max overlap
    # We previously used corr = np.correlate(x, 1 - x); bc = CURRENT_BETA * corr; then compute softmax.
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:

        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    """Maximum overlap integral for discrete step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass so that sum(h) == len(h)/2 (exactly)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.04, max_iter=15):
    """Projected gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]           # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # small random redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- simulated annealing ----------
def SA_optimize(h_init, rng, deadline):
    """Transfer‑based simulated annealing on a continuous vector."""
    n = h_init.size
    cur = h_init.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    start_T = max(0.01 * cur_c, 1e-5)
    it = 0
    last_improve = 0
    target = n / 2.0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # random direction of transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        if cur[src] <= 0.0 or cur[dst] >= 1.0:
            continue
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = start_T * (0.9995 ** it)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / (T + 1e-12)):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        # occasional re‑normalisation of the sum
        if (it & 0x7FF) == 0:  # every 2048 steps
            diff = target - cur.sum()
            if abs(diff) > 1e-12:
                i_adj = rng.integers(0, n)
                add = min(diff, 1.0 - cur[i_adj]) if diff > 0 else max(diff, -cur[i_adj])
                cur[i_adj] += add
        # restart if stuck
        if it - last_improve > 25000:
            cur = best.copy()
            cur_c = best_c
            last_improve = it
    return best, best_c

# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on a smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    n = 180                     # modest size – fast evaluation, still accurate
    target = n / 2.0

    best_h = None
    best_c = np.inf

    # ----- generate starter vectors -----
    starters = []

    # down‑sample parent if available
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size % 2 != 0:
            parent = parent[:-1]
        down = (parent[0::2] + parent[1::2]) / 2.0
        # ensure length matches n
        if down.size != n:
            xs = np.linspace(0.0, 2.0, down.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            down = np.interp(xt, xs, down)
        adjust_sum_exact(down, rng)
        starters.append(down)

    # binary random
    bin_vec = np.zeros(n)
    bin_vec[rng.choice(n, n // 2, replace=False)] = 1.0
    starters.append(bin_vec)

    # uniform fractional
    uf = rng.random(n)
    uf *= target / uf.sum()
    adjust_sum_exact(uf, rng)
    starters.append(uf)

    # squared bias fractional
    sb = rng.random(n) ** 2
    sb *= target / sb.sum()
    adjust_sum_exact(sb, rng)
    starters.append(sb)

    # exponential (Dirichlet‑like)
    expv = rng.exponential(scale=1.0, size=n)
    expv *= target / expv.sum()
    adjust_sum_exact(expv, rng)
    starters.append(expv)

    # ----- process each starter -----
    for h0 in starters:
        if time.time() > deadline:
            break
        adjust_sum_exact(h0, rng)

        # a quick gradient polishing
        h = gradient_refine(h0, rng, step=0.04, max_iter=15)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

        # limited simulated annealing (≈30 % of remaining time)
        remain = deadline - time.time()
        sub_dead = time.time() + min(remain * 0.3, 30.0)
        h_sa, c_sa = SA_optimize(h, rng, sub_dead)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        # final tiny gradient sweep
        h = gradient_refine(h_sa, rng, step=0.02, max_iter=20)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- global SA with remaining budget -----
    while time.time() < deadline:
        # random fresh start
        h0 = rng.random(n)
        h0 *= target / h0.sum()
        adjust_sum_exact(h0, rng)
        h = gradient_refine(h0, rng, step=0.03, max_iter=8)
        h_sa, c_sa = SA_optimize(h, rng, deadline)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        # tiny final polish
        h = gradient_refine(h_sa, rng, step=0.02, max_iter=10)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # if SA consumed everything we exit naturally
        if time.time() >= deadline:
            break

    # ----- optional smooth‑max SLSQP refinement if time permits -----
    if best_h is not None and time.time() < deadline:
        try:
            for beta in [5, 20, 80, 200, 500, 1000, 2000]:
                if time.time() >= deadline:
                    break
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=400)
                best_c = compute_c5(best_h)
        except Exception:
            pass

    # Fallback in the impossible case
    if best_h is None:
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
