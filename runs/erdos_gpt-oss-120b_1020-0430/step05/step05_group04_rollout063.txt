def sa_optimize(h, rng, max_iter, start_T, end_T):
    n = len(h)
    ones = list of indices where h=1; zeros similarly.
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    for it in range(max_iter):
        T = start_T + (end_T - start_T) * (it / max_iter) # linear cooling (or exponential)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0, 1
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1, 0
        # maybe time check break
    return best_h, best_c

def gradient_descent_refine(h, rng, total_iters, init_step=0.02):
    n = len(h)
    target = n/2
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    for it in range(total_iters):
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        c = corr[max_idx] * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # gradient for worst shift
        k = max_idx - (n - 1)
        if k >= 0:
            g = np.zeros_like(h)
            # for i <= n-1-k: g_i = (1 - h[i+k])
            g[:n - k] = 1.0 - h[k:]
            # for i >= k: g_i -= h[i - k]
            g[k:] -= h[:n - k]
        else:
            # negative shift: use symmetry
            k = -k
            # overlap for negative shift same as positive k
            g = np.zeros_like(h)
            g[:n - k] = 1.0 - h[k:]
            g[k:] -= h[:n - k]
        # update
        h = h - step * g
        # projection
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust by moving diff onto random entries
            # Could spread diff proportionally else do iterative approach as before
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        # reduce step gradually
        step *= 0.9995
        # time check
        # break if near deadline
    return best_h, best_c

def gradient_descent_refine(h, rng, max_iters, deadline):
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0.02
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        c = corr[idx] * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        k = idx - (n - 1)
        # compute gradient for shift k
        g = np.zeros_like(h)
        if k >= 0:
            # i in [0, n-1-k]
            g[:n - k] = 1.0 - h[k:]
            # i in [k, n-1] subtract
            g[k:] -= h[:n - k]
        else:

            g[:n - k] = 1.0 - h[k:]
            g[k:] -= h[:n - k]
        # update
        h = h - step * g
        # project
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        # decay step (optional)
        step *= 0.9995
    # after loop compute final c
    final_c = compute_c5(h)
    if final_c < best_c:
        best_h, best_c = h.copy(), final_c
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

# ----- utilities -------------------------------------------------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ----- smooth‑max SLSQP -----------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ----- heavy gradient descent -----------------------------------
def gradient_descent_refine(h, rng, max_iters, deadline):
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)
    step = 0.02
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        c = corr[idx] * (2.0 / n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        k = idx - (n - 1)
        if k < 0:
            k = -k
        g = np.zeros_like(h)
        g[: n - k] = 1.0 - h[k:]
        g[k:] -= h[: n - k]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        step *= 0.9995
    final_c = compute_c5(h)
    if final_c < best_c:
        best_c, best_h = final_c, h.copy()
    return best_h, best_c

# ----- main entry ------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    candidate_ns = [480, 360, 240, 180]
    candidate_ns = [n for n in candidate_ns if n % 2 == 0]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        start_vecs = []

        # parent interpolation if available
        if 'initial_h_values' in globals():
            p = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, p.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, p)
            start_vecs.append(interp)

        # a few random fractional starts
        for _ in range(2):
            rv = rng.random(n)
            rv = np.clip(rv, 0.0, 1.0)
            rv *= target / rv.sum()
            rv = np.clip(rv, 0.0, 1.0)
            adjust_sum_exact(rv, rng)
            start_vecs.append(rv)

        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h = h0.copy()
            # heavy gradient descent
            est_iter_time = 0.0003
            remaining = deadline - time.time()
            max_iters = int(min(50000, max(10, remaining / est_iter_time)))
            h_g, c_g = gradient_descent_refine(h, rng, max_iters, deadline)
            if c_g < best_c:
                best_h, best_c, best_n = h_g.copy(), c_g, n
            # short SLSQP refinement chain
            for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000):
                if time.time() > deadline:
                    break
                try:
                    h_s = slsqp_optimize(h_g, beta, rng, max_iter=400)
                except Exception:
                    continue
                c_s = compute_c5(h_s)
                if c_s < best_c:
                    best_h, best_c, best_n = h_s.copy(), c_s, n
                h_g = h_s

    # optional up‑sampling if time permits
    if best_h is not None and time.time() < deadline - 5 and best_n * 2 <= 720:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        est_iter_time = 0.0003
        max_iters = int(min(50000, max(10, (deadline - time.time()) / est_iter_time)))
        h_up, c_up = gradient_descent_refine(h_up, rng, max_iters, deadline)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
