    # start from parent if provided
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    low_ns = [64, 96, 128, 176, 224, 256]

    # ---------- many random restarts at low resolution ----------
    while time.time() - start < 0.78 * budget_s:
        n = int(rng.choice(low_ns))
        h = rng.random(n)
        h = enforce_sum(h, rng)
        # greedy optimisation
        h_opt, c_opt = greedy_hill_climb(
            h, rng, max_iters=int(3000 * 256 / n), min_eps=1e-12, eps_decay=0.5
        )
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, n

    # ---------- upsample to full resolution and deep refinement ----------
    if best_h is not None and time.time() - start < 0.92 * budget_s:
        target_n = 512
        if best_n != target_n:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
        else:
            h_up = best_h.copy()
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)

        # aggressive greedy refinement at full resolution
        h_up, c_up = greedy_hill_climb(
            h_up, rng, max_iters=5000, min_eps=1e-13, eps_decay=0.5
        )
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, target_n

        # continuation SLSQP with increasing beta
        for beta in (5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0):
            if time.time() - start > 0.97 * budget_s:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=300)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, target_n

        # final gradient polishing
        h_up = gradient_refine(h_up, rng, step=0.005, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # ---------- tiny stochastic tweaks until time runs out ----------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        # pick two random indices and move a small amount
        i = rng.integers(0, best_n)
        j = rng.integers(0, best_n)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random()
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # final sanity pass
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        final_c = compute_c5(best_h)
    else:
        final_c = np.inf

    return best_h.tolist(), float(final_c), int(best_h.size)
