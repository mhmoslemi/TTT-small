def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    idxs = np.where(corr == max_val)[0]
    # choose the middle index maybe
    idx = idxs[len(idxs)//2]
    k = abs(idx - (n - 1))
    # compute gradient g
    g = np.zeros_like(h)
    if k == 0:
        g = 1.0 - 2.0 * h  # as they did
    else:
        upper = n - k
        # for i in 0..upper-1: derivative w.r.t h[i] is (1 - h[i+k])
        g[:upper] += 1.0 - h[k:]
        # derivative w.r.t h[i + k] is -h[i]
        g[k:] -= h[:upper]
    return k, g, max_val

def continuous_sa(h, target, rng, deadline, max_iter=200000):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    # schedule: T0 = best_c * 0.02 maybe
    T0 = best_c * 0.02 + 1e-4
    # Use exponential decay: T = T0 * (decay ** i)
    decay = 0.99995
    i = 0
    while time.time() < deadline and i < max_iter:
        # pick two distinct indices
        i1, i2 = rng.choice(n, size=2, replace=False)
        # propose delta
        # compute max possible delta that keeps within [0,1] for both
        max_up = min(1.0 - cur_h[i1], cur_h[i2])  # moving from i1 to i2
        max_down = min(cur_h[i1], 1.0 - cur_h[i2])  # moving from i2 to i1
        # choose direction randomly
        if rng.random() < 0.5:
            # move from i1 to i2
            max_delta = max_up
        else:
            max_delta = max_down
            # swap indices to keep move direction same
            i1, i2 = i2, i1  # now moving from i1 to i2
        if max_delta <= 0:
            i += 1
            continue
        # delta magnitude
        delta = rng.random() * max_delta * 0.5  # maybe limited to half
        # apply
        cur_h[i1] += delta
        cur_h[i2] -= delta
        # compute new c5
        new_c = compute_c5(cur_h)
        # acceptance
        T = T0 * (decay ** i)
        if new_c < cur_c - 1e-12 or rng.random() < math.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            # keep change
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i1] -= delta
            cur_h[i2] += delta
        i += 1
    return best_h, best_c

import time, math, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def worst_shift_and_grad(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    idx = np.where(corr == max_val)[0][len(corr[corr == max_val]) // 2]
    k = abs(idx - (n - 1))
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        upper = n - k
        g[:upper] += 1.0 - h[k:]
        g[k:] -= h[:upper]
    return k, g, max_val

def enforce_sum(h, target, rng, max_iter=500):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h

def gradient_descent(h, target, rng, max_iter, step_initial, decay):
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iter):
        _, g, _ = worst_shift_and_grad(h)
        step = step_initial * (decay ** it)
        h = h - step * g
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum(h, target, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def continuous_sa(h, target, rng, deadline, max_iter=200000):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = cur_c * 0.02 + 1e-4
    decay = 0.99995
    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.choice(n, size=2, replace=False)
        # direction: move mass from i1 to i2
        max_up = min(1.0 - cur_h[i1], cur_h[i2])
        max_down = min(cur_h[i1], 1.0 - cur_h[i2])
        if rng.random() < 0.5:
            max_delta = max_up
        else:
            max_delta = max_down
            i1, i2 = i2, i1
        if max_delta <= 0:
            i += 1
            continue
        delta = rng.random() * max_delta * 0.5
        cur_h[i1] += delta
        cur_h[i2] -= delta
        new_c = compute_c5(cur_h)
        T = T0 * (decay ** i)
        if new_c < cur_c - 1e-12 or rng.random() < math.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i1] -= delta
            cur_h[i2] += delta
        i += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    # base size
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 360
        base_h = np.zeros(n0)
        base_h[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    n0 = base_h.size
    target0 = n0 / 2.0
    base_h = enforce_sum(base_h, target0, rng)

    # generate a few diverse starts at n0
    starts = [base_h]
    # random binary
    bin_rand = np.zeros(n0)
    bin_rand[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    starts.append(bin_rand)
    # random fractional
    frac = rng.random(n0)
    frac *= target0 / frac.sum()
    starts.append(enforce_sum(frac, target0, rng))
    # alternating pattern
    alt = np.tile([1.0, 0.0], n0 // 2)
    starts.append(alt)

    best_h = None
    best_c = np.inf
    best_n = None

    # optimise at base resolution
    for s in starts:
        if time.time() > deadline:
            break
        h = s.copy()
        h, c = gradient_descent(h, target0, rng, max_iter=2000,
                                step_initial=0.08, decay=0.996)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n0

        sub_dead = min(deadline, time.time() + 0.35 * (deadline - time.time()))
        h, c = continuous_sa(h, target0, rng, sub_dead, max_iter=150000)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n0

    # up‑sample to double resolution if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        target_up = n_up / 2.0
        h_up = enforce_sum(h_up, target_up, rng)

        h_up, c_up = gradient_descent(h_up, target_up, rng,
                                      max_iter=3000,
                                      step_initial=0.07, decay=0.995)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

        sub_dead = min(deadline, time.time() + 0.30 * (deadline - time.time()))
        h_up, c_up = continuous_sa(h_up, target_up, rng, sub_dead, max_iter=180000)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

    # final cleanup
    if best_h is None:
        best_n = 360
        best_h = np.tile([1.0, 0.0], best_n // 2).astype(float)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
