def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Adjust h in-place so that sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:
            continue
        h[i] += add
        diff = target - h.sum()
    # final safety clip
    np.clip(h, 0.0, 1.0, out=h)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # push the residual onto a random feasible entry
        for _ in range(100):
            i = rng.integers(0, n)
            if 0.0 <= h[i] + residual <= 1.0:
                h[i] += residual
                break
    return h

def hill_climb_targeted(h0, deadline, rng):

    Random pairwise hill‑climbing with a bias toward the worst shift.
    Returns the best vector found and its C5.

        m_star = int(np.argmax(corr))
        kstar = m_star - (n - 1)
        i_start = max(0, -kstar)
        i_end = min(n - 1, n - 1 - kstar)

        # decide whether to use a targeted move (most of the time)
