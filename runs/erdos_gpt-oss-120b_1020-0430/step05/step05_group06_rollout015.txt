Capped simplex projection algorithm: Given vector v, we need to find projection onto {x: 0 ≤ x_i ≤ 1, sum x_i = s}. This can be solved by a variant of water-filling: first clip to [0,1], then adjust if sum is not s; we can do:

- Else if sum > s, we need to reduce some components; we can solve by projecting onto simplex with upper bound 1.

- If sum < s, we need to increase some components.

We can use an algorithm:

function proj_capped_simplex(v, s):
    # Clip to [0,1] constraint
    v_clipped = np.clip(v, 0, 1)
    total = v_clipped.sum()
    if np.isclose(total, s):
        return v_clipped
    # If total > s, need to decrease some values.
    # Solve min ||x - v||^2 subject to 0 ≤ x ≤ 1, sum x = s.
    # Equivalent to find λ such that x_i = min(1, max(0, v_i - λ)), sum x_i = s.
    # This is water-filling with lower=0 and upper=1.

    # We need to find λ. This can be done by sorting v.

    # Classic method for projection onto simplex without upper bound solves x_i = max(v_i - λ, 0). With upper bound, we need to handle 1 cap.

    # We can treat as:
    # Let A = {i: v_i - λ ∈ (0,1)} => v_i - λ in (0,1).
    # Let B = {i: v_i - λ ≥ 1} => x_i=1.
    # Let C = {i: v_i - λ ≤ 0} => x_i=0.
    # Then sum_i x_i = sum_{i∈B} 1 + sum_{i∈A} (v_i - λ) = s.
    # So λ = (sum_{i∈A} v_i + |B| - s) / |A|.

    # We can find λ by iterative method: sort v descending and try thresholds.

    # Simpler approach: use binary search for λ between min(v)-1 and max(v). Since function f(λ) = sum_i min(1, max(0, v_i - λ)) is decreasing continuous. Find λ such that f(λ)=s.

    lo = -1.0  # ensure v_i - lo >1 for all i => f(lo)=n
    hi = np.max(v)  # v_i - hi <=0 => f(hi)=0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0, 1)
        total = x.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0, 1)
    return x

def project_onto_capped_simplex(v, s):
    # Goal: find x = min(1, max(0, v - lam)) such that sum(x)=s.
    # We'll binary search lam.
    n = v.shape[0]
    # Initial bounds
    # lam_low such that all v - lam_low >= 1 => lam_low <= min(v)-1
    lam_low = np.min(v) - 1.0
    # lam_high such that all v - lam_high <= 0 => lam_high >= max(v)
    lam_high = np.max(v)
    for _ in range(50):
        lam_mid = 0.5 * (lam_low + lam_high)
        x = np.clip(v - lam_mid, 0.0, 1.0)
        total = x.sum()
        if total > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    lam = 0.5 * (lam_low + lam_high)
    x = np.clip(v - lam, 0.0, 1.0)
    # final tweak for rounding errors
    diff = s - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff over nonboundary entries
        idx = np.where((x > 0) & (x < 1))[0]
        if idx.size > 0:
            adjust = diff / idx.size
            x[idx] += adjust
            # Clip again
            x = np.clip(x, 0.0, 1.0)
    return x

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            O = np.correlate(h, 1.0 - h, mode="full")
            cur_c5 = O.max() * (2.0 / n)
            O_max = O.max()
            exp_beta = np.exp(beta * (O - O_max))
            w = exp_beta / exp_beta.sum()
            conv_full = np.convolve(w, h, mode="full")
            conv_rev = np.convolve(w[::-1], h, mode="full")
            start = n - 1
            conv_full_mid = conv_full[start:start + n]
            conv_rev_mid = conv_rev[start:start + n]
            grad = 1.0 - conv_rev_mid - conv_full_mid
            alpha = 0.5
            # backtrack
            for _ in range(5):
                h_new = h - alpha * grad
                h_new = project_onto_capped_simplex(h_new, s)
                O_new = np.correlate(h_new, 1.0 - h_new, mode="full")
                new_c5 = O_new.max() * (2.0 / n)
                if new_c5 <= cur_c5:
                    h = h_new
                    cur_c5 = new_c5
                    break
                alpha *= 0.5
            if cur_c5 < best_c:
                best_c = cur_c5
                best_h = h.copy()
    return best_h, best_c

def max_shift_subgradient_optimize(h0, steps, rng, deadline):
    h = h0.copy()
    n = h.size
    s = n / 2.0
    best_c = compute_c5(h)
    for _ in range(steps):
        if time.time() > deadline:
            break
        O = np.correlate(h, 1.0 - h, mode="full")
        idx = O.argmax()
        cur_c5 = O[idx] * (2.0 / n)
        k = idx - (n - 1)
        grad = np.ones(n)
        if k >= 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            grad[kk:] -= h[:n - kk]
            grad[:n - kk] -= h[kk:]
        alpha = 0.5
        for _ in range(5):
            h_new = h - alpha * grad
            h_new = project_onto_capped_simplex(h_new, s)
            O_new = np.correlate(h_new, 1.0 - h_new, mode="full")
            new_c5 = O_new.max() * (2.0 / n)
            if new_c5 < cur_c5:
                h = h_new
                cur_c5 = new_c5
                break
            alpha *= 0.5
        if cur_c5 < best_c:
            best_c = cur_c5
            best_h = h.copy()
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
    else:
        n = 360
        parent = None

    s = n / 2.0
    best_h = None
    best_c = np.inf

    if parent is not None:
        c_parent = compute_c5(parent)
        if c_parent < best_c:
            best_c, best_h = c_parent, parent.copy()

    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    def improve_start(h_start):
        nonlocal best_h, best_c
        h_start = project_onto_capped_simplex(h_start, s)
        h_opt, c_opt = pgd_optimize(h_start, beta_sched, iters_per_beta=150,
                                    rng=rng, deadline=deadline)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    if parent is not None:
        improve_start(parent)

    # random continuous starts
    for _ in range(5):
        if time.time() > deadline:
            break
        h_rand = rng.normal(loc=0.5, scale=0.2, size=n)
        h_rand = np.clip(h_rand, 0.0, 1.0)
        improve_start(h_rand)

    # random binary starts with tiny noise
    for _ in range(3):
        if time.time() > deadline:
            break
        h_bin = np.zeros(n, dtype=float)
        idxs = rng.choice(n, size=n // 2, replace=False)
        h_bin[idxs] = 1.0
        h_start = h_bin + rng.normal(scale=0.01, size=n)
        h_start = np.clip(h_start, 0.0, 1.0)
        improve_start(h_start)

    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_h = project_onto_capped_simplex(best_h, s)

    # final fine‑grained subgradient refinement
    h_ref, c_ref = max_shift_subgradient_optimize(best_h, steps=100, rng=rng,
                                                  deadline=deadline)
    if c_ref < best_c:
        best_c, best_h = c_ref, h_ref.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, s)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
