    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- greedy binary improvement ----------
def greedy_improve(h, rng, max_iter=50, deadline=None):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 2500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

# ---------- binary simulated annealing ----------
def binary_sa_optim(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    i = 0
    while time.time() < deadline:
        T = max(1e-6, cur_c * 0.01 * (0.9999 ** i))
        i1 = rng.integers(0, n)
        while cur_h[i1] != 1:
            i1 = rng.integers(0, n)
        i0 = rng.integers(0, n)
        while cur_h[i0] != 0:
            i0 = rng.integers(0, n)
        cur_h[i1] = 0
        cur_h[i0] = 1
        c_new = compute_c5(cur_h)
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / T):
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i1] = 1
            cur_h[i0] = 0
        i += 1
        if i % 5000 == 0:
            cur_h = greedy_improve(cur_h, rng, max_iter=20, deadline=deadline)
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # primary resolution
    n_target = 720
    best_h = None
    best_c = np.inf
    best_n = n_target

    start_vecs = []
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
        interp = np.interp(xt, xs, parent)
        start_vecs.append(interp)
        idx = np.argpartition(interp, -n_target // 2)[-n_target // 2:]
        bin_vec = np.zeros(n_target)
        bin_vec[idx] = 1.0
        start_vecs.append(bin_vec)

    bin_rand = np.zeros(n_target)
    bin_rand[rng.choice(n_target, n_target // 2, replace=False)] = 1.0
    start_vecs.append(bin_rand)

    rand_frac = rng.random(n_target)
    target = n_target / 2.0
    rand_frac *= target / rand_frac.sum()
    rand_frac = np.clip(rand_frac, 0.0, 1.0)
    adjust_sum_exact(rand_frac, rng)
    start_vecs.append(rand_frac)

    for base_h in start_vecs:
        if time.time() > deadline:
            break
        h = base_h.copy()
        adjust_sum_exact(h, rng)
        h = np.clip(h, 0.0, 1.0)

        h = gradient_refine(h, rng, step=0.02, max_iter=30)

        betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
                 2000.0, 5000.0, 10000.0, 20000.0, 50000.0, 100000.0)
        for beta in betas:
            if time.time() > deadline:
                break
            max_iter = 400 if beta < 200 else 700
            if beta >= 2000:
                max_iter = 900
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            c_val = compute_c5(h)
            if c_val < best_c:
                best_h, best_c, best_n = h.copy(), c_val, n_target

        h = gradient_refine(h, rng, step=0.02, max_iter=40)
        c_val = compute_c5(h)
        if c_val < best_c:
            best_h, best_c, best_n = h.copy(), c_val, n_target

        # binary conversion
        idx = np.argpartition(h, -n_target // 2)[-n_target // 2:]
        h_bin = np.zeros(n_target)
        h_bin[idx] = 1.0

        # binary SA - multiple restarts
        remaining = deadline - time.time()
        for _ in range(4):
            if time.time() > deadline:
                break
            cur_start = h_bin.copy()
            for __ in range(5):
                i1 = rng.choice(np.flatnonzero(cur_start == 1))
                i0 = rng.choice(np.flatnonzero(cur_start == 0))
                cur_start[i1] = 0
                cur_start[i0] = 1
            sub_dead = min(deadline, time.time() + remaining * 0.2)
            h_sa, c_sa = binary_sa_optim(cur_start, rng, sub_dead)
            h_sa = greedy_improve(h_sa, rng, max_iter=100, deadline=deadline)
            c_sa = compute_c5(h_sa)
            if c_sa < best_c:
                best_h, best_c, best_n = h_sa.copy(), c_sa, n_target
            remaining = deadline - time.time()

    # fallback if nothing found
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
