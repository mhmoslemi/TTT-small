def compute_c5(h):
    n = h.size
    # compute correlation via FFT
    # h and g = 1-h
    g = 1.0 - h
    # compute circular correlation: cp[k] = sum_j h[j] * g[(j + k) % n]
    hp = np.fft.fft(h)
    gp = np.fft.fft(g)
    cp = np.real(np.fft.ifft(hp * np.conj(gp)))
    # cp[0] corresponds to shift 0
    max_cp = np.max(cp)
    return (2.0 / n) * max_cp

def gradient_refine_local(h, rng, step=0.04, max_iter=8):
    n = h.shape[0]
    target = n/2.0
    for it in range(max_iter):
        # compute full correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        # find worst shift (max corr)
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # compute gradient for shift k
        grad = np.zeros_like(h)
        if idx >= 0:  # positive shift k
            # grad = (1 - h_{i+k}) for i <= n-1-k ; - h_{i-k} for i >= k
            # implement slicing
            if k < n:
                grad[:n-k] = 1.0 - h[k:]
                grad[k:] -= h[:n-k]
        else:  # negative shift, same using k
            # treat as negative shift -k; gradient formula:
            # grad[j] = (1 - h[j+k]) for j >= k ; -h[j-k] for j <= n-1-k
            # where k = -idx > 0
            if k < n:
                grad[k:] = 1.0 - h[:n-k]
                grad[:n-k] -= h[k:]
        # step size decreasing
        step_sz = step * (0.5 ** it)
        h = h - step_sz * grad
        # clip
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # random redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time
import numpy as np
import scipy.optimize as opt

CURRENT_TARGET = None  # for equality constraint


def compute_c5(h):
    """Maximum overlap C5 for discrete step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    return (2.0 / n) * float(max_corr)


def adjust_sum_exact(h, rng):
    """Exact sum correction to target = len(h)/2 using random transfers."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def softmax_obj_val(x, beta):
    """Smooth max of full correlation (value only)."""
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    return float(m + np.log(w.sum()) / beta)


def softmax_obj_jac(x, beta):
    """Gradient of smooth max (full correlation version)."""
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    w_raw = np.exp(beta * (corr - m))
    w = w_raw / w_raw.sum()        # normalized weights
    # indices: shift k corresponds to corr index k+(n-1)
    offset = n - 1
    grad = np.zeros_like(x)
    # positive shifts (including zero)
    for k in range(0, n):
        wk = w[offset + k]
        if wk == 0.0:
            continue
        nk = n - k
        if nk > 0:
            grad[:nk] += wk * (1.0 - x[k:])          # 1 - h_{i+k}
            grad[k:]  -= wk * x[:nk]                # -h_{i}
    # negative shifts
    for k in range(1, n):
        wk = w[offset - k]
        if wk == 0.0:
            continue
        nk = n - k
        if nk > 0:
            grad[k:]  += wk * (1.0 - x[:nk])        # 1 - h_{i}
            grad[:nk]  -= wk * x[k:]                # -h_{i+k}
    return grad


def eq_constr_fun(x):
    return np.array([np.sum(x) - CURRENT_TARGET])


def eq_constr_jac(x):
    return np.ones((1, x.size))


def gradient_refine_local(h, rng, step=0.04, max_iter=8):
    """Projected gradient descent against worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr) - (n - 1))
        k = abs(idx)
        grad = np.zeros_like(h)
        if idx >= 0:
            if k < n:
                grad[: n - k] = 1.0 - h[k:]
                grad[k:] -= h[: n - k]
        else:
            if k < n:
                grad[k:] = 1.0 - h[: n - k]
                grad[: n - k] -= h[k:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def trust_optimize(h0, beta, rng, max_iter=250):
    """Constrained optimization with smooth max using trust-constr."""
    global CURRENT_TARGET
    n = h0.size
    CURRENT_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq_constr_fun, "jac": eq_constr_jac}
    try:
        res = opt.minimize(
            softmax_obj_val,
            h0,
            args=(beta,),
            method="trust-constr",
            jac=softmax_obj_jac,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "verbose": 0}
        )
        h = np.clip(res.x, 0.0, 1.0)
        adjust_sum_exact(h, rng)
    except Exception:
        h = h0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None
    n_list = [360, 240, 180]
    betas = [2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]

    for n in n_list:
        if time.time() > deadline:
            break
        target = n / 2.0
        # generate start vectors
        starts = []
        if parent is not None:

            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            starts.append(interp)
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            starts.append(bin_vec)
        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)
        # random fractional uniform
        uf = rng.random(n)
        uf *= target / uf.sum()
        uf = np.clip(uf, 0.0, 1.0)
        adjust_sum_exact(uf, rng)
        starts.append(uf)
        # random fractional squared bias
        sq = rng.random(n) ** 2
        sq *= target / sq.sum()
        sq = np.clip(sq, 0.0, 1.0)
        adjust_sum_exact(sq, rng)
        starts.append(sq)

        for h0 in starts:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            for beta in betas:
                if time.time() > deadline:
                    break
                h = trust_optimize(h, beta, rng, max_iter=250)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final local gradient refinement
            if time.time() > deadline:

            h = gradient_refine_local(h, rng, step=0.04, max_iter=8)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # optional upsampling if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        for beta in betas:
            if time.time() > deadline:
                break
            h_up = trust_optimize(h_up, beta, rng, max_iter=250)
        h_up = gradient_refine_local(h_up, rng, step=0.04, max_iter=6)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
