def binary_sa(h_bin, rng, time_deadline):
    n = len(h_bin)
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # compute start temperature based on cur_c
    init_T = max(cur_c * 0.01, 1e-4)
    T = init_T
    factor = 0.99995  # slow cooling

    iter = 0
    last_improve = 0
    while time.time() < time_deadline:
        # propose swap: pick one 1 and one 0
        i = rng.choice(np.flatnonzero(cur_h == 1))
        j = rng.choice(np.flatnonzero(cur_h == 0))
        # apply swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_c = new_c
            T *= factor
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
                last_improve = iter
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
            T *= factor
        iter += 1
        # restart maybe if stuck
        if iter - last_improve > 20000:
            # reinitialize random binary
            cur_h = np.zeros(n)
            cur_h[rng.choice(n, n//2, replace=False)] = 1.0
            cur_c = compute_c5(cur_h)
            last_improve = iter
    return best_h, best_c

def binary_sa_targeted(h_bin, rng, time_deadline):
    n = len(h_bin)
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    ones = list(np.flatnonzero(cur_h))
    zeros = list(np.flatnonzero(cur_h==0))

    init_T = max(cur_c * 0.01, 5e-5)  # small
    T = init_T
    factor = 0.9999

    it = 0
    last_best_it = 0
    while time.time() < time_deadline:
        # compute worst shift
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx  # could be negative

        # define possible moves
        # compute mask for S1 and S0
        i_range = np.arange(n)
        i_plus_k = i_range + k
        valid = (i_plus_k >= 0) & (i_plus_k < n)
        S1 = np.flatnonzero((cur_h == 1) & valid & (cur_h[i_plus_k] == 0))
        S0 = np.flatnonzero((cur_h == 0) & valid & (cur_h[i_plus_k] == 1))

        if S1.size > 0 and S0.size > 0:
            i = rng.choice(S1)
            j = rng.choice(S0)
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        # update ones/zeros lists
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)

        new_c = compute_c5(cur_h)
        delta = new_c - cur_c

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
                last_best_it = it
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)

        # cooling
        T *= factor
        it += 1
        if it - last_best_it > 50000:
            # restart random binary
            cur_h = np.zeros(n)
            cur_h[rng.choice(n, n//2, replace=False)] = 1.0
            cur_c = compute_c5(cur_h)
            ones = list(np.flatnonzero(cur_h))
            zeros = list(np.flatnonzero(cur_h==0))
            T = init_T
            last_best_it = it

    return best_h, best_c

def binary_sa_targeted(h_bin, rng, time_deadline, max_iter=300000):
    n = len(h_bin)
    # compute initial correlation
    cur_corr = np.correlate(h_bin, 1 - h_bin, mode='full')
    cur_c = cur_corr.max() * (2.0 / n)
    cur_idx = cur_corr.argmax() - (n - 1)
    best_h = h_bin.copy()
    best_c = cur_c

    # maintain ones and zeros as sets/lists
    ones = list(np.flatnonzero(h_bin))
    zeros = list(np.flatnonzero(h_bin == 0))

    # temperature schedule
    init_T = max(cur_c * 0.02, 5e-5)
    T = init_T
    factor = 0.9999

    it = 0
    last_improve = 0

    while time.time() < time_deadline and it < max_iter:
        k = cur_idx  # shift offset
        # compute candidate swap
        # generate S1, S0 if possible
        i_range = np.arange(n)
        i_plus_k = i_range + k
        valid = (i_plus_k >= 0) & (i_plus_k < n)

        S1 = np.flatnonzero((h_bin == 1) & valid & (h_bin[i_plus_k] == 0))
        S0 = np.flatnonzero((h_bin == 0) & valid & (h_bin[i_plus_k] == 1))

        if S1.size > 0 and S0.size > 0:
            i = rng.choice(S1)
            j = rng.choice(S0)
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # swap
        h_bin[i] = 0
        h_bin[j] = 1
        # update ones/zeros lists: remove i, add j; remove j, add i.
        # For ones list: remove i, append j
        # For zeros list: remove j, append i
        # Since we have small lists, use .remove (cost O(n)). Not huge.
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)

        # compute new correlation
        new_corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        new_c = new_corr.max() * (2.0 / n)
        new_idx = new_corr.argmax() - (n - 1)

        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_corr = new_corr
            cur_c = new_c
            cur_idx = new_idx
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
                last_improve = it
        else:
            # revert swap and lists
            h_bin[i] = 1
            h_bin[j] = 0
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
        # cooling
        T = T * factor
        it += 1
        if it - last_improve > 50000:
            # restart random binary
            h_bin = np.zeros(n)
            h_bin[rng.choice(n, n//2, replace=False)] = 1.0
            cur_corr = np.correlate(h_bin, 1 - h_bin, mode='full')
            cur_c = cur_corr.max() * (2.0 / n)
            cur_idx = cur_corr.argmax() - (n - 1)
            ones = list(np.flatnonzero(h_bin))
            zeros = list(np.flatnonzero(h_bin == 0))
            T = init_T
            last_improve = it

    return best_h, best_c

    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- binary simulated annealing ----------
def binary_sa_targeted(h_bin, rng, deadline, max_iter=500000):
    """SA on binary vector, proposing moves that address the current worst shift."""
    n = h_bin.size
    # initial correlation
    cur_corr = np.correlate(h_bin, 1.0 - h_bin, mode="full")
    cur_c = cur_corr.max() * (2.0 / n)
    cur_idx = cur_corr.argmax() - (n - 1)
    best_h = h_bin.copy()
    best_c = cur_c

    ones = list(np.flatnonzero(h_bin))
    zeros = list(np.flatnonzero(h_bin == 0))

    init_T = max(cur_c * 0.02, 5e-5)
    T = init_T
    factor = 0.9999

    it = 0
    last_improve = 0
    i_range = np.arange(n)

    while time.time() < deadline and it < max_iter:
        k = cur_idx

        i_plus_k = i_range + k
        valid = (i_plus_k >= 0) & (i_plus_k < n)
        if valid.any():
            iv = i_range[valid]
            ipk = i_plus_k[valid]
            mask1 = (h_bin[iv] == 1) & (h_bin[ipk] == 0)
            mask0 = (h_bin[iv] == 0) & (h_bin[ipk] == 1)
            S1 = iv[mask1]
            S0 = iv[mask0]
        else:
            S1 = np.empty(0, dtype=int)
            S0 = np.empty(0, dtype=int)

        if S1.size and S0.size:
            i = rng.choice(S1)
            j = rng.choice(S0)
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # perform swap
        h_bin[i] = 0
        h_bin[j] = 1
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)

        new_corr = np.correlate(h_bin, 1.0 - h_bin, mode="full")
        new_c = new_corr.max() * (2.0 / n)
        new_idx = new_corr.argmax() - (n - 1)

        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_corr = new_corr
            cur_c = new_c
            cur_idx = new_idx
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
                last_improve = it
        else:
            # revert swap and lists
            h_bin[i] = 1
            h_bin[j] = 0
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)

        T *= factor
        it += 1

        if it - last_improve > 50000:
            # random restart
            h_bin[:] = 0.0
            h_bin[rng.choice(n, n // 2, replace=False)] = 1.0
            cur_corr = np.correlate(h_bin, 1.0 - h_bin, mode="full")
            cur_c = cur_corr.max() * (2.0 / n)
            cur_idx = cur_corr.argmax() - (n - 1)
            ones = list(np.flatnonzero(h_bin))
            zeros = list(np.flatnonzero(h_bin == 0))
            T = init_T
            last_improve = it

    return best_h, best_c

# ---------- greedy local improvement for binary ----------
def greedy_improve_binary(h, rng, deadline):
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(corr.argmax())
        max_c = float(corr.max() * (2.0 / n))
        k = max_idx - (n - 1)

        i_range = np.arange(n)
        i_plus_k = i_range + k
        valid = (i_plus_k >= 0) & (i_plus_k < n)
        if not valid.any():
            break
        iv = i_range[valid]
        ipk = i_plus_k[valid]
        mask1 = (h[iv] == 1) & (h[ipk] == 0)
        mask0 = (h[iv] == 0) & (h[ipk] == 1)
        S1 = iv[mask1]
        S0 = iv[mask0]
        if S1.size == 0 or S0.size == 0:
            break

        best_pair = None
        best_c = max_c
        total = int(S1.size) * int(S0.size)
        max_samples = 5000
        if total <= max_samples:
            for i in S1:
                for j in S0:
                    h[i] = 0
                    h[j] = 1
                    c_new = compute_c5(h)
                    if c_new < best_c - 1e-12:
                        best_c = c_new
                        best_pair = (i, j)
                    h[i] = 1
                    h[j] = 0
        else:
            for _ in range(max_samples):
                i = rng.choice(S1)
                j = rng.choice(S0)
                h[i] = 0
                h[j] = 1
                c_new = compute_c5(h)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_pair = (i, j)
                h[i] = 1
                h[j] = 0

        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

# ---------- up‑sampling via linear interpolation ----------
def upsample_interpolate(h, rng):
    n0 = h.size
    n1 = n0 * 2
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    h_up = np.interp(x1, x0, h)
    adjust_sum_exact(h_up, rng)
    return h_up

def binary_from_fractional(frac, rng):
    n = frac.size
    target = n // 2
    idx = np.argpartition(frac, -target)[-target:]
    bin_vec = np.zeros(n)
    bin_vec[idx] = 1.0
    return bin_vec

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = None
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)

    # candidate dimensions
    n_cand = []
    if parent is not None:
        n_cand.append(parent.size)
        if parent.size * 2 <= 720:
            n_cand.append(parent.size * 2)
    if 720 not in n_cand:
        n_cand.append(720)
    n_cand = [n for n in n_cand if n % 2 == 0]
    n_cand = list(dict.fromkeys(n_cand))

    for n in n_cand:
        if time.time() > deadline:
            break

        start_vecs = []

        if parent is not None and parent.size == n:
            start_vecs.append(parent.copy())
            idx = np.argpartition(parent, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)

        if parent is not None and parent.size * 2 == n:
            interp = upsample_interpolate(parent, rng)
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)

        # random binary
        rand_bin = np.zeros(n)
        rand_bin[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(rand_bin)

        for vec in start_vecs:
            if time.time() > deadline:
                break

            # ensure binary start
            h_bin = np.rint(vec).astype(float)
            if int(h_bin.sum()) != n // 2:
                h_bin = np.zeros(n)
                h_bin[rng.choice(n, n // 2, replace=False)] = 1.0

            # simulated annealing
            h_sa, c_sa = binary_sa_targeted(h_bin, rng, deadline, max_iter=500000)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa.copy()
                best_n = n

            # greedy polishing (few seconds)
            if time.time() < deadline:
                gre_dead = min(deadline, time.time() + 5.0)
                h_g = greedy_improve_binary(best_h.copy(), rng, gre_dead)
                c_g = compute_c5(h_g)
                if c_g < best_c:
                    best_c = c_g
                    best_h = h_g.copy()
                    best_n = n

    # optional up‑sampling of the best solution
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1440:
        n_up = best_n * 2
        h_up_frac = upsample_interpolate(best_h, rng)
        h_up_bin = binary_from_fractional(h_up_frac, rng)

        h_up_sa, c_up_sa = binary_sa_targeted(h_up_bin, rng, deadline, max_iter=500000)
        if c_up_sa < best_c:
            best_c = c_up_sa
            best_h = h_up_sa.copy()
            best_n = n_up

        if time.time() < deadline:
            gre_dead = min(deadline, time.time() + 5.0)
            h_up_g = greedy_improve_binary(best_h.copy(), rng, gre_dead)
            c_up_g = compute_c5(h_up_g)
            if c_up_g < best_c:
                best_c = c_up_g
                best_h = h_up_g.copy()
                best_n = n_up

    # fallback (should never trigger)
    if best_h is None:
        best_n = 360
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact half‑ones for safety
    if int(best_h.sum()) != best_n // 2:
        idx = np.argpartition(best_h, -best_n // 2)[-best_n // 2:]
        best_h[:] = 0.0
        best_h[idx] = 1.0

    best_c = compute_c5(best_h)
    return best_h.tolist(), float(best_c), int(best_n)
