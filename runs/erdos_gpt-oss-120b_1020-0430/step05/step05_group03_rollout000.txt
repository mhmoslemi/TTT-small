def smooth_max_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # for stability: subtract max
    max_corr = np.max(beta * corr)
    # compute exp(beta*corr - max_beta_corr)
    exp_term = np.exp(beta * corr - max_corr)
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp  # shape (2n-1)
    # compute gradient
    grad = np.zeros_like(h)
    # s >= 0
    for s in range(0, n):
        ws = w[s + n - 1]
        if ws == 0:
            continue
        if s == 0:
            # for s=0, contributions: first part: grad[:n] += ws * (1 - h[:])
            # second part: grad[0:] += ws * (-h[:])  # same indices, do both
            grad += ws * (1 - h)  # first
            grad += ws * (-h)     # second
            # net = ws*(1-2h) same as earlier for s=0
        else:
            # first part: grad[:n-s] += ws * (1 - h[s:])
            grad[:n-s] += ws * (1 - h[s:])
            # second part: grad[s:] += ws * (-h[:n-s])
            grad[s:] += ws * (-h[:n-s])
    # s < 0 (t = 1..n-1)
    for t in range(1, n):
        ws = w[n - 1 - t]
        if ws == 0:
            continue
        # first part: grad[t:] += ws * (1 - h[:n-t])
        grad[t:] += ws * (1 - h[:n-t])
        # second part: grad[:n-t] += ws * (-h[t:])
        grad[:n-t] += ws * (-h[t:])
    # compute smooth max value
    smooth_val = (max_corr + np.log(sum_exp)) / beta
    return smooth_val, grad

import time, numpy as np
from scipy.optimize import minimize, differential_evolution, LinearConstraint

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def multi_transfer_move(h: np.ndarray, grad: np.ndarray,
                        m: int, scale: float) -> np.ndarray:
    if m <= 0:
        return h.copy()
    top = np.argpartition(-grad, m - 1)[:m]
    bot = np.argpartition(grad, m - 1)[:m]
    pos = h[top]
    cap = 1.0 - h[bot]
    total_pos = pos.sum()
    total_cap = cap.sum()
    if total_pos <= 0 or total_cap <= 0:
        return h.copy()
    eps = scale * min(total_pos, total_cap)
    delta_pos = -eps * (pos / total_pos)
    delta_bot =  eps * (cap / total_cap)
    nh = h.copy()
    nh[top] += delta_pos
    nh[bot] += delta_bot
    np.clip(nh, 0.0, 1.0, out=nh)
    return nh


def hill_climb_multi(h: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, step_scale: float = 0.7,
                     m_frac: float = 0.1):

    best_h, best_c = h.copy(), compute_c5(h)
    step = step_scale
    m = max(1, int(m_frac * n))
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh1 = guided_perturb(h, grad, rng, scale=step)
        c1 = compute_c5(nh1)
        nh2 = multi_transfer_move(h, grad, m, scale=step)
        c2 = compute_c5(nh2)
        if c1 < c2:
            nh, nc = nh1, c1
        else:
            nh, nc = nh2, c2
        if rng.random() < 0.02:  # occasional random transfer

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def smooth_max_and_grad(h: np.ndarray, beta: float):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_c = np.max(corr)
    exp_term = np.exp(beta * (corr - max_c))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp
    grad = np.zeros_like(h)
    # s >= 0
    for s in range(0, n):
        ws = w[s + n - 1]
        if ws == 0:
            continue
        if s == 0:
            grad += ws * (1 - 2 * h)
        else:
            grad[:n - s] += ws * (1 - h[s:])
            grad[s:]     += ws * (-h[:n - s])
    # s < 0
    for t in range(1, n):
        ws = w[n - 1 - t]
        if ws == 0:
            continue
        grad[t:]    += ws * (1 - h[:n - t])
        grad[:n - t] += ws * (-h[t:])
    smooth_val = max_c + np.log(sum_exp) / beta
    return smooth_val, grad


def trust_constr_optimize(h0: np.ndarray, beta: float, max_iter: int = 600) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    def obj(x):
        val, _ = smooth_max_and_grad(x, beta)
        return val
    def jac(x):
        _, g = smooth_max_and_grad(x, beta)
        return g
    res = minimize(obj, h0, method='trust-constr', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'verbose': 0,
                            'gtol': 1e-12, 'xtol': 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


def de_optimize(n: int, rng: np.random.Generator, max_iter: int = 30) -> np.ndarray:
    bounds = [(0.0, 1.0)] * n
    lin = LinearConstraint(np.ones((1, n)), [n / 2.0], [n / 2.0])
    result = differential_evolution(compute_c5, bounds, constraints=(lin,),
                                   maxiter=max_iter, polish=False,
                                   updating='deferred',
                                   seed=int(rng.random() * 2**32))
    return enforce_sum(result.x, rng)

        h = np.interp(x_new, x_old, best_h)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)

        h, _ = hill_climb_multi(h, rng, max_iters=2500,
                                step_scale=0.5, m_frac=0.1)
        update_candidate(h)

        for beta in (20., 200., 2000., 8000., 20000., 50000.):
            if time.time() - start > budget_s:
                break
            h = trust_constr_optimize(h, beta, max_iter=600)
            update_candidate(h)

        h = gradient_refine(h, rng, step=0.01, max_iter=15)
        update_candidate(h)
        best_h, best_c = h, compute_c5(h)

    # ---- final stochastic tweaks ----
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
