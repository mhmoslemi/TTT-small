import time
import numpy as np
from scipy.optimize import minimize

# ----- utilities -----
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ----- SLSQP with smooth max -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ----- gradient refinement (continuous) -----
def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ----- greedy hill climbing on binary vectors -----

    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode="full")

    best_h = h_bin.copy()
    best_worst = cur_worst

    zeros = list(np.flatnonzero(h_bin == 0))

    p = 359
    if n == p + 1:
        residues = { (a * a) % p for a in range(1, p) }
        leg = np.zeros(n)
        for i in range(p):
            if i in residues:
                leg[i] = 1.0
        leg[-1] = 1.0  # extra 1 to reach n/2 ones
        if leg.sum() == target:
            start_vectors.append(leg)

    # a few random binary starts
    for _ in range(3):
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vectors.append(bin_rand)

    for base_h in start_vectors:
        if time.time() > deadline:
            break
        h_bin = np.where(base_h > 0.5, 1.0, 0.0).copy()
        # ensure exact sum (should already hold)
        diff = target - h_bin.sum()
        if abs(diff) > 1e-12:
            # fix by flipping a few entries
            for i in range(n):
                if diff == 0:
                    break
                if diff > 0 and h_bin[i] == 0:
                    h_bin[i] = 1.0
                    diff -= 1
                elif diff < 0 and h_bin[i] == 1:
                    h_bin[i] = 0.0
                    diff += 1

        # greedy binary hill climbing
        sub_dead = min(deadline, time.time() + 0.6 * (deadline - time.time()))
        h_opt, worst = binary_hill_climb(h_bin, rng, sub_dead)
        c5 = worst * (2.0 / n)
        if c5 < best_c:
            best_h, best_c, best_n = h_opt.astype(float).copy(), c5, n

        # continuous refinement starting from the best binary so far
        h_frac = best_h.copy()
        h_frac = gradient_refine(h_frac, rng, step=0.02, max_iter=6)
        c5 = compute_c5(h_frac)
        if c5 < best_c:
            best_h, best_c, best_n = h_frac.copy(), c5, n

        # SLSQP cascade
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
            if time.time() > deadline:
                break
            try:
                h_frac = slsqp_optimize(h_frac, beta, rng, max_iter=600)
            except Exception:
                continue
            c5 = compute_c5(h_frac)
            if c5 < best_c:
                best_h, best_c, best_n = h_frac.copy(), c5, n

        # final gradient sweep
        h_frac = gradient_refine(h_frac, rng, step=0.02, max_iter=12)
        c5 = compute_c5(h_frac)
        if c5 < best_c:
            best_h, best_c, best_n = h_frac.copy(), c5, n

    # fallback if nothing succeeded
    if best_h is None:
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
