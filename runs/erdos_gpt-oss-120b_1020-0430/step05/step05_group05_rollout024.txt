from scipy.optimize import minimize

def compute_c5_full(h):
    n = h.size
    return float(np.max(np.correlate(h, 1 - h, mode='full')[n - 1:]) * (2.0 / n))

def smooth_max_obj_half(x, beta):
    full = np.concatenate([x, 1 - x])
    corr = np.correlate(full, 1 - full, mode='full')
    bc = beta * corr
    m = np.max(bc)
    # Use log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize_half(x0, betas, max_iter_per_beta, rng):
    # x0 is initial half array
    n = x0.size * 2
    bounds = [(0.0, 1.0)] * x0.size
    x = np.clip(x0, 0.0, 1.0)
    for beta in betas:
        args = (beta,)
        try:
            res = minimize(
                smooth_max_obj_half,
                x,
                args=args,
                method='SLSQP',
                bounds=bounds,
                options={'maxiter': max_iter_per_beta, 'ftol': 1e-12, 'disp': False}
            )
        except Exception:
            continue
        if not res.success:
            # ignore
            pass
        x = np.clip(res.x, 0.0, 1.0)
    return x

def hill_climb_binary_half(half, budget_seconds):
    n = half.size * 2
    full = np.concatenate([half, 1 - half]).astype(np.uint8)
    corr = np.correlate(full, 1 - full, mode='full')
    best_corr = corr.copy()
    best_full = full.copy()
    best_val = np.max(corr[n-1:])  # integer
    start = time.time()
    while time.time() - start < budget_seconds:
        i = rng.integers(0, half.size)
        # perform flips
        flip_one(full, corr, i, n)
        flip_one(full, corr, i + half.size, n)
        new_val = np.max(corr[n-1:])
        if new_val <= best_val:
            # accept
            best_val = new_val
            best_corr = corr.copy()
            best_full = full.copy()
        else:
            # reject: flip back
            flip_one(full, corr, i, n)
            flip_one(full, corr, i + half.size, n)
    # return best half
    return best_full[:half.size].astype(float), best_val * (2.0 / n)

        if parent_full.size != base_n:

import time
import numpy as np
from scipy.optimize import minimize

def half_to_full(half):
    return np.concatenate([half, 1 - half])

def compute_c5_from_full(full):
    n = full.size
    corr = np.correlate(full, 1 - full, mode="full")
    return float(np.max(corr[n - 1 :]) * (2.0 / n)), corr

def smooth_max_obj_half(x, beta):
    full = np.concatenate([x, 1 - x])
    corr = np.correlate(full, 1 - full, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize_half(x0, betas, max_iter, rng):
    bounds = [(0.0, 1.0)] * x0.size
    x = np.clip(x0, 0.0, 1.0)
    for beta in betas:
        try:
            res = minimize(
                smooth_max_obj_half,
                x,
                args=(beta,),
                method="SLSQP",
                bounds=bounds,
                options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
            )
            if res.success:
                x = np.clip(res.x, 0.0, 1.0)
        except Exception:
            continue
    return x

def fractional_to_binary_half(x):
    m = x.size
    k = m // 2
    idx = np.argpartition(x, -k)[-k:]
    bin_half = np.zeros_like(x)
    bin_half[idx] = 1.0
    return bin_half

def flip_one(full, corr, i, n):
    s = 1 - 2 * full[i]  # original value: 0->1, 1->-1

    if i:

def hill_climb_binary_half(half, time_budget, rng):
