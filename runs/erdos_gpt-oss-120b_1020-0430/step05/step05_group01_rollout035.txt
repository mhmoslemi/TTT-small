def hill_climb(h, O, current_max, rng, time_limit, start_time):
    while time.time() - start_time < time_limit:
        # pick swap candidate
        i, j = propose_swap(h, O, np.argmax(O), rng)
        # compute delta
        delta = delta_swap_O(h,i,j)
        # Compute new max efficiently: O_new = O + delta
        # we could compute new_max as max(O_new) but might also compute only needed.
        O_new = O + delta
        new_max = O_new.max()
        if new_max < current_max:
            # accept
            h[i] = 0
            h[j] = 1
            O = O_new
            current_max = new_max
        # else maybe accept maybe skip
    return h, O, current_max

def run(seed=42,budget_s=1100,…):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s * 0.97
    N = 512

    # initialize h0 binary
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != N:
            # upsample/interpolate
            x_old = np.linspace(0,2,init.size,endpoint=False)
            x_new = np.linspace(0,2,N,endpoint=False)
            init = np.interp(x_new,x_old,init)
        # binary conversion: choose top N/2 entries
        idx = np.argpartition(init, -N//2)[-N//2:]
        h = np.zeros(N, dtype=np.int8)
        h[idx] = 1
    else:
        # random binary
        h = np.zeros(N, dtype=np.int8)
        h[rng.choice(N, N//2, replace=False)] = 1

    # compute O
    O = np.correlate(h, 1 - h, mode='full')[N-1:].astype(np.int32)  # length N
    cur_max = int(O.max())
    best_h = h.copy()
    best_max = cur_max

    # precompute array of indices for delta if needed?
    # For delta, we use slices; no precomp needed.

    # Simulated annealing loop
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > total:
            break
        # compute temperature
        T = 10.0 * max(0.0, 1.0 - elapsed / total)  # linear cooling
        # worst shift index for biasing candidate generation
        k_max = int(np.argmax(O))
        # pick swap candidates
        # Build candidate sets for i and j
        N_range = np.arange(N)
        # i candidates
        i_mask = (h == 1) & (N_range + k_max < N) & (h[N_range + k_max] == 0)
        i_candidates = np.where(i_mask)[0]
        # j candidates
        cond1 = (h == 0) & (N_range + k_max < N) & (h[N_range + k_max] == 1)
        cond2 = (N_range >= k_max) & (h == 0) & (h[N_range - k_max] == 1)
        j_mask = cond1 | cond2
        j_candidates = np.where(j_mask)[0]

        if i_candidates.size > 0 and j_candidates.size > 0:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
            if i == j:
                # fallback
                i = rng.choice(np.where(h == 1)[0])
                j = rng.choice(np.where(h == 0)[0])
        else:
            # random pair
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # compute delta
        # Use slices as described
        delta = np.zeros(N, dtype=np.int32)
        # i contributions
        delta[:i+1] += h[i::-1]  # h[i], h[i-1], ..., h[0]
        delta[:N-i] += h[i:] - 1
        # j contributions
        delta[:N-j] += 1 - h[j:]
        delta[:j+1] -= h[j::-1]

        new_O = O + delta
        new_max = int(new_O.max())

        accept = False
        if new_max < cur_max:
            accept = True
        elif T > 1e-8:
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # apply swap
            h[i] = 0
            h[j] = 1
            O = new_O
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        # else do nothing

    # final evaluation
    final_c5 = best_max * (2.0 / N)
    # convert best_h to float list
    h_list = best_h.astype(float).tolist()
    return h_list, float(final_c5), N

import time
import numpy as np

def _compute_O(h: np.ndarray) -> np.ndarray:
    """Return O(k) = Σ h[i]*(1-h[i+k]) for k=0..N-1 (as int array)."""
    N = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr[N - 1 :].astype(np.int32)
    O[0] = 0
    return O


def _delta_swap(h: np.ndarray, i: int, j: int) -> np.ndarray:
    """Delta O(k) caused by swapping h[i]=1 with h[j]=0 (binary vector)."""
    N = h.size
    delta = np.zeros(N, dtype=np.int32)

    # i changes 1 → 0
    delta[: N - i] += h[i:] - 1
    if i > 0:
        delta[1 : i + 1] += h[i - 1 :: -1]

    # j changes 0 → 1
    delta[: N - j] += 1 - h[j:]
    if j > 0:
        delta[1 : j + 1] -= h[j - 1 :: -1]

    # the pair (i,j) is counted twice; correct once
    k0 = abs(j - i)
    delta[k0] += 1

    # O[0] must stay zero
    delta[0] = 0
    return delta


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_time = budget_s * 0.97  # reserve a little for final checks
    N = 512  # target resolution

    # ----- initialise binary vector ---------------------------------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != N:
            # linear interpolation to target size
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            init = np.interp(x_new, x_old, init)
        # take the largest N/2 entries as 1, rest 0
        idx = np.argpartition(init, -N // 2)[-N // 2 :]
        h = np.zeros(N, dtype=np.int8)
        h[idx] = 1
    else:
        h = np.zeros(N, dtype=np.int8)
        h[rng.choice(N, N // 2, replace=False)] = 1

    O = _compute_O(h)
    cur_max = int(O[1:].max())
    best_h = h.copy()
    best_max = cur_max

    # ----- main simulated‑annealing loop ---------------------------
    no_improve = 0
    while True:
        now = time.time()
        if now - start > total_time:
            break

        # temperature linearly cooled to zero
        elapsed = now - start
        T = 10.0 * max(0.0, 1.0 - elapsed / total_time)

        # worst shift (ignore k=0)
        k_max = int(np.argmax(O))

        # candidate i (1 that contributes to O(k_max))
        ones = np.where(h == 1)[0]
        i_mask = (ones + k_max < N) & (h[ones + k_max] == 0)
        i_cand = ones[i_mask]

        # candidate j (0 that would not increase O(k_max))
        zeros = np.where(h == 0)[0]
        cond1 = (zeros + k_max < N) & (h[zeros + k_max] == 1)
        cond2 = (zeros >= k_max) & (h[zeros - k_max] == 1)
        j_cand = zeros[cond1 | cond2]

        if i_cand.size and j_cand.size:
            i = int(rng.choice(i_cand))
            j = int(rng.choice(j_cand))
            if i == j:
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
        else:
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))

        delta = _delta_swap(h, i, j)
        O_new = O + delta
        new_max = int(O_new[1:].max())

        accept = False
        if new_max < cur_max:
            accept = True
        elif T > 1e-9:
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            h[i] = 0
            h[j] = 1
            O = O_new
            cur_max = new_max
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1

        # occasional random perturbation to escape plateaus
        if no_improve > 20000 and (now - start) < total_time - 5.0:
            # perform a small burst of random swaps (without checking)
            for _ in range(30):
                i = int(rng.choice(np.where(h == 1)[0]))
                j = int(rng.choice(np.where(h == 0)[0]))
                h[i] = 0
                h[j] = 1
            O = _compute_O(h)
            cur_max = int(O[1:].max())
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()

    # ----- final evaluation -----------------------------------------
    final_c5 = best_max * (2.0 / N)  # dx = 2/N
    return best_h.astype(float).tolist(), float(final_c5), int(N)
