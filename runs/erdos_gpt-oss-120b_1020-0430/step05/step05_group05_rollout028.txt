import time
import numpy as np
from scipy.optimize import minimize

# ----- globals for smooth‑max surrogate -----
CURRENT_BETA = None


def half_to_full(x):
    """Construct the full periodic vector from the half‑vector."""
    return np.concatenate([x, 1.0 - x])


def compute_c5_full(h):
    """Maximum overlap for the discrete problem."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def softmax_obj(x):
    """Smooth‑max of overlaps, used by SLSQP."""
    beta = CURRENT_BETA
    h = half_to_full(x)
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize_half(x0, beta, max_iter=1500):
    """SLSQP minimisation of the smooth‑max surrogate on the half‑vector."""
    global CURRENT_BETA
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * len(x0)
    res = minimize(
        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    return np.clip(res.x, 0.0, 1.0)


def fractional_to_binary_top_half(x):
    """Binary half‑vector: set the largest m/2 entries to 1."""
    m = x.size
    xb = np.zeros(m)
    idx = np.argpartition(x, -m // 2)[-m // 2 :]
    xb[idx] = 1.0
    return xb


def binary_swap_search(x, rng, deadline, max_swaps=20000):
    """
    Greedy random swaps on a binary half‑vector.
    Returns the (possibly) improved vector and its C5 value.
    """
    x = x.copy()
    m = x.shape[0]
    best_c = compute_c5_full(half_to_full(x))
    swaps = 0
    while time.time() < deadline and swaps < max_swaps:
        i, j = rng.choice(m, size=2, replace=False)
        if x[i] == x[j]:
            continue
        x[i], x[j] = x[j], x[i]  # perform swap
        c = compute_c5_full(half_to_full(x))
        if c < best_c:
            best_c = c
        else:
            x[i], x[j] = x[j], x[i]  # revert
        swaps += 1
    return x, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    phi = (np.sqrt(5) - 1) / 2  # golden‑ratio slope
    betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]
    target_c = 0.3808

    # ----- use parent construction if supplied -----

        best_h, best_c, best_n = init_h.copy(), init_c, init_h.size

    # ----- candidate problem sizes -----
    candidate_ns = [360, 720, 1080, 1440]
    # ensure the parent size is examined first
    if best_n is not None and best_n not in candidate_ns:
        candidate_ns.insert(0, best_n)

    for n in candidate_ns:
