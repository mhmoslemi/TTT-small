def project_onto_hyperplane_box(y, target, tol=1e-9, max_iter=60):
    # find lambda s.t. sum clip(y - lambda, 0, 1) = target.
    low = -np.max(y) - 1.0
    high = np.max(y) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < tol:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(y - (low + high) / 2.0, 0.0, 1.0)

Note: The monotonicity: As lambda increases, y - lambda decreases, thus sum after clipping decreases. So if s > target, we need increase lambda (i.e., move low up). Actually low starts low (negative), high positive. So if s > target, we need increase lambda (shrink sum) => set low = mid. If s < target, set high = mid.

def grad_smoothmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    exp_vals = np.exp(beta * (corr - max_corr))
    w = exp_vals / exp_vals.sum()
    grad = np.ones(n)  # from +1 term
    # w index mapping: w[k + n - 1] corresponds to lag k
    # loop over k from 0..n-1
    # positive lag
    for k in range(n):
        wk = w[n - 1 + k]
        if wk == 0.0:
            continue
        if k == 0:
            grad -= wk * h
        else:
            grad[: n - k] -= wk * h[k:]
    # negative lag
    for k in range(1, n):
        wk = w[n - 1 - k]
        if wk == 0.0:
            continue
        # k>0
        grad[k:] -= wk * h[: n - k]
    return grad

Note: This includes forward k=0 again; but we already processed wk for k=0 in positive loop. Negative loop starts from k=1 to avoid double counting zero.

def pgd_optimize(h0, beta, step, target, max_iter, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = grad_smoothmax(h, beta)
        y = h - step * grad
        h = project_onto_hyperplane_box(y, target)
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def grad_smoothmax(h, beta):

    corr = np.correlate(h, 1.0 - h, mode='full')

        if wk == 0.0:

        grad[k:] -= wk * h[: n - k]
    return grad

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return C₅ for the discretized step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def project_onto_hyperplane_box(y: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 40) -> np.ndarray:
    """
    Project y onto the set {x : sum(x)=target, 0≤x≤1}.
    Uses a binary search on the Lagrange multiplier.
    """
    low = -np.max(y) - 1.0   # sum will be n
    high = np.max(y) + 1.0   # sum will be 0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(y - (low + high) * 0.5, 0.0, 1.0)


def grad_smoothmax(h: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate:
        f(h) = (1/β) log Σ_k exp(β·corr_k)
    where corr_k = Σ_i h_i (1−h_{i+k}) (zero‑padded).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = corr.max()
    w = np.exp(beta * (corr - max_corr))
    w /= w.sum()                       # soft‑max weights, Σ w = 1
    grad = np.ones(n)                  # from the constant “+1” term in ∂corr/∂h_i
    # positive lags (k ≥ 0)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk == 0.0:
            continue
        if k == 0:
            grad -= 2.0 * wk * h        # both forward and backward coincide
        else:
            grad[: n - k] -= wk * h[k:]
    # negative lags (k ≤ -1)
    for k in range(1, n):
        wk = w[n - 1 - k]
        if wk == 0.0:
            continue
        grad[k:] -= wk * h[: n - k]
    return grad


def pgd_optimize(
    h0: np.ndarray,
    beta: float,
    step: float,
    target: float,
    max_iter: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max surrogate.
    Returns the best vector seen and its C₅ value.
    """
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        grad = grad_smoothmax(h, beta)
        y = h - step * grad
        h = project_onto_hyperplane_box(y, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        step *= 0.9995                 # very gentle decay
    return best_h, best_c
