def grad_worst_shift(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    # add shifted contributions
    grad[:n-k] += h[k:]   # from h_{i+k}
    grad[k:] += h[:n-k]   # from h_{i-k}
    grad *= -dx
    return grad

def project_sum_box(v, target, tol=1e-10, max_iter=100):
    # v is array
    lo = v.min() - 1.0  # if λ lower, v_i - λ >1 for all i, clipped to 1 => sum = n
    hi = v.max()        # if λ higher, all v_i - λ <=0 => sum=0
    # But we want sum target between 0 and n (target = n/2)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = mid
        else:
            hi = mid
    return proj
```

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_sum_box(v, target, tol=1e-12, max_iter=60):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = mid
        else:
            hi = mid
    return proj

def gradient_for_shift(h, k):
    n = h.size
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum precisely
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple correction
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def subgradient_descent(h0, max_iters, step0, rng):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_val = np.max(corr)
        cur_c = cur_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # get shift(s) achieving max
        max_idxs = np.where(corr == cur_val)[0]
        idx = max_idxs[0]
        shift = idx - (n - 1)
        k = abs(shift)
        grad = gradient_for_shift(h, k)
        # step schedule
        alpha = step0 * (1.0 - it / max_iters)
        if alpha <= 0:
            break
        h_new = h - alpha * grad
        # project
        h = project_sum_box(h_new, target)
        # occasional random small perturb
        if it % 500 == 0 and it > 0:
            i, j = rng.integers(0, n, size=2)
            eps = min(h[i], 1.0 - h[j]) * 0.1
            if eps > 0:
                h[i] -= eps
                h[j] += eps
                h = project_sum_box(h, target)
    return best_h, best_c

    best_h = h0.copy()

    # main optimization loop

    while time.time() - start < 0.70 * budget_s:

        # random perturb: swap epsilon between two random indices

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)

CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_box(v: np.ndarray, target: float, tol=1e-12, max_iter=30) -> np.ndarray:
    """Projection onto the intersection of box [0,1] and sum(v)=target."""
    lo = v.min() - 1.0  # ensures all entries would be clipped to 1
    hi = v.max()        # ensures all entries would be clipped to 0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = mid
        else:
            hi = mid
    return proj


def gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of O_k = Σ h_i (1‑h_{i+k}) (pre‑dx) w.r.t. h."""
    n = h.size
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def subgradient_descent(h0: np.ndarray, max_iters: int, step0: float, rng) -> tuple:
    """Projected sub‑gradient descent on max_k O_k."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_val = np.max(corr)
        cur_c = cur_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # shift achieving the max
        max_idx = np.where(corr == cur_val)[0][0]
        shift = max_idx - (n - 1)
        k = abs(shift)
        grad = gradient_for_shift(h, k)
        # simple linear step schedule
        alpha = step0 * (1.0 - it / max_iters)
        if alpha <= 0:
            break
        h_new = h - alpha * grad
        h = project_sum_box(h_new, target)
        # occasional tiny random transfer to escape plateaus
        if it % 500 == 0 and it > 0:
            i, j = rng.integers(0, n, size=2)
            eps = min(h[i], 1.0 - h[j]) * rng.random() * 0.1
            if eps > 0:
                h[i] -= eps
                h[j] += eps
                h = project_sum_box(h, target)
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target = n / 2.0

    # initialise from parent if present, otherwise random
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_sum_box(h0, target)
    else:
        h0 = rng.random(n)
        h0 = project_sum_box(h0, target)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---------- sub‑gradient phase (≈70 % of budget) ----------
    while time.time() - start < 0.70 * budget_s:
        # start from a perturbed copy of the current best
        h_start = best_h.copy()
        i, j = rng.integers(0, n, size=2)
        eps = min(h_start[i], 1.0 - h_start[j]) * rng.random()
        h_start[i] -= eps
        h_start[j] += eps
        h_start = project_sum_box(h_start, target)

        # run a modest descent
        cand_h, cand_c = subgradient_descent(h_start, max_iters=2000, step0=0.3, rng=rng)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()

    # ---------- SLSQP soft‑max refinement ----------
    for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0, 80000.0, 200000.0):
        if time.time() - start > 0.95 * budget_s:
            break
        refined = slsqp_optimize(best_h, beta, max_iter=300)
        rc = compute_c5(refined)
        if rc < best_c:
            best_c, best_h = rc, refined.copy()

    # final sanity projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
