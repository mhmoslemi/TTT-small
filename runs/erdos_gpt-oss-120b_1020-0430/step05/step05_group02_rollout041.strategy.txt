While the initial approach did use simulated annealing and SLSQP, it struggled to find configurations significantly better than the parent. The strategy introduces a multi-phase optimization that explicitly enforces separability between significant peaks in the function and its shifted versions. This addresses the key challenge of preventing the function from correlating too strongly with its shifts. We implement a two-phase method:

2. **Local Refinement**: Applies SLSQP after each restart to refine the solution

The mathematical formulation introduces a separability measure that penalizes configurations where the function becomes too similar to its shifts, which directly reduces the overlap. This is done via a soft penalty term that encourages diversity in peaks/troughs, while maintaining the integral constraint.

def compute_separability(h):
    n = h.size
    if n < 2:
        return -np.inf
    correlations = np.correlate(h, h, mode='full')
    return np.log(np.mean(correlations[0:n] - correlations[n-1:2*n-1] + 1e-10))

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0-h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def nelder_mead_optimize(x0, alpha=1.0, gamma=2.0, rho=0.5, sigma=0.5, max_iter=5000, tol=1e-6):
    n = len(x0)
    dimension = n
    x = x0.copy()
    best_x = x.copy()
    best_f = compute_separability(best_x)

    def objective(x):
        return compute_separability(x)  # Global scoring function

    current_time = time.time()
    iter_count = 0
    while iter_count < max_iter and time.time() < deadline:
        iter_count += 1
        vertices = [x.copy()]
        f_vals = []
        # Calculate function values for all vertices
        for i in range(dimension):
            xi = x.copy()
            xi[i] = 0.0
            new_f = objective(xi)
            if new_f < best_f:
                best_x = xi.copy()
                best_f = new_f
            vertices.append(xi)
            f_vals.append(new_f)

        # Sort vertices by objective value
        sorted_indices = np.argsort(f_vals)
        vertices = [vertices[i] for i in sorted_indices]
        f_vals = np.array(f_vals)[sorted_indices]
        best_f = f_vals[0]
        best_x_val = vertices[0]

        # Center of simplex
        center = np.mean(vertices[:-1], axis=0)

        # Reflection
        x_ref = center + alpha * (center - vertices[-1])
        f_ref = objective(x_ref)
        if f_ref < f_vals[-1] and f_ref > f_vals[0]:
            x = np.vstack([vertices[:-1], x_ref])
            continue

        # Expansion
        if f_ref < f_vals[0]:
            x_exp = center + gamma * (center - vertices[-1])
            f_exp = objective(x_exp)
            if f_exp < f_ref:
                vertices[-1][:] = x_exp

        # Reduction
        vertices_low = vertices[:int(dimension//2)]
        if allclose(vertices_low, vertices[-1]):
            # Reduce
            vertices = vertices_low + [vertices[-1] * (1 - rho - 1e-5) + vertices[-2] * rho]  # Simplified reduction
        else:

            x_contract = (1 + sigma) * center - sigma * vertices[-1]
            f_contract = objective(x_contract)
            if f_contract < f_vals[-1]:
                vertices[-1] = x_contract

        # Replace with current state
        for i in range(dimension):
            x[i] = vertices[i].copy()

    return best_x_val, best_f
