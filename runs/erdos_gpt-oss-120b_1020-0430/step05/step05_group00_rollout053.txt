import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# globals for SLSQP soft‑max
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            if h[i] >= 1.0:
                continue
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff = target - h.sum()
        else:
            if h[i] <= 0.0:
                continue
            sub = min(-diff, h[i])
            h[i] -= sub
            diff = target - h.sum()
    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def apply_delta(corr: np.ndarray, h: np.ndarray, b: np.ndarray,
                i: int, j: int) -> None:
    """
    In‑place update of full convolution `corr = h * b` after swapping
    a 1 at `i` with a 0 at `j`.  Arrays h,b are the *old* values.
    """
    n = h.size
    # -b shifted by i
    corr[i:i + n] -= b
    # +b shifted by j
    corr[j:j + n] += b
    # +h shifted by i
    corr[i:i + n] += h
    # -h shifted by j
    corr[j:j + n] -= h
    # conv(dh,db) contribution
    corr[n - 1] -= 2.0
    corr[i - j + n - 1] += 1.0
    corr[j - i + n - 1] += 1.0


def binary_sa(N: int, rng: np.random.Generator, time_budget: float,
              init_bin: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on binary vectors with exactly N/2 ones.
    Returns best vector and its C5.
    """
    start = time.time()
    best_h = None
    best_c = np.inf
    target_ones = N // 2

    while time.time() - start < time_budget:
        # ----- initialise a binary start -----
        if init_bin is not None:
            h = init_bin.copy()
            init_bin = None
        else:
            h = np.zeros(N, dtype=float)
            ones = rng.choice(N, target_ones, replace=False)
            h[ones] = 1.0

        b = 1.0 - h
        corr = np.correlate(h, b, mode="full")
        cur_c = np.max(corr) * (2.0 / N)

        max_iters_local = 50000
        for it in range(max_iters_local):
            if time.time() - start >= time_budget:
                break

            # temperature decreases linearly with global progress
            prog = (time.time() - start) / time_budget
            T = max(0.001, (1.0 - prog) * 0.03)

            # pick a random 1 and a random 0
            while True:
                i = rng.integers(N)
                if h[i] > 0.5:
                    break
            while True:
                j = rng.integers(N)
                if h[j] < 0.5:
                    break

            new_corr = corr.copy()
            apply_delta(new_corr, h, b, i, j)
            new_c = np.max(new_corr) * (2.0 / N)
            delta = new_c - cur_c

            if delta < 0.0 or rng.random() < np.exp(-delta / T):
                # accept swap
                h[i], h[j] = h[j], h[i]
                b[i], b[j] = b[j], b[i]
                corr = new_corr
                cur_c = new_c
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        # continue with a fresh start if time permits
    if best_h is None:
        # fallback: uniform half‑ones
        best_h = np.zeros(N, dtype=float)
        best_h[:target_ones] = 1.0
        best_c = compute_c5(best_h)
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # decide discretisation size
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        N = init_h.size
    else:
        N = 512

    # turn any initial float vector into a binary start (if possible)
    init_bin = None
    if 'initial_h_values' in globals():
        bin_vec = (init_h > 0.5).astype(float)
        target = N // 2
        diff = target - bin_vec.sum()
        if diff > 0:
            zeros = np.where(bin_vec == 0)[0]
            add = rng.choice(zeros, int(diff), replace=False)
            bin_vec[add] = 1.0
        elif diff < 0:
            ones = np.where(bin_vec == 1)[0]
            rem = rng.choice(ones, int(-diff), replace=False)
            bin_vec[rem] = 0.0
        init_bin = bin_vec

    # ------------------- binary SA (majority of time) --------------------
    time_sa = total_budget * 0.65
    best_h, best_c = binary_sa(N, rng, time_sa, init_bin)

    # ------------------- continuous refinement (remaining time) ----------
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 5.0:
        # soft‑max SLSQP with growing β
        for beta in (20., 80., 200., 800., 3000., 8000.):
            if time.time() - start > total_budget * 0.95:
                break
            best_h = slsqp_optimize(best_h, beta, rng,
                                    max_iter=800 if beta >= 200 else 500)
            c_val = compute_c5(best_h)
            if c_val < best_c:
                best_c = c_val

        # short gradient tidy‑up
        if time.time() - start < total_budget * 0.97:
            best_h = gradient_refine(best_h, rng, step=0.015, max_iter=10)
            c_val = compute_c5(best_h)
            if c_val < best_c:
                best_c = c_val

    # ------------------- final sanitisation ----------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
