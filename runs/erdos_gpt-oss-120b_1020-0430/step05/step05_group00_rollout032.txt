def overlap_vector(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    # length 2n-1; k from 0 to n-1 corresponds to corr[n-1: n-1 + n]
    return corr[n-1:]

def compute_c5(h):
    ov = overlap_vector(h)
    return np.max(ov) * (2.0 / h.size)

def gradient_descent(h0, max_iters, target_sum, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        grad = worst_gradient(h)
        # step size schedule: maybe step = 0.2 / sqrt(it+1) or 0.1 * (1 / (it+1))
        step = 0.3 / np.sqrt(it + 1)  # adjust
        # direction: negative gradient (to minimize)
        y = h - step * grad
        # project onto constraints
        h = project_onto_constraints(y, target_sum)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # occasional random perturbation to escape

            # add small random noise then project
            noise = (rng.random(h.size) - 0.5) * 0.02
            h = project_onto_constraints(h + noise, target_sum)
    return best_h, best_c

Let's attempt incremental update: For binary h, O_k = sum_i h_i * (1 - h_{i+k}); for each swap i (1->0) and j (0->1), effect on O_k can be computed as:

Let delta_i(k) = - (1 - h_{i+k}) after changing h_i from 1 to 0 (since term h_i term disappears). For i, before swap h_i=1, after h_i'=0, the term contribution for shift k is h_i*(1 - h_{i+k}) = (1 - h_{i+k}); after swap becomes 0. So delta_i(k) = - (1 - h_{i+k}). However note that for i+k may exceed n-1: then term not present originally (since correlation only up to n-1-k). But we need to handle only overlapping region length n - k for each k. For i >= n - k, previous term not present; so delta_i(k)=0. Also note that when we swap i, the term for shift -k (i.e., where h_{i-k} is multiplied by (1 - h_i)) also changes because h_i appears as (1 - h_i) factor in terms where it is shifted element. Indeed O_k includes h_{p}*(1 - h_{p+k}), but for shift -k (i.e., O_{-k}) we would have h_{p-k}*(1 - h_{p}) when p >= k. But O_k for k>=0 does not include (1 - h_i) as factor; only h_i is factor. However O_k for shift values includes term where i appears as h_{i - k} * (1 - h_i) for i>=k. In O_k sum we have terms over p = i - k. So swap at i changes O_k for k where p = i - k >= 0 => k <= i. So there are contributions for each k where i >= k: term is h_{i - k}*(1 - h_i). Changing h_i from 1 to 0 changes (1 - h_i) from 0 to 1, so delta_i'(k) = + h_{i - k}.

Thus total delta from swapping i (1->0) is:

delta_i(k) = - (1 - h_{i + k}) (if i + k < n) + h_{i - k} (if i - k >= 0).

Similarly for j (0 -> 1) contribution:

Before swap h_j=0; after 1. The contributions:

delta_j(k) = + (1 - h_{j + k}) (if j + k < n) - h_{j - k} (if j - k >= 0).

Combine both to get new O_k = O_k + delta_i(k) + delta_j(k).

Thus we can compute delta for all k efficiently using vectorized operations: compute arrays for k from 0 to n-1.

def swap_delta_O(h, i, j):
    n = len(h)
    k = np.arange(n)
    # For i to 0:
    # term1 = -(1 - h[i + k]) where i + k < n else 0
    term1 = np.zeros(n)
    idx = i + k < n
    term1[idx] = -(1 - h[i + k[idx]])
    # term2 = + h[i - k] where i - k >=0 else 0
    idx2 = i - k >= 0
    term2 = np.zeros(n)
    term2[idx2] = h[i - k[idx2]]
    delta_i = term1 + term2

    # For j from 0 to 1
    term1 = np.zeros(n)
    idx = j + k < n
    term1[idx] = (1 - h[j + k[idx]])
    term2 = np.zeros(n)
    idx2 = j - k >= 0
    term2[idx2] = -h[j - k[idx2]]
    delta_j = term1 + term2

    delta = delta_i + delta_j
    return delta

    start = time.time()
    # Parameters
    run_time = 10.0  # seconds per SA run
    while time.time() - start < total_budget:
        # initialize random binary vector
        h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, n//2, replace=False)
        h[ones_idx] = 1.0
        c_curr = compute_c5(h)
        # run SA for at most run_time seconds
        run_start = time.time()
        while time.time() - run_start < run_time and time.time() - start < total_budget:
            # compute temperature based on elapsed fraction of run_time
            t_elapsed = time.time() - run_start
            T = 0.02 * (1e-4 / 0.02) ** (t_elapsed / run_time)  # similar schedule
            # pick random swap
            i = rng.choice(np.where(h == 1)[0])
            j = rng.choice(np.where(h == 0)[0])
            # swap
            h[i] = 0.0; h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                if c_new < best_c:

                    # early break if good enough
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                # revert
                h[i] = 1.0; h[j] = 0.0
        # after this run, maybe continue to next run
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c

def fix_sum_binary(h, target, rng):
    # h is array of 0/1 possibly not exactly sum
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(h == 0)[0]
        chosen = rng.choice(zeros, diff, replace=False)
        h[chosen] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        chosen = rng.choice(ones, -diff, replace=False)
        h[chosen] = 0
    return h

import time
import numpy as np

# ------------------------------------------------------------
def overlap_vector(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h); returns O_k for k=0..n-1."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    return corr[n - 1 : n - 1 + n]


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function."""
    n = h.shape[0]
    return np.max(overlap_vector(h)) * (2.0 / n)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.shape[0]
    ov = overlap_vector(h)
    k = int(np.argmax(ov))
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def fix_sum_binary(h: np.ndarray, target: int, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = target while staying binary."""
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(h == 0)[0]
        chosen = rng.choice(zeros, diff, replace=False)
        h[chosen] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        chosen = rng.choice(ones, -diff, replace=False)
        h[chosen] = 0
    return h


def enforce_sum_float(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def project_onto_constraints(y: np.ndarray, target: float) -> np.ndarray:
    """Projection of y onto [0,1]^n ∩ {sum = target} by bisection."""
    lo, hi = -1e6, 1e6
    for _ in range(30):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
def projected_gradient_descent(
    h0: np.ndarray, target_sum: float, time_limit: float, rng: np.random.Generator
):
    """Simple projected gradient descent on the worst‑shift objective."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    while time.time() - start < time_limit:
        grad = worst_gradient(h)
        it += 1
        step = 0.4 / np.sqrt(it)
        y = h - step * grad
        h = project_onto_constraints(y, target_sum)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if it % 200 == 0 and rng.random() < 0.3:
            noise = (rng.random(h.size) - 0.5) * 0.01
            h = project_onto_constraints(h + noise, target_sum)
    return best_h, best_c
