import time, numpy as np
from scipy.optimize import minimize

def compute_c5_full(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def smooth_max_corr(h_half, beta):
    # h_half length M
    h = np.concatenate([h_half, 1.0 - h_half])
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize_half(h0, beta, max_iter=1500):
    M = h0.size
    bounds = [(0.0, 1.0)] * M
    # No constraints needed; integral automatically satisfied.
    res = minimize(smooth_max_corr, h0, args=(beta,), method='SLSQP',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    h_opt = np.clip(res.x, 0.0, 1.0)
    return h_opt

def gradient_refine_half(h_half, step=0.02, max_iter=30):
    M = h_half.size
    n = 2 * M
    for it in range(max_iter):
        # compute full h and correlation
        h = np.concatenate([h_half, 1.0 - h_half])
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        r = abs(lag)
        if r == 0:
            # gradient of self overlap: g_full = 1 - 2h
            g_full = 1.0 - 2.0 * h
        else:
            # compute gradient g_full using formula
            # first part: j in 0..n-r-1 term (1 - h[j+r])
            g_full = np.zeros_like(h)
            # part1:
            g_full[:n - r] = 1.0 - h[r:]
            # part2:
            g_full[r:] -= h[:n - r]
        # map to half variables:
        g_half = g_full[:M] - g_full[M:]
        # apply step
        step_sz = step * (0.5 ** it)
        h_half = h_half - step_sz * g_half
        # project onto [0,1]
        h_half = np.clip(h_half, 0.0, 1.0)
    return h_half

def binary_from_half(h_half):
    n = 2 * h_half.size
    h_full = np.concatenate([h_half, 1.0 - h_half])
    M = n // 2
    # select M largest entries
    idx = np.argpartition(h_full, -M)[-M:]
    bin_h = np.zeros_like(h_full)
    bin_h[idx] = 1.0
    # ensure exactly M ones (should be true)
    return bin_h

def binary_targeted_improve(bin_h, deadline, rng, max_swaps=2000):
    n = bin_h.size
    while time.time() < deadline and max_swaps > 0:
        max_swaps -= 1
        # compute correlation
        corr = np.correlate(bin_h, 1.0 - bin_h, mode='full')
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        k = abs(lag)
        if k == 0:
            break
        # compute neighbor sum for each position
        ar = np.arange(n)
        neigh = bin_h[(ar + k) % n] + bin_h[(ar - k) % n]
        ones = np.flatnonzero(bin_h == 1)
        zeros = np.flatnonzero(bin_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        # pick i from ones with minimal neighbor sum
        min_val = neigh[ones].min()
        cand_i = ones[neigh[ones] == min_val]
        i = rng.choice(cand_i)
        # pick j from zeros with maximal neighbor sum
        max_val = neigh[zeros].max()
        cand_j = zeros[neigh[zeros] == max_val]
        j = rng.choice(cand_j)
        # calculate delta in overlap of worst shift after swap
        # Original contribution of i: bin_h[i] * (1 - bin_h[(i±k)%n]) adds to overlap
        # But easier: compute delta = neigh[i] - neigh[j]
        delta = neigh[i] - neigh[j]
        if delta < 0:
            bin_h[i] = 0.0
            bin_h[j] = 1.0
        else:
            # try random swaps
            improved = False
            for _ in range(4):
                i2 = rng.choice(ones)
                j2 = rng.choice(zeros)
                neigh_i2 = neigh[i2]
                neigh_j2 = neigh[j2]
                if neigh_i2 - neigh_j2 < 0:
                    bin_h[i2] = 0.0
                    bin_h[j2] = 1.0
                    improved = True
                    break
            if not improved:
                break
    return bin_h

def upsample_half(h_half):
    M = h_half.size
    M2 = M * 2
    old_x = np.linspace(0.0, 1.0, M, endpoint=False)
    new_x = np.linspace(0.0, 1.0, M2, endpoint=False)
    h_half_up = np.interp(new_x, old_x, h_half)
    # Ensure bounds [0,1]
    h_half_up = np.clip(h_half_up, 0.0, 1.0)
    return h_half_up

def optimize_at_resolution(M, rng, deadline):
    # M = n//2
    n = 2 * M
    best_half = None
    best_c = np.inf
    # starting points
    starts = []
    # Use initial_h_values if present and size matches
    if 'initial_h_values' in globals():
        h_full_parent = np.array(initial_h_values, dtype=float)
        if h_full_parent.size == n:
            # extract half as first M
            starts.append(h_full_parent[:M].copy())
    # random starts
    for _ in range(4):
        h_half = rng.random(M)
        starts.append(h_half)
    # beta schedule
    betas = [5., 10., 20., 50., 100., 200., 500., 1000., 2000., 5000., 10000.]
    for h0 in starts:
        if time.time() >= deadline:
            break
        # Clip
        h = np.clip(h0, 0.0, 1.0)
        # SLSQP cascade
        for beta in betas:
            if time.time() >= deadline:
                break
            max_iter = 1200 if beta < 200 else 1800
            h = slsqp_optimize_half(h, beta, max_iter=max_iter)
            c = compute_c5_full(np.concatenate([h, 1.0 - h]))
            if c < best_c:
                best_half = h.copy()
                best_c = c
        # gradient refinement
        h = gradient_refine_half(h, step=0.03, max_iter=30)
        c = compute_c5_full(np.concatenate([h, 1.0 - h]))
        if c < best_c:
            best_half = h.copy()
            best_c = c
        # binary rounding and improve
        bin_h = binary_from_half(h)
        bin_h = binary_targeted_improve(bin_h, deadline, rng, max_swaps=5000)
        c_bin = compute_c5_full(bin_h)
        if c_bin < best_c:
            best_half = None  # keep full binary
            best_c = c_bin
            best_full_bin = bin_h.copy()
            # store full binary for return
            best_full = bin_h.copy()
    # return best solution at this resolution (full)
    if best_half is not None:
        full_h = np.concatenate([best_half, 1.0 - best_half])
    else:
        full_h = best_full
    return full_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97
    # Base resolution 360 points (M=180)
    M = 180
    best_h, best_c = None, np.inf
    # Run optimization at base resolution
    h_base, c_base = optimize_at_resolution(M, rng, deadline)
    best_h, best_c = h_base, c_base
    # If time permits, upsample to 720
    if time.time() < deadline - 30:  # leave time for refinement
        # Upsample half
        half_base = best_h[:M]  # first half of best solution
        half_up = upsample_half(half_base)
        M_up = half_up.size
        # Optimize at higher resolution
        h_up, c_up = optimize_at_resolution(M_up, rng, deadline)
        if c_up < best_c:
            best_h, best_c = h_up, c_up
    # Final cleanup: ensure bounds and sum
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure integral ~ 1 (symmetry ensures; but for binary may not if not symmetric)
    n_final = best_h.size
    # If not symmetric, enforce exact sum:
    total = best_h.sum() * (2.0 / n_final)
    if abs(total - 1.0) > 1e-9:
        # adjust by scaling? Since sum should be n/2 for symmetric double, but not for binary random maybe.
        # We'll adjust by adding/subtracting small amounts in a loop.
        target_sum = n_final / 2.0
        diff = target_sum - best_h.sum()
        # distribute difference across random indices
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n_final)
            if diff > 0:
                add = min(diff, 1.0 - best_h[i])
            else:
                add = max(diff, -best_h[i])
            best_h[i] += add
            diff = target_sum - best_h.sum()
    final_c = compute_c5_full(best_h)
    return best_h.tolist(), float(final_c), int(n_final)

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5_full(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def smooth_max_corr(h_half, beta):
    """Smooth max of overlaps for a half‑vector (symmetrised function)."""
    h = np.concatenate([h_half, 1.0 - h_half])
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def slsqp_optimize_half(h0, beta, max_iter=1500):
    M = h0.size
    bounds = [(0.0, 1.0)] * M
    res = minimize(
        smooth_max_corr,
        h0,
        args=(beta,),
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    return np.clip(res.x, 0.0, 1.0)

def gradient_refine_half(h_half, step=0.03, max_iter=30):
    M = h_half.size
    n = 2 * M
    for it in range(max_iter):
        h = np.concatenate([h_half, 1.0 - h_half])
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        r = abs(lag)
        if r == 0:
            g_full = 1.0 - 2.0 * h
        else:
            g_full = np.zeros_like(h)
            g_full[: n - r] = 1.0 - h[r:]
            g_full[r:] -= h[: n - r]
        g_half = g_full[:M] - g_full[M:]
        step_sz = step * (0.5 ** it)
        h_half = h_half - step_sz * g_half
        h_half = np.clip(h_half, 0.0, 1.0)
    return h_half

def binary_from_half(h_half):
    n = 2 * h_half.size
    h_full = np.concatenate([h_half, 1.0 - h_half])
    M = n // 2
    idx = np.argpartition(h_full, -M)[-M:]
    bin_h = np.zeros_like(h_full)
    bin_h[idx] = 1.0
    return bin_h

def binary_targeted_improve(bin_h, deadline, rng, max_swaps=4000):
    n = bin_h.size
    while time.time() < deadline and max_swaps > 0:
        max_swaps -= 1
        corr = np.correlate(bin_h, 1.0 - bin_h, mode="full")
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        k = abs(lag)
        if k == 0:
            break
        ar = np.arange(n)
        neigh = bin_h[(ar + k) % n] + bin_h[(ar - k) % n]
        ones = np.flatnonzero(bin_h == 1)
        zeros = np.flatnonzero(bin_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        min_val = neigh[ones].min()
        cand_i = ones[neigh[ones] == min_val]
        i = rng.choice(cand_i)
        max_val = neigh[zeros].max()
        cand_j = zeros[neigh[zeros] == max_val]
        j = rng.choice(cand_j)
        delta = neigh[i] - neigh[j]
        if delta < 0:
            bin_h[i] = 0.0
            bin_h[j] = 1.0
            continue
        improved = False
        for _ in range(3):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                bin_h[i2] = 0.0
                bin_h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return bin_h

def upsample_half(h_half):
    M = h_half.size
    M2 = M * 2
    old_x = np.linspace(0.0, 1.0, M, endpoint=False)
    new_x = np.linspace(0.0, 1.0, M2, endpoint=False)
    h_up = np.interp(new_x, old_x, h_half)
    return np.clip(h_up, 0.0, 1.0)

def optimize_at_resolution(M, rng, deadline):
    n = 2 * M
    best_half = None
    best_full = None
    best_c = np.inf

    starts = []
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size == n:
            starts.append(parent[:M].copy())
    for _ in range(6):
        starts.append(rng.random(M))

    betas = [5., 10., 20., 50., 100., 200., 500., 1000., 2000., 5000., 10000.]
    for h0 in starts:
        if time.time() >= deadline:
            break
        h = np.clip(h0, 0.0, 1.0)
        for beta in betas:
            if time.time() >= deadline:
                break
            max_iter = 1200 if beta < 200 else 1800
            h = slsqp_optimize_half(h, beta, max_iter=max_iter)
            c = compute_c5_full(np.concatenate([h, 1.0 - h]))
            if c < best_c:
                best_c = c
                best_half = h.copy()
                best_full = None
        h = gradient_refine_half(h, step=0.03, max_iter=30)
        c = compute_c5_full(np.concatenate([h, 1.0 - h]))
        if c < best_c:
            best_c = c
            best_half = h.copy()
            best_full = None
        bin_h = binary_from_half(h)
        bin_h = binary_targeted_improve(bin_h, deadline, rng, max_swaps=5000)
        c_bin = compute_c5_full(bin_h)
        if c_bin < best_c:
            best_c = c_bin
            best_half = None
            best_full = bin_h.copy()
    if best_half is not None:
        full_h = np.concatenate([best_half, 1.0 - best_half])
    elif best_full is not None:
        full_h = best_full
    else:
        # fallback to a simple symmetric step
        half = np.zeros(M)
        half[: M // 2] = 1.0
        full_h = np.concatenate([half, 1.0 - half])
    return full_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    M_base = 180
    best_h, best_c = optimize_at_resolution(M_base, rng, deadline)

    # optional up‑sampling if time permits
    if time.time() < deadline - 30:
        half_base = best_h[:M_base]
        half_up = upsample_half(half_base)
        M_up = half_up.size
        best_h_up, best_c_up = optimize_at_resolution(M_up, rng, deadline)
        if best_c_up < best_c:
            best_h, best_c = best_h_up, best_c_up

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = best_h.size
    target_sum = n_final / 2.0
    diff = target_sum - best_h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n_final)
        add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
        best_h[i] += add
        diff = target_sum - best_h.sum()
    final_c = compute_c5_full(best_h)
    return best_h.tolist(), float(final_c), int(n_final)
