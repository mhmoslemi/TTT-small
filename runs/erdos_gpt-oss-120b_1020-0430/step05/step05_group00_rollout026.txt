def softmax_c5_and_grad(h, beta):
    n = h.size
    dx = 2.0 / n
    # Compute O array via FFT
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1 - h, N)
    O = np.fft.irfft(H * G, N)[:2*n-1] * dx  # O_i values with dx factor included? Need to incorporate dx factor consistently.
    # Actually compute correlation of h and (1-h) without dx factor: corr = np.fft.ifft(H*G).real[:2n-1] ; then O_i = corr_i * dx.
    corr = np.fft.irfft(H * G, N)[:2*n-1]
    O = corr * dx
    Omax = O.max()
    exp_beta = np.exp(beta * (O - Omax))
    sum_exp = exp_beta.sum()
    p = exp_beta / sum_exp
    # gradient components
    # Compute conv1 = sum_i p_i * S_i h = convolution of h with p_rev (len 2n-1)

import numpy as np, time

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1 - h, N)
    corr = np.fft.irfft(H * G, N)[:2*n-1]
    return np.max(corr) * dx

def project_sum_bounds(h, target, rng):
    # enforce box constraints first
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # distribute proportionally
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - h
            total_slack = slack.sum()
            if total_slack > 0:
                h += diff * slack / total_slack
        else:
            slack = h
            total_slack = slack.sum()
            if total_slack > 0:
                h += diff * slack / total_slack
        h = np.clip(h, 0.0, 1.0)
    # final exact correction via random adjustments
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        n = h.size
        for _ in range(2000):
            if abs(diff) <= 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff = target - h.sum()
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff = target - h.sum()
    return h

def softmax_c5_and_grad(h, beta):
    n = h.size
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1 - h, N)
    corr = np.fft.irfft(H * G, N)[:2*n-1]
    O = corr * dx
    Omax = O.max()
    exp_beta = np.exp(beta * (O - Omax))
    sum_exp = exp_beta.sum()
    p = exp_beta / sum_exp
    # gradient
    # conv1 = sum_i p_i * S_i h = convolution of h with p_rev
    p_rev = p[::-1]
    conv1 = np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(p_rev, N), N)[:n]
    conv2 = np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(p, N), N)[:n]
    grad = dx * (1.0 - (conv1 + conv2))
    obj = (Omax + np.log(sum_exp)) / beta
    return obj, grad

def binary_sa_fast(n, rng, time_budget):
    start = time.time()
    target = n / 2
    # init random binary
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr

    # store indices
    ones_arr = np.sort(ones)
    zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
    zeros_arr = np.sort(zeros_arr)

    T0 = 0.01
    T_end = 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # pick random swap
        i_idx = rng.integers(len(ones_arr))
        j_idx = rng.integers(len(zeros_arr))
        i = ones_arr[i_idx]
        j = zeros_arr[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_curr = c_new
            # update indices
            ones_arr[i_idx] = j
            zeros_arr[j_idx] = i
            # keep arrays sorted for potential speed? But not necessary.
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    break
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def worst_gradient(h):
    n = h.size
    # compute correlation via FFT
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1 - h, N)
    corr = np.fft.irfft(H * G, N)[:2*n-1]
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        # For i in [0, n-k): grad[i] -= h[i+k]
        grad[:n - k] -= h[k:]
        # For i in [k, n): grad[i] -= h[i - k]
        grad[k:] -= h[:n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def gradient_descent_refine(h, rng, time_budget, target):
    start = time.time()
    n = h.size
    # schedule beta stages
    beta_stages = [(20., 2000), (200., 1500), (2000., 800), (8000., 400)]
    for beta, max_steps in beta_stages:
        # allocate steps based on remaining time
        remaining = time_budget - (time.time() - start)
        if remaining <= 0:
            break
        # approximate time per step ~ (maybe 0.001 s)
        # set steps = min(max_steps, int(remaining * 400))
        steps = int(min(max_steps, max(100, remaining * 400)))
        step_size = 0.1
        for _ in range(steps):
            obj, grad = softmax_c5_and_grad(h, beta)
            # gradient descent (minimize obj) -> move opposite to gradient
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            # enforce sum constraint
            h_new = project_sum_bounds(h_new, target, rng)
            h = h_new
            # decay step size slowly
            step_size *= 0.9995
            # early exit if time low
            if time.time() - start > time_budget:
                break
    return h

def upsample_solution(h, n_new, rng):
    n_old = h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n_new / 2.0
    h_new = project_sum_bounds(h_new, target, rng)
    return h_new

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1 - h, N)
    corr = np.fft.irfft(H * G, N)[:2 * n - 1]
    return np.max(corr) * dx

def project_sum_bounds(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - h
            tot = slack.sum()
            if tot > 0:
                h += diff * slack / tot
        else:
            slack = h
            tot = slack.sum()
            if tot > 0:
                h += diff * slack / tot
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        n = h.size
        for _ in range(2000):

            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff = target - h.sum()
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff = target - h.sum()
    return h

def softmax_c5_and_grad(h, beta):
    n = h.size
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1 - h, N)
    corr = np.fft.irfft(H * G, N)[:2 * n - 1]
    O = corr * dx
    Omax = O.max()
    exp_beta = np.exp(beta * (O - Omax))
    s = exp_beta.sum()
    p = exp_beta / s
    p_rev = p[::-1]
    conv1 = np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(p_rev, N), N)[:n]
    conv2 = np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(p, N), N)[:n]
    grad = dx * (1.0 - (conv1 + conv2))
    obj = (Omax + np.log(s)) / beta
    return obj, grad

def binary_sa_fast(n, rng, time_budget):
    start = time.time()
    target = n / 2
    h = np.zeros(n, dtype=float)

    best_h = h.copy()
    best_c = c_curr
    ones_arr = np.sort(ones)
    zeros_arr = np.sort(np.setdiff1d(np.arange(n), ones_arr))
    T0, Tend = 0.01, 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (Tend / T0) ** (elapsed / time_budget)
        i_idx = rng.integers(len(ones_arr))
        j_idx = rng.integers(len(zeros_arr))
        i, j = ones_arr[i_idx], zeros_arr[j_idx]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):

            ones_arr[i_idx] = j

                if best_c <= 0.3808:

            h[i] = 1.0
            h[j] = 0.0

def worst_gradient(h):
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1 - h, N)
    corr = np.fft.irfft(H * G, N)[:2 * n - 1]
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def gradient_descent_refine(h, rng, time_budget, target):

    beta_stages = [(20., 2000), (200., 1500), (2000., 800), (8000., 400)]

def upsample_solution(h, n_new, rng):
    n_old = h.size
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)

    h_new = np.clip(h_new, 0.0, 1.0)
