    start = time.time()
    deadline = start + budget_s * 0.99

    rng = np.random.default_rng(seed)

    n = 360 # default size maybe can also try other n like 720 if time permits
    # get initial binary

            h[idx] = 1
            cur_sum += 1

    deltas = []

def simulated_annealing(n, rng, max_iters, T_start=2.0, T_end=0.001):
    pad = n - 1
    idx = np.arange(2*pad+1)
    # initialize random binary h
    h = np.zeros(n, dtype=np.int8)
    h[: n//2] = 1
    rng.shuffle(h)
    # compute O
    O = np.correlate(h, 1 - h, mode='full')
    cur_max = O.max()
    best_max = cur_max
    best_h = h.copy()
    # padded
    h_padded = np.concatenate([np.zeros(pad, dtype=np.int8), h, np.zeros(pad, dtype=np.int8)])

        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]; j = zeros[j_idx]

        delta = tmp_i - tmp_j
        new_max = (O + delta).max()

def run(...):
    import time, numpy as np, scipy.optimize as opt

    start = time.time()
    deadline = start + budget_s * 0.99

    rng = np.random.default_rng(seed)

    best_c = np.inf
    best_h = None
    best_n = None

    # define compute_c5
    def compute_c5(h):
        n = h.size
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(corr) * (2.0 / n))

    # adjust sum exact as earlier
    def adjust_sum_exact(h):
        # h is 1D numpy array float
        target = .5 * h.size
        diff = target - h.sum()
        # iterate until diff near zero
        # just adjust random entries.
        n = h.size
        while abs(diff) > 1e-12:
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            if abs(add) < 1e-12:
                # Choose another i? break if cannot adjust
                break
            h[i] += add
            diff = target - h.sum()
        return h

    # Define SLSQP optimizer
    def slsqp_optimize(x0, beta, max_iter=800):
        n = x0.size
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
        # define objective
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            bc = beta * corr
            m = np.max(bc)
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta
        res = opt.minimize(obj, x0, method="SLSQP", bounds=bounds,
                           constraints=cons,
                           options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
        x = np.clip(res.x, 0.0, 1.0)
        adjust_sum_exact(x)
        return x

    # define SA for binary
    def sa_binary(n, max_iters, time_limit):
        # returns best_h (int8), best_max (int)
        pad = n - 1
        idx_arr = np.arange(2*pad+1, dtype=np.int32)
        target = n // 2
        # start from random binary
        h = np.zeros(n, dtype=np.int8)
        h[:target] = 1
        rng.shuffle(h)
        O = np.correlate(h, 1 - h, mode='full')
        cur_max = int(O.max())
        best_max = cur_max
        best_h = h.copy()
        h_padded = np.concatenate([np.zeros(pad, dtype=np.int8), h, np.zeros(pad, dtype=np.int8)])
        ones = list(np.flatnonzero(h))
        zeros = list(np.flatnonzero(1 - h))
        # temperature schedule
        T_start = 2.0
        T_end = 0.001
        for it in range(max_iters):
            if time.time() > time_limit:
                break
            # linear temperature
            T = T_start + (T_end - T_start) * (it / max_iters)
            i_idx = rng.integers(len(ones))
            j_idx = rng.integers(len(zeros))
            i = ones[i_idx]
            j = zeros[j_idx]
            # compute delta efficiently
            tmp_i = h_padded[i + idx_arr] + h_padded[i + 2*pad - idx_arr]
            tmp_j = h_padded[j + idx_arr] + h_padded[j + 2*pad - idx_arr]
            delta = tmp_i - tmp_j
            new_max = int((O + delta).max())
            delta_max = new_max - cur_max
            accept = False
            if delta_max <= 0:
                accept = True
            else:
                if rng.random() < np.exp(-delta_max / T):
                    accept = True
            if accept:
                O = O + delta
                cur_max = new_max
                # swap bits
                h[i] = 0
                h[j] = 1
                h_padded[pad + i] = 0
                h_padded[pad + j] = 1
                # update lists
                ones[i_idx] = j
                zeros[j_idx] = i
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
            # else no changes
        return best_h, best_max

    # Main loop over candidate n
    candidate_ns = [360, 720]  # can add more if time
    for n in candidate_ns:
        if time.time() > deadline:
            break
        # allocate time for this n: proportion of remaining time
        time_for_n = (deadline - time.time()) * 0.5  # half for SA, half for SLSQP maybe
        # SA restarts
        best_h_bin = None
        best_max_bin = None
        # we will run SA with restarts until time_for_n/2 used
        sa_deadline = time.time() + time_for_n * 0.5
        # in each restart run e.g., 20000 iterations (fast)
        restart_iters = 30000
        while time.time() < sa_deadline:
            h_bin, max_bin = sa_binary(n, restart_iters, sa_deadline)
            if best_max_bin is None or max_bin < best_max_bin:
                best_max_bin = max_bin
                best_h_bin = h_bin.copy()
        # after SA.
        if best_h_bin is None:
            # fallback: balanced binary
            h_bin = np.zeros(n, dtype=np.int8)
            h_bin[: n//2] = 1
            best_h_bin = h_bin.copy()
            best_max_bin = int(np.max(np.correlate(h_bin, 1 - h_bin, mode='full')))
        # compute C5 for this binary
        c5_bin = best_max_bin * (2.0 / n)
        if c5_bin < best_c:
            best_c = c5_bin
            best_h = best_h_bin.astype(float)
            best_n = n
            # best_max = best_max_bin

        # Now refine with SLSQP if time left
        # allocate remaining half of time_for_n
        if time.time() < deadline:
            # start from float version of best binary
            h_float = best_h_bin.astype(float)
            # adjust sum exactly
            adjust_sum_exact(h_float)
            # schedule beta values
            betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]
            for beta in betas:
                if time.time() > deadline:
                    break
                # max_iter can be adjusted based on beta; bigger beta may need more iter
                max_iter = 800 if beta >= 2000 else 500
                try:
                    h_float = slsqp_optimize(h_float, beta, max_iter=max_iter)
                except Exception:
                    continue
                c_new = compute_c5(h_float)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_float.copy()
                    best_n = n
            # final safety adjust
            best_h = np.clip(best_h, 0.0, 1.0)
            adjust_sum_exact(best_h)
    # Ensure final best exists
    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n//2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)

import time
import numpy as np
import scipy.optimize as opt

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if abs(add) < 1e-15:
            break
        h[i] += add
        diff = target - np.sum(h)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def smoothmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(x0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        smoothmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def sa_binary(n, max_iters, deadline, rng):
    pad = n - 1
    idx_arr = np.arange(2 * pad + 1, dtype=np.int32)
    target = n // 2
    h = np.zeros(n, dtype=np.int8)
    h[:target] = 1
    rng.shuffle(h)
    O = np.correlate(h, 1 - h, mode="full")
    cur_max = int(O.max())
    best_max = cur_max
    best_h = h.copy()
    h_padded = np.concatenate([np.zeros(pad, dtype=np.int8), h, np.zeros(pad, dtype=np.int8)])
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    T_start, T_end = 2.0, 0.001
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        tmp_i = h_padded[i + idx_arr] + h_padded[i + 2 * pad - idx_arr]
        tmp_j = h_padded[j + idx_arr] + h_padded[j + 2 * pad - idx_arr]
        delta = tmp_i - tmp_j
        new_max = int((O + delta).max())
        delta_max = new_max - cur_max
        accept = delta_max <= 0 or rng.random() < np.exp(-delta_max / T)
        if accept:
            O = O + delta
            cur_max = new_max
            h[i] = 0
            h[j] = 1
            h_padded[pad + i] = 0
            h_padded[pad + j] = 1
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    return best_h, best_max


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99
    TARGET = 0.3808

    best_h = None
    best_c = np.inf
    best_n = None

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        c_parent = compute_c5(parent)
        if c_parent < best_c:
            best_c = c_parent
            best_h = parent.copy()
            best_n = parent.size

    candidate_ns = [360, 720]

    for n in candidate_ns:
        if time.time() > deadline or best_c < TARGET:
            break

        time_left = deadline - time.time()
        time_sa = time_left * 0.4
        time_slsqp = time_left * 0.6

        # ---------- Simulated annealing ----------
        best_h_bin = None
        best_max_bin = None
        sa_dead = time.time() + time_sa
        while time.time() < sa_dead:
            # 20k iterations per restart (very fast)
            h_bin, max_bin = sa_binary(n, 20000, sa_dead, rng)
            if best_max_bin is None or max_bin < best_max_bin:
                best_max_bin = max_bin
                best_h_bin = h_bin.copy()
                if best_max_bin <= 68:
                    break
        if best_h_bin is None:
            h_bin = np.zeros(n, dtype=np.int8)
            h_bin[: n // 2] = 1
            best_h_bin = h_bin.copy()
            best_max_bin = int(np.max(np.correlate(h_bin, 1 - h_bin, mode="full")))

        c5_bin = best_max_bin * (2.0 / n)
        if c5_bin < best_c:
            best_c = c5_bin
            best_h = best_h_bin.astype(float).copy()
            best_n = n

        # ---------- SLSQP refinement ----------
        if time.time() < deadline:
            if best_h is not None and best_n == n:
                h_float = best_h.copy()
            else:
                h_float = best_h_bin.astype(float).copy()
            adjust_sum_exact(h_float, rng)
            betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]
            for beta in betas:
                if time.time() > deadline:
                    break
                max_iter = 1200 if beta >= 2000 else 800
                try:
                    h_float = slsqp_optimize(h_float, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c_new = compute_c5(h_float)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_float.copy()
                    best_n = n
            best_h = np.clip(best_h, 0.0, 1.0)
            adjust_sum_exact(best_h, rng)

    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
