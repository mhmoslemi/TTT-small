import time, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a step function on [0,2] with uniform sampling."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to satisfy Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (max) shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 20000, step_scale: float = 0.7,
                      time_limit: float = 1e9):
    """Hill‑climb using worst‑gradient transfers."""
    start = time.time()
    best_h, best_c = h.copy(), compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    it = 0
    while it < max_iters and (time.time() - start) < time_limit:
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
        it += 1
    return best_h, best_c


# ---------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- binary simulated annealing ----------
def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    best_h, best_c = None, np.inf
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i = rng.choice(ones_arr)
            j = rng.choice(zeros_arr)
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                # swap indices in the tracking arrays
                idx_i = np.where(ones_arr == i)[0][0]
                idx_j = np.where(zeros_arr == j)[0][0]
                ones_arr[idx_i] = j
                zeros_arr[idx_j] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
    return best_h, best_c


# ---------- resizing ----------
def resize_to_n(h: np.ndarray, n: int, rng: np.random.Generator) -> np.ndarray:
    x_old = np.linspace(0.0, 2.0, h.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return enforce_sum(h_new, rng)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start))

    best_h, best_c, best_n = None, np.inf, None

    # ---- use parent construction if present ----

    # ---- phase 1: binary SA on 256 points ----
    t_alloc = budget_s * 0.12
    if time_left() > t_alloc * 0.9:

                nonlocal best_h, best_c, best_n
                best_h, best_c, best_n = hc.copy(), cc, n
        # final hill‑climb
        hc, cc = guided_hill_climb(hc, rng, max_iters=10000,
                                   step_scale=0.5, time_limit=budget*0.2)
        return hc, cc

    remaining = time_left()
    if remaining > 1.0:
        hc, cc = refine_continuous(best_h, base_n, remaining)
        best_h, best_c = hc, cc

    # ---- phase 3: up‑sample to 512 and further refinement ----
    if best_n < 512 and time_left() > budget_s * 0.15:
        h_up = resize_to_n(best_h, 512, rng)
        best_h, best_c, best_n = h_up, compute_c5(h_up), 512
        remaining = time_left()

    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
