def gradient_step(h, step, rng):
    n = h.shape[0]
    target = n / 2.0
    # correlation
    corr = correlation_fft(h)
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    k = abs(lag)
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        upper = n - k
        g[:upper] += 1.0 - h[k:]    # 1 - h[i+k]
        g[k:] -= h[:upper]          # -h[i]
        # Since we consider both +k and -k shifts due to symmetry, gradient/g is correct as earlier
    # Apply step
    h = h - step * g
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # Project sum
    diff = target - h.sum()
    # Instead of random loops, we can adjust by adding diff equally to all (subject to bounds)
    # Simpler: use same random redistribution.
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def correlation_fft(h):
    n = h.shape[0]
    N = 2 * n - 1
    H = np.fft.rfft(np.concatenate([h, np.zeros(n - 1)]), n=N)
    B = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(n - 1)]), n=N)
    return np.fft.irfft(H * np.conj(B), n=N)

def compute_c5(h):
    n = h.shape[0]
    return float(correlation_fft(h).max() * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = correlation_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=1500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )

    lag = idx - (n - 1)

        h[i] += add
        diff = target - h.sum()
    return h


def gradient_step(h, step, rng):
    n = h.shape[0]
    target = n / 2.0
    corr = correlation_fft(h)
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    k = abs(lag)
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        upper = n - k
        g[:upper] += 1.0 - h[k:]
        g[k:] -= h[:upper]
    h_new = h - step * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_exact(h_new, target, rng)
    return h_new


def local_search(h0, rng, deadline):
    cur = h0.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    n = cur.size
    it = 0
    while time.time() < deadline:
        T = max(1e-12, (0.01 * best_c) * (0.995 ** it))
        old = cur.copy()
        if rng.random() < 0.3:
            i, j = rng.integers(0, n, size=2, replace=False)
            max_transfer = min(cur[i], 1.0 - cur[j])
            if max_transfer > 0:
                delta = rng.random() * max_transfer
                cur[i] -= delta
                cur[j] += delta
        else:
            step = 0.03 * (0.98 ** (it // 200))
            cur = gradient_step(cur, step, rng)
        new_c = compute_c5(cur)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if new_c < best_c:
                best, best_c = cur.copy(), new_c
        else:
            cur = old
        it += 1
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    n0 = 360
    target0 = n0 / 2.0
    start_vecs = []

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n0:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n0, endpoint=False)
            parent = np.interp(xt, xs, parent)
        start_vecs.append(parent.copy())
        idx = np.argpartition(parent, -n0 // 2)[-n0 // 2 :]
        bin_vec = np.zeros(n0)
        bin_vec[idx] = 1.0
        start_vecs.append(bin_vec)

    bin_rand = np.zeros(n0)
    bin_rand[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    start_vecs.append(bin_rand)

    rand_frac = rng.random(n0)
    rand_frac *= target0 / rand_frac.sum()
    rand_frac = np.clip(rand_frac, 0.0, 1.0)
    rand_frac = project_sum_exact(rand_frac, target0, rng)
    start_vecs.append(rand_frac)

    rand_frac2 = rng.random(n0) ** 2
    rand_frac2 *= target0 / rand_frac2.sum()
    rand_frac2 = np.clip(rand_frac2, 0.0, 1.0)
    rand_frac2 = project_sum_exact(rand_frac2, target0, rng)
    start_vecs.append(rand_frac2)

    for idx, h0 in enumerate(start_vecs):
        if time.time() >= deadline:
            break
        left = deadline - time.time()
        sub_dead = time.time() + max(5.0, left / (len(start_vecs) - idx))
        sub_dead = min(sub_dead, deadline)
        h0 = project_sum_exact(h0, target0, rng)
        h_best, c_best = local_search(h0, rng, sub_dead)
        if c_best < best_c:
            best_h, best_c, best_n = h_best.copy(), c_best, n0

    if best_h is not None and time.time() < deadline:
        n_up = best_n * 2
        if n_up <= 720:
            xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            target_up = n_up / 2.0
            h_up = project_sum_exact(h_up, target_up, rng)
            h_up, c_up = local_search(h_up, rng, deadline)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    if best_h is not None and time.time() < deadline:
        try:
            beta = 12000.0
            max_iter = 1800
            h_ref = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref
        except Exception:
            pass

    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_sum_exact(best_h, n / 2.0, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_exact(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
