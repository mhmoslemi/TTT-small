def simulated_annealing(h_init, time_limit):
    h = h_init.copy()
    n = len(h)
    n_ones = n//2
    corr = np.correlate(h, h, mode='full')
    min_corr = np.min(corr[np.arange(2*n-1) != n-1])
    max_overlap = n_ones - min_corr
    best_h = h.copy()
    best_overlap = max_overlap

    start = time.time()
    it = 0
    # schedule: T0 = 5.0, T_end = 1e-4, total_iters = approx 200000 (or time-based)
    while time.time() < time_limit:
        # choose random swap
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta_corr via local shift contributions
        # naive: compute new corr from scratch and get new max_overlap
        # easier: compute new_corr = np.correlate(updated_h, ...) each iteration
        # but that requires constructing updated_h (swap) each time, okay
        h[i], h[j] = 0, 1
        new_corr = np.correlate(h, h, mode='full')
        new_min_corr = np.min(new_corr[np.arange(2*n-1) != n-1])
        new_max_overlap = n_ones - new_min_corr

        delta_overlap = new_max_overlap - max_overlap
        # accept if new <= best, else accept with prob exp(-delta_overlap / T)
        if new_max_overlap <= max_overlap:
            accept = True
        else:
            # temperature schedule based on iteration count

        it += 1
        # optionally break if best_overlap <= target_alloted (target 68)

import time
import numpy as np

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """C5 for a (binary or real) step function h on [0,2] sampled uniformly."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def greedy_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Greedy construction of a binary vector with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # correlation of current vector with 1–h
    for _ in range(half):
        # O adjusted as if we inserted a 1 (adds 1 to all overlaps)
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            # window = O_adj shifted by p minus the current h contributions
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0  # zero shift must be 0 for binary vectors
    return h

def simulated_annealing(
    h_start: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    target_overlap: int,
) -> tuple[np.ndarray, int]:
    """Return (best_h, best_max_overlap) found before deadline."""
    n = h_start.size
    h = h_start.copy()
    O = np.correlate(h, 1 - h, mode="full")
    max_O = int(O.max())

    # initialise the "best so far"
    best_h = h.copy()
    best_max_O = max_O

    # maintain ones/zeros lists with O(1) updates
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    pos_in_ones = np.full(n, -1, dtype=int)
    pos_in_zeros = np.full(n, -1, dtype=int)
    for idx, i in enumerate(ones):
        pos_in_ones[i] = idx
    for idx, i in enumerate(zeros):
        pos_in_zeros[i] = idx

    it = 0
    start_temp = 5.0
    end_temp = 0.001
    max_iters = 2_000_000

    while time.time() < deadline:
        it += 1
        # pick a random 1↔0 swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        # apply swap
        h[i] = 0
        h[j] = 1
        O_new = np.correlate(h, 1 - h, mode="full")
        new_max_O = int(O_new.max())
        delta = new_max_O - max_O

        # temperature schedule
        progress = it / max_iters
        if progress > 1.0:
            progress = 1.0
        T = start_temp * (end_temp / start_temp) ** progress

        # acceptance test
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            # keep new state
            O = O_new
            max_O = new_max_O
            # update ones/zeros structures
            pos_i = pos_in_ones[i]
            pos_j = pos_in_zeros[j]
            ones[pos_i] = j
            zeros[pos_j] = i
            pos_in_ones[i] = -1
            pos_in_zeros[j] = -1
            pos_in_ones[j] = pos_i
            pos_in_zeros[i] = pos_j

            # record best
            if max_O < best_max_O:
                best_max_O = max_O
                best_h = h.copy()
                if best_max_O <= target_overlap:
                    break
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # early exit if time is about to run out
        if time.time() + 0.5 >= deadline:
            break

    return best_h, best_max_O

        best_c5 = best_max_O * (2.0 / n)
    else:
        # fallback: random binary
        best_h = np.zeros(n, dtype=np.int8)
        rand_ones = rng.choice(n, size=n_ones, replace=False)
        best_h[rand_ones] = 1
        best_max_O = int(np.max(np.correlate(best_h, 1 - best_h, mode="full")))
        best_c5 = best_max_O * (2.0 / n)

    # --------------------------------------------------------
    # Greedy binary construction (quick strong baseline)
    # --------------------------------------------------------
    if time.time() < deadline:
        h_greedy = greedy_binary(n, rng)
        c5_greedy = compute_c5(h_greedy)
        if c5_greedy < best_c5:
            best_h = h_greedy.copy()
            best_c5 = c5_greedy
            best_max_O = int(np.max(np.correlate(best_h, 1 - best_h, mode="full")))

    # --------------------------------------------------------
    # Simulated annealing starting from the greedy solution
    # --------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = deadline
        h_local, max_O_local = simulated_annealing(best_h, sa_deadline, rng, target_overlap)
        c5_local = max_O_local * (2.0 / n)
        if c5_local < best_c5:
            best_h = h_local.copy()
            best_c5 = c5_local
            best_max_O = max_O_local
            if best_max_O <= target_overlap:
                # reached desired bound
                return best_h.astype(float).tolist(), float(best_c5), int(n)

    # --------------------------------------------------------
    # Additional random restarts of SA until time runs out
    # --------------------------------------------------------
    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # allocate at most 30 s for the next SA run
        sa_time = min(30.0, remaining / 2.0)
        sa_dead = time.time() + sa_time

        # generate a random start (or a perturbed current best)
        if rng.random() < 0.4:
            # perturb current best slightly
            h_start = best_h.copy()
            perturb = rng.integers(5, 15)
            for _ in range(perturb):
                ones = np.flatnonzero(h_start)
                zeros = np.flatnonzero(1 - h_start)
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h_start[i], h_start[j] = 0, 1
        else:
            # fresh random binary vector
            h_start = np.zeros(n, dtype=np.int8)
            rand_ones = rng.choice(n, size=n_ones, replace=False)
            h_start[rand_ones] = 1

        h_loc, max_O_loc = simulated_annealing(h_start, sa_dead, rng, target_overlap)
        c5_loc = max_O_loc * (2.0 / n)
        if c5_loc < best_c5:
            best_h = h_loc.copy()
            best_c5 = c5_loc
            best_max_O = max_O_loc
            if best_max_O <= target_overlap:
                break

    # --------------------------------------------------------
    # Final safety adjustments and return
    # --------------------------------------------------------
    best_h = np.clip(best_h, 0, 1).astype(float)
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(n)
