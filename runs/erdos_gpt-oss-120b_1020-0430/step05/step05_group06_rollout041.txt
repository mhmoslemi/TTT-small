def local_improve_binary(h):
    n = h.size
    while True:
        # compute M vector
        corr = np.correlate(h, 1 - h, mode="full")
        M = corr[n-1:]  # length n
        max_M = M.max()
        # find all shifts achieving max
        ks = np.flatnonzero(M == max_M)
        improved = False
        for k in ks:
            # compute mismatches sets
            if k == 0:
                # For k=0, M_0 = sum_i h_i*(1-h_i) = 0 always because h_i*(1-h_i)=0. So max can't be at k=0 unless n=0.
                continue
            # mismatches where h[i]=1 and h[i+k]=0
            # consider slice up to n-k
            h_i = h[:n - k]
            h_i_k = h[k:]
            A = np.flatnonzero((h_i == 1) & (h_i_k == 0))
            B = np.flatnonzero((h_i == 0) & (h_i_k == 1))
            if A.size > 0 and B.size > 0:
                i = A[0]  # can random choose for variety
                j = B[0]
                # swap
                h[i] = 0
                h[j] = 1
                improved = True
                break
        if not improved:
            break
    return h, max_M
```

def greedy_binary(n, rng):

    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # correlation of all-zero vector

import time
import numpy as np

def compute_c5(h):
    """C5 for a step function sampled on n points."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000, tol=1e-12):
    """Exact correction so that Σh = n/2 (h may be float)."""
    n = h.size
    target = n / 2.0
    diff = target - np.sum(h)
    for _ in range(max_iter):
        if abs(diff) < tol:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def greedy_binary(n, rng):
    """Greedy construction of a binary vector with Σh = n/2."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # correlation of all‑zero vector
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0  # enforce O_0 = 0
    return h, O

def local_improve_binary(h, rng):
    """Iteratively apply beneficial swaps that reduce the current maximal overlap."""
    n = h.size
    while True:
        corr = np.correlate(h, 1 - h, mode="full")
        M = corr[n - 1:]               # shifts k >= 0
        max_M = int(M.max())
        if max_M == 0:
            break
        ks = np.flatnonzero(M == max_M)
        improved = False
        for k in ks:
            if k == 0:
                continue
            nk = n - k
            hi = h[:nk]
            hik = h[k:]
            A = np.flatnonzero((hi == 1) & (hik == 0))
            B = np.flatnonzero((hi == 0) & (hik == 1))
            if A.size and B.size:
                i = A[0]
                j = B[0]
                h[i] = 0
                h[j] = 1
                improved = True
                break
        if not improved:
            break
    # final max
    corr = np.correlate(h, 1 - h, mode="full")
    max_M = int(corr[n - 1:].max())
    return h, max_M

            h_imp2, max_M2 = local_improve_binary(h_rand, rng)
            c2 = max_M2 * 2.0 / n
            if c2 < best_c:
                best_c = c2
                best_h = h_imp2.astype(float).copy()
                best_n = n

    # ---- Parent solution fallback ----
    if best_h is None and "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        best_n = parent.size
        best_h = parent.copy()
        best_c = compute_c5(best_h)

    # ---- Final adjustments ----
    if best_h is None:
        # very last resort: simple half‑ones
        best_n = 360
        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    # binary solutions already have exact sum; adjust only if needed
    if abs(best_h.sum() - best_n / 2.0) > 1e-9:
        best_h = adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
