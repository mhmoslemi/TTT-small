    diff = target - y.sum()

        # distribute diff proportionally to slack
        slack = (y < upper) & (y > lower)
        if np.any(slack):
            y[slack] += diff / np.count_nonzero(slack)
    return y

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_onto_capped_simplex(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto {y: lower ≤ y ≤ upper, sum(y)=target}."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(v + mid, lower, upper)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v + (lo + hi) / 2.0, lower, upper)
    diff = target - y.sum()
    if abs(diff) > tol:
        slack = (y > lower) & (y < upper)
        if np.any(slack):
            y[slack] += diff / np.count_nonzero(slack)
    return y

                improved = True
                break
            alpha *= 0.5
        if not improved:
            # small perturbation
            i, j = rng.choice(n, size=2, replace=False)
            max_tr = min(h[i], 1.0 - h[j])
            if max_tr > 0:
                delta = max_tr * rng.random() * 0.2
                h[i] -= delta
                h[j] += delta
                h = project_onto_capped_simplex(h, target)
        beta = min(beta * 2.0, end_beta)
        steps -= 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    # lengths to try
    n_candidates = []
    if parent is not None:
        n_candidates.append(parent.size)
        if parent.size * 2 <= 720:
            n_candidates.append(parent.size * 2)
    for n in (240, 360, 480, 540, 600):
        if n not in n_candidates:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time.time() > deadline:
            break
        target = n / 2.0
        starts = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            starts.append(interp)

            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            starts.append(bin_vec)

            smoothed = np.convolve(parent, np.ones(5) / 5.0, mode="same")
            sm_interp = np.interp(xt, xs, smoothed)
            starts.append(sm_interp)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        # random continuous (uniform)
        starts.append(rng.random(n))

        # random continuous (biased)
        starts.append(rng.random(n) ** 2)

        for h0 in starts:
            if time.time() > deadline:
                break
            h0 = project_onto_capped_simplex(h0, target)

            # subgradient phase
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.5)
            h_sg, c_sg = subgradient_descent(h0, n, rng, sub_dead)
            if c_sg < best_c:
                best_h, best_c, best_n = h_sg.copy(), c_sg, n

            # smooth gradient refinement
            smooth_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
            h_ref = smooth_gradient_descent(h_sg, n, rng, smooth_dead)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c, best_n = h_ref.copy(), c_ref, n

    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_onto_capped_simplex(best_h, n / 2.0)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
