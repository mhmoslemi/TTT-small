function continuous_sa(n, h0, rng, time_budget):

- start = time.time()
- h = h0.copy()
- best_h = h.copy()
- best_c = compute_c5(h)
- While time remaining > small fraction (maybe >0.1 sec):
   - sample i, j distinct
   - max_delta = min(h[i], 1.0 - h[j]) if we want to move mass from i to j (i loses, j gains). Also can move opposite direction.
   - Decide direction: If random <0.5, move from i to j; else from j to i.
   - Choose eps = rng.random() * max_delta * step_scale, where step_scale maybe dynamic (temperature dependent).
   - h_new = h.copy(); h_new[i] -= eps; h_new[j] += eps
   - compute c_new = compute_c5(h_new)
   - delta = c_new - cur_c. If delta <=0 accept. Else accept with prob exp(-delta / T). Temperature schedule T decays over time. Choose T0 maybe 0.01, T_end maybe 1e-6. linear or exponential.
   - If accept, update h, cur_c.

   - If c_new < best_c: update best.

- Return best_h, best_c.

def continuous_sa(n, h0, rng, time_budget, current_best_c):

        fraction = elapsed / time_budget

        # choose two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:

        if rng.random() < 0.5:
            src, dst = i, j

    max_iter = 0

CURRENT_TARGET = None; CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp in stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly by redistribution
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # adjust randomly
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic helpers ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 500_000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (largest) overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(cur_h)
        nh = guided_perturb(cur_h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, cur_h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- simulated annealing for continuous vectors ----------
def continuous_sa(n: int, h0: np.ndarray, rng: np.random.Generator,
                 time_budget: float, current_best: float):
    """Temperature‑controlled SA that preserves the sum exactly."""
    start = time.time()
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    T0, T_end = 0.02, 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # decide direction of transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 0:
            continue
        scale = T / T0
        if scale > 1.0:
            scale = 1.0
        eps = max_eps * rng.random() * scale
        old_src, old_dst = h[src], h[dst]
        h[src] = old_src - eps
        h[dst] = old_dst + eps
        c_new = compute_c5(h)
        delta = c_new - cur_c
        accept = delta <= 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur_c = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
        else:
            h[src], h[dst] = old_src, old_dst
    return best_h, best_c


# ---------- soft‑max SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 200) -> np.ndarray:
    """Few‑iteration SLSQP on the smooth max‑approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 2000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
        it += 1
    return x
