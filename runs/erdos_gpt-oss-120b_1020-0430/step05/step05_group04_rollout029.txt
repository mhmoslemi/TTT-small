def smooth_max_and_grad(h, beta, lam):
    n = h.size
    w_target = n / 2.0
    f = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights
    beta_f = beta * f
    M = np.max(beta_f)
    exp_beta_f = np.exp(beta_f - M)
    sum_exp = np.sum(exp_beta_f)
    w = exp_beta_f / sum_exp  # shape L
    # compute convolution for gradient

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Basic utilities
# -------------------------------------------------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Shift mass between coordinates until Σh = n/2 exactly."""
    n = h.size
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
    return h


# -------------------------------------------------
# Convolution (FFT based, full mode)
# -------------------------------------------------
def conv_full(a, b):
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    conv = np.fft.irfft(fa * fb, N)[:n]
    return conv


# -------------------------------------------------
# Smooth‑max objective and its gradient
# -------------------------------------------------
def smooth_max_and_grad(h, beta, lam):
    """Return (obj, grad) for the log‑sum‑exp surrogate + sum‑penalty."""
    n = h.size
    w_target = n / 2.0

    # f_k = Σ h_i (1 - h_{i+k})
    f = np.correlate(h, 1.0 - h, mode="full")

    beta_f = beta * f
    M = np.max(beta_f)
    exp_beta_f = np.exp(beta_f - M)
    sum_exp = np.sum(exp_beta_f)
    w = exp_beta_f / sum_exp                     # softmax weights, sum to 1

    # needed convolutions: S1_i = Σ_k w_k h_{i+k}, S2_i = Σ_k w_k h_{i-k}
    offset = len(w) - 1           # = 2n-2
    conv_rev = conv_full(w[::-1], h)   # uses w reversed -> gives S1
    conv_fwd = conv_full(w, h)         # gives S2
    S1 = conv_rev[offset:offset + n]
    S2 = conv_fwd[offset:offset + n]

    g_f = 1.0 - S1 - S2                # gradient of smooth‑max part

    sum_diff = h.sum() - w_target
    grad = g_f + 2.0 * lam * sum_diff   # add penalty gradient

    obj = (M + np.log(sum_exp)) / beta + lam * sum_diff * sum_diff
    return obj, grad


def lbfgs_optimize(h0, beta, lam, rng, max_iter=200):
    """L‑BFGS‑B on the smooth‑max surrogate (box bounds only)."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return smooth_max_and_grad(x, beta, lam)

    res = minimize(fun, h0, method="L-BFGS-B", jac=True,
                   bounds=bounds,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


# -------------------------------------------------
# Simple projected gradient refinement on the worst shift
# -------------------------------------------------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)          # shift with max overlap
        k = abs(idx)

        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]           # (1 - h_{i+k})
            g[k:] -= h[:upper]                 # -h_{i-k}

        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
