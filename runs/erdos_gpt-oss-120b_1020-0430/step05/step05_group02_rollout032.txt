def gradient_descent_subspace(h, rng, target, max_iter, deadline):
    best_c = compute_c5(h)
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # compute worst shift correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (len(h)-1)
        k = abs(idx)
        n = len(h)
        # gradient g
        if k == 0:
            g = 1 - 2 * h
        else:
            upper = n - k
            g = np.zeros(n)
            g[:upper] = 1 - h[k:]
            g[k:] -= h[:upper]
        d = -g
        # compute max step
        pos = d > 1e-15
        neg = d < -1e-15
        max_step = np.inf
        if np.any(pos):
            step_pos = (1 - h[pos]) / d[pos]
            max_step = min(max_step, step_pos.min())
        if np.any(neg):
            step_neg = h[neg] / (-d[neg])
            max_step = min(max_step, step_neg.min())
        if max_step <= 0:
            # cannot move
            # random perturb and continue
            i = rng.integers(n)
            delta = (rng.random() - 0.5) * 0.001
            h[i] = np.clip(h[i] + delta, 0, 1)
            # enforce sum
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                j = rng.integers(n)
                add = min(diff, 1 - h[j]) if diff > 0 else max(diff, -h[j])
                h[j] += add
            continue
        alpha = 0.5 * max_step
        h = h + alpha * d
        # Clip due to numerical errors
        h = np.clip(h, 0.0, 1.0)
        # enforce sum drift
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            j = rng.integers(n)
            add = min(diff, 1 - h[j]) if diff > 0 else max(diff, -h[j])
            h[j] += add
        # evaluate
        c_val = np.max(corr) * (2.0 / n)  # we have corr already.
        if c_val + 1e-12 < best_c:
            best_c = c_val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:  # stagnation
                break
    return h, best_c

def binary_sa_with_time(h_bin, rng, deadline, max_iter=200000):
    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode='full')
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    T0 = cur_worst * 0.05 + 1e-4
    decay = 0.99995
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # remove i (1->0)
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i+1:]))
        c[start_i:start_i+n] -= hi_vec
        h_bin[i] = 0.0

        # add j (0->1)
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j+1:]))
        c[start_j:start_j+n] += hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_worst - cur_worst) / T):
                accept = True

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h_bin[j] = 0.0
            c[start_j:start_j+n] -= hj_vec
            h_bin[i] = 1.0
            c[start_i:start_i+n] += hi_vec
            no_improve += 1

        if no_improve > 50000:
            break
        it += 1

    return best_h, best_worst

        else:
            if T > 0 and rng.random() < np.exp(-(new_worst - cur_worst) / T):
                accept = True

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h_bin[j] = 0.0
            c[start_j : start_j + n] -= hj_vec
            h_bin[i] = 1.0
            c[start_i : start_i + n] += hi_vec
            no_improve += 1

        if no_improve > 50000:
            break
        it += 1
    return best_h, best_worst


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.97 * budget_s

    # base dimension
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        adjust_sum_exact(h0, rng)
    else:
        n = 360
        target = n / 2.0
        h0 = rng.random(n)
        h0 *= target / h0.sum()
        adjust_sum_exact(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)
    best_n = n

    # build diverse start vectors
    starts = []

    # 1) parent (fractional)
    starts.append(best_h.copy())

    # 2) binary top‑half of parent
    idx = np.argpartition(best_h, -n // 2)[-n // 2 :]
    bin_top = np.zeros(n)
    bin_top[idx] = 1.0
    starts.append(bin_top.copy())

    # 3) random binary
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append(bin_rand.copy())

    # 4) random fractional
    rand_frac = rng.random(n)
    rand_frac *= (n / 2.0) / rand_frac.sum()
    adjust_sum_exact(rand_frac, rng)
    starts.append(rand_frac.copy())

    # 5) two‑block binary pattern
    block = np.zeros(n)
    q = n // 4
    block[:q] = 1.0
    block[n // 2 : n // 2 + q] = 1.0
    starts.append(block.copy())

    # 6) smoothed random binary
    tmp = np.zeros(n)
    tmp[rng.choice(n, n // 2, replace=False)] = 1.0
    kernel = np.array([0.2, 0.6, 0.2])
    sm = np.convolve(np.r_[tmp[-1], tmp, tmp[0]], kernel, mode="valid")
    sm = np.clip(sm, 0.0, 1.0)
    adjust_sum_exact(sm, rng)
    starts.append(sm.copy())

    # ----- exploration loop -----
    for vec in starts:
        if time.time() > deadline:
            break
        h = vec.copy()
        adjust_sum_exact(h, rng)

        # aggressive gradient refinement
        h = gradient_refine(h, rng, step=0.07, max_iter=400)

        # modest hill climbing
        h, c_val = hill_climb(h, rng, max_iter=3000, deadline=deadline)

        # occasional SLSQP cascade
        for beta in (20, 200, 2000, 20000):
            if time.time() > deadline:
                break
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=200)
            except Exception:
                continue
            c_val = compute_c5(h)

        # final fine‑grained gradient pass
        h = gradient_refine(h, rng, step=0.03, max_iter=250)
        c_val = compute_c5(h)

        if c_val < best_c:
            best_h, best_c, best_n = h.copy(), c_val, n

    # ----- binary post‑processing if time permits -----
    if time.time() < deadline:
        # consider binary rounding of best solution
        bin_candidate = (best_h >= 0.5).astype(float)
        if np.max(np.abs(best_h - bin_candidate)) < 0.02:
            # run longer SA on the binary version
            bin_res, worst = binary_sa_with_time(
                bin_candidate.copy(), rng, deadline, max_iter=400000
            )
            c_bin = worst * (2.0 / n)
            if c_bin < best_c:
                best_h, best_c = bin_res.astype(float), c_bin

    # ----- optional up‑sampling refinement -----
    if time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        h_up = gradient_refine(h_up, rng, step=0.07, max_iter=300)
        h_up, c_up = hill_climb(h_up, rng, max_iter=2000, deadline=deadline)
        for beta in (20, 200, 2000):
            if time.time() > deadline:
                break
            try:
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=150)
            except Exception:
                continue
        h_up = gradient_refine(h_up, rng, step=0.04, max_iter=200)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
