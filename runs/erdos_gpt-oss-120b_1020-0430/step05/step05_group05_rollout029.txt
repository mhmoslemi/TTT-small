import time
import numpy as np
from scipy.optimize import minimize

CURRENT_BETA = None


def compute_c5_full(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k))dx, discretized."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def full_from_half(a: np.ndarray) -> np.ndarray:
    """Construct full period‑2 vector from half‑period a."""
    return np.concatenate([a, 1.0 - a])


def softmax_obj_half(a: np.ndarray) -> float:
    """Smooth‑max of overlaps for SLSQP (log‑sum‑exp)."""
    global CURRENT_BETA
    full = full_from_half(a)
    corr = np.correlate(full, 1.0 - full, mode="full")
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_half(a0: np.ndarray, beta: float, max_iter: int, deadline: float) -> np.ndarray:
    """SLSQP on smooth‑max surrogate, respecting bounds."""
    global CURRENT_BETA
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * a0.size
    # early exit if time is already exhausted
    if time.time() > deadline:
        return a0
    res = minimize(
        softmax_obj_half,
        a0,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    a_opt = np.clip(res.x, 0.0, 1.0)
    return a_opt


def gradient_descent_half(a0: np.ndarray, max_iter: int, step0: float, deadline: float):
    """Projected gradient descent on the worst shift."""
    M = a0.size
    n = 2 * M
    a = a0.copy()
    best_a = a.copy()
    best_c = compute_c5_full(full_from_half(a))
    step = step0
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        full_h = full_from_half(a)
        corr = np.correlate(full_h, 1.0 - full_h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)
        k = abs(shift)
        if k == 0:
            a += step * (np.random.rand(M) - 0.5)
            a = np.clip(a, 0.0, 1.0)
            step *= 0.99
            continue
        up = n - k
        g_full = np.zeros_like(full_h)
        g_full[:up] += 1.0 - full_h[k:]
        g_full[k:] -= full_h[:up]
        g_half = g_full[:M] - g_full[M:]
        a -= step * g_half
        a = np.clip(a, 0.0, 1.0)

        c = compute_c5_full(full_from_half(a))
        if c < best_c - 1e-12:
            best_c, best_a = c, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        step *= 0.995
        if no_improve > 3000:
            break
    return best_a, best_c


def binary_targeted_improve_full(h: np.ndarray, rng: np.random.Generator, deadline: float) -> np.ndarray:
    """Greedy swaps on a binary vector (size n) to lower the max overlap."""
    n = h.size
    # make a symmetric binary starter from the first half
    M = n // 2
    a = h[:M]
    idx = np.argpartition(a, -M)[-M:]
    a_bin = np.zeros_like(a)
    a_bin[idx] = 1.0
    h_bin = np.concatenate([a_bin, 1.0 - a_bin])

    while time.time() < deadline:
        corr = np.correlate(h_bin, 1.0 - h_bin, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)
        k = abs(shift)
        if k == 0:
            break

        ar = np.arange(n)
        neigh = h_bin[(ar + k) % n] + h_bin[(ar - k) % n]

        ones = np.flatnonzero(h_bin == 1)

        if ones.size == 0 or zeros.size == 0:

                h_bin[i2] = 0.0
                h_bin[j2] = 1.0
                improved = True
                break
        if not improved:
