def project_to_sum_and_bounds(x, target):
    # Find lambda s.t. sum(cap(x_i + λ,0,1)) = target
    # Use bisection with +/- large bounds.
    lo, hi = -1.0, 1.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return y

def optimize_h(initial_h, N, time_budget):
    start = time.time()
    best_h = initial_h.copy()
    best_c = compute_c5_fast(best_h)
    h = best_h.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0.05
    iteration = 0
    while time.time() - start < time_budget:
        iteration += 1
        # compute gradient of worst shift
        grad, k, idx = worst_shift_gradient(h)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
        # project to [0,1] and sum
        target = N / 2.0
        h_new = project_to_sum_and_bounds(h_new, target)
        # compute C5
        c = compute_c5_fast(h_new)
        # accept if better, otherwise maybe accept with some probability (simulated annealing)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        # always move? Might be beneficial to keep new as current even if worse for exploration? Use acceptance based on small probability.
        # Let's adopt simple rule: if c <= compute_c5_fast(h) + 1e-10 (i.e., non-worse), accept; else accept with small prob decaying.
        c_curr = compute_c5_fast(h)
        if c <= c_curr + 1e-12:
            h = h_new
        else:
            # probabilistic acceptance
            # Temperature schedule linear to residual time
            elapsed = time.time() - start
            remaining = time_budget - elapsed
            temp = max(1e-4, remaining / time_budget)  # decays to near 0
            if np.random.rand() < np.exp(-(c - c_curr) / (temp + 1e-12)):
                h = h_new
            else:
                # keep h unchanged
                pass
        # occasionally random perturb for diversification
        if iteration % 200 == 0:
            # random shuffle small mass
            diff = np.random.randn(N) * 0.01  # small noise
            h = project_to_sum_and_bounds(h + diff, target)
    return best_h, best_c

def simple_project(x, target):
    # clip to [0,1]
    y = np.clip(x, 0.0, 1.0)
    # adjust sum by adding constant and re-clipping iteratively
    for _ in range(5):
        diff = target - y.sum()
        if abs(diff) < 1e-12:
            break
        # distribute diff evenly among non-saturated entries
        mask = (y > 0.0) & (y < 1.0)
        n_free = np.count_nonzero(mask)
        if n_free == 0:
            # distribute to all entries
            y += diff / len(y)
            y = np.clip(y, 0.0, 1.0)
        else:
            delta = diff / n_free
            y[mask] += delta
            y = np.clip(y, 0.0, 1.0)
    # final correction: if still off by small amount, adjust randomly
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        # distribute to random indices that are not saturated in direction
        indices = np.arange(len(y))
        rng = np.random.default_rng()
        for _ in range(100):
            if abs(diff) < 1e-10:
                break
            i = rng.integers(len(y))
            if diff > 0 and y[i] < 1.0:
                add = min(diff, 1.0 - y[i])
                y[i] += add
                diff -= add
            elif diff < 0 and y[i] > 0.0:
                sub = min(-diff, y[i])
                y[i] -= sub
                diff += sub
    return y

def full_corr_fft(h):
    """Full linear correlation of h with 1-h using FFT (length 2*len-1)."""
    n = h.shape[0]
    a = h
    b_rev = (1.0 - h)[::-1]
    fa = np.fft.rfft(a, n * 2)
    fb = np.fft.rfft(b_rev, n * 2)
    conv = np.fft.irfft(fa * fb, n * 2)
    return conv[: 2 * n - 1]

def compute_c5_and_corr(h):
    """Return C5 value and full correlation array."""
    n = h.shape[0]
    corr = full_corr_fft(h)
    return np.max(corr) * (2.0 / n), corr

def project_to_sum_and_bounds(x, target):

        if s_lo > target:
            lo *= 2
        else:
            break

            lo = lam
        else:

    best_h, best_c = h.copy(), c_curr

    while time.time() - start < time_budget:
        it += 1
        grad, k = worst_shift_gradient(h, corr, idx)
        elapsed = time.time() - start
        frac = elapsed / time_budget
        step = 0.03 * (1 - frac) + 0.001 * frac
        h_new = h - step * grad
        h_new = project_to_sum_and_bounds(np.clip(h_new, 0.0, 1.0), N / 2.0)
        c_new, corr_new = compute_c5_and_corr(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if c_new <= c_curr:
            h, c_curr, corr, idx = h_new, c_new, corr_new, int(np.argmax(corr_new))
        else:
            temp = max(0.01, 1 - frac)
            if rng.random() < np.exp(-(c_new - c_curr) / (temp + 1e-12)):
                h, c_curr, corr, idx = h_new, c_new, corr_new, int(np.argmax(corr_new))
        if it % 300 == 0:
            noise = rng.normal(scale=0.01, size=N)
            h = project_to_sum_and_bounds(np.clip(h + noise, 0.0, 1.0), N / 2.0)
            c_curr, corr = compute_c5_and_corr(h)
            idx = int(np.argmax(corr))
    return best_h, best_c

# ---------- Main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # target resolution
    N_target = 1024
    # build initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        if n0 != N_target:
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N_target, endpoint=False)
            h = np.interp(x_new, x_old, h0)
        else:
            h = h0.copy()
    else:
        h = rng.random(N_target)
    h = project_to_sum_and_bounds(np.clip(h, 0.0, 1.0), N_target / 2.0)

    best_h, best_c = h.copy(), None
    c0, _ = compute_c5_and_corr(h)
    best_c = c0

    # Phase 1: several Adam restarts (≈70% time)
    phase1_end = start + total_budget * 0.70
    restart = 0
    while time.time() < phase1_end:
        remaining = phase1_end - time.time()
        max_iters = max(2000, int(min(15000, remaining * 1500)))
        # start point: best so far with slight jitter
        h_start = best_h + rng.normal(scale=0.03, size=N_target)
        h_start = project_to_sum_and_bounds(np.clip(h_start, 0.0, 1.0), N_target / 2.0)
        deadline = phase1_end
        h_opt, c_opt = adam_optimize(h_start, N_target, max_iters, deadline, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        restart += 1

    # Phase 2: subgradient descent on current best (≈20% time)
    phase2_end = start + total_budget * 0.90
    time_left = max(0.0, phase2_end - time.time())
    if time_left > 2.0:
        h_opt, c_opt = subgradient_descent(best_h, N_target, time_left, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # Phase 3: optional up‑sample to double resolution if time permits
    time_remaining = total_budget - (time.time() - start)
    if time_remaining > 6.0 and N_target * 2 <= 2048:
        N2 = N_target * 2
        x_old = np.linspace(0.0, 2.0, N_target, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_to_sum_and_bounds(np.clip(h2, 0.0, 1.0), N2 / 2.0)
        # quick subgradient on the higher resolution
        t_end = time.time() + time_remaining * 0.8
        iters = 0
        while time.time() < t_end:
            iters += 1
            grad, k = worst_shift_gradient(h2, full_corr_fft(h2), int(np.argmax(full_corr_fft(h2))))
            step = 0.02 / (1 + iters * 0.001)
            h2 = project_to_sum_and_bounds(np.clip(h2 - step * grad, 0.0, 1.0), N2 / 2.0)
        c2, _ = compute_c5_and_corr(h2)
        if c2 < best_c:
            best_c, best_h, N_target = c2, h2.copy(), N2

    # final sanitisation
    best_h = project_to_sum_and_bounds(np.clip(best_h, 0.0, 1.0), N_target / 2.0)
    final_c, _ = compute_c5_and_corr(best_h)
    return best_h.tolist(), float(final_c), int(N_target)
