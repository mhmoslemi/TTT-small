```
def gradient_refine(h, rng, step=0.02, iters=100):
    target = h.size / 2.0
    for _ in range(iters):
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr) - (h.size - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1 - 2*h
        else:
            upper = h.size - k
            g[:upper] = 1 - h[k:]
            g[k:] -= h[:upper]
        # step size schedule decreasing
        h_new = h - step * g
        # clip and adjust sum
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # adjust sum:
        diff = target - h_new.sum()
        # simple redistribution:
        for _ in range(200):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, h.size)
            add = min(diff, 1 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
            h_new[i] += add
            diff = target - h_new.sum()
        h = h_new
    return h
```

def sa_optimize(h0, rng, deadline):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    n = h.size
    target = n/2.0
    init_T = max(0.01 * cur_c, 1e-4)
    decay = 0.99995
    it = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # direction randomly i->j or j->i
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[src] -= delta
        h[dst] += delta
        c_new = compute_c5(h)
        T = init_T * (decay ** it) + 1e-12
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / T):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # revert
            h[src] += delta
            h[dst] -= delta
        it += 1
        # optional early exit if diff < tolerance? not needed.
    return best_h, best_c
```

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=5000):
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_frac(n, rng, power=1.0):
    x = rng.random(n) ** power
    target = n / 2.0
    x *= target / x.sum()
    x = np.clip(x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def gradient_refine(h, rng, step=0.02, iters=150):
    target = h.size / 2.0
    n = h.size
    for it in range(iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] = 1.0 - h[k:]
            g[k:] -= h[:upper]
        h_new = h - step * g
        np.clip(h_new, 0.0, 1.0, out=h_new)
        adjust_sum_exact(h_new, rng)
        h = h_new
    return h

def sa_optimize(h0, rng, deadline):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    init_T = max(0.02 * cur_c, 1e-4)
    T = init_T
    decay = 0.99995
    it = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        h[src] -= delta
        h[dst] += delta
        c_new = compute_c5(h)
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / T):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[src] += delta
            h[dst] -= delta
        T = max(T * decay, 1e-12)
        it += 1
    return best_h, best_c
