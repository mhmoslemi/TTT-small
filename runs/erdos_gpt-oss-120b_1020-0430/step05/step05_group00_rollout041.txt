import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# projection that respects the bounds and the integral constraint
def project_h(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    if "enforce_sum" in globals():
        return enforce_sum(h, rng)          # use the pre‑imported helper if present
    # fall‑back deterministic version
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def binary_sa(n: int, rng: np.random.Generator, time_budget: float):
    """Simulated‑annealing on binary vectors with exactly n/2 ones."""
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    all_idx = np.arange(n)

    while time.time() - start < time_budget:
        # ----- random binary start -----
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5(h)

        ones_arr = np.array(ones, copy=True)
        zeros_arr = np.setdiff1d(all_idx, ones_arr)

        swaps = 0
        max_swaps = 5000                     # limit per start to allow many restarts
        while time.time() - start < time_budget and swaps < max_swaps:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)

            i_idx = rng.integers(len(ones_arr))
            j_idx = rng.integers(len(zeros_arr))
            i, j = ones_arr[i_idx], zeros_arr[j_idx]

            # perform swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i_idx] = j
                zeros_arr[j_idx] = i
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                # revert swap
                h[i] = 1.0
                h[j] = 0.0
            swaps += 1

    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[n // 2 :] = 1.0
        best_c = compute_c5(best_h)
    return best_h, best_c


def projected_descent(h: np.ndarray, rng: np.random.Generator, time_limit: float):
    """Gradient‑descent with projection onto the feasible polytope."""
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()

    while time.time() - start < time_limit:
        # worst‑shift correlation and its gradient
        corr_full = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr_full) - (n - 1)
        k = abs(idx)

        grad = np.ones_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        # descent direction with zero sum
        d = -grad + np.mean(grad)

        # maximal feasible step size
        pos = d > 0
        neg = d < 0
        step_pos = np.min((1.0 - h[pos]) / d[pos]) if np.any(pos) else np.inf
        step_neg = np.min((-h[neg]) / d[neg]) if np.any(neg) else np.inf
        step_max = min(step_pos, step_neg)
        if step_max <= 0 or np.isinf(step_max):
            break

        step = step_max * 0.5                     # initial step

        for _ in range(8):                        # backtrack line search
            nh = project_h(h + step * d, rng)
            c = compute_c5(nh)
            if c < best_c:
                best_c, best_h, h = c, nh.copy(), nh
                improved = True

            if i != j:

                    nh[j] += eps
                    nh = project_h(nh, rng)
                    c = compute_c5(nh)
                    if c < best_c:
                        best_c, best_h, h = c, nh.copy(), nh
    return best_h, best_c
