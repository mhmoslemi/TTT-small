import time
import numpy as np
from scipy.optimize import minimize

# ----- global constants for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """C5 for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # smooth max = (log sum exp)/beta
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute to satisfy Σh = target exactly (target = n/2)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    max_iter: int = 12, init_step: float = 0.05) -> np.ndarray:
    """Projected gradient descent on the worst shift, with simple line search."""
    n = h.size
    target = n / 2.0
    step = init_step
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # gradient of the worst shift
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        cur_c5 = compute_c5(h)
        improved = False
        for _ in range(5):
            h_new = h - step * g
            h_new = np.clip(h_new, 0.0, 1.0)
            adjust_sum_exact(h_new, rng)
            new_c5 = compute_c5(h_new)
            if new_c5 < cur_c5 - 1e-12:
                h = h_new
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-6:
                break
    return h

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    """SLSQP on the smooth max (log‑sum‑exp) surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

    zeros = list(np.flatnonzero(h_bin == 0))

        j = rng.choice(zeros)

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.98 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None
    candidates = []
    max_top = 8

    # use parent if available
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        parent = adjust_sum_exact(parent, rng)
        c0 = compute_c5(parent)
        best_h, best_c, best_n = parent.copy(), c0, parent.size
        top_insert(candidates, (c0, parent.copy(), parent.size), max_top)

    # a set of admissible sizes (keep n moderate for speed)
    ns = [180, 200, 220, 240, 260, 300, 360, 420, 480, 560, 720]

    # main stochastic search
    while time.time() < deadline - 0.5:
        n = int(rng.choice(ns))
        target = n / 2.0

        # decide start vector
        if best_h is not None and rng.random() < 0.15 and best_n == n:
            h_start = best_h.copy()
        else:
            # random binary start
            h_start = np.zeros(n)
            h_start[rng.choice(n, n // 2, replace=False)] = 1.0

        # binary SA (time‑boxed)
        sub_dead = min(deadline, time.time() + 0.12)
        h_bin, worst = binary_sa(h_start.copy(), rng, sub_dead, max_iter=200000)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c - 1e-12:
            best_h, best_c, best_n = h_bin.copy(), c5_bin, n
        top_insert(candidates, (c5_bin, h_bin.copy(), n), max_top)

        # gradient refinement
        h_grad = gradient_refine(h_bin.copy(), rng, max_iter=12, init_step=0.05)
        c5_grad = compute_c5(h_grad)
        if c5_grad < best_c - 1e-12:
            best_h, best_c, best_n = h_grad.copy(), c5_grad, n
        top_insert(candidates, (c5_grad, h_grad.copy(), n), max_top)

        # light SLSQP cascade (few betas)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
            if time.time() > deadline:
                break
            try:
                h_opt = slsqp_optimize(h_grad.copy(), beta, rng, max_iter=180)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c - 1e-12:
                    best_h, best_c, best_n = h_opt.copy(), c5_opt, n
                top_insert(candidates, (c5_opt, h_opt.copy(), n), max_top)
                h_grad = h_opt  # continue from latest
            except Exception:
                continue

        # final gradient sweep
        h_grad = gradient_refine(h_grad, rng, max_iter=8, init_step=0.03)
        c5_final = compute_c5(h_grad)
        if c5_final < best_c - 1e-12:
            best_h, best_c, best_n = h_grad.copy(), c5_final, n
        top_insert(candidates, (c5_final, h_grad.copy(), n), max_top)

    # post‑processing of the best few candidates
    for c_val, h_val, n_val in candidates:
        if time.time() > deadline:
            break
        # optional up‑sampling if room remains
        if n_val * 2 <= 720 and time.time() < deadline - 1.0:
            xs = np.linspace(0.0, 2.0, n_val, endpoint=False)
            xt = np.linspace(0.0, 2.0, 2 * n_val, endpoint=False)
            h_up = np.interp(xt, xs, h_val)
            adjust_sum_exact(h_up, rng)
            h_up = gradient_refine(h_up, rng, max_iter=12, init_step=0.05)
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
                if time.time() > deadline:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=180)
            h_up = gradient_refine(h_up, rng, max_iter=8, init_step=0.03)
            c_up = compute_c5(h_up)
            if c_up < best_c - 1e-12:
                best_h, best_c, best_n = h_up.copy(), c_up, 2 * n_val
        else:
            # extra gradient passes on current resolution
            h_ref = gradient_refine(h_val.copy(), rng, max_iter=20, init_step=0.04)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c - 1e-12:
                best_h, best_c, best_n = h_ref.copy(), c_ref, n_val

    # fallback safety
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
