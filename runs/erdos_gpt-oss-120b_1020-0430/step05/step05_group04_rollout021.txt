def pgd_smooth_max(h0, n, target, beta_schedule, step0=0.05, max_iter_per_beta=200, rng=None):

Initialize h = h0.copy()

for beta in beta_schedule:

    for it in range(max_iter_per_beta):
        # compute correlation corr_k = sum_i h_i*(1 - h_{i+k}) for each shift k in -n+1..n-1
        # we can compute via np.correlate as before: corr = np.correlate(h, 1 - h, mode='full')
        corr = np.correlate(h, 1 - h, mode='full')
        # compute softmax weights p_k = exp(beta * corr) / sum exp
        # Use log-sum-exp trick for stability.
        max_corr = np.max(beta * corr)
        exp_vals = np.exp(beta * corr - max_corr)
        Z = exp_vals.sum()
        p = exp_vals / Z

        # gradient g = sum_k p_k * ((1 - h_{i+k}) - h_{i-k})
        g = np.zeros_like(h)
        # Use loops as earlier.
        # k index range -n+1...n-1; index in p is k+(n-1)
        for k in range(-n+1, n):
            pk = p[k + n - 1]
            if pk == 0.0:
                continue
            if k >= 0:
                if n-k > 0:
                    g[:n-k] += pk * (1 - h[k:])
                    g[k:] -= pk * h[:n-k]
            else:
                kneg = -k
                if n-kneg > 0:
                    g[kneg:] += pk * (1 - h[:n - kneg])
                    g[:n - kneg] -= pk * h[kneg:]
        # step size
        step = step0 * (0.9 ** it)  # decaying within beta stage
        h = h - step * g
        # project onto box and sum
        h = proj_box_sum(h, target)
    # optionally at end of beta stage, reduce step0 (but we already decayed per iter)

return h

def binary_sa_until(h, rng, deadline):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    ones = np.flatnonzero(cur_h >= 0.5)  # treat as ones
    zeros = np.flatnonzero(cur_h < 0.5)

    it = 0
    # compute initial temperature based on cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    while time.time() < deadline:
        # pick random 1 and 0 indices
        i = rng.choice(ones)
        j = rng.choice(zeros)

        cur_h[i] = 0.0
        cur_h[j] = 1.0

        c = compute_c5(cur_h)

        # temperature schedule
        T = T0 * (0.9999 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):

            # best

            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
        # optional early break if no improvement for many steps
        # but we'll rely on time.

# ---------- basic utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def proj_box_sum(x, target, tol=1e-12, max_iter=100):
    """Projection onto [0,1]^n ∩ {sum = target} via bisection on a shift λ."""
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

# ---------- PGD on smooth‑max surrogate ----------
def pgd_smooth_max(h0, n, target, beta_schedule, step0=0.05,
                   max_iter_per_beta=80):
    """Projected gradient descent on f = (1/β) log Σ exp(β·corr)."""
    h = h0.copy()
    for beta in beta_schedule:
        for _ in range(max_iter_per_beta):
            corr = np.correlate(h, 1.0 - h, mode="full")
            # stable soft‑max weights
            max_corr = np.max(corr)
            exp_vals = np.exp(beta * (corr - max_corr))
            Z = np.sum(exp_vals)
            p = exp_vals / Z

            g = np.zeros_like(h)
            # gradient: Σ_k p_k * ((1 - h_{i+k}) - h_{i-k})
            for k in range(-n + 1, n):
                pk = p[k + n - 1]
                if pk == 0.0:
                    continue
                if k >= 0:
                    nk = n - k
                    if nk > 0:
                        g[:nk] += pk * (1.0 - h[k:])
                        g[k:] -= pk * h[:nk]
                else:
                    kneg = -k
                    nk = n - kneg
                    if nk > 0:
                        g[kneg:] += pk * (1.0 - h[:nk])
                        g[:nk] -= pk * h[kneg:]
            h = h - step0 * g
            h = proj_box_sum(h, target)
    # final cleanup
    h = np.clip(h, 0.0, 1.0)
    adjust_sum_exact(h, np.random.default_rng())
    return h

# ---------- SLSQP on smooth‑max (optional fine‑tuning) ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=600):
    """SLSQP on the smooth‑max surrogate for a single β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- binary simulated‑annealing ----------
def binary_sa_until(h, rng, deadline):
    """Swap 1↔0 until time runs out; returns best binary vector seen."""
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        ones = np.where(cur_h == 1.0)[0]
        zeros = np.where(cur_h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5(cur_h)
        T = T0 * (0.9999 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c, best_h = c, cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
    return best_h, best_c

# ---------- greedy binary improvement ----------
def greedy_improve(h, rng, max_iter=50, deadline=None):
    """Local swap search that reduces the current worst shift."""
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c, best_pair = c_new, (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    # candidate grid sizes
    cand_ns = []
    if parent is not None:
        for div in (1, 2, 4):
            n = parent.size // div
            if n >= 180 and n % 2 == 0:
                cand_ns.append(n)
    else:
        cand_ns.extend([180, 240, 360])
    # add a few extra sizes (will be used only if time permits)
    for extra in (432, 504, 720):
        if extra >= 180 and extra % 2 == 0:
            cand_ns.append(extra)
    cand_ns = sorted(set(cand_ns))

    beta_sched = [5, 50, 500, 5000, 20000]

    for n in cand_ns:
        if time.time() > deadline:
            break
        target = n / 2.0

        # ----- build diverse starts -----
        starts = []

        # parent interpolation (if available)
        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            starts.append(interp)

        # random binary
        b = np.zeros(n)
        b[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(b)

        # random fractional
        f = rng.random(n)
        f *= target / f.sum()
        f = np.clip(f, 0.0, 1.0)
        adjust_sum_exact(f, rng)
        starts.append(f)

        # sinusoidal pattern (just for diversity)
        s = 0.5 * (1.0 + np.sin(2.0 * np.pi * np.arange(n) / n))
        s = np.clip(s, 0.0, 1.0)
        adjust_sum_exact(s, rng)
        starts.append(s)

        for h0 in starts:
            if time.time() > deadline:
                break
            h = h0.copy()
            adjust_sum_exact(h, rng)
            h = np.clip(h, 0.0, 1.0)

            # ----- PGD smooth‑max descent -----
            h = pgd_smooth_max(h, n, target, beta_sched,
                               step0=0.05, max_iter_per_beta=80)

            c = compute_c5(h)
            if c < best_c:
                best_c, best_h, best_n = c, h.copy(), n

            # ----- binary refinement if already binary -----
            if np.all((h <= 1e-8) | (h >= 1.0 - 1e-8)):
                # binary SA
                sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
                h_bin, c_bin = binary_sa_until(np.rint(h).astype(float), rng, sub_dead)
                if c_bin < best_c:
                    best_c, best_h, best_n = c_bin, h_bin.copy(), n
                h = h_bin.astype(float)

                # greedy swaps
                h = greedy_improve(h, rng, max_iter=100, deadline=deadline)
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h, best_n = c, h.copy(), n
                continue

            # ----- try rounding to binary and local improve -----
            h_bin = np.rint(h).astype(float)
            adjust_sum_exact(h_bin, rng)
            h_bin = greedy_improve(h_bin, rng, max_iter=100, deadline=deadline)
            c_bin = compute_c5(h_bin)
            if c_bin < best_c:
                best_c, best_h, best_n = c_bin, h_bin.copy(), n
                h = h_bin.astype(float)

    # ----- final SLSQP fine‑tuning on the current best -----
    if best_h is not None and time.time() < deadline:
        try:
            best_h = slsqp_optimize(best_h, beta=20000, rng=rng, max_iter=800)
            best_c = compute_c5(best_h)
        except Exception:
            pass

    # ----- optional up‑sampling and further refinement -----
    if (best_h is not None and time.time() < deadline and best_n * 2 <= 720):
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        target_up = n_up / 2.0
        adjust_sum_exact(h_up, rng)

        h_up = pgd_smooth_max(h_up, n_up, target_up, beta_sched,
                              step0=0.05, max_iter_per_beta=120)

        # binary SA if it became binary
        if np.all((h_up <= 1e-8) | (h_up >= 1.0 - 1e-8)):
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
            h_up, c_up = binary_sa_until(h_up, rng, sub_dead)
        else:
            h_bin = np.rint(h_up).astype(float)
            adjust_sum_exact(h_bin, rng)
            h_bin = greedy_improve(h_bin, rng, max_iter=120, deadline=deadline)
            c_up = compute_c5(h_bin)
            if c_up < best_c:
                best_h, best_c, best_n = h_bin, c_up, n_up
                h_up = h_bin

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ----- fallback in the extremely unlikely case no solution was found -----
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_n = n
        best_c = compute_c5(best_h)

    # final sanity checks
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
