def project_sum_box(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s

        if total > 0:
            h += diff * slack / total

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with spacing 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_box(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto [0,1]^n with sum = target (Euclidean projection)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        slack = 1.0 - h
        total = slack.sum()
        if total > 0:
            h += diff * slack / total
    else:
        slack = h
        total = slack.sum()
        if total > 0:
            h += diff * slack / total
    h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    resid = target - h.sum()
    if abs(resid) > 1e-9:
        mask = (h > 0) & (h < 1)
        cnt = mask.sum()
        if cnt:
            h[mask] += resid / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    return project_sum_box(h, n / 2.0)


# ---------- gradient utilities ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-5)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_box(h, n / 2.0)
    return h


# ---------- soft‑max (smoothed max) gradient ----------
def softmax_c5_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max of the overlap correlations:
    f_beta(h) = (1/beta) log sum_i exp(beta * corr_i).
    """
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    # stable softmax weights
    max_val = np.max(beta * corr)
    exp_vals = np.exp(beta * corr - max_val)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros_like(h)

    for idx, w in enumerate(weights):
        offset = idx - (n - 1)          # actual shift
        if offset >= 0:
            k = offset
            if k < n:
                grad[: n - k] += w * (1 - h[k:])
                grad[k:]     -= w * h[: n - k]
        else:
            k = -offset
            if k < n:
                grad[k:]     += w * (1 - h[: n - k])
                grad[: n - k] -= w * h[k:]
    return grad

    start = time.time()
    target_n = 512
    target_sum = target_n / 2.0

    # initialise from parent if present
    best_h = None
    best_c = float("inf")
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != target_n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_sum_box(h0, target_sum)
        best_h = h0
        best_c = compute_c5(h0)

    # ---------- main search ----------
    while time.time() - start < 0.94 * budget_s:
        # choose start point
        if best_h is not None and rng.random() < 0.78:
            h = best_h.copy()
            h += rng.normal(scale=0.05, size=target_n)
            h = np.clip(h, 0.0, 1.0)
            h = project_sum_box(h, target_sum)
        else:
            h = random_feasible_vector(target_n, rng)

        # aggressive hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=2500, step_scale=0.7)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # soft‑max refinement with increasing β
        for beta in (20., 80., 300., 1000., 3000., 10000.):
            step = 0.2
            for _ in range(140):
                grad = softmax_c5_gradient(h, beta)
                if np.linalg.norm(grad) < 1e-9:
                    break
                h_new = h - step * grad
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_sum_box(h_new, target_sum)
                c_new = compute_c5(h_new)

                if c_new < best_c:
                    best_h, best_c = h_new.copy(), c_new
                if c_new < c:
                    h, c = h_new, c_new
                    step = min(step * 1.02, 0.5)
                else:
                    step = max(step * 0.68, 5e-4)

                if time.time() - start > 0.93 * budget_s:
                    break
            if time.time() - start > 0.93 * budget_s:
                break

        # final fine‑tune
        h = gradient_refine(h, rng, step=0.015, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ---------- stochastic tail ----------
    while time.time() - start < 0.999 * budget_s:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(target_n)
