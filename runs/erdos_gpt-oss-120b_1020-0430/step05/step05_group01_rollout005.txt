def run(seed=42, budget_s=1100, **kwargs):
   rng = np.random.default_rng(seed)
   start = time.time()
   # get initial best

   if 'initial_h_values' in globals():
       h0 = np.array(initial_h_values, dtype=float)
   else:
       n = 512
       h0 = np.zeros(n)
       h0[rng.choice(n, n//2, replace=False)] = 1.0

   h0 = np.clip(h0, 0.0, 1.0)
   h0 = enforce_sum(h0, rng)
   best_h = h0.copy()
   best_c = compute_c5(best_h)

   # subgradient descent parameters
   n = best_h.size
   max_iter = 30000  # can adjust based on time

   step0 = 0.2

   for it in range(max_iter):
        if time.time() - start > budget_s * 0.80:
            break
        # compute gradient of worst shift
        g = worst_gradient(best_h)  # note: this recomputes correlation etc.
        step = step0 * (0.99 ** it)
        nh = best_h - step * g
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum(nh, rng)
        c = compute_c5(nh)
        if c < best_c:

        # optionally random perturb

import time
import numpy as np
from scipy.optimize import minimize

# --------------------- utility functions ---------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == len(h)/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    max_iter = 5000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # signed shift
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]          # subtract h[i+k] term
        grad[k:]    -= h[: n - k]       # subtract h[i-k] term
    return grad


# --------------------- SLSQP with soft‑max --------------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 1000) -> np.ndarray:
    """Run SLSQP on the soft‑max objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) <= 1e-12:
                break
    return x


# --------------------- Thue‑Morse generator ------------------------------

def thue_morse_bits(n: int) -> np.ndarray:
    """Return the first n bits of the Thue–Morse sequence (0/1)."""
    seq = np.array([0.0])
    while seq.size < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]


# --------------------- main routine ---------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial candidates ------------------------------------------------
    candidates = []

        h_init = np.clip(h_init, 0.0, 1.0)

        candidates.append((h_init, compute_c5(h_init)))

    # Thue–Morse (size 512, power of two)

    # random balanced binary (size 512)

    candidates.append((rand_bin, compute_c5(rand_bin)))

    # pick the best start
    best_h, best_c = min(candidates, key=lambda t: t[1])
    best_n = best_h.size

    # ---- subgradient descent (main phase) ---------------------------------
    max_iters = 20000
    step0 = 0.2
    it = 0

        grad = worst_gradient(best_h)          # dO_k/dh for worst shift

        if c < best_c:

            best_n = best_h.size

    for step in (0.05, 0.02, 0.01, 0.005):
        if time.time() - start > budget_s * 0.98:
            break
        for _ in range(300):
            grad = worst_gradient(best_h)
            nh = best_h - step * grad
            nh = np.clip(nh, 0.0, 1.0)
            nh = enforce_sum(nh, rng)
            c = compute_c5(nh)
            if c < best_c:
                best_c, best_h = c, nh.copy()
                if best_c <= 0.3808:
                    break

    # ---- random tweaking until time runs out --------------------------------
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps > 0:

                nh = best_h.copy()
                nh[i] -= eps

    # ---- final sanity enforcement -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
