def compute_c5_from_h(h):
    n = h.size
    # linear convolution using FFT of length 2n
    h_f = np.fft.rfft(h, n*2)
    g_f = np.fft.rfft(1 - h, n*2)

    # conv length 2n, but last element zero (since padding). Use conv[:2*n-1]
    conv = conv[:2*n-1]
    return np.max(conv) * (2.0 / n)

import time, math, numpy as np

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 (max overlap) via FFT."""
    n = h.shape[0]
    H = np.fft.rfft(h, n * 2)
    G = np.fft.rfft(1 - h, n * 2)
    conv = np.fft.irfft(H * G, n * 2)[: 2 * n - 1]
    return float(np.max(conv) * (2.0 / n))


def compute_c5_and_conv(h: np.ndarray):
    """Return (C5, full linear convolution array)."""
    n = h.shape[0]
    H = np.fft.rfft(h, n * 2)
    G = np.fft.rfft(1 - h, n * 2)
    conv = np.fft.irfft(H * G, n * 2)[: 2 * n - 1]
    return float(np.max(conv) * (2.0 / n)), conv


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force sum(h)=len(h)/2 while staying in [0,1] (random redistribution)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)

        new_c, conv = compute_c5_and_conv(h)
        delta = new_c - cur_c
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / time_budget)
        accept = (delta < 0) or (rng.random() < math.exp(-delta / (T + 1e-12)))

        if accept:
            cur_c = new_c
            # update candidate set for next iteration
            k = np.argmax(conv) - (n - 1)
            if k >= 0:
                mask = (h[: n - k] == 1) & (h[k:] == 0)
                cand_one = np.flatnonzero(mask)
            else:
                k2 = -k
                mask = (h[k2:] == 1) & (h[: n - k2] == 0)
                cand_one = np.flatnonzero(mask) + k2

            if new_c < best_c - 1e-12:
                best_c, best_h = new_c, h.copy()
        else:
            revert_swap_one_zero(h, i, j, ones, zeros, pos_one, pos_zero)
            # candidates unchanged

    return best_h, best_c


# ---------- gradient based helpers (continuous) ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 600) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = float('inf')
    best_n = None

    # ---- parent solution if available ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ---- try a binary rounding of the current best ----
    if best_h is not None:
        n = best_h.size
        idx = np.argpartition(best_h, -n // 2)[-n // 2 :]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin, c_bin, n

    # ---- binary simulated annealing (major chunk of time) ----
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 30.0:
        # choose resolution (prefer 512, but allow 256 if time limited)
        n_sa = 512
        if best_n is not None and best_n < 512:
            n_sa = best_n
        n_sa = max(128, (n_sa // 2) * 2)  # make even
        t_sa = remaining * 0.6
        h_sa, c_sa = binary_sa_optimize(n_sa, rng, t_sa)
        if c_sa < best_c:
            # upscale/downscale to original size if needed
            if best_n is not None and n_sa != best_n and best_n % n_sa == 0:
                factor = best_n // n_sa
                h_up = np.repeat(h_sa, factor)[:best_n]
                h_up = np.clip(h_up, 0.0, 1.0)
                h_up = enforce_sum(h_up, rng)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up, c_up, best_n
                else:
                    best_h, best_c, best_n = h_sa, c_sa, n_sa
            else:
                best_h, best_c, best_n = h_sa, c_sa, n_sa

    # ---- continuous refinement on the current best ----
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0 and best_h is not None:
        max_hill = int(min(3000, remaining * 2))
        best_h, best_c = guided_hill_climb(best_h, rng,
                                           max_iters=max_hill,
                                           step_scale=0.7)
        # SLSQP sweeps with increasing β
        for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
            if time.time() - start > budget_s * 0.95:
                break
            maxit = 800 if beta >= 200 else 500
            best_h = slsqp_optimize(best_h, beta, max_iter=maxit)
            best_c = compute_c5(best_h)
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=12)
        best_c = compute_c5(best_h)

    # ---- final stochastic nudges until time runs out ----
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ---- final sanity check ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
