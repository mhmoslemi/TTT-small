def binary_hill_climb(h, max_iters=10000):
    n = len(h)
    sum_h = n // 2
    conv = np.correlate(h, h, mode='full')
    best_c = (sum_h - conv.min()) * (2.0 / n)
    it = 0
    while it < max_iters:
        # find worst shift
        s_idx = conv.argmin()
        s0 = s_idx - (n - 1)
        # compute contributions
        contrib = contributions_at_shift(h, s0)
        # find i and j
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        # if no potential improvement, break
        i = ones_idx[np.argmin(contrib[ones_idx])]
        j = zeros_idx[np.argmax(contrib[zeros_idx])]
        if contrib[j] <= contrib[i]:
            break
        # compute delta
        delta = compute_delta_conv(h, i, j)  # O(n)
        new_conv = conv + delta
        new_min = new_conv.min()
        new_c = (sum_h - new_min) * (2.0 / n)
        if new_c < best_c - 1e-12:
            # accept swap
            h[i] = 0
            h[j] = 1
            conv = new_conv
            best_c = new_c
            # continue
        else:
            break
        it += 1
    return h, best_c

    delta = np.zeros(2*n-1, dtype=int)
    # forward/back contributions for i: subtract
    # loops over s where i+s in range:
    # for s = -(i) .. (n-1-i)
    start = -i
    end = n-1-i
    # s_idx = np.arange(start, end+1) + (n-1)
    # i_s_contrib = h[i + np.arange(start, end+1)]  # values of neighbor.
    # delta[s_idx] -= i_s_contrib
    # similarly for i - s:
    start2 = i-(n-1)
    end2 = i
    s_idx2 = np.arange(start2, end2+1) + (n-1)
    delta[s_idx2] -= h[i - np.arange(start2, end2+1)]
    # contributions from j: add
    start = -j
    end = n-1-j
    s_idx = np.arange(start, end+1) + (n-1)
    delta[s_idx] += h[j + np.arange(start, end+1)]
    start2 = j-(n-1)
    end2 = j
    s_idx2 = np.arange(start2, end2+1) + (n-1)

    idx2 = -s_ij + (n-1)
    delta[idx2] -= 1

    # set delta at s=0 to 0

def compute_delta_conv(h, i, j):
    n = h.shape[0]
    size = 2*n - 1
    delta = np.zeros(size, dtype=np.int32)
    # subtract contributions from i
    # forward shift:
    start = -i
    end = n - 1 - i
    s_idxs = np.arange(start, end+1) + (n-1)
    idxs = i + np.arange(start, end+1)  # neighbor indices
    delta[s_idxs] -= h[idxs]
    # backward shift:
    start2 = i - (n - 1)
    end2 = i
    s_idxs2 = np.arange(start2, end2+1) + (n-1)
    idxs2 = i - np.arange(start2, end2+1)
    delta[s_idxs2] -= h[idxs2]

    # add contributions from j
    start = -j
    end = n - 1 - j
    s_idxs = np.arange(start, end+1) + (n-1)
    idxs = j + np.arange(start, end+1)
    delta[s_idxs] += h[idxs]
    start2 = j - (n - 1)
    end2 = j
    s_idxs2 = np.arange(start2, end2+1) + (n-1)
    idxs2 = j - np.arange(start2, end2+1)
    delta[s_idxs2] += h[idxs2]

    # adjust cross terms
    s_ij = i - j
    idx1 = s_ij + (n-1)
    delta[idx1] -= 1
    idx2 = -s_ij + (n-1)
    delta[idx2] -= 1

    # zero out central shift
    delta[n-1] = 0
    return delta

def binary_hill_climb(h, max_iters=10000):
    n = h.shape[0]
    sum_h = n // 2
    conv = np.correlate(h, h, mode='full')
    conv = conv.astype(np.int32)
    best_c = (sum_h - conv.min()) * (2.0 / n)
    it = 0
    while it < max_iters:
        # worst shift
        s_idx = int(conv.argmin())
        s0 = s_idx - (n - 1)
        # contributions
        contrib = contributions_at_shift(h, s0)
        # indices of ones and zeros
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        # find i with minimal contrib among ones
        # Use np.argmin on contrib[ones]
        i = ones[int(np.argmin(contrib[ones]))]
        j = zeros[int(np.argmax(contrib[zeros]))]
        if contrib[j] <= contrib[i]:
            break
        # compute delta
        delta = compute_delta_conv(h, i, j)
        new_conv = conv + delta
        new_c = (sum_h - new_conv.min()) * (2.0 / n)
        if new_c < best_c - 1e-14:
            # accept
            h[i] = 0
            h[j] = 1
            conv = new_conv
            best_c = new_c
        else:
            break
        it += 1
    return h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global variables for the soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_continuous(x0: np.ndarray, beta: float,
                              rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def contributions_at_shift(h: np.ndarray, s: int) -> np.ndarray:
    """Contribution of each index to the auto‑correlation at shift s."""
    n = h.size
    contrib = np.zeros(n, dtype=np.int32)
    if s >= 0:
        contrib[: n - s] += h[s:]
        contrib[s:] += h[: n - s]
    else:
        t = -s
        contrib[t:] += h[: n - t]
        contrib[: n - t] += h[t:]
    return contrib


def compute_delta_conv(h: np.ndarray, i: int, j: int) -> np.ndarray:
    """Delta to the full auto‑correlation when swapping i (1) with j (0)."""
    n = h.shape[0]
    size = 2 * n - 1
    delta = np.zeros(size, dtype=np.int32)
    rng = np.arange

    # subtract contributions of i (which becomes 0)
    start = -i
    end = n - 1 - i
    s_idx = rng(start, end + 1) + (n - 1)
    idx = i + rng(start, end + 1)
    delta[s_idx] -= h[idx]

    start2 = i - (n - 1)
    end2 = i
    s_idx2 = rng(start2, end2 + 1) + (n - 1)
    idx2 = i - rng(start2, end2 + 1)
    delta[s_idx2] -= h[idx2]

    # add contributions of j (which becomes 1)
    start = -j
    end = n - 1 - j
    s_idx = rng(start, end + 1) + (n - 1)
    idx = j + rng(start, end + 1)
    delta[s_idx] += h[idx]

    start2 = j - (n - 1)
    end2 = j
    s_idx2 = rng(start2, end2 + 1) + (n - 1)
    idx2 = j - rng(start2, end2 + 1)
    delta[s_idx2] += h[idx2]

    # correct the two cross‑terms where the swapped positions meet
    s_ij = i - j
    delta[s_ij + (n - 1)] -= 1
    delta[-s_ij + (n - 1)] -= 1

    # keep the zero‑shift unchanged
    delta[n - 1] = 0
    return delta


def binary_hill_climb(h_bin: np.ndarray, max_iters: int = 20000):
    """
    Greedy hill‑climbing on a binary vector.
    Returns the improved vector and its C5 value.
    """
    n = h_bin.shape[0]
    sum_h = n // 2
    conv = np.correlate(h_bin, h_bin, mode="full").astype(np.int32)
    best_c = (sum_h - conv.min()) * (2.0 / n)

    it = 0
    while it < max_iters:
        # worst shift
        s_idx = int(conv.argmin())
        s0 = s_idx - (n - 1)

        # contributions of each index to that shift
        contrib = contributions_at_shift(h_bin, s0)

        ones = np.flatnonzero(h_bin)
        zeros = np.flatnonzero(1 - h_bin)

        i = ones[int(np.argmin(contrib[ones]))]
        j = zeros[int(np.argmax(contrib[zeros]))]

        if contrib[j] <= contrib[i]:
            break

        delta = compute_delta_conv(h_bin, i, j)
        new_conv = conv + delta
        new_c = (sum_h - new_conv.min()) * (2.0 / n)

        if new_c < best_c - 1e-14:
            h_bin[i] = 0
            h_bin[j] = 1
            conv = new_conv
            best_c = new_c
            it += 1
        else:
            break

    return h_bin, best_c


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ---- evaluate any supplied parent construction ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        if c0 < best_c:
            best_h, best_c, best_n = h0.copy(), c0, h0.size

        # a binary version of the parent (rounded)
        h_bin = (np.array(initial_h_values) > 0.5).astype(np.int32)
        target = h_bin.size // 2
        diff = target - h_bin.sum()
        if diff != 0:
            if diff > 0:
                zero_idx = np.flatnonzero(h_bin == 0)
                flip = rng.choice(zero_idx, diff, replace=False)
                h_bin[flip] = 1
            else:
                one_idx = np.flatnonzero(h_bin == 1)
                flip = rng.choice(one_idx, -diff, replace=False)
                h_bin[flip] = 0
        c_bin = compute_c5(h_bin.astype(float))
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin.astype(float), c_bin, h_bin.size

    # ---- sizes to explore ----
    sizes = [256, 384, 512]

    # ---- main random‑restart loop ----
    while time.time() - start < budget_s * 0.85:
        n = int(rng.choice(sizes))
        # random binary vector with exactly n/2 ones
        h = np.zeros(n, dtype=np.int32)
        h[rng.choice(n, n // 2, replace=False)] = 1

        # improve it with greedy binary hill climbing
        h_opt, c_opt = binary_hill_climb(h, max_iters=30000)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.astype(float), c_opt, n

        # if we are at a smaller resolution, also try up‑sampling to 512
        if n < 512:
            factor = 512 // n
            if factor * n == 512:
                h_up = np.repeat(h_opt, factor)
                h_up_opt, c_up_opt = binary_hill_climb(h_up, max_iters=30000)
                if c_up_opt < best_c:
                    best_h, best_c, best_n = h_up_opt.astype(float), c_up_opt, 512

    # ---- final polishing (continuous SLSQP) ----
    if best_h is not None:
        # if we already have a binary optimum, try a long binary climb first
        if np.all((best_h == 0) | (best_h == 1)):
            h_bin = best_h.astype(np.int32)
            h_bin, c_fin = binary_hill_climb(h_bin, max_iters=100000)
            best_h, best_c = h_bin.astype(float), c_fin

        # a few soft‑max SLSQP passes at increasing β
        for beta in (50.0, 500.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.98:
                break
            best_h = slsqp_optimize_continuous(best_h, beta, rng, max_iter=500)
            best_c = compute_c5(best_h)

    # ---- final sanity checks ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
