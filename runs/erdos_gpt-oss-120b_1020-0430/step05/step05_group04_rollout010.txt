def binary_sa_optimize(h, rng, deadline):
    n = h.size
    cur_h = h.copy()
    # compute initial correlation
    corr = np.correlate(cur_h, 1 - cur_h, mode='full')
    cur_val = corr.max()
    best_h = cur_h.copy()
    best_val = cur_val
    best_corr = corr.copy()
    # precompute position indices of ones and zeros
    ones = np.flatnonzero(cur_h)
    zeros = np.flatnonzero(cur_h == 0)
    # initialize temperature
    T0 = max(0.01 * cur_val, 1e-4)
    alpha = 0.9995
    it = 0
    last_improve = 0
    # loop
    while time.time() < deadline:
        # random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta using padded h
        # create padded h only once per iteration?
        # we can pad cur_h each iteration as an array distinct.
        h_ext = np.pad(cur_h, (n-1, n-1), mode='constant')
        delta = h_ext[i:i+2*n-1] - h_ext[j:j+2*n-1]
        new_corr = corr + delta
        new_val = new_corr.max()
        # temperature
        T = T0 * (alpha ** it)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if rng.random() < np.exp(-(new_val - cur_val) / T):
                accept = True
        if accept:
            # update structures
            cur_h[i] = 0
            cur_h[j] = 1
            # update ones and zeros lists efficiently
            # use python list for faster removal? Use sets?
            # we can replace ones list by python list and modify: find index of i in ones, replace with j.
            # Let's use numpy arrays but updates using list may be easier.
            # We'll keep python lists.
            # So find index of i in ones.
            idx_i = ones.index(i)
            ones[idx_i] = j
            idx_j = zeros.index(j)
            zeros[idx_j] = i
            corr = new_corr
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = cur_h.copy()
                best_corr = corr.copy()
                last_improve = it
        it += 1
        # break if many iterations without improvement
        if it - last_improve > 20000:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95

    # get parent if present
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        # ensure even length
        if parent.size % 2 != 0:
            parent = parent[:-1]
    else:
        parent = None

    # initialize best variables
    best_h = None
    best_c = np.inf
    best_n = None

    # function to evaluate and update best
    def maybe_update(h):
        nonlocal best_h, best_c, best_n
        c = compute_c5(h)
        if c < best_c:
            best_h = h.copy()
            best_c = c
            best_n = h.size

    # Stage 1: start from parent (or random) at size n0
    if parent is not None:
        n0 = parent.size
        # make binary version for SA
        # ensure sum = n/2
        # For binary, we need exactly n/2 ones. Use top n/2 values.
        idx_top = np.argpartition(parent, -n0//2)[-n0//2:]
        binary_h = np.zeros(n0, dtype=float)
        binary_h[idx_top] = 1.0
        maybe_update(binary_h)
        # Run SA on this binary at size n0 for limited time (e.g., 0.1 * remaining)
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.2)
        h_bin, best_val_raw = binary_sa_optimize(binary_h, rng, sub_deadline)
        # convert raw max to c5
        c_bin = best_val_raw * (2.0 / n0)
        maybe_update(h_bin)
        # proceed gradient refine on binary (as float)
        h_cont = h_bin.astype(float)
        h_cont = gradient_refine(h_cont, rng, step=0.02, max_iter=30)
        maybe_update(h_cont)
        # SLSQP refinement: call slsqp_optimize with multiple betas
        h_cont = slsqp_multi_optimize(h_cont, rng, n0, deadline)

import time, math
import numpy as np
from scipy.optimize import minimize

# ----- global vars for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact redistribution to satisfy Σh = n/2 (in‑place)."""
    n = h.size
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
    return h

def gradient_refine(h, rng, step=0.02, max_iter=30):
    """Project gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]   # + (1 - h[i+k])
            g[k:] -= h[:up]          # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    # exact sum correction
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def slsqp_multi_optimize(h, rng, deadline):
    """Cascade through a list of betas respecting the deadline."""
    betas = (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000)
    for beta in betas:
        if time.time() >= deadline:
            break
        # allocate iterations based on beta (more for larger beta)
        max_iter = 1200 if beta >= 2000 else 800
        h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
    return h

def binary_sa_optimize(h_bin, rng, deadline):
    """Simulated‑annealing on a binary vector, incremental correlation."""
    n = h_bin.size
    h = h_bin.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    cur_val = corr.max()
    best_h = h.copy()
    best_val = cur_val
    # maintain padded version for O(1) slice updates
    h_ext = np.pad(h, (n - 1, n - 1), mode="constant")
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    T0 = max(0.01 * cur_val, 1e-4)
    alpha = 0.9998
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        delta = h_ext[i:i + 2 * n - 1] - h_ext[j:j + 2 * n - 1]
        new_corr = corr + delta
        new_val = new_corr.max()
        T = T0 * (alpha ** it)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if rng.random() < math.exp(-(new_val - cur_val) / T):
                accept = True
        if accept:
            # swap in h and h_ext
            h[i] = 0.0
            h[j] = 1.0
            h_ext[n - 1 + i] = 0.0
            h_ext[n - 1 + j] = 1.0
            # update lists
            ones[i_idx] = j
            zeros[j_idx] = i
            corr = new_corr
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
                last_improve = it
        it += 1
        if it - last_improve > 50000:
            break
    return best_h, best_val
