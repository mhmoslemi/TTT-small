def slsqp_optimize_v(v0, beta, rng, max_iter):
    global CURRENT_BETA_CURR
    n = v0.size
    target = n/2.0
    def softmax_obj_v(x):
        h = np.repeat(x, repeat)
        corr = np.correlate(h, 1.0 - h, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    constraints = {'type':'eq','fun': lambda x: np.sum(x) - target}
    bounds = [(0.0,1.0)] * n
    res = minimize(softmax_obj_v, v0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0,1)
    # we may need to enforce sum due to numerical drift
    # adjust via enforce_sum
    x = enforce_sum(x, rng)
    return x
```

def block_aggregate_sum(arr, repeat):
    # arr length assumed multiple of repeat
    n_blocks = arr.size // repeat
    # Using reshape
    return arr.reshape((n_blocks, repeat)).sum(axis=1)

def guided_hill_climb_v(v, rng, max_iters=2000, step_scale=0.8):
    n = v.size
    repeat = CURRENT_REPEAT_V
    best_v = v.copy()
    best_c = compute_c5(np.repeat(v, repeat))
    for it in range(max_iters):
        h = np.repeat(v, repeat)
        grad_h = worst_gradient(h)
        grad_v = block_aggregate_sum(grad_h, repeat)
        # find max/min
        i = int(np.argmax(grad_v))
        j = int(np.argmin(grad_v))
        if i == j:
            # random perturb
            i, j = rng.integers(0, n, size=2)
        max_eps = min(v[i], 1.0 - v[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * step_scale
        v[i] -= eps
        v[j] += eps
        c = compute_c5(np.repeat(v, repeat))
        if c < best_c:
            best_c = c
            best_v = v.copy()
        # optional decay of step_scale
        if (it+1) % (max_iters//5+1) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_v, best_c
```

def guided_hill_climb_v(v, rng, max_iters=2000, step_scale=0.8):
    repeat = CURRENT_REPEAT_V
    best_v = v.copy()
    best_c = compute_c5(np.repeat(v, repeat))

        grad_h = worst_gradient(h)
        grad_v = block_aggregate_sum(grad_h)
        i = int(np.argmax(grad_v))
        j = int(np.argmin(grad_v))
        if i == j:
            # random distinct indices
            i, j = rng.integers(0, v.size, size=2)
            if i == j:

        v[i] -= eps
        v[j] += eps

        if c < best_c:
            best_c = c
            best_v = v.copy()
        # decay
        if (it + 1) % decay_interval == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_v, best_c

def sum_eq_constraint_v(x):

def softmax_obj_v(x):
    # x is v vector
    h = np.repeat(x, CURRENT_REPEAT_V)
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = CURRENT_BETA_V * corr
    m = np.max(bc)
    # compute softmax in a stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA_V

def slsqp_optimize_v(v0, beta, rng, max_iter=400):

    res = minimize(
        softmax_obj_v,
        v0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP on the interval representation
CURRENT_BETA_V = None
CURRENT_TARGET_V = None
CURRENT_REPEAT_V = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with N samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σarr = len(arr)/2 while keeping 0≤arr≤1."""
    n = arr.size
    target = n / 2.0
    diff = target - arr.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - arr[i]) if diff > 0 else max(diff, -arr[i])
        arr[i] += add
        diff = target - arr.sum()
    return arr


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def block_aggregate_sum(arr: np.ndarray) -> np.ndarray:
    """Sum of `arr` over blocks of size CURRENT_REPEAT_V."""
    repeat = CURRENT_REPEAT_V
    return arr.reshape((-1, repeat)).sum(axis=1)


def guided_hill_climb_v(
    v: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    step_scale: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Hill‑climb on the interval vector `v` using gradient transfers."""
    repeat = CURRENT_REPEAT_V
    best_v = v.copy()
    best_c = compute_c5(np.repeat(v, repeat))
    scale = step_scale
    decay_interval = max(1, max_iters // 5)

    for it in range(max_iters):
        h = np.repeat(v, repeat)
        grad_h = worst_gradient(h)
        grad_v = block_aggregate_sum(grad_h)

        i = int(np.argmax(grad_v))
        j = int(np.argmin(grad_v))
        if i == j:
            i, j = rng.integers(0, v.size, size=2)
            if i == j:
                continue

        max_eps = min(v[i], 1.0 - v[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * scale
        v[i] -= eps
        v[j] += eps

        c = compute_c5(np.repeat(v, repeat))
        if c < best_c:
            best_c = c
            best_v = v.copy()

        if (it + 1) % decay_interval == 0:
            scale = max(scale * 0.7, 1e-4)

    return best_v, best_c


def sum_eq_constraint_v(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET_V


def softmax_obj_v(x: np.ndarray) -> float:
    """Smooth approximation of max_k O_k for the interval vector."""
    h = np.repeat(x, CURRENT_REPEAT_V)
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = CURRENT_BETA_V * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA_V


def slsqp_optimize_v(
    v0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 300
) -> np.ndarray:
    """SLSQP on the interval vector using the soft‑max surrogate."""
    global CURRENT_BETA_V, CURRENT_TARGET_V
    n = v0.size
    CURRENT_BETA_V = beta
    CURRENT_TARGET_V = n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint_v}
    res = minimize(
        softmax_obj_v,
        v0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # enforce equality exactly (tiny numerical drift)
    diff = CURRENT_TARGET_V - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET_V - x.sum()
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    N = 512  # total samples
    # start with 32 intervals (repeat = 16)
    K = 32
    repeat = N // K
    global CURRENT_REPEAT_V
    CURRENT_REPEAT_V = repeat

    # initialise interval vector v
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != N:
            # interpolate to N points if necessary
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        v0 = h0.reshape((K, repeat)).mean(axis=1)
    else:
        v0 = np.zeros(K)
        idx = rng.choice(K, K // 2, replace=False)
        v0[idx] = 1.0

    v0 = np.clip(v0, 0.0, 1.0)
    v0 = enforce_sum(v0, rng)

    best_v = v0.copy()
    best_c = compute_c5(np.repeat(best_v, repeat))

    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.time() - start < budget_s * 0.99:
        # occasional random restart
        if rng.random() < 0.15:
            vr = rng.random(K)
            vr = enforce_sum(vr, rng)
            vr_opt, c_rand = guided_hill_climb_v(vr, rng, max_iters=500, step_scale=0.5)
            if c_rand < best_c:
                best_v, best_c = vr_opt, c_rand

        # guided hill‑climb on current best
        v_opt, c_opt = guided_hill_climb_v(
            best_v,
            rng,
            max_iters=int(2500 * 32 / K),
            step_scale=0.8,
        )
        if c_opt < best_c:
            best_c = c_opt
            best_v = v_opt

        # small random transfer perturbation
        if rng.random() < 0.3:
            i, j = rng.integers(0, K, size=2)
            if i != j:
                max_eps = min(best_v[i], 1.0 - best_v[j])
                eps = rng.random() * max_eps * 0.5
                best_v[i] -= eps
                best_v[j] += eps
                c_tmp = compute_c5(np.repeat(best_v, repeat))
                if c_tmp < best_c:
                    best_c = c_tmp

        # SLSQP refinements with increasing β
        for beta in (10.0, 100.0, 1000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.98:
                break
            best_v = slsqp_optimize_v(best_v, beta, rng, max_iter=300)
            c_tmp = compute_c5(np.repeat(best_v, repeat))
            if c_tmp < best_c:
                best_c = c_tmp

        # light gradient refinement
        v_ref, c_ref = guided_hill_climb_v(best_v, rng, max_iters=600, step_scale=0.4)
        if c_ref < best_c:
            best_c = c_ref
            best_v = v_ref

        # optional upscale to 64 intervals if time permits
        if K < 64 and time.time() - start > budget_s * 0.6:
            new_K = K * 2
            if N % new_K == 0:
                new_repeat = N // new_K
                x_old = np.arange(K)
                x_new = np.linspace(0, K, new_K, endpoint=False)
                v_interp = np.interp(x_new, x_old, best_v)
                CURRENT_REPEAT_V = new_repeat
                K = new_K
                repeat = new_repeat
                v_interp = np.clip(v_interp, 0.0, 1.0)
                v_interp = enforce_sum(v_interp, rng)
                best_v = v_interp
                best_c = compute_c5(np.repeat(best_v, repeat))

    # ------------------------------------------------------------------
    # Final fine‑tuning by pairwise epsilon swaps
    while time.time() - start < budget_s * 0.999:
        i = rng.integers(0, K)
        j = (i + 1) % K
        for sign in (-1, 1):
            eps = 0.001 * sign
            if 0.0 <= best_v[i] + eps <= 1.0 and 0.0 <= best_v[j] - eps <= 1.0:
                cand = best_v.copy()
                cand[i] += eps
                cand[j] -= eps
                c_cand = compute_c5(np.repeat(cand, repeat))
                if c_cand < best_c:
                    best_c = c_cand
                    best_v = cand

    # final sanitisation
    best_h = np.repeat(best_v, repeat)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(N)
