def delta_corr_swap(i, j, h):
    n = len(h)
    offset = n - 1
    delta = np.zeros(2*n-1, dtype=int)
    # i first
    start_i = offset - i
    end_i = offset + (n - i)
    delta[start_i:end_i] += 1
    delta[start_i:end_i] -= h[i:]  # h[i:l]
    # i second
    delta[offset + i - np.arange(n)] -= h
    # j first
    start_j = offset - j
    end_j = offset + (n - j)
    delta[start_j:end_j] -= 1
    delta[start_j:end_j] += h[j:]
    # j second
    delta[offset + j - np.arange(n)] += h

    # adjust for i+l=j case: l = j - i
    idx = offset + (j - i)
    if 0 <= idx < 2*n-1:
        delta[idx] += 1  # compensate

    return delta

def local_search(h, time_limit):
    n = h.size
    offset = n-1
    cur_corr = np.correlate(h, 1 - h, mode='full')
    cur_max = cur_corr.max()
    best_h = h.copy()
    best_max = cur_max
    no_improve = 0
    iter = 0
    delta = np.empty(2*n-1, dtype=np.int64)
    T0 = cur_max * 0.02 + 1e-6
    while time.time() < time_limit:
        # find worst shift
        idx = np.argmax(cur_corr)
        k = idx - offset
        # get candidate sets
        i_candidates, j_candidates = get_candidate_pairs(h, k)
        if i_candidates.size > 0 and j_candidates.size > 0:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        else:
            # random swap
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            i = rng.choice(zeros)
            j = rng.choice(ones)
        # compute delta
        apply_delta_swap(i, j, h, delta)
        new_corr = cur_corr + delta
        new_max = new_corr.max()
        # temperature
        T = T0 * (0.9995 ** iter) + 1e-12
        accept = new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / T)
        if accept:
            # commit swap
            h[i] = 1
            h[j] = 0
            cur_corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        iter += 1
        # optional restart if stuck
        if no_improve > 50000:
            break
    return best_h, best_max

def gradient_refine(h, rng, step=0.02, max_iter=15):
    n = h.shape[0]
    target = n / 2
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = idx  # shift
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            if k > 0:
                # For i from 0 to n - k -1: term from i and i+k
                # we derived earlier earlier gradient: g[:n - k] = 1 - h[k:] and g[k:] = -h[:n - k]
                g[: n - k] += 1.0 - h[k:]
                g[k:] -= h[: n - k]
            else:
                # k negative: treat similarly?
                # Use symmetry: correlation for -k same as for k swapping? Instead we can compute by shifting opposite.
                kk = -k
                g[kk:] += 1.0 - h[: n - kk]
                g[: n - kk] -= h[kk:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # adjust sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:

            for _ in range(200):

                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def local_search(h, deadline, rng):
    n = h.shape[0]
    offset = n - 1
    best_h = h.copy()
    cur_corr = np.correlate(h, 1 - h, mode='full')
    cur_max = cur_corr.max()
    best_max = cur_max
    best_h = h.copy()
    it = 0
    no_improve = 0
    # initial temperature maybe proportional to cur_max
    T0 = cur_max * 0.02 + 1e-6
    while time.time() < deadline:
        idx = np.argmax(cur_corr)
        k = idx - offset
        i_cands, j_cands = get_candidate_pairs(h, k)
        if i_cands.size > 0 and j_cands.size > 0:
            i = rng.choice(i_cands)
            j = rng.choice(j_cands)
        else:
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            if zeros.size == 0 or ones.size == 0:
                break
            i = rng.choice(zeros)
            j = rng.choice(ones)
        # swap
        h[i] = 1
        h[j] = 0
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = new_corr.max()
        # temperature schedule
        T = T0 * (0.9995 ** it) + 1e-12
        accept = new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / T)
        if accept:

import time
import numpy as np

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=15):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            kk = abs(k)
            upper = n - kk
            g[:upper] += 1.0 - h[kk:]
            g[kk:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def get_candidate_pairs(h, k):
    n = h.shape[0]
    if k >= 0:
        i_idx = np.arange(n - k)
        i_cands = i_idx[(h[i_idx] == 0) & (h[i_idx + k] == 1)]
        j_idx = np.arange(k, n)
        j_cands = j_idx[(h[j_idx] == 1) & (h[j_idx - k] == 0)]
    else:
        kp = -k
        i_idx = np.arange(kp, n)
        i_cands = i_idx[(h[i_idx] == 0) & (h[i_idx - kp] == 1)]
        j_idx = np.arange(kp, n)
        j_cands = j_idx[(h[j_idx] == 1) & (h[j_idx - kp] == 0)]
    return i_cands, j_cands

def local_search(h_start, deadline, rng):
    n = h_start.size
    offset = n - 1
    h = h_start.copy()
    cur_corr = np.correlate(h, 1 - h, mode="full")
    cur_max = cur_corr.max()
    best_h = h.copy()
    best_max = cur_max
    it = 0
    no_improve = 0
    T0 = cur_max * 0.02 + 1e-6
    while time.time() < deadline:
        idx = np.argmax(cur_corr)
        k = idx - offset
        i_cands, j_cands = get_candidate_pairs(h, k)
        if i_cands.size and j_cands.size:
            i = rng.choice(i_cands)
            j = rng.choice(j_cands)
        else:
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            if zeros.size == 0 or ones.size == 0:
                break
            i = rng.choice(zeros)
            j = rng.choice(ones)
        h[i] = 1
        h[j] = 0
        new_corr = np.correlate(h, 1 - h, mode="full")
        new_max = new_corr.max()
        T = T0 * (0.9995 ** it) + 1e-12
        accept = new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / T)
        if accept:
            cur_corr = new_corr
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 0
            h[j] = 1
            no_improve += 1
        it += 1
        if no_improve > 50000:
            break
    return best_h, best_max

def compute_c5(arr, n):
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(corr.max() * dx)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96  # small safety margin

    n = 720
    half = n // 2
    best_c = np.inf
    best_h = None

    # ----- generate initial binary starts -----
    start_vectors = []

    parent = globals().get("initial_h_values")
    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        up = np.interp(xt, xs, parent)
        idx = np.argpartition(up, -half)[-half:]
        bin_vec = np.zeros(n, dtype=np.int8)
        bin_vec[idx] = 1
        start_vectors.append(bin_vec)

    for _ in range(3):
        vec = np.zeros(n, dtype=np.int8)
        vec[rng.choice(n, half, replace=False)] = 1
        start_vectors.append(vec)

    # ----- run local searches on the starts -----
    for h0 in start_vectors:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # give each start up to 20 % of remaining time, capped at 120 s
        run_deadline = time.time() + min(remaining * 0.2, 120.0)
        h_best_loc, max_corr_loc = local_search(h0.copy(), run_deadline, rng)
        c_loc = max_corr_loc * (2.0 / n)
        if c_loc < best_c:
            best_c = c_loc
            best_h = h_best_loc.copy()

    # ----- additional random restarts while time permits -----
    while time.time() < deadline - 20.0:
        vec = np.zeros(n, dtype=np.int8)
        vec[rng.choice(n, half, replace=False)] = 1
        run_deadline = time.time() + min((deadline - time.time()) * 0.5, 90.0)
        h_best_loc, max_corr_loc = local_search(vec, run_deadline, rng)
        c_loc = max_corr_loc * (2.0 / n)
        if c_loc < best_c:
            best_c = c_loc
            best_h = h_best_loc.copy()

    # fallback if nothing succeeded
    if best_h is None:
        best_h = np.zeros(n, dtype=np.int8)
        best_h[:half] = 1

    # ----- fractional refinement -----
    h_float = best_h.astype(np.float64)
    h_float = gradient_refine(h_float, rng, step=0.02, max_iter=15)
    h_float = np.clip(h_float, 0.0, 1.0)
    h_float = adjust_sum_exact(h_float, rng)
    final_c = compute_c5(h_float, n)

    return h_float.tolist(), float(final_c), int(n)
