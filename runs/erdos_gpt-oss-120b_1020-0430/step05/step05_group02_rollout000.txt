def local_sa(h0, rng, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    cur_c = best_c
    # Initial temperature based on range of C values
    # Determine worst shift value O_max = cur_c; the potential improvement maybe up to cur_c - lower_bound_estimate
    T0 = cur_c * 0.1 + 1e-6 # some small baseline
    decay = 0.9995

    iteration = 0
    while time.time() < deadline:
        i, j = rng.integers(0,n, size=2)
        if i == j:
            continue
        # Ensure donor has higher value
        if h[i] < h[j]:
            i, j = j, i
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        # Choose delta proportion
        delta = rng.random() * max_delta * 0.5  # adjust factor
        if delta == 0:
            continue
        h[i] -= delta
        h[j] += delta
        cur_c_new = compute_c5(h)
        delta_c = cur_c_new - cur_c
        T = T0 * (decay ** iteration)
        accept = False
        if delta_c < -1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-delta_c / max(T, 1e-12)):
                accept = True
        if accept:
            cur_c = cur_c_new
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        iteration += 1
        # occasionally reheat (increase T) if stuck? maybe not needed.

    return best_h, best_c

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass to satisfy Σh = n/2 exactly (h stays in [0,1])."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random walk repair
    for _ in range(2000):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final clip for safety
    h.clip(0.0, 1.0, out=h)
    return h

def upsample(h: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of h from its current length to n_target points."""
    x_src = np.linspace(0.0, 2.0, h.size, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_tgt = np.interp(x_tgt, x_src, h)
    return h_tgt

def simulated_annealing(h0: np.ndarray, rng: np.random.Generator,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfers with a linear temperature schedule."""
    h = h0.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    total_time = max(deadline - start, 1e-12)

    iteration = 0
    while time.time() < deadline:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # ensure i is the donor (higher value)
        if h[i] < h[j]:
            i, j = j, i
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue

        # occasionally take a full transfer (swap) to explore binary space
        if rng.random() < 0.08:
            delta = max_delta
        else:
            delta = max_delta * rng.random()

        # apply move
        h[i] -= delta
        h[j] += delta

        new_c = compute_c5(h)
        dE = new_c - cur_c

        # linear temperature decreasing to zero at deadline
        remaining = deadline - time.time()
        T = cur_c * (remaining / total_time)  # cur_c is a convenient scale

        if dE < -1e-14 or rng.random() < np.exp(-dE / max(T, 1e-15)):
            cur_c = new_c
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert move
            h[i] += delta
            h[j] -= delta

        iteration += 1
    return best_h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    # initialise best holder
    best_h = None
    best_c = np.inf
    best_n = None

    # parent construction, if supplied
    parent = None
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)

    # --------------------------------------------------------------
    # Phase 1 – optimisation at n = 360 (or parent length)
    # --------------------------------------------------------------
    n0 = 360
    if parent is not None and parent.size != n0:
        # interpolate parent to n0 for consistency
        parent_n0 = upsample(parent, n0)
    else:
        parent_n0 = parent

    candidates = []

    if parent_n0 is not None:
        candidates.append(parent_n0.copy())

        # binary version of the parent (threshold at median)
        thresh = np.median(parent_n0)
        bin_parent = (parent_n0 >= thresh).astype(float)
        bin_parent = adjust_sum_exact(bin_parent, rng)
        candidates.append(bin_parent)

    # a few random starts
    for _ in range(2):
        h_rand = np.zeros(n0)
        h_rand[rng.choice(n0, n0 // 2, replace=False)] = 1.0
        candidates.append(h_rand)

    # allocate ~30 % of total time to phase 1
    phase1_deadline = start_time + 0.30 * budget_s

    for h0 in candidates:
        if time.time() >= phase1_deadline:
            break
        h0 = adjust_sum_exact(h0, rng)
        sub_dead = min(phase1_deadline, time.time() + 0.15 * budget_s)
        h_opt, c_opt = simulated_annealing(h0, rng, sub_dead)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, n0

    # --------------------------------------------------------------
    # Phase 2 – up‑sample best solution and continue refinement
    # --------------------------------------------------------------
    if best_h is not None and time.time() < deadline:
        n_up = best_n * 2
        if n_up <= 720:
            h_up = upsample(best_h, n_up)
            h_up = adjust_sum_exact(h_up, rng)
            # remaining time for the up‑sampled search
            sub_dead = deadline
            h_up_opt, c_up_opt = simulated_annealing(h_up, rng, sub_dead)
            if c_up_opt < best_c:
                best_h, best_c, best_n = h_up_opt.copy(), c_up_opt, n_up

    # --------------------------------------------------------------
    # Fallback – ensure a feasible output even if nothing succeeded
    # --------------------------------------------------------------
    if best_h is None:
        n_fallback = 360
        best_h = np.zeros(n_fallback)
        best_h[: n_fallback // 2] = 1.0
        best_h = adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n_fallback

    # final safety clipping and sum correction
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
