def simulated_annealing_binary(h_bool, T0, T_end, max_steps, rng):
    n = len(h_bool)
    pos = np.empty(n, dtype=int)
    ones = np.flatnonzero(h_bool)
    zeros = np.flatnonzero(~h_bool)
    pos[ones] = np.arange(len(ones))
    pos[zeros] = np.arange(len(zeros))
    # compute current C5
    h = h_bool.astype(float)
    cur_c = compute_c5_fft(h)
    best_c = cur_c
    best_h = h.copy()
    # Temperature schedule: exponential
    if max_steps == 0:
        return best_h, best_c
    for step in range(max_steps):
        T = T0 * (T_end / T0) ** (step / max_steps)  # exponentially cooling
        # propose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap in h_bool
        # perform swap in arrays and pos
        i_pos = pos[i]
        j_pos = pos[j]
        # swap indices in ones/zeros
        ones[i_pos] = j
        zeros[j_pos] = i
        # update bool
        h_bool[i] = False
        h_bool[j] = True
        # update pos
        pos[i] = j_pos
        pos[j] = i_pos
        # compute new C5
        h = h_bool.astype(float)
        new_c = compute_c5_fft(h)
        # accept?
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            # if new best
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            # swap back
            ones[i_pos] = i
            zeros[j_pos] = j
            h_bool[i] = True
            h_bool[j] = False
            pos[i] = i_pos
            pos[j] = j_pos
            # cur_c unchanged
        # maybe early exit if best_c <= target?
    return best_h, best_c

    cur_c = compute_c5_fft(h_float)
    best_c = cur_c
    n = h_float.shape[0]
    for step in range(max_steps):
        if time.time() > deadline:
            break
        # temperature schedule
        t = T0 * (T_end / T0) ** (step / max_steps)
        # pick i,j
        # make random choices using rng.integers over len(ones)
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        # update arrays (swap positions in ones and zeros)
        i_pos = pos[i]
        j_pos = pos[j]
        # swap indices in ones and zeros arrays
        ones[i_idx] = j
        zeros[j_idx] = i
        # update positions
        pos[i] = j_pos
        pos[j] = i_pos

        h_float[i] = 0.0
        h_float[j] = 1.0

            pos[j] = j_pos

            h_float[j] = 0.0

import time

# ---------- core evaluation ----------
def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)      # linear correlation (zero‑padded)
    # shifts 0..n‑1 are first n entries, negative shifts are the last n‑1 entries
    max_val = corr[:n].max()
    tail = corr[-(n - 1):].max()
    if tail > max_val:
        max_val = tail
    return float(max_val * (2.0 / n))

# ---------- utility ----------
def adjust_sum_exact(h, rng):
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(3000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- binary greedy improvement ----------
def binary_targeted_improve(h, rng, deadline):
    """Swap a 1 and a 0 that reduces the current worst overlap."""
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        neigh_one = neigh[ones]
        i = ones[np.argmin(neigh_one)]
        neigh_zero = neigh[zeros]
        j = zeros[np.argmax(neigh_zero)]
        delta = neigh[i] - neigh[j]
        if delta < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        # try a few random swaps as fallback
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

# ---------- simulated annealing for binary vectors ----------
def simulated_annealing_binary(h_float, ones, zeros, pos, T0, T_end, max_steps, rng, deadline):
    """SA swapping a 1 and a 0 while preserving the sum."""
    n = h_float.size
    cur_c = compute_c5(h_float)
    best_h = h_float.copy()
    best_c = cur_c
    # pre‑compute cooling factor to avoid repeated pow
    if max_steps > 0:
        log_ratio = np.log(T_end / T0)
    for step in range(max_steps):
        if time.time() > deadline:
            break
        # exponential cooling
        t = T0 * np.exp(log_ratio * (step / max_steps))
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # positions in the selector arrays
        i_pos = pos[i]
        j_pos = pos[j]
        # perform swap in selectors
        ones[i_idx] = j
        zeros[j_idx] = i
        pos[i] = j_pos
        pos[j] = i_pos
        # swap values
        h_float[i] = 0.0
        h_float[j] = 1.0
        new_c = compute_c5(h_float)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / t):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_float.copy()
        else:
            # revert swap
            ones[i_idx] = i
            zeros[j_idx] = j
            pos[i] = i_pos
            pos[j] = j_pos
            h_float[i] = 1.0
            h_float[j] = 0.0
    return best_h, best_c

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # base size from parent if provided
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        base_n = parent.size
    else:
        parent = None
        base_n = 360

    # set of n to try (all ≤ 720)
    n_vals = [base_n, 180, 240, 300, 480, 600, 720]
    n_vals = [n for n in n_vals if n <= 720]

    for n in n_vals:
        # number of random seeds per n (more for smaller n)
        seeds_per_n = 5 if n <= 360 else 3
        for _ in range(seeds_per_n):
            if time.time() > deadline:
                break

            # ----- initialise a binary vector -----
            if parent is not None and parent.size == n:
                h0 = parent.copy()
                # binarise if needed
                if not np.all(np.logical_or(h0 == 0.0, h0 == 1.0)):
                    idx = np.argpartition(h0, -n // 2)[-n // 2 :]
                    bin_h = np.zeros_like(h0)
                    bin_h[idx] = 1.0
                    h0 = bin_h
                adjust_sum_exact(h0, rng)
            else:
                h0 = np.zeros(n, dtype=float)
                h0[rng.choice(n, n // 2, replace=False)] = 1.0

            cur_c = compute_c5(h0)
            if cur_c < best_c:
                best_h, best_c, best_n = h0.copy(), cur_c, n

            # ----- simulated annealing -----
            h_float = h0.astype(np.float64)
            h_bool = (h_float > 0.5)
            ones = np.flatnonzero(h_bool)
            zeros = np.flatnonzero(~h_bool)
            pos = np.empty(n, dtype=int)
            pos[ones] = np.arange(len(ones))
            pos[zeros] = np.arange(len(zeros))

            # decide steps based on remaining time
            remaining = deadline - time.time()
            max_steps = 25000
            est = int(remaining * 3000)  # rough steps‑per‑second
            if est < max_steps:
                max_steps = max(1000, est)
            if max_steps <= 0:
                continue

            T0, T_end = 0.02, 0.0001
            sa_h, sa_c = simulated_annealing_binary(
                h_float, ones, zeros, pos, T0, T_end, max_steps, rng, deadline
            )
            adjust_sum_exact(sa_h, rng)
            cur_c = compute_c5(sa_h)
            if cur_c < best_c:
                best_h, best_c, best_n = sa_h.copy(), cur_c, n

            # ----- greedy binary improvement -----
            hb = binary_targeted_improve(sa_h, rng, deadline)
            c_b = compute_c5(hb)
            if c_b < best_c:
                best_h, best_c, best_n = hb.copy(), c_b, n

            # ----- gradient refinement (fractional) -----
            hg = gradient_refine(hb, rng, step=0.025, max_iter=30)
            adjust_sum_exact(hg, rng)
            c_g = compute_c5(hg)
            if c_g < best_c:
                best_h, best_c, best_n = hg.copy(), c_g, n

    # ----- optional up‑sampling refinement -----
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 720
        and (deadline - time.time()) > 20.0
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        idx = np.argpartition(h_up, -n_up // 2)[-n_up // 2 :]
        h_up_bin = np.zeros_like(h_up)
        h_up_bin[idx] = 1.0
        adjust_sum_exact(h_up_bin, rng)

        h_float = h_up_bin.astype(np.float64)
        h_bool = (h_float > 0.5)
        ones = np.flatnonzero(h_bool)
        zeros = np.flatnonzero(~h_bool)
        pos = np.empty(n_up, dtype=int)
        pos[ones] = np.arange(len(ones))
        pos[zeros] = np.arange(len(zeros))

        remaining = deadline - time.time()
        max_steps = 20000
        est = int(remaining * 3000)
        if est < max_steps:
            max_steps = max(1000, est)
        if max_steps > 0:
            sa_h, sa_c = simulated_annealing_binary(
                h_float, ones, zeros, pos, 0.02, 0.0001, max_steps, rng, deadline
            )
            adjust_sum_exact(sa_h, rng)
            c = compute_c5(sa_h)
            if c < best_c:
                best_h, best_c, best_n = sa_h.copy(), c, n_up

            hb = binary_targeted_improve(sa_h, rng, deadline)
            c_b = compute_c5(hb)
            if c_b < best_c:
                best_h, best_c, best_n = hb.copy(), c_b, n_up

            hg = gradient_refine(hb, rng, step=0.025, max_iter=30)
            adjust_sum_exact(hg, rng)
            c_g = compute_c5(hg)
            if c_g < best_c:
                best_h, best_c, best_n = hg.copy(), c_g, n_up

    # ----- final refinement on the best solution -----
    if best_h is not None and time.time() < deadline:
        n = best_n
        h_float = best_h.astype(np.float64)
        h_bool = (h_float > 0.5)
        ones = np.flatnonzero(h_bool)
        zeros = np.flatnonzero(~h_bool)
        pos = np.empty(n, dtype=int)
        pos[ones] = np.arange(len(ones))
        pos[zeros] = np.arange(len(zeros))
        remaining = deadline - time.time()
        max_steps = min(50000, int(remaining * 3000))
        if max_steps > 0:
            sa_h, _ = simulated_annealing_binary(
                h_float, ones, zeros, pos, 0.02, 0.0001, max_steps, rng, deadline
            )
            adjust_sum_exact(sa_h, rng)
            hb = binary_targeted_improve(sa_h, rng, deadline)
            hg = gradient_refine(hb, rng, step=0.025, max_iter=30)
            adjust_sum_exact(hg, rng)
            final_c = compute_c5(hg)
            if final_c < best_c:
                best_h, best_c, best_n = hg.copy(), final_c, n

    # ----- fallback -----
    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
