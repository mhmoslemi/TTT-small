import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# globals used by the smooth max optimiser
CURRENT_BETA = 1.0
CURRENT_LAMBDA = 0.0
CURRENT_N = 0
CURRENT_DX = 0.0


def compute_c5(h):
    """Maximum overlap for a step function sampled on a uniform grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Redistribute mass to satisfy Σh = n/2 exactly (maintaining bounds)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random walk redistribution
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(h, rng, step=0.03, max_iter=20):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:upper]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # enforce exact sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def fractional_sa_hill_climb(h, rng, time_deadline):
    """Transfer‑based simulated annealing on continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 12000:
            break
    return best, best_c


def binary_local_search(h_bin, rng, time_deadline, max_no_improve=5000):

def smooth_obj_grad(h):
    """Objective = smooth max of overlap + quadratic penalty on the sum."""
    global CURRENT_BETA, CURRENT_LAMBDA, CURRENT_N, CURRENT_DX
    n = CURRENT_N
    dx = CURRENT_DX
    # correlation without dx
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    c_raw = corr_raw * dx
    L = (1.0 / CURRENT_BETA) * logsumexp(CURRENT_BETA * c_raw)
    # normalized weights
    w = np.exp(CURRENT_BETA * (c_raw - L))
    # term_a and term_b via sliding dot products (O(n^2))
    one_minus_h = 1.0 - h
    w_full = w  # length 2n-1
    term_a = np.empty(n, dtype=np.float64)
    term_b = np.empty(n, dtype=np.float64)
    # slices are safe because 0 <= n-1-t <= n-1 and t+n-1 <= 2n-2
    for t in range(n):
        term_a[t] = np.dot(w_full[n - 1 - t : n - 1 - t + n], one_minus_h)
        term_b[t] = np.dot(w_full[t : t + n], h)
    grad = dx * (term_a - term_b)
    sum_err = h.sum() - n / 2.0
    grad += 2.0 * CURRENT_LAMBDA * sum_err
    # smooth max value + penalty
    obj = L + CURRENT_LAMBDA * sum_err * sum_err
    return obj, grad


def smooth_optimize(h, betas, rng, time_deadline):
    """Cascade of L‑BFGS‑B calls on the smoothed objective."""
    n = h.size
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() >= time_deadline:
            break
        global CURRENT_BETA, CURRENT_LAMBDA, CURRENT_N, CURRENT_DX
        CURRENT_BETA = beta
        CURRENT_LAMBDA = 1000.0
        CURRENT_N = n
        CURRENT_DX = 2.0 / n
        opts = {"maxiter": 250, "disp": False}
        try:
            res = minimize(smooth_obj_grad, h, method="L-BFGS-B", jac=True,
                           bounds=bounds, options=opts)
            h = np.clip(res.x, 0.0, 1.0)
            adjust_sum_exact(h, rng)
        except Exception:
            pass
    return h

        if target_n != n0:

            xt = np.linspace(0.0, 2.0, target_n, endpoint=False)
            h = np.interp(xt, xs, parent)
        else:
            h = parent.copy()
    else:
        target_n = 360
        h = np.zeros(target_n)
        h[: target_n // 2] = 1.0
    n = h.size
    adjust_sum_exact(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)
    best_n = n

    # ------------------------------------------------------------------
    # aggressive gradient refinement
    h = gradient_refine(h, rng, step=0.03, max_iter=30)
    c = compute_c5(h)
    if c < best_c:
        best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # fractional simulated annealing with remaining budget
    if time.time() < deadline:
        h, c = fractional_sa_hill_climb(h, rng, deadline)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # smooth optimisation cascade
    betas = [5, 20, 80, 200, 500, 2000, 5000, 10000]
    if time.time() < deadline:
        h = smooth_optimize(h, betas, rng, deadline)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # final gradient polishing
    if time.time() < deadline:
        h = gradient_refine(h, rng, step=0.025, max_iter=20)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ------------------------------------------------------------------
    # binary rounding and deterministic local search
    if time.time() < deadline:
        binary_h = np.zeros_like(h)
        idx = np.argpartition(h, -n // 2)[-n // 2 :]
        binary_h[idx] = 1.0
        adjust_sum_exact(binary_h, rng)
        c_bin = compute_c5(binary_h)
        if c_bin < best_c:
            best_h, best_c, best_n = binary_h.copy(), c_bin, n
        # local search (time‑aware)
        bin_dead = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        binary_h, c_bin = binary_local_search(binary_h, rng, bin_dead, max_no_improve=8000)
        if c_bin < best_c:
            best_h, best_c, best_n = binary_h.copy(), c_bin, n

    # ------------------------------------------------------------------
    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
