def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    # while diff not small:
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def line_search_hill_climb(h, rng, max_iters, time_limit):
    n = h.size
    best_c = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # random perturb maybe
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
        eps_max = min(h[i], 1 - h[j])
        if eps_max <= 0:
            continue
        # ternary search
        lo = 0.0
        hi = eps_max
        cur_best_c = best_c
        cur_best_eps = 0.0
        # compute number of ternary steps
        for _ in range(8):
            m1 = lo + (hi-lo)/3.0
            m2 = hi - (hi-lo)/3.0
            # evaluate at m1
            h_temp = h.copy()
            h_temp[i] -= m1
            h_temp[j] += m1
            c1 = compute_c5(h_temp)
            # evaluate at m2
            h_temp[i] = h[i] - m2
            h_temp[j] = h[j] + m2
            c2 = compute_c5(h_temp)
            if c1 < c2:
                hi = m2
                if c1 < cur_best_c:
                    cur_best_c, cur_best_eps = c1, m1
            else:
                lo = m1
                if c2 < cur_best_c:
                    cur_best_c, cur_best_eps = c2, m2
        # final check endpoints
        for eps in (lo, hi):
            h_temp = h.copy()
            h_temp[i] -= eps
            h_temp[j] += eps
            c = compute_c5(h_temp)
            if c < cur_best_c:
                cur_best_c, cur_best_eps = c, eps
        # if improvement
        if cur_best_c < best_c - 1e-12:
            # apply cur_best_eps
            h[i] -= cur_best_eps
            h[j] += cur_best_eps
            best_c = cur_best_c
            best_h = h.copy()
    return best_h, best_c

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

        best_h, best_c = h.copy(), c

    # ----- phase 3: soft‑max SLSQP passes -------------------------------
    phase3_deadline = start + total_budget * 0.90
    for beta in (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0):
        if time.time() > phase3_deadline:
            break
        h = slsqp_optimize(h, beta, rng, max_iter=400)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ----- phase 4: final random tweaks --------------------------------
    while time.time() - start < total_budget * 0.999:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            i = rng.integers(best_h.size)
            j = rng.integers(best_h.size)
            while j == i:
                j = rng.integers(best_h.size)
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.01
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c = nh.copy(), c

    # ----- sanitisation -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
