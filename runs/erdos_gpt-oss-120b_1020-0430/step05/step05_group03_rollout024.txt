def greedy_binary(n, rng):
    h = np.zeros(n, dtype=np.int8)
    ones = np.empty(0, dtype=np.int32)
    sum_prod = np.zeros(2*n-1, dtype=np.int16)
    half = n // 2
    for _ in range(half):
        zeros = np.where(h==0)[0]
        best_min = -1
        best_ps = []
        for p in zeros:
            if ones.size:
                contrib = np.bincount(n-1 - p + ones, minlength=2*n-1)
            else:
                contrib = np.zeros(2*n-1, dtype=np.int16)
            contrib[n-1] += 1
            new_min = np.min(sum_prod + contrib)

        sum_prod += contrib
        h[p] = 1

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1 - 2*h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

Guided_perturb:

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

        h[i] += add
        diff = target - h.sum()

def slsqp_optimize(h0, beta, max_iter=600):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())

def compute_c5(h: np.ndarray) -> float:

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size

    for _ in range(half):
        zeros = np.flatnonzero(h == 0)
        best_min = -1
        best_ps = []
        for p in zeros:
            if ones.size:
                contrib = np.bincount(n - 1 - p + ones, minlength=2 * n - 1)
                contrib = contrib.astype(np.int16)
            else:
                contrib = np.zeros(2 * n - 1, dtype=np.int16)
            contrib[n - 1] += 1
            new_min = np.min(sum_prod + contrib)
            if new_min > best_min:
                best_min, best_ps = new_min, [p]
            elif new_min == best_min:
                best_ps.append(p)
        p = rng.choice(best_ps)
        if ones.size:
            contrib = np.bincount(n - 1 - p + ones, minlength=2 * n - 1)
            contrib = contrib.astype(np.int16)
        else:
            contrib = np.zeros(2 * n - 1, dtype=np.int16)
        contrib[n - 1] += 1
        sum_prod += contrib
        h[p] = 1
        ones = np.append(ones, p)
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 600) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    best_h = None
    best_c = float("inf")

    if "initial_h_values" in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = enforce_sum(h_parent, rng)
        c_parent = compute_c5(h_parent)
        best_h, best_c = h_parent, c_parent

    max_restarts = 3
    for _ in range(max_restarts):
        if time.time() - start > 0.6 * budget_s:
            break
        h_bin = greedy_binary(n, rng)
        h = h_bin.astype(float)
        h, c = guided_hill_climb(h, rng, max_iters=2500, step_scale=0.7)
        for beta in (20., 80., 200., 800., 3000., 8000., 20000., 50000.):
            if time.time() - start > 0.9 * budget_s:
                break
            maxit = 800 if beta < 200 else 500
            h = slsqp_optimize(h, beta, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
        h = gradient_refine(h, rng, step=0.015, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    if best_h is None:
        h_bin = greedy_binary(n, rng)
        best_h = h_bin.astype(float)
        best_c = compute_c5(best_h)

    while time.time() - start < budget_s * 0.99:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
