def project_onto_box_sum(v, target):
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(60):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - low, 0.0, 1.0)
    # adjust small rounding
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # adjust first element
        idx = np.argmax(w) if diff > 0 else np.argmin(w)
        w[idx] = np.clip(w[idx] + diff, 0.0, 1.0)
    return w

CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_onto_box_sum(v, target):
    low, high = np.min(v) - 1.0, np.max(v)
    for _ in range(60):
        mid = (low + high) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - low, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(w) if diff > 0 else np.argmin(w)
        w[idx] = np.clip(w[idx] + diff, 0.0, 1.0)
    return w
