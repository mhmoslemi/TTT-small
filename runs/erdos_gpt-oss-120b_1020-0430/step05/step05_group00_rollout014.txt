def compute_c5(h):
    n = h.size
    rev_h = h[::-1]
    b = 1 - h
    L = 1 << ((2*n-1).bit_length())
    conv = np.fft.ifft(np.fft.fft(rev_h, L) * np.fft.fft(b, L)).real
    max_corr = conv[:2*n-1].max()
    return max_corr * (2.0 / n)
```

def hill_climb_binary(h, rng, max_no_improve=5000):
    n = h.size
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_h = h.copy()

    no_improve = 0

        j = rng.choice(zeros)

        h[i], h[j] = 0, 1

            no_improve = 0
        else:
            # revert
            h[i], h[j] = 1, 0
            no_improve += 1
    return best_h, best_c

    start = time.time()
    while time.time() - start < time_budget:

        h = np.zeros(n, dtype=float)

import time
import numpy as np

# ------------------------------------------------------------
def compute_c5_raw(h: np.ndarray) -> float:
    """Maximum of the linear correlation h * (1-h) (no dx factor)."""
    n = h.shape[0]
    rev_h = h[::-1]
    b = 1.0 - h
    # use FFT length 2*n (power of two for n up to 512)
    F1 = np.fft.fft(rev_h, n=2 * n)
    F2 = np.fft.fft(b, n=2 * n)
    conv = np.fft.ifft(F1 * F2).real
    return conv[: 2 * n - 1].max()


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 with proper scaling."""
    return compute_c5_raw(h) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 20) -> np.ndarray:
    """Projected gradient descent on the worst‑shift overlap."""
    n = h.size
    for _ in range(max_iter):
        grad = worst_gradient(h)
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        # exact sum correction (quick redistribution)
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            for __ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = n / 2.0 - h.sum()
        step *= 0.9
    return h

def sa_binary(n: int, rng: np.random.Generator, time_budget: float):
    """
    Simulated‑annealing on binary vectors with exactly n/2 ones.
    Returns best binary vector and its raw correlation max.
    """
    best_h = None
    best_raw = np.inf
    start = time.time()
    max_steps = 25000
    T0, T_end = 5.0, 1e-3

    while time.time() - start < time_budget:
        # --- random start ---
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        ones_list = list(ones)
        zeros_list = [i for i in range(n) if i not in ones]
        cur_raw = compute_c5_raw(h)

        # best of this run
        best_raw_run = cur_raw
        best_h_run = h.copy()

        for step in range(max_steps):
            if time.time() - start >= time_budget:
                break
            progress = step / max_steps
            T = T0 * (T_end / T0) ** progress

            # random swap
            pos_i = rng.integers(len(ones_list))
            pos_j = rng.integers(len(zeros_list))
            i = ones_list[pos_i]
            j = zeros_list[pos_j]

            # perform swap
            h[i] = 0.0
            h[j] = 1.0
            ones_list[pos_i] = j
            zeros_list[pos_j] = i

            new_raw = compute_c5_raw(h)
            delta = new_raw - cur_raw
            accept = False
            if delta <= 0.0:
                accept = True
            else:
                if rng.random() < np.exp(-delta / max(T, 1e-12)):
                    accept = True

            if accept:
                cur_raw = new_raw
                if cur_raw < best_raw_run:
                    best_raw_run = cur_raw
                    best_h_run = h.copy()
            else:
                # revert swap
                h[i] = 1.0
                h[j] = 0.0
                ones_list[pos_i] = i
                zeros_list[pos_j] = j

        if best_raw_run < best_raw:
            best_raw = best_raw_run
            best_h = best_h_run.copy()

    return best_h, best_raw

    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c), int(best_n)
