Define function move_h(h, grad, alpha):

  pos = grad > 0; neg = grad < 0

  total_out = np.sum(h[pos]); total_in = np.sum(1 - h[neg])

  if total_out == 0 or total_in == 0: return h (no move)

  eps = min(total_out, total_in) * alpha

  # scaling factor for each index
  if total_out > 0:
      h_new_pos = h[pos] * (1 - eps/total_out)
  else: h_new_pos = h[pos]

  if total_in > 0:
      h_new_neg = h[neg] + (1 - h[neg]) * (eps/total_in)
  else: h_new_neg = h[neg]

  h_new = h.copy()
  h_new[pos] = h_new_pos
  h_new[neg] = h_new_neg

  return h_new

import time
import numpy as np

# ----- global vars for SLSQP ----- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k with CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----- problem specific helpers ----- #
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        cap = (1.0 - h[i]) if diff > 0 else -h[i]
        if cap == 0.0:
            continue
        delta = min(abs(diff), abs(cap)) * np.sign(diff)
        h[i] += delta
        diff = target - h.sum()
    return h


def worst_shift_and_grad(h: np.ndarray):
    """Return (k, grad) for the shift attaining max overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return k, grad


def mass_transfer_move(h: np.ndarray, grad: np.ndarray, alpha: float,
                       target: float, rng: np.random.Generator) -> np.ndarray:
    """Move mass from positive‑gradient indices to negative‑gradient ones."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    if not pos.any() or not neg.any():
        return h.copy()
    total_out = np.sum(h[pos])
    total_in = np.sum(1.0 - h[neg])
    if total_out == 0.0 or total_in == 0.0:
        return h.copy()
    eps = min(total_out, total_in) * alpha
    h_new = h.copy()
    if total_out > 0.0:
        h_new[pos] = h[pos] * (1.0 - eps / total_out)
    if total_in > 0.0:
        h_new[neg] = h[neg] + (1.0 - h[neg]) * (eps / total_in)
    h_new = np.clip(h_new, 0.0, 1.0)
    # tiny sum correction
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new = enforce_sum(h_new, target, rng)
    return h_new


def random_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Small random transfer that preserves the total sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = max_eps * rng.random() * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2


def alpha_schedule(it: int, max_it: int):
    """Piecewise decreasing list of α values."""
    frac = it / max_it
    if frac < 0.3:
        return (1.0, 0.5, 0.25, 0.125, 0.0625)
    if frac < 0.6:
        return (0.5, 0.25, 0.125, 0.0625, 0.03125)
    if frac < 0.9:
        return (0.25, 0.125, 0.0625, 0.03125, 0.015625)
    return (0.125, 0.0625, 0.03125, 0.015625, 0.0078125)


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        _, grad = worst_shift_and_grad(h)
        gamma = step * (0.5 ** it)
        h = h - gamma * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = enforce_sum(h, target, rng)
    return h


def slsqp_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                 beta: float, max_iter: int) -> np.ndarray:
    """One SLSQP pass on the soft‑max approximation."""
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    from scipy.optimize import minimize  # local import to avoid unnecessary load
    n = h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x


# ----- main entry point ----- #
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target_sum = n / 2.0

    # ---- initialise h ---- #
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target_sum, rng)
    else:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    best_h = h.copy()
    best_c = compute_c5(best_h)

    max_iters = 50000  # generous but limited by time

    for it in range(max_iters):
        # time check
        if time.time() - start > budget_s * 0.99:
            break

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            # early exit if we already beat the target bound
            if best_c < 0.38079:
                break

        _, grad = worst_shift_and_grad(h)

        improved = False
        for alpha in alpha_schedule(it, max_iters):
            h_new = mass_transfer_move(h, grad, alpha, target_sum, rng)
            c_new = compute_c5(h_new)
            if c_new < c - 1e-12:
                h = h_new
                improved = True
                break

        if not improved:
            h2 = random_perturb(h, rng, scale=0.2)
            c2 = compute_c5(h2)
            if c2 < c - 1e-12:
                h = h2

    # ----- polish with gradient descent ----- #
    h = gradient_refine(best_h, target_sum, rng, step=0.015, max_iter=15)
    c = compute_c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # ----- final soft‑max SLSQP if time permits ----- #
    if time.time() - start < budget_s * 0.98:
        for beta in (500.0, 2000.0, 5000.0, 20000.0):
            if time.time() - start > budget_s * 0.99:
                break
            h_s = slsqp_refine(best_h, target_sum, rng, beta=beta, max_iter=400)
            c_s = compute_c5(h_s)
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()

    # final sanity enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
