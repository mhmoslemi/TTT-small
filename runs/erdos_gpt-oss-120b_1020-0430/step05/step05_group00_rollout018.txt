def compute_c5_fft(h):
    n = h.size
    L = 1 << ((2*n - 1).bit_length())  # next power of 2 >= 2n-1
    # could also use 2*n for speed.
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.real(np.fft.ifft(H * np.conj(G)))[:2*n-1]
    return corr.max() * (2.0 / n)

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using FFT (max linear correlation of h and 1‑h)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    return corr.max() * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while keeping 0≤h≤1."""
    target = h.size / 2.0
    diff = target - h.sum()
    it = 0
    max_it = 10000
    while abs(diff) > 1e-12 and it < max_it:
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    # tiny fallback – should rarely be needed
    if abs(diff) > 1e-9:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def random_h(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.5):
    """Hill‑climb using the worst‑shift gradient."""
    cur_h = h.copy()
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    for _ in range(max_iters):
        grad = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0.0:
            continue
        eps = max_transfer * rng.random() * step_scale
        # propose
        cur_h[i] -= eps
        cur_h[j] += eps
        new_c = compute_c5_fft(cur_h)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += eps
            cur_h[j] -= eps
    return best_h, best_c


def simulated_annealing_continuous(h_start: np.ndarray,
                                   rng: np.random.Generator,
                                   time_budget: float):
    """Continuous SA that preserves the exact sum (transfer between two sites)."""
    n = h_start.size
    h = h_start.copy()
    cur_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    # temperature schedule parameters
    T0 = 0.02
    T_end = 1e-5
    start = time.time()
    max_iters = int(time_budget * 5000) + 1  # generous upper bound
    for it in range(max_iters):
        if time.time() - start >= time_budget:
            break
        t = T0 * (1 - it / max_iters) + T_end * (it / max_iters)
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # random direction of transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= 0.0:
            continue
        eps = max_transfer * rng.random() * (t / T0) * 0.5
        if eps <= 0.0:
            continue
        # move
        h[src] -= eps
        h[dst] += eps
        new_c = compute_c5_fft(h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[src] += eps
            h[dst] -= eps
    return best_h, best_c

        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5_fft(h0)
        best_h, best_c = h0, c0

    # ----- simulated‑annealing phase (≈70 % of budget) -----
    time_for_sa = total_budget * 0.70
    sa_deadline = start_time + time_for_sa

    while time.time() < sa_deadline:
        # start from a promising point most of the time
        if best_h is not None and rng.random() < 0.6:
            h_start = best_h.copy()
            # small random disturbance to avoid exact same basin
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                max_transfer = min(h_start[i], 1.0 - h_start[j])
                if max_transfer > 0.0:
                    eps = max_transfer * rng.random() * 0.2
                    h_start[i] -= eps
                    h_start[j] += eps
        else:
            h_start = random_h(n, rng)

        remaining = sa_deadline - time.time()
        if remaining <= 0.0:
            break
        h_candidate, c_candidate = simulated_annealing_continuous(
            h_start, rng, remaining
        )
        if c_candidate < best_c:
            best_h, best_c = h_candidate, c_candidate

    # ----- gradient hill‑climb refinement (remaining budget) -----
    remaining = total_budget - (time.time() - start_time)
    if remaining > 0.0 and best_h is not None:
        max_iters = int(remaining * 2000)  # roughly 2000 iterations per second
        refined_h, refined_c = guided_hill_climb(
            best_h, rng, max_iters=max_iters, step_scale=0.5
        )
        if refined_c < best_c:
            best_h, best_c = refined_h, refined_c

    # ----- final sanitisation -----
    if best_h is None:
        best_h = random_h(n, rng)
        best_c = compute_c5_fft(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
