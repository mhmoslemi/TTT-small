def compute_c5(h: np.ndarray) -> float:
    """max_k ∫ h(x)(1‑h(x+k))dx for discrete step size."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Exact correction to enforce sum(h) = target."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

        while time.time() < time_limit:
            if not len(ones) or not len(zeros):
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            # perform swap
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            new_c = compute_c5(cur_h)
            if new_c < cur_c - 1e-12:
                cur_c = new_c
                maybe_update(cur_h, cur_c)
                # update index lists
                idx_i = np.where(ones == i)[0][0]
                ones[idx_i] = j
                idx_j = np.where(zeros == j)[0][0]
                zeros[idx_j] = i
            else:
                # revert swap
                cur_h[i] = 1.0
                cur_h[j] = 0.0
        return cur_h, cur_c

    # ------------------------------------------------------------------
    # helper: continuous local search preserving total sum
    def continuous_local_search(h_start, c_start, time_limit):
        cur_h = h_start.copy()
        cur_c = c_start
        step = 0.2
        n_local = cur_h.size
        while time.time() < time_limit and step > 1e-4:
            for _ in range(800):
                i, j = rng.choice(n_local, size=2, replace=False)
                delta = (rng.random() - 0.5) * 2.0 * step
                new_i = cur_h[i] + delta
                new_j = cur_h[j] - delta
                if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
                    continue
                cur_h[i] = new_i
                cur_h[j] = new_j
                new_c = compute_c5(cur_h)
                if new_c < cur_c - 1e-12:
                    cur_c = new_c
                    maybe_update(cur_h, cur_c)
                else:
                    cur_h[i] = new_i - delta
                    cur_h[j] = new_j + delta
            step *= 0.7
        # final exact sum correction (tiny drift possible)
        cur_h = adjust_sum_exact(cur_h, target_sum, rng)
        cur_c = compute_c5(cur_h)
        maybe_update(cur_h, cur_c)
        return cur_h, cur_c

    # ------------------------------------------------------------------
    # main stochastic search loop
    while time.time() < deadline:
        # --------------------------------------------------------------
        # generate a random binary vector with exact weight
        perm = rng.permutation(n)
        h_bin = np.zeros(n, dtype=float)
        h_bin[perm[:n // 2]] = 1.0
        c_bin = compute_c5(h_bin)
        maybe_update(h_bin, c_bin)

        # --------------------------------------------------------------
        # hill‑climbing on this seed (limited time)
        hc_dead = min(deadline, time.time() + 0.6)  # at most 0.6 s per seed
        h_hc, c_hc = hill_climb_swap(h_bin, c_bin, hc_dead)

        # --------------------------------------------------------------
        # continuous refinement if time permits
        if time.time() < deadline:
            cont_dead = min(deadline, time.time() + 0.4)
            h_cont, c_cont = continuous_local_search(h_hc, c_hc, cont_dead)

        # early stop if we already beat the target
        if best_c <= 0.3808:
            break

    # ------------------------------------------------------------------
    # final safety adjustments
    if best_h is None:
        # fallback to simple half‑interval
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_h = adjust_sum_exact(best_h, target_sum, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
