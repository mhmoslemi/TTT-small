def project_sum_box(x, target, tol=1e-9, max_iter=100):
    lo = -np.max(x)  # because x+lambda >=0 => lambda >= -x_i
    hi = 1.0 - np.min(x)  # x+lambda <=1 => lambda <= 1- x_i
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) < tol:
            return y
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(x + (lo+hi)/2.0,0.0,1.0)

def hill_climb(h, rng, max_iters, init_step=0.5):
    step = init_step
    best_h = h
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(best_h)
        nh = gradient_step(best_h, grad, step, rng)
        c = compute_c5(nh)
        if c < best_c - 1e-12:
            best_c = c
            best_h = nh
            # maybe keep step size same
        # periodic step decay
        if (it+1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def upscale(h, new_n):

def slsqp_optimize(h0, beta, max_iter=600):
    n = h0.size
    target = n/2
    bounds = [(0.0,1.0)] * n
    cons = {"type":"eq", "fun": lambda x: np.sum(x)-target}
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        # compute log-sum-exp / beta
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m))))/beta
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter":max_iter, "ftol":1e-12, "disp":False})
    x = np.clip(res.x,0,1)
    x = project_sum_box(x, target=target)
    return x

import time, numpy as np
from scipy.optimize import minimize

# -------------------- core utilities --------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with uniform sampling."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (maximal) shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # shift giving max C5
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]      -= h[: n - k]
    return grad


def project_sum_box(x: np.ndarray, target: float, tol: float = 1e-10,
                    max_iter: int = 80) -> np.ndarray:
    """Project x onto {0≤y≤1, Σy = target} by binary search on λ."""
    lo = -np.max(x)                # ensures x+λ≥0
    hi = 1.0 - np.min(x)           # ensures x+λ≤1
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    x = rng.random(n)
    return project_sum_box(x, target=n / 2.0)


# -------------------- local moves --------------------
def gradient_step(h: np.ndarray, grad: np.ndarray, step_scale: float,
                  rng: np.random.Generator) -> np.ndarray:
    """Transfer mass from high‑gradient entry to low‑gradient entry."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h
    delta_max = min(h[i], 1.0 - h[j])
    if delta_max <= 0.0:
        return h
    # try a few candidate transfers
    candidates = [step_scale * delta_max, delta_max * 0.5, delta_max * 0.1]
    best_h, best_c = h, compute_c5(h)
    tgt = h.size / 2.0
    for eps in candidates:
        if eps <= 0.0:
            continue
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = project_sum_box(nh, target=tgt)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh
    return best_h


def hill_climb(h: np.ndarray, rng: np.random.Generator,
               max_iters: int, init_step: float = 0.6) -> tuple:
    """Iterative gradient‑directed mass transfer."""
    step = init_step
    best_h, best_c = h, compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(best_h)
        nh = gradient_step(best_h, grad, step, rng)
        c = compute_c5(nh)
        if c < best_c - 1e-12:
            best_c, best_h = c, nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_sum_box(x, target=target)


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using diminishing steps along the worst‑gradient direction."""
    tgt = h.size / 2.0
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - step * (0.5 ** it) * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_box(h, target=tgt)
    return h


def upscale(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger sample size, followed by projection."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_sum_box(h_new, target=new_n / 2.0)


# -------------------- main driver --------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initial candidate
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_box(h, target=h.size / 2.0)
    else:
        h = random_feasible_vector(128, rng)

    best_h, best_c = h, compute_c5(h)

    # multi‑scale optimisation
    for n_target in (128, 256, 512):
        if time.time() - start > budget_s * 0.85:
            break
        if best_h.size != n_target:
            best_h = upscale(best_h, n_target)
        # aggressive hill‑climb
        iters = {128: 4000, 256: 2500, 512: 1500}[n_target]
        best_h, best_c = hill_climb(best_h, rng, max_iters=iters, init_step=0.6)

        # continuation of SLSQP passes with increasing β
        for beta in (20., 80., 200., 800., 3000., 8000., 20000., 50000.):
            if time.time() - start > budget_s * 0.95:
                break
            best_h = slsqp_optimize(best_h, beta,
                                    max_iter=800 if beta < 200 else 600)
            best_c = compute_c5(best_h)

        # final local tidy‑up
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=15)
        best_c = compute_c5(best_h)

    # stochastic fine‑tuning until time expires
    tgt = best_h.size / 2.0
    while time.time() - start < budget_s * 0.999:
        i, j = rng.integers(0, best_h.size, size=2)
        if i == j:
            continue
        delta_max = min(best_h[i], 1.0 - best_h[j])
        if delta_max <= 0.0:
            continue
        eps = rng.random() * delta_max * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = project_sum_box(nh, target=tgt)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # final sanity enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, target=best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
