def gradient_descent_sym(x0, n, rng, deadline):
    m = n // 2
    target_x = n / 4.0
    x = x0.copy()
    best_x = x.copy()
    best_c = compute_c5(sym_expand(x))
    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    step0 = 0.2
    for beta in betas:
        step = step0
        for it in range(10):
            if time.time() > deadline:
                return best_x, best_c
            h = sym_expand(x)
            # compute overlaps
            corr = np.correlate(h, 1.0 - h, mode='full')
            overlaps = corr * (2.0 / n)
            # compute softmax weights
            # scale to avoid overflow
            max_beta_overlap = np.max(beta * overlaps)
            w = np.exp(beta * overlaps - max_beta_overlap)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            # compute weighted sum of shifts
            weighted = np.zeros_like(h)
            # select top weights
            top_k = 80
            if top_k < w.size:
                idxs = np.argpartition(-w, top_k-1)[:top_k]
            else:
                idxs = np.arange(w.size)
            for idx in idxs:
                wk = w[idx]
                if wk < 1e-15:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    if n - k > 0:
                        weighted[:n - k] += wk * h[k:]
                        weighted[k:] += wk * h[:n - k]
                else:
                    k2 = -k
                    if n - k2 > 0:
                        weighted[k2:] += wk * h[:n - k2]
                        weighted[:n - k2] += wk * h[k2:]

            if c < best_c:
                best_c = c
                best_x = x.copy()

# ---------- helpers ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(arr, rng):
    """Force sum(arr) = len(arr)/2 exactly, staying in [0,1]."""
    n = arr.size
    target = n / 2.0
    diff = target - arr.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - arr[i])
        else:
            add = max(diff, -arr[i])
        arr[i] += add
        diff = target - arr.sum()
    return arr

def sym_expand(x):
    """Mirror x to obtain full even‑length vector."""
    return np.concatenate([x, x[::-1]])

def gradient_refine_full(h, rng, step=0.02, max_iter=15):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        np.clip(h, 0.0, 1.0, out=h)

                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

        step = step0
        for it in range(10):
            if time.time() > deadline:
                return best_x, best_c
            h = sym_expand(x)

            # overlaps (scaled)
            corr = np.correlate(h, 1.0 - h, mode="full")

            # softmax weights

            w = np.exp(beta * overlaps - max_beta)
            w_sum = w.sum()
            if w_sum == 0:

    return best_h.tolist(), float(final_c), int(best_n)
