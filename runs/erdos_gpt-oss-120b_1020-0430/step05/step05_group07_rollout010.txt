def subgradient_descent(h0, max_iter, step0, decay, target, deadline):
    h = h0.copy()
    n = h.size

        idx = np.argmax(corr)
        k = idx - (n - 1)
        if k == 0:

        else:
            grad = 1.0 - (np.roll(h, k) + np.roll(h, -k))
        step = step0 * (decay ** it)
        h = h - step * grad
        # projection
        h = project_to_sum_bounds(h, target)
    return h

import time
import numpy as np

# ------- Global placeholders (set inside run) -------
N_GLOBAL = None
L_GLOBAL = None

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT.
    Returns array of length 2*N_GLOBAL-1, lag index = idx-(N_GLOBAL-1)."""
    a = np.fft.rfft(h, L_GLOBAL)
    b = np.fft.rfft(1.0 - h, L_GLOBAL)
    c = np.fft.irfft(a * np.conj(b), L_GLOBAL)
    return c[: 2 * N_GLOBAL - 1]

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap C5 for current global discretisation."""
    corr = _compute_corr_fft(h)
    return float(corr.max() * (2.0 / N_GLOBAL))

def adjust_sum_exact(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass to hit the exact sum while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    # a few hundred attempts are enough in practice
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def subgradient_descent(h0: np.ndarray, max_iter: int, step0: float,
                        decay: float, target: float,
                        rng: np.random.Generator, deadline: float) -> np.ndarray:
    """Descent on the true max objective (piecewise linear)."""
    h = h0.copy()
    n = h.shape[0]
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        corr = _compute_corr_fft(h)
        idx = np.argmax(corr)
        k = idx - (n - 1)          # lag corresponding to max overlap
        grad = np.zeros_like(h)
        if k >= 0:
            if k < n:
                grad[: n - k] += 1.0 - h[k:]
                grad[k:]      -= h[: n - k]
        else:
            kk = -k
            if kk < n:
                grad[kk:]     += 1.0 - h[: n - kk]
                grad[: n - kk] -= h[kk:]
        step = step0 * (decay ** it)
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum_exact(h, target, rng)
    return h

def smooth_max_refine(h: np.ndarray, betas, step0: float,
                      target: float, rng: np.random.Generator,
                      deadline: float) -> np.ndarray:
    """Gradient descent on a log‑sum‑exp smoothing of the max."""
    n = h.shape[0]
    offset = n - 1
    for beta in betas:
        for _ in range(200):        # fixed inner iterations
            if time.time() >= deadline:
                break
            corr = _compute_corr_fft(h)          # length 2n‑1
            bc = beta * corr
            m = bc.max()
            w = np.exp(bc - m)
            w /= w.sum()                         # weights over all lags
            w_rev = w[::-1]

            ws = np.fft.rfft(w, L_GLOBAL)
            ws_rev = np.fft.rfft(w_rev, L_GLOBAL)
            hr = np.fft.rfft(h, L_GLOBAL)

            conv_minus = np.fft.irfft(ws * hr, L_GLOBAL)[: 2 * n - 1]
            conv_plus  = np.fft.irfft(ws_rev * hr, L_GLOBAL)[: 2 * n - 1]

            conv_sum = conv_minus[offset:offset + n] + conv_plus[offset:offset + n]
            grad = 1.0 - conv_sum                 # ∇ of smoothed objective
            h_new = h - step0 * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h = adjust_sum_exact(h_new, target, rng)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # Primary discretisation size (power of two for fast FFT)
    primary_n = 1024
    global N_GLOBAL, L_GLOBAL
    N_GLOBAL = primary_n
    L_GLOBAL = _next_pow2(2 * primary_n - 1)
    target_sum = primary_n / 2.0

    # Gather a few deterministic starts
    start_vecs = []

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, primary_n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        interp = adjust_sum_exact(interp, target_sum, rng)
        start_vecs.append(interp)

    # Simple half‑ones binary vector
    bin_half = np.zeros(primary_n)
    bin_half[: primary_n // 2] = 1.0

        h_up = adjust_sum_exact(h_up, target_sum, rng)

                                   decay=0.996, target=target_sum,
                                   rng=rng, deadline=deadline)

                                 step0=0.06, target=target_sum,
                                 rng=rng, deadline=deadline)

        if c_up < best_c:

            best_h = h_up.copy()

    L_GLOBAL = _next_pow2(2 * best_n - 1)

    best_h = adjust_sum_exact(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
