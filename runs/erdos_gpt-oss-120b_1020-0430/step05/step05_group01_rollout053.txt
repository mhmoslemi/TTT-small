def worst_gradient(h):

        grad[: n - k] -= h[k:]

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def proj_sum_box(y: np.ndarray, target: float) -> np.ndarray:
    """Project y onto the box [0,1]^n while forcing sum(y)=target."""
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        clipped = np.clip(y - mid, 0.0, 1.0)
        s = clipped.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1.0 - 1e-12)[0]
        else:
            cand = np.where(h > 1e-12)[0]
        if cand.size:
            h[cand[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def softmax_weights(O: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max probabilities for overlap values O."""
    bc = beta * O
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    return exp_bc / exp_bc.sum()


def grad_softmax(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max objective w.r.t. h.
    w are the soft‑max weights (sum to 1, length n).
    """
    n = h.size
    # term_const = sum_{k <= n-1-i} w_k
    term_const = np.cumsum(w)[::-1]                # shape (n,)
    # term_A = sum_{k <= n-1-i} w_k * h_{i+k}
    conv_rev_h = np.convolve(w[::-1], h)           # length 2n-1
    term_A = conv_rev_h[n - 1:]                    # tail of length n
    # term_B = sum_{k <= i} w_k * h_{i-k}
    conv_wh = np.convolve(w, h)                    # length 2n-1
    term_B = conv_wh[:n]                           # first n entries
    return term_const - term_A - term_B


def pgd_optimize(
    h0: np.ndarray,
    target: float,
    beta_seq,
    steps_per_beta: int,
    rng: np.random.Generator,
) -> np.ndarray:
    """Projected gradient descent on the soft‑max approximation."""
    h = h0.copy()
    n = h.size
    for beta in beta_seq:
        # step size shrinks with beta
        step = 0.2 / (beta ** 0.25) if beta > 0 else 0.2
        for _ in range(steps_per_beta):
            O = np.correlate(h, 1.0 - h, mode="full")[n - 1 :]  # O_k for k>=0
            w = softmax_weights(O, beta)
            grad = grad_softmax(h, w)
            h = proj_sum_box(h - step * grad, target)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    step_scale: float = 0.6,
):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

        h0 = np.clip(h0, 0.0, 1.0)
        h0 = proj_sum_box(h0, h0.size / 2.0)
        best_h, best_c5, best_n = h0.copy(), compute_c5(h0), h0.size

    beta_seq = (1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0)

    # coarse‑to‑fine resolutions
    for n in (128, 256, 512):
        if time_left() < 5.0:
            break
        target = n / 2.0
        steps_per_beta = max(30, int(200 * 256 / n))

        max_restarts = 4
        for _ in range(max_restarts):
            if time_left() < 5.0:
                break

            # initialise
            if best_h is not None:
                if best_n == n:
                    h = best_h.copy()
                else:
                    x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                    h = np.interp(x_new, x_old, best_h)
                    h = np.clip(h, 0.0, 1.0)
                    h = proj_sum_box(h, target)
            else:
                h = proj_sum_box(rng.random(n), target)

            # PGD on soft‑max
            h = pgd_optimize(h, target, beta_seq, steps_per_beta, rng)
            c = compute_c5(h)
            if c < best_c5:
                best_h, best_c5, best_n = h.copy(), c, n

            # final hill‑climb refinement
            hc, hc_val = guided_hill_climb(
                h, rng, max_iters=int(2000 * 256 / n), step_scale=0.6
            )
            if hc_val < best_c5:
                best_h, best_c5, best_n = hc.copy(), hc_val, n

    # up‑sample to full 512 if needed
    if best_h is not None and best_n < 512 and time_left() > 15.0:
        n = 512
        target = n / 2.0
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, best_h)
        h = np.clip(h, 0.0, 1.0)
        h = proj_sum_box(h, target)
        steps_per_beta = max(30, int(200 * 256 / n))
        h = pgd_optimize(h, target, beta_seq, steps_per_beta, rng)
        c = compute_c5(h)
        if c < best_c5:
            best_h, best_c5, best_n = h.copy(), c, n

        hc, hc_val = guided_hill_climb(
            h, rng, max_iters=4000, step_scale=0.6
        )
        if hc_val < best_c5:
            best_h, best_c5, best_n = hc.copy(), hc_val, n

    # stochastic tweaking until time expires
    while time_left() > 0.5 and best_h is not None:
        g = worst_gradient(best_h)
        nh = guided_perturb(best_h, g, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c5:
            best_h, best_c5 = nh.copy(), nc

    # final sanity check & exact sum enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_sum_box(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
