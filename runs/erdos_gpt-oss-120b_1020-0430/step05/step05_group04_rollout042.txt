def local_optimize_binary(h, target_M=None):
    n = h.size
    dx = 2.0 / n
    while True:
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlap = corr[n-1:] * dx  # k>=0
        k_max = np.argmax(overlap)
        M_max = overlap[k_max]
        if target_M is not None and M_max <= target_M:
            break
        # find mismatched 1s and 0s for this shift
        if k_max == 0:
            # no overlap; break
            break
        i_max = n - k_max
        # positions where h[i]==1 and h[i+k]==0
        S1 = np.flatnonzero((h[:i_max] == 1) & (h[k_max:] == 0))
        # positions where h[i]==0 and h[i+k]==1
        S0 = np.flatnonzero((h[:i_max] == 0) & (h[k_max:] == 1))

        best_M = M_max

                h_new[j] = 1

    n = len(h)
    const = n // 2
    h = h.copy()
    dot = np.correlate(h, h, mode='full')[n-1:]
    # Ensure dot is integer
    while True:
        # find shift with minimal dot (excluding index 0)
        k_min = np.argmin(dot[1:]) + 1
        min_dot = dot[k_min]
        current_M = const - min_dot
        # Build S1, S0 for this shift
        i_max = n - k_min
        # For S1: positions i in 0..i_max-1 where h[i]=1 and h[i+k]=0
        mask_S1 = (h[:i_max] == 1) & (h[k_min:] == 0)
        S1 = np.flatnonzero(mask_S1)
        mask_S0 = (h[:i_max] == 0) & (h[k_min:] == 1)
        S0 = np.flatnonzero(mask_S0)
        if S1.size == 0 or S0.size == 0:
            break
        # Enumerate all pairs (i, j) to find best reduction
        best_M = current_M
        best_pair = None
        # Precompute arrays for speed? We'll just loop.
        for i in S1:
            for j in S0:
                # compute delta for k>0
                delta = np.zeros(n, dtype=int)
                # i contributions
                if n - i > 1:
                    delta[1:n-i] -= h[i+1:]
                if i > 0:
                    delta[1:i+1] -= h[:i]
                # j contributions
                if n - j > 1:
                    delta[1:n-j] += h[j+1:]

        delta = np.zeros(n, dtype=int)

def improve_binary(h_start, time_deadline):
    h = h_start.copy()
    n = h.size
    const = n // 2
    dot = np.correlate(h, h, mode='full')[n-1:]  # int
    while True:
        if time.time() > time_deadline:
            break
        # identify worst shift
        # ignore k=0
        k_min = np.argmin(dot[1:]) + 1
        min_dot = dot[k_min]
        cur_M = const - min_dot
        # compute mismatched sets for this shift
        i_max = n - k_min
        mask_S1 = (h[:i_max] == 1) & (h[k_min:] == 0)
        mask_S0 = (h[:i_max] == 0) & (h[k_min:] == 1)
        S1 = np.flatnonzero(mask_S1)
        S0 = np.flatnonzero(mask_S0)
        if S1.size == 0 or S0.size == 0:
            break
        best_M = cur_M
        best_pair = None
        # Enumerate all pairs
        for i in S1:
            for j in S0:
                ndot = dot.copy()
                # apply modifications
                if n - i > 1:
                    ndot[1:n-i] -= h[i+1:]
                if i > 0:
                    ndot[1:i+1] -= h[:i]
                if n - j > 1:
                    ndot[1:n-j] += h[j+1:]
                if j > 0:
                    ndot[1:j+1] += h[:j]
                new_min = ndot[1:].min()
                new_M = const - new_min
                if new_M < best_M - 1e-12:
                    best_M = new_M
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        # perform swap
        h[i] = 0
        h[j] = 1
        # recompute dot
        dot = np.correlate(h, h, mode='full')[n-1:]
    final_M = const - dot[1:].min()
    final_c = final_M * (2.0 / n)
    return h, final_c

import time, numpy as np
from scipy.optimize import minimize

# ---------- global placeholders for constraints ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- utilities ----------
def compute_c5(h):
    """Exact C5 for a discrete step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth‑max (log‑sum‑exp) surrogate of max overlap."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stabilized log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    h = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny numerical drift)
    diff = CURRENT_TARGET - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = CURRENT_TARGET - h.sum()
    return h

def gradient_refine(h, rng, step=0.05, max_iter=60):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = np.argmax(corr) - (n - 1)            # signed shift with worst overlap
        k = abs(k)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]            # ∂/∂h_i for i < upper
            g[k:]     -= h[:upper]               # ∂/∂h_{i+k}
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        # re‑balance the sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
