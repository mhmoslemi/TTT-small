    for it in range(max_iter):
        # compute worst shift
        h_full = full_from_half(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        idx = np.argmax(corr) - (len(h_full) - 1)
        k = abs(idx)
        n = len(h_full)
        M = len(h_half)
        # compute arrays a,b,c,d
        # indices for a: (i + k) % n, for b: (i+M + k) % n, for c: (i - k) % n, for d: (i+M - k) % n
        # We can compute via vectorized indices
        i = np.arange(M)

        b_idx = (i + M + k) % n
        c_idx = (i - k) % n
        d_idx = (i + M - k) % n
        a = 1.0 - h_full[a_idx]
        b = 1.0 - h_full[b_idx]

import time
import numpy as np
from scipy.optimize import minimize

def full_from_half(h_half: np.ndarray) -> np.ndarray:
    """Construct the full period-2 function from the half representation."""
    return np.concatenate([h_half, 1.0 - h_half])

def compute_c5_full(h_full: np.ndarray) -> float:
    """Exact C5 for a given full-array h (period 2, N points)."""
    n = h_full.size
    corr = np.correlate(h_full, 1.0 - h_full, mode="full")
    return float(np.max(corr) * (2.0 / n))

def softmax_obj_half(h_half: np.ndarray, beta: float) -> float:
    """Smooth max via log-sum-exp for the half-variable representation."""
    h_full = full_from_half(h_half)
    corr = np.correlate(h_full, 1.0 - h_full, mode="full")
    bc = beta * corr
    m = np.max(bc)
    # stable log-sum-exp
    return float((m + np.log(np.sum(np.exp(bc - m)))) / beta)

def slsqp_optimize_half(h0: np.ndarray, beta: float, max_iter: int = 800) -> np.ndarray:
    """SLSQP optimization of the smooth-max surrogate on half variables."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return softmax_obj_half(x, beta)

    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    return np.clip(res.x, 0.0, 1.0)

def gradient_refine_half(h_half: np.ndarray, step: float = 0.1, max_iter: int = 30) -> np.ndarray:
    """Projected gradient descent on the worst shift (subgradient)."""
    for it in range(max_iter):
        h_full = full_from_half(h_half)
        n = h_full.size
        M = h_half.size
        corr = np.correlate(h_full, 1.0 - h_full, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        i = np.arange(M)
        a = 1.0 - h_full[(i + k) % n]
        b = 1.0 - h_full[(i + M + k) % n]
        c = h_full[(i - k) % n]
        d = h_full[(i + M - k) % n]

        delta = (a + d) - (b + c)  # ∂I_k/∂h_i
        step_sz = step * (0.5 ** it)
        h_half = h_half - step_sz * delta
        h_half = np.clip(h_half, 0.0, 1.0)
    return h_half

def binary_from_half(h_half: np.ndarray) -> np.ndarray:
    """Make a binary half-vector with exactly half the entries equal to 1."""
    M = h_half.size
    k = M // 2
    idx = np.argpartition(h_half, -k)[-k:]
    bin_h = np.zeros(M, dtype=float)
    bin_h[idx] = 1.0
    return bin_h

def binary_targeted_improve_half(h_half: np.ndarray, rng: np.random.Generator, deadline: float) -> np.ndarray:
    """Greedy swap improvement for binary half-vectors."""
    while time.time() < deadline:
        h_full = full_from_half(h_half)
        n = h_full.size
        M = h_half.size
        corr = np.correlate(h_full, 1.0 - h_full, mode="full")
        idx_max = np.argmax(corr) - (n - 1)
        s = abs(idx_max)
        if s == 0:
            break

        i = np.arange(M)
        a = 1.0 - h_full[(i + s) % n]
        b = 1.0 - h_full[(i + M + s) % n]
        c = h_full[(i - s) % n]
        d = h_full[(i + M - s) % n]

        contrib_one = a + d   # if h_i = 1
        contrib_zero = b + c  # if h_i = 0
        delta = contrib_one - contrib_zero

        zeros = np.where(h_half == 0.0)[0]
        ones = np.where(h_half == 1.0)[0]
        if zeros.size == 0 or ones.size == 0:
            break

        # candidates that actually improve the worst shift
        neg_zeros = zeros[delta[zeros] < 0]
        pos_ones = ones[delta[ones] > 0]

        if neg_zeros.size == 0 or pos_ones.size == 0:
            # try a few random swaps as last resort
            improved = False
            for _ in range(8):
                i_r = rng.choice(zeros)
                j_r = rng.choice(ones)
                if delta[i_r] - delta[j_r] < -1e-12:
                    h_half[i_r] = 1.0
                    h_half[j_r] = 0.0
                    improved = True
                    break
            if not improved:
                break
            continue

        i_best = neg_zeros[np.argmin(delta[neg_zeros])]
        j_best = pos_ones[np.argmax(delta[pos_ones])]
        if delta[i_best] - delta[j_best] < -1e-12:
            h_half[i_best] = 1.0
            h_half[j_best] = 0.0
            continue
        # fallback random attempts
        improved = False
        for _ in range(8):
            i_r = rng.choice(zeros)
            j_r = rng.choice(ones)
            if delta[i_r] - delta[j_r] < -1e-12:
                h_half[i_r] = 1.0
                h_half[j_r] = 0.0
                improved = True
                break
        if not improved:
            break
    return h_half

def optimize_half(h_start: np.ndarray, rng: np.random.Generator, deadline: float):
    """Full local search pipeline for a half-vector."""
    best_h = h_start.copy()
    best_c = compute_c5_full(full_from_half(best_h))

    betas = [5.0, 20.0, 100.0, 500.0, 2000.0, 10000.0]
    for beta in betas:
        if time.time() >= deadline:
            break
        h_mid = slsqp_optimize_half(best_h, beta, max_iter=800)
        c_mid = compute_c5_full(full_from_half(h_mid))
        if c_mid < best_c:
            best_h, best_c = h_mid, c_mid

    if time.time() < deadline:
        h_mid = gradient_refine_half(best_h, step=0.08, max_iter=20)
        c_mid = compute_c5_full(full_from_half(h_mid))
        if c_mid < best_c:
            best_h, best_c = h_mid, c_mid

    if time.time() < deadline:
        h_bin = binary_from_half(best_h)
        h_bin = binary_targeted_improve_half(h_bin, rng, deadline)
        c_bin = compute_c5_full(full_from_half(h_bin))
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin

    return best_h, best_c

def upsample_half(h_half: np.ndarray, factor: int = 2) -> np.ndarray:
    """Linear interpolation up‑sampling of the half‑vector."""
    M = h_half.size
    M_up = M * factor
    x_old = np.linspace(0.0, 1.0, M, endpoint=False)
    x_new = np.linspace(0.0, 1.0, M_up, endpoint=False)
    h_up = np.interp(x_new, x_old, h_half)
    return np.clip(h_up, 0.0, 1.0)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # Determine base size (must be even)
    if "initial_h_values" in globals():
        init_full = np.array(initial_h_values, dtype=float)
        n = init_full.size
        if n % 2 == 1:
            n -= 1
            init_full = init_full[:n]
    else:
        n = 360
        init_full = None

    if n % 2 == 1:
        n += 1
    M = n // 2

    # Build half‑vector starts
    starts = []
    if init_full is not None:
        h_half_parent = init_full[:M].copy()
        starts.append(h_half_parent)
        starts.append(binary_from_half(h_half_parent))

    # random binary halves
    for _ in range(3):
        hb = np.zeros(M)
        hb[rng.choice(M, M // 2, replace=False)] = 1.0
        starts.append(hb)

    # random fractional halves
    for _ in range(3):
        hf = rng.random(M)
        starts.append(hf)

    # heavy‑tailed fractional halves
    for _ in range(2):
        hf = rng.beta(0.5, 0.5, size=M)
        starts.append(hf)

    best_half = None
    best_c = np.inf

    # Main optimisation over all starts
    for h0 in starts:
        if time.time() >= deadline:
            break
        h0 = np.clip(h0, 0.0, 1.0)
        h_opt, c_opt = optimize_half(h0, rng, deadline)
        if c_opt < best_c:
            best_c, best_half = c_opt, h_opt.copy()

    # Compare with the raw initial full solution (if any)
    if init_full is not None:
        c_init = compute_c5_full(init_full)
        if c_init < best_c:
            best_c = c_init
            best_half = init_full[:M].copy()

    # Optional up‑sampling refinement
    if best_half is not None and time.time() < deadline:
        h_up = upsample_half(best_half, factor=2)
        h_opt_up, c_opt_up = optimize_half(h_up, rng, deadline)
        if c_opt_up < best_c:
            best_c, best_half = c_opt_up, h_opt_up.copy()

    # Fallback if everything failed
    if best_half is None:
        best_half = np.full(M, 0.5)
        best_c = compute_c5_full(full_from_half(best_half))

    best_full = full_from_half(best_half)
    best_full = np.clip(best_full, 0.0, 1.0)
    final_c = compute_c5_full(best_full)

    return best_full.tolist(), float(final_c), int(n)
