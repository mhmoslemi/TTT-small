def compute_corr_full(h):
    n = h.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    # compute via FFT
    H = np.fft.fft(h, m)
    G = np.fft.fft((1 - h)[::-1], m)
    conv = np.fft.ifft(H * G).real
    corr = conv[: 2*n - 1]
    # due to floating errors, round small negatives to zero
    # but keep as float
    return corr

def anneal(h_start, time_limit, rng):
    n = len(h_start)
    corr = compute_corr_full(h_start)
    cur_max = corr.max()
    h = h_start.copy()
    ones = [i for i, val in enumerate(h) if val ==1]
    zeros = [i for i, val in enumerate(h) if val ==0]
    pos_one = {i: idx for idx, i in enumerate(ones)}
    pos_zero = {i: idx for idx, i in enumerate(zeros)}
    best_h = h.copy()
    best_c = cur_max * (2.0 / n)
    start = time.time()
    iter = 0
    # set parameters
    T0 = 1.0
    T_end = 0.001
    max_iter = 1000000  # large
    while True:
        if time.time() - start > time_limit:
            break
        if iter >= max_iter:
            break
        # compute temperature
        frac = (time.time() - start) / time_limit
        T = T0 * (T_end / T0) ** frac
        # pick i, j
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # compute new corr via copy
        new_corr = corr.copy()
        # apply delta via slice modifications
        # i_f_removal
        new_corr[n-1 : 2*n-1-i] -= (1 - h[i:])
        # i_s_add
        new_corr[n-1-i : n] += h[i::-1]
        # j_f_add
        new_corr[n-1 : 2*n-1-j] += (1 - h[j:])
        # j_s_removal
        new_corr[n-1-j : n] -= h[j::-1]
        new_max = new_corr.max()
        # acceptance
        if new_max < cur_max or rng.random() < math.exp(-(new_max - cur_max) / T):
            # accept
            # update h and lists
            h[i] = 0.0
            h[j] = 1.0
            # update ones list
            pos_i = pos_one[i]
            ones[pos_i] = j
            pos_one.pop(i)
            pos_one[j] = pos_i
            # update zeros list
            pos_j = pos_zero[j]
            zeros[pos_j] = i
            pos_zero.pop(j)
            pos_zero[i] = pos_j
            # update corr and cur_max
            corr = new_corr
            cur_max = new_max
            # record best
            if cur_max * (2.0 / n) < best_c:
                best_c = cur_max * (2.0 / n)
                best_h = h.copy()
        iter += 1
    return best_h, best_c

import time
import math
import numpy as np

# ----- correlation utilities -------------------------------------------------
def _full_corr_fft(h: np.ndarray) -> np.ndarray:
    """Return full correlation of h with (1-h) using zero padding (length 2*len-1)."""
    n = h.size
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.fft(h, m)
    G = np.fft.fft((1.0 - h)[::-1], m)
    conv = np.fft.ifft(H * G).real
    return conv[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    """C5 for a binary/real step function on [0,2] (dx = 2/n)."""
    corr = _full_corr_fft(h)
    return float(corr.max() * (2.0 / h.size))


# ----- greedy improvement for the worst shift --------------------------------
def greedy_improve(h: np.ndarray, rng: np.random.Generator, deadline: float):
    """Perform a few targeted swaps that lower the current worst overlap."""
    n = h.size
    for _ in range(20):
        if time.time() > deadline:
            break
        corr = _full_corr_fft(h)
        idx_max = int(corr.argmax())
        shift = idx_max - (n - 1)
        if shift == 0:
            break
        k = abs(shift)
        ar = np.arange(n)
        neigh = h[(ar + k) % n] + h[(ar - k) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(neigh[ones])]
        j = zeros[np.argmax(neigh[zeros])]
        new_corr = corr.copy()
        # i : 1 -> 0
        new_corr[n - 1 : 2 * n - 1 - i] -= (1.0 - h[i:])
        new_corr[n - 1 - i : n] += h[i::-1]
        # j : 0 -> 1
        new_corr[n - 1 : 2 * n - 1 - j] += (1.0 - h[j:])
        new_corr[n - 1 - j : n] -= h[j::-1]
        if new_corr.max() < corr.max():
            h[i] = 0.0
            h[j] = 1.0
        else:
            break
    return h, compute_c5(h)


# ----- simulated annealing on binary vectors ---------------------------------
def _anneal_binary(
    h_start: np.ndarray, time_limit: float, rng: np.random.Generator
):
    """Binary simulated annealing using exact delta updates."""
    n = h_start.size
    h = h_start.copy()
    corr = _full_corr_fft(h)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_c = cur_max * (2.0 / n)

    # maintain index lists and position maps for O(1) swaps
    ones = [i for i, v in enumerate(h) if v == 1.0]
    zeros = [i for i, v in enumerate(h) if v == 0.0]
    pos_one = {i: idx for idx, i in enumerate(ones)}
    pos_zero = {i: idx for idx, i in enumerate(zeros)}

    start = time.time()
    T0, T_end = 1.0, 0.001

    while time.time() < start + time_limit:
        # temperature schedule (exponential decay)
        frac = (time.time() - start) / time_limit
        T = T0 * (T_end / T0) ** frac

        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        # build new correlation by applying delta
        new_corr = corr.copy()
        # i : 1 -> 0
        new_corr[n - 1 : 2 * n - 1 - i] -= (1.0 - h[i:])
        new_corr[n - 1 - i : n] += h[i::-1]
        # j : 0 -> 1
        new_corr[n - 1 : 2 * n - 1 - j] += (1.0 - h[j:])
        new_corr[n - 1 - j : n] -= h[j::-1]

        new_max = float(new_corr.max())

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # update index structures
            pi = pos_one[i]
            pj = pos_zero[j]

            ones[pi] = j
            pos_one.pop(i)
            pos_one[j] = pi

            zeros[pj] = i
            pos_zero.pop(j)
            pos_zero[i] = pj

            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_max = new_max

            c_val = cur_max * (2.0 / n)
            if c_val < best_c:
                best_c = c_val
                best_h = h.copy()
    return best_h, best_c


# ----- main entry point -------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # ----- Build an initial binary vector (length 720 if possible) -------------
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
        # keep the top half entries as ones
        idx = np.argpartition(parent, -n0 // 2)[-n0 // 2 :]
        bin_parent = np.zeros_like(parent, dtype=float)
        bin_parent[idx] = 1.0
        # up‑sample to 720 if the size divides evenly; otherwise fall back to 720 random
        if 720 % n0 == 0:
            repeats = 720 // n0
            h0 = np.repeat(bin_parent, repeats)
            n = h0.size
        else:
            n = 720
            h0 = np.zeros(n, dtype=float)
            h0[rng.choice(n, n // 2, replace=False)] = 1.0
    else:
        n = 720
        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    best_h = None
    best_c = np.inf
    best_n = n

    # ----- Simulated annealing (single long run) -----------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        h_opt, c_opt = _anneal_binary(h0, remaining * 0.7, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
            best_n = n

    # ----- Small greedy polishing on the best found solution ------------------
    if time.time() < deadline:
        h_pol, c_pol = greedy_improve(best_h.copy(), rng, deadline)
        if c_pol < best_c:
            best_c, best_h = c_pol, h_pol.copy()

    # ----- Final safety check -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    # binary vectors already satisfy the sum constraint exactly
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
