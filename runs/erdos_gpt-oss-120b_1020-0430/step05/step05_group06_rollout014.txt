        # use self-correlation based SA hill climbing

def adjust_sum_exact(h, rng, tol=1e-12):
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    # attempt to adjust diff by adding to random entries that can accept
    for _ in range(5000):
        if abs(diff) < tol:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff -= add
        else:
            sub = min(-diff, h[i])
            if sub > 0:
                h[i] -= sub
                diff += sub
    # final fallback: distribute remaining diff uniformly small
    if abs(diff) >= tol:
        # adjust first element within bounds
        i = 0
        h[i] += diff
    return h

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def correlation_fft(h):
    """Linear correlation of h with (1-h) using FFT (zero padded)."""
    n = h.size
    h_pad = np.concatenate([h, np.zeros(n, dtype=h.dtype)])
    g = 1.0 - h
    g_pad = np.concatenate([g, np.zeros(n, dtype=h.dtype)])
    corr = np.real(np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(g_pad))))
    return corr[: 2 * n - 1]


def compute_c5(h):
    """C5 = max correlation * (2/n)."""
    n = h.shape[0]
    max_corr = correlation_fft(h).max()
    return float(max_corr * (2.0 / n))


def adjust_sum_exact(h, rng, tol=1e-12):
    """Exact correction so that sum(h) = n/2, staying in [0,1]."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(5000):
        if abs(diff) < tol:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff -= add
        else:
            sub = min(-diff, h[i])
            if sub > 0:
                h[i] -= sub
                diff += sub
    # tiny fallback
    if abs(diff) >= tol:
        i = 0
        h[i] += diff
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max surrogate (using correlation_fft) for SLSQP."""
    corr = correlation_fft(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=600):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def hill_climb_binary(n, h_initial, idx, deadline, rng):
    """SA hill‑climb on binary vectors, maximizing the minimal self‑correlation."""
    h = h_initial.copy()
    S = np.correlate(h, h, mode="full")
    min_S = S.min()
    best_h = h.copy()
    best_min_S = min_S

    pad = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
    temp = 2.0
    cooling = 0.9999

    while time.time() < deadline:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        base_i = n - 1 + i
        base_j = n - 1 + j

        hi_total = pad[base_i + idx] + pad[base_i - idx]
        hj_total = pad[base_j + idx] + pad[base_j - idx]
        delta_S = hj_total - hi_total
        new_S = S + delta_S
        new_min_S = new_S.min()
        delta_min_S = new_min_S - min_S

        accept = False
        if delta_min_S > 0:
            accept = True
        else:
            if temp > 1e-12 and rng.random() < np.exp(delta_min_S / temp):
                accept = True

        if accept:
            h[i] = 0.0
            h[j] = 1.0
            pad[base_i] = 0.0
            pad[base_j] = 1.0
            S = new_S
            min_S = new_min_S
            if min_S > best_min_S:
                best_min_S = min_S
                best_h = h.copy()

        temp *= cooling

    return best_h, best_min_S


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # problem size
    n = 720
    target_sum = n / 2.0
    idx = np.arange(-(n - 1), n, dtype=int)  # length 2n‑1

    best_h = None
    best_c = np.inf
    best_min_S = -np.inf

    # consider the provided parent if its length matches n
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size == n:
            c_parent = compute_c5(parent)
            if c_parent < best_c:
                best_c = c_parent
                best_h = parent.copy()
                best_min_S = np.correlate(parent, parent, mode="full").min()

    # allocate time: 65% hill‑climbing, rest for refinement
    total_time = deadline - start
    refine_time = total_time * 0.35
    hc_deadline = deadline - refine_time

    # random binary hill‑climbing restarts
    while time.time() < hc_deadline:
        perm = rng.permutation(n)
        h0 = np.zeros(n, dtype=float)
        h0[perm[: int(target_sum)]] = 1.0
        h_local, min_S_local = hill_climb_binary(n, h0, idx, hc_deadline, rng)
        c_local = (target_sum - min_S_local) * (2.0 / n)
        if c_local < best_c:
            best_c = c_local
            best_h = h_local.copy()
            best_min_S = min_S_local

    # fallback if nothing found yet
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: int(target_sum)] = 1.0
        best_min_S = np.correlate(best_h, best_h, mode="full").min()
        best_c = (target_sum - best_min_S) * (2.0 / n)

    # fine‑tune with SLSQP (smooth‑max surrogate)
    h_float = best_h.astype(float).copy()
    adjust_sum_exact(h_float, rng)

    betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]
    while time.time() < deadline:
        for beta in betas:
            if time.time() >= deadline:
                break
            max_iter = 600 if beta < 200 else 800
            try:
                h_float = slsqp_optimize(h_float, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            c_new = compute_c5(h_float)
            if c_new < best_c:
                best_c = c_new
                best_h = h_float.copy()
                best_min_S = np.correlate(best_h, best_h, mode="full").min()
        break  # one pass is sufficient

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
