import time

def adjust_sum_exact(h, rng):
    """Modify h in‑place so that sum(h) == len(h)/2 exactly,
    keeping 0 ≤ h[i] ≤ 1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # greedy random adjustments
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(h, rng, step=0.02, max_iter=30):
    """Projected sub‑gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]   # contribution of (1 - h[i+k])
            g[k:] -= h[:upper]          # contribution of -h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def fractional_sa(start_h, rng, sub_deadline):
    """Simulated‑annealing on continuous h (preserves sum)."""
    n = start_h.size
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    total = max(sub_deadline - start, 1e-12)
    # initial temperature – a fraction of current objective
    T0 = max(1e-4, cur_c5 * 0.5)

    while time.time() < sub_deadline:
        now = time.time()
        elapsed = now - start
        frac = elapsed / total
        T = T0 * (1.0 - frac) + 1e-12          # linear cooling
        eps_scale = max(0.01, 1.0 - frac)      # shrink step size

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        eps_max = min(h[i], 1.0 - h[j])
        if eps_max < 1e-12:
            continue
        eps = eps_max * eps_scale * rng.random()

        h[i] -= eps
        h[j] += eps

        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps

    # final local polish
    best_h = gradient_refine(best_h, rng, step=0.02, max_iter=80)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

        best_h = gradient_refine(best_h, rng, step=0.02, max_iter=120)
        best_c5 = compute_c5(best_h)

    # safety: enforce constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(n)
