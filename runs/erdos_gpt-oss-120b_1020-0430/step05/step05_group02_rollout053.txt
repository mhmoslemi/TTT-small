def conv_fft(a, b):
    n = a.size + b.size - 1
    nfft = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    c = np.fft.irfft(C, nfft)
    return c[:n]

def conv_fft(a, b):
    n = a.size + b.size - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    c = np.fft.irfft(C, nfft)
    return c[:n]

import time
import numpy as np

# ----- helper functions -----
def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    n = a.size + b.size - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    c = np.fft.irfft(C, nfft)
    return c[:n]

def adjust_sum_exact(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass so that Σh == target, keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5_fast(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx, using the discretised version."""
    I = conv_fft(h, (1.0 - h)[::-1])           # raw sum, length 2n‑1
    I_norm = I * (2.0 / h.size)               # integral value in [0,1]
    return float(I_norm.max())

# ----- main optimisation routine -----
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # parent vector if supplied
    parent = None
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)

    candidate_ns = [360, 240, 180]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0

        # ---- generate a few diverse starts ----
        starts = []

        if parent is not None:
            xs_p = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xs_n = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xs_n, xs_p, parent)
            starts.append(interp.copy())
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            starts.append(bin_vec)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        starts.append(rand_frac)

        # alternating binary pattern
        starts.append(np.tile([1.0, 0.0], n // 2))

        for h0 in starts:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, target, rng)

            c = compute_c5_fast(h)
            if c < best_c:
                best_c, best_h, best_n = c, h.copy(), n

            # ---- gradient‑based descent on smoothed max of I ----
            beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 800.0, 3200.0, 12800.0]
            base_iters = int(250 * (360.0 / n))  # more steps for smaller n

            for beta in beta_sched:
                if time.time() > deadline:
                    break
                iters = base_iters
                step = 0.5 / (np.sqrt(beta) + 0.1)

                for it in range(iters):
                    if time.time() > deadline:
                        break

                    # I (raw sum) and its normalised version (integral)
                    I_raw = conv_fft(h, (1.0 - h)[::-1])
                    I = I_raw * (2.0 / n)            # in [0,1]

                    # weights concentrate on large I (worst shift)
                    w = np.exp(beta * I)
                    w_sum = w.sum()
                    if w_sum == 0 or not np.isfinite(w_sum):
                        w = np.full_like(w, 1.0 / w.size)
                    else:
                        w /= w_sum

                    # convolutions needed for the gradient
                    conv_wh = conv_fft(w, h)           # length 3n‑2
                    w_rev = w[::-1]
                    conv_wrev_h = conv_fft(w_rev, h)

                    off = n - 1
                    conv2 = conv_wh[off:off + n]       # Σ w_s h_{j‑s}
                    conv1 = conv_wrev_h[off:off + n]   # Σ w_s h_{j+s}

                    g = 1.0 - conv2 - conv1               # Σ w_s ∂I_s/∂h_j
                    # g already sums to zero analytically; enforce numerically
                    g -= g.mean()

                    # gradient descent on max(I)
                    h_new = h - step * g
                    np.clip(h_new, 0.0, 1.0, out=h_new)
                    adjust_sum_exact(h_new, target, rng)
                    h = h_new

                    # occasional evaluation
                    if it % 100 == 0:
                        c_cur = I.max()
                        if c_cur < best_c:
                            best_c, best_h, best_n = c_cur, h.copy(), n

                # final evaluation after this beta
                I_raw = conv_fft(h, (1.0 - h)[::-1])
                c_cur = (I_raw * (2.0 / n)).max()
                if c_cur < best_c:
                    best_c, best_h, best_n = c_cur, h.copy(), n

    # ----- fallback (should never trigger) -----
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, n / 2.0, rng)
        best_c = compute_c5_fast(best_h)

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, best_n / 2.0, rng)
    final_c = compute_c5_fast(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
