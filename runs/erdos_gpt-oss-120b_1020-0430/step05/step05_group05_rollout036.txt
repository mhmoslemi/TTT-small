def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s*0.99

    # set n=360
    n = 360
    M = n // 2

    # Precompute constant arrays maybe need for gradient in CSV? We'll compute fresh each iteration.

    def half_to_full(half):
        return np.concatenate([half, 1.0 - half])

    def compute_c5(full):
        n = full.size
        # use FFT correlation for speed
        H = np.fft.fft(full)
        G = np.fft.fft(1.0 - full)
        corr = np.real(np.fft.ifft(H * np.conj(G)))
        return float(np.max(corr) * (2.0 / n))

    # gradient descent
    def gradient_descent_half(half, betas, steps_per_beta, rng):
        M = half.size
        n = 2*M
        # pre allocate arrays? fine
        for beta in betas:
            # adapt step size
            step = 0.5 / (1.0 + beta / 1000.0)
            for _ in range(steps_per_beta):
                full = np.concatenate([half, 1.0 - half])
                # FFTs
                H = np.fft.fft(full)
                G = np.fft.fft(1.0 - full)
                corr = np.real(np.fft.ifft(H * np.conj(G)))
                # compute w safely
                max_corr = np.max(corr)
                exp_shift = np.exp(beta * (corr - max_corr))
                sum_exp = np.sum(exp_shift)
                w = exp_shift / sum_exp
                # compute conv_fwd
                W = np.fft.fft(w)
                conv_fwd = np.real(np.fft.ifft(W * H))
                # w_rev
                w_rev = np.empty_like(w)
                w_rev[0] = w[0]
                w_rev[1:] = w[:0:-1]
                W_rev = np.fft.fft(w_rev)
                conv_rev = np.real(np.fft.ifft(W_rev * H))
                g_full = 1.0 - (conv_fwd + conv_rev)
                g_half = g_full[:M] - g_full[M:]
                # update
                half -= step * g_half
                np.clip(half, 0.0, 1.0, out=half)
                # optional early exit if time
                if time.time() > deadline:
                    return half
        return half

    # binary improvement
    # define binary_targeted_improve similar to earlier code but to keep in separate function maybe simpler: we'll implement a simple greedy: repeatedly find shift with max overlap in current binary, then swap a 1 that contributes most to that shift with a 0 that contributes least.

    def binary_targeted_improve(bin_h, deadline):
        n = bin_h.size
        # Use while loop until time or no improvement
        while time.time() < deadline:
            # compute corr
            H = np.fft.fft(bin_h)
            G = np.fft.fft(1.0 - bin_h)
            corr = np.real(np.fft.ifft(H * np.conj(G)))
            max_idx = np.argmax(corr)
            # shift = max_idx (mod n)
            shift = max_idx
            # if shift == 0: break
            if shift == 0:
                break
            # compute neighbor sums for this shift
            # For each i, neighbor value = bin_h[(i+shift)%n] + bin_h[(i-shift)%n]
            # This can be computed via roll
            plus = np.roll(bin_h, -shift)
            minus = np.roll(bin_h, shift)
            neigh = plus + minus
            # ones indices
            ones = np.where(bin_h > 0.5)[0]
            zeros = np.where(bin_h < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            # candidate i with minimal neigh among ones
            i = ones[np.argmin(neigh[ones])]
            # candidate j with maximal neigh among zeros
            j = zeros[np.argmax(neigh[zeros])]
            # compute delta = neigh[i] - neigh[j]
            if neigh[i] - neigh[j] < 0:
                # perform swap
                bin_h[i] = 0.0
                bin_h[j] = 1.0
                continue
            # else try random swaps
            improved = False
            for _ in range(5):
                i2 = rng.choice(ones)
                j2 = rng.choice(zeros)
                if neigh[i2] - neigh[j2] < 0:
                    bin_h[i2] = 0.0
                    bin_h[j2] = 1.0
                    improved = True
                    break
            if not improved:
                break
        return bin_h

    # binary conversion
    def fractional_to_binary(full):
        # set top n/2 largest entries to 1
        n = full.size
        half = n // 2
        idx = np.argpartition(full, -half)[-half:]
        bin_full = np.zeros_like(full)
        bin_full[idx] = 1.0
        return bin_full

    # main search
    best_h = None
    best_c = np.inf

    # betas list and steps per beta
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]
    steps_per_beta = 30  # each beta 30 steps

    # set up start halves
    start_halves = []
    # parent if available
    if "initial_h_values" in globals():
        parent_full = np.array(initial_h_values, dtype=float)
        # ensure length n
        if parent_full.size != n:
            # if different size, maybe we can interpolate to n
            xs = np.linspace(0.0, 2.0, parent_full.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            parent_full = np.interp(xt, xs, parent_full)
        # take first half
        start_halves.append(parent_full[:M].copy())
    # random halves
    for _ in range(5):
        start_halves.append(rng.random(M))

    for half_init in start_halves:
        if time.time() > deadline:
            break
        half = half_init.copy()
        # gradient descent
        half = gradient_descent_half(half, betas, steps_per_beta, rng)
        # evaluate
        full = half_to_full(half)
        c = compute_c5(full)
        if c < best_c:
            best_c = c
            best_h = full.copy()
        # binary improvement
        bin_h = fractional_to_binary(full)
        bin_h = binary_targeted_improve(bin_h, deadline)
        c_bin = compute_c5(bin_h)
        if c_bin < best_c:
            best_c = c_bin
            best_h = bin_h.copy()

    # optional upsampling
    if best_h is not None and time.time() < deadline and n*2 <= 720:
        # upsample best_h to n_up = 720
        n_up = n * 2
        M_up = n_up // 2
        # interpolate half part
        xs = np.linspace(0.0, 1.0, M, endpoint=False)
        xt = np.linspace(0.0, 1.0, M_up, endpoint=False)
        half_up = np.interp(xt, xs, best_h[:M])
        # gradient descent on upsampled half
        half_up = gradient_descent_half(half_up, betas, steps_per_beta, rng)
        full_up = np.concatenate([half_up, 1.0 - half_up])
        c_up = compute_c5(full_up)
        if c_up < best_c:
            best_c = c_up
            best_h = full_up.copy()
        # binary improvement on upsampled
        bin_up = fractional_to_binary(full_up)
        bin_up = binary_targeted_improve(bin_up, deadline)
        c_bin_up = compute_c5(bin_up)
        if c_bin_up < best_c:
            best_c = c_bin_up
            best_h = bin_up.copy()
        n = n_up

    # fallback if none
    if best_h is None:
        n = 360
        half = np.full(n//2, 0.5)
        best_h = half_to_full(half)
        best_c = compute_c5(best_h)

    # final adjust: ensure values within [0,1]
    best_h = np.clip(best_h, 0.0, 1.0)
    # compute final C5
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np

def half_to_full(half):
    """Map M‑vector to full periodic 2M‑vector with antisymmetry."""
    return np.concatenate([half, 1.0 - half])

def compute_c5(full):
    """Exact max overlap using linear correlation (np.correlate)."""
    n = full.size
    corr = np.correlate(full, 1.0 - full, mode="full")
    return float(np.max(corr) * (2.0 / n))

def smooth_weights(corr, beta):
    """Log‑sum‑exp weights for a given β."""
    m = np.max(corr)
    exp_shift = np.exp(beta * (corr - m))
    return exp_shift / np.sum(exp_shift)

def grad_full_from_corr(full, w):
    """Gradient of smooth‑max w.r.t. full vector using circular convolution."""
    n = full.size
    H = np.fft.fft(full)
    W = np.fft.fft(w)
    conv_fwd = np.real(np.fft.ifft(W * H))
    # reversed w (w[0] stays, others reversed)
    w_rev = np.empty_like(w)
    w_rev[0] = w[0]
    if n > 1:
        w_rev[1:] = w[:0:-1]
    W_rev = np.fft.fft(w_rev)
    conv_rev = np.real(np.fft.ifft(W_rev * H))
    return 1.0 - (conv_fwd + conv_rev)

def gradient_descent_half(half, betas, steps_per_beta, deadline):
    """Projected gradient descent on half‑vector using smooth‑max surrogate."""
    M = half.size
    n = 2 * M
    for beta in betas:
        step = 0.5 / (1.0 + beta / 1000.0)
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return half
            full = half_to_full(half)
            # circular correlation via FFT
            H = np.fft.fft(full)
            G = np.fft.fft(1.0 - full)
            corr = np.real(np.fft.ifft(H * np.conj(G)))
            w = smooth_weights(corr, beta)
            g_full = grad_full_from_corr(full, w)
            g_half = g_full[:M] - g_full[M:]
            half -= step * g_half
            np.clip(half, 0.0, 1.0, out=half)
    return half

def fractional_to_binary(full):
    """Exact binary vector with n/2 ones (largest entries become 1)."""
    n = full.size
    half = n // 2
    idx = np.argpartition(full, -half)[-half:]
    bin_full = np.zeros_like(full)
    bin_full[idx] = 1.0
    return bin_full

def binary_targeted_improve(bin_h, deadline):
    """Greedy swap improvement for binary vectors."""
    n = bin_h.size
    while time.time() < deadline:
        corr = np.correlate(bin_h, 1.0 - bin_h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)
        k = abs(shift)
        if k == 0:
            break
        plus = np.roll(bin_h, -k)
        minus = np.roll(bin_h, k)
        neigh = plus + minus
        ones = np.where(bin_h > 0.5)[0]
        zeros = np.where(bin_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(neigh[ones])]
        j = zeros[np.argmax(neigh[zeros])]
        if neigh[i] - neigh[j] < 0:
            bin_h[i] = 0.0
            bin_h[j] = 1.0
            continue
        improved = False
        for _ in range(5):
            i2 = np.random.choice(ones)
            j2 = np.random.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                bin_h[i2] = 0.0
                bin_h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return bin_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    n = 360                # base resolution, even
    M = n // 2

    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0,
             200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]
    steps_per_beta = 70

    # collect starting half‑vectors
    starts = []
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            parent = np.interp(xt, xs, parent)
        starts.append(parent[:M].copy())
    for _ in range(30):
        starts.append(rng.random(M))

    best_h = None
    best_c = np.inf

    for half0 in starts:
        if time.time() > deadline:
            break
        half = half0.copy()
        half = gradient_descent_half(half, betas, steps_per_beta, deadline)
        full = half_to_full(half)
        c = compute_c5(full)
        if c < best_c:
            best_c, best_h = c, full.copy()
        bin_h = fractional_to_binary(full)
        bin_h = binary_targeted_improve(bin_h, deadline)
        c_bin = compute_c5(bin_h)
        if c_bin < best_c:
            best_c, best_h = c_bin, bin_h.copy()

    # optional up‑sampling to 720 points
    if best_h is not None and time.time() < deadline and n * 2 <= 720:
        n_up = n * 2
        M_up = n_up // 2
        xs_half = np.linspace(0.0, 1.0, M, endpoint=False)
        xt_half = np.linspace(0.0, 1.0, M_up, endpoint=False)
        half_up = np.interp(xt_half, xs_half, best_h[:M])
        half_up = gradient_descent_half(half_up, betas, steps_per_beta, deadline)
        full_up = half_to_full(half_up)
        c_up = compute_c5(full_up)
        if c_up < best_c:
            best_c, best_h = c_up, full_up.copy()
        bin_up = fractional_to_binary(full_up)
        bin_up = binary_targeted_improve(bin_up, deadline)
        c_bin_up = compute_c5(bin_up)
        if c_bin_up < best_c:
            best_c, best_h = c_bin_up, bin_up.copy()
        n = n_up

    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
