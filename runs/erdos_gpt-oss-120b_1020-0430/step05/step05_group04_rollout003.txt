def simulated_annealing_binary(h, time_deadline):
    n = len(h)
    dx = 2.0/n
    g = 1 - h
    conv = np.correlate(h, g, mode='full')
    cur_c = conv.max() * dx
    best_c = cur_c
    best_h = h.copy()
    best_conv = conv.copy()
    T0 = cur_c * 0.1 + 1e-5
    decay = 0.9999
    iteration = 0
    while time.time() < time_deadline:
        # choose swap
        i = np.random.choice(np.flatnonzero(h))
        j = np.random.choice(np.flatnonzero(g))
        # new conv copy
        new_conv = conv.copy()
        # apply updates
        # use vectorized
        idx_i = np.arange(n) - i + (n-1)
        np.add.at(new_conv, idx_i, -g)
        idx_j = np.arange(n) - j + (n-1)
        np.add.at(new_conv, idx_j, g)
        idx_i2 = np.arange(n) - i + (n-1)
        np.add.at(new_conv, idx_i2, h)
        idx_j2 = np.arange(n) - j + (n-1)
        np.add.at(new_conv, idx_j2, -h)

        # cross terms:
        new_conv[n-1] += -2
        d = i - j
        new_conv[n-1 + d] += 1
        new_conv[n-1 - d] += 1  # because (j-i) = -d
        new_c = new_conv.max() * dx

        # acceptance
        T = T0 * (decay ** iteration)
        if new_c < cur_c or np.random.rand() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            # accept
            h[i] = 0
            h[j] = 1
            g[i] = 1
            g[j] = 0
            conv = new_conv
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                best_conv = conv.copy()
        iteration += 1

    return best_h, best_c

def binary_sa_fixed_time(h_bin, rng, time_deadline):

    n = h_bin.size
    h = h_bin.astype(np.int8).copy()
    g = 1 - h
