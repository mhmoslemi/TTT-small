    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.99 * budget_s

    # choose n = 360 (if parent exists with different n, we may adapt)
    if 'initial_h_values' in globals():

        h = rng.random(n)
        h = h / h.sum() * (n/2)
        # clip to [0,1] and adjust...
        # Use adjust_sum_exact.
        parent = h

    # Evaluate parent
    best_h = parent.copy()
    best_c = compute_c5(best_h)

    # Stage1: SLSQP with smooth-max cascade (multiple betas) with more iterations
    # Might need to set options more robustly.
    # We'll use function smooth_max_obj with current beta.

    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = n/2

    def sum_eq_constraint(x):
        return np.sum(x) - CURRENT_TARGET

    def smooth_max_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    # We'll iteratively increase beta.
    # Start with betas list maybe [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]

    def slsqp_optimize(h0, beta, max_iter=800):

        CURRENT_BETA = beta

        res = minimize(
            smooth_max_obj,
            h0,

            bounds=bounds,

        adjust_sum_exact(x, rng)

    h_cur = best_h.copy()
    for beta in [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]:

        def anneal(h, deadline):
            cur_h = h.copy()
            cur_c = compute_c5(cur_h)
            best_h_local = cur_h.copy()
            best_c_local = cur_c
            n = cur_h.size
            target = n/2
            start = time.time()

            # set temperature schedule
            T0 = 0.005  # initial temperature
            # Also define minimal temperature
            while time.time() < deadline:

                elapsed = time.time() - start
                frac = elapsed / total

                # randomly choose i,j
                i = rng.integers(n)
                j = rng.integers(n-1)
                if j >= i:
                    j += 1

                max_up = 1.0 - cur_h[i]
                max_down = cur_h[j]
                # if both zero, skip
                if max_up <= 0 or max_down <= 0:
                    continue
                # sample delta uniformly
                delta = rng.random() * min(max_up, max_down)
                # do move
                cur_h[i] += delta
                cur_h[j] -= delta
                new_c = compute_c5(cur_h)
                dE = new_c - cur_c
                if dE < -1e-12 or rng.random() < np.exp(-dE / (T + 1e-12)):
                    # accept
                    cur_c = new_c
                    if new_c < best_c_local:
                        best_c_local = new_c
                        best_h_local = cur_h.copy()
                else:
                    # revert
                    cur_h[i] -= delta
                    cur_h[j] += delta
                # occasional rebalancing to ensure sum exactly (due to floating errors)
                if (elapsed % 5.0) < 0.01:  # every ~5 seconds
                    # adjust small diff
                    diff = target - cur_h.sum()
                    if abs(diff) > 1e-12:
                        # distribute diff to random indices
                        for _ in range(10):
                            k = rng.integers(n)

                            else:
                                add = max(diff, -cur_h[k])
                            cur_h[k] += add
                            diff = target - cur_h.sum()
            return best_h_local, best_c_local

        best_h, best_c = anneal(best_h, anneal_deadline)

    # final safety adjust sum
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)

import time
import numpy as np
from scipy.optimize import minimize

# Global parameters for the smooth‑max surrogate
CURR_TARGET = 0.0
CURR_BETA = 1.0

def sum_eq_constraint(x):
    return np.sum(x) - CURR_TARGET

def smooth_max_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURR_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, numerically stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURR_BETA

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Iteratively fix Σh = n/2 using random adjustments."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """Smooth‑max SLSQP step for a given beta."""
    global CURR_TARGET, CURR_BETA
    CURR_BETA = beta
    CURR_TARGET = h0.size / 2.0
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_max_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def anneal(h_start, rng, deadline):
    """Simple pair‑wise simulated‑annealing on the feasible set."""
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = cur_h.size
    target = n / 2.0
    start = time.time()
    last_reb = start
    T0 = 0.005  # initial temperature
    while True:
        now = time.time()
        if now >= deadline:
            break
        total = deadline - start
        elapsed = now - start
        frac = elapsed / total if total > 0 else 0.0
        T = T0 * max(0.0, 1.0 - frac)

        # pick distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        max_up = 1.0 - cur_h[i]
        max_down = cur_h[j]
        if max_up <= 0 or max_down <= 0:
            continue
        delta = rng.random() * min(max_up, max_down)
        cur_h[i] += delta
        cur_h[j] -= delta
        new_c = compute_c5(cur_h)
        dE = new_c - cur_c
        accept = False
        if dE < -1e-12:
            accept = True
        elif T > 0:
            if rng.random() < np.exp(-dE / T):
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] -= delta
            cur_h[j] += delta

        # re‑balance the exact sum every few seconds
        if now - last_reb >= 5.0:
            diff = target - cur_h.sum()
            if abs(diff) > 1e-12:
                for _ in range(5):
                    k = rng.integers(n)
                    if diff > 0:
                        add = min(diff, 1.0 - cur_h[k])
                    else:
                        add = max(diff, -cur_h[k])
                    cur_h[k] += add
                    diff = target - cur_h.sum()
                    if abs(diff) < 1e-12:
                        break
            last_reb = now
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        np.clip(h0, 0.0, 1.0, out=h0)
        adjust_sum_exact(h0, rng)
    else:
        n = 360
        h0 = rng.random(n)
        h0 = h0 / h0.sum() * (n / 2.0)
        adjust_sum_exact(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- smooth‑max cascade on several starts ---------------------------
    betas = (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000)
    starts = [h0]
    # two additional random starts
    for _ in range(2):
        if time.time() >= deadline:
            break
        h_rand = rng.random(n)
        h_rand = h_rand / h_rand.sum() * (n / 2.0)
        adjust_sum_exact(h_rand, rng)
        starts.append(h_rand)

    for h_start in starts:
        if time.time() >= deadline:
            break
        h_cur = h_start.copy()
        for beta in betas:
            if time.time() >= deadline:
                break
            try:
                h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=800 if beta >= 2000 else 600)
            except Exception:
                continue
            c_val = compute_c5(h_cur)
            if c_val < best_c:
                best_c = c_val
                best_h = h_cur.copy()

    # ----- simulated annealing --------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        h_a, c_a = anneal(best_h, rng, anneal_deadline)
        if c_a < best_c:
            best_c = c_a
            best_h = h_a

    # ----- final short SLSQP refinement ------------------------------------
    if time.time() < deadline:
        try:
            best_h = slsqp_optimize(best_h, 20000, rng, max_iter=400)
        except Exception:
            pass
        best_c = compute_c5(best_h)

    # ----- final feasibility fix -----------------------------------------
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
