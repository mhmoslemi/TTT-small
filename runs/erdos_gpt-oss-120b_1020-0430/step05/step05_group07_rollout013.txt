def penalty_obj_and_grad(x, beta, mu, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    # softmax value: (max + log sum exp)/beta
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    softmax_val = (max_bc + np.log(sum_exp)) / beta
    # weights

def gradient_refine(h, rng, step=0.02, max_iter=15):

        else:

            cur[j] -= delta

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def compute_c5(h):
    """Maximum overlap C5 for a step function given by array h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng, max_iter=2000, tol=1e-12):
    """Exact sum correction to enforce Σh = n/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(h, rng, step=0.02, max_iter=15):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            s = k
            g = np.empty_like(h)
            # first term (1 - h[i+s])
            g[: n - s] = 1.0 - h[s:]
            g[n - s :] = 0.0
            # second term -h[i-s]
            g[:s] -= 0.0
            g[s:] -= h[: n - s]

        decay = 0.5 ** it
        h = h - step * decay * g
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(100):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def fractional_sa(h, rng, deadline):
    """Transfer‑based simulated annealing on a continuous vector."""
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    n = cur.size
    init_T = max(0.05 * cur_c, 1e-4)
    it = 0
    check_interval = 2000
    while True:
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0.0 or cur[j] >= 1.0:
            continue
        max_transfer = min(cur[i], 1.0 - cur[j])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        cur[i] -= delta
        cur[j] += delta
        c = compute_c5(cur)
        T = init_T * (0.99995 ** it)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / (T + 1e-12)):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it % check_interval == 0 and time.time() > deadline:
            break
    return best, best_c


def penalty_obj_and_grad(x, beta, mu, target):
    """Soft‑max surrogate + quadratic sum‑penalty; returns (f, grad)."""
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    w = exp_bc / sum_exp

    grad = np.zeros_like(x)
    n1 = n - 1
    # accumulate weighted derivatives
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        shift = idx - n1
        if shift >= 0:
            s = shift
            if s < n:
                grad[: n - s] += weight * (1.0 - x[s:])
                grad[s:] -= weight * x[: n - s]
        else:
            s = -shift
            if s < n:
                grad[s:] += weight * (1.0 - x[: n - s])
                grad[: n - s] -= weight * x[s:]

    sum_diff = np.sum(x) - target
    grad += 2.0 * mu * sum_diff
    soft_val = (max_bc + np.log(sum_exp)) / beta
    f = soft_val + mu * sum_diff * sum_diff
    return f, grad


def lbfgsb_optimize(h, rng, deadline):
    """Continuation L‑BFGS‑B using increasing beta values."""
    n = h.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best = h.copy()
    best_c = compute_c5(best)
    betas = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]

    for beta in betas:
        if time.time() > deadline:
            break
        mu = 1e3 if beta < 2000 else 1e4
        max_iter = 200 if beta < 200 else 400
        try:
            res = minimize(
                penalty_obj_and_grad,
                best,
                args=(beta, mu, target),
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        h_new = np.clip(res.x, 0.0, 1.0)
        adjust_sum_exact(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best = h_new
    return best, best_c


def random_binary(n, rng):
    v = np.zeros(n)
    v[rng.choice(n, n // 2, replace=False)] = 1.0
    return v


def random_fractional(n, rng):
    v = rng.random(n)
    v *= (n / 2.0) / v.sum()
    v = np.clip(v, 0.0, 1.0)
    adjust_sum_exact(v, rng)
    return v


def random_fractional_sq(n, rng):
    v = rng.random(n) ** 2
    v *= (n / 2.0) / v.sum()
    v = np.clip(v, 0.0, 1.0)
    adjust_sum_exact(v, rng)
    return v


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # Choose a fairly fine sampling – 720 points works well within the budget.
    n = 720
    if n % 2 != 0:
        n += 1

    starts = []

    # Interpolated parent (if available) and its binarised version
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        starts.append(interp)

        bin_v = np.zeros(n)
        idx = np.argpartition(interp, -n // 2)[-n // 2 :]
        bin_v[idx] = 1.0
        starts.append(bin_v)

    # Additional diverse starts
    for _ in range(2):
        starts.append(random_binary(n, rng))
    for _ in range(2):
        starts.append(random_fractional(n, rng))
    for _ in range(2):
        starts.append(random_fractional_sq(n, rng))

    best_h = None
    best_c = float("inf")

    for init_h in starts:
        if time.time() >= deadline:
            break
        h0 = init_h.copy()
        adjust_sum_exact(h0, rng)

        # ---- Simulated annealing (large chunk of remaining time) ----
        remaining = deadline - time.time()
        sa_time = min(remaining * 0.30, 300.0)  # cap to 5 minutes
        sub_dead = time.time() + sa_time
        h_sa, c_sa = fractional_sa(h0, rng, sub_dead)
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa

        # ---- L‑BFGS‑B continuation ----
        if time.time() >= deadline:
            break
        h_opt, c_opt = lbfgsb_optimize(best_h, rng, deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

        # ---- Final gradient refinement ----
        if time.time() < deadline:
            h_ref = gradient_refine(best_h, rng, step=0.02, max_iter=15)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

        # ---- Small random perturbations + brief re‑optimisation ----
        if time.time() < deadline:
            for _ in range(3):
                if time.time() >= deadline:
                    break
                pert = best_h + rng.normal(scale=0.02, size=n)
                pert = np.clip(pert, 0.0, 1.0)
                adjust_sum_exact(pert, rng)
                h_pert, c_pert = lbfgsb_optimize(pert, rng, deadline)
                if c_pert < best_c:
                    best_h, best_c = h_pert.copy(), c_pert

    # ----- Fallback if nothing succeeded -----
    if best_h is None:
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # Final sanity checks
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
