def project_onto_box_sum(v, target):
    # v is real vector length n
    # wants x_i = min(max(v_i - lam,0),1)
    # find lam such that sum_i x_i = target
    # lam can be binary search between min(v_i-1) and max(v_i)
    lo = v.min() - 1.0  # if lam low, then v_i - lam >1 so clipped to 1
    hi = v.max()  # if lam high, v_i - lam < 0 => 0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # adjust for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff to non-boundary entries (<=1 and >=0)
        # find indices where 0 < x_i < 1
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if np.any(mask):
            # add equally
            add = diff / mask.sum()
            x[mask] += add
            # clip again and recompute diff if needed
            x = np.clip(x, 0.0, 1.0)
        else:
            # handle edge case: all at boundaries, adjust smallest gap
            # fallback: enforce sum by adjusting a random index
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def project_onto_box_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    # adjust small remainder
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # find non-bound indices
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust random index
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def slsqp_optimize(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n / 2.0

    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # compute log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def sum_eq_constraint(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # fix sum using water-filling (project)
        x = project_onto_box_sum(x, target)
    return x

import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    step = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c

def softmax_gradient(h, beta):
    n = h.size
    O = np.correlate(h, 1.0 - h, mode='full')
    Omax = O.max()
    exp_term = np.exp(beta * (O - Omax))
    w = exp_term / exp_term.sum()
    grad = np.zeros_like(h)
    wk = w[n - 1]
    grad += wk * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk == 0.0:
            continue
        grad += wk
        grad[: n - k] -= wk * h[k:]
        grad[k:] -= wk * h[: n - k]
    return grad

def slsqp_optimize(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n / 2.0
    def soft_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def eq_con(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": eq_con}
    res = minimize(soft_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = project_onto_box_sum(h0, target)
    else:
        n = 512
        target = n / 2.0
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    beta_list = [20, 100, 500, 2000, 8000, 20000]
    step_map = {20:0.5, 100:0.4, 500:0.3, 2000:0.2, 8000:0.15, 20000:0.1}
    for beta in beta_list:
        if time.time() - start > budget_s * 0.78:
            break
        iters = int(800 * 256 / n)
        if beta >= 2000:
            iters = int(1200 * 256 / n)
        step0 = step_map[beta]
        h = best_h.copy()
        for it in range(iters):
            if time.time() - start > budget_s * 0.78:
                break
            grad = softmax_gradient(h, beta)
            step = step0 * (1.0 - it / iters)
            v = h - step * grad
            h = project_onto_box_sum(v, target)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
        if time.time() - start < budget_s * 0.85:
            hc, cc = guided_hill_climb(h, rng,
                max_iters=int(800 * 256 / n), step_scale=0.5)
            if cc < best_c:
                best_h, best_c = hc.copy(), cc
    # final SLSQP refinements
    for beta in (2000, 8000, 20000, 50000):
        if time.time() - start > budget_s * 0.90:
            break
        h = slsqp_optimize(best_h, beta, rng, max_iter=800)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c
    if time.time() - start < budget_s * 0.95:
        hc, cc = guided_hill_climb(best_h, rng,
            max_iters=int(1200 * 256 / n), step_scale=0.4)
        if cc < best_c:
            best_h, best_c = hc.copy(), cc
    # small stochastic polishing
    while time.time() - start < budget_s * 0.999:

        nc = compute_c5(nh)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
