def project_onto_box_and_sum(v, target, max_iter=50, tol=1e-10):
    # v: real vector
    # target: desired sum
    lo = -np.max(v)  # ensure v+lo >=0? Actually lower bound for lambda is such that all entries after clipping to 0 => we need lambda >= -max(v) maybe.
    hi = 1.0 - np.min(v)  # ensure v+hi <=1
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) < tol:
            return y
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)
```

import time

# ------------------------------------------------------------
def project_onto_box_and_sum(v, target, max_iter=60):
    """Euclidean projection of v onto [0,1]^n with sum=v_target."""
    # lo => all zeros, hi => all ones
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def worst_gradient(h):
    """Gradient of the worst (max) overlap shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad
