import time, numpy as np, math
from scipy.optimize import minimize

def half_to_full(x):
    return np.concatenate([x, 1.0 - x])

def compute_c5_full(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def compute_c5_half(x):
    return compute_c5_full(half_to_full(x))

CURRENT_BETA = None

def smoothmax_corr_half(x):
    full = half_to_full(x)
    corr = np.correlate(full, 1.0 - full, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # logsumexp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_half(x0, beta, max_iter):
    global CURRENT_BETA
    CURRENT_BETA = beta
    n = x0.size * 2
    bounds = [(0.0, 1.0)] * x0.size
    # SLSQP may try to change x by small amounts
    res = minimize(
        smoothmax_corr_half,
        x0,
        method='SLSQP',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    return x

def gradient_refine_half(x, step, max_iter):
    M = x.size
    n = 2 * M
    for it in range(max_iter):
        full = half_to_full(x)
        corr = np.correlate(full, 1.0 - full, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        k = abs(k)
        # compute neighbor sum for full
        idx_plus = (np.arange(n) + k) % n
        idx_minus = (np.arange(n) - k) % n
        neighbor_sum = full[idx_plus] + full[idx_minus]  # shape n
        grad_half = neighbor_sum[:M] - neighbor_sum[M:]
        # step size decreasing
        step_sz = step * (0.5 ** it)
        x = x + step_sz * grad_half
        x = np.clip(x, 0.0, 1.0)
    return x

def binary_improve_half(x_half, deadline):
    # x_half is binary (0/1). full = half + complement.
    M = x_half.size
    n = 2 * M
    x = x_half.copy()
    while time.time() < deadline:
        full = half_to_full(x)
        corr = np.correlate(full, 1.0 - full, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        k = abs(k)
        if k == 0:
            break
        # neighbor sum for full
        idx_plus = (np.arange(n) + k) % n
        idx_minus = (np.arange(n) - k) % n
        neighbor_sum = full[idx_plus] + full[idx_minus]
        # compute neighbor sum for half positions (i) as before
        neigh_half = neighbor_sum[:M] - neighbor_sum[M:]  # g_half = (h_i+h_{i+k} + h_i+h_{i-k}) - same for complement?
        # However we need value of contribution to S_k from each half position.
        # Contribution to S_k from a half index i when it's 1: h[i]h[i+k] + h[i]h[i-k] (since both directions)
        # When half[i] ==1, full[i]=1, full[i+M]=0. Contribution from i to S_k: full[i]*full[i+k] + full[i]*full[i-k] .
        # But full[i] is 1, so contribution = full[i+k] + full[i-k]. Similarly, if half[i] ==0 (full[i]=0), contribution = 0.
        # For the complement (full[i+M]), contribution to S_k when it's 1 (i.e., half[i]==0) is similar.
        # However for swapping zero and one in half, delta S_k = (contrib_when_one) - (contrib_when_zero). That equals neighbor_sum[i] - neighbor_sum[i+M] = grad_half[i].
        # So we can use grad_half = neighbor_sum[:M] - neighbor_sum[M:].
        grad_half = neighbor_sum[:M] - neighbor_sum[M:]
        # indices of ones and zeros in half
        ones = np.flatnonzero(x == 1)
        zeros = np.flatnonzero(x == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        # For ones, we want low contribution (i.e., small grad_half) to replace.
        i = ones[np.argmin(grad_half[ones])]
        # For zeros, we want high contribution (large grad_half) to replace.
        j = zeros[np.argmax(grad_half[zeros])]
        if grad_half[j] > grad_half[i]:
            x[i] = 0
            x[j] = 1
            continue
        # else try random swap; break if none improve
        improved = False
        for _ in range(5):
            i2 = ones[np.random.randint(ones.size)]
            j2 = zeros[np.random.randint(zeros.size)]
            if grad_half[j2] > grad_half[i2]:
                x[i2] = 0
                x[j2] = 1
                improved = True
                break
        if not improved:
            break
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----- helpers for the symmetric representation -----
def half_to_full(x):
    """Construct full period-2 vector from half-vector obeying h(x+1)=1-h(x)."""
    return np.concatenate([x, 1.0 - x])


def compute_c5_full(h):
    """Max overlap C5 for a full vector (linear correlation, as used in evaluation)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def compute_c5_half(x):
    """C5 for a half-vector (full is built internally)."""
    return compute_c5_full(half_to_full(x))


# ----- smooth‑max surrogate for SLSQP -----
CURRENT_BETA = None


def _smoothmax_half(x):
    full = half_to_full(x)
    corr = np.correlate(full, 1.0 - full, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_half(x0, beta, max_iter):
    """SLSQP on the smooth‑max objective for a half‑vector."""
    global CURRENT_BETA
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * x0.size
    res = minimize(
        _smoothmax_half,
        x0,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    return np.clip(res.x, 0.0, 1.0)


# ----- projected gradient refinement (no sum constraint needed) -----
def gradient_refine_half(x, step, max_iter):
    M = x.size
    n = 2 * M
    for it in range(max_iter):
        full = half_to_full(x)
        corr = np.correlate(full, 1.0 - full, mode="full")
        idx = np.argmax(corr)
        k = abs(idx - (n - 1))
        # neighbour sum for every full index
        idx_plus = (np.arange(n) + k) % n
        idx_minus = (np.arange(n) - k) % n
        neigh = full[idx_plus] + full[idx_minus]
        grad_half = neigh[:M] - neigh[M:]
        step_sz = step * (0.5 ** it)
        x = x + step_sz * grad_half
        x = np.clip(x, 0.0, 1.0)
    return x


# ----- binary improvement respecting symmetry -----
def binary_improve_half(x_half, deadline):
    """Swap 0/1 entries in half to lower the worst overlap (symmetry preserved)."""
    M = x_half.size
    n = 2 * M
    x = x_half.copy()
    rng = np.random.default_rng()
    cnt = 0
    while time.time() < deadline and cnt < 2000:
        full = half_to_full(x)
        corr = np.correlate(full, 1.0 - full, mode="full")
        idx = np.argmax(corr)
        k = abs(idx - (n - 1))
        if k == 0:
            break
        idx_plus = (np.arange(n) + k) % n
        idx_minus = (np.arange(n) - k) % n
        neigh = full[idx_plus] + full[idx_minus]
        grad_half = neigh[:M] - neigh[M:]

        ones = np.flatnonzero(x == 1.0)
        zeros = np.flatnonzero(x == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break

        i = ones[np.argmin(grad_half[ones])]
        j = zeros[np.argmax(grad_half[zeros])]
        if grad_half[j] > grad_half[i]:
            x[i] = 0.0
            x[j] = 1.0
            cnt += 1
            continue

        # try a few random swaps as a last resort
        improved = False
        for _ in range(5):
            i2 = ones[rng.integers(ones.size)]
            j2 = zeros[rng.integers(zeros.size)]
            if grad_half[j2] > grad_half[i2]:
                x[i2] = 0.0
                x[j2] = 1.0
                improved = True
                break
        if not improved:
            break
        cnt += 1
    return x


# ----- main routine ---------------------------------------------------

    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin
    rng = np.random.default_rng(seed)

    # base resolution
    n = 360

            starts.append(half_parent)

    for _ in range(3):
        starts.append(rng.random(M))

    # random binary halves with exactly M/2 ones
    for _ in range(3):
        half_bin = np.zeros(M)
        half_bin[rng.choice(M, M // 2, replace=False)] = 1.0
        starts.append(half_bin)

    for _ in range(2):
        starts.append(rng.beta(0.5, 0.5, size=M))

    best_half = None
    best_c = np.inf

    for half0 in starts:

        x = np.clip(half0, 0.0, 1.0)

        # SLSQP cascade
        for beta in betas:

            max_iter = 1200 if beta < 200 else 1800

            c = compute_c5_half(x)
            if c < best_c:

                best_half = x.copy()

        # gradient polishing
        x = gradient_refine_half(x, step=0.03, max_iter=30)

        if c < best_c:

        x_bin = (x >= 0.5).astype(float)

    # ----- optional up‑sampling if time permits -----
    if best_half is not None and time.time() < deadline and n * 2 <= 720:
        n_up = n * 2

        xs = np.linspace(0.0, 1.0, M, endpoint=False)
        xt = np.linspace(0.0, 1.0, M_up, endpoint=False)
        half_up = np.interp(xt, xs, best_half)
        half_up = np.clip(half_up, 0.0, 1.0)

        for beta in betas:
            if time.time() > deadline:
                break
            try:
                half_up = slsqp_optimize_half(half_up, beta, max_iter=2000)
            except Exception:
                continue
            c = compute_c5_half(half_up)
            if c < best_c:
                best_c = c
                best_half = half_up.copy()
                n = n_up

        c = compute_c5_half(half_up)
        if c < best_c:
            best_c = c
            best_half = half_up.copy()
            n = n_up

        half_up_bin = (half_up >= 0.5).astype(float)
        half_up_bin = binary_improve_half(half_up_bin, deadline)

    # ----- fallback if nothing succeeded -----
    if best_half is None:

        best_half[: M // 2] = 1.0

    # ----- final output -----
    final_full = half_to_full(best_half)
    final_c = compute_c5_full(final_full)
    final_full = np.clip(final_full, 0.0, 1.0)

    return final_full.tolist(), float(final_c), int(n)
