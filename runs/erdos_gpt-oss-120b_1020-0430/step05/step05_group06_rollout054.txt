CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x): return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # project exactly via proj_box_sum just to be safe
    x = proj_box_sum(x, CURRENT_TARGET)
    return x
```

def subgradient_descent(h0, max_iters, alpha0, target, rng, deadline):

    h = h0.copy()
    # project initial to feasible (box+sum)
    h = proj_box_sum(h, target)
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        # convert lag index to k
        lag_idx = np.argmax(corr)
        k = lag_idx - (n - 1)
        # collect all ks near max within tolerance
        tol = 1e-12
        ks = np.where(corr >= max_corr - tol)[0] - (n - 1)
        # compute gradient as average
        grad = np.zeros_like(h)
        for kk in ks:
            grad += gradient_of_corr_at_k(h, kk)
        grad /= len(ks)
        # step size
        step = alpha0 / np.sqrt(t + 1)
        # update and project
        h_new = h - step * grad
        h = proj_box_sum(h_new, target)
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

    else:
        j = -k
        nj = n - j
        if nj:
            grad[j:] += (1.0 - h[:nj])
            grad[:nj] += -h[j:]
    return grad

def subgradient_descent(h0, max_iters, alpha0, target, rng, deadline):
    n = h0.size
    h = proj_box_sum(h0, target)
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        lag_idx = np.argmax(corr)
        # collect all k tied for max (within tiny tolerance)
        tol = 1e-12
        ks = np.where(corr >= max_corr - tol)[0] - (n - 1)
        grad = np.zeros_like(h)
        for k in ks:
            grad += gradient_of_corr_at_k(h, k)
        grad /= len(ks)
        step = alpha0 / np.sqrt(t + 1.0)
        h = proj_box_sum(h - step * grad, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = proj_box_sum(x, CURRENT_TARGET)
    return x

def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start:start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(
            h.astype(np.int32), (1 - h).astype(np.int32), mode="full"
        )
        O[n - 1] = 0
    return h.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98

    # Determine dimension
    if "initial_h_values" in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        n = h_parent.size
    else:
        n = 360
        h_parent = None
    target = n / 2.0

    best_h = None
    best_c = np.inf

    # parent candidate
    if h_parent is not None:
        c_par = compute_c5(h_parent)
        if c_par < best_c:
            best_c = c_par
            best_h = h_parent.copy()

    # generate starting vectors
    starts = []
    if h_parent is not None:
        starts.append(h_parent.copy())
    try:
        starts.append(greedy_binary(n, rng))
    except Exception:
        pass
    for _ in range(3):
        idx = rng.permutation(n)[:int(target)]
        hb = np.zeros(n, dtype=float)
        hb[idx] = 1.0
        starts.append(hb)
    for _ in range(2):
        ru = rng.random(n)
        ru = proj_box_sum(ru, target)
        starts.append(ru)

    # subgradient descent on each start
    for h0 in starts:
        if time.time() > deadline:
            break
        h_opt, c_opt = subgradient_descent(
            h0, max_iters=50000, alpha0=1.0, target=target, rng=rng, deadline=deadline
        )
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # random restarts while time permits
    while time.time() < deadline - 5.0:
        if rng.random() < 0.5:
            idx = rng.permutation(n)[:int(target)]
            h0 = np.zeros(n, dtype=float)
            h0[idx] = 1.0
        else:
            h0 = proj_box_sum(rng.random(n), target)
        h_opt, c_opt = subgradient_descent(
            h0, max_iters=20000, alpha0=1.0, target=target, rng=rng, deadline=deadline
        )
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # smooth‑max SLSQP refinement
    if best_h is not None and time.time() < deadline:
        h_ref = best_h.copy()
        for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000):
            if time.time() > deadline:
                break
            h_ref = slsqp_optimize(h_ref, beta, rng, max_iter=800)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()

    # final safety projection and evaluation
    best_h = proj_box_sum(np.clip(best_h, 0.0, 1.0), target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
