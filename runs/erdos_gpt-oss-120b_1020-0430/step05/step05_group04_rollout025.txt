def delta_for_swap(h, i, j, k):
    N = h.size
    delta = 0
    # j at p=j: h[j+k] is 1 by mismatched definition
    if j + k < N:

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h):
    """Exact C5 from correlation (used only for final reporting)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Redistribute values so that Σh = n/2 (in‑place). Works for binary or
    fractional arrays."""
    n = h.size
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
    return h


def compute_autocorr(h):
    """Linear autocorrelation via FFT (length 2n)."""
    n = h.size
    H = np.fft.fft(h, n=2 * n)
    conv = np.fft.ifft(H * np.conj(H)).real
    return conv


def get_worst_shift(conv):
    """Return (k, min_overlap, overlaps) for k≥1 giving the smallest overlap."""
    n = (len(conv) - 1) // 2
    overlaps = conv[n - 1 : n - 1 + n]          # overlaps for k = 0..n-1
    nonzero = overlaps[1:]                     # discard k = 0
    min_overlap = nonzero.min()
    k = int(np.argmin(nonzero) + 1)            # shift with minimal overlap
    return k, min_overlap, overlaps


def get_mismatch_sets(h, k):
    """Indices i where h[i]=1, h[i+k]=0 and j where h[j]=0, h[j+k]=1."""
    n = h.size
    valid = np.arange(n - k)
    hi = h[valid]
    hik = h[valid + k]
    S1 = valid[(hi == 1) & (hik == 0)]
    S0 = valid[(hi == 0) & (hik == 1)]
    return S1, S0


def delta_for_swap(h, i, j, k):
    """Change in overlap for shift k when swapping i (1→0) with j (0→1)."""
    n = h.size
    delta = 0
    # contribution from j at position j (j+k) – always 1 because j∈S0
    delta += 1
    # contribution from j at position j−k (if in range)
    if j - k >= 0:
        delta += h[j - k]
    # contribution loss from i at position i−k (if in range)
    if i - k >= 0:
        delta -= h[i - k]
    return int(delta)


def improve_h(h, rng, time_deadline):
    """Greedy local improvement targeting the current worst shift."""
    n = h.size
    best_h = h.copy()
    best_c = 1.0 - (2.0 / n) * get_worst_shift(compute_autocorr(h))[1]

    no_improve = 0
    while time.time() < time_deadline:
        conv = compute_autocorr(h)
        k, min_overlap, _ = get_worst_shift(conv)
        cur_c = 1.0 - (2.0 / n) * min_overlap
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        S1, S0 = get_mismatch_sets(h, k)
        if S1.size == 0 or S0.size == 0:
            break

        total_pairs = int(S1.size) * int(S0.size)
        samples = min(20000, total_pairs) if total_pairs else 0
        best_delta = 0
        best_pair = None

        for _ in range(samples):
            i = rng.choice(S1)
            j = rng.choice(S0)
            d = delta_for_swap(h, i, j, k)
            if d > best_delta:
                best_delta = d
                best_pair = (i, j)
                if best_delta == 2:
                    break

        if best_delta <= 0:
            no_improve += 1
            if no_improve >= 5:
                break
            # random perturbation to escape a plateau
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(h == 0)
            if ones.size and zeros.size:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i] = 0
                h[j] = 1
            continue

        # apply the improving swap
        i, j = best_pair
        h[i] = 0
        h[j] = 1
        no_improve = 0

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s * 0.98

    best_h = None
    best_c = np.inf
    best_n = None

    # --------------------------------------------------------
    # Determine candidate sizes
    # --------------------------------------------------------
    candidate_sizes = []

    if "initial_h_values" in globals() and initial_h_values is not None:
        n_parent = len(initial_h_values)
        candidate_sizes.append(n_parent)
        if n_parent * 2 <= 720:
            candidate_sizes.append(n_parent * 2)
        half = (n_parent // 2) // 2 * 2
        if half >= 120:
            candidate_sizes.append(half)

    extra = [480, 600, 720]
    for s in extra:
        if s not in candidate_sizes and s <= 720:
            candidate_sizes.append(s)

    # keep only reasonable even sizes
    candidate_sizes = sorted({s for s in candidate_sizes if s % 2 == 0 and 120 <= s <= 720})

    # --------------------------------------------------------
    # Process each size
    # --------------------------------------------------------
    for idx, n in enumerate(candidate_sizes):
        if time.time() >= overall_deadline:
            break

        # time share for remaining sizes
        remaining = len(candidate_sizes) - idx
        size_deadline = time.time() + (overall_deadline - time.time()) / remaining

        trials = 0
        while time.time() < size_deadline and trials < 4:
            trials += 1

            # ----- initialise a start vector for this size -----
            if (
                "initial_h_values" in globals()
                and initial_h_values is not None
                and n == len(initial_h_values)
            ):
                h0 = np.rint(initial_h_values).astype(float)
                # ensure exact number of ones
                target = n // 2
                diff = target - h0.sum()
                if diff > 0:
                    zeros = np.flatnonzero(h0 == 0)
                    h0[rng.choice(zeros, int(diff), replace=False)] = 1.0
                elif diff < 0:
                    ones = np.flatnonzero(h0 == 1)
                    h0[rng.choice(ones, int(-diff), replace=False)] = 0.0
            else:
                h0 = np.zeros(n, dtype=float)
                h0[rng.choice(n, n // 2, replace=False)] = 1.0

            # ----- run local improvement -----
            trial_deadline = min(
                size_deadline,
                time.time() + (size_deadline - time.time()) * 0.5,
            )
            h_opt, c_opt = improve_h(h0.copy(), rng, trial_deadline)

            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n

    # --------------------------------------------------------
    # Fallback (should never happen)
    # --------------------------------------------------------
    if best_h is None:
        best_n = 180
        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
        best_c = compute_c5(best_h)

    # --------------------------------------------------------
    # Final safety handling
    # --------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
