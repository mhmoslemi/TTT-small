import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def compute_c5(h):
    """Return max overlap integral (C5) for a sample array h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng, target=None):
    """Force Σh == target while staying in [0,1]."""
    n = h.size
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def slsqp_optimize(h0, beta, rng, max_iter=2000):
    """SLSQP with smooth‑max surrogate."""
    n = h0.size
    target = n / 2.0

    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # log‑sum‑exp for smooth max
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def sum_eq(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    constraints = {"type": "eq", "fun": sum_eq}
    res = minimize(
        smooth_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return adjust_sum_exact(x, rng, target)


def gradient_refine(h, rng, step=0.02, max_iter=30):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            up = n - k
            g = np.zeros_like(h)
            g[:up] += 1.0 - h[k:]      # derivative part 1
            g[k:] -= h[:up]            # derivative part 2
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(150):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def fractional_to_binary(h, rng):
    """Take the n/2 largest entries as ones."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return adjust_sum_exact(bin_h, rng)


def binary_targeted_improve(h, rng, deadline):
    """Greedy swaps that lower the worst overlap."""
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        # best 1 to remove
        neigh_one = neigh[ones]
        i = ones[neigh_one.argmin()]
        # best 0 to add
        neigh_zero = neigh[zeros]
        j = zeros[neigh_zero.argmax()]
        delta = neigh[i] - neigh[j]
        if delta < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        # try a few random alternatives
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h


def local_fractional_search(h, rng, deadline, moves=300):
    """Simple random local moves that accept improvements."""
    n = h.size
    best_c = compute_c5(h)
    while time.time() < deadline and moves > 0:
        moves -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or h[i] < 1e-12 or h[j] > 1 - 1e-12:
            continue
        # random delta respecting bounds
        max_delta = min(h[i], 1.0 - h[j])
        delta = max_delta * rng.random()
        if delta < 1e-8:
            continue
        h2 = h.copy()
        h2[i] -= delta
        h2[j] += delta
        c2 = compute_c5(h2)
        if c2 < best_c:
            h, best_c = h2, c2
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ------------------------------------------------------------------
    # Base size and initial vectors
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        n0 = h_parent.size
    else:
        n0 = 360
        h_parent = np.zeros(n0)
        h_parent[: n0 // 2] = 1.0
        rng.shuffle(h_parent)

    adjust_sum_exact(h_parent, rng)

    best_h, best_c, best_n = None, np.inf, None

    # ------------------------------------------------------------------
    # Build a modest set of start vectors
    # ------------------------------------------------------------------
    starts = [h_parent.copy(), fractional_to_binary(h_parent, rng)]

    for _ in range(5):
        # random fractional
        hf = rng.random(n0)
        hf *= (n0 / 2) / hf.sum()
        hf = np.clip(hf, 0.0, 1.0)
        adjust_sum_exact(hf, rng)
        starts.append(hf)

    for _ in range(5):
        # random binary
        hb = np.zeros(n0)
        hb[rng.choice(n0, n0 // 2, replace=False)] = 1.0
        starts.append(hb)

    betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
             2000.0, 5000.0, 10000.0, 20000.0, 50000.0]

    # ------------------------------------------------------------------
    # Optimize each start
    # ------------------------------------------------------------------
    for base in starts:
        if time.time() > deadline:
            break
        h = base.copy()
        for beta in betas:
            if time.time() > deadline:
                break
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=2500)
            except Exception:
                continue
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n0
        # gradient polishing
        h = gradient_refine(h, rng, step=0.02, max_iter=40)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n0
        # binary refinement
        hb = fractional_to_binary(h, rng)
        hb = binary_targeted_improve(hb, rng, deadline)
        c_b = compute_c5(hb)
        if c_b < best_c:
            best_h, best_c, best_n = hb.copy(), c_b, n0

    # ------------------------------------------------------------------
    # Upsample if time permits
    # ------------------------------------------------------------------
    if best_h is not None and time.time() < deadline:
        n_up = best_n * 2
        if n_up <= 720:
            xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            adjust_sum_exact(h_up, rng)
            for beta in betas:
                if time.time() > deadline:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=2500)
            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=50)
            h_up = local_fractional_search(h_up, rng, deadline, moves=500)
            hb_up = fractional_to_binary(h_up, rng)
            hb_up = binary_targeted_improve(hb_up, rng, deadline)
            c_up = compute_c5(hb_up)
            if c_up < best_c:
                best_h, best_c, best_n = hb_up, c_up, n_up

    # ------------------------------------------------------------------
    # Fallback if nothing succeeded
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = n0
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Final clean‑up and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
