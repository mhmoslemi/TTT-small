def binary_hill_climb(N, rng, time_limit):
    h = np.zeros(N, dtype=np.float64)
    ones_idx = rng.choice(N, N//2, replace=False)
    h[ones_idx] = 1.0
    zeros_idx = np.setdiff1d(np.arange(N), ones_idx)
    # compute initial maxc
    maxc = np.max(compute_correlation_fft(h)) * (2.0/N)
    best_h = h.copy()
    best_c = maxc
    T = 0.001
    cooling = 0.9999
    start = time.time()
    iter = 0
    while time.time() - start < time_limit:
        # randomly choose i from ones, j from zeros
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new maxc
        maxc_candidate = np.max(compute_correlation_fft(h)) * (2.0/N)
        delta = maxc_candidate - maxc
        if delta < 0 or rng.random() < math.exp(-delta / T):
            # accept
            maxc = maxc_candidate
            # update list
            # swap indices in arrays
            # remove i from ones and add to zeros; same for j
            # we can use numpy set operations
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            # better: maintain as python lists for O(1) removal using swap

            # However updates are cheap relative to FFT maybe. We'll accept O(N) set recompute for simplicity? N=512, not big.
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # record best

            best_c = maxc
            best_h = h.copy()
        # temperature update
        T *= cooling
        iter += 1
        # periodically recalc ones_idx to avoid drift due to floating errors; not needed due to binary
    return best_h, best_c

import time, math, numpy as np
from scipy.optimize import minimize, LinearConstraint

# ---------- basic tools ----------
def correlation_full_fft(h: np.ndarray) -> np.ndarray:
    """Linear (full) correlation of h with (1‑h) using FFT."""
    n = h.size
    n2 = 2 * n
    H = np.fft.fft(h, n=n2)
    G = np.fft.fft((1 - h)[::-1], n=n2)
    conv = np.fft.ifft(H * G).real
    conv[conv < 0] = 0.0
    return conv


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    n = h.size
    return np.max(correlation_full_fft(h)) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    eps = 1e-12
    diff = target - h.sum()
    # simple random redistribution
    it = 0
    while abs(diff) > eps and it < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > eps:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


# ---------- gradient utilities ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining max C5."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ---------- binary hill‑climb ----------
def binary_hill_climb(n: int, rng: np.random.Generator, time_limit: float):
    """Search for a good 0/1 pattern using simulated annealing."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    zeros = np.setdiff1d(np.arange(n), ones)
    ones_list = list(ones)
    zeros_list = list(zeros)

    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c

    T = 0.001            # initial temperature
    cooling = 0.9999
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c

        if delta < 0 or rng.random() < math.exp(-delta / T):
            # accept
            cur_c = new_c
            ones_list.remove(i)
            zeros_list.append(i)
            zeros_list.remove(j)
            ones_list.append(j)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

        T *= cooling
        it += 1
    return best_h, best_c


# ---------- fractional refinement ----------
def refine_hill_climb(h: np.ndarray, rng: np.random.Generator,
                     time_limit: float, init_scale: float = 0.5):
    """Guided stochastic hill‑climb with a time budget."""
    best_h, best_c = h.copy(), compute_c5(h)
    cur_h, cur_c = h.copy(), best_c
    scale = init_scale
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        grad = worst_gradient(cur_h)
        nh = guided_perturb(cur_h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
        if c < cur_c:
            cur_c, cur_h = c, nh
        if (it + 1) % 200 == 0:
            scale = max(scale * 0.9, 1e-4)
        it += 1
    return best_h, best_c


# ---------- soft‑max SLSQP ----------
def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 600) -> np.ndarray:
    """SLSQP refinement of the soft‑max smoothed C5."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    N = 512

    # ---- binary global search (≈45 % of the budget) ----
    binary_time = budget_s * 0.45
    best_bin, best_c_bin = binary_hill_climb(N, rng, binary_time)

    # ---- fractional fine‑tuning (≈35 % of the budget) ----
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining <= 0.0:
        best_h = best_bin
        final_c = compute_c5(best_h)
        return best_h.tolist(), float(final_c), int(N)

    # small random relaxation to leave the binary corners
    h_frac = best_bin.astype(float)
    h_frac += rng.uniform(-0.005, 0.005, size=N)
    h_frac = np.clip(h_frac, 0.0, 1.0)
    h_frac = enforce_sum(h_frac, rng)

    # hill‑climb with time budget
    refine_time = remaining * 0.6
    best_frac, best_c_frac = refine_hill_climb(h_frac, rng, refine_time, init_scale=0.6)

    # update global best
    if best_c_frac < best_c_bin:
        best_h, best_c = best_frac.copy(), best_c_frac
    else:
        best_h, best_c = best_bin.copy(), best_c_bin

    # ---- SLSQP soft‑max passes (≈15 % of the budget) ----
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    betas = (20., 80., 200., 800., 3000., 8000., 20000.)
    for beta in betas:
        if remaining < 5.0:
            break
        maxit = 800 if beta >= 200 else 500
        best_h = slsqp_optimize(best_h, beta, max_iter=maxit)
        cur_c = compute_c5(best_h)
        if cur_c < best_c:
            best_c = cur_c
        remaining = budget_s - (time.time() - start)

    # ---- final stochastic tweaks until time expires ----
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(N)
