def enforce_sum(h, target):
    # project h onto [0,1] and sum=target
    # using water-filling
    tol = 1e-12
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < tol:
            break
        # capacities for adding: 1 - h
        if diff > 0:
            caps = 1 - h
        else:
            caps = -h
        total = caps.sum()
        if total <= 0:
            break
        alpha = diff / total
        h = h + caps * alpha
        # clip after each step
        np.clip(h, 0, 1, out=h)
    # final correction: maybe adjust by random small shifts
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # small adjust
        i = np.random.default_rng().integers(0, h.size)
        # adjust limited by bounds
        if diff > 0:
            add = min(diff, 1 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
    return h

def binary_hill_climb(h_bin, rng, max_swaps=20000):

    n = h_bin.size

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol":1e-12, "disp":False})
    x = np.clip(res.x, 0., 1.)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Project h onto [0,1] and enforce Σh = target (water‑filling)."""
    tol = 1e-12
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        caps = 1.0 - h if diff > 0 else -h
        total = caps.sum()
        if total <= 0:
            break
        alpha = diff / total
        h = h + caps * alpha
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimization for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator, max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, target: float, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum(h, target, rng)
    return h


def binary_hill_climb(h_bin: np.ndarray, rng: np.random.Generator, time_limit: float):
    """Swap‑based hill‑climb for binary vectors using O(1) delta updates."""
    n = h_bin.size
    O_raw = np.correlate(h_bin, 1 - h_bin, mode="full")
    max_O = O_raw.max()
    ks = np.arange(-(n - 1), n, dtype=int)
    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))
    while time.time() < time_limit:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        ip = i + ks
        im = i - ks
        sum_i = np.where((ip >= 0) & (ip < n), h_bin[ip], 0) + np.where((im >= 0) & (im < n), h_bin[im], 0)
        jp = j + ks
        jm = j - ks
        sum_j = np.where((jp >= 0) & (jp < n), h_bin[jp], 0) + np.where((jm >= 0) & (jm < n), h_bin[jm], 0)
        delta = sum_i - sum_j
        new_max_O = np.max(O_raw + delta)
        if new_max_O < max_O - 1e-12:
            O_raw += delta
            max_O = new_max_O
            h_bin[i] = 0
            h_bin[j] = 1
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
    c5 = max_O * (2.0 / n)
    return h_bin.astype(float), c5
