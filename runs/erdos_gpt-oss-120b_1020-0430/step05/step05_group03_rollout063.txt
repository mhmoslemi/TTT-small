def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.time()
    rng = np.random.default_rng(seed)
    n = 512
    dx = 2.0 / n
    # initialize binary h0
    # if initial_h_values exists, use its values to get binary starting point:
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # pick top n//2 entries as ones
        idx_sorted = np.argsort(-h0)  # descending
        top = idx_sorted[:n//2]
        h = np.zeros(n, dtype=np.int8)
        h[top] = 1
    else:
        # random
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, n//2, replace=False)] = 1

    # compute initial convolution C
    C = np.convolve(h, h, mode='full')
    # ensure length 2n-1
    # compute current C5
    minC = C[n:].min()
    cur_c5 = ((n/2) - minC) * dx
    best_h = h.copy()
    best_c5 = cur_c5

    # maintain ones_arr and pos_index
    ones_arr = np.flatnonzero(h).astype(np.int64)  # length m
    pos_index = np.full(n, -1, dtype=np.int64)
    pos_index[ones_arr] = np.arange(ones_arr.size, dtype=np.int64)

    # set parameters for annealing/hill climb
    # For hill climbing: use simple greedy; if no improvement for many attempts, random restart
    attempts_without_improve = 0
    max_stale = 20000  # after this, random restart
    # We may also cool temperature for occasional uphill: implement small temperature schedule
    T = 0.001  # initial temperature
    # temperature decay per iteration maybe T *= 0.9999

    # loop
    while time.time() - start < budget_s * 0.97:
        # random swap
        # pick a random one
        idx_one = rng.integers(0, ones_arr.size)
        i = ones_arr[idx_one]
        # pick a random zero
        j = rng.integers(0, n)
        while h[j] == 1:
            j = rng.integers(0, n)

        # perform removal of i
        offs = i - ones_arr
        idx = offs + n - 1
        C[idx] -= 1
        mask = offs != 0
        if mask.any():
            C[-offs[mask] + n - 1] -= 1

        # update ones_arr: replace i with j
        pos_index[i] = -1
        pos_index[j] = idx_one
        ones_arr[idx_one] = j
        h[i] = 0
        h[j] = 1

        # perform addition of j
        offs2 = j - ones_arr
        idx2 = offs2 + n - 1
        C[idx2] += 1
        mask2 = offs2 != 0
        if mask2.any():
            C[-offs2[mask2] + n - 1] += 1

        # compute new C5
        minC_new = C[n:].min()
        new_c5 = ((n/2) - minC_new) * dx

        # decide acceptance
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # metropolis
            delta = new_c5 - cur_c5
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            attempts_without_improve = 0
