def binary_sa(h_init, rng, time_deadline):
    h = h_init.copy()
    n = h.size
    cur_corr = compute_corr_fft(h)
    cur_min_corr = cur_corr[n-1+1 : n-1+n].min()
    best_h = h.copy()
    best_min_corr = cur_min_corr
    T0 = (cur_min_corr) * 0.1 + 1.0   # some positive
    it = 0
    while time.time() < time_deadline:
        # temperature schedule
        T = T0 * (0.9995 ** it)
        # pick random 1 and 0 indices
        idx1 = rng.choice(np.flatnonzero(h))
        idx0 = rng.choice(np.flatnonzero(1 - h))
        # swap
        h[idx1] = 0
        h[idx0] = 1
        # compute new corr
        new_corr = compute_corr_fft(h)
        new_min_corr = new_corr[n-1+1 : n-1+n].min()
        delta = new_min_corr - cur_min_corr
        if delta > 0 or rng.random() < np.exp(delta / T):
            cur_corr = new_corr
            cur_min_corr = new_min_corr
            if new_min_corr > best_min_corr:
                best_min_corr = new_min_corr
                best_h = h.copy()
        else:
            # revert swap
            h[idx1] = 1
            h[idx0] = 0
        it += 1
        if it % 1000 == 0 and time.time() >= time_deadline:
            break
    # compute best_c5
    best_c5 = 1 - (2.0 / n) * best_min_corr
    return best_h, best_c5
```

def binary_sa_opt(h_init, rng, deadline):
    h = h_init.copy()
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    rev = h[::-1]
    F1 = np.fft.rfft(h, L)
    F2 = np.fft.rfft(rev, L)
    conv = np.fft.irfft(F1 * F2, L)
    # min corr
    min_corr = conv[n:2*n-1].min()
    best_h = h.copy()
    best_min_corr = min_corr
    # ones and zeros lists
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    # temperature
    T0 = max(10.0, min_corr * 0.1) # maybe
    it = 0
    while time.time() < deadline:
        T = T0 * (0.9995 ** it)
        # random swap
        idx_one = rng.integers(len(ones))
        idx_zero = rng.integers(len(zeros))
        i = ones[idx_one]
        j = zeros[idx_zero]
        # perform swap in h
        h[i] = 0
        h[j] = 1
        # recompute correlation
        rev = h[::-1]
        F1 = np.fft.rfft(h, L)
        F2 = np.fft.rfft(rev, L)
        conv = np.fft.irfft(F1 * F2, L)
        new_min_corr = conv[n:2*n-1].min()
        delta = new_min_corr - min_corr
        if delta > 0 or rng.random() < np.exp(delta / T):
            # accept
            min_corr = new_min_corr
            # update ones, zeros
            ones[idx_one] = j
            zeros[idx_zero] = i
            if new_min_corr > best_min_corr:
                best_min_corr = new_min_corr
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
        it += 1
        # optional: break if no improvement for many steps
    best_c5 = 1 - (2.0 / n) * best_min_corr
    return best_h, best_c5
```

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def adjust_sum_exact(h, rng, max_iter=2000):
    """Redistribute mass so that Σh = n/2 exactly while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5_fast(h):
    """Maximum overlap C5 for a step function sampled on a uniform grid."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    rev = h[::-1]
    Fh = np.fft.rfft(h, L)
    Fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(Fh * Fr, L)           # linear correlation (length L)
    # correlation for shifts 0 … n‑1  (center at index n‑1)
    corr = conv[n - 1 : n - 1 + n]            # shape (n,)
    O = h.sum() - corr                        # O_k = Σh – correlation_k
    max_O = O.max()
    return float(max_O * (2.0 / n))

# Main optimization routine
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97   # leave a small safety margin

    # --------------------------------------------------------
    # Grid setup (use n = 720 for a good trade‑off)
    # --------------------------------------------------------
    n = 720
    dx = 2.0 / n
    xs = np.linspace(0.0, 2.0, n, endpoint=False)

    # --------------------------------------------------------
    # Initial vector (interpolated parent if available)
    # --------------------------------------------------------
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        h = np.interp(xs, xs_parent, parent)
    else:
        h = np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)
    adjust_sum_exact(h, rng)

    best_h = h.copy()
    best_c = compute_c5_fast(best_h)

    # --------------------------------------------------------
    # Fourier‑based low‑dimensional optimisation (L‑BFGS‑B)
    # --------------------------------------------------------
    K = 12                                 # number of cosine modes
    ks = np.arange(1, K + 1)[:, None]     # shape (K,1)
    C = np.cos(np.pi * ks * xs)           # shape (K,n)
    C = C.T                                # shape (n,K)

    mu = 500.0                             # penalty weight for bound violations

    def penalty(vec):
        low = np.maximum(-vec, 0.0)
        high = np.maximum(vec - 1.0, 0.0)
        return np.sum(low * low + high * high)

    def obj_a(a):
        # a : (K,) coefficient vector
        vec = 0.5 + C @ a                  # construct h(x)
        p = mu * penalty(vec)
        c = compute_c5_fast(vec)
        return c + p

    # multiple random restarts while we have time
    ft_deadline = deadline * 0.80
    while time.time() < ft_deadline:
        a0 = rng.normal(scale=0.08, size=K)
        bounds = [(-0.6, 0.6)] * K
        res = minimize(
            obj_a,
            a0,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 2500, 'ftol': 1e-12, 'disp': False}
        )
        a_opt = res.x
        h_opt = 0.5 + C @ a_opt
        h_opt = np.clip(h_opt, 0.0, 1.0)
        adjust_sum_exact(h_opt, rng)
        c_opt = compute_c5_fast(h_opt)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
        if best_c < 0.3808:
            break

    # --------------------------------------------------------
    # Sub‑gradient refinement on the worst shift
    # --------------------------------------------------------
    # pre‑compute FFT size for the loop
    L_fft = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        # correlation of h with itself (linear)
        Fh = np.fft.rfft(best_h, L_fft)
        Fr = np.fft.rfft(best_h[::-1], L_fft)
        conv = np.fft.irfft(Fh * Fr, L_fft)
        corr = conv[n - 1 : n - 1 + n]          # correlation for shifts 0…n‑1
        O = best_h.sum() - corr
        k = int(O.argmax())                     # worst shift (0 ≤ k < n)
        if k == 0:
            grad = 1.0 - 2.0 * best_h
        else:
            # forward shift by k
            fwd = np.empty(n)
            fwd[: n - k] = best_h[k:]
            fwd[n - k :] = 0.0
            # backward shift by k
            bwd = np.empty(n)
            bwd[k:] = best_h[: n - k]
            bwd[: k] = 0.0
            grad = 1.0 - fwd - bwd

        # step‑size schedule (linear decay)
        elapsed = time.time() - start_time
        total = deadline - start_time
        eta = 0.5 * (1.0 - elapsed / total) + 0.001

        best_h = best_h - eta * grad
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)

        c_now = compute_c5_fast(best_h)
        if c_now < best_c:

        it += 1
        # early exit if we already beat the target
        if best_c < 0.3808:
            break

    # --------------------------------------------------------
    # Final sanitisation and return
    # --------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5_fast(best_h)

    return best_h.tolist(), float(final_c), int(n)
