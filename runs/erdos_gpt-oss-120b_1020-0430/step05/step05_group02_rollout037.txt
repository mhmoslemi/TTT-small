def binary_sa(n, rng, deadline):
    # initialize h random
    h = np.zeros(n, dtype=np.int8)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1
    # prepare lists and positions
    ones = list(ones_idx)
    zeros = [i for i in range(n) if h[i] == 0]
    pos_one = np.full(n, -1, dtype=np.int32)
    pos_zero = np.full(n, -1, dtype=np.int32)
    for p, idx in enumerate(ones):
        pos_one[idx] = p
    for p, idx in enumerate(zeros):
        pos_zero[idx] = p

    # compute initial c
    c = np.correlate(h, h, mode='full')
    cur_min = min(c[:n-1].min(), c[n:].min())
    cur_c5 = (n/2 - cur_min) * (2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5

    start = time.time()
    # temperature parameters
    T0 = 0.003  # maybe adjust
    while time.time() < deadline:
        # progress fraction
        progress = (time.time() - start) / (deadline - start)
        T = T0 * (1 - progress) + 1e-12

        if not ones or not zeros:
            break

        i = rng.choice(ones)
        j = rng.choice(zeros)
        # store previous min, c5
        prev_c5 = cur_c5

        # removal i
        # update c using h before change
        # idx i removal
        delta = -1
        # left part
        if i > 0:
            left_idx = np.arange(i) - i + (n - 1)
            c[left_idx] += delta * h[:i]
        # right part
        right_len = n - i
        c[np.arange(right_len) + (n - 1)] += delta * h[i:]

        # update lists and h
        # remove i from ones, add to zeros
        pos = pos_one[i]
        last_one = ones[-1]
        ones[pos] = last_one
        pos_one[last_one] = pos
        ones.pop()
        pos_one[i] = -1
        pos_zero[i] = len(zeros)
        zeros.append(i)

        h[i] = 0

        # addition j
        delta = +1
        if j > 0:
            left_idx = np.arange(j) - j + (n - 1)
            c[left_idx] += delta * h[:j]
        right_len = n - j
        c[np.arange(right_len) + (n - 1)] += delta * h[j:]

        # update lists and h
        pos = pos_zero[j]
        last_zero = zeros[-1]
        zeros[pos] = last_zero
        pos_zero[last_zero] = pos
        zeros.pop()
        pos_zero[j] = -1
        pos_one[j] = len(ones)
        ones.append(j)

        h[j] = 1

        # compute new min
        cur_min = c[:n-1].min()
        v2 = c[n:].min()
        if v2 < cur_min:
            cur_min = v2
        cur_c5 = (n/2 - cur_min) * (2.0 / n)

        # acceptance
        if cur_c5 <= prev_c5 or rng.random() < np.exp(-(cur_c5 - prev_c5) / T):
            # accept; update best
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert swap: reverse addition then removal
            # revert addition j (remove 1)
            delta = -1
            if j > 0:
                left_idx = np.arange(j) - j + (n - 1)
                c[left_idx] += delta * h[:j]
            right_len = n - j
            c[np.arange(right_len) + (n - 1)] += delta * h[j:]

            # update lists and h for revert j
            pos = pos_one[j]
            last_one = ones[-1]
            ones[pos] = last_one
            pos_one[last_one] = pos
            ones.pop()
            pos_one[j] = -1
            pos_zero[j] = len(zeros)
            zeros.append(j)

            h[j] = 0

            # revert removal i (add 1 back)
            delta = +1
            if i > 0:
                left_idx = np.arange(i) - i + (n - 1)
                c[left_idx] += delta * h[:i]
            right_len = n - i
            c[np.arange(right_len) + (n - 1)] += delta * h[i:]

            # update lists for i
            pos = pos_zero[i]
            last_zero = zeros[-1]
            zeros[pos] = last_zero
            pos_zero[last_zero] = pos
            zeros.pop()
            pos_zero[i] = -1
            pos_one[i] = len(ones)
            ones.append(i)

            h[i] = 1

            # cur_c5 remains prev_c5
            cur_c5 = prev_c5

    return best_h.astype(float), best_c5

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    iteration = 0
    # simple loop
    while abs(diff) > 1e-12 and iteration < 2000:
        i = rng.integers(0, n)
        # compute max allowable add
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 1e-12:
            h[i] += add
            diff = target - h.sum()
        iteration += 1
    # final check
    # may enforce small positivity/negativity clamping
    h[:] = np.clip(h, 0.0, 1.0)

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum exactly
    adjust_sum_exact(x, rng)
    return x

def binary_sa(n, rng, deadline):
    # Initialize random binary vector
    h = np.zeros(n, dtype=np.int8)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1
    # lists and position arrays
    ones = list(ones_idx)
    zeros = [i for i in range(n) if h[i] == 0]
    pos_one = np.full(n, -1, dtype=np.int32)
    pos_zero = np.full(n, -1, dtype=np.int32)
    for p, idx in enumerate(ones):
        pos_one[idx] = p
    for p, idx in enumerate(zeros):
        pos_zero[idx] = p
    # initial correlation h*h
    c = np.correlate(h, h, mode='full')
    # compute current C5
    min_corr = c[:n-1].min()
    v2 = c[n:].min()
    if v2 < min_corr:
        min_corr = v2
    cur_c5 = (n/2 - min_corr) * (2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5
    # precompute indices
    left_idx = [np.arange(i) - i + (n - 1) for i in range(n)]
    right_idx = [np.arange(n - i) + (n - 1) for i in range(n)]
    start = time.time()
    while time.time() < deadline:
        # compute temperature
        progress = (time.time() - start) / (deadline - start)
        T = 0.003 * (1.0 - progress) + 1e-12
        # pick random swap
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        prev_c5 = cur_c5
        # removal of i (1->0)
        delta = -1
        if i > 0:
            c[left_idx[i]] += delta * h[:i]
        c[right_idx[i]] += delta * h[i:]
        # update lists for removal i
        pos = pos_one[i]
        last_one = ones[-1]
        ones[pos] = last_one
        pos_one[last_one] = pos
        ones.pop()
        pos_one[i] = -1
        pos_zero[i] = len(zeros)
        zeros.append(i)
        h[i] = 0
        # addition of j (0->1)
        delta = 1
        if j > 0:
            c[left_idx[j]] += delta * h[:j]
        c[right_idx[j]] += delta * h[j:]
        # update lists for addition j
        pos = pos_zero[j]
        last_zero = zeros[-1]
        zeros[pos] = last_zero
        pos_zero[last_zero] = pos
        zeros.pop()
        pos_zero[j] = -1
        pos_one[j] = len(ones)
        ones.append(j)
        h[j] = 1
        # compute new C5
        min_corr = c[:n-1].min()
        v2 = c[n:].min()
        if v2 < min_corr:
            min_corr = v2
        cur_c5 = (n/2 - min_corr) * (2.0 / n)
        # acceptance
        if cur_c5 <= prev_c5 or rng.random() < np.exp(-(cur_c5 - prev_c5) / T):
            # accept
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert swap
            # revert addition j (remove 1)
            delta = -1
            if j > 0:
                c[left_idx[j]] += delta * h[:j]
            c[right_idx[j]] += delta * h[j:]
            # revert list for j removal from ones
            pos = pos_one[j]
            last_one = ones[-1]
            ones[pos] = last_one
            pos_one[last_one] = pos
            ones.pop()
            pos_one[j] = -1
            pos_zero[j] = len(zeros)
            zeros.append(j)
            h[j] = 0
            # revert removal i (add 1 back)
            delta = 1
            if i > 0:
                c[left_idx[i]] += delta * h[:i]
            c[right_idx[i]] += delta * h[i:]
            # revert list for i add to zeros
            pos = pos_zero[i]
            last_zero = zeros[-1]
            zeros[pos] = last_zero
            pos_zero[last_zero] = pos
            zeros.pop()
            pos_zero[i] = -1
            pos_one[i] = len(ones)
            ones.append(i)
            h[i] = 1
            cur_c5 = prev_c5
    # convert best_h to float array
    return best_h.astype(float), best_c5

import time, numpy as np
from scipy.optimize import minimize

# ----- global variables for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 2000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 1e-12:
            h[i] += add
            diff = target - h.sum()
        it += 1
    h[:] = np.clip(h, 0.0, 1.0)

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for __ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa(n, rng, deadline):
    # random balanced binary start
    h = np.zeros(n, dtype=np.int8)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1
    ones = list(ones_idx)
    zeros = [i for i in range(n) if h[i] == 0]
    pos_one = np.full(n, -1, dtype=np.int32)
    pos_zero = np.full(n, -1, dtype=np.int32)
    for p, i in enumerate(ones):
        pos_one[i] = p
    for p, i in enumerate(zeros):
        pos_zero[i] = p
    c = np.correlate(h, h, mode="full")
    # initial C5 from min autocorrelation
    min_corr = c[:n - 1].min()
    v2 = c[n:].min()
    if v2 < min_corr:
        min_corr = v2
    cur_c5 = (n / 2 - min_corr) * (2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5
    # pre‑compute index slices
    left_idx = [np.arange(i) - i + (n - 1) for i in range(n)]
    right_idx = [np.arange(n - i) + (n - 1) for i in range(n)]
    start = time.time()
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = 0.003 * (1.0 - progress) + 1e-12
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        prev_c5 = cur_c5
        # remove i
        delta = -1
        if i > 0:
            c[left_idx[i]] += delta * h[:i]
        c[right_idx[i]] += delta * h[i:]
        pos = pos_one[i]
        last_one = ones[-1]
        ones[pos] = last_one
        pos_one[last_one] = pos
        ones.pop()
        pos_one[i] = -1
        pos_zero[i] = len(zeros)
        zeros.append(i)
        h[i] = 0
        # add j
        delta = 1
        if j > 0:
            c[left_idx[j]] += delta * h[:j]
        c[right_idx[j]] += delta * h[j:]
        pos = pos_zero[j]
        last_zero = zeros[-1]
        zeros[pos] = last_zero
        pos_zero[last_zero] = pos
        zeros.pop()
        pos_zero[j] = -1
        pos_one[j] = len(ones)
        ones.append(j)
        h[j] = 1
        # evaluate
        min_corr = c[:n - 1].min()
        v2 = c[n:].min()
        if v2 < min_corr:
            min_corr = v2
        cur_c5 = (n / 2 - min_corr) * (2.0 / n)
        # accept / reject
        if cur_c5 <= prev_c5 or rng.random() < np.exp(-(cur_c5 - prev_c5) / T):
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert addition j
            delta = -1
            if j > 0:
                c[left_idx[j]] += delta * h[:j]
            c[right_idx[j]] += delta * h[j:]
            pos = pos_one[j]
            last_one = ones[-1]
            ones[pos] = last_one
            pos_one[last_one] = pos
            ones.pop()
            pos_one[j] = -1
            pos_zero[j] = len(zeros)
            zeros.append(j)
            h[j] = 0
            # revert removal i
            delta = 1
            if i > 0:
                c[left_idx[i]] += delta * h[:i]
            c[right_idx[i]] += delta * h[i:]
            pos = pos_zero[i]
            last_zero = zeros[-1]
            zeros[pos] = last_zero
            pos_zero[last_zero] = pos
            zeros.pop()
            pos_zero[i] = -1
            pos_one[i] = len(ones)
            ones.append(i)
            h[i] = 1
            cur_c5 = prev_c5
    return best_h.astype(float), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.96 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # if a parent construction is supplied, start from it
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        best_h = parent.copy()
        best_n = parent.size
        best_c = compute_c5(best_h)

    candidate_ns = [720, 360, 240, 180]
    for n in candidate_ns:
        if time.time() >= deadline:
            break
        # time allocation for this n
        remaining = deadline - time.time()
        if remaining < 30:
            break
        # binary simulated annealing (major work)
        sa_deadline = time.time() + min(remaining * 0.6, 300)
        h_bin, _ = binary_sa(n, rng, sa_deadline)
        c5 = compute_c5(h_bin)
        if c5 < best_c:
            best_h, best_c, best_n = h_bin.copy(), c5, n
        # gradient refinement on the binary result
        while time.time() < deadline:
            h_grad = gradient_refine(best_h.copy(), rng, step=0.01, max_iter=12)
            c5g = compute_c5(h_grad)
            if c5g < best_c - 1e-12:
                best_h, best_c = h_grad, c5g
                continue
            break
        # SLSQP cascade for final polishing
        if best_h is not None and best_h.size == n and time.time() < deadline:
            h_cur = best_h.copy()
            for beta in (20, 80, 200, 500, 1000, 2000, 5000, 10000):
                if time.time() >= deadline:
                    break

                    h_cur = slsqp_optimize(h_cur, beta, rng, max_iter=400)
                except Exception:

                if c5_s < best_c:
                    best_h, best_c = h_cur.copy(), c5_s
                # small gradient sweep after each beta
                h_cur = gradient_refine(h_cur, rng, step=0.01, max_iter=4)
                c5_s2 = compute_c5(h_cur)

                    best_h, best_c = h_cur.copy(), c5_s2
