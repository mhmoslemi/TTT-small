```
def compute_c5_bin(h):
    n = h.shape[0]
    # correlation (circular) of h with (1-h):
    # can compute via FFT: conv = ifft(fft(h) * fft(1-h)).real
    # but (1-h) FFT = delta function - fft(h). Actually easier: compute conv_self = ifft(fft(h)*fft(h)).real
    # then overlap = (n/2 - conv_self) * (2/n)
    h_fft = np.fft.rfft(h)
    conv_self = np.fft.irfft(h_fft * h_fft, n=n)
    min_self = np.rint(conv_self).astype(int).min()
    C = 1 - 2 * min_self / n
    return C, min_self
```

def anneal_one(start_h, total_iters, deadline):
    h = start_h.copy()
    # compute conv_self
    h_fft = np.fft.rfft(h)
    conv_self = np.rint(np.fft.irfft(h_fft * h_fft, n=n)).astype(np.int32)
    min_self = int(conv_self.min())
    best_local_h = h.copy()
    best_local_min = min_self
    # schedule
    T_start = 5.0
    T_end = 0.001
    for t in range(total_iters):
        if time.time() >= deadline:
            break
        # propose swap
        # choose i where h[i]=1
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)
        # compute delta
        i_plus = shift_plus[i]
        i_minus = shift_minus[i]
        j_plus = shift_plus[j]
        j_minus = shift_minus[j]

        # use int16 for array operations
        delta = -(h[i_plus].astype(np.int16) + h[i_minus].astype(np.int16)) + (h[j_plus].astype(np.int16) + h[j_minus].astype(np.int16))
        # corrections where j_plus == i or j_minus == i
        mask = (j_plus == i)
        delta[mask] -= 1
        mask2 = (j_minus == i)
        delta[mask2] -= 1

        # zero out k=0
        delta[0] = 0
        # fix middle if even
        if n % 2 == 0:
            k_mid = n // 2
            # compute proper delta for k_mid
            val_i_mid = h[(i + k_mid) % n]
            val_j_mid = h[(j + k_mid) % n]
            delta_mid = -int(val_i_mid) + int(val_j_mid)
            if (j + k_mid) % n == i:
                delta_mid -= 1
            delta[k_mid] = delta_mid

        new_conv = conv_self + delta
        new_min_self = int(new_conv.min())
        delta_s = new_min_self - min_self
        # temperature
        T = T_start * (T_end / T_start) ** (t / total_iters)
        accept = False
        if delta_s >= 0:
            accept = True
        else:
            if rng.random() < np.exp(delta_s / T):
                accept = True
        if accept:
            # swap bits
            h[i] = 0
            h[j] = 1
            conv_self = new_conv
            min_self = new_min_self
            if min_self > best_local_min:
                best_local_min = min_self
                best_local_h = h.copy()
        # optional early break if cannot improve further? We'll ignore.
    return best_local_h, best_local_min

def improve_binary_greedy(h, n, rng, deadline):
    # h is binary int8 array
    while time.time() < deadline:
        # compute conv_self and min_self
        h_fft = np.fft.rfft(h)
        conv_self = np.rint(np.fft.irfft(h_fft * h_fft, n=n)).astype(np.int32)
        min_self = conv_self.min()
        # find shifts achieving min_self
        worst_shifts = np.where(conv_self == min_self)[0]
        # pick a shift randomly
        k = rng.choice(worst_shifts)
        # neighbors: for each index i, define neigh = h[i] + h[(i+k)%n]
        # A swap 1<->0 that increases number of pairs with this shift should increase min_self
        # compute neighbor sum for each i: h[i] + h[i+k]
        # We want to turn a 1 with low neighbor into 0, and a 0 with high neighbor into 1.
        # Compute neighbor sum array
        neigh = h + h[(np.arange(n) + k) % n]
        # indices of ones and zeros
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        # among ones, find i where neigh[i] minimal
        i = ones[np.argmin(neigh[ones])]
        # among zeros, find j where neigh[j] maximal
        j = zeros[np.argmax(neigh[zeros])]
        # compute effect of swapping i and j on conv_self at shift k: currently
        # contribution to conv_self[k] from i: h[i] * h[i+k] = h[i+k] (since h[i]=1)
        # after swap: 0*... = 0, decrease by h[i+k]
        # contribution from j: h[j] * h[j+k] = 0; after swap becomes 1 * h[j+k] = h[j+k]; increase by h[j+k]
        # overall change = -h[i+k] + h[j+k]
        if - (h[(i + k) % n]) + h[(j + k) % n] > 0:
            # improvement, perform swap
            h[i] = 0
            h[j] = 1
            # continue loops
        else:
            # no improvement possible for this shift
            break
    return h

import time
import numpy as np

# ---------- helpers ----------
def compute_c5(h):
    """Exact overlap bound using full correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact_fractional(h, rng):
    """Project h onto the simplex Σh = n/2 with 0≤h≤1."""
    n = h.size
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(h, rng, step=0.02, max_iter=20):
    """Projected gradient descent on the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k != 0:
            up = n - k
            g[:up] += 1.0 - h[k:]        # ∂O/∂h_i for i in [0,up)
            g[k:] -= h[:up]              # ∂O/∂h_{i+k}
        # projected step
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(150):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def anneal_fractional(h0, total_iters, deadline, rng):
    """Simulated annealing on fractional h (0≤h≤1, Σh=n/2)."""
    h = h0.copy()
    n = h.size
    target = n / 2.0

    # initial score
    fft_h = np.fft.rfft(h)
    conv = np.fft.irfft(fft_h * fft_h, n=n)
    min_self = conv.min()
    if min_self < 0:
        min_self = 0.0
    cur_c = 1.0 - 2.0 * min_self / n
    best_h = h.copy()
    best_c = cur_c

    T0 = 0.2
    T1 = 1e-6

    for t in range(total_iters):
        if time.time() >= deadline:
            break

        # pick i with h[i]>0 and j with h[j]<1
        i = rng.integers(n)
        for _ in range(10):
            if h[i] > 1e-12:
                break
            i = rng.integers(n)
        j = rng.integers(n)
        for _ in range(10):
            if h[j] < 1.0 - 1e-12:
                break
            j = rng.integers(n)

        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue

        # temperature and step scaling
        ratio = (T0 * (T1 / T0) ** (t / total_iters)) / T0
        delta = max_delta * rng.random() * ratio
        if delta < 1e-12:
            continue

        # propose move
        h[i] -= delta
        h[j] += delta

        fft_h = np.fft.rfft(h)
        conv = np.fft.irfft(fft_h * fft_h, n=n)
        min_self_new = conv.min()
        if min_self_new < 0:
            min_self_new = 0.0
        new_c = 1.0 - 2.0 * min_self_new / n

                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
