import time
import numpy as np
from scipy.optimize import minimize

# ---------- global placeholders for constraints ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx, using linear correlation."""
    n = h.size
    dx = 2.0 / n
    # linear correlation via FFT (zero‑padded to 2n)
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.real(np.fft.ifft(H * np.conj(G)))  # length 2n
    return float(np.max(corr) * dx)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth max of overlaps via log‑sum‑exp (beta from CURRENT_BETA)."""
    n = x.size
    dx = 2.0 / n
    L = 2 * n
    H = np.fft.fft(x, L)
    G = np.fft.fft(1.0 - x, L)
    corr = np.real(np.fft.ifft(H * np.conj(G)))  # linear correlation
    ov = corr * dx                         # overlaps
    bc = CURRENT_BETA * ov
    m = np.max(bc)
    # log‑sum‑exp, stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def adjust_sum_exact(h, rng, max_iter=2000):
    """Force Σh = n/2 while staying in [0,1] by random tweaks."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=30):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    dx = 2.0 / n
    for it in range(max_iter):
        # linear correlation using FFT
        L = 2 * n
        H = np.fft.fft(h, L)
        G = np.fft.fft(1.0 - h, L)
        corr = np.real(np.fft.ifft(H * np.conj(G)))  # length 2n
        idx = np.argmax(corr)
        shift = idx - (n - 1)                # shift in [-(n-1), n-1]
        k = abs(shift)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]            # + (1 - h[i+k])
            g[k:] -= h[:up]                  # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # re‑balance sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(150):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def fractional_to_binary(h, rng):
    """Produce a binary vector with exactly n/2 ones by thresholding."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    adjust_sum_exact(bin_h, rng)
    return bin_h

def binary_targeted_improve(h, rng, deadline):
    """Greedy swapping to reduce the worst overlap (integer case)."""
    n = h.size
    while time.time() < deadline:
        # linear correlation (exact, integer)
        L = 2 * n
        H = np.fft.fft(h, L)
        G = np.fft.fft(1.0 - h, L)
        corr = np.real(np.fft.ifft(H * np.conj(G))) * (2.0 / n)
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift == 0:
            break
        k = abs(shift)
        # neighbours for the shift
        ar = np.arange(n)
        neigh = h[(ar + k) % n] + h[(ar - k) % n]

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        neigh_one = neigh[ones]
        min_val = neigh_one.min()
        cand_i = ones[neigh_one == min_val]
        i = rng.choice(cand_i)

        neigh_zero = neigh[zeros]
        max_val = neigh_zero.max()
        cand_j = zeros[neigh_zero == max_val]
        j = rng.choice(cand_j)

        # swap only if it improves the worst shift
        if neigh[i] - neigh[j] < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue

        # try a few random swaps as fallback
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

def slsqp_optimize(h0, beta, rng, max_iter=1800):
    """SLSQP on the smooth‑max surrogate for a given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    adjust_sum_exact(x, rng)
    return x
