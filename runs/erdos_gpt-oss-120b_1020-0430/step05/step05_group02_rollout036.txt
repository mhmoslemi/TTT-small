def compute_c5(h):
    n = h.shape[0]
    # zero pad to length 2*n
    H = np.fft.rfft(h, n=2*n)
    G = np.fft.rfft(1.0 - h, n=2*n)
    corr = np.fft.irfft(H * G, n=2*n)
    # correlation in "full" mode gives length 2*n-1; but using convolution orientation: corr[k] = sum_i h[i] * (1-h)[k-i]
    # Actually convolution of h and (1-h) reversed? Let's use np.convolve
    # Alternative: np.correlate is sum_i h[i] * (1-h)[i+k-N+1]? However the value of max is same as using convolution with reversed second argument.
    # Simpler: implement with np.convolve using h and (1-h)[::-1], then take appropriate slice.
    # Let's verify: np.correlate(h, 1-h, mode='full') = np.convolve(h, (1-h)[::-1], mode='full')
    # So we need to compute conv = np.fft.ifft(np.fft.fft(h, size) * np.fft.fft((1-h)[::-1], size))
    # Use real FFT.
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=2*n)
    R = np.fft.rfft(rev, n=2*n)
    conv = np.fft.irfft(H * R, n=2*n)
    # conv length 2*n
    # For full mode, length = 2*n -1, but irfft returns length 2*n. The last element is zero (due to zero padding). We can ignore last.
    max_val = conv[:2*n-1].max()
    return max_val * (2.0 / n)

def binary_sa_fixed(h_bin, rng, deadline, max_iter=500000):
    n = h_bin.size
    target = n // 2
    c = np.correlate(h_bin, h_bin, mode="full")
    # c length 2n-1
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    # Temperature schedule: T = init_T * (cool_factor ** it)
    # Let init_T = cur_worst * 0.01 + 1e-4 maybe.
    init_T = cur_worst * 0.02 + 1e-4
    cool = 0.99997 # slower cooling

    for it in range(max_iter):
        now = time.time()
        if now >= deadline:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta updates:
        # compute hi_vec (length n): contributions of i (1->0) to c:
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i+1:]))
        diff_i = -1.0
        c[start_i:start_i + n] += diff_i * hi_vec
        h_bin[i] = 0.0

        # compute contributions of j (0->1) after update of h_bin.
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j+1:]))
        diff_j = 1.0
        c[start_j:start_j + n] += diff_j * hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        # acceptance
        T = init_T * (cool ** it)
        if new_worst < cur_worst - 1e-12 or rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12)):
            # accept
            cur_worst = new_worst
            cur_min = new_min
            # update ones & zeros lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
        else:
            # reject: revert
            # revert j
            h_bin[j] = 0.0
            c[start_j:start_j + n] -= diff_j * hj_vec
            # revert i
            h_bin[i] = 1.0
            c[start_i:start_i + n] -= diff_i * hi_vec

    # return best_h (binary)
    return best_h, best_worst

def compute_c5(h):
    n = h.shape[0]
    # Use fft convolution: h * (1-h)[::-1]
    rev = (1.0 - h)[::-1]
    # Use rfft with length 2n
    H = np.fft.rfft(h, n=2*n)
    R = np.fft.rfft(rev, n=2*n)
    conv = np.fft.irfft(H * R, n=2*n)
    # conv length 2n
    max_val = conv[:2*n-1].max()
    return float(max_val * (2.0 / n))

def binary_sa_fixed(h_bin, rng, deadline, max_iter=500000):
    n = h_bin.size
    target = n // 2
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    init_T = max(1e-4, cur_worst * 0.02)
    cool = 0.99997

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # compute hi_vec before changing h_bin[i]
        # hi_vec length n: h_bin[i], h_bin[i-1],...,h_bin[0], h_bin[i+1],...,h_bin[n-1]
        hi_vec = np.empty(n, dtype=h_bin.dtype)
        hi_vec[:i+1] = h_bin[i::-1]
        hi_vec[i+1:] = h_bin[i+1:]

        start_i = n - 1 - i
        c[start_i:start_i + n] -= hi_vec  # diff_i = -1

        h_bin[i] = 0.0

        # compute hj_vec after i set to 0
        hj_vec = np.empty(n, dtype=h_bin.dtype)
        hj_vec[:j+1] = h_bin[j::-1]
        hj_vec[j+1:] = h_bin[j+1:]

        start_j = n - 1 - j
        c[start_j:start_j + n] += hj_vec  # diff_j = +1

        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        T = init_T * (cool ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            # accept with prob exp(-(new_worst - cur_worst)/T)
            if rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
        else:
            # revert
            # revert j
            h_bin[j] = 0.0
            c[start_j:start_j + n] -= hj_vec
            # revert i
            h_bin[i] = 1.0
            c[start_i:start_i + n] += hi_vec

    return best_h, best_worst

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)

    candidate_ns = [360, 240, 180]

    for n in candidate_ns:
        if time.time() >= deadline:
            break
        target = n // 2

        start_vectors = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vectors.append(interp)  # fractional

            idx = np.argpartition(interp, -target)[-target:]
            bin_vec = np.zeros(n, dtype=float)
            bin_vec[idx] = 1.0
            start_vectors.append(bin_vec)  # binary

        # random binary (two)
        for _ in range(2):
            bin_rand = np.zeros(n, dtype=float)
            bin_rand[rng.choice(n, target, replace=False)] = 1.0
            start_vectors.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        start_vectors.append(rand_frac)

        # alternating pattern
        alt_bin = np.tile([1.0, 0.0], target)
        if alt_bin.size < n:
            alt_bin = np.concatenate([alt_bin, np.zeros(n - alt_bin.size)])
        start_vectors.append(alt_bin)

        for base_h in start_vectors:
            if time.time() >= deadline:
                break
            h = np.clip(base_h, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            is_binary = np.all((h == 0) | (h == 1))

            if is_binary:
                # allocate iterations based on remaining time
                remaining = deadline - time.time()
                max_iter = int(min(800000, max(1000, remaining * 1500)))
                h_bin, raw_worst = binary_sa_fixed(h.copy(), rng, deadline, max_iter=max_iter)
                c5 = raw_worst * (2.0 / n)
                if c5 < best_c:
                    best_h, best_c, best_n = h_bin.astype(float).copy(), c5, n
                h = h_bin.astype(float)
            else:
                c5 = compute_c5(h)
                if c5 < best_c:
                    best_h, best_c, best_n = h.copy(), c5, n

            # gradient refinement
            h = gradient_refine_custom(h, rng, step=0.02, max_iter=20)
            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

            # SLSQP cascade (only for fractional vectors)
            if not is_binary:
                for beta in (
                    5.0,
                    20.0,
                    80.0,
                    200.0,
                    500.0,
                    1000.0,
                    2000.0,
                    5000.0,
                    10000.0,
                ):
                    if time.time() >= deadline:
                        break
                    try:
                        h = slsqp_optimize_custom(h, beta, rng, max_iter=800)
                    except Exception:
                        continue
                    c5 = compute_c5(h)
                    if c5 < best_c:
                        best_h, best_c, best_n = h.copy(), c5, n

                # final gradient pass
                h = gradient_refine_custom(h, rng, step=0.02, max_iter=20)
                c5 = compute_c5(h)
                if c5 < best_c:
                    best_h, best_c, best_n = h.copy(), c5, n

    # Upsample for extra refinement if possible
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        h_up = gradient_refine_custom(h_up, rng, step=0.02, max_iter=20)
        for beta in (
            5.0,
            20.0,
            80.0,
            200.0,
            500.0,
            1000.0,
            2000.0,
            5000.0,
            10000.0,
        ):
            if time.time() >= deadline:
                break
            h_up = slsqp_optimize_custom(h_up, beta, rng, max_iter=800)
        h_up = gradient_refine_custom(h_up, rng, step=0.02, max_iter=20)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # fallback
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
