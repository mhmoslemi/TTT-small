import cvxpy as cp
def solve_binary_milp(n, time_limit=None):
    h = cp.Variable(n, boolean=True)
    Y = cp.Variable((n, n), boolean=True)
    t = cp.Variable(nonneg=True)
    constraints = []
    # sum h = n/2
    constraints.append(cp.sum(h) == n/2)
    # product constraints
    # Y <= h_i
    # Y <= h_j
    # Y >= h_i + h_j - 1
    # We can vectorize:
    # For each i, j: constraints as above.
    # Use broadcasting: not straightforward in cvxpy; loops needed.
    for i in range(n):
        constraints.append(Y[i, i] == h[i])
    # For i<j constraints:
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(Y[i, j] <= h[i])
            constraints.append(Y[i, j] <= h[j])
            constraints.append(Y[i, j] >= h[i] + h[j] - 1)
            # Symmetry: Y[j,i] = Y[i,j] perhaps to reduce variables
            constraints.append(Y[j, i] == Y[i, j])
    # shift constraints
    half = n/2
    for k in range(1, n):
        indices = [(i, i+k) for i in range(n - k)]
        if not indices:
            continue
        sum_y = cp.sum(cp.vstack([Y[i, i+k] for i in range(n - k)]))
        constraints.append(sum_y >= half - t)
    # minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    if time_limit is not None:
        prob.solve(solver=cp.GLPK_MI, glpk={'tm_lim': int(time_limit*1000)}) # glpk expects ms?
    else:
        prob.solve(solver=cp.GLPK_MI)
    # retrieve solution if status ok
    if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
        h_val = np.round(h.value).astype(int)
        return h_val, float(t.value)
    else:
        return None, None

def binary_sa_hill_climb(h_bin, rng, max_swaps, first_temp=None):
    n = len(h_bin)
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h))
    zeros = list(np.flatnonzero(cur_h==0))
    # initial temperature perhaps proportional to cur_c (like 0.01*cur_c)
    init_T = first_temp if first_temp else max(0.01 * cur_c, 1e-4)
    for it in range(max_swaps):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            # accept
            cur_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # reject
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        adjust_sum_exact(x, rng)
    return x

def gradient_refine(h, rng, step=0.02, max_iter=30):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]           # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            adjust_sum_exact(h, rng)
    return h

def binary_sa_hill_climb(h_bin, rng, deadline, max_swaps=80000):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h))
    zeros = list(np.flatnonzero(1 - cur_h))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while it < max_swaps and time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
    return best_h, best_c

def greedy_improve(h, rng, max_iter=50, deadline=None):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

                adjust_sum_exact(h_bin, rng)

        adjust_sum_exact(x, rng)
    return x

def gradient_refine(h, rng, step=0.02, max_iter=30):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            adjust_sum_exact(h, rng)
    return h

def binary_sa_hill_climb(h_bin, rng, deadline, max_swaps=50000):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h))
    zeros = list(np.flatnonzero(1 - cur_h))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while it < max_swaps and time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
    return best_h, best_c

def greedy_improve(h, rng, max_iter=50, deadline=None):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if "initial_h_values" in globals() else None
    candidate_ns = [720, 600, 480, 360, 240, 180]
    candidate_ns = [n for n in candidate_ns if n % 2 == 0]

    beta_seq = (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000)

    for n in candidate_ns:
        if time.time() > deadline:
            break

        start_vecs = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            interp = np.clip(interp, 0.0, 1.0)
            adjust_sum_exact(interp, rng)
            start_vecs.append(interp)                     # fractional
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec.copy())              # binary

        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand.copy())                  # random binary

        rand_frac = rng.random(n)
        target = n / 2.0
        rand_frac *= target / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)
        start_vecs.append(rand_frac)                        # random fractional

        for base_h in start_vecs:
            if time.time() > deadline:
                break
            h = base_h.copy()
            adjust_sum_exact(h, rng)
            h = np.clip(h, 0.0, 1.0)

            # Binary local search
            if np.all((h == 0) | (h == 1)):
                now = time.time()
                time_left = deadline - now
                sa_time = min(time_left * 0.5, max(5.0, time_left - 20.0))
                sa_dead = now + sa_time
                est_swaps = max(20000, min(40000, int(sa_time / 0.001)))
                h_bin, c_bin = binary_sa_hill_climb(h.astype(int), rng, sa_dead, max_swaps=est_swaps)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h_bin = greedy_improve(h_bin, rng, max_iter=80, deadline=deadline)
                c_bin = compute_c5(h_bin)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin.astype(float)

            # Early gradient refinement
            h = gradient_refine(h, rng, step=0.02, max_iter=20)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # SLSQP cascade
            for beta in beta_seq:
                if time.time() > deadline:
                    break
                max_iter = 1000 if beta >= 2000 else 600
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # Final refinement
            h = gradient_refine(h, rng, step=0.02, max_iter=30)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # Greedy if binary at end
            if np.all((np.abs(h - 0) < 1e-12) | (np.abs(h - 1) < 1e-12)):
                h_bin = np.rint(h).astype(float)
                adjust_sum_exact(h_bin, rng)
                h_bin = greedy_improve(h_bin, rng, max_iter=80, deadline=deadline)
                c = compute_c5(h_bin)
                if c < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c, n
                h = h_bin

    # Optional up‑sampling if we still have time and n is modest
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=30)
        for beta in beta_seq:
            if time.time() > deadline:
                break
            max_iter = 1000 if beta >= 2000 else 600
            try:
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=max_iter)
            except Exception:
                pass
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=30)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
