import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 = max_k ∫ h (1-h(·+k)) dx."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


# ---------- deterministic hill‑climb on worst shift ----------
def hill_climb_worst(h: np.ndarray, rng: np.random.Generator,
                    max_iters: int = 200) -> np.ndarray:
    """Repeatedly move maximal feasible ε from best‑gradient to worst‑gradient
    index while it improves C5."""
    n = h.size
    for _ in range(max_iters):
        L = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(L) - (n - 1)
        k = abs(idx)
        grad = np.ones_like(h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:]       -= h[: n - k]
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        eps = min(h[i], 1.0 - h[j])
        if eps <= 1e-12:
            break
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        new_c = np.max(np.correlate(nh, 1.0 - nh, mode="full")) * (2.0 / n)
        old_c = np.max(L) * (2.0 / n)
        if new_c < old_c - 1e-12:
            h = nh
        else:
            break
    return h

    # ---------- up‑sample to 512 points if beneficial --------------------
    if best_h is not None and best_n < 512 and time.time() - start < budget_s * 0.95:
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, 512, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        h_up = hill_climb_worst(h_up, rng, max_iters=400)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, 512

    # ---------- final stochastic tweaks until time expires ---------------
    while time.time() - start < budget_s * 0.998:
        if best_h is None:
            break
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity
    if best_h is None:
        # fallback: simple half‑interval step function
        n = 512
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
