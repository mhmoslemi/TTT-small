    n = len(arr)

        i = rng.integers(0, n)

            cur_min = new_min
            cur_worst = target - cur_min
            improved = True
        else:
            # revert swap
            # revert via opposite operation
            # remove added contribution
            c[start_j:start_j+n] -= hj_vec
            h_bin[j] = 0.0
            # restore i
            c[start_i:start_i+n] += hi_vec
            h_bin[i] = 1.0
            # revert lists
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
    return h_bin, cur_worst

import time
import numpy as np

def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def adjust_sum_exact(arr, target, rng):
    diff = target - arr.sum()
    n = arr.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        maxadd = 1.0 - arr[i] if diff > 0 else -arr[i]
        add = min(diff, maxadd) if diff > 0 else max(diff, maxadd)
        arr[i] += add
        diff = target - arr.sum()
    np.clip(arr, 0.0, 1.0, out=arr)
    return arr

def binary_sa_timed(h_bin, rng, deadline, max_iter):
    n = h_bin.size
    target = n * 0.5
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst
    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))
    T0 = cur_worst * 0.05 + 1e-4
    decay = 0.99995
    it = 0
    no_improve = 0
    while it < max_iter and time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # remove i
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        c[start_i : start_i + n] -= hi_vec
        h_bin[i] = 0.0

        # add j
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j + 1 :]))
        c[start_j : start_j + n] += hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h_bin[j] = 0.0
            c[start_j : start_j + n] -= hj_vec
            h_bin[i] = 1.0
            c[start_i : start_i + n] += hi_vec
            no_improve += 1

        if no_improve > 50000:
            break
        it += 1
    return best_h.astype(float), best_worst

def greedy_improve_binary(h_bin, rng, max_steps):
    n = h_bin.size
    target = n * 0.5
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))
    for _ in range(max_steps):
        min_idx = np.argmin(c)
        k = min_idx - (n - 1)

        # try a deterministic improving swap
        i_swap, j_swap = None, None
        for i in ones:
            j = i + k
            if 0 <= j < n and h_bin[j] == 0:
                i_swap, j_swap = i, j
                break
        if i_swap is None:
            i_swap = rng.choice(ones)
            j_swap = rng.choice(zeros)

        i, j = i_swap, j_swap
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        c[start_i : start_i + n] -= hi_vec
        h_bin[i] = 0.0
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j + 1 :]))
        c[start_j : start_j + n] += hj_vec
        h_bin[j] = 1.0
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)

        new_min = c.min()
        if new_min > cur_min + 1e-12:
            cur_min = new_min
            cur_worst = target - cur_min
        else:
            # revert
            c[start_j : start_j + n] -= hj_vec
            h_bin[j] = 0.0
            c[start_i : start_i + n] += hi_vec
            h_bin[i] = 1.0
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
    return h_bin, cur_worst

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    best_h = None

    candidate_ns = [720, 540, 360]

    for n in candidate_ns:
        if time.time() >= deadline:

        target = n * 0.5

        start_vectors = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            idx = np.argpartition(interp, -int(target))[-int(target) :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vectors.append(bin_vec)

        for _ in range(2):
            bin_rand = np.zeros(n)
            bin_rand[rng.choice(n, int(target), replace=False)] = 1.0
            start_vectors.append(bin_rand)

        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        adjust_sum_exact(rand_frac, target, rng)
        idx = np.argpartition(rand_frac, -int(target))[-int(target) :]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        start_vectors.append(bin_vec)

        for base_h in start_vectors:
            if time.time() >= deadline:
                break
            h_bin = base_h.copy()

            # iteration budget per size
            if n >= 720:
                max_iter = 120000
            elif n >= 540:
                max_iter = 80000
            else:
                max_iter = 60000
            # scale down if very little time left
            if deadline - time.time() < 2.0:
                max_iter = int(max_iter * (deadline - time.time()) / 2.0)

            h_res, worst = binary_sa_timed(h_bin, rng, deadline, max_iter)
            c5 = worst * (2.0 / n)

    # optional up‑sampling
    if best_h is not None and time.time() < deadline:

        adjust_sum_exact(best_h, best_n * 0.5, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
