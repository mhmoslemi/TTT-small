import time
import math
import numpy as np
import cvxpy as cp

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx, using discrete correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Exact sum correction to enforce Σh = n/2 for binary/float vectors."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


# ---------- MILP attempt ----------
def attempt_milp(n: int, Omax_target: int, time_limit: float, rng: np.random.Generator):
    """
    Try to find a binary vector of length n with exactly n/2 ones
    such that for all k with n-k > Omax_target we have
        sum_{i=0}^{n-k-1} (x_i - x_i x_{i+k}) <= Omax_target.
    Returns the vector if found, otherwise None.
    """
    half = n // 2
    x = cp.Variable(n, boolean=True)
    constraints = [cp.sum(x) == half]

    # Only need constraints for k where the overlap length exceeds Omax_target.
    for k in range(1, n):
        L = n - k
        if L <= Omax_target:
            continue
        x_sub = x[:L]
        x_shift = x[k:]
        w = cp.Variable(L, boolean=True)
        constraints.append(w <= x_sub)
        constraints.append(w <= x_shift)
        constraints.append(w >= x_sub + x_shift - 1)
        constraints.append(cp.sum(x_sub - w) <= Omax_target)

    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.CBC, maxTime=time_limit, verbose=False)
    except Exception:
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE) or x.value is None:
        return None

    xv = np.rint(x.value).astype(float)
    cur = int(round(xv.sum()))
    if cur != half:  # tiny rounding fix if needed
        diff = half - cur
        if diff > 0:
            zeros = np.where(xv < 0.5)[0]
            xv[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.where(xv > 0.5)[0]
            xv[ones[: -diff]] = 0.0
    return xv


# ---------- greedy / hill‑climb fallback ----------
def greedy_binary(n: int, rng: np.random.Generator):
    """Construct a binary vector with n/2 ones by greedy minimisation of max O."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # overlap of all‑zero vector
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0  # enforce O_0 = 0
    return h


def hill_climb(h_bin: np.ndarray, deadline: float):
    """Simple hill‑climb on binary vectors (swap 1↔0)."""
    n = h_bin.size
    best_h = h_bin.copy()
    best_O = np.max(
        np.correlate(best_h.astype(np.int32), (1 - best_h).astype(np.int32), mode="full")
    )
    cur_h = h_bin.copy()
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    rng = np.random.default_rng()
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        cur_h[i] = 0
        cur_h[j] = 1
        O_new = np.max(
            np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode="full")
        )
        if O_new < best_O:
            best_O = O_new
            best_h = cur_h.copy()
            ones[i_idx] = j
            zeros[j_idx] = i
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h


# ---------- main entry ----------

        if rem < 20.0:

        time_limit = min(rem * 0.5, 300.0)

        if sol is not None:
            c = compute_c5(sol)
            if c < best_c:
                best_h = sol
                best_c = c
                best_n = n

    # ----- fallback greedy / hill‑climb if needed -----
    if best_h is None or best_c > target_c:
        # use a few different lengths, preferring the length of the parent if known
        candidate_ns = [360, 240, 300]
        for n in candidate_ns:
            if time.time() > deadline:

            # greedy construction
            h_bin = greedy_binary(n, rng)
            c_greedy = compute_c5(h_bin.astype(float))
            if c_greedy < best_c:

                best_c = c_greedy

            # hill‑climb refinement
            hc_dead = deadline
            h_hc = hill_climb(h_bin.copy(), hc_dead)
            c_hc = compute_c5(h_hc.astype(float))

    best_h = np.clip(best_h, 0.0, 1.0)
