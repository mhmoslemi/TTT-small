def max_corr_fft(h):
    n = h.size
    # zero pad to size 2*n
    size = 2*n
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]), n=size)
    G = np.fft.rfft(np.concatenate([1-h, np.zeros(n)]), n=size)
    corr = np.fft.irfft(H * np.conj(G), n=size)
    # The full correlation length 2n-1, ignore trailing zero? We'll just take first 2n-1 entries.
    return np.max(corr[:2*n-1])

def anneal_binary(h_bin, rng, time_budget):
    start = time.time()
    n = h_bin.shape[0]
    h = h_bin.copy()
    best = h.copy()
    best_val = max_corr_fft(h)
    cur_val = best_val
    # maintain lists
    ones = np.flatnonzero(h).tolist()
    zeros = np.flatnonzero(1 - h).tolist()
    # dictionaries for index -> position
    pos_one = {idx: i for i, idx in enumerate(ones)}
    pos_zero = {idx: i for i, idx in enumerate(zeros)}
    # temperature schedule: start with T0 = cur_val * 0.5 maybe
    T0 = cur_val * 0.5
    # iteration count for dynamic T might be slower.
    while time.time() - start < time_budget:
        # random choice
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_val = max_corr_fft(h)
        if new_val < cur_val:
            accept = True
        else:
            elapsed = time.time() - start
            T = T0 * (1 - elapsed / time_budget)
            if T <= 0:
                accept = False
            else:
                accept = rng.random() < np.exp(-(new_val - cur_val) / T)
        if accept:

            # update ones and zeros lists and dicts
            # i becomes zero, j becomes one
            # update ones list: replace i with j
            i_pos = pos_one[i]
            ones[i_pos] = j
            pos_one[j] = i_pos

        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # occasional occasional re-sync (if dict size mismatched)
        # not needed
    return best, best_val

def max_corr_fft(h):
    n = h.shape[0]
    size = 2 * n
    # use real FFT for speed
    h_pad = np.concatenate([h, np.zeros(n)], dtype=float)
    g_pad = np.concatenate([1.0 - h, np.zeros(n)], dtype=float)
    H = np.fft.rfft(h_pad)
    G = np.fft.rfft(g_pad)
    corr = np.fft.irfft(H * np.conj(G), n=size)
    # take first 2*n - 1 elements
    return np.max(corr[: 2 * n - 1])

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # While diff not close to zero, adjust a random index.
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def gradient_refine(h, rng, step=0.015, max_iter=10):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add

def anneal_binary(h0, rng, time_budget):
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_val = max_corr_fft(h)
    cur_val = best_val

    ones = np.flatnonzero(h).tolist()
    zeros = np.flatnonzero(1.0 - h).tolist()
    pos_one = {idx: i for i, idx in enumerate(ones)}
    pos_zero = {idx: i for i, idx in enumerate(zeros)}
    T0 = best_val * 0.5 if best_val > 0 else 1.0

    while time.time() - start < time_budget:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_val = max_corr_fft(h)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            elapsed = time.time() - start
            T = T0 * (1 - elapsed / time_budget)
            if T > 0:
                accept = rng.random() < np.exp(-(new_val - cur_val) / T)
        if accept:
            cur_val = new_val
            # update lists/dicts
            # i becomes zero
            i_pos = pos_one.pop(i)
            ones[i_pos] = j
            pos_one[j] = i_pos
            # j becomes one
            j_pos = pos_zero.pop(j)
            zeros[j_pos] = i
            pos_zero[i] = j_pos
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_val

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    return max_corr_fft(h) * (2.0 / n)


def max_corr_fft(h: np.ndarray) -> float:
    n = h.shape[0]
    size = 2 * n
    h_pad = np.concatenate([h, np.zeros(n, dtype=h.dtype)])
    g_pad = np.concatenate([1.0 - h, np.zeros(n, dtype=h.dtype)])
    H = np.fft.rfft(h_pad)
    G = np.fft.rfft(g_pad)
    corr = np.fft.irfft(H * np.conj(G), n=size)
    return np.max(corr[: 2 * n - 1])


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def anneal_binary(h0: np.ndarray, rng: np.random.Generator,
                  time_budget: float) -> tuple[np.ndarray, float]:
    start = time.time()
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_val = max_corr_fft(h)
    cur_val = best_val

    ones = np.flatnonzero(h).tolist()
    zeros = np.flatnonzero(1.0 - h).tolist()
    pos_one = {idx: i for i, idx in enumerate(ones)}
    pos_zero = {idx: i for i, idx in enumerate(zeros)}
    T0 = best_val * 0.5 if best_val > 0 else 1.0

    while time.time() - start < time_budget:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_val = max_corr_fft(h)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            elapsed = time.time() - start
            T = T0 * (1 - elapsed / time_budget)
            if T > 0:
                accept = rng.random() < np.exp(-(new_val - cur_val) / T)
        if accept:
            cur_val = new_val
            i_pos = pos_one.pop(i)
            ones[i_pos] = j
            pos_one[j] = i_pos
            j_pos = pos_zero.pop(j)
            zeros[j_pos] = i
            pos_zero[i] = j_pos
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.shape[0]
        idx = np.argsort(-init_h)[: n // 2]
        h_bin = np.zeros_like(init_h, dtype=float)
        h_bin[idx] = 1.0
    else:
        n = 512
        h_bin = np.zeros(n, dtype=float)
        h_bin[rng.choice(n, n // 2, replace=False)] = 1.0

    total_budget = budget_s

    def time_left():
        return total_budget - (time.time() - start)

    # Phase 1: binary simulated annealing (≈70% of budget)
    anneal_budget = total_budget * 0.70
    best_h_bin, _ = anneal_binary(h_bin, rng, anneal_budget)

    best_h = best_h_bin.astype(float)
    best_c5 = compute_c5(best_h)

    # Phase 2: continuous refinement with soft‑max SLSQP
    beta_seq = [10.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]
    for beta in beta_seq:
        if time_left() < 5.0:
            break
        max_iter = 400 if beta < 500 else 300
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        cur_c5 = compute_c5(best_h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
        # gentle gradient polishing
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=5)
        cur_c5 = compute_c5(best_h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5

    # Final greedy swap improvement (zero‑temperature hill climb)
    cur_val = max_corr_fft(best_h)
    while time_left() > 0.5:
        ones = np.flatnonzero(best_h > 0.0)
        zeros = np.flatnonzero(best_h < 1.0)
        if not len(ones) or not len(zeros):
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best_h[i] = 0.0
        best_h[j] = 1.0
        new_val = max_corr_fft(best_h)
        if new_val < cur_val:
            cur_val = new_val
            best_c5 = compute_c5(best_h)
        else:
            best_h[i] = 1.0
            best_h[j] = 0.0

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.size)
