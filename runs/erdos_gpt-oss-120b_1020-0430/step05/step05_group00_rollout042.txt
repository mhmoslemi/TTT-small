def binary_sa_fast(n, rng, time_budget, target=0.3808):
    dx = 2.0 / n
    start = time.time()
    best_h = None
    best_c = np.inf
    # SA schedule
    T0 = 0.05
    T_end = 1e-4
    while time.time() - start < time_budget:
        # random binary start
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, n//2, replace=False)
        h[ones] = 1.0
        # compute initial O array
        corr = np.correlate(h, 1 - h, mode='full')
        O = corr[n-1:] * dx  # length n
        c_curr = O.max()
        # maintain ones_arr and zeros_arr for selecting indices
        ones_arr = np.flatnonzero(h > 0.5)
        zeros_arr = np.flatnonzero(h < 0.5)
        # inner loop steps
        # decide number of steps based on remaining time
        # For simplicity, let's have a fixed inner steps but break if time exceeds
        inner_steps = 20000
        for step in range(inner_steps):
            elapsed = time.time() - start
            if elapsed >= time_budget:
                break
            # temperature
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            # select random 1 and 0
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]
            # compute delta_A and delta_B
            delta_A = np.zeros(n, dtype=np.float64)
            delta_A[: n - i] -= 1.0
            delta_A[: n - j] += 1.0
            delta_B = np.zeros(n, dtype=np.float64)
            delta_B[: n - i] -= h[i : n]  # h[i+i_s]
            delta_B[: n - j] += h[j : n]
            delta_B[: i + 1] -= h[i::-1]   # h[i-s]
            delta_B[: j + 1] += h[j::-1]
            delta_B[0] = 0.0
            delta_O = (delta_A - delta_B) * dx
            O_new = O + delta_O
            c_new = O_new.max()
            delta = c_new - c_curr
            accept = False
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True
            if accept:
                # apply update
                O = O_new
                c_curr = c_new
                # swap bits in h
                h[i] = 0.0
                h[j] = 1.0
                # update ones_arr and zeros_arr
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                # update best
                if c_curr < best_c:
                    best_c = c_curr
                    best_h = h.copy()
                    if best_c <= target:
                        return best_h, best_c
            # else reject: keep h and O unchanged
        # continue new start if time left
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Global vars for SLSQP soft‑max optimizer
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def binary_sa_fast(n: int, rng: np.random.Generator,
                   time_budget: float, target: float = 0.3808):
    """Fast simulated‑annealing for binary vectors using O(n) updates."""
    dx = 2.0 / n
    start = time.time()
    best_h = None
    best_c = np.inf
    T0 = 0.05
    T_end = 1e-4
    inner_steps = 50000

    while time.time() - start < time_budget:
        # random binary start
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0

        corr = np.correlate(h, 1 - h, mode="full")
        O = corr[n - 1:] * dx          # O[s] for s=0..n-1
        c_curr = O.max()

        ones_arr = np.flatnonzero(h > 0.5)
        zeros_arr = np.flatnonzero(h < 0.5)

        for step in range(inner_steps):
            elapsed = time.time() - start
            if elapsed >= time_budget:
                break
            T = T0 * (T_end / T0) ** (elapsed / time_budget)

            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = int(ones_arr[i_pos])
            j = int(zeros_arr[j_pos])

            # delta_A
            delta_A = np.zeros(n, dtype=np.float64)
            delta_A[: n - i] -= 1.0
            delta_A[: n - j] += 1.0

            # delta_B (double‑count s=0 later corrected)
            delta_B = np.zeros(n, dtype=np.float64)
            delta_B[: n - i] -= h[i:n]
            delta_B[: n - j] += h[j:n]
            delta_B[: i + 1] -= h[i::-1]
            delta_B[: j + 1] += h[j::-1]
            delta_B[0] = 0.0  # fix double count at s=0

            delta_O = (delta_A - delta_B) * dx
            O_new = O + delta_O
            c_new = O_new.max()
            delta = c_new - c_curr
            accept = False
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True
            if accept:
                O = O_new
                c_curr = c_new
                h[i] = 0.0
                h[j] = 1.0
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_curr < best_c:
                    best_c = c_curr
                    best_h = h.copy()
                    if best_c <= target:
                        return best_h, best_c
        # next restart
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # ---- initial solution from parent (if any) ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ---- binary simulated annealing (fast, main effort) ----
    n_sa = best_n if best_n is not None else 512
    time_binary = total_budget * 0.80
    h_bin, c_bin = binary_sa_fast(n_sa, rng, time_binary, target=0.3808)
    if h_bin is not None and c_bin < best_c:
        best_h, best_c, best_n = h_bin, c_bin, n_sa

    # ---- remaining time for continuous refinement ----
    remaining = total_budget - (time.time() - start)
    if remaining > 0 and best_h is not None:
        # guided hill‑climb (fast)
        max_iters = int(min(3000, remaining * 250))
        h = best_h.copy()
        h, c = guided_hill_climb(h, rng, max_iters=max_iters, step_scale=0.7)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # soft‑max SLSQP passes with increasing β
        for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
            if time.time() - start > total_budget * 0.95:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=800 if beta >= 200 else 500)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

        # gradient tidy‑up
        if time.time() - start < total_budget * 0.97:
            h = gradient_refine(h, rng, step=0.015, max_iter=10)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

    # ---- stochastic tweaking until budget closes ----
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ---- final sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
