def project_onto_capped_simplex(v, L):
    # ensure 0 ≤ x_i ≤ 1
    # goal: minimize ||x - v||^2 s.t. sum x = L, 0 ≤ x ≤ 1
    # Equivalent to find threshold λ such that x_i = min(1, max(0, v_i - λ)) and sum x_i = L.
    # Solve for λ using bisection.
    # Since function sum_i min(1, max(0, v_i - λ)) is decreasing in λ, we can binary search.
    # λ lower bound = min(v_i - 1) (since if λ too low, many v_i - λ >1 => clip to 1)
    # λ upper bound = max(v_i) (since if λ >= max(v_i), all become 0)
    # We'll perform binary search for number of iterations (e.g., 40)
    lo = np.min(v - 1)
    hi = np.max(v)
    for _ in range(40):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0, 1)
        s = np.sum(x)
        if s > L:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0, 1)
    # due to numeric error, adjust sum exactly:
    diff = np.sum(x) - L
    if abs(diff) > 1e-8:
        # adjust by distributing diff over entries not at bounds
        # find indices where 0 < x_i < 1
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if np.any(mask):
            x[mask] -= diff / np.sum(mask)
            x = np.clip(x, 0, 1)
        else:
            # fallback to adjust using random
            # use adjust_sum_exact
            # We'll ignore as should be small
            pass
    return x

def anneal_optimize(h0, L, deadline, rng):
    n = h0.size
    h = h0.copy()
    # ensure feasible
    # maybe already feasible
    current_c = compute_c5(h)
    best_h = h.copy()
    best_c = current_c
    # Estimate iteration time
    t0 = time.time()
    # Perform a small sample to estimate time per iter
    for _ in range(5):
        i = rng.integers(n); j = rng.integers(n)
        if i == j: continue
        max_inc = 1 - h[i]
        max_dec = h[j]
        if max_inc <= 0 or max_dec <= 0:
            continue
        delta = min(max_inc, max_dec) * rng.random()
        if delta < 1e-12:
            continue
        h[i] += delta; h[j] -= delta
        new_c = compute_c5(h)
        # revert for measurement
        h[i] -= delta; h[j] += delta
    time_per_iter = (time.time() - t0) / 5.0
    # Parameters
    total_iters = int((deadline - time.time()) / time_per_iter)
    if total_iters < 1:
        return best_h, best_c
    # Temperature schedule
    T0 = max( (best_c) * 0.05, 1e-3)  # maybe 0.02
    T_end = 1e-8

        if (it & 1023) == 0:

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Maximum overlap integral for discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact correction of Σh = n/2 by random local moves."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def greedy_binary(n, rng):
    """Simple greedy construction of a binary vector."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h.astype(float)

def anneal_optimize(h_init, deadline, rng):
    """Simulated‑annealing refinement (swap‑based)."""
    n = h_init.size
    h = h_init.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    start = time.time()
    T0 = max(0.02, best_c * 0.05)
    T_end = 1e-7

    while time.time() < deadline:
        # temperature by elapsed fraction
        frac = (time.time() - start) / (deadline - start)
        T = T0 * (T_end / T0) ** frac

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        if max_inc <= 0.0 or max_dec <= 0.0:
            continue
        delta = rng.random() * min(max_inc, max_dec)
        if delta < 1e-14:
            continue

        h[i] += delta
        h[j] -= delta
        new_c = compute_c5(h)

        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 360
        h0 = greedy_binary(n, rng)

    adjust_sum_exact(h0, rng)
    best_h = np.clip(h0, 0.0, 1.0)
    best_c = compute_c5(best_h)

    # ----- SLSQP cascade (smooth‑max) ---------------------------------
    if time.time() < deadline:
        h_float = best_h.copy()
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() >= deadline:
                break
            max_iter = 600 if beta >= 200 else 400
            try:
                h_float = slsqp_optimize(h_float, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            c_now = compute_c5(h_float)
            if c_now < best_c:
                best_c = c_now
                best_h = h_float.copy()

    # ----- Simulated annealing -----------------------------------------
    if time.time() < deadline:
        anneal_h, anneal_c = anneal_optimize(best_h, deadline, rng)
        if anneal_c < best_c:
            best_c = anneal_c
            best_h = anneal_h.copy()

    # ----- Final high‑beta SLSQP refinement -----------------------------
    if time.time() < deadline:
        h_float = best_h.copy()
        for beta in (2000.0, 5000.0, 8000.0):
            if time.time() >= deadline:
                break
            try:
                h_float = slsqp_optimize(h_float, beta, rng, max_iter=1000)
            except Exception:
                continue
            c_now = compute_c5(h_float)
            if c_now < best_c:
                best_c = c_now
                best_h = h_float.copy()

    # ----- safety clip & final evaluation -------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
