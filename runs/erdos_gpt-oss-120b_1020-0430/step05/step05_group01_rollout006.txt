def random_transfer(h, rng, scale):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j: return h.copy()
    # Move from i to j (positive direction)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
        return new_h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
