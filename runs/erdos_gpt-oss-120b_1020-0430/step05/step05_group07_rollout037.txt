def grad_worst_shift(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    s = idx  # signed shift
    g = np.zeros(n)
    if s >= 0:
        # first term: (j >= s)*(1 - h[j - s])
        if s < n:
            g[s:] += 1 - h[:n - s]
        # second term: -(j + s < n)*h[j + s]
        if s < n:
            g[:n - s] -= h[s:]
    else:
        t = -s
        # first term: (j < n - t)*(1 - h[j + t])
        if t < n:
            g[:n - t] += 1 - h[t:]
        # second term: -(j >= t)*h[j - t]
        if t < n:
            g[t:] -= h[:n - t]
    return g

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Maximum overlap for a step function sampled by h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Redistribute mass so that Σh = n/2 exactly (while staying in [0,1])."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k sum_i h_i (1-h_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp, stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP on the smooth max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (simple redistribution)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def grad_worst_shift(h):
    """Gradient of the overlap for the shift(s) attaining the maximal value."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(corr)
    # all shifts attaining the max (within a tiny tolerance)
    idxs = np.where(corr >= maxc - 1e-12)[0] - (n - 1)
    g = np.zeros_like(h)
    for s in idxs:
        if s >= 0:
            if s < n:
                g[s:] += 1.0 - h[: n - s]          # (j>=s)*(1 - h[j-s])
                g[: n - s] -= h[s:]                # -(j+s<n)*h[j+s]
        else:
            t = -s
            if t < n:
                g[: n - t] += 1.0 - h[t:]          # (j<n-t)*(1 - h[j+t])
                g[t:] -= h[: n - t]                # -(j>=t)*h[j-t]
    return g


def gradient_refine_fixed(h, rng, step=0.05, max_iter=20):
    """Projected gradient descent using the correct worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        g = grad_worst_shift(h)
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def fractional_sa_hill_climb(h, rng, deadline):
    """Simple transfer‑based simulated annealing on continuous vectors."""
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(cur.size, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 15000:
            break
    return best, best_c


def optimize_candidate(base_h, rng, deadline):
    """Apply a sequence of refinements to a starting vector."""
    h = base_h.copy()
    adjust_sum_exact(h, rng)

    # a few gradient steps (aggressive)
    h = gradient_refine_fixed(h, rng, step=0.07, max_iter=25)

    # short simulated‑annealing phase
    sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.15)
    h, _ = fractional_sa_hill_climb(h, rng, sub_dead)

    # SLSQP cascade with increasing smoothness
    betas = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 2000.0, 5000.0, 10000.0]
    for beta in betas:
        if time.time() > deadline:
            break
        max_iter = 1200 if beta >= 2000 else 800
        try:
            h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
        except Exception:
            continue

    # final fine‑grained gradient refinement
    h = gradient_refine_fixed(h, rng, step=0.04, max_iter=15)
    return h, compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    n = 360  # base resolution

    # possible starters
    starters = []

    # parent if available
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        starters.append(interp)

    # random binary starter
    bin_vec = np.zeros(n)
    bin_vec[rng.choice(n, n // 2, replace=False)] = 1.0
    starters.append(bin_vec)

    # random fractional (uniform)
    uf = rng.random(n)
    uf *= (n / 2.0) / uf.sum()
    uf = np.clip(uf, 0.0, 1.0)
    adjust_sum_exact(uf, rng)
    starters.append(uf)

    # random fractional (biased)
    bf = rng.random(n) ** 2
    bf *= (n / 2.0) / bf.sum()
    bf = np.clip(bf, 0.0, 1.0)
    adjust_sum_exact(bf, rng)
    starters.append(bf)

    # Dirichlet‑based starter (adds diversity)
    df = rng.dirichlet(alpha=np.ones(n)) * (n / 2.0)
    df = np.clip(df, 0.0, 1.0)
    adjust_sum_exact(df, rng)
    starters.append(df)

    # Main optimisation loop – keep trying new starts until time runs out
    while time.time() < deadline - 5.0:
        base_h = rng.choice(starters)
        # occasional perturbation to escape plateaus
        if rng.random() < 0.3:
            sigma = 0.08
            pert = base_h + rng.normal(0.0, sigma, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            adjust_sum_exact(pert, rng)
            base_h = pert

        h_opt, c_opt = optimize_candidate(base_h, rng, deadline)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # Upsampling refinement if time permits
    if best_h is not None and time.time() < deadline and best_h.size * 2 <= 720:
        n_up = best_h.size * 2
        xs = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        # higher‑resolution gradient + SA
        h_up = gradient_refine_fixed(h_up, rng, step=0.05, max_iter=30)
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
        h_up, _ = fractional_sa_hill_climb(h_up, rng, sub_dead)

        # final SLSQP passes (very smooth)
        for beta in [2000.0, 5000.0, 10000.0]:
            if time.time() > deadline:
                break
            try:
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=1500)
            except Exception:
                pass

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # Fallback in the extremely unlikely event that nothing was found
    if best_h is None:
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
