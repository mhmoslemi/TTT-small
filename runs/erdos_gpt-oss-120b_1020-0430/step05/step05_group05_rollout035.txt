def local_search_binary(h, rng, deadline):
    n = h.size
    sum_h = n // 2
    # compute conv using FFT
    conv = np.fft.ifft(np.fft.fft(h) * np.fft.fft(h).conj()).real
    conv = np.rint(conv).astype(np.int64)  # length n
    best_min = conv.min()
    # precompute list of indices of ones and zeros (update after each swap)
    while time.time() < deadline:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        # sample
        s = min(30, len(ones), len(zeros))
        sample_ones = rng.choice(ones, s, replace=False)
        sample_zeros = rng.choice(zeros, s, replace=False)
        # compute rolls
        # shape (s,n)
        shifts_one = np.stack([np.roll(h, -i) for i in sample_ones], axis=0)
        shifts_zero = np.stack([np.roll(h, -j) for j in sample_zeros], axis=0)
        # delta shape (s,s,n)
        delta = -shifts_one[:, None, :] + shifts_zero[None, :, :]   # int64
        # new min
        # broadcast conv to shape (1,1,n)
        new_min_mat = conv[None, None, :] + delta
        # compute min across last axis
        new_min = new_min_mat.min(axis=2)   # shape (s,s)
        # find best improvement
        max_new_min = new_min.max()
        if max_new_min <= best_min:
            # maybe random restart or small random perturbations to escape stagnation
            # Perform random swap to explore
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # apply anyway (but log? Not beneficial)
            # Skip
            continue
        # find indices
        idx = np.unravel_index(np.argmax(new_min), new_min.shape)
        i = sample_ones[idx[0]]
        j = sample_zeros[idx[1]]
        # apply swap
        h[i] = 0
        h[j] = 1
        # update conv using delta for selected pair
        delta_selected = delta[idx[0], idx[1], :]
        conv = conv + delta_selected
        # update best_min
        best_min = max_new_min
    return h, conv
```

import time
import numpy as np

def _compute_c5_fft(h: np.ndarray) -> float:
    """Exact C5 value using FFT (h can be 0/1 float)."""
    n = h.size
    sum_h = h.sum()
    conv = np.fft.ifft(np.fft.fft(h.astype(np.float64)) *
                       np.fft.fft(h.astype(np.float64)).conj()).real
    conv = np.rint(conv).astype(np.int64)
    min_conv = conv.min()
    return float((sum_h - min_conv) * (2.0 / n))


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # safety margin

    # -------------------------------------------------
    # target resolution (larger n gives finer possibilities)
    n_target = 720

    # -------------------------------------------------
    # build initial binary seed, preferably using the provided parent
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
        # binary version of parent (top half entries become 1)
        idx = np.argpartition(parent, -n0 // 2)[-n0 // 2 :]
        h0 = np.zeros(n0, dtype=np.int8)
        h0[idx] = 1
        if n_target != n0:
            if n_target % n0 == 0:
                factor = n_target // n0
                h0 = np.repeat(h0, factor)
            else:
                xs = np.linspace(0.0, 2.0, n0, endpoint=False)
                xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
                interp = np.interp(xt, xs, h0.astype(np.float64))
                idx2 = np.argpartition(interp, -n_target // 2)[-n_target // 2 :]
                h0 = np.zeros(n_target, dtype=np.int8)
                h0[idx2] = 1
        h = h0
    else:
        # pure random binary start
        h = np.zeros(n_target, dtype=np.int8)
        h[rng.choice(n_target, n_target // 2, replace=False)] = 1

    # -------------------------------------------------
    # globals for the search
    best_h = None
    best_c5 = np.inf
    best_min = -1  # max min(conv) seen so far

    # -------------------------------------------------
    # main multi‑restart loop
    max_stagnation = 3000          # iterations without improvement before restart
    sample_sz = 25                 # number of 1s/0s sampled each inner step
    target_min_needed = int(np.ceil((1 - 0.3808) * (n_target // 2)))  # 223 for n=720

    while time.time() < deadline:
        # optional early exit if we already reached the theoretical target
        if best_min >= target_min_needed:
            break

        # initialise a fresh binary vector for this restart
        h = np.zeros(n_target, dtype=np.int8)
        h[rng.choice(n_target, n_target // 2, replace=False)] = 1

        # compute its autocorrelation via FFT (integer values)
        fft_h = np.fft.fft(h.astype(np.float64))
        conv = np.rint(np.fft.ifft(fft_h * fft_h.conj()).real).astype(np.int64)
        cur_min = conv.min()
        stagn = 0

        while time.time() < deadline and stagn < max_stagnation:
            ones_idx = np.flatnonzero(h)
            zeros_idx = np.flatnonzero(h == 0)
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break

            s = min(sample_sz, ones_idx.size, zeros_idx.size)
            sample_ones = rng.choice(ones_idx, s, replace=False)
            sample_zeros = rng.choice(zeros_idx, s, replace=False)

            # rolls as int64 for exact arithmetic
            shifts_one = np.stack([np.roll(h, -i) for i in sample_ones], axis=0).astype(np.int64)
            shifts_zero = np.stack([np.roll(h, -j) for j in sample_zeros], axis=0).astype(np.int64)

            # delta[k] = -h[i+k] + h[j+k] for each sampled pair (i,j)
            delta = -shifts_one[:, None, :] + shifts_zero[None, :, :]          # shape (s, s, n)

            # candidate new minima for each pair
            new_min_mat = conv[None, None, :] + delta                        # shape (s, s, n)
            cand_min = new_min_mat.min(axis=2)                               # shape (s, s)

            # best improvement among the sampled pairs
            best_pair_val = cand_min.max()
            if best_pair_val > cur_min:
                pair_pos = np.unravel_index(np.argmax(cand_min), cand_pos := cand_min.shape)
                i = sample_ones[pair_pos[0]]
                j = sample_zeros[pair_pos[1]]

                # apply swap
                h[i] = 0
                h[j] = 1

                # update autocorrelation exactly
                conv = conv + delta[pair_pos[0], pair_pos[1], :]
                cur_min = best_pair_val
                stagn = 0

                # global best check
                c5_val = ((n_target // 2) - cur_min) * (2.0 / n_target)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h.copy()
                    best_min = cur_min
            else:
                stagn += 1

        # end of this restart – continue with a fresh one
        # (no explicit action needed)

    # -------------------------------------------------
    # fallback if nothing was discovered (should not happen)
    if best_h is None:
        best_h = np.zeros(n_target, dtype=np.int8)
        best_h[: n_target // 2] = 1
        best_c5 = _compute_c5_fft(best_h.astype(np.float64))

    # final clean‑up and conversion to the required format
    best_h = np.clip(best_h, 0, 1).astype(np.float64)
    final_c5 = _compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c5), int(n_target)
