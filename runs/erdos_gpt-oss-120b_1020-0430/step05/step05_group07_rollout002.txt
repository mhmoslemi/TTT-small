    n = h.size
    for it in range(max_iter):

        idx = np.argmax(corr) - (n-1)
        k = (idx) % n # we treat shift positive
        # compute mismatched indices
        # i such that h[i] > 0 (or > 1e-12) and h[(i+k)%n] < 1 (or < 1-1e-12)

    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    # Use SLSQP
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # repair sum exactly:
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def fractional_sa(h, rng, time_deadline):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    start_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < time_deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        # randomly decide direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = start_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            # revert
            cur[src] += delta
            cur[dst] -= delta
        # maybe early stop if no improvement for many iter? Not needed

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Maximum overlap for step function samples h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth max via log‑sum‑exp, scaled by CURRENT_BETA."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def adjust_sum_exact(h, rng, max_iter=2000):
    """Redistribute mass to satisfy Σh = n/2 exactly (maintaining bounds)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def binary_sa(h_bin, rng, deadline):
    """Swap‑based simulated annealing on binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    it = 0
    start_T = max(0.01 * cur_c, 1e-4)
    while time.time() < deadline:
        it += 1
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = start_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_c

def fractional_sa(h, rng, deadline):
    """Transfer‑based simulated annealing on continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    start_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = start_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            cur[src] += delta
            cur[dst] -= delta
    return best, best_c

def gradient_refine(h, rng, step=0.03, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:upper]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def worst_shift_smoothing(h, rng, max_iter=20):

def upsample(h, new_n):
    """Linear interpolation of h from its current length to new_n points on [0,2)."""
    old_n = h.size
    xs = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xt = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(xt, xs, h)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # Base sample sizes (all even)
    base_ns = [240, 300, 360, 480, 720]

    for n in base_ns:
        if time.time() > deadline:
            break

        # Build a modest pool of starting vectors
        start_vecs = []

        if 'initial_h_values' in globals():

            if parent.size != n:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                interp = np.interp(xt, xs, parent)
                start_vecs.append(interp)
                idx = np.argpartition(interp, -n // 2)[-n // 2:]
                bin_vec = np.zeros(n)
                bin_vec[idx] = 1.0
                start_vecs.append(bin_vec)
            else:
                start_vecs.append(parent.copy())
                bin_vec = np.zeros(n)
                bin_vec[: n // 2] = 1.0
                start_vecs.append(bin_vec)

        # Random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)

        # Random fractional (uniform)
        frac = rng.random(n)
        frac *= (n / 2) / frac.sum()
        frac = np.clip(frac, 0.0, 1.0)
        adjust_sum_exact(frac, rng)
        start_vecs.append(frac)

        # Random fractional (biased toward 0)
        frac2 = rng.random(n) ** 2
        frac2 *= (n / 2) / frac2.sum()
        frac2 = np.clip(frac2, 0.0, 1.0)
        adjust_sum_exact(frac2, rng)
        start_vecs.append(frac2)

        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # Binary SA if already binary
            if np.all((h == 0) | (h == 1)):
