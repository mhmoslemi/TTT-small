# globals used by the SLSQP soft‑max surrogate
CURRENT_TARGET = None
CURRENT_BETA = None
CURRENT_LAMBDA = 0.0


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # iteratively fix the deficit/excess using random feasible positions
    while abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1.0 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
        else:
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift that attains the maximum overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer a small amount from a positive‑gradient index to a negative‑gradient index."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1500, step_scale: float = 0.7):
    """Stochastic hill‑climb driven by the worst‑shift gradient."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def softmax_obj(x: np.ndarray) -> float:
    """Smooth approximation of max_k O_k with an optional binary‑penalty."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    sm = (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    penalty = CURRENT_LAMBDA * np.sum(x * (1.0 - x))
    return sm + penalty


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP for a given β (and global λ)."""
    global CURRENT_TARGET, CURRENT_BETA, CURRENT_LAMBDA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # correct any tiny sum error
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                cand = np.where(x < 1.0 - 1e-12)[0]
                if cand.size == 0:
                    break
                i = rng.choice(cand)
                add = min(diff, 1.0 - x[i])
            else:
                cand = np.where(x > 1e-12)[0]
                if cand.size == 0:
                    break
                i = rng.choice(cand)
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 6) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    cand = np.where(h < 1.0 - 1e-12)[0]
                    if cand.size == 0:
                        break
                    i = rng.choice(cand)
                    add = min(diff, 1.0 - h[i])
                else:
                    cand = np.where(h > 1e-12)[0]
                    if cand.size == 0:
                        break
                    i = rng.choice(cand)
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def upsample(h: np.ndarray, target_n: int, rng: np.random.Generator) -> np.ndarray:
    """Linear interpolation from len(h) to target_n, then enforce the integral constraint."""
    n = h.size
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, rng)
    return h_new


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # initialise from parent if available
    if "initial_h_values" in globals():
        hv = np.array(initial_h_values, dtype=float)
        hv = np.clip(hv, 0.0, 1.0)
        hv = enforce_sum(hv, rng)
        best_h, best_c, best_n = hv.copy(), compute_c5(hv), hv.size

    # helper to record a candidate
    def consider(hc, n):
        nonlocal best_h, best_c, best_n
        if hc < best_c - 1e-12:
            best_h, best_c, best_n = h.copy(), hc, n

    # ------------------------------------------------------------------
    # main multi‑resolution search
    cand_ns = [256, 384, 512]
    while time.time() - start < 0.78 * budget_s:
        n = int(rng.choice(cand_ns))
        # initialise h for this n
        if best_h is not None and best_n == n:
            h = best_h.copy()
        elif best_h is not None:
            h = upsample(best_h, n, rng)
        else:
            h = np.zeros(n)
            h[rng.choice(n, n // 2, replace=False)] = 1.0

        # deterministic hill‑climb
        h, c = max_gain_hill_climb(h, rng, max_iters=int(3000 * 256 / n))
        consider(c, n)

        # stochastic guided hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=int(1500 * 256 / n), step_scale=0.8)
        consider(c, n)

        # soft‑max SLSQP passes
        for beta in (200.0, 2000.0, 8000.0, 20000.0):
            if time.time() - start > 0.88 * budget_s:
                break
            CURRENT_LAMBDA = 0.001  # tiny binary‑penalty
            h = slsqp_optimize(h, beta, rng,
                               max_iter=800 if beta >= 2000 else 500)
            c = compute_c5(h)
            consider(c, n)

        # gradient polishing
        h = gradient_refine(h, rng, step=0.008, max_iter=8)
        c = compute_c5(h)
        consider(c, n)

        # a few random guided perturbations
        for _ in range(30):
            grad = worst_gradient(h)
            nh = guided_perturb(h, grad, rng, scale=0.02)
            nc = compute_c5(nh)
            if nc < best_c:
                h = nh
                consider(nc, n)

    # ------------------------------------------------------------------
    # Upsample to higher resolution if time permits
    if best_h is not None and time.time() - start < 0.95 * budget_s:
        target_n = 1024
        h = upsample(best_h, target_n, rng)

        # a compact deterministic climb at high resolution
        h, c = max_gain_hill_climb(h, rng, max_iters=int(1500 * 256 / target_n))
        consider(c, target_n)

        # short guided climb
        h, c = guided_hill_climb(h, rng, max_iters=int(800 * 256 / target_n), step_scale=0.7)
        consider(c, target_n)

        # one SLSQP refinement (moderate β)
        for beta in (2000.0, 8000.0, 20000.0):
            if time.time() - start > budget_s * 0.99:
                break
            CURRENT_LAMBDA = 0.0005
            h = slsqp_optimize(h, beta, rng, max_iter=500)
            c = compute_c5(h)
            consider(c, target_n)

        # final gradient tidy‑up
        h = gradient_refine(h, rng, step=0.005, max_iter=5)
        c = compute_c5(h)
        consider(c, target_n)

        best_h = h.copy()
        best_c = c
        best_n = target_n

    # ------------------------------------------------------------------
    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
