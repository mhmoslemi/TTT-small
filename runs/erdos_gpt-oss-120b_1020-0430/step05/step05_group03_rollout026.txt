def compute_c5_fft(h):
    n = h.size
    size = 2*n
    hpad = np.concatenate([h, np.zeros(n)])
    gpad = np.concatenate([1 - h, np.zeros(n)])
    H = np.fft.rfft(hpad)
    G = np.fft.rfft(gpad)
    corr = np.fft.irfft(H * np.conj(G), n=size)
    # take first 2n-1 entries
    corr = corr[:2*n-1]
    return np.max(corr) * (2.0 / n)

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    nh = symmetrize(nh)
    nh = enforce_sum(nh, rng)
    return nh

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution, LinearConstraint

# ---------- core utilities ----------
def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h and (1‑h) via FFT."""
    n = h.size
    size = 2 * n
    hpad = np.concatenate([h, np.zeros(n)])
    gpad = np.concatenate([1.0 - h, np.zeros(n)])
    H = np.fft.rfft(hpad)
    G = np.fft.rfft(gpad)
    corr = np.fft.irfft(H * np.conj(G), n=size)
    return corr[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    it = 0
    while abs(diff) > 1e-12 and it < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h


def symmetrize(h: np.ndarray) -> np.ndarray:
    """Mirror symmetry h(x)=h(2‑x)."""
    return (h + h[::-1]) / 2.0


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random feasible vector, then symmetrized."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)
    h = symmetrize(h)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift achieving the maximal overlap."""
    n = h.size
    corr = compute_corr_fft(h)
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    nh = symmetrize(nh)
    nh = enforce_sum(nh, rng)
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = symmetrize(h)
        h = enforce_sum(h, rng)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax_obj(x):
        corr = compute_corr_fft(x)
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = symmetrize(x)
    return enforce_sum(x, np.random.default_rng())


def de_optimize(n: int, rng: np.random.Generator,
                max_iter: int = 30) -> np.ndarray:
    """Differential evolution for a small n."""
    bounds = [(0.0, 1.0)] * n
    constraint = LinearConstraint(np.ones((1, n)), [n / 2.0], [n / 2.0])
    result = differential_evolution(compute_c5,
                                    bounds,
                                    constraints=(constraint,),
                                    maxiter=max_iter,
                                    polish=False,
                                    updating='deferred',
                                    seed=int(rng.random() * 2**32))
    return enforce_sum(result.x, rng)


def upsample(h_small: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from size len(h_small) to target_n."""
    n = h_small.size
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h_small)
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

                   scale: float = 0.01) -> np.ndarray:

    nh = h + noise
    nh = np.clip(nh, 0.0, 1.0)
    nh = symmetrize(nh)
    nh = enforce_sum(nh, rng)
