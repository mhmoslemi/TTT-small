def compute_c5_fft(h, dx):
    n = h.size
    nfft = 1 << ((2*n - 1).bit_length())  # next power of 2
    # compute FFT
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1 - h, n=nfft)
    # correlation via product H * conj(G)
    # but using rfft: correlation via IFFT of H * conj(G)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)
    # length nfft; linear correlation length = 2*n-1
    corr = corr[: 2*n - 1]
    idx = int(np.argmax(corr) - (n - 1))
    c5 = corr.max() * dx
    return c5, idx

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 using np.correlate."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def compute_c5_fft(h: np.ndarray, dx: float):
    """Fast C5 and worst–shift index using FFT."""
    n = h.size
    nfft = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1.0 - h, n=nfft)
    corr = np.fft.irfft(H * np.conj(G), n=nfft)
    corr = corr[: 2 * n - 1]
    idx = int(np.argmax(corr) - (n - 1))
    return float(corr.max() * dx), idx


def gradient_worst_shift(h: np.ndarray, idx: int):
    """Sub‑gradient of the worst shift overlap (no dx factor)."""
    n = h.size
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros_like(h)
    g[: n - k] = 1.0 - h[k:]
    g[k:] -= h[: n - k]
    return g


def enforce_sum(h: np.ndarray, rng: np.random.Generator):
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 2000:
        if diff > 0:
            i = rng.integers(0, n)
            cap = 1.0 - h[i]
            if cap <= 0:
                attempts += 1
                continue
            add = min(diff, cap)
        else:
            i = rng.integers(0, n)
            cap = h[i]
            if cap <= 0:
                attempts += 1
                continue
            add = -min(-diff, cap)
        h[i] += add
        diff = target - h.sum()
        attempts = 0
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def pgd_restart(h0: np.ndarray, rng: np.random.Generator,
                deadline: float, dx: float):
    """Projected sub‑gradient descent with transfer moves."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    while time.time() < deadline and it < 20000:
        c, idx = compute_c5_fft(h, dx)
        if c < best_c:
            best_c, best_h = c, h.copy()
        grad = gradient_worst_shift(h, idx)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            pos = np.where(grad > 1e-12)[0]
            neg = np.where(grad < -1e-12)[0]
            if pos.size and neg.size:
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_eps = min(h[i], 1.0 - h[j])
        if max_eps > 0:
            step = 0.5 / np.sqrt(it + 1)
            eps = rng.random() * max_eps * step
            h[i] -= eps
            h[j] += eps
            h[i] = np.clip(h[i], 0.0, 1.0)
            h[j] = np.clip(h[j], 0.0, 1.0)
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * 0.01
                h[i] -= eps
                h[j] += eps
        it += 1
    final_c = compute_c5(best_h)
    return best_h, final_c
