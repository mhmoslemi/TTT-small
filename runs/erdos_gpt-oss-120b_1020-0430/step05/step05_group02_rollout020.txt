def hill_climb(h, rng, deadline, max_iters=None):
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    n = len(h)
    while time.time() < deadline and (max_iters is None or it < max_iters):
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # amount δ
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        max_move = min(max_inc, max_dec)
        if max_move <= 1e-12:
            continue
        # choose δ random from (0, max_move)
        delta = rng.random() * max_move
        # apply
        h[i] += delta
        h[j] -= delta
        # evaluate
        c = compute_c5_fast(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta
        it += 1
    return best_h, best_c

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()

    while abs(diff) > 1e-12:
        i = rng.integers(0, n)

            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:

        diff = target - h.sum()

    start = time.time()
    deadline = start + 0.99 * budget_s

    parent = globals().get('initial_h_values', None)
    if parent is not None:
        parent = np.array(parent, dtype=float)

    best_h = None
    best_c = np.inf
    best_n = None

    candidate_ns = [360, 480, 720]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        bases = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            bases.append(np.interp(xt, xs, parent))

        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        bases.append(rand_frac)

        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        bases.append(bin_rand)

        alt_bin = np.tile([1.0, 0.0], n // 2)
        bases.append(alt_bin.astype(float))

        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        bases.append(0.5 + 0.5 * np.cos(np.pi * xs))

        tri_h = np.where(xs <= 1.0, xs, 2.0 - xs)
        bases.append(tri_h)

        for base in bases:
            if time.time() > deadline:
                break
            h = base.copy()
            np.clip(h, 0.0, 1.0, out=h)
            adjust_sum_exact(h, rng)

            h = gradient_refine(h, rng, step=0.04, max_iter=12)
            c = compute_c5_fft(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # hill‑climbing refinement (≈5 s per start)
            sub_dead = min(deadline, time.time() + 5.0)
            hc_h, hc_c = hill_climb(h, rng, sub_dead)
            if hc_c < best_c:
                best_h, best_c, best_n = hc_h.copy(), hc_c, n

    # final focused hill‑climb on the current best
    if best_h is not None and time.time() < deadline:
        best_h = gradient_refine(best_h, rng, step=0.02, max_iter=12)
        best_h = hill_climb(best_h, rng, deadline)[0]

    # optional up‑sampling for a last push
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 960
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        h_up = hill_climb(h_up, rng, deadline)[0]
        c_up = compute_c5_fft(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_n = n
        best_c = compute_c5_fft(best_h)

    np.clip(best_h, 0.0, 1.0, out=best_h)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
