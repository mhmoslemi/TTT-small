def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    best_c = np.inf
    best_h = None
    # baseline
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # enforce sum exactly
        h0 = enforce_sum(np.clip(h0,0,1), n/2, rng)
        c0 = compute_c5_general(h0)
        best_h, best_c = h0, c0
    # coarse DE
    M = 32
    target_M = M/2
    # initialize population
    popsize = 30
    # random initial pop: ensure sum constraint
    pop = []
    for _ in range(popsize):
        a = rng.random(M)
        a = enforce_sum(a, target_M, rng)
        pop.append(a)
    pop = np.array(pop)
    best_idx = np.argmin([compute_c5_block(a, M) for a in pop])
    best_a = pop[best_idx].copy()
    best_c_block = compute_c5_block(best_a, M)
    de_end_time = start + min(300, budget_s * 0.25)
    while time.time() < de_end_time:
        # generation
        for i in range(popsize):
            # select distinct a,b,c
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec, b_vec, c_vec = pop[a_idx], pop[b_idx], pop[c_idx]
            F = 0.8  # could random
            mutant = a_vec + F * (b_vec - c_vec)
            # clip
            mutant = np.clip(mutant, 0.0, 1.0)
            # crossover
            CR = 0.9
            cross = rng.random(M) < CR
            if not np.any(cross):
                cross[rng.integers(0,M)] = True
            trial = np.where(cross, mutant, pop[i])
            # enforce sum
            trial = enforce_sum(trial, target_M, rng)
            # evaluate
            trial_c = compute_c5_block(trial, M)
            if trial_c < compute_c5_block(pop[i], M):
                pop[i] = trial
                if trial_c < best_c_block:
                    best_c_block = trial_c
                    best_a = trial.copy()
        # optional early break if time near end
        if time.time() > de_end_time - 0.5:
            break
    # Upsample best_a to n=512
    repeat_factor = n // M
    h = np.repeat(best_a, repeat_factor)
    h = enforce_sum(np.clip(h,0,1), n/2, rng)
    best_h = h.copy()
    best_c = compute_c5_general(best_h)
    # SLSQP passes
    # set global target and beta later inside slsqp_optimize
    # We'll use beta list, maybe up to 1e5
    beta_list = [20., 200., 2000., 8000., 20000., 80000., 200000.]

            break
        h = slsqp_optimize(h, beta, rng, max_iter=800)
        c = compute_c5_general(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Guided hill climb refined
    while time.time() - start < budget_s * 0.95:
        # set max_iters based on remaining time proportionally
        remaining = budget_s - (time.time() - start)

import time, numpy as np
from scipy.optimize import minimize

# globals for soft‑max SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def compute_c5_block(a: np.ndarray, M: int) -> float:
    """C5 for a block‑constant vector of length M."""
    corr = np.correlate(a, 1.0 - a, mode="full")
    return np.max(corr) * (2.0 / M)

def enforce_sum(v: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust v so that sum(v)=target while staying in [0,1]."""
    diff = target - v.sum()
    n = v.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth max of overlaps for given beta."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP for a fixed β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, CURRENT_TARGET, rng)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.5):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512                      # final resolution
    target_n = n / 2.0

    # ----- baseline -------------------------------------------------
    best_h = None
    best_c = np.inf
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_n, rng)
        best_h = h0.copy()
        best_c = compute_c5(best_h)

    # ----- DE on block representation --------------------------------
    M = 32                       # number of blocks (divides 512)
    target_M = M / 2.0
    popsize = 30
    pop = []
    for _ in range(popsize):
        a = rng.random(M)
        a = enforce_sum(a, target_M, rng)
        pop.append(a)
    pop = np.array(pop)
    best_idx = np.argmin([compute_c5_block(a, M) for a in pop])
    best_a = pop[best_idx].copy()
    best_c_block = compute_c5_block(best_a, M)

    de_time = min(300.0, budget_s * 0.25)
    de_end = start + de_time
    F = 0.8
    CR = 0.9
    while time.time() < de_end:
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])
            mutant = np.clip(mutant, 0.0, 1.0)
            mask = rng.random(M) < CR
            if not np.any(mask):
                mask[rng.integers(0, M)] = True
            trial = np.where(mask, mutant, pop[i])
            trial = enforce_sum(trial, target_M, rng)
            trial_c = compute_c5_block(trial, M)
            cur_c = compute_c5_block(pop[i], M)
            if trial_c < cur_c:
                pop[i] = trial
                if trial_c < best_c_block:
                    best_c_block = trial_c
                    best_a = trial.copy()
        if time.time() > de_end - 0.5:
            break

    # ----- upsample to full resolution -------------------------------
    repeat = n // M
    h = np.repeat(best_a, repeat)
    h = enforce_sum(np.clip(h, 0.0, 1.0), target_n, rng)
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- SLSQP refinement (soft‑max) -------------------------------
    for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0, 80000.0, 200000.0):
        if time.time() - start > budget_s * 0.90:
            break
        h = slsqp_optimize(h, beta, rng, max_iter=800)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- final hill‑climb -----------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 0:
        iters = int(3000 * 256 / n)  # ~1500 for n=512
        h, c = guided_hill_climb(h, rng, max_iters=iters, step_scale=0.5)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- tiny stochastic tweaks until time runs out ---------------
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ----- final sanity ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_n, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
