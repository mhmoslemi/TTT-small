def subgradient_descent(h, rng, steps, step_init, target_sum, max_time=None):
    t0 = time.time()
    n = len(h)
    sum_h = target_sum # constant = n/2
    step = step_init
    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(steps):
        if max_time and time.time() - t0 > max_time:
            break
        # compute correlation
        conv = np.correlate(h, h, mode='full')
        S = sum_h - conv  # array length 2n-1
        maxS = S.max()
        # indices where S is near max (tolerance)
        idxs = np.where(S >= maxS - 1e-12)[0]
        # compute gradient sum across these shifts
        g = np.zeros_like(h)
        for idx in idxs:
            s = idx - (n - 1)
            # gradient for shift s
            if s >= 0:
                right = np.concatenate([h[s:], np.zeros(s)])
                left = np.concatenate([np.zeros(s), h[:n-s]])
            else:
                sp = -s
                right = np.concatenate([np.zeros(sp), h[:n-sp]])
                left = np.concatenate([h[sp:], np.zeros(sp)])
            g += 1 - right - left
        # average
        g /= len(idxs)
        # step
        h = h - step * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        h = project_sum(h, sum_h)
        # evaluate
        c = compute_c5(h)
        if c < best_val:
            best_val = c
            best_h = h.copy()
        # update step (e.g., exponential decay)
        step *= 0.9995  # slight decay
    return best_h, best_val

def slsqp_refine(h0, beta, target):
    n = len(h0)
    cons = {'type':'eq','fun': lambda x: np.sum(x) - target}
    bounds = [(0.,1.)]*n
    res = minimize(lambda x: smooth_max_obj(x, beta, target), h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':500,'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0,1)

    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap C5 for a discretised step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Adjust h so that sum(h) == target while keeping 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # one‑pass proportional redistribution (repeated a few times for stability)
    for _ in range(3):
        if diff > 0:
            cap = 1.0 - h
            cap_sum = cap.sum()
            if cap_sum < 1e-12:
                break
            h += diff * cap / cap_sum
        else:
            cap = h
            cap_sum = cap.sum()
            if cap_sum < 1e-12:
                break
            h += diff * cap / cap_sum  # diff is negative, so this subtracts
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def shift_arrays(s, h):
    """Return (right, left) arrays for gradient of shift s."""
    n = h.size
    if s >= 0:
        right = np.concatenate([h[s:], np.zeros(s)])
        left = np.concatenate([np.zeros(s), h[:n - s]])
    else:
        sp = -s
        right = np.concatenate([np.zeros(sp), h[:n - sp]])
        left = np.concatenate([h[sp:], np.zeros(sp)])
    return right, left

def subgradient_descent(h, rng, steps, step0, target):
    """Projected sub‑gradient descent on max_{k} S_k where S_k = target - conv."""
    n = h.size
    step = step0
    best_h = h.copy()
    best_val = compute_c5(h)
    sum_h = target
    for it in range(steps):
        # correlation of h with itself
        conv = np.correlate(h, h, mode="full")
        S = sum_h - conv
        maxS = S.max()
        # indices attaining (near) the maximum
        idxs = np.where(S >= maxS - 1e-12)[0]
        # gradient (average over worst shifts)
        g = np.zeros_like(h)
        for idx in idxs:
            s = idx - (n - 1)
            r, l = shift_arrays(s, h)
            g += 1.0 - r - l
        g /= len(idxs)
        # simple descent step
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        h = project_sum(h, sum_h)
        # evaluate
        cur_val = compute_c5(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        step *= 0.9995  # gradual decay
    return best_h, best_val

def smooth_max_obj(x, beta, target):
    """Smooth approximation of max_k S_k with S_k = target - conv."""
    conv = np.correlate(x, x, mode="full")
    S = target - conv
    m = S.max()
    return (m + np.log(np.sum(np.exp(beta * (S - m))))) / beta

def slsqp_refine(h0, beta, target, rng):
    """One SLSQP refinement minimizing the smooth max."""
    n = h0.size
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lambda x: smooth_max_obj(x, beta, target),
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 800, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_sum(x, target)
    return x

def run(seed=42, budget_s=1100, **kwargs):

    deadline = start + budget_s * 0.99

    # best solution seen so far

    # optional parent solution

            return project_sum(interp, target)

        # first try the interpolated parent

    best_h = project_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)
