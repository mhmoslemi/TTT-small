def project_to_sum_bounds(v, target, tol=1e-9, max_iter=100):
    lo = -np.max(v) - 1  # ensure v_i+lo <=0 for all i.
    hi = 1 - np.min(v) + 1 # ensure v_i+hi >=1 for all.
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        x = np.clip(v + mid, 0, 1)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + lo, 0, 1)

def binary_sa_optimize(x0, rng, deadline, max_no_improve=200000):
    n = x0.shape[0]
    m = n // 2
    # compute corr (int)
    corr = np.convolve(x0, x0, mode='full')
    center = n - 1
    # compute current c5
    min_corr = np.min(np.concatenate([corr[:center], corr[center+1:]]))
    cur_c5 = (m - min_corr) * (2.0 / n)
    best_c5 = cur_c5
    best_x = x0.copy()
    # sets
    ones = set(np.flatnonzero(x0).tolist())
    zeros = set(np.flatnonzero(1 - x0).tolist())
    it = 0
    last_improve = 0
    init_T = max(0.01 * cur_c5, 1e-4)
    while time.time() < deadline:
        # maybe occasional restart if stagnation
        if it - last_improve > max_no_improve:
            # restart with random binary
            x0 = np.zeros(n, dtype=int)
            sel = rng.choice(n, m, replace=False)
            x0[sel] = 1
            corr = np.convolve(x0, x0, mode='full')
            ones = set(sel.tolist())
            zeros = set(np.arange(n).tolist()) - ones
            min_corr = np.min(np.concatenate([corr[:center], corr[center+1:]]))
            cur_c5 = (m - min_corr) * (2.0 / n)
            it = 0
            last_improve = 0
            init_T = max(0.01 * cur_c5, 1e-4)
            continue

        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))

        # Prepare array of ones (excluding i)
        arr = np.array(list(ones), dtype=int)
        diff = arr - i
        mask = diff != 0
        diff_nz = diff[mask]
        idx1 = diff_nz + center
        idx2 = 2*center - idx1
        # apply removal
        corr[idx1] -= 1
        corr[idx2] -= 1

        # Build arr_wo_i for addition
        arr_wo_i = arr[mask]
        diff2 = arr_wo_i - j
        idx1b = diff2 + center
        idx2b = 2*center - idx1b
        corr[idx1b] += 1
        corr[idx2b] += 1

        # compute new_c5
        min_corr_new = np.min(np.concatenate([corr[:center], corr[center+1:]]))
        new_c5 = (m - min_corr_new) * (2.0 / n)

        # temperature
        T = init_T * (0.9995 ** it) + 1e-12
        accept = new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T)
        if accept:
            # update sets
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            cur_c5 = new_c5
            # update min_corr (not needed)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_x = np.zeros(n, dtype=int)
                best_x[list(ones)] = 1
                last_improve = it
        else:
            # revert removal/addition
            # revert addition (subtract what we added)
            corr[idx1b] -= 1
            corr[idx2b] -= 1
            # revert removal (add back what we removed)
            corr[idx1] += 1
            corr[idx2] += 1
        it += 1
    return best_x, best_c5

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap for step function samples h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_to_sum_bounds(v, target, tol=1e-10, max_iter=80):
    """Euclidean projection of v onto {0≤x≤1, sum x = target}."""
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + lo, 0.0, 1.0)

def gradient_refine(h, step=0.03, max_iter=20):
    """Projected gradient descent on the max‑overlap objective."""
    n = h.shape[0]
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        sz = step * (0.5 ** it)
        h = h - sz * g
        h = project_to_sum_bounds(h, target)
    return h

def binary_sa_optimize(x0, rng, deadline, max_no_improve=200000):
    """Swap‑based simulated annealing on binary vectors (sum = n/2)."""
    n = x0.shape[0]
    m = n // 2
    cur = x0.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    ones = list(np.flatnonzero(cur))
    zeros = list(np.flatnonzero(1 - cur))
    init_T = max(0.01 * cur_c5, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        if it - last_improve > max_no_improve:
            # random restart
            cur[:] = 0
            sel = rng.choice(n, m, replace=False)
            cur[sel] = 1
            cur_c5 = compute_c5(cur)
            ones = list(np.flatnonzero(cur))
            zeros = list(np.flatnonzero(1 - cur))
            last_improve = it
            continue
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        cur[i] = 0
        cur[j] = 1
        new_c5 = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T)
        if accept:
            cur_c5 = new_c5
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
                last_improve = it
        else:
            # revert swap
            cur[i] = 1
            cur[j] = 0
        it += 1
    return best, best_c5
