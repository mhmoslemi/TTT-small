def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # if initial present, use it
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        init = np.clip(init, 0, 1)  # ensure bounds
        init = enforce_sum(init, rng)
        best_h = init.copy()
        best_c = compute_c5(best_h)
        best_n = best_h.size
    else:
        best_h = None

    # stage 1: low resolution random restarts
    low_ns = [128, 192, 256]
    # keep total time under e.g., 0.5*budget
    while time.time() - start < 0.45 * budget_s:
        n = int(rng.choice(low_ns))
        # generate random binary h
        h = np.zeros(n)
        h[rng.choice(n, n//2, replace=False)] = 1.0
        # hill climb moderate
        h, c = guided_hill_climb(h, rng, max_iters=int(2500 * 128 / n), step_scale=0.9)
        # optionally SLSQP
        for beta in (20., 200., 2000., 8000.):
            if time.time() - start > 0.45 * budget_s:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=400)
            c = compute_c5(h)
        # gradient refine
        h = gradient_refine(h, rng, step=0.02, max_iter=8)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # stage 2: upscale to 512 if not already
    target_n = 512
    if best_n != target_n:
        # upscale via linear interpolation
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0, 1)
        h_up = enforce_sum(h_up, rng)
    else:
        h_up = best_h.copy()

    # stage 2: high resolution refinement
    # apply hill climbing with many iterations
    h_up, c_up = guided_hill_climb(h_up, rng,
                                   max_iters=int(15000 * 256 / target_n), step_scale=0.8)
    if c_up < best_c:
        best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # SLSQP passes with increasing beta
    for beta in (20., 200., 2000., 8000., 20000., 50000.):
        if time.time() - start > 0.95 * budget_s:
            break
        h_up = slsqp_optimize(h_up, beta, rng, max_iter=800 if beta > 2000 else 500)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # gradient refine heavily
    h_up = gradient_refine(h_up, rng, step=0.005, max_iter=30)
    c_up = compute_c5(h_up)
    if c_up < best_c:
        best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # final stochastic tweaking (small perturbations)
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.01)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    best_h = np.clip(best_h, 0, 1)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.7):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        # periodically decay scale
        if (it+1) % decay == 0:
            scale = max(scale * 0.8, 1e-4)
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h)=len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- SLSQP with soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)

            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.8, 1e-4)
    return best_h, best_c

        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
    else:
        h_up = best_h.copy()

    # ---------- high‑resolution refinement ----------
    # aggressive hill‑climb
    h_up, c_up = guided_hill_climb(
        h_up, rng, max_iters=int(8000 * 256 / target_n), step_scale=0.8
    )
    if c_up < best_c:
        best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # gradient polish
    h_up = gradient_refine(h_up, rng, step=0.015, max_iter=30)
    c_up = compute_c5(h_up)
    if c_up < best_c:
        best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # soft‑max SLSQP passes with higher betas
    for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0, 80000.0):
        if time.time() - start > 0.90 * budget_s:
            break
        max_it = 800 if beta > 2000 else 500
        h_up = slsqp_optimize(h_up, beta, rng, max_iter=max_it)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # final gradient tidy‑up
    h_up = gradient_refine(h_up, rng, step=0.005, max_iter=30)
    c_up = compute_c5(h_up)
    if c_up < best_c:
        best_h, best_c, best_n = h_up.copy(), c_up, target_n

    # ---------- simulated‑annealing style tweaking ----------
    cur_h, cur_c = best_h.copy(), best_c
    temp = 0.02
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(cur_h)
        nh = guided_perturb(cur_h, grad, rng, scale=temp)
        nc = compute_c5(nh)
        if nc < cur_c or rng.random() < np.exp((cur_c - nc) / (temp * 0.1)):
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        temp = max(temp * 0.9995, 0.0005)

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
