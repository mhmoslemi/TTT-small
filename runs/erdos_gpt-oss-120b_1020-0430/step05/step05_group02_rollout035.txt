import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_sum_bounds(h, target, tol=1e-12, max_iter=30):
    """Euclidean projection onto [0,1]^n with sum == target."""
    # binary search for lambda such that sum(clamp(h - λ,0,1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(h - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > tol:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_worst_shift(h, k):
    n = h.size
    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        up = n - k
        g[:up] += 1.0 - h[k:]
        g[k:] += -h[:up]
    else:  # k < 0
        kk = -k
        up = n - kk
        g[kk:] += 1.0 - h[:up]
        g[:up] += -h[kk:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.99 * budget_s

    # ----- choose discretisation size -----
    n = 360
    target = n / 2.0

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            xs = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(xt, xs, h0)
        h = np.clip(h0, 0.0, 1.0)
    else:
        h = rng.random(n)
        h = h * target / h.sum()
    h = project_sum_bounds(h, target)

    # small random jitter to escape potential stagnation
    jitter = (rng.random(n) - 0.5) * 0.02
    h = np.clip(h + jitter, 0.0, 1.0)
    h = project_sum_bounds(h, target)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ----- main gradient‑descent on worst shift -----
    step = 0.5
    beta = 0.9985  # step decay
    it = 0
    no_imp = 0
    max_iters = 30000

    while time.time() < deadline and it < max_iters:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = corr.argmax()
        k = max_idx - (n - 1)               # shift corresponding to worst overlap
        g = gradient_worst_shift(h, k)

        # normalise step direction
        norm = np.linalg.norm(g)
        if norm == 0:
            break
        h_new = h - step * g / norm
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum_bounds(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        # keep improvement when it occurs
        if c5_new <= best_c5:
            h = h_new

        step *= beta
        it += 1

        # occasional reset if stuck
        if no_imp > 5000:
            h = best_h + (rng.random(n) - 0.5) * 0.01
            h = np.clip(h, 0.0, 1.0)
            h = project_sum_bounds(h, target)
            no_imp = 0

    # ----- optional up‑sampling and fine‑tuning -----
    if time.time() < deadline and n * 2 <= 720:
        n2 = n * 2
        target2 = n2 / 2.0
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = project_sum_bounds(h_up, target2)

        step2 = 0.5
        it2 = 0
        max_iters2 = 20000
        while time.time() < deadline and it2 < max_iters2:
            corr = np.correlate(h_up, 1.0 - h_up, mode="full")
            max_idx = corr.argmax()
            k = max_idx - (n2 - 1)
            g = gradient_worst_shift(h_up, k)
            norm = np.linalg.norm(g)
            if norm == 0:
                break
            h_new = h_up - step2 * g / norm
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_sum_bounds(h_new, target2)

            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_up = h_new
                no_imp = 0
            else:
                no_imp += 1

            h_up = h_new
            step2 *= beta
            it2 += 1
            if no_imp > 5000:
                h_up = best_h + (rng.random(n2) - 0.5) * 0.01
                h_up = np.clip(h_up, 0.0, 1.0)
                h_up = project_sum_bounds(h_up, target2)
                no_imp = 0

    # final safety projection
    best_h = project_sum_bounds(best_h, best_h.size / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.size)
