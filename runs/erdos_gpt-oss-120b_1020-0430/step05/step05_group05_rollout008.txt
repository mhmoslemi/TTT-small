        # choose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Swap and compute new C

        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        elapsed = time.time() - start_time

def simulated_annealing_binary(h, time_limit):
    start = time.time()
    n = h.size
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    cur_c = compute_c5(h)
    best_c = cur_c
    best_h = h.copy()
    # schedule: T = T0 * (1 - (elapsed / (time_limit/2))) for first half, then restart T0 again for second half
    T0 = 0.02
    half = time_limit / 2
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        # compute temperature
        if elapsed < half:
            T = T0 * (1 - elapsed / half)
        else:
            T = T0 * (elapsed - half) / half  # ramp up again
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = int(ones[idx_i])
        j = int(zeros[idx_j])
        # perform swap

        h[j] = 1.0

def gradient_descent_refine(h, num_iters, step0, rng):
    n = h.size
    target = n/2
    for it in range(num_iters):
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr) - (n-1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1 - 2*h
        else:
            # for i from 0 to n-k-1, gradient involves (1 - h[i+k]) - h[i]
            # Actually gradient from worst shift s = k: derivative w.r.t. h[i] = (1 - h[i+k]) if i + k < n else 0
            # plus derivative w.r.t. h[i] = -h[i - k] if i - k >= 0 else 0
            # So compute
            g[:n-k] += 1 - h[k:]
            g[k:] -= h[:n-k]
        # step with decreasing step size
        step = step0 * (0.5**it)
        h = h - step * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # correct sum
        diff = target - h.sum()
        # adjust a random index
        for _ in range(200):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h
```

def gradient_descent_refine(h, num_iters, step0, rng):
    n = h.size
    target = n / 2.0
    for it in range(num_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step = step0 * (0.5 ** it)
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        # adjust sum
        for _ in range(200):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h
```

def simulated_annealing_binary(h, time_limit, rng):
    n = h.size
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    cur_c = compute_c5(h)
    best_c = cur_c
    best_h = h.copy()
    start = time.time()
    T0 = 0.04
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        # exponential cooling
        T = T0 * np.exp(-5.0 * elapsed / time_limit)
        # random swap indices
        idx_i = rng.integers(0, len(ones))
        idx_j = rng.integers(0, len(zeros))
        i = int(ones[idx_i])
        j = int(zeros[idx_j])
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            # update index arrays
            ones[idx_i] = j
            zeros[idx_j] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        if best_c <= 0.3808:
            break
    return best_h, best_c
```

        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h

def fractional_to_binary(h):
    n = h.size
    half = n // 2
    idx = np.argpartition(h, -half)[-half:]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h

def simulated_annealing_binary(h, time_limit, rng):
    n = h.size
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    cur_c = compute_c5(h)
    best_c = cur_c
    best_h = h.copy()
    start = time.time()
    T0 = 0.04
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T0 * np.exp(-5.0 * elapsed / time_limit)
        idx_i = rng.integers(0, len(ones))
        idx_j = rng.integers(0, len(zeros))
        i = int(ones[idx_i])
        j = int(zeros[idx_j])
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        elif T > 0 and rng.random() < np.exp(-delta / T):
            accept = True
        if accept:
            cur_c = new_c
            ones[idx_i] = j
            zeros[idx_j] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        if best_c <= 0.3808:
            break
    return best_h, best_c

def gradient_descent_refine(h, num_iters, step0, rng):
    n = h.size
    target = n / 2.0
    for it in range(num_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step = step0 * (0.5 ** it)
        h = h - step * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        for _ in range(200):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
    else:
        parent = None
        n = 360

    best_h = None
    best_c = np.inf

    seeds = []
    if parent is not None:
        seeds.append(fractional_to_binary(parent))
        seeds.append(parent.copy())
    for _ in range(5):
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        seeds.append(h)

    for s_idx, seed_h in enumerate(seeds):
        if time.time() > deadline:
            break
        adjust_sum_exact(seed_h, rng)
        remaining = deadline - time.time()
        slice_time = min(25.0, max(5.0, remaining / (len(seeds) - s_idx + 0.5)))
        if slice_time <= 0:
            break
        h_opt, c_opt = simulated_annealing_binary(seed_h.copy(), slice_time, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        if best_c <= 0.3808:
            break

    # Upsample and continuous refinement if time permits
    if best_h is not None and time.time() < deadline:
        n_up = min(720, n * 2)
        if n_up > n:
            xs = np.linspace(0.0, 2.0, n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            adjust_sum_exact(h_up, rng)
            h_up = gradient_descent_refine(h_up, num_iters=40, step0=0.02, rng=rng)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_c, best_h, n = c_up, h_up.copy(), n_up

    # Final high‑resolution binary annealing if time left
    if time.time() < deadline:
        n_high = 720
        remaining = deadline - time.time()
        if remaining >= 20.0:
            for _ in range(3):
                if time.time() > deadline:
                    break
                hb = np.zeros(n_high)
                hb[rng.choice(n_high, n_high // 2, replace=False)] = 1.0
                slice_time = min(remaining / (3 - _), 30.0)
                h_opt, c_opt = simulated_annealing_binary(hb, slice_time, rng)
                if c_opt < best_c:
                    best_c, best_h, n = c_opt, h_opt.copy(), n_high
                remaining = deadline - time.time()
                if best_c <= 0.3808:
                    break

    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
