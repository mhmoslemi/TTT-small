import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h)=len(h)/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
        else:
            dec = min(-diff, h[i])
            h[i] -= dec
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny correction to satisfy the sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- simulated annealing ----------
def make_move(cur_h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = cur_h.size
    move_type = rng.choice(['transfer', 'swap', 'block', 'window'],
                           p=[0.4, 0.1, 0.2, 0.3])
    if move_type == 'transfer':
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n - 1)
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            return cur_h.copy()
        eps = max_eps * rng.random() * 0.5
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        return nh
    if move_type == 'swap':
        i, j = rng.choice(n, 2, replace=False)
        nh = cur_h.copy()
        nh[i], nh[j] = nh[j], nh[i]
        return nh
    if move_type == 'block':
        L = rng.integers(2, min(30, n))
        s = rng.integers(0, n - L + 1)
        sub = cur_h[s:s+L].copy()
        rng.shuffle(sub)
        nh = cur_h.copy()
        nh[s:s+L] = sub
        return nh
    # window redistribution
    L = rng.integers(2, min(30, n))
    s = rng.integers(0, n - L + 1)
    sub = cur_h[s:s+L]
    m = sub.sum()
    if m >= L - 1e-12:  # all ones, nothing to change
        return cur_h.copy()
    rand = rng.random(L)
    new_sub = rand * (m / rand.sum())
    nh = cur_h.copy()
    nh[s:s+L] = new_sub
    return nh


def simulated_annealing(start_h: np.ndarray, rng: np.random.Generator,
                        time_limit: float) -> tuple:
    """Return (best_h, best_c) found before time_limit seconds elapse."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    start = time.time()
    # SA runs for ~75% of allotted time
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = 0.005 * (1.0 - frac) + 1e-6 * frac
        new_h = make_move(cur_h, rng)
        # enforce constraints (fast correction)
        np.clip(new_h, 0.0, 1.0, out=new_h)
        new_h = enforce_sum(new_h, rng)
        new_c = compute_c5(new_h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    # final sanity

    best_h = enforce_sum(best_h, rng)
