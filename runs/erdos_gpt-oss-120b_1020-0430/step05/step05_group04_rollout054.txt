def project_box_sum(v, target, tol=1e-12, max_iter=60):
    """Project v onto [0,1]^n with exactly sum(target) (binary box + sum)."""
    lo = -np.max(v) - 10.0
    hi = 1.0 - np.min(v) + 10.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

                return h
            # autocorrelation of h
            H = np.fft.rfft(h, n=L)
            Hr = np.fft.rfft(h[::-1], n=L)
            corr = np.fft.irfft(H * Hr, n=L)[: 2 * n - 1]
            # weights (soft‑max over -corr)
            cmin = corr.min()
            w = np.exp(-beta * (corr - cmin))
            w /= w.sum()
            # gradient via convolutions
            W = np.fft.rfft(w, n=L)
            W_rev = np.fft.rfft(w[::-1], n=L)
            conv1 = np.fft.irfft(W * H, n=L)[: 3 * n - 2]
            conv_rev = np.fft.irfft(W_rev * H, n=L)[: 3 * n - 2]
            mid = n - 1
            g = -(conv1[mid : mid + n] + conv_rev[mid : mid + n])
            # step and projection
            h = project_box_sum(h - lr_base * g, target)
    return h

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.96  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if "initial_h_values" in globals() else None

    # candidate sample sizes (even)
    cand = []
    if parent is not None:
        n0 = parent.size
        if n0 % 2 == 0:
            cand.append(n0)
        if n0 * 2 <= 720 and (n0 * 2) % 2 == 0:
            cand.append(n0 * 2)
    for n in (720, 540, 360, 240):
        if n % 2 == 0:
            cand.append(n)
    cand = [c for c in dict.fromkeys(cand)]  # preserve order, remove dupes

    # loop over candidate sizes
    for n in cand:
        if time.time() > deadline:
            break
        target = n // 2
        L = 1 << ((3 * n - 2).bit_length())
        # prepare a few seeds per size
        seeds = []
        # parent‑based seeds
        if parent is not None:
            if n == parent.size:
                seeds.append(parent.copy())
            elif n == parent.size * 2:
                seeds.append(np.repeat(parent, 2))
        # binary seed (top half ones)
        bin0 = np.zeros(n)
        bin0[:target] = 1.0
        seeds.append(bin0)
        # random fractional seeds
        for _ in range(2):
            rand = rng.random(n)
            rand = project_box_sum(rand, target)
            seeds.append(rand)

        for h0 in seeds:
            if time.time() > deadline:
                break
            # smooth‑max gradient descent
            beta_seq = (50, 200, 800, 2000, 8000, 20000)
            h = smoothmax_grad_descent(h0, beta_seq, deadline)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n
                if best_c < 0.3808:
                    return best_h.tolist(), float(best_c), int(best_n)

            # optional binary rounding and quick evaluation
            if not np.all((h == 0) | (h == 1)):
                idx = np.argpartition(h, -target)[-target:]
                h_bin = np.zeros_like(h)
                h_bin[idx] = 1.0
                c_bin = compute_c5(h_bin)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                    if best_c < 0.3808:
                        return best_h.tolist(), float(best_c), int(best_n)

            # final SLSQP refinement with a strong beta
            if time.time() < deadline:
                h_ref = slsqp_optimize(h, beta=5000, max_iter=300)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c, best_n = h_ref.copy(), c_ref, n
                    if best_c < 0.3808:
                        return best_h.tolist(), float(best_c), int(best_n)

    # fallback simple construction if nothing found
    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_box_sum(best_h, n // 2)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_n // 2)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
