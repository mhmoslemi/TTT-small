import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # If diff is very small, return
    if abs(diff) < 1e-12:
        return h
    # We'll adjust iteratively up to 2000 times
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        # compute allowable add or sub
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # Final clip
    h = np.clip(h, 0.0, 1.0)
    # Might still have tiny diff due to clipping, correct by distributing
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # simple uniform small adjustment
        for _ in range(2000):
            i = rng.integers(0, n)
            if diff > 0 and h[i] < 1.0:
                step = min(diff, 1.0 - h[i])
            elif diff < 0 and h[i] > 0.0:
                step = max(diff, -h[i])
            else:
                continue
            h[i] += step
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def smooth_max_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = beta * corr
    m = bc.max()
    # log-sum-exp
    s = np.exp(bc - m).sum()
    return (m + np.log(s)) / beta

def sum_eq_constraint(x, target):
    return np.sum(x) - target

def slsqp_optimize(x0, beta, rng, max_iter=500):
    n = x0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint, 'args': (target,)}
    res = opt.minimize(smooth_max_obj, x0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=[cons],
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    x = adjust_sum_exact(x, rng)
    return x

def continuous_hill_climb(h, cur_c, rng, deadline):
    n = h.size
    cur = h.copy()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        # max epsilon
        eps_max = min(cur[src], 1.0 - cur[dst])
        if eps_max <= 0:
            continue
        eps = rng.random() * eps_max
        # apply
        cur[src] -= eps
        cur[dst] += eps
        new_c = compute_c5(cur)
        if new_c < cur_c - 1e-12:
            cur_c = new_c
        else:
            # revert
            cur[src] += eps
            cur[dst] -= eps
    return cur, cur_c

import time
import numpy as np
import scipy.optimize as opt

TARGET = 0.3808  # early exit target


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(200):
            i = rng.integers(0, n)
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
            elif diff < 0 and h[i] > 0.0:
                add = max(diff, -h[i])
            else:
                continue
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def smooth_max_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = bc.max()
    s = np.exp(bc - m).sum()
    return (m + np.log(s)) / beta


def sum_eq_constraint(x, target):
    return np.sum(x) - target


def slsqp_optimize(x0, beta, rng, max_iter=500):
    n = x0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint, "args": (target,)}
    res = opt.minimize(
        smooth_max_obj,
        x0,
        args=(beta,),
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": int(max_iter), "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = adjust_sum_exact(x, rng)
    return x


def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(
            h.astype(np.int32), (1 - h).astype(np.int32), mode="full"
        )
        O[n - 1] = 0
    return h, O


def random_feasible(n, rng):
    target = n / 2.0
    h = rng.random(n)
    s = target / h.sum()
    h = h * s
    h = np.clip(h, 0.0, 1.0)
    h = adjust_sum_exact(h, rng)
    return h


def continuous_hill_climb(h, cur_c, rng, deadline):
    n = h.size
    cur = h.copy()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        eps_max = min(cur[src], 1.0 - cur[dst])
        if eps_max <= 0:
            continue
        eps = rng.random() * eps_max
        cur[src] -= eps
        cur[dst] += eps
        new_c = compute_c5(cur)
        if new_c < cur_c - 1e-12:
            cur_c = new_c
        else:
            cur[src] += eps
            cur[dst] -= eps
    return cur, cur_c


def to_binary_by_rank(h):
    n = h.size
    k = n // 2
    idx = np.argpartition(-h, k)[:k]
    bin_h = np.zeros(n, dtype=np.int8)
    bin_h[idx] = 1
    return bin_h


def hill_climb_binary(h_bin, n, rng, deadline):
    cur = h_bin.copy()
    O = np.correlate(cur.astype(np.int32), (1 - cur).astype(np.int32), mode="full")
    cur_max = O.max()
    while time.time() < deadline:
        ones = np.flatnonzero(cur == 1)
        zeros = np.flatnonzero(cur == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        O_new = np.correlate(cur.astype(np.int32), (1 - cur).astype(np.int32), mode="full")
        new_max = O_new.max()
        if new_max < cur_max:
            cur_max = new_max
        else:
            cur[i] = 1
            cur[j] = 0
    return cur, cur_max

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98

    best_h = None
    best_c = np.inf
    best_n = None

    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        parent = adjust_sum_exact(parent, rng)
        c_parent = compute_c5(parent)

        if best_c <= TARGET:

    candidate_ns = [360, 720]

    for n in candidate_ns:

            break
        starts = []
        if parent is not None and parent.size == n:
            starts.append(parent.copy())
        if n <= 720:
            h_bin, _ = greedy_binary(n, rng)
            starts.append(h_bin.astype(float))
        for _ in range(2):
            starts.append(random_feasible(n, rng))

        for h0 in starts:
            if time.time() >= deadline:
                break
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = adjust_sum_exact(h0, rng)
            cur_h = h0.copy()
            cur_c = compute_c5(cur_h)
            if cur_c < best_c - 1e-12:
                best_h, best_c, best_n = cur_h.copy(), cur_c, n
                if best_c <= TARGET:
                    return best_h.tolist(), float(best_c), int(best_n)

            betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]

                max_iter = 300 if beta < 200 else 600

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
