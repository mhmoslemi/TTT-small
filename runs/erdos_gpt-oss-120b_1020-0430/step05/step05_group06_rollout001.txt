def continuous_refine(h0, beta_seq, rng, deadline):
   h = h0.copy()
   target = h.size / 2.0
   # ensure sum exactly initially
   adjust_sum_exact(h, rng)
   for beta in beta_seq:
        if time.time() >= deadline:
            break
        # define objective with penalty
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            bc = beta * corr
            max_bc = np.max(bc)
            # using logsumexp
            return (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / beta + 1e3 * (np.sum(x) - target) ** 2
        bounds = [(0.0, 1.0)] * h.size
        # use L-BFGS-B
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 300, 'ftol': 1e-12, 'disp': False})
        h = np.clip(res.x, 0.0, 1.0)
        # Optional: reproject sum exactly after each beta
        adjust_sum_exact(h, rng)
   return h

def simulated_annealing_binary(h0, rng, start, deadline):
    # h0 binary (0/1) array
    h = h0.copy()
    n = h.size
    cur_c = compute_c5(h.astype(float))
    best_h = h.copy()
    best_c = cur_c

    # estimate temperature
    deltas = []
    for _ in range(50):
        # random swap
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h.astype(float))
        deltas.append(c - cur_c)
        h[i], h[j] = h[j], h[i]
    pos = [d for d in deltas if d > 0]
    if pos:
        T_start = max(np.mean(pos), 1e-6)
    else:
        T_start = 1e-6
    T_end = 1e-9

    while time.time() < deadline:
        # compute temperature based on fraction of time elapsed from start to deadline
        fraction = (time.time() - start) / (deadline - start)
        T = T_start * (T_end / T_start) ** fraction
        # propose swap
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h.astype(float))
        accepted = False
        if new_c < cur_c:
            accepted = True
        else:
            delta = new_c - cur_c
            if T > 0 and rng.random() < np.exp(-delta / T):
                accepted = True
        if accepted:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP surrogate
CURRENT_BETA = None
CURRENT_TARGET = None

def compute_c5(h):
    """Return C5 for a step function h (float array)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact correction to enforce sum(h)=n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth‑max surrogate with global beta/target."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """Run SLSQP on smooth‑max surrogate for a given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def greedy_binary(n, rng):
    """Greedy construction of a binary vector with exactly n/2 ones."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h

def simulated_annealing_binary(h0, rng, sa_start, sa_deadline):
    """SA on binary vectors (0/1 float) respecting sum constraint."""
    h = h0.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    # estimate starting temperature
    deltas = []
    for _ in range(50):
        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        deltas.append(c - cur_c)
        h[i], h[j] = h[j], h[i]
    pos = [d for d in deltas if d > 0]
    T_start = max(np.mean(pos) if pos else 1e-6, 1e-6)
    T_end = 1e-9

    while time.time() < sa_deadline:
        frac = (time.time() - sa_start) / (sa_deadline - sa_start)
        T = T_start * (T_end / T_start) ** frac

        i = rng.integers(n)
        while h[i] < 0.5:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] > 0.5:
            j = rng.integers(n)
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            delta = new_c - cur_c
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # parent solution, if provided
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        c_parent = compute_c5(parent)
        best_h = parent.copy()
        best_c = c_parent
        best_n = parent.size

    # candidate sample sizes (large first)
    for n in (720, 360):
        if time.time() > deadline - 5:
            break

        # ----- greedy binary start -----
        h_bin = greedy_binary(n, rng).astype(float)
        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_h = h_bin.copy()
            best_c = c_bin
            best_n = n

        # ----- simulated annealing -----
        remaining = deadline - time.time()
        if remaining > 15:
            # allocate up to 60% of remaining time for SA
            sa_time = min(remaining * 0.6, 300.0)
            sa_deadline = time.time() + sa_time
            sa_start = time.time()
            # ensure binary start
            h_start = (h_bin > 0.5).astype(float)
            h_best_sa, c_best_sa = simulated_annealing_binary(h_start, rng, sa_start, sa_deadline)
            if c_best_sa < best_c:
                best_h = h_best_sa.copy()
                best_c = c_best_sa
                best_n = n

        # ----- continuous refinement (SLSQP cascade) -----
        remaining = deadline - time.time()
        if remaining > 10 and best_h is not None and best_n == n:
            # make sure sum constraint exact before refinement
            adjust_sum_exact(best_h, rng)
            beta_seq = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0)
            for beta in beta_seq:
                if time.time() >= deadline:
                    break
                max_iter = 800 if beta >= 2000 else 600
                try:
                    best_h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c_new = compute_c5(best_h)
                if c_new < best_c:
                    best_c = c_new
                    best_n = n

        # keep best so far across n
        if best_h is not None:
            best_n = best_h.size

    # final safety adjustments
    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_n = n
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
