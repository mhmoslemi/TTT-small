def sa_optimize(h, rng, deadline):
    n = h.size
    corr, cur_c = compute_c5_fft(h)
    # effective temperature initial derived from cur_c
    T = cur_c * 0.05 + 1e-6
    # cooling factor
    cooling = 0.99995
    iter = 0
    while time.time() < deadline:
        # decide to try worst-shift or random pair
        if rng.random() < 0.7:
            idx_max = np.argmax(corr)
            k = idx_max - (n - 1)
            if k >= 0:
                i_low = 0
                i_high = n - 1 - k
            else:
                i_low = -k
                i_high = n - 1
            if i_low > i_high:
                continue
            i = rng.integers(i_low, i_high + 1)
            j = i + k
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
        # delta
        delta_max = min(h[i], 1 - h[j])
        if delta_max <= 1e-12:
            continue
        # sample delta ~ exponential or uniform
        delta = delta_max * rng.random()
        # apply update
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # compute new
        corr_new, new_c = compute_c5_fft(h_new)
        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            h = h_new
            corr = corr_new
            cur_c = new_c
        # cooling
        T = max(T * cooling, 1e-9)
        iter += 1
    return h, cur_c

def compute_c5_corr(h):
    n = h.shape[0]
    L = 2*n - 1
    L2 = 1 << (L-1).bit_length()
    # reversed (1 - h)
    b = np.flip(1.0 - h)
    # Compute via FFT
    H = np.fft.fft(h, L2)
    B = np.fft.fft(b, L2)
    c = np.fft.ifft(H * B).real[:L]
    return c, c.max() * (2.0 / n)

    target = n / 2.0
    for it in range(steps):
        corr, cur_c = compute_c5_corr(h)

            if i_low > i_high:
                continue
            i = rng.integers(i_low, i_high + 1)
            j = i + k
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
        # decide move or swap
        if rng.random() < 0.5:
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 1e-15:
                continue
            delta = delta_max * rng.random()
            h[i] -= delta
            h[j] += delta
            moved = True
        else:
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            moved = False
        corr_new, new_c = compute_c5_corr(h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
                accept = True
        if accept:
            corr, cur_c = corr_new, new_c
        else:
            if moved:
                h[i] += delta
                h[j] -= delta
            else:
                h[i], h[j] = h[j], h[i]
        T = max(T * cool, 1e-12)
    return h, cur_c

# ----- SLSQP helpers (from earlier solution) -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.99 * budget_s

    # initial vector
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 360
        h = np.zeros(n0)
        h[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    adjust_sum_exact(h, n0 / 2.0, rng)

    # quick gradient refinement
    h = gradient_refine_custom(h, rng, steps=12)
    _, cur_c = compute_c5_corr(h)
    best_h, best_c, best_n = h.copy(), cur_c, n0

    # SA on original size (about 60% of time)
    sa_dead = min(deadline, time.time() + (deadline - start) * 0.6)
    h, cur_c = sa_optimize(h, rng, sa_dead)
    if cur_c < best_c:
        best_h, best_c, best_n = h.copy(), cur_c, n0

    # post‑SA gradient
    h = gradient_refine_custom(h, rng, steps=8)
    _, cur_c = compute_c5_corr(h)
    if cur_c < best_c:
        best_h, best_c, best_n = h.copy(), cur_c, n0

    # optional SLSQP if time permits
    if time.time() < deadline:
        for beta in (5.0, 20.0, 80.0):
            if time.time() >= deadline:
                break
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=250)
            except Exception:
                continue
            _, cur_c = compute_c5_corr(h)
            if cur_c < best_c:
                best_h, best_c, best_n = h.copy(), cur_c, n0

    # up‑sample and repeat if time remains
    if time.time() < deadline and n0 * 2 <= 720:
        n_up = n0 * 2
        xs = np.linspace(0.0, 2.0, n0, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, n_up / 2.0, rng)

        h_up = gradient_refine_custom(h_up, rng, steps=8)
        _, cur_c_up = compute_c5_corr(h_up)
        # SA on upsampled vector (remaining time)
        h_up, cur_c_up = sa_optimize(h_up, rng, deadline)
        h_up = gradient_refine_custom(h_up, rng, steps=6)
        _, cur_c_up = compute_c5_corr(h_up)
        if cur_c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), cur_c_up, n_up
        # final quick SLSQP on upsampled if possible
        if time.time() < deadline:
            for beta in (5.0, 20.0):
                if time.time() >= deadline:
                    break
                try:
                    h_up = slsqp_optimize(best_h, beta, rng, max_iter=200)
                except Exception:
                    continue
                _, cur_c_up = compute_c5_corr(h_up)
                if cur_c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), cur_c_up, best_n

    # final safety adjustments
    adjust_sum_exact(best_h, best_n / 2.0, rng)
    _, final_c = compute_c5_corr(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
