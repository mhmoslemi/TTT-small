def gradient_smooth_max(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta_corr = beta * corr
    max_val = np.max(beta_corr)
    exp_shifted = np.exp(beta_corr - max_val)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros_like(h)
    for shift in range(-n+1, n):
        w = weights[shift + n - 1]
        if w == 0.0:
            continue
        if shift >= 0:
            l = n - shift
            # left indices 0..l-1
            grad[:l] += w * (1.0 - h[shift:])   # shape l
            grad[shift:] -= w * h[:l]
        else:
            s = -shift
            l = n - s
            # right indices s..n-1
            grad[s:] += w * (1.0 - h[:l])      # h[:l] length l
            grad[:l] -= w * h[s:]
    return grad
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Redistribute mass so that sum(h) = n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- gradient descent on worst shift ----------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected sub‑gradient descent using the actual worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            if upper > 0:
                g[:upper] += 1.0 - h[k:]
                g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # enforce the sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- smooth‑max gradient ----------
def grad_smooth_max(h, beta):
    """Gradient of the smooth‑max of overlap (log‑sum‑exp)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    beta_corr = beta * corr
    max_val = np.max(beta_corr)
    exp_shifted = np.exp(beta_corr - max_val)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp

    grad = np.zeros_like(h)
    for shift in range(-n + 1, n):
        w = weights[shift + n - 1]
        if w == 0.0:
            continue
        if shift >= 0:
            l = n - shift
            if l > 0:
                grad[:l] += w * (1.0 - h[shift:])
                grad[shift:] -= w * h[:l]
        else:
            s = -shift
            l = n - s
            if l > 0:
                grad[s:] += w * (1.0 - h[:l])
                grad[:l] -= w * h[s:]
    return grad

# ---------- L‑BFGS‑B with penalty ----------
def lbfgsb_optimize(h0, beta, rng, mu=1e3, max_iter=600):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def fun_and_grad(x):
        # smooth‑max part
        corr = np.correlate(x, 1.0 - x, mode="full")
        beta_corr = beta * corr
        max_val = np.max(beta_corr)
        exp_shifted = np.exp(beta_corr - max_val)
        sum_exp = np.sum(exp_shifted)
        smooth = (max_val + np.log(sum_exp)) / beta
        weights = exp_shifted / sum_exp

        # gradient of smooth‑max
        grad = np.zeros_like(x)
        for shift in range(-n + 1, n):
            w = weights[shift + n - 1]
            if w == 0.0:
                continue
            if shift >= 0:
                l = n - shift
                if l > 0:
                    grad[:l] += w * (1.0 - x[shift:])
                    grad[shift:] -= w * x[:l]
            else:
                s = -shift
                l = n - s
                if l > 0:
                    grad[s:] += w * (1.0 - x[:l])
                    grad[:l] -= w * x[s:]

        # penalty for sum constraint
        diff = np.sum(x) - target
        smooth += mu * diff * diff
        grad += 2.0 * mu * diff
        return smooth, grad

    res = minimize(fun_and_grad, h0, method="L-BFGS-B", jac=True,
                   bounds=bounds,
                   options={"maxiter": max_iter, "ftol": 1e-13, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations (keep total work modest)
    candidate_ns = [360, 480, 600]

    # optional parent vector
    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        start_vecs = []

        # interpolate parent if available
        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp)                                 # fractional
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)                                # binary

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        start_vecs.append(rand_frac)

        # alternating binary pattern
        alt_bin = np.tile([1.0, 0.0], n // 2)
        start_vecs.append(alt_bin)

        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # initial evaluation
            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

            # a few gradient refinements
            h = gradient_refine(h, rng, step=0.02, max_iter=10)
            c5 = compute_c5(h)
            if c5 < best_c:

            # cascade of L‑BFGS‑B with increasing smoothing
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
                if time.time() > deadline:

                max_iter = 400 if beta < 200 else 600
                if beta >= 2000:

                h = lbfgsb_optimize(h, beta, rng, mu=1e3, max_iter=max_iter)
                c5 = compute_c5(h)
                if c5 < best_c:

            # final gradient sweep

            c5 = compute_c5(h)
            if c5 < best_c:

    # -------- optional up‑sampling --------
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)

        adjust_sum_exact(h_up, rng)

            if time.time() > deadline:
