def proj_capped_simplex(v, target):
    n = v.shape[0]
    # Clip between 0 and 1 initially? Actually we want to find lambda s.t. sum(clip(v - lambda, 0,1)) = target.
    # We'll bisection on lambda.
    lo = np.min(v - 1) - 1e-12   # ensure sum>target at lo
    hi = np.max(v) + 1e-12       # sum<target at hi
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Due to numerical errors, adjust small diff
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # add diff to some entries not at bounds
        idx = np.where((x > 0) & (x < 1))[0]
        if idx.size > 0:
            # distribute uniformly
            add = diff / idx.size
            x[idx] += add
            # clip again
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback to simple redistribution
            pass
    return x

def compute_softmax_weights(corr_raw, dx, beta):
    # Compute w = -dx * softmax(beta*(1 - corr_raw*dx))
    u = beta * (1 - corr_raw * dx)
    max_u = np.max(u)
    exp_u = np.exp(u - max_u)
    sum_exp = np.sum(exp_u)
    p = exp_u / sum_exp
    w = -dx * p
    return w

def smooth_max_descent(h0, target, dx, deadline):
    h = proj_capped_simplex(np.clip(h0,0,1), target)
    best_h, best_c = h, compute_c5(h)
    step_scale = 0.5
    beta = 1.0
    iteration = 0
    while time.time() < deadline:
        corr_raw = np.correlate(h, 1 - h, mode='full')
        C = corr_raw * dx
        u = beta * (1 - C)
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        p = exp_u / np.sum(exp_u)   # softmax
        w = -p
        rev_w = w[::-1]
        # Compute term1 and term2 via convolution
        term1 = np.convolve(1 - h, rev_w, mode='full')[len(w)//2 : len(w)//2 + len(h)]
        term2 = np.convolve(h, w, mode='full')[len(w)//2 : len(w)//2 + len(h)]
        update = step_scale * (term1 - term2)
        h = proj_capped_simplex(h - update, target)
        iteration += 1
        # adjust beta and step_scale over time
        if iteration % 200 == 0:
            beta = min(beta * 1.2, 5000.0)
            step_scale = max(step_scale * 0.95, 0.02)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        # maybe random perturbation occasionally
        if iteration % 1000 == 0:
            # small noise
            noise = np.random.default_rng().normal(scale=0.01, size=len(h))
            h = proj_capped_simplex(h + noise, target)
    return best_h, best_c

import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _proj_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto {0≤x≤1, sum x = target}."""
    lo = np.min(v) - 1.0          # sum = n  > target
    hi = np.max(v)                # sum = 0  < target
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute remaining mass among non‑boundary entries
        idx = np.where((x > 0.0) & (x < 1.0))[0]
        if idx.size:
            add = diff / idx.size
            x[idx] += add
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _smooth_max_descent(h0: np.ndarray, target: float, dx: float,
                        deadline: float, rng: np.random.Generator):
    """Projected gradient on a smooth max surrogate."""
    n = h0.size
    h = _proj_capped_simplex(np.clip(h0, 0.0, 1.0), target)
    best_h, best_c = h.copy(), _compute_c5(h)

    L_corr = 2 * n - 1
    nfft_corr = _next_pow2(L_corr)
    L_conv = 3 * n - 2
    nfft_conv = _next_pow2(L_conv)

    step = 0.5           # multiplied by the raw diff (term1-term2)
    beta = 1.0
    it = 0
    while time.time() < deadline:
        # ----- C (overlap) via FFT correlation -----
        h_fft_corr = np.fft.rfft(h, n=nfft_corr)
        rev1mh = (1.0 - h)[::-1]
        rev_fft = np.fft.rfft(rev1mh, n=nfft_corr)
        corr_raw = np.fft.irfft(h_fft_corr * rev_fft, n=nfft_corr)[:L_corr]
        C = corr_raw * dx

        # ----- softmax weights -----
        u = beta * (1.0 - C)
        mu = np.max(u)
        exp_u = np.exp(u - mu)
        p = exp_u / np.sum(exp_u)
        w = -p                       # sum(w) = -1

        # ----- convolutions for gradient -----
        a_fft = np.fft.rfft(1.0 - h, n=nfft_conv)
        h_fft = np.fft.rfft(h, n=nfft_conv)
        w_fft = np.fft.rfft(w, n=nfft_conv)

        conv_a = np.fft.irfft(a_fft * w_fft, n=nfft_conv)[:L_conv]
        conv_h = np.fft.irfft(h_fft * w_fft, n=nfft_conv)[:L_conv]

        off = n - 1
        term1 = conv_a[off:off + n]
        term2 = conv_h[off:off + n]
        diff = term1 - term2

        # ----- projected update -----
        h = _proj_capped_simplex(h - step * diff, target)

        it += 1
        if it % 200 == 0:
            beta = min(beta * 1.2, 5000.0)
            step = max(step * 0.95, 0.02)
            cur_c = _compute_c5(h)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        if it % 1000 == 0:
            # small random nudge to escape shallow basins
            h = _proj_capped_simplex(h + rng.normal(scale=0.01, size=n), target)

    return best_h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # basic parameters
    n = 360
    dx = 2.0 / n
    target = n / 2.0

    best_h, best_c = None, np.inf

    # make a small pool of initial vectors
    init_pool = []

    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        init_pool.append(np.interp(xt, xs, parent))

    # random continuous
    for _ in range(3):
        rc = rng.random(n)
        rc = rc * (target / rc.sum())
        rc = np.clip(rc, 0.0, 1.0)
        rc = _proj_capped_simplex(rc, target)
        init_pool.append(rc)

    # random binary with a touch of smoothing
    for _ in range(2):
        bin_vec = np.zeros(n)
        bin_vec[rng.choice(n, n // 2, replace=False)] = 1.0
        # smear a little
        bl = np.convolve(bin_vec, np.ones(3) / 3, mode='same')
        bl = _proj_capped_simplex(bl, target)
        init_pool.append(bl)

    # main restart loop
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 2.0:
            break
        # allocate about a third of the remaining time to this restart
        sub_dead = time.time() + remaining * 0.33
        h0 = init_pool[rng.integers(len(init_pool))]
        h_opt, c_opt = _smooth_max_descent(h0, target, dx, sub_dead, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # optional up‑sampling refinement if time permits
    if best_h is not None and time.time() < deadline and best_h.size * 2 <= 720:
        n2 = best_h.size * 2
        xs = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        h_up = _proj_capped_simplex(h_up, target2)
        sub_dead = deadline
        h_up, c_up = _smooth_max_descent(h_up, target2, dx2, sub_dead, rng)
        if c_up < best_c:
            best_c, best_h = c_up, h_up

    # final safety fallback
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = _proj_capped_simplex(best_h, n / 2)
        best_c = _compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _proj_capped_simplex(best_h, best_h.size / 2.0)
    final_c = _compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
