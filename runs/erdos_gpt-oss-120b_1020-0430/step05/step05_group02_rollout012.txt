        # random pair perturbation
        i, j = rng.choice(len(h), size=2, replace=False)
        # choose delta bounded by 0≤h_i+delta≤1 and 0≤h_j-delta≤1 (to maintain sum)

def gradient_refine(h, rng, step_start=0.02, step_decay=0.5, max_iter=30):
    n = h.size
    target = n/2
    step = step_start
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum exact
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust using random indices
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        step *= step_decay
    return h
```

def slsqp_optimize(h0, beta, rng, max_iter=1000):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    cons = {"type":"eq","fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0.0,1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0,n)
        add = min(diff,1.0-x[i]) if diff>0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x
```

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for constraints used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Return the C5 value for a discretised step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def sum_eq_constraint(x):
    """Equality constraint: sum(x) == CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ h_i (1‑h_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # (m + log Σ exp(bc‑m)) / β  =  smooth max
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def adjust_sum_exact(h, rng, max_iter=2000):
    """Exactly enforce Σh = target by random walk on coordinates."""
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP optimisation of the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    x = adjust_sum_exact(x, rng)
    return x


def gradient_refine(h, rng, step_start=0.02, step_decay=0.5, max_iter=30):
    """Projected gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    step = step_start
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        # enforce exact sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = adjust_sum_exact(h, rng)
        step *= step_decay
    return h


def pairwise_opt(h, rng, max_iter=200):
    """
    Greedy pairwise updates using the gradient of the worst shift.
    Keeps the total sum unchanged.
    """
    n = h.size
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            upper = n - k
            g = np.zeros_like(h)
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        pos = np.argmax(g)
        neg = np.argmin(g)
        if g[pos] <= 0 or g[neg] >= 0:
            break
        delta = min(h[pos], 1.0 - h[neg])
        if delta <= 0:
            break
        # move half of the feasible amount for stability
        delta *= 0.5
        h[pos] -= delta
        h[neg] += delta
    return h
