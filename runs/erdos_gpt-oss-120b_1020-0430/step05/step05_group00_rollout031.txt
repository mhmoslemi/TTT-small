def simulated_annealing_continuous(h, rng, time_budget, start_time):
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    # parameters
    T0 = 0.01
    T_end = 1e-5
    elapsed = time.time() - start_time
    while elapsed < time_budget:
        # decay based on total allowed time
        t = elapsed / time_budget
        T = T0 * (T_end / T0) ** t
        # perform guided perturb
        grad = worst_gradient(cur_h)
        # scale: we can use epsilon = min(cur_h[i_max], 1-cur_h[i_min]) * rng.random() * (0.1 * (1-t) + 0.01)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        max_eps = min(cur_h[i_max], 1.0 - cur_h[i_min])
        if max_eps <= 0:
            # fallback random pair
            i_max, i_min = rng.integers(0, cur_h.size, size=2)
            max_eps = min(cur_h[i_max], 1.0 - cur_h[i_min])
        eps = max_eps * rng.random() * (0.05 * (1 - t) + 0.005)
        new_h = cur_h.copy()
        new_h[i_max] -= eps
        new_h[i_min] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        # periodically randomizing large epsilon
        if rng.random() < 0.001:
            # large random shuffle: pick random subset and reassign values while preserving sum
            # choose K indices
            K = max(2, cur_h.size // 100)
            idx = rng.choice(cur_h.size, K, replace=False)
            vals = cur_h[idx]
            # random Dirichlet for these K values
            rand = rng.random(K)
            total = vals.sum()
            # reassign values proportionally to random
            new_vals = rand / rand.sum() * total
            # Clip to [0,1]
            excess = new_vals - 1.0
            # If any >1, bring back to 1 and distribute excess
            # simple approach: clip and then fix sum later using enforce_sum on full vector.
            new_h = cur_h.copy()
            new_h[idx] = new_vals
            new_h = enforce_sum(new_h, rng)
            new_c = compute_c5(new_h)
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        elapsed = time.time() - start_time
    return best_h, best_c

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def simulated_annealing_continuous(start_h, rng, time_budget, global_start):
    best_h = start_h.copy()
    best_c = compute_c5(best_h)
    cur_h = start_h.copy()
    cur_c = best_c
    n = cur_h.size
    elapsed = time.time() - global_start
    while elapsed < time_budget:
        # progress relative to allocated time
        t = (elapsed) / time_budget
        # temperature schedule
        T0 = 0.02
        T_end = 1e-5
        T = T0 * (T_end / T0) ** t
        # gradient based perturb
        grad = worst_gradient(cur_h)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        max_eps = min(cur_h[i_max], 1.0 - cur_h[i_min])
        if max_eps <= 0:
            # choose random pair
            i_max, i_min = rng.integers(0, n, size=2)
            max_eps = min(cur_h[i_max], 1.0 - cur_h[i_min])
        eps = max_eps * rng.random() * (0.05 * (1 - t) + 0.005)
        new_h = cur_h.copy()
        new_h[i_max] -= eps
        new_h[i_min] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-14)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        # occasional random perturb larger
        if rng.random() < 0.001:
            # random reallocation among a subset
            K = max(2, n // 100)
            idx = rng.choice(n, K, replace=False)
            vals = cur_h[idx]
            total = vals.sum()
            # random redistribution using Dirichlet
            rand = rng.random(K)
            new_vals = rand / rand.sum() * total
            # clip and fix sum later
            new_h2 = cur_h.copy()
            new_h2[idx] = new_vals
            new_h2 = enforce_sum(new_h2, rng)
            new_c2 = compute_c5(new_h2)
            cur_h, cur_c = new_h2, new_c2
            if new_c2 < best_c:
                best_h, best_c = new_h2.copy(), new_c2
        elapsed = time.time() - global_start
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {'type': 'eq', 'fun': sum_eq}
    from scipy.optimize import minimize
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff using random indices
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Fast C5 via FFT correlation."""
    n = h.size
    L = 2 * n - 1
    Lfft = 1 << (L - 1).bit_length()
    H = np.fft.rfft(h, Lfft)
    G = np.fft.rfft(1.0 - h, Lfft)
    corr = np.fft.irfft(H * np.conj(G), Lfft)[:L]
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to have sum = n/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 20000:
        if diff > 0:
            i = rng.integers(0, h.size)
            room = 1.0 - h[i]
            if room <= 0:
                it += 1
                continue
            add = min(diff, room)
            h[i] += add
            diff -= add
        else:
            i = rng.integers(0, h.size)
            have = h[i]
            if have <= 0:
                it += 1
                continue
            sub = min(-diff, have)
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        h[0] += diff
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (max) overlap shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb_time(h0: np.ndarray, rng: np.random.Generator,
                    time_limit: float):
    start = time.time()
    best_h, best_c = h0.copy(), compute_c5(h0)
    cur_h, cur_c = best_h.copy(), best_c
    it = 0
    while time.time() - start < time_limit:
        grad = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        eps = min(cur_h[i], 1.0 - cur_h[j]) * 0.5 * (0.99 ** it)
        if eps <= 0:
            break
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
        if new_c < best_c:
            best_h, best_c = new_h.copy(), new_c
        it += 1
    return best_h, best_c


def simulated_annealing_continuous(start_h: np.ndarray,
                                   rng: np.random.Generator,
                                   time_budget: float):
    best_h = start_h.copy()
    best_c = compute_c5(best_h)
    cur_h, cur_c = best_h.copy(), best_c
    n = cur_h.size
    chunk_start = time.time()
    while time.time() - chunk_start < time_budget:
        elapsed = time.time() - chunk_start
        t = elapsed / time_budget
        T0, T1 = 0.02, 1e-5
        T = T0 * (T1 / T0) ** t
        grad = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            i, j = rng.integers(0, n, size=2)
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
        eps = max_eps * rng.random() * (0.05 * (1 - t) + 0.005)
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-14)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:

        if rng.random() < 0.001:

            cur_h, cur_c = new_h2, new_c2
            if new_c2 < best_c:
                best_h, best_c = new_h2.copy(), new_c2
    return best_h, best_c


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 500):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 2000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
        it += 1
    return x

def upsample(h: np.ndarray, new_n: int):
    x_old = np.linspace(0.0, 2.0, h.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new
