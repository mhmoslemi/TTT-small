def perturb_and_refine(h, n, rng, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = 0.1
    while time.time() < deadline and step > 1e-4:
        # propose perturb
        h_new = best_h.copy()
        i = rng.integers(n)
        delta = (rng.random() - 0.5) * 2 * step
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        # adjust sum to maintain exact total by adjusting another random index j != i
        diff = best_h.sum() - h_new.sum()
        if abs(diff) > 1e-12:
            # adjust j
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            # add diff to h_new[j] within bounds
            add = diff
            # clamp
            lower = -h_new[j]
            upper = 1.0 - h_new[j]
            add = min(max(add, lower), upper)
            h_new[j] += add
            # if diff not fully corrected, continue adjusting by more indices (but diff small)
        # ensure sum exactly via adjust_sum_exact
        adjust_sum_exact(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            # optionally increase step?
        else:
            # optionally reduce step
            step *= 0.99
    return best_h, best_c

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Exact correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

CURRENT_TARGET = 0.0
CURRENT_BETA = 1.0


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

        overall_best_h = np.zeros(overall_best_n, dtype=float)
        overall_best_h[: overall_best_n // 2] = 1.0
        adjust_sum_exact(overall_best_h, rng)
        overall_best_c = compute_c5(overall_best_h)

    # final safety adjustments
    overall_best_h = np.clip(overall_best_h, 0.0, 1.0)
    adjust_sum_exact(overall_best_h, rng)
    final_c = compute_c5(overall_best_h)

    return overall_best_h.tolist(), float(final_c), int(overall_best_n)
