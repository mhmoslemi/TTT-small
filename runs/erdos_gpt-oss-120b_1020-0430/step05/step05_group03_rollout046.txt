import time, math, numpy as np
from scipy.optimize import differential_evolution, LinearConstraint, minimize

# definitions: compute_c5, enforce_sum, worst_gradient, guided_perturb, guided_hill_climb, gradient_refine, slsqp_optimize

def de_optimize_blocks(target_n, n_blocks, rng, max_iter=150):
    if target_n % n_blocks != 0:
        return None
    block_size = target_n // n_blocks
    # define bound and constraint
    def obj(b):
        h = np.repeat(b, block_size)
        return compute_c5(h)
    bounds = [(0.0, 1.0)] * n_blocks
    constraint = LinearConstraint(np.ones((1, n_blocks)), [n_blocks/2], [n_blocks/2])
    result = differential_evolution(obj, bounds, constraints=(constraint,),
                                    maxiter=max_iter, polish=False,
                                    updating='deferred', seed=int(rng.integers(1<<32)))
    b_opt = np.clip(result.x, 0.0, 1.0)
    h = np.repeat(b_opt, block_size)
    h = enforce_sum(h, rng)
    return h

def run(seed=42, budget_s=1100, **kwargs):
   rng = np.random.default_rng(seed)
   start=time.time()

   # start with parent if available, else produce initial via block DE
   best_h = None
   best_c = float('inf')
   best_n = None

   if 'initial_h_values' in globals():
       h0 = np.array(initial_h_values, dtype=float)
       h0 = np.clip(h0,0,1)
       h0 = enforce_sum(h0, rng)
       best_h = h0
       best_c = compute_c5(h0)
       best_n = h0.size

   # try block DE for various block configs on n=512
   target_n = 512
   block_configs = [8, 16, 32, 64] # 64 gives block size 8
   for n_blocks in block_configs:
       if time.time() - start > budget_s * 0.6:
           break
       h = de_optimize_blocks(target_n, n_blocks, rng, max_iter=200)
       if h is None: continue
       c = compute_c5(h)
       # hill climb + SLSQP
       h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.7)
       for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
           if time.time() - start > budget_s * 0.8: break
           h = slsqp_optimize(h, beta, max_iter=500)
           c = compute_c5(h)
           if c < best_c:
               best_h, best_c, best_n = h.copy(), c, target_n
       # gradient refine
       h = gradient_refine(h, rng, step=0.015, max_iter=12)
       c = compute_c5(h)
       if c < best_c:
           best_h, best_c, best_n = h.copy(), c, target_n

   # If still not good enough, run a modified guided hill climbing from best solution with small perturbations (like previous final tweaks)
   while time.time() - start < budget_s * 0.99:
       # random perturbation from best: transfer epsilon from random positive grad to random negative
       grad = worst_gradient(best_h)
       nh = guided_perturb(best_h, grad, rng, scale=0.02)
       nc = compute_c5(nh)
       if nc < best_c:
           best_h, best_c = nh.copy(), nc

   # final adjust
   best_h = np.clip(best_h, 0, 1)
   best_h = enforce_sum(best_h, rng)
   final_c = compute_c5(best_h)
   return best_h.tolist(), float(final_c), int(best_h.size)

import time, numpy as np
from scipy.optimize import differential_evolution, LinearConstraint, minimize

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift attaining max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from max‑gradient index to min‑gradient index."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using worst‑shift gradient with diminishing step."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


# ---------- block‑level DE ----------
def de_optimize_blocks(target_n: int, n_blocks: int,
                       rng: np.random.Generator, max_iter: int = 200) -> np.ndarray:
    """DE on a piecewise‑constant representation with n_blocks blocks."""
    if target_n % n_blocks != 0:
        return None
    block_size = target_n // n_blocks
    def obj(b):
        h = np.repeat(b, block_size)
        return compute_c5(h)
    bounds = [(0.0, 1.0)] * n_blocks
    A = np.ones((1, n_blocks))
    constraint = LinearConstraint(A, [n_blocks / 2], [n_blocks / 2])
    result = differential_evolution(obj, bounds,
                                    constraints=(constraint,),
                                    maxiter=max_iter,
                                    polish=False,
                                    updating='deferred',
                                    seed=int(rng.integers(1 << 32)))
    b_opt = np.clip(result.x, 0.0, 1.0)
    h = np.repeat(b_opt, block_size)
    h = enforce_sum(h, rng)
    return h
