def project_onto_capped_simplex(v, target, ub=1.0):
    """Project v onto {x: 0<=x<=ub, sum x = target}."""
    # Clip to [0, ub] initially? Actually we need to find tau such that x = clip(v - tau, 0, ub) satisfies sum x=target.
    # This is a water-filling problem.
    # Use algorithm: sort v_i; find threshold tau.
    # It's similar to projection onto simplex with upper bound.
    # Implementation approach: use binary search on tau.
    lo = np.min(v - ub) - 1.0  # lower bound for tau
    hi = np.max(v) + 1.0        # upper bound for tau
    for _ in range(50):
        tau = (lo + hi) / 2.0
        x = np.clip(v - tau, 0, ub)
        s = x.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - (lo+hi)/2.0, 0, ub)
    # due to numerical error adjust sum to target by distributing residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust by adding diff to some elements not at bounds
        free = (x > 0) & (x < ub)
        if np.any(free):
            # distribute equally to free positions
            inc = diff / free.sum()
            x[free] += inc
            # clip again
            x = np.clip(x, 0, ub)
        else:
            # fallback: adjust all
            x += diff / len(x)
            x = np.clip(x, 0, ub)
    return x

def project_onto_capped_simplex(v, target, ub=1.0):
    """Project onto [0, ub] and sum=target."""
    # Ensure target in [0, n*ub]
    # Use binary search on tau: x = clip(v - tau, 0, ub). Goal sum = target.
    lo = -ub
    hi = ub
    for _ in range(50):
        tau = (lo + hi) / 2.0
        x = np.clip(v - tau, 0.0, ub)
        s = x.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - (lo + hi) / 2.0, 0.0, ub)
    # adjust residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff proportionally among free variables
        free = (x > 1e-12) & (x < ub - 1e-12)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            x = np.clip(x, 0.0, ub)
        else:
            # if no free variables, just adjust one variable within bounds
            # find index where we can add/subtract
            idx = np.argmax(x) if diff > 0 else np.argmin(x)
            x[idx] = np.clip(x[idx] + diff, 0.0, ub)
    return x

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_onto_capped_simplex(v, target, ub=1.0):
    """Project onto {x: 0<=x<=ub, sum x = target} using binary search."""
    lo, hi = -1e6, 1e6
    for _ in range(30):
        tau = (lo + hi) * 0.5
        x = np.clip(v - tau, 0.0, ub)
        if x.sum() > target:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - (lo + hi) * 0.5, 0.0, ub)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0) & (x < ub)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            x = np.clip(x, 0.0, ub)
        else:
            idx = np.argmax(x) if diff > 0 else np.argmin(x)
            x[idx] = np.clip(x[idx] + diff, 0.0, ub)
    return x

def compute_grad(h, beta, target, dx):
    """Gradient of the soft‑max max_k O_k (including dx factor)."""
    n = h.shape[0]
    # autocorrelation R_k = sum_i h_i h_{i+k}
    h_pad = np.concatenate([h, np.zeros(n)])
    fft_h = np.fft.rfft(h_pad)
    R = np.fft.irfft(fft_h * np.conj(fft_h))
    R = R[:n]  # k = 0..n‑1
    O_pos = target - R
    O_full = np.concatenate([O_pos[:0:-1], O_pos])  # length 2n‑1, symmetric
    max_O = np.max(O_full)
    exp_beta = np.exp(beta * (O_full - max_O))
    w = exp_beta / exp_beta.sum()
    # convolution w * h, we need the slice starting at offset n‑1
    conv_len = 3 * n - 2
    nfft = 1 << (conv_len - 1).bit_length()
    w_pad = np.pad(w, (0, nfft - w.shape[0]))
    h_pad2 = np.pad(h, (0, nfft - n))
    fft_w = np.fft.rfft(w_pad)
    fft_h2 = np.fft.rfft(h_pad2)
    conv = np.fft.irfft(fft_w * fft_h2, nfft)[:conv_len]
    start = n - 1
    conv_slice = conv[start:start + n]
    grad = dx * (1.0 - 2.0 * conv_slice)
    return grad

def worst_gradient(h):
    """Gradient of the worst‑shift overlap O_k (used for final hill‑climb)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.2):
    """Move ε from a positive‑gradient entry to a negative‑gradient entry."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
    else:
        # fallback: 512 points, random start
        n = 512
        h = rng.random(n)

    n = h.size
    target = n / 2.0
    dx = 2.0 / n
    h = project_onto_capped_simplex(h, target, ub=1.0)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    beta_schedule = [5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]
    iters_per_beta = 4000

    for beta in beta_schedule:
        step = 0.5
        for it in range(iters_per_beta):
            if time.time() - start > budget_s * 0.97:
                break
            grad = compute_grad(best_h, beta, target, dx)
            nh = best_h - step * grad
            nh = project_onto_capped_simplex(nh, target, ub=1.0)

            # occasional jitter
            if rng.random() < 0.01:
                noise = rng.normal(scale=0.01, size=n)
                nh = np.clip(nh + noise, 0.0, 1.0)
                nh = project_onto_capped_simplex(nh, target, ub=1.0)

            if it % 20 == 0:
                c_val = compute_c5(nh)
                if c_val < best_c:
                    best_c = c_val
                    best_h = nh.copy()

            step *= 0.9997
        # end of beta loop: one more check
        c_val = compute_c5(best_h)
        if c_val < best_c:
            best_c = c_val

    # final local refinement using worst‑gradient perturbations
    for _ in range(3000):
        if time.time() - start > budget_s * 0.995:
            break
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.1)
        c_val = compute_c5(nh)
        if c_val < best_c:
            best_c = c_val
            best_h = nh.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, target, ub=1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
