def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    from scipy.optimize import minimize

    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512

    # init h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0.copy()
    else:
        h0 = np.zeros(n)
        h0[rng.choice(n, n//2, replace=False)] = 1.0
        best_h = h0.copy()
    best_c = compute_c5(best_h)

    # SA
    current_h = best_h.copy()
    current_c = best_c

    # parameters
    T0 = 0.01
    step_scale = 0.2
    min_step = 1e-5
    decay = 0.9995
    # get time limit for SA
    while time.time() - start < budget_s * 0.85:
        elapsed = time.time() - start
        T = max(1e-12, T0 * (1 - elapsed / (budget_s * 0.85)))
        # compute gradient of worst shift
        grad = worst_gradient(current_h)
        # select candidates
        top_k = max(1, n // 20)  # 5%
        pos_idx = np.argpartition(-grad, top_k-1)[:top_k]
        neg_idx = np.argpartition(grad, top_k-1)[:top_k]
        i = int(rng.choice(pos_idx))
        j = int(rng.choice(neg_idx))
        if i == j:
            continue
        max_eps = min(current_h[i], 1.0 - current_h[j])
        if max_eps <= 1e-12:
            continue
        eps = rng.random() * step_scale * max_eps
        if eps < 1e-12:
            continue
        new_h = current_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_h = new_h
            current_c = new_c
            # update best
            if new_c < best_c:
                best_h = new_h.copy()
                best_c = new_c
                # record improvement time
        # decay step scale
        step_scale = max(min_step, step_scale * decay)
        # occasional enforce sum drift
        if (int(elapsed * 100) % 1000) == 0:
            diff = n/2 - current_h.sum()
            if abs(diff) > 1e-12:
                current_h = enforce_sum(current_h, rng)
                current_c = compute_c5(current_h)

from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        # choose indices with extreme gradient
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # fallback random
            i, j = rng.integers(0, n, size=2)
            while j == i:
                j = rng.integers(0, n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            # pick another pair randomly
            continue
        eps = rng.random() * max_eps * step_scale
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps

    return h

# Globals for SLSQP softmax
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA

            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0.copy()
    else:
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
        best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- simulated annealing ----
    current_h = best_h.copy()
    current_c = best_c
    it = 0
    T0 = 0.01
    step_scale = 0.2
    min_step = 1e-5
    decay = 0.9995
    sa_frac = 0.80
    while time.time() - start < budget_s * sa_frac:
        it += 1
        elapsed = time.time() - start
        T = max(1e-12, T0 * (1.0 - elapsed / (budget_s * sa_frac)))
        grad = worst_gradient(current_h)
        top_k = max(1, n // 20)          # 5 % of indices
        if np.allclose(grad, 0):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        else:
            pos_idx = np.argpartition(-grad, top_k - 1)[:top_k]
            neg_idx = np.argpartition(grad, top_k - 1)[:top_k]
            i = int(rng.choice(pos_idx))
            j = int(rng.choice(neg_idx))
            if i == j:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
        max_eps = min(current_h[i], 1.0 - current_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * step_scale * max_eps
        if eps < 1e-12:
            continue
        new_h = current_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - current_c
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            current_h = new_h
            current_c = new_c
            if new_c < best_c:
                best_h = new_h.copy()
                best_c = new_c
        step_scale = max(min_step, step_scale * decay)
        if it % 2000 == 0:                # occasional drift correction
            diff = n / 2.0 - current_h.sum()
            if abs(diff) > 1e-12:
                current_h = enforce_sum(current_h, rng)
                current_c = compute_c5(current_h)

    # ---- guided hill‑climb refinement ----
    if budget_s - (time.time() - start) > 5.0:
        hc, cc = guided_hill_climb(best_h, rng,
                                   max_iters=int(6000 * 256 / n),
                                   step_scale=0.5)
        if cc < best_c:
            best_h, best_c = hc, cc

    # ---- soft‑max SLSQP polishing ----
    if budget_s - (time.time() - start) > 5.0:
        for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0, 50000.0):
            if time.time() - start > budget_s * 0.99:
                break
            max_iter = 1500 if beta < 5000 else 1000
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
            best_c = compute_c5(best_h)

    # ---- final gradient polishing ----
    if budget_s - (time.time() - start) > 2.0:
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=8)
        best_c = compute_c5(best_h)

    # ---- stochastic tweaking until the very end ----
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            break
        eps = rng.random() * max_eps * 0.02
        if eps < 1e-12:
            continue
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_h, best_c = new_h, new_c

    # ---- final sanity check ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
