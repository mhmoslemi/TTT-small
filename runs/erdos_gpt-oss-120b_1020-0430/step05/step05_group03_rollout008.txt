def enforce_sum_simple(h):
    target = len(h)/2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        free = 1.0 - h
        total = free.sum()
        if total == 0:
            return h  # cannot increase
        h += diff * free / total
    else:
        free = h
        total = free.sum()
        if total == 0:
            return h
        h += diff * free / total
    return h

    rng = np.random.default_rng()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iter):
        g = worst_gradient(h)  # gradient of C5
        # descent direction is -g

        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh

            step *= 1.05
        else:
            step *= 0.5
            # If step too small, break
            if step < 1e-6:
                break
    return best_h, best_c

def smoothed_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    # compute weights safely
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()
    grad = np.zeros(n)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += wk * (1 - 2 * h)
        elif k > 0:
            # grad[:n-k] += wk*(1 - h[k:])
            # grad[k:]   += wk*(-h[:n-k])
            grad[:n - k] += wk * (1 - h[k:])
            grad[k:] += wk * (-h[:n - k])
        else:  # k < 0
            l = -k
            grad[l:] += wk * (1 - h[:n - l])
            grad[:n - l] += wk * (-h[l:])
    return grad

def enforce_sum_simple(h):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        free = 1.0 - h
        total = free.sum()
        if total == 0:
            return h
        h += diff * free / total
    else:
        free = h
        total = free.sum()
        if total == 0:
            return h
        h += diff * free / total
    # due to floating errors, adjust a random index a bit
    # ensure bounds
    # sometimes after update some entries may slightly exceed bounds due to rounding, clip
    h = np.clip(h, 0.0, 1.0)
    # final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # add diff to a random not at bound
        candidates = np.where((h > 1e-12) & (h < 1-1e-12))[0]
        if candidates.size == 0:
            # fallback to arbitrary
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        else:
            idx = candidates[rng.integers(candidates.size)]
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

import time, numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum_simple(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to satisfy Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            free = 1.0 - h
            tot = free.sum()
            if tot > 0:
                h += diff * free / tot
        else:
            free = h
            tot = free.sum()
            if tot > 0:
                h += diff * free / tot
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            cand = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
            if cand.size == 0:
                idx = rng.integers(n)
            else:
                idx = cand[rng.integers(cand.size)]
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    x = rng.random(n)
    x = x * (n / 2.0) / x.sum()
    x = np.clip(x, 0.0, 1.0)
    return enforce_sum_simple(x, rng)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the max‑overlap (C₅)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def pgd_optimize(h0: np.ndarray, rng: np.random.Generator,
                 max_iter: int = 1500, step0: float = 0.5):
    """Projected gradient descent on C₅ using worst_gradient."""
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    step = step0
    for _ in range(max_iter):
        g = worst_gradient(h)
        nh = h - step * g
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum_simple(nh, rng)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
            step *= 1.05
        else:
            step *= 0.5
            if step < 1e-8:
                break
    return best_h, best_c

def smoothed_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of soft‑max C₅ for temperature β."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    w = np.exp(bc - m)
    w /= w.sum()
    grad = np.zeros(n)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += wk * (1 - 2 * h)
        elif k > 0:
            grad[: n - k] += wk * (1 - h[k:])
            grad[k:]      += wk * (-h[: n - k])
        else:
            l = -k
            grad[l:]     += wk * (1 - h[: n - l])
            grad[: n - l] += wk * (-h[l:])
    return grad

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initial solution -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum_simple(h0, rng)
        best_h, best_c = h0.copy(), compute_c5(h0)
    else:
        n0 = 512
        h0 = random_feasible_vector(n0, rng)
        best_h, best_c = h0.copy(), compute_c5(h0)

    def try_update(h):
        nonlocal best_h, best_c
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        return c

    # ----- annealed smoothed gradient descent -----
    betas = np.geomspace(5, 5000, num=6)
    for beta in betas:
        step = 0.2 / np.sqrt(beta)
        it = 0
        while True:
            if time.time() - start > 0.78 * budget_s or it >= 500:
                break
            g = smoothed_gradient(best_h, beta)
            nh = best_h - step * g
            nh = np.clip(nh, 0.0, 1.0)
            nh = enforce_sum_simple(nh, rng)
            try_update(nh)
            it += 1
        if time.time() - start > 0.78 * budget_s:
            break

    # ----- fine‑grained projected gradient descent -----
    pgd_h, pgd_c = pgd_optimize(best_h, rng, max_iter=1200, step0=0.5)
    if pgd_c < best_c:
        best_h, best_c = pgd_h.copy(), pgd_c

    # ----- final stochastic tweaks -----
    while time.time() - start < budget_s * 0.99:
        scale = max(0.02, 0.4 * (1 - (time.time() - start) / budget_s))
        nh = guided_perturb(best_h, worst_gradient(best_h), rng, scale=scale)
        try_update(nh)

    # ----- sanity check & return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_simple(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
