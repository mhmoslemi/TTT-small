def smooth_obj_and_grad(x, n, beta, target, P):
    h = x
    sum_h = h.sum()
    diff = sum_h - target
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    # compute smooth max
    # use logsumexp
    lse = logsumexp(beta * corr) / beta
    f = (2.0 / n) * lse + P * diff * diff
    # gradient:
    # compute w_soft = exp(beta * corr - max_beta) / sum(exp(beta*corr - max_beta))
    max_beta = np.max(beta * corr)
    w_raw = np.exp(beta * corr - max_beta)
    Z = np.sum(w_raw)
    w_soft = w_raw / Z
    # compute grad_corr = weighted sum of d corr / dh (weighted shift)
    # compute weighted shifts: grad_corr_i = sum_l w_soft[l] * b[i + l] (if inside)
    # Use convolution as described
    w_rev = w_soft[::-1]
    out = np.convolve(w_rev, b, mode='full')
    # indices from 2*n-2 to 3*n-2-1 inclusive
    start = 2 * n - 2
    grad_corr = out[start:start + n]
    grad = (2.0 / n) * grad_corr + 2.0 * P * diff
    return f, grad

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ---------- utilities ----------
def compute_c5(h):
    """Exact C5 for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng, max_iter=2000, eps=1e-12):
    """Redistribute values to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        i = rng.integers(0, n)
        if diff > 0:
            delta = min(diff, 1.0 - h[i])
        else:
            delta = max(diff, -h[i])
        h[i] += delta
        diff = target - h.sum()
    return h


def weighted_corr_grad(h, w_soft):
    """Gradient of weighted correlation = sum_l w_l * sum_i h_i (1-h_{i+l})."""
    # g_i = sum_l w_l * (1 - h_{i+l})  (out‑of‑range h taken as 0)
    b = 1.0 - h
    w_rev = w_soft[::-1]
    out = np.convolve(w_rev, b, mode="full")
    n = h.size
    start = 2 * n - 2
    return out[start:start + n]


def smooth_obj_and_grad(x, n, beta, target, penalty):
    """Smooth‐max objective (log‑sum‑exp) with quadratic sum penalty."""
    h = x
    sum_h = h.sum()
    diff = sum_h - target
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")
    # smooth max term
    lse = logsumexp(beta * corr) / beta
    f = (2.0 / n) * lse + penalty * diff * diff
    # softmax weights
    max_beta = np.max(beta * corr)
    w_raw = np.exp(beta * corr - max_beta)
    Z = w_raw.sum()
    w_soft = w_raw / Z
    # gradient of smooth max term
    grad_corr = weighted_corr_grad(h, w_soft)
    grad = (2.0 / n) * grad_corr + 2.0 * penalty * diff
    return f, grad


def gradient_refine_max(h, rng, step=0.02, max_iter=12):
    """Projected subgradient descent on the true worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]            # - h[i]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            adjust_sum_exact(h, rng, max_iter=200, eps=1e-12)
    return h


def lbfgsb_cascade(h0, n, rng, deadline):
    """Run a continuation of L‑BFGS‑B optimisations over increasing β."""
    target = n / 2.0
    h = h0.copy()
    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]
    for beta in betas:
        if time.time() > deadline:
            break
        # penalty grows mildly with β
        pen = 1000.0 + 0.1 * beta
        if beta < 200:
            max_iter = 200
        elif beta < 2000:
            max_iter = 400
        else:
            max_iter = 800
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                lambda x: smooth_obj_and_grad(x, n, beta, target, pen),
                h,
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        h = np.clip(res.x, 0.0, 1.0)
        adjust_sum_exact(h, rng, max_iter=500, eps=1e-12)
    return h


def generate_initials(n, rng):
    """Return a list of diverse initial vectors for size n."""
    target = n / 2.0
    init_list = []

    # parent if available
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        init_list.append(interp)  # fractional
        # binary version of the interpolation
        idx = np.argpartition(interp, -n // 2)[-n // 2 :]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        init_list.append(bin_vec)

    # random binary
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    init_list.append(bin_rand)

    # random fractional (Dirichlet‑like)
    rand_frac = rng.random(n)
    rand_frac *= target / rand_frac.sum()
    rand_frac = np.clip(rand_frac, 0.0, 1.0)
    adjust_sum_exact(rand_frac, rng)
    init_list.append(rand_frac)

    # sinusoidal pattern
    phi = rng.random() * 2.0 * np.pi
    sin_vec = 0.5 + 0.5 * np.sin(2.0 * np.pi * np.arange(n) / n + phi)
    sin_vec = np.clip(sin_vec, 0.0, 1.0)
    adjust_sum_exact(sin_vec, rng)
    init_list.append(sin_vec)

    return init_list


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  # small safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate sizes
    if "initial_h_values" in globals():
        parent_n = len(initial_h_values)
        cand_ns = [parent_n, max(240, parent_n // 2), 180]
    else:
        cand_ns = [360, 240, 180]
    cand_ns = [n for n in cand_ns if n % 2 == 0]

    for n in cand_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        init_vecs = generate_initials(n, rng)

        for h0 in init_vecs:
            if time.time() > deadline:
                break
            h = h0.copy()
            adjust_sum_exact(h, rng)

            # quick subgradient cleanup
            h = gradient_refine_max(h, rng, step=0.02, max_iter=8)

            # L‑BFGS‑B continuation
            h = lbfgsb_cascade(h, n, rng, deadline)

            # final subgradient polish
            h = gradient_refine_max(h, rng, step=0.02, max_iter=20)

            c = compute_c5(h)
            if c < best_c:
                best_h = h.copy()
                best_c = c
                best_n = n

    # ----- progressive up‑sampling -----
    while best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        # duplicate each entry
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)

        # limited cascade on the finer grid
        h_up = gradient_refine_max(h_up, rng, step=0.02, max_iter=8)
        h_up = lbfgsb_cascade(h_up, n_up, rng, deadline)
        h_up = gradient_refine_max(h_up, rng, step=0.02, max_iter=20)

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up
        else:
            # no improvement – stop up‑sampling loop
            break

    # ----- fallback if nothing found -----
    if best_h is None:
        best_n = 180
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
