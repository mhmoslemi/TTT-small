def gradient_step(h, beta):
    O_vec = np.correlate(h, 1 - h, mode='full')[n-1:]
    maxO = O_vec.max()
    # compute weights
    maxO_shift = O_vec.max()
    exp_vals = np.exp(beta * (O_vec - maxO_shift))
    weights = exp_vals
    # Normalize
    Z = weights.sum()
    p = weights / Z
    # compute gradient
    g = np.zeros_like(h)
    # k=0
    wk = p[0]
    if wk != 0:
        g += wk * (1 - h)
    for k in range(1, n):
        wk = p[k]
        if wk == 0:
            continue
        g[:n - k] += wk * (1 - h[k:])
        g[k:] -= wk * h[:n - k]
    return g, maxO
```

import time
import numpy as np

# -----------------------------------------------------------
# Helper: projection onto [0,1] box and sum constraint Σx = target
def _project_box_sum(y, target):
    low, high = -1.0, np.max(y)  # low guarantees sum > target, high sum = 0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(y - lam, 0.0, 1.0)

# -----------------------------------------------------------
# Smooth‑max and gradient of max_k ∫ h(1‑h_k)
def _smooth_and_grad(h, beta):
    n = h.size
    O_vec = np.correlate(h, 1.0 - h, mode="full")[n - 1 :]  # k = 0..n‑1
    maxO = O_vec.max()
    # soft‑max weights (log‑sum‑exp trick)
    exp_vals = np.exp(beta * (O_vec - maxO))
    Z = exp_vals.sum()
    p = exp_vals / Z                     # Σ p = 1
    # gradient of the soft‑max objective
    g = np.zeros_like(h)
    w0 = p[0]
    if w0:
        g += w0 * (1.0 - h)
    for k in range(1, n):
        wk = p[k]
        if wk == 0.0:
            continue
        g[: n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[: n - k]
    smooth = maxO + np.log(Z) / beta
    return g, smooth, maxO

# -----------------------------------------------------------

def _descend(h, beta, target, deadline, max_iters=1000):

    # initial smooth value
    _, smooth_best, maxO_best = _smooth_and_grad(h, beta)
    best_h = h.copy()
    best_c = maxO_best * (2.0 / n)

        if time.time() > deadline:
            break
        g, smooth_curr, maxO_curr = _smooth_and_grad(h, beta)
        # line search on the smooth objective
        step_cur = step
        improved = False
        while step_cur > 1e-9:
            y = h - step_cur * g
            y = _project_box_sum(y, target)
            _, smooth_y, maxO_y = _smooth_and_grad(y, beta)
            if smooth_y < smooth_best - 1e-12:
                h = y
                smooth_best = smooth_y
                maxO_best = maxO_y

                best_c = maxO_y * (2.0 / n)

            step_cur *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-9:
                break
    return best_h, best_c

# -----------------------------------------------------------
# Main entry point
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate discretisations
    candidate_ns = [360, 720, 1080]

    # pre‑compute parent if supplied
    parent_vec = None
    if "initial_h_values" in globals():
        parent_vec = np.array(initial_h_values, dtype=float)

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0

        # number of restarts, tuned by remaining time
        time_left = deadline - time.time()
        restarts = 3
        if n > 720 and time_left < 120:
            restarts = 1

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ----- initial vector -----
            if parent_vec is not None and parent_vec.size == n:
                h0 = parent_vec.copy()
            elif parent_vec is not None and n % parent_vec.size == 0:
                factor = n // parent_vec.size
                h0 = np.repeat(parent_vec, factor)
                h0 = _project_box_sum(h0, target)
            else:
                # random start then project to feasible set
                h0 = _project_box_sum(rng.random(n), target)

            # ----- multi‑beta descent -----
            betas = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]
            h_cur = h0
            for beta in betas:
                if time.time() > deadline:
                    break
                # iteration budget shrinks for large betas
                max_it = 800 if beta < 200 else 1200
                # ensure we have enough time for the batch
                if (deadline - time.time()) < 5.0:
                    break
                h_cur, c_cur = _descend(h_cur, beta, target, deadline, max_iters=max_it)
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = h_cur.copy()
                    best_n = n

    # ----- fallback -----
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)

    # final safety projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
