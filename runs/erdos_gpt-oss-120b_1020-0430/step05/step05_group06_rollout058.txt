def compute_c5(h):
    n = h.size
    # h double; compute correlation of h and 1-h
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

import time
import numpy as np
from scipy.optimize import minimize

# globals for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx  (discretized)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Exact adjustment to enforce Σh = n/2 (h in [0,1])."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth‑max surrogate for max_k corr."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def update_O_swap(O, h_before, a, b):
    """
    Update autocorrelation O after swapping a (0→1) and b (1→0).
    h_before[a]==0, h_before[b]==1.
    """
    n = h_before.size
    ones = np.flatnonzero(h_before)          # includes b, not a
    mask = ones != b
    others = ones[mask]                      # all ones except b
    # distances to a (new 1) and b (old 1)
    d_new = np.abs(others - a)
    d_old = np.abs(others - b)
    idx_new_pos = d_new + n - 1
    idx_new_neg = (n - 1) - d_new
    idx_old_pos = d_old + n - 1
    idx_old_neg = (n - 1) - d_old
    np.add.at(O, idx_new_pos, 1)
    np.add.at(O, idx_new_neg, 1)
    np.add.at(O, idx_old_pos, -1)
    np.add.at(O, idx_old_neg, -1)
    return O


def hill_climb_targeted(h_bin, deadline):
    """
    Focused hill‑climb: repeatedly improve the minimal autocorrelation O[d].
    Returns best binary vector found and its C5 value.
    """
    n = h_bin.size
    S = n // 2
    O = np.correlate(h_bin, h_bin, mode="full")  # int64
    cur_h = h_bin.copy()
    best_h = cur_h.copy()
    best_min_O = O.min()
    best_c = S - best_min_O
    no_imp = 0
    while time.time() < deadline:

        min_O = best_min_O

        if d_min >= 0:
