    ell_vec = np.arange(2*n-1) - (n-1)

    best_h = h.copy()
    best_c = corr.max() * (2.0/n)
    while time.time() < deadline:
        cur_max = corr.max()
        cur_c = cur_max * (2.0/n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            if best_c <= 0.3808:

        shift = ell_vec[idx]  # shift = idx - (n-1)

            hj = h[:range_len]

        Z_mask = (hi == 0) & (hj == 1)

            j_local = rng.choice(Z_idx)

            if shift >= 0:
                i = i_local
                j = j_local + shift
            else:

def binary_search(h, rng, deadline):
    n = len(h)
    ell_vec = np.arange(2*n-1) - (n-1)
    corr = np.correlate(h, 1 - h, mode='full')
    corr[n-1] = 0
    best_c = corr.max() * (2.0/n)
    best_h = h.copy()
    while time.time() < deadline:
        max_corr = corr.max()
        cur_c = max_corr * (2.0/n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            # early exit if good
            if best_c <= 0.3808:
                break
        # indices of max
        idx = np.argmax(corr)
        shift = ell_vec[idx]
        # compute O, Z
        if shift >= 0:
            length = n - shift
            hi = h[:length]
            hj = h[shift:]
        else:
            length = n + shift
            hi = h[-shift:]
            hj = h[:length]
        O_mask = (hi == 1) & (hj == 0)
        Z_mask = (hi == 0) & (hj == 1)
        made_move = False
        if O_mask.any() and Z_mask.any():
            i_local = rng.choice(np.flatnonzero(O_mask))
            j_local = rng.choice(np.flatnonzero(Z_mask))
            if shift >= 0:
                i = i_local
                j = j_local + shift
            else:
                i = i_local - shift
                j = j_local
            # attempt swap
            old_max = max_corr
            apply_change(corr, h, i, -1, ell_vec)
            apply_change(corr, h, j, +1, ell_vec)
            new_max = corr.max()
            if new_max <= old_max:
                made_move = True
            else:
                # revert
                apply_change(corr, h, j, -1, ell_vec)
                apply_change(corr, h, i, +1, ell_vec)
        if not made_move:
            # try second worst
            top_idx = np.argpartition(corr, -2)[-2:]
            for idx2 in top_idx:
                if idx2 == idx:
                    continue
                shift2 = ell_vec[idx2]
                if shift2 >= 0:
                    length = n - shift2
                    hi = h[:length]
                    hj = h[shift2:]
                else:
                    length = n + shift2
                    hi = h[-shift2:]
                    hj = h[:length]
                O_mask = (hi == 1) & (hj == 0)
                Z_mask = (hi == 0) & (hj == 1)
                if O_mask.any() and Z_mask.any():
                    i_local = rng.choice(np.flatnonzero(O_mask))
                    j_local = rng.choice(np.flatnonzero(Z_mask))
                    if shift2 >= 0:
                        i = i_local
                        j = j_local + shift2
                    else:
                        i = i_local - shift2
                        j = j_local
                    old_max = max_corr
                    apply_change(corr, h, i, -1, ell_vec)
                    apply_change(corr, h, j, +1, ell_vec)
                    new_max = corr.max()
                    if new_max <= old_max:
                        made_move = True
                    else:
                        apply_change(corr, h, j, -1, ell_vec)
                        apply_change(corr, h, i, +1, ell_vec)
                    break
        if not made_move:
            # fallback: pick i from O if any else any 1; pick j any 0
            if O_mask.any():
                i_local = rng.choice(np.flatnonzero(O_mask))
                if shift >= 0:
                    i = i_local
                else:
                    i = i_local - shift
            else:
                i = rng.choice(np.flatnonzero(h))
            j = rng.choice(np.flatnonzero(1 - h))
            old_max = max_corr
            apply_change(corr, h, i, -1, ell_vec)
            apply_change(corr, h, j, +1, ell_vec)

            else:
                # revert
                apply_change(corr, h, j, -1, ell_vec)
                apply_change(corr, h, i, +1, ell_vec)
        # loop continues
    return best_h, best_c

import time
import numpy as np

def compute_c5(arr):
    n = arr.size
    corr = np.correlate(arr.astype(float), 1.0 - arr.astype(float), mode="full")
    return float(corr.max() * (2.0 / n))

def apply_change(corr, h, p, delta, ell):
    """Sequentially add (+1) or remove (-1) a unit at position p."""
    n = h.size
    # first index contribution
    mask1 = (p + ell >= 0) & (p + ell < n)
    term1 = 1 - h[p + ell[mask1]]
    corr[mask1] += delta * term1
    # second index contribution
    mask2 = (p - ell >= 0) & (p - ell < n)
    term2 = h[p - ell[mask2]]
    corr[mask2] -= delta * term2
    # update the value itself
    h[p] = 1 if delta > 0 else 0
    # zero‑shift must stay zero for binary vectors
    corr[n - 1] = 0

def binary_local_search(h, rng, deadline):
    """Greedy / SA style search on binary vectors."""
    n = h.size
    ell = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1 - h, mode="full")
    corr[n - 1] = 0  # enforce zero shift
    best_h = h.copy()
    best_c = corr.max() * (2.0 / n)
    no_improve = 0

    while time.time() < deadline:
        max_corr = corr.max()
        cur_c = max_corr * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
            if best_c <= 0.3808:
                break

        idx = np.argmax(corr)
        shift = ell[idx]

        # helpers to extract O/Z masks for a given shift
        def masks_for(s):
            if s >= 0:
                L = n - s
                hi = h[:L]
                hj = h[s:]
            else:
                L = n + s
                hi = h[-s:]
                hj = h[:L]
            O = (hi == 1) & (hj == 0)
            Z = (hi == 0) & (hj == 1)
            return O, Z, L

        O_mask, Z_mask, L = masks_for(shift)

        made_move = False

        # try targeted swap on worst shift
        if O_mask.any() and Z_mask.any():
            i_loc = rng.choice(np.flatnonzero(O_mask))
            j_loc = rng.choice(np.flatnonzero(Z_mask))
            if shift >= 0:
                i = i_loc
                j = j_loc + shift
            else:
                i = i_loc - shift
                j = j_loc
            if i != j:
                old_max = max_corr
                apply_change(corr, h, i, -1, ell)
                apply_change(corr, h, j, +1, ell)
                new_max = corr.max()
                if new_max <= old_max:
                    made_move = True
                else:
                    # revert
                    apply_change(corr, h, j, -1, ell)
                    apply_change(corr, h, i, +1, ell)

        # second‑worst shift
        if not made_move:
            top_idxs = np.argpartition(corr, -2)[-2:]
            for idx2 in top_idxs:
                if idx2 == idx:
                    continue
                shift2 = ell[idx2]
                O_mask2, Z_mask2, L2 = masks_for(shift2)
                if O_mask2.any() and Z_mask2.any():
                    i_loc = rng.choice(np.flatnonzero(O_mask2))
                    j_loc = rng.choice(np.flatnonzero(Z_mask2))
                    if shift2 >= 0:
                        i = i_loc
                        j = j_loc + shift2
                    else:
                        i = i_loc - shift2
                        j = j_loc
                    if i != j:
                        old_max = max_corr
                        apply_change(corr, h, i, -1, ell)
                        apply_change(corr, h, j, +1, ell)
                        new_max = corr.max()
                        if new_max <= old_max:
                            made_move = True
                        else:
                            apply_change(corr, h, j, -1, ell)
                            apply_change(corr, h, i, +1, ell)
                    break

        # fallback random move (try to reduce the current worst lag)
        if not made_move:
            # pick a contributing one if possible
            if O_mask.any():
                i_loc = rng.choice(np.flatnonzero(O_mask))
                i = i_loc if shift >= 0 else i_loc - shift
            else:
                i = rng.choice(np.flatnonzero(h))
            j = rng.choice(np.flatnonzero(1 - h))
            if i != j:
                old_max = max_corr
                apply_change(corr, h, i, -1, ell)
                apply_change(corr, h, j, +1, ell)
                new_max = corr.max()
                if new_max <= old_max:
                    made_move = True
                else:
                    # revert
                    apply_change(corr, h, j, -1, ell)
                    apply_change(corr, h, i, +1, ell)

        if made_move:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 2000:
                # random perturbation to escape a plateau
                i = rng.choice(np.flatnonzero(h))
                j = rng.choice(np.flatnonzero(1 - h))
                apply_change(corr, h, i, -1, ell)
                apply_change(corr, h, j, +1, ell)
                no_improve = 0

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # target resolution (even)
    n = 960
    if n % 2:
        n += 1

    # initialise from parent if present, otherwise random binary
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        idx = np.argpartition(interp, -n // 2)[-n // 2:]
        h = np.zeros(n, dtype=np.uint8)
        h[idx] = 1
    else:
        h = np.zeros(n, dtype=np.uint8)
        h[rng.choice(n, n // 2, replace=False)] = 1

    best_h, best_c = binary_local_search(h, rng, deadline)

    # final sanity checks
    best_h = np.clip(best_h, 0, 1)
    final_c = compute_c5(best_h.astype(float))

    return best_h.tolist(), float(final_c), int(n)
