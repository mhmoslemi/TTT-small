def binary_sa_fast(h_bin, rng, deadline):
    n = len(h_bin)
    offset = n - 1
    # compute A
    A = np.correlate(h_bin, h_bin, mode='full')
    min_A = A.min()
    cur_c = (n/2 - min_A) * (2.0 / n)
    best_h = h_bin.copy()
    best_c = cur_c

    # maintain ones and zeros as Python lists (or numpy arrays)
    ones = np.flatnonzero(h_bin).tolist()
    zeros = np.flatnonzero(1 - h_bin).tolist()

    it = 0
    init_T = max(0.01 * cur_c, 1e-4)  # maybe scaling
    while time.time() < deadline:
        it += 1
        T = init_T * (0.9995 ** it) + 1e-12
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # compute delta updates
        # Exclude i and j from p list
        # Use numpy array of p_sel
        p_ones = np.array(ones, dtype=int)
        mask = (p_ones != i) & (p_ones != j)
        p_sel = p_ones[mask]
        if p_sel.size > 0:
            idx_i = offset + (p_sel - i)
            idx_j = offset + (p_sel - j)
            A[idx_i] -= 1
            A[idx_j] += 1

        # compute new min_A
        new_min_A = A.min()
        new_c = (n/2 - new_min_A) * (2.0 / n)

        # Decide acceptance
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            # accept with probability exp(-(new_c - cur_c)/T)
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True

        if accept:
            # commit swap
            h_bin[i] = 0
            h_bin[j] = 1
            # update ones and zeros lists
            # remove i from ones, add j
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            cur_c = new_c
            min_A = new_min_A
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_bin.copy()
        else:
            # revert changes on A
            if p_sel.size > 0:
                A[idx_i] += 1
                A[idx_j] -= 1
            # h unchanged, ones/zeros unchanged

    return best_h, best_c

def greedy_improve(h, rng, max_iter=30, deadline=None):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time()>deadline: break
        corr = np.correlate(h, 1-h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0/n)
        k = max_idx - (n - 1)
        # find mismatched positions
        i_range = np.arange(n)
        # condition where h[i]=1 and i+k in range and h[i+k]=0
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        # sample some swaps
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 300
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=30):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]           # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_fast(h_bin, rng, deadline):
    n = len(h_bin)
    offset = n - 1
    A = np.correlate(h_bin, h_bin, mode='full')
    min_A = A.min()
    cur_c = (n / 2.0 - min_A) * (2.0 / n)
    best_h = h_bin.copy()
    best_c = cur_c

    ones = np.flatnonzero(h_bin).tolist()
    zeros = np.flatnonzero(1 - h_bin).tolist()
    it = 0
    init_T = max(0.02 * cur_c, 5e-4)

    while time.time() < deadline:
        it += 1
        T = init_T * (0.9995 ** it) + 1e-12
        i = rng.choice(ones)
        j = rng.choice(zeros)

        if len(ones) > 2:
            p_arr = np.array(ones, dtype=int)
            mask = (p_arr != i) & (p_arr != j)
            p_sel = p_arr[mask]
            idx_i = offset + (p_sel - i)
            idx_j = offset + (p_sel - j)
            A[idx_i] -= 1
            A[idx_j] += 1
        else:
            p_sel = np.array([], dtype=int)

        new_min_A = A.min()
        new_c = (n / 2.0 - new_min_A) * (2.0 / n)

        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True

        if accept:
            h_bin[i] = 0
            h_bin[j] = 1
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            cur_c = new_c
            min_A = new_min_A
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_bin.copy()
        else:
            if p_sel.size > 0:
                A[idx_i] += 1
                A[idx_j] -= 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = None
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    cand = []
    if parent is not None:
        parent_n = parent.size
        cand.append(parent_n)
        for k in (parent_n // 2, parent_n * 2 // 3, parent_n * 3 // 4):
            if k >= 120 and k % 2 == 0:
                cand.append(k)
    for n in (720, 600, 480, 360, 240, 180):
        if n >= 120 and n % 2 == 0:
            cand.append(n)
    cand = sorted(set(cand), reverse=True)

    for idx, n in enumerate(cand):
        if time.time() >= deadline:
            break
        remaining = len(cand) - idx
        time_slice = (deadline - time.time()) / remaining
        # initial binary vector
        if parent is not None and n == parent.size:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            idx_on = np.argpartition(interp, -n // 2)[-n // 2:]
            h_bin = np.zeros(n, dtype=np.uint8)
            h_bin[idx_on] = 1
        else:
            h_bin = np.zeros(n, dtype=np.uint8)
            h_bin[rng.choice(n, n // 2, replace=False)] = 1

        # SA phase
        sa_dead = time.time() + time_slice * 0.85
        best_bin, best_bin_c = binary_sa_fast(h_bin.copy(), rng, sa_dead)

        # Gradient refinement on float version
        h_f = best_bin.astype(float)
        h_f = gradient_refine(h_f, rng, step=0.02, max_iter=30)
        adjust_sum_exact(h_f, rng)
        c_f = compute_c5(h_f)

        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()
            best_n = n

    # Upsample if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        h_up = np.clip(h_up, 0.0, 1.0)
        adjust_sum_exact(h_up, rng)
        # SA on upsampled binary version (thresholded)
        h_up_bin = (h_up > 0.5).astype(np.uint8)
        sa_dead = deadline
        best_up_bin, _ = binary_sa_fast(h_up_bin, rng, sa_dead)
        h_up_f = best_up_bin.astype(float)
        h_up_f = gradient_refine(h_up_f, rng, step=0.02, max_iter=30)
        adjust_sum_exact(h_up_f, rng)
        c_up = compute_c5(h_up_f)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up_f.copy()
            best_n = n_up

    if best_h is None:
        best_n = 180
        best_h = np.zeros(best_n, dtype=float)
        best_h[:best_n // 2] = 1.0

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
