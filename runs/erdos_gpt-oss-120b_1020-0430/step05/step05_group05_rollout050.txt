def anneal(h, rng, time_limit):
    n = len(h)
    best_h = h.copy()
    best_min = np.min(autocorr(h))
    # compute correlation once
    cur_corr = autocorr(h)
    cur_min = cur_corr.min()
    best_corr = cur_corr.copy()
    start = time.time()
    temperature = 0.1 * n  # initial scale maybe?
    while time.time() < time_limit:
        # pick random 1 and 0
        i = rng.choice(np.where(h==1)[0])
        j = rng.choice(np.where(h==0)[0])
        # swap
        h[i], h[j] = 0, 1
        # recompute correlation
        corr = autocorr(h)
        min_corr = corr.min()
        # simple acceptance if min_corr >= cur_min - temp*... else maybe revert
        if min_corr > cur_min or rng.random() < np.exp((min_corr - cur_min) / max(temperature, 1e-12)):
            cur_corr = corr
            cur_min = min_corr
            if min_corr > best_min:
                best_min = min_corr
                best_h = h.copy()
                best_corr = corr.copy()
        else:
            # revert swap
            h[i], h[j] = 1, 0
        # cooling
        # maybe reduce temperature slowly
        temperature *= 0.99999
    return best_h, best_min
```

def swap_delta(f, h, i, j):
    n = len(h)
    # indices for all k
    ks = np.arange(n)
    # compute (i+ks) % n
    ip = (i + ks) % n
    im = (i - ks) % n
    jp = (j + ks) % n
    jm = (j - ks) % n
    delta = -h[ip] - h[im] + h[jp] + h[jm]
    delta[0] = 0
    return delta
```

import time
import numpy as np

def autocorr(h: np.ndarray) -> np.ndarray:
    """Cyclic autocorrelation of a binary (or 0/1) vector."""
    n = h.shape[0]
    ft = np.fft.fft(h)
    corr = np.fft.ifft(ft * ft).real
    # round to nearest integer to avoid floating noise
    return np.rint(corr).astype(int)

def swap_delta(f: np.ndarray, h: np.ndarray, i: int, j: int) -> np.ndarray:
    """Delta of autocorrelation f when swapping h[i] (1) with h[j] (0)."""
    n = h.size
    ks = np.arange(n)
    ip = (i + ks) % n
    im = (i - ks) % n
    jp = (j + ks) % n
    jm = (j - ks) % n
    delta = -h[ip] - h[im] + h[jp] + h[jm]
    delta[0] = 0  # shift 0 must stay n/2
    return delta

def compute_c5_from_f(f: np.ndarray, n: int) -> float:
    """C5 from autocorrelation f (cyclic)."""
    min_f = f.min()
    omax = n / 2 - min_f
    return omax * 2.0 / n

def compute_c5(h: np.ndarray) -> float:
    """Reference C5 using np.correlate (as in the original evaluation)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

    """Deterministic greedy improvement using exact delta updates."""
    n = h.size
    f = autocorr(h)
    f[0] = n // 2

        cur_min = f.min()
        K = np.where(f == cur_min)[0]
        K = K[K != 0]  # ignore shift 0
        if K.size == 0:

        # scores for ones (how much they participate in the worst shifts)
        mask_one = np.zeros(n, dtype=int)
        mask_zero = np.zeros(n, dtype=int)
        for k in K:

            hk_b = np.roll(h, k)
            mask_one += (h & hk_f) + (h & hk_b)
            mask_zero += ((1 - h) & hk_f) + ((1 - h) & hk_b)
        # best candidates
        max_score = mask_one.max()
        min_score = mask_zero.min()
        ones_cand = np.flatnonzero((h == 1) & (mask_one == max_score))
        zeros_cand = np.flatnonzero((h == 0) & (mask_zero == min_score))
        if ones_cand.size == 0 or zeros_cand.size == 0:
            break
        improved = False
        for _ in range(6):

            j = rng.choice(zeros_cand)

def anneal(h: np.ndarray, rng: np.random.Generator, deadline: float):
    """Simple simulated annealing using exact delta updates."""
    n = h.size
    f = autocorr(h)
    f[0] = n // 2
    cur_min = f.min()
    best_h = h.copy()
    best_min = cur_min
    temperature = 0.2 * n  # initial temperature

        j = rng.choice(np.flatnonzero(1 - h))

        new_min = min(f[0], (f[1:] + delta[1:]).min())
        accept = False

    else:
        n_candidates = [360, 720]

    for n in n_candidates:
        # Prepare a small pool of initial seeds
        seeds = []

        # Parent-derived binary (if applicable)
        if 'initial_h_values' in globals() and n == len(initial_h_values):
            parent = np.array(initial_h_values, dtype=float)
            idx = np.argpartition(parent, -n // 2)[-n // 2 :]
            hb = np.zeros(n, dtype=np.int8)
            hb[idx] = 1
            seeds.append(hb)

        # Random binary seeds
        for _ in range(4):
            hb = np.zeros(n, dtype=np.int8)
            hb[rng.choice(n, n // 2, replace=False)] = 1
            seeds.append(hb)

        # Fractional random seeds (turned binary later)
        for _ in range(2):
            hf = rng.random(n)
            hf *= (n / 2) / hf.sum()
            hf = np.clip(hf, 0.0, 1.0)
            adjust_sum_exact(hf, rng)
            # convert to binary by top half
            idx = np.argpartition(hf, -n // 2)[-n // 2 :]
            hb = np.zeros(n, dtype=np.int8)
            hb[idx] = 1
            seeds.append(hb)

        # Greedy improvement on each seed
        for seed_h in seeds:
            if time.time() > deadline:
                break
            h_opt, f_opt = greedy_improve(seed_h.copy(), rng, deadline)
            c = compute_c5_from_f(f_opt, n)
            if c < best_c:
                best_c = c
                best_h = h_opt.copy()
                best_n = n

        # Additional random restarts while time permits
        extra_trials = 0
        while time.time() < deadline and extra_trials < 30:
            hb = np.zeros(n, dtype=np.int8)
            hb[rng.choice(n, n // 2, replace=False)] = 1
            h_opt, f_opt = greedy_improve(hb, rng, deadline)
            c = compute_c5_from_f(f_opt, n)
            if c < best_c:
                best_c = c
                best_h = h_opt.copy()
                best_n = n
            extra_trials += 1

    # Final annealing refinement on the best found binary solution
    if best_h is not None and time.time() < deadline:
        h_ann, min_f = anneal(best_h.copy(), rng, deadline)
        c_ann = compute_c5_from_f(autocorr(h_ann), best_n)
        if c_ann < best_c:
            best_c = c_ann
            best_h = h_ann.copy()

    # Safety cleanup: ensure constraints are met exactly
    if best_h is None:
        # fallback: simple half‑ones pattern
        best_n = 360
        best_h = np.zeros(best_n, dtype=np.int8)
        best_h[: best_n // 2] = 1
    best_h = np.clip(best_h, 0, 1).astype(float)
    # enforce exact sum
    target = best_n / 2.0
    diff = target - best_h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, best_n)
        add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
        best_h[i] += add
        diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
