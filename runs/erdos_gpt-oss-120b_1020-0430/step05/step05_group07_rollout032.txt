import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.03, max_iter=8):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = last_imp = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c, best_h = c, cur_h.copy()
                last_imp = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_imp > 8000:
            break
    return best_h, best_c

def fractional_sa_long(h, rng, deadline, max_iter=500000):
    cur = h.copy()
    cur_c = compute_c5(cur)
    best, best_c = cur.copy(), cur_c
    it = 0
    init_T = max(0.01 * cur_c, 1e-5)
    n = len(cur)
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_t = min(cur[src], 1.0 - cur[dst])
        if max_t <= 0:
            it += 1
            continue
        delta = max_t * (rng.random() ** 0.5)
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9998 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c, best = c, cur.copy()
                init_T = max(0.01 * cur_c, 1e-5)
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    n_base = 360
    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    start_vecs = []
    if parent is not None:
        xs = np.linspace(0, 2, parent.size, endpoint=False)
        xt = np.linspace(0, 2, n_base, endpoint=False)
        interp = np.interp(xt, xs, parent)
        start_vecs.append(interp)
        idx = np.argpartition(interp, -n_base // 2)[-n_base // 2:]
        bin_vec = np.zeros(n_base)
        bin_vec[idx] = 1.0
        start_vecs.append(bin_vec)

    bin_rand = np.zeros(n_base)
    bin_rand[rng.choice(n_base, n_base // 2, replace=False)] = 1.0
    start_vecs.append(bin_rand)

    target = n_base / 2.0
    rand_frac = rng.random(n_base)
    rand_frac *= target / rand_frac.sum()
    adjust_sum_exact(rand_frac, rng)
    start_vecs.append(rand_frac)

    rand_f2 = rng.random(n_base) ** 2
    rand_f2 *= target / rand_f2.sum()
    adjust_sum_exact(rand_f2, rng)
    start_vecs.append(rand_f2)

    best_h = None
    best_c = np.inf
    best_n = n_base

    for h0 in start_vecs:
        if time.time() > deadline:
            break
        h = np.clip(h0, 0.0, 1.0)
        adjust_sum_exact(h, rng)
        h = gradient_refine(h, rng, step=0.04, max_iter=6)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n_base

        sub_dead = min(deadline, time.time() + min(5.0, (deadline - time.time()) * 0.2))
        h, c = fractional_sa_long(h, rng, sub_dead)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n_base

        if deadline - time.time() > 5:
            for beta in [10.0, 30.0, 100.0, 300.0, 1000.0]:
                if time.time() > deadline:
                    break
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=500)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n_base

        h = gradient_refine(h, rng, step=0.02, max_iter=8)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n_base

    if best_h is not None and time.time() < deadline:
        n_up = min(720, best_h.shape[0] * 2)
        xs = np.linspace(0, 2, best_h.shape[0], endpoint=False)
        xt = np.linspace(0, 2, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        h_up, c_up = fractional_sa_long(h_up, rng, deadline)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

        if time.time() < deadline:
            idx = np.argpartition(best_h, -best_h.shape[0] // 2)[-best_h.shape[0] // 2:]
            bin_h = np.zeros_like(best_h)
            bin_h[idx] = 1.0
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.2)
            bin_best, bin_c = binary_sa_hill_climb(bin_h.copy(), rng, sub_dead)
            if bin_c < best_c:
                best_h, best_c, best_n = bin_best.astype(float), bin_c, bin_best.shape[0]

        best_h = gradient_refine(best_h, rng, step=0.02, max_iter=10)
        best_c = compute_c5(best_h)

    if best_h is None:
        n = n_base
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.shape[0])
