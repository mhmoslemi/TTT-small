    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0.1
    while time.time() < time_deadline:
        A = np.correlate(h, h, mode='full')
        Apos = A[n-1:]
        minA = Apos.min()
        ks = np.where(Apos == minA)[0]
        # compute gradient g
        g = np.zeros_like(h)
        for k in ks:
            if k > 0:
                g[:n-k] += h[k:]
                g[k:] += h[:n-k]
        # zero mean to preserve sum
        g = g - g.mean()
        # apply step
        h_new = h + step * g
        # clip

            h = h_new
        else:
            step *= 0.5
            if step < 1e-6:
                break
        # decaying step
        step *= 0.99
    return best_h, best_c

    deadline = time.time() + time_limit
    # compute initial A and minA
    A = np.correlate(h, h, mode='full')
    Apos = A[n-1:]
    minA = int(Apos.min())
    cur_c = 1.0 - (2.0 * minA) / n
    best_h = h.copy()
    best_c = cur_c
    step = 0.1
    while time.time() < deadline:
        # find worst ks
        ks = np.where(Apos == minA)[0]
        # compute gradient g
        g = np.zeros_like(h)
        for k in ks:
            if k == 0:
                continue
            g[:n-k] += h[k:]
            g[k:] += h[:n-k]
        # zero-mean
        g -= g.mean()
        if np.allclose(g, 0):
            break
        h_new = h + step * g
        # clip and adjust sum
        h_new = np.clip(h_new, 0.0, 1.0)
        adjust_sum_exact(h_new, rng)
        # recompute A and minA
        A_new = np.correlate(h_new, h_new, mode='full')
        Apos_new = A_new[n-1:]
        minA_new = int(Apos_new.min())

            A = A_new

            step *= 1.05
        else:
            step *= 0.5
            if step < 1e-6:

import time
import numpy as np
from scipy.optimize import minimize

# ----- globals for smooth‑max surrogate -----
CURRENT_TARGET = None
CURRENT_BETA = None

# ----- utilities -----
def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth‑max of the overlap values."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def gradient_minA_ascent(h, n, rng, time_limit):
    """Increase the minimal auto‑correlation (thus lower C₅)."""
    deadline = time.time() + time_limit

        ks = np.where(Apos == minA)[0]          # shifts with minimal overlap
        g = np.zeros_like(h)

                continue

        A_new = np.correlate(h_new, h_new, mode="full")
        Apos_new = A_new[n - 1 :]
        minA_new = int(Apos_new.min())
        c_new = 1.0 - (2.0 * minA_new) / n
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if c_new < cur_c - 1e-12:
            h, A, Apos, minA, cur_c = h_new, A_new, Apos_new, minA_new, c_new
            step *= 1.05
        else:
            step *= 0.5
            if step < 1e-6:
                break
        step *= 0.99
    return best_h, best_c

# ----- main entry -----
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        parent = np.asarray(parent, dtype=float)

    # candidate sample sizes (even, ≤720)
    cand_ns = [720, 600, 480, 360, 240, 180]
    cand_ns = [n for n in cand_ns if n % 2 == 0]

    for n in cand_ns:
        if time.time() > deadline:
            break

        # ----- generate starting vectors -----
        start_vecs = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp)

        # three random fractional starts
        for _ in range(3):
            rnd = rng.random(n)
            rnd *= (n / 2.0) / rnd.sum()
            rnd = np.clip(rnd, 0.0, 1.0)
            adjust_sum_exact(rnd, rng)
            start_vecs.append(rnd)

        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h = np.clip(h0.copy(), 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # ----- SLSQP cascade (smooth‑max) -----
            betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
                     2000.0, 5000.0, 10000.0, 20000.0)
            for beta in betas:
                if time.time() > deadline:
                    break
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=500)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c - 1e-12:
                    best_c, best_h, best_n = c, h.copy(), n

            # ----- targeted gradient ascent on min‑auto‑correlation -----
            remaining = deadline - time.time()
            if remaining > 0.02:
                h_opt, c_opt = gradient_minA_ascent(h, n, rng, remaining)
                if c_opt < best_c - 1e-12:
                    best_c, best_h, best_n = c_opt, h_opt.copy(), n

    # ----- fallback if nothing succeeded -----
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    # ----- final safety checks -----
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
