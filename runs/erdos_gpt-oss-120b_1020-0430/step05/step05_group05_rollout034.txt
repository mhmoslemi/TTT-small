def smoothmax_optimize(h, rng, max_iters=2000, init_lr=0.01, beta0=5.0, beta_multiplier=1.05, max_beta=20000.0):
    n = h.size; target = n/2.0.
    m = np.zeros_like(h); v = np.zeros_like(h)
    beta = beta0
    lr = init_lr
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(1, max_iters+1):
        H = np.fft.fft(h)
        C = np.real(np.fft.ifft(np.abs(H)**2))
        O = target - C  # length n
        maxO = O.max()
        # Evaluate c
        c_curr = maxO * (2.0 / n)
        if c_curr < best_c:
            best_c = c_curr; best_h = h.copy()
        # compute w for smoothmax
        exp_beta = np.exp(beta * (O - maxO))
        w = exp_beta / exp_beta.sum()
        # gradient
        W = np.fft.fft(w)
        conv_minus = np.real(np.fft.ifft(W * H))
        w_rev = np.concatenate([w[:1], w[:0:-1]])  # w_rev[0]=w[0]; w_rev[1:]=w[-1:1:-1]
        W_rev = np.fft.fft(w_rev)
        conv_plus = np.real(np.fft.ifft(W_rev * H))
        grad = 1.0 - conv_plus - conv_minus
        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad**2)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = h - step
        # projection and sum adjust
        h = np.clip(h, 0.0, 1.0)
        # adjust sum efficiently:
        diff = target - h.sum()
        # We can distribute diff across random indices not at bounds:
        # Use simple loop:
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
        # update beta
        if beta < max_beta:
            beta = min(beta * beta_multiplier, max_beta)
        # optional learning rate decay
        if t % 200 == 0:
            lr *= 0.9
        # time check: break if out of budget
    return best_h, best_c

import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def fractional_to_binary(h, rng):
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    adjust_sum_exact(bin_h, rng)
    return bin_h

def binary_targeted_improve(h, rng, deadline):
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        cand_i = ones[neigh[ones] == neigh[ones].min()]
        i = rng.choice(cand_i)
        cand_j = zeros[neigh[zeros] == neigh[zeros].max()]
        j = rng.choice(cand_j)
        delta = neigh[i] - neigh[j]
        if delta < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            d = neigh[i2] - neigh[j2]
            if d < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

def smoothmax_optimize(h0, rng, max_iters=2000, init_lr=0.01):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta = 5.0
    max_beta = 20000.0
    lr = init_lr
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(1, max_iters + 1):
        H = np.fft.fft(h)
        C = np.real(np.fft.ifft(np.abs(H)**2))
        O = target - C
        maxO = O.max()
        c_curr = maxO * (2.0 / n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h.copy()
        # smoothmax weights
        maxO_s = maxO  # for stability
        exp_beta = np.exp(beta * (O - maxO_s))
        w = exp_beta / exp_beta.sum()
        W = np.fft.fft(w)
        conv_minus = np.real(np.fft.ifft(W * H))
        w_rev = np.concatenate([w[:1], w[:0:-1]])
        W_rev = np.fft.fft(w_rev)
        conv_plus = np.real(np.fft.ifft(W_rev * H))
        grad = 1.0 - conv_plus - conv_minus
        # Adam

import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def fractional_to_binary(h, rng):
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return adjust_sum_exact(bin_h, rng)

def binary_targeted_improve(h, rng, deadline):
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones[neigh[ones] == neigh[ones].min()])
        j = rng.choice(zeros[neigh[zeros] == neigh[zeros].max()])
        if neigh[i] - neigh[j] < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                break
        else:
            break
    return h

def smoothmax_optimize(h0, rng, max_iters=1500, init_lr=0.02):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    m = np.zeros_like(h)

    beta = 5.0
    max_beta = 20000.0
    lr = init_lr
    best_h = h.copy()
    best_c = compute_c5(h)
    for t in range(1, max_iters + 1):
        H = np.fft.fft(h)
        C = np.real(np.fft.ifft(np.abs(H) ** 2))
        O = target - C
        maxO = O.max()
        c_cur = maxO * (2.0 / n)
        if c_cur < best_c:
            best_c, best_h = c_cur, h.copy()
        exp_beta = np.exp(beta * (O - maxO))
        w = exp_beta / exp_beta.sum()
        W = np.fft.fft(w)

        w_rev = np.concatenate([w[:1], w[:0:-1]])
