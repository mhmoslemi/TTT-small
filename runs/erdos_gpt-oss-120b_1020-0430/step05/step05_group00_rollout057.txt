def binary_sa_fast(N, rng, time_budget):
    start = time.time()
    # initial random
    h = np.zeros(N, dtype=int)
    ones = rng.choice(N, N//2, replace=False)
    h[ones] = 1
    zeros = np.setdiff1d(np.arange(N), ones)  # but we need arrays
    ones_arr = ones.copy()
    zeros_arr = np.setdiff1d(np.arange(N), ones) # also copy
    O = np.correlate(h, 1 - h, mode='full')[N-1:]
    best = h.copy()
    best_max = O.max()
    K = np.arange(1, N, dtype=int)
    # temperature schedule
    T0 = 0.5
    T_end = 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end/T0)**(elapsed/time_budget)
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = int(ones_arr[i_pos])
        j = int(zeros_arr[j_pos])
        # compute delta
        ip = i + K
        im = i - K
        jp = j + K
        jm = j - K
        term_i_plus = np.where(ip < N, h[ip], 0)
        term_i_minus = np.where(im >= 0, h[im], 0)
        term_j_plus = np.where(jp < N, h[jp], 0)
        term_j_minus = np.where(jm >= 0, h[jm], 0)
        delta = (term_i_plus + term_i_minus) - (term_j_plus + term_j_minus)
        # compute new max
        O_new = O.copy()
        O_new[1:] = O[1:] + delta
        O_new[0] = 0
        new_max = O_new.max()
        delta_max = new_max - O.max()
        if delta_max <= 0 or rng.random() < np.exp(-delta_max / max(T, 1e-12)):
            # accept
            h[i] = 0
            h[j] = 1
            # update arrays
            ones_arr[i_pos] = j
            zeros_arr[j_pos] = i
            O = O_new
            if new_max < best_max:
                best_max = new_max
                best = h.copy()
                # early stop if below target
                if best_max * (2.0 / N) <= 0.3808:
                    return best, best_max * (2.0 / N)
    return best, best_max * (2.0 / N)

def binary_sa_fast(N, rng, time_budget):
    start = time.time()
    # init random binary with half ones
    h = np.zeros(N, dtype=np.int8)
    ones = rng.choice(N, N // 2, replace=False)
    h[ones] = 1
    # create arrays of ones and zeros positions
    ones_arr = ones.copy()
    zeros_arr = np.setdiff1d(np.arange(N, dtype=np.int64), ones)  # could be large but okay
    # compute O
    O = compute_O_array(h)
    current_max = O.max()
    best_h = h.copy()
    best_max = current_max
    K = np.arange(1, N, dtype=np.int64)
    T0 = 0.7
    T_end = 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # select random 1 and 0
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = int(ones_arr[i_pos])
        j = int(zeros_arr[j_pos])
        # compute delta
        ip = i + K
        im = i - K
        jp = j + K
        jm = j - K
        term_i_plus = np.where(ip < N, h[ip], 0)
        term_i_minus = np.where(im >= 0, h[im], 0)
        term_j_plus = np.where(jp < N, h[jp], 0)
        term_j_minus = np.where(jm >= 0, h[jm], 0)
        delta = (term_i_plus + term_i_minus) - (term_j_plus + term_j_minus)
        cand = O[1:] + delta
        new_max = cand.max()
        delta_max = new_max - current_max
        if delta_max <= 0 or rng.random() < np.exp(-delta_max / max(T, 1e-12)):

            h[i] = 0
            h[j] = 1
            # update position arrays
            ones_arr[i_pos] = j

                best_max = new_max
                best_h = h.copy()

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for a step function on [0,2]."""
    cor = np.correlate(h, 1.0 - h, mode="full")
    return np.max(cor) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

# ----------------------------------------------------------------------
def compute_O_array(h: np.ndarray) -> np.ndarray:
    """O_k = sum_i h_i (1 - h_{i+k}) for k = 0..N-1 (int version for binary)."""
    h_i = h.astype(np.int64)
    cor = np.correlate(h_i, 1 - h_i, mode="full")
    O = cor[h_i.size - 1 :].astype(np.int64)   # length N, O[0] == 0
    return O

def binary_sa_fast(N: int, rng: np.random.Generator, time_budget: float):
    """Simulated‑annealing for binary vectors with exact sum constraint.
       Uses O(N) incremental updates for the overlap."""
    start = time.time()
    # initialise a random half‑filled binary vector
    h = np.zeros(N, dtype=np.int8)
    ones = rng.choice(N, N // 2, replace=False)
    h[ones] = 1
    ones_arr = ones.copy()
    zeros_arr = np.setdiff1d(np.arange(N, dtype=np.int64), ones)
    O = compute_O_array(h)
    current_max = O.max()
    best_h = h.copy()
    best_max = current_max
    K = np.arange(1, N, dtype=np.int64)
    delta = np.empty(N - 1, dtype=np.int64)

    T0 = 0.7
    T_end = 1e-6

    target_c5 = 0.3808

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)

        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = int(ones_arr[i_pos])
        j = int(zeros_arr[j_pos])

        # build delta = (h_{i+k}+h_{i-k})-(h_{j+k}+h_{j-k})
        delta.fill(0)
        # i contributions (+)
        if i < N - 1:
            delta[: N - i - 1] += h[i + 1 :]
        if i > 0:
            delta[:i] += h[i - 1 :: -1]          # reversed prefix
        # j contributions (‑)
        if j < N - 1:
            delta[: N - j - 1] -= h[j + 1 :]
        if j > 0:
            delta[:j] -= h[j - 1 :: -1]

        # new maximum after the swap
        cand = O[1:] + delta
        new_max = cand.max()
        delta_max = new_max - current_max

        if delta_max <= 0 or rng.random() < np.exp(-delta_max / max(T, 1e-12)):
            # accept move
            h[i] = 0
            h[j] = 1
            ones_arr[i_pos] = j
            zeros_arr[j_pos] = i
            O[1:] = cand
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                if best_max * (2.0 / N) <= target_c5:
                    return best_h, best_max * (2.0 / N)

    return best_h, best_max * (2.0 / N)
