def smooth_max_and_grad(h, beta, target_sum, gamma):
    n = h.size
    y = np.correlate(h, 1.0 - h, mode='full')
    max_y = y.max()
    diff = y - max_y
    # w = exp(beta*diff)
    w = np.exp(beta * diff)
    w_sum = w.sum()
    w_norm = w / w_sum
    # smooth max value:
    smax = max_y + np.log(w_sum) / beta
    # gradient of smooth max part:
    c = 1.0 - 2.0 * h
    grad_smax = np.correlate(w_norm, c, mode='valid')
    # penalty gradient:
    sum_diff = h.sum() - target_sum
    grad_pen = 2.0 * gamma * sum_diff
    grad = grad_smax + grad_pen
    # Add gradient of penalty w.r.t each component is same grad_pen
    # Actually grad_pen is scalar; need vector of same size:
    grad = grad_smax + grad_pen  # broadcasting adds scalar to each entry
    # total objective:
    obj = smax + gamma * (sum_diff ** 2)
    return obj, grad
```

def lbfgsb_optimize(h0, beta, target_sum, gamma, max_iter, rng):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        obj, grad = smooth_max_and_grad(x, beta, target_sum, gamma)
        return obj, grad
    res = minimize(fun, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum exactly
    adjust_sum_exact(x, rng)
    return x

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact sum correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def smooth_max_and_grad(h, beta, target_sum, gamma):
    """Smooth max of the overlap vector and its gradient."""
    y = np.correlate(h, 1.0 - h, mode="full")
    max_y = y.max()
    diff = y - max_y
    # stable softmax weights
    w = np.exp(beta * diff)
    w_sum = w.sum()
    w_norm = w / w_sum
    smax = max_y + np.log(w_sum) / beta
    c = 1.0 - 2.0 * h  # derivative of each term inside the correlation
    grad_smax = np.correlate(w_norm, c, mode="valid")
    sum_diff = h.sum() - target_sum
    grad = grad_smax + 2.0 * gamma * sum_diff   # penalty gradient adds same scalar to each entry
    obj = smax + gamma * (sum_diff ** 2)
    return obj, grad

def lbfgsb_optimize(h0, beta, target_sum, gamma, max_iter, rng):
    """L‑BFGS‑B on the smooth‑max surrogate with a quadratic sum penalty."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        obj, grad = smooth_max_and_grad(x, beta, target_sum, gamma)
        return obj, grad

    res = minimize(
        fun,
        h0,
        method="L-BFGS-B",
        jac=True,
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # Determine problem size
    if "initial_h_values" in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        n = h_parent.size
        best_h = h_parent.copy()
        best_c = compute_c5(best_h)
    else:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)

    target_sum = n / 2.0
    beta_seq = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]
    gamma = 10.0

    # Refine the parent (or default) solution through the beta cascade
    h_cur = best_h.copy()
    for beta in beta_seq:
        if time.time() > deadline:
            break
        h_cur = lbfgsb_optimize(h_cur, beta, target_sum, gamma, max_iter=600, rng=rng)
        c_val = compute_c5(h_cur)
        if c_val < best_c - 1e-12:
            best_c = c_val
            best_h = h_cur.copy()

    # Random restarts with the same cascade
    while time.time() < deadline:
        # binary start with half ones
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h0[idx] = 1.0
        # small perturbation to escape symmetries
        h0 += rng.uniform(-0.07, 0.07, size=n)
        h0 = np.clip(h0, 0.0, 1.0)
        adjust_sum_exact(h0, rng)

        h_cur = h0.copy()
        for beta in beta_seq:
            if time.time() > deadline:
                break
            h_cur = lbfgsb_optimize(h_cur, beta, target_sum, gamma, max_iter=500, rng=rng)
        c_val = compute_c5(h_cur)
        if c_val < best_c:
            best_c = c_val
            best_h = h_cur.copy()

    # Final polishing on the best solution using a higher iteration budget
    if time.time() < deadline - 5:
        for beta in (5000, 10000):
            if time.time() > deadline:
                break
            best_h = lbfgsb_optimize(best_h, beta, target_sum, gamma, max_iter=1500, rng=rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
