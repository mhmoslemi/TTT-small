def gradient_refine_cont(h, rng, step=0.015, max_iter=10):
    N = h.size
    for it in range(max_iter):
        # compute current worst shift k
        corr = full_correlation_fft(h, 1 - h)  # length 2N
        # find index of max correlation
        idx = np.argmax(corr)  # idx in [0, 2N-1]
        k = idx - (N - 1)
        k_abs = abs(k)
        # compute gradient
        g = np.zeros_like(h)
        if k_abs == 0:
            g = 1.0 - 2.0 * h
        else:
            # For shift k, O_k = sum_i h_i * (1 - h_{i+k})
            # gradient of O_k w.r.t h_i: (1 - h_{i+k}) - h_{i-k}
            # Derivation: O = Σ_i h_i - Σ_i h_i h_{i+k}
            # dO/dh_i = 1 - h_{i+k} - h_{i-k}
            # but we want gradient of max O_k; using worst shift k.
            # We'll use that.
            # compute term1 = 1 - h[(i+k)%N]
            term1 = 1.0 - np.roll(h, -k_abs)  # roll forward
            term2 = -np.roll(h, k_abs)        # -h_{i-k}
            g = term1 + term2  # shape N
        # update
        h = h - (step * (0.5 ** it)) * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        target = N / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution similar to earlier
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, N)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def sum_eq_constraint(x):

def softmax_obj(x):
    corr = full_correlation_fft(x, 1.0 - x)  # returns length 2N
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute smoothmax using log-sum-exp stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    N = h0.size
    CURRENT_TARGET = N / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * N
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # adjust randomly
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, N)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global parameters for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def full_correlation_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (full) correlation of a and b using FFT."""
    n = a.shape[0]
    m = 2 * n
    A_rev = np.fft.rfft(a[::-1], n=m)
    B = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(A_rev * B, n=m)
    return conv[: 2 * n - 1]  # length 2n‑1


def compute_c5_fft(h: np.ndarray) -> float:
    """C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    corr = full_correlation_fft(h, 1.0 - h)
    return float(corr.max() * (2.0 / n))


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = full_correlation_fft(x, 1.0 - x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return float((m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA)


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Smooth‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine_cont(h: np.ndarray, rng: np.random.Generator,
                         step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune a continuous h using the worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = full_correlation_fft(h, 1.0 - h)
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        k = abs(shift)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            if shift >= 0:
                term1 = 1.0 - np.roll(h, -shift)
                term2 = -np.roll(h, shift)
            else:
                s = -shift
                term1 = 1.0 - np.roll(h, s)
                term2 = -np.roll(h, -s)
            g = term1 + term2
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:

def binary_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      time_limit: float) -> tuple:
    """Greedy hill‑climb on binary vectors using C5 as objective."""
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    start = time.time()
    while time.time() - start < time_limit:
        # current correlation and C5
        corr = full_correlation_fft(h, 1.0 - h)
        cur_c = float(corr.max() * (2.0 / n))
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)  # may be negative
        # build candidate indices for this shift
        if shift >= 0:
            i_vals = np.arange(n - shift)
            h_i = h[i_vals]
            h_s = h[i_vals + shift]
        else:
            k = -shift
            i_vals = np.arange(k, n)
            h_i = h[i_vals]
            h_s = h[i_vals - k]
        cand_i = i_vals[(h_i == 0) & (h_s == 1)]
        cand_j = i_vals[(h_i == 1) & (h_s == 0)]
        if cand_i.size == 0 or cand_j.size == 0:
            break  # no improving move possible for this shift
        improved = False
        for _ in range(5):  # a few attempts
            i = int(rng.choice(cand_i))
            j = int(rng.choice(cand_j))
            h[i] = 1.0
            h[j] = 0.0
            new_c = compute_c5_fft(h)
            if new_c < cur_c - 1e-12:
                cur_c = new_c
                improved = True
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                break
            # revert
            h[i] = 0.0
            h[j] = 1.0
        if not improved:
            break
    return best_h, best_c


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
