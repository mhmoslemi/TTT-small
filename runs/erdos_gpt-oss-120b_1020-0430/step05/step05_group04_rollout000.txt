def smooth_max_obj_grad(h, beta):
    n = h.shape[0]
    O = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    maxO = O.max()
    exp_term = np.exp(beta * (O - maxO))
    w = exp_term / exp_term.sum()
    g = np.ones(n)
    # loop over lag
    for lag in range(-(n-1), n):
        weight = w[lag + n - 1]
        if weight == 0.0:
            continue
        if lag >= 0:
            # forward: h[i+lag]
            g[:n - lag] -= weight * h[lag:]
            # backward: h[i-lag]
            g[lag:] -= weight * h[:n - lag]
        else:
            k = -lag
            # For negative lag, forward/backward similar but reversed direction
            # Here lag <0: forward term corresponds to h[i+lag] i+lag = i-k, valid for i>=k
            # We'll treat as:
            # forward: i from k..n-1 have i+lag = i-k => subtract weight * h[i - k]
            g[k:] -= weight * h[:n - k]
            # backward: i from 0..n-1-k have i - lag = i + k => subtract weight * h[i + k]
            g[:n - k] -= weight * h[k:]
    # after loop, gradient is ready.
    # value:
    val = (maxO + np.log(np.sum(exp_term))) / beta
    return val, g
```

Check for double counting: For lag>0 we subtract weight * h[lag:] from g[:n-lag]; also subtract weight * h[:n-lag] from g[lag:]. For lag<0 we execute similar subtract. This ensures that each pair (i,i+lag) counted twice across lag and -lag. But that's correct as the derivative formula includes both terms.

Potential double subtract: For lag>0, i will be part of forward subtraction for some i and backward for some other i; but that's acceptable.

We also need to ensure at lag=0, both forward and backward use same weight: code for lag>=0 uses both slices; for lag=0, slices are [:] and [:], so g -= weight * h twice (once forward, once backward). This yields g = 1 - 2 weight * h. Since sum of weight over all lags = 1, total coefficient for h at lag=0 is -2 * w_0. This matches derivative of O(0) (which is 1 - 2 h_i). Correct.

Thus gradient is correct.

        idx = np.argmax(corr)
        k = idx - (n - 1)
        # find mismatched positions
        if k >= 0:
            S1 = np.flatnonzero((h[:n - k] == 1) & (h[k:] == 0))
            S0 = np.flatnonzero((h[:n - k] == 0) & (h[k:] == 1))
        else:
            kabs = -k
            S1 = np.flatnonzero((h[kabs:] == 1) & (h[:n - kabs] == 0))
            S0 = np.flatnonzero((h[kabs:] == 0) & (h[:n - kabs] == 1))
        if S1.size == 0 or S0.size == 0:
            break
        # try random subset of swaps
        improve = False
        for _ in range(min(200, S1.size * S0.size)):
            i = rng.choice(S1)
            j = rng.choice(S0)
            h[i] = 0
            h[j] = 1
            c = compute_c5(h)
            if c < best_c - 1e-12:
                best_c = c
                improve = True
                no_improve = 0
                # update mismatches sets maybe skip
                break
            else:
                # revert
                h[i] = 1
                h[j] = 0
        if not improve:
            no_improve += 1
    return h, best_c

import time, numpy as np

# ---------------------------------------------------------
# Helper utilities
# ---------------------------------------------------------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Shift mass to make sum(h) == n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def smooth_max_obj_grad(h, beta):
    """Smooth max of overlap integrals and its gradient."""
    n = h.shape[0]
    O = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = O.max()
    exp_term = np.exp(beta * (O - m))
    w = exp_term / exp_term.sum()
    g = np.ones(n)
    # iterate over all lags
    for lag in range(-(n - 1), n):
        weight = w[lag + n - 1]
        if weight == 0.0:
            continue
        if lag >= 0:
            g[: n - lag] -= weight * h[lag:]
            g[lag:]      -= weight * h[: n - lag]
        else:
            k = -lag
            g[k:]       -= weight * h[: n - k]
            g[: n - k]  -= weight * h[k:]
    val = (m + np.log(np.sum(exp_term))) / beta
    return val, g

def grad_descent_smooth_max(h, betas, step0, steps_per_beta, rng):
    """Stage‑1 descent on smooth‑max surrogate."""
    for i, beta in enumerate(betas):
        step = step0 * (0.6 ** i)  # decay with beta index
        for _ in range(steps_per_beta):
            _, g = smooth_max_obj_grad(h, beta)
            h -= step * g
            np.clip(h, 0.0, 1.0, out=h)
            adjust_sum_exact(h, rng)
    return h

def grad_refine_worst(h, rng, step=0.02, max_iter=15):
    """Projected gradient on the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.ones(n)
        if k == 0:
            g -= 2.0 * h
        else:
            kabs = abs(k)
            if k > 0:
                g[: n - kabs] -= h[kabs:]
                g[kabs:]      -= h[: n - kabs]
            else:
                g[kabs:]      -= h[: n - kabs]
                g[: n - kabs] -= h[kabs:]
        step_sz = step * (0.5 ** it)
        h -= step_sz * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa(h_bin, rng, time_dead):
    """Simulated‑annealing on binary vectors."""
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur))
    zeros = list(np.flatnonzero(1 - cur))
    it = 0
    while time.time() < time_dead:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        c = compute_c5(cur)
        T = max(1e-6, 0.1 * cur_c * np.exp(-it / 20000.0))
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it > 200000:  # safety break
            break
    return best, best_c

def greedy_binary(h, rng, max_no_improve=2000):
    """Swap‑based local improvement on binary vectors."""
    n = h.size
    best_c = compute_c5(h)
    no_improve = 0
    while no_improve < max_no_improve:

                improved = True
                no_improve = 0
                break
            h[i] = 1
            h[j] = 0
        if not improved:
            no_improve += 1
