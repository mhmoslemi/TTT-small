import time
import numpy as np
from scipy.optimize import minimize

# ---------- global vars for soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] using linear correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max O_k via log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def proj_capped(x: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0 ≤ y ≤ 1, sum(y)=target} via binary search on λ."""
    lo = np.min(x - 1.0) - 1.0   # ensures sum > target
    hi = np.max(x)              # ensures sum < target
    for _ in range(50):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if needed
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        idx = np.where((y > 0.0) & (y < 1.0))[0]
        if idx.size == 0:
            idx = np.arange(y.size)
        i = idx[0]
        y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of O_k for the shift attaining the current maximum."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)      # shift index (may be negative)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 5000, step_scale: float = 0.6):
    """Monotone hill‑climb using gradient‑directed mass transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(cur_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # fallback random pair
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step_scale
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5(nh)
        if c < cur_c:
            cur_h, cur_c = nh, c
            if c < best_c:
                best_c, best_h = c, nh.copy()
        # decay step_scale occasionally
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def random_feasible(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2 via projection."""
    x = rng.random(n)  # uniform in [0,1]
    return proj_capped(x, n / 2.0)

    start = time.time()

    # ---- initialise ----------------------------------------------------
    if 'initial_h_values' in globals():

        h0 = random_feasible(rng, n)

    best_h = h0.copy()

    elapsed = time.time() - start

    while time.time() - start < total_budget * 0.70:

            max_eps = min(h_tmp[i], 1.0 - h_tmp[j])

            h_tmp[j] += eps
        h_tmp = proj_capped(h_tmp, n / 2.0)

    # ---- Phase 3: soft‑max SLSQP refinement ----------------------------

        h_up = np.clip(h_up, 0.0, 1.0)

        # hill‑climb at higher resolution
        elapsed = time.time() - start
        remaining = total_budget - elapsed
        iters_up = int(min(30000, remaining * 500))
        h_opt, c_opt = guided_hill_climb(h_up, rng,

        if c_opt < best_c:
