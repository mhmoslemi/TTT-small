    half = half.copy()

import time
import numpy as np
import scipy.optimize as opt

# ----- Global for smooth‑max -----
CURRENT_BETA = 1.0

def half_to_full(half):
    """Construct full symmetric vector of length 2*len(half)."""
    return np.concatenate((half, 1.0 - half[::-1]))

def compute_c5(full):
    """Maximum overlap integral for a full vector."""
    n = full.size
    corr = np.correlate(full, 1.0 - full, mode="full")
    return float(np.max(corr) * (2.0 / n))

def smooth_max_obj_half(half):
    """Smooth‑max surrogate of the overlap for a half‑vector."""
    global CURRENT_BETA
    full = half_to_full(half)
    corr = np.correlate(full, 1.0 - full, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def lbfgsb_optimize_half(h0, beta, max_iter=400):
    """L‑BFGS‑B on the smooth‑max surrogate for a half‑vector."""
    global CURRENT_BETA
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * h0.size
    res = opt.minimize(
        smooth_max_obj_half,
        h0,
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    return np.clip(res.x, 0.0, 1.0)

def worst_shift(full):
    """Return signed shift (index‑offset) that attains the worst overlap."""
    n = full.size
    corr = np.correlate(full, 1.0 - full, mode="full")
    idx = np.argmax(corr)
    return idx - (n - 1)

def gradient_half(full, shift):
    """Gradient of the worst‑shift overlap w.r.t. the half‑vector."""
    n = full.size
    idx = np.arange(n)
    i_plus = idx + shift
    i_minus = idx - shift
    term1 = np.where((i_plus >= 0) & (i_plus < n), 1.0 - full[i_plus], 0.0)
    term2 = np.where((i_minus >= 0) & (i_minus < n), full[i_minus], 0.0)
    g_full = term1 - term2
    m = n // 2
    # g_half = g_full[:m] - g_full[n-1:n-1-m:-1]
    return g_full[:m] - g_full[n - 1 : n - 1 - m : -1]

def gradient_refine_half(half, rng, step_init=0.05, max_iter=200):
    """Projected gradient descent against the current worst shift."""
    for it in range(max_iter):
        full = half_to_full(half)
        shift = worst_shift(full)
        grad = gradient_half(full, shift)
        step = step_init * (0.5 ** it)
        half -= step * grad
        half = np.clip(half, 0.0, 1.0)
        # occasional tiny random perturbation to escape flat regions
        if rng.random() < 0.01:
            half += rng.normal(scale=0.01, size=half.shape)
            half = np.clip(half, 0.0, 1.0)
        if np.linalg.norm(step * grad) < 1e-9:
            break
    return half

def half_to_binary(half):
    """Binary rounding of a half‑vector (top half entries become 1)."""
    m = half.size
    k = m // 2
    idx = np.argpartition(half, -k)[-k:]
    b = np.zeros_like(half)
    b[idx] = 1.0
    return b

def binary_targeted_improve(h, rng, deadline):
    """Greedy swaps on a binary full vector to lower the worst overlap."""
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        shift = abs(shift)
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        # Choose the 1 with minimal neighbour sum and 0 with maximal neighbour sum
        i = ones[neigh[ones].argmin()]
        j = zeros[neigh[zeros].argmax()]
        if neigh[i] - neigh[j] < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h
