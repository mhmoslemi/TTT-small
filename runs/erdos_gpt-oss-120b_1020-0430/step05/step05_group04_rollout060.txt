def greedy_binary(h, rng, deadline):
    n = h.size
    dx = 2.0 / n
    c_full = np.correlate(h, h, mode='full')
    c = c_full[n-1:].astype(int)
    missing = 1.0 - c * dx
    missing[0] = 0.0
    best_missing = missing.max()
    best_h = h.copy()
    # loop
    while time.time() < deadline:
        worst_k = int(np.argmax(missing))
        worst_val = missing[worst_k]
        if worst_val <= 0.3808:  # target reached
            break
        # compute match for worst_k only (to pick i/j)
        match = np.zeros(n, dtype=int)
        if worst_k != 0:
            match[:n-worst_k] += h[worst_k:]
            match[worst_k:] += h[:n-worst_k]
        # pick i,j
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        i_min = ones[np.argmin(match[ones])]
        j_max = zeros[np.argmax(match[zeros])]
        if match[j_max] <= match[i_min]:
            # greedy stuck; break
            break
        # compute contributions for i_min and j_max (full)
        contrib_i = np.zeros(n, dtype=int)
        contrib_j = np.zeros(n, dtype=int)
        # k=0
        contrib_i[0] = 1
        contrib_j[0] = 1
        # forward/backward as above
        # use slices
        if n - i_min - 1 > 0:
            contrib_i[1:n-i_min] = h[i_min+1:]
        if i_min > 0:
            contrib_i[1:i_min+1] += h[i_min-1::-1]
        if n - j_max - 1 > 0:
            contrib_j[1:n-j_max] = h[j_max+1:]
        if j_max > 0:
            contrib_j[1:j_max+1] += h[j_max-1::-1]
        # swap h
        h[i_min] = 0
        h[j_max] = 1
        # update c
        c += -contrib_i + contrib_j
        # recompute missing
        missing = 1.0 - c * dx
        missing[0] = 0.0
        cur_max = missing.max()
        if cur_max < best_missing:
            best_missing = cur_max
            best_h = h.copy()
    return best_h, best_missing

def binary_sa_opt(h, rng, deadline):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, h, mode='full')[n-1:].astype(int)
    cur_min = c[1:].min() if n > 1 else c[0]
    cur_val = 1.0 - cur_min * dx
    best_h = h.copy()
    best_val = cur_val
    T = max(cur_val * 0.5, 1e-6)
    cooling = 0.9997
    while time.time() < deadline:
        # pick random swap
        ones_idx = np.flatnonzero(h)
        zeros_idx = np.flatnonzero(1 - h)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # contributions
        contrib_i = contributions(h, i)
        contrib_j = contributions(h, j)
        c_new = c - contrib_i + contrib_j
        new_min = c_new[1:].min() if n > 1 else c_new[0]
        new_val = 1.0 - new_min * dx
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            h[i] = 0
            h[j] = 1
            c = c_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        T = max(T * cooling, 1e-12)
    return best_h, best_val

import time
import numpy as np

# ---------- helpers ----------
def contributions(h: np.ndarray, idx: int) -> np.ndarray:
    """Return integer array of length n where entry k is the contribution
    of a 1 at position idx to the autocorrelation count c_k.
    k=0 contributes 1 (self‑pair), for k>0 includes forward and backward neighbour."""
    n = h.shape[0]
    contrib = np.zeros(n, dtype=np.int32)
    contrib[0] = 1
    forward_len = n - idx - 1
    if forward_len > 0:
        contrib[1 : forward_len + 1] = h[idx + 1 : idx + 1 + forward_len]
    if idx > 0:
        contrib[1 : idx + 1] += h[idx - 1 :: -1]
    return contrib


def binary_sa_opt(h: np.ndarray, rng: np.random.Generator, deadline: float):
    """Simulated‑annealing for binary h to minimise max_k ∫h(1‑h_k).
    Returns best vector and its C5 value."""
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, h, mode="full")[n - 1 :].astype(np.int32)
    cur_min = c[1:].min() if n > 1 else c[0]
    cur_val = 1.0 - cur_min * dx
    best_h, best_val = h.copy(), cur_val
    T = max(cur_val * 0.5, 1e-6)
    cooling = 0.9997

    while time.time() < deadline:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        ci = contributions(h, i)
        cj = contributions(h, j)
        c_new = c - ci + cj

        new_min = c_new[1:].min() if n > 1 else c_new[0]
        new_val = 1.0 - new_min * dx
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            h[i] = 0
            h[j] = 1
            c = c_new
            cur_val = new_val
            if cur_val < best_val:
                best_val, best_h = cur_val, h.copy()

        T = max(T * cooling, 1e-12)

    return best_h, best_val


def hill_climb_locals(h: np.ndarray, rng: np.random.Generator, deadline: float):
    """Deterministic hill climbing: accept only moves that improve C5."""
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, h, mode="full")[n - 1 :].astype(np.int32)

    while time.time() < deadline:

def final_c5_from_h(h: np.ndarray) -> float:
    n = h.shape[0]
