import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def proj_sum(y, target, eps=1e-10, max_iter=50):
    lo, hi = -1.0, 1.0
    # Ensure bounds
    # Bisection
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - lo, 0.0, 1.0)
    # final correction
    diff = target - x.sum()
    if abs(diff) > eps:
        # distribute diff among free variables
        mask = (x > eps) & (x < 1 - eps)
        if mask.any():
            delta = diff / mask.sum()
            x[mask] += delta
            x = np.clip(x, 0.0, 1.0)
    return x

def adjust_sum_exact(x, target, rng, max_iter=2000):
    diff = target - x.sum()
    n = x.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def proj_sum(y: np.ndarray, target: float, eps: float = 1e-10, max_iter: int = 50) -> np.ndarray:
    """Project y onto {0≤x≤1, sum(x)=target} using bisection on a shift λ."""
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - lo, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > eps:
        mask = (x > eps) & (x < 1.0 - eps)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def adjust_sum_exact(x: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Exact correction of Σx to target by moving mass on random coordinates."""
    diff = target - x.sum()
    n = x.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ----- discretisation -----
    n = 720                     # chosen sample size
    target = n / 2.0

    # ----- initialise from parent if present -----
    if "initial_h_values" in globals():
        init0 = np.array(initial_h_values, dtype=float)
        n0 = init0.size
        if n % n0 == 0:
            h = np.repeat(init0, n // n0)
        else:
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, init0)
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    h = proj_sum(h, target)
    h = adjust_sum_exact(h, target, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- optimisation schedule -----
    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]
    iters_per_beta = [300] * 8 + [1000]   # more effort at the final sharp beta

    restarts = 0
    while time.time() < deadline and restarts < 3:
        cur = best_h.copy()
        for beta, iters in zip(betas, iters_per_beta):
            if time.time() > deadline:
                break

            # smooth‑max value (log‑sum‑exp)
            corr = np.correlate(cur, 1 - cur, mode="full")
            m = np.max(corr)
            smooth = (np.log(np.exp(beta * corr - beta * m).sum()) / beta) + m
            alpha = 0.5 / np.sqrt(beta)

            for _ in range(iters):
                if time.time() > deadline:
                    break

                # gradient of smooth‑max
                corr = np.correlate(cur, 1 - cur, mode="full")
                m = np.max(corr)
                w = np.exp(beta * corr - beta * m)
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.ones_like(w) / w.size
                else:
                    w /= w_sum

                conv_full = np.convolve(w, cur, mode="full")
                conv1 = conv_full[n - 1 : n - 1 + n]
                conv2 = np.convolve(w[::-1], cur, mode="full")[n - 1 : n - 1 + n]
                grad = 1.0 - conv1 - conv2

                # simple backtracking line‑search
                step = alpha
                improved = False
                for _ in range(10):
                    y = cur - step * grad
                    x_new = proj_sum(y, target)
                    corr_new = np.correlate(x_new, 1 - x_new, mode="full")
                    m_new = np.max(corr_new)
                    smooth_new = (np.log(np.exp(beta * corr_new - beta * m_new).sum()) / beta) + m_new
                    if smooth_new <= smooth - 1e-12:
                        cur = x_new
                        smooth = smooth_new
                        improved = True
                        break
                    step *= 0.5
                if not improved:
                    break

            # mild random jitter before next beta
            if time.time() > deadline:
                break
            jitter = rng.normal(scale=0.02 / np.sqrt(beta), size=n)
            cur = proj_sum(cur + jitter, target)

        c_val = compute_c5(cur)
        if c_val < best_c:
            best_c = c_val
            best_h = cur.copy()
            if best_c < 0.3808:
                break

        # prepare next restart (perturb current best)
        perturb = rng.normal(scale=0.05, size=n)
        best_h = proj_sum(best_h + perturb, target)
        best_h = adjust_sum_exact(best_h, target, rng)
        restarts += 1

    # ----- final polishing -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_sum(best_h, target)
    best_h = adjust_sum_exact(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
