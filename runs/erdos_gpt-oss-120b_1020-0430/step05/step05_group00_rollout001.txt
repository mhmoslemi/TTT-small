def binary_sa_global(n, rng, time_budget):
    start = time.time()
    # initialize vector
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr

    # precompute lists
    ones_arr = np.where(h==1)[0]
    zeros_arr = np.where(h==0)[0]

    T0 = 0.1
    T_min = 1e-5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(T_min, T0 * (T_min/T0) ** (elapsed / time_budget))  # exponential cooling
        # choose swap
        i = rng.choice(ones_arr)
        j = rng.choice(zeros_arr)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            c_curr = c_new
            # update arrays
            # replace i in ones_arr with j
            # replace j in zeros_arr with i
            idx_i = np.where(ones_arr == i)[0][0]
            idx_j = np.where(zeros_arr == j)[0][0]
            ones_arr[idx_i] = j
            zeros_arr[idx_j] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                # possible early exit if below target
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def binary_sa_single_run(n, rng, time_budget):
    # start with random binary vector
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr

    # maintain lists
    ones_list = ones.tolist()
    zeros_list = [i for i in range(n) if h[i] == 0.0]
    # ensure length sums
    T0 = 0.2
    T_min = 1e-6
    start = time.time()
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(T_min, T0 * (T_min/T0) ** (elapsed / time_budget))
        i_idx = rng.integers(len(ones_list))
        j_idx = rng.integers(len(zeros_list))
        i = ones_list[i_idx]
        j = zeros_list[j_idx]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        accept = delta <= 0.0 or rng.random() < np.exp(-delta / T)
        if accept:
            c_curr = c_new
            # update lists
            ones_list[i_idx] = j
            zeros_list[j_idx] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    break
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c
```

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def to_binary_by_top_half(arr: np.ndarray) -> np.ndarray:
    """Return binary vector with exactly half ones (top half of values)."""
    n = arr.size
    k = n // 2
    # argpartition for efficiency
    idx = np.argpartition(-arr, k - 1)[:k]
    bin_arr = np.zeros_like(arr)
    bin_arr[idx] = 1.0
    return bin_arr


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator,
                   scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def binary_sa_single_run(n: int, rng: np.random.Generator, time_budget: float):
    """Simulated‑annealing on binary vectors with exactly n/2 ones."""
    start = time.time()
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr

    ones_list = ones.tolist()
    all_idx = np.arange(n)
    zeros = np.setdiff1d(all_idx, ones, assume_unique=True)
    zeros_list = zeros.tolist()

    T0 = 0.2
    T_min = 1e-6
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = max(T_min, T0 * (T_min / T0) ** (elapsed / time_budget))
        i_idx = rng.integers(len(ones_list))
        j_idx = rng.integers(len(zeros_list))
        i = ones_list[i_idx]
        j = zeros_list[j_idx]

        # swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            c_curr = c_new
            ones_list[i_idx] = j
            zeros_list[j_idx] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # Use parent construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        # enforce exact half‑ones binary projection if needed
        if abs(h0.sum() - h0.size / 2) > 1e-12:
            h0 = to_binary_by_top_half(h0)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # ---------- binary SA on 512 points ----------
    n = 512
    time_sa = total * 0.70
    run_time = 12.0
    n_runs = max(1, int(time_sa // run_time))
    run_time = time_sa / n_runs  # adjust to fill the allocated time

    for _ in range(n_runs):
        if time.time() - start > total * 0.70:
            break
        h_bin, c_bin = binary_sa_single_run(n, rng, run_time)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin.copy(), c_bin, n
            if best_c <= 0.3808:
                break

    # ---------- guided hill‑climb on current best ----------
    remaining = total - (time.time() - start)
    if remaining > total * 0.05:
        iters = int(min(20000, remaining * 3000))
        h_hc, c_hc = guided_hill_climb(best_h.copy(), rng,
                                        max_iters=iters, step_scale=0.5)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc

    # ---------- up‑sample to double resolution ----------
    remaining = total - (time.time() - start)
    if remaining > total * 0.05:
        n_up = best_n * 2
        if n_up <= 1024:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_interp = np.interp(x_up, x_old, best_h)
            h_up = to_binary_by_top_half(h_interp)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ---------- short binary SA with remaining time ----------
    remaining = total - (time.time() - start)
    if remaining > total * 0.02:
        h_bin2, c_bin2 = binary_sa_single_run(best_n, rng, remaining)
        if c_bin2 < best_c:
            best_h, best_c = h_bin2.copy(), c_bin2

    # ---------- final sanitisation ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = to_binary_by_top_half(best_h)  # enforce exact half‑ones
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
