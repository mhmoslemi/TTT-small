Define enforce_sum:

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # tolerance
    eps = 1e-12
    max_iter = 10000
    it = 0
    while abs(diff) > eps and it < max_iter:
        # choose random index where adjustment possible
        if diff > 0:
            # need to add
            idxs = np.where(h < 1 - eps)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            # need to subtract
            idxs = np.where(h > eps)[0]
            if idxs.size == 0:
                break
            i = rng.choice(idxs)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        it += 1
    return h

# *** Erdős C5 optimizer ***
import time, numpy as np
from scipy.optimize import minimize, LinearConstraint

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    eps = 1e-12
    it = 0
    while abs(diff) > eps and it < 10000:
        if diff > 0:
            cand = np.where(h < 1.0 - eps)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            cand = np.where(h > eps)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        it += 1
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


# ---------- correlation helpers (FFT) ----------
def full_corr(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear correlation of a and b (mode='full'), via FFT."""
    n = a.size + b.size - 1
    fsize = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, fsize)
    Fb = np.fft.rfft(b[::-1], fsize)
    conv = np.fft.irfft(Fa * Fb, fsize)[:n]
    return conv


def valid_corr(q: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Correlation of q with v, cropped to length len(v) (valid part)."""
    n = v.size
    conv = full_corr(q, v)
    start = n - 1
    return conv[start:start + n]


# ---------- gradient of soft‑max max‑overlap ----------
def sgd_softmax(h: np.ndarray, rng: np.random.Generator,
                betas, steps_per_beta: int = 150, step0: float = 0.8) -> np.ndarray:
    """Projected stochastic gradient descent on the softmax approximation."""
    h = enforce_sum(np.clip(h, 0.0, 1.0), rng)
    n = h.size
    for beta in betas:
        step = step0
        for _ in range(steps_per_beta):
            O = full_corr(h, 1.0 - h)                 # length 2n‑1
            maxO = O.max()
            exp_shift = beta * (O - maxO)             # ≤0
            exp_beta = np.exp(exp_shift)
            sum_exp = exp_beta.sum()
            f = (maxO + np.log(sum_exp)) / beta       # soft‑max value
            q = exp_beta / sum_exp                    # probability vector, length 2n‑1

            # gradient of soft‑max (vector length n)
            term1 = valid_corr(q, 1.0 - h)            # Σ q_k (1‑h_{i+k})
            term2 = valid_corr(q[::-1], h)           # Σ q_{‑k} h_{i‑k}
            grad = term1 - term2

            # back‑tracking line search
            improved = False
            for _ in range(5):
                h_new = h - step * grad
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = enforce_sum(h_new, rng)

                O2 = full_corr(h_new, 1.0 - h_new)
                maxO2 = O2.max()
                exp_shift2 = beta * (O2 - maxO2)
                exp_beta2 = np.exp(exp_shift2)
                sum_exp2 = exp_beta2.sum()
                f2 = (maxO2 + np.log(sum_exp2)) / beta

                if f2 < f - 1e-14:
                    h, f = h_new, f2
                    step *= 1.05
                    improved = True
                    break
                step *= 0.5
            if not improved and step < 1e-8:
                break
    return enforce_sum(np.clip(h, 0.0, 1.0), rng)


# ---------- hill‑climb utilities (same as previous) ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst shift (the shift giving maximal C5)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Simple hill‑climb using worst‑gradient directed moves."""
    best_h, best_c = h.copy(), compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- SLSQP (optional final polish) ----------
def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 600) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(beta * corr)
        z = np.exp(beta * corr - m)
        return (m + np.log(z.sum())) / beta
    res = minimize(softmax, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # one final projection onto the sum‑constraint
    return enforce_sum(x, np.random.default_rng())
