def gradient_refine(h, rng, step=0.02, max_iter=50):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            # part for i = 0 .. n-1-k
            g[:n - k] += 1.0 - h[k:]
            # part for i = k .. n-1
            g[k:] -= h[:n - k]
        # step schedule
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        np.clip(h, 0.0, 1.0, out=h)

        diff = target - h.sum()

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.shape[0]
    target = n / 2.0

def gradient_refine(h, rng, step=0.02, max_iter=30):
    n = h.shape[0]
    target = n / 2.0
    for it in range(max_iter):

def softmax_obj(x, beta, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = beta * corr
    m = np.max(bc)
    # avoid overflow
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def sum_eq_constraint(x, target):
    return np.sum(x) - target

def slsqp_optimize(h, beta, rng, max_iter=800):
    n = h.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint, 'args': (target,)}
    res = minimize(softmax_obj, h, args=(beta, target), method='SLSQP',
                   bounds=bounds, constraints=[cons],
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def gradient_refine(
    h: np.ndarray,
    rng: np.random.Generator,
    step: float = 0.02,
    max_iter: int = 30,
) -> np.ndarray:
    n = h.shape[0]
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                if add != 0.0:
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
    return h

def softmax_obj(x: np.ndarray, beta: float, target: float) -> float:
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def sum_eq_constraint(x: np.ndarray, target: float) -> float:
    return np.sum(x) - target

def slsqp_optimize(
    h: np.ndarray,
    beta: float,
    rng: np.random.Generator,
    max_iter: int = 800,
) -> np.ndarray:
    n = h.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint, "args": (target,)}
    res = minimize(
        softmax_obj,
        h,
        args=(beta, target),
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96

    # --- initial construction -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 360
        h = np.zeros(n0)

    # SLSQP cascade with increasing beta

        adjust_sum_exact(h_frac, rng)
