def compute_c5(h):
    n = h.shape[0]
    f = h.astype(np.float64)
    g = 1.0 - f
    m = 1 << ((2*n - 1).bit_length())
    F = np.fft.rfft(f, n=m)
    G = np.fft.rfft(g[::-1], n=m)
    conv = np.fft.irfft(F * G, n=m)[:2*n-1]
    max_corr = conv.max()
    return max_corr * (2.0 / n)

def binary_greedy_improve(h, rng, deadline):
    n = h.size
    while time.time() < deadline:
        corr = compute_corr_fft(h)
        idx_max = int(corr.argmax())
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        # compute neighbor sums for this shift
        # use modular indexing
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        # ones and zeros indices
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        # best i, j
        min_neigh_i = neigh[ones].min()
        max_neigh_j = neigh[zeros].max()
        # find all i with min, j with max. Choose random from them.
        cand_i = ones[neigh[ones] == min_neigh_i]
        cand_j = zeros[neigh[zeros] == max_neigh_j]
        i = rng.choice(cand_i)
        j = rng.choice(cand_j)
        delta = min_neigh_i - max_neigh_j
        if delta < 0:
            h[i] = 0
            h[j] = 1
            continue
        # try random swaps up to few attempts for possible improvement
        improved = False
        for _ in range(5):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            delta2 = (neigh[i2] - neigh[j2])
            if delta2 < 0:
                h[i2] = 0
                h[j2] = 1
                improved = True
                break
        if not improved:
            break
    return h

def hill_climb_fractional(h, rng, deadline):
    n = h.size
    best = h.copy()
    best_c = compute_c5(h)
    while time.time() < deadline:
        # propose change by swapping a fraction of two entries
        i, j = rng.choice(n, size=2, replace=False)
        if i == j: continue
        hi, hj = h[i], h[j]
        # propose new values by moving epsilon from i to j
        eps = (rng.random() - 0.5) * 0.2  # range -0.1 to 0.1
        # adjust eps to keep within [0,1]
        eps = max(-hi, min(eps, 1 - hi))  # ensure hi+eps in [0,1]
        eps = max(-(1 - hj), min(eps, hj))  # ensure hj-eps in [0,1]
        if abs(eps) < 1e-8:
            continue
        h_new = h.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best = h_new
            h = h_new
        # else maybe accept with some probability? but we can skip.
    return best

def compute_corr_full(h):
    n = h.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    f = h.astype(np.float64)
    g = (1.0 - f)[::-1]
    F = np.fft.rfft(f, m)
    G = np.fft.rfft(g, m)
    conv = np.fft.irfft(F * G, m)[:2*n-1]
    return conv

def compute_c5(h):
    n = h.shape[0]
    conv = compute_corr_full(h)
    return float(conv.max() * (2.0 / n))

import time
import numpy as np
from scipy.optimize import minimize

# ----- globals for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """smooth max of overlaps via log-sum-exp (β controlled globally)."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def adjust_sum_exact(h, rng):
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_corr_full(h):
    """Full linear correlation of h with (1‑h) via FFT."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    f = h.astype(np.float64)
    g = (1.0 - f)[::-1]
    F = np.fft.rfft(f, m)
    G = np.fft.rfft(g, m)
    conv = np.fft.irfft(F * G, m)[: 2 * n - 1]
    return conv.real

def compute_c5(h):
    """C5 = (2/n)·max_{k} correlation."""
    n = h.shape[0]
    conv = compute_corr_full(h)
    return float(conv.max() * (2.0 / n))

def gradient_refine(h, rng, step=0.02, max_iter=20):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            up = n - k
            g = np.zeros_like(h)
            g[:up] += 1.0 - h[k:]   # ∂/∂h[i] of h[i](1‑h[i+k])
            g[k:] -= h[:up]        # ∂/∂h[i+k] part
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def fractional_to_binary(h, rng):
    """Keep exactly n/2 ones – take the largest entries."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h

def binary_greedy_improve(h, rng, deadline):
    """Greedy swaps that lower the current worst overlap."""
    n = h.size
    while time.time() < deadline:
        conv = compute_corr_full(h)
        idx_max = int(conv.argmax())
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        min_nei = neigh[ones].min()
        max_nei = neigh[zeros].max()
        cand_i = ones[neigh[ones] == min_nei]
        cand_j = zeros[neigh[zeros] == max_nei]
        i = rng.choice(cand_i)
        j = rng.choice(cand_j)
        if min_nei - max_nei < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue
        improved = False
        for _ in range(5):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

def hill_climb_fractional(h, rng, deadline):
    """Small random moves while keeping the sum exact."""
    n = h.size
    best = h.copy()
    best_c = compute_c5(h)
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h[i], h[j]
        eps = (rng.random() - 0.5) * 0.2  # ≤±0.1
        eps = max(-hi, min(eps, 1.0 - hi))
        eps = max(-(1.0 - hj), min(eps, hj))
        if abs(eps) < 1e-8:
            continue
        h_new = h.copy()
        h_new[i] = hi + eps
        h_new[j] = hj - eps
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best = h_new
            h = h_new
    return best
