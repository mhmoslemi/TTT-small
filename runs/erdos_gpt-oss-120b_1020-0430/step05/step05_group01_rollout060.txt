def compute_O_and_grad(h, beta):
    n = h.size
    # O data
    O_corr = np.correlate(h, 1.0 - h, mode='full')
    O = O_corr[n-1:]  # length n
    max_O = O.max()
    exp_terms = np.exp(beta * (O - max_O))
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp  # shape (n,)
    # gradient initial zeros
    grad = np.zeros_like(h)
    for k in range(n):
        wk = w[k]
        if wk == 0.0:
            continue
        # forward part:
        if k < n:
            # i from 0 to n-1-k
            i_end = n - k
            grad[:i_end] += wk * (1.0 - h[k:])  # (1 - h[i+k])
        # backward:
        if k > 0:
            # i from k to n-1
            grad[k:] += wk * (-h[:n-k])  # -h[i-k]
    return max_O, grad

def compute_O_and_grad(h, beta):
    n = h.size
    # O via correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    O = corr[n-1:]  # shape n, includes k=0,...,n-1
    max_O = O.max()
    # softmax weights
    # m = max_O
    exp_terms = np.exp(beta * (O - max_O))
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp
    # gradient
    grad = np.zeros_like(h)
    # k=0 special
    wk0 = w[0]
    if wk0 != 0.0:
        grad += wk0 * (1.0 - 2.0 * h)
    # other k
    for k in range(1, n):
        wk = w[k]
        if wk == 0.0:
            continue
        # forward
        i_end = n - k
        grad[:i_end] += wk * (1.0 - h[k:])
        # backward
        grad[k:] += wk * (-h[:i_end])
    # loss
    loss = max_O + (np.log(sum_exp) / beta)  # equivalent previously
    return loss, grad

def enforce_sum_fast(h, target):
    # adjust h so sum equals target while staying within [0,1].
    # iterate few times.
    max_iters=10
    for _ in range(max_iters):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1)
        count = mask.sum()
        if count == 0:
            # all at bounds, distribute diff by adding to random ones respecting bounds
            # fallback to random method
            break
        delta = diff / count
        h[mask] += delta
        # clip any that go out of bounds
        np.clip(h, 0.0, 1.0, out=h)

    diff = target - h.sum()
    if abs(diff) > 1e-12:

        while abs(diff) > 1e-12:
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

import time
import numpy as np

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ----------------------------------------------------------------------
def compute_max_O_and_grad(h: np.ndarray, beta: float):
    """Return max O_k and gradient of the soft‑max approximation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    O = corr[n - 1 :]  # O_k for k = 0 … n‑1
    max_O = O.max()
    # soft‑max weights
    exp_terms = np.exp(beta * (O - max_O))
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp

    grad = np.zeros_like(h)
    # k = 0 (special)
    wk0 = w[0]
    if wk0 != 0.0:
        grad += wk0 * (1.0 - 2.0 * h)

    # k > 0
    for k in range(1, n):
        wk = w[k]
        if wk == 0.0:
            continue
        i_end = n - k
        grad[:i_end] += wk * (1.0 - h[k:])
        grad[k:] += wk * (-h[:i_end])
    return max_O, grad


def gradient_descent_step(h, beta, step, target, rng):
    """One projected gradient step with soft‑max smoothing."""
    max_O, grad = compute_max_O_and_grad(h, beta)
    # project gradient to keep the sum invariant
    grad -= grad.mean()
    h_new = h - step * grad
    np.clip(h_new, 0.0, 1.0, out=h_new)
    enforce_sum(h_new, rng)
    return h_new, max_O


# ----------------------------------------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    O = corr[n - 1 :]  # non‑negative shifts
    k = int(np.argmax(O))
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        i_end = n - k
        grad[:i_end] -= h[k:]
        grad[k:] -= h[:i_end]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator, max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        O = corr[n - 1 :]
        k = int(np.argmax(O))
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            i_end = n - k
            g[:i_end] = 1.0 - h[k:]
            g[k:] -= h[:i_end]
        h = h - step * g
        np.clip(h, 0.0, 1.0, out=h)
        enforce_sum(h, rng)
    return h


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target = n / 2.0

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        enforce_sum(h, rng)
    else:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        enforce_sum(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # projected gradient descent with increasing β
    beta_schedule = [5, 20, 100, 500, 2000, 8000, 20000]
    base_step = 0.10
    max_iter_per_beta = 2000

    for beta in beta_schedule:
        step = base_step * (5.0 / beta)  # scale step inversely with β
        if step < 1e-5:
            step = 1e-5
        it = 0
        while it < max_iter_per_beta and time.time() - start < budget_s * 0.85:
            h, max_O = gradient_descent_step(h, beta, step, target, rng)
            c = max_O * (2.0 / n)
            if c < best_c:
                best_c, best_h = c, h.copy()
            it += 1

    # ------------------------------------------------------------------
    # final hill‑climb and polishing if time permits
    now = time.time()
    if now - start < budget_s * 0.95:
        remaining = budget_s - (now - start)
        # allocate up to 30 s for hill‑climb
        hc_iters = int(min(5000, 30 * 5000 / 30 * remaining / 30))
        best_h, best_c = guided_hill_climb(best_h, rng, max_iters=hc_iters, step_scale=0.6)

    if time.time() - start < budget_s * 0.98:
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=8)
        best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # stochastic tweaking until time expires
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_c, best_h = nc, nh.copy()

    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
