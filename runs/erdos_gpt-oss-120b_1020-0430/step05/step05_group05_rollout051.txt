import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def fractional_to_binary(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    # tiny correction in case of rounding issues
    diff = n / 2 - bin_h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - bin_h[i]) if diff > 0 else max(diff, -bin_h[i])
        bin_h[i] += add
        diff = n / 2 - bin_h.sum()
    return bin_h

def greedy_improve(h: np.ndarray, rng: np.random.Generator, max_passes: int = 2000) -> np.ndarray:
    n = h.size
    for _ in range(max_passes):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        shift = abs(shift)
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(neigh[ones])]
        j = zeros[np.argmax(neigh[zeros])]
        if neigh[i] - neigh[j] < 0:
            h[i] = 0.0
            h[j] = 1.0
        else:
            improved = False
            for __ in range(3):
                i2 = rng.choice(ones)
                j2 = rng.choice(zeros)
                if neigh[i2] - neigh[j2] < 0:
                    h[i2] = 0.0
                    h[j2] = 1.0
                    improved = True
                    break
            if not improved:
                break
    return h

def sa_optimize(start_h: np.ndarray, rng: np.random.Generator, deadline: float) -> tuple:
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T0, T_end = 0.01, 1e-6
    start = time.time()
    total = max(deadline - start, 1e-12)
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = T0 * (T_end / T0) ** frac
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = delta <= 0.0 or rng.random() < np.exp(-delta / T)
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        # occasional short greedy pass
        if rng.random() < 0.001:
            cur_h = greedy_improve(cur_h, rng, max_passes=200)
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs) -> tuple:
    start_global = time.time()
    deadline_global = start_global + budget_s * 0.99
    rng = np.random.default_rng(seed)

    # base size (use parent if present)
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
        h0_360 = fractional_to_binary(parent, rng)
    else:
        n0 = 360
        h0_360 = np.zeros(n0, dtype=float)
        h0_360[rng.choice(n0, n0 // 2, replace=False)] = 1.0

    # up‑sample to gain extra resolution
    n = n0 * 2
    h0 = np.repeat(h0_360, 2)                # binary, sum = n/2

    best_h_overall = None
    best_c_overall = np.inf
    attempts = 2  # parent‑derived + one random start

    for attempt in range(attempts):
        if time.time() >= deadline_global:
            break
        # use parent‑derived on first attempt, random on second
        if attempt == 0:
            h_start = h0.copy()
        else:
            # fresh random binary of size n
            h_start = np.zeros(n, dtype=float)
            h_start[rng.choice(n, n // 2, replace=False)] = 1.0

        # a quick greedy polishing
        h_start = greedy_improve(h_start, rng, max_passes=2000)
        c_start = compute_c5(h_start)
        if c_start < best_c_overall:
            best_c_overall = c_start
            best_h_overall = h_start.copy()

        # allocate a slice of remaining time to SA
        remaining = deadline_global - time.time()
        if remaining < 1.0:
            continue
        slice_deadline = time.time() + min(remaining * 0.6, 60.0)

        h_sa, c_sa = sa_optimize(h_start, rng, slice_deadline)
        if c_sa < best_c_overall:
            best_c_overall = c_sa
            best_h_overall = h_sa.copy()

        # final short greedy refinement
        h_final = greedy_improve(best_h_overall, rng, max_passes=2000)
        c_final = compute_c5(h_final)
        if c_final < best_c_overall:
            best_c_overall = c_final
            best_h_overall = h_final.copy()

    # safety clip and exact‑sum fix
    best_h_overall = np.clip(best_h_overall, 0.0, 1.0)
    # exact sum (should already be exact)
    diff = best_h_overall.size / 2 - best_h_overall.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, best_h_overall.size)
        add = min(diff, 1.0 - best_h_overall[i]) if diff > 0 else max(diff, -best_h_overall[i])
        best_h_overall[i] += add
        diff = best_h_overall.size / 2 - best_h_overall.sum()

    final_c = compute_c5(best_h_overall)
    return best_h_overall.tolist(), float(final_c), int(best_h_overall.size)
