def simulated_annealing_binary(h_start, deadline, rng):
    h = h_start.copy()
    n = h.size
    cur_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c
    # prepare ones, zeros lists/dicts
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    pos_one = {idx: i for i, idx in enumerate(ones)}
    pos_zero = {idx: i for i, idx in enumerate(zeros)}
    # temperature schedule
    T0 = 0.1
    iter = 0
    while time.time() < deadline:
        iter += 1
        # exponential cooling
        T = T0 * (0.99995 ** iter)
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # apply swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new cost
        new_c = compute_c5_fft(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            # acceptance probability
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            cur_c = new_c
            # update ones/zeros lists/dicts
            pos_i = pos_one[i]
            pos_j = pos_zero[j]
            ones[pos_i] = j
            zeros[pos_j] = i
            pos_one[j] = pos_i
            pos_zero[i] = pos_j
            del pos_one[i]
            del pos_zero[j]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # optional early break if best_c already < target threshold 0.3808
        if best_c < 0.3808:
            # we can break early
            break
    return best_h, best_c

def upsample_and_optimize(h, n_up, deadline, rng):
    n = h.size
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
    h_up = np.interp(xt, xs, h)
    # ensure range
    h_up = np.clip(h_up, 0.0, 1.0)
    # adjust sum (target = n_up/2)
    # get current sum
    target = n_up / 2.0
    diff = target - h_up.sum()
    # while diff !=0, adjust random entries within bounds
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n_up)
        if diff > 0:
            add = min(diff, 1.0 - h_up[i])
        else:
            add = max(diff, -h_up[i])
        h_up[i] += add
        diff = target - h_up.sum()
    # Convert to binary (top half)
    h_bin = to_binary(h_up, rng)
    # Now run SA
    return simulated_annealing_binary(h_bin, deadline, rng)

import time
import numpy as np

def compute_c5_fft(h: np.ndarray) -> float:
    """Maximum overlap ∫ h(x)(1‑h(x+k))dx using FFT (full correlation)."""
    n = h.size
    m = 2 * n
    H = np.fft.fft(h, m)
    G = np.fft.fft(1.0 - h, m)
    conv = np.fft.ifft(H * G).real[: 2 * n - 1]
    return float(conv.max() * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> None:
    """Modify h in‑place so that Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()


def to_binary(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a binary vector with exactly n/2 ones (top half entries of h)."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros(n, dtype=float)
    bin_h[idx] = 1.0
    # binary already has correct sum, but keep guard
    adjust_sum_exact(bin_h, rng)
    return bin_h


def simulated_annealing_binary(h_start: np.ndarray, deadline: float,
                               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """SA on binary vectors (swap 1↔0) to minimise max overlap."""
    h = h_start.copy()
    n = h.size
    cur_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = cur_c

    # maintain mutable lists and position maps for O(1) swaps
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    pos_one = {v: i for i, v in enumerate(ones)}
    pos_zero = {v: i for i, v in enumerate(zeros)}

    T0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        T = T0 * (0.9999 ** it)

        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5_fft(h)
        delta = new_c - cur_c

        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_c = new_c
            # update bookkeeping
            pi = pos_one[i]
            pj = pos_zero[j]
            ones[pi] = j
            zeros[pj] = i
            pos_one[j] = pi
            pos_zero[i] = pj
            del pos_one[i]
            del pos_zero[j]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

        # early exit if already excellent
        if best_c < 0.3808:
            break

    return best_h, best_c


def binary_targeted_improve(h: np.ndarray, deadline: float,
                            rng: np.random.Generator) -> np.ndarray:
    """Greedy swaps that lower the worst‑shift overlap (non‑SA)."""
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break

        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        neigh_one = neigh[ones]
        i_cand = ones[neigh_one == neigh_one.min()]
        i = rng.choice(i_cand)

        neigh_zero = neigh[zeros]
        j_cand = zeros[neigh_zero == neigh_zero.max()]
        j = rng.choice(j_cand)

        delta = neigh[i] - neigh[j]          # negative → improvement
        if delta < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue

        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h


def upsample_and_optimize(h: np.ndarray, n_up: int, deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Linear up‑sampling to n_up, binarise, then SA (plus greedy polish)."""
    n = h.size
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
    h_up = np.interp(xt, xs, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    adjust_sum_exact(h_up, rng)
    h_bin = to_binary(h_up, rng)

    h_opt, c_opt = simulated_annealing_binary(h_bin, deadline, rng)
    # final greedy polishing if time permits
    if time.time() < deadline:
        h_opt = binary_targeted_improve(h_opt, deadline, rng)
        c_opt = compute_c5_fft(h_opt)
    return h_opt, c_opt


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ----- base vector -----
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
        n = base_h.size
    else:
        n = 360
        base_h = rng.random(n)
        base_h *= (n / 2.0) / base_h.sum()
        base_h = np.clip(base_h, 0.0, 1.0)
        adjust_sum_exact(base_h, rng)

    # ----- starting seeds -----
    starts = [to_binary(base_h, rng)]
    for _ in range(3):
        hb = np.zeros(n)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(hb)

    best_h = None
    best_c = np.inf
    best_n = n

    # ----- SA on each start (time‑split) -----
    total_starts = len(starts)
    for idx, h0 in enumerate(starts):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        alloc = remaining / (total_starts - idx)
        sub_deadline = time.time() + alloc
        h_opt, c_opt = simulated_annealing_binary(h0, sub_deadline, rng)
        if c_opt < best_c:
            best_c, best_h, best_n = c_opt, h_opt.copy(), n

    # ----- greedy polish on the best binary vector -----
    if best_h is not None and time.time() < deadline:
        best_h = binary_targeted_improve(best_h, deadline, rng)
        best_c = compute_c5_fft(best_h)

    # ----- optional up‑sampling refinement -----
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 720
    ):
        h_up, c_up = upsample_and_optimize(best_h, best_n * 2, deadline, rng)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, best_n * 2

    # ----- final sanity checks -----
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - best_n / 2) > 1e-12:
        adjust_sum_exact(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
