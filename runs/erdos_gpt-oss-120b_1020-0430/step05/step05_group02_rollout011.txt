def binary_sa(h_bin, rng, deadline, max_swaps=500000):
    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode='full')
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = np.flatnonzero(h_bin == 1).tolist()
    zeros = np.flatnonzero(h_bin == 0).tolist()

    T0 = max(cur_worst * 0.05, 1e-5)
    decay = 0.99996
    T = T0

    it = 0
    while time.time() < deadline and it < max_swaps:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # flip i from 1 to 0
        diff_i = -1.0
        left_len = i + 1
        start = n - 1 - i
        c[start : start + left_len] += diff_i * h_bin[i::-1]
        c[start + left_len : start + n] += diff_i * h_bin[i+1:]

        h_bin[i] = 0.0

        # flip j from 0 to 1 (after i change, h_bin updated)
        diff_j = 1.0
        left_len_j = j + 1
        start_j = n - 1 - j
        c[start_j : start_j + left_len_j] += diff_j * h_bin[j::-1]
        c[start_j + left_len_j : start_j + n] += diff_j * h_bin[j+1:]

        h_bin[j] = 1.0

        # compute new worst
        new_min = c.min()
        new_worst = target - new_min

        # accept/reject: if improvement accept; else accept with probability
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12))

        if accept:
            cur_worst = new_worst

def binary_sa(h_bin, rng, deadline, max_swaps=500000):
    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    T = max(cur_worst * 0.2, 1e-4)
    decay = 0.99996
    it = 0

    while time.time() < deadline and it < max_swaps:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # flip i (1->0)
        diff_i = -1.0
        left_len_i = i + 1
        start_i = n - 1 - i
        c[start_i : start_i + left_len_i] += diff_i * h_bin[i::-1]
        c[start_i + left_len_i : start_i + n] += diff_i * h_bin[i + 1 :]
        h_bin[i] = 0.0

        # flip j (0->1) using updated h_bin
        diff_j = 1.0
        left_len_j = j + 1
        start_j = n - 1 - j
        c[start_j : start_j + left_len_j] += diff_j * h_bin[j::-1]
        c[start_j + left_len_j : start_j + n] += diff_j * h_bin[j + 1 :]
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12))

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
        else:
            # revert j (1->0)
            diff_j_rev = -1.0
            c[start_j : start_j + left_len_j] += diff_j_rev * h_bin[j::-1]
            c[start_j + left_len_j : start_j + n] += diff_j_rev * h_bin[j + 1 :]
            h_bin[j] = 0.0
            # revert i (0->1)

            c[start_i : start_i + left_len_i] += diff_i_rev * h_bin[i::-1]
            c[start_i + left_len_i : start_i + n] += diff_i_rev * h_bin[i + 1 :]
            h_bin[i] = 1.0
        T *= decay
        it += 1

    return best_h, best_worst

        adjust_sum_exact(h_up, rng)

        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # fallback (should never happen)
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
