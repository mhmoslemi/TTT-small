def aggregated_gradient(h, m=3):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # exclude zero shift? Possibly include.
    idx = np.argpartition(-corr, m-1)[:m]  # top m indices

    shifts = idx - (n-1)
    grad = np.zeros_like(h)
    for k in np.abs(shifts):
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
    return grad

def compute_c5_fft(h):
    n = h.shape[0]
    f = h.astype(np.float64)
    g = 1 - h
    # reverse g to get correlation
    g_rev = g[::-1]
    nfft = 1 << (2 * n - 1).bit_length()  # next power of 2
    F = np.fft.rfft(f, nfft)
    G = np.fft.rfft(g_rev, nfft)

def guided_perturb(h, grad, rng, scale):
    pos = np.argmax(grad)
    neg = np.argmin(grad)
    if pos == neg:
        return h.copy()
    max_eps = min(h[pos], 1.0 - h[neg])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[pos] -= eps
    nh[neg] += eps
    return nh

def aggregated_gradient(h, m=3):
    n = h.shape[0]
    corr = correlation_fft(h)
    # get top m indices (largest corr). If fewer than m, just all.
    m = min(m, corr.size)
    idx = np.argpartition(-corr, m-1)[:m]
    # shift = idx - (n-1)
    shifts = np.abs(idx - (n - 1)).astype(int)
    grad = np.zeros_like(h)
    for k in np.unique(shifts):
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
    return grad
```

import time
import numpy as np
import scipy.optimize as opt

# ---------- basic helpers ----------
def correlation_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    f = h.astype(np.float64)
    g = 1.0 - h
    nfft = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n-1
    F = np.fft.rfft(f, nfft)
    G = np.fft.rfft(g, nfft)
    corr = np.fft.irfft(F * np.conj(G), nfft)
    return corr[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.shape[0]
    return correlation_fft(h).max() * (2.0 / n)


def enforce_sum(h: np.ndarray) -> np.ndarray:
    """Project onto {0<=h<=1, sum(h)=len(h)/2} via bisection."""
    n = h.shape[0]
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with exact sum = n/2."""
    h = rng.random(n)
    return enforce_sum(h)

def aggregated_gradient(h: np.ndarray, m: int = 3) -> np.ndarray:
    """
    Gradient of the max overlap approximation.
    Sum of gradients for the m largest correlation shifts.

    n = h.shape[0]
    corr = correlation_fft(h)
    m = min(m, corr.size)
    idx = np.argpartition(-corr, m - 1)[:m]
    shifts = np.abs(idx - (n - 1)).astype(int)
    grad = np.zeros_like(h)
    for k in np.unique(shifts):
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient index."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def random_perturb(h: np.ndarray, rng: np.random.Generator,
                   scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a random index to another random index."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ---------- soft‑max + SLSQP ----------
def softmax_c5(h: np.ndarray, beta: float) -> float:
    """Smooth approximation of max correlation."""
    corr = correlation_fft(h)
    bc = beta * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 200) -> np.ndarray:
    """SLSQP refinement using the soft‑max objective."""
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = opt.minimize(
        lambda x: softmax_c5(x, beta),
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-10, "disp": False},
    )
    return enforce_sum(np.clip(res.x, 0.0, 1.0))


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    budget = budget_s

    # ---- initial solution (use parent if present) ----
    n = 512
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = enforce_sum(h0)
        best_h = h0.copy()
        best_c = compute_c5(best_h)
    else:
        best_h = random_feasible_vector(n, rng)
        best_c = compute_c5(best_h)

    # ---- phase 1: many random restarts with short hill‑climbs ----
    phase1_end = start + budget * 0.35
    while time.time() < phase1_end:
        h = random_feasible_vector(n, rng)
        c_h = compute_c5(h)
        scale = 0.5
        decay = 0.995
        for _ in range(180):
            grad = aggregated_gradient(h, m=3)
            nh = guided_perturb(h, grad, rng, scale)
            c_n = compute_c5(nh)
            if c_n < c_h:
                h, c_h = nh, c_n
                if c_n < best_c:
                    best_h, best_c = nh.copy(), c_n
            scale *= decay
        # occasional soft‑max refinement
        if time.time() < phase1_end - 2.0:
            for beta in (120.0, 600.0, 2500.0):
                h2 = slsqp_optimize(h, beta, max_iter=150)
                c2 = compute_c5(h2)
                if c2 < best_c:
                    best_h, best_c = h2.copy(), c2
                if c2 < c_h:
                    h, c_h = h2, c2

    # ---- phase 2: refined hill‑climb on the current best ----
    phase2_end = start + budget * 0.70
    h = best_h.copy()
    c_h = best_c
    scale = 0.3
    decay = 0.998
    while time.time() < phase2_end:
        if rng.random() < 0.12:
            nh = random_perturb(h, rng, scale)
        else:
            grad = aggregated_gradient(h, m=4)
            nh = guided_perturb(h, grad, rng, scale)
        c_n = compute_c5(nh)
        if c_n < c_h:
            h, c_h = nh, c_n
            if c_n < best_c:
                best_h, best_c = nh.copy(), c_n
        scale *= decay
        # sporadic SLSQP steps
        if rng.random() < 0.07:
            for beta in (800.0, 3000.0, 10000.0):
                h2 = slsqp_optimize(h, beta, max_iter=250)
                c2 = compute_c5(h2)
                if c2 < best_c:
                    best_h, best_c = h2.copy(), c2
                if c2 < c_h:
                    h, c_h = h2, c2

    # ---- optional up‑sampling to 1024 if time permits ----
    if time.time() < start + budget * 0.90:
        n2 = 1024
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = enforce_sum(h_up)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up.copy(), c_up
        # brief refinement at higher resolution
        scale2 = 0.25
        while time.time() < start + budget * 0.98:
            grad = aggregated_gradient(best_h, m=5)
            nh = guided_perturb(best_h, grad, rng, scale2)
            c_n = compute_c5(nh)
            if c_n < best_c:
                best_h, best_c = nh.copy(), c_n
            scale2 *= 0.9995
            # occasional soft‑max
            if rng.random() < 0.04:
                for beta in (2000.0, 8000.0, 24000.0):
                    h2 = slsqp_optimize(best_h, beta, max_iter=200)
                    c2 = compute_c5(h2)
                    if c2 < best_c:
                        best_h, best_c = h2.copy(), c2

    # ---- final stochastic tweaks until the deadline ----
    while time.time() - start < budget * 0.999:
        if rng.random() < 0.85:
            grad = aggregated_gradient(best_h, m=5)
            nh = guided_perturb(best_h, grad, rng, scale=0.02)
        else:
            nh = random_perturb(best_h, rng, scale=0.02)
        c_n = compute_c5(nh)
        if c_n < best_c:
            best_h, best_c = nh.copy(), c_n

    # final sanity check
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0))
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
