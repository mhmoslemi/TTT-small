def project_to_box_and_sum(v, target):
    # v is float array.
    # Find lambda such that sum_i clip(v_i - lambda, 0, 1) = target.
    # Use binary search on lambda in range [-1, 1] maybe.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def gradient_refine(h, rng, step=0.005, max_iter=200):
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (h.size - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: h.size - k] += 1.0 - h[k:]
            g[k:] -= h[: h.size - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = h.size / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, h.size)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

import time
import numpy as np

# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (maximal) overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in [0,1] so that sum(h)=target (random redistribution)."""
    diff = target - h.sum()
    # quick random redistribution, limited to a few passes
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add > 0:
                h[i] += add
                diff -= add
        else:
            sub = max(diff, -h[i])
            if sub < 0:
                h[i] += sub
                diff -= sub
    # final tiny correction if needed
    if abs(diff) > 1e-12:
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator,
                   scale: float = 0.5) -> np.ndarray:
    """Transfer epsilon from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def round_to_binary(h: np.ndarray) -> np.ndarray:
    """Binary vector with exactly half the entries equal to 1."""
    n = h.size
    half = n // 2
    idx = np.argpartition(h, -half)[-half:]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h


# --------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ---- problem size -------------------------------------------------
    n = 512
    target_sum = n / 2.0

    # ---- initial solution ---------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_sum, rng)
    else:
        h0 = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h0[ones] = 1.0

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- subgradient descent (majority of time) -----------------------
    sub_time = total_budget * 0.70
    a0 = 0.6
    t = 1
    eval_interval = 5000
    h = best_h.copy()
    while time.time() - start < sub_time:
        grad = worst_gradient(h)
        step = a0 / np.sqrt(t)
        h -= step * grad
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum(h, target_sum, rng)

        if t % eval_interval == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
        t += 1

    # ---- rounding to binary (quick check) ----------------------------
    bin_h = round_to_binary(best_h)
    bin_h = enforce_sum(bin_h, target_sum, rng)  # should already hold
    c_bin = compute_c5(bin_h)
    if c_bin < best_c:
        best_c, best_h = c_bin, bin_h.copy()

    # ---- final local refinement ---------------------------------------
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        # a quick guided hill‑climb
        max_iters = int(min(3000, remaining * 300))
        best_h, best_c = guided_hill_climb(best_h, rng,
                                           max_iters=max_iters, step_scale=0.7)

    # ---- final sanitisation -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
