def softmax_obj_and_grad(h, beta):
    n = h.size
    S = h.sum()  # should be target
    # compute autocorrelation via FFT
    L = 1 << ((2*n-1 + n -1 -1).bit_length())
    H = np.fft.rfft(h, n=L)  # rfft length L//2+1
    # compute corr_h linear correlation: using zero pad: correlation = ifft( H * conj(H) )
    corr = np.fft.irfft(H * np.conj(H), n=L)[:2*n-1]  # length 2n-1
    O = S - corr  # O_k for k from -(n-1) to n-1; we ignore mapping, as O length 2n-1 correlates with shift indices
    # compute softmax
    m = np.max(O)
    exp_beta = np.exp(beta * (O - m))
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp  # length 2n-1
    # compute gradient via convolutions
    # w_rev
    w_rev = w[::-1]
    # convolution lengths: w length = 2n-1, h length n, L2 = next power of two >= (2n-1)+n-1 = 3n-2
    L2 = 1 << ((3*n - 2 - 1).bit_length())
    # fft w and w_rev
    W = np.fft.rfft(w, n=L2)
    W_rev = np.fft.rfft(w_rev, n=L2)
    H2 = np.fft.rfft(h, n=L2)
    conv1 = np.fft.irfft(W * H2, n=L2)  # length L2
    conv2 = np.fft.irfft(W_rev * H2, n=L2)
    # take slices
    start = n - 1
    conv1_slice = conv1[start:start + n]
    conv2_slice = conv2[start:start + n]
    grad = 1.0 - conv1_slice - conv2_slice
    f = (m + np.log(sum_exp)) / beta
    return f, grad

def project_onto_box_sum(v, target):
    # v is numpy array
    # solve x = clamp(v - lam, 0, 1) with sum = target.
    # Use bisection for lam.
    lo = v.min() - 1.0
    hi = v.max()
    # maintain tolerance
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust for any small error: compute diff = target - x.sum()
    # if diff > 1e-12, distribute among non-bound elements proportionally, but diff should be minimal.
    # We'll just adjust the first free element.
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # try to add small diff to one element that is not at bound
        # find indices where 0 < x[i] < 1
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size > 0:
            i = free[0]
            x[i] += diff
            x[i] = np.clip(x[i], 0.0, 1.0)
        else:
            # all at bounds, adjust by distributing proportionally
            x += diff / v.size
            x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0<=x<=1, sum(x)=target}.
    Uses bisection on the Lagrange multiplier.
    """
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    # tiny adjustment for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            i = free[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            x += diff / v.size
            x = np.clip(x, 0.0, 1.0)
    return x


def softmax_obj_and_grad(h: np.ndarray, beta: float):
    """
    Soft‑max approximation of max_k O_k and its gradient.
    O_k = Σ_i h_i (1−h_{i+k}) = S − Σ_i h_i h_{i+k}
    """
    n = h.size
    # linear autocorrelation h * h
    L1 = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L1)
    corr = np.fft.irfft(H * np.conj(H), n=L1)[: 2 * n - 1]
    S = h.sum()
    O = S - corr
    m = np.max(O)
    exp_beta = np.exp(beta * (O - m))
    w = exp_beta / exp_beta.sum()
    w_rev = w[::-1]

    # convolution w * h and w_rev * h
    L2 = 1 << ((3 * n - 2).bit_length())
    W = np.fft.rfft(w, n=L2)
    W_rev = np.fft.rfft(w_rev, n=L2)
    H2 = np.fft.rfft(h, n=L2)
    conv1 = np.fft.irfft(W * H2, n=L2)[: 3 * n - 2]
    conv2 = np.fft.irfft(W_rev * H2, n=L2)[: 3 * n - 2]
    start = n - 1
    conv1_slice = conv1[start : start + n]
    conv2_slice = conv2[start : start + n]

    grad = 1.0 - conv1_slice - conv2_slice
    f = (m + np.log(exp_beta.sum())) / beta
    return f, grad

    steps_per_beta: int,
    target: float,
):

                g[: n - kk] -= h[kk:]
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        if g[i] <= 0 or g[j] >= 0:
            break
        eps = min(h[i], 1.0 - h[j])
        if eps <= 0:
            break
        h[i] -= eps
        h[j] += eps
        # sum unchanged, values stay in [0,1]
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = project_onto_box_sum(np.clip(h0, 0.0, 1.0), target)
        best_h = h0.copy()
        best_c = compute_c5(best_h)
    else:
        n = 256
        target = n / 2.0
        h0 = np.zeros(n)
        idx = rng.choice(n, int(target), replace=False)
        h0[idx] = 1.0
        best_h = h0.copy()
        best_c = compute_c5(best_h)

    # ---- upscale to 512 if necessary ------------------------------------
    if n < 512:
        n_target = 512
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_onto_box_sum(np.clip(h_up, 0.0, 1.0), n_target / 2.0)
        best_h = h_up
        n = n_target
        target = n / 2.0
        best_c = compute_c5(best_h)

    # ---- optimisation parameters -----------------------------------------
    betas = [10.0, 50.0, 200.0, 800.0, 3200.0, 12800.0]
    steps_per_beta = 400          # total ~2400 gradient steps
    init_step = 0.5
    step_decay = 0.99

    # ---- main improvement loop -------------------------------------------
    while time.time() - start < budget_s * 0.94:
        # random perturbation of the current best
        if rng.random() < 0.2:
            h_start = np.zeros(n)
            idx = rng.choice(n, int(target), replace=False)
            h_start[idx] = 1.0
        else:
            h_start = best_h.copy()
            noise = rng.normal(scale=0.02, size=n)
            h_start = project_onto_box_sum(h_start + noise, target)

        # soft‑max gradient descent
        h_opt = projected_gradient_descent(
            h_start,
            betas,
            init_step,
            step_decay,
            steps_per_beta,
            target,
        )

        # exact max‑overlap polishing
        h_opt = exact_max_hill_climb(h_opt, target, max_iters=200)

        c = compute_c5(h_opt)
        if c < best_c:
            best_h, best_c = h_opt.copy(), c

    # ---- final fine‑tuning ------------------------------------------------
    while time.time() - start < budget_s * 0.999:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = min(best_h[i], 1.0 - best_h[j])
        if eps <= 1e-12:
            continue
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = project_onto_box_sum(cand, target)
        c = compute_c5(cand)
        if c < best_c:
            best_h, best_c = cand, c

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
