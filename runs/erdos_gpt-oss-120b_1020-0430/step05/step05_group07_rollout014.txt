import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility ----------

def next_pow2(x):
    return 1 << (x - 1).bit_length()

def adjust_sum_exact(h, rng, max_iter=5000):
    """Redistribute to achieve exact sum target while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- C5 evaluation using FFT (linear convolution) ----------

class C5Computer:
    def __init__(self, n):
        self.n = n
        self.L = next_pow2(2 * n - 1)

    def corr(self, h):
        """Linear convolution of h and (1-h). Returns length 2n-1 array."""
        fh = np.fft.rfft(h, n=self.L)
        f1mh = np.fft.rfft(1.0 - h, n=self.L)
        conv = np.fft.irfft(fh * f1mh, n=self.L)
        return conv[: 2 * self.n - 1]

    def max_overlap(self, h):
        """Maximum overlap C5 value."""
        return float(self.corr(h).max() * (2.0 / self.n))

    def worst_shift(self, h):
        """Index of shift achieving max overlap (relative to -n+1..n-1)."""
        corr = self.corr(h)
        idx = corr.argmax() - (self.n - 1)
        return idx

# ---------- Gradient refinement ----------

def gradient_refine(h, comp, rng, step=0.04, max_iter=30):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        idx = comp.worst_shift(h)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        # simple backtracking line search
        lam = step * (0.8 ** it)
        h_new = h - lam * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = adjust_sum_exact(h_new, rng)
        if comp.max_overlap(h_new) < comp.max_overlap(h):
            h = h_new
    return h

# ---------- Simulated annealing (binary) ----------

def binary_sa(h_bin, comp, rng, deadline):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = comp.max_overlap(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    init_T = max(0.01 * cur_c, 1e-5)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        c = comp.max_overlap(cur)
        T = init_T * (0.9994 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it - last_improve > 12000:
            break
    return best, best_c

# ---------- Simulated annealing (fractional) ----------

def fractional_sa(h, comp, rng, deadline):
    n = h.size
    cur = h.copy()
    cur_c = comp.max_overlap(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-5)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = comp.max_overlap(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 25000:
            break
    return best, best_c

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        smooth_max_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    x = adjust_sum_exact(x, rng)
    return x

# ---------- Main driver ----------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # Helper to generate a fresh C5Computer after size change
    def get_comp(n):
        return C5Computer(n)

    # ---------- Initial size loops ----------
    for n in (360, 720):
        if time.time() >= deadline:
            break
        comp = get_comp(n)

        # Prepare initial candidates
        candidates = []

        # 1. Interpolated parent if present
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            candidates.append(interp)

            # binary version from interpolation (top n/2 values)
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            candidates.append(bin_vec)

        # 2. Random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        candidates.append(bin_rand)

        # 3. Random fractional (uniform)
        rand_frac = rng.random(n)
        rand_frac *= (n / 2.0) / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        candidates.append(rand_frac)

        # 4. Random fractional (biased square)
        rand_frac2 = rng.random(n) ** 2
        rand_frac2 *= (n / 2.0) / rand_frac2.sum()
        rand_frac2 = np.clip(rand_frac2, 0.0, 1.0)
        candidates.append(rand_frac2)

        # 5. Small perturbation of best so far (if any)
        if best_h is not None and best_n == n:
            pert = best_h + 0.1 * (rng.random(n) - 0.5)
            pert = np.clip(pert, 0.0, 1.0)
            candidates.append(pert)

        for h0 in candidates:
            if time.time() >= deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            h = adjust_sum_exact(h, rng)

            # Gradient pre‑refine
            h = gradient_refine(h, comp, rng, step=0.05, max_iter=20)

            # Binary SA if essentially binary
            if np.all((h == 0) | (h == 1)):
                sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.30)
                h_bin, c_bin = binary_sa(h.astype(int).copy(), comp, rng, sub_dead)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.astype(float).copy(), c_bin, n
                h = h_bin.astype(float)

            # Fractional SA (major search)
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.40)
            h_frac, c_frac = fractional_sa(h.copy(), comp, rng, sub_dead)
            if c_frac < best_c:
                best_h, best_c, best_n = h_frac.copy(), c_frac, n
            h = h_frac

            # Post‑SA gradient
            h = gradient_refine(h, comp, rng, step=0.04, max_iter=30)
            c = comp.max_overlap(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # Smooth‑max cascade (light)
            for beta in (10.0, 100.0, 500.0, 2000.0):
                if time.time() >= deadline:
                    break
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=600)
                except Exception:
                    continue
                c = comp.max_overlap(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # Final gradient
            h = gradient_refine(h, comp, rng, step=0.03, max_iter=20)
            c = comp.max_overlap(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # Fallback (should never trigger)
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = adjust_sum_exact(best_h, rng)
        best_c = get_comp(n).max_overlap(best_h)

    # Final sanity clip
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h, rng)

    final_c = get_comp(best_n).max_overlap(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
