def gradient_refine_extended(h, rng, step=0.03, max_iter=80):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        # compute corr and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # compute gradient g: shape n
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            # contributions from first part
            g[:up] += 1.0 - h[k:]
            # contributions from second part
            g[k:] -= h[:up]
        # step size schedule: perhaps step * (0.6 ** it)
        step_sz = step * (0.7 ** it)
        # update
        h = h - step_sz * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # adjust sum exactly (simple random fix)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

    last_improve = 0

        # Choose transfer direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)

                best = cur.copy()
                last_improve = it

            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 20000:

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> None:
    """Redistribute mass so that sum(h) == n/2 while keeping 0<=h<=1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return
    # simple random walk repair
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def fractional_sa_hill_climb(h: np.ndarray, rng: np.random.Generator,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Random-transfer simulated annealing on continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 20000:
            break
    return best, best_c

def binary_sa_hill_climb(h_bin: np.ndarray, rng: np.random.Generator,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Swap-based simulated annealing on binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 15000:
            break
    return best_h, best_c

def gen_start_vecs(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse set of initial vectors for size n."""
    vecs = []
    # parent interpolation if available
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        vecs.append(interp)
        idx = np.argpartition(interp, -n // 2)[-n // 2:]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        vecs.append(bin_vec)
    # random binary
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    vecs.append(bin_rand)
    # random fractional (uniform)
    rand_frac = rng.random(n)
    rand_frac *= (n / 2.0) / rand_frac.sum()
    adjust_sum_exact(rand_frac, rng)
    vecs.append(rand_frac)
    # random fractional (biased)
    rand_frac2 = rng.random(n) ** 2
    rand_frac2 *= (n / 2.0) / rand_frac2.sum()
    adjust_sum_exact(rand_frac2, rng)
    vecs.append(rand_frac2)
    # sinusoidal template
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    sin_vec = 0.5 + 0.5 * np.cos(np.pi * xs)  # period 2, stays in [0,1]
    sin_vec = np.clip(sin_vec, 0.0, 1.0)
    adjust_sum_exact(sin_vec, rng)
    vecs.append(sin_vec)
    return vecs

def improve_at_n(n: int, rng: np.random.Generator, deadline: float,
                 best_h_ref: list, best_c_ref: list, best_n_ref: list) -> bool:
    """
    Optimize for a given n. Updates best_h_ref[0], best_c_ref[0], best_n_ref[0].
    Returns True if target bound < 0.3808 has been reached.
    """
    start_vecs = gen_start_vecs(n, rng)
    target_reached = False
    for h0 in start_vecs:
        if time.time() >= deadline:
            break
        h = h0.copy()
        adjust_sum_exact(h, rng)
        # binary SA if applicable
        if np.all((h == 0) | (h == 1)):
            sub_dead = min(deadline, time.time() + min(20.0, (deadline - time.time()) * 0.2))
            h, _ = binary_sa_hill_climb(h, rng, sub_dead)
        # fractional SA
        sub_dead = min(deadline, time.time() + min(30.0, (deadline - time.time()) * 0.3))
        h, _ = fractional_sa_hill_climb(h, rng, sub_dead)
        # gradient refinement
        h = gradient_refine_extended(h, rng, step=0.03, max_iter=80)
        c = compute_c5(h)
        if c < best_c_ref[0]:
            best_h_ref[0] = h.copy()
            best_c_ref[0] = c
            best_n_ref[0] = n
        if c < 0.3808:
            target_reached = True
            break
    # extra polishing on the current best of this n
    if not target_reached and best_h_ref[0] is not None and best_h_ref[0].size == n:
        for _ in range(3):
            if time.time() >= deadline:
                break
            sub_dead = min(deadline, time.time() + min(20.0, (deadline - time.time()) * 0.2))
            best_h_ref[0], _ = fractional_sa_hill_climb(best_h_ref[0], rng, sub_dead)
            best_h_ref[0] = gradient_refine_extended(best_h_ref[0], rng, step=0.03, max_iter=80)
            c = compute_c5(best_h_ref[0])
            if c < best_c_ref[0]:
                best_c_ref[0] = c
                best_n_ref[0] = n
            if c < 0.3808:
                target_reached = True
                break
    return target_reached

def run(seed: int = 42, budget_s: int = 1100, **kwargs) -> tuple[list, float, int]:
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99  # safety margin

    # containers for the current best solution
    best_h = [None]      # list used for mutable reference
    best_c = [np.inf]
    best_n = [None]

    # evaluate parent if supplied
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        adjust_sum_exact(h_parent, rng)
        c_parent = compute_c5(h_parent)
        best_h[0] = h_parent.copy()
        best_c[0] = c_parent
        best_n[0] = h_parent.size

    # primary size (use parent size if present, else 360)
    base_n = len(initial_h_values) if 'initial_h_values' in globals() else 360
    if base_n % 2:
        base_n += 1

    # first optimisation round at base_n
    if time.time() < deadline:
        if improve_at_n(base_n, rng, deadline, best_h, best_c, best_n):
            # target already met
            pass

    # second round: upsample if time permits
    if best_h[0] is not None and time.time() < deadline and best_n[0] * 2 <= 720:
        n_up = best_n[0] * 2
        xs = np.linspace(0.0, 2.0, best_n[0], endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h[0])
        adjust_sum_exact(h_up, rng)
        # polish the upsampled function
        while time.time() < deadline:
            remaining = deadline - time.time()
            if remaining < 5.0:
                break
            sub_dead = time.time() + min(30.0, remaining * 0.5)
            h_up, _ = fractional_sa_hill_climb(h_up, rng, sub_dead)
            h_up = gradient_refine_extended(h_up, rng, step=0.03, max_iter=80)
            c_up = compute_c5(h_up)
            if c_up < best_c[0]:
                best_h[0] = h_up.copy()
                best_c[0] = c_up
                best_n[0] = n_up
                if c_up < 0.3808:
                    break

    # fallback if nothing found
    if best_h[0] is None:
        n = base_n
        best_h[0] = np.zeros(n)
        best_h[0][: n // 2] = 1.0
        adjust_sum_exact(best_h[0], rng)
        best_c[0] = compute_c5(best_h[0])
        best_n[0] = n

    # final clean‑up
    adjust_sum_exact(best_h[0], rng)
    final_c = compute_c5(best_h[0])

    return best_h[0].tolist(), float(final_c), int(best_n[0])
