def adjust_sum_exact(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # for diff>0 need to add to some entries <1
    # compute capacity = sum(1 - h)
    if diff > 0:
        capacity = np.sum(1.0 - h)
        if capacity < diff - 1e-9:
            # cannot satisfy due to rounding, just set to target
            diff = capacity
        # distribute proportionally
        # avoid scaling too much for near-1 entries, simple: add diff * (1 - h) / capacity
        add = diff * (1.0 - h) / capacity
        h += add
    else:
        # diff <0 need to subtract from entries >0
        capacity = np.sum(h)
        sub = -diff * h / capacity
        h -= sub

    # It's possible due to rounding not exactly target; could iterate again limited times.
    # For safety, we can call adjust_sum_exact recursively a max of few times.

    return h

import time
import numpy as np

def _adjust_sum_exact(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Redistribute to satisfy Σh = target exactly (within 1e-12)."""
    for _ in range(3):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = np.sum(1.0 - h)
            if cap < 1e-12:
                break
            h += diff * (1.0 - h) / cap
        else:
            cap = np.sum(h)
            if cap < 1e-12:
                break
            h += diff * h / cap
        np.clip(h, 0.0, 1.0, out=h)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        n = h.size
        for i in rng.permutation(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h


def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 for a given discretisation."""
    n = h.size
    target = n / 2.0
    autocorr = np.fft.ifft(np.fft.fft(h) * np.conj(np.fft.fft(h))).real
    max_corr = target - np.min(autocorr)          # max Σ h_i (1-h_{i+k})
    return (2.0 / n) * max_corr                  # the overlap integral


def _gd_anneal(h0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max of the overlap.
    Returns the best (h, C5) seen before the deadline.
    """
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    # best so far
    best_h = h.copy()
    best_c = _c5_from_h(h)

    schedule = [
        (5, 400),
        (20, 300),
        (80, 300),
        (200, 250),
        (500, 250),
        (1000, 200),
        (2000, 150),
        (5000, 120),
        (10000, 100),
    ]

    for beta, max_it in schedule:
        if time.time() > deadline:
            break
        step0 = 0.5 / (beta ** 0.3)          # heuristic base step size
        for it in range(max_it):
            if time.time() > deadline:
                break

            H = np.fft.fft(h)
            autocorr = np.fft.ifft(H * np.conj(H)).real
            corr = target - autocorr          # Σ h_i (1‑h_{i+k})
            c5 = (2.0 / n) * np.max(corr)
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()

            # smooth‑max weights
            bc = beta * corr
            m = np.max(bc)
            exp_bc = np.exp(bc - m)
            w = exp_bc / np.sum(exp_bc)

            # gradient of smooth‑max (w symmetric → -2 conv(w, h))
            grad = -2.0 * np.fft.ifft(np.fft.fft(w) * H).real

            step = step0 / (1.0 + it * 0.01)
            h_new = h - step * grad
            np.clip(h_new, 0.0, 1.0, out=h_new)
            h_new = _adjust_sum_exact(h_new, target, rng)
            h = h_new

    # final evaluation
    final_c = _c5_from_h(h)
    if final_c < best_c:
        best_c = final_c
        best_h = h.copy()
    return best_h, best_c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.99 * budget_s

    # ----- initial discretisation size -----
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
    else:
        parent = None
        n0 = 240

    target0 = n0 / 2.0
    best_h = None
    best_c = np.inf

    # ----- several independent starts at the base resolution -----
    for attempt in range(5):
        if time.time() > deadline:
            break
        if attempt == 0 and parent is not None:
            h0 = parent.copy()
        else:
            h0 = rng.random(n0)
            h0 *= target0 / h0.sum()
            h0 = _adjust_sum_exact(h0, target0, rng)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _adjust_sum_exact(h0, target0, rng)

        h_opt, c_opt = _gd_anneal(h0, deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # ----- progressive up‑sampling (720 → 1440) -----
    n = n0
    h = best_h if best_h is not None else np.full(n0, 0.5)
    while time.time() < deadline and n * 2 <= 1440:
        n2 = n * 2
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(xt, xs, h)
        target2 = n2 / 2.0
        h_up = _adjust_sum_exact(h_up, target2, rng)

        # two tries at the higher resolution
        best_h2 = None
        best_c2 = np.inf
        for sub_attempt in range(2):
            if time.time() > deadline:
                break
            h0 = h_up.copy()
            if sub_attempt == 1:                     # slight random nudging
                h0 += rng.normal(scale=0.01, size=n2)
                np.clip(h0, 0.0, 1.0, out=h0)
                h0 = _adjust_sum_exact(h0, target2, rng)

            h_opt, c_opt = _gd_anneal(h0, deadline, rng)
            if c_opt < best_c2:
                best_c2 = c_opt
                best_h2 = h_opt.copy()

        if best_h2 is not None and best_c2 < best_c:
            best_c = best_c2
            best_h = best_h2.copy()
        # continue up‑sampling from the interpolated (not yet optimised) version
        n = n2
        h = h_up

    # ----- fallback if everything failed -----
    if best_h is None:
        n = n0
        best_h = np.tile([1.0, 0.0], n // 2).astype(float)
        best_h = _adjust_sum_exact(best_h, n / 2.0, rng)
        best_c = _c5_from_h(best_h)

    # final safety pass
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _adjust_sum_exact(best_h, best_h.size / 2.0, rng)
    final_c = _c5_from_h(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
