def gradient_step_optimize(h, rng, max_iters=2000, frac=0.5):
    n = len(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # compute gradient of O_k:
        if k == 0:
            grad = 1 - 2*h   # O_0 = sum_i h_i*(1 - h_i) i.e. 1 - 2h_i? Wait O_0 = sum_i h_i (1 - h_i) = sum_i (h_i - h_i^2). gradient = 1 - 2 h_i. yes.
        else:
            grad = np.ones(n)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        # sum of grad should be zero (numerical errors we can correct)
        # compute max step
        pos = grad > 1e-12
        neg = grad < -1e-12
        # for positive grad, moving -alpha*grad decreases h[i]
        # bound: h_i - alpha*grad_i >= 0 => alpha <= h_i / grad_i
        max_alpha_pos = np.min(h[pos] / grad[pos]) if np.any(pos) else np.inf
        # for negative grad, moving -alpha*grad_i increases h_i
        # bound: h_i - alpha*grad_i <= 1 => alpha <= (1 - h_i)/(-grad_i)
        max_alpha_neg = np.min((1 - h[neg]) / (-grad[neg])) if np.any(neg) else np.inf
        max_alpha = min(max_alpha_pos, max_alpha_neg)
        if max_alpha <= 0:
            break
        alpha = frac * max_alpha
        # do update
        h_new = h - alpha * grad
        # clamp to [0,1] (should be within)
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce sum? Since we used step respecting sum, rounding might cause tiny diff.
        diff = (n/2) - h_new.sum()
        if abs(diff) > 1e-10:
            # adjust a random index
            i = rng.integers(0, n)
            add = min(diff, 1 - h_new[i]) if diff > 0 else max(diff, -h_new[i])
            h_new[i] += add
        # compute new C5

            h = h_new
        else:
            # maybe reduce frac and try again
            frac *= 0.5
            if frac < 1e-4:
                break
    return h

    n = h0.size
    h = h0.copy()
    for it in range(max_iters):

        old_c = np.max(corr) * (2.0 / n)
        # select top shifts
        if top_shifts >= n:
            ks = [abs(i-(n-1)) for i in range(len(corr))]
        else:
            # partial sort
            idxs = np.argpartition(-corr, top_shifts-1)[:top_shifts]
            ks = np.abs(idxs - (n-1))
        # randomly choose one
        k = int(rng.choice(ks))
        # gradient
        if k == 0:
            grad = 1 - 2 * h
        else:

            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]

        pos = grad > 1e-12

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random walk to fix the sum
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the current worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.ones_like(h)
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])

        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

                   rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    """Soft‑max approximation of max_k O_k, solved by SLSQP."""

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    def sum_eq_constraint(x):

    n = h0.size
    h = h0.copy()
    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        old_c = np.max(corr) * (2.0 / n)
