import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):

    target = n / 2.0
    # Clip to [0,1] first (should be already)
    h = np.clip(h, 0.0, 1.0)

    # tolerance
    tol = 1e-12

            add = min(diff, 1.0 - h[i])
        else:

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)  # shift index
    k = idx
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else: # k < 0
        kk = -k
        grad[kk:] -= h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad

Now guided_perturb:

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def binary_swap_hill_climb(bin_h, rng, time_limit, start_time):
    n = bin_h.size
    best_h = bin_h.copy()
    best_c = compute_c5(bin_h)
    # Precompute indices arrays for ones and zeros each iteration (update after swap)
    while time.time() - start_time < time_limit:
        ones = np.flatnonzero(bin_h == 1)
        zeros = np.flatnonzero(bin_h == 0)
        if ones.size == 0 or zeros.size == 0:

        bin_h[j] = 1.0

        else:
            # revert
            bin_h[i] = 1.0
            bin_h[j] = 0.0
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with n points."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project h onto the hyperplane Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    tol = 1e-12
    # simple redistribution loop
    it = 0
    while abs(diff) > tol and it < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    if abs(diff) > tol:
        # final tweak on the first coordinate
        h[0] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # shift index (could be negative)
    k = idx
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:  # k < 0
        kk = -k
        grad[kk:] -= h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def binary_swap_hill_climb(bin_h: np.ndarray, rng: np.random.Generator,
                          deadline: float, start_time: float):
    """Swap a 1‑entry with a 0‑entry, accepting only improving moves."""
    n = bin_h.size
    best_h = bin_h.copy()
    best_c = compute_c5(bin_h)
    while time.time() - start_time < deadline:
        ones = np.flatnonzero(bin_h == 1)
        zeros = np.flatnonzero(bin_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        bin_h[i] = 0.0
        bin_h[j] = 1.0
        c = compute_c5(bin_h)
        if c < best_c:
            best_c, best_h = c, bin_h.copy()
        else:
            bin_h[i] = 1.0
            bin_h[j] = 0.0
    return best_h, best_c


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 600):
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny numerical drift)
    x = enforce_sum(x, np.random.default_rng())
    return x
