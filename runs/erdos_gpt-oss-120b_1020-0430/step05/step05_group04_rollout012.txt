def smooth_max_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * (2.0 / n)
    # shift for conv: w = exp(beta*C) normalized
    w = np.exp(beta * C - np.max(beta * C))  # for stability
    w = w / w.sum()
    # compute term1 via conv with reversed w
    w_rev = w[::-1]
    term1 = np.convolve(w_rev, 1.0 - h, mode='full')
    term1 = term1[n - 1 : n - 1 + n]  # pick indices i + (n-1)
    # compute term2 via conv
    term2 = np.convolve(w, h, mode='full')
    term2 = term2[n - 1 : n - 1 + n]
    grad = (2.0 / n) * (term1 - term2)
    return grad

def binary_sa(h_init, rng, time_deadline, init_T=None):
    n = h_init.size
    h = h_init.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # lists of indices of ones and zeros
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    if init_T is None:
        init_T = max(0.01 * cur_c, 1e-5)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        c = compute_c5(h)
        # temperature schedule: exponential decay
        T = init_T * np.exp(-1e-5 * it)  # adapt slower
        # acceptance
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            # update ones and zeros
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                last_improve = it
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        it += 1
        if it - last_improve > 20000:
            # reheat
            init_T = max(0.01 * cur_c, 1e-5)
            last_improve = it
    return best_h, best_c

def binary_sa(h0, rng, deadline):
    h = h0.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # determine initial temperature
    init_T = max(0.01 * cur_c, 1e-5)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        # pick i,j with opposite values
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        # exponential decay
        T = init_T * np.exp(-1e-5 * it)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
                last_improve = it
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        it += 1
        # reheat if stuck
        if it - last_improve > 20000:
            init_T = max(0.01 * cur_c, 1e-5)
            last_improve = it
    return best_h, best_c

def smooth_max_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * (2.0 / n)
    # numerical stability: subtract max
    max_betaC = np.max(beta * C)
    exp_betaC = np.exp(beta * C - max_betaC)
    w = exp_betaC / exp_betaC.sum()
    # term1 via convolution with reversed w
    w_rev = w[::-1]
    term1 = np.convolve(w_rev, 1.0 - h, mode='full')
    term1 = term1[n - 1 : n - 1 + n]
    term2 = np.convolve(w, h, mode='full')
    term2 = term2[n - 1 : n - 1 + n]
    grad = (2.0 / n) * (term1 - term2)
    return grad

def compute_c5(h):
    """Maximum overlap integral for a step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Force sum(h)=n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # fallback (very unlikely)
    if abs(diff) > 1e-9:
        h[:] = np.clip(h, 0.0, 1.0)
        total = h.sum()
        if total > 0:
            h *= target / total
        else:
            h[: n // 2] = 1.0
    return h

def smooth_max_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")

    max_bc = np.max(beta * C)

        time_per_start = max(0.05, (deadline - time.time()) * 0.5 / (len(starters) * remaining_n))

        for h0 in starters:
            if time.time() > deadline:
                break
            h = h0.copy()
            adjust_sum_exact(h, rng)
            h = np.clip(h, 0.0, 1.0)

            if np.all((h == 0) | (h == 1)):
                # binary SA
                sub_dead = min(deadline, time.time() + time_per_start)
                h_bin, c_bin = binary_sa(h, rng, sub_dead)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                # greedy polish
                h_pol = greedy_improve(h_bin.copy(), rng, max_iter=100, deadline=deadline)
                c_pol = compute_c5(h_pol)
                if c_pol < best_c:
                    best_h, best_c, best_n = h_pol.copy(), c_pol, n
            else:
                # continuous descent + rounding
                sub_dead = min(deadline, time.time() + time_per_start)
                h_cont = smooth_max_descent(h, rng, sub_dead)
                c_cont = compute_c5(h_cont)
                if c_cont < best_c:
                    best_h, best_c, best_n = h_cont.copy(), c_cont, n
                # round to binary and polish
                h_bin = np.rint(h_cont).astype(float)
                adjust_sum_exact(h_bin, rng)
                h_bin = greedy_improve(h_bin, rng, max_iter=100, deadline=deadline)
                c_bin = compute_c5(h_bin)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n

    # ------------------------------------------------------------------
    # optional up‑sampling of the current best
    if (
        best_h is not None
        and time.time() + 30 < deadline
        and best_n * 2 <= 960
    ):
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        # sum already matches target after repeat
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
        h_up_bin, c_up = binary_sa(h_up, rng, sub_dead)
        if c_up < best_c:
            best_h, best_c, best_n = h_up_bin.copy(), c_up, n_up
        h_up_bin = greedy_improve(h_up_bin, rng, max_iter=200, deadline=deadline)
        c_up = compute_c5(h_up_bin)
        if c_up < best_c:
            best_h, best_c, best_n = h_up_bin.copy(), c_up, n_up

    # ------------------------------------------------------------------
    if best_h is None:
        # fallback simple half‑interval
        n = 180
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
