def slsqp_refine(h, beta, max_iter):
    n = h.size
    target = n/2.0
    bounds = [(0.0, 1.0)]*n
    cons = {"type":"eq","fun":lambda x: np.sum(x)-target}
    def objective(x):
        corr = np.correlate(x, 1-x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # compute log-sum-exp
        lse = m + np.log(np.sum(np.exp(bc - m)))
        return lse / beta
    res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol":1e-12, "disp":False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def slsqp_optimize(h0, beta, max_iter=800):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter,
                                              "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()
    # initial vector
    if 'initial_h_values' in globals():

    else:

        h[rng.choice(n, n//2, replace=False)] = 1.0

    # correlation for binary

    k_min = int(np.argmin(corr))
    min_corr = corr[k_min]
    best_c5 = 1.0 - (2.0 / n) * min_corr
    best_h = h.copy()
    # maintain ones/zeros
    ones = [int(i) for i in range(n) if h[i] > 0.5]
    zeros = [int(i) for i in range(n) if h[i] <= 0.5]
    pos_one = {i: idx for idx, i in enumerate(ones)}
    pos_zero = {i: idx for idx, i in enumerate(zeros)}
    idx_arr = np.arange(n)
    no_improve = 0
    target_c5 = 0.3808

    while time.time() - start < budget_s * 0.85:
        if best_c5 < target_c5:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # delta for current worst shift
        d = 0.0
        if i + k_min < n:
            d -= h[i + k_min]
        if j + k_min < n:
            d += h[j + k_min]
        if i - k_min >= 0:
            d -= h[i - k_min]
        if j - k_min >= 0:
            d += h[j - k_min]
        if d > 0.0:

            jp = j + idx_arr
            im = i - idx_arr
            jm = j - idx_arr
            delta_full = np.zeros(n)
            mask = ip < n
            delta_full[mask] -= h[ip[mask]]
            mask = jp < n

            del pos_zero[j]

            best_h = h.copy()
            no_improve = 0

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function on [0,2] with dx = 2/len(h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 800) -> np.ndarray:
    """Smooth‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    target_c5 = 0.3808

    # ---- initial continuous candidate (if any) ----
    if "initial_h_values" in globals():
        h_cont = np.array(initial_h_values, dtype=float)
        h_cont = np.clip(h_cont, 0.0, 1.0)
        h_cont = enforce_sum(h_cont, rng)
    else:
        n = 256
        h_cont = rng.random(n)
        h_cont *= (n / 2.0) / h_cont.sum()
        h_cont = np.clip(h_cont, 0.0, 1.0)
        h_cont = enforce_sum(h_cont, rng)

    n = h_cont.size
    best_h = h_cont.copy()
    best_c5 = compute_c5(best_h)

    # ---- binary version for aggressive hill climbing ----
    order = np.argsort(-h_cont)
    h = np.zeros(n, dtype=float)
    h[order[:n // 2]] = 1.0  # binary start

    corr = np.correlate(h, h, mode="full")[n - 1 :]
    k_min = int(np.argmin(corr))
    min_corr = corr[k_min]
    bin_best_c5 = 1.0 - (2.0 / n) * min_corr
    bin_best_h = h.copy()

    # bookkeeping for swaps
    ones = [int(i) for i in range(n) if h[i] > 0.5]
    zeros = [int(i) for i in range(n) if h[i] <= 0.5]
    pos_one = {i: idx for idx, i in enumerate(ones)}
    pos_zero = {i: idx for idx, i in enumerate(zeros)}
    idx_arr = np.arange(n)
    no_improve = 0

    # ---- greedy binary hill climb (focus on current worst shift) ----
    while time.time() - start < budget_s * 0.85:
        if bin_best_c5 < target_c5:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        # delta for current worst shift (linear, zero‑padded)
        d = 0.0
        if i + k_min < n:
            d -= h[i + k_min]
        if j + k_min < n:
            d += h[j + k_min]
        if i - k_min >= 0:
            d -= h[i - k_min]

            d += h[j - k_min]

            bin_best_h = h.copy()
            no_improve = 0
        else:

            j = zeros[rng.integers(len(zeros))]
            ip = i + idx_arr
            jp = j + idx_arr
            im = i - idx_arr
            jm = j - idx_arr
            delta_full = np.zeros(n)
            mask = ip < n
            delta_full[mask] -= h[ip[mask]]
            mask = jp < n
            delta_full[mask] += h[jp[mask]]
            mask = im >= 0
            delta_full[mask] -= h[im[mask]]
            mask = jm >= 0
            delta_full[mask] += h[jm[mask]]
            corr += delta_full
            h[i] = 0.0
            h[j] = 1.0
            p_i = pos_one[i]
            p_j = pos_zero[j]
            ones[p_i] = j
            pos_one[j] = p_i
            zeros[p_j] = i
            pos_zero[i] = p_j
            del pos_one[i]
            del pos_zero[j]
            k_min = int(np.argmin(corr))
            min_corr = corr[k_min]
            new_c5 = 1.0 - (2.0 / n) * min_corr
            if new_c5 < bin_best_c5:
                bin_best_c5 = new_c5
                bin_best_h = h.copy()
            no_improve = 0

    # ---- choose the better of the continuous and binary candidates ----
    if bin_best_c5 < best_c5:
        best_h = bin_best_h.copy()
        best_c5 = bin_best_c5

    # ---- smooth‑max SLSQP refinement (continuous) ----
    for beta in (20., 100., 400., 1500., 6000., 20000., 80000.):
        if time.time() - start > budget_s * 0.96:
            break
        best_h = slsqp_optimize(best_h, beta, max_iter=1000)
        best_c5 = compute_c5(best_h)

    # ---- final guided hill‑climb with worst‑gradient ----
    while time.time() - start < budget_s * 0.998:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c5:
            best_h, best_c5 = nh.copy(), nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
