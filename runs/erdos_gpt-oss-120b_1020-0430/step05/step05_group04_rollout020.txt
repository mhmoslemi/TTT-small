import time
import numpy as np

# ---------- core utilities ----------
def compute_raw_max(h: np.ndarray) -> int:
    """Maximum overlap count for a binary vector h."""
    # h is binary (0/1); 1-h also binary
    corr = np.correlate(h, 1 - h, mode="full")
    return int(corr.max())


def get_worst_shifts_and_sets(h: np.ndarray):
    """Return (max_raw, array_of_shifts) where the shift(s) attain the max overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_raw = int(corr.max())
    idxs = np.where(corr == max_raw)[0]
    ks = idxs - (n - 1)          # shift = index - (n-1)
    return max_raw, ks


def get_mismatch_sets(h: np.ndarray, k: int):
    """For a given shift k, return (S1, S0):
       S1 = positions i where h[i]=1 and h[i+k]=0 (valid indices)
       S0 = positions i where h[i]=0 and h[i+k]=1 (valid indices)
    """
    n = h.size
    idx = np.arange(n)
    if k >= 0:
        mask = (idx + k < n)
        s1 = idx[(h == 1) & mask & (h[idx + k] == 0)]
        s0 = idx[(h == 0) & mask & (h[idx + k] == 1)]
    else:
        kk = -k
        mask = (idx - kk >= 0)
        s1 = idx[(h == 1) & mask & (h[idx - kk] == 0)]
        s0 = idx[(h == 0) & mask & (h[idx - kk] == 1)]
    return s1, s0


def local_search(h: np.ndarray, rng: np.random.Generator,
                 start_time: float, deadline: float):
    """Improve binary vector h until deadline. Returns (best_h, best_raw)."""
    n = h.size
    best_h = h.copy()
    best_raw = compute_raw_max(h)

    while time.time() < deadline:
        max_raw, ks = get_worst_shifts_and_sets(h)
        if max_raw < best_raw:
            best_raw = max_raw
            best_h = h.copy()

        # try deterministic improvement on the worst shift(s)
        improved = False
        ks_shuffled = ks.copy()
        rng.shuffle(ks_shuffled)

        for k in ks_shuffled:
            s1, s0 = get_mismatch_sets(h, k)
            if s1.size == 0 or s0.size == 0:
                continue
            product = s1.size * s0.size
            max_eval = 20000  # cap of full enumeration

            if product <= max_eval:
                # exhaustive search
                best_swap = None
                best_new = max_raw
                for i in s1:
                    for j in s0:
                        h[i] = 0
                        h[j] = 1
                        new_raw = compute_raw_max(h)
                        if new_raw < best_new:
                            best_new = new_raw
                            best_swap = (i, j)
                        h[i] = 1
                        h[j] = 0
                if best_swap is not None:
                    i, j = best_swap
                    h[i] = 0
                    h[j] = 1
                    improved = True
                    break

                trials = max_eval
                best_swap = None
                best_new = max_raw
                for _ in range(trials):
                    i = rng.choice(s1)
                    j = rng.choice(s0)
                    h[i] = 0
                    h[j] = 1
                    new_raw = compute_raw_max(h)
                    if new_raw < best_new:
                        best_new = new_raw
                        best_swap = (i, j)
                    h[i] = 1
                    h[j] = 0
                if best_swap is not None:
                    i, j = best_swap
                    h[i] = 0
                    h[j] = 1
                    improved = True
                    break

        elapsed = time.time() - start_time
        total = deadline - start_time

            j = rng.choice(zeros)

            h[j] = 1
            new_raw = compute_raw_max(h)
            delta = new_raw - max_raw
            accept = False
            if delta <= 0:
                accept = True
            else:
                if rng.random() < np.exp(-delta / T):
                    accept = True
            if accept:
                max_raw = new_raw
                if new_raw < best_raw:
                    best_raw = new_raw
                    best_h = h.copy()
                # update ones/zeros for next iteration
                ones = np.flatnonzero(h == 1)
                zeros = np.flatnonzero(h == 0)
            else:
                h[i] = 1
                h[j] = 0

    return best_h, best_raw


# ---------- main entry ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # ---- initial binary vector (from parent if available) ----
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
        idx = np.argpartition(parent, -n // 2)[-n // 2 :]
        start_vec = np.zeros(n, dtype=int)
        start_vec[idx] = 1
        init_vectors = [start_vec]
    else:
        n = 360
        init_vectors = []

    best_h = None
    best_raw = None

    # ---- multiple restarts until time runs out ----
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # allocate a fraction of the remaining budget to this run
        sub_dead = time.time() + remaining * 0.35

        if init_vectors:
            h0 = init_vectors.pop(0)
        else:
            h0 = np.zeros(n, dtype=int)
            h0[rng.choice(n, n // 2, replace=False)] = 1

        h_loc, raw_loc = local_search(h0.copy(), rng, start_time, sub_dead)

        if best_raw is None or raw_loc < best_raw:
            best_raw = raw_loc
            best_h = h_loc.copy()

        # early stop if we already beat the known target (raw ≤ floor(0.3808·n/2))
        target_raw = int(np.floor(0.3808 * n / 2.0 + 1e-12))
        if best_raw is not None and best_raw <= target_raw:
            # we keep searching a bit more for possible further improvement
            pass

    # ---- up‑sampling stage (if time permits) ----
    if best_h is not None and time.time() < deadline and best_h.size * 2 <= 720:
        n_up = best_h.size * 2
        h_up = np.repeat(best_h, 2)  # still binary
        sub_dead = time.time() + (deadline - time.time()) * 0.5
        h_up, raw_up = local_search(h_up.copy(), rng, start_time, sub_dead)
        if raw_up < best_raw:
            best_raw = raw_up
            best_h = h_up.copy()
            n = n_up

    # ---- fallback in the unlikely event no solution was found ----
    if best_h is None:
        n = 180
        best_h = np.zeros(n, dtype=int)
        best_h[: n // 2] = 1
        best_raw = compute_raw_max(best_h)

    final_c = best_raw * 2.0 / best_h.size
    return best_h.astype(float).tolist(), float(final_c), int(best_h.size)
