def get_worst_shift_and_corr(h):

def enforce_sum(h, rng=None):
    n = h.size
    target = n / 2.0
    # clip initially
    h = np.clip(h, 0.0, 1.0)
    # compute current sum
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    # we need to adjust
    # binary search for tau
    lower = -1.0  # enough negative
    upper = 1.0   # enough positive
    # But to guarantee brackets: If sum > target, we need to subtract some; tau positive reduces values. If sum < target, tau negative increases values.
    # Use bounds based on current values
    # For safety, set lower = -2, upper = 2.
    for _ in range(60):
        tau = (lower + upper) / 2.0
        h_adj = np.clip(h - tau, 0.0, 1.0)
        s = h_adj.sum()
        if s > target:
            lower = tau
        else:
            upper = tau
    h = np.clip(h - tau, 0.0, 1.0)
    # after binary search, due to numerical error may not be exact; adjust residual with random moves
    diff = target - h.sum()
    if abs(diff) > 1e-10 and rng is not None:
        # fix residual by random adjustments
        while abs(diff) > 1e-12:
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

import time

def _fft_len(n):
    return 1 << ((2 * n - 1).bit_length())

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.size
    L = _fft_len(n)
    H = np.fft.rfft(np.concatenate([h, np.zeros(L - n)]))
    G = np.fft.rfft(np.concatenate([1 - h, np.zeros(L - n)]))
    corr = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def _worst_shift_and_corr(h: np.ndarray):
    """Return (k, full correlation) for the shift attaining the max."""
    n = h.size
    L = _fft_len(n)
    H = np.fft.rfft(np.concatenate([h, np.zeros(L - n)]))
    G = np.fft.rfft(np.concatenate([1 - h, np.zeros(L - n)]))
    corr = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    idx = int(np.argmax(corr))
    k = abs(idx - (n - 1))
    return k, corr


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    k, _ = _worst_shift_and_corr(h)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def enforce_sum(h: np.ndarray, rng: np.random.Generator = None) -> np.ndarray:
    """Project h onto {0≤x≤1, Σx = n/2}."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    # binary search for tau in h_adj = clip(h - tau,0,1)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        tau = (lo + hi) * 0.5
        adj = np.clip(h - tau, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(h - tau, 0.0, 1.0)
    diff = target - h.sum()
    if rng is not None and abs(diff) > 1e-12:
        while abs(diff) > 1e-12:
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def greedy_pairwise_exchange(h: np.ndarray, rng: np.random.Generator,
                             max_iters: int = 5000,
                             time_budget: float = np.inf):
    """Greedy mass transfer from high‑gradient to low‑gradient indices."""
    start = time.time()
    cur_c = compute_c5(h)
    n = h.size
    for _ in range(max_iters):
        if time.time() - start > time_budget:
            break
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        delta = grad[i] - grad[j]
        if delta <= 1e-12:
            break
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            break
        eps = max_eps
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            continue
        # revert and try half
        h[i] += eps
        h[j] -= eps
        eps = max_eps * 0.5
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            continue
        # revert and stop
        h[i] += eps
        h[j] -= eps
        break
    return h, cur_c


def continuous_refine(h: np.ndarray, rng: np.random.Generator,
                     time_budget: float):
    """Gradient‑guided refinement + greedy exchange."""
    start = time.time()
    cur_c = compute_c5(h)
    step = 0.02
    n = h.size
    # gradient descent phase
    for it in range(5000):
        if time.time() - start > time_budget:

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(h[i], 1.0 - h[j])

            eps = max(eps, 1e-8)
            h[i] -= eps
            h[j] += eps

            h = enforce_sum(h, rng)

            else:

    if remaining > 0:

                                            max_iters=2000,
                                            time_budget=remaining)
    return h, cur_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    budget = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # parent construction, if any
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0, c0, h0.size

    # Phase 1: binary simulated annealing (≈30% time)
    phase1_time = budget * 0.30
    n_sa = max(256, best_n if best_n else 256)
    if n_sa % 2:
        n_sa += 1
    h_bin, c_bin = binary_sa(n_sa, rng, phase1_time)
    if c_bin < best_c:
        best_h, best_c, best_n = h_bin, c_bin, n_sa

    # Phase 2: continuous refinement at current resolution (≈40% remaining)
    remaining = budget - (time.time() - start)
    if remaining > 0:
        ref_time = remaining * 0.40
        h_ref, c_ref = continuous_refine(best_h.copy(), rng, ref_time)
        if c_ref < best_c:
            best_h, best_c, best_n = h_ref, c_ref, best_n

    # Phase 3: up‑sample and refine (≈30% of what is left)
    remaining = budget - (time.time() - start)
    if remaining > 0 and best_n * 2 <= 2048:
        up_time = remaining * 0.30
        n_up = best_n * 2
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        h_up, c_up = continuous_refine(h_up, rng, up_time)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # Phase 4: light stochastic tweaks until time expires
    while time.time() - start < budget * 0.99:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
