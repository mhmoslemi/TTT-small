def update_counts_swap(i, j, h, counts):
    n = len(h)
    deltas = np.zeros(n, dtype=int)

    # i left
    for s in range(0, n - i):
        deltas[s] -= (1 - h[i + s])

    # i right (as RHS)
    for s in range(1, i + 1):
        p = i - s
        if p == j:
            continue
        deltas[s] += h[p]

    # j left (skip s=0)
    for s in range(1, n - j):
        rhs = h[j + s]
        if j + s == i:
            rhs = 0
        deltas[s] += (1 - rhs)

    # j right
    for s in range(1, j + 1):
        p = j - s
        if p == i:
            continue
        deltas[s] -= h[p]

    # ensure delta[0] = 0
    deltas[0] = 0

    # Apply deltas
    counts += deltas

    # swap bits
    h[i] = 0
    h[j] = 1

def delta_swap(i, j, h):
    n = h.shape[0]
    delta = np.zeros(n, dtype=np.int32)

    # i left
    delta[:n - i] -= (1 - h[i:])

    # i right (i as RHS)
    if i > 0:
        delta[1:i + 1] += h[i - 1::-1]  # reversed

    # j left (skip s=0)
    if j < n - 1:   # if there is any s>=1
        rhs = h[j + 1:]  # length n-j-1
        # adjust for i position
        if i > j:
            idx = i - j - 1  # position in rhs array (0-index)
            # ensure idx within length
            if 0 <= idx < rhs.shape[0]:

                # We'll compute base = 1 - rhs, then add 1 at idx.
                base = 1 - rhs
                base[idx] = 1  # because rhs becomes 0 -> 1 - 0 =1
            else:
                base = 1 - rhs

def smooth_max_obj_and_grad(x, beta, lam, target):
    n = x.shape[0]
    # compute g
    corr = np.correlate(x, 1 - x, mode='full')
    g = corr[n-1:].astype(np.float64)  # length n
    bc = beta * g
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    Z = np.sum(exp_bc)
    w = exp_bc / Z
    f = (max_bc + np.log(Z)) / beta
    # penalty
    diff = np.sum(x) - target
    f_total = f + lam * diff * diff

    # gradient
    grad = np.zeros_like(x, dtype=np.float64)
    # for each s, add w[s] * (1 - x[i+s]) for i+s < n
    # vectorized loop:
    for s in range(n):
        ws = w[s]
        if ws == 0.0:
            continue
        # left part
        if s < n:
            grad[:n - s] += ws * (1.0 - x[s:])
        # right part (subtraction)
        if s > 0:
            grad[s:] -= ws * x[:n - s]
    # add penalty gradient
    grad += 2.0 * lam * diff * 1.0
    return f_total, grad

def compute_c5(arr):
    """C5 for a (float) array."""
    n = arr.size
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * (2.0 / n))


def compute_counts(h):
    """Linear counts for binary h (0/1)."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    return corr[n - 1 :].astype(np.int32)  # s = 0..n-1


def adjust_sum_exact(v, rng):
    """Redistribute mass so that Σ v = n/2 exactly (bounds respected)."""
    n = v.size
    target = n / 2.0
    diff = target - v.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - v[i]) if diff > 0 else max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v


def delta_swap(i, j, h):
    """Delta of counts for swapping a 1 at i with a 0 at j (binary h)."""
    n = h.shape[0]
    delta = np.zeros(n, dtype=np.int32)

    # i as left element
    delta[: n - i] -= (1 - h[i:])

    # i as RHS (right side of a pair)
    if i > 0:
        delta[1 : i + 1] += h[i - 1 :: -1]

    # j as left (skip s=0)
    if j < n - 1:
        rhs = h[j + 1 :]  # length n-j-1, corresponds to s=1..
        if i > j:
            idx = i - j - 1
            if 0 <= idx < rhs.shape[0]:
                base = 1 - rhs
                base[idx] = 1  # after swap RHS becomes 0 → (1‑0)=1
            else:
                base = 1 - rhs
        else:
            base = 1 - rhs
        delta[1 : n - j] += base

    # j as RHS
    if j > 0:
        arr = h[j - 1 :: -1]  # length j
        delta[1 : j + 1] -= arr
        if i < j:
            idx = j - i - 1
            if 0 <= idx < arr.shape[0]:
                delta[j - i] += 1  # cancel subtraction of the pair (i,j)

    delta[0] = 0
    return delta


def search_binary(n, rng, deadline):
    """Heuristic binary search for low max‑overlap."""
    # start with a random balanced binary vector
    h = np.zeros(n, dtype=np.uint8)
    h[rng.choice(n, n // 2, replace=False)] = 1

    counts = compute_counts(h)
    max_cnt = int(counts.max())
    best_h = h.copy()
    best_max = max_cnt
    target_max = int(np.floor(TARGET_C * n / 2))

    # temperature for occasional uphill moves
    T0 = max(1.0, max_cnt - target_max)
    it = 0
    while time.time() < deadline:
        it += 1
        T = max(0.01, T0 * (0.9995 ** it))

        max_cnt = int(counts.max())
        if max_cnt <= target_max:
            best_h = h.copy()
            best_max = max_cnt
            break

        s_max = int(counts.argmax())

        # indices that contribute to the worst shift
        mask = (h[: n - s_max] == 1) & (h[s_max:] == 0)
        cand_i = np.flatnonzero(mask)
        if cand_i.size == 0:
            i = rng.choice(np.flatnonzero(h == 1))
        else:
            i = rng.choice(cand_i)

        # try a few zeros to improve the worst shift
        zeros = np.flatnonzero(h == 0)
        rng.shuffle(zeros)
        best_delta = None
        best_new = max_cnt
        best_j = None
        for j in zeros[:5]:
            if j == i:
                continue
            d = delta_swap(i, j, h)
            new_max = int((counts + d).max())
            if new_max < best_new:
                best_new = new_max
                best_delta = d
                best_j = j
                if new_max <= max_cnt:
                    break

        if best_j is not None and (best_new < max_cnt or rng.random() < np.exp(-(best_new - max_cnt) / T)):
            counts += best_delta
            h[i] = 0
            h[best_j] = 1
            max_cnt = best_new
            if max_cnt < best_max:
                best_max = max_cnt
                best_h = h.copy()
        else:
            # random uphill/neutral move
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            d = delta_swap(i2, j2, h)
            new_max = int((counts + d).max())
            if new_max < max_cnt or rng.random() < np.exp(-(new_max - max_cnt) / T):
                counts += d
                h[i2] = 0
                h[j2] = 1
                max_cnt = new_max
                if max_cnt < best_max:
                    best_max = max_cnt
                    best_h = h.copy()

    # exact sum correction (should already be satisfied)
    best_h = adjust_sum_exact(best_h.astype(float), rng)
    return best_h.astype(float), best_max


def smooth_max_obj_grad(x, beta, lam, target):
    """Smooth max + quadratic penalty; also returns gradient."""
    n = x.size
    # g_s = ∑ x_i (1‑x_{i+s})
    corr = np.correlate(x, 1.0 - x, mode="full")
    g = corr[n - 1 :].astype(np.float64)  # length n

    bc = beta * g
    m = np.max(bc)
    e = np.exp(bc - m)
    Z = np.sum(e)
    w = e / Z                         # softmax weights
    f = (m + np.log(Z)) / beta        # smooth maximum

    diff = np.sum(x) - target
    f_total = f + lam * diff * diff

    # gradient of g_s
    grad = np.zeros_like(x)
    for s in range(n):
        ws = w[s]
        if ws == 0.0:
            continue
        # left contribution (1‑x_{i+s})
        if s < n:
            grad[: n - s] += ws * (1.0 - x[s:])
        # right contribution (‑x_{i‑s})
        if s > 0:
            grad[s:] -= ws * x[: n - s]

    grad += 2.0 * lam * diff
    return f_total, grad


def lbfgs_optimize(x0, beta, lam, rng, deadline):
    """L‑BFGS‑B on smooth max with penalty; stops at deadline."""
    n = x0.size
    target = n / 2.0
    max_iter = 800
    # ensure time permits at least a few iterations
    while time.time() < deadline and max_iter > 0:
        res = minimize(
            fun=lambda x: smooth_max_obj_grad(x, beta, lam, target),
            x0=x0,
            method="L-BFGS-B",
            jac=True,
            bounds=[(0.0, 1.0)] * n,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = adjust_sum_exact(x, rng)
        # next stage will increase parameters, so break after one call
        break
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    # candidate lengths (all even)
    lengths = [360, 420, 480, 540, 600, 720]

    best_h = None
    best_c = np.inf
    best_n = None

    # quick correctness check for delta_swap (once, if time permits)
    if time.time() + 0.5 < deadline:
        n_test = 30
        h_test = rng.integers(0, 2, size=n_test, dtype=np.uint8)
        # force exact half ones
        target_half = n_test // 2
        cur = int(h_test.sum())
        if cur > target_half:
            idx = np.flatnonzero(h_test == 1)[: cur - target_half]
            h_test[idx] = 0
        elif cur < target_half:
            idx = np.flatnonzero(h_test == 0)[: target_half - cur]
            h_test[idx] = 1
        i = rng.choice(np.flatnonzero(h_test == 1))
        j = rng.choice(np.flatnonzero(h_test == 0))
        d = delta_swap(i, j, h_test)
        h2 = h_test.copy()
        h2[i] = 0
        h2[j] = 1
        if not np.array_equal(compute_counts(h_test) + d, compute_counts(h2)):
            raise AssertionError("delta_swap verification failed")

    for n in lengths:
        if time.time() >= deadline:
            break
        # allocate time slice for this length
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.7

        # --- binary stage -------------------------------------------------
        h_bin, max_cnt = search_binary(n, rng, sub_deadline)
        c_bin = 2.0 * max_cnt / n
        if c_bin < best_c:
            best_c, best_h, best_n = c_bin, h_bin.copy(), n

        # --- fractional refinement ----------------------------------------
        if time.time() >= deadline:
            break
        # start from binary (float) vector
        h_frac = h_bin.copy()
        # three L‑BFGS‑B stages with increasing aggressiveness
        stages = [
            (800.0, 1000.0, 300),
            (2000.0, 5000.0, 300),
            (8000.0, 20000.0, 300),
        ]
        for beta, lam, iters in stages:
            if time.time() >= deadline:
                break
            stage_dead = time.time() + min(iters * 0.005, deadline - time.time())
            h_frac = lbfgs_optimize(h_frac, beta, lam, rng, stage_dead)

        c_frac = compute_c5(h_frac)
        if c_frac < best_c:
            best_c, best_h, best_n = c_frac, h_frac.copy(), n

        # occasional short random fractional SA to shake out of plateaus
        if time.time() < deadline:
            cur = h_frac.copy()
            cur_c = compute_c5(cur)
            T = max(1e-3, cur_c * 0.01)
            it = 0
            while time.time() < deadline and it < 12000:
                i, j = rng.choice(n, size=2, replace=False)
                if cur[i] == cur[j]:
                    continue
                src, dst = (i, j) if rng.random() < 0.5 else (j, i)
                max_transfer = min(cur[src], 1.0 - cur[dst])
                if max_transfer <= 0:
                    continue
                delta = rng.random() * max_transfer
                cur[src] -= delta
                cur[dst] += delta
                new_c = compute_c5(cur)
                if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
                    cur_c = new_c
                    if cur_c < best_c:
                        best_c, best_h, best_n = cur_c, cur.copy(), n
                else:
                    cur[src] += delta
                    cur[dst] -= delta
                it += 1
                T *= 0.9995

    # fallback to parent if nothing better was found
    if best_h is None:
        if "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
            best_h = parent
            best_n = parent.size
            best_c = compute_c5(parent)
        else:
            n = 180
            best_h = np.zeros(n, dtype=float)
            best_h[: n // 2] = 1.0
            best_n = n
            best_c = compute_c5(best_h)

    # final sanity checks
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact integral
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        # simple redistribution
        zeros = np.where(best_h < 1.0)[0]
        ones = np.where(best_h > 0.0)[0]
        if diff > 0 and zeros.size:
            for i in zeros[: int(np.ceil(diff))]:
                best_h[i] = 1.0
        elif diff < 0 and ones.size:
            for i in ones[: int(np.ceil(-diff))]:
                best_h[i] = 0.0

    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
