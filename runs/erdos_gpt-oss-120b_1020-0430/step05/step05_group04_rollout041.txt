```python
import cvxpy as cp

def solve_milp(n, time_limit):
    x = cp.Variable(n, boolean=True)
    # Compute number of shifts to consider
    ks = list(range(1, n//2+1))  # exclude 0
    # Create y dict: y[(i,k)] variable
    y = {}
    constraints = [cp.sum(x) == n//2]
    for k in ks:
        for i in range(n):
            y_ik = cp.Variable(boolean=True)
            y[(i,k)] = y_ik
            constraints += [
                y_ik <= x[i],
                y_ik <= x[(i+k) % n],
                y_ik >= x[i] + x[(i+k) % n] - 1
            ]
    # Z variable
    Z = cp.Variable(integer=True)  # Actually Z integer, but treat as continuous.
    # For each k compute S_k as sum_i y[(i,k)].
    for k in ks:
        S_k = cp.sum([y[(i,k)] for i in range(n)])
        constraints.append(Z <= S_k)
    # Also Z <= n/2
    constraints.append(Z <= n/2)
    # Z >= 0
    constraints.append(Z >= 0)
    # Objective maximize Z
    prob = cp.Problem(cp.Maximize(Z), constraints)
    # Solve with CBC and time limit
    try:
        prob.solve(solver=cp.CBC, maximumSeconds=time_limit, verbose=False)
    except Exception:
        prob.solve(solver=cp.CBC, verbose=False)  # no time limit
    # Retrieve x values
    if x.value is None:
        return None
    x_vals = np.rint(np.array(x.value)).astype(float)  # should be 0 or 1
    return x_vals
```

```
def gradient_refine(h, max_iter=20, step=0.02):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0*h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * (0.5**it) * g
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute evenly across all entries for simplicity
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h
```

```
def lbfgsb_optimize(h0, beta, rng, max_iter=200, penalty=1000.0):
    n = h0.size
    target = n/2.0
    def obj(h):
        # enforce bounds inside but also penalty for sum
        corr = np.correlate(h, 1.0 - h, mode='full')
        bc = beta * corr
        m = np.max(bc)
        smooth = (m + np.log(np.sum(np.exp(bc - m)))) / beta
        sum_pen = penalty * (np.sum(h) - target) ** 2
        return smooth + sum_pen
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    h = np.clip(res.x, 0.0, 1.0)
    # final sum correction
    adjust_sum_exact(h, rng)
    return h
```

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_fractional_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h = h * target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    adjust_sum_exact(h, rng)
    return h

            g[k:] -= h[:upper]
        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            free = (h > 0.0) & (h < 1.0)
            if free.any():
                h[free] += diff / free.sum()
                h = np.clip(h, 0.0, 1.0)
            else:
                adjust_sum_exact(h, rng)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # base number of points
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        if n % 2:
            n += 1
        n = min(n, 720)
    else:
        n = 360
    if n % 2:
        n += 1

    # evaluate parent if present
    if "initial_h_values" in globals():
        parent_h = np.array(initial_h_values, dtype=float)
        parent_h = np.clip(parent_h, 0.0, 1.0)
        adjust_sum_exact(parent_h, rng)
        c_par = compute_c5(parent_h)
        if c_par < best_c:
            best_c, best_h, best_n = c_par, parent_h.copy(), n

    betas = (5.0, 20.0, 80.0, 200.0)
    max_restarts = 30
    restart = 0
    while time.time() < deadline and restart < max_restarts:
        restart += 1
        h0 = best_h.copy() if restart == 1 and best_h is not None else random_fractional_vector(n, rng)
        h = h0.copy()
        for beta in betas:
            if time.time() > deadline:
                break
            h = smoothmax_gradient_descent(h, beta, rng, step0=0.05, max_iter=20)
        if time.time() < deadline:
            h = gradient_refine(h, rng, step=0.02, max_iter=30)
        adjust_sum_exact(h, rng)
        h = np.clip(h, 0.0, 1.0)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), n
        if best_c < 0.38080:
            break

    # optional up‑sampling
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        h = h_up.copy()
        for beta in betas:
            if time.time() > deadline:
                break
            h = smoothmax_gradient_descent(h, beta, rng, step0=0.05, max_iter=20)
        if time.time() < deadline:
            h = gradient_refine(h, rng, step=0.02, max_iter=30)
        adjust_sum_exact(h, rng)
        h = np.clip(h, 0.0, 1.0)
        c_up = compute_c5(h)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h.copy(), n_up

    # fallback
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
