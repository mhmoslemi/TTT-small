def binary_sa_time(N, init_h, rng, time_budget):
    start = time.time()
    best_h = init_h.copy()
    best_c = compute_c5(best_h.astype(float))
    h = best_h.copy()
    c_curr = best_c
    # precompute list of indices with ones and zeros
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h) # or those where h==0
    # loop
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        # simulate SA temperature decreasing
        T = 0.02 * (1e-4 / 0.02) ** (elapsed / time_budget)  # as before
        # pick random pair
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]
        # swap
        h[i] = 0
        h[j] = 1
        c_new = compute_c5(h.astype(float))
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            c_curr = c_new
            # update ones/zeros lists
            ones[i_pos] = j
            zeros[j_pos] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                if best_c <= 0.3808:  # early exit threshold
                    return best_h, best_c
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] (h in [0,1], sum(h)=len/2)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in‑place so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution, fast because diff quickly shrinks
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k = ∫h(1−h(·+k))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, step_scale: float = 0.7):
    """Hill‑climb using gradient‑directed ε‑transfers."""
    best_h, best_c = h.copy(), compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- Soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(x0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    """SLSQP with soft‑max surrogate for max_k O_k."""
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, x0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple random fix (few iterations suffice)
        for _ in range(2000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            if add != 0.0:
                x[i] += add
                diff = CURRENT_TARGET - x.sum()
                if abs(diff) < 1e-12:
                    break
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] = 1.0 - h[k:]
            g[k:] = 1.0 - h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- Binary simulated annealing ----------
def binary_sa_time(N: int, init_h: np.ndarray,
                  rng: np.random.Generator, time_budget: float,
                  early_target: float = 0.3808):
    """SA on binary vectors with exact N/2 ones."""
    start = time.time()
    h = init_h.copy()
    c_curr = compute_c5(h.astype(float))
    best_h, best_c = h.copy(), c_curr
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i, j = ones[i_pos], zeros[j_pos]
        # swap
        h[i] = 0
        h[j] = 1
        c_new = compute_c5(h.astype(float))
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            c_curr = c_new
            ones[i_pos] = j
            zeros[j_pos] = i
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= early_target:
                    return best_h, best_c
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best_h, best_c


# ---------- Main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ----- problem size -----
    N = 512
    target = N // 2

    # ----- initial binary from initial_h_values if present -----
    if 'initial_h_values' in globals():
        vals = np.array(initial_h_values, dtype=float)
        order = np.argsort(-vals)
        init_bin = np.zeros(N, dtype=np.int8)
        init_bin[order[:target]] = 1
    else:
        init_bin = np.zeros(N, dtype=np.int8)
        init_bin[rng.choice(N, target, replace=False)] = 1

    best_h = init_bin.astype(float)
    best_c = compute_c5(best_h)

    # ----- binary SA (≈65% of time) -----
    time_binary = total_budget * 0.65
    bin_h, bin_c = binary_sa_time(N, init_bin, rng, time_binary, early_target=0.3808)
    if bin_c < best_c:
        best_h, best_c = bin_h.astype(float), bin_c

    # ----- remaining time for continuous refinement -----
    remaining = total_budget - (time.time() - start)
    if remaining > 1.0:
        # start from best binary (as float)
        h = best_h.copy()
        # guided hill‑climb
        max_hc_iters = int(min(5000, remaining * 600))
        h, c = guided_hill_climb(h, rng, max_iters=max_hc_iters, step_scale=0.7)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # soft‑max SLSQP passes with increasing β
        for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
            if time.time() - start > total_budget * 0.95:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=800 if beta < 200 else 1200)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

        # final gradient tidy‑up (if time permits)
        if time.time() - start < total_budget * 0.97:
            h = gradient_refine(best_h, rng, step=0.01, max_iter=8)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

    # ----- stochastic tiny tweaks until budget expires -----
    while time.time() - start < total_budget * 0.999:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * 0.02
                best_h[i] -= eps
                best_h[j] += eps
                best_h = np.clip(best_h, 0.0, 1.0)
                best_h = enforce_sum(best_h, rng)
                best_c = compute_c5(best_h)

    # ----- final sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
