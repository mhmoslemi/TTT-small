def compute_delta_swap(h, i, j):
    n = h.size
    # Create array of indices for shift offsets
    ks = np.arange(-n+1, n)  # length 2n-1
    # For i:
    idx_i_plus = (i + ks) % n
    idx_i_minus = (i - ks) % n
    delta_i = -(1 - h[idx_i_plus]) + h[idx_i_minus]
    # For j:
    idx_j_plus = (j + ks) % n
    idx_j_minus = (j - ks) % n
    delta_j = (1 - h[idx_j_plus]) - h[idx_j_minus]
    delta = delta_i + delta_j
    return delta

import time
import numpy as np

# -------------------------------------------------
# Core binary correlation utilities
# -------------------------------------------------
def compute_O(h):
    """Periodic correlation O[k] = sum_i h_i (1-h_{i+k}) for k in -n+1..n-1."""
    return np.correlate(h, 1 - h, mode="full").astype(np.int64)


def delta_swap(h, i, j, ks, center):
    """Delta of O after swapping 1 at i with 0 at j (i!=j)."""
    # delta = h[i+ks] + h[i-ks] - h[j+ks] - h[j-ks]
    delta = h[(i + ks) % h.size] + h[(i - ks) % h.size] - h[(j + ks) % h.size] - h[(j - ks) % h.size]
    delta[center] = 0  # shift 0 must stay zero
    return delta


def worst_shift(O, center):
    idx = int(O.argmax())
    k = idx - center
    return idx, k


def mismatch_sets(h, k):
    """Return indices S1 (1->0) and S0 (0->1) for shift k."""
    n = h.size
    i = np.arange(n)
    i_k = (i + k) % n
    S1 = i[(h == 1) & (h[i_k] == 0)]
    S0 = i[(h == 0) & (h[i_k] == 1)]
    return S1, S0


def greedy_local_search(h, O, ks, center, rng, time_deadline):
    """Iterative improvement focusing on the current worst shift."""
    while time.time() < time_deadline:
        idx, k = worst_shift(O, center)
        if idx == center:  # all overlaps zero
            break
        S1, S0 = mismatch_sets(h, k)
        if S1.size == 0 or S0.size == 0:
            break

        max_samples = 200
        total_pairs = int(S1.size) * int(S0.size)
        if total_pairs <= max_samples:
            pair_is = np.repeat(S1, S0.size)
            pair_js = np.tile(S0, S1.size)
        else:
            pair_is = rng.choice(S1, size=max_samples, replace=True)
            pair_js = rng.choice(S0, size=max_samples, replace=True)

        cur_max = int(O.max())
        best_new = cur_max
        best_i = best_j = None
        best_delta = None

        for i, j in zip(pair_is, pair_js):
            d = delta_swap(h, i, j, ks, center)
            new_max = int((O + d).max())
            if new_max < best_new:
                best_new = new_max
                best_i, best_j = i, j
                best_delta = d
                if best_new < cur_max - 2:  # good enough early exit
                    break

        if best_i is None:
            break  # no improving swap found

        # apply best swap
        O += best_delta
        h[best_i] = 0
        h[best_j] = 1
    return h, O


def simulated_annealing(h, O, ks, center, rng, steps, T0, decay, time_deadline):
    """Simple SA on binary space using O(n) delta updates."""
    cur_max = int(O.max())
    T = T0
    for _ in range(steps):
        if time.time() > time_deadline:
            break
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        d = delta_swap(h, i, j, ks, center)
        new_max = int((O + d).max())
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-12)):
                accept = True
        if accept:
            O += d
            cur_max = new_max
            h[i] = 0
            h[j] = 1
        T *= decay
    return h, O

    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # baseline from possible parent
    if 'initial_h_values' in globals():

        init_c = compute_c5(init_h)

    # candidate sample sizes (even)

    if 'initial_h_values' in globals():

    cand_ns = [n for n in cand_ns if n % 2 == 0]

    for n in cand_ns:
        if time.time() > deadline:
            break
        ks = np.arange(-n + 1, n, dtype=np.int64)
        center = n - 1

        # prepare a starting binary vector from parent if sizes match or by interpolation
        start_vectors = []

        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            if parent.size != n:
                # interpolate parent to length n and binarize by taking largest n/2 values
                x_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                x_target = np.linspace(0.0, 2.0, n, endpoint=False)
                interp = np.interp(x_target, x_parent, parent)
                idx = np.argpartition(interp, -n // 2)[-n // 2:]
                bin_vec = np.zeros(n, dtype=np.int64)
                bin_vec[idx] = 1
            else:

                bin_vec = (parent > 0.5).astype(np.int64)
                # enforce exact half ones

                if cur != n // 2:

                    zeros = np.flatnonzero(1 - bin_vec)

            O = compute_O(h)

            # 1) greedy local improvement
            h, O = greedy_local_search(h, O, ks, center, rng, deadline)
            c_val = O.max() * (2.0 / n)

                best_n = n

                best_h = h.astype(float).copy()
                best_n = n

    # -------------------------------------------------
    # Optional up‑sampling of the best solution

        h_up, O_up = greedy_local_search(h_up, O_up, ks_up, center_up, rng, deadline)
        c_up = O_up.max() * (2.0 / n_up)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up.astype(float).copy()
            best_n = n_up

        )

            best_c = c_up
            best_h = h_up.astype(float).copy()
            best_n = n_up

        return best_h.tolist(), float(final_c), int(best_n)

    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
