def gradient_maximin(h0, rng, max_iters=200, beta_start=5.0, beta_factor=1.3, beta_max=5000.0, lr=5.0, lr_decay=0.995):
    n = h0.size
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    beta = beta_start
    for it in range(max_iters):
        # compute inner correlation
        inner = np.correlate(h, h, mode='full') * dx
        w_raw = np.exp(-beta * inner)
        w = w_raw / w_raw.sum()
        # compute gradient
        g = np.zeros_like(h)
        wk0 = w[n-1]
        g += 2.0 * wk0 * h
        # contributions for k>0
        for k in range(1, n):
            wk = w[n-1 + k]  # same as w[n-1 - k] due to symmetry
            if wk == 0.0:
                continue
            # slice updates
            g[:n - k] += wk * h[k:]
            g[k:] += wk * h[:n - k]
        g *= dx
        # gradient ascent step
        step = lr * g
        h = h + step
        # clip
        h = np.clip(h, 0.0, 1.0)
        # adjust sum to target
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # update beta and lr
        beta = min(beta * beta_factor, beta_max)
        lr *= lr_decay
        # early stopping if lr becomes tiny
        if lr < 1e-6:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99
    best_h = None
    best_c = np.inf
    best_n = None
    # list of n values
    n_vals = [300, 360, 420, 480, 600]
    for n in n_vals:
        if time.time() > deadline:
            break
        target_sum = n / 2.0
        # generate start vectors
        start_vecs = []
        # parent interpolation
        if 'initial_h_values' in globals():
            parent_arr = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent_arr.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent_arr)
            start_vecs.append(interp)
            # binary version of interpolation: take highest values
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)
        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)
        # random uniform fractional
        rand_frac = rng.random(n)
        rand_frac = rand_frac * (target_sum / rand_frac.sum())
        adjust_sum_exact(rand_frac, rng)
        start_vecs.append(rand_frac)
        # random biased fractional (square)
        rand_frac2 = rng.random(n) ** 2
        rand_frac2 = rand_frac2 * (target_sum / rand_frac2.sum())
        adjust_sum_exact(rand_frac2, rng)
        start_vecs.append(rand_frac2)
        # structured half-half block
        block = np.zeros(n)
        block[: n // 2] = 1.0
        start_vecs.append(block)
        # loop over starts
        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)
            # gradient maximin
            max_iters = 150
            # use early stop if time running out
            if time.time() + max_iters * 0.005 > deadline:  # rough estimate 5ms per iter
                max_iters = max(10, int((deadline - time.time()) / 0.005))
            h_best, c_best = gradient_maximin(h, rng, max_iters=max_iters)
            if c_best < best_c:
                best_c = c_best
                best_h = h_best.copy()
                best_n = n
    # Upsample refinement if time remains
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        # run gradient on upsampled
        max_iters = 200
        if time.time() + max_iters * 0.005 < deadline:
            h_up, c_up = gradient_maximin(h_up, rng, max_iters=max_iters)
            if c_up < best_c:
                best_h = h_up
                best_c = c_up
                best_n = n_up
    # final ensure sum correct and clip
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)

        final_c = compute_c5(best_h)

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap for step function samples h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Redistribute mass to satisfy Σh = n/2 exactly, respecting bounds."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_maximin(
    h0,
    rng,
    max_iters=400,
    beta_start=2.0,
    beta_factor=1.7,
    beta_max=5000.0,
    lr=8.0,
    lr_decay=0.99,
    time_deadline=None,
):
    """Projected gradient ascent on the smoothed min autocorrelation."""
    n = h0.size
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    beta = beta_start
    for it in range(max_iters):
        if time_deadline is not None and time.time() > time_deadline:
            break
        # autocorrelation (integral)
        inner = np.correlate(h, h, mode="full") * dx
        # smooth min weights (shifted to avoid underflow)
        shift = inner.min()
        w_raw = np.exp(-beta * (inner - shift))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum
        # gradient of the smoothed min
        g = np.zeros_like(h)
        wk0 = w[n - 1]
        g += 2.0 * wk0 * h
        for k in range(1, n):
            wk = w[n - 1 + k]  # same as w[n-1-k] because of symmetry
            if wk == 0.0:
                continue
            g[: n - k] += wk * h[k:]
            g[k:] += wk * h[: n - k]
        g *= dx
        # ascent step
        h += lr * g
        h = np.clip(h, 0.0, 1.0)
        adjust_sum_exact(h, rng)
        # record improvement
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # update schedule
        beta = min(beta * beta_factor, beta_max)
        lr *= lr_decay
        if lr < 1e-6:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # list of sample sizes to explore (even numbers give exact target sum)
    n_vals = [240, 300, 360, 420, 480, 600, 720]

    for n in n_vals:
        if time.time() > deadline:
            break
        target = n / 2.0
        start_vecs = []

        # parent interpolation (if available)
        if "initial_h_values" in globals():
            parent = np.asarray(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp)
            # binary version of interpolated values
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)

        # random binary vector

        rand_frac2 = rng.random(n) ** 2

        adjust_sum_exact(h_up, rng)

        remaining = deadline - time.time()

            time_deadline=deadline,
        )
        if c_up_opt < best_c:

            best_h = h_up_opt.copy()
            best_n = n_up

    # final safety projection
    if best_h is None:
        best_n = 180
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
