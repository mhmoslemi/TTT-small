def slsqp_optimize(h0, beta, rng, max_iter=600):
    n = h0.size
    target = n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': lambda x, t=target: np.sum(x) - t}
    def obj(x, b=beta):
        corr = np.correlate(x, 1.-x, mode='full')
        bc = b * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / b
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':max_iter,'ftol':1e-12,'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction as earlier
    diff = target - x.sum()
    # We can do a simple loop distributing diff.
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0,n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000, tol=1e-12):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < tol:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=20):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # re‑project sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def slsqp_optimize(h0, beta, rng, max_iter=800):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t}
    def obj(x, b=beta):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = b * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / b
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def optimize_pipeline(h0, rng, n, deadline):
    h = h0.copy()
    adjust_sum_exact(h, rng)
    betas = [50, 200, 800, 3000, 12000]
    for beta in betas:
        if time.time() > deadline:
            break
        it = 800 if beta < 3000 else 1200
        h = slsqp_optimize(h, beta, rng, max_iter=it)
    if time.time() < deadline:
        h = gradient_refine(h, rng, step=0.02, max_iter=25)
    return h

        fr = fr * target / fr.sum()
        fr = np.clip(fr, 0.0, 1.0)
        adjust_sum_exact(fr, rng)
        starts.append(fr)

        # random fractional (squared)
        fr2 = rng.random(n) ** 2
        fr2 = fr2 * target / fr2.sum()
        fr2 = np.clip(fr2, 0.0, 1.0)
        adjust_sum_exact(fr2, rng)
        starts.append(fr2)

        # random fractional (sqrt)
        fr3 = np.sqrt(rng.random(n))
        fr3 = fr3 * target / fr3.sum()
        fr3 = np.clip(fr3, 0.0, 1.0)
        adjust_sum_exact(fr3, rng)
        starts.append(fr3)

        for h0 in starts:
            if time.time() > deadline - 5.0:
                break
            h_opt = quick_optimize(h0, rng, n, deadline)
            c = compute_c5(h_opt)
            if c < best_c:
                best_h, best_c, best_n = h_opt.copy(), c, n

            # perturb current best a little
            if best_h is not None and time.time() < deadline - 2.0:
                pert = best_h.copy()
                pert += rng.normal(scale=0.06, size=pert.shape)
                pert = np.clip(pert, 0.0, 1.0)
                adjust_sum_exact(pert, rng)
                pert_opt = quick_optimize(pert, rng, best_n, deadline)
                c2 = compute_c5(pert_opt)
                if c2 < best_c:
                    best_h, best_c, best_n = pert_opt.copy(), c2, best_n

    # --- final refinement at higher resolution ---
    while time.time() < deadline - 3.0:
        if best_h is None:
            break
        n_new = best_n * 2
        if n_new > 720:
            break
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_new, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        h_up = full_optimize(h_up, rng, n_new, deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_new
        else:
            break

    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
