    size = 2*n - 1
    c = np.zeros(size)

def coordinate_descent(h, rng, deadline, max_iter=200000):
    n = h.shape[0]
    target = n / 2.0
    O = np.correlate(h, 1.0 - h, mode='full')
    Omax = O.max()
    no_improve = 0
    it = 0
    while time.time() < deadline and it < max_iter:
        idx = O.argmax()
        k = idx - (n - 1)
        # compute c_i_k vector for all i
        if k >= 0:
            term1 = np.empty(n)
            term1[:n - k] = 1.0 - h[k:]
            term1[n - k:] = 0.0
            term2 = np.empty(n)
            term2[:k] = 0.0
            term2[k:] = h[:n - k]
        else:
            k2 = -k
            term1 = np.empty(n)
            term1[:k2] = 0.0
            term1[k2:] = 1.0 - h[:n - k2]
            term2 = np.empty(n)
            term2[:n - k2] = h[k2:]
            term2[n - k2:] = 0.0
        c = term1 - term2
        inc = np.where((c < -1e-12) & (h < 1.0 - 1e-12))[0]
        dec = np.where((c > 1e-12) & (h > 1e-12))[0]
        if inc.size == 0 or dec.size == 0:
            break
        i = rng.choice(inc)
        j = rng.choice(dec)
        dmax = min(1.0 - h[i], h[j])
        if dmax <= 1e-12:
            # shouldn't happen but skip
            it += 1
            continue
        # compute contribution vectors for i and j
        ci = contrib_vector(h, i)
        cj = contrib_vector(h, j)
        # try full move first
        d = dmax
        accepted = False
        while d > 1e-10:
            delta_O = d * (ci - cj)
            Onew = O + delta_O
            new_max = Onew.max()
            if new_max < Omax - 1e-12:
                O = Onew
                Omax = new_max
                h[i] += d
                h[j] -= d
                accepted = True
                break
            d *= 0.5
        if not accepted:
            # no improvement, maybe try other pair later
            no_improve += 1
            if no_improve > 5000:
                break
        else:
            no_improve = 0
        # occasional sum correction
        if (it & 0xFF) == 0:
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                # adjust a random index
                idx_adj = rng.integers(0, n)
                add = min(diff, 1.0 - h[idx_adj]) if diff > 0 else max(diff, -h[idx_adj])
                h[idx_adj] += add
        it += 1
    return h, O

def contrib_vector(h, i):
    n = h.shape[0]
    size = 2*n - 1
    c = np.zeros(size)
    # term1: (1 - h[i + t]) for t where i+t < n
    max_t1 = n - 1 - i
    # t goes from -i to max_t1
    t_start1 = -i
    # indices in c
    idx_start1 = t_start1 + (n - 1)
    # Fill segment
    if max_t1 >= -i:
        # length = max_t1 - (-i) + 1 = max_t1 + i + 1
        length1 = max_t1 + i + 1
        c[idx_start1:idx_start1+length1] = 1.0 - h[i + np.arange(-i, max_t1+1)]

    length2 = t_max2 - t_min2 + 1
    c[idx_start2:idx_start2+length2] += -h[i - np.arange(t_min2, t_max2+1)]
    return c

    n = h.size
    target = n/2.0
    for it in range(max_iter):
        O = np.correlate(h, 1.0 - h, mode='full')
        idx = O.argmax()
        k = idx - (n - 1)
        # gradient g_i: derivative of O_max? Actually we want gradient of O_k

        if k >= 0:

from scipy.optimize import minimize

# ---------- global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    h[:] = np.clip(h, 0.0, 1.0)
    return h

# ---------- contribution of a single index ----------
def contrib_vector(h, i):
    n = h.size
    size = 2 * n - 1
    c = np.zeros(size)

    # term1: (1 - h[i + t]) where i + t in [0, n-1]
    max_t1 = n - 1 - i               # highest t where i+t < n
    t_start1 = -i
    length1 = max_t1 + i + 1         # always n
    idx_start1 = t_start1 + (n - 1)
    c[idx_start1:idx_start1 + length1] = 1.0 - h[i + np.arange(-i, max_t1 + 1)]

    # term2: -h[i - t] where i - t in [0, n-1]
    t_min2 = i - (n - 1)
    length2 = n                     # also always n
    idx_start2 = t_min2 + (n - 1)
    c[idx_start2:idx_start2 + length2] += -h[i - np.arange(t_min2, i + 1)]

    return c

# ---------- coordinate descent targeting the worst shift ----------
def coordinate_descent(h, rng, deadline, max_iter=200000):
    n = h.size
    target = n / 2.0
    O = np.correlate(h, 1.0 - h, mode="full")
    Omax = O.max()
    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iter:
        idx = O.argmax()
        k = idx - (n - 1)

        # derivative of the worst shift for all positions
        if k >= 0:
            term1 = np.empty(n)
            term1[:n - k] = 1.0 - h[k:]
            term1[n - k:] = 0.0
            term2 = np.empty(n)
            term2[:k] = 0.0
            term2[k:] = h[:n - k]
        else:
            k2 = -k
            term1 = np.empty(n)
            term1[:k2] = 0.0
            term1[k2:] = 1.0 - h[:n - k2]
            term2 = np.empty(n)
            term2[:n - k2] = h[k2:]
            term2[n - k2:] = 0.0
        c = term1 - term2

        inc = np.where((c < -1e-12) & (h < 1.0 - 1e-12))[0]
        dec = np.where((c > 1e-12) & (h > 1e-12))[0]
        if inc.size == 0 or dec.size == 0:
            break

        i = rng.choice(inc)
        j = rng.choice(dec)
        dmax = min(1.0 - h[i], h[j])
        if dmax <= 1e-12:
            it += 1
            continue

        ci = contrib_vector(h, i)
        cj = contrib_vector(h, j)

        d = dmax
        accepted = False
        while d > 1e-10:
            delta_O = d * (ci - cj)
            Onew = O + delta_O
            new_max = Onew.max()
            if new_max < Omax - 1e-12:
                O = Onew
                Omax = new_max
                h[i] += d
                h[j] -= d
                accepted = True
                break
            d *= 0.5

        if not accepted:
            no_improve += 1
            if no_improve > 5000:
                break
        else:
            no_improve = 0

        # occasional sum‑correction to avoid drift
        if (it & 0xFF) == 0:
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                idx_adj = rng.integers(0, n)
                add = min(diff, 1.0 - h[idx_adj]) if diff > 0 else max(diff, -h[idx_adj])
                h[idx_adj] += add

        it += 1
    return h, O

# ---------- gradient refinement for the worst shift ----------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        O = np.correlate(h, 1.0 - h, mode="full")
        idx = O.argmax()
        k = idx - (n - 1)

        if k >= 0:
            g = np.empty(n)
            g[:n - k] = 1.0 - h[k:]
            g[n - k:] = 0.0
            g2 = np.empty(n)
            g2[:k] = 0.0
            g2[k:] = -h[:n - k]
            g += g2
        else:
            k2 = -k
            g = np.empty(n)
            g[:k2] = 0.0
            g[k2:] = 1.0 - h[:n - k2]
            g2 = np.empty(n)
            g2[:n - k2] = -h[k2:]
            g2[n - k2:] = 0.0
            g += g2

        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            i_adj = rng.integers(0, n)
            add = min(diff, 1.0 - h[i_adj]) if diff > 0 else max(diff, -h[i_adj])
            h[i_adj] += add
    return h

# ---------- SLSQP with smoothed max (log‑sum‑exp) ----------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum enforcement
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

    rng = np.random.default_rng(seed)

    deadline = start_time + 0.99 * budget_s

    candidate_ns = [360, 240, 180]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        start_vectors = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vectors.append(interp)                       # fractional
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vectors.append(bin_vec)                       # binary

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vectors.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        start_vectors.append(rand_frac)

        # alternating binary pattern
        alt_bin = np.tile([1.0, 0.0], n // 2)
        start_vectors.append(alt_bin)

        for base_h in start_vectors:
            if time.time() > deadline:
                break
            h = np.clip(base_h.astype(float), 0.0, 1.0)
            adjust_sum_exact(h, rng)

            c0 = compute_c5(h)
            if c0 < best_c:
                best_h, best_c, best_n = h.copy(), c0, n

            # ---- coordinate descent ----
            h_cd, O = coordinate_descent(h.copy(), rng, deadline, max_iter=200000)
            # fix possible tiny sum drift
            adjust_sum_exact(h_cd, rng)
            c_cd = compute_c5(h_cd)
            if c_cd < best_c:
                best_h, best_c, best_n = h_cd.copy(), c_cd, n

            # ---- gradient refinement ----
            h_gr = gradient_refine(h_cd, rng, step=0.02, max_iter=8)
            c_gr = compute_c5(h_gr)
            if c_gr < best_c:
                best_h, best_c, best_n = h_gr.copy(), c_gr, n

            # ---- SLSQP cascade ----
            temp_h = h_gr
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
                if time.time() > deadline:
                    break
                max_iter = 400 if beta < 200 else 600
                if beta >= 2000:
                    max_iter = 800
                try:
                    temp_h = slsqp_optimize(temp_h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c_tmp = compute_c5(temp_h)
                if c_tmp < best_c:
                    best_h, best_c, best_n = temp_h.copy(), c_tmp, n

            # final gradient sweep
            temp_h = gradient_refine(temp_h, rng, step=0.02, max_iter=12)
            c_final = compute_c5(temp_h)
            if c_final < best_c:
                best_h, best_c, best_n = temp_h.copy(), c_final, n

    # optional up‑sampling if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        # brief refinement
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=8)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # fallback in the unlikely event no solution was found
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
