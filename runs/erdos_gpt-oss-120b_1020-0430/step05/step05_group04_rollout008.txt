def gradient_smoothmax(h, beta):
    n = h.size
    v = np.correlate(h, 1 - h, mode='full')
    w = np.exp(beta * v - np.max(beta * v)) # subtract max to prevent overflow
    w_sum = w.sum()
    w = w / w_sum
    # now compute gradient
    g = np.zeros_like(h, dtype=float)
    # shift i ranges
    for shift_idx, wi in enumerate(w):
        i = shift_idx - (n - 1)  # shift offset
        if wi == 0:
            continue
        # first part: (1 - h_{j+i})
        if i >= 0:
            g[:n - i] += wi * (1 - h[i:])
        else:
            g[-i:] += wi * (1 - h[:n + i])
        # second part: - h_{j-i}
        if i >= 0:
            g[i:] -= wi * h[:n - i]
        else:
            g[:n + i] -= wi * h[-i:]
    return g

def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # maintain list of ones and zeros
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    # temperature schedule: initial T = cur_c * 0.05 or similar
    init_T = max(1e-4, cur_c * 0.02)
    iter_count = 0
    while time.time() < deadline:
        # choose random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        # acceptance
        T = init_T * (0.9995 ** iter_count) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            # accept
            cur_c = c
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                # if no improvement for long, break?
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
        iter_count += 1
        # early stop if no improvement in many steps
        if iter_count % 5000 == 0 and best_c == cur_c:
            break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ------------------ basic utilities ------------------
def compute_c5(h):
    """max overlap integral for discrete h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Modify h in-place so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ------------------ smooth‑max gradient ------------------
def gradient_smoothmax(h, beta):
    """Gradient of (1/β)log∑exp(β·overlap) wrt h."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")        # overlaps for all integer shifts
    m = np.max(beta * v)
    exp_term = np.exp(beta * v - m)
    w = exp_term / exp_term.sum()                   # soft‑max weights, sum to 1
    g = np.zeros_like(h, dtype=np.float64)
    one_minus_h = 1.0 - h
    for shift_idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        i = shift_idx - (n - 1)                     # signed shift
        if i >= 0:
            g[: n - i] += wi * one_minus_h[i:]
            g[i :]    -= wi * h[: n - i]
        else:
            ii = -i
            g[ii :]   += wi * one_minus_h[: n - ii]
            g[: n - ii] -= wi * h[ii :]
    return g

def worst_shift_gradient(h):
    """Sub‑gradient using the currently worst shift(s)."""
    n = h.size
    v = np.correlate(h, 1.0 - h, mode="full")
    max_val = v.max()
    idxs = np.where(v == max_val)[0]
    g = np.zeros_like(h, dtype=np.float64)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            g[: n - k] += 1.0 - h[k:]
            g[k :]    -= h[: n - k]
        else:
            kk = -k
            g[kk :]   += 1.0 - h[: n - kk]
            g[: n - kk] -= h[kk :]
    g /= max(1, len(idxs))
    return g

# ------------------ descent helpers ------------------
def gradient_descent_smoothmax(h, betas, base_step, it_per_beta, rng, deadline):
    """Run projected gradient descent on smooth‑max for a schedule of β."""
    n = h.size
    for beta in betas:
        for it in range(it_per_beta):
            if time.time() > deadline:
                return h
            g = gradient_smoothmax(h, beta)
            if np.linalg.norm(g) < 1e-14:
                continue
            best_h = h
            best_c = compute_c5(h)
            for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
                step = base_step * mult
                cand = h - step * g
                np.clip(cand, 0.0, 1.0, out=cand)
                adjust_sum_exact(cand, rng)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    best_h = cand
            h = best_h
    return h

def worst_shift_refine(h, rng, base_step, iters, deadline):

# ------------------ SLSQP smooth‑max cascade ------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    # exact sum fix
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa_hill_climb(h_bin, rng, deadline):
    """Simulated‑annealing swap hill‑climb on binary vectors."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(1e-4, cur_c * 0.02)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                it = 0  # reset stagnation counter
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it > 8000:
            break
    return best_h, best_c

        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
