    corr = np.correlate(h, 1-h, mode='full')
    # convert to float
    c = corr
    # find indices of top M values
    idx = np.argpartition(c, -M)[-M:]
    # weight equally
    grad = np.zeros_like(h)
    for j in idx:
        shift = j - (len(h) - 1)  # shift offset
        if shift == 0:
            grad += 1 - 2*h
        else:
            # for i where i+shift in range
            if shift > 0:
                # i from 0..n-shift-1: g_i += 1 - h[i+shift]
                grad[:len(h)-shift] += 1 - h[shift:]

                # i from 0..n-s-1: g_i -= h[i+s]

    grad /= M
    return grad

import time, numpy as np, scipy.optimize as opt

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # slight clip to enforce bounds
    np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_top_m(h, M=5):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    # c = corr (unscaled)
    flat = corr.ravel()
    idx = np.argpartition(flat, -M)[-M:]  # indices of top M
    grads = np.zeros(n, dtype=float)
    offset = n - 1
    for j in idx:
        shift = j - offset
        if shift == 0:
            grads += 1.0 - 2.0 * h
        elif shift > 0:
            s = shift
            grads[:n - s] += 1.0 - h[s:]
            grads[s:] -= h[:n - s]
        else:
            s = -shift
            grads[s:] += 1.0 - h[:n - s]
            grads[:n - s] -= h[s:]
    grads /= M
    return grads

def projected_gradient_descent(h0, rng, max_iter, init_step, M=5, time_deadline=None):
    h = h0.copy()
    step = init_step
    for it in range(max_iter):
        if time_deadline is not None and time.time() > time_deadline:
            break
        grad = grad_top_m(h, M=M)
        h -= step * grad
        # clip
        np.clip(h, 0.0, 1.0, out=h)
        # adjust sum
        adjust_sum_exact(h, rng)
        # decay step
        step *= 0.995
    return h

def fractional_sa_hill_climb(h0, rng, deadline):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[src] -= delta
        h[dst] += delta
        c = compute_c5(h)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
                last_improve = it
        else:
            # revert
            h[src] += delta
            h[dst] -= delta
        it += 1
        if it - last_improve > 20000:
            break
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=800):
    # Use log-sum-exp as smooth objective
    n = h0.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        # scaled by (2/n) not needed for argmin
        b_corr = beta * corr
        m = np.max(b_corr)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(b_corr - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target}
    # Use L-BFGS-B method with bounds and constraints? Actually L-BFGS-B can't handle constraints. Use SLSQP.
    res = opt.minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol':1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def upsample_h(h, new_n):
    n = h.size
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xt = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_up = np.interp(xt, xs, h)
    return h_up

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h):
    """Maximum overlap for discrete step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Enforce Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_top_m(h, M=5):
    """Gradient of the average of the M worst shifts."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argpartition(corr, -M)[-M:]
    grad = np.zeros(n, dtype=float)
    offset = n - 1
    for j in idx:
        shift = j - offset
        if shift == 0:
            grad += 1.0 - 2.0 * h
        elif shift > 0:
            s = shift
            grad[: n - s] += 1.0 - h[s:]
            grad[s:] -= h[: n - s]
        else:
            s = -shift
            grad[s:] += 1.0 - h[: n - s]
            grad[: n - s] -= h[s:]
    grad /= M
    return grad

def projected_gradient_descent(h0, rng, max_iter, init_step, M=5, deadline=None):
    """Simple projected GD using the top‑M worst‑shift gradient."""
    h = h0.copy()
    step = init_step
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g = grad_top_m(h, M=M)
        h -= step * g
        np.clip(h, 0.0, 1.0, out=h)
        adjust_sum_exact(h, rng)
        step *= 0.995
    return h

def fractional_sa_hill_climb(h0, rng, deadline):
    """Continuous SA that moves mass between two coordinates."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    n = h.size
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        h[src] -= delta
        h[dst] += delta
        c = compute_c5(h)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
                last_improve = it
        else:
            h[src] += delta
            h[dst] -= delta
        it += 1
        if it - last_improve > 20000:
            break
    return best_h, best_c

def binary_sa_hill_climb(h_bin, rng, deadline):

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """Smooth‑max minimisation via SLSQP (log‑sum‑exp surrogate)."""
    n = h0.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = opt.minimize(
        obj, h0, method="SLSQP", bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def upsample_h(h, new_n):
    """Linear interpolation to a finer grid on [0,2)."""
    n = h.size
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xt = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xt, xs, h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    n_levels = [360, 540, 720]

    for n in n_levels:
        if time.time() > deadline:
            break

        # ----- generate starting vectors -----
        starters = []

        if "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
            if parent.size != n:
                p = upsample_h(parent, n)
            else:
                p = parent.copy()
            starters.append(p)

        # alternating binary (exact half)
        alt = np.zeros(n)
        alt[::2] = 1.0
        if n % 2 == 1:
            alt[-1] = 0.5
        adjust_sum_exact(alt, rng)
        starters.append(alt)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starters.append(bin_rand)

        # random fractional uniform
        uf = rng.random(n)
        uf *= (n / 2.0) / uf.sum()
        np.clip(uf, 0.0, 1.0, out=uf)
        adjust_sum_exact(uf, rng)
        starters.append(uf)

        # random fractional squared
        sf = rng.random(n) ** 2
        sf *= (n / 2.0) / sf.sum()
        np.clip(sf, 0.0, 1.0, out=sf)
        adjust_sum_exact(sf, rng)
        starters.append(sf)

        # ----- optimise each start -----
        for h0 in starters:
            if time.time() > deadline:
                break
            h = h0.copy()
            adjust_sum_exact(h, rng)

            # binary SA if already binary
            if np.all((h == 0) | (h == 1)):
                bin_dead = min(deadline, time.time() + (deadline - time.time()) * 0.20)
                h_bin, c_bin = binary_sa_hill_climb(h.astype(float), rng, bin_dead)
                h = h_bin.astype(float)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n

            # projected gradient descent (coarse)
            gd_dead = min(deadline, time.time() + (deadline - time.time()) * 0.15)
            h = projected_gradient_descent(h, rng, max_iter=800, init_step=0.2, M=7,
                                            deadline=gd_dead)

            # fractional SA
            sa_dead = min(deadline, time.time() + (deadline - time.time()) * 0.25)
            h, c = fractional_sa_hill_climb(h, rng, sa_dead)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # SLSQP cascade
            betas = [10, 50, 200, 800, 3000, 10000]
            for beta in betas:
                if time.time() > deadline:
                    break
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=1200)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final projected gradient
            h = projected_gradient_descent(h, rng, max_iter=500, init_step=0.1, M=5,
                                            deadline=deadline)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # ----- optional up‑sampling if time left -----
    if best_h is not None and time.time() < deadline and best_n < 720:
        new_n = min(720, best_n * 2)
        h_up = upsample_h(best_h, new_n)
        adjust_sum_exact(h_up, rng)
        gd_dead = min(deadline, time.time() + (deadline - time.time()) * 0.30)
        h_up = projected_gradient_descent(h_up, rng, max_iter=800, init_step=0.2, M=7,
                                          deadline=gd_dead)
        sa_dead = min(deadline, time.time() + (deadline - time.time()) * 0.30)
        h_up, c_up = fractional_sa_hill_climb(h_up, rng, sa_dead)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, new_n

    # ----- fallback (should never trigger) -----
    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
