def binary_sa_n(n, time_budget, rng):

    start = time.time()

        ones_idx = rng.choice(n, n//2, replace=False)
        h[ones_idx] = 1
        # compute O

        cur_max = O.max()
        # compute padded_h
        pad = np.zeros(n-1, dtype=np.int8)
        padded = np.concatenate([pad, h, pad])
        # maintain lists of one and zero indices for fast sampling
        ones = list(ones_idx)
        zeros = [i for i in range(n) if h[i]==0]  # could compute via set difference
        # initialize temperature schedule parameters
        T0 = 1.0
        # total moves per run maybe n*10? Instead we base on time left.
        iter_start = time.time()
        moves = 0
        while time.time() - iter_start < time_budget * 0.5 and time.time() - start < time_budget:
            # compute temperature based on moves
            T = T0 * (0.9995 ** moves)
            # choose i,j
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            # compute delta

import time, math, numpy as np
from scipy.optimize import minimize, LinearConstraint

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ---------- gradient based local search ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


# ---------- binary simulated annealing ----------
def binary_sa(n: int, time_budget: float, rng: np.random.Generator):
    """Binary (0/1) simulated annealing for C5."""
    best_h = None
    best_max = np.iinfo(np.int16).max
    start = time.time()

    while time.time() - start < time_budget:
        # --- random start ---
        h = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1

        # correlation (int16 to avoid overflow)
        O = np.correlate(h, 1 - h, mode="full").astype(np.int16)
        cur_max = int(O.max())

        # padded view for fast slice access
        pad = np.zeros(n - 1, dtype=np.int8)
        padded = np.concatenate([pad, h, pad])

        inner_start = time.time()
        moves = 0
        T0 = 2.0

        # run inner SA until either local time slice expires or overall budget expires
        while (time.time() - start < time_budget and
               time.time() - inner_start < 0.5 * time_budget):
            T = T0 * (0.99995 ** moves)

            # pick a random 1 and a random 0
            i = rng.integers(n)
            while h[i] == 0:
                i = rng.integers(n)
            j = rng.integers(n)
            while h[j] == 1:
                j = rng.integers(n)

            # slices (promote to int16 before addition)
            ai = padded[i : i + 2 * n - 1].astype(np.int16)
            aj = padded[j : j + 2 * n - 1].astype(np.int16)
            delta = (ai + ai[::-1]) - (aj + aj[::-1])

            new_max = int((O + delta).max())
            accept = (new_max <= cur_max or
                      rng.random() < math.exp(-(new_max - cur_max) / max(T, 1e-12)))

            if accept:
                O = O + delta
                cur_max = new_max
                # swap bits
                h[i] = 0
                h[j] = 1
                padded[i + n - 1] = 0
                padded[j + n - 1] = 1
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
            moves += 1

    # fallback if nothing found
    if best_h is None:
        best_h = np.zeros(n, dtype=np.float64)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)
        return best_h, best_c
    best_c = best_max * (2.0 / n)
    return best_h.astype(np.float64), best_c

        best_h, best_c = h_cur.copy(), c_cur

    # increasing‑β SLSQP passes
    betas = (20., 80., 200., 800., 3000., 8000., 20000., 80000., 200000.)
    for beta in betas:
        if remaining() < 0.03 * budget_s:
            break
        maxit = 600 if beta >= 8000 else 400
        h_cur = slsqp_optimize(best_h, beta, max_iter=maxit)
        c_cur = compute_c5(h_cur)
        if c_cur < best_c:
            best_h, best_c = h_cur.copy(), c_cur

    # final gradient tidy‑up
    h_cur = gradient_refine(best_h, rng, step=0.01, max_iter=15)
    c_cur = compute_c5(h_cur)
    if c_cur < best_c:
        best_h, best_c = h_cur.copy(), c_cur

    # ---- stochastic tweaks until budget expires ----
    while remaining() > 0.01 * budget_s:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
