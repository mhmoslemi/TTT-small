def binary_targeted_improve(h, rng, deadline):
    """Greedy swaps that lower the worst overlap."""
    n = h.size

        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:

        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        neigh_one = neigh[ones]
        min_val = neigh_one.min()
        cand_i = ones[neigh_one == min_val]
        i = rng.choice(cand_i)

        neigh_zero = neigh[zeros]
        max_val = neigh_zero.max()
        cand_j = zeros[neigh_zero == max_val]
        j = rng.choice(cand_j)

        delta = neigh[i] - neigh[j]          # <0 improves worst shift

            if d < 0:
                h[i2] = 0.0

def run(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.95   # margin

    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        # binary from initial: pick largest n/2 entries
        init = np.array(initial_h_values, dtype=float)
        idx = np.argpartition(init, -n//2)[-n//2:]
        h = np.zeros(n, dtype=int)
        h[idx] = 1
    else:
        n = 360
        h = np.zeros(n, dtype=int)
        h[rng.choice(n, n//2, replace=False)] = 1

    # compute initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    curr_c = float(corr.max() * (2.0 / n))
    best_h = h.copy()
    best_c = curr_c

    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))

    right = np.arange(n, dtype=int)
    n2m1 = 2 * n - 1

    # annealing parameters
    T0 = 0.03
    T_min = 0.001
    total_time = deadline - start
    iters = 0
    last_full = 0
    full_recalc_interval = 2000

    while time.time() < deadline:
        elapsed = time.time() - start
        T = max(T_min, T0 * (1 - elapsed / total_time))

        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # delta corr
        d_corr = np.zeros(n2m1, dtype=np.float64)

        # p = i
        p = i
        delta_p = -1
        hp = h[p]  # =1
        idx = right - p + n - 1
        d_corr[idx] += delta_p * (1 - h[right])

        # term2+term3 for right==i
        idx_i = i - p + n - 1  # = n-1
        d_corr[idx_i] += - hp * (-1) - delta_p * (-1)
        # for right==j
        idx_j = j - p + n - 1
        d_corr[idx_j] += - hp * (+1) - delta_p * (+1)

        # p = j
        p = j
        delta_p = +1
        hp = h[p]  # =0
        idx = right - p + n - 1
        d_corr[idx] += delta_p * (1 - h[right])

        # term2+term3 for right==i
        idx_i = i - p + n - 1
        d_corr[idx_i] += - hp * (-1) - delta_p * (-1)
        # right==j
        idx_j = j - p + n - 1
        d_corr[idx_j] += - hp * (+1) - delta_p * (+1)

        # new correlation
        new_corr = corr + d_corr
        new_c = float(new_corr.max() * (2.0 / n))

        # acceptance
        if new_c <= curr_c or rng.random() < math.exp(-(new_c - curr_c) / T):
            # accept
            # swap in h
            h[i] = 0
            h[j] = 1
            # update ones/zeros lists
            # replace i with j in ones
            idx_i_list = ones.index(i)
            ones[idx_i_list] = j
            # replace j with i in zeros
            idx_j_list = zeros.index(j)
            zeros[idx_j_list] = i

            corr = new_corr
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()

        iters += 1
        if iters - last_full >= full_recalc_interval:
            # recompute correlation from scratch for numerical stability
            corr = np.correlate(h, 1 - h, mode='full')
            curr_c = float(corr.max() * (2.0 / n))
            last_full = iters

    # final greedy reduction on best_h (or current h)
    # We'll apply greedy_reduce on best_h to try improvement within remaining time
    h = best_h.copy()
    # greedy reduction
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        s = idx_max - (n - 1)
        if s == 0:
            break
        inds = np.arange(n)
        valid = (inds + s >= 0) & (inds + s < n)
        # i's where h[i]==1 and h[i+s]==0
        i_candidates = np.where(valid & (h == 1) & (h[inds + s] == 0))[0]
        # j's where h[j]==0 and h[j+s]==1
        j_candidates = np.where(valid & (h == 0) & (h[inds + s] == 1))[0]
        if i_candidates.size == 0 or j_candidates.size == 0:
            break
        i_sel = rng.choice(i_candidates)
        j_sel = rng.choice(j_candidates)
        h[i_sel] = 0
        h[j_sel] = 1
        # continue loop

    final_c = float(np.correlate(h, 1 - h, mode='full').max() * (2.0 / n))

    # Ensure constraints
    assert h.sum() == n // 2
    assert np.all((h == 0) | (h == 1))

    return h.tolist(), final_c, int(n)

def compute_c5(h):

def greedy_reduce(h, rng, deadline):
    n = h.size
    ar = np.arange(n)
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        s = idx_max - (n - 1)
        if s == 0:

            break
        i = rng.choice(i_cand)
        j = rng.choice(j_cand)

import time
import math
import numpy as np


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a binary step function."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def greedy_reduce(h: np.ndarray, rng: np.random.Generator, deadline: float) -> np.ndarray:
    """Deterministic swaps that lower the current worst shift."""
    n = h.size
    ar = np.arange(n, dtype=int)
    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = int(np.argmax(corr))
        s = idx_max - (n - 1)
        if s == 0:
            break

        valid = (ar + s >= 0) & (ar + s < n)
        if not np.any(valid):
            break
        idxs = ar[valid]
        h_valid = h[idxs]
        h_shift = h[idxs + s]

        i_cand = idxs[(h_valid == 1) & (h_shift == 0)]
        j_cand = idxs[(h_valid == 0) & (h_shift == 1)]
        if i_cand.size == 0 or j_cand.size == 0:
            break

        i = int(rng.choice(i_cand))
        j = int(rng.choice(j_cand))
        h[i] = 0
        h[j] = 1
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.95  # keep a small safety margin

    # ----- initial binary vector -----

        init = np.array(initial_h_values, dtype=float)

        h[idx] = 1

    # ----- simulated annealing -----
