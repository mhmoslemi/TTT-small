def simulated_annealing_binary(h0, start_temp, end_temp, max_steps, rng, deadline):
    n = h0.size
    h = h0.copy()
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    curr_max = max_corr_bin(h)
    curr_c5 = curr_max * (2.0 / n)
    best_h = h.copy()
    best_c5 = curr_c5
    best_max = curr_max
    step = 0
    while step < max_steps and time.time() < deadline:
        T = start_temp + (end_temp - start_temp) * (step / max_steps)
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_max = max_corr_bin(h)
        new_c5 = new_max * (2.0 / n)
        delta = new_c5 - curr_c5
        accept = False
        if delta <= 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            accept = True
        if accept:
            curr_c5 = new_c5
            curr_max = new_max
            # update lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
                best_max = curr_max
        else:
            # revert
            h[i] = 1
            h[j] = 0
        step += 1
    return best_h, best_c5
```

def slsqp_cascade(h0, betas, rng, deadline):
    h = h0.copy()
    adjust_sum_exact(h, rng)
    n = h.size
    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        CURRENT_BETA = beta
        max_iter = 400 if beta <= 200 else 800 if beta <= 5000 else 1200
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
        try:

                softmax_obj,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        h = np.clip(res.x, 0.0, 1.0)
        adjust_sum_exact(h, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

import time
import numpy as np
import math
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr[n - 1] = 0.0
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def softmax_obj(x):
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    corr[n - 1] = 0.0
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def max_corr_bin(h):
    n = h.size
    corr = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode='full')
    corr[n - 1] = 0
    return corr.max()

def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start:start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode='full')
        O[n - 1] = 0
    return h, O

def simulated_annealing_binary(h0, start_temp, end_temp, max_steps, rng, deadline):
    n = h0.size
    h = h0.copy()
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    if not ones or not zeros:
        return h.copy(), compute_c5(h.astype(float))
    curr_max = max_corr_bin(h)
    curr_c5 = curr_max * (2.0 / n)
    best_h = h.copy()
    best_c5 = curr_c5
    step = 0
    while step < max_steps and time.time() < deadline:
        T = start_temp + (end_temp - start_temp) * (step / max_steps)
        # pick random 1 and 0
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_max = max_corr_bin(h)
        new_c5 = new_max * (2.0 / n)
        delta = new_c5 - curr_c5
        accept = False
        if delta <= 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            accept = True
        if accept:
            curr_c5 = new_c5
            curr_max = new_max
            # update lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        step += 1
    return best_h, best_c5

def slsqp_cascade(h0, betas, rng, deadline):
    global CURRENT_TARGET, CURRENT_BETA
    h = h0.copy()
    adjust_sum_exact(h, rng)
    n = h.size
    CURRENT_TARGET = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        CURRENT_BETA = beta
        if beta <= 200:
            max_iter = 400
        elif beta <= 5000:
            max_iter = 800
        else:
            max_iter = 1200
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
        try:
            res = minimize(
                softmax_obj,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        h = np.clip(res.x, 0.0, 1.0)
        adjust_sum_exact(h, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

import time
import numpy as np
import math
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr[n - 1] = 0.0
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


def softmax_obj(x):
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    corr[n - 1] = 0.0
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def max_corr_bin(h):
    n = h.size
    corr = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
    corr[n - 1] = 0
    return corr.max()


def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start:start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h, O


def simulated_annealing_binary(h0, start_temp, end_temp, rng, deadline):
    n = h0.size
    h = h0.copy()
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))
    if not ones or not zeros:
        return h.copy(), compute_c5(h.astype(float))
    curr_max = max_corr_bin(h)
    curr_c5 = curr_max * (2.0 / n)
    best_h = h.copy()
    best_c5 = curr_c5
    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        fraction = elapsed / (deadline - start_time)
        T = start_temp + (end_temp - start_temp) * fraction
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        new_max = max_corr_bin(h)
        new_c5 = new_max * (2.0 / n)
        delta = new_c5 - curr_c5
        accept = False
        if delta <= 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            accept = True
        if accept:
            curr_c5 = new_c5
            curr_max = new_max
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c5


def slsqp_cascade(h0, betas, rng, deadline):
    global CURRENT_TARGET, CURRENT_BETA
    h = h0.copy()
    adjust_sum_exact(h, rng)
    n = h.size
    CURRENT_TARGET = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        CURRENT_BETA = beta
        if beta <= 200:
            max_iter = 400
        elif beta <= 5000:
            max_iter = 800
        else:
            max_iter = 1200
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
        try:
            res = minimize(
                softmax_obj,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        h = np.clip(res.x, 0.0, 1.0)
        adjust_sum_exact(h, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
