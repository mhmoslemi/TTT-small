def project_onto_bounds_and_sum(v, target):
    # Solve for lambda such that sum_i min(1, max(0, v_i - lambda)) = target
    # Equivalent to find lambda where after shifting down by lambda, clip to [0,1], sum equals target.
    # This is monotonic decreasing in lambda: larger lambda => smaller sum.
    # Use bisection with bounds: lambda_low = min(v)-1 (so that after shift v_i - lambda_low >= 1 for all? Actually if lambda low very negative, then v_i - lambda >> 1 => clips to 1, sum = n.)
    # lambda_high = max(v) (so v_i - lambda <=0 for all => sum=0).
    # Use binary search for tolerance 1e-12.
    # Ensure target within [0,n].
    l = np.min(v) - 1.0
    r = np.max(v)
    for _ in range(60):
        mid = (l + r) / 2
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            l = mid
        else:
            r = mid
    h = np.clip(v - (l + r) / 2, 0.0, 1.0)
    # after projection, due to rounding sum may be slightly off; adjust small diff
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly across interior (not at bounds) indices
        # find indices where 0 < h_i < 1
        interior = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if interior.size > 0:
            # distribute equally
            add = diff / interior.size
            h[interior] += add
            # clip again
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: adjust randomly
            for _ in range(1000):
                i = np.random.randint(n)
                if diff > 0 and h[i] < 1:
                    inc = min(diff, 1 - h[i])
                    h[i] += inc
                    diff -= inc
                elif diff < 0 and h[i] > 0:
                    dec = min(-diff, h[i])
                    h[i] -= dec
                    diff += dec
                if abs(diff) < 1e-12:
                    break
    return h

def pgd_optimize(h0, beta, num_iters, step_size):
    h = h0.copy()
    for it in range(num_iters):
        y = np.correlate(h, 1-h, mode='full')
        # compute softmax weights
        maxy = np.max(beta*y)
        w = np.exp(beta*y - maxy)
        w = w / w.sum()

        term1_full = np.convolve(w, 1-h)
        term2_full = np.correlate(w, h, mode='full')

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")

    """Gradient of the worst shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer a small amount from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def pgd_optimize(h0: np.ndarray, beta: float, iters: int, step: float, rng: np.random.Generator) -> np.ndarray:
    """Projected gradient descent on the soft‑max approximation of max overlap."""
    h = h0.copy()
    v = np.zeros_like(h)
    mu = 0.9
    target = h.size / 2.0
    offset = h.size - 1
    for _ in range(iters):
        corr = np.correlate(h, 1 - h, mode="full")
        # soft‑max weights
        max_bc = np.max(beta * corr)
        w = np.exp(beta * corr - max_bc)
        w /= w.sum()
        term1_full = np.convolve(w, 1 - h)
        term2_full = np.correlate(w, h, mode="full")
        term1 = term1_full[offset:offset + h.size]
        term2 = term2_full[offset:offset + h.size]
        grad = term1 - term2
        v = mu * v + grad
        h = h - step * v
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(10):
                i = rng.integers(0, h.size)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                if add != 0:
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
    return h


def upsample_and_refine(h: np.ndarray, rng: np.random.Generator, time_limit: float) -> np.ndarray:
    """If time permits, double the resolution and run a short PGD refinement."""
    n = h.size
    if n * 2 > 1024 or time_limit < 5.0:
        return h
    n_up = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    np.clip(h_up, 0.0, 1.0, out=h_up)
    target_up = n_up / 2.0
    enforce_sum_inplace(h_up, target_up, rng)

    beta_seq = [20.0, 200.0, 2000.0, 8000.0]
    step_factor = 0.5
    for beta in beta_seq:
        step = step_factor / beta
        # allocate a modest number of iterations per beta
        iters = 1200
        h_up = pgd_optimize(h_up, beta, iters, step, rng)
    enforce_sum_inplace(h_up, target_up, rng)
    return h_up


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    elapsed = lambda: time.time() - start

    # ----- initialise ----- #
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 512
        h0 = rng.random(n)
    np.clip(h0, 0.0, 1.0, out=h0)
    enforce_sum_inplace(h0, n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- soft‑max PGD with increasing β ----- #
    beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0]
    for beta in beta_schedule:
        if elapsed() > budget_s * 0.80:
            break
        iters = 1800                     # fixed budget per β (fast enough)
        step = 0.5 / beta
        h_new = pgd_optimize(best_h, beta, iters, step, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # ----- guided hill‑climb on the worst shift ----- #
    remaining = budget_s - elapsed()
    max_iters = int(min(5000, remaining * 2000))
    h = best_h.copy()
    for it in range(max_iters):
        grad, _ = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=0.5)
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c = nh.copy(), c
            h = nh
        else:
            h = nh if rng.random() < 0.01 else h  # occasional acceptance

        if (it + 1) % 200 == 0 and elapsed() > budget_s * 0.98:
            break

    # ----- optional up‑sampling and refinement ----- #
    if elapsed() < budget_s * 0.99:
        h_up = upsample_and_refine(best_h, rng, budget_s - elapsed())
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # ----- final sanitisation ----- #
    np.clip(best_h, 0.0, 1.0, out=best_h)
    enforce_sum_inplace(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
