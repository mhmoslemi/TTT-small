def binary_sa(n, rng, time_budget):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0 = 0.05
    T_end = 1e-4
    while time.time() - start < time_budget:
        # random binary start
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n//2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5_binary(h)
        # maybe limit inner loops by time
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end/T0) ** (elapsed / time_budget)
            # propose swap
            i = rng.integers(n)
            while h[i] == 0:
                i = rng.integers(n)
            j = rng.integers(n)
            while h[j] == 1:
                j = rng.integers(n)
            # swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5_binary(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                # revert
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        # fallback: deterministic half ones contiguous
        best_h = np.zeros(n)
        best_h[:n//2] = 1.0
        best_c = compute_c5_binary(best_h)
    return best_h, best_c

import time

from scipy.optimize import minimize

# ---------- global for soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for any step function."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def compute_c5_binary(h: np.ndarray) -> float:
    """C5 for binary vector using overlap of ones."""
    n = h.size
    overlap = np.correlate(h, h, mode="full")
    overlap = overlap[n - 1 :]  # shift k = 0..n-1
    O = (n / 2) - overlap
    Omax = O.max()
    return Omax * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
