def compute_c5_and_shift(h):
    n = h.size
    L = 2*n
    hf = np.fft.rfft(h, L)
    bf = np.fft.rfft(1 - h, L)
    conv = np.fft.irfft(hf * bf, L)[:2*n-1]
    max_idx = np.argmax(conv)
    max_val = conv[max_idx] * (2.0 / n)
    shift = max_idx - (n - 1)
    return max_val, shift, conv

def simulated_annealing(h, rng, max_iters, start_temp, end_temp, time_limit, start_time):
    n = h.size
    best_h = h.copy()
    best_c, _, _ = compute_c5_and_shift(h)
    T0 = start_temp
    Tf = end_temp
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # temperature schedule: exponential
        T = T0 * ((Tf/T0) ** (i / max_iters))
        # propose swap
        i1 = rng.integers(n)
        i2 = rng.integers(n-1)
        if i2 >= i1:
            i2 += 1
        if h[i1] == h[i2]:
            continue
        # swap
        h[i1], h[i2] = h[i2], h[i1]
        c, _, _ = compute_c5_and_shift(h)
        delta = c - best_c
        # if improve accept and update best
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # accept with metropolis
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # keep new state
            pass
        else:
            # revert swap
            h[i1], h[i2] = h[i2], h[i1]

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    dx = 2.0 / n

    def compute_c5_and_shift(h):
        L = 2 * n
        hf = np.fft.rfft(h.astype(np.float64), L)
        bf = np.fft.rfft((1 - h).astype(np.float64), L)
        conv = np.fft.irfft(hf * bf, L)[:2 * n - 1]
        idx = np.argmax(conv)
        max_val = conv[idx] * dx
        shift = idx - (n - 1)
        return max_val, shift, conv

    def repair_worst_shift(h, max_iters=2000):
        # will also check time
        for it in range(max_iters):
            if time.time() - start > budget_s * 0.95:
                break
            max_val, shift, _ = compute_c5_and_shift(h)
            if max_val == 0:
                break
            k = shift
            if k >= 0:
                seg1 = h[:n - k]
                seg2 = h[k:]
            else:
                kk = -k
                seg1 = h[kk:]
                seg2 = h[:n - kk]
            mis1 = np.where((seg1 == 1) & (seg2 == 0))[0]
            mis0 = np.where((seg1 == 0) & (seg2 == 1))[0]
            if mis1.size and mis0.size:
                i_rel = mis1[rng.integers(mis1.size)]
                j_rel = mis0[rng.integers(mis0.size)]
                if k >= 0:
                    i = i_rel
                    j = j_rel + k
                else:
                    i = i_rel + kk
                    j = j_rel
                # swap
                h[i] = 0
                h[j] = 1
            else:
                break
        return h

    # generate start candidates
    start_candidates = []

    # if initial vector exists, binarize
    if 'initial_h_values' in globals():
        init_arr = np.array(initial_h_values, dtype=np.float64)

    # A random initial (generated later per iteration)

def slsqp_optimize(h0, beta, max_iter):
    n = h0.size
    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        hf = np.fft.rfft(x, 2*n)
        bf = np.fft.rfft(1 - x, 2*n)
        conv = np.fft.irfft(hf * bf, 2*n)[:2*n-1]
        # softmax using beta
        m = conv.max()
        # compute log-sum-exp stable
        lse = m + np.log(np.sum(np.exp(beta*(conv - m)))) / beta
        # Wait: original softmax objective returns (m + log sum exp(beta*(c - m))) / beta
        # Actually they compute bc = beta*c; then m = max(bc); return (m + log sum exp(bc - m)) / beta
        # So we do:
        bc = beta * conv
        mb = bc.max()
        return (mb + np.log(np.sum(np.exp(bc - mb)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly (may be slight deviation)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust by distributing small amounts to random indices not at bounds
        for _ in range(1000):
            i = rng.integers(0, n)
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                add = max(diff, -x[i])
                x[i] += add
                diff -= add
            if abs(diff) < 1e-12:
                break
    return x

    dx = 2.0 / n

    def compute_c5_and_shift(h):
        L = 2*n
        hf = np.fft.rfft(h.astype(np.float64), L)
        bf = np.fft.rfft((1.0 - h).astype(np.float64), L)
        conv = np.fft.irfft(hf * bf, L)[:2*n - 1]

        max_val = float(conv[idx] * dx)
        shift = idx - (n - 1)
        return max_val, shift, conv

    h0: np.ndarray,
    beta: float,
    rng: np.random.Generator,
    n: int,
    max_iter: int = 200,
):

    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def softmax_obj(x):

def simulated_annealing(
    h: np.ndarray,
    rng: np.random.Generator,
    n: int,
    start_time: float,
    budget_s: float,
):
    """Metropolis SA on binary vectors (swap‑based)."""
    cur_h = h.copy()
    cur_c, _, _ = compute_c5_and_shift(cur_h, n)
    best_h = cur_h.copy()
    best_c = cur_c
    total_steps = 200_000
    T0, Tf = 5.0, 0.01

    for step in range(total_steps):
        if time.time() - start_time > budget_s * 0.9:
            break
        T = T0 * ((Tf / T0) ** (step / total_steps))
        i1 = rng.integers(n)
        i2 = rng.integers(n - 1)
        if i2 >= i1:
            i2 += 1
        if cur_h[i1] == cur_h[i2]:
            continue
        cur_h[i1], cur_h[i2] = cur_h[i2], cur_h[i1]
        new_c, _, _ = compute_c5_and_shift(cur_h, n)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < math.exp(-delta / T):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i1], cur_h[i2] = cur_h[i2], cur_h[i1]
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    best_h = None
    best_c = np.inf

    # ---- initial candidates -------------------------------------------------
    candidates = []

    if "initial_h_values" in globals():
        init_arr = np.array(initial_h_values, dtype=np.float64)
        idx_desc = np.argsort(-init_arr)
        bin_arr = np.zeros_like(init_arr, dtype=np.uint8)
        bin_arr[idx_desc[: n // 2]] = 1
        candidates.append(bin_arr)

    # Thue‑Morse (parity of popcount)
    tm = np.empty(n, dtype=np.uint8)
    for i in range(n):
        tm[i] = i.bit_count() & 1
    diff = (n // 2) - tm.sum()
    if diff > 0:
        tm[np.where(tm == 0)[0][:diff]] = 1
    elif diff < 0:
        tm[np.where(tm == 1)[0][: -diff]] = 0
    candidates.append(tm)

    for cand in candidates:
        cval, _, _ = compute_c5_and_shift(cand, n)
        if cval < best_c:
            best_c = cval
            best_h = cand.copy()

    # ---- random‑restart SA + repair ----------------------------------------
    while time.time() - start < budget_s * 0.9:
        # random binary start
        h = np.zeros(n, dtype=np.uint8)
        h[rng.choice(n, n // 2, replace=False)] = 1

        # quick deterministic repair
        h = repair_worst_shift(h, rng, start, budget_s, max_iters=500)

        cur_c, _, _ = compute_c5_and_shift(h, n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        # simulated annealing
        h_sa, c_sa = simulated_annealing(h, rng, n, start, budget_s)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()

        # final repair for this run
        h_sa = repair_worst_shift(h_sa, rng, start, budget_s, max_iters=200)
        c_fin, _, _ = compute_c5_and_shift(h_sa, n)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_sa.copy()

    # ---- optional continuous refinement ------------------------------------
    if time.time() - start < budget_s * 0.98:
        h_cont = best_h.astype(np.float64)
        for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0):
            if time.time() - start > budget_s * 0.999:
                break
            h_cont = slsqp_optimize(h_cont, beta, rng, n, max_iter=200)
        final_c, _, _ = compute_c5_and_shift(h_cont, n)
        best_h = h_cont
        best_c = final_c

    # final sanity check
    final_c, _, _ = compute_c5_and_shift(best_h, n)
    return best_h.astype(float).tolist(), float(final_c), int(best_h.size)
