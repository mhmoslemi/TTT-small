import time, numpy as np
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # move mass from random indices to others until diff satisfied
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-5)
    return best_h, best_c

def gradient_refine(h, rng, step=0.015, max_iter=10):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def thue_morse(n):
    # n must be power of 2
    a = np.array([0], dtype=float)
    while a.size < n:
        a = np.concatenate([a, 1 - a])
    return a[:n]

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine initial h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h,0.0,1.0)
        # enforce sum precisely
        h = enforce_sum(h, rng)
        n = h.size
    else:
        n = 512
        # Use Thue-Morse binary pattern
        h = thue_morse(n)
        # Add small random perturbation to break symmetry
        perturb = rng.random(n) * 0.01
        h = np.clip(h + perturb, 0.0, 1.0)
        h = enforce_sum(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Annealing parameters
    T0 = best_c * 0.5 + 0.01  # start temperature a bit larger
    T_end = best_c * 1e-5 + 1e-7
    # time allocation
    max_time = budget_s * 0.95
    it = 0
    cur_c = best_c
    cur_h = h.copy()
    while time.time() - start < max_time:
        # progress fraction based on time elapsed
        elapsed = time.time() - start
        prog = elapsed / max_time
        T = T0 * (T_end / T0) ** prog
        # pick i, j random
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * (T / T0)
        if eps <= 0:
            continue
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        new_c = compute_c5(nh)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_h, cur_c = nh, new_c
            if new_c < best_c:
                best_c, best_h = new_c, nh.copy()
        it += 1

    # final local refinement
    best_h = gradient_refine(best_h, rng, step=0.01, max_iter=15)
    best_c = compute_c5(best_h)
    # further small hill climb
    best_h, best_c = guided_hill_climb(best_h, rng, max_iters=2000, step_scale=0.4)

    # ensure constraints
    best_h = np.clip(best_h,0.0,1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple loop
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-5)
    return best_h, best_c

def gradient_refine(h, rng, step=0.015, max_iter=10):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def thue_morse(n):
    a = np.array([0.0])
    while a.size < n:
        a = np.concatenate([a, 1.0 - a])
    return a[:n]

        if eps <= 0:
            continue
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        new_c = compute_c5(nh)
        delta = new_c - cur_c

            if new_c < best_c:

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap O_k for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σ h = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of O_k for the shift attaining the current maximum."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:

    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using the worst‑shift gradient."""
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-5)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def thue_morse(n: int) -> np.ndarray:
    """Binary Thue–Morse sequence of length n (n a power of two)."""
    a = np.array([0.0])
    while a.size < n:
        a = np.concatenate([a, 1.0 - a])
    return a[:n]

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
        n = h.size
    else:
        n = 512
        h = thue_morse(n)
        h += rng.random(n) * 0.01          # tiny break of symmetry
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- simulated annealing -----------------------------------------
    T0 = best_c * 0.5 + 0.01                # initial temperature
    T_end = best_c * 1e-5 + 1e-7
    max_time = budget_s * 0.95
    cur_h = h.copy()
    cur_c = best_c

    while time.time() - start < max_time:
        elapsed = time.time() - start
        prog = elapsed / max_time
        T = T0 * (T_end / T0) ** prog      # exponential cooling

        # 10 % of the time use a pure random transfer for diversity
        if rng.random() < 0.1:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * (T / T0)
            if eps <= 0.0:
                continue
            nh = cur_h.copy()
            nh[i] -= eps
            nh[j] += eps
        else:
            grad = worst_gradient(cur_h)
            nh = guided_perturb(cur_h, grad, rng, scale=T / T0)
            if np.array_equal(nh, cur_h):
                continue

        new_c = compute_c5(nh)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_h, cur_c = nh, new_c
            if new_c < best_c:
                best_c, best_h = new_c, nh.copy()

    # ---- final polishing -----------------------------------------------
    best_h = gradient_refine(best_h, rng, step=0.01, max_iter=15)
    best_h, best_c = guided_hill_climb(best_h, rng,
                                       max_iters=3000, step_scale=0.4)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
