import time, numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _project_sum_box(h: np.ndarray, target: float) -> np.ndarray:
    """Projection of h onto [0,1] hyperplane sum=h_target via binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def enforce_sum(h: np.ndarray, rng=None) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    return _project_sum_box(h, n * 0.5)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    h *= (n * 0.5) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- soft‑max objective ----------
def _softmax_weights(h: np.ndarray, beta: float):
    corr = np.correlate(h, 1 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / exp_bc.sum()
    return w, corr


def _softmax_gradient_fft(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of log‑sum‑exp/β using FFT (vectorised, O(n log n))."""
    n = h.size
    L = 2 * n - 1
    L_fft = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, n=L_fft)
    W = np.fft.rfft(w, n=L_fft)
    conv_wh = np.fft.irfft(H * W, n=L_fft)[:L]
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=L_fft)
    conv_wrev_h = np.fft.irfft(H * W_rev, n=L_fft)[:L]
    start = n - 1
    A = conv_wh[start:start + n]          # Σ w_s h_{i‑s}
    B = conv_wrev_h[start:start + n]      # Σ w_s h_{i+s}
    grad = 1.0 - B - A
    return grad


def gradient_descent_softmax(h0: np.ndarray, rng: np.random.Generator,
                            beta_schedule, iters_per_beta: int = 200,
                            step0: float = 0.8):
    """Smooth‑max minimisation with increasing β."""
    h = h0.copy()
    n = h.size
    target = n * 0.5
    best_h = h.copy()
    best_c = compute_c5(h)
    c_curr = best_c
    for beta in beta_schedule:
        step = step0 / np.sqrt(beta)
        for _ in range(iters_per_beta):
            w, _ = _softmax_weights(h, beta)
            grad = _softmax_gradient_fft(h, w)
            h_new = h - step * grad
            h_new = enforce_sum(h_new, rng)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
            if c_new <= c_curr:
                h, c_curr = h_new, c_new
    return best_h, best_c


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # start from parent if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c = h0.copy(), compute_c5(h0)
    else:
        best_h = None
        best_c = np.inf

    # ensure a baseline solution exists
    n_main = 256
    if best_h is None:
        best_h = random_binary_vector(n_main, rng)
        best_c = compute_c5(best_h)

    # ----- multi‑restart soft‑max + hill‑climb -----
    beta_sched = [5, 20, 100, 500, 2000, 8000, 30000, 80000]
    while time.time() - start < 0.60 * budget_s:
        h0 = random_binary_vector(n_main, rng) if rng.random() < 0.5 else random_feasible_vector(n_main, rng)
        h_opt, c_opt = gradient_descent_softmax(h0, rng, beta_sched, iters_per_beta=200, step0=0.8)
        h_opt, c_opt = guided_hill_climb(h_opt, rng, max_iters=1500, step_scale=0.6)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # ----- up‑sample to 512 and refine -----
    if time.time() - start < 0.85 * budget_s:
        target_n = 512
        if best_h.size != target_n:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
        else:
            h_up = best_h.copy()
        h_up = enforce_sum(h_up, rng)
        up_beta = [20, 200, 2000, 20000, 80000]
        h_up, c_up = gradient_descent_softmax(h_up, rng, up_beta, iters_per_beta=300, step0=0.5)
        h_up, c_up = guided_hill_climb(h_up, rng, max_iters=2000, step_scale=0.5)
        if c_up < best_c:
            best_h, best_c = h_up.copy(), c_up

    # ----- final stochastic tweaks -----
    while time.time() - start < budget_s * 0.99:
        i = rng.integers(0, best_h.size)
        j = rng.integers(0, best_h.size)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = enforce_sum(nh, rng)
        c_new = compute_c5(nh)
        if c_new < best_c:
            best_h, best_c = nh, c_new

    # final sanity
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
