def smooth_min_obj_and_grad(h, beta, gamma):
    n = h.shape[0]
    target = n * 0.5
    sum_h = np.sum(h)
    diff = sum_h - target
    # s correlation
    s = np.convolve(h, h, mode='full')
    # stable compute
    min_s = np.min(s)
    # compute shifted exponents
    exp_term = np.exp(-beta * (s - min_s))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    # objective
    f = -min_s + (np.log(sum_exp) / beta) + gamma * diff * diff
    # w_sym
    w_sym = w + w[::-1]
    # gradient of smooth min: -conv(w_sym, h)
    conv_full = np.convolve(w_sym, h, mode='full')
    grad = -conv_full[n-1:n-1+n]  # slice length n
    # penalty gradient
    grad += 2 * gamma * diff
    return f, grad

def smooth_min_obj_and_grad(h, beta, gamma):
    n = h.shape[0]
    target = n * 0.5
    sum_h = np.sum(h)
    diff = sum_h - target
    # correlation h with h
    s = np.convolve(h, h, mode='full')
    min_s = np.min(s)
    exp_term = np.exp(-beta * (s - min_s))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    # objective = -min_s + (log sum_exp)/beta + gamma * diff^2
    f = -min_s + np.log(sum_exp) / beta + gamma * diff * diff
    # compute gradient: -(conv(w_sym, h)) + 2*gamma*diff
    w_sym = w + w[::-1]
    conv_full = np.convolve(w_sym, h, mode='full')
    start = 2 * n - 2
    grad = -conv_full[start:start + n]
    grad += 2 * gamma * diff
    return f, grad

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop
    it = 0
    while abs(diff) > 1e-12 and it < 2000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ---------- smooth‑min objective ----------
def smooth_min_obj_and_grad(h: np.ndarray, beta: float, gamma: float):
    """Smooth‑min of overlaps (max of c_k) with quadratic sum‑penalty."""
    n = h.shape[0]
    target = n * 0.5
    sum_h = np.sum(h)
    diff = sum_h - target

    # correlation h with itself
    s = np.convolve(h, h, mode='full')
    min_s = np.min(s)

    # stable soft‑min weights
    exp_term = np.exp(-beta * (s - min_s))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp

    # objective value (smooth‑min of -s_k) + penalty
    f = -min_s + np.log(sum_exp) / beta + gamma * diff * diff

    # gradient: -(conv(w_sym, h)) + 2*gamma*diff
    w_sym = w + w[::-1]                     # length 2n‑1
    conv_full = np.convolve(w_sym, h, mode='full')
    start = 2 * n - 2                        # index aligning to shift 0
    grad = -conv_full[start:start + n]
    grad += 2.0 * gamma * diff
    return f, grad


def opt_l_bfgs(h0: np.ndarray, beta: float, gamma: float,
               maxiter: int = 300) -> np.ndarray:
    """L‑BFGS‑B on the smoothed objective (bounds [0,1])."""
    n = h0.shape[0]
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        f, g = smooth_min_obj_and_grad(x, beta, gamma)
        return f, g

    res = minimize(fun, h0, method='L-BFGS-B', jac=True,
                   bounds=bounds,
                   options={'maxiter': maxiter,
                            'ftol': 1e-12,
                            'disp': False})
    h_opt = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum after optimization
    h_opt = enforce_sum(h_opt, np.random.default_rng())
    return h_opt
