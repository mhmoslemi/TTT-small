def gradient_smooth_max(h,beta):
    n = h.size
    a = 1.0 - h
    c = np.correlate(h, a, mode='full')
    bc = beta * c
    m = np.max(bc)
    Z = np.exp(bc - m)
    Z_sum = np.sum(Z)
    w = Z / Z_sum
    # w corresponds to indices - (n-1) .. (n-1)
    # Compute grad = sum_t w[t] * (a shifted by t) - sum_t w[t] * (h shifted by -t)
    grad = np.zeros_like(h, dtype=float)
    # Loop over t
    # shift forward t>=0: a segment
    for t in range(-(n-1), n):
        wt = w[t + (n-1)]
        if wt == 0:
            continue
        if t >= 0:
            # a[i+t] for i = 0..n-1-t
            grad[:n-t] += wt * a[t:]
            # -h[i-t] for i=t..n-1 i.e. i-t>=0 => i>=t
            grad[t:] -= wt * h[:n-t]
        else:
            # t negative
            tp = -t
            # a[i - tp] for i >= tp -> i - tp >= 0
            grad[tp:] += wt * a[:n-tp]
            # -h[i + tp] for i <= n-1 - tp
            grad[:n-tp] -= wt * h[tp:]
    return grad

def gradient_smooth_max_fft(h,beta):
    n = h.size
    a = 1.0 - h
    c = np.correlate(h, a, mode='full')
    bc = beta * c
    m = np.max(bc)
    Z = np.exp(bc - m)
    w = Z / Z.sum()
    w_rev = w[::-1]
    # convolve
    wa = np.convolve(w_rev, a, mode='full')
    wh = np.convolve(w, h, mode='full')
    offset = n - 1
    grad = wa[offset:offset + n] - wh[offset:offset + n]
    return grad

Fractional SA code:

def fractional_sa(h, rng, deadline):
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(len(cur), size=2, replace=False)
        # transfer from i to j or j to i randomly
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        # ensure sum unchanged; after delta, sum unchanged.
        c = compute_c5(cur)
        T = init_T * (0.9995**it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            # revert
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 20000:
            break
    return best, best_c

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_sum_bounds(x, target, rng=None):
    """
    Project x onto the box [0,1] and the hyperplane sum(x)=target.
    Uses a bisection on the Lagrange multiplier lambda for the affine shift.
    """
    lo, hi = -2.0, 2.0  # wide enough for any feasible shift
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(x + mid, 0.0, 1.0)
        s = z.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    z = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-12:
        # try to fix tiny residuals
        if rng is None:
            rng = np.random.default_rng()
        free = np.where((z > 0.0) & (z < 1.0))[0]
        if free.size:
            add = diff / free.size
            z[free] = np.clip(z[free] + add, 0.0, 1.0)
        diff = target - z.sum()
        for _ in range(200):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, len(z))
            delta = min(diff, 1.0 - z[i]) if diff > 0 else max(diff, -z[i])
            z[i] += delta
            diff = target - z.sum()
    return z

def smooth_max_obj(h, beta):
    """Smooth approximation of the max overlap via log‑sum‑exp."""
    c = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * c
    m = np.max(bc)
    z = np.exp(bc - m)
    return (m + np.log(z.sum())) / beta

def gradient_smooth_max_fft(h, beta):
    """
    Gradient of the smooth max objective.
    Uses linear convolutions (via np.convolve) for efficiency.
    """
    n = h.size
    a = 1.0 - h
    c = np.correlate(h, a, mode="full")
    bc = beta * c
    m = np.max(bc)
    w = np.exp(bc - m)
    w /= w.sum()
    w_rev = w[::-1]
    wa = np.convolve(w_rev, a, mode="full")
    wh = np.convolve(w, h, mode="full")
    off = n - 1
    grad = wa[off:off + n] - wh[off:off + n]
    return grad

def projected_gradient_descent(h, beta, target, max_iters=80, init_step=0.5, rng=None):
    """Projected gradient descent on the smooth max surrogate."""
    if rng is None:
        rng = np.random.default_rng()
    step = init_step
    for it in range(max_iters):
        grad = gradient_smooth_max_fft(h, beta)
        # simple decreasing step schedule
        h_new = project_sum_bounds(h - step * grad, target, rng=rng)
        # optional acceptance test (skip for speed)
        h = h_new
        step *= 0.97
    return h

def random_feasible(rng, n):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    return project_sum_bounds(rng.random(n), n / 2.0, rng=rng)

def fractional_sa(h, rng, deadline):
    """Transfer‑based simulated annealing on continuous vectors."""
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_imp = 0
    n = len(cur)
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_imp = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_imp > 20000:
            break
    return best, best_c

def gradient_refine(h, target, rng, steps=8, step_sz=0.02):
    """Projected gradient descent against the worst shift."""
    n = h.size
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        elif k > 0:
            g[:n - k] += 1.0 - h[k:]
            g[k:] -= h[:n - k]
        else:
            kk = -k
            g[kk:] += 1.0 - h[:n - kk]
            g[:n - kk] -= h[kk:]
        h = h - step_sz * g
        h = project_sum_bounds(h, target, rng=rng)
    return h
