def compute_slopes(i, j, h):
    n = h.size
    s = np.arange(1, n)  # shape (n-1,)
    left_i = np.where(s <= i, h[i - s], 0.0)
    right_i = np.where(s <= n - 1 - i, h[i + s], 0.0)
    left_j = np.where(s <= j, h[j - s], 0.0)
    right_j = np.where(s <= n - 1 - j, h[j + s], 0.0)
    slopes = left_i + right_i - left_j - right_j
    # slopes for shift 0 linear term:
    slope0_lin = 2 * (h[i] - h[j])
    return slope0_lin, slopes
```

import time, numpy as np, scipy.optimize as opt

def enforce_sum(h, rng):
    """Adjust h so that sum(h)=len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_O_raw(h):
    """Correlation O_raw = np.correlate(h,1-h,mode='full') (no dx)."""
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    """Discrete C5 with correct scaling."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

    right_i = np.where(s <= n - 1 - i, h[i + s], 0.0)
    left_j = np.where(s <= j, h[j - s], 0.0)
    right_j = np.where(s <= n - 1 - j, h[j + s], 0.0)
    slopes = left_i + right_i - left_j - right_j
    slope0_lin = 2.0 * (h[i] - h[j])          # linear term for O0
    return slope0_lin, slopes

def build_slope_raw(slope0_lin, slopes):
    """Create full‑length slope array (both signs) for O_raw."""
    n = slopes.size + 1
    m = 2 * n - 1
    sr = np.empty(m, dtype=float)
    mid = n - 1
    sr[mid] = slope0_lin
    sr[mid + 1 :] = slopes
    sr[:mid] = slopes[::-1]
    return sr

def max_c5_with_eps(O_raw, slope_raw, eps):
    """Maximum O after applying epsilon (includes O0 quadratic term)."""
    O_new = O_raw + eps * slope_raw
    O_new[len(O_raw) // 2] -= 2.0 * eps * eps
    return O_new.max()

def ternary_search_eps(eps_max, O_raw, slope_raw):
    lo, hi = 0.0, eps_max
    for _ in range(15):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        c1 = max_c5_with_eps(O_raw, slope_raw, m1)
        c2 = max_c5_with_eps(O_raw, slope_raw, m2)
        if c1 < c2:
            hi = m2
        else:
            lo = m1
    eps_opt = (lo + hi) / 2.0
    c_opt = max_c5_with_eps(O_raw, slope_raw, eps_opt)
    return eps_opt, c_opt

# ---------- SLSQP (soft‑max) ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        n = 512
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    n = h.size
    O_raw = compute_O_raw(h)
    cur_c_raw = O_raw.max()
    best_h = h.copy()
    best_c_raw = cur_c_raw

    no_improve = 0

    # main guided‑move loop
    while time.time() - start < budget_s * 0.92:
        i, j, k, cur_c_raw = worst_info(h, O_raw)

        if i == j:
            # random pair to break symmetry
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

        eps_max = min(h[i], 1.0 - h[j])
        if eps_max < 1e-12:
            no_improve += 1
            if no_improve > 300:
                # random shuffle to escape
                for _ in range(5):
                    a, b = rng.integers(0, n, size=2)
                    if a != b:
                        h[a], h[b] = h[b], h[a]
                O_raw = compute_O_raw(h)
                cur_c_raw = O_raw.max()
                no_improve = 0
            continue

        slope0_lin, slopes = compute_slopes(i, j, h)
        slope_raw = build_slope_raw(slope0_lin, slopes)

        eps_opt, c_opt = ternary_search_eps(eps_max, O_raw, slope_raw)

        if c_opt < cur_c_raw - 1e-12:
            # apply move
            h[i] -= eps_opt
            h[j] += eps_opt
            O_raw += eps_opt * slope_raw
            O_raw[len(O_raw) // 2] -= 2.0 * eps_opt * eps_opt
            cur_c_raw = O_raw.max()
            if cur_c_raw < best_c_raw - 1e-12:
                best_c_raw = cur_c_raw
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            # full random swap and recompute O_raw
            a, b = rng.integers(0, n, size=2)
            if a != b:
                delta = min(h[a], 1.0 - h[b])
                if delta > 0:
                    h[a] -= delta
                    h[b] += delta
            O_raw = compute_O_raw(h)
            cur_c_raw = O_raw.max()
            no_improve = 0

    # final soft‑max SLSQP passes (if time permits)
    if time.time() - start < budget_s * 0.96:
        for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0):
            if time.time() - start > budget_s * 0.99:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=400)
            cur_c = compute_c5(best_h)
            if cur_c * (n / 2.0) < best_c_raw:
                best_c_raw = cur_c * (n / 2.0)
                # keep best_h

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
