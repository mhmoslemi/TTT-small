def auto_corr(h):
    # circular autocorrelation via FFT
    # return arr length n, C[0]=weight
    h0 = np.asarray(h, dtype=float)
    n = h0.size
    # using FFT
    H = np.fft.rfft(h0)
    C = np.fft.irfft(np.abs(H)**2, n=n)
    # Round to nearest int? maybe small rounding errors; we can keep as float.
    return C

        J1 = zeros[a[zeros] == 1]

    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, best_n)
        add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
        best_h[i] += add
        diff = target - best_h.sum()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
