def project_sum_bounds(x, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=50):
    lo = lower - np.max(x)  # maybe negative enough
    hi = upper - np.min(x)  # positive enough
    for _ in range(max_iter):
        mid = (lo + hi)/2
        y = np.clip(x + mid, lower, upper)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
        if abs(s - target) < tol:
            break
    return y
```

def project_sum_bounds(x, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(x) - 1.0
    hi = upper - np.min(x) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, lower, upper)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
        if abs(s - target) <= tol:
            break
    y = np.clip(x + (lo+hi)/2.0, lower, upper)
    # ensure sum exactly target by adjusting first element marginally
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # add diff to a not full element
        idx = np.where((y > lower) & (y < upper))[0]
        if idx.size > 0:
            y[idx[0]] += diff
        else:
            y[0] += diff  # may go out of bounds slightly, but diff small
    return y
```

import time
import numpy as np

# ------------------------------------------------------------
def compute_c5_numpy(h: np.ndarray) -> float:
    """Exact C5 using np.correlate (full) and Δx = 2/N."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear convolution of h with reversed (1‑h) using FFT."""
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, n=size)
    g = (1.0 - h)[::-1]
    fg = np.fft.rfft(g, n=size)
    conv = np.fft.irfft(fh * fg, n=size)[: 2 * n - 1]
    return conv


def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using the FFT‑based convolution."""
    conv = compute_corr_fft(h)
    return np.max(conv) * (2.0 / h.size)


def project_sum_fixed(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h in‑place so that sum(h) = target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    if diff > 0:
        for i in range(n):
            space = 1.0 - h[i]
            if space <= 0.0:
                continue
            delta = diff if diff < space else space
            h[i] += delta
            diff -= delta
            if diff <= 1e-12:
                break
    else:
        for i in range(n):
            avail = h[i]
            if avail <= 0.0:
                continue
            delta = -diff if -diff < avail else avail
            h[i] -= delta
            diff += delta
            if diff >= -1e-12:
                break
    return h


def worst_gradient(h: np.ndarray, conv: np.ndarray = None):
    """Return gradient of the worst shift and the shift amount k."""
    n = h.size
    if conv is None:
        conv = compute_corr_fft(h)
    idx = int(np.argmax(conv))
    k = abs(idx - (n - 1))
    grad = np.ones_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad, k


def gradient_descent(
    h0: np.ndarray,
    target: float,
    max_iters: int,
    alpha0: float = 0.2,
    alpha_decay: float = 0.99,
):
    """Projected gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5_numpy(h)
    alpha = alpha0
    for _ in range(max_iters):
        conv = compute_corr_fft(h)
        c_curr = np.max(conv) * dx
        grad, _ = worst_gradient(h, conv)
        h_new = h - alpha * grad
        h_new = project_sum_fixed(h_new, target)
        conv_new = compute_corr_fft(h_new)
        c_new = np.max(conv_new) * dx
        if c_new < c_curr:
            h = h_new
        else:
            alpha *= alpha_decay
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if alpha < 1e-6:
            alpha = alpha0
    return best_h, best_c


def random_transfer_search(h: np.ndarray, steps: int, rng: np.random.Generator):
    """Small stochastic refinement: move mass from one index to another."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fft(best_h)
    for _ in range(steps):
        i, j = rng.choice(n, size=2, replace=False)
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5_fft(nh)
        if c < best_c:
            best_c, best_h = c, nh
    return best_h


def init_h_interpolated(N: int, rng: np.random.Generator):
    """Interpolate the parent solution to size N and project the sum."""
    old_vals = np.array(initial_h_values, dtype=float)
    old_n = old_vals.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, N, endpoint=False)
    h = np.interp(new_x, old_x, old_vals)
    h = np.clip(h, 0.0, 1.0)
    return project_sum_fixed(h, N / 2.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    start = time.time()
    time_budget = budget_s

    best_h = None

        if c_opt < best_c:

        if best_c <= 0.3808:

    # Final refinement phase
    if best_h is not None:
