    n = h.size

    corr = np.correlate(h, g, mode='full')

    grad = conv1[n-1: n-1 + n] - conv2[n-1: n-1 + n]  # slice length n
    return grad

import time, numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_bounds(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and enforce Σh = n/2 by moving mass between free entries."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop (fast enough for n≤512)
    while abs(diff) > 1e-12:
        if diff > 0:
            idxs = np.where(h < 1 - 1e-12)[0]
            if not len(idxs):
                break
            i = rng.choice(idxs)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff = target - h.sum()
        else:  # diff < 0
            idxs = np.where(h > 1e-12)[0]
            if not len(idxs):
                break
            i = rng.choice(idxs)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff = target - h.sum()
    return h


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    return project_sum_bounds(h, rng)


# ---------- worst‑shift gradient ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)        # shift k
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h, best_c = h.copy(), compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- smooth (soft‑max) gradient ----------
def softmax_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    n = h.size
    g = 1 - h
    corr = np.correlate(h, g, mode='full')
    b = beta * corr
    m = np.max(b)
    w = np.exp(b - m)
    w /= w.sum()
    conv1 = np.convolve(w, g, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    start = n - 1
    grad = conv1[start:start + n] - conv2[start:start + n]
    return grad


def smooth_descent(h: np.ndarray, rng: np.random.Generator,
                   beta: float, step: float, iters: int):
    for _ in range(iters):
        grad = softmax_gradient(h, beta)
        h = h - step * grad
        h = project_sum_bounds(h, rng)
    return h

                               (200.0, 0.015, 120),
                               (800.0, 0.008, 80)):
            if time.time() - start > 0.88 * budget_s:
                break
            h = smooth_descent(h, rng, beta, step, it)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # final tidying hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=800, step_scale=0.6)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ---- up‑sample to 512 if beneficial ----
    if best_n < 512 and time.time() - start < 0.95 * budget_s:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, 512, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_sum_bounds(h_up, rng)
        h_up, c_up = guided_hill_climb(h_up, rng, max_iters=3000, step_scale=0.6)

        for beta, step, it in ((20.0, 0.04, 200),
                               (200.0, 0.015, 120),
                               (2000.0, 0.006, 80)):
            if time.time() - start > budget_s:
                break
            h_up = smooth_descent(h_up, rng, beta, step, it)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, 512

    # ---- final random perturbations ----
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    best_h = project_sum_bounds(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
