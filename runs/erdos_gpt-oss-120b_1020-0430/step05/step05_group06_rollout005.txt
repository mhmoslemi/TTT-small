def smooth_max_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    # subtract max for stability
    m = np.max(beta * corr)
    # compute exp(beta * corr - m) to avoid overflow
    exp_term = np.exp(beta * corr - m)
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp  # shape (2n-1)

    # compute conv_fwd = w * h (as described)
    w_vec = w  # index offset n-1
    conv_fwd = np.convolve(w_vec, h, mode='full')
    w_rev_vec = w_vec[::-1]
    conv_rev = np.convolve(w_rev_vec, h, mode='full')
    # slice indices
    idx_start = n - 1
    idx_end = idx_start + n
    conv_fwd_mid = conv_fwd[idx_start:idx_end]
    conv_rev_mid = conv_rev[idx_start:idx_end]
    grad = 1.0 - conv_fwd_mid - conv_rev_mid

    # smooth max value
    smooth_val = (m + np.log(sum_exp)) / beta
    return smooth_val, grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Global parameters for constraints ----------
CURRENT_TARGET = None  # sum constraint (n/2)
CURRENT_BETA = None    # smooth‑max temperature

# ---------- Helper utilities ----------
def adjust_sum_exact(h, rng):
    """Exact correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

# ---------- Smooth‑max objective with exact gradient ----------
def _smooth_val_and_grad(x):
    """Return (smooth‑max value, gradient) for the current beta."""
    beta = CURRENT_BETA
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    # stability shift
    m = np.max(beta * corr)
    exp_term = np.exp(beta * corr - m)
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # weights for each lag
    # convolution to obtain Σ w_i * h_{k+i}
    conv_fwd = np.convolve(w, x, mode='full')
    conv_rev = np.convolve(w[::-1], x, mode='full')
    start = n - 1
    end = start + n
    grad = 1.0 - conv_fwd[start:end] - conv_rev[start:end]
    smooth_val = (m + np.log(sum_exp)) / beta
    return smooth_val, grad

def smooth_objective(x):
    val, _ = _smooth_val_and_grad(x)
    return val

def smooth_jac(x):
    _, grad = _smooth_val_and_grad(x)
    return grad

# ---------- SLSQP optimizer that uses the gradient ----------
def slsqp_optimize_with_grad(h0, beta, rng, max_iter=200):
    global CURRENT_BETA
    CURRENT_BETA = beta
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    opts = {"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    res = minimize(
        smooth_objective,
        h0,
        method="SLSQP",
        jac=smooth_jac,
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

# ---------- Greedy binary construction (same as previous code) ----------
def greedy_binary(n, rng):
    """Construct a binary vector of length n with n/2 ones, minimizing max overlap."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        # O_adj = O + 1 works because adding a 1 at position p adds 1 to every lag
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h, O
