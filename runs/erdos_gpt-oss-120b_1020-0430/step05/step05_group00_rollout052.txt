def binary_sa_512(n, rng, time_budget):

   start = time.time()
   best_h = None
   best_c = np.inf

   # initial random start
   h = np.zeros(n, dtype=float)
   ones = rng.choice(n, n // 2, replace=False)
   h[ones] = 1.0

   c_curr = compute_c5(h)
   if c_curr < best_c:
       best_c, best_h = c_curr, h.copy()

   T0 = 0.02
   T_end = 1e-5
   iter = 0
   while True:
       elapsed = time.time() - start
       if elapsed >= time_budget:
           break
       # cooling temperature
       T = T0 * (T_end / T0) ** (elapsed / time_budget)
       # propose swap
       # select random 1 index and random 0 index
       i = rng.choice(np.where(h == 1.0)[0])
       j = rng.choice(np.where(h == 0.0)[0])
       # swap
       h[i] = 0.0
       h[j] = 1.0
       c_new = compute_c5(h)
       delta = c_new - c_curr
       if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
           c_curr = c_new
           if c_new < best_c:
               best_c, best_h = c_new, h.copy()
               if best_c <= 0.3808:
                   return best_h, best_c
       else:
           # revert
           h[i] = 1.0
           h[j] = 0.0
       iter += 1
   return best_h, best_c

    mid = n // 2

    if start_h is not None:

        bin_h = np.where(start_h >= 0.5, 1.0, 0.0)
        # enforce exact sum
        current_sum = bin_h.sum()

            # adjust by moving entries near 0.5
            diff = int(mid - current_sum)
            if diff > 0:
                # need to set diff zeros to 1 (choose those with highest start_h among zeros)
                zero_idx = np.flatnonzero(bin_h == 0)
                # sort zero_idx by start_h descending

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

# ---------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- binary simulated annealing (fast swaps) ----------
def binary_sa_optimize(n: int, rng: np.random.Generator,
                       time_budget: float, start_h: np.ndarray = None):
    start = time.time()
    mid = n // 2
    if start_h is not None:
        bin_h = np.where(start_h >= 0.5, 1.0, 0.0)
        cur_sum = int(bin_h.sum())
        if cur_sum != mid:
            diff = mid - cur_sum
            if diff > 0:
                zero_idx = np.flatnonzero(bin_h == 0)
                top = zero_idx[np.argpartition(start_h[zero_idx], -diff)[-diff:]]
                bin_h[top] = 1.0
            else:
                one_idx = np.flatnonzero(bin_h == 1)
                diff = -diff
                low = one_idx[np.argpartition(start_h[one_idx], diff)[:diff]]
                bin_h[low] = 0.0
        h = bin_h.astype(float)
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        rng.shuffle(ones)
        rng.shuffle(zeros)
        order = np.concatenate([ones, zeros])
    else:
        order = rng.permutation(n)
        h = np.zeros(n, dtype=float)
        h[order[:mid]] = 1.0

    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr

    T0, T_end = 0.02, 1e-5
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        i_idx = rng.integers(0, mid)
        j_idx = rng.integers(mid, n)
        i = order[i_idx]
        j = order[j_idx]
        h[i] = 0.0
        h[j] = 1.0
        order[i_idx], order[j_idx] = order[j_idx], order[i_idx]
        c_new = compute_c5(h)
        delta = c_new - c_curr
        accept = delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12))
        if accept:
            c_curr = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        else:
            h[i] = 1.0
            h[j] = 0.0
            order[i_idx], order[j_idx] = order[j_idx], order[i_idx]
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # decide resolution
    n = 512
    best_h = None
    best_c = np.inf

    # use parent if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)

    time_binary = total_budget * 0.60

    if c_bin < best_c:

    # remaining time for continuous refinement
    elapsed = time.time() - start_time
    remaining = total_budget - elapsed
    if remaining > 20.0 and best_h is not None:
        # guided hill‑climb
        max_iters = int(min(3000, remaining * 250))
        h_gc, c_gc = guided_hill_climb(best_h.copy(), rng,
                                       max_iters=max_iters, step_scale=0.7)
        if c_gc < best_c:
            best_h, best_c = h_gc, c_gc
        elapsed = time.time() - start_time
        remaining = total_budget - elapsed

        # soft‑max SLSQP passes
        betas = (20., 80., 200., 800., 3000., 8000.)
        for beta in betas:
            if time.time() - start_time > total_budget * 0.95:

            c_sl = compute_c5(h_sl)

        # gradient tidy‑up
        if time.time() - start_time < total_budget * 0.97:
            h_gr = gradient_refine(best_h.copy(), rng,
                                   step=0.015, max_iter=15)
            c_gr = compute_c5(h_gr)
            if c_gr < best_c:

    # final stochastic tweaks until time up
    while time.time() - start_time < total_budget * 0.999 and best_h is not None:

        nh = guided_perturb(best_h, grad, rng, scale=0.02)

    # final sanitisation

    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
