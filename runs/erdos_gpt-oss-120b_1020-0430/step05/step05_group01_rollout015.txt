import time
import numpy as np

TARGET_C = 0.3808  # early exit target


def compute_c5(v: np.ndarray) -> float:
    """Maximum overlap integral for a binary step function (discrete)."""
    n = v.shape[0]
    # correlation = convolution of v with reverse of (1-v)
    w_rev = (1.0 - v)[::-1]
    n_full = 2 * n - 1
    vf = np.fft.rfft(v, n=n_full)
    wf = np.fft.rfft(w_rev, n=n_full)
    conv = np.fft.irfft(vf * wf, n=n_full)
    return np.max(conv) * (2.0 / n)


def random_balanced_vector(N: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector of length N with exactly N/2 ones."""
    half = N // 2
    v = np.zeros(N, dtype=float)
    v[rng.choice(N, half, replace=False)] = 1.0
    return v


def thue_morse(N: int) -> np.ndarray:
    """Thue‑Morse binary sequence of length N (N must be a power of two)."""
    return np.array([bin(i).count("1") % 2 for i in range(N)], dtype=float)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    N = 512
    half = N // 2

    # ---------- initialise best solution ----------
    best_h = None
    best_c = np.inf

    # use parent if supplied
    if "initial_h_values" in globals():
        v0 = np.array(initial_h_values, dtype=float)
        v0 = (v0 >= 0.5).astype(float)          # binarise
        s = int(v0.sum())
        diff = half - s
        if diff > 0:
            zeros = np.where(v0 == 0)[0]
            v0[rng.choice(zeros, diff, replace=False)] = 1.0
        elif diff < 0:
            ones = np.where(v0 == 1)[0]
            v0[rng.choice(ones, -diff, replace=False)] = 0.0
        c0 = compute_c5(v0)
        best_h, best_c = v0.copy(), c0

    # Thue‑Morse
    tm = thue_morse(N)
    c_tm = compute_c5(tm)
    if c_tm < best_c:
        best_h, best_c = tm.copy(), c_tm

    # a few random starts
    for _ in range(3):
        v = random_balanced_vector(N, rng)
        c = compute_c5(v)
        if c < best_c:
            best_h, best_c = v.copy(), c

    # ---------- simulated annealing ----------
    while time.time() - start < budget_s * 0.99:
        if best_c <= TARGET_C:
            break

        # start point for this run
        if rng.random() < 0.3 and best_h is not None:
            v = best_h.copy()
            cur_c = best_c
        else:
            v = random_balanced_vector(N, rng)
            cur_c = compute_c5(v)

        # temperature schedule
        T = 0.02                     # initial temperature
        decay = 0.99995

        # inner annealing loop
        while time.time() - start < budget_s * 0.99:
            # pick a random 1 and a random 0
            i1 = rng.integers(N)
            while v[i1] == 0.0:
                i1 = rng.integers(N)
            i0 = rng.integers(N)
            while v[i0] == 1.0:
                i0 = rng.integers(N)

            # perform the swap
            v[i1] = 0.0
            v[i0] = 1.0

            new_c = compute_c5(v)
            delta = new_c - cur_c

            if delta <= 0.0 or rng.random() < np.exp(-delta / T):
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = v.copy()
                    if best_c <= TARGET_C:
                        break
                # swap kept
            else:
                # revert
                v[i1] = 1.0
                v[i0] = 0.0

            T = max(1e-6, T * decay)

        # if we reached target inside inner loop, break outer as well
        if best_c <= TARGET_C:
            break

    # ---------- final tidy up ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact half‑ones (tiny rounding issues)
    v_bin = (best_h >= 0.5).astype(float)
    s = int(v_bin.sum())
    if s != half:
        if s > half:
            ones = np.where(v_bin == 1)[0]
            v_bin[rng.choice(ones, s - half, replace=False)] = 0.0
        else:
            zeros = np.where(v_bin == 0)[0]
            v_bin[rng.choice(zeros, half - s, replace=False)] = 1.0
    best_h = v_bin
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
