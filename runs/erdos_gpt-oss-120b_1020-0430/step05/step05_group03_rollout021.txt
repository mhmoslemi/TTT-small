import time, math, numpy as np
from scipy.optimize import minimize, LinearConstraint

# ---------- utilities ----------
def compute_c5_direct(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2] (used for final evaluation)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def compute_c5_fast(h: np.ndarray) -> float:
    """Fast C5 estimate using FFT (max O_k = 1 - 2*min_corr/n)."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two >= 2n-1
    f = np.fft.rfft(h, m)
    conv = np.fft.irfft(f * np.conj(f), m)[: 2 * n - 1]   # autocorrelation of h
    min_corr = np.min(conv)
    # guard against tiny negative round‑off
    if min_corr < 0:
        min_corr = 0.0
    return 1.0 - 2.0 * min_corr / n


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop – fast because diff shrinks quickly
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5_direct(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5_direct(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # log‑sum‑exp for smooth max
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(
        softmax, h0, method="SLSQP",
        bounds=bounds, constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


def anneal_binary(h: np.ndarray, total_iters: int, T0: float,
                  T_end: float, rng: np.random.Generator,
                  time_limit: float) -> tuple:
    """Simulated‑annealing on binary vectors."""
    n = h.size
    c_fast = compute_c5_fast(h)
    best_h = h.copy()
    best_c_direct = compute_c5_direct(h)

    log_T0 = math.log(T0)
    log_T_end = math.log(T_end)
    log_factor = (log_T_end - log_T0) / total_iters

    for it in range(total_iters):
        if time.time() >= time_limit:
            break
        T = math.exp(log_T0 + log_factor * it)
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue
        # propose swap
        h[i], h[j] = h[j], h[i]
        new_fast = compute_c5_fast(h)
        delta = new_fast - c_fast
        if delta <= 0.0 or rng.random() < math.exp(-delta / T):
            c_fast = new_fast
            if c_fast < best_c_direct - 1e-9:
                # re‑evaluate exactly
                c_direct = compute_c5_direct(h)
                if c_direct < best_c_direct:
                    best_c_direct = c_direct
                    best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]

    return best_h, best_c_direct


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_limit = start + budget_s

    n = 512
    target_sum = n / 2.0

    # initial solution -------------------------------------------------
    if "initial_h_values" in globals():
        best_h = np.array(initial_h_values, dtype=float)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5_direct(best_h)
    else:
        best_h = random_feasible_vector(n, rng)
        best_c = compute_c5_direct(best_h)

    # ---- binary simulated annealing (≈ 30 % of time) ------------------
    anneal_deadline = start + budget_s * 0.30
    max_restarts = 12
    iters_per_restart = 50000
    for _ in range(max_restarts):
        if time.time() >= anneal_deadline:
            break
        h_bin = random_binary_vector(n, rng)
        best_h_bin, best_c_bin = anneal_binary(
            h_bin, total_iters=iters_per_restart,
            T0=5e-3, T_end=1e-6, rng=rng,
            time_limit=anneal_deadline
        )
        if best_c_bin < best_c:
            best_h = best_h_bin.astype(float)
            best_c = best_c_bin

    # ---- continuous refinement (≈ 55 % of time) -----------------------
    refine_deadline = start + budget_s * 0.85
    while time.time() < refine_deadline:
        # a chunk of guided hill‑climb
        h_new, c_new = guided_hill_climb(best_h, rng, max_iters=3000, step_scale=0.7)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

        # SLSQP passes with increasing β
        for beta in (20., 80., 200., 800., 3000., 8000., 20000., 50000.):
            if time.time() >= refine_deadline:
                break
            maxit = 1200 if beta >= 3000 else 800
            best_h = slsqp_optimize(best_h, beta, max_iter=maxit)
            c_new = compute_c5_direct(best_h)
            if c_new < best_c:
                best_c = c_new

        # fine gradient tidy‑up
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=15)
        c_new = compute_c5_direct(best_h)
        if c_new < best_c:
            best_c = c_new

    # ---- final stochastic tweaks until time runs out -----------------
    while time.time() < time_limit * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5_direct(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # final sanity ------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_direct(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
