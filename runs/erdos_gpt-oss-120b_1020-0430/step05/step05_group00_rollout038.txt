def run(seed=42,budget_s=1100,...):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    n = 512
    # initial best
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0_bin = np.zeros_like(h0)
        idx = np.argsort(-h0)
        h0_bin[idx[:n//2]] = 1.0
        best_h = h0_bin
        best_c = compute_c5_fft(best_h)
    else:
        # random binary
        ones = rng.choice(n, n//2, replace=False)
        best_h = np.zeros(n)
        best_h[ones] = 1.0
        best_c = compute_c5_fft(best_h)

    # SA loop
    time_sa = total * 0.8 # allocate 80%
    T0 = 0.04
    Tend = 1e-4
    # current state
    h_curr = best_h.copy()
    c_curr = best_c
    while time.time() - start < time_sa:
        elapsed = time.time() - start
        T = T0 * (Tend/T0) ** (elapsed / time_sa)
        # compute worst shift
        # compute correlation full for h_curr, but we already have c_curr; need index k0
        # Let's compute correlation once per iteration? That's O(n log n). Accept.
        # Could store previous corr to reuse after swap? But we need new anyway on each iteration.
        corr = get_corr_fft(h_curr)  # returns full correlation array speed.
        idx_max = np.argmax(corr)
        k0 = idx_max - (n - 1)
        # compute candidate swap
        if k0 != 0:
            # forward shift
            if k0 > 0:
                h_shift = np.concatenate([h_curr[k0:], np.zeros(k0)])
            else:
                # negative shift treat similarly (swap direction)
                # shift by -k: h_shift = zeros(-k) + h_curr[:n + k]
                k_abs = -k0
                h_shift = np.concatenate([np.zeros(k_abs), h_curr[:n - k_abs]])
            # zeros/h ones
            i_cand = np.where((h_curr == 1.0) & (h_shift == 1.0))[0]
            j_cand = np.where((h_curr == 0.0) & (1 - h_shift == 1.0))[0]
            if i_cand.size > 0 and j_cand.size > 0:
                i = rng.choice(i_cand)
                j = rng.choice(j_cand)
                # propose swap
                h_new = h_curr.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                c_new = compute_c5_fft(h_new)
                # acceptance
                if c_new < c_curr or rng.random() < np.exp(-(c_new - c_curr)/max(T,1e-12)):
                    h_curr, c_curr = h_new, c_new
                    if c_new < best_c:
                        best_h, best_c = h_new.copy(), c_new
                continue  # next iteration
        # fallback random swap
        ones_idx = np.where(h_curr == 1.0)[0]
        zeros_idx = np.where(h_curr == 0.0)[0]
        i = rng.choice(ones_idx)

def binary_sa(n, rng, time_budget, h_start=None):
    start = time.time()
    target_ones = n // 2
    # initial binary
    if h_start is not None:
        h_curr = h_start.copy()
    else:
        h_curr = np.zeros(n)
        ones = rng.choice(n, target_ones, replace=False)
        h_curr[ones] = 1.0
    c_curr = compute_c5(h_curr)
    best_h = h_curr.copy()
    best_c = c_curr
    steps_no_improve = 0
    restart_limit = max(20000, n*40)  # heuristically
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        # compute temperature schedule: T decreasing
        T0 = 0.04
        Tend = 1e-4
        T = T0 * (Tend/T0) ** (elapsed / time_budget)
        # compute correlation and worst shift
        corr = compute_corr_fft(h_curr)
        idx_max = np.argmax(corr)
        k0 = idx_max - (n - 1)
        # try guided swap
        swap_done = False
        if k0 != 0:
            if k0 > 0:
                h_shift = np.concatenate([h_curr[k0:], np.zeros(k0)])
            else:
                kk = -k0
                h_shift = np.concatenate([np.zeros(kk), h_curr[:n-kk]])
            i_cand = np.where((h_curr == 1.0) & (h_shift == 1.0))[0]
            j_cand = np.where((h_curr == 0.0) & ((1 - h_shift) == 1.0))[0]
            if i_cand.size and j_cand.size:
                i = rng.choice(i_cand)
                j = rng.choice(j_cand)
                # swap
                h_new = h_curr.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                c_new = compute_c5(h_new)
                swap_done = True
        if not swap_done:
            # random swap
            ones_idx = np.where(h_curr == 1.0)[0]
            zeros_idx = np.where(h_curr == 0.0)[0]
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            h_new = h_curr.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            c_new = compute_c5(h_new)
        # acceptance
        if c_new < c_curr or rng.random() < np.exp(-(c_new - c_curr)/max(T,1e-12)):
            h_curr = h_new
            c_curr = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                steps_no_improve = 0
            else:
                steps_no_improve += 1
        else:
            steps_no_improve += 1
        # restart if stuck
        if steps_no_improve > restart_limit:
            # reinitialize random
            h_curr = np.zeros(n)
            ones = rng.choice(n, target_ones, replace=False)
            h_curr[ones] = 1.0
            c_curr = compute_c5(h_curr)
            steps_no_improve = 0
    return best_h, best_c

def worst_gradient(h):
    n = h.size
    corr = compute_corr_fft(h)
    idx_max = np.argmax(corr)
    k = idx_max - (n - 1)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        # forward shift
        if k > 0:

            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[kk:] -= h[: n - kk]
            grad[: n - kk] -= h[kk:]
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# Global variables for the soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def _fft_len(n: int) -> int:
    """Small helper: next power of two ≥ 2n‑1."""
    return 1 << ((2 * n - 1).bit_length())


def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1‑h) using FFT."""
    n = h.size
    L = _fft_len(n)
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    return corr[: 2 * n - 1]


def compute_c5(h: np.ndarray) -> float:
    """Discrete C₅ for a step function on [0,2]."""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        if k > 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            kk = -k
            grad[kk:] -= h[: n - kk]
            grad[: n - kk] -= h[kk:]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = compute_corr_fft(h)
        idx = np.argmax(corr)
        k = idx - (n - 1)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            if k > 0:
                g[: n - k] += 1.0 - h[k:]
                g[k:] -= h[: n - k]
            else:
                kk = -k
                g[kk:] += 1.0 - h[: n - kk]
                g[: n - kk] -= h[kk:]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = n / 2.0 - h.sum()
    return h


# ----------------------------------------------------------------------
# Soft‑max SLSQP helpers
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


# ----------------------------------------------------------------------
def binary_sa(n: int, rng: np.random.Generator, time_budget: float,
              h_start: np.ndarray = None):
    """Guided simulated‑annealing on binary vectors (0/1) with exactly n/2 ones."""
    start = time.time()
    target_ones = n // 2
    if h_start is not None:
        h = h_start.copy()
    else:
        h = np.zeros(n)
        ones = rng.choice(n, target_ones, replace=False)
        h[ones] = 1.0
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    steps_no_improve = 0
    restart_limit = max(20000, n * 40)

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T0, Tend = 0.04, 1e-4
        T = T0 * (Tend / T0) ** (elapsed / time_budget)

        corr = compute_corr_fft(h)
        idx = np.argmax(corr)
        k0 = idx - (n - 1)

        made_swap = False
        if k0 != 0:
            if k0 > 0:
                h_shift = np.concatenate([h[k0:], np.zeros(k0)])
            else:
                kk = -k0
                h_shift = np.concatenate([np.zeros(kk), h[: n - kk]])
            i_cand = np.where((h == 1.0) & (h_shift == 1.0))[0]
            j_cand = np.where((h == 0.0) & ((1.0 - h_shift) == 1.0))[0]
            if i_cand.size and j_cand.size:
                i = rng.choice(i_cand)
                j = rng.choice(j_cand)
                h_new = h.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                c_new = compute_c5(h_new)
                made_swap = True
        if not made_swap:
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            h_new = h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            c_new = compute_c5(h_new)

        if c_new < c or rng.random() < np.exp(-(c_new - c) / max(T, 1e-12)):
            h, c = h_new, c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
                steps_no_improve = 0
            else:
                steps_no_improve += 1
        else:
            steps_no_improve += 1

        if steps_no_improve > restart_limit:
            h = np.zeros(n)
            ones = rng.choice(n, target_ones, replace=False)
            h[ones] = 1.0
            c = compute_c5(h)
            steps_no_improve = 0

    return best_h, best_c


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    base_n = 512

    # ----- initial continuous start -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        h0 = np.random.random(base_n)
        h0 = enforce_sum(h0, rng)

    best_h, best_c, best_n = h0.copy(), compute_c5(h0), base_n

    # ----- binary simulated annealing (≈50% of time) -----
    time_binary = total * 0.50
    # create a binary start from the best continuous solution
    idx = np.argsort(-h0)
    h_bin0 = np.zeros(base_n)
    h_bin0[idx[: base_n // 2]] = 1.0
    h_bin, c_bin = binary_sa(base_n, rng, time_binary, h_start=h_bin0)
    if c_bin < best_c:
        best_h, best_c, best_n = h_bin.astype(float), c_bin, base_n

    # ----- continuous refinement (remaining time) -----
    remaining = total - (time.time() - start)
    if remaining > 0:
        # guided hill‑climb
        max_iters = int(min(3000, remaining * 400))
        h_hc, c_hc = guided_hill_climb(best_h, rng, max_iters=max_iters, step_scale=0.8)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

        # soft‑max SLSQP passes
        for beta in (20., 80., 200., 800., 3000.):
            if time.time() - start > total * 0.95:
                break
            max_iter = 600 if beta >= 200 else 400
            h_opt = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt.copy(), c_opt

        # gradient tidy‑up
        if time.time() - start < total * 0.97:
            h_gr = gradient_refine(best_h, rng, step=0.015, max_iter=15)
            c_gr = compute_c5(h_gr)
            if c_gr < best_c:
                best_h, best_c = h_gr.copy(), c_gr

        # ----- up‑sampling to double resolution -----
        if (time.time() - start) < total * 0.99 and best_n * 2 <= 1024:
            n_up = best_n * 2
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = enforce_sum(h_up, rng)

            # hill‑climb on the up‑sampled array
            max_iters_up = int(min(2000, (total - (time.time() - start)) * 200))
            h_up, c_up = guided_hill_climb(
                h_up, rng, max_iters=max_iters_up, step_scale=0.7
            )
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

            # SLSQP on the up‑sampled array
            for beta in (20., 200., 2000., 8000.):
                if time.time() - start > total:
                    break
                h_up_opt = slsqp_optimize(best_h, beta, rng, max_iter=500)
                c_up_opt = compute_c5(h_up_opt)
                if c_up_opt < best_c:
                    best_h, best_c, best_n = h_up_opt.copy(), c_up_opt, best_h.size

            # final gradient tidy‑up on up‑sampled
            if time.time() - start < total * 0.999:
                h_up_gr = gradient_refine(best_h, rng, step=0.01, max_iter=8)
                c_up_gr = compute_c5(h_up_gr)
                if c_up_gr < best_c:
                    best_h, best_c, best_n = (
                        h_up_gr.copy(),
                        c_up_gr,
                        best_h.size,
                    )

    # ----- stochastic tweaking until budget expires -----
    while time.time() - start < total * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ----- final sanitisation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
