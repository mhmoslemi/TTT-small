def simulated_annealing_binary(h, rng, deadline, start_T=None, decay=0.9995, max_iter=None):
    n = h.size
    ones = [i for i, val in enumerate(h) if val > 0.5]
    zeros = [i for i, val in enumerate(h) if val <= 0.5]
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = start_T if start_T is not None else max(0.001 * cur_c, 1e-6)
    it = 0
    while time.time() < deadline:
        if max_iter is not None and it >= max_iter:
            break
        # pick random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new C
        new_c = compute_c5(h)
        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            # accept
            cur_c = new_c
            # update lists
            # replace i with j in ones list, and j with i in zeros list
            # but need to find index positions in lists: they are known by using position arrays?

            h[j] = 0.0
        # update best

            best_h = h.copy()
        # temperature decay
        T *= decay

        it += 1
    return best_h, best_c

def sa_binary(h, rng, deadline, init_T=None, decay=0.9995, max_iter=200000):
    n = h.size
    # maintain ones and zeros lists with pos arrays
    ones = np.flatnonzero(h > 0.5).tolist()
    zeros = np.flatnonzero(h <= 0.5).tolist()
    pos_one = np.full(n, -1, dtype=int)
    pos_zero = np.full(n, -1, dtype=int)
    for idx,i in enumerate(ones):
        pos_one[i] = idx
    for idx,i in enumerate(zeros):
        pos_zero[i] = idx
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    if init_T is None:
        init_T = max(0.001 * cur_c, 1e-6)
    T = init_T
    it = 0
    while time.time() < deadline and it < max_iter:
        # choose random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            # Metropolis criteria
            if T > 0:
                if rng.random() < np.exp(-(new_c - cur_c) / T):
                    accept = True
        if accept:
            cur_c = new_c
            # update lists and pos arrays
            # swap i (now zero) with j (now one)
            # In ones list, replace index of i with j
            p_i = pos_one[i]  # index in ones list
            ones[p_i] = j
            pos_one[j] = p_i
            pos_one[i] = -1
            # In zeros list, replace index of j with i
            p_j = pos_zero[j]
            zeros[p_j] = i
            pos_zero[i] = p_j
            pos_zero[j] = -1
            # update best
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                # optionally adjust when best_c is very low to reset temperature
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # decay temperature
        T *= decay
        it += 1
    return best_h, best_c

def hill_climb_binary(h, rng, max_no_improve=5000):
    n = h.size
    cur_c = compute_c5(h)
    no_improve = 0
    while no_improve < max_no_improve:
        # compute worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # find mismatched ones and zeros for this shift
        i_range = np.arange(n)
        # positions i where i+k in range
        cond_i = (i_range + k >= 0) & (i_range + k < n)
        # ones in mismatched: h[i]==1 and h[i+k]==0
        mism_one = i_range[cond_i & (h == 1) & (h[i_range + k] == 0)]
        # zeros mismatched: h[i]==0 and h[i+k]==1
        mism_zero = i_range[cond_i & (h == 0) & (h[i_range + k] == 1)]
        if mism_one.size == 0 or mism_zero.size == 0:
            break
        # try random pair
        i = mism_one[rng.integers(mism_one.size)]
        j = mism_zero[rng.integers(mism_zero.size)]
        # swap and compute new c
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        if new_c < cur_c - 1e-12:
            cur_c = new_c
            no_improve = 0
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, cur_c

def run_sa_until_deadline(h, rng, deadline, max_iter=10**9):
    n = h.size
    # Ensure binary
    # For safety, convert any values >0.5 to 1, else 0
    h = (h > 0.5).astype(float)
    ones = np.flatnonzero(h == 1).tolist()
    zeros = np.flatnonzero(h == 0).tolist()
    pos_one = np.full(n, -1, dtype=int)
    pos_zero = np.full(n, -1, dtype=int)
    for idx, i in enumerate(ones):
        pos_one[i] = idx
    for idx, i in enumerate(zeros):
        pos_zero[i] = idx
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = max(0.01 * cur_c, 1e-6)
    decay = 0.9997  # slower cooling
    it = 0
    while time.time() < deadline and it < max_iter:
        # choose random indices
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if T > 1e-12 and rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            # update lists pos arrays
            p_i = pos_one[i]  # index in ones list
            # replace i with j in ones
            ones[p_i] = j
            pos_one[j] = p_i
            pos_one[i] = -1
            p_j = pos_zero[j]
            zeros[p_j] = i
            pos_zero[i] = p_j
            pos_zero[j] = -1
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T *= decay
        it += 1
    return best_h, best_c

def hill_climb_binary(h, rng, max_no_improve=2000):
    n = h.size
    cur_c = compute_c5(h)
    no_improve = 0
    while no_improve < max_no_improve:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # compute mismatches for shift k
        i_range = np.arange(n)
        valid = (i_range + k >= 0) & (i_range + k < n)
        # mismatch where h[i]==1 and h[i+k]==0
        mism_one = i_range[valid & (h == 1) & (h[i_range + k] == 0)]
        mism_zero = i_range[valid & (h == 0) & (h[i_range + k] == 1)]
        if mism_one.size == 0 or mism_zero.size == 0:
            break
        # choose random pair
        i = mism_one[rng.integers(mism_one.size)]
        j = mism_zero[rng.integers(mism_zero.size)]
        # try swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        if new_c < cur_c - 1e-12:
            cur_c = new_c
            no_improve = 0
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, cur_c

import time
import numpy as np

# ---------- helpers ----------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute a tiny amount to enforce Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
    return h


# ---------- simulated annealing for binary vectors ----------
def sa_binary(h0: np.ndarray, rng: np.random.Generator,
              deadline: float, max_iter: int) -> tuple[np.ndarray, float]:
    """Binary simulated annealing with fixed swaps. Returns best (h, C5)."""
    n = h0.size
    # enforce exact binary representation
    h = (h0 > 0.5).astype(float)
    ones = np.flatnonzero(h == 1).tolist()
    zeros = np.flatnonzero(h == 0).tolist()
    pos_one = np.full(n, -1, dtype=int)
    pos_zero = np.full(n, -1, dtype=int)
    for idx, i in enumerate(ones):
        pos_one[i] = idx
    for idx, i in enumerate(zeros):
        pos_zero[i] = idx

    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    T = max(0.01 * cur_c, 1e-6)
    decay = 0.9997

    it = 0
    while it < max_iter and time.time() < deadline:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)

        accept = False
        if new_c < cur_c:
            accept = True
        elif T > 1e-12 and rng.random() < np.exp(-(new_c - cur_c) / T):
            accept = True

        if accept:
            cur_c = new_c
            # update position structures
            p_i = pos_one[i]                     # index of i in ones list
            ones[p_i] = j
            pos_one[j] = p_i
            pos_one[i] = -1

            p_j = pos_zero[j]                    # index of j in zeros list
            zeros[p_j] = i
            pos_zero[i] = p_j
            pos_zero[j] = -1

            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

        T *= decay
        it += 1
    return best_h, best_c


# ---------- greedy hill climbing for binary vectors ----------
def hill_climb_binary(h: np.ndarray, rng: np.random.Generator,
                     max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """Local search focusing on the current worst shift."""
    n = h.size
    cur_c = compute_c5(h)
    no_imp = 0
    while no_imp < max_no_improve:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)

        i_range = np.arange(n)
        valid = (i_range + k >= 0) & (i_range + k < n)

        mism_one = i_range[valid & (h == 1) & (h[i_range + k] == 0)]
        mism_zero = i_range[valid & (h == 0) & (h[i_range + k] == 1)]

        if mism_one.size == 0 or mism_zero.size == 0:
            break

        i = mism_one[rng.integers(mism_one.size)]
        j = mism_zero[rng.integers(mism_zero.size)]

        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)

        if new_c < cur_c - 1e-12:
            cur_c = new_c
            no_imp = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_imp += 1
    return h, cur_c


# ---------- main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # parent solution, if provided
    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        adjust_sum_exact(parent, rng)

    # ----- binary start generation helpers -----
    def binary_from_parent(arr: np.ndarray) -> np.ndarray:
        n = arr.size
        bin_h = np.zeros(n, dtype=float)
        idx = np.argpartition(arr, -n // 2)[-n // 2 :]
        bin_h[idx] = 1.0
        return bin_h

    # ----- attempt schedule (n, iterations per start, starts) -----
    schedule = [
        (360, 500_000, 2),
        (240, 300_000, 2),
        (180, 200_000, 2),
    ]

    for n, max_iter, n_starts in schedule:
        if time.time() >= deadline:
            break
        for s in range(n_starts):
            if time.time() >= deadline:
                break
            # initial vector
            if parent is not None and parent.size == n and s == 0:
                h0 = binary_from_parent(parent)
            else:
                h0 = np.zeros(n, dtype=float)
                h0[rng.choice(n, n // 2, replace=False)] = 1.0

            # simulated annealing
            sub_dead = deadline  # SA stops early on time limit
            h_best, c_best = sa_binary(h0, rng, sub_dead, max_iter)

            # greedy local improvement
            h_best, c_best = hill_climb_binary(h_best, rng, max_no_improve=2000)

            if c_best < best_c:
                best_h, best_c, best_n = h_best.copy(), c_best, n

    # ----- upsample best binary solution to n=720 and refine -----
    if best_h is not None and best_n is not None and best_n * 2 <= 720:
        if time.time() < deadline:
            n_up = best_n * 2
            h_up = np.repeat(best_h, 2)  # keeps exact half‑ones property
            # SA on the up‑sampled vector
            h_up, c_up = sa_binary(h_up, rng, deadline, 800_000)
            h_up, c_up = hill_climb_binary(h_up, rng, max_no_improve=3000)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # ----- fallback (very small binary vector) -----
    if best_h is None:
        n = 180
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        best_h, best_c, best_n = h, compute_c5(h), n

    # final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
