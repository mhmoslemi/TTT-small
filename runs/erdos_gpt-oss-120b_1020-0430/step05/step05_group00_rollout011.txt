def smoothed_max_corr_and_grad(h, beta):
    # compute corr via FFT (linear)
    n = h.size
    size = 1 << ((2*n-1).bit_length())
    fh = np.fft.rfft(h, size)
    fg = np.fft.rfft(1 - h, size)
    corr = np.fft.irfft(fh * fg, size)[:2*n-1]
    # compute softmax weights
    # compute max for numeric stability:
    m = np.max(beta * corr)
    # we should compute w = exp(beta*corr - m) ; then normalized
    exp_vals = np.exp(beta * corr - m)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp  # shape (2n-1)
    # smooth max value f = (log sum_exp + m) / beta
    f = (np.log(sum_exp) + m) / beta
    # compute gradient
    # b = 1 - h
    b = 1 - h
    # conv w with b -> term1
    # convolve using FFT for speed:
    # pad w to size:
    fw = np.fft.rfft(w, size)
    term1_full = np.fft.irfft(fw * np.fft.rfft(b, size), size)[:2*n-1]
    term1 = term1_full[(n-1):(n-1)+n]
    # conv w_rev with a:
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, size)
    term2_full = np.fft.irfft(fw_rev * np.fft.rfft(h, size), size)[:2*n-1]
    term2 = term2_full[(n-1):(n-1)+n]
    grad = term1 - term2
    return f, grad

def project_to_constraints(h, target, rng):
    # ensure within [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    diff = target - h.sum()
    # use iterative random adjust
    n = h.size
    # while diff bigger than tolerance:
    # do up to some max iter
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        add = diff
        # clamp add to be feasible:
        if add > 0:
            add = min(add, 1.0 - h[i])
        else:
            add = max(add, -h[i])
        if abs(add) < 1e-15:
            # can't adjust this i, pick another
            it += 1
            continue
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h

def binary_hill_climb(h, rng, time_budget):
    start = time.time()
    n = h.size
    # ensure binary
    h = (h > 0.5).astype(float)
    c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = c
    # maintain lists
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    while time.time() - start < time_budget:
        # try random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute new c using direct compute
        h[i] = 0.0; h[j] = 1.0
        new_c = compute_c5_fft(h)
        if new_c < c:
            c = new_c
            # update lists
            idx_i = ones.index(i)
            ones[idx_i] = j
            idx_j = zeros.index(j)
            zeros[idx_j] = i
            if c < best_c:
                best_c = c
                best_h = h.copy()

            # revert
            h[i] = 1.0; h[j] = 0.0
        # optionally break early if best_c <= threshold
        if best_c <= 0.3808:

    return best_h, best_c

import time

# -------------------------------------------------
# Utility functions
def compute_c5_fft(h: np.ndarray) -> float:
    """Exact C5 for a (binary or continuous) step function on [0,2] using FFT."""
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, size)
    fg = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(fh * fg, size)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)


def project_to_constraints(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to `target` by random feasible moves."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) <= 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

# Smoothed max and its gradient (log‑sum‑exp)
def smoothed_max_corr_and_grad(h: np.ndarray, beta: float):
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    # correlation
    fh = np.fft.rfft(h, size)
    fg = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(fh * fg, size)[: 2 * n - 1]
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    w = exp_bc / sum_exp               # softmax weights, length 2n‑1
    f = (np.log(sum_exp) + m) / beta   # smooth max value
    # gradient via convolutions
    b = 1.0 - h
    fw = np.fft.rfft(w, size)
    term1_full = np.fft.irfft(fw * np.fft.rfft(b, size), size)[: 2 * n - 1]
    term1 = term1_full[(n - 1) : (n - 1) + n]
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, size)
    term2_full = np.fft.irfft(fw_rev * np.fft.rfft(h, size), size)[: 2 * n - 1]
    term2 = term2_full[(n - 1) : (n - 1) + n]
    grad = term1 - term2
    return f, grad

# Binary simulated annealing (many short runs)
def binary_sa_multi(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    best_h = None
    best_c = np.inf
    target = n // 2
    inner_limit = 10.0  # seconds per SA run

    while time.time() - start < time_budget:
        # random binary start
        h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, target, replace=False)
        h[ones_idx] = 1.0
        c = compute_c5_fft(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= 0.3808:
                return best_h, best_c

        ones = list(ones_idx)
        zeros = [i for i in range(n) if h[i] == 0.0]

        run_start = time.time()
        while time.time() - start < time_budget and time.time() - run_start < inner_limit:
            # temperature schedule (geometric)
            frac = (time.time() - start) / time_budget
            T = 0.02 * (1e-6 / 0.02) ** frac

            i = rng.choice(ones)   # current 1
            j = rng.choice(zeros)  # current 0

            # propose swap
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5_fft(h)
            delta = new_c - c
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                # accept
                c = new_c
                # update index lists
                idx_i = ones.index(i)
                ones[idx_i] = j
                idx_j = zeros.index(j)
                zeros[idx_j] = i
                if c < best_c:
                    best_c, best_h = c, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                # reject: revert
                h[i] = 1.0
                h[j] = 0.0

    if best_h is None:
        h = np.zeros(n)
        h[:target] = 1.0
        best_h = h
        best_c = compute_c5_fft(best_h)
    return best_h, best_c

# Pure hill‑climb on binary vectors (only improving swaps)
def binary_hill_climb(h_start: np.ndarray, rng: np.random.Generator, time_budget: float):
    start = time.time()
    n = h_start.size
    h = (h_start > 0.5).astype(float)
    c = compute_c5_fft(h)
    best_h, best_c = h.copy(), c
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))

    while time.time() - start < time_budget:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5_fft(h)
        if new_c < c:
            c = new_c
            idx_i = ones.index(i)
            ones[idx_i] = j
            idx_j = zeros.index(j)
            zeros[idx_j] = i
            if c < best_c:
                best_c, best_h = c, h.copy()
                if best_c <= 0.3808:
                    break
        else:
            h[i] = 1.0
            h[j] = 0.0
        if best_c <= 0.3808:
            break
    return best_h, best_c

def gradient_descent_smoothing(h0: np.ndarray, rng: np.random.Generator,
                               time_budget: float, target: float):

    h = h0.copy()
    n = h.size
    best_h, best_c = h.copy(), compute_c5_fft(h)
    it = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget

        h -= step * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_to_constraints(h, target, rng)

            if c < best_c:

                if best_c <= 0.3808:
                    break
