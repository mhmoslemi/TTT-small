def simulated_annealing(bits, rng, time_deadline):
    cur_bits = bits.copy()
    cur_score = evaluate_bits(cur_bits)
    best_bits = cur_bits.copy()
    best_score = cur_score
    N_iter = 0
    # hyperparameters:
    T0 = 0.02
    alpha = 0.99995
    T = T0
    while time.time() < time_deadline:
        i = rng.integers(0, half)
        # flip bit i
        cur_bits[i] = 1 - cur_bits[i]  # toggle
        # evaluate
        new_score = evaluate_bits(cur_bits)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_bits = cur_bits.copy()
        else:
            # revert
            cur_bits[i] = 1 - cur_bits[i]
        # update temperature
        T *= alpha
        if T < 1e-9:
            T = 1e-9
        N_iter += 1
    return best_bits, best_score, N_iter
```

    best = cur.copy()
    best_score = cur_score
    N = cur.size

    alpha = 0.99995
    while time.time() < deadline:
        # select indices i,j with opposite bits
        i = rng.integers(N)
        j = rng.integers(N)
        if cur[i] == cur[j]:
            continue
        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_score = evaluate_binary(cur)
        delta = new_score - cur_score
        if delta <= T:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]
        T *= alpha
        if T < 1e-9:
            T = 0.0
    return best, best_score
```

        # binary selection of top N_up/2 values
        target = N_up // 2
        idx = np.argpartition(h_up_f, -target)[-target:]
        h_up_int = np.zeros(N_up, dtype=np.int8)
        h_up_int[idx] = 1
        # local search at higher resolution

        if c_up_best < best_c:
            best_c, best_h = c_up_best, h_up_best
            N = N_up

    # final compute exact C
    final_c = compute_c5_from_h(best_h.astype(np.float64))

    return best_h.astype(np.float64).tolist(), float(final_c), int(N)

import time
import numpy as np

def _c5_from_float(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _greedy_improve(h_int, stop):
    """Iteratively swap a 1 with a 0 to improve the worst shift."""
    n = h_int.shape[0]
    while time.time() < stop:
        h = h_int.astype(np.float64)
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        k = abs(shift)
        if k == 0:
            break
        neigh = h_int + np.roll(h_int, k)
        ones = np.flatnonzero(h_int)
        zeros = np.flatnonzero(h_int == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        min_one = neigh[ones].min()
        i_cand = ones[neigh[ones] == min_one]
        i = i_cand[0]
        max_zero = neigh[zeros].max()
        j_cand = zeros[neigh[zeros] == max_zero]
        j = j_cand[0]
        if neigh[i] < neigh[j]:
            h_int[i] = 0
            h_int[j] = 1
            continue
        break
    return h_int

def _local_search(start_h, rng, deadline):
    """Greedy improve + random perturbations until time runs out."""
    best_h = start_h.copy()
    best_c = _c5_from_float(best_h.astype(np.float64))
    cur = start_h.copy()
    while time.time() < deadline:
        cur = _greedy_improve(cur, deadline)
        cur_c = _c5_from_float(cur.astype(np.float64))
        if cur_c < best_c:
            best_c, best_h = cur_c, cur.copy()
        if time.time() >= deadline:
            break
        # random perturbation: swap a random 1 with a random 0
        ones = np.flatnonzero(cur)
        zeros = np.flatnonzero(cur == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i], cur[j] = 0, 1
    return best_h, best_c
