def directed_swap(b, rng):
    n = b.size
    corr = np.correlate(b, 1 - b, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    # Determine slices
    if shift >= 0:
        a = b[:n - shift]
        b2 = b[shift:]
        offset_a = 0
        offset_b2 = shift
    else:
        s = -shift
        a = b[s:]
        b2 = b[:n - s]
        offset_a = s
        offset_b2 = 0
        shift = s
    # Find masks
    mask1 = (a == 1) & (b2 == 0)
    mask2 = (a == 0) & (b2 == 1)
    if mask1.any() and mask2.any():
        idx1 = rng.choice(np.flatnonzero(mask1))
        idx2 = rng.choice(np.flatnonzero(mask2))
        i = offset_a + idx1

import time, numpy as np
from scipy.optimize import minimize, LinearConstraint

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute to achieve Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb moving mass from positive to negative gradient entries."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using worst‑shift gradient with diminishing step."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP minimisation of smooth max (log‑sum‑exp) with fixed β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # target resolution
    target_n = 1024
    # initialise from parent if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # upsample / downsample to target_n
        if h0.size != target_n:
            x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
            h0 = np.interp(x_new, x_old, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c = h0, compute_c5(h0)
    else:
        h0 = random_feasible_vector(target_n, rng)
        best_h, best_c = h0, compute_c5(h0)

    # ---- main stochastic optimisation loop ----
    # allocate ~85% of time to exploration, rest to final polishing
    explore_deadline = start + 0.85 * budget_s
    while time.time() < explore_deadline:
        # decide whether to restart or perturb current best
        fresh = rng.random() < 0.15
        if fresh:
            h = random_feasible_vector(target_n, rng)
        else:
            # slight random perturbation of best_h
            h = best_h.copy()
            # pick a few random indices and jitter them within bounds
            for _ in range(3):
                i = rng.integers(0, target_n)
                delta = (rng.random() - 0.5) * 0.2
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h = enforce_sum(h, rng)

        # aggressive hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=2500, step_scale=0.7)

        # SLSQP passes with increasing β
        for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
            if time.time() > explore_deadline:
                break
            # allocate a few hundred iter for larger β
            iters = 800 if beta >= 200 else 500
            h = slsqp_optimize(h, beta, max_iter=iters)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

        # fine gradient tidy‑up
        h = gradient_refine(h, rng, step=0.015, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ---- final polishing -------------------------------------------------
    # up‑sample if we ever dropped below target_n (not the case here)
    if best_h.size < target_n:
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        best_h = np.interp(x_new, x_old, best_h)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)

    # final stochastic tweaks until time runs out
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
