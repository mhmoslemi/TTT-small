    return np.correlate(h, 1-h, mode='full')

def compute_c5(h):
    corr = compute_corr(h)
    return float(np.max(corr) * 2.0 / h.size)

def gradient_smooth_max(h, beta):
    n = h.size
    corr = compute_corr(h)
    # weights
    scaled = beta * corr
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = np.sum(w)
    w /= w_sum
    # compute grad
    dx = 2.0 / n
    grad = np.zeros(n, dtype=float)
    # iterate over k
    for k in range(-n+1, n):
        wk = w[k + n - 1]
        if wk == 0.0:
            continue
        if k == 0:
            grad += wk * 2 * h
        else:
            # roll both directions
            roll_minus = np.roll(h, -k)
            roll_plus = np.roll(h, k)
            grad += wk * (roll_minus + roll_plus)
    grad *= -dx
    return grad

def project_box_sum(h):
    # Clamp to [0,1] then adjust sum via simple redistribution.
    h = np.clip(h, 0.0, 1.0)
    target = h.size / 2.0
    current = np.sum(h)
    diff = target - current
    if abs(diff) < 1e-12:
        return h
    # choose direction
    if diff > 0:
        # Need to increase sum: add diff equally to all indices that are not at 1, up to 1.
        # Simple loop adding small epsilon until done.
        # Use while diff>0 and there is capacity
        # We'll use np.where each iteration; but diff may be small.

        cap = 1.0 - h

        if total_cap <= 0:

def project_box_sum(h):
    # h: np.ndarray float
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    # tolerance
    tol = 1e-12
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        if diff > 0:
            cap = 1.0 - h
        else:
            cap = h
        total_cap = np.sum(cap)
        if total_cap == 0:
            break
        # distribute diff proportionally to capacities
        delta = diff * (cap / total_cap)
        h += delta
        diff = target - np.sum(h)
    # final clip to be safe
    np.clip(h, 0.0, 1.0, out=h)
    return h

def random_swap_local(h, steps, rng):
    n = h.size
    best_h = h
    best_c = compute_c5(h)
    for _ in range(steps):
        i, j = rng.choice(n, 2, replace=False)
        if h[i] == 1.0 and h[j] == 0.0:
            continue  # cannot move
        # direction: move small amount from i to j
        max_move = min(1.0 - h[i], h[j])
        if max_move <= 0:
            continue
        delta = max_move * rng.random()
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new
    return best_h

def gradient_smooth_max(h, beta):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    # compute weights
    scaled = beta * corr
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / np.sum(exp_scaled)
    # compute term2 (i - k)
    # convolution with w
    N = n + (2*n - 1) - 1
    # Compute FFT of h and w (both zero-padded to N)
    H = np.fft.fft(h, N)
    W = np.fft.fft(w, N)
    conv = np.fft.ifft(H * W).real
    term2 = conv[n-1:n-1+n]  # length n
    # compute term1 (i + k) via w_rev
    w_rev = w[::-1]
    W_rev = np.fft.fft(w_rev, N)
    conv_rev = np.fft.ifft(H * W_rev).real
    term1 = conv_rev[n-1:n-1+n]
    grad = - (2.0 / n) * (term1 + term2)  # dx = 2/n
    return grad

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discretised step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _project_box_sum(h: np.ndarray) -> np.ndarray:
    """Clamp to [0,1] and adjust to satisfy Σh = n/2 exactly."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    tol = 1e-12
    max_iter = 500
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        if diff > 0:
            cap = 1.0 - h
        else:
            cap = h
        total_cap = np.sum(cap)
        if total_cap == 0:
            break
        delta = diff * (cap / total_cap)
        h += delta
        diff = target - np.sum(h)
    # final safety clip
    np.clip(h, 0.0, 1.0, out=h)
    return h


def _gradient_smooth_max(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of β⁻¹ log‑sum‑expₖ β·corrₖ (corrₖ = Σ h[i](1‑h[i+k]))."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)                     # wₖ ≥ 0, Σ wₖ = 1

    # convolution with w (term for i‑k) and with reversed w (term for i+k)
    N = 3 * n - 2                       # length of full linear convolution
    H = np.fft.fft(h, N)
    W = np.fft.fft(w, N)
    conv2 = np.fft.ifft(H * W).real      # conv with w
    term2 = conv2[n - 1 : n - 1 + n]    # Σₖ wₖ h_{i‑k}

    W_rev = np.fft.fft(w[::-1], N)
    conv1 = np.fft.ifft(H * W_rev).real  # conv with w reversed
    term1 = conv1[n - 1 : n - 1 + n]    # Σₖ wₖ h_{i+k}

    grad = -dx * (term1 + term2)        # ∂/∂h_i
    return grad


def _random_swap_local(h: np.ndarray, steps: int, rng: np.random.Generator) -> np.ndarray:
    """Small two‑index swaps that keep the sum constant."""
    n = h.size
    best_h = h.copy()
    best_c = _compute_c5(best_h)
    for _ in range(steps):
        i, j = rng.choice(n, 2, replace=False)
        # direction: move from i to j (or vice‑versa) whichever is feasible
        max_move = min(1.0 - h[i], h[j])
        if max_move <= 0:
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 0:
                continue
            delta = max_move * rng.random()
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            delta = max_move * rng.random()
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        c_new = _compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new
    return best_h


def _optimize_n(n: int, init_h: np.ndarray, rng: np.random.Generator,
                deadline: float) -> tuple[np.ndarray, float]:
    """Continuation on β with projected gradient descent."""
    h = init_h.copy()
    best_h = h.copy()
    best_c = _compute_c5(best_h)

    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        step = 0.5 / np.sqrt(beta)
        max_iter = int(2000 / (beta ** 0.3)) + 1
        for it in range(max_iter):
            if time.time() >= deadline:
                break
            grad = _gradient_smooth_max(h, beta)
            grad -= grad.mean()               # keep the sum invariant
            h = h - step * grad
            h = _project_box_sum(h)

            # occasional evaluation
            if (it & 7) == 0:
                c_val = _compute_c5(h)
                if c_val < best_c:
                    best_c = c_val
                    best_h = h.copy()

        # small jitter around the current best to avoid stagnation
        if time.time() >= deadline:
            break
        jitter = rng.normal(scale=0.02, size=n)
        h = best_h + jitter
        h = _project_box_sum(h)
        c_val = _compute_c5(h)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()

    # final local refinement
    if time.time() < deadline:
        best_h = _random_swap_local(best_h, steps=2000, rng=rng)
        best_c = _compute_c5(best_h)

    return best_h, best_c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    global_best_h = None
    global_best_c = np.inf
    global_best_n = None

    # try the two standard discretisations
    candidate_ns = [360, 720]

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        # initialise h for this n
        if "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
            if parent.size == n:
                h0 = parent.copy()
            else:
                h0 = rng.random(n)
        else:
            h0 = rng.random(n)

        h0 = _project_box_sum(h0)
        # quick sanity evaluation
        c0 = _compute_c5(h0)
        if c0 < global_best_c:
            global_best_c = c0
            global_best_h = h0.copy()
            global_best_n = n

        # run the continuation optimiser
        h_opt, c_opt = _optimize_n(n, h0, rng, deadline)

        if c_opt < global_best_c:
            global_best_c = c_opt
            global_best_h = h_opt.copy()
            global_best_n = n

    # final polishing – ensure constraints are met exactly
    if global_best_h is None:
        # fallback: simple half‑ones
        n = 360
        global_best_n = n
        global_best_h = np.zeros(n, dtype=float)
        global_best_h[: n // 2] = 1.0
    else:
        global_best_h = np.clip(global_best_h, 0.0, 1.0)
        global_best_h = _project_box_sum(global_best_h)

    final_c = _compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(global_best_n)
