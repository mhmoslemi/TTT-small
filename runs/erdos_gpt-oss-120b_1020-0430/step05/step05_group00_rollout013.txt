def compute_c5_fft(h):
    N = len(h)
    a = h
    b = 1 - h
    L = 1 << (2*N - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)
    # conv length L
    # O_k values: conv[N-1: N-1 + N]
    vals = conv[N-1: N-1+N]
    # max of O_k
    c5 = np.max(vals) * (2.0 / N)
    return c5

def binary_sa_fft(n, rng, time_budget):
    start = time.time()
    best_h = None
    best_c = np.inf
    # Pre-calc time limit
    while time.time() - start < time_budget:
        # random binary start
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n//2, replace=False)
        h[ones] = 1.0
        c = compute_c5_fft(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            # break early if sufficiently good (maybe < 0.3808)
            if best_c <= 0.3808:
                return best_h, best_c
        # inner loop with SA
        T0 = 0.02
        T_end = 1e-4
        # maintain indices arrays
        ones_arr = np.copy(ones)
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        # Pre-calc exponential maybe done each iteration
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            # pick random swap
            i_idx = rng.integers(len(ones_arr))
            j_idx = rng.integers(len(zeros_arr))
            i = ones_arr[i_idx]
            j = zeros_arr[j_idx]
            # swap
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5_fft(h)
            delta = c_new - c
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c = c_new
                # update arrays

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def linear_corr_fft(h: np.ndarray) -> np.ndarray:
    n = h.size
    a = h
    b = 1.0 - h
    L = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)
    return conv[: 2 * n - 1]


def compute_c5_fft(h: np.ndarray) -> float:
    n = h.size
    corr = linear_corr_fft(h)
    vals = corr[n - 1 : n - 1 + n]  # O_k for k=0..n-1
    return np.max(vals) * (2.0 / n)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = linear_corr_fft(h)
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    grad[: n - k] += 1.0 - h[k:]
    grad[k:] -= h[: n - k]
    return grad


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = linear_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    scale = step_scale
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5_fft(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % 200 == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = n / 2.0 - h.sum()
    return h


def binary_sa_fft(n: int, rng: np.random.Generator, time_budget: float):
    start = time.time()
    best_h = None
    best_c = np.inf
    T0, T_end = 0.02, 1e-4
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        c_curr = compute_c5_fft(h)
        if c_curr < best_c:
            best_c, best_h = c_curr, h.copy()
            if best_c <= 0.3808:
                return best_h, best_c
        ones_arr = ones.copy()
        zeros_arr = np.setdiff1d(np.arange(n), ones_arr)
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            T = T0 * (T_end / T0) ** (elapsed / time_budget)
            i_pos = rng.integers(len(ones_arr))
            j_pos = rng.integers(len(zeros_arr))
            i = ones_arr[i_pos]
            j = zeros_arr[j_pos]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5_fft(h)
            delta = c_new - c_curr
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c_curr = c_new
                ones_arr[i_pos] = j
                zeros_arr[j_pos] = i
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        return best_h, best_c
            else:
                h[i] = 1.0
                h[j] = 0.0
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_c = compute_c5_fft(best_h)

    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
