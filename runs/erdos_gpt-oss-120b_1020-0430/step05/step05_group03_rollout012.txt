def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # if diff positive (need add mass), add from those not at 1
    # else subtract from those not at 0.
    while abs(diff) > 1e-12:
        if diff > 0:
            # choose index where h < 1
            candidates = np.where(h < 1 - 1e-12)[0]
            if candidates.size == 0:
                break
            i = rng.choice(candidates)
            add = min(diff, 1 - h[i])
            h[i] += add

            if candidates.size == 0:
                break
            i = rng.choice(candidates)
            sub = min(-diff, h[i])

            diff += sub
    return h

import time
import numpy as np

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple recursive redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1 - 1e-12)[0]
            if not cand.size:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            cand = np.where(h > 1e-12)[0]
            if not cand.size:
                break
            i = rng.choice(cand)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h


def worst_gradient(h: np.ndarray):
    """Gradient of the worst‑shift overlap and its value."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = corr.max()
    idx = np.argmax(corr) - (n - 1)          # shift index (can be negative)
    k = idx
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:                                   # k < 0
        kk = -k
        grad[kk:] -= h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad, k, max_corr * (2.0 / n)


def big_move(h: np.ndarray, grad: np.ndarray, factor: float):
    """Transfer mass from positive‑gradient to negative‑gradient entries.
    `factor` ∈ (0,1] scales the total transferred amount."""
    pos = grad > 1e-12
    neg = grad < -1e-12
    total_pos = h[pos].sum()
    total_neg = (1.0 - h[neg]).sum()
    D_max = min(total_pos, total_neg)
    if D_max <= 0:
        return None
    D = D_max * factor
    h_new = h.copy()
    if total_pos > 0:
        h_new[pos] -= D * (h[pos] / total_pos)
    if total_neg > 0:
        h_new[neg] += D * ((1.0 - h[neg]) / total_neg)
    return np.clip(h_new, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)

                    step: float = 0.01, max_iter: int = 10) -> np.ndarray:
    """Few gentle gradient steps after main search."""

        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h
