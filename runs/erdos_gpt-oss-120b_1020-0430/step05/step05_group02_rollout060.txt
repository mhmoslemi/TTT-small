def smooth_max_and_grad(h, beta):
    n = h.size
    g = np.correlate(h, 1.0 - h, mode='full')
    # g_len = 2*n-1
    max_g = np.max(g)
    # compute stable log-sum-exp
    # Compute w via exponentials; use np.exp(beta*(g - max_g)), then normalize.
    exp_term = np.exp(beta * (g - max_g))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    # gradient
    # compute convolution/correlation
    # forward sum: sum_i w_i * h_{k+i}
    forward_corr = np.correlate(w, h, mode='full')
    # backward sum: sum_i w_i * h_{k-i} = sum_i w_{-i} * h_{k+i} = correlation with reversed w
    backward_corr = np.correlate(w[::-1], h, mode='full')
    # offsets
    offset = n - 1
    grad = 1.0 - forward_corr[offset:offset + n] - backward_corr[offset:offset + n]
    # Ensure gradient shape n
    # Now enforce equality constraint: we need to stay on hyperplane sum = n/2. The gradient of smoothed max w.r.t h is as above. For projected gradient descent, we subtract the mean of grad to make it orthogonal to vector of ones. That ensures step does not change sum (first order).
    grad_proj = grad - np.mean(grad)
    # also enforce box constraints with projection after step.
    return max_g, grad, grad_proj

def gradient_refine_simple(h, rng, max_iter=12, step0=0.02):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        # compute correlation g = np.correlate(h, 1-h, mode='full')
        g = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(g) - (n - 1)
        k = abs(idx)
        # compute gradient vector gvec
        gvec = np.zeros_like(h)
        if k == 0:
            gvec[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            # h[:upper] vs h[k:]
            # contribution: for j from 0 to upper-1:
            # derivative w.r.t h_j: (1 - h_{j+k}) if j+k < n else 1 (since h_{j+k}=0)
            # but easier: compute using formula: for j in 0..n-1:
            # derivative = (1 - h_{j+k}) if j+k < n else 1   - (h_{j-k} if j-k>=0 else 0)
            # This is equivalent to:
            # gvec[:upper] += 1 - h[k:]
            # gvec[upper:] += 1   (since j+k out of bound)
            # then subtract h[:n-k] from gvec[k:] .
            # Implement:
            gvec[:upper] += 1.0 - h[k:]
            gvec[upper:] += 1.0
            gvec[k:] -= h[:upper]
        step = step0 * (0.5 ** it)
        h = h - step * gvec
        # clip
        np.clip(h, 0.0, 1.0, out=h)
        # adjust sum to target
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust using random entries
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for the discrete vector h."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Exact projection of h onto the hyperplane sum(h)=n/2 while respecting [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            delta = min(diff, 1.0 - h[i])
        else:
            delta = max(diff, -h[i])
        h[i] += delta
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h


def gradient_refine_simple(h: np.ndarray, rng: np.random.Generator,
                           max_iter: int = 12, step0: float = 0.02) -> np.ndarray:
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        shift = abs(idx)

        # gradient = 1 - h_{k+shift} - h_{k-shift}
        grad = np.full(n, 1.0)
        if shift < n:
            grad[: n - shift] -= h[shift:]
        if shift > 0:
            grad[shift:] -= h[: n - shift]

        step = step0 * (0.5 ** it)
        h = h - step * grad
        np.clip(h, 0.0, 1.0, out=h)

        # enforce exact sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                delta = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += delta
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def smoothed_max_descent(h: np.ndarray, rng: np.random.Generator,
                         deadline: float) -> np.ndarray:
    """Descent on the log‑sum‑exp surrogate of the max‑overlap."""
    n = h.size
    target = n / 2.0
    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]

    for beta in betas:
        if time.time() > deadline:
            break
        # iteration budget scaled with n
        if n <= 360:
            it_max = 80
        elif n <= 720:
            it_max = 40
        else:
            it_max = 20
        for it in range(it_max):
            if time.time() > deadline:
                break

            # correlation g_i = ∫ h (1‑h shifted)
            g = np.correlate(h, 1.0 - h, mode="full")
            max_g = np.max(g)                     # for stable weights
            exp_term = np.exp(beta * (g - max_g))
            w = exp_term / np.sum(exp_term)

            # forward and backward weighted sums
            fwd = np.correlate(w, h, mode="full")
            bwd = np.correlate(w[::-1], h, mode="full")
            off = n - 1
            grad = 1.0 - fwd[off:off + n] - bwd[off:off + n]

            # project onto sum‑hyperplane
            grad -= np.mean(grad)

            # step size, decreasing with iteration
            step = (1.0 / (beta + 1.0)) * (0.7 ** it)
            h_new = h - step * grad
            np.clip(h_new, 0.0, 1.0, out=h_new)

            # exact sum correction
            diff = target - h_new.sum()
            if abs(diff) > 1e-12:
                for _ in range(200):
                    i = rng.integers(0, n)
                    if diff > 0:
                        delta = min(diff, 1.0 - h_new[i])
                    else:
                        delta = max(diff, -h_new[i])
                    if delta == 0:
                        continue
                    h_new[i] += delta
                    diff = target - h_new.sum()
                    if abs(diff) < 1e-12:
                        break
                if abs(diff) > 1e-12:
                    adjust_sum_exact(h_new, rng)

            h = h_new
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.97 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # --- candidate sizes -------------------------------------------------
    base_ns = [360, 240, 180]

    for n in base_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        candidates = []

        # parent if supplied
        if "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            candidates.append(interp)                     # fractional
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            candidates.append(bin_vec)                    # binary

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        candidates.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        candidates.append(rand_frac)

        # alternating 1‑0 pattern
        alt = np.tile([1.0, 0.0], n // 2)
        candidates.append(alt)

        for h in candidates:
            if time.time() > deadline:
                break
            h = np.clip(h, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # simple gradient sweep
            h = gradient_refine_simple(h, rng, max_iter=12, step0=0.02)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # smoothed‑max descent
            h = smoothed_max_descent(h, rng, deadline)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # --- up‑sampling phase ------------------------------------------------
    up_sizes = [720, 1080]
    for n_up in up_sizes:
        if best_h is None or time.time() > deadline:
            break
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        h_up = gradient_refine_simple(h_up, rng, max_iter=12, step0=0.015)
        h_up = smoothed_max_descent(h_up, rng, deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # --- fallback (should never be needed) -------------------------------
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
