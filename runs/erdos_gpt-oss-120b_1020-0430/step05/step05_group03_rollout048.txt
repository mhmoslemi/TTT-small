def project_to_feasible(h, target):
    # clip to [0,1]
    # if sum already equal target, return.
    # else binary search for lambda in low/high.
    lo = -np.max(h) - 1.0  # safe lower bound to push all to 0
    hi = 1.0 - np.min(h) + 1.0  # push all to 1
    for _ in range(50):
        lam = (lo + hi) / 2
        hp = np.clip(h + lam, 0.0, 1.0)
        s = hp.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(h + (lo+hi)/2, 0.0, 1.0)

Implementation of grad:

def compute_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    maxc = np.max(corr)
    w = np.exp(beta * (corr - maxc))
    w /= w.sum()
    grad = np.zeros(n)
    mid = n - 1
    # vectorized loop: iterate offset
    for offset in range(-(n-1), n):
        ws = w[offset + mid]
        if ws == 0.0:
            continue
        if offset >= 0:
            k = offset
            grad[:n-k] += ws * (1 - h[k:])
            grad[k:]   += ws * (-h[:n-k])
        else:
            t = -offset
            grad[t:]   += ws * (1 - h[:n-t])
            grad[:n-t] += ws * (-h[t:])
    return grad

def gradient_descent(h0, rng, beta_seq, steps_seq, step_size_base=0.02):
    h = h0.copy()
    target = h.size / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta, steps in zip(beta_seq, steps_seq):
        step_size = step_size_base / np.sqrt(beta)  # adapt
        v = np.zeros_like(h)
        for it in range(steps):
            grad = compute_grad(h, beta)
            v = 0.9 * v - step_size * grad
            h_new = h + v
            h_new = project_to_feasible(h_new, target)
            # occasionally random perturb
            if rng.random() < 0.005:
                # random swap: pick i,j and swap small epsilon
                i = rng.integers(0, h_new.size)
                j = rng.integers(0, h_new.size)
                eps = min(h_new[i], 1.0 - h_new[j]) * 0.5
                if eps > 1e-6:
                    h_new[i] -= eps
                    h_new[j] += eps
            # evaluate
            c = compute_c5(h_new)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c

import time
import numpy as np

# --------------------------------------------------------------
# basic utilities
# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of h onto {0≤h≤1, sum h = target}."""
    # Clip first to keep within bounds; the sum may still be off.
    hp = np.clip(h, 0.0, 1.0)
    s = hp.sum()
    if abs(s - target) <= 1e-12:
        return hp
    # binary search for lambda so that sum clip(h+λ) = target
    lo = -np.max(h) - 1.0   # guarantee all become 0
    hi = 1.0 - np.min(h) + 1.0  # guarantee all become 1
    for _ in range(50):
        lam = (lo + hi) * 0.5
        hp = np.clip(h + lam, 0.0, 1.0)
        s = hp.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


# --------------------------------------------------------------
# gradient of smoothed max (log‑sum‑exp)
# --------------------------------------------------------------
def compute_grad(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/β)·log‑sum‑exp(β·corr) w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    maxc = np.max(corr)
    w = np.exp(beta * (corr - maxc))
    w /= w.sum()
    grad = np.zeros(n, dtype=float)
    mid = n - 1
    for offset in range(-(n - 1), n):
        ws = w[offset + mid]
        if ws == 0.0:
            continue
        if offset >= 0:
            k = offset
            grad[: n - k] += ws * (1.0 - h[k:])
            grad[k:]      += ws * (-h[: n - k])
        else:
            t = -offset
            grad[t:]      += ws * (1.0 - h[: n - t])
            grad[: n - t] += ws * (-h[t:])
    return grad


# --------------------------------------------------------------
# simple momentum gradient descent with continuation on β
# --------------------------------------------------------------
def gradient_descent(
    h0: np.ndarray,
    rng: np.random.Generator,
    beta_seq,
    steps_seq,
    start_time,
    budget,
    step_base: float = 0.03,
):
    h = h0.copy()
    target = h.size / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta, steps in zip(beta_seq, steps_seq):
        step = step_base / np.sqrt(beta)
        v = np.zeros_like(h)
        for _ in range(steps):
            if time.time() - start_time > budget * 0.95:
                return best_h, best_c
            grad = compute_grad(h, beta)
            v = 0.9 * v - step * grad
            h_new = project_to_feasible(h + v, target)

            # occasional tiny random transfer to escape plateaus
            if rng.random() < 0.004:
                i = rng.integers(0, h_new.size)
                j = rng.integers(0, h_new.size)
                eps = min(h_new[i], 1.0 - h_new[j]) * 0.4
                if eps > 1e-8:
                    h_new[i] -= eps
                    h_new[j] += eps
                    h_new = project_to_feasible(h_new, target)

            c = compute_c5(h_new)
            if c < best_c:
                best_c, best_h = c, h_new.copy()
            h = h_new
    return best_h, best_c


# --------------------------------------------------------------
# local hill‑climb based on the worst‑shift gradient
# --------------------------------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the overlap for the shift attaining max C5."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)   # signed shift
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]      -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move a small ε from a most‑positive‑gradient index to a most‑negative."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.5):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    target = n / 2.0
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        # gradual step‑size decay
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    # ensure feasibility
    best_h = project_to_feasible(best_h, target)
    return best_h, best_c
