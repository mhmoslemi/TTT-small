def compute_c5(h):
   n = h.size
   corr = np.correlate(h, 1.0 - h, mode='full')
   return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final roundoff
    if abs(diff) > 1e-12:
        # distribute remaining diff evenly
        h += diff / n
        # clip again and adjust
        h = np.clip(h, 0.0, 1.0)
        # recompute diff
        diff = target - h.sum()
        # if still off, do simple correction on a single index
        if abs(diff) > 1e-12:
            i = rng.integers(0, n)
            h[i] += diff
            h[i] = max(0.0, min(1.0, h[i]))
    return h

CURRENT_TARGET = None
CURRENT_BETA = None
CURRENT_LAMBDA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def smooth_obj(x):
    # smooth max + lambda binary penalty
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # logsumexp
    smooth_max = (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    bin_pen = CURRENT_LAMBDA * np.sum(x * (1.0 - x))
    return smooth_max + bin_pen

def slsqp_optimize(h0, beta, lam, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA, CURRENT_LAMBDA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    CURRENT_LAMBDA = lam
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(smooth_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple correction: adjust random indices
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # + (1 - h[i+k])
            g[k:] -= h[:upper]               # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust diff by random swapping until exact
            for _ in range(200):

def compute_c5(h):

def adjust_sum_exact(h, rng):
    """Force Σh = n/2 while keeping 0≤h≤1 (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final safety
    if abs(diff) > 1e-12:

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:

CURRENT_TARGET = None
CURRENT_BETA = None
CURRENT_LAMBDA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    smooth_max = (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
    bin_pen = CURRENT_LAMBDA * np.sum(x * (1.0 - x))
    return smooth_max + bin_pen

def slsqp_optimize(h0, beta, lam, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA, CURRENT_LAMBDA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    CURRENT_LAMBDA = lam
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    # parent (if any)
    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
    else:
        n0 = 360

    # candidate sizes (even only)
    cand = [n0]
    half = n0 // 2
    if half >= 120:
        cand.append(half)
    if 240 not in cand:
        cand.append(240)
    cand = sorted({n if n % 2 == 0 else n + 1 for n in cand}, reverse=True)

    best_h, best_c, best_n = None, np.inf, None

    # schedule: (beta, lambda, max_iter)
    schedule = [
        (5, 0.0, 200),
        (20, 0.0, 200),
        (80, 0.0, 200),
        (200, 0.0, 200),
        (500, 5.0, 200),
        (1000, 10.0, 200),
        (2000, 20.0, 200),
        (5000, 40.0, 200),
    ]

    for n in cand:
        if time.time() > deadline:
            break

        # preparation of starting vectors
        starts = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            starts.append(np.interp(xt, xs, parent))

        starts.append(random_frac(n, rng))
        starts.append(sinusoid_frac(n, rng))

        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand.astype(float))

        for base_h in starts:
            if time.time() > deadline:
                break
            h = base_h.copy()
            adjust_sum_exact(h, rng)
            h = np.clip(h, 0.0, 1.0)

            # SLSQP cascade
            for beta, lam, itmax in schedule:
                if time.time() > deadline:
                    break
                h = slsqp_optimize(h, beta, lam, rng, max_iter=itmax)

            # final gradient refinement
            h = gradient_refine(h, rng, step=0.02, max_iter=30)

            # evaluate fractional candidate
            c_frac = compute_c5(h)
            if c_frac < best_c:
                best_h, best_c, best_n = h.copy(), c_frac, n
                if best_c <= 0.3808:
                    return best_h.tolist(), float(best_c), int(best_n)

            # binary rounding & greedy improvement
            idx = np.argpartition(h, -n // 2)[-n // 2:]
            h_bin = np.zeros(n)
            h_bin[idx] = 1.0
            adjust_sum_exact(h_bin, rng)
            h_bin = greedy_improve(h_bin, rng, max_iter=200, deadline=deadline)
            c_bin = compute_c5(h_bin)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin.copy(), c_bin, n
                if best_c <= 0.3808:
                    return best_h.tolist(), float(best_c), int(best_n)

    # optional up‑sampling of best binary solution
    if (
        best_h is not None
        and time.time() < deadline
        and best_n is not None
        and best_n * 2 <= 720
    ):
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        h_up = greedy_improve(h_up, rng, max_iter=300, deadline=deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # fallback if nothing found
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
