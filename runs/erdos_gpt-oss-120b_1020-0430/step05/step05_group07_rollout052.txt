def softmax_obj(x, beta):
    corr = np.correlate(x, 1 - x, mode='full')
    exponent = beta * corr
    m = np.max(exponent)
    exp_shift = np.exp(exponent - m)
    return (m + np.log(np.sum(exp_shift))) / beta

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Maximum overlap for discretized step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass so that sum(h) = n/2 while respecting bounds."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.03, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        elif idx >= 0:  # positive shift
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        else:  # negative shift
            upper = n - k
            g[:upper] += 1.0 - h[:upper]
            g[k:] -= h[k:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def softmax_obj(x, beta):
    """Smooth approximation of max_corr."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    exp_val = beta * corr
    m = np.max(exp_val)
    sum_exp = np.sum(np.exp(exp_val - m))
    return (m + np.log(sum_exp)) / beta

def softmax_grad(x, beta):
    """Gradient of the smooth max_corr."""
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    exp_val = beta * corr
    m = np.max(exp_val)
    w = np.exp(exp_val - m)
    w_sum = np.sum(w)
    w_norm = w / w_sum
    g = np.zeros_like(x)
    for idx in range(2 * n - 1):
        d = idx - (n - 1)
        wd = w_norm[idx]
        if d == 0:
            g += wd * (1.0 - 2.0 * x)
        elif d > 0:
            upper = n - d
            g[:upper] += wd * (1.0 - x[d:])
            g[d:] -= wd * x[:upper]
        else:
            d_abs = -d
            upper = n - d_abs
            g[d_abs:] += wd * (1.0 - x[:upper])
            g[:upper] -= wd * x[d_abs:]
    return g

def slsqp_optimize_analytic(h0, beta, rng, max_iter=800):
    """SLSQP using analytic gradient for the smooth max."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_cons = {
        "type": "eq",
        "fun": lambda x: np.sum(x) - target,
        "jac": lambda x: np.ones_like(x),
    }

    def fun(x):
        return softmax_obj(x, beta)

    def jac(x):
        return softmax_grad(x, beta)

    res = minimize(
        fun,
        h0,
        method="SLSQP",
        jac=jac,
        bounds=bounds,
        constraints=[eq_cons],
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x
