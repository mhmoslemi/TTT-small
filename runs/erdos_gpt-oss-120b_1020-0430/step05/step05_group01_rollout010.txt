def compute_c5_fft(h):
    n = len(h)
    L = 2*n - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)
    corr = np.fft.irfft(H * G, L)
    return np.max(corr) * (2.0 / n)

def binary_anneal(h, rng, time_budget):
    start = time.time()
    n = h.shape[0]
    current = h.copy()
    cur_c = compute_c5_fft(current)
    best = current.copy()
    best_c = cur_c

    # maintain lists
    ones = [int(i) for i in np.flatnonzero(current == 1)]
    zeros = [int(i) for i in np.flatnonzero(current == 0)]

    T = cur_c  # initial temperature
    Tmin = 1e-8

    while time.time() - start < time_budget:
        # random swap
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = ones[idx_i]
        j = zeros[idx_j]

        # perform swap
        current[i] = 0.0
        current[j] = 1.0

        new_c = compute_c5_fft(current)

        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            # accept
            cur_c = new_c
            # update lists
            ones[idx_i] = j
            zeros[idx_j] = i
            if cur_c < best_c:
                best_c = cur_c
                best = current.copy()
        else:
            # revert swap
            current[i] = 1.0
            current[j] = 0.0

        # cooling
        T = max(T * 0.9995, Tmin)

    return best, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def compute_c5_fft(h: np.ndarray) -> float:
    """Fast C5 using FFT (linear correlation)."""
    n = h.shape[0]
    L = 2 * n - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * G, L)
    return float(np.max(corr) * (2.0 / n))


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def binary_anneal(h_start: np.ndarray, rng: np.random.Generator,
                 time_budget: float) -> tuple[np.ndarray, float]:
    """Random swap annealing on a binary vector."""
    start = time.time()
    n = h_start.size
    cur = h_start.copy()
    cur_c = compute_c5_fft(cur)
    best = cur.copy()
    best_c = cur_c

    # maintain index lists
    ones = [int(i) for i in np.flatnonzero(cur == 1)]
    zeros = [int(i) for i in np.flatnonzero(cur == 0)]

    T = cur_c if cur_c > 0 else 1.0
    Tmin = 1e-8

    while time.time() - start < time_budget:
        # pick swap indices
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))

        j = zeros[idx_j]

        cur[i] = 0.0
        cur[j] = 1.0

        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        c0 = compute_c5_fft(h0)
        best_h, best_c = h0, c0

    # ----- binary annealing phase (≈70 % of budget) -----
    anneal_deadline = start + 0.70 * budget_s
    while time.time() < anneal_deadline:
        # random binary start
        h_bin = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, size=n // 2, replace=False)
        h_bin[ones_idx] = 1.0

        remaining = anneal_deadline - time.time()
        run_time = min(30.0, remaining)
        if run_time < 5.0:
            break

        h_best_bin, c_best_bin = binary_anneal(h_bin, rng, run_time)
        if c_best_bin < best_c:
            best_h, best_c = h_best_bin.copy(), c_best_bin

    # ----- continuous refinement phase (≈30 % of budget) -----
    time_left = start + budget_s - time.time()
    if time_left > 5.0 and best_h is not None:
        h_cont = best_h.astype(float)

        # gradient polish
        h_cont = gradient_refine(h_cont, rng, step=0.015, max_iter=15)
        c_cont = compute_c5_fft(h_cont)
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont

        # soft‑max SLSQP passes with increasing β
        for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0):
            if time.time() >= start + 0.99 * budget_s:
                break
            max_iter = 200 if beta < 200 else 150
            h_cont = slsqp_optimize(h_cont, beta, rng, max_iter=max_iter)
            c_cont = compute_c5_fft(h_cont)
            if c_cont < best_c:
                best_h, best_c = h_cont, c_cont

        # final fine‑grained gradient
        h_cont = gradient_refine(best_h, rng, step=0.005, max_iter=10)
        c_cont = compute_c5_fft(h_cont)
        if c_cont < best_c:
            best_h, best_c = h_cont, c_cont

    # ----- final sanity check -----
    if best_h is None:
        # fallback: random binary
        best_h = np.zeros(n, dtype=float)
        best_h[rng.choice(n, n // 2, replace=False)] = 1.0
        best_c = compute_c5_fft(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
