def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to add diff
            # indices where h < 1
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            add = min(diff, 1 - h[i])
            h[i] += add
            diff = target - h.sum()
        else:
            # need to subtract -diff

def compute_c5(h):
    """Exact C5 for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Modify h in-place so that sum(h) = len(h)/2 exactly."""
    target = h.size / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:

            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h
