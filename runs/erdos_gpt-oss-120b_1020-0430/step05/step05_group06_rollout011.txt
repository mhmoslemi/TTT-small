import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h):
    """C5 = max_k ∫h(x)(1-h(x+k))dx using discrete correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def compute_O(h):
    """Correlation vector O_k = Σ h_i (1-h_{i+k}) (unnormalized)."""
    return np.correlate(h, 1.0 - h, mode="full")

def adjust_sum_exact(h, rng):
    """Exact correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

# ----- smooth‑max SLSQP -------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

# ----- greedy binary construction -----
def greedy_binary(n, rng):
    """Simple greedy binary construction (half ones)."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h

# ----- max‑overlap evaluation for a tentative move -----
def compute_new_max(O, h, h_rev, src, dst, delta):
    """Maximum overlap after moving `delta` from src to dst (using current h)."""
    n = h.size
    cur = O.max()
    # src contributions
    s1 = O[n - 1 - src : n - 1 - src + n] - delta * (1 - h)
    s2 = O[src : src + n] + delta * h_rev
    # dst contributions
    d1 = O[n - 1 - dst : n - 1 - dst + n] + delta * (1 - h)
    d2 = O[dst : dst + n] - delta * h_rev
    m = cur
    tmp = s1.max()
    if tmp > m: m = tmp
    tmp = s2.max()
    if tmp > m: m = tmp
    tmp = d1.max()
    if tmp > m: m = tmp
    tmp = d2.max()
    if tmp > m: m = tmp
    return m

        # optional SLSQP on the refined 2n solution
        if time.time() < deadline - 6:
            h_opt2 = best_h2.copy()
            adjust_sum_exact(h_opt2, rng)
            for beta in (10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0):
                if time.time() > deadline - 6:
                    break
                try:
                    h_opt2 = slsqp_optimize(h_opt2, beta, rng, max_iter=200)
                except Exception:
                    continue
                c_val2 = compute_c5(h_opt2)
                if c_val2 < best_c:
                    best_c = c_val2
                    best_h = h_opt2.copy()
                    best_n = n2

    # ---- final safety adjustments ----
    if best_h is None:
        # fallback trivial solution
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_n = n
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
