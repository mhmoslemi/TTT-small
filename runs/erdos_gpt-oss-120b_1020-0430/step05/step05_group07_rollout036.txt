def generate_starts(n, rng, parent=None):
    starts = []
    # Binary random
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n//2, replace=False)] = 1.0
    starts.append(bin_rand)

    # Fractional uniform random scaled to sum target
    target = n/2.0
    rand_frac = rng.random(n)
    rand_frac *= target / rand_frac.sum()
    adjust_sum_exact(rand_frac, rng)
    starts.append(rand_frac)

    # Fractional square random (biased)
    rand_frac2 = rng.random(n)**2
    rand_frac2 *= target / rand_frac2.sum()
    adjust_sum_exact(rand_frac2, rng)
    starts.append(rand_frac2)

    # if parent provided (size parent_n):
    If parent is not None:
        # interpolate to n
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        starts.append(interp.copy())
        # binary version: keep top n/2 values to 1.
        idx = np.argpartition(interp, -n//2)[-n//2:]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        starts.append(bin_vec)

    # symmetrical twin: half block of 1's then zeros? E.g., first n/2 ones, rest zeros.
    block = np.concatenate([np.ones(n//2), np.zeros(n - n//2)])
    starts.append(block)

    return starts

import time
import numpy as np
from scipy.optimize import minimize

# ----- globals used by callbacks -----
CURRENT_BETA = None
CURRENT_TARGET = None

# ----- basic utilities -----
def compute_c5(h):
    """Maximum overlap integral for step function h (discrete)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Redistribute mass to satisfy Σh = n/2 exactly (keeping 0≤h≤1)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


# ----- smooth‑max objective and its gradient (used by SLSQP) -----
def smooth_max_objective(h):
    """Smooth approximation of max_k np.correlate(h,1-h)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_corr = np.max(corr)
    exp = np.exp(CURRENT_BETA * (corr - max_corr))
    sumexp = np.sum(exp)
    return float(max_corr + np.log(sumexp) / CURRENT_BETA)


def smooth_max_gradient(h):
    """Gradient of the smooth‑max w.r.t. h."""
    n = h.size
    dx = 2.0 / n
    # correlation without dx (used only to get same shape as weights)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    exp = np.exp(CURRENT_BETA * (corr - max_corr))
    sumexp = np.sum(exp)
    w = exp / sumexp  # softmax weights, sum to 1

    term1 = np.zeros(n)
    term2 = np.zeros(n)
    # shifts from -(n-1) … (n-1)
    for idx, s in enumerate(range(-(n - 1), n)):
        weight = w[idx]
        if weight == 0.0:
            continue
        if s >= 0:
            i_end = n - s
            term1[:i_end] += weight * (1.0 - h[s:i_end + s])
            term2[s:] += weight * h[:i_end]
        else:
            t = -s
            i_end = n - t
            term1[t:] += weight * (1.0 - h[:i_end])
            term2[:i_end] += weight * h[t:]
    return dx * (term1 - term2)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def sum_eq_jac(x):
    return np.ones_like(x)


# ----- gradient descent on the worst shift -----
def gradient_descent_worst(h, rng, max_iter=20, step0=0.04):
    """Projected gradient descent against the currently worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            i_end = n - k
            g[:i_end] = 1.0 - h[k:]
            g[k:] -= h[:i_end]
        step = step0 * (0.7 ** it)
        h -= step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ----- optimization driver for a single start vector -----
def optimize_h(h0, rng, deadline):
    """SLSQP with smooth‑max surrogate followed by worst‑shift GD."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    adjust_sum_exact(h, rng)

    global CURRENT_TARGET, CURRENT_BETA
    CURRENT_TARGET = target

    betas = [1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 5000.0]
    for beta in betas:
        if time.time() > deadline:
            break
        CURRENT_BETA = beta
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint, "jac": sum_eq_jac}
        max_iter = 80 if beta < 2000.0 else 150
        try:
            res = minimize(
                fun=smooth_max_objective,
                x0=h,
                method="SLSQP",
                jac=smooth_max_gradient,
                bounds=bounds,
                constraints=cons,
                options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
            )
            h = np.clip(res.x, 0.0, 1.0)
            adjust_sum_exact(h, rng)
        except Exception:
            pass

    # final polishing with a very high beta
    if time.time() < deadline:
        CURRENT_BETA = 10000.0
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint, "jac": sum_eq_jac}
        try:
            res = minimize(
                fun=smooth_max_objective,
                x0=h,
                method="SLSQP",
                jac=smooth_max_gradient,
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 80, "ftol": 1e-12, "disp": False},
            )
            h = np.clip(res.x, 0.0, 1.0)
            adjust_sum_exact(h, rng)
        except Exception:
            pass

    # gradient descent on the worst shift
    if time.time() < deadline:
        h = gradient_descent_worst(h, rng, max_iter=30, step0=0.05)

    adjust_sum_exact(h, rng)
    return h


# ----- generation of diverse start vectors -----
def generate_starts(n, rng, parent=None):
    starts = []

    # binary random
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append(bin_rand)

    # uniform fractional
    target = n / 2.0
    uf = rng.random(n)
    uf *= target / uf.sum()
    adjust_sum_exact(uf, rng)
    starts.append(uf)

    # biased fractional (square)
    bf = rng.random(n) ** 2
    bf *= target / bf.sum()
    adjust_sum_exact(bf, rng)
    starts.append(bf)

    # simple block
    block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
    starts.append(block)

    if parent is not None:
        xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xt, xs, parent)
        starts.append(interp.copy())
        idx = np.argpartition(interp, -n // 2)[-n // 2:]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        starts.append(bin_vec)

    return starts


# ----- binary‑only simulated annealing (optional final polishing) -----
def binary_sa_opt(h_bin, rng, deadline):
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        if not ones or not zeros:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 15000:
            break
    return best_h, best_c
