    def adam_optimize(h_half, betas, max_iter_per_beta, deadline):
        lr_initial = 0.05
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = np.zeros_like(h_half)
        v = np.zeros_like(h_half)
        t = 0
        for beta in betas:
            for _ in range(max_iter_per_beta):
                if time.time() > deadline:
                    return h_half
                t += 1
                # compute gradient
                full = full_from_half(h_half)
                n = full.size
                dx = 2.0 / n
                corr = np.correlate(full, 1.0 - full, mode='full') * dx
                w = np.exp(beta * corr)
                sum_w = np.sum(w)

                conv_minus = np.convolve(w, full, mode='full')
                conv_plus = np.convolve(w[::-1], full, mode='full')

def adam_optimize(h_half, betas, max_iter_per_beta, deadline):
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h_half)
    v = np.zeros_like(h_half)
    t = 0
    for beta in betas:
        for _ in range(max_iter_per_beta):
            if time.time() > deadline:
                return h_half
            t += 1
            full = full_from_half(h_half)
            n = full.size
            dx = 2.0 / n
            corr = np.correlate(full, 1.0 - full, mode='full') * dx
            max_corr = np.max(corr)
            w = np.exp(beta * (corr - max_corr))
            sum_w = np.sum(w)
            conv_minus = np.convolve(w, full, mode='full')
            conv_plus = np.convolve(w[::-1], full, mode='full')
            mid = n - 1
            cov_minus = conv_minus[mid:mid + n]
            cov_plus = conv_plus[mid:mid + n]
            grad_full = 1.0 - (cov_minus + cov_plus) / sum_w
            m_half = grad_full[:h_half.size] - grad_full[-1: -h_half.size - 1: -1]
            m = beta1 * m + (1 - beta1) * m_half
            v = beta2 * v + (1 - beta2) * (m_half ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            h_half = h_half - lr * m_hat / (np.sqrt(v_hat) + eps)
            np.clip(h_half, 0.0, 1.0, out=h_half)
    return h_half

import time
import numpy as np

def full_from_half(h_half: np.ndarray) -> np.ndarray:
    """Construct full vector of length 2*m from half vector using
    h = [h_half, 1 - h_half[::-1]] which automatically satisfies the sum constraint."""
    return np.concatenate([h_half, 1.0 - h_half[::-1]])


def compute_c5_full(full: np.ndarray) -> float:
    """Maximum overlap integral for the discretised problem."""
    n = full.size
    corr = np.correlate(full, 1.0 - full, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adam_optimize(
    h_half: np.ndarray,
    betas: np.ndarray,
    max_iter_per_beta: int,
    deadline: float,
) -> np.ndarray:
    """Adam on the smooth‑max surrogate (log‑sum‑exp) of the max‑overlap."""
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(h_half)
    v = np.zeros_like(h_half)
    t = 0

    for beta in betas:
        for _ in range(max_iter_per_beta):
            if time.time() > deadline:
                return h_half
            t += 1

            full = full_from_half(h_half)
            n = full.size
            dx = 2.0 / n

            # correlation for every shift (linear, zero‑padded)
            corr = np.correlate(full, 1.0 - full, mode="full") * dx
            max_corr = np.max(corr)
            w = np.exp(beta * (corr - max_corr))          # safe scaling
            sum_w = np.sum(w)

            # convolution w * full  and w_rev * full
            conv_minus = np.convolve(w, full, mode="full")
            conv_plus = np.convolve(w[::-1], full, mode="full")
            mid = n - 1
            cov_minus = conv_minus[mid : mid + n]
            cov_plus = conv_plus[mid : mid + n]

            # gradient for full vector
            grad_full = 1.0 - (cov_minus + cov_plus) / sum_w

            # map to half variables (symmetry derivative)
            grad_half = grad_full[: h_half.size] - grad_full[-1 : -h_half.size - 1 : -1]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_half
            v = beta2 * v + (1.0 - beta2) * (grad_half ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            h_half = h_half - lr * m_hat / (np.sqrt(v_hat) + eps)
            np.clip(h_half, 0.0, 1.0, out=h_half)

    return h_half


def binary_targeted_improve(h: np.ndarray, rng: np.random.Generator, deadline: float) -> np.ndarray:
    """Greedy swap improvement for binary vectors."""
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break

        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        i = rng.choice(ones)
        j = rng.choice(zeros)

        if neigh[i] - neigh[j] < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue

        improved = False
        for _ in range(5):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] - neigh[j2] < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

    n_final = best_full.size
    target = n_final / 2.0
    diff = target - best_full.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n_final)
            add = min(diff, 1.0 - best_full[i]) if diff > 0 else max(diff, -best_full[i])
            best_full[i] += add
            diff = target - best_full.sum()
            if abs(diff) < 1e-12:
                break

    best_full = np.clip(best_full, 0.0, 1.0)
    final_c = compute_c5_full(best_full)

    return best_full.tolist(), float(final_c), int(n_final)
