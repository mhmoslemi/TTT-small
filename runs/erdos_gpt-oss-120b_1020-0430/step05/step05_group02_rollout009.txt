# ---------- utilities -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx, using FFT for speed."""
    n = h.size
    conv = np.real(np.fft.ifft(np.fft.fft(h) * np.fft.fft(1.0 - h)))
    return float(conv.max() * (2.0 / n))


def project_onto_box_and_sum(v: np.ndarray, target: float, max_iter: int = 200) -> np.ndarray:
    """Project v onto [0,1]^n intersect {sum = target} (L2 projection)."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if not np.any(mask):
            break
        delta = diff / mask.sum()
        x[mask] = np.clip(x[mask] + delta, 0.0, 1.0)
    # tiny residual fix
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        i = np.argmin(x) if diff > 0 else np.argmax(x)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
            x[i] += add
        else:
            sub = min(-diff, x[i])
            x[i] -= sub
    return x


def smooth_max_gradient(h: np.ndarray, beta: float):
    """Gradient of f_beta = (1/beta) log-sum-exp(beta * conv) w.r.t. h."""
    n = h.size
    h_fft = np.fft.fft(h)
    o_fft = np.fft.fft(1.0 - h)

    conv = np.real(np.fft.ifft(h_fft * o_fft))

    a = beta * conv
    a_max = np.max(a)
    exp_a = np.exp(a - a_max)
    w = exp_a / exp_a.sum()               # softmax weights

    w_fft = np.fft.fft(w)
    term1 = np.real(np.fft.ifft(w_fft * o_fft))
    term2 = np.real(np.fft.ifft(w_fft * h_fft))
    grad = term1 - term2
    return grad, conv, w


def smooth_max_optimize_one_beta(h: np.ndarray, beta: float, target: float,
                                 max_iter: int, deadline: float) -> np.ndarray:
    """One‑beta descent on the smooth max, with projection onto constraints."""
    step = 0.2 / np.sqrt(beta + 1e-12)
    for _ in range(max_iter):
        if time.time() > deadline:
            break

        grad, conv, _ = smooth_max_gradient(h, beta)

        a = beta * conv
        a_max = np.max(a)
        f_cur = (a_max + np.log(np.sum(np.exp(a - a_max)))) / beta

        h_new = h - step * grad
        h_new = project_onto_box_and_sum(h_new, target)

        # objective for the proposal
        h_fft = np.fft.fft(h_new)
        o_fft = np.fft.fft(1.0 - h_new)
        conv_new = np.real(np.fft.ifft(h_fft * o_fft))
        a_new = beta * conv_new
        a_new_max = np.max(a_new)
        f_new = (a_new_max + np.log(np.sum(np.exp(a_new - a_new_max)))) / beta

        if f_new < f_cur - 1e-12:
            h = h_new
            step *= 1.02
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return h


def smooth_max_optimize(h: np.ndarray, beta_seq, target: float,
                        max_iter_per_beta: int, deadline: float) -> np.ndarray:
    """Cascade over increasing betas."""
    for beta in beta_seq:
        if time.time() > deadline:
            break
        h = smooth_max_optimize_one_beta(h, beta, target,
                                         max_iter_per_beta, deadline)
    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        conv = np.real(np.fft.ifft(np.fft.fft(h) *
                                   np.fft.fft(1.0 - h)))
        k = int(conv.argmax())
        # gradient of the worst term
        g = (1.0 - np.roll(h, -k)) - np.roll(h, k)
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_and_sum(h, target)
    return h

        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = project_onto_box_and_sum(h_up, n_up / 2.0)

        h_up = smooth_max_optimize(h_up, beta_seq, n_up / 2.0,
                                   max_iter_per_beta=150,
                                   deadline=deadline)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # safety fallback
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_onto_box_and_sum(best_h, n / 2.0)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_and_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
