def binary_sa(n, rng, time_budget, init_h=None):

    start = time.time()
    best_h = None
    best_c = np.inf

    # Prepare initial h
    if init_h is not None:
        h = init_h.copy()
        # Ensure binary: enforce exactly n/2 ones: if sum not exact due to rounding?
        # We'll force via enforce_sum_bin
        h = enforce_sum_bin(h, rng) # New function that transforms to binary with exact count.
    else:
        # random binary
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    # Setup indices
    ones_arr = np.where(h == 1.0)[0]
    zeros_arr = np.where(h == 0.0)[0]

    c_curr = compute_c5(h)

    if c_curr < best_c:
        best_h, best_c = h.copy(), c_curr

    # parameters
    T0 = 0.02
    T_end = 1e-5
    # tracking for restarts
    iters_since_improve = 0
    max_no_improve = 20000  # after that, restart

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        T = T0 * (T_end / T0) ** (elapsed / time_budget)
        # sample swap
        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = ones_arr[i_pos]
        j = zeros_arr[j_pos]
        # apply swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr
        accept = False
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            accept = True
            c_curr = c_new
            # update index arrays
            ones_arr[i_pos] = j
            zeros_arr[j_pos] = i
            iters_since_improve += 1
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
                iters_since_improve = 0
        if not accept:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        if iters_since_improve >= max_no_improve:
            # restart random binary
            h = np.zeros(n, dtype=float)
            ones = rng.choice(n, n // 2, replace=False)
            h[ones] = 1.0
            ones_arr = np.where(h == 1.0)[0]
            zeros_arr = np.where(h == 0.0)[0]
            c_curr = compute_c5(h)
            iters_since_improve = 0
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
    # Return best_h and best_c
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for the soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C₅ for a step function on [0,2] with step size 2/n."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust a float vector to satisfy Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def enforce_sum_bin(h: np.ndarray) -> np.ndarray:
    """Convert (possibly non‑binary) array to binary with exactly n/2 ones."""
    n = h.size
    target = n // 2
    idx = np.argsort(-h)  # descending
    bin_h = np.zeros_like(h, dtype=float)
    bin_h[idx[:target]] = 1.0
    return bin_h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def binary_sa(n: int, rng: np.random.Generator, time_budget: float,
              init_h: np.ndarray = None):
    """Simulated‑annealing on binary vectors with exactly n/2 ones."""
    start = time.time()
    best_h = None
    best_c = np.inf

    # initialise
    if init_h is not None:
        h = enforce_sum_bin(init_h)
    else:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    ones_arr = np.where(h == 1.0)[0]
    zeros_arr = np.where(h == 0.0)[0]
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr

    T0 = 0.02
    T_end = 1e-5
    iters_no_improve = 0
    max_no_improve = 20000

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        T = T0 * (T_end / T0) ** (elapsed / time_budget)

        i_pos = rng.integers(len(ones_arr))
        j_pos = rng.integers(len(zeros_arr))
        i = ones_arr[i_pos]
        j = zeros_arr[j_pos]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c_curr

        accept = False
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            accept = True
            c_curr = c_new
            ones_arr[i_pos] = j
            zeros_arr[j_pos] = i
            iters_no_improve += 1
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
                iters_no_improve = 0
        if not accept:
            h[i] = 1.0
            h[j] = 0.0

        if iters_no_improve >= max_no_improve:
            # restart
            h = np.zeros(n, dtype=float)
            ones = rng.choice(n, n // 2, replace=False)
            h[ones] = 1.0
            ones_arr = np.where(h == 1.0)[0]
            zeros_arr = np.where(h == 0.0)[0]
            c_curr = compute_c5(h)
            iters_no_improve = 0
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # -------------------------------------------------------------
    # Determine base size and possibly a starting vector
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values, dtype=float)
        n = base_h.size
        base_h = np.clip(base_h, 0.0, 1.0)
        base_h = enforce_sum(base_h, rng)
        best_h = base_h.copy()
        best_c = compute_c5(best_h)
        best_n = n
    else:
        n = 512
        best_h = None
        best_c = np.inf
        best_n = None

    # -------------------------------------------------------------
    # Binary simulated annealing (major share of time)
    time_sa = total_budget * 0.75
    init_bin = None
    if best_h is not None:
        # turn current float solution into a binary start
        init_bin = enforce_sum_bin(best_h)
    h_bin, c_bin = binary_sa(n, rng, time_sa, init_h=init_bin)
    if c_bin < best_c:
        best_h, best_c, best_n = h_bin.copy(), c_bin, n

    # -------------------------------------------------------------
    # Continuous refinement with soft‑max SLSQP (remaining time)
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        h_cont = best_h.astype(float).copy()
        # small random perturbation to leave the binary manifold
        h_cont += rng.uniform(-0.01, 0.01, size=h_cont.shape)
        h_cont = np.clip(h_cont, 0.0, 1.0)
        h_cont = enforce_sum(h_cont, rng)

        betas = (10.0, 50.0, 200.0, 800.0, 3000.0, 10000.0, 30000.0)
        for beta in betas:
            if time.time() - start > total_budget * 0.95:
                break
            h_cont = slsqp_optimize(h_cont, beta, rng, max_iter=800)
            c = compute_c5(h_cont)
            if c < best_c:
                best_h, best_c = h_cont.copy(), c

        if time.time() - start < total_budget * 0.97:
            h_cont = gradient_refine(h_cont, rng, step=0.01, max_iter=12)
            c = compute_c5(h_cont)
            if c < best_c:
                best_h, best_c = h_cont.copy(), c

    # -------------------------------------------------------------
    # Light final tweaks until budget expires
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # -------------------------------------------------------------
    # Sanitise output
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
