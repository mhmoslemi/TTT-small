            if time.time() > deadline:
                break
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=1500)
            except Exception:
                continue
            maybe_update(h)
        h = gradient_refine(h, rng, step=0.025, max_iter=30)
        maybe_update(h)
        hb = fractional_to_binary(h, rng)
        hb = binary_targeted_improve_nonperiodic(hb, rng, deadline)
        maybe_update(hb)

    if best_h is not None and time.time() < deadline:
        n_up = best_n * 2
        if n_up <= 720:
            xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            adjust_sum_exact(h_up, rng)
            for beta in betas:
                if time.time() > deadline:
                    break
                try:
                    h_up = slsqp_optimize(h_up, beta, rng, max_iter=2000)
                except Exception:
                    continue
                maybe_update(h_up)
            h_up = gradient_refine(h_up, rng, step=0.025, max_iter=40)
            maybe_update(h_up)
            hb_up = fractional_to_binary(h_up, rng)
            hb_up = binary_targeted_improve_nonperiodic(hb_up, rng, deadline)
            maybe_update(hb_up)

    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
