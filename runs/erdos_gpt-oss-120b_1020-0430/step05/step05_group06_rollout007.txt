def sa_optimize(h, max_iter, start_T, end_T, start_step, end_step, rng, deadline):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = len(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # schedule
        t = it / max_iter
        T = start_T * (1 - t) + end_T * t
        step = start_step * (1 - t) + end_step * t
        # pick random i,j distinct
        i, j = rng.choice(n, size=2, replace=False)
        # compute max permissible delta so that new values within [0,1]
        # delta adds to i and subtracts from j: hi' = hi + d, hj' = hj - d
        max_pos = min(step, 1.0 - cur_h[i], cur_h[j]) # Actually d <= 1 - hi (to not exceed) and d <= hj (to not go negative)
        max_neg = min(step, cur_h[i], 1.0 - cur_h[j]) # d negative magnitude
        # sample delta uniformly in [-max_neg, max_pos]
        if max_pos + max_neg == 0:
            continue
        d = rng.uniform(-max_neg, max_pos)
        if abs(d) < 1e-12:
            continue
        new_h = cur_h.copy()
        new_h[i] += d
        new_h[j] -= d
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c

def subgradient_optimize(h, rng, deadline, max_steps=5000):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = len(h)

        if time.time() > deadline:

        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_c = np.max(corr)
        # find lag(s) achieving max
        # compute lag offset index: lag index = argmax - (n-1)
        idxs = np.flatnonzero(corr == max_c)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h):
    """Exact C5 for a step function h sampled on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Adjust h in-place so that sum(h) == len(h)/2 exactly, keeping 0<=h<=1."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    # simple iterative correction; diff shrinks quickly
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def _smoothmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp in a numerically stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _smoothmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

# ---------- simulated annealing ----------
def sa_optimize(h_start, rng, deadline):
    """Simple SA that preserves the sum exactly."""
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = cur_h.size

    start_T, end_T = 0.02, 0.0002
    start_step, end_step = 0.30, 0.01
    it = 0
    while time.time() < deadline:
        t_frac = it / 2_000_000  # linear schedule over ~2M iters
        if t_frac > 1.0:
            t_frac = 1.0
        T = start_T * (1.0 - t_frac) + end_T * t_frac
        step = start_step * (1.0 - t_frac) + end_step * t_frac

        i, j = rng.choice(n, size=2, replace=False)
        max_pos = min(step, 1.0 - cur_h[i], cur_h[j])
        max_neg = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_pos + max_neg == 0:
            it += 1
            continue
        d = rng.uniform(-max_neg, max_pos)
        new_h = cur_h.copy()
        new_h[i] += d
        new_h[j] -= d
        new_c = compute_c5(new_h)

        accept = False
        if new_c < cur_c:
            accept = True
        else:
            prob = np.exp(-(new_c - cur_c) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        it += 1
    return best_h, best_c

# ---------- subgradient descent ----------
def subgradient_optimize(h_start, rng, deadline,
                         max_steps=50000, init_alpha=0.20, final_alpha=0.001):
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = cur_h.size
    target = n / 2.0

    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_corr = np.max(corr)
        # choose a worst lag at random (if several)
        worst_idxs = np.flatnonzero(corr == max_corr)
        lag_idx = rng.choice(worst_idxs)
        lag = lag_idx - (n - 1)  # shift relative to zero

        # gradient of the worst correlation (subgradient)
        h_fwd = np.roll(cur_h, -lag)
        h_bwd = np.roll(cur_h, lag)
        g = 1.0 - h_fwd - h_bwd

        alpha = init_alpha * (1.0 - step / max_steps) + final_alpha * (step / max_steps)
        new_h = cur_h - alpha * g
        new_h = np.clip(new_h, 0.0, 1.0)
        adjust_sum_exact(new_h, rng)

        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

def random_feasible_h(n, rng):
    x = rng.random(n)
    x = x / x.sum() * (n / 2.0)
    adjust_sum_exact(x, rng)
