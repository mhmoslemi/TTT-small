def simulated_annealing_binary(h_bin, n, rng, deadline):
    # h_bin: binary vector (0/1)
    current = h_bin.copy()
    best = current.copy()
    cur_val = compute_c5(current)
    best_val = cur_val
    # precompute lists of indices of ones and zeros for speed; update after each swap.
    ones = np.flatnonzero(current == 1)
    zeros = np.flatnonzero(current == 0)
    # schedule parameters
    startT = 0.01  # maybe moderate
    endT = 1e-5
    total_time = deadline - time.time()
    # we can base T on fraction of total time passed.
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        T = startT * (endT/startT)**frac  # exponential decay
        # pick a random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # swap
        current[i] = 0
        current[j] = 1
        # compute new val
        new_val = compute_c5(current)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_val = new_val
            # update lists
            ones[i_idx] = j
            zeros[j_idx] = i
            # update best
            if cur_val < best_val:
                best_val = cur_val
                best = current.copy()
        else:
            # revert swap
            current[i] = 1
            current[j] = 0
    return best, best_val

def smoothmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    max_bc = np.max(bc)
    lse = max_bc + np.log(np.sum(np.exp(bc - max_bc)))

def smoothmax_grad(x):
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / np.sum(exp_bc)
    # compute convolution
    w_rev = w[::-1]
    s1 = np.convolve(x, w_rev, mode='same')

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Exact sum correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h


def greedy_binary(n, rng):
    """Construct a binary sequence minimizing max_k Σ h_i(1‑h_{i+k})."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start:start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h, O


def hill_climb(h_bin, max_O, n, rng, deadline):
    """Simple hill‑climb on binary vectors (swap 1↔0)."""
    best_h = h_bin.copy()
    best_max = max_O
    cur_h = h_bin.copy()
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        cur_h[i] = 0
        cur_h[j] = 1
        O_new = np.max(
            np.correlate(
                cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode="full"
            )
        )
        if O_new < best_max:
            best_max = O_new
            best_h = cur_h.copy()
            ones[i_idx] = j
            zeros[j_idx] = i
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_max


def smoothmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA


def smoothmax_grad(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / np.sum(exp_bc)          # softmax weights, sum to 1
    w_rev = w[::-1]
    s1 = np.convolve(x, w_rev, mode="same")
    s2 = np.convolve(x, w, mode="same")
    return 1.0 - s1 - s2                # gradient of smooth‑max


def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on smooth‑max surrogate (with analytic gradient)."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(
        smoothmax_obj,
        h0,
        method="SLSQP",
        jac=smoothmax_grad,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def simulated_annealing_binary(h_bin, rng, deadline):
    """SA on binary vectors (pair swaps) to lower C5."""
    start = time.time()
    cur = h_bin.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    T0, Tf = 0.02, 1e-5
    while time.time() < deadline:
        elapsed = time.time() - start
        if elapsed == 0.0:
            continue
        frac = elapsed / (deadline - start)
        T = T0 * (Tf / T0) ** frac
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i, j = ones[i_idx], zeros[j_idx]
        cur[i] = 0
        cur[j] = 1
        new_val = compute_c5(cur)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    return best, best_val

        if c_val < best_c:
            best_c = c_val
            best_h = h_bin.astype(float).copy()
            best_n = n

        # ----- hill climbing -----
        if time.time() < deadline:
            hc_deadline = deadline
            h_hc, max_O_hc = hill_climb(h_bin.copy(), max_O, n, rng, hc_deadline)
            c_hc = max_O_hc * (2.0 / n)
            if c_hc < best_c:
                best_c = c_hc
                best_h = h_hc.astype(float).copy()
                best_n = n

        # ----- simulated annealing -----
        if time.time() < deadline:
            sa_deadline = min(deadline, time.time() + 0.12 * (deadline - time.time()))
            # start from the best binary we have for this n
            start_bin = (
                best_h.astype(np.int8)
                if best_n == n and best_h is not None
                else h_bin.copy()
            )
            bin_sa, val_sa = simulated_annealing_binary(start_bin, rng, sa_deadline)
            if val_sa < best_c:
                best_c = val_sa
                best_h = bin_sa.astype(float).copy()
                best_n = n

        # ----- smooth‑max SLSQP cascade -----
        if (
            time.time() < deadline
            and best_h is not None
            and best_n == n
        ):
            h_float = best_h.copy()
            adjust_sum_exact(h_float, rng)
            for beta in (
                5.0,
                20.0,
                80.0,
                200.0,
                500.0,
                1000.0,
                2000.0,
                5000.0,
                10000.0,
            ):
                if time.time() >= deadline:
                    break
                max_iter = 300 if beta < 200 else 600
                if beta >= 2000:
                    max_iter = 800
                if beta >= 5000:
                    max_iter = 600
                try:
                    h_float = slsqp_optimize(h_float, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c_new = compute_c5(h_float)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_float.copy()
                    best_n = n

    # ----- fallback -----
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)

    # final correction and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
