def slsqp_optimize(h0,beta,rng,max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    cons = {"type":"eq","fun":sum_eq_constraint}
    res = minimize(smooth_max_obj, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0.0,1.0)
    # adjust sum exactly
    diff = CURRENT_TARGET - x.sum()
    # make small adjustments
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0,n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_descent_worst(h,rng,step_start=0.05,max_iter=20):
    n = h.size
    target = n/2.0
    step = step_start
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]  # derivative wrt h_i for i in [0, upper-1]
            g[k:] -= h[:upper]       # derivative wrt h_{i+k}
        # step size decays
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum after each step
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0,n)
                add = min(diff, 1.0 - h[i]) if diff>0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        step *= 0.9  # decay
    return h

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        # exponential cooling
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 10000:  # longer patience
            break
    return best_h, best_c

def greedy_improve(h, rng, max_iter=100, deadline=None):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        # 1 at i and 0 at i+k (if valid)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        # 0 at i and 1 at i+k
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        # Sample pairs
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 2000
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i,j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i,j)
        if best_pair is None:
            break
        i,j = best_pair
        h[i] = 0
        h[j] = 1
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global state for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass to enforce Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def smooth_max_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_max_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_descent_worst(h: np.ndarray, rng: np.random.Generator,
                          step_start: float = 0.05, max_iter: int = 20) -> np.ndarray:
    n = h.size
    target = n / 2.0
    step = step_start
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        step *= 0.9
    return h

def binary_sa_hill_climb(h_bin: np.ndarray, rng: np.random.Generator,
                         time_deadline: float):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 10000:
            break
    return best_h, best_c

def greedy_improve(h: np.ndarray, rng: np.random.Generator,
                   max_iter: int = 100, deadline: float = None) -> np.ndarray:
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 2000
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.98

    best_h = None
    best_c = np.inf
    best_n = None

    # Candidate resolutions (even numbers)
    candidate_ns = [360, 480, 600, 720]

        if time.time() > deadline:
            break

        # Build a modest set of initial vectors for this n
        init_vecs = []

        if parent is not None:

            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            interp = np.clip(interp, 0.0, 1.0)
            adjust_sum_exact(interp, rng)
            init_vecs.append(interp)

            # binary version of interpolated parent
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_parent = np.zeros(n)
            bin_parent[idx] = 1.0
            adjust_sum_exact(bin_parent, rng)
            init_vecs.append(bin_parent)

        # random binary vectors
        for _ in range(2):
            bin_rand = np.zeros(n)
            bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            init_vecs.append(bin_rand)

        # random fractional vectors
        for _ in range(2):

            rand = np.clip(rand, 0.0, 1.0)
            rand *= (n / 2.0) / rand.sum()
            adjust_sum_exact(rand, rng)
            init_vecs.append(rand)

        # ------------------------------------------------------------------
        # Quick evaluation and keep the top few
        scored = []
        for vec in init_vecs:

        scored.sort(key=lambda x: x[1])
        top_k = 2 if len(scored) > 2 else len(scored)

        for base_h, base_c in scored[:top_k]:
            if time.time() > deadline:
                break
            h = base_h.copy()

            h = np.clip(h, 0.0, 1.0)

                h = h_bin.astype(float)

            # Gradient descent focusing on worst shift
            h = gradient_descent_worst(h, rng, step_start=0.05, max_iter=30)
            c = compute_c5(h)
            if c < best_c:

            # SLSQP cascade with increasing β
            beta_seq = [200, 1000, 5000, 20000, 50000]
            for beta in beta_seq:
                if time.time() > deadline:
                    break
                iters = 400 if beta < 5000 else 800
                h = slsqp_optimize(h, beta, rng, max_iter=iters)
                c = compute_c5(h)
                if c < best_c:

            # Final gradient polishing
            h = gradient_descent_worst(h, rng, step_start=0.02, max_iter=20)

            # Record best after this candidate
