def compute_c5_fft(h):
    n = h.shape[0]
    m = 2*n
    h_fft = np.fft.rfft(h, m)
    opp_fft = np.fft.rfft(1 - h, m)
    corr = np.fft.irfft(h_fft * np.conj(opp_fft), m)
    # The correlation for k from -(n-1) to n-1 is at indices:
    # corr[k + n - 1] = sum_i h[i] * (1 - h[i+k]) for - (n-1) <= k <= n-1.
    # We need max over all k. So compute max(corr)
    max_corr = np.max(corr)
    return float(max_corr * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    # Clamp diff to within [-1,1] maybe.
    while abs(diff) > 1e-12:
        # choose a random index where we can add diff without violating bounds
        if diff > 0:
            # need to increase some entries
            candidates = np.where(h < 1 - 1e-12)[0]
            if not candidates.size:
                break
            i = rng.choice(candidates)
            delta = min(diff, 1 - h[i])
        else:
            candidates = np.where(h > 1e-12)[0]
            if not candidates.size:
                break
            i = rng.choice(candidates)
            delta = max(diff, -h[i])
        h[i] += delta
        diff = target - h.sum()
    return h

def compute_c5(h):
    n = h.shape[0]
    L = 2 * n
    # reverse (1-h)
    b_rev = (1.0 - h)[::-1]
    a_fft = np.fft.rfft(h, L)
    b_fft = np.fft.rfft(b_rev, L)
    conv = np.fft.irfft(a_fft * b_fft, L)
    max_corr = np.max(conv)
    return float(max_corr * (2.0 / n))

def slsqp_optimize(x0, beta, rng, max_iter=800):
    n = x0.size
    target = n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(lambda x: smooth_max_obj(x, beta),
                   x0,
                   method="SLSQP",
                   bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)

def greedy_binary(n, rng):
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # correlation of all‑zero vector
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0  # enforce O_0 = 0
    return h, O

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# globals used by the SLSQP callbacks
GLOBAL_TARGET = None
GLOBAL_BETA = None

def adjust_sum_exact(h, rng):
    """Force sum(h) = len(h)/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple iterative fix; diff is tiny after optimisations
    while abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            delta = min(diff, 1 - h[i])
        else:
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            delta = max(diff, -h[i])
        h[i] += delta
        diff = target - h.sum()
    return h

def compute_c5(h):
    """Exact C5 using FFT‑based linear correlation."""
    n = h.shape[0]
    L = 2 * n
    b_rev = (1.0 - h)[::-1]               # reverse (1‑h) for correlation
    a_fft = np.fft.rfft(h, L)
    b_fft = np.fft.rfft(b_rev, L)
    conv = np.fft.irfft(a_fft * b_fft, L)
    max_corr = np.max(conv)
    return float(max_corr * (2.0 / n))

def sum_eq_constraint(x):
    """Equality constraint Σx = n/2 (uses GLOBAL_TARGET)."""
    return np.sum(x) - GLOBAL_TARGET

def smooth_max_obj(x):
    """Smooth‑max surrogate Σ_k exp(β c_k) with β = GLOBAL_BETA."""
    n = x.shape[0]
    L = 2 * n
    b_rev = (1.0 - x)[::-1]
    a_fft = np.fft.rfft(x, L)
    b_fft = np.fft.rfft(b_rev, L)
    conv = np.fft.irfft(a_fft * b_fft, L)   # full correlation values
    return logsumexp(GLOBAL_BETA * conv) / GLOBAL_BETA

def slsqp_optimize(x0, beta, rng, max_iter=800):
    """SLSQP minimisation of the smooth‑max objective for a given β."""
    global GLOBAL_TARGET, GLOBAL_BETA
    n = x0.size
    GLOBAL_TARGET = n / 2.0
    GLOBAL_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_max_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def greedy_binary(n, rng):
    """Greedy construction of a binary vector (0/1) with Σ= n/2."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)        # correlation of all‑zero vector
    for _ in range(L):
        O_adj = O + 1                               # pretend we would add a 1 at each position
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h   # new correlation after setting h[p]=1
            w = window.max()
            if (best_w is None) or (w < best_w):
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0                               # enforce zero at lag 0
    return h, O

def binary_local_opt(h0, rng, deadline):
    """Hill‑climbing on binary vectors (swap 1↔0) until the deadline."""
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    ones = [int(i) for i in np.flatnonzero(h == 1)]
    zeros = [int(i) for i in np.flatnonzero(h == 0)]
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0
        h[j] = 1
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            # update the index lists
            idx_i = ones.index(i)
            idx_j = zeros.index(j)
            ones[idx_i] = j
            zeros[idx_j] = i
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.96   # safety margin

    n = 360
    best_h = None
    best_c = np.inf

    # helper to record a new global best
    def record_candidate(h):
        nonlocal best_h, best_c
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        return c

    # 1. start from supplied construction if any
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        adjust_sum_exact(h0, rng)
        record_candidate(h0)

    # 2. greedy binary construction
    if time.time() < deadline:
        h_bin, _ = greedy_binary(n, rng)
        h_bin_f = h_bin.astype(float)
        record_candidate(h_bin_f)

        # 3. binary local optimisation (30 % of remaining time)
        rem = deadline - time.time()
        bin_dead = time.time() + rem * 0.3
        h_opt_bin, _ = binary_local_opt(h_bin_f, rng, bin_dead)
        record_candidate(h_opt_bin)

    # 4. SLSQP refinement from the best binary/continuous candidate
    if best_h is not None and time.time() < deadline:
        x = best_h.astype(float)
        adjust_sum_exact(x, rng)
        beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000]
        for beta in beta_schedule:
            if time.time() >= deadline:
                break
            max_iter = 600 if beta < 200 else 800
            try:
                x = slsqp_optimize(x, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            record_candidate(x)

    # 5. a few random restarts with SLSQP (use remaining time)
    while time.time() < deadline:
        # allocate a small chunk per restart
        chunk = min(30.0, deadline - time.time())
        if chunk < 5.0:
            break
        # random binary start + tiny Gaussian perturbation
        start_vec = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        start_vec[idx] = 1.0
        noise = rng.normal(scale=0.02, size=n)
        start_vec = np.clip(start_vec + noise, 0.0, 1.0)
        adjust_sum_exact(start_vec, rng)

        # run the beta cascade within the allocated chunk
        sub_dead = time.time() + chunk
        x = start_vec
        for beta in beta_schedule:
            if time.time() >= sub_dead:
                break
            max_iter = 600 if beta < 200 else 800
            try:
                x = slsqp_optimize(x, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            record_candidate(x)
        # continue loop for another restart if time permits

    # fallback – uniform half‑filled vector
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
