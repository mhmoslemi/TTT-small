def lp_step(h, rng):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    grads = np.empty((2*n-1, n), dtype=float)
    for idx, k in enumerate(range(-(n-1), n)):
        g = np.zeros(n)
        if k >= 0:
            g[:n - k] += 1 - h[k:]
            g[k:] -= h[:n - k]
        else:
            l = -k
            g[l:] += 1 - h[:n - l]
            g[:n - l] -= h[l:]
        grads[idx] = g
    c = np.zeros(n+1)
    c[-1] = 1.0
    A_ub = np.hstack([grads, -np.ones((2*n-1, 1), dtype=float)])
    b_ub = -corr + grads @ h
    A_eq = np.zeros((1, n+1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = n / 2.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return h
    h_new = res.x[:n]
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, rng)
    return h_new

import time, numpy as np
from scipy.optimize import linprog, minimize

# ---- basic utilities -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift with maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Shift ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h, best_c = h.copy(), compute_c5(h)
    scale = step_scale

            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:

def lp_step(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """One convex‑approximation iteration (min‑max LP)."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    grads = np.empty((2 * n - 1, n), dtype=float)
    for idx, k in enumerate(range(-(n - 1), n)):
        g = np.zeros(n)
        if k >= 0:
            g[: n - k] += 1 - h[k:]
            g[k:] -= h[: n - k]
        else:
            l = -k
            g[l:] += 1 - h[: n - l]
            g[: n - l] -= h[l:]
        grads[idx] = g
    # LP: minimize t  s.t.  grad·h - t <= -corr + grad·h_cur  for all k
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([grads, -np.ones((2 * n - 1, 1), dtype=float)])
    b_ub = -corr + grads @ h
    A_eq = np.zeros((1, n + 1), dtype=float)

                  bounds=bounds, method="highs")

            continue

        # 4) occasional random restart with brief refinement
        if rng.random() < 0.01:
            h_rand = np.zeros(n)
            h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            h_rand = enforce_sum(h_rand, rng)
            h_tmp, c_tmp = guided_hill_climb(h_rand, rng,
                                             max_iters=500,
                                             step_scale=0.5)
            if c_tmp < best_c:
                best_h, best_c = h_tmp, c_tmp
                continue

    # final soft‑max polishing if time permits
    if time.time() - start < budget_s:
        for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0):
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=300)
            best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
