def gradient_refine(h, rng, step=0.01, max_iter=20):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        grad = np.ones_like(h)
        if k == 0:
            grad = 1 - 2*h
        elif k > 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:  # k <0
            k2 = -k
            grad[k2:] -= h[: n - k2]
            grad[: n - k2] -= h[k2:]
        # gradient step
        h = h - step * grad
        # clip 0/1
        h = np.clip(h, 0.0, 1.0)
        # enforce sum by adjusting difference
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff using random indices
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def softmax_obj(x):

def slsqp_optimize(x0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction if needed
    diff = CURRENT_TARGET - x.sum()

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Global parameters for SLSQP soft‑max optimization
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def shift_h(h: np.ndarray, k: int) -> np.ndarray:
    """Shift h by k with zero padding (k can be negative)."""
    n = h.shape[0]
    out = np.empty(n, dtype=h.dtype)
    if k >= 0:
        out[: n - k] = h[k:]
        out[n - k :] = 0
    else:
        k2 = -k
        out[k2:] = h[: n - k2]
        out[:k2] = 0
    return out


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(x0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimization for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = x0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.02, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        grad = np.ones_like(h)
        if k == 0:
            grad = 1 - 2 * h
        elif k > 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            k2 = -k
            grad[k2:] -= h[: n - k2]
            grad[: n - k2] -= h[k2:]
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def binary_sa_guided(n: int, start_h: np.ndarray, rng: np.random.Generator, time_budget: float):
    """Simulated‑annealing on binary vectors with a guided swap."""
    cur_h = start_h.copy()
    ones_set = set(np.flatnonzero(cur_h))
    zeros_set = set(range(n)) - ones_set
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    T0, T_end = 0.2, 1e-5
    start = time.time()
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (T_end / T0) ** frac

        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx

        i = None
        if rng.random() < max(0.2, 1 - frac):
            if k >= 0:
                reg_len = n - k
                if reg_len > 0:
                    region = np.arange(reg_len)
                    cand_i = region[(cur_h[region] == 0) & (cur_h[region + k] == 1)]
                else:
                    cand_i = np.array([], dtype=int)
            else:
                k2 = -k
                reg_len = n - k2
                if reg_len > 0:
                    region = np.arange(k2, n)
                    cand_i = region[(cur_h[region] == 0) & (cur_h[region - k2] == 1)]
                else:
                    cand_i = np.array([], dtype=int)

            if cand_i.size > 0:
                i = int(rng.choice(cand_i))
                h_shift = shift_h(cur_h, k)
                cand_j = np.where((cur_h == 1) & (h_shift == 0))[0]
                if cand_j.size > 0:
                    j = int(rng.choice(cand_j))
                else:
                    i = None

        if i is None:
            i_one = int(rng.choice(list(ones_set)))
            j_zero = int(rng.choice(list(zeros_set)))
            cur_h[i_one] = 0
            cur_h[j_zero] = 1
            ones_set.remove(i_one); zeros_set.add(i_one)
            ones_set.add(j_zero); zeros_set.remove(j_zero)
        else:
            cur_h[i] = 1
            cur_h[j] = 0
            zeros_set.remove(i); ones_set.add(i)
            ones_set.remove(j); zeros_set.add(j)

        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            if i is None:
                cur_h[i_one] = 1
                cur_h[j_zero] = 0
                ones_set.add(i_one); zeros_set.remove(i_one)
                ones_set.remove(j_zero); zeros_set.add(j_zero)
            else:
                cur_h[i] = 0
                cur_h[j] = 1
                zeros_set.add(i); ones_set.remove(i)
                ones_set.add(j); zeros_set.remove(j)

    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ---- initial binary vector ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        n = h0.size
        idx = np.argsort(-h0)
        start_h = np.zeros(n, dtype=int)
        start_h[idx[: n // 2]] = 1
    else:
        n = 512
        start_h = np.zeros(n, dtype=int)
        start_h[rng.choice(n, n // 2, replace=False)] = 1

    # ---- binary simulated annealing (guided) ---------------------------------
    time_sa = total_budget * 0.55
    best_h_bin, best_c_bin = binary_sa_guided(n, start_h, rng, time_sa)

    best_h = best_h_bin.astype(float)
    best_c = best_c_bin

    # ---- continuous soft‑max refinement ---------------------------------------
    remaining = total_budget - (time.time() - start)
    if remaining > 5.0:
        betas = [20., 80., 200., 800., 3000., 8000.]
        for beta in betas:
            if time.time() - start > total_budget * 0.95:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=400)
            c = compute_c5(best_h)
            if c < best_c:
                best_c = c

    # ---- gradient tidy‑up ------------------------------------------------------
    remaining = total_budget - (time.time() - start)
    if remaining > 2.0:
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=8)
        best_c = compute_c5(best_h)

    # ---- optional up‑sampling --------------------------------------------------
    remaining = total_budget - (time.time() - start)
    if remaining > 20.0 and n * 2 <= 1024:
        n_up = n * 2
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        h_up = gradient_refine(h_up, rng, step=0.01, max_iter=6)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, n = h_up, c_up, n_up

    # ---- final sanitisation ----------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
