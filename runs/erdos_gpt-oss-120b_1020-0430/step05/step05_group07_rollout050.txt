def gradient_refine(h, max_iter=20, step=0.1):
    n = len(h)
    target = n/2.0
    for it in range(max_iter):
        # compute corr h with 1-h
        corr = np.correlate(h, 1.0 - h, mode='full')
        # find worst shift (max)
        idx = np.argmax(corr)
        k = idx - (n - 1)
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0*h
        else:
            # compute contributions
            # for i where i+k in range
            # -h_{i+k} term
            g[:n - k] -= h[k:]
            # for i where i-k in range
            g[k:] -= h[:n - k]
            # +1 term for all i
            g += 1.0
        # step size
        alpha = step * (0.7 ** it)
        h = h - alpha * g
        # project onto [0,1]
        h = np.clip(h, 0.0, 1.0)
        # adjust sum to target
        cur_sum = h.sum()
        diff = target - cur_sum
        if abs(diff) > 1e-12:
            # redistribute diff uniformly across non-boundary positions
            # Choose direction: add to those <1, subtract from >0.
            # Simple approach: modify all entries proportionally.
            # We'll adjust by adding diff/n to all entries, then clip and repeat.
            # Use loop to converge.
            for _ in range(10):
                add = diff / n
                h += add
                h = np.clip(h, 0.0, 1.0)
                new_sum = h.sum()
                new_diff = target - new_sum
                if abs(new_diff) < 1e-12:
                    break
                diff = new_diff
    return h

def adjust_sum_exact(h, target, rng, max_iter=2000):
    diff = target - h.sum()
    n = len(h)
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

import time
import numpy as np

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def adjust_sum_exact(h, target, rng, max_iter=2000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def assign_by_c(c, target):
    idx = np.argsort(-c)               # descending
    n = len(c)
    h = np.zeros(n, dtype=float)
    full = int(np.floor(target))
    if full > 0:
        h[idx[:full]] = 1.0
    rem = target - full
    if rem > 1e-12 and full < n:
        h[idx[full]] = rem
    return h

def neighbor_sum(h, shift):
    n = h.size
    if shift >= 0:
        left = np.concatenate((np.zeros(shift), h[:n - shift]))
        right = np.concatenate((h[shift:], np.zeros(shift)))
    else:
        s = -shift
        left = np.concatenate((h[s:], np.zeros(s)))
        right = np.concatenate((np.zeros(s), h[:n - s]))
    return left + right

def linear_iterate(h, max_iter, tol, rng):
    n = h.size
    target = n / 2.0
    dx = 2.0 / n
    for _ in range(max_iter):
        autoc = np.correlate(h, h, mode="full") * dx
        min_val = autoc.min()
        idx = autoc.argmin()
        shift = idx - (n - 1)
        c = neighbor_sum(h, shift)
        new_h = assign_by_c(c, target)
        new_autoc = np.correlate(new_h, new_h, mode="full") * dx
        improvement = new_autoc.min() - min_val
        if improvement < tol:
            break
        h = new_h
    return h

def gradient_refine(h, max_iter, step, rng):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        g = np.ones_like(h)
        if k == 0:
            g -= 2.0 * h
        else:
            if k > 0:
                g[: n - k] -= h[k:]
                g[k:] -= h[: n - k]
            else:
                s = -k
                g[s:] -= h[: n - s]
                g[: n - s] -= h[s:]
        alpha = step * (0.9 ** it)
        h = h - alpha * g
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum_exact(h, target, rng)
    return h

def simple_sa(h, rng, deadline, max_steps=4000):
    n = h.size
    target = n / 2.0
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline and it < max_steps:
        i, j = rng.choice(n, size=2, replace=False)
        max_tr = min(cur[i], 1.0 - cur[j])
        if max_tr <= 0.0:
            it += 1
            continue
        delta = rng.random() * max_tr
        cur[i] -= delta
        cur[j] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            cur[i] += delta
            cur[j] -= delta
        it += 1
    return best, best_c

def upsample_refine(h, rng, deadline):
    n = h.size
    if n * 2 > 720:
        return h
    n_up = n * 2
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
    h_up = np.interp(xt, xs, h)
    target = n_up / 2.0
    h_up = adjust_sum_exact(h_up, target, rng)
    h_up = linear_iterate(h_up, max_iter=100, tol=1e-5, rng=rng)
    h_up = gradient_refine(h_up, max_iter=20, step=0.2, rng=rng)
    if time.time() < deadline:
        h_up = simple_sa(h_up, rng, deadline)[0]
    return h_up

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = float('inf')
    best_n = None

    n_list = [360, 240, 180]
    for n in n_list:
        if time.time() >= deadline:
            break
        target = n / 2.0
        init_vecs = []

        # parent interpolation if available
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            if parent.size != n:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                interp = np.interp(xt, xs, parent)
                init_vecs.append(interp)

        # random binary
        bin_vec = np.zeros(n)
        bin_vec[rng.choice(n, n // 2, replace=False)] = 1.0
        init_vecs.append(bin_vec)

        # random fractional (uniform)
        rand_frac = rng.random(n)
        rand_frac = rand_frac * (target / rand_frac.sum())
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        rand_frac = adjust_sum_exact(rand_frac, target, rng)
        init_vecs.append(rand_frac)

        # random fractional (biased)
        rand_bias = rng.random(n) ** 2
        rand_bias = rand_bias * (target / rand_bias.sum())
        rand_bias = np.clip(rand_bias, 0.0, 1.0)
        rand_bias = adjust_sum_exact(rand_bias, target, rng)
        init_vecs.append(rand_bias)

        for h0 in init_vecs:
            if time.time() >= deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            h = adjust_sum_exact(h, target, rng)

            # linear improvement iterations
            h = linear_iterate(h, max_iter=200, tol=1e-6, rng=rng)

            # gradient refinement
            h = gradient_refine(h, max_iter=30, step=0.2, rng=rng)

            # short simulated annealing
            if time.time() < deadline:
                h, _ = simple_sa(h, rng, deadline, max_steps=3000)

            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                best_n = n

    # optional upsampling of the current best
    if best_h is not None and time.time() < deadline:
        h_up = upsample_refine(best_h, rng, deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up.copy()
            best_n = best_h.size

    if best_h is None:

        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = adjust_sum_exact(best_h, n / 2.0, rng)
        best_c = compute_c5(best_h)

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
