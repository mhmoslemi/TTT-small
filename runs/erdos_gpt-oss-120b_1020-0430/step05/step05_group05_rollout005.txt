def compute_c5_half(h_half):
    full = np.concatenate([h_half, 1 - h_half[::-1]])
    n = full.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    a = np.fft.rfft(full, n=L)
    b = np.fft.rfft((1 - full)[::-1], n=L)  # reversed
    conv = np.fft.irfft(a * b, n=L)[:2*n-1]  # take first 2n-1 elements
    max_val = conv.max()
    return float(max_val * (2.0 / n))

import time
import numpy as np
from scipy.optimize import minimize

# ----- basic helpers ---------------------------------------------------------
def full_from_half(h_half: np.ndarray) -> np.ndarray:
    """Construct the full 2‑periodic symmetric vector."""
    return np.concatenate([h_half, 1.0 - h_half[::-1]])


def compute_c5_half(h_half: np.ndarray) -> float:
    """C₅ for a symmetric half‑vector (linear correlation as used by the evaluator)."""
    full = full_from_half(h_half)
    n = full.size
    corr = np.correlate(full, 1.0 - full, mode="full")
    return float(corr.max() * (2.0 / n))


def smooth_max_obj(h_half: np.ndarray, beta: float) -> float:
    """Smooth‑max surrogate of the max‑overlap for SLSQP."""
    full = full_from_half(h_half)
    corr = np.correlate(full, 1.0 - full, mode="full")
    vals = beta * corr
    m = np.max(vals)
    return float((m + np.log(np.sum(np.exp(vals - m)))) / beta)


# ----- gradient based polishing -----------------------------------------------
def gradient_polish(h_half: np.ndarray, rng: np.random.Generator,
                    deadline: float, max_iter: int = 20) -> np.ndarray:
    """Projected gradient descent on the worst shift."""
    m = h_half.size
    while time.time() < deadline and max_iter > 0:
        full = full_from_half(h_half)
        n = full.size
        corr = np.correlate(full, 1.0 - full, mode="full")
        idx = np.argmax(corr)
        shift = abs(idx - (n - 1))

        if shift == 0:
            g_full = 1.0 - 2.0 * full
        else:
            # (1 - h[i+shift]) - h[i-shift]
            g_full = (1.0 - np.roll(full, -shift)) - np.roll(full, shift)

        rev_idx = np.arange(n - 1, -1, -1)[:m]        # indices of the mirrored half
        grad_half = g_full[:m] - g_full[rev_idx]

        step = 0.06 * (0.5 ** (max_iter - 1))
        h_half = np.clip(h_half - step * grad_half, 0.0, 1.0)

        max_iter -= 1
    return h_half

def binary_targeted_improve_half(hb: np.ndarray, rng: np.random.Generator,
                                 deadline: float) -> np.ndarray:
    """Swap 0/1 in the half‑vector to lower the worst overlap."""
    m = hb.size
    while time.time() < deadline:
        full = full_from_half(hb)
        n = full.size
        corr = np.correlate(full, 1.0 - full, mode="full")
        idx = np.argmax(corr)
        shift = abs(idx - (n - 1))
        if shift == 0:
            g_full = 1.0 - 2.0 * full
        else:
            g_full = (1.0 - np.roll(full, -shift)) - np.roll(full, shift)

        rev_idx = np.arange(n - 1, -1, -1)[:m]
        grad_half = g_full[:m] - g_full[rev_idx]

        ones = np.flatnonzero(hb == 1)
        zeros = np.flatnonzero(hb == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        i = ones[np.argmax(grad_half[ones])]   # flipping this 1 down helps most
        j = zeros[np.argmin(grad_half[zeros])]  # flipping this 0 up hurts least

        if grad_half[i] > grad_half[j]:
            hb[i] = 0.0
            hb[j] = 1.0
            continue

        # random rescue attempts
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if grad_half[i2] > grad_half[j2]:
                hb[i2] = 0.0
                hb[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return hb

        c_up = compute_c5_half(hb_up)
        if c_up < best_c:
            best_c = c_up
            best_h_half = hb_up.copy()
            best_n = best_n * 2

    # ---- final assembly -------------------------------------------------------
    if best_h_half is None:
        # fallback: simple block
        best_h_half = np.concatenate([np.ones(m // 2), np.zeros(m - m // 2)])

    best_full = full_from_half(best_h_half)
    best_full = np.clip(best_full, 0.0, 1.0)
    final_c = compute_c5_half(best_h_half)

    return best_full.tolist(), float(final_c), int(best_n)
