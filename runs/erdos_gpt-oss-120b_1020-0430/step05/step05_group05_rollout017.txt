def binary_hill_climb(h, deadline):
    n = h.shape[0]
    best_c = compute_c5_fft(h)
    # Precompute indices of ones and zeros
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    # We'll keep updating ones and zeros after swaps
    while time.time() < deadline:
        # pick random i from ones, j from zeros
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5_fft(h)
        if c < best_c - 1e-8:
            best_c = c
            # update ones and zeros lists
            # remove i from ones, add i to zeros; similarly for j
            ones = ones[ones != i]  # inefficient; maybe use set
        else:
            # reject swap: revert
            h[i] = 1.0
            h[j] = 0.0
        # optionally, break if no improvement after many attempts
    return h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Maximum overlap integral for discrete h (full correlation)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth max via log-sum-exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP on smooth-max surrogate; returns feasible h."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    try:
        res = minimize(
            softmax_obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = np.clip(h0, 0.0, 1.0)
    # exact sum correction by water‑filling
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def adjust_sum_exact(h, rng):
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=20):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k != 0:
            up = n - k
            g[:up] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:up]           # - h[i]
        # (k==0 case gives g = 1-2h, but max never occurs at k=0)
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(150):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def fractional_to_binary(h, rng):
    """Keep exactly n/2 ones – take the largest entries."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    adjust_sum_exact(bin_h, rng)
    return bin_h

def binary_targeted_improve(h, rng, deadline):
    """Greedy swaps that lower the worst overlap."""
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        shift = abs(shift)
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = h[(ar + shift) % n] + h[(ar - shift) % n]

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        neigh_one = neigh[ones]
        i_cand = ones[neigh_one == neigh_one.min()]
        i = rng.choice(i_cand)

        neigh_zero = neigh[zeros]
        j_cand = zeros[neigh_zero == neigh_zero.max()]
        j = rng.choice(j_cand)

        delta = neigh[i] - neigh[j]
        if delta < 0:
            h[i] = 0.0
            h[j] = 1.0
            continue

        # try a few random swaps as a last resort
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            d = neigh[i2] - neigh[j2]
            if d < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

def binary_hill_climb(h, rng, deadline, batch=10, max_no_improve=200):
    """Random swap hill‑climbing on binary h."""
    n = h.size
    best_c = compute_c5(h)

    ones = [int(i) for i in np.flatnonzero(h == 1)]
    zeros = [int(i) for i in np.flatnonzero(h == 0)]
    one_pos = {i: idx for idx, i in enumerate(ones)}
    zero_pos = {i: idx for idx, i in enumerate(zeros)}

    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        best_swap = None
        best_c_swap = best_c
        for _ in range(batch):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            if c_new < best_c_swap - 1e-12:
                best_c_swap = c_new
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            # update bookkeeping
            i_pos = one_pos.pop(i)
            j_pos = zero_pos.pop(j)
            ones[i_pos] = j
            one_pos[j] = i_pos
            zeros[j_pos] = i
            zero_pos[i] = j_pos
            best_c = best_c_swap
            no_improve = 0
        else:
            no_improve += 1
    return h, best_c

            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=1200)

        h_up = gradient_refine(h_up, rng, step=0.025, max_iter=30)
        h_up = binary_targeted_improve(fractional_to_binary(h_up, rng), rng, deadline)

        # final hill‑climb on up‑sampled binary solution
        if time.time() < deadline:
            h_up, c_up = binary_hill_climb(h_up, rng, deadline)
        else:
            c_up = compute_c5(h_up)

        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ----- fallback if nothing found -----
    if best_h is None:
        best_n = n
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
