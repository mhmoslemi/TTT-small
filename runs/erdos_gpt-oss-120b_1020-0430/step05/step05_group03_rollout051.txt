    # Solve for λ such that sum(clip(v - λ, 0, 1)) = target.

    # Use bisection over λ in range [min(v)-1, max(v)].
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for h on [0,2] with uniform spacing."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_capped_simplex(v: np.ndarray, target: float, eps: float = 1e-12) -> np.ndarray:
    """Projection of v onto {x : sum x = target, 0 <= x <= 1} (L2)."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    # bisection for lambda such that sum clip(v - lam) = target
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def worst_gradient_multi(h: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """Average gradient of all shifts attaining the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxc = np.max(corr)
    active = np.where(np.abs(corr - maxc) <= tol)[0]
    grad = np.zeros_like(h, dtype=float)
    for idx in active:
        k = idx - (n - 1)          # shift (negative or positive)
        if k == 0:
            grad += 1.0 - 2.0 * h
        elif k > 0:
            # positive shift
            grad[: n - k] += 1.0 - h[k:]
            grad[k:]     -= h[: n - k]
        else:  # k < 0
            t = -k
            # negative shift
            grad[t:]     += 1.0 - h[: n - t]
            grad[: n - t] -= h[t:]
    grad /= max(1, len(active))
    return grad


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_onto_capped_simplex(v, n / 2.0)


def gradient_descent_proj(h0: np.ndarray, rng: np.random.Generator,
                         max_iters: int, step0: float,
                         target: float, deadline: float) -> (np.ndarray, float):
    """Projected gradient descent on the worst‑shift overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        grad = worst_gradient_multi(h)
        h_new = h - step * grad
        h_new = project_onto_capped_simplex(h_new, target)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.1, 1.0)   # modest acceleration
        else:
            step *= 0.7
            if rng.random() < 0.02:      # occasional random nudge
                h_tmp = h + rng.normal(scale=0.01, size=h.shape)
                h_tmp = project_onto_capped_simplex(h_tmp, target)
                c2 = compute_c5(h_tmp)
                if c2 < best_c:
                    best_c = c2
                    best_h = h_tmp.copy()
                    h = h_tmp
        if step < 1e-6:
            step = step0
    return best_h, best_c


# ---------- main driver ----------
