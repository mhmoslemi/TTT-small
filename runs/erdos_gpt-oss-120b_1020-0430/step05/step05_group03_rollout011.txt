import time
import numpy as np

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x | 0<=x<=1, sum x = target_sum}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Clamp to [0,1] and enforce the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    target = h.size / 2.0
    return project_onto_sum(h, target)


# ---------- smooth objective (soft‑max) ----------
def smooth_obj_and_grad(h: np.ndarray, beta: float):
    """Return (soft‑max objective, gradient) for given beta."""
    n = h.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    # FFT of h and reversed h
    H = np.fft.fft(h, m)
    H_rev = np.fft.fft(h[::-1], m)

    # linear autocorrelation via convolution
    conv = np.fft.ifft(H * H_rev, m).real[: 2 * n - 1]
    autocorr = conv[n - 1 - np.arange(n)]

    # soft‑max weights w_k ∝ exp(-β * autocorr_k)
    exp_arg = -beta * autocorr
    max_exp = np.max(exp_arg)
    shifted = np.exp(exp_arg - max_exp)
    sum_shifted = shifted.sum()
    w = shifted / sum_shifted

    # gradient = - (w * h convolved) - (w_rev * h convolved)
    W = np.fft.fft(w, m)
    conv_minus = np.fft.ifft(W * H, m).real[: 2 * n - 1]          # w ⋆ h
    W_rev = np.fft.fft(w[::-1], m)
    conv_plus = np.fft.ifft(W_rev * H, m).real[: 2 * n - 1]       # w_rev ⋆ h
    grad = - (conv_minus[:n] + conv_plus[n - 1 : n - 1 + n])

    # objective value (soft‑max of corr = S - autocorr, S constant)
    obj = (np.log(sum_shifted) + max_exp) / beta
    return obj, grad


# ---------- hill‑climb helpers ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift attaining the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]     -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1500, step_scale: float = 0.6):
    """Simple hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 8)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    start = time.time()
    time_limit = lambda frac: start + budget_s * frac

    # ----- initialise -----

        h0 = np.array(initial_h_values, dtype=float)
        h0 = enforce_sum(h0, rng)

    else:

        h0 *= (n0 / 2.0) / h0.sum()
        h0 = enforce_sum(h0, rng)
        best_h, best_c = h0, compute_c5(h0)

    n = best_h.size
    target_sum = n / 2.0

    # ----- smooth‑gradient descent over increasing β -----
    beta_schedule = [5., 20., 80., 200., 800., 3000., 8000., 20000., 50000.]
    for beta in beta_schedule:
        if time.time() > time_limit(0.80):
            break
        step = 0.2 / np.sqrt(beta)          # heuristic step size
        max_iters = 2500
        for it in range(max_iters):
            if time.time() > time_limit(0.80):
                break
            _, grad = smooth_obj_and_grad(best_h, beta)
            cand = best_h - step * grad
            cand = enforce_sum(cand, rng)
            c_val = compute_c5(cand)
            if c_val < best_c:
                best_c, best_h = c_val, cand.copy()
                step = min(step * 1.2, 1.0)    # modest step increase on improvement
            else:
                step *= 0.5
            if step < 1e-7:
                break

    # ----- aggressive local hill‑climb -----
    if time.time() < time_limit(0.90):
        best_h, best_c = guided_hill_climb(best_h, rng,

    # ----- final stochastic tweaks -----
    while time.time() < time_limit(0.99):
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ----- sanity fix -----
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
