    k = worst shift (abs idx)
    g = np.ones_like(h)
    if k > 0:
        n = h.size
        g[:n - k] -= h[k:]

    # Clip to [0,1]

    # Adjust sum to target exactly
    diff = target - new_h.sum()

        i = rng.integers(0, len(h))
        possible = (1.0 - new_h[i]) if diff > 0 else (-new_h[i])
        add = min(diff, possible) if diff > 0 else max(diff, possible)
        new_h[i] += add
        diff = target - new_h.sum()
    return new_h

def subgrad_optimize(h, target, rng, max_iters, deadline):
    n = h.size
    step0 = 0.5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # compute gradient
        g = np.ones_like(h)
        if k > 0:
            g[:n-k] -= h[k:]
            g[k:] -= h[:n-k]
        else:
            # k==0
            g = 1.0 - 2.0 * h
        step = step0 / np.sqrt(it + 1)
        h_new = h - step * g
        # clip and adjust sum
        h_new = np.clip(h_new, 0.0, 1.0)
        # adjust sum
        diff = target - h_new.sum()
        # adjust using random indices loop (like adjust_sum_exact)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h_new[i])
            else:
                add = max(diff, -h_new[i])
            h_new[i] += add
            diff = target - h_new.sum()
        h = h_new
    return h

import time

# ---------- helper functions ----------
def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx with discrete sampling."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> None:
    """Modify h in‑place so that Σh = n/2 exactly while staying in [0,1]."""
    n = h.size
    target = n / 2.0

            break

        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

def subgrad_optimize(
    h: np.ndarray,
    target: float,
    rng: np.random.Generator,
    max_iters: int,
    deadline: float,
) -> np.ndarray:
    """Projected subgradient descent on the worst shift."""
    n = h.size
    step0 = 0.5
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        # gradient of f_k = Σ h_i (1-h_{i+k})  (including sum term)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones_like(h)
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]

        h_new = np.clip(h_new, 0.0, 1.0)
        adjust_sum_exact(h_new, rng)

CURRENT_BETA = None

def softmax_obj(x):

    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
