        k0 = int(np.argmax(O_corr))
        O0_corr = O_corr[k0]
        grad = worst_gradient(h)
        # choose pos/neg idxs
        pos_idxs = np.argpartition(-grad, min(5, n-1))[:5]
        neg_idxs = np.argpartition(grad, min(5, n-1))[:5]
        H = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
        best_impr = 0.0
        best_move = None
        # evaluate pairs
        for i in pos_idxs:
            for j in neg_idxs:
                if i == j:

                max_eps = min(h[i], 1.0 - h[j])

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for the discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def enforce_sum_precise(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(1000):
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            idx = rng.choice(idxs)
            add = min(diff, 1.0 - h[idx])
        else:
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            idx = rng.choice(idxs)
            add = max(diff, -h[idx])
        h[idx] += add
        diff = target - h.sum()
    return h


def improve_h(h: np.ndarray, rng: np.random.Generator, deadline: float) -> np.ndarray:
    """Iterative line‑search hill‑climb using optimal ε transfers."""
    n = h.size
    ks = np.arange(-(n - 1), n, dtype=int)          # length 2n‑1
    target_sum = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)

    while time.time() < deadline:
        # current correlation (no dx)
        O = np.correlate(h, 1.0 - h, mode="full")
        k0 = int(np.argmax(O))
        O0 = O[k0]

        # gradient for the worst shift
        grad = worst_gradient(h)
        # candidate indices (top/low gradients)
        top = min(5, n - 1)
        pos_idx = np.argpartition(-grad, top)[:top]
        neg_idx = np.argpartition(grad, top)[:top]

        # padded h for fast shifted access
        H = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])

        best_impr = 0.0
        best_move = None

        for i in pos_idx:
            for j in neg_idx:
                if i == j:
                    continue
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 1e-12:
                    continue
                pi, pj = i + n - 1, j + n - 1
                d = (H[pi + ks] - H[pj + ks]) + (H[pi - ks] - H[pj - ks])
                d0 = d[k0]
                if d0 >= -1e-12:          # would not reduce the worst value
                    continue
                dd = d - d0
                diff = O0 - O
                mask = (dd > 1e-12) & (diff > 1e-12)
                if np.any(mask):
                    eps_opt = min((diff[mask] / dd[mask]).min(), max_eps)
                else:
                    eps_opt = max_eps
                new_Omax = np.max(O + eps_opt * d)
                impr = O0 - new_Omax
                if impr > best_impr + 1e-12:
                    best_impr = impr
                    best_move = (i, j, eps_opt, new_Omax)

        if best_move is not None:
            i, j, eps_opt, new_Omax = best_move
            h[i] -= eps_opt
            h[j] += eps_opt
            h[i] = max(0.0, min(1.0, h[i]))
            h[j] = max(0.0, min(1.0, h[j]))
            # tiny sum correction
            diff = target_sum - h.sum()
            if abs(diff) > 1e-12:
                idx = rng.integers(0, n)
                cap = (1.0 - h[idx]) if diff > 0 else -h[idx]
                adj = min(abs(diff), abs(cap))
                h[idx] += np.sign(diff) * adj
            cur_c = new_Omax * (2.0 / n)
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
        else:
            # random tiny transfer to escape plateaus
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 1e-12:
                    eps_opt = max_eps * rng.random() * 0.1
                    h[i] -= eps_opt
                    h[j] += eps_opt
                    diff = target_sum - h.sum()
                    if abs(diff) > 1e-12:
                        idx = rng.integers(0, n)
                        cap = (1.0 - h[idx]) if diff > 0 else -h[idx]
                        adj = min(abs(diff), abs(cap))
                        h[idx] += np.sign(diff) * adj
                    cur_c = compute_c5(h)
                    if cur_c < best_c - 1e-12:
                        best_c = cur_c
                        best_h = h.copy()
    return best_h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial construction -------------------------------------------------
