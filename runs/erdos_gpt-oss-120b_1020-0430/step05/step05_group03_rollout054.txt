def softmax_c5(x, beta):
    corr = np.correlate(x, 1 - x, mode="full")
    vals = corr * beta
    # log-sum-exp
    m = np.max(vals)
    return (m + np.log(np.sum(np.exp(vals - m)))) / beta

def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, max_iter=400):
    n = h0.size
    target = n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        vals = beta * corr
        m = np.max(vals)
        return (m + np.log(np.sum(np.exp(vals - m)))) / beta
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum due to tolerance
    return enforce_sum(x, np.random.default_rng())

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def binarize_by_top(x: np.ndarray) -> np.ndarray:
    """Turn a vector into binary by setting top half entries to 1."""
    n = x.size
    k = n // 2
    idx = np.argsort(-x)[:k]
    b = np.zeros(n, dtype=float)
    b[idx] = 1.0
    return b


def upsample_h(h: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of h to a larger grid."""
    n = h.size
    if n == n_target:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return np.clip(h_new, 0.0, 1.0)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr) - (n - 1)  # shift giving max C5
    grad = np.zeros_like(h)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        if k > 0:
            grad[: n - k] = 1.0 - h[k:]
            grad[k:] -= h[: n - k]
        else:
            k = -k
            grad[k:] = 1.0 - h[: n - k]
            grad[: n - k] -= h[k:]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator,
                   scale: float = 0.02) -> np.ndarray:
    """Small move along gradient direction while preserving sum."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

                     max_iters: int = 20000,
                     init_h: np.ndarray = None) -> tuple:
    """Max‑min self‑overlap via swaps on a binary vector."""
    if init_h is not None:
        h = (init_h >= 0.5).astype(float)
        cur = int(h.sum())
        target = n // 2
        if cur > target:
            idx = np.where(h == 1)[0]
            h[rng.choice(idx, cur - target, replace=False)] = 0.0
        elif cur < target:
            idx = np.where(h == 0)[0]
            h[rng.choice(idx, target - cur, replace=False)] = 1.0
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    conv = np.correlate(h, h, mode="full").astype(np.int64)
    offset = n - 1
    ones = set(np.where(h == 1)[0].tolist())
    best_h = h.copy()
    best_c = 1 - conv.min() * (2.0 / n)

    for it in range(max_iters):
        i = rng.choice(list(ones))
        while True:
            j = rng.integers(n)
            if j not in ones:
                break

        new_conv = conv.copy()
        # remove i
        for t in ones:
            if t == i:
                new_conv[offset] -= 1
            else:
                new_conv[i - t + offset] -= 1
                new_conv[t - i + offset] -= 1
        # add j
        for t in ones:
            if t == i:

            new_conv[j - t + offset] += 1
            new_conv[t - j + offset] += 1
        new_conv[offset] += 1  # self‑pair of j

        cur_c = 1 - conv.min() * (2.0 / n)
        new_c = 1 - new_conv.min() * (2.0 / n)

        # Metropolis acceptance
        T = 0.02 * (max_iters - it) / max_iters + 1e-6
        accept = new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T)

        if accept:
            h[i] = 0.0
            h[j] = 1.0
            conv = new_conv
            ones.remove(i)
            ones.add(j)
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
    return best_h, best_c


# ---------- continuous refinement ----------
def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        vals = beta * corr
        m = np.max(vals)
        return (m + np.log(np.sum(np.exp(vals - m)))) / beta

    rng = np.random.default_rng()
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, rng)


def continuous_refine(h: np.ndarray, rng: np.random.Generator,
                     beta_seq=None) -> np.ndarray:
    """Apply a sequence of SLSQP refinements."""
    if beta_seq is None:
        beta_seq = [20., 80., 200., 800., 3000., 8000., 20000.]
    for beta in beta_seq:
        h = slsqp_optimize(h, beta, max_iter=500)
    return h


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = float("inf")
    best_n = None

    def try_update(h):
        nonlocal best_h, best_c, best_n
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_n = c, h.copy(), h.size

    # ---- small scale binary search (n=64) ----
    n0 = 64
    for _ in range(3):
        if time.time() - start > budget_s * 0.1:
            break
        h0, _ = binary_hill_climb(n0, rng, max_iters=5000)
        try_update(h0)
        if time.time() - start < budget_s * 0.2:
            h1 = continuous_refine(h0, rng, beta_seq=[20., 80., 200., 800.])
            try_update(h1)

    # ---- upscale progressively ----
    for n_target in (128, 256):
        if best_h is None:
            h_start = np.zeros(n_target)
            h_start[rng.choice(n_target, n_target // 2, replace=False)] = 1.0
        else:
            h_start = upsample_h(best_h, n_target)
            h_start = binarize_by_top(h_start)

        max_it = 8000 if n_target == 128 else 12000
        h_opt, _ = binary_hill_climb(n_target, rng, max_iters=max_it, init_h=h_start)
        try_update(h_opt)

        if time.time() - start < budget_s * 0.8:
            h_cont = continuous_refine(h_opt, rng,
                                       beta_seq=[20., 80., 200., 800., 3000., 8000.])
            try_update(h_cont)

    # ---- final level n=512 ----
    n_big = 512
    if best_h is None or best_h.size != n_big:
        if best_h is not None:
            h_start = upsample_h(best_h, n_big)
            h_start = binarize_by_top(h_start)
        else:
            h_start = np.zeros(n_big)
            h_start[rng.choice(n_big, n_big // 2, replace=False)] = 1.0
    else:
        h_start = best_h.copy()

    # blend with parent if present
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        h_start = (h_start + parent) / 2.0
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = enforce_sum(h_start, rng)

    h_opt, _ = binary_hill_climb(n_big, rng, max_iters=25000, init_h=h_start)
    try_update(h_opt)

    if time.time() - start < budget_s * 0.95:
        h_ref = continuous_refine(h_opt, rng,
                                  beta_seq=[20., 80., 200., 800., 3000., 8000., 20000.])
        try_update(h_ref)

    # ---- final stochastic tweaks ----
    while time.time() - start < budget_s * 0.999:
        if best_h is None:
            break
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanitisation
    if best_h is None:
        best_h = np.zeros(128)
        best_h[rng.choice(128, 64, replace=False)] = 1.0
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
