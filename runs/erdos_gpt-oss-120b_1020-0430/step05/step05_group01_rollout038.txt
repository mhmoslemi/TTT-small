import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions (top level) ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x: np.ndarray, target: float) -> float:
    return np.sum(x) - target


def softmax_obj(x: np.ndarray, beta: float) -> float:
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


# ---------- main run ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.size
        target_sum = n / 2.0
        diff = target_sum - h.sum()
        while abs(diff) > 1e-12:
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target_sum - h.sum()
    else:
        n = 512
        h = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        target_sum = n / 2.0

    # ----- correlation A = h * h (full) -----
    A = np.correlate(h, h, mode="full")               # length 2n-1
    cur_c5 = 1.0 - (2.0 / n) * A.min()
    best_h = h.copy()
    best_c5 = cur_c5

    L = 2 * n - 1
    deltaA = np.empty(L, dtype=np.float64)
    mid = n - 1

    # ----- annealing / directed hill‑climb -----
    max_time = budget_s * 0.78                     # reserve time for final polishing
    it = 0
    while time.time() - start < max_time:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / max_time
        # temperature decays linearly
        T = 1e-4 * (1 - frac) + 1e-7 * frac

        # decide move type
        if rng.random() < 0.7:
            # gradient‑directed move
            s_idx = np.argmin(A)
            s_shift = s_idx - mid
            s_abs = abs(s_shift)
            if s_abs == 0:

            else:

                cont[s_abs:] += h[: n - s_abs]

            mask_i = h > 1e-12
            if not np.any(mask_i):

    # ----- final sanity check -----
    best_h = np.clip(best_h, 0.0, 1.0)
