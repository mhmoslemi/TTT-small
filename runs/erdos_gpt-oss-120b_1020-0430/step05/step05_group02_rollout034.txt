def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def binary_sa(h_bin, rng, deadline, max_iter=200000):
    n = h_bin.size
    target = n / 2.0
    corr = np.correlate(h_bin, h_bin, mode='full')
    cur_min = corr.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    T0 = cur_worst * 0.5 + 1e-6
    decay = 0.99999
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # Remove i
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i+1:]))
        diff_i = -1.0
        corr[start_i:start_i+n] += diff_i * hi_vec
        h_bin[i] = 0.0

        # Add j
        start_j = n - 1 - j
        # note h_bin now has i=0
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j+1:]))
        diff_j = 1.0
        corr[start_j:start_j+n] += diff_j * hj_vec
        h_bin[j] = 1.0

        new_min = corr.min()
        new_worst = target - new_min

        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)

def binary_sa(h_bin, rng, deadline, max_iter=200000):
    n = h_bin.size
    target = n / 2.0
    corr = np.correlate(h_bin, h_bin, mode="full")
    cur_min = corr.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    T0 = cur_worst * 0.5 + 1e-6
    decay = 0.99999
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # remove i (1->0)
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        diff_i = -1.0

        h_bin[i] = 0.0

        start_j = n - 1 - j

        c5 = worst * (2.0 / n)
        if c5 < best_c:
            best_h = bin_h.astype(float).copy()
            best_c = c5
            best_n = n

    # ---------- continuous refinement ----------
    h_frac = best_h.copy()
    h_frac = gradient_refine(h_frac, rng, step=0.02, max_iter=20)
    c5 = compute_c5(h_frac)
    if c5 < best_c:
        best_h, best_c = h_frac.copy(), c5

    beta_schedule = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0)
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            h_frac = slsqp_optimize(h_frac, beta, rng, max_iter=800)
        except Exception:
            continue
        c5 = compute_c5(h_frac)
        if c5 < best_c:
            best_h, best_c = h_frac.copy(), c5

    h_frac = gradient_refine(h_frac, rng, step=0.015, max_iter=12)
    c5 = compute_c5(h_frac)
    if c5 < best_c:
        best_h, best_c = h_frac.copy(), c5

    # ---------- optional up‑sampling ----------
    if best_h is not None and best_n * 2 <= 720 and time.time() < deadline - 5.0:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        for beta in (20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)
        h_up = gradient_refine(h_up, rng, step=0.015, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ---------- final safety ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
