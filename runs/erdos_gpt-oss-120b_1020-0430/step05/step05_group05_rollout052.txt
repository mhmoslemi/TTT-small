def compute_corr_self(h):
    n = len(h)
    L = 1 << (2*n-1).bit_length()  # next power of two >= 2n
    H = np.fft.rfft(h, n=L)
    corr = np.fft.irfft(H * np.conj(H), n=L)
    # linear correlation is first 2*n-1 entries
    return np.rint(corr[:2*n-1]).astype(np.int64)  # rounding

def simulated_annealing(h, rng, start, deadline, target_overlap):
    n = h.shape[0]
    best_h = h.copy()
    best_ov = max_overlap_from_h(h)
    current_ov = best_ov
    # precompute ones and zeros
    # but we recompute each iteration
    T0 = 2.0
    T_end = 1e-4
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)  # fraction of total SA time
        T = T0 * ((T_end / T0) ** frac)
        # get current ones and zeros
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_ov = max_overlap_from_h(h)
        delta = new_ov - current_ov
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            current_ov = new_ov
            if current_ov < best_ov:
                best_ov = current_ov
                best_h = h.copy()
                if best_ov <= target_overlap:
                    break
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # check time
        if time.time() >= deadline:
            break
    return best_h, best_ov

import time
import numpy as np

def _corr_self(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with itself (integer values)."""
    n = h.shape[0]
    L = 2 * n
    H = np.fft.fft(h, L)
    c = np.fft.ifft(H * np.conj(H)).real
    return np.rint(c[: 2 * n - 1]).astype(np.int64)


def _max_overlap(h: np.ndarray) -> int:
    """Maximum overlap max_k ∫ h(x)(1‑h(x+k))dx expressed as integer count."""
    n = h.shape[0]
    corr = _corr_self(h)
    # ignore the zero shift (corr[n‑1] = number of ones = n/2)
    corr[n - 1] = n  # a value larger than any possible min_corr
    min_corr = corr.min()
    return n // 2 - int(min_corr)


def _greedy_improve(h: np.ndarray, rng: np.random.Generator,
                    deadline: float) -> np.ndarray:
    """Simple greedy swaps that never increase the worst overlap."""
    n = h.shape[0]
    while time.time() < deadline:
        corr = _corr_self(h)
        corr[n - 1] = n
        min_corr = corr.min()
        max_overlap = n // 2 - int(min_corr)
        # shift achieving the worst overlap
        idx = np.where(corr == min_corr)[0][0]
        shift = idx - (n - 1)
        k = abs(shift)
        if k == 0:
            break
        up = n - k
        # positions that contribute to the worst overlap
        ones_local = np.where((h[:up] == 1) & (h[k:] == 0))[0]
        zeros_local = np.where((h[:up] == 0) & (h[k:] == 1))[0]
        if ones_local.size == 0 or zeros_local.size == 0:
            break
        i_loc = rng.choice(ones_local)
        j_loc = rng.choice(zeros_local) + k
        h[i_loc] = 0.0
        h[j_loc] = 1.0
    return h


def _simulated_annealing(h: np.ndarray, rng: np.random.Generator,
                         start: float, deadline: float,
                         target_overlap: int) -> tuple[np.ndarray, int]:
    """SA on binary vectors; returns best vector seen and its max overlap."""
    n = h.shape[0]
    best_h = h.copy()
    best_ov = _max_overlap(h)
    cur_ov = best_ov

    T0, T_end = 2.0, 1e-4
    total_time = deadline - start

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        T = T0 * ((T_end / T0) ** frac)

        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        new_ov = _max_overlap(h)
        delta = new_ov - cur_ov

        accept = False
        if delta <= 0:
            accept = True
        elif T > 0:
            accept = rng.random() < np.exp(-delta / T)

        if accept:
            cur_ov = new_ov
            if cur_ov < best_ov:
                best_ov = cur_ov
                best_h = h.copy()
                if best_ov <= target_overlap:
                    break
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_ov


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99  # safety margin

    # ----- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
        # binary start: take the largest n/2 entries
        idx = np.argpartition(parent, -n // 2)[-n // 2 :]
        h = np.zeros(n, dtype=np.float64)
        h[idx] = 1.0
    else:
        n = 360
        h = np.zeros(n, dtype=np.float64)
        h[rng.choice(n, n // 2, replace=False)] = 1.0

    # ----- greedy polishing --------------------------------------------
    h = _greedy_improve(h, rng, deadline)

    # ----- simulated annealing -----------------------------------------
    target_overlap = int(np.floor(0.3808 * n / 2))      # for n=360 → 68
    h, best_ov = _simulated_annealing(h, rng, start_time, deadline,
                                      target_overlap)

    best_c = best_ov * (2.0 / n)

    # ----- if not good enough, try double resolution --------------------
    if best_c > 0.3808 and time.time() < deadline:
        n2 = n * 2
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(xt, xs, h)                     # fractional values
        # binaryisation
        idx2 = np.argpartition(h_up, -n2 // 2)[-n2 // 2 :]
        h2 = np.zeros(n2, dtype=np.float64)
        h2[idx2] = 1.0

        # polish at larger scale
        h2 = _greedy_improve(h2, rng, deadline)
        target_overlap2 = int(np.floor(0.3808 * n2 / 2))  # for n=720 → 137
        h2, best_ov2 = _simulated_annealing(h2, rng, time.time(),
                                            deadline, target_overlap2)
        c2 = best_ov2 * (2.0 / n2)
        if c2 < best_c:
            h, best_c = h2, c2
            n = n2

    # ----- final sanity check -------------------------------------------
    h = np.rint(np.clip(h, 0, 1)).astype(np.float64)
    # enforce exact half‑ones if rounding drifted
    deficit = int(n // 2 - h.sum())
    if deficit > 0:
        zeros = np.flatnonzero(h == 0)
        h[rng.choice(zeros, deficit, replace=False)] = 1.0
    elif deficit < 0:
        ones = np.flatnonzero(h == 1)
        h[rng.choice(ones, -deficit, replace=False)] = 0.0

    # recompute final C5
    final_ov = _max_overlap(h)
    final_c = final_ov * (2.0 / n)

    return h.tolist(), float(final_c), int(n)
