import time, numpy as np, scipy.optimize as opt

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type":"eq","fun":sum_eq_constraint}
    res = opt.minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                       constraints=cons, options={"maxiter":max_iter,"ftol":1e-12,"disp":False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12: break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0

        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])

            thr_j = n - 1 - j
            delta_S = (t_arr <= thr_j).astype(np.int32) - (t_arr <= thr_i).astype(np.int32)

            new_max_O = new_O.max()

            else:

                    accept = True
            if accept:
                O = new_O
                A += delta_A
                cur_max_O = new_max_O
                # swap indices in sets
                ones_set[i_idx] = j
                zeros_set[j_idx] = i
                # swap values in h
                h[i] = 0.0
                h[j] = 1.0
                cur_c5 = cur_max_O * (2.0 / n)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
    return best_h, best_c5

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

import time
import numpy as np
import scipy.optimize as opt

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def binary_sa_incremental(n: int, rng: np.random.Generator,
                         time_budget: float):
    start = time.time()
    best_h = None
    best_c5 = np.inf
    starts = []
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size == n:
            hb = (h0 >= 0.5).astype(float)
            diff = hb.sum() - n / 2
            if diff > 0:
                ones = np.where(hb == 1)[0]
                hb[rng.choice(ones, int(diff), replace=False)] = 0
            elif diff < 0:
                zeros = np.where(hb == 0)[0]
                hb[rng.choice(zeros, int(-diff), replace=False)] = 1
            starts.append(hb)
    for _ in range(3):
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(h)

    per_start = time_budget / len(starts)
    t_arr = np.arange(n, dtype=np.int32)

    for h in starts:
        is_one = h.astype(bool)
        ones_set = np.where(is_one)[0].astype(np.int32)
        zeros_set = np.where(~is_one)[0].astype(np.int32)

        # autocorrelation A
        H = np.fft.rfft(h, n=2 * n - 1)
        ac_full = np.fft.irfft(H * np.conj(H), n=2 * n - 1).real
        A = np.rint(ac_full[n - 1 :: -1]).astype(np.int32)  # length n

        # prefix sums S
        csum = np.cumsum(h, dtype=np.int32)
        S = csum[::-1]  # S[t] = sum_{i=0}^{n-1-t} h_i

        O = S - A
        cur_max_O = O.max()
        cur_c5 = cur_max_O * (2.0 / n)
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, h.copy()

        start_loop = time.time()
        while time.time() - start_loop < per_start:
            elapsed = time.time() - start_loop
            frac = elapsed / per_start
            T = 0.02 * (1e-4 / 0.02) ** frac

            i_idx = rng.integers(len(ones_set))
            j_idx = rng.integers(len(zeros_set))
            i = int(ones_set[i_idx])
            j = int(zeros_set[j_idx])

            # other ones without i
            if i_idx == 0:
                p = ones_set[1:].copy()
            elif i_idx == len(ones_set) - 1:
                p = ones_set[:-1].copy()
            else:
                p = np.concatenate((ones_set[:i_idx], ones_set[i_idx + 1:]))

            lags_i = np.abs(i - p).astype(np.int32)
            lags_j = np.abs(j - p).astype(np.int32)
            cnt_i = np.bincount(lags_i, minlength=n)
            cnt_j = np.bincount(lags_j, minlength=n)
            delta_A = cnt_j - cnt_i

            thr_i = n - 1 - i
            thr_j = n - 1 - j
            delta_S = (t_arr <= thr_j) - (t_arr <= thr_i)
            delta_O = delta_S - delta_A

            new_max_O = new_O.max()

                cur_max_O = new_max_O
                ones_set[i_idx] = j
                zeros_set[j_idx] = i
                h[i] = 0.0
                h[j] = 1.0
                cur_c5 = cur_max_O * (2.0 / n)
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()
    return best_h, best_c5
