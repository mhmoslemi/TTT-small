def adjust_sum_exact(h, rng):
    # ensure sum = target = n/2
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # if diff positive, need to add mass; if negative, remove.
    # We can distribute diff across random indices handling bounds.
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            # add up to 1 - h[i]
            can = 1.0 - h[i]
            if can <= 0:
                continue
            add = min(diff, can)
            h[i] += add
            diff -= add
        else:
            # remove up to h[i]
            can = h[i]
            if can <= 0:
                continue
            sub = min(-diff, can)
            h[i] -= sub
            diff += sub

        else:

            # g[i] for i < up: derivative includes + (1 - h[i+k])
            g[:up] += 1.0 - h[k:]
            # g[i+k] includes - h[i] for i < up
            g[k:] -= h[:up]
        # step size decays
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # correct sum exactly
        diff = target - h.sum()

            # distribute diff by random indices similar to adjust_sum_exact
            for _ in range(200):

                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

        if max_transfer <= 0:
            continue

        h[i] -= delta
        h[j] += delta

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap for step function samples h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=5000):
    """Force sum(h) = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            can = 1.0 - h[i]
            if can <= 0:
                continue
            add = min(diff, can)
            h[i] += add
            diff -= add
        else:
            can = h[i]
            if can <= 0:
                continue
            sub = min(-diff, can)
            h[i] -= sub
            diff += sub
    return h

    """Simulated annealing on binary vectors."""

    cur_c = compute_c5(cur)

    zeros = list(np.flatnonzero(cur == 0))

    it = 0
    last_imp = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_imp = it
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it - last_imp > 8000:
            break
    return best.astype(float), best_c

def fractional_sa_hill_climb(h, rng, deadline):
    """Simulated annealing on continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_imp = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_tr = min(cur[src], 1.0 - cur[dst])
        if max_tr <= 0:
            continue
        delta = rng.random() * max_tr
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_imp = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_imp > 12000:
            break
    return best, best_c

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected GD against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def gradient_transfer_refine(h, rng, deadline):
    """Move mass from positions that increase the worst overlap to those that decrease it."""
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        if idx == 0:
            # find second worst if the worst is the trivial shift
            idx = np.argsort(-corr)[1] - (n - 1)
        k = abs(idx)
        if k == 0:
            break
        g = np.zeros(n)
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
        donors = np.flatnonzero(g > 1e-12)
        receivers = np.flatnonzero(g < -1e-12)
        if donors.size == 0 or receivers.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(receivers)
        max_tr = min(h[i], 1.0 - h[j])
        if max_tr <= 0:
            continue
        delta = rng.random() * max_tr
        h[i] -= delta
        h[j] += delta
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if "initial_h_values" in globals() else None
    ns = [360, 540, 720]
    for n in ns:
        if time.time() > deadline:
            break

        start_vecs = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp)

            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)

        # random fractional (uniform)
        rand_frac = rng.random(n)
        rand_frac *= (n / 2) / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)
        start_vecs.append(rand_frac)

        # random fractional (biased)
        rand_frac2 = rng.random(n) ** 2
        rand_frac2 *= (n / 2) / rand_frac2.sum()
        rand_frac2 = np.clip(rand_frac2, 0.0, 1.0)
        adjust_sum_exact(rand_frac2, rng)
        start_vecs.append(rand_frac2)

        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # binary SA if applicable
            if np.all((h == 0) | (h == 1)):
                sub_dead = min(deadline, time.time() + min(10.0, (deadline - time.time()) * 0.1))
                h_bin, c_bin = binary_sa_hill_climb(h.copy(), rng, sub_dead)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin.astype(float)

            # gradient‑transfer refinement (time‑limited)
            remaining = deadline - time.time()
            if remaining > 0:
                sub_dead = time.time() + min(5.0, remaining * 0.07)
                h = gradient_transfer_refine(h, rng, sub_dead)

            # fractional SA (time‑limited)
            remaining = deadline - time.time()
            if remaining > 0:
                sub_dead = time.time() + min(12.0, remaining * 0.15)
                h_frac, c_frac = fractional_sa_hill_climb(h.copy(), rng, sub_dead)
                if c_frac < best_c:
                    best_h, best_c, best_n = h_frac.copy(), c_frac, n
                h = h_frac

            # final gradient refinement
            h = gradient_refine(h, rng, step=0.02, max_iter=15)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # Upsample if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1080:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        remaining = deadline - time.time()
        if remaining > 0:
            sub_dead = time.time() + min(8.0, remaining * 0.2)
            h_up = gradient_transfer_refine(h_up, rng, sub_dead)

        remaining = deadline - time.time()
        if remaining > 0:
            sub_dead = time.time() + min(15.0, remaining * 0.3)
            h_up, c_up = fractional_sa_hill_climb(h_up, rng, sub_dead)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    # Fallback
    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
