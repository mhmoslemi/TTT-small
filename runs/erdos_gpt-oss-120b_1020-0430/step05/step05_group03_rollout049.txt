import time, math, numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
def compute_c5(h):
    """C5 for a step function discretized on [0,2]. h can be any numeric array."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_gradient(h):
    """Gradient of the worst‑shift overlap and the shift index."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)               # signed shift giving the maximum overlap
    grad = np.ones_like(h, dtype=float)
    if k != 0:
        kk = abs(k)
        grad[: n - kk] -= h[kk:]
        grad[kk:]      -= h[: n - kk]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad, k

def binary_from_float(x):
    """Return a binary vector with exactly n/2 ones, taken from the largest entries of x."""
    n = x.size
    idx = np.argpartition(x, -n // 2)[-n // 2:]
    b = np.zeros(n, dtype=np.uint8)
    b[idx] = 1
    return b

# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial binary vector -------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        best_h = binary_from_float(init)
    else:
        n = 512
        best_h = np.zeros(n, dtype=np.uint8)
        best_h[rng.choice(n, n // 2, replace=False)] = 1

    best_c = compute_c5(best_h.astype(float))

    # current state for simulated annealing
    cur_h = best_h.copy()
    cur_c = best_c

    # temperature schedule
    temp = best_c * 0.02 + 1e-6
    iter_cnt = 0
    last_restart = 0

    # ------------------------------------------------------------
    while time.time() - start < budget_s * 0.985:
        # ---- gradient‑guided local swaps ------------------------------------
        grad, _ = worst_gradient(cur_h.astype(float))
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        # sort candidates by gradient magnitude
        o_order = ones[np.argsort(-grad[ones])]
        z_order = zeros[np.argsort(grad[zeros])]
        K = min(12, o_order.size, z_order.size)
        improved = False
        best_swap = None
        best_swap_c = cur_c

        for i in o_order[:K]:
            for j in z_order[:K]:
                if grad[i] <= grad[j]:
                    continue
                # try swap i<->j
                cur_h[i] = 0
                cur_h[j] = 1
                c_new = compute_c5(cur_h.astype(float))
                if c_new < best_swap_c - 1e-12:
                    best_swap_c = c_new
                    best_swap = (i, j)
                    improved = True
                # revert
                cur_h[i] = 1
                cur_h[j] = 0

        if improved:
            i, j = best_swap
            cur_h[i] = 0
            cur_h[j] = 1
            cur_c = best_swap_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
            temp *= 0.995
            iter_cnt += 1
            continue

        # ---- simulated‑annealing random swap ---------------------------------
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c_new = compute_c5(cur_h.astype(float))
        delta = c_new - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:

            if rng.random() < prob:

        if accept:

            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            # revert the random move
            cur_h[i] = 1
            cur_h[j] = 0
        temp *= 0.9995
        iter_cnt += 1

        if iter_cnt - last_restart > 50000:

            cur_c = best_c
            last_restart = iter_cnt

        # ---- occasional continuous refinement (soft‑max SLSQP) ---------------
        if iter_cnt % 7000 == 0 and time.time() - start < budget_s * 0.90:
            # start from the best binary, allow fractional adjustments
            x = best_h.astype(float)
            target = n / 2.0
            cons = {"type": "eq", "fun": lambda v: np.sum(v) - target}
            bounds = [(0.0, 1.0)] * n

            def soft_obj(v, beta):

                bc = beta * corr
                m = np.max(bc)
                return (m + np.log(np.sum(np.exp(bc - m)))) / beta

        if max_eps <= 0:

        if c_tmp < best_c:
            best_c = c_tmp
            best_h = binary_from_float(h_float)

    # ---- final sanity --------------------------------------------------------
    best_h = np.clip(best_h, 0, 1).astype(np.uint8)
    final_c = compute_c5(best_h.astype(float))

    return best_h.tolist(), float(final_c), int(best_h.size)
