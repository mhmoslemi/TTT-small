def compute_c5_fft(h):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    # compute h padded
    H = np.fft.rfft(h, n=L)
    rev = (1.0 - h)[::-1]
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)[: 2*n-1]
    # Each element corresponds to sum_i h[i] * (1-h[i+k])
    # compute max
    max_val = np.max(conv)
    return float(max_val * (2.0 / n))

    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # identify ones and zeros indices arrays for fast random sampling
    ones = np.flatnonzero(cur_h)
    zeros = np.flatnonzero(1 - cur_h)

    # Determine initial temperature: maybe 0.05 * cur_c (heuristic) or higher.
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        # choose a random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5_fft(cur_h)
        # acceptance
        T = T0 * (0.99995 ** it) + 1e-12  # slower cooling
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            # update ones zeros
            ones[ones == i] = j  # replace i with j (but careful with array semantics)
            # Use list for dynamic modifications?

def compute_c5_fft(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(h, n=L)
    B = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(A * B, n=L)[:2 * n - 1]
    return float(conv.max() * (2.0 / n))

def fractional_sa_core(h, cur_c, deadline, rng):
    n = h.size
    best_h = h.copy()
    best_c = cur_c
    cur = h.copy()
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5_fft(cur)
        T = T0 * (0.99995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c)/T):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 20000:
            T0 = max(0.01 * cur_c, 1e-4)
            last_improve = it
    return best_h, best_c

import time
import numpy as np

def compute_c5_fft(h):
    """Maximum overlap using FFT (linear correlation)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n-1
    A = np.fft.rfft(h, n=L)
    B = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(A * B, n=L)[: 2 * n - 1]
    return float(conv.max() * (2.0 / n))


def adjust_sum_exact(arr, rng, max_iter=5000):
    """Redistribute mass so that sum(arr) == len(arr)/2, respecting 0<=arr<=1."""
    target = arr.size / 2.0
    diff = target - arr.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, arr.size)
        if diff > 0:
            add = min(diff, 1.0 - arr[i])
            if add <= 0:
                continue
            arr[i] += add
        else:
            add = max(diff, -arr[i])
            if add >= 0:
                continue
            arr[i] += add
        diff = target - arr.sum()
    return arr


def binary_sa_core(h, cur_c, deadline, rng):
    """Swap‑based simulated annealing for a binary vector."""
    n = h.size
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    best_h = h.copy()
    best_c = cur_c
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # perform swap
        h[i] = 0
        h[j] = 1
        c = compute_c5_fft(h)
        T = T0 * (0.99995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                last_improve = it
        else:
            h[i] = 1
            h[j] = 0
        it += 1
        if it - last_improve > 200000:
            T0 = max(0.01 * cur_c, 1e-4)
            last_improve = it
    return best_h, best_c


def binary_sa_random_restart(n, deadline, rng):
    """Repeated short SA runs from random starts (and optionally parent)."""
    best_h = None
    best_c = np.inf

    # optionally seed with the parent construction (if provided)
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            parent = np.interp(xt, xs, parent)
        idx = np.argpartition(parent, -n // 2)[-n // 2:]
        bin_parent = np.zeros(n, dtype=np.int8)
        bin_parent[idx] = 1
        cur_c = compute_c5_fft(bin_parent)
        inner_dead = min(deadline, time.time() + max(5.0, (deadline - time.time()) * 0.05))
        bh, bc = binary_sa_core(bin_parent.copy(), cur_c, inner_dead, rng)
        if bc < best_c:
            best_c, best_h = bc, bh

    while time.time() < deadline:
        remaining = deadline - time.time()
        sub_time = min(5.0, remaining * 0.2)
        inner_dead = time.time() + sub_time
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, n // 2, replace=False)] = 1
        cur_c = compute_c5_fft(h)
        bh, bc = binary_sa_core(h, cur_c, inner_dead, rng)
        if bc < best_c:
            best_c, best_h = bc, bh
    return best_h, best_c


def fractional_sa_core(h, cur_c, deadline, rng):
    """Transfer‑based SA on a continuous vector (preserves total mass)."""
    n = h.size
    best_h = h.copy()
    best_c = cur_c
    cur = h.copy()
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_tr = min(cur[src], 1.0 - cur[dst])
        if max_tr <= 0:
            continue
        delta = rng.random() * max_tr
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5_fft(cur)
        T = T0 * (0.99995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 20000:
            T0 = max(0.01 * cur_c, 1e-4)
            last_improve = it
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.shape[0]
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
