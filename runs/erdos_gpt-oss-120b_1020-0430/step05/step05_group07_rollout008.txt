    start = time.time()
    deadline = start + budget_s * 0.99

    x = np.arange(n) * (2.0 / n)   # points from 0 to <2
    # choose N harmonics

    # Build Fourier basis matrix A
    cos_terms = np.column_stack([np.cos(np.pi * k * x) for k in range(1, N+1)])
    sin_terms = np.column_stack([np.sin(np.pi * k * x) for k in range(1, N+1)])
    A = np.concatenate([cos_terms, sin_terms], axis=1)   # shape (n, 2N)
    p = A.shape[1]
    AtA = A.T @ A
    inv_AtA = np.linalg.inv(AtA)

    target_integral = 1.0  # not needed because average is 0.5

    def c_to_h(c):
        # compute h = 0.5 + A @ c
        h = 0.5 + A @ c
        # clip to [0,1]
        h_clip = np.clip(h, 0.0, 1.0)
        # project back to coefficients that produce h_clip
        # residual = h_clip - 0.5
        res = h_clip - 0.5
        c_proj = inv_AtA @ (A.T @ res)
        # recompute h from projected c (should equal h_clip within tolerance)
        h_proj = 0.5 + A @ c_proj
        # ensure numerically in [0,1]
        h_proj = np.clip(h_proj, 0.0, 1.0)
        return h_proj, c_proj
