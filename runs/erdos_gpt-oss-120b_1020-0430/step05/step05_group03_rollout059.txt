def evaluate_pattern(pattern, n, dx):
    tiled = np.tile(pattern, n // len(pattern)).astype(float)
    # amplitude of FFT length 2*n-1
    L = 2*n - 1
    F = np.fft.rfft(tiled, n=L)
    G = np.fft.rfft(1.0 - tiled, n=L)
    corr = np.fft.irfft(F * np.conj(G), n=L)
    return np.max(corr) * dx

import time, itertools, numpy as np, scipy.optimize as opt

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = opt.minimize(softmax, h0, method="SLSQP", bounds=bounds,
                       constraints=cons,
                       options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def compute_cross_binary(h: np.ndarray) -> np.ndarray:
    """Autocorrelation for binary h (cross[k] = Σ h_i h_{i+k})."""
    conv = np.convolve(h, h[::-1], mode="full")
    n = h.size
    # cross[0] = conv[n-1]; cross decreasing slice
    return conv[n - 1 :: -1][:n]


def binary_local_search(h: np.ndarray, rng: np.random.Generator,
                        time_limit: float) -> np.ndarray:
    """Improve binary h by swapping to raise the minimal autocorrelation."""
    n = h.size
    best_h = h.copy()
    best_cross = compute_cross_binary(best_h)
    best_min = best_cross.min()
    start = time.time()
    idx = np.arange(n)
    while time.time() < time_limit:
        cross = compute_cross_binary(best_h)
        min_cross = cross.min()
        k = int(np.argmin(cross))
        if k == 0:
            break
        # candidates that would improve shift k
        i_cand = idx[(best_h == 1) & (idx + k < n) & (best_h[idx + k] == 0)]
        j_cand = idx[(best_h == 0) & (idx + k < n) & (best_h[idx + k] == 1)]
        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
        else:
            i = rng.choice(np.where(best_h == 1)[0])
            j = rng.choice(np.where(best_h == 0)[0])
        best_h[i], best_h[j] = 0, 1
        new_cross = compute_cross_binary(best_h)
        new_min = new_cross.min()
        if new_min >= min_cross:
            if new_min > best_min:
                best_min = new_min
        else:
            best_h[i], best_h[j] = 1, 0
    return best_h


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)

    else:

        h0[rng.choice(n, n // 2, replace=False)] = 1.0
        best_h = h0
        best_c = compute_c5(best_h)
    n = best_h.size
    dx = 2.0 / n

    # ----- exhaustive search over small periods -----
    period_candidates = (8, 16)
    for p in period_candidates:
        if n % p:
            continue
        half = p // 2
        repeats = n // p
        # time guard
        if time.time() - start > budget_s * 0.15:
            break
        for comb in itertools.combinations(range(p), half):
            pattern = np.zeros(p, dtype=float)
            pattern[list(comb)] = 1.0
            tiled = np.tile(pattern, repeats)
            c = compute_c5(tiled)
            if c < best_c:
                best_c = c
                best_h = tiled
    # ----- binary local improvement -----
    time_for_bin = start + budget_s * 0.25
    if time.time() < time_for_bin:
        bin_h = (best_h > 0.5).astype(np.int8)
        # enforce exact half ones
        cur = bin_h.sum()
        if cur != n // 2:
            diff = n // 2 - cur
            if diff > 0:
                zero_idx = np.where(bin_h == 0)[0]
                bin_h[rng.choice(zero_idx, diff, replace=False)] = 1
            else:
                one_idx = np.where(bin_h == 1)[0]
                bin_h[rng.choice(one_idx, -diff, replace=False)] = 0
        bin_h = binary_local_search(bin_h, rng, time_for_bin)
        c = compute_c5(bin_h.astype(float))
        if c < best_c:
            best_c = c
            best_h = bin_h.astype(float)

    # ----- gradient‑based refinement -----
    # main hill‑climb
    max_hc_iters = 3000
    if time.time() - start < budget_s * 0.7:
        best_h, hc_c = guided_hill_climb(best_h, rng,
                                         max_iters=max_hc_iters,
                                         step_scale=0.7)
        if hc_c < best_c:
            best_c, best_h = hc_c, best_h.copy()

    # soft‑max SLSQP passes
    betas = (20., 80., 200., 800., 3000., 8000., 20000.)
    for beta in betas:
        if time.time() - start > budget_s * 0.9:
            break
        maxit = 800 if beta >= 200 else 500
        best_h = slsqp_optimize(best_h, beta, max_iter=maxit)
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c

    # final tidy‑up
    best_h = gradient_refine(best_h, rng, step=0.015, max_iter=12)
    best_c = compute_c5(best_h)

        nh = guided_perturb(best_h, grad, rng, scale=0.02)

        if nc < best_c:

    best_h = enforce_sum(best_h, rng)
