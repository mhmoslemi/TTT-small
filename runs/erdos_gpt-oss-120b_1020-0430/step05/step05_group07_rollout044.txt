def solve_binary_milp(n, time_limit):
    import cvxpy as cp, numpy as np, time

    # variables
    h = cp.Variable(n, boolean=True)
    T = cp.Variable()
    # pairwise y variables as a dict or array
    # Since indices varying per shift, we can flatten.

    # Let's create a list of (i,k) pairs and mapping to index.
    pair_indices = []
    for k in range(-(n-1), n):
        # For each k define i range
        imin = max(0, -k)
        imax = min(n, n - k)  # exclusive
        for i in range(imin, imax):
            pair_indices.append((i, i + k, k))
    m = len(pair_indices)
    y = cp.Variable(m, boolean=True)
    # Constraints for each pair: y <= h_i, y <= h_j, y >= h_i + h_j - 1
    constraints = [cp.sum(h) == n/2, T >= 0]

    for idx, (i, j, k) in enumerate(pair_indices):
        constraints.append(y[idx] <= h[i])
        constraints.append(y[idx] <= h[j])
        constraints.append(y[idx] >= h[i] + h[j] - 1)

    # For each shift k, sum y for that k >= n/2 - T
    # Build mapping shift to list of idx
    shift_to_idx = {}
    for idx, (i, j, k) in enumerate(pair_indices):
        shift_to_idx.setdefault(k, []).append(idx)
    for k, idxs in shift_to_idx.items():
        constraints.append(cp.sum(y[idxs]) >= n/2 - T)

    # Objective
    obj = cp.Minimize(T)

    prob = cp.Problem(obj, constraints)

    # Solve with solver
    # maybe set "max_iters" or "TimeLimit" but not all support.
    try:
        prob.solve(solver=cp.GLPK_MI, warm_start=True, msg=False)
    except cp.error.SolverError:
        # fallback: try ECOS_BB
        prob.solve(solver=cp.ECOS_BB, warm_start=True, msg=False)

    # After solving check if status is optimal/optimal_inaccurate.
    if h.value is None:
        return None, None
    h_sol = np.round(h.value).astype(float)  # binary
    T_val = T.value
    return h_sol, T_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # Try MILP for n = 180, then 144, 120.
    for n in [180, 144, 120]:
        if time.time() > deadline:
            break
        # time left for MILP
        time_left = deadline - time.time()
        # We'll allocate at most a portion e.g., min(300, time_left/2)
        # Not essential; just call solve.
        h_binary, T_val = solve_binary_milp(n, time_left)
        if h_binary is None:
            continue
        c = compute_c5(h_binary)
        if c < best_c:
            best_h, best_c, best_n = h_binary.copy(), c, n

        # Interpolate and refine if time permits
        # Upsample to n_up = n*2 if <= 720 (upper bound)
        n_up = n * 2
        if n_up <= 720 and time.time() < deadline:
            xs = np.linspace(0.0, 2.0, n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            # ensure bounds and sum
            adjust_sum_exact(h_up, rng)
            # refinement: gradient + SA cycles
            # Use small iteration counts to stay within time.
            # Use function gradient_refine defined earlier similar to previous code.
            # Use fractional_sa_hill_climb
            # We'll call them with time subdeadline.
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
            h_up = gradient_refine(h_up, rng, step=0.04, max_iter=15)
            h_up, c_up = fractional_sa_hill_climb(h_up, rng, sub_dead)
            # final gradient
            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # fallback: if no solution found, use simple uniform 0-1 half.
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n//2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)

import time, numpy as np, cvxpy as cp

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, iters=2000):
    """Redistribute mass so that sum(h) == len(h)/2 (within 1e-12)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(iters):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.03, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:]      -= h[:upper]           # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = adjust_sum_exact(h, rng, iters=200)
    return h

def fractional_sa_hill_climb(h, rng, deadline):
    """Transfer‑based simulated annealing on continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 12000:
            break
    return best, best_c

def solve_binary_milp(n, time_limit):
    """Solve the binary Erdős min‑overlap problem for given n."""
    # Build pair list
    pair_idx = []                # (i, j)
    shift_to_idxs = {}
    for k in range(-(n - 1), n):
        imin = max(0, -k)
        imax = min(n, n - k)
        for i in range(imin, imax):
            j = i + k
            idx = len(pair_idx)
            pair_idx.append((i, j))
            shift_to_idxs.setdefault(k, []).append(idx)

    m = len(pair_idx)
    h = cp.Variable(n, boolean=True)
    y = cp.Variable(m, boolean=True)
    T = cp.Variable()
    constraints = [cp.sum(h) == n / 2.0, T >= 0]

    # AND constraints
    for idx, (i, j) in enumerate(pair_idx):
        constraints.append(y[idx] <= h[i])
        constraints.append(y[idx] <= h[j])
        constraints.append(y[idx] >= h[i] + h[j] - 1)

    # Shift constraints: sum_i y_{i,k} >= n/2 - T
    rhs = n / 2.0 - T
    for idxs in shift_to_idxs.values():
        constraints.append(cp.sum(y[idxs]) >= rhs)

    prob = cp.Problem(cp.Minimize(T), constraints)
    # Try GLPK_MI with a time limit; fall back to ECOS_BB.
    try:
        prob.solve(solver=cp.GLPK_MI,
                    glpk={'tm_lim': int(time_limit * 1000)},
                    warm_start=True, msg=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS_BB, warm_start=True, msg=False)
        except Exception:
            return None, None

    if h.value is None:
        return None, None
    # Ensure binary (tiny numerical noise)
    h_sol = np.rint(np.clip(h.value, 0, 1)).astype(float)
    T_val = float(T.value) if T.value is not None else None
    return h_sol, T_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # Try MILP on a few sizes (larger first)
    for n in (180, 144, 120):
        if time.time() > deadline:
            break
        time_left = deadline - time.time()
        if time_left < 30:
            break
        h_bin, _ = solve_binary_milp(n, time_left * 0.5)
        if h_bin is None:
            continue
        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin.copy(), c_bin, n

        # Upsample and refine if we still have time
        n_up = n * 2
        if n_up <= 720 and time.time() < deadline:
            xs = np.linspace(0.0, 2.0, n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            adjust_sum_exact(h_up, rng)

            # Gradient step
            h_up = gradient_refine(h_up, rng, step=0.04, max_iter=15)

            # Fractional SA with remaining time slice
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.35)
            h_up, c_up = fractional_sa_hill_climb(h_up, rng, sub_dead)

            # Final gentle gradient
            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # Fallback simple construction
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
