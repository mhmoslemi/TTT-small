def compute_corr_fft(h):
    n = h.shape[0]
    size = 2*n - 1
    N = 1 << (size - 1).bit_length()
    # pack
    h_pad = np.zeros(N)
    g_pad = np.zeros(N)
    h_pad[:n] = h
    g_pad[:n] = 1.0 - h
    # reverse g for correlation
    g_rev = g_pad[::-1]
    # FFT
    H = np.fft.rfft(h_pad)
    G = np.fft.rfft(g_rev)
    conv = np.fft.irfft(H * G, n=N)
    return conv[:size]
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global helpers ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Correlation h with (1‑h) using FFT (matches np.correlate mode='full')."""
    n = h.shape[0]
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    h_pad = np.zeros(N)
    h_pad[:n] = h
    g_rev = np.zeros(N)
    g_rev[:n] = (1.0 - h)[::-1]               # reverse only the meaningful part
    H = np.fft.rfft(h_pad)
    G = np.fft.rfft(g_rev)
    conv = np.fft.irfft(H * G, n=N)
    return conv[:size]


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 20000:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:
        # small deterministic fix
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h[0] += diff
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = compute_corr_fft(h)
    idx = np.argmax(corr)
    k = abs(idx - (n - 1))
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-5)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = compute_corr_fft(h)
        idx = np.argmax(corr)
        k = abs(idx - (n - 1))
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    budget = budget_s

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
    else:
        n0 = 512
        h0 = np.zeros(n0)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0
        h0 = enforce_sum(h0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)
    best_n = best_h.size

    # ----- stage 1 (512‑point intensive refinement) -----
    stage1_end = start + 0.60 * budget
    while time.time() < stage1_end:
        # hill‑climb
        h_new, c_new = guided_hill_climb(best_h.copy(), rng,
                                         max_iters=4000, step_scale=0.8)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new

        # soft‑max SLSQP passes
        for beta in (10.0, 30.0, 100.0, 300.0, 1000.0,
                     3000.0, 10000.0, 30000.0, 100000.0):
            if time.time() >= stage1_end:
                break
            max_iter = 800 if beta < 300 else 500
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
            c_new = compute_c5(best_h)
            if c_new < best_c:
                best_c = c_new

        # gradient tidy‑up
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=12)
        c_new = compute_c5(best_h)
        if c_new < best_c:
            best_c = c_new

        # occasional random swap
        if rng.random() < 0.2:
            i, j = rng.integers(0, best_n, size=2)
            best_h[i], best_h[j] = best_h[j], best_h[i]
            best_h = enforce_sum(best_h, rng)
            c_new = compute_c5(best_h)
            if c_new < best_c:
                best_c = c_new

    # ----- up‑sample to 1024 if time allows -----
    if time.time() < start + 0.95 * budget:
        target_n = 1024
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, target_n
        else:
            best_n = best_h.size

        # ----- stage 2 (1024‑point refinement) -----
        stage2_end = start + 0.99 * budget
        while time.time() < stage2_end:
            # hill‑climb
            max_iters = int(6000 * 256 / best_n) + 500
            h_new, c_new = guided_hill_climb(best_h.copy(), rng,
                                             max_iters=max_iters,
                                             step_scale=0.7)
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new

            # SLSQP passes (moderate β)
            for beta in (10.0, 30.0, 100.0, 300.0, 1000.0,
                         3000.0, 10000.0, 30000.0):
                if time.time() >= stage2_end:
                    break
                max_iter = 600 if beta < 300 else 400
                best_h = slsqp_optimize(best_h, beta, rng, max_iter=max_iter)
                c_new = compute_c5(best_h)
                if c_new < best_c:
                    best_c = c_new

            # gradient polish
            best_h = gradient_refine(best_h, rng, step=0.005, max_iter=15)
            c_new = compute_c5(best_h)
            if c_new < best_c:
                best_c = c_new

    # ----- final stochastic tweaking until timeout -----
    while time.time() - start < budget:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.01)
        nh = enforce_sum(nh, rng)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
