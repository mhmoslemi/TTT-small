import time
import numpy as np

def _compute_inner_corr(h: np.ndarray) -> np.ndarray:
    """Circular autocorrelation of a binary vector using FFT (int values)."""
    # FFT works with float, result should be integer up to rounding error
    f = np.fft.rfft(h)
    corr = np.fft.irfft(f * np.conj(f), n=h.size)
    return np.rint(corr).astype(np.int64)  # inner[k] = sum_i h[i] * h[i+k]

def _greedy_improve(h, inner, idx_arr, rng, deadline, best_h, best_min):
    """Try to improve the worst shift by a deterministic swap."""
    n = h.size
    while time.time() < deadline:
        s = int(inner.argmin())
        if s == 0:  # cannot improve shift 0
            break
        # neighbour count = h[i+s] + h[i-s] for each i
        neigh = np.roll(h, -s) + np.roll(h, s)
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        neigh_one = neigh[ones]
        min_val = int(neigh_one.min())
        cand_i = ones[neigh_one == min_val]
        i = int(rng.choice(cand_i))
        neigh_zero = neigh[zeros]
        max_val = int(neigh_zero.max())
        cand_j = zeros[neigh_zero == max_val]
        j = int(rng.choice(cand_j))

        # compute effect of swapping i (1) with j (0)
        delta = (
            -h[(i + idx_arr) % n]
            - h[(i - idx_arr) % n]
            + h[(j + idx_arr) % n]
            + h[(j - idx_arr) % n]
        )
        new_inner = inner + delta
        new_min = int(new_inner.min())
        if new_min > int(inner.min()):
            # perform swap
            h[i] = 0
            h[j] = 1
            inner = new_inner
            if new_min > best_min:
                best_min = new_min
                best_h = h.copy()
        else:
            break
    return h, inner, best_h, best_min

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ---------- choose size ----------
    TARGET_N = 720  # we aim for the largest allowed size for a finer bound
    # ---------- initial binary ----------
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
        # make binary from parent (top half entries)
        bin_parent = np.zeros(n0, dtype=np.int8)
        idx = np.argpartition(parent, -n0 // 2)[-n0 // 2 :]
        bin_parent[idx] = 1
        # repeat / truncate to reach TARGET_N
        repeats = TARGET_N // n0
        remainder = TARGET_N % n0
        h = np.tile(bin_parent, repeats)
        if remainder:
            h = np.concatenate([h, bin_parent[:remainder]])
    else:
        h = np.zeros(TARGET_N, dtype=np.int8)
        h[rng.choice(TARGET_N, TARGET_N // 2, replace=False)] = 1

    n = h.size
    w = n // 2
    # ensure exact weight
    diff = w - int(h.sum())
    if diff != 0:
        zeros = np.flatnonzero(h == 0)
        ones = np.flatnonzero(h == 1)
        if diff > 0:
            sel = rng.choice(zeros, diff, replace=False)
            h[sel] = 1
        else:
            sel = rng.choice(ones, -diff, replace=False)
            h[sel] = 0

    # ---------- pre‑computations ----------
    idx_arr = np.arange(n, dtype=np.int64)
    inner = _compute_inner_corr(h)

    best_min = int(inner.min())
    best_h = h.copy()

    # ---------- greedy deterministic improvement ----------
    h, inner, best_h, best_min = _greedy_improve(
        h, inner, idx_arr, rng, deadline, best_h, best_min
    )

    # early exit if already good enough
    target_min = int(np.ceil((1.0 - 0.3808) * w))
    if best_min >= target_min:
        c5 = 1.0 - best_min / w
        return best_h.astype(float).tolist(), float(c5), int(n)

    # ---------- simulated annealing ----------
    # temperature schedule (linear cooling)
    T0 = 5.0
    Tmin = 1e-4
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (budget_s * 0.99)
        T = max(Tmin, T0 * (1.0 - fraction))

        # pick random 1 and random 0
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))

        # effect of swapping i and j
        delta = (
            -h[(i + idx_arr) % n]
            - h[(i - idx_arr) % n]
            + h[(j + idx_arr) % n]
            + h[(j - idx_arr) % n]
        )
        new_inner = inner + delta
        new_min = int(new_inner.min())
        cur_min = int(inner.min())
        delta_min = new_min - cur_min

        accept = False
        if delta_min > 0:
            accept = True
        else:
            # metropolis acceptance
            if T > 0 and rng.random() < np.exp(delta_min / T):
                accept = True

        if accept:
            # apply swap
            h[i] = 0
            h[j] = 1
            inner = new_inner
            if new_min > best_min:
                best_min = new_min
                best_h = h.copy()
                if best_min >= target_min:
                    break  # achieved target

    # ---------- final bound ----------
    c5 = 1.0 - best_min / w
    # final sanity checks
    best_h = np.clip(best_h, 0, 1)
    assert int(best_h.sum()) == w
    return best_h.astype(float).tolist(), float(c5), int(n)
