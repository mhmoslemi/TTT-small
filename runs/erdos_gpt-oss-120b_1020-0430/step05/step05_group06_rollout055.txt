def smoothmax_and_grad(h, beta):
    n = len(h)
    a = 1.0 - h
    corr = np.correlate(h, a, mode='full') * dx
    # compute w
    # using stable: maxv = np.max(beta*corr); w = np.exp(beta*corr - maxv); w /= w.sum()
    maxv = np.max(beta*corr)
    expb = np.exp(beta*corr - maxv)
    w = expb / expb.sum()
    f = (maxv + np.log(expb.sum())) / beta
    # gradient
    grad = np.zeros_like(h)
    # s from -(n-1) to (n-1)
    for s in range(-(n-1), n):
        w_s = w[s + (n-1)]
        if w_s == 0:
            continue
        if s >= 0:
            # i in 0..n-1-s
            i_end = n - s
            # vectorized:
            # grad[i] += w_s * (1 - h[i+s])
            # grad[i+s] -= w_s * h[i]
            # Use slicing
            grad[:i_end] += w_s * a[s:]  # a[i+s] = 1 - h[i+s]
            grad[s:] -= w_s * h[:i_end]
        else:
            t = -s
            i_end = n - t
            # for s negative, corr[s] = sum_{i=0}^{n-1-t} h[i+t] * a[i]
            # contribution: ∂/∂h[j] of term = a[j-t] if j>=t (for h[i+t]) and -h[j+t] if j < n-t (for a[i])
            # Equivalent to swapping roles: we can treat it similarly to s positive but swapping h and a.
            # Let's derive: sum_{i=0}^{n-1-t} h[i+t] * a[i]; derivative wrt h[j]:
            # For j = i+t → i = j - t, valid if j - t >=0 and <= n-1-t => j >= t and j <= n-1
            # contribution = a[j - t]

def smoothmax_obj_grad(h, beta, p, target, dx):
    n = h.shape[0]
    a = 1.0 - h
    # compute correlation
    corr = np.correlate(h, a, mode='full') * dx
    # compute softmax weights
    # to avoid overflow:
    y = beta * corr
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    sum_exp = np.sum(exp_y)
    w = exp_y / sum_exp
    # smooth max value
    f = (maxy + np.log(sum_exp)) / beta
    # penalty
    s = np.sum(h)
    diff = s - target
    f += p * diff * diff
    # gradient
    grad = np.zeros_like(h)
    # w_s = w[shift + n-1]
    # loops
    for sft in range(-(n-1), n):
        ws = w[sft + (n-1)]
        if ws == 0.0:
            continue
        if sft >= 0:
            end = n - sft
            # grad[:end] += ws * a[sft:]
            grad[:end] += ws * a[sft:]
            # grad[sft:] -= ws * h[:end]
            grad[sft:] -= ws * h[:end]

            end = n - t
            grad[t:] += ws * a[:end]
            grad[:end] -= ws * h[t:]
    grad *= dx

    grad += 2.0 * p * diff
    return f, grad

import time
import numpy as np
from scipy.optimize import minimize

def adjust_sum_exact(h, rng):
    """Project h onto the hyperplane sum(h)=len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def compute_c5(h):
    """Exact C5 value for a discrete step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def smoothmax_obj_grad(h, beta, p, target, dx):
    """Smooth‑max objective (log‑sum‑exp) with sum‑penalty and its gradient."""
    n = h.shape[0]
    a = 1.0 - h
    corr = np.correlate(h, a, mode="full") * dx

    y = beta * corr
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    sumexp = np.sum(exp_y)
    w = exp_y / sumexp

    f = (maxy + np.log(sumexp)) / beta
    diff = np.sum(h) - target
    f += p * diff * diff

    grad = np.zeros_like(h)
    for s in range(-(n - 1), n):

            grad[:end] -= ws * h[t:]
    grad *= dx
    grad += 2.0 * p * diff
    return f, grad
