def simulated_annealing_binary(h, rng, time_deadline):
    n = h.size
    # Compute initial corr and max_corr
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    cur_c5 = max_corr * (2.0 / n)
    # Lists of ones and zeros
    ones = np.flatnonzero(h == 1).tolist()
    zeros = np.flatnonzero(h == 0).tolist()
    # Temperature schedule
    T = cur_c5 * 0.05 + 1e-12  # adjust
    # iteration count
    it = 0
    best_c5 = cur_c5
    best_h = h.copy()
    best_corr = corr.copy()
    while time.time() < time_deadline and it < 500000:
        # Choose random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Evaluate new corr
        # Use copy method
        new_corr = corr.copy()
        # val_i = 1, val_j = 0 scenario
        # Add contributions for i
        n = h.size
        new_corr[(n-1)-i:(n-1)-i + n] += h
        new_corr[i:i+n] += h[::-1]
        # No subtraction for j because h[j]==0
        # compute new max
        new_max_corr = new_corr.max()
        new_c5 = new_max_corr * (2.0 / n)
        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # update state
            corr = new_corr
            max_corr = new_max_corr
            cur_c5 = new_c5
            # update lists and h
            h[i] = 0
            h[j] = 1
            # move i from ones to zeros, j from zeros to ones
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            # track best
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_corr = corr.copy()
                # optional early exit if good enough
        # cooling
        it += 1
        T *= 0.99999  # adjust
    # return best
    return best_h, best_c5

import time
import numpy as np
from scipy.optimize import minimize

# ---------- globals for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- core utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=30):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:upper]            # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- binary simulated annealing ----------
def sa_binary_optimize(h, rng, deadline):
    """Binary (0/1) simulated‑annealing using O(n) updates."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_c5 = corr.max() * (2.0 / n)
    best_c5 = cur_c5
    best_h = h.copy()
    best_corr = corr.copy()
    ones = np.flatnonzero(h == 1).tolist()
    zeros = np.flatnonzero(h == 0).tolist()
    T0 = max(cur_c5 * 0.02, 1e-12)
    T = T0
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # copy correlation
        new_corr = corr.copy()
        # contributions for i (currently 1)
        new_corr[(n - 1) - i : (n - 1) - i + n] += h
        new_corr[i : i + n] += h[::-1]
        # contributions for j (currently 0) are subtracted (i.e., +0 - 0), so just remove nothing
        # but we also need to subtract j's future contribution (it will become 1)
        # Net delta = contrib_i - contrib_j; since j is 0, contrib_j = 0
        new_max = new_corr.max()
        new_c5 = new_max * (2.0 / n)
        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            corr = new_corr
            cur_c5 = new_c5
            # perform swap
            h[i] = 0.0
            h[j] = 1.0
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()
                best_corr = corr.copy()
                last_improve = it
        it += 1
        T *= 0.99998
        if it - last_improve > 20000:
            T = T0
            last_improve = it
    return best_h, best_c5

    # candidate sizes (even)
    cand_sizes = []
    if parent is not None:
        cand_sizes.append(parent.size)
        cand_sizes.append(int(parent.size * 1.5) // 2 * 2)  # 1.5× rounded to even
    cand_sizes.extend([540, 720])
    cand_sizes = [n for n in cand_sizes if n % 2 == 0]
    cand_sizes = list(dict.fromkeys(cand_sizes))  # unique, preserve order

    for n in cand_sizes:
        if time.time() >= deadline:
            break
        # ---- build starting binary vector ----
        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            base = np.interp(xt, xs, parent)
        else:
            base = rng.random(n)
        # pick top n/2 entries
        idx = np.argpartition(base, -n // 2)[-n // 2:]
        h = np.zeros(n)
        h[idx] = 1.0
        adjust_sum_exact(h, rng)  # safety

        # ---- binary simulated annealing ----
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
        h_sa, c_sa = sa_binary_optimize(h.copy(), rng, sub_dead)
        if c_sa < best_c:
            best_h, best_c, best_n = h_sa.copy(), c_sa, n

        # ---- gradient refinement (continuous) ----
        h_grad = gradient_refine(best_h.copy(), rng, step=0.02, max_iter=30)
        c_grad = compute_c5(h_grad)
        if c_grad < best_c:
            best_h, best_c, best_n = h_grad.copy(), c_grad, n

        # ---- SLSQP cascade (soft‑max) ----
        betas = (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 50000.0)
        for beta in betas:
            if time.time() >= deadline:
                break
            max_iter = 400 if beta < 200 else 600
            if beta >= 20000:
                max_iter = 800
            try:
                h_opt = slsqp_optimize(best_h.copy(), beta, rng, max_iter=max_iter)
            except Exception:
                continue
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n

        # ---- final gradient polish ----
        h_final = gradient_refine(best_h.copy(), rng, step=0.02, max_iter=40)
        c_final = compute_c5(h_final)
        if c_final < best_c:
            best_h, best_c, best_n = h_final.copy(), c_final, n

    # ---- optional up‑sampling if time permits ----
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)
        adjust_sum_exact(h_up, rng)
        # brief SA on the up‑sampled version
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
        h_up_sa, c_up_sa = sa_binary_optimize(h_up.copy(), rng, sub_dead)
        if c_up_sa < best_c:
            best_h, best_c, best_n = h_up_sa, c_up_sa, n_up
        # final refinement
        h_up_ref = gradient_refine(best_h.copy(), rng, step=0.02, max_iter=30)
        c_up_ref = compute_c5(h_up_ref)
        if c_up_ref < best_c:
            best_h, best_c, best_n = h_up_ref, c_up_ref, n_up

    # ---- fallback in the extremely unlikely case ----
    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    # ----- final safety handling -----
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
