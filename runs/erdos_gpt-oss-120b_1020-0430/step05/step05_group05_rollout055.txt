import time, numpy as np

def _c5_fft(h):
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, size)
    f1 = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(np.conj(fh) * f1, size)
    return float(np.max(corr[: 2 * n - 1]) * (2.0 / n))

def _sa(h, deadline, rng):
    n = h.shape[0]
    cur = h.copy()
    cur_c = _c5_fft(cur)
    best = cur.copy()
    best_c = cur_c
    T = 0.02
    while time.time() < deadline:

            i = rng.integers(0, n)
        j = rng.integers(0, n)

            cur[i] = 1.0
            cur[j] = 0.0

def _polish(h, deadline, rng):

    cur = h.copy()
    cur_c = _c5_fft(cur)

        fh = np.fft.rfft(cur, size)
        f1 = np.fft.rfft(1.0 - cur, size)

        idx = np.argmax(corr[: 2 * n - 1])
        shift = idx - (n - 1)
        if shift == 0:
            break
        k = abs(shift)
        if shift > 0:
            bad = np.where((cur[:n - k] == 1) & (cur[k:] == 0))[0]
            good = np.where((cur[:n - k] == 0) & (cur[k:] == 1))[0]
        else:
            bad = np.where((cur[k:] == 1) & (cur[:n - k] == 0))[0] + k
            good = np.where((cur[k:] == 0) & (cur[:n - k] == 1))[0] + k
        if bad.size == 0 or good.size == 0:
            break
        i = rng.choice(bad)
        j = rng.choice(good)
        cur[i] = 0.0
        cur[j] = 1.0
        new_c = _c5_fft(cur)
        if new_c < cur_c:
            cur_c = new_c
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            break
    return cur, cur_c

        h0[idx] = 1.0
    else:
        n0 = 360
        h0 = np.zeros(n0)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0

    n = n0 * 2
    h = np.repeat(h0, 2)
    best_h = h.copy()
    best_c = _c5_fft(best_h)
    # main loop
    while time.time() < deadline:
        remaining = deadline - time.time()

            break

        slice_time = min(30.0, remaining)

        if rng.random() < 0.7:

            start_h[rng.choice(n, n // 2, replace=False)] = 1.0

        else:

            idx = rng.choice(ones, cur_sum - target, replace=False)
            best_h[idx] = 0.0

    return best_h.tolist(), float(final_c), int(n)
