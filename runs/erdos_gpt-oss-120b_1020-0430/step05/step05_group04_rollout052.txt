def greedy_improve(h, rng, max_iter=50, deadline=None, max_samples=2000):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        # indices
        i_range = np.arange(n)
        # Determine S1,S0 as in previous code
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        # sample pairs
        if total_pairs <= max_samples:
            # evaluate all
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

def gradient_refine(h, rng, init_step=0.05, max_iter=30):
    n = h.size
    target = n / 2.0
    step = init_step
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # part from (1 - h[i+k])
            g[k:] -= h[:upper]               # part from -h[i] in product
        # step schedule: decay by factor 0.7 per iteration perhaps.
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        # adjust sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # perform few adjustments
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
        # decay step
        step *= 0.7
    return h

import time
import numpy as np
from scipy.optimize import minimize

# global helpers for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def gradient_refine(h, rng, init_step=0.05, max_iter=30):
    n = h.size
    target = n / 2.0
    step = init_step
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break

    return h

def greedy_improve(h, rng, max_iter=50, deadline=None, max_samples=2000):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)

        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break

        total_pairs = int(S1.size) * int(S0.size)
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]

        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

        alt = np.tile([1, 0], n // 2).astype(float)
        start_vecs.append(alt)

        # Thue‑Morse based binary
        tm = np.array([bin(i).count("1") % 2 for i in range(n)], dtype=float)
        ones = np.sum(tm)
        if ones != n / 2:
            if ones > n / 2:
                idx = np.flatnonzero(tm == 1)
                flip = rng.choice(idx, int(ones - n / 2), replace=False)
                tm[flip] = 0.0
            else:
                idx = np.flatnonzero(tm == 0)
                flip = rng.choice(idx, int(n / 2 - ones), replace=False)
                tm[flip] = 1.0
        start_vecs.append(tm)

        for base_h in start_vecs:
            if time.time() > deadline:
                break
            h = base_h.copy()
            adjust_sum_exact(h, rng)
            h = np.clip(h, 0.0, 1.0)

            # binary branch
            if np.all((h == 0) | (h == 1)):
                sub_dead = min(deadline, time.time() + min(5.0, (deadline - time.time()) * 0.25))
                h_bin, c_bin = binary_sa(h.copy(), rng, sub_dead)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin.astype(float)

            else:
                # continuous refinement
                h = gradient_refine(h, rng, init_step=0.05, max_iter=20)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

                # smooth‑max cascade
                for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000):
                    if time.time() > deadline:
                        break
                    max_iter = 300 if beta < 200 else 600
                    if beta >= 2000:
                        max_iter = 800
                    try:
                        h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                    except Exception:
                        continue
                    c = compute_c5(h)
                    if c < best_c:
                        best_h, best_c, best_n = h.copy(), c, n

                # final gradient
                h = gradient_refine(h, rng, init_step=0.02, max_iter=30)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

                # turn into binary and finish
                if not np.all((np.abs(h) < 1e-12) | (np.abs(h - 1) < 1e-12)):
                    idx = np.argpartition(h, -n // 2)[-n // 2:]
                    h_bin = np.zeros(n)
                    h_bin[idx] = 1.0
                    adjust_sum_exact(h_bin, rng)

                    sub_dead = min(deadline, time.time() + min(5.0, (deadline - time.time()) * 0.25))
                    h_bin, c_bin = binary_sa(h_bin.copy(), rng, sub_dead)
                    if c_bin < best_c:
                        best_h, best_c, best_n = h_bin.copy(), c_bin, n

                    h_bin = greedy_improve(
                        h_bin, rng, max_iter=50, deadline=deadline, max_samples=2000
                    )
                    c_bin = compute_c5(h_bin)
                    if c_bin < best_c:
                        best_h, best_c, best_n = h_bin.copy(), c_bin, n
                    h = h_bin

                # optional extra greedy on a binary solution
                if np.all((np.abs(h) < 1e-12) | (np.abs(h - 1) < 1e-12)):
                    h = greedy_improve(
                        h, rng, max_iter=30, deadline=deadline, max_samples=1500
                    )
                    c = compute_c5(h)
                    if c < best_c:
                        best_h, best_c, best_n = h.copy(), c, n

    # up‑sample best result if time permits
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 720
    ):
        n_up = best_n * 2
        x_orig = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_up, x_orig, best_h)
        adjust_sum_exact(h_up, rng)

        h_up = gradient_refine(h_up, rng, init_step=0.05, max_iter=20)
        for beta in (200, 500, 1000, 2000, 5000, 10000, 20000):
            if time.time() > deadline:
                break
            try:
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
            except Exception:
                continue
        h_up = gradient_refine(h_up, rng, init_step=0.02, max_iter=30)

        if not np.all((np.abs(h_up) < 1e-12) | (np.abs(h_up - 1) < 1e-12)):
            idx = np.argpartition(h_up, -n_up // 2)[-n_up // 2:]
            h_bin_up = np.zeros(n_up)
            h_bin_up[idx] = 1.0
            adjust_sum_exact(h_bin_up, rng)
            sub_dead = min(deadline, time.time() + min(5.0, (deadline - time.time()) * 0.2))
            h_bin_up, c_bin_up = binary_sa(h_bin_up, rng, sub_dead)
            if c_bin_up < best_c:
                best_h, best_c, best_n = h_bin_up.copy(), c_bin_up, n_up
            h_bin_up = greedy_improve(
                h_bin_up, rng, max_iter=50, deadline=deadline, max_samples=2000
            )
            c_bin_up = compute_c5(h_bin_up)
            if c_bin_up < best_c:
                best_h, best_c, best_n = h_bin_up.copy(), c_bin_up, n_up
        else:
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # fallback if nothing found
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
