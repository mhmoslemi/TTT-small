def binary_sa(h_bin, rng, deadline, max_iter=200000, decay=0.99995, early_stop=500000):
    n = h_bin.size
    target = n/2.0
    c = np.correlate(h_bin, h_bin, mode='full')
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    T0 = cur_worst * 0.05 + 1e-4
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # hi update
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i+1:]))
        diff_i = -1.0
        c[start_i:start_i+n] += diff_i * hi_vec
        h_bin[i] = 0.0

        # hj update
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j+1:]))
        diff_j = 1.0
        c[start_j:start_j+n] += diff_j * hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert changes
            h_bin[j] = 0.0
            c[start_j:start_j+n] -= diff_j * hj_vec
            h_bin[i] = 1.0
            c[start_i:start_i+n] -= diff_i * hi_vec
            no_improve += 1

        if no_improve > early_stop:
            break
        it += 1

    return best_h, best_worst

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa(h_bin, rng, deadline, max_iter=300000, decay=0.999995, early_stop=500000):
    n = h_bin.size
    target = n / 2.0
    c = np.correlate(h_bin, h_bin, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h_bin.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h_bin == 1))
    zeros = list(np.flatnonzero(h_bin == 0))

    T0 = cur_worst * 0.05 + 1e-4
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # i: 1 -> 0
        start_i = n - 1 - i
        hi_vec = np.concatenate((h_bin[i::-1], h_bin[i + 1 :]))
        diff_i = -1.0
        c[start_i : start_i + n] += diff_i * hi_vec
        h_bin[i] = 0.0

        # j: 0 -> 1 (on updated h_bin)
        start_j = n - 1 - j
        hj_vec = np.concatenate((h_bin[j::-1], h_bin[j + 1 :]))
        diff_j = 1.0
        c[start_j : start_j + n] += diff_j * hj_vec
        h_bin[j] = 1.0

        new_min = c.min()
        new_worst = target - new_min

        T = T0 * (decay ** it)
        accept = False
        if new_worst < cur_worst - 1e-12:
            accept = True
        else:
            if rng.random() < np.exp(-(new_worst - cur_worst) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_worst = new_worst
            cur_min = new_min
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_worst < best_worst - 1e-12:
                best_worst = cur_worst
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h_bin[j] = 0.0
            c[start_j : start_j + n] -= diff_j * hj_vec
            h_bin[i] = 1.0
            c[start_i : start_i + n] -= diff_i * hi_vec
            no_improve += 1

        if no_improve > early_stop:
            break
        it += 1

    return best_h, best_worst

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.95 * budget_s

    best_h = None
    best_c = float("inf")
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if "initial_h_values" in globals() else None
    base_n = parent.size if parent is not None else 360

    candidate_ns = [base_n]
    if base_n * 1.5 <= 720:
        candidate_ns.append(int(base_n * 1.5))
    if base_n * 2 <= 720:
        candidate_ns.append(base_n * 2)

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target_sum = n / 2.0

        starts = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            starts.append(interp)

        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        rand_frac = rng.random(n)
        rand_frac *= target_sum / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        starts.append(rand_frac)

        alt_bin = np.tile([1.0, 0.0], n // 2)
        starts.append(alt_bin)

        # parametric two‑step shapes
        for t in np.linspace(0.0, 0.5, 9)[1:-1]:
            a = (0.5 - t) / (1.0 - t) if t < 1.0 else 0.0
            xs = np.linspace(0.0, 2.0, n, endpoint=False)
            h_par = np.where((xs < a) | (xs > 2.0 - a), 1.0, t)
            adjust_sum_exact(h_par, rng)
            starts.append(h_par)

        for h0 in starts:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

            if np.all((h == 0) | (h == 1)):
                sub_dead = min(deadline, time.time() + 0.6 * (deadline - time.time()))
                h_bin, worst = binary_sa(h.copy(), rng, sub_dead, max_iter=300000)
                c5 = worst * (2.0 / n)
                if c5 < best_c:
                    best_h, best_c, best_n = h_bin.astype(float).copy(), c5, n
                h = h_bin.astype(float)
            else:
                h = gradient_refine(h, rng, step=0.02, max_iter=12)
                c5 = compute_c5(h)
                if c5 < best_c:
                    best_h, best_c, best_n = h.copy(), c5, n

            # SLSQP cascade
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
                if time.time() > deadline:
                    break
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=800)
                except Exception:
                    continue
                c5 = compute_c5(h)
                if c5 < best_c:
                    best_h, best_c, best_n = h.copy(), c5, n

            h = gradient_refine(h, rng, step=0.02, max_iter=12)
            c5 = compute_c5(h)
            if c5 < best_c:
                best_h, best_c, best_n = h.copy(), c5, n

        # optional up‑sampling
        if (
            best_h is not None
            and time.time() < deadline
            and best_n == n
            and n * 2 <= 720
        ):
            n_up = n * 2
            xs = np.linspace(0.0, 2.0, n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(xt, xs, best_h)
            adjust_sum_exact(h_up, rng)

            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0):
                if time.time() > deadline:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)
            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up, c_up, n_up

    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
