```python
def anneal_binary(h, rng, max_iter, temp_start, temp_end):
    best = h.copy()
    best_c = compute_c5_binary(best)
    cur = h.copy()
    cur_c = best_c
    for it in range(max_iter):
        # temperature schedule
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))  # exponential cooling
        # propose a swap: choose i where cur[i]==1, j where cur[j]==0
        i = rng.choice(np.where(cur == 1)[0])
        j = rng.choice(np.where(cur == 0)[0])
        # swap
        cur[i], cur[j] = 0, 1
        new_c = compute_c5_binary(cur)
        # acceptance probability
        if new_c < cur_c or rng.random() < math.exp((cur_c - new_c) / t):
            cur_c = new_c
            # update best if improved
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            # revert swap
            cur[i], cur[j] = 1, 0
        # maybe early break if best_c below threshold
    return best, best_c
```

```python
def compute_c5_fft(h):
    n = h.size
    dx = 2.0 / n
    # zero pad to length 2n
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    Corr = np.fft.irfft(H * np.conj(H))
    # Corr length 2n-1, first n entries correspond to shifts 0..n-1
    corr_hh = Corr[:n]
    min_corr = np.min(corr_hh)
    return 1 - dx * min_corr
```

def run(seed=42, budget_s=1100, **kwargs):

        best_c = float('inf')
        best_n = None

    # Phase 1: binary annealing on n_small=256 or 128 if time low
    n_small = 256
    time_phase1 = start + 0.35 * budget_s
    while time.time() < time_phase1:
        # set max_iters based on remaining time
        remaining = time_phase1 - time.time()
        # allocate maybe 0.1 sec per 1000 iter? Not needed.
        max_iters = 200000
        # we might adjust max_iters to ensure full run ends before time
        h_bin, c_bin = binary_anneal(n_small, rng, max_iters, time_phase1)
        if c_bin < best_c:
            best_h = h_bin.astype(float)
            best_c = c_bin
            best_n = n_small

    # Phase 2: Upsample to 512 if not already
    target_n = 512
    if best_h is not None and best_n != target_n:
        # Upsample by repeating pattern (if best_n divides target)
        if target_n % best_n == 0:
            repeats = target_n // best_n
            up_h = np.tile(best_h, repeats)
        else:
            # linear interpolation
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
            up_h = np.interp(x_new, x_old, best_h)
        up_h = np.clip(up_h, 0.0, 1.0)
        up_h = enforce_sum(up_h, rng)
        best_h = up_h
        best_n = target_n
        best_c = compute_c5(best_h)

    # Phase 3: Local refinement (guided hill climb, SLSQP, gradient refine)
    # We'll allocate remaining time with flexibility.
    time_end = start + budget_s * 0.95
    while time.time() < time_end:
        # use guided hill climb
        h_hc, c_hc = guided_hill_climb(best_h, rng, max_iters=2000, step_scale=0.6)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc
        # SLSQP with increasing beta
        for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
            if time.time() >= time_end:
                break
            maxit = 800 if beta >= 200 else 500
            best_h = slsqp_optimize(best_h, beta, max_iter=maxit)
            best_c = compute_c5(best_h)
        # gradient refine
        best_h = gradient_refine(best_h, rng, step=0.015, max_iter=12)
        best_c = compute_c5(best_h)

    # Final quick tweaks
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # final enforce
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time, math, numpy as np
from scipy.optimize import minimize

# -------------------- utilities --------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple redistribution loop – fast because diff shrinks quickly
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift that attains max C5."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using worst‑gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr * (2.0 / n)          # convert to C(k) values
        m = np.max(bc)
        # log‑sum‑exp safely
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


def upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid (periodic endpoints not wrapped)."""
    old_n = h.size
    if new_n % old_n == 0:
        # simple tiling when divisor matches
        return np.tile(h, new_n // old_n)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    start = time.time()

    # ---- initial solution -------------------------------------------------
    if "initial_h_values" in globals():

        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = best_h.size
    else:
        best_n = 256
        best_h = random_feasible_vector(best_n, rng)
        best_c = compute_c5(best_h)

    # ---- global stochastic search (medium resolution) --------------------
    time_global_end = start + 0.35 * budget_s
    n_pool = (128, 192, 256, 384)
    while time.time() < time_global_end:

        if c < best_c:

    # ---- upsample to full resolution --------------------------------------
    target_n = 512
    if best_n != target_n:
        h_up = upsample(best_h, target_n)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        best_h, best_c, best_n = h_up, compute_c5(h_up), target_n

    # ---- intensive local refinement (full resolution) --------------------
    time_refine_end = start + 0.90 * budget_s
    beta_schedule = (20., 80., 200., 800., 3000., 8000., 20000., 50000.)
    while time.time() < time_refine_end:
        # aggressive hill‑climb
        h, c = guided_hill_climb(best_h, rng,

        if c < best_c:
            best_h, best_c = h, c

        # sequence of SLSQP passes with increasing β
        for beta in beta_schedule:
            if time.time() >= time_refine_end:
                break

            best_h = slsqp_optimize(best_h, beta, max_iter=maxit)

    while time.time() - start < budget_s * 0.999:

        nc = compute_c5(nh)
        if nc < best_c:

    # ---- sanity enforcement ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
