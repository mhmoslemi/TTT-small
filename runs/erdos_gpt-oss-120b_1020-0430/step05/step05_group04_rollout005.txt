    factor = 2.0 / n
    nfft = 1 << ((3*n - 2).bit_length())
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.02
    t = 0
    beta1_pow = 1.0
    beta2_pow = 1.0
    for beta in beta_seq:
        # perhaps set max_iter for each beta based on availability: e.g., 2000
        max_iter = 2000
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            t += 1
            beta1_pow *= beta1
            beta2_pow *= beta2
            # compute correlation O
            H = np.fft.fft(h, nfft)
            B = np.fft.fft(1.0 - h, nfft)
            O_un = np.fft.ifft(H * np.conj(B)).real[:2*n-1]
            O = O_un * factor
            # weight
            maxO = O.max()
            w_raw = np.exp(beta * (O - maxO))
            w_sum = w_raw.sum()
            if w_sum == 0:
                w = np.full_like(w_raw, 1.0 / w_raw.size)
            else:
                w = w_raw / w_sum
            # convolution for gradient
            # conv w with (1 - h)
            W = np.fft.fft(w, nfft)
            G_add_full = np.fft.ifft(W * np.fft.fft(1.0 - h, nfft)).real
            # conv w_rev with h
            w_rev = w[::-1]
            W_rev = np.fft.fft(w_rev, nfft)
            G_sub_full = np.fft.ifft(W_rev * H).real
            start = n - 1
            end = start + n
            g_add = G_add_full[start:end]
            g_sub = G_sub_full[start:end]
            grad = factor * (g_add - g_sub)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1_pow)
            v_hat = v / (1 - beta2_pow)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            # Clip to [0,1]
            np.clip(h_new, 0.0, 1.0, out=h_new)
            # Adjust sum exactly
            adjust_sum_exact(h_new, rng)
            h = h_new
    return h

import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass so that sum(h) = n/2 exactly (in‑place)."""
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def adam_optimize_with_budget(h0, beta_seq, iters_per_beta, rng, deadline):

    n = h0.shape[0]

    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.02

    beta2_pow = 1.0

    for beta in beta_seq:
        for _ in range(iters_per_beta):

                return h
            beta1_pow *= beta1
            beta2_pow *= beta2

            O_un = np.fft.ifft(H * np.conj(B)).real[: 2 * n - 1]
