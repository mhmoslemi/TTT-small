def binary_local_search(h, deadline):
    n = h.size
    # compute correlation
    C = np.correlate(h, 1-h, mode='full')
    cur_max = C.max()
    ks = np.arange(-(n-1), n)
    zero_idx = n-1
    while time.time() < deadline:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1-h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        # compute delta
        delta = np.zeros_like(C)
        # i + k
        ip = i + ks
        mask = (ip >= 0) & (ip < n)
        delta[mask] += h[ip[mask]]
        # i - k
        im = i - ks
        mask = (im >= 0) & (im < n)
        delta[mask] += h[im[mask]]
        # j + k
        jp = j + ks
        mask = (jp >= 0) & (jp < n)
        delta[mask] -= h[jp[mask]]
        # j - k
        jm = j - ks
        mask = (jm >= 0) & (jm < n)
        delta[mask] -= h[jm[mask]]
        # -2 at zero
        delta[zero_idx] -= 2.0
        # cross terms
        diff = i - j
        idx1 = diff + zero_idx
        idx2 = -diff + zero_idx
        delta[idx1] += 1.0
        delta[idx2] += 1.0
        # evaluate new max
        new_max = (C + delta).max()
        # accept if improves
        if new_max < cur_max - 1e-12:
            # update
            h[i] = 0
            h[j] = 1
            C += delta
            cur_max = new_max
            # continue (maybe small improvement)
        # else possibly try next random swap.
    return h, C, cur_max

import time
import numpy as np
from scipy.optimize import minimize

# Global variables used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    """Return C5 = max overlap * (2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng, max_iter=2000):
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def fractional_to_binary(h, rng):
    """Return a binary vector with exactly n/2 ones (largest entries kept)."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    adjust_sum_exact(bin_h, rng)
    return bin_h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max via log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=800):
    """Optimize the smooth‑max surrogate with SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def binary_local_search(h, rng, deadline):
    """Hill‑climbing on binary vectors using the exact correlation delta."""
    n = h.size
    C = np.correlate(h, 1.0 - h, mode="full")
    cur_max = C.max()
    ks = np.arange(-(n - 1), n)          # length 2n‑1
    zero_idx = n - 1
    no_improve = 0
    max_no_improve = 50000

    while time.time() < deadline and no_improve < max_no_improve:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))

        delta = np.zeros_like(C)

        ip = i + ks
        mask = (ip >= 0) & (ip < n)
        delta[mask] += h[ip[mask]]

        im = i - ks
        mask = (im >= 0) & (im < n)
        delta[mask] += h[im[mask]]

        jp = j + ks
        mask = (jp >= 0) & (jp < n)
        delta[mask] -= h[jp[mask]]

        jm = j - ks
        mask = (jm >= 0) & (jm < n)
        delta[mask] -= h[jm[mask]]

        delta[zero_idx] -= 2.0

        diff = i - j
        idx1 = diff + zero_idx
        idx2 = -diff + zero_idx
        delta[idx1] += 1.0
        delta[idx2] += 1.0

        new_max = (C + delta).max()
        if new_max < cur_max - 1e-15:
            h[i] = 0.0
            h[j] = 1.0
            C += delta
            cur_max = new_max
            no_improve = 0
        else:
            no_improve += 1
