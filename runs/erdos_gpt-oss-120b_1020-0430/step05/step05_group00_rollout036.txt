def fft_convolve(x, y):

def fft_convolve(a, b):
    n = a.size + b.size - 1
    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(np.concatenate([a, np.zeros(size - a.size, dtype=a.dtype)]))
    B = np.fft.rfft(np.concatenate([b, np.zeros(size - b.size, dtype=b.dtype)]))
    C = A * B
    c = np.fft.irfft(C, n=size)[:n]
    return c

def adam_refine(h, rng, time_budget, start_time):
    N = h.shape[0]
    h = h.astype(float)
    target_sum = N / 2.0
    m = np.zeros(N, dtype=float)
    v = np.zeros(N, dtype=float)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    # initial lr
    lr0 = 0.5
    while True:
        now = time.time()
        if now - start_time >= time_budget:
            break
        t += 1
        # find worst shift
        S_corr = fft_convolve(h[::-1], h)
        min_idx = np.argmin(S_corr)
        k = min_idx - (N - 1)
        k_abs = abs(k)
        if k_abs == 0:
            D = 2 * h
        else:
            D = np.zeros(N, dtype=float)
            D[:N - k_abs] += h[k_abs:]
            D[k_abs:] += h[:N - k_abs]
        grad = 1.0 - D
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 * (0.5 ** (t // 2000))  # decay every 2000 steps
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        # clip
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce sum
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-12:
            h_new += diff / N
            h_new = np.clip(h_new, 0.0, 1.0)
        h = h_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            # early stop if below target
            if best_c <= 0.3808:
                break
    return best_h, best_c

import time
import numpy as np


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real 1‑D arrays via FFT."""
    n = a.size + b.size - 1
    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(np.concatenate([a, np.zeros(size - a.size, dtype=a.dtype)]))
    B = np.fft.rfft(np.concatenate([b, np.zeros(size - b.size, dtype=b.dtype)]))
    C = A * B
    c = np.fft.irfft(C, n=size)[:n]
    return c


def _c5_value(h: np.ndarray) -> float:
    """C₅ = maxₖ ∫ h(x)(1−h(x+k))dx for a step function represented by h."""
    N = h.shape[0]
    # Sₖ = Σ_i h_i·h_{i+k}  (linear convolution of h_rev and h)
    S = _fft_convolve(h[::-1], h)
    min_S = np.min(S)
    if min_S < 0:
        min_S = 0.0
    return 1.0 - (2.0 * min_S) / N


def _random_binary(N: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly N/2 ones."""
    h = np.zeros(N, dtype=int)
    idx = rng.choice(N, N // 2, replace=False)
    h[idx] = 1
    return h


def _greedy_local_opt(h_bin: np.ndarray, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy improvement: repeatedly swap the best 1/0 pair for the worst shift."""
    N = h_bin.shape[0]
    h = h_bin.copy()
    while True:
        S = _fft_convolve(h[::-1], h)
        min_idx = np.argmin(S)
        k = min_idx - (N - 1)
        k_abs = abs(k)

        if k_abs == 0:
            D = 2 * h
        else:
            D = np.empty(N, dtype=int)
            D.fill(0)
            D[: N - k_abs] += h[k_abs:]
            D[k_abs:] += h[: N - k_abs]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        i_one = ones[np.argmin(D[ones])]
        j_zero = zeros[np.argmax(D[zeros])]
        delta = D[j_zero] - D[i_one]      # increase in S_k if we swap

        if delta <= 0:
            break

        h[i_one] = 0
        h[j_zero] = 1

    return h, _c5_value(h)


def _adam_refine(
    h0: np.ndarray,
    rng: np.random.Generator,
    time_budget: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Adam‑style sub‑gradient descent on the worst shift."""
    N = h0.shape[0]
    h = h0.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N, dtype=float)
    v = np.zeros(N, dtype=float)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5
    best_h, best_c = h.copy(), _c5_value(h)
    t = 0

    while time.time() - start_time < time_budget:
        t += 1
        S = _fft_convolve(h[::-1], h)
        min_idx = np.argmin(S)
        k = min_idx - (N - 1)
        k_abs = abs(k)

        if k_abs == 0:
            D = 2.0 * h
        else:
            D = np.zeros(N, dtype=float)
            D[: N - k_abs] += h[k_abs:]
            D[k_abs:] += h[: N - k_abs]

        grad = 1.0 - D                      # sub‑gradient of the worst shift
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = lr0 * (0.5 ** (t // 2000))      # simple decay
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)

        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h += diff / N
            h = np.clip(h, 0.0, 1.0)

        c = _c5_value(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            if best_c <= 0.3808:
                break

    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs) -> tuple[list, float, int]:
    rng = np.random.default_rng(seed)
    start = time.time()
    budget = budget_s

    # ------------------------------------------------------------------
    # Determine problem size
    if 'initial_h_values' in globals():
        N = len(initial_h_values)
        N -= N % 2                     # ensure even
    else:
        N = 512

    best_h = None
    best_c = np.inf

    # ------------------------------------------------------------------
    # Evaluate parent construction if present
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        target = N / 2.0
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            h0 += diff / N
            h0 = np.clip(h0, 0.0, 1.0)
        c0 = _c5_value(h0)
        best_h, best_c = h0, c0
        if best_c <= 0.3808:
            return best_h.tolist(), float(best_c), int(N)

    # ------------------------------------------------------------------
    # Greedy local search with many random restarts (≈75 % of time)
    time_for_greedy = budget * 0.75
    while time.time() - start < time_for_greedy:
        h_start = _random_binary(N, rng)
        h_loc, c_loc = _greedy_local_opt(h_start, rng)
        if c_loc < best_c:
            best_c, best_h = c_loc, h_loc.astype(float)
            if best_c <= 0.3808:
                break

    # ------------------------------------------------------------------
    # Adam‑style refinement on the current best (≈60 % of remaining time)
    remaining = budget - (time.time() - start)
    if best_h is not None and remaining > 1.0:
        adam_time = remaining * 0.6
        adam_start = time.time()
        h_ref, c_ref = _adam_refine(best_h, rng, adam_time, adam_start)
        if c_ref < best_c:
            best_c, best_h = c_ref, h_ref

    # ------------------------------------------------------------------
    # Simple random‑swap hill climbing on the best solution (rest of time)
    remaining = budget - (time.time() - start)
    if best_h is not None and remaining > 0.5:
        h = best_h.copy()
        c = best_c
        N = h.shape[0]
        while time.time() - start < budget * 0.99:
            i, j = rng.integers(N, size=2)
            if i == j or h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]          # swap
            c_new = _c5_value(h)
            if c_new < c:
                c = c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
                    if best_c <= 0.3808:
                        break
            else:
                h[i], h[j] = h[j], h[i]      # revert

    # ------------------------------------------------------------------
    # Optional up‑sampling to 1024 points if time permits
    remaining = budget - (time.time() - start)
    if remaining > 2.0 and N == 512:
        N2 = 1024
        x_old = np.linspace(0.0, 2.0, N, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        target2 = N2 / 2.0
        diff = target2 - h_up.sum()
        if abs(diff) > 1e-12:
            h_up += diff / N2
            h_up = np.clip(h_up, 0.0, 1.0)

        up_start = time.time()
        up_time = remaining * 0.6
        h_up_ref, c_up = _adam_refine(h_up, rng, up_time, up_start)
        if c_up < best_c:
            best_c, best_h, N = c_up, h_up_ref, N2

        # final short random‑swap tweak after up‑sampling
        remaining = budget - (time.time() - start)
        if remaining > 0.5:
            h = best_h.copy()
            c = best_c
            while time.time() - start < budget * 0.999:
                i, j = rng.integers(N, size=2)
                if i == j or h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                c_new = _c5_value(h)
                if c_new < c:
                    c = c_new
                    if c_new < best_c:
                        best_c, best_h = c_new, h.copy()
                        if best_c <= 0.3808:
                            break
                else:
                    h[i], h[j] = h[j], h[i]

    # ------------------------------------------------------------------
    # Final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    target = N / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        best_h += diff / N
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c = _c5_value(best_h)
    return best_h.tolist(), float(final_c), int(N)
