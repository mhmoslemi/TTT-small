def subgradient_descent(h0, target, deadline, max_iter=2000):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    while time.time() < deadline and it < max_iter:

def project_onto_sum_bounds(v, target, tol=1e-12, max_iter=100):

    lo = -1.0
    hi = 2.0

        else:

import time, math

# ---------- utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_to_sum_and_bounds(v, target, rng=None):
    """Projection of v onto {0≤h≤1, sum(h)=target} via bisection on a Lagrange multiplier."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s < target:
            hi = lam
        else:
            lo = lam
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if rng is not None and abs(diff) > 1e-12:
        n = h.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def subgradient_descent(h0, target, deadline, rng, max_iter=2000):
    """Projected sub‑gradient descent on the worst‑shift overlap."""
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    n = h.size
    it = 0
    while time.time() < deadline and it < max_iter:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k > 0:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        elif k < 0:
            ka = -k
            g[ka:] += 1.0 - h[: n - ka]
            g[: n - ka] -= h[ka:]

        h = project_to_sum_and_bounds(v, target, rng)
        c = compute_c5(h)
        if c < best_c:

        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = float("inf")
    best_n = None

    # prefer the largest feasible size for finer search
    candidate_ns = [720, 540, 360]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0

        # build a handful of initial vectors
        init_vecs = []

        if "initial_h_values" in globals():
            parent = np.asarray(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            init_vecs.append(interp)                     # fractional
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            init_vecs.append(bin_vec)                    # binary

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        init_vecs.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= target / rand_frac.sum()
        init_vecs.append(rand_frac)

        # alternating binary pattern
        alt_bin = np.tile([1.0, 0.0], n // 2)
        init_vecs.append(alt_bin)

        for h0 in init_vecs:
            if time.time() > deadline:
                break
            h0 = project_to_sum_and_bounds(h0, target, rng)

            # sub‑gradient phase (takes ~40 % of remaining budget)
            sub_dead = min(deadline, time.time() + 0.4 * (deadline - time.time()))
            h_sub, c_sub = subgradient_descent(h0, target, sub_dead, rng, max_iter=5000)
            if c_sub < best_c:
                best_h, best_c, best_n = h_sub.copy(), c_sub, n

            # SLSQP polishing if time permits
            if time.time() < deadline:
                h_ref = best_h.copy()
                for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
                             2000.0, 5000.0, 10000.0, 20000.0, 50000.0):
                    if time.time() > deadline:
                        break
                    try:
                        h_ref = slsqp_optimize(h_ref, beta, rng, max_iter=800)
                    except Exception:
                        continue
                    c = compute_c5(h_ref)
                    if c < best_c:
                        best_h, best_c, best_n = h_ref.copy(), c, n
                # final gradient sweep
                if time.time() < deadline:
                    h_ref = gradient_refine(best_h, rng, step=0.02, max_iter=12)
                    c = compute_c5(h_ref)
                    if c < best_c:
                        best_h, best_c, best_n = h_ref.copy(), c, n

    # fallback – simple alternating pattern
    if best_h is None:

        best_h = np.tile([1.0, 0.0], best_n // 2).astype(float)
