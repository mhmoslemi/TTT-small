def local_search_binary(h, rng, deadline):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    curr_h = h.copy()
    curr_c = best_c
    # Temperature schedule
    T0 = (best_c - 0.35) if best_c > 0.35 else 0.01
    T = T0
    iter = 0
    while time.time() < deadline:
        # pick random 1 and 0
        ones = np.flatnonzero(curr_h)
        zeros = np.flatnonzero(1 - curr_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        new_h = curr_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)
        if new_c < curr_c or rng.random() < np.exp((curr_c - new_c) / max(T, 1e-12)):
            curr_h = new_h
            curr_c = new_c
            if new_c < best_c:
                best_h = new_h.copy()
                best_c = new_c
        # update temperature
        iter += 1
        if iter % 1000 == 0:
            # reduce temperature
            T *= 0.9
    return best_h, best_c
```

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        parent = np.array(initial_h_values, dtype=float)
        start_bin = binary_from_fractional(parent)
    else:
        n0 = 360
        start_bin = None

    best_h = None
    best_c = np.inf
    best_n = None

    # Prepare seeds
    seeds = []
    if start_bin is not None:
        seeds.append(start_bin.copy())
    # add random binary seeds
    num_random = 8
    for _ in range(num_random):
        hb = np.zeros(n0)
        hb[rng.choice(n0, n0 // 2, replace=False)] = 1.0
        seeds.append(hb)

    # iterate over seeds
    for seed_h in seeds:
        if time.time() >= deadline:
            break
        # local search
        hb, bc = local_search_binary(seed_h, rng, deadline)
        # greedy polish
        hb = greedy_improvement_binary(hb, rng, deadline)
        bc = compute_c5(hb)
        if bc < best_c:
            best_h, best_c, best_n = hb.copy(), bc, n0

    # Upsample to 720 if possible
    if best_h is not None and best_n * 2 <= 720 and time.time() < deadline:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up_f = np.interp(xt, xs, best_h)
        # binary conversion
        h_up = binary_from_fractional(h_up_f)
        # local search
        hb_up, bc_up = local_search_binary(h_up, rng, deadline)
        hb_up = greedy_improvement_binary(hb_up, rng, deadline)
        bc_up = compute_c5(hb_up)
        if bc_up < best_c:
            best_h, best_c, best_n = hb_up, bc_up, n_up

    # If still none (unlikely), fallback simple alternating

        best_n = n0
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        best_c = compute_c5(best_h)

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

def greedy_improvement_binary(h, rng, deadline):
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        # compute neighbor sums for each position: sum of h shifted by +shift and -shift (where defined)
        ar = np.arange(n)
        # for each i, the contribution to overlap for shift is h[i] * h[i+shift]
        # When swapping, we want to increase overlap i.e., increase h[i] and h[i+shift] together.
        # We'll compute for each i the sum of h at neighbors at distance shift.
        neigh = np.zeros(n)
        # positions i where i+shift < n
        mask1 = ar + shift < n
        neigh[mask1] += h[ar[mask1] + shift]
        # positions i where i-shift >=0
        mask2 = ar - shift >= 0
        neigh[mask2] += h[ar[mask2] - shift]
        # For a 1 we want small neighbor sum, for a 0 we want large neighbor sum
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(neigh[ones])]
        j = zeros[np.argmax(neigh[zeros])]
        # Evaluate delta
        # compute new overlap max after swap? Instead just swap if delta improves worst shift:
        # We'll approximate using neighbor sum: if neigh[i] < neigh[j], swap.
        if neigh[i] < neigh[j]:
            h[i] = 0.0
            h[j] = 1.0
            continue
        # else try random swaps
        improved = False
        for _ in range(5):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] < neigh[j2]:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

Implementation of fractional gradient descent:

Goal: minimize max_overlap = max_k integral of h*(1 - shift h). Equivalent to maximizing min_overlap = min_k sum_i h[i]*h[i+k].

We can approximate gradient of max_overlap: compute the shift k* that attains max_overlap (or compute shift with max "C5", which is worst shift). The gradient with respect to h is: for each position i, derivative = (1 - h[i+k*]) - (something?) Actually compute c5 = ∑ h[i](1 - h[i+k*]) = sum_i h[i] - sum_i h[i]h[i+k*] . derivative wrt h[i] = 1 - h[i+k*] - h[i-k*] (since the term h[i-k*] appears in sum_i h[i-k*]h[i]). Wait differentiate sum_i h[i]h[i+k*] wrt h[i] contributes h[i+k*] (term where i is left index) plus h[i-k*] (term where i is right index). So derivative of C5 = 1 - h[i+k*] - h[i-k*] . So gradient = 1 - h[i+k*] - h[i-k*].

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for a step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def binary_from_fractional(h):
    """Convert a vector to binary by taking the largest n/2 entries."""
    n = h.size
    idx = np.argpartition(h, -n // 2)[-n // 2 :]
    bh = np.zeros_like(h)
    bh[idx] = 1.0
    return bh

def adjust_sum_exact(h, rng, target):
    """Project a vector onto {0,1}^n with total sum = target (integer)."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def greedy_improve_binary(h, rng, deadline):
    """Deterministic swaps that improve the worst shift."""
    n = h.size
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:
            shift = -shift
        if shift == 0:
            break
        ar = np.arange(n)
        neigh = np.zeros(n)
        m1 = ar + shift < n
        neigh[m1] += h[ar[m1] + shift]
        m2 = ar - shift >= 0
        neigh[m2] += h[ar[m2] - shift]
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(neigh[ones])]
        j = zeros[np.argmax(neigh[zeros])]
        if neigh[i] < neigh[j]:
            h[i] = 0.0
            h[j] = 1.0
            continue
        improved = False
        for _ in range(5):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh[i2] < neigh[j2]:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

def local_search_binary(h, rng, deadline):
    """Simulated‑annealing / hill‑climbing on binary vectors."""
    n = h.size
    target = n // 2
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    # temperature schedule
    T = max(best_c - 0.35, 0.005)
    it = 0
    while time.time() < deadline:
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1 - cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)
        improve = new_c < cur_c
        if not improve:
            prob = np.exp((cur_c - new_c) / max(T, 1e-12))
            improve = rng.random() < prob
        if improve:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        it += 1
        if it % 1000 == 0:
            T *= 0.9
    return best_h, best_c

def gradient_descent_frac(h, rng, deadline, max_iter=200):
    """Projected gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        grad = np.ones_like(h)
        if k >= 0:
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        else:
            k = -k
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        step = 0.02 * (0.95 ** it)
        h -= step * grad
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple projection onto the hyperplane Σh = target
            for _ in range(10):
                i = rng.integers(0, n)
                add = min(diff, 1 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # base size
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        parent = np.array(initial_h_values, dtype=float)
        start_bin = binary_from_fractional(parent)
    else:
        n0 = 360
        start_bin = None

    best_h = None
    best_c = np.inf
    best_n = None

    # collect initial binary seeds
    seeds = []
    if start_bin is not None:
        seeds.append(start_bin.copy())
    for _ in range(8):
        hb = np.zeros(n0)
        hb[rng.choice(n0, n0 // 2, replace=False)] = 1.0
        seeds.append(hb)

    # explore each seed
    for sb in seeds:
        if time.time() >= deadline:
            break
        hb, bc = local_search_binary(sb, rng, deadline)
        hb = greedy_improve_binary(hb, rng, deadline)
        bc = compute_c5(hb)
        if bc < best_c:
            best_h, best_c, best_n = hb.copy(), bc, n0

    # optional gradient refinement on the best binary (fractional improvement)
    if best_h is not None and time.time() < deadline:
        hb_frac = gradient_descent_frac(best_h.astype(float), rng, deadline)
        bc_frac = compute_c5(hb_frac)
        if bc_frac < best_c:
            best_h, best_c, best_n = hb_frac, bc_frac, best_n

    # up‑sample to 720 and repeat if time permits
    if best_h is not None and best_n * 2 <= 720 and time.time() < deadline:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_interp = np.interp(xt, xs, best_h)
        h_up = binary_from_fractional(h_interp)
        hb, bc = local_search_binary(h_up, rng, deadline)
        hb = greedy_improve_binary(hb, rng, deadline)
        bc = compute_c5(hb)
        if bc < best_c:
            best_h, best_c, best_n = hb.copy(), bc, n_up
        # final fractional gradient step on up‑sampled solution
        if time.time() < deadline:
            hb_frac = gradient_descent_frac(best_h.astype(float), rng, deadline)
            bc_frac = compute_c5(hb_frac)
            if bc_frac < best_c:
                best_h, best_c, best_n = hb_frac, bc_frac, best_n

    # fallback simple construction
    if best_h is None:
        best_n = n0
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
