def binary_anneal(h, rng, deadline):
    n = h.size
    s = np.arange(1, n)
    I = autocorr_binary(h)
    O_counts_current = (n // 2) - I.min()
    O_current = (2.0 / n) * O_counts_current
    best_h = h.copy()
    best_O_counts = O_counts_current
    O_initial = O_current
    start = time.time()
    # loops
    while time.time() < deadline:
        # pick i (one) and j (zero)
        i = rng.integers(0, n)
        if h[i] == 0:
            continue
        j = rng.integers(0, n)
        if h[j] == 1:
            continue
        # compute dI
        ip = (i + s) % n
        im = (i - s) % n
        jp = (j + s) % n
        jm = (j - s) % n
        dI = - h[ip] - h[im] + h[jp] + h[jm]
        I_new_min = I[0]
        # compute I[1:] + dI min
        # Using np.minimum reduce? We'll compute directly.
        min_rest = (I[1:] + dI).min()
        if min_rest < I_new_min:
            I_new_min = min_rest
        O_counts_new = (n // 2) - I_new_min
        O_new = (2.0 / n) * O_counts_new
        delta = O_new - O_current
        # temperature
        elapsed = time.time() - start
        frac = elapsed / (deadline - start) if deadline > start else 1.0
        T = max(1e-6, (O_initial - O_current) * (1.0 - frac) + 1e-6)
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            h[i] = 0
            h[j] = 1
            I[1:] += dI
            O_current = O_new
            O_counts_current = O_counts_new
            if O_counts_current < best_O_counts:
                best_O_counts = O_counts_current
                best_h = h.copy()
                # optionally break if perfect or target reached
                if (2.0 / n) * best_O_counts <= 0.3808:
                    # early exit: we have sufficient bound
                    break
        # else reject: continue
    return best_h

    start = time.time()

            if parent_bin.sum() != n//2:

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h):
    """Maximum overlap integral for a step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng, max_iter=2000):
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def autocorr_binary(h):
    """Cyclic autocorrelation I(s)=∑ h_i h_{i+s} for binary h."""
    n = h.size
    fft_h = np.fft.fft(h)
    corr = np.fft.ifft(fft_h * np.conj(fft_h)).real
    return np.rint(corr).astype(np.int64)


# ---------- SLSQP surrogate optimisation ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k ∫ h(1‑h_k)."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=1200):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


# ---------- binary simulated‑annealing ----------
def binary_anneal(h, rng, deadline):
    """Monotone‑inspired simulated annealing on a binary vector."""
    n = h.size
    s = np.arange(1, n)
    I = autocorr_binary(h)
    I_min = I.min()
    O_counts = (n // 2) - I_min
    O_curr = (2.0 / n) * O_counts
    best_h = h.copy()
    best_counts = O_counts
    I0 = I[0]  # always n/2
    start = time.time()
    O_initial = O_curr

    while time.time() < deadline:
        # pick a random 1‑0 pair
        i = rng.integers(0, n)
        if h[i] == 0:
            continue
        j = rng.integers(0, n)
        if h[j] == 1:
            continue

        # delta autocorrelation for all non‑zero shifts
        ip = (i + s) % n
        im = (i - s) % n
        jp = (j + s) % n
        jm = (j - s) % n
        dI = - h[ip] - h[im] + h[jp] + h[jm]

        # prospective minimum after swap
        I_new_min = I0
        rest_min = (I[1:] + dI).min()
        if rest_min < I_new_min:
            I_new_min = rest_min
        O_counts_new = (n // 2) - I_new_min
        O_new = (2.0 / n) * O_counts_new
        delta = O_new - O_curr

        # simple temperature schedule
        elapsed = time.time() - start
        frac = elapsed / (deadline - start) if deadline > start else 1.0
        T = max(1e-6, (O_initial - O_curr) * (1.0 - frac) + 1e-6)

        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            h[i] = 0
            h[j] = 1
            I[1:] += dI
            O_curr = O_new
            I_min = I_new_min
            if O_counts_new < best_counts:
                best_counts = O_counts_new
                best_h = h.copy()
                if (2.0 / n) * best_counts <= 0.3808:
                    break
    return best_h


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # base resolution
    n = 360
    best_h = None
    best_c = np.inf
    best_n = n

    # use parent if available
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=np.int8)
        if parent.size == n:
            # force binary and correct sum
            parent_bin = (parent > 0.5).astype(np.int8)
            diff = int(parent_bin.sum() - n // 2)
            if diff > 0:
                ones = np.flatnonzero(parent_bin)
                turn = rng.choice(ones, diff, replace=False)
                parent_bin[turn] = 0
            elif diff < 0:
                zeros = np.flatnonzero(1 - parent_bin)
                turn = rng.choice(zeros, -diff, replace=False)
                parent_bin[turn] = 1
            # anneal starting from parent
            h_local = binary_anneal(parent_bin.copy(), rng, deadline)
            c_local = compute_c5(h_local.astype(float))
            if c_local < best_c:
                best_h, best_c, best_n = h_local.astype(float), c_local, n
                if best_c <= 0.3808:
                    return best_h.tolist(), float(best_c), int(best_n)

    # repeated random restarts
    while time.time() < deadline:
        h = np.zeros(n, dtype=np.int8)
        ones_idx = rng.choice(n, n // 2, replace=False)
        h[ones_idx] = 1
        h_local = binary_anneal(h, rng, deadline)
        c_local = compute_c5(h_local.astype(float))
        if c_local < best_c:
            best_h, best_c, best_n = h_local.astype(float), c_local, n
            if best_c <= 0.3808:
                break
        # stop early if little time left
        if time.time() > deadline - 5:
            break

    # optional up‑sampling + smooth refinement if time permits
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 720
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        for beta in (20.0, 200.0, 2000.0):
            if time.time() > deadline:
                break
            try:
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=1200)
            except Exception:
                pass
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # fallback if nothing was found
    if best_h is None:
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    # final sanitisation
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
