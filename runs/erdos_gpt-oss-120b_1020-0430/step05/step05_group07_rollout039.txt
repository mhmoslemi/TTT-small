def convolve_fft(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n-1).bit_length()
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    fc = fa * fb
    conv = np.fft.irfft(fc, n=size)[:n]
    return conv

def proj_sum_box(v, target, atol=1e-12, max_iter=50):

    low = -1.0

    high = 1.0

    for _ in range(max_iter):

        lam = (low + high) / 2

        x = np.clip(v + lam, 0.0, 1.0)

        s = x.sum()

        if abs(s - target) <= atol:

            return x

        if s > target:

            high = lam

        else:

            low = lam

    return np.clip(v + (low + high) / 2, 0.0, 1.0)

import time
import numpy as np

# ---------- helper utilities ----------
def convolve_fft(a, b):
    """Linear convolution using FFT (real input)."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    fc = fa * fb
    conv = np.fft.irfft(fc, n=size)[:n]
    return conv

def proj_sum_box(v, target, tol=1e-12, max_iter=60):
    """Project v onto {0<=x<=1, sum(x)=target} by shifting and clipping."""
    low, high = -1.0, 1.0
    # expand bounds if necessary
    while np.clip(v + low, 0.0, 1.0).sum() > target:
        low *= 2.0
    while np.clip(v + high, 0.0, 1.0).sum() < target:
        high *= 2.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            high = mid
        else:
            low = mid
    x = np.clip(v + (low + high) * 0.5, 0.0, 1.0)
    # final tiny fix
    diff = target - x.sum()
    if abs(diff) > tol:
        free = (x > tol) & (x < 1.0 - tol)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h):
    """Exact C5 value for vector h (full correlation)."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / N))

def gradient_refine(h, step=0.03, max_iter=12, rng=None):
    """Subgradient descent targeting the worst shift."""
    N = h.size
    target = N / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (N - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = N - k
            g[:up] += 1.0 - h[k:]          # (1 - h[i+k])
            g[k:]  -= h[:up]               # -h[i-k]
        cur_step = step * (0.5 ** it)
        h = h - cur_step * g
        h = np.clip(h, 0.0, 1.0)
        h = proj_sum_box(h, target)
    return h

def smooth_grad_optimize(h, betas, steps_per_beta, step0, lam_var, rng, deadline):
    """Smooth‑max gradient descent with continuation on beta."""
    N = h.size
    target = N / 2.0
    for beta in betas:
        if time.time() > deadline:
            break
        for it in range(steps_per_beta):
            if time.time() > deadline:
                break
            # correlation (full) for smooth‑max weights
            corr = np.correlate(h, 1.0 - h, mode="full")
            maxc = corr.max()
            w = np.exp(beta * (corr - maxc))
            w /= w.sum()
            # convolutions needed for gradient
            conv_wh   = convolve_fft(w, h)
            conv_wrev = convolve_fft(w[::-1], h)
            off = N - 1
            f = conv_wrev[off:off + N]
            g = conv_wh[off:off + N]
            grad = 1.0 - (f + g)                     # ∇ smooth‑max
            # add variance‑enhancing term
            grad += -lam_var * (h - h.mean())
            cur_step = step0 / np.sqrt(it + 1)
            h_new = h - cur_step * grad
            h = proj_sum_box(h_new, target)
    return h
