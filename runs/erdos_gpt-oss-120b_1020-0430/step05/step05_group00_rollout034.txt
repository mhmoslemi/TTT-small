```
def project_to_sum(h, target):
    # ensure 0<=h<=1 and sum = target
    # first clip
    h = np.clip(h, 0, 1)
    # compute diff
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # sort indices by how far they are from bound in the direction needed.
    if diff > 0:
        # need to increase sum: sort by current value ascending (more room)
        inc = np.sort(np.argsort(h))
        for i in inc:
            room = 1.0 - h[i]
            add = min(room, diff)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # need to decrease sum: sort by current value descending (more room)
        dec = np.argsort(-h)
        for i in dec:
            room = h[i]
            sub = min(room, -diff)
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h
```

def gradient_softmax_descent(h, beta, rng, max_iters=500):
    N = h.size
    target = N/2.0
    for it in range(max_iters):
        O_full = np.correlate(h, 1 - h, mode='full')
        # compute softmax weights
        # use logsumexp for numeric stability
        max_O = np.max(beta * O_full)
        # log-sum-exp not needed for weights; we can compute w = np.exp(beta*O_full - max) then normalize
        w = np.exp(beta * O_full - max_O)
        w_sum = w.sum()
        w /= w_sum
        # compute gradient g_i = sum_{t=i}^{i+N-1} w[t] * (1 - h[t-i])
        # compute through matrix multiplication using stride
        # Simple loop:
        g = np.empty(N, dtype=float)
        one_minus_h = 1.0 - h
        # Pre compute convolution via np.dot sliding
        # Use list comprehension
        for i in range(N):
            g[i] = np.dot(w[i:i+N], one_minus_h)
        # step size schedule
        alpha = 0.5 / (it + 1)  # decreasing
        h = h - alpha * g
        # clip and enforce sum
        h = np.clip(h, 0.0, 1.0)
        # enforce exact sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust proportionally to (1 - h) or h
            # simple random redistribution
            # use same enforce_sum function (which returns h)
            h = enforce_sum(h, rng)
        # optionally break if gradient norm small
        if np.linalg.norm(g) < 1e-6:
            break
    return h

import time
import numpy as np
import scipy.optimize as opt
import scipy.special as sp

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def gradient_softmax_descent(h: np.ndarray, beta: float,
                             rng: np.random.Generator,
                             max_iters: int = 300) -> np.ndarray:
    """Gradient descent on the soft‑max approximation."""
    n = h.size
    target = n / 2.0
    one_minus_h = 1.0 - h
    for it in range(max_iters):
        O = np.correlate(h, 1.0 - h, mode="full")
        # soft‑max weights
        maxb = np.max(beta * O)
        w = np.exp(beta * O - maxb)
        w /= w.sum()
        # gradient: g_i = Σ_{t=i}^{i+n-1} w_t * (1 - h_{t-i})
        g = np.empty(n, dtype=float)
        for i in range(n):
            g[i] = np.dot(w[i:i + n], one_minus_h)
        # step size
        step = 0.5 / (it + 1)
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        # enforce exact sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution
            h = enforce_sum(h, rng)
        one_minus_h = 1.0 - h
        if np.linalg.norm(g) < 1e-7:
            break
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    target_time = budget_s

    # ----- prepare initial h (down‑sample to 256 for speed) -----
    N_target = 256
    if "initial_h_values" in globals():
        h_full = np.array(initial_h_values, dtype=float)
        if h_full.size != N_target:
            factor = h_full.size // N_target
            h0 = h_full[: factor * N_target].reshape(N_target, factor).mean(axis=1)
        else:
            h0 = h_full.copy()
    else:
        h0 = np.full(N_target, 0.5, dtype=float)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- main search (random restarts + soft‑max gradient descent) -----
    beta_schedule = [5., 20., 80., 200., 800., 3000., 8000., 15000.]
    while time.time() - start < target_time * 0.85:
        # perturb current best to explore new basin
        h = best_h + rng.normal(0.0, 0.2, size=best_h.size)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)

        for beta in beta_schedule:
            if time.time() - start >= target_time * 0.95:
                break
            # a few gradient steps at this beta
            h = gradient_softmax_descent(h, beta, rng, max_iters=300)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
                if best_c <= 0.3808:
                    break
        # occasional SLSQP refinement after a full beta sweep
        if time.time() - start < target_time * 0.97:
            h_ref = slsqp_optimize(best_h, 2000., rng, max_iter=800)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # ----- final local polishing -----
    if time.time() - start < target_time * 0.99:

                                         max_iters=2000, step_scale=0.7)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
