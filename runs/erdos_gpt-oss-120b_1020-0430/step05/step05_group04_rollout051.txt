def greedy_improve(h, rng, max_iter=50, deadline=None):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        k = max_idx - (n - 1)
        # positions that cause overlap for this shift:
        if k >= 0:
            i_range = np.arange(n - k)
            mismatch_1 = i_range[(h == 1) & ((h[i_range + k]) == 0)]
            mismatch_0 = i_range[(h == 0) & ((h[i_range + k]) == 1)]
        else:
            shift = -k
            i_range = np.arange(shift, n)
            mismatch_1 = i_range[(h == 1) & ((h[i_range - shift]) == 0)]
            mismatch_0 = i_range[(h == 0) & ((h[i_range - shift]) == 1)]
        if mismatch_1.size == 0 or mismatch_0.size == 0:
            break
        # try sample pairs
        total_pairs = int(mismatch_1.size) * int(mismatch_0.size)
        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in mismatch_1 for j in mismatch_0]
        else:
            pairs = [(int(rng.choice(mismatch_1)), int(rng.choice(mismatch_0))) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            # swap
            h[i] = 0
            h[j] = 1
            c_new = compute_c5(h)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
            # revert
            h[i] = 1
            h[j] = 0
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Redistribute mass so that Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tiny correction if needed
    if abs(diff) > 1e-12:
        h[0] += diff
    return h


def smooth_max_opt(h, beta, rng, steps=50, step_size=0.1):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    n = h.size
    target = n / 2.0
    for _ in range(steps):
        overlaps = np.correlate(h, 1.0 - h, mode="full")
        max_o = np.max(overlaps)
        w = np.exp(beta * (overlaps - max_o))
        w /= np.sum(w)
        grad = np.convolve(w, 1.0 - h, mode="full")[n - 1 : n - 1 + n]
        h = h - step_size * grad
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def binary_sa_opt(h_bin, rng, time_deadline):
    """Simulated‑annealing on binary vectors, focusing on the worst shift."""
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    init_T = max(0.1 * cur_c, 1e-4)
    it = 0
    while time.time() < time_deadline and it < 5000:
        # worst shift
        corr = np.correlate(cur, 1 - cur, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        # mismatched positions for this shift
        if k >= 0:
            i_range = np.arange(n - k)
            mismatch_1 = i_range[(cur[i_range] == 1) & (cur[i_range + k] == 0)]
            mismatch_0 = i_range[(cur[i_range] == 0) & (cur[i_range + k] == 1)]
        else:
            s = -k
            i_range = np.arange(s, n)
            mismatch_1 = i_range[(cur[i_range] == 1) & (cur[i_range - s] == 0)]
            mismatch_0 = i_range[(cur[i_range] == 0) & (cur[i_range - s] == 1)]
        if mismatch_1.size == 0 or mismatch_0.size == 0:
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = int(rng.choice(mismatch_1))
            j = int(rng.choice(mismatch_0))
        cur[i] = 0
        cur[j] = 1
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c, best = c, cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
    return best, best_c


def greedy_improve(h, rng, max_iter=30, deadline=None):
    """Local swaps that lower the current worst overlap."""
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        if k >= 0:
            i_range = np.arange(n - k)
            mis_1 = i_range[(h == 1) & (h[i_range + k] == 0)]
            mis_0 = i_range[(h == 0) & (h[i_range + k] == 1)]
        else:
            s = -k
            i_range = np.arange(s, n)
            mis_1 = i_range[(h == 1) & (h[i_range - s] == 0)]
            mis_0 = i_range[(h == 0) & (h[i_range - s] == 1)]
        if mis_1.size == 0 or mis_0.size == 0:
            break
        total = int(mis_1.size) * int(mis_0.size)
        samples = 500
        if total <= samples:
            pairs = [(int(i), int(j)) for i in mis_1 for j in mis_0]
        else:
            pairs = [(int(rng.choice(mis_1)), int(rng.choice(mis_0))) for _ in range(samples)]
        cur_c = compute_c5(h)
        best_pair = None
        best_c = cur_c
        for i, j in pairs:
            h[i] = 0
            h[j] = 1
            c = compute_c5(h)
            if c < best_c - 1e-12:
                best_c, best_pair = c, (i, j)
            h[i] = 1
            h[j] = 0
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h
