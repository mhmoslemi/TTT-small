def guided_hill_climb(h, rng, max_iters=1000, step_scale=0.8):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * scale
        h[i] -= eps
        h[j] += eps
        # ensure within bounds due to eps selection
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if (it+1) % decay_interval == 0:
            scale = max(scale * 0.9, 1e-4)
    return best_h, best_c
```

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global vars for SLSQP soft‑max optimisation
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k (corrected)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.empty_like(h, dtype=float)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[:] = 0.0
        # part from h_i (1 - h_{i+k})
        grad[: n - k] = 1.0 - h[k:]
        # part from -h_{i-k}
        grad[k:] -= h[: n - k]
    return grad


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7,
                      time_limit: float = None, start_time: float = None):
    """Hill‑climb using the worst‑shift gradient."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        if time_limit is not None and (time.time() - start_time) >= time_limit:
            break
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * scale
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if (it + 1) % decay == 0:
            scale = max(scale * 0.9, 1e-4)
    return best_h, best_c

    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    best_h = None

    best_n = None

    # ---- initial solution from parent (if any) ----

    # hill‑climb on up‑sampled version

        )

        h_opt, c_opt = h, compute_c5(h)

    for beta in (20.0, 200.0, 2000.0, 8000.0):
        if time.time() - start > total_budget * 0.98:
            break
        h_opt = slsqp_optimize(h_opt, beta, rng, max_iter=600)
        c_opt = compute_c5(h_opt)

    # ---- final stochastic tweaking until time runs out ----
    while time.time() - start < total_budget * 0.999:
        grad = worst_gradient(h_opt)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        max_eps = min(h_opt[i], 1.0 - h_opt[j])
        if max_eps <= 0.0:
            break
        eps = rng.random() * max_eps * 0.02
        h_opt[i] -= eps
        h_opt[j] += eps
        c_new = compute_c5(h_opt)
        if c_new < c_opt:
            c_opt = c_new
        else:
            # revert if not improvement
            h_opt[i] += eps
            h_opt[j] -= eps

    # ---- final sanitisation ----
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = enforce_sum(h_opt, rng)
    final_c = compute_c5(h_opt)

    return h_opt.tolist(), float(final_c), int(h_opt.size)
