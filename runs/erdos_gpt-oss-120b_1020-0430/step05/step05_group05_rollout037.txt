import time, numpy as np
from scipy.optimize import minimize

def compute_c5_full(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def compute_c5_half(a):
    h = np.concatenate([a, 1.0 - a])
    return compute_c5_full(h)

def smooth_max_obj_half(a, beta):
    h = np.concatenate([a, 1.0 - a])
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def lbfgs_optimize(a0, beta, max_iter):
    bounds = [(0.0, 1.0)] * a0.size
    res = minimize(lambda x: smooth_max_obj_half(x, beta), a0, method='L-BFGS-B',
                    bounds=bounds, options={'maxiter': max_iter, 'ftol':1e-12, 'disp': False})
    return np.clip(res.x, 0.0, 1.0)

def gradient_refine_half(a, rng, step=0.05, max_iter=15):
    M = a.size
    n = 2 * M
    for it in range(max_iter):
        h = np.concatenate([a, 1.0 - a])
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            break
        ar = np.arange(M)
        neigh_i = h[(ar + k) % n] + h[(ar - k) % n]
        neigh_iM = h[(ar + M + k) % n] + h[(ar + M - k) % n]
        diff = neigh_i - neigh_iM
        g = -diff
        step_sz = step * (0.5 ** it)
        a = a - step_sz * g
        a = np.clip(a, 0.0, 1.0)
    return a

def binary_greedy_half(a, rng, deadline):

    n = 2 * M
    while time.time() < deadline:

        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)

        diff = neigh_i - neigh_iM
        ones = np.flatnonzero(a == 1)
        zeros = np.flatnonzero(a == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(diff[ones])]

            a[i], a[j] = 0, 1

        improved = False

            i2 = rng.choice(ones)

            if diff[i2] < diff[j2]:
                a[i2], a[j2] = 0, 1
                improved = True
                break
        if not improved:
            break
    return a

def upsample_half(a):
    M = a.size
    M_up = M * 2
    # Interpolate a defined on M points to M_up points using linear interpolation over [0,1)
    x_old = np.linspace(0.0, 1.0, M, endpoint=False)

import time
import numpy as np
from scipy.optimize import minimize

def _c5_full(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _c5_half(a):
    h = np.concatenate([a, 1.0 - a])
    return _c5_full(h)

def _smooth_obj_half(a, beta):
    h = np.concatenate([a, 1.0 - a])
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def _lbfgs_opt(a0, beta, max_iter):
    bounds = [(0.0, 1.0)] * a0.size
    res = minimize(lambda x: _smooth_obj_half(x, beta), a0,
                   method="L-BFGS-B", bounds=bounds,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    return np.clip(res.x, 0.0, 1.0)

def _grad_refine_half(a, rng, step=0.06, max_iter=20):
    M = a.size
    n = 2 * M
    for it in range(max_iter):
        h = np.concatenate([a, 1.0 - a])
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            break
        ar = np.arange(M)
        neigh_i = h[(ar + k) % n] + h[(ar - k) % n]
        neigh_iM = h[(ar + M + k) % n] + h[(ar + M - k) % n]
        diff = neigh_i - neigh_iM
        g = -diff
        a = a - step * (0.5 ** it) * g
        a = np.clip(a, 0.0, 1.0)
    return a

def _binary_greedy_half(a, rng, deadline):
    M = a.size
    n = 2 * M
    while time.time() < deadline:
        h = np.concatenate([a, 1.0 - a])
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        shift = abs(idx)
        if shift == 0:
            break
        ar = np.arange(M)
        neigh_i = h[(ar + shift) % n] + h[(ar - shift) % n]
        neigh_iM = h[(ar + M + shift) % n] + h[(ar + M - shift) % n]
        diff = neigh_i - neigh_iM
        ones = np.flatnonzero(a == 1)
        zeros = np.flatnonzero(a == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[np.argmin(diff[ones])]
        j = zeros[np.argmax(diff[zeros])]
        if diff[i] < diff[j]:
            a[i], a[j] = 0.0, 1.0
            continue
        improved = False
        for _ in range(8):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if diff[i2] < diff[j2]:
                a[i2], a[j2] = 0.0, 1.0
                improved = True
                break
        if not improved:
            break
    return a

def _upsample_half(a):
    M = a.size
    M2 = M * 2
    x_old = np.linspace(0.0, 1.0, M, endpoint=False)
    x_new = np.linspace(0.0, 1.0, M2, endpoint=False)
    a2 = np.interp(x_new, x_old, a)
    return np.clip(a2, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98

    n = 360
    M = n // 2
    starts = []

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size == n:
            half_sym = (parent[:M] + (1.0 - parent[M:])) * 0.5
            starts.append(np.clip(half_sym, 0.0, 1.0))
            bin_half = np.zeros(M, dtype=float)
            idx = np.argpartition(half_sym, -M // 2)[-M // 2 :]
            bin_half[idx] = 1.0
            starts.append(bin_half)

    for _ in range(4):
        a = np.zeros(M, dtype=float)
        a[rng.choice(M, M // 2, replace=False)] = 1.0
        starts.append(a)

    for _ in range(4):
        a = rng.random(M)
        starts.append(np.clip(a, 0.0, 1.0))

    x = np.linspace(0.0, 1.0, M, endpoint=False)
    a = 0.5 + 0.5 * np.sin(2 * np.pi * x)
    starts.append(np.clip(a, 0.0, 1.0))

    betas = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 2000.0, 5000.0, 20000.0]
    best_c = np.inf
    best_a = None
    best_n = n

    for a0 in starts:
        if time.time() > deadline:
            break
        a = a0.copy()
        for beta in betas:
            if time.time() > deadline:
                break
            max_iter = 300 if beta < 200 else 600
            a = _lbfgs_opt(a, beta, max_iter)
            c = _c5_half(a)
            if c < best_c:
                best_c, best_a = c, a.copy()
        a = _grad_refine_half(a, rng, step=0.06, max_iter=20)
        c = _c5_half(a)
        if c < best_c:
            best_c, best_a = c, a.copy()
        # binary phase
        bin_a = np.zeros(M, dtype=float)
        idx = np.argpartition(a, -M // 2)[-M // 2 :]
        bin_a[idx] = 1.0
        bin_a = _binary_greedy_half(bin_a, rng, deadline)
        c_bin = _c5_half(bin_a)
        if c_bin < best_c:
            best_c, best_a = c_bin, bin_a.copy()

    # optional up‑sampling
    if best_a is not None and time.time() < deadline and best_n * 2 <= 720:
        a_up = _upsample_half(best_a)
        for beta in betas:
            if time.time() > deadline:

            max_iter = 500 if beta < 200 else 800
            a_up = _lbfgs_opt(a_up, beta, max_iter)
        a_up = _grad_refine_half(a_up, rng, step=0.05, max_iter=30)
        bin_up = np.zeros(a_up.size, dtype=float)
        idx = np.argpartition(a_up, -a_up.size // 2)[-a_up.size // 2 :]
        bin_up[idx] = 1.0
        bin_up = _binary_greedy_half(bin_up, rng, deadline)
        c_up = _c5_half(bin_up)
        if c_up < best_c:
            best_c, best_a, best_n = c_up, bin_up.copy(), best_n * 2

    if best_a is None:
        best_a = np.zeros(M, dtype=float)
        best_a[: M // 2] = 1.0

    h_full = np.concatenate([best_a, 1.0 - best_a])
    h_full = np.clip(h_full, 0.0, 1.0)
    final_c = _c5_full(h_full)
    return h_full.tolist(), float(final_c), int(best_n)
