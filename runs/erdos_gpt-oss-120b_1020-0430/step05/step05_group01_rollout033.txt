def compute_c5_fast(h):
    n = h.size
    nfft = 1 << (2*n-1).bit_length()
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1-h, n=nfft)
    corr = np.fft.irfft(H * G, n=nfft)[:2*n-1]
    return corr.max() * (2.0 / n)

import time, numpy as np
from scipy.optimize import minimize

# global placeholders for softmax SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def correlation_full(h, nfft=None):
    n = h.size
    if nfft is None:
        nfft = 1 << (2*n-1).bit_length()
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft((1 - h)[::-1], n=nfft)
    corr = np.fft.irfft(H * G, n=nfft)[:2*n-1]
    return corr

def compute_c5_fast(h, nfft=None):
    n = h.size
    corr = correlation_full(h, nfft)
    return corr.max() * (2.0 / n)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple correction loop
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h, corr=None):
    n = h.size
    if corr is None:
        corr = correlation_full(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += 1.0 - h[:-k]
        grad[:n - k] -= h[k:]
    else:
        kk = -k
        grad[:n - kk] += 1.0 - h[kk:]
        grad[kk:] -= h[:n - kk]
    return grad, k, corr

def compute_diff_full(h, i, j):
    n = h.size
    k = np.arange(n)
    i_plus = i + k
    i_minus = i - k
    j_plus = j + k
    j_minus = j - k
    h_i_plus = np.where((i_plus >= 0) & (i_plus < n), h[i_plus], 0.0)
    h_i_minus = np.where((i_minus >= 0) & (i_minus < n), h[i_minus], 0.0)
    h_j_plus = np.where((j_plus >= 0) & (j_plus < n), h[j_plus], 0.0)
    h_j_minus = np.where((j_minus >= 0) & (j_minus < n), h[j_minus], 0.0)
    diff_pos = h_i_plus + h_i_minus - h_j_plus - h_j_minus
    diff_full = np.empty(2*n - 1, dtype=float)
    diff_full[n-1:] = diff_pos
    diff_full[:n-1] = diff_pos[1:][::-1]
    return diff_full

def optimal_epsilon(corr, diff, eps_max, tol=1e-8, max_iter=30):
    lo = 0.0
    hi = eps_max
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        C = corr + diff * mid
        idx = np.argmax(C)
        b = diff[idx]
        if b > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < tol:
            break
    return (lo + hi) * 0.5

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x, dx):
    corr = correlation_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    # log-sum-exp
    val = (m + np.log(np.exp(bc - m).sum())) / CURRENT_BETA
    return val * dx

def slsqp_optimize(h0, beta, rng, max_iter=200, dx=1.0):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(lambda x: softmax_obj(x, dx), h0, method="SLSQP",
                  bounds=bounds, constraints=cons,
                  options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple fix
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

import time
import numpy as np
from scipy.optimize import minimize

# globals for soft‑max SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def correlation_full(h, nfft=None):
    """Full correlation of h with (1‑h) using FFT."""
    n = h.size
    if nfft is None:
        nfft = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft((1.0 - h)[::-1], n=nfft)
    corr = np.fft.irfft(H * G, n=nfft)[: 2 * n - 1]
    return corr


def compute_c5_fast(h, nfft=None):
    """Discrete C₅ for a step function."""
    n = h.size
    corr = correlation_full(h, nfft)
    return corr.max() * (2.0 / n)


def enforce_sum(h, rng):
    """Adjust h so that Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h, corr=None):
    """Gradient of the worst‑shift overlap."""
    n = h.size
    if corr is None:
        corr = correlation_full(h)
    idx = np.argmax(corr)
    k = idx - (n - 1)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += 1.0 - h[:-k]
        grad[: n - k] -= h[k:]
    else:
        kk = -k
        grad[: n - kk] += 1.0 - h[kk:]
        grad[kk:] -= h[: n - kk]
    return grad, k, corr


def compute_diff_full(h, i, j):
    """diff[k] = h[i+k]+h[i‑k]−h[j+k]−h[j‑k] (zero outside)."""
    n = h.size
    k = np.arange(n)
    i_plus = i + k
    i_minus = i - k
    j_plus = j + k
    j_minus = j - k
    h_ip = np.where((i_plus >= 0) & (i_plus < n), h[i_plus], 0.0)
    h_im = np.where((i_minus >= 0) & (i_minus < n), h[i_minus], 0.0)
    h_jp = np.where((j_plus >= 0) & (j_plus < n), h[j_plus], 0.0)
    h_jm = np.where((j_minus >= 0) & (j_minus < n), h[j_minus], 0.0)
    diff_pos = h_ip + h_im - h_jp - h_jm
    diff_full = np.empty(2 * n - 1, dtype=float)
    diff_full[n - 1 :] = diff_pos
    diff_full[: n - 1] = diff_pos[1:][::-1]
    return diff_full


def optimal_epsilon(corr, diff, eps_max, tol=1e-8, max_iter=30):
    """Bisection to minimise max_k (corr_k + diff_k·ε) over ε∈[0,eps_max]."""
    lo, hi = 0.0, eps_max
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        C = corr + diff * mid
        idx = np.argmax(C)
        b = diff[idx]
        if b > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < tol:
            break
    return (lo + hi) * 0.5


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x, dx):
    corr = correlation_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    val = (m + np.log(np.exp(bc - m).sum())) / CURRENT_BETA
    return val * dx


def slsqp_optimize(h0, beta, rng, max_iter=250, dx=1.0):
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(lambda x: softmax_obj(x, dx), h0, method="SLSQP",
                  bounds=bounds, constraints=cons,
                  options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
