def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # Finally ensure rounding errors:
    h[:] = np.clip(h, 0.0, 1.0)
    return h

def binary_sa_run(n, rng, sub_deadline, init_h=None):
    target = n / 2.0
    if init_h is None:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n//2, replace=False)] = 1.0
    else:
        h = init_h.astype(float).copy()
        # Ensure sum target (should be already), but adjust if required.
        adjust_sum_exact(h, rng)
    c = np.correlate(h, h, mode='full')
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h.copy()
    best_worst = cur_worst
    ones = list(np.flatnonzero(h == 1.0))
    zeros = list(np.flatnonzero(h == 0.0))
    # temperature schedule
    T = cur_worst * 0.5 + 1e-6  # initial temperature
    decay = 0.9999
    while time.time() < sub_deadline:
        old_worst = cur_worst
        # random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # hi_vec before change
        hi_vec = np.concatenate((h[i::-1], h[i+1:]))
        start_i = n - 1 - i
        c[start_i:start_i + n] -= hi_vec  # diff = -1
        h[i] = 0.0
        # after removal, compute hj_vec
        hj_vec = np.concatenate((h[j::-1], h[j+1:]))
        start_j = n - 1 - j
        c[start_j:start_j + n] += hj_vec  # diff = +1
        h[j] = 1.0
        # new worst
        cur_min = c.min()
        cur_worst = target - cur_min
        if cur_worst < best_worst - 1e-12:
            best_worst = cur_worst
            best_h = h.copy()
        delta = cur_worst - old_worst
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            # revert updates
            # revert c for i (add back hi_vec)
            c[start_i:start_i + n] += hi_vec
            # revert c for j (subtract hj_vec)
            c[start_j:start_j + n] -= hj_vec
            h[i] = 1.0
            h[j] = 0.0
            cur_worst = old_worst
            # lists unchanged
        T *= decay
    return best_h, best_worst

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = n/2 exactly by moving mass between coordinates."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    h[:] = np.clip(h, 0.0, 1.0)
    return h

def binary_sa_run(
    n: int,
    rng: np.random.Generator,
    sub_deadline: float,
    init_h: np.ndarray | None = None,
):
    """Simulated annealing on binary vectors (swap 0↔1)."""
    target = n / 2.0
    if init_h is None:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    else:
        h = init_h.astype(float).copy()
        adjust_sum_exact(h, rng)

    c = np.correlate(h, h, mode="full")
    cur_min = c.min()
    cur_worst = target - cur_min
    best_h = h.copy()
    best_worst = cur_worst

    ones = list(np.flatnonzero(h == 1.0))
    zeros = list(np.flatnonzero(h == 0.0))

    T = max(cur_worst * 0.5, 1e-6)
    decay = 0.9999

    while time.time() < sub_deadline:
        old_worst = cur_worst

        i = rng.choice(ones)
        j = rng.choice(zeros)

        # remove i (1→0)
        hi_vec = np.concatenate((h[i::-1], h[i + 1 :]))
        start_i = n - 1 - i
        c[start_i : start_i + n] -= hi_vec
        h[i] = 0.0

        # add j (0→1)
        hj_vec = np.concatenate((h[j::-1], h[j + 1 :]))
        start_j = n - 1 - j
        c[start_j : start_j + n] += hj_vec
        h[j] = 1.0

        cur_min = c.min()
        cur_worst = target - cur_min

        if cur_worst < best_worst - 1e-12:
            best_worst = cur_worst
            best_h = h.copy()

        delta = cur_worst - old_worst
        accept = delta <= 0 or rng.random() < np.exp(-delta / T)

        if accept:
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            # revert
            c[start_i : start_i + n] += hi_vec
            c[start_j : start_j + n] -= hj_vec
            h[i] = 1.0
            h[j] = 0.0
            cur_worst = old_worst

        T *= decay

    return best_h, best_worst

def greedy_improve(
    h: np.ndarray,
    rng: np.random.Generator,
    max_swaps: int,
    deadline: float,
):
    """Local swaps that increase the worst (minimum) autocorrelation."""
    n = h.size
    target = n / 2.0
    while max_swaps > 0 and time.time() < deadline:
        corr = np.correlate(h, h, mode="full")
        min_idx = int(np.argmin(corr))
        k = min_idx - (n - 1)  # shift amount

        if k == 0:
            break

        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_end <= i_start:
            break

        idx_range = np.arange(i_start, i_end)
        hi = h[idx_range]
        hk = h[idx_range + k]

        mismatched_1 = idx_range[(hi == 1) & (hk == 0)]
        mismatched_0 = idx_range[(hi == 0) & (hk == 1)]

        if mismatched_1.size == 0 or mismatched_0.size == 0:
            break

        i_swap = rng.choice(mismatched_1)
        j_swap = rng.choice(mismatched_0)

        h[i_swap] = 0.0
        h[j_swap] = 1.0
        max_swaps -= 1

    return h, compute_c5(h)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # Parent solution if provided
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        parent = adjust_sum_exact(parent, rng)
        best_h = parent.copy()
        best_c = compute_c5(best_h)
        best_n = parent.shape[0]

    # Candidate discretisations
    n_candidates = [360, 240, 180]
    if best_n is not None and best_n not in n_candidates:
        n_candidates.insert(0, best_n)

    # Main simulated‑annealing loop
    while time.time() + 2.0 < deadline:
        for n in n_candidates:
            if time.time() + 2.0 >= deadline:
                break

            # Warm‑start: binary version of current best (if same size)
            init_h = None
            if best_h is not None and best_h.size == n:
                idx = np.argpartition(best_h, -n // 2)[-n // 2 :]
                bin_vec = np.zeros(n, dtype=float)
                bin_vec[idx] = 1.0
                init_h = bin_vec

            sub_deadline = time.time() + 1.0
            h_sa, worst_sa = binary_sa_run(n, rng, sub_deadline, init_h)
            c5_sa = worst_sa * (2.0 / n)

            if c5_sa < best_c - 1e-12:
                # Try a quick greedy refinement
                greedy_deadline = deadline - 0.5
                h_g, c5_g = greedy_improve(
                    h_sa.copy(), rng, max_swaps=200, deadline=greedy_deadline
                )
                if c5_g < best_c - 1e-12:
                    best_h, best_c, best_n = h_g, c5_g, n
                else:
                    best_h, best_c, best_n = h_sa, c5_sa, n

    # Optional up‑sampling if time permits
    if (
        best_h is not None
        and time.time() + 1.0 < deadline
        and best_n * 2 <= 720
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        greedy_deadline = deadline - 0.5
        h_up, c5_up = greedy_improve(
            h_up, rng, max_swaps=500, deadline=greedy_deadline
        )
        if c5_up < best_c:
            best_h, best_c, best_n = h_up, c5_up, n_up

    # Fallback if nothing succeeded
    if best_h is None:
        best_n = 180
        best_h = np.zeros(best_n, dtype=float)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
