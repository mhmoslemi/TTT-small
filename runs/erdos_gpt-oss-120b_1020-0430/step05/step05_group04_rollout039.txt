def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95

    best_h = None
    best_c = float('inf')
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    candidate_ns = []
    if parent is not None:
        candidate_ns = [parent.size, max(240, parent.size//2), 180]
    else:
        candidate_ns = [360, 240, 180]
    candidate_ns = [n for n in candidate_ns if n%2==0]

    # Initial best from parent if exists
    if parent is not None:
        # Ensure sum constraint (should be already)
        p_c = compute_c5(parent)
        best_h = parent.copy()
        best_c = p_c
        best_n = parent.size

    restart = 0

import time
import numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass to enforce Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=30):
    """Projected gradient descent using the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:]     -= h[:upper]        # - h[i]
        h = h - (step * (0.5 ** it)) * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def smoothed_descent(h, target, rng, deadline):
    """Gradient descent on a soft‑max surrogate of the max overlap."""
    n = h.size
    betas = [1, 5, 20, 80, 200, 500, 2000, 5000, 20000, 50000, 100000]
    steps_per = [30, 30, 30, 30, 30, 30, 20, 20, 15, 10, 10]
    base_step = 0.3
    for beta, steps in zip(betas, steps_per):
        if time.time() > deadline:
            break
        step = base_step / (beta ** 0.2 + 1e-12)
        for _ in range(steps):
            if time.time() > deadline:
                break
            # overlaps for all non‑negative shifts
            conv_hh = np.convolve(h, h[::-1], mode="full")
            overlaps = target - conv_hh[n - 1 : n - 1 + n]
            # soft‑max weights
            scaled = beta * overlaps
            m = np.max(scaled)
            exp_term = np.exp(scaled - m)
            p = exp_term / exp_term.sum()
            # gradient of the surrogate
            p_rev = p[::-1]
            conv1 = np.convolve(p_rev, h, mode="full")
            term1 = conv1[n - 1 : n - 1 + n]
            conv2 = np.convolve(p, h, mode="full")
            term2 = conv2[:n]
            g = np.ones(n) - term1 - term2
            h = h - step * g
            np.clip(h, 0.0, 1.0, out=h)
            adjust_sum_exact(h, rng)
            step *= 0.93
    return h
