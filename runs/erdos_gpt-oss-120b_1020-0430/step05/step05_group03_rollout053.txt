    # we need to add diff to some entries (if diff>0) not at 1, else subtract from entries >0
    # water-filling: sort free entries by distance to bound.
    # For diff>0: can increase each free (v < 1) up to 1.

def subgradient_descent(h0, rng, max_iters, step0):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    target = h.size / 2.0
    step = step0
    for it in range(max_iters):
        grad = worst_gradient(h)
        # step size schedule: step0 / sqrt(it+1)
        step = step0 / np.sqrt(it + 1)
        h_new = h - step * grad

            best_h = h_new.copy()

        h = h_new

def slsqp_optimize(h0, beta, max_iter=600):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    def softmax(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad

def project_onto_box_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(v - mid, 0.0, 1.0)

def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_onto_box_and_sum(v, n / 2.0)

def random_binary_half(n: int, rng: np.random.Generator) -> np.ndarray:
    v = np.zeros(n)
    v[rng.choice(n, n // 2, replace=False)] = 1.0
    return v

def subgradient_descent(h0: np.ndarray, rng, max_iters: int, step0: float):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = np.ones_like(h)
            grad[: n - k] -= h[k:]
            grad[k:] -= h[: n - k]
        step = step0 / np.sqrt(it + 1)
        h = h - step * grad
        h = project_onto_box_and_sum(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 600):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_and_sum(x, target)

def gradient_refine(h: np.ndarray, rng, step: float = 0.015, max_iter: int = 12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_and_sum(h, target)
    return h

        h_opt, c_opt = subgradient_descent(h_start, rng, max_iters, step0=1.0)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, n
        if time.time() - start > 0.6 * budget_s and best_n >= 128:
            h_ref = slsqp_optimize(best_h, beta=400.0, max_iter=500)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    if best_n < 512 and time.time() - start < 0.92 * budget_s:
        h_up = upscale(best_h, 512)
        h_up_opt, c_up_opt = subgradient_descent(h_up, rng, 12000, step0=0.7)
        if c_up_opt < best_c:
            best_h, best_c, best_n = h_up_opt.copy(), c_up_opt, 512
        h_up_opt = gradient_refine(best_h, rng, step=0.02, max_iter=10)
        c_up_opt = compute_c5(h_up_opt)
        if c_up_opt < best_c:
            best_h, best_c, best_n = h_up_opt.copy(), c_up_opt, 512

    if best_n == 512:
        for beta in (200., 800., 2000., 8000., 20000.):
            if time.time() - start > 0.97 * budget_s:
                break
            h_ref = slsqp_optimize(best_h, beta=beta, max_iter=500)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    while time.time() - start < budget_s * 0.998:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            break
        eps = rng.random() * max_eps * 0.05
        h_new = best_h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        h_new = project_onto_box_and_sum(h_new, best_h.size / 2.0)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    best_h = project_onto_box_and_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
