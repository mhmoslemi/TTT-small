def compute_c5(h):

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Adjust h so that sum(h) = len(h)/2 exactly, staying in [0,1]."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def greedy_binary(n, rng):
    """Construct binary h with exactly n/2 ones minimizing max correlation."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h, O

def hill_climb(h_bin, max_O, n, rng, deadline):
    """Random swaps on binary vector to reduce max correlation."""
    best_h = h_bin.copy()
    best_max = max_O
    cur_h = h_bin.copy()
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        cur_h[i] = 0
        cur_h[j] = 1
        O_new = np.max(
            np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode="full")
        )
        if O_new < best_max:
            best_max = O_new
            best_h = cur_h.copy()
            ones[ones == i] = j
            zeros[zeros == j] = i
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_max

def slsqp_optimize_h(h0, beta, max_iter):
    n = h0.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return x
