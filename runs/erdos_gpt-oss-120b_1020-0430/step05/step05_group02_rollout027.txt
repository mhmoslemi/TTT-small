class SoftmaxWithGrad:
    __slots__ = ('beta',)
    def __init__(self, beta):
        self.beta = beta
    def __call__(self, h):
        n = h.size
        corr = np.correlate(h, 1.0 - h, mode='full')
        beta_corr = self.beta * corr
        max_bc = np.max(beta_corr)
        exp_args = beta_corr - max_bc
        exps = np.exp(exp_args)
        sum_exps = np.sum(exps)
        w = exps / sum_exps
        obj = np.dot(w, corr)
        # convolution for gradient
        a = 1.0 - h
        # term1 = conv of w_rev with a
        w_rev = w[::-1]
        conv1 = np.convolve(w_rev, a, mode='full')
        term1 = conv1[n-1 : n-1 + n]
        conv2 = np.convolve(w, h, mode='full')
        term2 = conv2[n-1 : n-1 + n]
        grad = term1 - term2
        return obj, grad
```

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # exact sum correction
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time
import numpy as np
import scipy.optimize as opt

CURRENT_TARGET = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


class SoftmaxWithGrad:
    __slots__ = ("beta",)

    def __init__(self, beta):
        self.beta = beta

    def __call__(self, h):
        n = h.size
        corr = np.correlate(h, 1.0 - h, mode="full")
        beta_corr = self.beta * corr
        max_bc = np.max(beta_corr)
        exp_args = beta_corr - max_bc
        exps = np.exp(exp_args)
        sum_exps = np.sum(exps)
        w = exps / sum_exps
        obj = np.dot(w, corr)
        a = 1.0 - h
        w_rev = w[::-1]
        conv1 = np.convolve(w_rev, a, mode="full")
        term1 = conv1[n - 1 : n - 1 + n]
        conv2 = np.convolve(w, h, mode="full")
        term2 = conv2[n - 1 : n - 1 + n]
        grad = term1 - term2
        return obj, grad


def slsqp_optimize_with_softmax(h0, beta, rng, max_iter=300):
    n = h0.size
    global CURRENT_TARGET
    CURRENT_TARGET = n / 2.0
    prob = SoftmaxWithGrad(beta)
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        prob,
        h0,
        method="SLSQP",
        jac=True,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x
