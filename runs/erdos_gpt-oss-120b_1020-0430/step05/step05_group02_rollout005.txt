CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    L = 2*n
    A = np.fft.rfft(h, n=L)
    B = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(A * np.conj(B), n=L)
    max_corr = np.max(corr)
    return float(max_corr * (2.0 / n))

def adjust_sum_exact(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def softmax_obj(x):
    # global CURRENT_BETA
    A = np.fft.rfft(x, n=2*x.size)
    B = np.fft.rfft(1.0 - x, n=2*x.size)

import time, numpy as np
from scipy.optimize import minimize

# globals for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """Maximum overlap using FFT (linear correlation)."""
    n = h.size
    L = 2 * n
    A = np.fft.rfft(h, n=L)
    B = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(A * np.conj(B), n=L)
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, target, rng):
    """Redistribute to satisfy Σh = target exactly."""
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    L = 2 * x.size
    A = np.fft.rfft(x, n=L)
    B = np.fft.rfft(1.0 - x, n=L)
    corr = np.fft.irfft(A * np.conj(B), n=L)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, n / 2.0, rng)
    return x

def worst_shift_and_grad(h):
    """Return shift k with maximal overlap and gradient of that overlap."""
    n = h.size
    L = 2 * n
    A = np.fft.rfft(h, n=L)
    B = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(A * np.conj(B), n=L)
    idx = np.argmax(corr)
    k = idx - (n - 1)          # shift index (may be negative)
    g = np.ones_like(h)       # ∂/∂h of Σh term
    if k >= 0:
        upper = n - k
        g[:upper] -= h[k:]
        g[k:]    -= h[:upper]
    else:
        kk = -k
        upper = n - kk
        g[:upper] -= h[kk:]
        g[kk:]    -= h[:upper]
    return k, g

def gradient_descent(h, rng, steps=30, step_start=0.6, step_decay=0.92):
    """Projected gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    step = step_start
    for _ in range(steps):
        _, g = worst_shift_and_grad(h)
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        adjust_sum_exact(h, target, rng)
        step *= step_decay
    return h

def binary_hill_climb(h_bin, rng, max_trials=3000):
    """Simple hill‑climb on binary vectors by swapping 1↔0."""
    best = h_bin.copy()
    best_c = compute_c5(best)
    n = best.size
    for _ in range(max_trials):
        i = rng.integers(0, n)
        if best[i] == 0:
            continue
        j = rng.integers(0, n)
        if best[j] == 1:
            continue
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        c = compute_c5(cand)
        if c < best_c - 1e-12:
            best, best_c = cand, c
    return best, best_c
