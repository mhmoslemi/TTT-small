import time
import numpy as np

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    eps = 1e-12
    diff = target - h.sum()
    while abs(diff) > eps:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < eps:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # ---- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
    else:
        n = 512
        h = rng.random(n)
        h = np.clip(h, 0.0, 1.0)
    n = h.size
    h = enforce_sum(h, rng)

    # ---- initial autocorrelation ---------------------------------------
    S = np.correlate(h, h, mode="full")[n - 1 :]  # S[k] = sum_i h_i h_{i+k}
    minS = S.min()
    best_h = h.copy()
    best_minS = minS

    # ---- parameters ----------------------------------------------------
    max_scale = 0.5
    min_scale = 1e-4
    no_improve = 0
    shake_thresh = 20000  # iterations without improvement before a shake

    # ---- main stochastic hill‑climb ------------------------------------
    while time.time() - start < total * 0.98:
        elapsed = time.time() - start
        t_frac = elapsed / total
        step_scale = max_scale * (min_scale / max_scale) ** t_frac

        # decide move type
        if rng.random() < 0.85:            # gradient‑guided move
            k0 = int(np.argmin(S))
            if k0 == 0:
                grad = 2.0 * h
            else:
                grad = np.zeros(n)
                grad[: n - k0] += h[k0:]
                grad[k0:] += h[: n - k0]
            i = int(np.argmin(grad))
            j = int(np.argmax(grad))
            if i == j:
                continue
        else:                               # random move
            i = rng.integers(0, n)
            trials = 0
            while h[i] <= 0.0 and trials < 10:
                i = rng.integers(0, n)
                trials += 1
            j = rng.integers(0, n)
            trials = 0
            while h[j] >= 1.0 and trials < 10:
                j = rng.integers(0, n)
                trials += 1
            if i == j:
                continue

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random() * step_scale
        if eps < 1e-12:
            continue

        # ---- try the move ------------------------------------------------
        S_old = S.copy()
        hi, hj = h[i], h[j]
        di, dj = -eps, eps

        # contributions using the *old* h values
        S[: i + 1] += di * h[i :: -1]
        S[i:] += di * h[i:]
        S[: j + 1] += dj * h[j :: -1]
        S[j:] += dj * h[j:]
        S[0] += di * di + dj * dj
        if i != j:
            S[abs(i - j)] += di * dj

        h[i] = hi - eps
        h[j] = hj + eps

        new_minS = S.min()
        if new_minS > minS + 1e-12:          # improvement
            minS = new_minS
            no_improve = 0
            if new_minS > best_minS + 1e-12:
                best_minS = new_minS
                best_h = h.copy()
        else:                                 # reject
            S[:] = S_old
            h[i] = hi
            h[j] = hj
            no_improve += 1

        # ---- shake to escape stagnation ---------------------------------
        if no_improve >= shake_thresh:
            h = best_h + rng.normal(scale=0.02, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
            S = np.correlate(h, h, mode="full")[n - 1 :]
            minS = S.min()
            no_improve = 0

    # ---- final sanity ---------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = 1.0 - (2.0 * best_minS) / n
    return best_h.tolist(), float(final_c), int(n)
