import time, numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {0<=x<=1, sum x = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ---------- gradient of smoothed max ----------
def grad_from_weights(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Weighted sum of shift‑gradients, w corresponds to shifts -(n-1)..+(n-1)."""
    n = h.size
    grad = np.zeros_like(h)

    # shift 0
    w0 = w[n - 1]
    grad += w0 * (1.0 - 2.0 * h)

    # positive and negative shifts
    for s in range(1, n):
        wp = w[n - 1 + s]   # +s
        if wp != 0.0:
            grad[: n - s] += wp * (1.0 - h[s:])
            grad[s:]      -= wp * h[: n - s]

        wn = w[n - 1 - s]   # -s
        if wn != 0.0:
            grad[: n - s] -= wn * h[s:]
            if n - 2 * s > 0:
                grad[s : n - s] += wn * (1.0 - h[: n - 2 * s])
    return grad


# ---------- local hill‑climb (same as previous code) ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

    start = time.time()

    # target size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_capped_simplex(h0, n / 2.0)
        inits = [h0]
    else:
        n = 512
        inits = []

    # a few extra starts (binary and random)
    def init_binary():
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        return h

    def init_random():
        v = rng.random(n)
        return project_onto_capped_simplex(v, n / 2.0)

    for _ in range(3):
        inits.append(init_binary())
    for _ in range(2):
        inits.append(init_random())

    best_h = None
    best_c = np.inf

    # ----- smoothed‑gradient descent on each start -----
    for h in inits:
        if time.time() - start > budget_s * 0.88:
            break
        eta = 0.8
        mu = 0.9
        vel = np.zeros_like(h)
        beta = 5.0
        for it in range(2000):
            if time.time() - start > budget_s * 0.88:
                break
            corr = np.correlate(h, 1 - h, mode="full")
            bc = beta * corr
            m = np.max(bc)
            exp_bc = np.exp(bc - m)
            w = exp_bc / exp_bc.sum()
            grad = grad_from_weights(h, w)
            vel = mu * vel - eta * grad
            h = project_onto_capped_simplex(h + vel, n / 2.0)
            if (it + 1) % 400 == 0:
                eta *= 0.7
                beta = min(beta * 2.0, 2000.0)

        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # brief hill‑climb refinement
        h2, c2 = guided_hill_climb(best_h, rng, max_iters=800, step_scale=0.6)
        if c2 < best_c:
            best_h, best_c = h2.copy(), c2

    # ----- final stochastic tweaks until time out -----
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
