Implementation of smoothmax objective:

```
def smoothmax_obj(x, beta):
    N = x.size
    fft_len = 1 << ((2*N-1).bit_length())
    # precompute fft_len once per call? Not efficient; we can compute inside as it's constant.
    # But they call with varying x; constructing fft_len each time fine.
    fx = np.fft.rfft(x, n=fft_len)
    fg = np.fft.rfft(1.0 - x, n=fft_len)
    corr = np.fft.irfft(fx * np.conj(fg), n=fft_len)[:2*N-1]
    bc = beta * corr
    max_bc = bc.max()
    # log-sum-exp
    val = (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / beta
    return float(val)
```

def correlation_fft(x):
    N = x.shape[0]
    M = 1 << ((2*N-1).bit_length())
    fx = np.fft.rfft(x, n=M)
    fg = np.fft.rfft(1.0 - x, n=M)
    corr = np.fft.irfft(fx * np.conj(fg), n=M)[:2*N-1]
    return corr

def convolve_fft(a, b, M):
    fa = np.fft.rfft(a, n=M)
    fb = np.fft.rfft(b, n=M)
    conv = np.fft.irfft(fa * fb, n=M)[:len(a)+len(b)-1]
    return conv
```

Implementation details for binary_local_search:

Given binary vector h (0/1).

Compute corr = correlation_fft(h, M). Determine max_corr and shift index idx_max (0..2n-2). Convert shift s = idx_max - (n-1). Overlap interval for shift s: for s >=0, consider indices i=0..n-1-s; for s<0, consider i=-s..n-1. But correlation at shift s for s>=0 is as defined; for s<0, same formula but from index.

Now we need to find a swap (i,j) where h[i]=1, h[j]=0, such that the overlap at shift s reduces.

Contribution to overlap at shift s from index i (if i within overlap range) is h[i] * (1 - h[i+s]) unless i+s out of range. Since h[i] = 1, contribution is (1 - h[i+s]) which either 0 or 1 depending on h[i+s] (if i+s < n). For shift s>0: i can be 0..n-1-s; for shift s<0: i can be -s..n-1.

Thus we can compute for each i within overlap region the neighbor value h[i+s] if s>=0 else h[i+s]. Actually for s negative, i+s is lower index.

Goal: find a 1 at i where neighbor h[i+s] = 0 (so contribution 1) and a 0 at j where neighbor h[j+s] = 1 (so swapping reduces overlap by 2?). Let's compute effect of swapping:

- Before: contribution at i = 1 * (1 - h[i+s]) = 1 if h[i+s]=0 else 0.
- After swapping i's 1 with 0, i becomes 0 => contribution = 0.
- Similarly, at j (originally 0) after swapping becomes 1: contribution = 1 * (1 - h[j+s]) = 1 if h[j+s]=0 else 0.

Thus delta = new - old = - (1 - h[i+s]) + (1 - h[j+s]) = h[i+s] - h[j+s].

Thus swap improves (decreases overlap) if h[i+s] < h[j+s] (i.e., neighbor of i is 0 and neighbor of j is 1). So we need to find a 1 at i with neighbor 0, and a 0 at j with neighbor 1.

Thus algorithm: find indices of ones where neighbor is 0 (call set S1), zeros where neighbor is 1 (set S0). If both nonempty, choose any i in S1 and j in S0, swap them.

We also need to ensure i and j are not overlapping with each other or causing out-of-range neighbors accordingly.
```

def binary_local_improve(h, rng, deadline):
    N = h.size
    M = compute_fft_len(N)
    while time.time() < deadline:
        corr = correlation_fft(h, M)
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        s = idx - (N - 1)
        if s == 0:
            break
        # Determine valid range for i and j (overlap indices)
        if s > 0:
            i_range = np.arange(0, N - s)
            j_range = i_range  # same range
        else:
            i_range = np.arange(-s, N)
            j_range = i_range
        # For each i in i_range where h[i]=1 and h[i+s]=0
        # We can compute neighbor = h[i+s]
        neighbor_i = h[i_range + s]
        # Candidate i's
        cand_i = i_range[(h[i_range]==1) & (neighbor_i==0)]
        if cand_i.size == 0:
            # no improving swap, break
            break
        # For zeros with neighbor=1
        neighbor_j = h[j_range + s]
        cand_j = j_range[(h[j_range]==0) & (neighbor_j==1)]
        if cand_j.size == 0:
            break
        i = rng.choice(cand_i)
        j = rng.choice(cand_j)
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        # continue
    return h

    # correlation
    fx = np.fft.rfft(x, n=M_corr)
    fg = np.fft.rfft(1.0 - x, n=M_corr)
    corr = np.fft.irfft(fx * np.conj(fg), n=M_corr)[:2*N-1]  # length 2N-1
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    Z = exp_bc.sum()
    p = exp_bc / Z
    # convolution for term3
    fp = np.fft.rfft(p, n=M_conv)
    fx_conv = np.fft.rfft(x, n=M_conv)

import time
import numpy as np

# ----- basic helpers -------------------------------------------------
def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2]."""
    N = h.size
    M = 1 << ((2 * N - 1).bit_length())
    fh = np.fft.rfft(h, n=M)
    fg = np.fft.rfft(1.0 - h, n=M)
    corr = np.fft.irfft(fh * np.conj(fg), n=M)[: 2 * N - 1]
    return float(np.max(corr) * (2.0 / N))


def correlation_fft(x, M):
    """Linear correlation of x with (1-x) using FFT, length M >= 2*len(x)-1."""
    fx = np.fft.rfft(x, n=M)
    fg = np.fft.rfft(1.0 - x, n=M)
    corr = np.fft.irfft(fx * np.conj(fg), n=M)[: 2 * x.size - 1]
    return corr


def convolve_fft(a, b, M):
    """Linear convolution via FFT, M >= len(a)+len(b)-1."""
    fa = np.fft.rfft(a, n=M)
    fb = np.fft.rfft(b, n=M)
    conv = np.fft.irfft(fa * fb, n=M)[: len(a) + len(b) - 1]
    return conv


def adjust_sum_exact(x, rng):
    """Force Σx = len(x)/2 while staying in [0,1]."""
    N = x.size
    target = N / 2.0
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, N)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x


def fractional_to_binary(h, rng):
    """Take largest N/2 entries of h as 1, others 0 (exact sum)."""
    N = h.size
    idx = np.argpartition(h, -N // 2)[-N // 2 :]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    adjust_sum_exact(bin_h, rng)
    return bin_h


# ----- smooth‑max gradient step --------------------------------------
def smoothmax_step(x, beta, step, rng):
    N = x.size
    # length for correlation (2N-1) and convolution (3N-2)
    M_corr = 1 << ((2 * N - 1).bit_length())
    M_conv = 1 << ((3 * N - 2).bit_length())

    # correlation for gradient
    corr = correlation_fft(x, M_corr)
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    Z = exp_bc.sum()
    p = exp_bc / Z

    # FFTs for convolutions
    x_fft = np.fft.rfft(x, n=M_conv)
    p_fft = np.fft.rfft(p, n=M_conv)
    conv_fwd = np.fft.irfft(p_fft * x_fft, n=M_conv)[: len(p) + N - 1]
    term3 = conv_fwd[N - 1 : N - 1 + N]

    p_rev = p[::-1]
    p_rev_fft = np.fft.rfft(p_rev, n=M_conv)
    conv_rev = np.fft.irfft(p_rev_fft * x_fft, n=M_conv)[: len(p_rev) + N - 1]
    term2 = conv_rev[N - 1 : N - 1 + N]

    grad = 1.0 - term2 - term3
    x_new = x - step * grad
    np.clip(x_new, 0.0, 1.0, out=x_new)

    # enforce exact sum
    target = N / 2.0
    diff = target - x_new.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, N)
            if diff > 0:
                add = min(diff, 1.0 - x_new[i])
            else:
                add = max(diff, -x_new[i])
            x_new[i] += add
            diff = target - x_new.sum()

    # compute new C5
    corr_new = correlation_fft(x_new, M_corr)
    c5 = float(np.max(corr_new) * (2.0 / N))
    return x_new, c5


# ----- gradient descent with smooth‑max schedule ----------------------
def gradient_descent(x0, betas, steps_per_beta, step0, rng, deadline):
    x = x0.copy()
    best_x = x.copy()
    best_c = compute_c5(x)
    for beta in betas:
        step = step0
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_x, best_c
            x, c = smoothmax_step(x, beta, step, rng)
            if c < best_c:
                best_c, best_x = c, x.copy()
            step *= 0.985  # gradual decay
    return best_x, best_c


# ----- binary local improvement (linear case) ------------------------
def binary_local_improve(h, rng, deadline):
    N = h.size
    while time.time() < deadline:
        corr = correlation_fft(h, 1 << ((2 * N - 1).bit_length()))
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        s = idx - (N - 1)
        if s == 0:
            break
        if s > 0:
            rng_range = np.arange(0, N - s)
        else:
            rng_range = np.arange(-s, N)
        neigh = h[rng_range + s]
        cand_i = rng_range[(h[rng_range] == 1) & (neigh == 0)]
        if cand_i.size == 0:
            break
        cand_j = rng_range[(h[rng_range] == 0) & (neigh == 1)]
        if cand_j.size == 0:
            break
        i = rng.choice(cand_i)
        j = rng.choice(cand_j)
        h[i] = 0.0
        h[j] = 1.0
    return h
