def binary_local_search(h, rng, deadline):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    # precompute ones and zeros lists for fast selection
    while time.time() < deadline:
        # compute correlation
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_bad = corr.max()
        # If max_bad is equal to target? keep record.
        # Identify shifts achieving max_bad
        best_shifts_idx = np.where(corr == max_bad)[0]
        k_idx = rng.choice(best_shifts_idx)
        # Convert index to shift value: shift = k_idx - (n-1)
        shift = int(k_idx - (n - 1))
        # compute ones and zeros arrays
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        # Determine bad ones for this shift
        if shift >= 0:
            # condition: i+shift >= n OR cur_h[i+shift]==0
            bad_mask = (ones + shift >= n) | (cur_h[ones + shift] == 0)
            # good zeros: j+shift < n AND cur_h[j+shift]==1
            good_mask = (zeros + shift < n) & (cur_h[zeros + shift] == 1)
        else:
            # shift negative: i+shift < 0 => out of bounds => bad
            s = -shift
            bad_mask = (ones < s) | (cur_h[ones - s] == 0)
            good_mask = (zeros >= s) & (cur_h[zeros - s] == 1)
        bad_ones = ones[bad_mask]
        good_zeros = zeros[good_mask]
        if len(bad_ones) == 0 or len(good_zeros) == 0:
            # cannot improve this shift directly, perform random swap
            i = rng.choice(ones)
            j = rng.choice(zeros)
        else:
            i = rng.choice(bad_ones)
            j = rng.choice(good_zeros)
        # Swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # compute new c
        new_c = compute_c5(cur_h)
        # Accept if new_c <= cur_c + tolerance maybe
        if new_c <= cur_c + 1e-12:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # Possibly break if best_c is already low enough
        if best_c < 0.3808:
            break
    return best_h, best_c
```

def compute_c5_fft(h):
    n = h.size
    nfft = 1
    while nfft < 2*n:
        nfft <<= 1
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * G, nfft)
    # conv length nfft; we need the first 2n-1 entries (full correlation)
    # With zero padding, conv[i] corresponds to sum_j h[j] * (1-h[i-j]) for i index.
    # The correlation we need is conv[0:2n-1]
    # But we can just take conv[:2*n-1].
    corr = conv[: 2*n - 1]
    return float(corr.max())

import time
import numpy as np

def correlation_array(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    nfft = 1
    while nfft < 2 * n:
        nfft <<= 1
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * np.conj(G), nfft)[: 2 * n - 1]
    return conv

def compute_c5(h):
    """Maximum overlap integral for a sampled step function."""
    n = h.shape[0]
    max_bad = correlation_array(h).max()
    return float(max_bad * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact redistribution to satisfy Σh = n/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def fractional_sa_hill_climb(h, rng, deadline):
    """Transfer‑based simulated annealing for continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                it = 0
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if best_c * (cur.size / 2.0) < 0.3808:
            break
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = float('inf')
    best_n = None

    # candidate sample sizes
    n_candidates = [360, 480, 600, 720]

    for n in n_candidates:
        if time.time() >= deadline:
            break
        start_vecs = []

        # parent‑derived start (if available)
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)

        # random binary starts
        for _ in range(2):
            bin_rand = np.zeros(n)
            bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            start_vecs.append(bin_rand)

        # biased random binary (top‑half of random uniforms)
        vals = rng.random(n)
        idx = np.argpartition(vals, -n // 2)[-n // 2 :]
        bin_bi = np.zeros(n)
        bin_bi[idx] = 1.0
        start_vecs.append(bin_bi)

        for h0 in start_vecs:
            if time.time() >= deadline:
                break
            h0 = (h0 > 0.5).astype(float)   # ensure strict binary
            # allocate a modest time slice per start
            remaining = deadline - time.time()
            per_dead = time.time() + min(remaining * 0.3, 20.0)
            h_opt, c_opt = binary_local_search(h0, rng, per_dead)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n
            # early exit if already very good
            if best_c < 0.3808:
                break

    if best_h is not None and time.time() < deadline:
        h_frac = best_h.astype(float)
        # small random jitter to escape pure binary basin
        jitter = rng.random(best_n) * 0.02
        h_frac = np.clip(h_frac + jitter, 0.0, 1.0)
        adjust_sum_exact(h_frac, rng)
        h_frac, c_frac = fractional_sa_hill_climb(h_frac, rng, deadline)
        if c_frac < best_c:
            best_h, best_c, best_n = h_frac.copy(), c_frac, best_n

    # final safety checks
    if best_h is None:
        # fallback simple half‑ones
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        if abs(best_h.sum() - best_n / 2.0) > 1e-9:
            adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
