def compute_gradient(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift == 0:
        grad[:] = 1 - 2*h
    elif shift > 0:
        k = shift
        grad[:n-k] = (1 - h[k:]) - h[:n-k]
        # grad[n-k:] = 0
    else: # shift < 0
        k = -shift
        grad[k:] = (1 - h[:n-k]) - h[k:]
        # grad[:k] = 0
    return grad

def adjust_sum_exact(h, target_sum, rng):
    diff = target_sum - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            # add up to min(diff, 1 - h[i])
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target_sum - h.sum()
    # after loop, may still be diff if all bounds reached but should not happen given target within bounds.
    return h

Now implement compute_c5 quickly:

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

Now for gradient step schedule:

def gradient_refine(h, rng, step, max_iter):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        grad = compute_gradient(h)
        # step size decreasing
        step_sz = step * (0.5 ** it)
        h = h - step_sz * grad
        # project
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust by small corrections
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

Now implement fractional SA:

def fractional_sa_hill_climb(h, rng, deadline):
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # random direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        # propose delta uniformly from (0, max_transfer)
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        # cooling schedule
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            # revert
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 15000:
            break
    return best, best_c

Now define local_optimize:

def local_optimize(n, h0, rng, deadline):
    target = n / 2.0
    # ensure feasible
    h = np.clip(h0, 0.0, 1.0)
    h = adjust_sum_exact(h, target, rng)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    # We'll perform repeated cycles until deadline
    while time.time() < deadline:
        # gradient refinement
        h = gradient_refine(h, rng, step=0.03, max_iter=8)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # SA with sub-deadline
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate perhaps 0.4*remaining to SA
        sub_dead = time.time() + min(remaining * 0.5, 30.0)  # max 30 sec per SA
        h_sa, c_sa = fractional_sa_hill_climb(h, rng, sub_dead)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        # set h to the best of h and h_sa for next cycle
        h = h_sa if c_sa <= c else h
        # if not improved for many cycles break
        if time.time() + 1.0 > deadline:
            break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP smooth max
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap C5 for a step function sampled in h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Shift mass to make sum(h) == target while respecting 0/1 bounds."""
    diff = target - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def compute_gradient(h: np.ndarray) -> np.ndarray:
    """Subgradient of the worst overlap w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        k = shift
        m = n - k
        grad[:m] = (1.0 - h[k:]) - h[:m]
    else:  # shift < 0
        k = -shift
        m = n - k
        grad[k:] = (1.0 - h[:m]) - h[k:]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.03, max_iter: int = 12) -> np.ndarray:
    """Projected gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        grad = compute_gradient(h)
        step_sz = step * (0.5 ** it)
        h -= step_sz * grad
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ----- SLSQP smooth‑max helpers -----
def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 800) -> np.ndarray:
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = adjust_sum_exact(x, CURRENT_TARGET, rng)
    return x


# ----- Fractional simulated annealing -----
def fractional_sa_hill_climb(h: np.ndarray, rng: np.random.Generator,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Transfer‑based SA on continuous vectors."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c, best = c, cur.copy()
                last_improve = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_improve > 15000:
            break
    return best, best_c


def local_optimize(n: int, h0: np.ndarray, rng: np.random.Generator,
                   deadline: float) -> tuple[np.ndarray, float]:
    """Repeated gradient+SA cycles until deadline."""
    target = n / 2.0
    h = np.clip(h0, 0.0, 1.0)
    h = adjust_sum_exact(h, target, rng)
    best_h, best_c = h.copy(), compute_c5(h)

    while time.time() < deadline:
        # gradient refinement
        h = gradient_refine(h, rng, step=0.03, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

        # allocate time for a SA run
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        sub_dead = time.time() + min(remaining * 0.5, 30.0)
        h_sa, c_sa = fractional_sa_hill_climb(h, rng, sub_dead)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        h = h_sa if c_sa <= c else h

    return best_h, best_c
