def hill_climb(h0, rng, time_budget):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    last_improve = 0
    # schedule parameters
    n = h.size
    while time.time() - start < time_budget:
        # compute gradient
        grad = worst_gradient(h)
        # determine scale: dec as sqrt of time proportion
        elapsed = time.time() - start
        proportion = elapsed / time_budget
        scale = max(0.5 * (0.9 ** (it // 2000)), 0.005)  # constant lower bound
        # decide perturb type
        if rng.random() < 0.85:
            nh = guided_perturb(h, grad, rng, scale=scale)
        else:
            i, j = rng.choice(n, 2, replace=False)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * 0.5
                nh = h.copy()
                nh[i] -= eps
                nh[j] += eps
            else:
                nh = h
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
            last_improve = it
            # early exit if surpass threshold
            if best_c <= 0.3808:
                break
        else:
            # maybe accept if random chance to escape
            if rng.random() < 0.001:
                h = nh
        it += 1
        # restart if no improvement for many iterations
        if it - last_improve > 10000:
            # perturb best to new start
            h = best_h + rng.normal(scale=0.02, size=n)
            np.clip(h, 0.0, 1.0, out=h)
            enforce_sum(h, rng)
            last_improve = it
    return best_h, best_c

import time, numpy as np, scipy.optimize

Define compute_c5(h): corr = np.correlate(h, 1 - h, mode='full'); return np.max(corr) * (2.0 / h.size)

Define enforce_sum(h, rng): n = h.size; target = n/2.0; diff = target - h.sum(); while abs(diff) > 1e-12: i = rng.integers(0, n); add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i]); h[i] += add; diff = target - h.sum(); return h

Define sum_eq_constraint(x): return np.sum(x) - CURRENT_TARGET

Define softmax_obj(x): corr = np.correlate(x, 1-x, mode='full'); bc = CURRENT_BETA * corr; m = np.max(bc); return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

import time
import numpy as np
from scipy.optimize import minimize

# Global variables used by the soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        # add as much as possible without breaking bounds
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # quick redistribution
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb(h0: np.ndarray, rng: np.random.Generator, time_budget: float) -> tuple[np.ndarray, float]:
    """Guided hill‑climb with occasional random transfers."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    last_improve = 0
    n = h.size
    while time.time() - start < time_budget:
        grad = worst_gradient(h)
        # scaling slowly decays, never below a small floor
        scale = max(0.5 * (0.9 ** (it // 2000)), 0.004)
        if rng.random() < 0.85:
            nh = guided_perturb(h, grad, rng, scale=scale)
        else:
            i, j = rng.choice(n, 2, replace=False)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * 0.5
                nh = h.copy()
                nh[i] -= eps
                nh[j] += eps
            else:
                nh = h
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
            last_improve = it
            if best_c <= 0.3808:
                break
        else:
            # occasional acceptance to escape flat plateaus
            if rng.random() < 0.001:
                h = nh
        it += 1
        if it - last_improve > 12000:
            # restart from best with a small Gaussian bump
            h = best_h + rng.normal(scale=0.02, size=n)
            np.clip(h, 0.0, 1.0, out=h)
            h = enforce_sum(h, rng)
            last_improve = it
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        np.clip(h, 0.0, 1.0, out=h)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h
