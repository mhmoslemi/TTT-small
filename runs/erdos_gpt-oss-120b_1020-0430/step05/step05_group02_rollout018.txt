def smoothed_max_and_grad(h, beta):
    corr = np.correlate(h, 1-h, mode='full')  # length 2n-1
    maxc = np.max(corr)
    # compute log-sum-exp
    exps = np.exp(beta * (corr - maxc))
    sum_exps = exps.sum()
    f = (maxc + np.log(sum_exps) / beta)  # this is value of smooth max
    w = exps / sum_exps
    # compute convolution for gradient
    ones_minus_h = 1 - h
    conv1 = np.convolve(w, ones_minus_h, mode='full')
    conv2 = np.convolve(w, h, mode='full')
    offset = len(h) - 1
    grad = conv1[offset: offset + len(h)] - conv2[offset: offset + len(h)]
    return f, grad

def project_sum_bounds(x, target, tol=1e-12):
    # x is np.array; modifies in place
    # clip to [0,1]
    np.clip(x, 0.0, 1.0, out=x)
    diff = target - x.sum()
    iteration = 0
    while abs(diff) > tol and iteration < 1000:
        if diff > 0:
            # can increase those < 1
            mask = x < 1 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        cnt = np.count_nonzero(mask)
        add = diff / cnt
        # limit add to not exceed bounds
        if diff > 0:
            np.minimum(x + add, 1.0, out=x)
        else:
            np.maximum(x + add, 0.0, out=x)
        diff = target - x.sum()
        iteration += 1
    return x

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_sum_bounds(x, target, tol=1e-12, max_iter=1000):
    """Project x onto {0<=x<=1, sum(x)=target}."""
    np.clip(x, 0.0, 1.0, out=x)
    diff = target - x.sum()
    it = 0
    while abs(diff) > tol and it < max_iter:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        add = diff / cnt
        if diff > 0:
            np.minimum(x + add, 1.0, out=x)
        else:
            np.maximum(x + add, 0.0, out=x)
        diff = target - x.sum()
        it += 1
    return x

# ---------- smoothed max and its gradient ----------
def smooth_max_and_grad(h, beta):
    """Return smooth max value (without dx) and gradient w.r.t h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")               # length 2n-1
    m = np.max(corr)
    exp_terms = np.exp(beta * (corr - m))
    w = exp_terms / exp_terms.sum()                           # weights, sum to 1
    # gradient: sum_s w_s * ((1-h[i+s]) - h[i-s])
    conv1 = np.convolve(w, 1.0 - h, mode="full")
    conv2 = np.convolve(w, h, mode="full")
    off = n - 1
    grad = conv1[off:off + n] - conv2[off:off + n]
    smooth_val = (m + np.log(exp_terms.sum()) / beta)
    return smooth_val, grad

# ---------- worst‑shift projected gradient ----------
def gradient_refine(h, rng, step=0.02, max_iter=6):
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        project_sum_bounds(h, target, max_iter=200)
    return h

    # random fractional
    rnd = rng.random(n)
    rnd *= target / rnd.sum()
    project_sum_bounds(rnd, target)
    starts.append(rnd)
    # random binary
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append(bin_rand)
    # alternating binary
    alt = np.tile([1.0, 0.0], n // 2)
    if n % 2:
        alt = np.append(alt, 1.0)  # keep sum ≈ target when odd, will be projected
    starts.append(alt)
    # constant 0.5 (already satisfies sum)
    const = np.full(n, 0.5)
    starts.append(const)
    return starts

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + 0.99 * budget_s

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate resolutions (smaller first for speed, larger later)
    candidate_ns = [360, 240, 180, 300]

    for n in candidate_ns:
        if time.time() >= deadline:
            break
        starts = generate_starts(n, parent, rng)
        for h0 in starts:
            if time.time() >= deadline:
                break
            # ensure feasibility before passing to optimizer
            h0 = np.clip(h0, 0.0, 1.0)
            project_sum_bounds(h0, n / 2.0)
            best_h, best_c = optimize_h(h0, n, rng, deadline, best_c, best_h)
            best_n = n if best_h is not None else best_n

    # optional up‑sampling for extra refinement
    if (best_h is not None) and (time.time() < deadline) and (best_n * 2 <= 720):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        project_sum_bounds(h_up, n_up / 2.0)
        best_h, best_c = optimize_h(h_up, n_up, rng, deadline, best_c, best_h)
        best_n = n_up

    # final safety projection and evaluation
    if best_h is None:
        # fall back to trivial construction
        best_n = 180
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        project_sum_bounds(best_h, best_n / 2.0)

    best_h = np.clip(best_h, 0.0, 1.0)
    project_sum_bounds(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
