def gradient_refine_half(h_half, rng, step=0.02, max_iter=20):
    n = len(h_half) * 2
    for it in range(max_iter):
        full = np.concatenate([h_half, 1 - h_half])
        corr = np.correlate(full, 1 - full, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0: # ignore
            continue
        # compute gradient g for full h
        # g_i = -(h_{i+k} + h_{i-k})
        # We only need to compute updated half after update.
        # Compute shifted arrays
        h = full
        h_plus = np.take(h, (np.arange(n) + k) % n)
        h_minus = np.take(h, (np.arange(n) - k) % n)
        g_full = -(h_plus + h_minus)
        # step size decays
        step_sz = step * (0.5 ** it)
        h_new = h - step_sz * g_full
        # clip
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce symmetry: average half
        h_half = (h_new[:len(h_half)] + (1 - h_new[len(h_half):])) / 2.0
    return h_half

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97  #Reserve margin
    best_c = np.inf
    best_h_full = None
    best_n = None

    n = 360
    M = n // 2

    # Seed constructors
    seeds = []

    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            # if size mismatch, maybe take first n entries or adjust
            parent = parent[:n]
        # compute half by averaging with symmetric complement
        h_half_parent = (parent[:M] + (1 - parent[M:])) / 2.0
        seeds.append(h_half_parent)
        # binary version of h_half_parent
        idx = np.argpartition(h_half_parent, -M//2)[-M//2:]
        h_half_bin = np.zeros(M)
        h_half_bin[idx] = 1.0
        seeds.append(h_half_bin)
    # Random uniform half
    for _ in range(2):
        h = rng.random(M)
        seeds.append(h)
    # Random beta distribution
    for _ in range(2):
        h = rng.beta(0.5, 0.5, size=M)
        seeds.append(h)

    betas = [5., 20., 80., 200., 500., 1000., 2000., 5000.]
    max_iter_base = 300

    # Helper: compute c5
    def compute_c5_full(full):
        corr = np.correlate(full, 1 - full, mode='full')
        max_corr = np.max(corr)
        return float(max_corr * (2.0 / full.size))

    # Optimization for each seed
    for h_half in seeds:
        if time.time() > deadline:
            break
        # Ensure values in [0,1]
        h_half = np.clip(h_half, 0.0, 1.0)
        # SLSQP cascade using L-BFGS-B for speed
        for beta in betas:
            if time.time() > deadline:
                break
            # Define objective
            def obj(x, beta=beta):
                full = np.concatenate([x, 1 - x])
                corr = np.correlate(full, 1 - full, mode='full')
                # Using logsumexp for smooth max
                return logsumexp(beta * corr) / beta
            # Optimize with L-BFGS-B
            max_iter = int(max_iter_base * (beta / 5.0))
            # limit max_iter to reasonable cap
            if max_iter > 2000:
                max_iter = 2000
            try:
                res = minimize(obj, h_half, method='L-BFGS-B', bounds=[(0.0, 1.0)]*M,
                               options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
                h_half = np.clip(res.x, 0.0, 1.0)
            except Exception:
                # on failure, continue with current h_half
                pass
        # Evaluate after smooth optimization
        full = np.concatenate([h_half, 1 - h_half])
        c = compute_c5_full(full)
        if c < best_c:
            best_c, best_h_full, best_n = c, full.copy(), n

        # Gradient refinement
        h_half = gradient_refine_half(h_half, rng, step=0.04, max_iter=15)
        full = np.concatenate([h_half, 1 - h_half])
        c = compute_c5_full(full)

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5_full(full):
    corr = np.correlate(full, 1 - full, mode='full')
    return float(np.max(corr) * (2.0 / full.size))

def gradient_refine_half(h_half, rng, step=0.02, max_iter=20):
    M = h_half.size
    for it in range(max_iter):
        n = M * 2
        full = np.concatenate([h_half, 1 - h_half])
        corr = np.correlate(full, 1 - full, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            continue
        # compute shifted versions
        ar = np.arange(n)
        h = full
        h_plus = h[(ar + k) % n]
        h_minus = h[(ar - k) % n]
        g_full = -(h_plus + h_minus)
        step_sz = step * (0.5 ** it)
        h_new = h - step_sz * g_full
        h_new = np.clip(h_new, 0.0, 1.0)
        # restore symmetry by averaging halves
        h_half = (h_new[:M] + (1 - h_new[M:])) / 2.0
    return h_half

def binary_targeted_improve_half(h_half, rng, deadline):
    M = h_half.size
    n = M * 2
    while time.time() < deadline:
        full = np.concatenate([h_half, 1 - h_half])
        corr = np.correlate(full, 1 - full, mode='full')
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        k = abs(shift)
        if k == 0:
            break
        ar = np.arange(n)
        h = full
        neigh = h[(ar + k) % n] + h[(ar - k) % n]
        neigh_half = neigh[:M]
        ones = np.flatnonzero(h_half == 1)
        zeros = np.flatnonzero(h_half == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        neigh_one = neigh_half[ones]
        min_val = neigh_one.min()
        cand_i = ones[neigh_one == min_val]
        i = rng.choice(cand_i)
        neigh_zero = neigh_half[zeros]
        max_val = neigh_zero.max()
        cand_j = zeros[neigh_zero == max_val]
        j = rng.choice(cand_j)
        if neigh_half[i] - neigh_half[j] < 0:
            h_half[i] = 0.0
            h_half[j] = 1.0
            continue
        # try random swaps
        improved = False
        for _ in range(5):
            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            if neigh_half[i2] - neigh_half[j2] < 0:
                h_half[i2] = 0.0
                h_half[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h_half

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    n = 360
    M = n // 2

    best_c = np.inf
    best_h_full = None
    best_n = None

    # Prepare seeds for h_half
    seeds = []
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            parent = parent[:n]
        # symmetrized half
        h_half_parent = (parent[:M] + (1 - parent[M:])) / 2.0
        seeds.append(h_half_parent)
        # binary half from parent
        idx = np.argpartition(h_half_parent, -M//2)[-M//2:]
        h_half_bin = np.zeros(M)
        h_half_bin[idx] = 1.0
        seeds.append(h_half_bin)

    # random uniform halves
    for _ in range(2):
        seeds.append(rng.random(M))
    # random beta halves
    for _ in range(2):
        seeds.append(rng.beta(0.5, 0.5, size=M))

    betas = [5., 20., 80., 200., 500., 1000., 2000., 5000.]
    max_iter_base = 250

    for h_half in seeds:
        if time.time() > deadline:
            break
        h_half = np.clip(h_half, 0.0, 1.0)

        # Smooth max cascade using L-BFGS-B
        for beta in betas:
            if time.time() > deadline:
                break

            def obj(x, beta=beta):
                full = np.concatenate([x, 1 - x])
                corr = np.correlate(full, 1 - full, mode='full')
                return logsumexp(beta * corr) / beta

            max_iter = int(max_iter_base * (beta / 5.0))
            if max_iter > 2000:
                max_iter = 2000

                               bounds=[(0.0, 1.0)] * M,

                h_half = np.clip(res.x, 0.0, 1.0)
            except Exception:
                pass

        full = np.concatenate([h_half, 1 - h_half])

    if best_h_full is not None and time.time() < deadline and best_n * 2 <= 720:

        M_up = M_cur * 2
        x_old = np.linspace(0, 1, M_cur, endpoint=False)
        x_new = np.linspace(0, 1, M_up, endpoint=False)
        h_half_up = np.interp(x_new, x_old, h_half)
        h_half_up = np.clip(h_half_up, 0.0, 1.0)

        n_up = M_up * 2
        for beta in betas:
            if time.time() > deadline:
                break

            def obj_up(x, beta=beta):
                full = np.concatenate([x, 1 - x])
                corr = np.convolve(full, 1 - full, mode='full')
                # Using np.convolve is equivalent to correlate for real sequences
                return logsumexp(beta * corr) / beta

            max_iter = int(max_iter_base * (beta / 5.0))
            if max_iter > 2000:
                max_iter = 2000

            try:
                res = minimize(obj_up, h_half_up, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * M_up,
                               options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
                h_half_up = np.clip(res.x, 0.0, 1.0)
            except Exception:
                pass

        h_half_up = gradient_refine_half(h_half_up, rng, step=0.04, max_iter=10)
        full_up = np.concatenate([h_half_up, 1 - h_half_up])
        c_up = compute_c5_full(full_up)
        if c_up < best_c:
            best_c, best_h_full, best_n = c_up, full_up.copy(), n_up

        # Binary improvement for upsampled
        idx = np.argpartition(h_half_up, -M_up//2)[-M_up//2:]
        h_half_up_bin = np.zeros(M_up)
        h_half_up_bin[idx] = 1.0
        h_half_up_bin = binary_targeted_improve_half(h_half_up_bin, rng, deadline)
        full_up = np.concatenate([h_half_up_bin, 1 - h_half_up_bin])
        c_up = compute_c5_full(full_up)
        if c_up < best_c:
            best_c, best_h_full, best_n = c_up, full_up.copy(), n_up

    # Fallback if nothing found
    if best_h_full is None:
        best_n = 360
        half = np.zeros(best_n // 2)
        half[:best_n // 4] = 1.0
        best_h_full = np.concatenate([half, 1 - half])

    best_h_full = np.clip(best_h_full, 0.0, 1.0)
