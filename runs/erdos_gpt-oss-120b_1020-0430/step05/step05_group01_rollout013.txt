def slsqp_optimize(h0, beta, rng, max_iter=300):
    n = h0.size
    target = n/2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # compute log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def constr(x):
        return np.sum(x) - target
    bounds = [(0.0,1.0)]*n
    cons = {'type':'eq','fun':constr}
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0.0,1.0)
    # enforce sum if needed
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # shift using projection
        x = project_sum_one(x, target)
    return x

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def project_sum_one(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto [0,1] with sum = target via bisection on λ."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        if proj.sum() > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + lo, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 0.0) & (proj < 1.0)
        if mask.any():
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def enforce_sum(v: np.ndarray, target: float) -> np.ndarray:
    return project_sum_one(v, target)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def symmetrize(h: np.ndarray) -> np.ndarray:
    return (h + h[::-1]) * 0.5

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * c
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def cons(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints={"type": "eq", "fun": cons},
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    if abs(x.sum() - target) > 1e-12:
        x = enforce_sum(x, target)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, n / 2.0)
    else:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    h = symmetrize(h)                     # enforce symmetry early
    curr_c = compute_c5(h)
    best_h, best_c = h.copy(), curr_c

    # ----- short SLSQP refinements ---------------------------------------
    for beta in (100.0, 1000.0, 5000.0):
        if time.time() - start > 0.15 * budget_s:
            break
        h = slsqp_optimize(h, beta, rng, max_iter=300)
        h = symmetrize(h)
        curr_c = compute_c5(h)
        if curr_c < best_c:
            best_h, best_c = h.copy(), curr_c

    # ----- main stochastic hill climbing ---------------------------------
    init_scale, final_scale = 0.2, 1e-5
    max_iters = 300000
    for it in range(max_iters):
        if time.time() - start > 0.95 * budget_s:
            break
        frac = (time.time() - start) / budget_s
        scale = init_scale * (final_scale / init_scale) ** frac

        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale)
        nh = symmetrize(nh)
        c_nh = compute_c5(nh)
        if c_nh < best_c:
            best_h, best_c = nh.copy(), c_nh
        if c_nh < curr_c:
            h, curr_c = nh, c_nh

        # occasional random transfer
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 0.0:
                    eps = max_eps * rng.random() * scale * 5.0
                    nh2 = h.copy()
                    nh2[i] -= eps
                    nh2[j] += eps
                    nh2 = symmetrize(nh2)
                    c2 = compute_c5(nh2)
                    if c2 < best_c:
                        best_h, best_c = nh2.copy(), c2
                    if c2 < curr_c:
                        h, curr_c = nh2, c2

        # periodic SLSQP refinement
        if (it + 1) % 5000 == 0:
            if time.time() - start > 0.90 * budget_s:
                continue
            h = slsqp_optimize(h, 2000.0, rng, max_iter=200)
            h = symmetrize(h)
            curr_c = compute_c5(h)
            if curr_c < best_c:
                best_h, best_c = h.copy(), curr_c

    # ----- final high‑beta SLSQP passes ----------------------------------
    for beta in (5000.0, 15000.0, 50000.0, 150000.0):
        if time.time() - start > 0.99 * budget_s:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)
        best_h = symmetrize(best_h)
        best_c = compute_c5(best_h)

    # ----- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
