def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ----- global placeholders for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5_fft(h: np.ndarray) -> float:
    """Exact C5 using circular autocorrelation via FFT."""
    n = h.size
    H = np.fft.fft(h)
    S = np.real(np.fft.ifft(np.abs(H) ** 2))
    min_S = np.min(S)
    if min_S < 0:
        min_S = 0.0
    return 1.0 - (2.0 / n) * min_S

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass to meet Σh = n/2 exactly (keeps 0≤h≤1)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # smooth max of corr = h·(1−h shifted)
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    """SLSQP on smooth-max surrogate, then exact sum repair."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Projected gradient descent against worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def simulated_annealing_h(h0: np.ndarray, rng: np.random.Generator,
                          deadline: float) -> tuple[np.ndarray, float]:
    """SA on fractional h, preserving sum exactly."""
    h = h0.copy()
    n = h.size
    cur_c5 = compute_c5_fft(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_h, best_c5
    T0 = max(1e-4, cur_c5 * 0.5)
    it = 0
    while True:
        now = time.time()
        if now >= deadline or deadline - now < 5.0:
            break
        elapsed = now - start
        T = max(1e-8, T0 * (1.0 - elapsed / total))
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # decide direction of mass transfer
        if rng.random() < 0.5:
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta * (T / T0)
            new_i, new_j = h[i] + delta, h[j] - delta
        else:
            max_delta = min(1.0 - h[j], h[i])
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta * (T / T0)
            new_i, new_j = h[i] - delta, h[j] + delta
        old_i, old_j = h[i], h[j]
        h[i], h[j] = new_i, new_j
        new_c5 = compute_c5_fft(h)
        dC = new_c5 - cur_c5
        accept = False
        if dC <= 0:
            accept = True
        else:
            prob = np.exp(-dC / T)
            if rng.random() < prob:
                accept = True
        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = old_i, old_j
        it += 1
    return best_h, best_c5

def binary_sa_h(h_bin: np.ndarray, rng: np.random.Generator,
                deadline: float) -> tuple[np.ndarray, float]:
    """SA on binary vectors (swap 0↔1)."""
    n = h_bin.size
    cur_c5 = compute_c5_fft(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_h, best_c5
    T0 = max(1e-4, cur_c5 * 0.5)
    while True:
        now = time.time()
        if now >= deadline or deadline - now < 2.0:
            break
        elapsed = now - start
        T = max(1e-8, T0 * (1.0 - elapsed / total))
        # pick a random 1 and a random 0
        i = rng.integers(0, n)
        while h_bin[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h_bin[j] == 1:
            j = rng.integers(0, n)
        # swap
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5_fft(h_bin)
        dC = new_c5 - cur_c5
        accept = False
        if dC <= 0:
            accept = True
        else:
            prob = np.exp(-dC / T)
            if rng.random() < prob:
                accept = True
        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + 0.99 * budget_s

    # ----- main parameter -----
    n = 720  # target resolution

    # ----- initialise h -----
