def project_onto_simplex_capped(v, target):
    # v is array (float), we want x = min(1, max(0, v - lambda)) such that sum(x) = target.
    # lambda can be found via binary search.
    lo = -1.0  # lambda low ensures v - lambda > 1 => truncated to 1 => sum maybe > target.
    hi = 1.0   # lambda high ensures v - lambda < 0 => truncated to 0 => sum maybe < target.
    # Actually we can start lo = -np.max(v) - 1; hi = np.max(v) maybe.
    # We'll just set lo = -2, hi = 2 for safety.
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2, 0, 1)
    # after binary search, adjust small rounding error:
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # adjust by distributing diff among non-bound entries
        non_bounds = (x > 0) & (x < 1)
        m = non_bounds.sum()
        if m > 0:
            x[non_bounds] += diff / m
            # clip again
            x = np.clip(x, 0, 1)
    return x

def gradient_descent(h, target, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step_factor = 1.0
    iteration = 0
    n = h.size
    while time.time() < deadline:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        # update best
        if max_c < best_c - 1e-12:
            best_c = max_c
            best_h = h.copy()
        # compute lag
        lag = max_idx - (n - 1)
        # compute gradient
        b = 1 - h
        g = np.zeros_like(h)
        if lag >= 0:
            l = lag
            g[:n-l] = b[l:]
            g[l:] -= h[:n-l]
        else:
            l = -lag
            g[l:] = b[:n-l]
            g[:n-l] -= h[l:]
        max_abs_g = np.max(np.abs(g))
        if max_abs_g < 1e-12:
            break
        # step size
        alpha = 0.5 / max_abs_g
        alpha *= step_factor
        # update
        h_new = h - alpha * g
        # projection
        h = project_onto_simplex_capped(h_new, target)
        # decay step factor

            h = project_onto_simplex_capped(h, target)
        # early stop if no improvement after many iterations?
        # Not needed; loop will eventually reach deadline.
    return best_h, best_c

Define slsqp_optimize_beta(h0, beta, rng, max_iter=300):

    n = h0.size
    # initial x = h0 (float)
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - n/2}
    # objective
    def softmax_obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        # subtract max for numerics
        m = np.max(c)
        # compute log-sum-exp scaled by beta
        # to avoid overflow, factor beta inside exp
        # c * beta may be large; compute as exp(beta*(c - m)) etc.
        # But using np.exp(beta*(c-m)) may overflow if beta large and (c-m) approx 0. We'll use np.exp(beta*(c - m)) which is safe because argument <=0.
        # compute lse = m + log(sum(exp(beta*(c - m)))) / beta
        # final smooth max = (m + (1/beta)*log(sum(exp(beta*(c - m)))))
        lse = np.log(np.sum(np.exp(beta * (c - m)))) / beta + m

def gradient_descent(h0, target, rng, deadline, max_iters=3000):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step_factor = 1.0
    n = h.size
    iter = 0
    while time.time() < deadline and iter < max_iters:
        # correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c = corr[max_idx] * (2.0 / n)
        if max_c < best_c - 1e-12:
            best_c = max_c
            best_h = h.copy()
        lag = max_idx - (n - 1)
        b = 1 - h
        g = np.zeros_like(h)
        if lag >= 0:
            l = lag
            g[:n-l] = b[l:]
            g[l:] -= h[:n-l]
        else:
            l = -lag
            g[l:] = b[:n-l]
            g[:n-l] -= h[l:]
        max_abs_g = np.max(np.abs(g))
        if max_abs_g < 1e-12:
            break
        alpha = 0.5 / max_abs_g * step_factor
        h_new = h - alpha * g
        h = project_onto_simplex_capped(h_new, target)
        step_factor *= 0.9995
        iter += 1

            h = np.clip(h + noise, 0, 1)
            h = project_onto_simplex_capped(h, target)
    return best_h, best_c

        m = np.max(c)
        # compute smooth max
        # Use stable exp
        exp_vals = np.exp(beta * (c - m))
        # prevent overflow: if beta large, still okay because c - m <=0.
        return m + np.log(np.sum(exp_vals)) / beta
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_simplex_capped(x, target)

import time

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def project_onto_simplex_capped(v, target):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        nb = (x > 0.0) & (x < 1.0)
        m = nb.sum()
        if m:
            x[nb] += diff / m
            x = np.clip(x, 0.0, 1.0)
    return x

    best_h, best_c = h.copy(), compute_c5(h)
    step_factor = 1.0
    n = h.size
