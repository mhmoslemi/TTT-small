def hill_climb_local(h, deadline):
    n = h.size
    max_O = np.max(np.correlate(h, 1-h, mode='full')) # int
    ones = set(np.flatnonzero(h))
    zeros = set(np.flatnonzero(1-h))
    while time.time() < deadline:
        # compute O array and index of max
        corr = np.correlate(h, 1-h, mode='full')
        max_O = corr.max()
        # get list of k where corr == max_O
        ks = np.flatnonzero(corr == max_O) - (n-1)
        # cannot process all ks; choose a random one
        k = ks[np.random.randint(len(ks))]
        # For this k, find candidate i (ones that contribute)
        # contributions if i <= n-1-k and h[i]==1 and (1 - h[i+k]) ==1 => h[i+k]==0
        # So i where h[i]==1 and (i+k >= n or h[i+k]==0). Since i+k>=n => h[i+k] out-of-bound => treat as 0.
        # Compute list
        i_candidates = []
        for i in ones:
            if i + k < n:
                if h[i+k] == 0:
                    i_candidates.append(i)
            else:
                # out-of-bounds case always contributes
                i_candidates.append(i)
        # Similarly j candidates (zeros) where swapping them might reduce O
        # For zero j, after swap to 1, j will contribute to O(k) if (j+k < n and h[j+k]==0) or j+k >= n
        j_candidates = []
        for j in zeros:
            # before swap, zero does not contribute
            # after swap to 1, it would contribute if h[j+k]==0 or out-of-bound
            if j + k < n:
                if h[j+k] == 0:
                    j_candidates.append(j)
            else:
                j_candidates.append(j)
        if not i_candidates or not j_candidates:
            # fallback random swap
            i = np.random.choice(list(ones))
            j = np.random.choice(list(zeros))
        else:
            i = i_candidates[np.random.randint(len(i_candidates))]
            j = j_candidates[np.random.randint(len(j_candidates))]
        # perform swap
        h[i] = 0; h[j] = 1
        # recompute new max
        new_corr = np.correlate(h, 1-h, mode='full')
        new_max = new_corr.max()
        if new_max < max_O:
            # accept swap and update sets
            ones.remove(i); ones.add(j)
            zeros.remove(j); zeros.add(i)
            max_O = new_max
        else:
            # revert
            h[i] = 1; h[j] = 0
        # Optionally break loop if no improvement found after many attempts
    return h, max_O

    n = h.size
    max_O = np.max(np.correlate(h, 1-h, mode='full'))
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1-h)
    stagnation = 0
    while time.time() < deadline and stagnation < 10000:
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        # swap
        h[i] = 0; h[j] = 1
        new_corr = np.correlate(h, 1-h, mode='full')
        new_max = new_corr.max()
        if new_max < max_O:
            max_O = new_max
            # update lists
            ones[ones == i] = j
            zeros[zeros == j] = i
            stagnation = 0
        else:
            # revert
            h[i] = 1; h[j] = 0
            stagnation += 1
    return h, max_O

import time
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact sum correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def smooth_max_step(x, beta, step, rng):
    n = x.shape[0]
    # current O(k)
    O = np.correlate(x, 1.0 - x, mode="full")
    m = O.max()
    # stable softmax weights
    exp_vals = np.exp(beta * (O - m))
    w = exp_vals / exp_vals.sum()
    term1 = np.full(n, 1.0)
    term2 = np.zeros(n)

    # k >= 0
    for k in range(n):
        wk = w[k + n - 1]
        if wk == 0.0:
            continue
        # subtract wk * x[i+k] where i+k < n  -> i < n-k
        if k < n:
            term1[: n - k] -= wk * x[k:]
        # add wk * x[i-k] where i >= k
        if k == 0:
            term2 += wk * x
        else:
            term2[k:] += wk * x[: n - k]

    # k < 0
    for k in range(-n + 1, 0):
        wk = w[k + n - 1]
        if wk == 0.0:
            continue
        off = -k
        # subtract wk * x[i-off] for i >= off
        term1[off:] -= wk * x[: n - off]
        # add wk * x[i+off] for i < n-off
        term2[: n - off] += wk * x[off:]

    grad = term1 - term2
    x_new = x - step * grad
    x_new = np.clip(x_new, 0.0, 1.0)
    adjust_sum_exact(x_new, rng)
    return x_new

def smooth_max_optimize(x0, beta_schedule, step_base, max_iter_beta, deadline, rng):
    x = x0.copy()
    best_x = x.copy()
    best_c = compute_c5(x)
    for beta in beta_schedule:
        step = step_base / np.sqrt(beta + 1e-12)
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                return best_x, best_c
            x = smooth_max_step(x, beta, step, rng)
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_x = x.copy()
    return best_x, best_c

def greedy_binary(n, rng):
    """Greedy construction of a binary sequence."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0
    return h

def hill_climb_random(h_start, deadline, rng):
    """Simple random swap hill‑climb on binary vectors."""
    h = h_start.copy()
    n = h.shape[0]
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    max_O = np.max(np.correlate(h, 1 - h, mode="full"))
    stagnation = 0
    while time.time() < deadline and stagnation < 15000:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_O = np.max(np.correlate(h, 1 - h, mode="full"))
        if new_O < max_O:
            max_O = new_O
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            stagnation = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            stagnation += 1
    return h, max_O

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # Determine n from parent if available
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
        best_h = parent.copy()
        best_c = compute_c5(best_h)
    else:
        n = 360
        best_h = None
        best_c = np.inf

    # -------------------------------------------------
    # 1. Greedy binary + local hill climbing (multiple restarts)
    # -------------------------------------------------
    while time.time() < deadline - 80:  # reserve time for refinement
        h_bin = greedy_binary(n, rng)
        sub_dead = time.time() + 2.5
        h_hc, max_O = hill_climb_random(h_bin, sub_dead, rng)
        c_val = max_O * (2.0 / n)
        if c_val < best_c:
            best_c = c_val
            best_h = h_hc.astype(float)
        # occasional random binary restart
        if rng.random() < 0.3:
            h_rand = np.zeros(n, dtype=float)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_rand[idx] = 1.0
            sub_dead = time.time() + 2.5
            h_hc, max_O = hill_climb_random(h_rand, sub_dead, rng)
            c_val = max_O * (2.0 / n)
            if c_val < best_c:
                best_c = c_val
                best_h = h_hc.astype(float)

    # -------------------------------------------------
    # 2. Smooth‑max projected gradient descent (multiple beta stages)
    # -------------------------------------------------
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)

    beta_sched = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
                  2000.0, 5000.0, 10000.0, 20000.0]
    step_base = 0.3
    max_iter_beta = 400

    best_h, best_c = smooth_max_optimize(
        best_h,
        beta_sched,
        step_base,
        max_iter_beta,
        deadline,
        rng,
    )

    # -------------------------------------------------
    # 3. Optional extra runs from random starts if time permits
    # -------------------------------------------------
    while time.time() < deadline:
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h0[idx] = 1.0
        adjust_sum_exact(h0, rng)
        h_opt, c_opt = smooth_max_optimize(
            h0,
            beta_sched,
            step_base,
            max_iter_beta,
            deadline,
            rng,
        )
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
