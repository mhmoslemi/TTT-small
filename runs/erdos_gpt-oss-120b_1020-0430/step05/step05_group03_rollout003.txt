import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {0≤x≤1, sum x = target} by binary search on λ."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_to_sum_box(v, n / 2.0)


# ---------- smooth‑min (max‑overlap) gradient ----------
def gradient_of_smooth_min_w(h: np.ndarray, beta: float) -> np.ndarray:
    """∇ of -1/β·log∑exp(-β·w_k) where w_k = Σ h_i·h_{i+k} (circular)."""
    n = h.size
    H = np.fft.rfft(h)
    w = np.fft.irfft(H * np.conj(H), n=n)
    max_w = np.max(w)
    shift = w - max_w
    weights = np.exp(-beta * shift)          # stability by shifting
    W = np.fft.rfft(weights)
    conv1 = np.fft.irfft(W * H, n=n)
    rev = np.empty_like(weights)
    rev[0] = weights[0]
    rev[1:] = weights[:0:-1]
    W_rev = np.fft.rfft(rev)
    conv2 = np.fft.irfft(W_rev * H, n=n)
    return conv1 + conv2


def smooth_min_w_val(h: np.ndarray, beta: float) -> float:
    """Smooth approximation of min_k w_k."""
    n = h.size
    H = np.fft.rfft(h)
    w = np.fft.irfft(H * np.conj(H), n=n)
    max_w = np.max(w)
    shift = w - max_w
    log_sum = np.log(np.sum(np.exp(-beta * shift)))
    return max_w - log_sum / beta


# ---------- worst‑shift utilities ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    import time
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 512
    target = n / 2.0

    # start from parent if present
    if "initial_h_values" in globals():
        h = np.clip(np.array(initial_h_values, dtype=float), 0.0, 1.0)
        h = project_to_sum_box(h, target)
    else:
        h = random_feasible_vector(n, rng)

    best_h, best_c = h.copy(), compute_c5(h)

    beta_schedule = [5., 20., 80., 200., 800., 2000., 8000., 20000.]

    for beta in beta_schedule:
        if time.time() - start > budget_s * 0.80:
            break
        remaining = budget_s - (time.time() - start)
        max_iters = min(2500, int(remaining * 5))
        step = 0.5 / np.sqrt(beta)
        for it in range(max_iters):
            grad = gradient_of_smooth_min_w(h, beta)
            h_new = project_to_sum_box(h + step * grad, target)
            # accept if smooth‑min improves
            if smooth_min_w_val(h_new, beta) >= smooth_min_w_val(h, beta):
                h = h_new
            if (it & 31) == 0:
                c_val = compute_c5(h)
                if c_val < best_c:
                    best_c, best_h = c_val, h.copy()
            if (it + 1) % 200 == 0:
                step *= 0.9
            if time.time() - start > budget_s * 0.85:
                break

    # ----- local refinement with hill‑climb -----
    remaining = budget_s - (time.time() - start)
    if remaining > 4.0:
        hc_iters = int(min(5000, remaining * 5))
        h_hc, c_hc = guided_hill_climb(best_h, rng,
                                        max_iters=hc_iters, step_scale=0.6)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

    # ----- final stochastic tweaks -----
    while time.time() - start < budget_s * 0.995:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_c, best_h = nc, nh.copy()

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum_box(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
