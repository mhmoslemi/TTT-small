def combined_gradient(h, K=3):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    # compute shift ks for top K
    # note corr length = 2*n-1; index shift = idx - (n-1); take absolute
    idxs = np.argpartition(-corr, K)[:K]  # get K largest values
    grads = np.zeros_like(h)
    for idx in idxs:
        k = abs(idx - (n - 1))
        if k == 0:
            grads += (1.0 - 2.0 * h)
        else:
            # add gradient for this k
            tmp = np.ones_like(h)
            tmp[: n - k] -= h[k:]
            tmp[k:]       -= h[: n - k]
            grads += tmp
    return grads / K
```

def softmax_corr(x, beta):
    corr = np.correlate(x, 1-x, mode='full')
    # compute log-sum-exp and gradient
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    f = (max_bc + np.log(sum_exp)) / beta
    # compute weights
    w = exp_bc / sum_exp
    # compute gradient of each corr_i
    # We can compute accumulate: 
    n = x.size
    grad = np.zeros_like(x)
    for idx, weight in enumerate(w):
        k = idx - (n - 1)
        kk = abs(k)
        if kk == 0:
            grad += weight * (1 - 2*x)
        else:
            # shift contributions: for i in [0,n-kk-1] grad[i] -= weight * x[i+kk]
            # and for i in [kk,n-1] grad[i] -= weight * x[i-kk]
            grad[: n - kk] -= weight * x[kk:]
            grad[kk:]      -= weight * x[: n - kk]
    return f, grad

```
def softmax_obj_and_grad(x, beta):
    n = x.size
    corr = np.correlate(x, 1-x, mode='full')
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sumexp = np.exp(max_bc) * np.sum(exp_bc) # Actually bc - max_bc then exp: sumexp = sum(exp(bc - max_bc))
    f = (max_bc + np.log(sumexp)) / beta
    w = exp_bc / sumexp
    grad = np.zeros_like(x)
    # iterate over each shift
    # discount computational cost: we can use for loop over idx, weight.
    for idx, weight in enumerate(w):
        k = idx - (n - 1)
        kk = abs(k)
        if kk == 0:
            grad += weight * (1 - 2*x)
        else:
            grad[: n - kk] -= weight * x[kk:]
            grad[kk:]      -= weight * x[: n - kk]
    return f, grad
```

But note sumexp is sum(exp_bc), not multiplied by exp(max_bc) because we subtracted max_bc before exponent, so sumexp = sum(exp_bc). In f, we need (max_bc + log(sumexp))/beta. That's correct.

def slsqp_optimize_with_grad(h0, beta, max_iter=400):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0,1.0)]*n
    cons = {"type":"eq","fun":lambda x: np.sum(x)-target}
    def fun(x):
        f,_ = softmax_obj_and_grad(x, beta)
        return f
    def jac(x):
        _,g = softmax_obj_and_grad(x, beta)
        return g
    res = minimize(fun, h0, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=cons, options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0,1)
    return enforce_sum(x, np.random.default_rng())

Better implement a more efficient function: Let's implement enforce_sum which modifies h in place to satisfy sum constraint while respecting bounds, using a simple iterative approach:

- Compute diff = target - sum(h). While |diff| > tolerance:
   - Determine indices where we can increase (h < 1) if diff>0, or decrease (h>0) if diff<0.
   - If diff>0: compute total capacity = sum(1 - h[i] for i in inc_idx). If total capacity <=0, break.
     Then add delta = diff * (1 - h[i]) / total_capacity to each inc_idx (proportionally). But that may overshoot for some if delta > capacity; but we can clip.

Simplify: Use scaling method: after sum diff, we adjust all entries uniformly while clipping: e.g., set h = h + diff/n; then clip to [0,1]; if after clipping sum still not target, repeat.

```
def enforce_sum(h, rng, tol=1e-12, max_iter=1000):
    n = h.size
    target = n/2.0
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) <= tol:
            break
        # distribute diff uniformly
        h += diff / n
        # clip
        np.clip(h, 0.0, 1.0, out=h)
    # final correction if needed
    diff = target - h.sum()
    # if still small, do random redistribution as fallback
    while abs(diff) > tol:
        i = rng.integers(0,n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
```

import time, numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator,
                tol: float = 1e-12, max_iter: int = 1000) -> np.ndarray:
    """Adjust h in-place so that Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) <= tol:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    while abs(diff) > tol:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ = n/2."""
    h = rng.random(n)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def hill_climb_line_search(h: np.ndarray, rng: np.random.Generator,
                           max_iters: int = 2000, step_scale: float = 0.6):
    """Gradient‑guided hill‑climb with a few line‑search candidates."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    fractions = np.array([0.03, 0.1, 0.25, 0.5, 0.75, 1.0])
    for it in range(max_iters):
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps <= 0:
                continue
        eps_base = max_eps * step_scale
        improved = False
        for f in fractions:
            eps = eps_base * f
            if eps <= 0:
                continue
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            c = compute_c5(nh)
            if c < best_c:
                best_c, best_h = c, nh
                improved = True
                break
        if not improved:
            # no progress – shrink step_scale a bit
            step_scale = max(step_scale * 0.9, 1e-5)
        if (it + 1) % 200 == 0:
            step_scale = max(step_scale * 0.8, 1e-5)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        grad = worst_gradient(h)
        h = h - (step * (0.5 ** it)) * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 600) -> np.ndarray:
    """SLSQP refinement using a smooth approximation of the max‑overlap."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}

    def softmax(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(softmax, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, np.random.default_rng())


def cosine_pattern(n: int, p: int) -> np.ndarray:
    """Periodic cosine pattern of period p, automatically in [0,1] and mean 0.5."""
    i = np.arange(n)
    h = 0.5 + 0.5 * np.cos(2.0 * np.pi * i / p)
    return np.clip(h, 0.0, 1.0)


# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    def time_left():
        return total - (time.time() - start)

    # ---- start from parent if available ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0, compute_c5(h0), h0.size
    else:
        best_n = 128
        best_h = random_feasible_vector(best_n, rng)
        best_c = compute_c5(best_h)

    # ---- Phase 1: search on small/medium sizes ----
    small_ns = [64, 96, 128, 192]
    period_choices = [8, 12, 16, 20, 24, 32, 40]
    for n in small_ns:
        if time_left() < 0.07 * total:
            break
        for typ in ("random", "binary", "cosine"):
            if time_left() < 0.025 * total:
                break
            if typ == "random":
                h = random_feasible_vector(n, rng)
            elif typ == "binary":
                h = np.zeros(n)
                h[rng.choice(n, n // 2, replace=False)] = 1.0
            else:  # cosine
                p = rng.choice(period_choices)
                h = cosine_pattern(n, p)
                h = enforce_sum(h, rng)
            # hill‑climb with line search
            h, c = hill_climb_line_search(h, rng, max_iters=800, step_scale=0.7)
            # SLSQP passes (soft‑max smoothing)
            for beta in (10.0, 30.0, 80.0, 200.0, 800.0):
                if time_left() < 0.015 * total:
                    break
                h = slsqp_optimize(h, beta, max_iter=200)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n
            # short gradient tidy‑up
            h = gradient_refine(h, rng, step=0.015, max_iter=4)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # ---- Phase 2: upscale to full resolution (512) ----
    target_n = 512
    if best_n < target_n and time_left() > 0.2 * total:
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h = np.interp(x_new, x_old, best_h)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)

        h, c = hill_climb_line_search(h, rng, max_iters=2500, step_scale=0.6)

        for beta in (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0, 20000.0):
            if time_left() < 0.015 * total:
                break
            maxit = 800 if beta >= 200 else 500
            h = slsqp_optimize(h, beta, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, target_n

        h = gradient_refine(h, rng, step=0.01, max_iter=10)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, target_n

    # ---- Phase 3: final stochastic annealing tweaks ----
    T = 0.005
    while time_left() > 0.15:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps > 0:
            eps = max_eps * rng.random() * 0.02
            nh = best_h.copy()
            nh[i] -= eps
            nh[j] += eps
            nc = compute_c5(nh)
            if nc < best_c or rng.random() < np.exp(-(nc - best_c) / T):
                best_h, best_c = nh, nc
        else:
            # fallback random pair
            i = rng.integers(0, best_h.size)
            j = rng.integers(0, best_h.size)
            if i != j:
                max_eps = min(best_h[i], 1.0 - best_h[j])
                if max_eps > 0:
                    eps = max_eps * rng.random() * 0.02
                    nh = best_h.copy()
                    nh[i] -= eps
                    nh[j] += eps
                    nc = compute_c5(nh)
                    if nc < best_c or rng.random() < np.exp(-(nc - best_c) / T):
                        best_h, best_c = nh, nc
        T = max(T * 0.9999, 1e-6)

    # ---- final sanity and return ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
