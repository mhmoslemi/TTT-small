import time
import numpy as np
import scipy.optimize as opt

# ---------- basic helpers ----------
def compute_c5(h):
    """C5 for a discrete vector h on [0,2] with dx=2/len(h)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):
    """Redistribute mass to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
    return h

# ---------- binary max‑min hill climbing ----------
def hill_climb_binary_maximin(h0, rng, max_iters, deadline):
    """Improve a binary vector by swaps that raise the worst autocorrelation."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c

    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break

        conv = np.correlate(h, h, mode="full")

    # final safety cleaning
    adjust_sum_exact(best_h, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
