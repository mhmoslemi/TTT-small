def subgradient_descent(h, max_iters, init_alpha, rng):
    n = h.size
    target = n/2
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        O = corr[n-1:]  # length n
        k = int(np.argmax(O))
        O_max = O[k]
        if O_max < best_c:
            best_c = O_max
            best_h = h.copy()
        # gradient
        g = np.ones(n)
        if k > 0:
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
        else:
            g -= 2 * h
        # compute safe alpha
        pos = g > 1e-15
        neg = g < -1e-15
        alpha_max = np.inf
        if np.any(pos):
            # h[pos] / g[pos] safe as g[pos] > 0
            alpha_max = min(alpha_max, np.min(h[pos] / g[pos]))
        if np.any(neg):
            alpha_max = min(alpha_max, np.min((1.0 - h[neg]) / (-g[neg])))
        # schedule
        alpha = min(init_alpha / (it + 1), alpha_max) * 0.9
        if alpha <= 0:
            continue
        h = h - alpha * g
        # optional enforce sum correction
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust random index
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
        # maybe after many iterations break if O_max pretty low
    # final
    final_c = compute_c5(best_h)
    return best_h, final_c

def slsqp_optimize_local(h0, beta, rng, max_iter=300):
    n = h0.size
    target = n/2.0
    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        O = corr[n-1:]  # k>=0
        # multiply by beta for scaling
        bc = beta * O
        m = np.max(bc)
        # stable log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    # constraint
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0,1.0)] * n
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust one entry
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
    return x

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if abs(add) > 0:
            h[i] += add
            diff = target - h.sum()
        else:
            # pick another index if this one cannot move
            continue
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k for the current h."""
    n = h.size
    # correlation for O_k (k ≥ 0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    O = corr[n - 1 :]  # O[0] corresponds to shift 0
    k = int(np.argmax(O))
    g = np.ones(n)
    if k > 0:
        g[: n - k] -= h[k:]
        g[k:] -= h[: n - k]
    else:  # k == 0
        g -= 2.0 * h
    return g, O[k]


def subgradient_descent(
    h0: np.ndarray,
    max_iters: int,
    init_alpha: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max_k O_k."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    # O for the initial point
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c = corr[n - 1 :].max()
    for it in range(max_iters):
        # compute O_k and its worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        O = corr[n - 1 :]  # length n, shifts 0..n-1
        k = int(np.argmax(O))
        O_max = O[k]
        if O_max < best_c:
            best_c = O_max
            best_h = h.copy()
        # gradient for this worst shift
        g = np.ones(n)
        if k > 0:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
        else:
            g -= 2.0 * h
        # safe step size (stay in [0,1])
        pos = g > 1e-15
        neg = g < -1e-15
        alpha_max = np.inf
        if np.any(pos):
            alpha_max = min(alpha_max, np.min(h[pos] / g[pos]))
        if np.any(neg):
            alpha_max = min(alpha_max, np.min((1.0 - h[neg]) / (-g[neg])))
        # schedule
        alpha = min(init_alpha / (it + 1), alpha_max) * 0.9
        if alpha <= 0:
            continue
        # update (guaranteed feasible)
        h = h - alpha * g
        # tiny drift correction for the sum constraint
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
    # final evaluation of the best point found
    final_c = compute_c5(best_h)
    return best_h, final_c


def slsqp_optimize_local(
    h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 300
) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0

    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        O = corr[n - 1 :]  # shifts 0..n‑1
        bc = beta * O
        m = np.max(bc)
        # stable log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final sum repair
    x = enforce_sum(x, target, rng)
    return x


def upsample(h: np.ndarray, new_n: int, rng: np.random.Generator) -> np.ndarray:
    """Linear interpolation to a finer grid, with sum repair."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = new_n / 2.0
    h_new = enforce_sum(h_new, target, rng)
    return h_new


# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial construction ------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        n = h.size
        h = enforce_sum(h, n / 2.0, rng)
    else:
        n = 512
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- sub‑gradient phase (major work) ------------------------------------
    elapsed = time.time() - start
    time_left = budget_s * 0.70 - elapsed
    if time_left > 1.0:
        # rough per‑iteration cost estimate
        per_iter = 0.0003 if n <= 512 else 0.001
        max_iters = int(min(80000, max(0, time_left / per_iter)))
        if max_iters > 0:
            best_h, best_c = subgradient_descent(best_h, max_iters, 0.5, rng)

    # ---- occasional SLSQP refinement ----------------------------------------
    elapsed = time.time() - start
    if time.time() - start < budget_s * 0.90:
        for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0):
            if time.time() - start > budget_s * 0.90:
                break
            best_h = slsqp_optimize_local(best_h, beta, rng, max_iter=400)
            cand_c = compute_c5(best_h)
            if cand_c < best_c:
                best_c = cand_c

    # ---- up‑sample and refine at higher resolution if time permits ---------
    elapsed = time.time() - start
    if time.time() - start < budget_s - 5.0:
        new_n = max(1024, n * 2)
        h_up = upsample(best_h, new_n, rng)
        # sub‑gradient on finer grid
        elapsed = time.time() - start
        time_left = budget_s * 0.95 - elapsed
        if time_left > 5.0:
            per_iter_up = 0.001
            max_iters_up = int(min(20000, max(0, time_left / per_iter_up)))
            if max_iters_up > 0:
                h_up, c_up = subgradient_descent(h_up, max_iters_up, 0.5, rng)
                if c_up < best_c:
                    best_h, best_c = h_up, c_up
        # final SLSQP passes on the fine grid
        if time.time() - start < budget_s - 2.0:
            for beta in (20.0, 200.0, 2000.0, 8000.0, 20000.0):
                if time.time() - start > budget_s:
                    break
                best_h = slsqp_optimize_local(best_h, beta, rng, max_iter=300)
                cand_c = compute_c5(best_h)
                if cand_c < best_c:
                    best_c = cand_c

    # ---- final sanity check ---------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = best_h.size
    best_h = enforce_sum(best_h, n_final / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n_final)
