import time, numpy as np

# ---------------------------------------------------------------
# basic utilities
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto {0<=x_i<=1, sum x = target}."""
    lo = -np.max(v) - 1.0          # guarantees sum=0
    hi = 1.0 - np.min(v) + 1.0     # guarantees sum=n
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_box_simplex(v, n / 2.0)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap (the shift giving max C5)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

    # ----- FFT helpers (same n throughout the first phase) ----------
    M = 1 << ((3 * n - 1).bit_length())          # power of two ≥ 3n‑2
    ones_fft = np.fft.fft(np.ones(n), M)

    # ----- schedule of smooth‑max betas -------------------------------
    betas = [5., 20., 80., 200., 800., 3000., 8000., 20000.]
    for beta in betas:
        if time.time() - start > budget_s * 0.90:
            break
        max_iters = 4000 if beta < 200 else 2000
        step = 0.5
        for it in range(max_iters):
            if time.time() - start > budget_s * 0.90:
                break

            # FFT of current h
            H = np.fft.fft(h, M)

            # linear autocorrelation of h (zero‑padded)
            autocorr = np.real(np.fft.ifft(H * np.conj(H)))[: 2 * n - 1]
            Ck = dx * (target - autocorr)                # length 2n-1

            # smooth‑max weights
            mC = np.max(Ck)
            w = np.exp(beta * (Ck - mC))
            w /= w.sum()

            # FFTs of w and its reversal
            W = np.fft.fft(w, M)
            rev_w = w[::-1]
            W_rev = np.fft.fft(rev_w, M)

            # three linear convolutions (w*h, rev_w*h, w*1)
            conv_w_h = np.real(np.fft.ifft(W * H))[: 3 * n - 2]
            conv_rev_h = np.real(np.fft.ifft(W_rev * H))[: 3 * n - 2]
            conv_w_1 = np.real(np.fft.ifft(W * ones_fft))[: 3 * n - 2]

            start_idx = 2 * n - 2
            A_raw = conv_w_1[start_idx: start_idx + n]   # sum_w over valid window
            B_raw = conv_rev_h[start_idx: start_idx + n] # Σ w_k h_{j+k}
            C_raw = conv_w_h[start_idx: start_idx + n]   # Σ w_k h_{j‑k}

            grad = dx * (A_raw - B_raw - C_raw)

            # gradient step + projection
            h = project_box_simplex(h - step * grad, target)

            # occasional evaluation
            if (it & 31) == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()

            # slowly decay step size
            step *= 0.9995

        # final check after each beta
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- up‑sample to 512 points if needed ------------------------
    if best_h.size < 512 and time.time() - start < budget_s * 0.96:
        n_old = best_h.size
        n = 512
        x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, best_h)
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        h = project_box_simplex(h, target)
        dx = 2.0 / n
        # recompute FFT helper for new size
        M = 1 << ((3 * n - 1).bit_length())
        ones_fft = np.fft.fft(np.ones(n), M)

        # short refinement at higher resolution
        for beta in (200., 2000., 8000.):
            if time.time() - start > budget_s * 0.98:
                break
            step = 0.5
            for _ in range(800):
                H = np.fft.fft(h, M)
                autocorr = np.real(np.fft.ifft(H * np.conj(H)))[: 2 * n - 1]
                Ck = dx * (target - autocorr)
                mC = np.max(Ck)
                w = np.exp(beta * (Ck - mC))
                w /= w.sum()
                W = np.fft.fft(w, M)
                rev_w = w[::-1]
                W_rev = np.fft.fft(rev_w, M)

                conv_w_h = np.real(np.fft.ifft(W * H))[: 3 * n - 2]
                conv_rev_h = np.real(np.fft.ifft(W_rev * H))[: 3 * n - 2]
                conv_w_1 = np.real(np.fft.ifft(W * ones_fft))[: 3 * n - 2]

                start_idx = 2 * n - 2
                A_raw = conv_w_1[start_idx: start_idx + n]
                B_raw = conv_rev_h[start_idx: start_idx + n]
                C_raw = conv_w_h[start_idx: start_idx + n]

                grad = dx * (A_raw - B_raw - C_raw)
                h = project_box_simplex(h - step * grad, target)
                step *= 0.999

                if time.time() - start > budget_s * 0.98:
                    break
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # ----- final stochastic perturbations ---------------------------
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh

    # final sanity projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_simplex(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
