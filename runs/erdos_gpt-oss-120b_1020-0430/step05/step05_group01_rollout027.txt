CURRENT_TARGET = None, CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # small correction
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    max_it = 5 * n
    while abs(diff) > 1e-12 and it < max_it:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 1e-15:
            h[i] += add
            diff = target - h.sum()
        it += 1
    if abs(diff) > 1e-12:
        # tiny residual fix on a random index
        i = rng.integers(0, n)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# --------- soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 200) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------- simulated‑annealing at a given resolution ----------
def search_at_resolution(n: int, h_start: np.ndarray, rng: np.random.Generator,
                         time_limit: float):
    """SA/gradient search for size n, returns best (h, c)."""
    # interpolate if needed
    if h_start.size != n:
        x_old = np.linspace(0.0, 2.0, h_start.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, h_start)
    else:
        h = h_start.copy()
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, rng)

    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c

    # temperature schedule
    T0 = max(0.1, best_c * 0.5)
    T1 = 1e-8
    start = time.time()

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + T1 * frac

        move_type = rng.random()
        if move_type < 0.5:                     # gradient‑guided epsilon move
            grad = worst_gradient(h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
        elif move_type < 0.8:                   # random pair epsilon move
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
        else:                                   # direct swap
            i, j = rng.integers(0, n, size=2)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            c_new = compute_c5(h_new)
            if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-12)):
                h = h_new
                cur_c = c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h_new.copy()
            continue

        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        scale = max(0.01, T / max(T0, 1e-12))
        eps = rng.random() * max_eps * scale
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)

        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-12)):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()

    return best_h, best_c


# --------- random binary starter ----------
def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


# ------------------- main entry -------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # initial solution -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c = h0.copy(), compute_c5(h0)
    else:
        h0 = random_binary(512, rng)
        best_h, best_c = h0.copy(), compute_c5(h0)

    # multi‑resolution SA ---------------------------------------------
    # low resolution (128)
    rem = budget_s - (time.time() - start_total)
    if rem > 8.0:
        low_time = rem * 0.25
        h_low, c_low = search_at_resolution(128, best_h, rng, low_time)
        if c_low < best_c:
            best_h, best_c = h_low, c_low

    # medium resolution (256)
    rem = budget_s - (time.time() - start_total)
    if rem > 8.0:
        mid_time = rem * 0.35
        h_mid, c_mid = search_at_resolution(256, best_h, rng, mid_time)
        if c_mid < best_c:
            best_h, best_c = h_mid, c_mid

    # high resolution (512)
    rem = budget_s - (time.time() - start_total)
    if rem > 8.0:
        high_time = rem * 0.35
        h_high, c_high = search_at_resolution(512, best_h, rng, high_time)
        if c_high < best_c:
            best_h, best_c = h_high, c_high

    # final soft‑max SLSQP polishing -----------------------------------
    rem = budget_s - (time.time() - start_total)
    if rem > 2.0:
        for beta in (20.0, 200.0, 2000.0, 8000.0):
            if time.time() - start_total > budget_s * 0.98:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=200)
            best_c = compute_c5(best_h)

    # ensure 512‑point representation ---------------------------------
    if best_h.size != 512:
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, 512, endpoint=False)
        best_h = np.interp(x_new, x_old, best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
