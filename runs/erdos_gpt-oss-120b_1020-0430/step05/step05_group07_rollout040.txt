def project_onto_capped_simplex(y, target):
    # Find lambda via bisection between lower and upper bounds.
    # lower bound: lambda_low = min(y) - 1 (ensures all y_i - lambda > 1 => clamp to 1 -> sum = n). Actually lambda too low leads to all x_i = 1, sum=n.
    # upper bound: lambda_high = max(y) (ensures all x_i = 0).
    lo = np.min(y) - 1
    hi = np.max(y)
    for _ in range(50):
        lam = (lo + hi) / 2
        x = np.clip(y - lam, 0, 1)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo+hi)/2, 0, 1)
```

The monotonicity: As λ increases, y_i - λ decreases, so clamp results decrease, sum decreases. So if sum > target, need higher λ.

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h):
    """Maximum overlap integral for discrete step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def project_onto_capped_simplex(y, target):
    """Euclidean projection of y onto {0<=x<=1, sum x = target}."""
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

# ---------- gradient based local refinement ----------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        s = np.argmax(corr) - (n - 1)
        g = np.ones_like(h)
        if s >= 0:
            g[:n - s] -= h[s:]
            g[s:]     -= h[:n - s]
        else:
            k = -s
            g[k:]     -= h[:n - k]
            g[:n - k] -= h[k:]
        step_sz = step * (0.5 ** it)
        y = h - step_sz * g
        h = project_onto_capped_simplex(y, target)
    return h

# ---------- fractional simulated annealing ----------
def fractional_sa_hill_climb(h, rng, deadline):
    """Transfer‑based simulated annealing on a fractional vector."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_imp = 0
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        src, dst = (i, j) if rng.random() < 0.5 else (j, i)
        max_transfer = min(cur[src], 1.0 - cur[dst])
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        cur[src] -= delta
        cur[dst] += delta
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_imp = it
        else:
            cur[src] += delta
            cur[dst] -= delta
        it += 1
        if it - last_imp > 15000:
            break
    return best, best_c

# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation to max_k ∫h(1−h(·+k))."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP minimization of the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum correction via projection (fast)
    x = project_onto_capped_simplex(x, CURRENT_TARGET)
    return x

# ---------- subgradient descent on the worst shift ----------
def subgradient_descent(h0, rng, deadline):
    """Subgradient method targeting the current worst shift."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()

        # start point: jittered best or fresh random
        if rng.random() < 0.6:
            init = best_h + rng.normal(scale=0.05, size=n)
        else:
            init = rng.random(n)
        init = np.clip(init, 0.0, 1.0)
        init = project_onto_capped_simplex(init, target)

        h_opt, c_opt = subgradient_descent(init, rng, sub_dead)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        # short gradient refinement
        if time.time() < deadline - 5.0:
            h_ref = gradient_refine(best_h, rng, step=0.02, max_iter=12)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

        # light fractional SA
        if time.time() < deadline - 4.0:
            sa_dead = time.time() + min(5.0, deadline - time.time() - 2.0)
            h_sa, c_sa = fractional_sa_hill_climb(best_h.copy(), rng, sa_dead)
            if c_sa < best_c:
                best_c, best_h = c_sa, h_sa.copy()

    # final smooth‑max SLSQP cascade
    if time.time() < deadline - 1.0:
        h = best_h.copy()
        for beta in [50.0, 200.0, 500.0, 2000.0, 5000.0, 10000.0]:
            if time.time() > deadline:
                break
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=800)
            except Exception:
                continue
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # final projection and result
    best_h = project_onto_capped_simplex(best_h, n / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
