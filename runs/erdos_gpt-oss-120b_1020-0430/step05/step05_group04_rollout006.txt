import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    init_T = max(0.05 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        c = compute_c5(cur)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = c < cur_c or rng.random() < np.exp(-(c - cur_c) / T)
        if accept:
            cur_c = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c < best_c:
                best_c = c
                best = cur.copy()
                last_improve = it
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best, best_c

def greedy_improve(h, rng, max_iter=50, deadline=None):
    n = h.size
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        i_range = np.arange(n)
        cond_i = (h == 1) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 0)
        S1 = np.flatnonzero(cond_i)
        cond_j = (h == 0) & (i_range + k >= 0) & (i_range + k < n) & (h[i_range + k] == 1)
        S0 = np.flatnonzero(cond_j)
        if S1.size == 0 or S0.size == 0:
            break
        total_pairs = int(S1.size) * int(S0.size)
        max_samples = 2000
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = corr[max_idx] * (2.0 / n)
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

    h = h0.copy()
    adjust_sum_exact(h, rng)
    h = np.clip(h, 0.0, 1.0)
    h = gradient_refine(h, rng, step=0.025, max_iter=15)

    for it in range(2000):
        if time.time() > deadline:

        step = step_max * (0.95 ** it)
        h_new = best_h.copy()
        i = rng.integers(0, n)

        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:

    return best_h, best_c

def optimize_binary(h_bin, n, rng, deadline):
    cur_h = np.rint(h_bin).astype(float)

        if time.time() > deadline:
            break
        max_iter = 400 if beta < 200 else 800
        if beta >= 20000:
            max_iter = 1200
        try:
            h = slsqp_optimize(h, beta, rng, max_iter)
        except Exception:
            continue
        h = gradient_refine(h, rng, step=0.025, max_iter=12)
    best_h, best_c = h, compute_c5(h)
    step_max = 0.05
    for it in range(2000):
        if time.time() > deadline:
            break
        step = step_max * (0.95 ** it)
        h_new = best_h.copy()
        i = rng.integers(0, n)
        delta = rng.uniform(-step, step)
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        adjust_sum_exact(h_new, rng)
        h_new = gradient_refine(h_new, rng, step=0.025, max_iter=5)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            best_c, best_h = c_new, h_new
    return best_h, best_c

def optimize_binary(h_bin, n, rng, deadline):
    cur_h = np.rint(h_bin).astype(float)
    cur_h = np.clip(cur_h, 0.0, 1.0)
    cur_h, cur_c = greedy_improve(cur_h, rng, max_iter=200, deadline=deadline), compute_c5(cur_h)
    best_h, best_c = cur_h, cur_c
    time_left = deadline - time.time()
    if time_left > 0:
        sub_dead = time.time() + min(time_left * 0.5, 30.0)
        h_sa, c_sa = binary_sa_hill_climb(best_h, rng, sub_dead)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa
    if np.all((np.abs(best_h) < 1e-12) | (np.abs(best_h - 1) < 1e-12)):
        best_h = greedy_improve(best_h, rng, max_iter=200, deadline=deadline)
        best_c = compute_c5(best_h)
    return best_h, best_c

def upsample_linear(h, factor):
    n = h.size
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n * factor, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98
    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None
    cand = []
    if parent is not None:
        cand.append(parent.size)
    cand.extend([360, 480, 600, 720])
    cand = sorted(set([n for n in cand if n % 2 == 0]))
    best_h = None
    best_c = np.inf
    best_n = 0
    for n in cand:
        if time.time() > deadline:
            break
        starts = []
        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            starts.append(interp)
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            starts.append(bin_vec)
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)
        rand_frac = rng.random(n)
        rand_frac *= (n / 2) / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)
        starts.append(rand_frac)
        for h0 in starts:
            if time.time() > deadline:
                break
            h = h0.copy()
            adjust_sum_exact(h, rng)
            h = np.clip(h, 0.0, 1.0)
            if np.all((np.abs(h) < 1e-12) | (np.abs(h - 1) < 1e-12)):
                h_opt, c_opt = optimize_binary(h, n, rng, deadline)
            else:
                h_opt, c_opt = optimize_fractional(h, n, rng, deadline)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        up_h = upsample_linear(best_h, 2)
        adjust_sum_exact(up_h, rng)
        up_h, up_c = optimize_fractional(up_h, best_n * 2, rng, deadline)
        if up_c < best_c:
            best_h, best_c, best_n = up_h, up_c, best_n * 2
    if best_h is None:
        best_n = 180
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
