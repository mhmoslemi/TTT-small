import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        cor = diff / n
        h += cor
        h = np.clip(h, 0.0, 1.0)
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    dec_int = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            nh = perturb_h(h, rng, scale=step)
            c = compute_c5(nh)
            if c < best_c:
                best_c, h = c, nh
                improved = True
                break
        if not improved and rng.random() < 0.02:
            nh = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(nh)
            if c < best_c:
                best_c, h = c, nh
                improved = True
        if (it + 1) % dec_int == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def anneal_binary(h, c, max_time, rng, n):
    best_h = h.copy()
    best_c = c.max() * (2.0 / n)
    cur_c = best_c
    T = best_c * 0.5
    it = 0
    start = time.time()
    while time.time() - start < max_time:
        if T > 0:
            T *= 0.9995
        # pick random 1 and 0 positions
        i = rng.integers(0, n)
        while h[i] != 1:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 0:
            j = rng.integers(0, n)
        hf = h.astype(float)
        b = 1.0 - hf
        slice_h_i = slice(i, i + n)
        slice_h_j = slice(j, j + n)
        slice_b_i = slice((n - 1 - i), (n - 1 - i + n))
        slice_b_j = slice((n - 1 - j), (n - 1 - j + n))
        # apply forward deltas
        c[slice_h_i] -= b
        c[slice_h_j] += b
        c[slice_b_i] += hf
        c[slice_b_j] -= hf
        # swap
        h[i] = 0
        h[j] = 1
        new_c = c.max() * (2.0 / n)
        delta = new_c - cur_c
        accept = delta <= 0 or (T > 0 and rng.random() < np.exp(-delta / T))
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            # revert deltas
            c[slice_h_i] += b
            c[slice_h_j] -= b
            c[slice_b_i] -= hf
            c[slice_b_j] += hf
        it += 1
        if best_c < 0.3808:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initialise binary h
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        idx = np.argsort(-init)
        h = np.zeros(n, dtype=np.int8)
        h[idx[: n // 2]] = 1
    else:
        n = 360
        h = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1
    # correlation array for binary annealing
    h_float = h.astype(float)
    c = np.correlate(h_float, 1.0 - h_float, mode="full")
    # annealing phase
    elapsed = time.time() - start
    anneal_time = min(budget_s * 0.55, max(5.0, budget_s - elapsed - 15.0))
    h, _ = anneal_binary(h, c, anneal_time, rng, n)
    # hill‑climb fine‑tune (fractional)
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        h_float = h.astype(float)
        max_iters = int(remaining * 200)  # heuristic scaling
        h_float, _ = hill_climb_multi(h_float, rng, max_iters=max_iters,
                                      init_step=0.3, n_trials=5)
        h = np.clip(h_float, 0.0, 1.0)
        h = enforce_sum(h, n / 2.0, rng)
    final_c = compute_c5(h)
    return h.tolist(), float(final_c), int(n)
