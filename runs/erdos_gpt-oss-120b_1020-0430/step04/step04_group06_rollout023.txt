def gradient_softmax(h, beta):
    n = h.shape[0]
    # correlation
    f = np.correlate(h, 1.0 - h, mode='full')
    # stable softmax

def enforce_sum_fast(h, target):
    # ensure 0 <= h <=1 and sum = target
    # Clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Determine direction: if diff > 0, need to increase some values
    if diff > 0:
        # slack = 1 - h
        slack = 1.0 - h
        total_slack = slack.sum()
        if total_slack > 0:
            add = diff * slack / total_slack
            h = h + add
            # clip again to ensure not exceed 1 due to numerical errors
            h = np.clip(h, 0.0, 1.0)
    else:
        # need to decrease some values, capacity = h
        capacity = h
        total_cap = capacity.sum()
        if total_cap > 0:
            sub = diff * capacity / total_cap  # diff negative, sub negative
            h = h + sub
            h = np.clip(h, 0.0, 1.0)
    # After this, sum might be slightly off; correct with small adjustment
    diff2 = target - h.sum()
    if abs(diff2) > 1e-9:
        # distribute uniformly (small adjust)
        h += diff2 / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

    f = np.correlate(h, 1.0 - h, mode='full')
    max_f = np.max(f)
    # compute stabilized softmax weights
    w = np.exp(beta * (f - max_f))

import time
import numpy as np

# -------------------------------------------------------
#  Utility functions
# -------------------------------------------------------

def enforce_sum_fast(h, target):
    """Project h onto the box [0,1] and adjust to have sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0.0:
        slack = 1.0 - h
        total_slack = slack.sum()
        if total_slack > 0.0:
            h += diff * slack / total_slack
    else:
        cap = h
        total_cap = cap.sum()
        if total_cap > 0.0:
            h += diff * cap / total_cap
    h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-10:
        h += diff2 / h.size
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Generate a random feasible h with Σh = n/2 and 0≤h≤1."""
    h = rng.random(n)
    return enforce_sum_fast(h, n / 2.0)


def gradient_softmax(h, beta):
    """Gradient of the smooth max (softmax) of overlap values."""
    n = h.shape[0]
    f = np.correlate(h, 1.0 - h, mode="full")
    max_f = np.max(f)
    # stable softmax weights
    w = np.exp(beta * (f - max_f))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # compute convolutions needed for the gradient
    conv_plus = np.convolve(w[::-1], h, mode="full")
    c_plus = conv_plus[n - 1 : 2 * n - 1]
    conv_minus = np.convolve(w, h, mode="full")
    c_minus = conv_minus[n - 1 : 2 * n - 1]
    g = np.ones(n) - c_plus - c_minus
    return g


def perturb_small(h, rng, scale=0.02):
    """Add small Gaussian noise and re‑project."""
    h_new = h + rng.normal(scale=scale, size=h.shape)
    h_new = np.clip(h_new, 0.0, 1.0)
    return enforce_sum_fast(h_new, h.size / 2.0)


def max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift


def pair_transfer_optimize(
    h0, rng, max_iters=5000, time_limit=30.0, init_factor=0.5, decay=0.999
):
    """Specialised local search moving mass from i to i+shift to lower max overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift, dtype=int)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n, dtype=int)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        # safety clipping (sum stays exact)
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    return best_h, best_c

#  Main optimisation routine
# -------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---------------------------------------------------
    #  Initialise h
    # ---------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum_fast(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---------------------------------------------------
    #  Gradient‑based smooth‑max optimisation
    # ---------------------------------------------------
    betas = [5, 20, 80, 200, 800, 2000, 5000, 10000]
    steps_per_beta = 12000
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    target_sum = n / 2.0

    for beta in betas:
        if time.time() - start_time > budget_s * 0.80:
            break
        # start from a slight perturbation of the current best
        h = perturb_small(best_h, rng, scale=0.02)
        m = np.zeros(n)
        v = np.zeros(n)
        # learning rate scales with beta
        lr = 0.2 / np.sqrt(beta)
        for t in range(1, steps_per_beta + 1):
            if time.time() - start_time > budget_s * 0.80:
                break
            g = gradient_softmax(h, beta)
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum_fast(h, target_sum)
            if t % 50 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        # final evaluation for this beta
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # ---------------------------------------------------
    #  Pair‑transfer refinement (use remaining time)
    # ---------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 15.0:
        # try a few independent attempts within the remaining budget
        attempts = max(1, int(remaining // 30))
        per_attempt = remaining / (attempts + 1)
        for _ in range(attempts):
            start_vec = perturb_small(best_h, rng, scale=0.05) if rng.random() < 0.6 else best_h.copy()
            h_opt, c_opt = pair_transfer_optimize(
                start_vec,
                rng,
                max_iters=20000,
                time_limit=per_attempt,
                init_factor=0.5,
                decay=0.999,
            )
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
        # final longer run on the current best
        h_opt, c_opt = pair_transfer_optimize(
            best_h,
            rng,
            max_iters=50000,
            time_limit=remaining * 0.9,
            init_factor=0.5,
            decay=0.999,
        )
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # ---------------------------------------------------
    #  Final clean‑up and return
    # ---------------------------------------------------
    best_h = enforce_sum_fast(np.clip(best_h, 0.0, 1.0), target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
