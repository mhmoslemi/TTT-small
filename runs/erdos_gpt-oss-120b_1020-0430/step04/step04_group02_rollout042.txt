def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

def gradient_softmax(h, beta):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(beta * corr)
    exps = np.exp(beta * corr - max_val)
    p = exps / np.sum(exps)
    L = next_pow_two(3 * n - 2)
    ft_pr = np.fft.rfft(p[::-1], n=L)
    ft_one = np.fft.rfft(1.0 - h, n=L)
    conv1 = np.fft.irfft(ft_pr * ft_one, n=L)[:n]
    ft_p = np.fft.rfft(p, n=L)
    ft_h = np.fft.rfft(h, n=L)
    conv2 = np.fft.irfft(ft_p * ft_h, n=L)[:n]
    return dx * (conv1 - conv2)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 540
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            old_x = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            h = np.interp(new_x, old_x, h0)
        else:
            h = h0.copy()
    else:
        h = random_feasible_vector(n, rng)
    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000]

    while time.time() - start < budget_s * 0.80:
        # subgradient descent on current worst shift
        lr = 0.5 * (2.0 / n)
        for _ in range(200):
            if time.time() - start > budget_s * 0.80:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = int(np.argmax(corr))
            k_star = idx - (n - 1)
            g = np.zeros_like(h)
            if k_star > 0:
                g[: n - k_star] += (1.0 - h[k_star:])
                g[k_star:] -= h[: n - k_star]
            elif k_star < 0:
                k = -k_star
                g[k:] += (1.0 - h[: n - k])
                g[: n - k] -= h[k:]
            h_new = project_box_sum(h - lr * g, n / 2.0)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
                h = h_new
                lr = min(lr * 1.2, 2.0 * (2.0 / n))
            else:
                lr *= 0.7

        # softmax gradient descent
        for beta in beta_schedule:
            if time.time() - start > budget_s * 0.80:
                break
            lr = 0.5 * (2.0 / n)
            for _ in range(150):
                if time.time() - start > budget_s * 0.80:
                    break
                grad = gradient_softmax(h, beta)
                h_new = project_box_sum(h - lr * grad, n / 2.0)
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    best_c, best_h = c_new, h_new.copy()
                    h = h_new
                    lr = min(lr * 1.2, 2.0 * (2.0 / n))
                else:
                    lr *= 0.7

        # local hill climbing
        for _ in range(200):
            if time.time() - start > budget_s * 0.90:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = int(np.argmax(corr))
            k_star = idx - (n - 1)
            if rng.random() < 0.3:
                cand = targeted_move(h, rng, k_star, scale=0.2)
            else:
                cand = perturb_h(h, rng, scale=0.2)
            c_cand = compute_c5(cand)
            if c_cand < best_c:
                best_c, best_h = c_cand, cand.copy()
                h = cand

        if rng.random() < 0.05:
            h = perturb_h(best_h, rng, scale=0.3)

    # final SLSQP refinement
    try:
        for beta in beta_schedule:
            if time.time() - start > budget_s:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=300)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()
    except Exception:
        pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
