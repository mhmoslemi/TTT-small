            break

```
def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)
```

def compute_c5_and_conv(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L] * (2.0 / n)
    max_idx = np.argmax(conv)
    max_val = conv[max_idx]
    return max_val, conv, max_idx

import time
import numpy as np

def compute_c5_and_conv(h):
    """Return (c_max, full_conv_array, index_of_max) for binary step h."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L] * (2.0 / n)
    max_idx = int(np.argmax(conv))
    return conv[max_idx], conv, max_idx

def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    ones = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n, dtype=float)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total = budget_s

    # primary resolution
    n = 720
    # initialise from parent if possible
    if "initial_h_values" in globals():
        ih = np.asarray(initial_h_values, dtype=float)
        if ih.shape[0] == n:
            idx = np.argsort(-ih)[:n // 2]
            cur_h = np.zeros(n, dtype=float)
            cur_h[idx] = 1.0
        else:
            cur_h = random_binary_vector(n, rng)
    else:
        cur_h = random_binary_vector(n, rng)

    cur_c, cur_conv, cur_max_idx = compute_c5_and_conv(cur_h)

    # index bookkeeping for O(1) swaps
    ones = list(np.where(cur_h == 1)[0])
    zeros = list(np.where(cur_h == 0)[0])
    pos_one = np.empty(n, dtype=int)
    pos_zero = np.empty(n, dtype=int)
    for i, v in enumerate(ones):
        pos_one[v] = i
    for i, v in enumerate(zeros):
        pos_zero[v] = i

    best_h = cur_h.copy()
    best_c = cur_c
    best_n = n

    # SA parameters
    init_factor = 0.5
    final_factor = 0.001
    no_improve = 0
    restart_limit = 200000
    target = 0.3808

    while time.time() - start < total * 0.98:
        elapsed = time.time() - start
        frac = elapsed / total
        init_temp = cur_c * init_factor
        final_temp = cur_c * final_factor
        temp = init_temp * (1 - frac) + final_temp * frac

        # choose swap type
        if rng.random() < 0.25:  # targeted move
            k = cur_max_idx - (n - 1)
            if k >= 0:
                i_range = np.arange(k, n)
                h_i = cur_h[i_range]
                h_j = cur_h[i_range - k]
            else:
                i_range = np.arange(0, n + k)
                h_i = cur_h[i_range]
                h_j = cur_h[i_range - k]
            cand_mask = (h_i == 1) & (h_j == 0)
            cand_i = i_range[cand_mask]
            if cand_i.size:
                i1 = int(rng.choice(cand_i))
                i0 = zeros[rng.integers(len(zeros))]
            else:
                i1 = ones[rng.integers(len(ones))]
                i0 = zeros[rng.integers(len(zeros))]
        else:                     # random move
            i1 = ones[rng.integers(len(ones))]
            i0 = zeros[rng.integers(len(zeros))]

        if i1 == i0:
            continue

        # apply tentative swap
        cur_h[i1] = 0.0
        cur_h[i0] = 1.0
        new_c, new_conv, new_max_idx = compute_c5_and_conv(cur_h)
        delta = new_c - cur_c
        accept = delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp))
        if accept:
            # update bookkeeping
            pos_i1 = pos_one[i1]
            pos_i0 = pos_zero[i0]
            ones[pos_i1] = i0
            zeros[pos_i0] = i1
            pos_one[i0] = pos_i1
            pos_zero[i1] = pos_i0

            cur_c, cur_conv, cur_max_idx = new_c, new_conv, new_max_idx
            no_improve = 0
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
                if best_c <= target:
                    break
        else:
            # revert
            cur_h[i1] = 1.0
            cur_h[i0] = 0.0
            no_improve += 1

        if no_improve > restart_limit:
            # random restart
            cur_h = random_binary_vector(n, rng)
            cur_c, cur_conv, cur_max_idx = compute_c5_and_conv(cur_h)
            ones = list(np.where(cur_h == 1)[0])
            zeros = list(np.where(cur_h == 0)[0])
            pos_one = np.empty(n, dtype=int)
            pos_zero = np.empty(n, dtype=int)
            for i, v in enumerate(ones):
                pos_one[v] = i
            for i, v in enumerate(zeros):
                pos_zero[v] = i
            no_improve = 0

    # optional up‑sampling if time permits
    remaining = total - (time.time() - start)
    if remaining > total * 0.05 and best_n < 960:
        n2 = min(960, best_n * 2)
        factor = n2 // best_n
        if factor >= 2 and n2 == best_n * factor:
            cur_h = np.repeat(best_h, factor)
            n = n2
            cur_c, cur_conv, cur_max_idx = compute_c5_and_conv(cur_h)
            ones = list(np.where(cur_h == 1)[0])
            zeros = list(np.where(cur_h == 0)[0])
            pos_one = np.empty(n, dtype=int)
            pos_zero = np.empty(n, dtype=int)
            for i, v in enumerate(ones):
                pos_one[v] = i
            for i, v in enumerate(zeros):
                pos_zero[v] = i

            start_up = time.time()
            while time.time() - start_up < remaining * 0.9:
                elapsed = time.time() - start_up
                frac = elapsed / remaining
                init_temp = cur_c * init_factor
                final_temp = cur_c * final_factor
                temp = init_temp * (1 - frac) + final_temp * frac

                i1 = ones[rng.integers(len(ones))]
                i0 = zeros[rng.integers(len(zeros))]
                if i1 == i0:
                    continue
                cur_h[i1] = 0.0
                cur_h[i0] = 1.0
                new_c, new_conv, new_max_idx = compute_c5_and_conv(cur_h)
                delta = new_c - cur_c
                accept = delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp))
                if accept:
                    pos_i1 = pos_one[i1]
                    pos_i0 = pos_zero[i0]
                    ones[pos_i1] = i0
                    zeros[pos_i0] = i1
                    pos_one[i0] = pos_i1
                    pos_zero[i1] = pos_i0

                    cur_c, cur_conv, cur_max_idx = new_c, new_conv, new_max_idx
                    if cur_c < best_c:
                        best_c, best_h, best_n = cur_c, cur_h.copy(), n
                        if best_c <= target:
                            break
                else:
                    cur_h[i1] = 1.0
                    cur_h[i0] = 0.0

    final_c = compute_c5_and_conv(best_h)[0]
    return best_h.tolist(), float(final_c), int(best_n)
