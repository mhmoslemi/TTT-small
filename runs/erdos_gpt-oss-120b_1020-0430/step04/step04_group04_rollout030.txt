def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def softmax_grad(x):
    n = x.size
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    maxb = np.max(bc)
    expb = np.exp(bc - maxb)
    sumexp = np.sum(expb)
    w = expb / sumexp  # shape 2n-1
    # compute conv_fwd = conv(w_rev, x); conv_rev = conv(w, x)
    w_rev = w[::-1]
    L = len(w) + n - 1
    m = 1 << ((L - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, m)
    W = np.fft.rfft(w, m)
    W_rev = np.fft.rfft(w_rev, m)
    conv_rev = np.fft.irfft(W * X, m)[:L]
    conv_fwd = np.fft.irfft(W_rev * X, m)[:L]
    offset = n - 1
    sum_plus = conv_fwd[offset:offset + n]
    sum_minus = conv_rev[offset:offset + n]
    grad = 1.0 - sum_plus - sum_minus
    return grad

import time, numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def correlation(x):
    """Full correlation of x and (1-x) via FFT (no dx scaling)."""
    n = x.size
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    conv = correlation(h)
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Project h onto [0,1] and adjust to achieve Σh = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            nh = perturb_h(h, rng, scale=step)
            c = compute_c5(nh)
            if c < best_c:
                best_c, h = c, nh
                improved = True
                break
        if not improved and rng.random() < 0.02:
            nh = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(nh)
            if c < best_c:
                best_c, h = c, nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h, cur_c = best_h.copy(), best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        nh = perturb_h(cur_h, rng, scale=scale)
        nc = compute_c5(nh)
        delta = nc - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = nh, nc
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def softmax_obj(x):
    corr = correlation(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def softmax_grad(x):
    n = x.size
    corr = correlation(x)
    bc = CURRENT_BETA * corr
    maxb = np.max(bc)
    expb = np.exp(bc - maxb)
    sumexp = np.sum(expb)
    w = expb / sumexp                   # length 2n-1
    w_rev = w[::-1]

    L = len(w) + n - 1
    m = 1 << ((L - 1).bit_length())
    X = np.fft.rfft(x, m)
    W = np.fft.rfft(w, m)
    W_rev = np.fft.rfft(w_rev, m)

    conv_rev = np.fft.irfft(W * X, m)[:L]
    conv_fwd = np.fft.irfft(W_rev * X, m)[:L]
    off = n - 1
    sum_plus = conv_fwd[off:off + n]
    sum_minus = conv_rev[off:off + n]
    grad = 1.0 - sum_plus - sum_minus
    return grad


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        jac=softmax_grad,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    def time_left():
        return budget_s - (time.time() - start)

    # ---- initial resolution (360) ----
    n0 = 360
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n0 / 2.0, rng)
    else:
        h0 = random_feasible_vector(n0, rng)

    xs = (np.arange(n0) + 0.5) * (2.0 / n0)
    candidates = [h0]

    # triangular
    tri = np.where(xs <= 1, xs, 2 - xs)
    candidates.append(enforce_sum(tri, n0 / 2.0, rng))
    # cosine bump
    cosine = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    candidates.append(enforce_sum(cosine, n0 / 2.0, rng))
    # sinusoids
    for f in range(1, 5):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(s, n0 / 2.0, rng))
        candidates.append(enforce_sum(c, n0 / 2.0, rng))
    # random
    for _ in range(8):
        candidates.append(random_feasible_vector(n0, rng))

    best_h, best_c = None, np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ---- phase 1: local improvement (≈20% budget) ----
    while time_left() > budget_s * 0.2:
        start_vec = perturb_h(best_h, rng, scale=0.4)
        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=2000,
                                      init_step=0.5, n_trials=8)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
        h_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iter=5000,
                                         init_temp=1e-2,
                                         final_temp=1e-5,
                                         init_scale=0.5,
                                         final_scale=0.02)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        if time_left() < 30:
            break

    # ---- phase 2: SLSQP smoothing (mid budget) ----
    for beta in (200, 800, 3000, 10000, 30000):
        if time_left() < 30:
            break
        try:
            h_opt = slsqp_optimize(best_h, beta, max_iter=1500)
        except Exception:
            continue
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ---- phase 3: up‑sample and fine‑tune (remaining budget) ----
    if time_left() > budget_s * 0.2:
        target_n = max(720, n0 * 2)
        if time_left() > budget_s * 0.4 and target_n < 1080:
            target_n = 1080
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, target_n / 2.0, rng)

        up_h, _ = hill_climb_multi(up_h, rng, max_iters=3000,
                                   init_step=0.4, n_trials=8)
        up_h, _ = simulated_annealing(up_h, rng,
                                      max_iter=8000,
                                      init_temp=5e-3,
                                      final_temp=1e-5,
                                      init_scale=0.4,
                                      final_scale=0.02)

        for beta in (500, 2000, 8000, 30000, 80000):
            if time_left() < 10:
                break
            try:
                up_h = slsqp_optimize(up_h, beta, max_iter=2000)
            except Exception:
                continue

        up_h, _ = hill_climb_multi(up_h, rng, max_iters=2000,
                                   init_step=0.3, n_trials=6)

        final_c = compute_c5(up_h)
        if final_c < best_c:
            best_c, best_h, n0 = final_c, up_h.copy(), target_n

    # ---- final cleanup ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n0 / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n0)
