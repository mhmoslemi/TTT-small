def compute_gradient(h, beta):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # shape (2n-1)
    w = np.exp(beta * corr)
    w_sum = w.sum()
    w = w / w_sum
    grad = np.zeros_like(h)
    # compute contributions
    # for lag = i - (n-1) where i index in w
    for idx in range(2*n-1):
        w_l = w[idx]
        if w_l == 0:
            continue
        lag = idx - (n-1)
        if lag >= 0:
            if lag < n:
                # term for j where j+lag in range:
                grad[:n - lag] -= dx * w_l * h[lag:]
                # term for j where j-lag in range:
                grad[lag:] -= dx * w_l * h[:n - lag]

import time
import numpy as np

# ----------- basic utilities -------------------------------------------------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Project x onto [lo,hi] while forcing sum(x)=target."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


# ----------- gradient based smoothing ---------------------------------------
def _gradient_step(h, beta, step):
    """One gradient-descent step for the soft‑max approximation."""
    n = h.size
    dx = 2.0 / n
    # correlation for current h
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    # soft‑max weights
    w = np.exp(beta * (corr - m))
    w /= w.sum()

    grad = np.zeros_like(h)
    # w index 0 corresponds to lag = -(n-1)
    offset = n - 1
    for idx, w_l in enumerate(w):
        if w_l == 0.0:
            continue
        lag = idx - offset
        if lag >= 0:
            if lag < n:
                grad[: n - lag] -= dx * w_l * h[lag:]
                grad[lag:]     -= dx * w_l * h[: n - lag]
        else:
            L = -lag
            if L < n:
                grad[L:]      -= dx * w_l * h[: n - L]
                grad[: n - L] -= dx * w_l * h[L:]

    new_h = h - step * grad
    new_h = project_box_sum(new_h, n / 2.0)
    # compute new objective (actual max overlap)
    new_corr = np.correlate(new_h, 1.0 - new_h, mode="full") * dx
    new_val = np.max(new_corr)
    return new_h, new_val


def smooth_descent(start_h, beta, step_init, steps, rng):
    """Run a short gradient descent with adaptive step size."""
    h = start_h.copy()
    best_val = compute_c5(h)
    step = step_init
    for _ in range(steps):
        new_h, new_val = _gradient_step(h, beta, step)
        if new_val < best_val:
            h, best_val = new_h, new_val
            step = min(step * 1.1, step_init * 2.0)
        else:
            step *= 0.6
            if rng.random() < 0.07:
                cand = perturb_h(h, rng, scale=step * 2.0)
                cand = project_box_sum(cand, h.size / 2.0)
                cand_val = compute_c5(cand)
                if cand_val < best_val:
                    h, best_val = cand, cand_val
                    step *= 1.3
        if step < 1e-8:
            break
    return h, best_val


def hill_climb_targeted(h_start, rng, max_iters=300, init_step=0.4, n_trials=3, p_target=0.25):
    """Shortened version of the earlier hill‑climb."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.shape[0]
    decay_interval = max(1, max_iters // 8)

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.75, 1e-5)
    return h, best_c
