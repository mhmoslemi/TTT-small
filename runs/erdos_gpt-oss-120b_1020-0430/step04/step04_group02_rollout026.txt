def improve_move(h, k, rng, scale=0.2):
    n = len(h)
    # valid i for shift k
    if k >=0:
        idx_range = np.arange(n - k)
        i_idx = idx_range
        i2_idx = i_idx + k
    else:
        k_abs = -k
        idx_range = np.arange(k_abs, n)
        i_idx = idx_range
        i2_idx = i_idx - k_abs
    # compute sums
    sums = h[i_idx] + h[i2_idx]
    # choose i_low where sums is minimal and both positions not near bounds to allow increase
    # ensure h[i] < 1 and h[i2] < 1.
    low_mask = (sums < np.percentile(sums, 30)) & (h[i_idx] < 1.0) & (h[i2_idx] < 1.0)
    # high_mask for decreasing
    high_mask = (sums > np.percentile(sums, 70)) & (h[i_idx] > 0.0) & (h[i2_idx] > 0.0)
    if not low_mask.any() or not high_mask.any():
        return h.copy()
    i_low = rng.choice(i_idx[low_mask])
    i_low2 = i_low + k if k >=0 else i_low - k_abs
    i_high = rng.choice(i_idx[high_mask])
    i_high2 = i_high + k if k >=0 else i_high - k_abs

    max_eps = min(1.0 - h[i_low],
                  1.0 - h[i_low2],
                  h[i_high],
                  h[i_high2])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low2] += eps
    new_h[i_high] -= eps
    new_h[i_high2] -= eps
    return new_h

def hill_climb_enhanced(h_start, rng, max_iters=2000, init_step=0.5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size
    decay_interval = max(1, max_iters // 20)
    for it in range(max_iters):
        # compute current max shift (argmax of corr)
        corr = np.correlate(h, 1 - h, mode='full')
        idx = int(np.argmax(corr))
        k_star = idx - (n - 1)
        # try multiple moves
        improved = False
        for _ in range(3):
            if rng.random() < 0.5:
                cand = improve_move(h, k_star, rng, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:

                c = compute_c5(cand)
                if c < best_c:
                    best_c, h = c, cand
        if (it+1) % decay_interval == 0:
            step = max(step * 0.9, 1e-5)
    return h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ----- basic utilities -------------------------------------------------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Project onto [lo,hi] while forcing sum(x)=target."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    """Uniform random vector then project to the simplex with box constraints."""
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Move a random epsilon from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Maximum overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----- targeted improvement move ---------------------------------------
def improve_move(h, k, rng, scale=0.2):
    """Increase product on the worst shift k (where overlap is maximal)."""
    n = h.size
    if k == 0:
        return perturb_h(h, rng, scale)

    if k > 0:
        i_idx = np.arange(n - k)
        i2_idx = i_idx + k
    else:
        k_abs = -k
        i_idx = np.arange(k_abs, n)
        i2_idx = i_idx - k_abs

    sums = h[i_idx] + h[i2_idx]

    # low-sum positions (room to increase)
    low_mask = (sums < np.percentile(sums, 40)) & (h[i_idx] < 0.999) & (h[i2_idx] < 0.999)
    # high-sum positions (room to decrease)
    high_mask = (sums > np.percentile(sums, 60)) & (h[i_idx] > 1e-8) & (h[i2_idx] > 1e-8)

    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)

    i_low = rng.choice(i_idx[low_mask])
    i_low2 = i_low + k if k > 0 else i_low - (-k)
    i_high = rng.choice(i_idx[high_mask])
    i_high2 = i_high + k if k > 0 else i_high - (-k)

    max_eps = min(1.0 - h[i_low], 1.0 - h[i_low2], h[i_high], h[i_high2])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)

    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low2] += eps
    new_h[i_high] -= eps
    new_h[i_high2] -= eps
    return new_h


# ----- enhanced hill climbing -------------------------------------------
def hill_climb_enhanced(h_start, rng, max_iters=2000, init_step=0.5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size
    decay_interval = max(1, max_iters // 20)

    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k_star = idx - (n - 1)

        improved = False
        for _ in range(3):
            if rng.random() < 0.5:
                cand = improve_move(h, k_star, rng, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break

        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.9, 1e-5)

    return h, best_c


# ----- SLSQP surrogate (softmax) ---------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)
