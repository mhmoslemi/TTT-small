        x_sample = hyperrect[0] + (hyperrect[1]-hyperrect[0]) * np.random.rand(len(h))
        
        # Gradient estimation via Anderson acceleration
        grad_est = stochastic_gradient(x_sample, n)
        
        # Stochastic gradient descent step
        x_new = project_to_bounds(
            x_sample - 0.05 * (np.random.rand(len(h)) < 0.5) * grad_est,
            (0,1)
        )
        
        # New function evaluation
        f_val_new = compute_c5(x_new)
        
        # Update hyperrect bounds
        update_hyperrect(lower_bound, upper_bound, x_new, f_val_new, ...)
        
        # Update Kriging model
        for landmark in krig_landmarks:
            update_kernel_predictions(landmark, x_new, f_val_new)
    
    return x_opt
