def compute_c5(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    R = np.fft.rfft(rev, L)
    conv = np.fft.irfft(H * R, L)[:2*n-1]
    max_val = np.max(conv)
    return max_val * (2.0 / n)
```

```
class C5Computer:
    def __init__(self, n):
        self.n = n
        self.L = 1 << ((2*n - 1).bit_length())
    def compute(self, h):
        H = np.fft.rfft(h, self.L)
        rev = (1 - h)[::-1]
        R = np.fft.rfft(rev, self.L)
        conv = np.fft.irfft(H * R, self.L)[:2*self.n - 1]
        return conv.max() * (2.0 / self.n)
```

    comp = C5Computer(n)
    cur_h = h_start.copy()
    cur_c5 = comp.compute(cur_h)

def perturb_h(h, rng, scale=0.1):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:

    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Utility functions
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(20000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary_vector(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


# --------------------------------------------------------------
# C5 computation (FFT based)
# --------------------------------------------------------------
class C5Computer:
    def __init__(self, n):
        self.n = n
        self.L = 1 << ((2 * n - 1).bit_length())

    def compute(self, h):
        H = np.fft.rfft(h, self.L)
        rev = (1.0 - h)[::-1]
        R = np.fft.rfft(rev, self.L)
        conv = np.fft.irfft(H * R, self.L)[: 2 * self.n - 1]
        return conv.max() * (2.0 / self.n)


# --------------------------------------------------------------
# Simulated annealing on binary vectors
# --------------------------------------------------------------
def binary_sa(h_start, rng, deadline, comp):
    n = h_start.shape[0]
    cur_h = h_start.copy()
    cur_c5 = comp.compute(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    T0 = cur_c5 * 0.5 + 1e-8
    Tf = 1e-6

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T0 * ((Tf / T0) ** frac)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if cur_h[i] == cur_h[j]:
            continue
        if cur_h[i] < cur_h[j]:
            i, j = j, i  # i has 1, j has 0

        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = comp.compute(cur_h)
        delta = new_c5 - cur_c5

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_c5


# --------------------------------------------------------------
# Small continuous hill‑climb (post‑SA)
# --------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters, comp):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = comp.compute(h)
    step = 0.5
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        new_c5 = comp.compute(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c5


# --------------------------------------------------------------
# Smooth surrogate for SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.fft.irfft(
        np.fft.rfft(x, 1 << ((2 * x.size - 1).bit_length()))
        * np.fft.rfft(((1.0 - x)[::-1]), 1 << ((2 * x.size - 1).bit_length())),
        1 << ((2 * x.size - 1).bit_length()),
    )[: 2 * x.size - 1]
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ----------------------------------------------------------

        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
    else:
        n = 720
        h_bin = random_binary_vector(n, rng)

    comp = C5Computer(n)

    best_h = h_bin.copy()
    best_c5 = comp.compute(best_h)

    # ----------------------------------------------------------
    # Phase 1 – Binary simulated annealing (≈60% budget)
    # ----------------------------------------------------------
    sa_deadline = start_time + 0.60 * budget_s
    best_h, best_c5 = binary_sa(best_h, rng, sa_deadline, comp)

    # ----------------------------------------------------------
    # Phase 2 – Continuous hill‑climb (≈20% budget)
    # ----------------------------------------------------------
    hc_deadline = start_time + 0.85 * budget_s
    if time.time() < hc_deadline:
        remaining_iters = 5000
        best_h, best_c5 = hill_climb(best_h, rng, remaining_iters, comp)

    # ----------------------------------------------------------
    # Phase 3 – SLSQP polishing (remaining time)
    # ----------------------------------------------------------
    if time.time() < deadline:
        target = n / 2.0
        diff = target - best_h.sum()
        if abs(diff) > 1e-9:
            i = rng.integers(0, n)
            best_h[i] = np.clip(best_h[i] + diff, 0.0, 1.0)
        for beta in [200.0, 800.0, 2000.0, 5000.0]:
            if time.time() >= deadline:
                break
            best_h = slsqp_optimize(best_h, beta, max_iter=250)
            best_c5 = comp.compute(best_h)

    # ----------------------------------------------------------
    # Final clean‑up
    # ----------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c5 = comp.compute(best_h)

    return best_h.tolist(), float(final_c5), int(n)
