```python
def trust_constr_optimize(h0, beta, max_iter=500):
    n = h0.size
    target = n/2
    # Define objective: smoothed max via log-sum-exp with parameter beta
    def obj(x):
        # ensure x in bounds 0-1? We'll handle bounds separately.
        corr = np.correlate(x, 1 - x, mode='full')
        # shape 2n-1
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    # Gradient? Not needed for trust-constr if gradient approximated numerically
    bounds = [(0.0, 1.0)] * n
    constraints = {'type':'eq','fun': lambda x: np.sum(x) - target}
    res = minimize(obj, h0, method='trust-constr',
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': max_iter, 'gtol':1e-12, 'xtol':1e-12, 'verbose': 0})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum same as before.
    # ...
    return x
```

```python
def pair_transfer_optimize(h0, rng, max_iters=10000, time_limit=30.0, init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        # compute current max shift and correlation
        max_val, shift = max_overlap_and_shift(h)  # shift index
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # Determine i_range, j_range
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        # compute second max product
        if prod.size > 1:
            # set the max to -inf temporarily
            second_max = np.max(np.delete(prod, idx_max))
        else:
            second_max = 0.0
        # compute epsilon to bring max product down to second_max
        # new product = (h_i - eps)*(1 - h_j) = second_max
        # eps = (prod_max - second_max) / (1 - h_j)
        prod_max = prod[idx_max]
        if 1.0 - h[j] > 1e-12:
            eps_target = (prod_max - second_max) / (1.0 - h[j])
        else:
            eps_target = 0.0
        # ensure eps_target within bounds
        eps = min(max_eps, eps_target)
        # optionally ensure eps positive
        if eps <= 0.0:
            # fallback to factor*max_eps
            eps = factor * max_eps
        # apply transfer
        h[i] -= eps
        h[j] += eps
        # clip
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-6, factor * decay)
    # final evaluation
    final_c = compute_c5(best_h)
    return best_h, final_c
```

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=4000):
    """Project h onto the simplex Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        # final correction
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector with Σh = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.4
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Small transfer preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2500, init_step=0.5, n_trials=8):
    """Hill climb with several random tries per iteration."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 12)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.75, 1e-4)
    return h, best_c

def max_overlap_and_shift(h):
    """Return (max correlation value, shift index) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)


# ----- pair‑transfer optimizer with line‑search epsilon --------------------
def pair_transfer_optimize(
    h0, rng, max_iters=20000, time_limit=30.0, init_factor=0.5, decay=0.99
):
    """Move mass from i to i+shift to lower the current max overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        # index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        # second largest product (fallback 0)
        if prod.size > 1:
            second_max = np.max(np.delete(prod, idx_max))
        else:
            second_max = 0.0
        prod_max = prod[idx_max]
        denom = 1.0 - h[j]
        if denom > 1e-12:
            eps_target = (prod_max - second_max) / denom
        else:
            eps_target = 0.0
        eps = min(max_eps, eps_target) if eps_target > 0 else factor * max_eps
        if eps <= 0.0:
            eps = factor * max_eps
        h[i] -= eps
        h[j] += eps
        # safety clipping
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-6, factor * decay)
    final_c = compute_c5(best_h)
    return best_h, final_c

CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    """Refine h using SLSQP on the log‑sum‑exp surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny correction)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
