def perturb_h_pair(h, rng, scale=0.2):

   n = h.size

   i = rng.integers(n)
   j = rng.integers(n)
   while j == i:
       j = rng.integers(n)
   max_eps = min(h[i], 1 - h[j])
   if max_eps <= 0:
       return h.copy()
   eps = (rng.random() - 0.5) * 2 * max_eps * scale
   # ensure not going outside bounds: we limited to max_eps
   new_h = h.copy()
   new_h[i] -= eps
   new_h[j] += eps
   return new_h

def targeted_move(h, rng, k_star, scale=0.2):
   n = h.size
   k = abs(k_star)
   if k == 0 or k >= n:
       return perturb_h_pair(h, rng, scale)
   # compute masks
   low_mask = (h[:n-k] < 0.5) & (h[k:] < 0.5)
   high_mask = (h[:n-k] > 0.5) & (h[k:] > 0.5)
   if not low_mask.any() or not high_mask.any():
       return perturb_h_pair(h, rng, scale)
   i_low = rng.choice(np.flatnonzero(low_mask))
   i_high = rng.choice(np.flatnonzero(high_mask))
   max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
   )
   if max_eps <= 0.0:
       return perturb_h_pair(h, rng, scale)
   eps = (rng.random() - 0.5) * 2 * max_eps * scale
   new_h = h.copy()
   new_h[i_low] += eps
   new_h[i_low + k] += eps
   new_h[i_high] -= eps
   new_h[i_high + k] -= eps
   return new_h

def softmax_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        if y.sum() > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


# ---------- basic moves ----------
def perturb_h_pair(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k_star, scale=0.2):
    n = h.size
    k = abs(k_star)
    if k == 0 or k >= n:
        return perturb_h_pair(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h_pair(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h_pair(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


# ---------- evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---------- SLSQP surrogate ----------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- hill climbing ----------
def hill_climb(start_h, rng, max_iters=2000, init_step=0.3, target_ratio=0.25, decay_interval=150):
    h = start_h.copy()
    n = h.size
    best_c = compute_c5(h)
    step = init_step
    for it in range(max_iters):
        # current shift with max overlap
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr)) - (n - 1)

        if rng.random() < target_ratio:
            cand = targeted_move(h, rng, k_star, scale=step)
        else:
            cand = perturb_h_pair(h, rng, scale=step)

        c = compute_c5(cand)
        if c < best_c:
            best_c, h = c, cand

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return h, best_c
