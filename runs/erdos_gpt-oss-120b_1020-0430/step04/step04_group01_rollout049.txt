    h = h0.copy()

# Basic linear algebra utilities (FFT based)
# --------------------------------------------------------------
def conv_full(a, b):
    """Linear convolution of a and b (real vectors)."""
    n = a.size + b.size - 1
    L = 1 << ((n).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[:n]
    return conv


def correlation(a, b):
    """Linear correlation of a with b: sum_i a[i] * b[i+k] (zero padding)."""
    return conv_full(a, b[::-1])


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.size
    vals = correlation(h, 1.0 - h)
    return vals.max() * (2.0 / n)

# --------------------------------------------------------------
def project_sum_box(x, target, max_iter=50):
    """Euclidean projection of x onto {0<=y<=1, sum(y)=target}."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return y

    candidates.append(project_sum_box(block, target))

    # alternating 0/1
    alt = np.tile([0.0, 1.0], n // 2)
    candidates.append(project_sum_box(alt, target))

    # a few random vectors
    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    # ----- keep the best from the pool -----
    best_h, best_c = None, float('inf')
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- Phase 1: Projected Gradient Descent (multi‑start) -----
    phase1_end = start_time + 0.65 * budget_s
    betas = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
    for h_start in candidates:
        if time.time() >= phase1_end:
            break
        # random circular shift to diversify the start
        shift = rng.integers(0, n)
        h_shifted = np.roll(h_start, shift)
        h_opt, c_opt = pgd_optimize(
            h_shifted, betas, target, phase1_end,
            steps_per_beta=800, init_lr=0.2
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- Phase 2: Binary Simulated Annealing -----
    # create binary start from the current best (top n/2 entries)
    bin_start = np.zeros(n)
    bin_start[np.argsort(-best_h)[: n // 2]] = 1.0
    sa_end = start_time + 0.90 * budget_s
    bin_h, bin_c = binary_sa(bin_start, rng, sa_end)
    if bin_c < best_c:
        best_c, best_h = bin_c, bin_h.copy()

    # ----- Phase 3: Optional up‑sampling and fine polishing -----
    if time.time() < deadline and best_h.size * 2 <= 1440:
        factor = 2
        h_fine = np.repeat(best_h, factor)
        n_fine = h_fine.size
        target_fine = n_fine / 2.0
        h_fine = project_sum_box(h_fine, target_fine)
        fine_end = start_time + budget_s
        h_opt, c_opt = pgd_optimize(
            h_fine, betas, target_fine, fine_end,
            steps_per_beta=800, init_lr=0.1
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- final cleanup -----
    best_h = project_sum_box(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
