def trust_constr_optimize(h0, beta, rng, max_iter=200):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        corr = np.correlate(x, 1 - x, mode='full')
        C = corr
        maxexp = np.max(beta * C)
        expb = np.exp(beta * C - maxexp)
        sumexp = np.exp(maxexp) * np.sum(expb)  # but we can compute faster: sumexp = np.sum(expb) * np.exp(maxexp)
        # Actually f = (log(sumexp)) / beta = (maxexp + log(np.sum(expb))) / beta
        f = (maxexp + np.log(np.sum(expb))) / beta
        return f * (2.0 / n)  # scaling by dx
    def jac(x):
        corr = np.correlate(x, 1 - x, mode='full')
        C = corr
        maxexp = np.max(beta * C)
        expb = np.exp(beta * C - maxexp)
        sumexp = np.sum(expb)
        w = expb / sumexp  # shape (2n-1,)
        n = x.size
        grad = np.zeros_like(x)
        # lags: - (n-1) .. (n-1)
        for idx, ell in enumerate(range(- (n-1), n)):
            weight = w[idx]
            if ell >= 0:
                i = slice(0, n - ell)
                j = slice(ell, n)
                grad[i] += weight * (1 - x[j])
                grad[j] += weight * (-x[i])
            else:
                d = -ell
                i = slice(d, n)
                j = slice(0, n - d)
                grad[i] += weight * (1 - x[j])
                grad[j] += weight * (-x[i])
        return grad * (2.0 / n)
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target,
            'jac': lambda x: np.ones_like(x)}
    res = minimize(fun, h0, method='trust-constr', jac=jac,
                   hess='2-point', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'gtol': 1e-12, 'xtol': 1e-12, 'verbose': 0})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly again
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust with small epsilon distribution
        for _ in range(5000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x
```

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector with entries in [0,1] and sum=n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def trust_constr_optimize(h0: np.ndarray, beta: float,
                         rng: np.random.Generator, max_iter: int = 200) -> np.ndarray:
    """Soft‑max optimisation using trust‑region constrained method."""
    n = h0.size
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        corr = np.correlate(x, 1 - x, mode='full')
        C = corr
        mb = beta * C.max()
        expb = np.exp(beta * C - mb)
        f = (mb + np.log(expb.sum())) / beta
        return f * dx
    def jac(x):
        corr = np.correlate(x, 1 - x, mode='full')
        C = corr
        mb = beta * C.max()
        expb = np.exp(beta * C - mb)
        w = expb / expb.sum()               # weights for each lag
        grad = np.zeros_like(x)
        # lags from -(n-1) .. (n-1)
        for idx, ell in enumerate(range(-(n - 1), n)):
            wt = w[idx]
            if ell >= 0:
                i = slice(0, n - ell)
                j = slice(ell, n)
                grad[i] += wt * (1 - x[j])
                grad[j] += wt * (-x[i])
            else:
                d = -ell
                i = slice(d, n)
                j = slice(0, n - d)
                grad[i] += wt * (1 - x[j])
                grad[j] += wt * (-x[i])
        return grad * dx
    cons = {'type': 'eq',
            'fun': lambda x: np.sum(x) - target,
            'jac': lambda x: np.ones_like(x)}
    res = minimize(fun, h0, method='trust-constr', jac=jac,
                   hess='2-point', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'gtol': 1e-12,
                            'xtol': 1e-12, 'verbose': 0})
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, rng)

def schedule_opt(h0: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Run a cascade of trust‑constr optimisations with increasing β."""
    h = h0.copy()
    for beta in (5., 20., 80., 200., 800., 3000., 8000.):
        maxit = 150 if beta < 200 else 300
        h = trust_constr_optimize(h, beta, rng, max_iter=maxit)
    return h
