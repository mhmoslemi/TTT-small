import time
import numpy as np

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    m = 1 << ((2 * n - 2).bit_length())     # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, m)
    R = np.abs(H) ** 2
    conv = np.fft.irfft(R, m)[: 2 * n - 1]
    corr = n / 2.0 - conv
    return np.max(corr) * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ----------------------------------------------------
    if "initial_h_values" in globals():
        old = np.array(initial_h_values, dtype=float)
        n0 = old.size
        target_n = max(720, min(1200, n0 * 3 // 2))
        xs_old = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs_new = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h = np.interp(xs_new, xs_old, old)
    else:
        target_n = 960
        h = np.full(target_n, 0.5)

    target_sum = target_n / 2.0
    h = enforce_sum(h, target_sum, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    n = target_n
    m_fft = 1 << ((2 * n - 2).bit_length())

    beta = 5.0
    step = 0.2
    it = 0
    beta_inc_interval = 2000

    # ---- main projected‑gradient loop ---------------------------------
    while time.time() - start < budget_s * 0.9:
        # correlation for current h
        H = np.fft.rfft(h, m_fft)
        conv = np.fft.irfft(np.abs(H) ** 2, m_fft)[: 2 * n - 1]
        corr = target_sum - conv
        c = np.max(corr) * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

        # smooth weights (log‑sum‑exp trick)
        scaled = beta * corr
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_pos = w[n - 1 :]                     # k ≥ 0, length n

        # gradient via two linear convolutions
        g2 = np.convolve(w_pos, h)[:n]
        g1 = np.convolve(w_pos, h[::-1])[n - 1 :: -1]
        grad = g1 + g2

        # normalized step
        norm = np.linalg.norm(grad)
        if norm > 0:
            h = h + (step / norm) * grad

        # box & sum projection
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target_sum, rng)

        # schedule adjustments
        it += 1
        if it % beta_inc_interval == 0:
            beta = min(beta * 1.5, 5000.0)
            step = max(step * 0.8, 1e-4)

        # occasional stochastic jitter to escape shallow traps
        if rng.random() < 0.01:
            h += rng.normal(scale=0.005, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, target_sum, rng)

    # ---- optional fine‑tuning with a short hill‑climb -----------------
    remaining = budget_s - (time.time() - start)
    if remaining > 20.0:
        # simple pairwise transfer hill‑climb (few hundred trials)
        def perturb(vec, scale=0.1):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                return vec
            max_eps = min(vec[i], 1.0 - vec[j])
            if max_eps <= 0:
                return vec
            eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
            new = vec.copy()
            new[i] -= eps
            new[j] += eps
            return new

        for _ in range(2000):

            cand = enforce_sum(cand, target_sum, rng)
            cval = compute_c5(cand)

    # ---- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c = compute_c5(best_h)
