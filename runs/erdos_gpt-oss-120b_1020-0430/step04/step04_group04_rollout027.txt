def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    # convolution using FFT
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    fft_h = np.fft.rfft(h, m)
    fft_rev = np.fft.rfft(rev, m)
    conv = np.fft.irfft(fft_h * fft_rev, m)[:L]
    # max over non-negative shift -> index off = n-1 to L-1
    off = n - 1
    max_val = np.max(conv[off:]) * (2.0 / n)
    return max_val

def simulated_annealing_limited(h0, rng, max_iter, init_temp, final_temp, init_scale, final_scale, time_limit, start_time):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    for i in range(max_iter):
        now = time.time()
        if now - start_time > time_limit:
            break
        t_frac = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t_frac
        scale = init_scale * (final_scale / init_scale) ** t_frac
        new_h = perturb_h(cur_h, rng, scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()

def perturb_h(h, rng, scale):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure within bounds
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # no need to clip; values should be within [0,1]
    return new_h

def enforce_sum(h, target, rng):
    # Clip then adjust to meet target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # while diff not zero adjust a random index
    # but to be efficient use vector adjustments
    # Use weighted adjustment approach
    if abs(diff) < 1e-12:
        return h
    # Identify free indices (not at bounds)
    free = (h > 1e-12) & (h < 1 - 1e-12)
    free_idx = np.where(free)[0]
    if free_idx.size > 0:
        per = diff / free_idx.size
        h[free_idx] += per
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            return h
    # fallback: adjust all
    per = diff / n
    h += per
    h = np.clip(h, 0.0, 1.0)
    # final correction for any residual
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # adjust a random index up to bound
        for _ in range(100):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global state for the SLSQP surrogate
CURRENT_TARGET_ = None
CURRENT_BETA_ = None


def sum_eq_constraint(x):
    """Equality constraint Σx = target."""
    return np.sum(x) - CURRENT_TARGET_


def softmax_obj(x):
    """Smooth approximation of max_k correlation using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA_ * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA_


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())          # next power of two ≥ L
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng):
    """Clip h to [0,1] and adjust to reach Σh = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return h
    # First try to distribute among interior points
    free = (h > 1e-12) & (h < 1 - 1e-12)
    idx = np.where(free)[0]
    if idx.size:
        per = diff / idx.size
        h[idx] += per
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            return h
    # Fallback: uniform adjustment
    per = diff / n
    h += per
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Final tiny correction
    if abs(diff) > 1e-12:
        for _ in range(200):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = v / v.sum() * target
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    for _ in range(500):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - v[i]) if diff > 0 else max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v


def perturb_h(h, rng, scale=0.2):
    """Transfer ε from one coordinate to another, preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_limited(h0, rng, max_iters, step, time_limit, start_time):
    """Simple stochastic hill‑climb with a time budget."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    step_sz = step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        new_h = perturb_h(best_h, rng, scale=step_sz)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h
        if (it + 1) % decay == 0:
            step_sz = max(step_sz * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing_limited(h0, rng, max_iter, init_temp, final_temp,
                                init_scale, final_scale, time_limit, start_time):
    """Simulated‑annealing with a hard time limit."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h, cur_c = best_h.copy(), best_c
    for i in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def upsample_h(h, old_n, new_n, rng):
    """Linear interpolation to a finer grid and re‑normalisation."""
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    up_h = np.interp(new_x, old_x, h)
    up_h = enforce_sum(up_h, new_n / 2.0, rng)
    return up_h


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine using a smooth surrogate and SLSQP."""
    global CURRENT_TARGET_, CURRENT_BETA_
    n = h0.size
    CURRENT_TARGET_ = n / 2.0
    CURRENT_BETA_ = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET_ - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET_ - x.sum()
    return x
