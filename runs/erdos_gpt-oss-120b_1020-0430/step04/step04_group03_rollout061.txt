def enforce_sum(h, target, rng):
    diff = target - h.sum()
    # Spread diff equally
    h += diff / h.size
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # Recompute diff
    diff = target - h.sum()
    # If still not zero, adjust by random picks
    max_iter = 1000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def gradient_smoothed(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    exp_term = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_term)
    p = exp_term / sum_exp
    smooth_val = np.dot(p, corr)
    a = p * (1.0 + beta * (corr - smooth_val))
    grad = np.zeros(n)
    # iterate over shifts
    for idx, w in enumerate(a):
        if w == 0.0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] += w * (1.0 - h[:n - s_abs])
            grad[:n - s_abs] -= w * h[s_abs:]
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    """Project h onto [0,1]^n with sum exactly target."""
    diff = target - h.sum()
    # simple equal redistribution
    if abs(diff) > 1e-12:
        h += diff / h.size
    h = np.clip(h, 0.0, 1.0)
    # fix remaining error with random moves
    diff = target - h.sum()
    max_iter = 2000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
    # final correction (rare)
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def compute_c5(h):
    """Discrete C5 for vector h on [0,2] with step 2/n."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=800):
    """Deterministic single‑pair transfer improvement."""
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

# ---------- smoothed gradient descent ----------
def gradient_smoothed(h, beta):
    """Gradient of the soft‑max approximation of max correlation."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    exp_term = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_term)
    p = exp_term / sum_exp
    smooth_val = np.dot(p, corr)
    a = p * (1.0 + beta * (corr - smooth_val))
    grad = np.zeros(n)
    for idx, w in enumerate(a):
        if w == 0.0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] += w * (1.0 - h[:n - s_abs])
            grad[:n - s_abs] -= w * h[s_abs:]
    return grad

def smooth_descent(h, beta_seq, rng, time_limit, start_time, budget_s):
    """Gradient descent on the smoothed max, with decreasing step sizes."""
    target = h.size / 2.0
    for beta in beta_seq:
        # adapt step size to beta
        step_base = 0.5 / (beta + 1.0)
        # number of inner steps inversely related to beta
        steps = max(5, int(200 / (beta ** 0.25)))
        for _ in range(steps):
            if time.time() - start_time > time_limit:
                return h
            grad = gradient_smoothed(h, beta)
            h = h - step_base * grad
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, target, rng)
    return h

# ---------- simulated annealing ----------
def anneal_optimize(h, rng, time_limit, start_time):
    n = h.size
    target = n / 2.0
    best = h.copy()
    best_c = compute_c5(best)
    cur = h.copy()
    cur_c = best_c
    T0 = 0.01 * best_c + 1e-6
    k = 0
    while time.time() - start_time < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_max = min(cur[i], 1.0 - cur[j])
        if eps_max <= 0:
            continue
        eps = eps_max * rng.random()
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        # sum unchanged, no clipping needed
        cand_c = compute_c5(cand)
        delta = cand_c - cur_c
        T = T0 * (0.995 ** k)
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur, cur_c = cand, cand_c
            if cand_c < best_c:
                best, best_c = cand, cand_c
        k += 1
    return best, best_c

# ---------- interpolation helpers ----------
def upscale_interpolate(h, new_n):
    old_n = h.size
    xs_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    xs_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(xs_new, xs_old, h)
    return h_new

def downscale_interpolate(h, new_n):
    # simple down‑sampling via interpolation (preserves sum approx)
    old_n = h.size
    xs_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    xs_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(xs_new, xs_old, h)
    return h_new
