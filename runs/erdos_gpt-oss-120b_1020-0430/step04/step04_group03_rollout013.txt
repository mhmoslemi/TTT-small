def softmax_gradient(h, beta):
    n = h.size
    t = 1 - h
    corr = np.correlate(h, t, mode='full')
    bc = beta * corr
    max_bc = np.max(bc)

def proj_onto_sum_and_bounds(v, target, max_iter=60):
    n = v.size
    low = v.min() - 1.0
    high = v.max()
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low+high)/2, 0.0, 1.0)
```

def proj_onto_sum_and_bounds(v, target, max_iter=60):
    n = v.size
    low = v.min() - 1.0
    high = v.max()
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

def softmax_gradient(h, beta):
    n = h.size
    t = 1.0 - h
    corr = np.correlate(h, t, mode='full')
    bc = beta * corr
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    p = exp_bc / exp_bc.sum()
    p_rev = p[::-1]
    conv1 = np.convolve(p_rev, t, mode='full')
    conv2 = np.convolve(p, h, mode='full')
    start = n - 1
    grad = conv1[start:start + n] - conv2[start:start + n]
    return grad
```

import time, numpy as np

# ---------- basic utilities ----------
def proj_onto_sum_and_bounds(v, target, max_iter=60):
    low = v.min() - 1.0
    high = v.max()
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

def enforce_sum(h, target, rng=None):
    return proj_onto_sum_and_bounds(h, target)

def random_feasible_vector(n, rng):
    v = rng.random(n)
    return proj_onto_sum_and_bounds(v, n / 2.0)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1.0 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            eps = max_eps * factor
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            if compute_c5(nh) + 1e-12 < compute_c5(h):
                h = nh
                improved = True
                break
        if not improved:
            break
    return h

# ---------- softmax gradient ----------
def softmax_gradient(h, beta):
    n = h.size
    t = 1.0 - h
    corr = np.correlate(h, t, mode='full')
    bc = beta * corr
    m = bc.max()
    exp_bc = np.exp(bc - m)
    p = exp_bc / exp_bc.sum()
    p_rev = p[::-1]
    conv1 = np.convolve(p_rev, t, mode='full')
    conv2 = np.convolve(p, h, mode='full')
    start = n - 1
    grad = conv1[start:start + n] - conv2[start:start + n]
    return grad

def gradient_descent_softmax(h0, beta, steps, step_size, target):
    h = h0.copy()
    for _ in range(steps):
        grad = softmax_gradient(h, beta)
        nh = h - step_size * grad
        nh = proj_onto_sum_and_bounds(nh, target)
        if compute_c5(nh) < compute_c5(h) - 1e-12:
            h = nh
            step_size *= 1.05
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                break
    return h

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n / 2.0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- stochastic greedy + softmax polishing ----
    while time.time() - start_time < budget_s * 0.70:
        if rng.random() < 0.7 and best_h is not None:
            start = perturb_random(best_h, rng, scale=0.5)
        else:
            start = random_feasible_vector(n, rng)

        local = greedy_optimize(start, max_steps=5000)
        c_local = compute_c5(local)
        if c_local < best_c:
            best_h, best_c = local, c_local

        # softmax polishing with increasing beta
        for beta in (200, 800, 2000, 5000):
            if time.time() - start_time > budget_s * 0.85:
                break
            step0 = 0.5 / (beta / 200.0)
            refined = gradient_descent_softmax(best_h, beta, steps=30,
                                              step_size=step0, target=n / 2.0)
            c_ref = compute_c5(refined)
            if c_ref < best_c:
                best_h, best_c = refined, c_ref

    # ---- final tiny perturbation search ----
    while time.time() - start_time < budget_s * 0.99:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
