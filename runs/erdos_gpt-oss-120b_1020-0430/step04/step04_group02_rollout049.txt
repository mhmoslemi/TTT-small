import time
import numpy as np

# ---------- basic helpers ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Project x onto [lo,hi] while fixing sum(x)=target."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def compute_corr(h):
    """Full linear correlation of h with (1-h)."""
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5_from_corr(corr, n):
    """Return C5 given correlation array."""
    return corr.max() * (2.0 / n)


# ---------- incremental correlation update ----------
def delta_corr_from_moves(entries, h, arange_n, n):
    """
    entries: list of (idx, delta_val) where sum(delta_val)=0.
    Returns the change to the correlation array caused by applying the deltas.
    """
    size = 2 * n - 1
    delta = np.zeros(size, dtype=np.float64)

    # S1 and S2 using vectorised addition
    for idx, dv in entries:
        if dv == 0.0:
            continue
        arr = dv * h  # length n
        # S1: pos = idx - t + n-1  => idx_offset - t
        idx_off = idx + n - 1
        pos1 = idx_off - arange_n
        mask1 = (pos1 >= 0) & (pos1 < size)
        delta[pos1[mask1]] += arr[mask1]

        # S2: pos = t - idx + n-1  => t + (n-1-idx)
        off2 = n - 1 - idx
        pos2 = arange_n + off2
        mask2 = (pos2 >= 0) & (pos2 < size)
        delta[pos2[mask2]] += arr[mask2]

    # S3: ordered pairs of deltas
    m = len(entries)
    for a in range(m):
        idx_a, dv_a = entries[a]
        for b in range(m):
            idx_b, dv_b = entries[b]
            k = idx_b - idx_a
            pos = k + n - 1
            delta[pos] += dv_a * dv_b

    return -delta  # because corr = sum h*(1-h_shift) and we changed the product term

        d_corr = delta_corr_from_moves(entries, h, arange_n, n)
        cand_corr = corr + d_corr
        cand_c5 = cand_corr.max() * (2.0 / n)

        delta = cand_c5 - cur_c5
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-16)):
            # accept
            for idx, dv in entries:
                h[idx] += dv
            corr = cand_corr
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

        # occasional full recompute to avoid drift
        if iteration % reeval_interval == 0:
            corr = compute_corr(h)
            cur_c5 = compute_c5_from_corr(corr, n)

    return best_h, best_c5


# ---------- optional SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp in a numerically stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    from scipy.optimize import minimize
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: uniform random feasible vector of size 360
        n0 = 360
        x = rng.random(n0)
        h0 = project_box_sum(x, n0 / 2.0)

    n0 = h0.size
    # upscale to a finer grid if time permits
    target_n = max(720, n0 * 2)  # at least 720 points
    if target_n > 1500:
        target_n = 1500
    if target_n != n0:
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_up = np.interp(new_x, old_x, h0)
        h_up = project_box_sum(h_up, target_n / 2.0)
        h0 = h_up

    # ---- simulated annealing ----
    best_h, best_c5 = anneal_h(h0, budget_s, rng)

    # ---- optional SLSQP refinement (use remaining time) ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.05:
        beta_list = [20, 80, 200, 500, 2000, 5000]
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.99:
                break
            try:
                ref_h = slsqp_optimize(best_h, beta, max_iter=200)
            except Exception:
                continue
            ref_corr = compute_corr(ref_h)
            ref_c5 = compute_c5_from_corr(ref_corr, ref_h.size)
            if ref_c5 < best_c5:
                best_c5 = ref_c5
                best_h = ref_h.copy()

    # final projection and clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c5 = compute_c5_from_corr(compute_corr(best_h), best_h.size)

    return best_h.tolist(), float(final_c5), int(best_h.size)
