    for s,e in zip(starts, ends):

        h[mask] = 1.0
    # enforce sum to N/2:
    target = N//2   # N is even (720) so N/2 = 360

        # adjust near boundaries: find indices where xs close to any interval boundary
        # Compute distances to nearest boundary
        # Combine start and end boundaries.
        boundaries = np.concatenate([starts, ends])
        # Compute distance of each index to any boundary
        dists = np.min(np.abs(xs[:, None] - boundaries[None, :]), axis=1)

            to_turn_on = idx_sorted[:target - cur]
            h[to_turn_on] = 1.0

def stick_break(v0, v1):
    return np.array([v0, (1 - v0) * v1, (1 - v0) * (1 - v1)])

def build_h_from_params(v, N, xs):
    # v is 4-d array
    l = stick_break(v[0], v[1])
    g = stick_break(v[2], v[3])
    starts = np.array([0.0, l[0] + g[0], l[0] + g[0] + l[1] + g[1]])
    ends = starts + l
    h = np.zeros(N, dtype=float)
    for s, e in zip(starts, ends):
        # note that due to floating errors, maybe e > 2 (should not)
        mask = (xs >= s) & (xs < e)
        h[mask] = 1.0
    # adjust sum to N/2
    target = N // 2
    cur = int(h.sum())
    if cur != target:
        # compute distance to nearest boundary
        boundaries = np.concatenate([starts, ends])
        dists = np.min(np.abs(xs[:, None] - boundaries[None, :]), axis=1)
        idx_sorted = np.argsort(dists)
        if cur > target:
            h[idx_sorted[:cur - target]] = 0.0
        else:
            h[idx_sorted[:target - cur]] = 1.0
    return h

def compute_c5_fft(h, N, fft_len):
    H = np.fft.rfft(h, n=fft_len)
    G = np.fft.rfft(1.0 - h, n=fft_len)
    conv = np.fft.irfft(H * np.conj(G), n=fft_len)[:2 * N - 1]
    return np.max(conv) * (2.0 / N)

def softmax_obj(v, N, xs, fft_len, beta):

def stick_break_multi(v):
    # v is array of length M-1
    # generate lengths that sum to 1

    G = np.fft.rfft(1.0 - h, n=fft_len)

    # due to possible large beta, we need stable computing
    # subtract m for stability
    sum_exp = np.sum(np.exp(beta * (conv - m)))
    return m + np.log(sum_exp) / beta

import numpy as np
from scipy.optimize import differential_evolution, minimize

# ----- global configuration (filled in run) -----
_global_N = None          # number of sample points
_global_xs = None         # sample positions
_global_fft_len = None    # length for FFT (power of two)
_global_beta = None       # temperature for soft‑max
M = 4                     # number of intervals (chosen for flexibility)


# ----- helper functions (must be top‑level) -----
def stick_break_multi(v):
    """Stick‑breaking transform: v in [0,1]^(M-1) → lengths summing to 1."""
    lengths = np.empty(len(v) + 1, dtype=float)
    prod = 1.0
    for i, vi in enumerate(v):
        lengths[i] = vi * prod
        prod *= (1.0 - vi)
    lengths[-1] = prod
    return lengths


def build_h_from_params(v):
    """Construct binary vector h for given parameter vector v."""
    N = _global_N
    xs = _global_xs

    v_int = v[: M - 1]
    v_gap = v[M - 1 :]
    l = stick_break_multi(v_int)          # interval lengths (sum 1)
    g = stick_break_multi(v_gap)          # gap lengths (sum 1)

    # starts of the intervals
    starts = np.empty(M, dtype=float)
    starts[0] = 0.0
    for i in range(1, M):
        starts[i] = starts[i - 1] + l[i - 1] + g[i - 1]
    ends = starts + l

    h = np.zeros(N, dtype=float)
    for s, e in zip(starts, ends):
        mask = (xs >= s) & (xs < e)
        h[mask] = 1.0

    # --- enforce exact integral (N/2 ones) -----------------
    target = N // 2
    cur = int(h.sum())
    if cur != target:
        # distances to the nearest interval boundary
        boundaries = np.concatenate([starts, ends])
        dists = np.min(np.abs(xs[:, None] - boundaries[None, :]), axis=1)
        idx = np.argsort(dists)
        if cur > target:
            h[idx[: cur - target]] = 0.0
        else:
            h[idx[: target - cur]] = 1.0
    return h


def compute_c5_fast_global(h):
    """Linear overlap C5 using an FFT (O(N log N))."""
    N = _global_N
    L = _global_fft_len
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * np.conj(G), n=L)[: 2 * N - 1]
    return np.max(conv) * (2.0 / N)


def objective_c5_global(v):
    """Objective for DE: the exact C5."""
    h = build_h_from_params(v)
    return compute_c5_fast_global(h)


def softmax_obj_global(v):
    """Smooth approximation of max used for SLSQP polishing."""
    beta = _global_beta
    h = build_h_from_params(v)
    N = _global_N
    L = _global_fft_len
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * np.conj(G), n=L)[: 2 * N - 1] * (2.0 / N)
    m = np.max(conv)
    # stable log‑sum‑exp
    sum_exp = np.sum(np.exp(beta * (conv - m)))
    return m + np.log(sum_exp) / beta
