def anneal(h0, time_limit):
    n = h0.size
    N = 1 << ((2*n - 1).bit_length())
    h = h0.copy()
    # compute initial C5 and shift
    a = np.fft.rfft(h, N)
    b = np.fft.rfft(1 - h, N)
    conv = np.fft.irfft(a * np.conj(b), N)[:2*n-1]
    idx = conv.argmax()
    s = idx - (n - 1)
    cur_c = conv[idx] * (2.0 / n)
    # best so far for this run
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    # temperature schedule: start_temp = 0.01, end_temp = 1e-5
    start_temp = 0.01
    end_temp = 1e-5
    total = max(time_limit, 1e-9)
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        temp = start_temp * (end_temp / start_temp) ** (elapsed / total)
        # compute gradient for worst shift s (already computed)
        # compute gradient
        grad = np.zeros(n)
        if s >= 0:
            grad[:n - s] += (1 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] += (1 - h[:n - s_abs])
            grad[:n - s_abs] -= h[s_abs:]
        # pick move
        pos_idx = np.where((grad > 0) & (h > 0))[0]
        neg_idx = np.where((grad < 0) & (h < 1))[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            # random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            max_eps = min(h[i], 1 - h[j])
            eps = max_eps * rng.random() * 0.5
        else:
            i = pos_idx[np.argmax(grad[pos_idx])]
            j = neg_idx[np.argmin(grad[neg_idx])]
            max_eps = min(h[i], 1 - h[j])
            # sample epsilon fraction
            eps = max_eps * rng.random()
        # new h
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        # compute new C5 and shift
        a2 = np.fft.rfft(nh, N)
        b2 = np.fft.rfft(1 - nh, N)
        conv2 = np.fft.irfft(a2 * np.conj(b2), N)[:2*n-1]
        idx2 = conv2.argmax()
        s2 = idx2 - (n - 1)
        new_c = conv2[idx2] * (2.0 / n)
        delta = new_c - cur_c
        # accept if better or with prob
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            h = nh
            cur_c = new_c
            s = s2
            if new_c < best_c:
                best_c = new_c
                best_h = nh.copy()
    return best_h, best_c

def hierarchical_optimize(rng, total_budget):
    # allocate fractions
    fine_n = 720
    coarse_n = 180
    # allocate times
    t0 = time.time()
    # allocate 0.2 of budget for coarse
    coarse_time = total_budget * 0.2
    # create initial coarse vector
    if 'initial_h_values' in globals():
        init_fine = np.array(initial_h_values, dtype=float)
        # downsample to coarse
        if init_fine.size == fine_n:
            coarse_init = init_fine.reshape(coarse_n, fine_n // coarse_n).mean(axis=1)
        else:
            coarse_init = random_feasible_vector(coarse_n, rng)
    else:
        coarse_init = random_feasible_vector(coarse_n, rng)
    coarse_init = enforce_sum(coarse_init, coarse_n/2., rng)
    # run anneal on coarse
    coarse_best_h, coarse_best_c = anneal(coarse_init, coarse_time)
    # if time left allocate medium
    elapsed = time.time() - t0
    remaining = total_budget - elapsed
    if remaining <= 0:
        # enlarge to fine via upsample and return
        best_h_fine = np.repeat(coarse_best_h, fine_n // coarse_n)
        best_h_fine = enforce_sum(best_h_fine, fine_n/2., rng)
        best_c = compute_c5_fft(best_h_fine)
        return best_h_fine, best_c
    medium_n = 360
    # upsample coarse_best_h to medium
    repeat_factor = medium_n // coarse_n
    medium_h = np.repeat(coarse_best_h, repeat_factor)
    medium_h = enforce_sum(medium_h, medium_n/2., rng)
    medium_time = total_budget * 0.3
    medium_best_h, medium_best_c = anneal(medium_h, medium_time)
    # upsample to fine
    repeat_factor2 = fine_n // medium_n
    fine_h = np.repeat(medium_best_h, repeat_factor2)
    fine_h = enforce_sum(fine_h, fine_n/2., rng)
    elapsed = time.time() - t0
    remaining = total_budget - elapsed
    if remaining <= 0:
        best_c = compute_c5_fft(fine_h)
        return fine_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # Enforce sum exactly after optimization
    # we can reuse enforce_sum
    rng_local = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng_local)
    return x

def greedy_optimize(h, max_steps=200):
    n = h.size
    for _ in range(max_steps):
        # worst shift info
        N = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(h, N)
        B = np.fft.rfft(1 - h, N)
        conv = np.fft.irfft(A * np.conj(B), N)[:2*n-1]
        idx = np.argmax(conv)
        s = idx - (n-1)
        # compute gradient
        grad = np.zeros(n)
        if s >= 0:
            grad[:n-s] += (1 - h[s:])
            grad[s:] -= h[:n-s]
        else:
            s_abs = -s
            grad[s_abs:] += (1 - h[:n-s_abs])
            grad[:n-s_abs] -= h[s_abs:]
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1-1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        # compute new C5
        A2 = np.fft.rfft(nh, N)
        B2 = np.fft.rfft(1 - nh, N)
        conv2 = np.fft.irfft(A2 * np.conj(B2), N)[:2*n-1]
        if conv2.max() < conv.max():
            h = nh
            continue
        # try half
        eps = max_eps * 0.5
        nh[i] = h[i] - eps
        nh[j] = h[j] + eps
        A2 = np.fft.rfft(nh, N)
        B2 = np.fft.rfft(1 - nh, N)
        conv2 = np.fft.irfft(A2 * np.conj(B2), N)[:2*n-1]
        if conv2.max() < conv.max():
            h = nh
            continue
        break
    return h
```

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def anneal(h0, time_limit, rng):
    n = h0.size
    N = 1 << ((2*n - 1).bit_length())
    h = h0.copy()
    # initial correlation and shift
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1 - h, N)
    conv = np.fft.irfft(A * np.conj(B), N)[:2*n-1]
    idx = conv.argmax()
    s = idx - (n - 1)
    cur_c = conv[idx] * (2.0 / n)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    start_temp = 0.01
    end_temp = 1e-5
    total = max(time_limit, 1e-9)
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        temp = start_temp * (end_temp / start_temp) ** (elapsed / total)
        # gradient for worst shift
        grad = np.zeros(n)
        if s >= 0:
            grad[:n - s] += (1 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] += (1 - h[:n - s_abs])
            grad[:n - s_abs] -= h[s_abs:]
        pos_idx = np.where((grad > 0) & (h > 0))[0]
        neg_idx = np.where((grad < 0) & (h < 1))[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            max_eps = min(h[i], 1 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.5
        else:
            i = pos_idx[np.argmax(grad[pos_idx])]
            j = neg_idx[np.argmin(grad[neg_idx])]
            max_eps = min(h[i], 1 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random()
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        # compute new correlation
        A2 = np.fft.rfft(nh, N)
        B2 = np.fft.rfft(1 - nh, N)
        conv2 = np.fft.irfft(A2 * np.conj(B2), N)[:2*n-1]
        idx2 = conv2.argmax()
        s2 = idx2 - (n - 1)
        new_c = conv2[idx2] * (2.0 / n)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            h = nh
            cur_c = new_c
            s = s2
            if new_c < best_c:
                best_c = new_c
                best_h = nh.copy()
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

# ---------- constraint handling ----------
def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

# ---------- evaluation ----------
def compute_c5_fft(h):
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1 - h, N)
    conv = np.fft.irfft(A * np.conj(B), N)[: 2 * n - 1]
    return conv.max() * (2.0 / n)

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

# ---------- greedy refinement ----------
def greedy_optimize(h, max_steps=200):
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    for _ in range(max_steps):
        A = np.fft.rfft(h, N)
        B = np.fft.rfft(1 - h, N)
        conv = np.fft.irfft(A * np.conj(B), N)[: 2 * n - 1]
        idx = np.argmax(conv)
        s = idx - (n - 1)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1 - h[j])
        if max_eps < 1e-12:
            break
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        A2 = np.fft.rfft(nh, N)
        B2 = np.fft.rfft(1 - nh, N)
        conv2 = np.fft.irfft(A2 * np.conj(B2), N)[: 2 * n - 1]
        if conv2.max() < conv.max():
            h = nh
            continue
        eps = max_eps * 0.5
        nh[i] = h[i] - eps
        nh[j] = h[j] + eps
        A2 = np.fft.rfft(nh, N)
        B2 = np.fft.rfft(1 - nh, N)
        conv2 = np.fft.irfft(A2 * np.conj(B2), N)[: 2 * n - 1]
        if conv2.max() < conv.max():
            h = nh
            continue
        break
    return h

# ---------- simulated annealing ----------
def anneal(h0, time_limit, rng):
    n = h0.size
    N = 1 << ((2 * n - 1).bit_length())
    h = h0.copy()
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1 - h, N)
    conv = np.fft.irfft(A * np.conj(B), N)[: 2 * n - 1]
    idx = conv.argmax()
    s = idx - (n - 1)
    cur_c = conv[idx] * (2.0 / n)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    start_temp, end_temp = 0.05, 1e-5
    total = max(time_limit, 1e-9)
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        temp = start_temp * (end_temp / start_temp) ** (elapsed / total)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 0) & (h > 0))[0]
        neg = np.where((grad < 0) & (h < 1))[0]
        if pos.size == 0 or neg.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            max_eps = min(h[i], 1 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.5
        else:
            i = pos[np.argmax(grad[pos])]
            j = neg[np.argmin(grad[neg])]
            max_eps = min(h[i], 1 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random()
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        A2 = np.fft.rfft(nh, N)
        B2 = np.fft.rfft(1 - nh, N)
        conv2 = np.fft.irfft(A2 * np.conj(B2), N)[: 2 * n - 1]
        idx2 = conv2.argmax()
        s2 = idx2 - (n - 1)
        new_c = conv2[idx2] * (2.0 / n)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            h = nh
            cur_c = new_c
            s = s2
            if new_c < best_c:
                best_c = new_c
                best_h = nh.copy()
    return best_h, best_c

# ---------- random perturbation ----------
def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- SLSQP polishing ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng_local = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng_local)
    return x

# ---------- hierarchical optimisation ----------
def hierarchical_optimize(rng, total_budget):
    fine_n = 720
    coarse_n = 180
    t_start = time.time()

    # initial coarse vector (down‑sample if possible)
    if 'initial_h_values' in globals():
        init_fine = np.array(initial_h_values, dtype=float)
        if init_fine.size == fine_n and fine_n % coarse_n == 0:
            factor = fine_n // coarse_n
            coarse_init = init_fine.reshape(coarse_n, factor).mean(axis=1)
        else:
            coarse_init = random_feasible_vector(coarse_n, rng)
    else:
        coarse_init = random_feasible_vector(coarse_n, rng)
    coarse_init = enforce_sum(coarse_init, coarse_n / 2.0, rng)

    # coarse stage
    coarse_time = total_budget * 0.20
    coarse_best_h, _ = anneal(coarse_init, coarse_time, rng)

    # medium stage (n = 360)
    elapsed = time.time() - t_start
    if total_budget - elapsed <= 0:
        fine_h = np.repeat(coarse_best_h, fine_n // coarse_n)
        fine_h = enforce_sum(fine_h, fine_n / 2.0, rng)
        return fine_h, compute_c5_fft(fine_h)

    medium_n = 360
    repeat_factor = medium_n // coarse_n
    medium_h = np.repeat(coarse_best_h, repeat_factor)
    medium_h = enforce_sum(medium_h, medium_n / 2.0, rng)
    medium_time = total_budget * 0.30
    medium_best_h, _ = anneal(medium_h, medium_time, rng)

    # fine stage (n = 720)
    elapsed = time.time() - t_start
    if total_budget - elapsed <= 0:
        fine_h = np.repeat(medium_best_h, fine_n // medium_n)
        fine_h = enforce_sum(fine_h, fine_n / 2.0, rng)
        return fine_h, compute_c5_fft(fine_h)

    repeat_factor2 = fine_n // medium_n
    fine_h = np.repeat(medium_best_h, repeat_factor2)
    fine_h = enforce_sum(fine_h, fine_n / 2.0, rng)
    fine_time = total_budget - (time.time() - t_start)
    fine_best_h, _ = anneal(fine_h, fine_time, rng)

    # consider the original fine vector if it exists
    if 'initial_h_values' in globals():
        init_fine = np.array(initial_h_values, dtype=float)
        if init_fine.size == fine_n:
            init_fine = enforce_sum(init_fine, fine_n / 2.0, rng)
            if compute_c5_fft(init_fine) < compute_c5_fft(fine_best_h):
                fine_best_h = init_fine
    return fine_best_h, compute_c5_fft(fine_best_h)

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # hierarchical search
    h_best, _ = hierarchical_optimize(rng, budget_s)

    # deterministic greedy polishing
    h_best = greedy_optimize(h_best, max_steps=500)

    # final SLSQP polishing if time permits
    elapsed = time.time() - start
    if budget_s - elapsed > 12:
        try:
            for beta in (200, 800, 2000, 5000):
                if time.time() - start > budget_s * 0.99:
                    break
                refined = slsqp_optimize(h_best, beta, max_iter=500)
                if compute_c5_fft(refined) < compute_c5_fft(h_best):
                    h_best = refined
        except Exception:
            pass

    # final clean‑up
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = enforce_sum(h_best, h_best.size / 2.0, rng)
    final_c = compute_c5_fft(h_best)
    return h_best.tolist(), float(final_c), int(h_best.size)
