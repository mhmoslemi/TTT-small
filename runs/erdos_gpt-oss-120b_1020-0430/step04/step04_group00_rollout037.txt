def compute_delta_vector(h, p, q):
    n = h.shape[0]
    L = 2*n - 1
    s = np.arange(-(n-1), n)   # shape (L,)
    offset = n-1
    # Build extended array with zeros at both ends of length 3*n-2.
    h_ext = np.empty(3*n-2, dtype=h.dtype)
    # Fill zeros first and last n-1 elements
    h_ext[:n-1] = 0.0
    h_ext[n-1:n-1+n] = h
    h_ext[n-1+n:] = 0.0
    # vector accesses
    idx_ps = p + s + offset   # shape L
    idx_qms = q - s + offset
    idx_pms = p - s + offset
    idx_qps = q + s + offset
    delta = -h_ext[idx_ps] + h_ext[idx_qms] - h_ext[idx_pms] + h_ext[idx_qps]
    return delta

But note that building h_ext inside each call O(3n) ~ O(n), fine.

def compute_delta(h, p, q):
    n = h.shape[0]
    L = 2*n - 1
    delta = np.zeros(L, dtype=np.int16)
    # A
    if p < n:    # always true
        delta[n-1-p : 2*n-1-p] -= h[0:n-p]
    # B
    # reversed slice length n:
    # compute rev = h[q+n-1 : q-1 : -1] if q>0 else h[n-1::-1]
    if q > 0:

    return delta

def random_shake(h, rng, frac=0.05):
    n = len(h)
    k = int(frac*n)
    # Choose k indices from ones to turn to zero, and k from zeros to turn to one.
    ones_idx = np.where(h==1)[0]
    zeros_idx = np.where(h==0)[0]
    if len(ones_idx) < k or len(zeros_idx) < k:
        return h
    p_sel = rng.choice(ones_idx, k, replace=False)
    q_sel = rng.choice(zeros_idx, k, replace=False)
    h[p_sel] = 0
    h[q_sel] = 1
    return h

def binary_hill_climb(h, rng, deadline):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()
    no_improve = 0
    shake_interval = 20000
    while time.time() < deadline:
        # sample swap
        p = rng.integers(n)
        while h[p] == 0:
            p = rng.integers(n)
        q = rng.integers(n)

            q = rng.integers(n)
        delta = compute_delta(h, p, q)

            corr = new_corr
            cur_max = new_max
            no_improve = 0

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact sum correction for a real vector (Σh = n/2)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- gradient refinement ----------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- binary hill‑climb ----------
def compute_delta(h, p, q):
    """Delta of the (1‑h)·h correlation for swapping a 1 at p with a 0 at q."""
    n = h.shape[0]
    L = 2 * n - 1
    delta = np.zeros(L, dtype=np.int16)
    rev = h[::-1]                     # reversed full vector

    # A: -h[p+s]   (length n-p)
    start_a = n - 1 - p
    delta[start_a : start_a + (n - p)] -= h[0 : n - p]

    # B: +h[q‑s]   (full reversed)
    delta[q : q + n] += rev

    # C: -h[p‑s]   (full reversed)
    delta[p : p + n] -= rev

    # D: +h[q+s]   (full forward)
    start_d = n - 1 - q
    delta[start_d : start_d + n] += h

    return delta

def random_shake(h, rng, frac=0.03):
    n = h.shape[0]
    k = max(1, int(frac * n))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) < k or len(zeros) < k:
        return h
    p_sel = rng.choice(ones, k, replace=False)
    q_sel = rng.choice(zeros, k, replace=False)
    h[p_sel] = 0
    h[q_sel] = 1
    return h

def binary_hill_climb(h0, rng, deadline):
    """Intensive swap‑based hill climbing on a binary vector."""
    n = h0.size
    h = h0.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    cur_max = int(corr.max())
    best_max = cur_max
    best_h = h.copy()
    no_improve = 0
    shake_interval = 20000

    while time.time() < deadline:
        # pick a random 1 and a random 0
        p = rng.integers(n)
        while h[p] == 0:
            p = rng.integers(n)
        q = rng.integers(n)
        while h[q] == 1:
            q = rng.integers(n)

        delta = compute_delta(h, p, q)
        new_corr = corr - delta
        new_max = int(new_corr.max())

        if new_max < cur_max:
            # accept swap
            h[p] = 0
            h[q] = 1
            corr = new_corr
            cur_max = new_max
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= shake_interval:
                h = random_shake(h, rng, frac=0.02)
                corr = np.correlate(h, 1 - h, mode="full")
                cur_max = int(corr.max())
                no_improve = 0

    return best_h, best_max

            h_interp = rng.random(target_n)
        idx = np.argpartition(h_interp, -target_n // 2)[-target_n // 2:]
        h_binary = np.zeros(target_n, dtype=np.int8)
        h_binary[idx] = 1

    n = target_n

    # ----- stage 1 : binary hill climb -----
    best_h_bin, best_max_int = binary_hill_climb(h_binary, rng, deadline)
    best_c = best_max_int * (2.0 / n)
    best_h = best_h_bin.astype(np.float64)
    best_n = n

    # early exit if already good enough
    if best_c <= 0.3808 and time.time() < deadline:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        final_c = compute_c5(best_h)
        return best_h.tolist(), float(final_c), int(best_n)

    # ----- stage 2 : continuous refinement -----
    # a few gradient steps before SLSQP
    best_h = gradient_refine(best_h, rng, step=0.02, max_iter=8)
    c = compute_c5(best_h)
    if c < best_c:
        best_c = c
        best_h = best_h.copy()

    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0)
    for beta in betas:
        if time.time() >= deadline:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=1000)
        except Exception:
            continue
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c
            best_h = best_h.copy()

    # final gradient sweep
    if time.time() < deadline:
        best_h = gradient_refine(best_h, rng, step=0.02, max_iter=12)
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c
            best_h = best_h.copy()

    # ----- final safety adjustments -----
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
