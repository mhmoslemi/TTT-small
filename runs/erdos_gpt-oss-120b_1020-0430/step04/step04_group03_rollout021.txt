        else:
            accept = rng.random() < np.exp(-(new_c - cur_c) / T)

        if accept:
            h = nh
            cur_c = new_c
            if new_c < best_c:
                best_h = nh.copy()
                best_c = new_c
                last_improve = time.time()
        iter_cnt += 1

        # occasional restart if stuck
        if time.time() - last_improve > 30 and (budget_s - (time.time() - start)) > 60:
            h = best_h.copy()
            # apply a few random perturbations to shake out of flat region
            for _ in range(8):
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 0:
                    continue
                eps = max_eps * 0.3 * rng.random()
                h[i] -= eps
                h[j] += eps
            cur_c = compute_c5(h)
            last_improve = time.time()

    # ----- deterministic polishing -----
    best_h = greedy_optimize(best_h, max_steps=15000)
    best_c = compute_c5(best_h)

    # ----- final fine‑grained micro‑search (remaining budget) -----
    micro_cut = budget_s * 0.98
    while time.time() - start < micro_cut:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * 0.02 * rng.random()
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        new_c = compute_c5(nh)
        if new_c < best_c:
            best_h = nh
            best_c = new_c

    # ----- final cleanup -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
