    t0 = time.time()

    # Determine base n0 = 360 or transform

        n = h.size

    best_h, best_c = h, compute_c5(h)

    # define beta schedule
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]

    # Main loop for stages: size list = [360, 720, 1440]
    for target_n in [360, 720, 1440]:
        if target_n != n:
            # upscale
            old_x = (np.arange(n) + 0.5) * (2.0 / n)

def compute_c5_corr(h):
    n = h.size
    g = 1 - h
    # reverse g for correlation
    g_rev = g[::-1]
    # compute convolution by FFT
    nfft = 1
    while nfft < 2*n:
        nfft <<= 1
    # Compute FFTs with complex numbers
    H = np.fft.fft(h, nfft)
    G = np.fft.fft(g_rev, nfft)
    conv = np.fft.ifft(H * G).real
    corr = conv[:2*n-1]
    # compute max correlation * (2/n)
    c = np.max(corr) * (2.0 / n)
    return c, corr

def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def compute_c5_corr(h):
    """Return (C5, full correlation array) using FFT for speed."""
    n = h.size
    g = 1.0 - h
    g_rev = g[::-1]
    # power‑of‑two length for FFT
    nfft = 1
    while nfft < 2 * n:
        nfft <<= 1
    H = np.fft.fft(h, nfft)
    G = np.fft.fft(g_rev, nfft)
    conv = np.fft.ifft(H * G).real
    corr = conv[: 2 * n - 1]
    c5 = np.max(corr) * (2.0 / n)
    return c5, corr

def compute_c5(h):
    """Return C5 (max overlap) for vector h."""
    n = h.size
    g = 1.0 - h
    g_rev = g[::-1]
    nfft = 1
    while nfft < 2 * n:
        nfft <<= 1
    H = np.fft.fft(h, nfft)
    G = np.fft.fft(g_rev, nfft)
    conv = np.fft.ifft(H * G).real
    corr = conv[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    """Bias the transfer toward indices that dominate the worst lag."""
    n = h.size
    if lag >= n:
        return perturb_h(h, rng, scale)
    valid = np.arange(n - lag)
    contrib = h[valid] * (1.0 - h[valid + lag])
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    """Time‑aware SA using the exact correlation."""
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.02, 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        # identify worst lag
        lag = np.argmax(cur_corr) - (n - 1)
        if lag < 0:
            lag = -lag
        if rng.random() < 0.6:
            new_h = targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def upsample_h(h, target_n, rng):
    n = h.size
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    new_h = np.interp(new_x, old_x, h)
    new_h = enforce_sum(new_h, target_n / 2.0, rng)
    return new_h

        # allocate time for this grid size

    h_cur = best_h.copy()

    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
