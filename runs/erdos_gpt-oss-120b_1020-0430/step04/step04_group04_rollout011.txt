def compute_c5(h):
    n = h.shape[0]
    rev = (1-h)[::-1]
    L = 2*n-1
    m = 1 << ((L-1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def enforce_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    diff = target - current
    if abs(diff) < 1e-12:
        return h
    # Distribute diff to interior points (0<h<1) equally
    free = (h > 1e-12) & (h < 1-1e-12)
    idx = np.where(free)[0]
    if idx.size > 0:
        per = diff / idx.size
        h[idx] += per
        h = np.clip(h, 0.0, 1.0)
    else:
        # fallback: adjust all entries proportionally
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # final correction if needed (rare)
    cur = h.sum()
    if abs(cur - target) > 1e-9:
        # adjust smallest necessary
        for i in range(len(h)):
            adjust = min(target - cur, 1.0 - h[i]) if target - cur > 0 else max(target - cur, -h[i])
            h[i] += adjust
            cur = h.sum()
            if abs(cur - target) < 1e-12:
                break
    return h

def random_binary_vector(n, rng):
    ones = n // 2
    zeros = n - ones
    arr = np.array([1.0]*ones + [0.0]*zeros)
    rng.shuffle(arr)
    return arr

def simulated_annealing_binary(h0, rng, max_time, start_temp=0.01, end_temp=1e-5):
    start = time.time()
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    n = h0.size
    # compute number of iterations based on time
    # We'll loop until time passes max_time
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= max_time:
            break
        # compute temperature based on elapsed fraction; linear cooling
        fraction = elapsed / max_time
        T = start_temp * (end_temp/start_temp) ** fraction
        # propose swap
        # select random indices i (1) and j (0)
        # ensure we pick appropriate indices
        # To speed up, select until we get such indices
        i = rng.integers(0, n)
        while cur_h[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while cur_h[j] == 1:
            j = rng.integers(0, n)
        # swap
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

def hill_climb_swaps(h0, rng, max_iter=2000):
    h = h0.copy()
    n = h.size
    best_c = compute_c5(h)
    for it in range(max_iter):
        # try random swap that improves
        i = rng.integers(0, n)
        while h[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] == 1:
            j = rng.integers(0, n)
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c = compute_c5(new_h)
        if new_c < best_c:
            h, best_c = new_h, new_c
    return h, best_c

def slsqp_refine(h0, beta, max_iter=300):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    # define closure to compute objective uses global CURRENT_BETA? Better to pass beta through closure.
    # But we need to give to minimize function a function that captures beta.
    # We can define a function inside slsqp_refine (allowed as local function, not closure at top-level? It's okay as inner function).
    # In top-level code we have runs inside function, local functions are fine.
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(obj, h0, method="SLSQP", bounds=bounds, constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # spread diff uniformly over interior points
        free = (x > 1e-12) & (x < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size > 0:
            per = diff / idx.size
            x[idx] += per
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# Core utilities
# -------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for discrete h on [0,2]."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to exact sum=target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    free = (h > 1e-12) & (h < 1.0 - 1e-12)
    idx = np.where(free)[0]
    if idx.size:
        per = diff / idx.size
        h[idx] += per
        h = np.clip(h, 0.0, 1.0)
    else:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = h.sum()
    if abs(cur - target) > 1e-10:
        for i in range(h.size):
            if abs(cur - target) < 1e-12:
                break
            adj = min(target - cur, 1.0 - h[i]) if target - cur > 0 else max(target - cur, -h[i])
            h[i] += adj
            cur = h.sum()
    return h


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones (n even)."""
    ones = n // 2
    zeros = n - ones
    arr = np.array([1.0] * ones + [0.0] * zeros)
    rng.shuffle(arr)
    return arr


# -------------------------------------------------------------
# Stochastic optimisation
# -------------------------------------------------------------
def simulated_annealing_binary(h0: np.ndarray, rng: np.random.Generator,
                               max_time: float, start_temp=0.02,
                               end_temp=5e-5) -> tuple:
    """SA on binary vectors (swap a 1 with a 0)."""
    start = time.time()
    best_h, best_c = h0.copy(), compute_c5(h0)
    cur_h, cur_c = best_h.copy(), best_c
    n = h0.size
    while True:
        now = time.time()
        if now - start >= max_time:
            break
        # linear‑exponential cooling
        frac = (now - start) / max_time
        T = start_temp * (end_temp / start_temp) ** frac
        # pick a random 1‑index and a random 0‑index
        i = rng.integers(0, n)
        while cur_h[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while cur_h[j] == 1:
            j = rng.integers(0, n)
        new_h = cur_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c


def hill_climb_swaps(h0: np.ndarray, rng: np.random.Generator,
                     max_iter: int) -> tuple:
    """Greedy swaps that strictly improve the objective."""
    h = h0.copy()
    best_c = compute_c5(h)
    n = h.size
    for _ in range(max_iter):
        i = rng.integers(0, n)
        while h[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] == 1:
            j = rng.integers(0, n)
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c = compute_c5(new_h)
        if new_c < best_c:
            h, best_c = new_h, new_c
    return h, best_c


def slsqp_refine(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    """SLSQP optimisation of a smooth surrogate (log‑sum‑exp)."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            x[idx] += per
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x

# Main entry point
# -------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # base resolution – enough points to capture fine structure
    n = 720
    target_sum = n / 2.0

    # ---------------------------------------------------------
    # 1. Initialise with a random feasible binary vector
    # ---------------------------------------------------------
    h_best = random_binary_vector(n, rng)
    best_c = compute_c5(h_best)

    # ---------------------------------------------------------
    # 2. Simulated annealing (≈50 % of the budget)
    # ---------------------------------------------------------
    sa_time = budget_s * 0.5
    h_sa, c_sa = simulated_annealing_binary(
        h_best, rng, max_time=sa_time, start_temp=0.02, end_temp=5e-5
    )
    if c_sa < best_c:
        h_best, best_c = h_sa, c_sa

    # ---------------------------------------------------------
    # 3. Greedy hill‑climbing on swaps (≈30 % of the budget)
    # ---------------------------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    hill_time = remaining * 0.4
    # rough conversion to iteration count (≈3 µs per eval on this hardware)
    max_iters = max(2000, int(hill_time / 0.0003))
    h_hc, c_hc = hill_climb_swaps(h_best, rng, max_iter=max_iters)
    if c_hc < best_c:
        h_best, best_c = h_hc, c_hc

    # ---------------------------------------------------------
    # 4. Continuous refinement with SLSQP (few β values)
    # ---------------------------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        for beta in (100, 500, 2000, 5000):
            if time.time() - start >= budget_s:
                break
            h_ref = slsqp_refine(h_best, beta, max_iter=400)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                h_best, best_c = h_ref, c_ref

    # ---------------------------------------------------------
    # 5. Optional up‑sampling and final polish (≈last 10 %)
    # ---------------------------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 30.0:
        target_n = min(1024, n * 2)
        if target_n != n:

            up_h = np.interp(new_x, old_x, h_best)

            # quick hill‑climb on the finer grid

            up_h, c_up = hill_climb_swaps(up_h, rng, max_iter=max_iters_up)
            if c_up < best_c:
                h_best, best_c, n = up_h, c_up, target_n

    # ---------------------------------------------------------
    # 6. Final sanity checks
    # ---------------------------------------------------------
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = enforce_sum(h_best, n / 2.0)
    final_c = compute_c5(h_best)

    return h_best.tolist(), float(final_c), int(n)
