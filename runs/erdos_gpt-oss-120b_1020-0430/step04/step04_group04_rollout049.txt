    L = 2*n - 1
    size_max = max(L, 3*n - 2)
    m = 1 << ((size_max - 1).bit_length())

    rev = (1 - h)[::-1]
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]  # c_i

def project_box_simplex(v, target):
    # v shape (n,)
    x = np.clip(v, 0.0, 1.0)
    deficit = target - x.sum()
    # Iteratively distribute deficit across free variables
    # Use tolerance
    tol = 1e-12
    max_iter = 100
    for _ in range(max_iter):
        if abs(deficit) <= tol:
            break
        free = (x > 1e-12) & (x < 1 - 1e-12)
        free_cnt = free.sum()
        if free_cnt == 0:
            # all at bounds; distribute to nearest bound direction
            # If deficit >0, try to increase some bound that is not at 1
            # but all at bound either 0 or 1, so infeasible. Should not happen.
            break
        adjust = deficit / free_cnt
        # Apply adjustment
        x[free] += adjust
        # Clip again
        # some may exceed bounds; set those to bound and update deficit
        # Snap to bounds
        over = x > 1.0
        under = x < 0.0
        if over.any():
            x[over] = 1.0
        if under.any():
            x[under] = 0.0
        new_deficit = target - x.sum()
        # If deficit didn't change much due to clipping, break
        if abs(new_deficit - deficit) < 1e-12:
            # Could be due to numeric; break
            deficit = new_deficit
            break
        deficit = new_deficit
    # final clipping
    x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np

# ---------- basic helpers -------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def random_feasible_vector(n, rng):
    """Uniform random feasible point (0≤h≤1, sum= n/2)."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # simple scaling to hit the target, then fix bounds
    h *= target / h.sum()
    # ggf still outside bounds after scaling – fix by water‑filling
    return project_box_simplex(h, target)


def project_box_simplex(v, target):
    """Projection onto {0≤x≤1, sum(x)=target}."""
    x = np.clip(v, 0.0, 1.0)
    deficit = target - x.sum()
    tol = 1e-12
    max_iter = 100
    for _ in range(max_iter):
        if abs(deficit) <= tol:
            break
        free = (x > 1e-12) & (x < 1 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = deficit / cnt
        x[free] += delta
        # clip again – newly saturated variables will be removed next loop
        np.clip(x, 0.0, 1.0, out=x)
        deficit = target - x.sum()
    # final safety clip
    np.clip(x, 0.0, 1.0, out=x)
    return x


# ---------- gradient of smooth max ----------------------------------------
def compute_gradient(h, beta):
    """Gradient of (1/β)·log‑sum‑exp of the overlap convolution."""
    n = h.shape[0]
    L = 2 * n - 1
    size = max(L, 3 * n - 2)
    m = 1 << ((size - 1).bit_length())

    # FFT of h and its reverse (1‑h)
    rev = (1 - h)[::-1]
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]          # raw overlaps (no dx)

    # soft‑max weights
    maxc = np.max(conv)
    exps = np.exp(beta * (conv - maxc))
    w = exps / exps.sum()                     # length L
    w_rev = w[::-1]

    # FFTs needed for convolutions with weights
    W = np.fft.rfft(w, m)
    W_rev = np.fft.rfft(w_rev, m)
    H_one = np.fft.rfft(1 - h, m)

    # convolutions (full length L+N-1)
    conv1_full = np.fft.irfft(W_rev * H_one, m)[: (L + n - 1)]
    conv2_full = np.fft.irfft(W * H, m)[: (L + n - 1)]

    start = n - 1
    conv1 = conv1_full[start:start + n]
    conv2 = conv2_full[start:start + n]
    return conv1 - conv2


# ---------- PGD optimizer -------------------------------------------------
def pgd_optimize(h0, beta, max_iter, step0):
    """Projected gradient descent on the smooth surrogate."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    for i in range(max_iter):
        grad = compute_gradient(h, beta)
        step = step0 / np.sqrt(i + 1.0)
        h_new = h - step * grad
        h_new = project_box_simplex(h_new, target)
        if np.max(np.abs(h_new - h)) < 1e-10:
            h = h_new
            break
        h = h_new
    return h

    best = h.copy()
    best_c = compute_c5(best)
    n = h.size
    for _ in range(iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(best[i], 1.0 - best[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        c = compute_c5(cand)
        if c < best_c:
            best, best_c = cand, c
    return best, best_c


# ---------- main driver ---------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ----- starting point ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_simplex(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # schedule of smoothness parameters
    betas = [5, 25, 125, 625, 2500, 5000]
    iters_per_beta = [2000, 2000, 2000, 2000, 2000, 2000]   # generous but limited by time

    # ----- main stochastic search ----------------------------------------
    while time.time() - start_time < budget_s * 0.65:
        # start from a perturbed best or a fresh random point
        if rng.random() < 0.6:
            start = best_h + rng.normal(scale=0.03, size=n)
        else:
            start = random_feasible_vector(n, rng)
        start = project_box_simplex(start, n / 2.0)

        h = start
        # run PGD through the β ladder
        for beta, max_it in zip(betas, iters_per_beta):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 2.0:
                break
            # bound per‑β iterations by remaining budget (very rough)
            use_it = min(max_it, int(remaining * 30))
            if use_it <= 0:
                break
            h = pgd_optimize(h, beta, use_it, step0=1.0)

        # final exact hill‑climb
        h, c = hill_climb_c5(h, rng, iters=1200, scale=0.3)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- optional up‑sampling ------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.2:
        target_n = 720
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = project_box_simplex(up_h, target_n / 2.0)

            # repeat PGD on finer grid (fewer iterations per β)
            fine_betas = [5, 25, 125, 625, 2500, 5000]
            for beta in fine_betas:
                remaining = budget_s - (time.time() - start_time)
                if remaining < 1.0:
                    break
                it_fine = min(1200, int(remaining * 20))
                up_h = pgd_optimize(up_h, beta, it_fine, step0=0.8)

            up_h, c_up = hill_climb_c5(up_h, rng, iters=1500, scale=0.25)
            if c_up < best_c:
                best_c, best_h, n = c_up, up_h.copy(), target_n

    # ----- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_simplex(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
