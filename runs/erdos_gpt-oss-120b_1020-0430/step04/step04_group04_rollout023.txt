def targeted_move(h, idx_max, rng, scale):
    n = len(h)
    k = idx_max - (n-1)
    # compute contributions array
    if k >= 0:
        # shift right: i from 0..n-1-k, h[i+k] exists
        contrib = np.empty(n)
        contrib[:n-k] = h[:n-k] * (1 - h[k:])
        contrib[n-k:] = h[n-k:]  # since shifted h =0
    else:
        # k <0, shift left: i from -k..n-1, h[i+k] exists
        kk = -k
        contrib = np.empty(n)
        contrib[kk:] = h[kk:] * (1 - h[:n-kk])
        contrib[:kk] = h[:kk]
    # now pick i with high contrib and j with low contrib (but also need respecting bounds)
    # get sorted indices
    high_idx = np.argpartition(contrib, -max(1, int(0.02*n)))[-int(0.02*n):]  # top 2%
    low_idx = np.argpartition(contrib, int(0.02*n))[:int(0.02*n)]
    # random choice
    i = rng.choice(high_idx)
    j = rng.choice(low_idx)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

Define compute_c5:

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv.max() * (2.0 / n)
```

def targeted_move(h, idx_max, rng, scale=0.2):
    n = h.shape[0]
    k = idx_max - (n - 1)
    contrib = contribution_for_shift(h, k)
    # sizes for top/bottom fractions
    top_n = max(1, int(0.02 * n))
    low_n = top_n
    # get indices
    high_idx = np.argpartition(contrib, -top_n)[-top_n:]
    low_idx = np.argpartition(contrib, low_n)[:low_n]
    # choose random i/j
    i = int(rng.choice(high_idx))
    j = int(rng.choice(low_idx))
    # compute feasible epsilon
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_improved(h_start, rng, max_iters=2000, step_scale=0.5, p_target=0.5):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    no_imp = 0
    for it in range(max_iters):
        # compute correlation and max idx
        corr = compute_corr(h)
        idx_max = int(corr.argmax())
        if rng.random() < p_target:
            new_h = targeted_move(h, idx_max, rng, scale=step_scale)
        else:
            new_h = perturb_h(h, rng, scale=step_scale)
        new_c = compute_c5(new_h)
        if new_c < cur_c - 1e-12:
            h, cur_c = new_h, new_c
            no_imp = 0
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        else:
            no_imp += 1
        if no_imp > 200:
            # try a random restart from best
            h = perturb_h(best_h, rng, scale=step_scale)
            cur_c = compute_c5(h)
            no_imp = 0
        # decay step scale gradually
        if (it + 1) % (max_iters // 5) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=200):
    n = h0.size
    target = n/2.0
    bounds = [(0.0,1.0)]*n
    # define objective as smooth max using log-sum-exp
    def obj(x):
        corr = np.correlate(x, 1-x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m))))/beta
    def eq(x):
        return np.sum(x) - target
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints={'type':'eq','fun':eq},
                   options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0.0,1.0)
    # enforce sum again
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple correction
        # distribute diff uniformly
        x += diff / n
        x = np.clip(x,0.0,1.0)
    return x

import numpy as np

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h to have sum == target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv.max() * (2.0 / n)


def compute_corr(h):
    """Correlation array of h with (1-h)."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv


def contribution_for_shift(h, k):
    """Elementwise contributions h[i]*(1-h[i+k]) for a given shift k."""
    n = h.shape[0]
    if k >= 0:
        contrib = np.empty(n)
        contrib[:n - k] = h[:n - k] * (1 - h[k:])
        contrib[n - k:] = h[n - k:]  # outside range: (1-h)=1
    else:
        kk = -k
        contrib = np.empty(n)
        contrib[kk:] = h[kk:] * (1 - h[:n - kk])
        contrib[:kk] = h[:kk]
    return contrib


def targeted_move(h, idx_max, rng, scale=0.2):
    """Move mass from a high‑contribution site (for the worst shift) to a low‑contribution site."""
    n = h.shape[0]
    k = idx_max - (n - 1)
    contrib = contribution_for_shift(h, k)
    top_n = max(1, int(0.02 * n))
    low_n = top_n
    high_idx = np.argpartition(contrib, -top_n)[-top_n:]
    low_idx = np.argpartition(contrib, low_n)[:low_n]
    i = int(rng.choice(high_idx))
    j = int(rng.choice(low_idx))
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_improved(h_start, rng, max_iters=2000, step_scale=0.5,
                        p_target=0.4, p_swap=0.2):
    """Monotone hill‑climb using random, swap and targeted moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    no_imp = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        idx_max = int(corr.argmax())
        r = rng.random()
        if r < p_target:
            new_h = targeted_move(h, idx_max, rng, scale=step_scale)
        elif r < p_target + p_swap:
            i, j = rng.choice(n, size=2, replace=False)
            new_h = h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
        else:
            new_h = perturb_h(h, rng, scale=step_scale)
        new_c = compute_c5(new_h)
        if new_c < cur_c - 1e-12:
            h, cur_c = new_h, new_c
            no_imp = 0
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        else:
            no_imp += 1
        if no_imp > 300:
            h = perturb_h(best_h, rng, scale=step_scale)
            cur_c = compute_c5(h)
            no_imp = 0
        if (it + 1) % (max_iters // 5) == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h0, rng, max_iter=3000, init_temp=1e-2, final_temp=1e-6,
                        init_scale=0.5, final_scale=0.02):
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new_h = perturb_h(cur_h, rng, scale=scale)

            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=200):
    """Smooth surrogate optimisation using SLSQP."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def eq(x):
        return np.sum(x) - target
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints={'type': 'eq', 'fun': eq},
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x


def make_spike(n, rng, w_ratio=0.12):
    """Single narrow spike of height 1."""
    w = max(1, int(w_ratio * n))
    a = (1 - w_ratio) / (2 - w_ratio)  # initial background level
    h = np.full(n, a)
    h[:w] = 1.0
    h = enforce_sum(h, n / 2.0, rng)
    return h


def make_two_spikes(n, rng, w_ratio=0.10, gap_ratio=0.10):
    """Two spikes separated by a gap."""
    w = max(1, int(w_ratio * n))
    gap = max(1, int(gap_ratio * n))
    a = (1 - 2 * w_ratio) / (2 - 2 * w_ratio) if 2 * w_ratio < 1 else 0.0
    h = np.full(n, a)
    h[:w] = 1.0
    start2 = w + gap
    end2 = start2 + w
    if end2 <= n:
        h[start2:end2] = 1.0
    else:
        h[start2:] = 1.0
    h = enforce_sum(h, n / 2.0, rng)
    return h

# Main entry point
# --------------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Base dimension and initial vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ------------------------------------------------------------------
    # Build a diverse pool of candidates
    # ------------------------------------------------------------------
    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    # triangular bump (integrates to 1)
    tri = np.where(xs <= 1, xs, 2 - xs)
    candidates.append(enforce_sum(tri, n / 2.0, rng))

    # cosine bump
    cosine = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    candidates.append(enforce_sum(cosine, n / 2.0, rng))

    # linear ramps
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1 - xs / 2.0, n / 2.0, rng))

    # sinusoidal phases
    for f in (1, 2, 3):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(s, n / 2.0, rng))
        candidates.append(enforce_sum(c, n / 2.0, rng))

    # spike‑based shapes
    candidates.append(make_spike(n, rng, w_ratio=0.12))
    candidates.append(make_two_spikes(n, rng, w_ratio=0.10, gap_ratio=0.10))

    # random feasible vectors
    for _ in range(6):
        candidates.append(random_feasible_vector(n, rng))

    # ------------------------------------------------------------------
    # Pick the best of the initial pool
    # ------------------------------------------------------------------
    best_h = None
    best_c = np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop (improved hill‑climb, SA, SLSQP)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.70:
        # start from a perturbed best or a fresh random vector
        if rng.random() < 0.8:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        # improved hill‑climb
        h_hc, c_hc = hill_climb_improved(start_vec, rng,
                                         max_iters=1500,
                                         step_scale=0.4,
                                         p_target=0.5,
                                         p_swap=0.2)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # occasional simulated annealing
        if rng.random() < 0.25:
            h_sa, c_sa = simulated_annealing(best_h, rng,
                                             max_iter=2500,
                                             init_temp=5e-3,
                                             final_temp=1e-6,
                                             init_scale=0.4,
                                             final_scale=0.02)
            if c_sa < best_c:
                best_c, best_h = c_sa, h_sa.copy()

        # SLSQP refinement with increasing beta
        for beta in (200, 800, 2000):
            h_ref = slsqp_optimize(best_h, beta, max_iter=200)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # ------------------------------------------------------------------
    # Optional up‑sampling and fine‑tuning if time permits
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.15:
        n2 = min(720, n * 2)
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(n2) + 0.5) * (2.0 / n2)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, n2 / 2.0, rng)

        up_h, c_up = hill_climb_improved(up_h, rng,
                                         max_iters=1200,
                                         step_scale=0.3,
                                         p_target=0.5,
                                         p_swap=0.2)
        if c_up < best_c:
            best_c, best_h, n = c_up, up_h.copy(), n2

        # final SLSQP passes on the refined grid
        for beta in (200, 800, 2000):
            h_ref = slsqp_optimize(best_h, beta, max_iter=200)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()
                n = best_h.size

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
