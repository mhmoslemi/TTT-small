import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate for max correlation."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact C5 for a binary / step vector."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv.max() * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to reach exact sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_binary_vector(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h, cur_c = best_h.copy(), best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def build_h_from_lengths(one_lengths, zero_lengths, N, xs):
    """Create step vector from alternating lengths (measure units)."""
    h = np.zeros(N)
    pos = 0.0
    K = len(one_lengths)
    for i in range(K):
        start, end = pos, pos + one_lengths[i]
        h[(xs >= start) & (xs < end)] = 1.0
        pos = end
        # add following zero interval (including trailing zero after last block)
        pos += zero_lengths[i]
    return h


# ----------------------------------------------------------------------
# Main routine
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # --------------------------------------------------------------
    # Base grid size
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        N = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, N / 2.0, rng)
    else:
        N = 720
        h0 = random_binary_vector(N, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # --------------------------------------------------------------
    # Small pool of handcrafted candidates
    # --------------------------------------------------------------
    xs = (np.arange(N) + 0.5) * (2.0 / N)
    cand = []

    # triangular bump
    tri = np.where(xs <= 1, xs, 2 - xs)
    cand.append(enforce_sum(tri, N / 2.0, rng))

    # cosine bump
    cos = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    cand.append(enforce_sum(cos, N / 2.0, rng))

    # linear ramps
    cand.append(enforce_sum(xs / 2.0, N / 2.0, rng))
    cand.append(enforce_sum(1 - xs / 2.0, N / 2.0, rng))

    # random binaries
    for _ in range(8):
        cand.append(random_binary_vector(N, rng))

    for c in cand:
        cval = compute_c5(c)
        if cval < best_c:
            best_c, best_h = cval, c.copy()

    # --------------------------------------------------------------
    # Random interval search (2–4 one intervals)
    # --------------------------------------------------------------
    interval_budget = budget_s * 0.25
    while time.time() - start_time < interval_budget:
        K = rng.integers(2, 5)  # number of one‑interval blocks
        one_len = rng.dirichlet(np.ones(K))
        zero_len = rng.dirichlet(np.ones(K))
        h = build_h_from_lengths(one_len, zero_len, N, xs)
        h = enforce_sum(h, N / 2.0, rng)
        cval = compute_c5(h)
        if cval < best_c:
            best_c, best_h = cval, h.copy()

    # --------------------------------------------------------------
    # Main optimisation loop (hill‑climb → SA → SLSQP)
    # --------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.75:
        # hill‑climb
        h_hc, c_hc = hill_climb_multi(best_h, rng, max_iters=4000, init_step=0.4, n_trials=8)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # simulated annealing
        h_sa, c_sa = simulated_annealing(
            best_h,
            rng,
            max_iter=12000,
            init_temp=5e-3,
            final_temp=1e-6,
            init_scale=0.3,
            final_scale=0.02,
        )
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        # SLSQP ladder with increasing smoothness
        for beta in (50, 200, 800, 2000, 5000, 10000):
            if time.time() - start_time > budget_s * 0.75:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=600)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # --------------------------------------------------------------
    # Optional up‑sampling and refinement if time permits
    # --------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.1:
        target_N = min(1440, N * 2)
        if target_N != N:
            old_xs = xs
            new_xs = (np.arange(target_N) + 0.5) * (2.0 / target_N)
            up_h = np.interp(new_xs, old_xs, best_h)
            up_h = enforce_sum(up_h, target_N / 2.0, rng)

            # refinement on finer grid
            up_h, _ = hill_climb_multi(up_h, rng, max_iters=3000, init_step=0.3, n_trials=6)
            up_h, _ = simulated_annealing(
                up_h,
                rng,
                max_iter=6000,
                init_temp=2e-3,
                final_temp=1e-6,
                init_scale=0.3,
                final_scale=0.02,
            )
            for beta in (200, 800, 2000, 5000, 10000):
                if time.time() - start_time > budget_s:
                    break
                up_h = slsqp_optimize(up_h, beta, max_iter=500)
                c_up = compute_c5(up_h)
                if c_up < best_c:
                    best_c, best_h, N = c_up, up_h.copy(), target_N
            xs = new_xs

    # --------------------------------------------------------------
    # Final clean‑up and return
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, N / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(N)
