def correlation_fft(h):
    n = h.size
    m = 1 << (2 * n - 1).bit_length()  # next power of 2 >= 2n-1
    h_fft = np.fft.rfft(h, n=m)
    g = 1 - h
    g_fft = np.fft.rfft(g, n=m)
    prod = h_fft * g_fft
    conv = np.fft.irfft(prod, n=m)
    return conv[:2*n-1]  # length 2n-1

def softmax_gradient(h, beta):
    n = h.size
    C = correlation_fft(h)
    # compute weights
    # to avoid overflow, use stable exponent: subtract max before exp
    maxC = np.max(C)
    exp_terms = np.exp(beta * (C - maxC))
    sum_exp = np.sum(exp_terms)
    w = exp_terms / sum_exp  # shape 2n-1
    # compute sum_plus and sum_minus via FFT
    m = 1 << ((2*n-1).bit_length())
    # compute FFT of w and w_rev
    w_fft = np.fft.rfft(w, n=m)
    w_rev_fft = np.fft.rfft(w[::-1], n=m)
    h_fft = np.fft.rfft(h, n=m)
    conv_rev_full = np.fft.irfft(w_rev_fft * h_fft, n=m)
    conv_full = np.fft.irfft(w_fft * h_fft, n=m)
    conv_rev = conv_rev_full[:2*n-1]
    conv = conv_full[:2*n-1]
    sum_plus = conv_rev[n-1:n-1+n]
    sum_minus = conv[n-1:n-1+n]
    grad = 1.0 - (sum_plus + sum_minus)
    # compute objective value f = (maxC + log(sum_exp))/beta
    # Actually f = (1/beta) * (maxC + log(sum_exp)) + maybe maxC?
    # Derivation: f = (1/beta) * log(sum_i exp(beta*C_i))
    # Using stable: log(sum exp(beta*C)) = maxC + log(sum_exp)
    f_val = (maxC + np.log(sum_exp)) / beta
    return grad, f_val, C

def proj_to_simplex_box(v, target):

def correlation_fft(h):
    n = h.size
    m = 1 << ((2 * n - 1).bit_length())
    h_fft = np.fft.rfft(h, n=m)
    g_fft = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(h_fft * g_fft, n=m)
    return conv[:2*n-1]

def softmax_gradient(h, beta):

    C = correlation_fft(h)  # length 2n-1

    # stable weights

import time
import numpy as np


def _project_to_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x|0≤x≤1, sum x = target} using bisection."""
    lo = -np.max(v) - 1.0          # ensures all v+lo ≤ 0 → sum = 0
    hi = 2.0 - np.min(v)           # ensures all v+hi ≥ 1 → sum = len(v)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return _project_to_box_simplex(v, n / 2.0)


def _correlation_fft(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.size
    m = 1 << ((2 * n - 1).bit_length())
    hf = np.fft.rfft(h, n=m)
    gf = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(hf * gf, n=m)
    return conv[: 2 * n - 1]


def _softmax_gradient(h: np.ndarray, beta: float):
    """Gradient of (1/β)·log ∑ exp(β·C) where C is the correlation vector."""
    n = h.size
    C = _correlation_fft(h)            # length 2n‑1
    maxC = np.max(C)
    # stable weights
    w = np.exp(beta * (C - maxC))
    w_sum = w.sum()
    w /= w_sum

    m = 1 << ((2 * n - 1).bit_length())
    w_fft = np.fft.rfft(w, n=m)
    w_rev_fft = np.fft.rfft(w[::-1], n=m)
    h_fft = np.fft.rfft(h, n=m)

    conv_rev_full = np.fft.irfft(w_rev_fft * h_fft, n=m)
    conv_full = np.fft.irfft(w_fft * h_fft, n=m)

    conv_rev = conv_rev_full[: 2 * n - 1]
    conv = conv_full[: 2 * n - 1]

    start = n - 1
    sum_plus = conv_rev[start : start + n]   # Σ w_L h_{i+L}
    sum_minus = conv[start : start + n]      # Σ w_L h_{i‑L}
    grad = 1.0 - (sum_plus + sum_minus)

    f_val = (maxC + np.log(w_sum)) / beta
    return grad, f_val, C


def _gradient_descent_optimize(
    h: np.ndarray,
    beta_list,
    iters_per_beta: int,
    rng: np.random.Generator,
    time_limit: float,
):
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = np.max(_correlation_fft(h)) * (2.0 / n)

    for beta in beta_list:
        step = 0.5 / np.sqrt(beta + 1.0)
        for _ in range(iters_per_beta):
            if time.time() > time_limit:
                return best_h, best_c
            grad, _, C = _softmax_gradient(h, beta)
            v = h - step * grad
            h = _project_to_box_simplex(v, target)
            cur_c = np.max(C) * (2.0 / n)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            step *= 0.9999
    return best_h, best_c
