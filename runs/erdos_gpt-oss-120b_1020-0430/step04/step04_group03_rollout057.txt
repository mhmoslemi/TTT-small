def project_capped_simplex(v, target):
    n = v.size
    low = v.min() - 1.0  # ensures sum > target
    high = v.max()       # ensures sum < target
    for _ in range(60):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    proj = np.clip(v - ((low + high) / 2.0), 0.0, 1.0)
    # adjust minor residual
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # distribute diff to non-bound elements
        idx = np.where((proj > 0) & (proj < 1))[0]
        if idx.size > 0:
            # add diff equally to all
            add = diff / idx.size
            proj[idx] += add
            proj = np.clip(proj, 0.0, 1.0)
    return proj
```

def softmax_obj_and_grad(h, beta):
    f = np.correlate(h, 1 - h, mode='full')
    f_max = f.max()
    exp_f = np.exp(beta * (f - f_max))
    sum_exp = exp_f.sum()
    w = exp_f / sum_exp
    # gradient
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    start = h.size - 1
    end = start + h.size
    grad = -(conv1[start:end] + conv2[start:end])
    # objective
    obj = (np.log(sum_exp) + beta * f_max) / beta
    return obj, grad
```

import time, numpy as np

def project_capped_simplex(v, target, max_iter=60):
    low = v.min() - 1.0
    high = v.max()
    # bisection
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    λ = (low + high) / 2.0
    proj = np.clip(v - λ, 0.0, 1.0)
    # adjust residual
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        idx = np.where((proj > 0) & (proj < 1))[0]
        if idx.size > 0:
            add = diff / idx.size
            proj[idx] += add
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_capped_simplex(v, n/2.0)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

def softmax_obj_and_grad(h, beta):
    f = np.correlate(h, 1 - h, mode='full')
    f_max = f.max()
    exp_f = np.exp(beta * (f - f_max))
    sum_exp = exp_f.sum()
    w = exp_f / sum_exp
    # gradient via convolutions
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    start = h.size - 1
    end = start + h.size
    grad = -(conv1[start:end] + conv2[start:end])
    obj = (np.log(sum_exp) + beta * f_max) / beta
    return obj, grad

def pgd_optimize(h0, beta, max_iters, lr, target_sum, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_obj, _ = softmax_obj_and_grad(h, beta)
    no_improve = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        obj, grad = softmax_obj_and_grad(h, beta)

            best_h = h.copy()
            no_improve = 0

        h = project_capped_simplex(y, target_sum)
    # final evaluate best_h (ensure feasible)
    best_h = project_capped_simplex(best_h, target_sum)
    return best_h

import time, numpy as np

def project_capped_simplex(v, target, max_iter=60):
    low = v.min() - 1.0
    high = v.max()
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    proj = np.clip(v - lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        idx = np.where((proj > 0) & (proj < 1))[0]
        if idx.size:
            add = diff / idx.size
            proj[idx] += add
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_capped_simplex(v, n / 2.0)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

def softmax_obj_and_grad(h, beta):
    f = np.correlate(h, 1 - h, mode='full')
    f_max = f.max()
    exp_f = np.exp(beta * (f - f_max))
    sum_exp = exp_f.sum()
    w = exp_f / sum_exp
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    start = h.size - 1
    end = start + h.size
    grad = -(conv1[start:end] + conv2[start:end])
    obj = (np.log(sum_exp) + beta * f_max) / beta
    return obj, grad

def pgd_optimize(h0, beta, max_iters, lr, target_sum, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_obj, _ = softmax_obj_and_grad(h, beta)
    no_improve = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        obj, grad = softmax_obj_and_grad(h, beta)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 200:
            break
        y = h - lr * grad
        h = project_capped_simplex(y, target_sum)
    best_h = project_capped_simplex(best_h, target_sum)
    return best_h
