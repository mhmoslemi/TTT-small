def pair_transfer_opt(h, rng, time_limit):
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = 0.5
    decay = 0.995
    steps = 0
    # initial compute
    correlation = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(correlation))
    shift = idx - (n - 1)
    max_corr = correlation[idx]
    while time.time() - start < time_limit:
        # maybe recompute every 30 steps
        if steps % 30 == 0:
            correlation = np.correlate(h, 1.0 - h, mode='full')
            idx = int(np.argmax(correlation))
            shift = idx - (n - 1)
            max_corr = correlation[idx]
            cur_c = max_corr * (2.0 / n)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        # compute derivative for this shift
        s = shift
        if s >= 0:
            d = np.empty(n, dtype=float)
            d[:n-s] = 1.0 - h[s:]
            d[s:] -= h[:n-s]
        else:
            t = -s
            d = np.empty(n, dtype=float)
            d[t:] = 1.0 - h[:n-t]
            d[:n-t] -= h[t:]
        # find candidates
        # indices where we can decrease (h>0) and d>0
        dec_candidates = np.where((d > 1e-12) & (h > 1e-12))[0]
        inc_candidates = np.where((d < -1e-12) & (h < 1.0 - 1e-12))[0]
        if dec_candidates.size == 0 or inc_candidates.size == 0:
            # random transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = dec_candidates[np.argmax(d[dec_candidates])]
            j = inc_candidates[np.argmin(d[inc_candidates])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            # try random
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                steps += 1
                continue
        eps = factor * max_eps
        if eps < 1e-12:
            eps = max_eps * 0.5
        h[i] -= eps
        h[j] += eps
        # clip
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        # optional periodic enforce sum to correct floating drift
        if steps % 200 == 0:
            target = n / 2.0
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                # distribute diff uniformly
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
        factor = max(1e-4, factor * decay)
        steps += 1
    # final check
    final_c = compute_c5(best_h)
    return best_h, final_c

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth surrogate for SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) using log-sum-exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target (0≤h≤1)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h with Σh = n/2 and 0≤h≤1."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(2000):
            if abs(diff) < 1e-14:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def slsqp_optimize(h0, beta, max_iter=150):
    """Refine h using SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def max_corr_shift(h):
    """Return (shift, max_corr) for the current h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (h.size - 1)
    return shift, corr[idx]


def pair_transfer_opt(h0, rng, time_limit):
    """Greedy mass‑transfer targeting the worst shift."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = 0.5
    decay = 0.995
    steps = 0

    # initial worst shift
    shift, max_corr = max_corr_shift(h)

    while time.time() - start < time_limit:
        # periodic recompute of worst shift
        if steps % 30 == 0:
            shift, max_corr = max_corr_shift(h)
            cur_c = max_corr * (2.0 / n)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()

        s = shift
        # derivative of f_s w.r.t each h_i (increase direction)
        if s >= 0:
            d = np.empty(n, dtype=float)
            d[: n - s] = 1.0 - h[s:]
            d[s:] -= h[: n - s]
        else:
            t = -s
            d = np.empty(n, dtype=float)
            d[t:] = 1.0 - h[: n - t]
            d[: n - t] -= h[t:]

        # candidates for decreasing / increasing
        dec = np.where((d > 1e-12) & (h > 1e-12))[0]
        inc = np.where((d < -1e-12) & (h < 1.0 - 1e-12))[0]

        if dec.size == 0 or inc.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = dec[np.argmax(d[dec])]
            j = inc[np.argmin(d[inc])]

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                steps += 1
                continue

        eps = factor * max_eps
        if eps < 1e-12:
            eps = max_eps * 0.5

        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))

        # occasional re‑normalisation to avoid drift
        if steps % 200 == 0:
            target = n / 2.0
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)

        factor = max(1e-4, factor * decay)
        steps += 1

    final_c = compute_c5(best_h)
    return best_h, final_c
