def targeted_perturb_worst_lag(h, lag, rng, max_factor=0.5):
    n = h.shape[0]
    if lag >=0:
        i = rng.integers(0, n-lag)
        j = i + lag
    else:
        lag_abs = -lag
        i = rng.integers(lag_abs, n)  # i > lag_abs - 1
        j = i - lag_abs
    # Ensure we move from lower to higher
    if h[i] > h[j]:
        i, j = j, i
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * max_factor
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing(h_start, rng, time_start, time_budget):
    h = h_start.copy()
    cur_corr = compute_corr_fft(h)
    cur_c = np.max(cur_corr)
    best_h, best_c = h.copy(), cur_c
    n = h.shape[0]
    # Determine max_iters based on time budget maybe 100000?
    max_iters = 200000  # large.
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = 0.02 * (0.001/0.02) ** (it / max_iters)  # exponential drop.

        idx = np.argmax(cur_corr)
        lag = idx - (n - 1)
        # choose perturb

            new_h = targeted_perturb_worst_lag(h, lag, rng, max_factor=0.5)
        else:

    """Linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    n_fft = 2 * n
    H = np.fft.rfft(h, n=n_fft)
    R = np.fft.rfft(rev, n=n_fft)
    conv = np.fft.irfft(H * R, n=n_fft)
    corr = conv[: 2 * n - 1]
    return corr

def perturb_h(h, rng, scale=0.5):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()

    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

            # optional single SLSQP refinement (lightweight)
            try:
                ref_h = slsqp_optimize_one_beta(best_h, 800, rng, max_iter=100)
                ref_c = compute_c5(ref_h)
                if ref_c < best_c:
                    best_h, best_c = ref_h, ref_c
            except Exception:
                pass

    # ----- final greedy hill‑climb -----
    cur_h = best_h.copy()
    cur_c = best_c
    step = 0.3
    while time.time() < t_end:
        new_h = perturb_h(cur_h, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        step *= 0.9995
        if step < 1e-5:
            step = 0.3

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
