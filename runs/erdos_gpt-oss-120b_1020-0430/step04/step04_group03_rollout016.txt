def compute_c5(h, fft_n=None):
    n = h.size
    if fft_n is None:
        fft_n = 1 << ((2*n-1).bit_length())
    # compute correlation via fft
    rev = np.flip(1.0 - h)
    # zero pad both to fft_n
    H = np.fft.rfft(h, fft_n)
    R = np.fft.rfft(rev, fft_n)
    conv = np.fft.irfft(H * R, fft_n)
    # conv length fft_n, we need indices 0..2n-2
    conv = conv[:2*n-1]
    # max
    max_corr = conv.max()
    return max_corr * (2.0 / n)

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple redistribution loop
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# fast C5 using FFT
def compute_c5(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    rev = np.flip(1.0 - h)
    H = np.fft.rfft(h, N)
    R = np.fft.rfft(rev, N)
    conv = np.fft.irfft(H * R, N)[: 2 * n - 1]
    return conv.max() * (2.0 / n)

def worst_shift_info(h):
    n = h.size
    N = 1 << ((2 * n - 1).bit_length())
    rev = np.flip(1.0 - h)
    H = np.fft.rfft(h, N)
    R = np.fft.rfft(rev, N)
    conv = np.fft.irfft(H * R, N)[: 2 * n - 1]
    idx = int(conv.argmax())
    s = idx - (n - 1)
    return s, idx, conv[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    """Greedy deterministic improvement on the worst shift."""
    for _ in range(max_steps):
        s, _, cur_val = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

# ---------- SLSQP polishing ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- Simulated annealing ----------
def anneal_optimize(h0, time_limit, start_time, rng):
    n = h0.size
    target = n / 2.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    T0 = 0.006  # initial temperature
    while time.time() < time_limit:
        elapsed = time.time() - start_time
        frac = (elapsed) / (time_limit - start_time + 1e-12)
        T = T0 * (1.0 - frac)  # linear cooling
        scale = 0.2 * (1.0 - frac) + 0.01
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * scale * rng.random()
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        c_new = compute_c5(nh)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-18)):
            cur_h, cur_c = nh, c_new
            if c_new < best_c:
                best_h, best_c = nh.copy(), c_new
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 720  # fixed resolution

    # initial vector

    # Phase 2: simulated annealing
    remain = budget_s - (time.time() - start)

    best_h = enforce_sum(best_h, n / 2.0, rng)
