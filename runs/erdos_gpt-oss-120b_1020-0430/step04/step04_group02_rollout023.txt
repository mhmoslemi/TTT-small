def pgd_optimize(h0, beta, max_iter=400, step=0.5):
    n = len(h0)
    h = h0.copy()
    k_vals = np.arange(-(n-1), n)
    i_idx = np.arange(n)[:, None]
    plus_idx = (i_idx + k_vals) % n
    minus_idx = (i_idx - k_vals) % n
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Softmax weights
        max_bc = np.max(beta * corr)
        w = np.exp(beta * corr - max_bc)
        w_sum = np.sum(w)
        w /= w_sum
        # Compute gradient
        # w shape (2n-1,)
        w_row = w[None, :]  # (1, 2n-1)
        h_plus = h[plus_idx]  # (n, 2n-1)
        h_minus = h[minus_idx]
        grad = np.sum(w_row * (1.0 - h_plus - h_minus), axis=1)
        grad *= (2.0 / n)  # scale factor
        # step update
        h_new = h - step * grad
        # projection
        h_new = project_box_sum(h_new, n / 2.0)
        # convergence
        if np.max(np.abs(h_new - h)) < 1e-12:
            break
        h = h_new
        step *= 0.99
    return h

import time
import numpy as np

# ---------- helpers ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i_low] += eps
    nh[i_low + k] += eps
    nh[i_high] -= eps
    nh[i_high + k] -= eps
    return nh

def hill_climb_targeted(h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]
    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return h, best_c

# ---------- PGD on smooth max ----------
def pgd_optimize(h0, beta, max_iter=200, step=0.5):
    n = h0.size
    h = h0.copy()
    s_vals = np.arange(-(n - 1), n)
    mod_idx = np.mod(s_vals, n)
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        w = np.exp(beta * (corr - max_corr))
        w /= w.sum()
        w_mod = np.bincount(mod_idx, weights=w, minlength=n)
        fft_h = np.fft.fft(h)
        fft_w = np.fft.fft(w_mod)
        conv_minus = np.real(np.fft.ifft(fft_h * fft_w))
        w_rev = w_mod[::-1]
        fft_w_rev = np.fft.fft(w_rev)
        conv_plus = np.real(np.fft.ifft(fft_h * fft_w_rev))
        grad = 1.0 - conv_plus - conv_minus
        grad *= (2.0 / n)
        h_new = h - step * grad
        h_new = project_box_sum(h_new, n / 2.0)
        if np.max(np.abs(h_new - h)) < 1e-12:
            h = h_new
            break
        h = h_new
        step *= 0.995
    return h
