def subgradient_descent(h, rng, max_iter, step0, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # compute correlation and shift
        corr = np.correlate(h, 1-h, mode='full')
        # naive correlation (size 2n-1)
        # find max
        idx = np.argmax(corr)
        k = idx - (n - 1)  # shift
        # compute gradient
        grad = np.zeros_like(h)
        # for i where i + k within range
        if k >= 0:
            grad[:n-k] = 1 - h[k:]
            grad[k:] -= h[:n-k]
        else:
            k2 = -k
            grad[k2:] = 1 - h[:n-k2]
            grad[:n-k2] -= h[k2:]
        # update
        # step size decays
        step = step0 / np.sqrt(it+1)
        new_h = h - step * grad
        # project onto [0,1] via clipping
        new_h = np.clip(new_h, 0.0, 1.0)
        # enforce sum
        new_h = enforce_sum(new_h, n/2.0, rng)
        h = new_h
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # occasional random perturb to escape local minima
        if rng.random() < 0.01:
            h = perturb_h(h, rng, scale=0.1)
    return best_h, best_c
```

def compute_c5(h):

    rev = (1 - h)[::-1]
    L = 2 * n - 1

    H = np.fft.rfft(h, m)

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Find indices that can be adjusted in needed direction
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        if not np.any(free):
            break
        # distribute diff equally among free indices
        per = diff / free.sum()
        # ensure we don't exceed bounds drastically
        adjust = np.minimum(per, 1.0 - h) if diff > 0 else np.maximum(per, -h)
        # apply
        h[free] += adjust[free]
        diff = target - h.sum()
    # final clip and adjust small residual
    h = np.clip(h, 0.0, 1.0)
    # small residual fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute arbitrarily
        idx = np.arange(n)
        for i in idx:
            if abs(diff) < 1e-12:
                break
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def subgradient_descent(h0, rng, max_iter=1500, step0=0.5, time_limit=None):
    h = h0.copy()
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iter):
        if time_limit is not None and time.time() > time_limit:
            break
        # correlation via FFT
        conv = full_corr(h)
        idx = np.argmax(conv)
        k = idx - (n - 1)
        # gradient
        grad = np.zeros_like(h)
        if k >= 0:
            i = np.arange(n - k)
            j = i + k
        else:
            i = np.arange(-k, n)
            j = i + k
        grad[i] += 1 - h[j]
        grad[j] += -h[i]
        # step size decays
        step = step0 / np.sqrt(it + 1)
        new_h = h - step * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target, rng)
        h = new_h
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # occasional random perturb
        if rng.random() < 0.005:
            h = perturb_h(h, rng, scale=0.1)
    return best_h, best_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # fix sum drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple correction
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np

# ----------- Global parameters for the smooth surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint for SLSQP: sum(x) == CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k correlation using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def full_corr(h):
    """Full correlation array without the dx scaling."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector with Σh = n/2."""
    h = rng.random(n)
    h = h / h.sum() * (n / 2.0)
    return enforce_sum(h, n / 2.0, rng)


def slsqp_optimize(h0, beta, max_iter=200):
    """SLSQP refinement of the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    from scipy.optimize import minimize

    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def subgradient_descent(h0, rng, max_iter=1500, step0=0.5, time_limit=None):
    """Projected sub‑gradient descent on the true max‑overlap objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iter):
        if (time_limit is not None) and (time.time() > time_limit):
            break
        conv = full_corr(h)
        idx = np.argmax(conv)
        k = idx - (n - 1)  # shift giving the maximum
        # gradient of the selected correlation term
        grad = np.zeros_like(h)
        if k >= 0:
            i = np.arange(n - k)
            j = i + k
        else:
            i = np.arange(-k, n)
            j = i + k
        grad[i] += 1.0 - h[j]
        grad[j] += -h[i]
        # step size decay
        step = step0 / np.sqrt(it + 1)
        new_h = h - step * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target, rng)
        h = new_h
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # occasional random shake
        if rng.random() < 0.005:
            h = perturb_h(h, rng, scale=0.1)
    return best_h, best_c

    """
    Search for a step function h on [0,2] with a low Erdős C₅ bound.
    Returns (list of h values, C5 bound, number of points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # choose resolution
    n = 720 if budget_s > 800 else 360
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    # ----- generate initial pool of candidates -----
    candidates = []

    # deterministic shapes

    candidates.append(enforce_sum(tri, n / 2.0, rng))
    cos = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    candidates.append(enforce_sum(cos, n / 2.0, rng))

    # include parent construction if supplied

        h_parent = np.array(initial_h_values, dtype=float)

        candidates.append(h_parent)

    # low‑dimensional Fourier family
    K = 4
    X = np.column_stack([np.cos(k * np.pi * (xs - 1)) for k in range(1, K + 1)])
    for _ in range(30):
        a = rng.uniform(-0.45, 0.45, K)
        h = 0.5 + X.dot(a)

    while time.time() - start < budget_s * 0.9:
        # sub‑gradient descent
        limit = start + budget_s * 0.85

                                         step0=0.5, time_limit=limit)
