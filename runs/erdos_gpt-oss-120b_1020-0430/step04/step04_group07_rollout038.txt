def greedy_swap(h, shift, rng):
    n = h.size
    if shift >= 0:
        limit = n - shift
        hi = h[:limit]
        hshift = h[shift:]
        mismatched = (hi == 1) & (hshift == 0)
        good = (hi == 0) & (hshift == 1)
        mis_idx = np.nonzero(mismatched)[0]
        good_idx = np.nonzero(good)[0]
        if mis_idx.size == 0 or good_idx.size == 0:
            return False, h
        i = rng.choice(mis_idx)
        j = rng.choice(good_idx)
        # swap
        new_h = h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        # note: i and j are within same range; ensure distinct; they can be equal? They cannot because mismatched implies h[i]=1 and good implies h[j]=0. So distinct.
        return True, new_h
    else:
        t = -shift
        limit = n - t

def gradient_refine(h, rng, step=0.015, max_iter=10):
    n = h.size
    for it in range(max_iter):
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if idx >= 0:  # shift >=0
            # O_k = sum_{i=0}^{n-1-k} h_i*(1 - h_{i+k})
            # derivative w.r.t h_i for i in [0, n-1-k] includes (1 - h_{i+k})
            # plus contributions from shift -k? Actually gradient of max O_k with respect to h? approx gradient of O_k
            # For shift >=0, derivative of O_k w.r.t h_i:
            # For i in 0..n-1-k: dO/dh_i = (1 - h_{i+k})
            # For i in k..n-1: dO/dh_i = -h_{i-k}
            g[:n - idx] = 1 - h[idx:n]  # first part
            g[idx:] -= h[:n - idx]    # second part
        else:  # shift negative
            t = -idx
            # O_{-t} = sum_{i=t}^{n-1} h_i * (1 - h_{i-t})
            g[t:] = 1 - h[:n - t]
            g[:n - t] -= h[t:]
        # update
        h = h - step * (0.5 ** it) * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # enforce sum
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def softmax_obj(x):
    global CURRENT_TARGET, CURRENT_BETA
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)

Observations: The gradient of O_k (a linear function) w.r.t h:

For shift k ≥ 0: O_k = sum_{i=0}^{n-1-k} h_i * (1 - h_{i+k}) * dx.

Derivative w.r.t h_i:

- For i in [0,n-1-k]: derivative = (1 - h_{i+k}) * dx.
- For i in [k, n-1]: derivative from term where i = i' + k? Actually term h_{i' } * (1 - h_{i'+k}) includes -h_{i'+k} * dx. So derivative w.r.t h_{i} appears as -h_{i - k} * dx for i ≥ k.

Thus gradient g[i] = (1 - h[i+k]) if i <= n-1-k else 0, minus h[i-k] if i >= k else 0.

Thus code:

if k >=0:
    g[:n - k] = 1 - h[k:]
    g[k:] -= h[:n - k]

If k <0: let t = -k positive. O_{-t} = sum_{i=t}^{n-1} h_i * (1 - h_{i - t}). Derivative:
- For i in [t, n-1]: derivative = (1 - h_{i - t}) * dx
- For i in [0, n-1 - t]: -h_{i + t} * dx (since i + t corresponds to index that appears as partner in the previous term). Equivalent to same pattern but shifted.

Thus for k negative:

t = -k
g[t:] = 1 - h[:n - t]
g[:n - t] -= h[t:]

import time, numpy as np
from scipy.optimize import minimize

# ---------- global variables for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) > 0:
            h[i] += add
            diff = target - h.sum()
    return h


def sum_eq_constraint(x: np.ndarray) -> float:
    global CURRENT_TARGET
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    global CURRENT_BETA
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def greedy_swap(h: np.ndarray, shift: int, rng: np.random.Generator):
    """One‑sided transfer that reduces the worst shift, if possible."""
    n = h.size
    if shift == 0:
        return False, h
    if shift > 0:
        lim = n - shift
        hi = h[:lim]
        hs = h[shift:]
        mis = (hi >= 0.5) & (hs < 0.5)
        good = (hi < 0.5) & (hs >= 0.5)
        mis_idx = np.nonzero(mis)[0]
        good_idx = np.nonzero(good)[0]
        if mis_idx.size == 0 or good_idx.size == 0:
            return False, h
        i = rng.choice(mis_idx)
        j = rng.choice(good_idx)
        new_h = h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        return True, new_h
    else:
        t = -shift
        lim = n - t
        hi = h[t:]          # indices t..n-1
        hs = h[:lim]        # 0..n-1-t
        mis = (hi < 0.5) & (hs >= 0.5)
        good = (hi >= 0.5) & (hs < 0.5)
        mis_idx = np.nonzero(mis)[0] + t
        good_idx = np.nonzero(good)[0] + t
        if mis_idx.size == 0 or good_idx.size == 0:
            return False, h
        i = rng.choice(mis_idx)
        j = rng.choice(good_idx)
        new_h = h.copy()
        new_h[i] = 1.0
        new_h[j] = 0.0
        return True, new_h


def random_swap(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap a random zero with a random one (preserves total mass)."""
    zeros = np.where(h < 0.5)[0]
    ones = np.where(h > 0.5)[0]
    if zeros.size == 0 or ones.size == 0:
        return h
    i = rng.choice(zeros)
    j = rng.choice(ones)
    new_h = h.copy()
    new_h[i] = 1.0
    new_h[j] = 0.0
    return new_h


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = int(np.argmax(corr)) - (n - 1)  # signed shift
        g = np.zeros_like(h)
        if idx >= 0:
            k = idx
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        else:
            t = -idx
            g[t:] = 1.0 - h[: n - t]
            g[: n - t] -= h[t:]
        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                if abs(add) > 0:
                    h[i] += add
                    diff = target - h.sum()
    return h
