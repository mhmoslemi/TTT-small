    n = h.shape[0]
    m = 2*n - 1
    # compute convolution of h and (1-h) reversed?
    # correlation: corr[k] = sum_i h[i] * (1-h[i+k]) where i+k < n else (1-h[i+k]) = 0

    rev = (1 - h)[::-1]
    # compute via FFT

    f1 = np.fft.rfft(h, n=N)

import time, numpy as np, scipy.optimize as opt

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    N = 1 << ((2*n - 1).bit_length())
    f1 = np.fft.rfft(h, n=N)
    f2 = np.fft.rfft(rev, n=N)
    conv = np.fft.irfft(f1 * f2, n=N)[:2*n-1]
    return conv.max() * (2.0 / n)

def enforce_sum(h, rng, target=None):
    n = h.shape[0]
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            max_eps = 1.0 - h[i]
            add = min(diff, max_eps)
        else:
            max_eps = h[i]
            add = max(diff, -max_eps)
        h[i] += add
        diff = target - h.sum()
    return h

def worst_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-5)
    return best_h, best_c

def gradient_refine(h, rng, step=0.015, max_iter=10):
    n = h.shape[0]
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # quick redistribution
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def hill_climb_binary(h, max_iters, rng):
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    for _ in range(max_iters):
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(cur_h)
        if new_c < cur_c - 1e-12:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # initial from parent if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0.copy()
        best_n = h0.size
        best_c = compute_c5(best_h)

    n_candidates = [96, 120, 144, 168, 192, 216, 240, 256, 288, 312, 336, 360, 384]

    # Phase 1: many randomized local searches
    while time.time() - start < 0.55 * budget_s:
        n = int(rng.choice(n_candidates))
        fresh = rng.random() < 0.2
        if not fresh and best_h is not None:
            if best_n == n:
                h = best_h.copy()
            else:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
        else:
            # random start
            h = rng.random(n)
            h *= (n / 2.0) / h.sum()
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)

        # guided hill climb
        h, c = guided_hill_climb(h, rng, max_iters=1400, step_scale=0.7)
        # optional gradient refinement
        h = gradient_refine(h, rng, step=0.015, max_iter=6)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # binary hill climb if n even
        if n % 2 == 0:
            h_bin = np.zeros(n, dtype=float)
            h_bin[rng.choice(n, n // 2, replace=False)] = 1.0
            h_bin, c_bin = hill_climb_binary(h_bin, max_iters=6000, rng=rng)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin.copy(), c_bin, n
                # refine binary solution a little
                h_bin, _ = guided_hill_climb(best_h, rng, max_iters=800, step_scale=0.5)
                best_h = h_bin
                best_c = compute_c5(best_h)

    # Phase 2: soft‑max SLSQP refinement on best solution
    if best_h is not None:
        for beta in (20., 80., 200., 800., 2000., 8000., 20000.):
            if time.time() - start > 0.80 * budget_s:
                break
            maxit = 1200 if beta >= 200 else 800
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=maxit)
            best_c = compute_c5(best_h)

    # Phase 3: up‑sampling and further refinement
    while time.time() - start < 0.95 * budget_s:
        factor = int(rng.choice([2, 3, 4]))
        n_up = best_n * factor
        if n_up > 512:
            break
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)
        # local hill climb at higher resolution
        h_up, c_up = guided_hill_climb(h_up, rng, max_iters=2000, step_scale=0.6)
        # SLSQP passes
        for beta in (20., 200., 2000., 8000.):
            if time.time() - start > budget_s:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)
        h_up = gradient_refine(h_up, rng, step=0.01, max_iter=8)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # Final stochastic fine‑tuning
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:

    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

import time, numpy as np, scipy.optimize as opt

# globals used by the soft‑max SLSQP objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function on [0,2] using FFT for speed."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    f1 = np.fft.rfft(h, n=N)
    f2 = np.fft.rfft(rev, n=N)
    conv = np.fft.irfft(f1 * f2, n=N)[: 2 * n - 1]
    return conv.max() * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator, target: float = None) -> np.ndarray:
    """Adjust h so that sum(h) == target while staying in [0,1]."""
    n = h.shape[0]
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            max_eps = 1.0 - h[i]
            add = min(diff, max_eps)
        else:
            max_eps = h[i]
            add = max(diff, -max_eps)
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-5)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def hill_climb_binary(h: np.ndarray, max_iters: int, rng: np.random.Generator):
    """Hill‑climb on a binary vector (exactly n/2 ones)."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    for _ in range(max_iters):
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c = compute_c5(cur_h)
        if new_c < cur_c - 1e-12:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # --- start from parent construction if available -----------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0.copy()
        best_n = h0.size
        best_c = compute_c5(best_h)

    n_candidates = [96, 120, 144, 168, 192, 216, 240, 256,
                    288, 312, 336, 360, 384]

    # ------------------- Phase 1: random local searches -------------
    while time.time() - start < 0.55 * budget_s:
        n = int(rng.choice(n_candidates))
        fresh = rng.random() < 0.2

        if not fresh and best_h is not None:
            if best_n == n:
                h = best_h.copy()
            else:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
        else:
            h = rng.random(n)
            h *= (n / 2.0) / h.sum()
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)

        # guided hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=1400, step_scale=0.7)
        # short gradient tidy‑up
        h = gradient_refine(h, rng, step=0.015, max_iter=6)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # binary hill‑climb (optional)
        if n % 2 == 0:
            hb = np.zeros(n, dtype=float)
            hb[rng.choice(n, n // 2, replace=False)] = 1.0
            hb, c_bin = hill_climb_binary(hb, max_iters=6000, rng=rng)
            if c_bin < best_c:
                best_h, best_c, best_n = hb.copy(), c_bin, n
                # a quick continuous refinement
                hb, _ = guided_hill_climb(best_h, rng, max_iters=800, step_scale=0.5)
                best_h = hb
                best_c = compute_c5(best_h)

    # ------------------- Phase 2: SLSQP soft‑max refinement -------
    if best_h is not None:
        for beta in (20., 80., 200., 800., 2000., 8000., 20000.):
            if time.time() - start > 0.80 * budget_s:
                break
            maxit = 1200 if beta >= 200 else 800
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=maxit)
            best_c = compute_c5(best_h)

    # ------------------- Phase 3: Upsampling & finer search -------
    while time.time() - start < 0.95 * budget_s:
        factor = int(rng.choice([2, 3, 4]))
        n_up = best_n * factor
        if n_up > 512:
            break
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, rng)

        h_up, c_up = guided_hill_climb(h_up, rng, max_iters=2000, step_scale=0.6)
        for beta in (20., 200., 2000., 8000.):
            if time.time() - start > budget_s:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)
        h_up = gradient_refine(h_up, rng, step=0.01, max_iter=8)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ------------------- Final stochastic tweaks -------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ------------------- Sanity & return ---------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
