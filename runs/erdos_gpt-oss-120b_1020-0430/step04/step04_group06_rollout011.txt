import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def triangular_h(n: int) -> np.ndarray:
    """Symmetric triangular step function with ∫h = 1 and values in [0,1].

    h[i] = (2*i+1)/n for i < n/2,
           (2*(n-i)-1)/n for i >= n/2.
    """
    i = np.arange(n, dtype=np.float64)
    half = n // 2
    h = np.empty(n, dtype=np.float64)
    # first half (increasing)
    h[:half] = (2 * i[:half] + 1) / n
    # second half (decreasing)
    h[half:] = (2 * (n - i[half:]) - 1) / n
    return h

    def pair_transfer_opt(h, steps, beta):
        """Move mass from a high‑h index to a low‑h index that contributes most
