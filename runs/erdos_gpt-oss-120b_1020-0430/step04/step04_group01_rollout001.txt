def enforce_sum(h, target, rng):
    # iterative adjusting to meet sum while staying in bounds
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:
            # try another index
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def targeted_local_search(h, rng, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    while time.time() < deadline:
        conv = correlate_full(best_h)
        k_max = np.argmax(conv)
        # compute contribution to this shift
        rolled = np.roll(best_h, -k_max)
        contrib = best_h * (1.0 - rolled)
        # find i_max (largest contribution)
        i_max = np.argmax(contrib)
        # find j_min (smallest contribution, perhaps zero)
        # mask to avoid same index
        # maybe random choose among low contributions
        # get indices sorted by contribution
        # we can take j candidates: where contrib is small
        # we'll select random low
        # e.g., choose an index where contrib <= percentile
        threshold = np.percentile(contrib, 10)  # bottom 10%
        low_idxs = np.where(contrib <= threshold)[0]
        if low_idxs.size == 0:
            low_idxs = np.arange(n)
        j = rng.choice(low_idxs)
        if i_max == j:
            # pick another j
            j = rng.integers(0, n)
            if j == i_max:
                continue
        # amount to transfer
        max_eps = min(best_h[i_max], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        # transfer a fraction of contribution at i_max
        eps = min(0.5 * contrib[i_max], max_eps)  # reduce at most 50% of contribution
        # also ensure eps not too small
        if eps < 1e-8:
            eps = max_eps * 0.1
        # perform transfer
        new_h = best_h.copy()
        new_h[i_max] -= eps
        new_h[j] += eps
        # ensure bounds
        new_h = np.clip(new_h, 0.0, 1.0)
        # re-enforce sum (should be same but numeric errors)
        new_h = enforce_sum(new_h, n/2.0, rng)
        c = compute_c5(new_h)
        if c < best_c:
            best_h, best_c = new_h, c
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    target = h0.size/2.0
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        conv = correlate_full(x)
        bc = beta * conv
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m))))/beta
    bounds = [(0.0,1.0)]*h0.size
    cons = {'type':'eq','fun':sum_eq}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter':max_iter,'ftol':1e-12,'disp':False})
    x = np.clip(res.x,0.0,1.0)
    # enforce sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x
```

def shift_contributions(h, k):
    n = h.size
    if k >= 0:
        contrib = np.empty(n, dtype=h.dtype)
        contrib[:n - k] = h[:n - k] * (1 - h[k:])
        contrib[n - k:] = h[n - k:]
    else:
        kp = -k
        contrib = np.empty(n, dtype=h.dtype)
        contrib[:n - kp] = h[kp:] * (1 - h[:n - kp])
        contrib[n - kp:] = h[:kp]
    return contrib

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in-place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        enforce_sum(h, target, rng)
    return h


def correlate_full(h):
    """Full linear correlation of h with (1-h) via FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def shift_contributions(h, k):
    """Contribution of each index to correlation at integer shift k."""
    n = h.size
    contrib = np.empty(n, dtype=h.dtype)
    if k >= 0:
        nk = n - k
        contrib[:nk] = h[:nk] * (1.0 - h[k:])
        contrib[nk:] = h[nk:]
    else:
        kp = -k
        nk = n - kp
        contrib[:nk] = h[kp:] * (1.0 - h[:nk])
        contrib[nk:] = h[:kp]
    return contrib

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on log‑sum‑exp smooth approximation of the max."""
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def softmax_obj(x):
        conv = correlate_full(x)
        bc = beta * conv
        m = bc.max()
        # stable log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        enforce_sum(x, target, rng)
    return x

def targeted_local_search(h, rng, deadline):
    """Iterative reduction of the dominant shift by moving mass from bad to good sites."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = best_h.size
