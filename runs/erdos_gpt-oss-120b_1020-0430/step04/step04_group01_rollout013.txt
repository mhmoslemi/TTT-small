def projected_subgradient_optimization(h0, target_sum, time_deadline):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    h = h0.copy()
    step0 = 1.0  # initial step size (maybe 0.5)
    it = 0
    while time.time() < time_deadline:
        # compute correlation
        corr = correlate_full(h)
        cmax = corr.max()
        if cmax < best_c:
            best_c = cmax
            best_h = h.copy()
        # subgradient weights
        active = np.where(corr >= cmax - 1e-12)[0]
        m = active.size
        if m == 0:
            break
        w = np.zeros_like(corr)
        w[active] = 1.0 / m
        # compute gradient using FFT
        L = 1 << ((2 * h.size - 1).bit_length())
        H = np.fft.rfft(h, L)
        W = np.fft.rfft(w, L)
        conv_hw = np.fft.irfft(H * W, L)[: 2 * h.size - 1]
        # reversed w
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, L)
        conv_rev = np.fft.irfft(H * W_rev, L)[: 2 * h.size - 1]
        # gradient
        t = np.arange(h.size) + (h.size - 1)
        grad = 1.0 - (conv_hw[t] + conv_rev[t])
        # step size schedule
        step = step0 / np.sqrt(it + 1)
        # update and project
        v = h - step * grad
        h = project_to_feasible(v, target_sum)
        it += 1
    return best_h, best_c
```

    L = 1 << ((2 * n - 1).bit_length())

    return conv[: 2 * n - 1]


def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/n)."""
    return correlate_full(h).max() * (2.0 / h.size)


def project_to_feasible(v, target):
    """
    Euclidean projection of v onto the intersection of the box [0,1]^n
    and the hyperplane sum(x)=target.
    """
    lo, hi = -1.0, 2.0
    # bisection on the Lagrange multiplier
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    proj = np.clip(v - lam, 0.0, 1.0)

    # tiny correction to hit the target exactly
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        nfree = free.sum()
        if nfree:
            proj[free] += diff / nfree
            proj = np.clip(proj, 0.0, 1.0)
    return proj


# --------------------------------------------------------------
# Subgradient optimizer for max‑overlap (min‑max) problem
# --------------------------------------------------------------
def projected_subgradient_optimization(h0, target, deadline):
    n = h0.size
    L = 1 << ((2 * n - 1).bit_length())
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    h = h0.copy()
    it = 0
    step0 = 1.0

    while time.time() < deadline:
        corr = correlate_full(h)                     # length 2n‑1
        cmax = corr.max()
        c_c5 = cmax * (2.0 / n)
        if c_c5 < best_c:
            best_c = c_c5
            best_h = h.copy()

        # indices attaining the maximum (subgradient set)
        active = np.where(corr >= cmax - 1e-12)[0]
        m = active.size
        if m == 0:
            break
        w = np.zeros_like(corr)
        w[active] = 1.0 / m

        # gradient of the subgradient surrogate
        H = np.fft.rfft(h, L)
        W = np.fft.rfft(w, L)
        conv_hw = np.fft.irfft(H * W, L)[: 2 * n - 1]
        W_rev = np.fft.rfft(w[::-1], L)
        conv_rev = np.fft.irfft(H * W_rev, L)[: 2 * n - 1]
        t = np.arange(n) + (n - 1)
        grad = 1.0 - (conv_hw[t] + conv_rev[t])

        # step and projection
        step = step0 / np.sqrt(it + 1)
        v = h - step * grad
        h = project_to_feasible(v, target)

        it += 1

    return best_h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ----------------------------------------------------------
    # 1. Base resolution and initial candidate
    # ----------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_feasible(h0, n / 2.0)
    else:
        n = 720
        h0 = project_to_feasible(rng.random(n), n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----------------------------------------------------------
    # 2. Build a diverse pool of starting points
    # ----------------------------------------------------------
    starts = [h0]

    xs = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n

    # triangular
    tri = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    starts.append(project_to_feasible(tri, n / 2.0))

    # sinusoidal / cosinusoidal families
    for f in (0.5, 1.0, 2.0, 3.0):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        starts.append(project_to_feasible(s, n / 2.0))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        starts.append(project_to_feasible(c, n / 2.0))

    # binary block (half 1, half 0)
    blk = np.zeros(n)
    blk[: n // 2] = 1.0
    starts.append(project_to_feasible(blk, n / 2.0))

    # alternating 0/1
    alt = np.tile([0.0, 1.0], n // 2)
    starts.append(project_to_feasible(alt, n / 2.0))

    # a few random feasible vectors
    while len(starts) < 30 and time.time() < deadline:
        rnd = project_to_feasible(rng.random(n), n / 2.0)
        starts.append(rnd)

    # ----------------------------------------------------------
    # 3. Optimize each start with subgradient method
    # ----------------------------------------------------------
    for h_start in starts:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # give each start at most 5 s or 70 % of the remaining time
        slice_deadline = time.time() + min(remaining * 0.7, 5.0)
        h_opt, c_opt = projected_subgradient_optimization(
            h_start, n / 2.0, slice_deadline
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----------------------------------------------------------
    # 4. Final refinement using a smooth soft‑max surrogate
    # ----------------------------------------------------------
    if time.time() < deadline:
        beta = 5000.0
        h = best_h.copy()
        L = 1 << ((2 * n - 1).bit_length())
        it = 0
        while time.time() < deadline and it < 4000:
            corr = correlate_full(h)
            max_corr = corr.max()
            w = np.exp(beta * (corr - max_corr))
            w /= w.sum()

            H = np.fft.rfft(h, L)
            W = np.fft.rfft(w, L)
            conv_hw = np.fft.irfft(H * W, L)[: 2 * n - 1]
            W_rev = np.fft.rfft(w[::-1], L)
            conv_rev = np.fft.irfft(H * W_rev, L)[: 2 * n - 1]
            t = np.arange(n) + (n - 1)
            grad = 1.0 - (conv_hw[t] + conv_rev[t])

            step = 0.5 / np.sqrt(it + 1)
            v = h - step * grad
            h = project_to_feasible(v, n / 2.0)

            if it % 200 == 0:
                c_val = compute_c5(h)
                if c_val < best_c:
                    best_c, best_h = c_val, h.copy()
            it += 1

    # ----------------------------------------------------------
    # 5. Clean up and return
    # ----------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
