def random_transfer(h, rng, scale):

    i = rng.integers(n)

def targeted_transfer(h, rng, lag, scale, top_frac=0.2):

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # choose random index that can adjust in direction of diff
        i = rng.integers(0, n)
        if diff > 0:
            avail = 1.0 - h[i]
            if avail <= 0:
                continue
            add = min(diff, avail)
        else:
            avail = h[i]
            if avail <= 0:
                continue
            add = max(diff, -avail)
        h[i] += add
        diff = target - h.sum()
    # final correction
    if abs(diff) > 1e-12:
        # distribute uniformly
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            avail = 1.0 - h[i]
            if avail <= 0:
                continue
            add = min(diff, avail)
        else:
            avail = h[i]
            if avail <= 0:
                continue
            add = max(diff, -avail)
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
    return h

# ---------- correlation ----------
def full_corr_fft(h):
    """Full correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    B = np.fft.rfft(np.flip(1.0 - h), m)
    conv = np.fft.irfft(H * B, m)[: 2 * n - 1]
    return conv

def compute_c5_fast(h):
    """C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated."""
    n = h.size
    corr = full_corr_fft(h)
    return np.max(corr) * (2.0 / n)

# ---------- perturbations ----------
def random_transfer(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def targeted_transfer(h, rng, lag, scale=0.2, top_frac=0.2):
    n = h.size
    if lag >= n or lag <= -n:
        return random_transfer(h, rng, scale)
    if lag >= 0:
        contrib = h[: n - lag] * (1.0 - h[lag:])
        idx = np.arange(n - lag)
    else:
        l = -lag
        contrib = h[l:] * (1.0 - h[: n - l])
        idx = np.arange(l, n)
    if contrib.size == 0:
        return random_transfer(h, rng, scale)
    k = max(1, int(len(idx) * top_frac))
    top_idx = idx[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = idx[np.argpartition(contrib, k - 1)[:k]]
    if rng.random() < 0.7:
        i = rng.choice(top_idx)
        j = rng.choice(bot_idx)
    else:
        i = rng.choice(top_idx)
        j = rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return random_transfer(h, rng, scale)
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- SLSQP refined soft‑max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # quick fix
        n = x.size
        correction = diff / n
        x += correction
        x = np.clip(x, 0.0, 1.0)
    return x

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    # --- choose resolution ---
    n = 720
    target_sum = n / 2.0

    # --- initialise ---
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        if n0 != n:
            old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            h0 = np.interp(new_x, old_x, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_sum, rng)
    else:
        # start from a half‑filled binary pattern plus tiny noise
        h0 = np.zeros(n)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)
        # add small uniform noise and re‑project
        h0 += (rng.random(n) - 0.5) * 0.02
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_sum, rng)

    best_h = h0.copy()
    best_c = compute_c5_fast(best_h)
    cur_h = best_h.copy()
    cur_c = best_c

    # --- simulated annealing on fine grid ---
    max_iters = 600_000
    T0, Tf = 0.02, 0.00002
    scale_start, scale_end = 0.3, 0.001
    it = 0
    while it < max_iters and time.time() - t0 < budget_s * 0.85:
        frac = it / max_iters
        T = T0 * (Tf / T0) ** frac
        scale = scale_start * (scale_end / scale_start) ** frac
        # identify worst lag
        corr = full_corr_fft(cur_h)
        lag = np.argmax(corr) - (n - 1)
        if rng.random() < 0.6:
            new_h = targeted_transfer(cur_h, rng, lag, scale)
        else:
            new_h = random_transfer(cur_h, rng, scale)
        new_c = compute_c5_fast(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-15)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        it += 1

    # --- SLSQP refinement ladder (beta sweep) ---
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    for beta in beta_list:
        if time.time() - t0 >= budget_s * 0.95:
            break
        try:
            refined = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=200)
        except Exception:
            continue
        refined_c = compute_c5_fast(refined)
        if refined_c < best_c:
            best_h, best_c = refined.copy(), refined_c

    # --- final light hill‑climb ---
    cur_h = best_h.copy()
    cur_c = best_c
    step = 0.2
    for _ in range(8000):
        if time.time() - t0 > budget_s * 0.99:
            break
        nh = random_transfer(cur_h, rng, scale=step)
        nc = compute_c5_fast(nh)
        if nc < cur_c:
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        step *= 0.9998

    # --- final clean‑up ---
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5_fast(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
