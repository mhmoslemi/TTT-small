def max_overlap_and_shift(h):

def pair_transfer_optimize(h0, rng, max_iters=10000, time_limit=None, init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    while True:
        if max_iters is not None and max_iters <= 0:
            break
        if time_limit is not None and time.time() - start >= time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
        # compute i_range, j_range
        if shift >= 0:
            i_range = np.arange(n - shift)  # i from 0 to n-shift-1
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)  # shift negative
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            if max_iters is not None:
                max_iters -= 1
            continue
        # product = h[i] * (1 - h[j])
        prod = h[i_range] * (1.0 - h[j_range])
        # find index of maximum product
        idx_max = int(np.argmax(prod))
        i = i_range[idx_max]
        j = j_range[idx_max]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            if max_iters is not None:
                max_iters -= 1
            continue
        eps = factor * max_eps
        # adjust epsilon
        h[i] -= eps
        h[j] += eps
        # keep within bounds: just clip
        if h[i] < 0.0:
            h[i] = 0.0
        if h[j] > 1.0:
            h[j] = 1.0
        # ensure sum is exact: because we moved eps from i to j, total sum unchanged exactly (unless clipping). With clipping we may need to adjust: if clipping occurred we can add diff to some other index.
        # Simpler: after each eps move, we can avoid clipping by ensuring eps <= max_eps; since max_eps = min(h[i], 1-h[j]), moving eps ensures both stay within bounds.
        # So fine.

        factor = max(1e-6, factor * decay)
        if max_iters is not None:
            max_iters -= 1
    # after loop, compute final best
    final_c = compute_c5(best_h)
    return best_h, final_c

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
#  Helper functions
# --------------------------------------------------------------
def _project_sum_box(v, target, max_iter=60):
    """Project v onto the hyperplane sum(v)=target with 0<=v<=1 by binary search."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] projected onto sum=n/2."""
    v = rng.random(n)
    return _project_sum_box(v, n / 2.0)


def _fft_corr(h):
    """Linear correlation of h with (1-h) using FFT, returns full length 2n-1 array."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    corr = np.fft.irfft(H * G, n=m)[: 2 * n - 1]
    return corr


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.shape[0]
    corr = _fft_corr(h)
    return np.max(corr) * (2.0 / n)


def max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    n = h.shape[0]
    corr = _fft_corr(h)
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (n - 1)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small mass ε from i to j (preserves sum)."""
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple hill‑climb using perturbations."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = 0.7
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            nh = perturb_h(h, rng, scale=step)
            c = compute_c5(nh)
            if c < best_c:
                best_c, h = c, nh
                improved = True
                break
        if not improved:
            step = max(step * decay, 1e-4)
    return h, best_c


def pair_transfer_optimize(
    h0,
    rng,
    max_iters=None,
    time_limit=None,
    init_factor=0.5,
    decay=0.999,
):
    """Local search moving mass from i to j that most contributes to the worst shift."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    it = 0
    while True:
        if max_iters is not None and it >= max_iters:
            break
        if time_limit is not None and (time.time() - start) >= time_limit:
            break

        cur_c = max_val * (2.0 / n)

        idx_max = int(np.argmax(prod))

        max_eps = min(h[i], 1.0 - h[j])

#  Smooth surrogate for SLSQP (optional, kept for possible final refinement)

CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = _fft_corr(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=200):
    """Refine h using SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final projection onto sum box (tiny numeric drift possible)
    x = _project_sum_box(x, n / 2.0)
    return x
