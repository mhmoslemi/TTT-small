def gradient_surrogate(h, beta):
    n = h.size
    # compute S_k
    conv = correlate_full(h)  # length 2n-1
    S = conv[n-1:]  # non-negative shifts
    # compute weights
    bc = beta * S
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()
    # compute term1 and term2
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    # vectorized loop
    for k, wk in enumerate(w):
        if wk == 0.0:
            continue
        if k == 0:
            term1 += wk * h
            term2 += wk * h
        else:
            term1[:n-k] += wk * h[k:]
            term2[k:] += wk * h[:n-k]
    grad = 1.0 - term1 - term2
    return grad

    L = 1 << ((2 * n - 1).bit_length())

    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]

import time
import numpy as np

def project_box_sum(v, target):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        mask = (x > 0) & (x < 1)
        if np.any(mask):
            x[mask] += diff / np.count_nonzero(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def random_binary_vector(n, rng):
    target = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, target, replace=False)
    h[idx] = 1.0
    return h

def fft_convolve(a, b):
    n = a.size + b.size - 1
    L = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    c = np.fft.irfft(fa * fb, L)[:n]
    return c

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

def gradient_surrogate(h, beta):
    n = h.size
    conv = correlate_full(h)
    S = conv[n - 1 :]                      # non‑negative shifts
    bc = beta * S
    m = bc.max()
    w = np.exp(bc - m)
    w /= w.sum()
    term2 = fft_convolve(w, h)[:n]        # sum_k w_k * h_{j‑k}
    term1 = fft_convolve(w[::-1], h)[n - 1 : n - 1 + n]  # sum_k w_k * h_{j+k}
    grad = 1.0 - term1 - term2
    return grad

def perturb_h(h, rng, scale=0.2):
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c
