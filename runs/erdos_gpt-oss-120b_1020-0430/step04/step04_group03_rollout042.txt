def project_onto_capped_simplex(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0, 1)
    # adjust may have slight sum error due to tolerance, do final fix:
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff to indices not at bound
        mask = (x > 0) & (x < 1)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0, 1)  # ensure still within [0,1]
    return x

import time, numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def project_to_capped_simplex(v, target):
    """Projection of v onto {x: sum(x)=target, 0<=x<=1}."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h):
    """Discrete C5 for vector h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def greedy_optimize(h, max_steps=2000):
    """Deterministic greedy improvement."""
    for _ in range(max_steps):
        s, _, cur = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # smooth max via log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # decide problem size
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 720

    target_sum = n / 2.0
    best_h = None
    best_c = float('inf')

    # ---- 1) low‑dimensional parametric scan (three‑level symmetric shape) ---
    a_vals = np.linspace(0.01, 0.49, 49)          # fraction of each edge interval
    edge_vals = np.linspace(0.0, 1.0, 101)        # height on the edge blocks
    for a in a_vals:
        edge_len = int(round(a * n / 2))
        if edge_len < 1 or 2 * edge_len >= n:
            continue
        for h_edge in edge_vals:
            h_mid = (0.5 - a * h_edge) / (1.0 - a)
            if not (0.0 <= h_mid <= 1.0):
                continue
            h = np.full(n, h_mid, dtype=float)
            h[:edge_len] = h_edge
            h[-edge_len:] = h_edge
            h = project_to_capped_simplex(h, target_sum)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
        if time.time() - start_time > budget_s * 0.30:
            break

    # fallback to a random feasible vector if needed
    if best_h is None:
        best_h = random_feasible_vector(n, rng)
        best_c = compute_c5(best_h)

    # ---- 2) iterative local improvement (greedy + SLSQP + annealing) ---
    while time.time() - start_time < budget_s * 0.85:
        # generate a start point
        if rng.random() < 0.6:
            start = perturb_random(best_h, rng, scale=0.4)
        else:
            start = random_feasible_vector(n, rng)
        # greedy refinement
        local = greedy_optimize(start, max_steps=4000)
        c_local = compute_c5(local)
        if c_local < best_c:
            best_c, best_h = c_local, local.copy()
        # occasional SLSQP polishing
        if rng.random() < 0.35:
            for beta in (500, 2000, 5000, 10000, 20000):
                if time.time() - start_time > budget_s * 0.95:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=1000)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_c, best_h = c_ref, refined.copy()
        # tiny simulated‑annealing moves
        if rng.random() < 0.25:
            cur = best_h.copy()
            cur_c = best_c
            T = 0.01 * cur_c
            for _ in range(20):
                cand = perturb_random(cur, rng, scale=0.12)
                cand_c = compute_c5(cand)
                if cand_c < cur_c or rng.random() < np.exp(-(cand_c - cur_c) / T):
                    cur, cur_c = cand, cand_c
                    if cur_c < best_c:
                        best_c, best_h = cur_c, cur.copy()
                T *= 0.94
        # early exit if we already beat target
        if best_c < 0.3808:
            break

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_capped_simplex(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
