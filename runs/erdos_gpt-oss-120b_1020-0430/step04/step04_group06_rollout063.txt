def correlation_full(a, b):
    n = a.size
    m = b.size
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)

    C = A * B
    corr = np.fft.irfft(C, N)[:size]
    return corr

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0, init_factor=0.5, decay=0.999):
   h = h0.copy()
   n = h.size
   best_h = h.copy()
   best_c = compute_c5(h)
   start = time.time()
   factor = init_factor
   for it in range(max_iters):
       if time.time() - start > time_limit:
           break
       # compute max overlap shift and its value
       corr = correlation_full(h, 1 - h)
       idx = int(np.argmax(corr))
       s = idx - (n - 1)
       # produce i_range, j_range
       if s >= 0:
           i_range = np.arange(n - s)
           j_range = i_range + s
       else:
           s_abs = -s
           i_range = np.arange(s_abs, n)
           j_range = i_range - s_abs
       if i_range.size == 0:
           factor = init_factor
           continue
       prod = h[i_range] * (1.0 - h[j_range])
       idx_max = int(np.argmax(prod))
       i = int(i_range[idx_max])
       j = int(j_range[idx_max])
       max_eps = min(h[i], 1.0 - h[j])
       if max_eps <= 0:
           factor = init_factor
           continue
       eps = factor * max_eps
       if eps <= 0:
           factor = init_factor
           continue
       h[i] -= eps
       h[j] += eps
       # clip
       h[i] = max(0.0, min(1.0, h[i]))
       h[j] = max(0.0, min(1.0, h[j]))
       factor = max(1e-4, factor * decay)
       # track best
       cur_c = compute_c5(h)
       if cur_c < best_c:
           best_c = cur_c
           best_h = h.copy()
   return best_h, best_c

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction if needed
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:

import time
import numpy as np

def correlation_full(a,b):
    n = a.size; m = b.size
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b[::-1], N)
    C = A * B
    corr = np.fft.irfft(C, N)[:size]
    return corr

def compute_c5(h):
    corr = correlation_full(h, 1.0 - h)
    return np.max(corr) * (2.0 / h.size)

def project_box_sum(v, target):
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(30):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def pgd_optimize(h0, target, max_iters, step0, decay, time_limit):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    n = h.size
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = correlation_full(h, 1 - h)
        cur_c = np.max(corr) * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        idx = int(np.argmax(corr))
        s = idx - (n - 1)
        if s >= 0:
            hf = np.empty_like(h)
            hf[:n-s] = h[s:]
            hf[n-s:] = 0.0
            hb = np.empty_like(h)
            hb[s:] = h[:n-s]
            hb[:s] = 0.0
        else:
            s_abs = -s
            hf = np.empty_like(h)
            hf[s_abs:] = h[:n-s_abs]
            hf[:s_abs] = 0.0
            hb = np.empty_like(h)
            hb[:n-s_abs] = h[s_abs:]
            hb[n-s_abs:] = 0.0
        g = 1.0 - (hf + hb)
        v = h - step * g
        h = project_box_sum(v, target)
        step *= decay
    # final check
    final_c = compute_c5(best_h)
    return best_h, final_c

def max_overlap_and_shift(h):
    corr = correlation_full(h, 1 - h)
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    shift = idx - (h.size - 1)
    return max_val, shift

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0, init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            s_abs = -shift
            i_range = np.arange(s_abs, n)
            j_range = i_range - s_abs
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    return best_h, best_c

def anneal_optimize(h_start, rng, max_moves, time_limit):
    h = h_start.copy()
    n = h.size
    # make sure h is binary for swaps
    h_bin = (h > 0.5).astype(float)
    # adjust sum exactly (should be n/2)
    target = n/2
    diff = target - h_bin.sum()
    if diff != 0:
        # fix by setting diff indices to 1 or 0
        idx = rng.choice(n, size=abs(int(diff)), replace=False)
        if diff > 0:
            h_bin[idx] = 1.0
        else:
            h_bin[idx] = 0.0
    h = h_bin
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    for move in range(max_moves):
        if time.time() - start > time_limit:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        # acceptance
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            # temperature schedule
            temp = max(1e-4, (max_moves - move) / max_moves)
            if rng.random() >= np.exp(-(c - best_c) / temp):
                # revert swap
                h[i], h[j] = h[j], h[i]
    return best_h, best_c

def generate_candidates(rng, n, target):
    candidates = []
    # binary random
    h0 = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h0[ones] = 1.0
    candidates.append(h0)
    # Beatty sequence
    alpha = (np.sqrt(5) - 1) / 2  # approx 0.618
    xs = np.arange(n)
    hb = (np.mod(xs * alpha, 1.0) < 0.5).astype(float)
    candidates.append(enforce_sum(hb, target, rng))
    # sinusoidal thresholded
    for freq in (2,3,5):
        hs = (np.sin(2*np.pi*freq*xs/n) > 0).astype(float)
        candidates.append(enforce_sum(hs, target, rng))
    # random real vectors
    for _ in range(2):
        h = rng.random(n)
        candidates.append(enforce_sum(h, target, rng))
    return candidates

import time
import numpy as np

def correlation_full(a, b):
    """Linear correlation a * b (full mode) via FFT."""
    n = a.size
    m = b.size
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b[::-1], N)
    C = A * B
    corr = np.fft.irfft(C, N)[:size]
    return corr

def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2]."""
    corr = correlation_full(h, 1.0 - h)
    return np.max(corr) * (2.0 / h.size)

def project_box_sum(v, target):
    """Projection of v onto {0≤x≤1, sum(x)=target} using bisection."""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(30):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def enforce_sum(h, target, rng, max_iter=2000):
    """Clamp to [0,1] and adjust to hit the exact sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def pgd_optimize(h0, target, max_iters, step0, decay, time_limit):
    """Projected gradient descent on the worst–overlap shift."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    n = h.size
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = correlation_full(h, 1 - h)
        cur_c = np.max(corr) * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        idx = int(np.argmax(corr))
        s = idx - (n - 1)
        if s >= 0:
            hf = np.empty_like(h)
            hf[:n - s] = h[s:]
            hf[n - s:] = 0.0
            hb = np.empty_like(h)
            hb[s:] = h[:n - s]
            hb[:s] = 0.0
        else:
            s_abs = -s
            hf = np.empty_like(h)
            hf[s_abs:] = h[:n - s_abs]
            hf[:s_abs] = 0.0
            hb = np.empty_like(h)
            hb[:n - s_abs] = h[s_abs:]
            hb[n - s_abs:] = 0.0
        g = 1.0 - (hf + hb)
        v = h - step * g
        h = project_box_sum(v, target)
        step *= decay
    return best_h, best_c

def max_overlap_and_shift(h):
    corr = correlation_full(h, 1 - h)
    idx = int(np.argmax(corr))
    return corr[idx], idx - (h.size - 1)

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0, init_factor=0.5, decay=0.999):
    """Specialised local search moving mass from i to i+shift."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            s_abs = -shift
            i_range = np.arange(s_abs, n)
            j_range = i_range - s_abs
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    return best_h, best_c

def generate_candidates(rng, n, target):
    """A small diverse pool of starter vectors."""
    cand = []
    # random binary
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    cand.append(h)
    # Beatty (Sturmian) sequence
    alpha = (np.sqrt(5) - 1) / 2
    hb = (np.mod(np.arange(n) * alpha, 1.0) < 0.5).astype(float)
    cand.append(enforce_sum(hb, target, rng))
    # simple thresholded sines
    xs = np.arange(n)
    for f in (2, 3, 5):
        hs = (np.sin(2 * np.pi * f * xs / n) > 0).astype(float)
        cand.append(enforce_sum(hs, target, rng))
    # few random real vectors
    for _ in range(2):
        h = rng.random(n)
        cand.append(enforce_sum(h, target, rng))
    return cand
