def project_bounds_sum(v, target, lo=0.0, hi=1.0, tol=1e-12, max_iter=100):
    # find lambda such that sum(clip(v + lam, lo, hi)) = target
    low = -np.max(v - lo)  # ensure v+lam <= lo?
    high = hi - np.min(v)  # ensure v+lam >= hi?
    # Actually need lower bound lam such that all v+lam <= lo (i.e., v+lam <= lo => lam <= lo - v_i for all i => lam <= min_i (lo - v_i) = lo - max(v)
    # So low = lo - max(v). high = hi - min(v).
    low = lo - np.max(v)
    high = hi - np.min(v)
    for _ in range(max_iter):
        mid = (low + high) / 2
        proj = np.clip(v + mid, lo, hi)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s < target:
            low = mid
        else:
            high = mid
    # after loop, return proj
    return np.clip(v + (low+high)/2, lo, hi)
```

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust so that sum(h) == target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final correction if needed
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h with Σh = n/2 and 0≤h≤1."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):

        j = rng.integers(0, n)
        if i == j:

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple hill‑climb with several random perturbations per iteration."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):

                h = new_h

CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    target = CURRENT_TARGET
    diff = target - x.sum()

    return x

def max_overlap_and_shift(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    final_c = compute_c5(best_h)
    return best_h, final_c


# --------------------------------------------------------------
#  Projection onto box [0,1] with exact sum
# --------------------------------------------------------------
def project_bounds_sum(v, target, lo=0.0, hi=1.0, tol=1e-12, max_iter=100):
    """Find lambda so that sum(clip(v+lambda, lo, hi)) = target."""
    low = lo - np.max(v)
    high = hi - np.min(v)
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        proj = np.clip(v + mid, lo, hi)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) / 2.0, lo, hi)

#  Balancing step: maximise the minimum autocorrelation
# --------------------------------------------------------------
def balance_min_corr(h, target, max_iters, time_limit):
    """Gradient‑ascent on the minimum autocorrelation (equivalently minimise C5)."""
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha = 0.08
    decay = 0.9995
    n = h.size
    pad_len = n - 1
    while True:
        if time.time() - start > time_limit:
            break
        # autocorrelation (full)
        autocorr = np.correlate(h, h, mode="full")
        s_star = int(np.argmin(autocorr)) - (n - 1)
        # build padded array once
        pad = np.concatenate([np.zeros(pad_len), h, np.zeros(pad_len)])
        idx = pad_len + s_star
        v_pos = pad[idx: idx + n]                     # h shifted by s_star
        v_neg = pad[pad_len - s_star: pad_len - s_star + n]  # h shifted by -s_star
        g = v_pos + v_neg
        d = g - np.mean(g)          # zero‑sum direction
        h_new = h + alpha * d
        h = project_bounds_sum(h_new, target)
        # occasional exact evaluation
        if (np.random.rand() < 0.02):
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        alpha *= decay
        if alpha < 1e-6:
            alpha = 0.0  # stop moving
    # final check
    final_c = compute_c5(best_h)
    return best_h, final_c
