def gradient_worst_lag(h, lag):
    n = h.size
    G = np.zeros(n)
    if lag >= 0:
        # i from 0 to n-lag-1: + (1 - h[i+lag])
        G[:n-lag] = 1 - h[lag:]
        # i from lag to n-1: - h[i-lag]
        G[lag:] -= h[:n-lag]
    else:
        L = -lag
        # i from L to n-1: + (1 - h[i-L])
        G[L:] = 1 - h[:n-L]
        # i from 0 to n-L-1: - h[i+L]
        G[:n-L] -= h[L:]
    return G

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def gradient_worst_lag(h, lag):
    n = h.size
    G = np.zeros(n)
    if lag >= 0:
        G[: n - lag] = 1 - h[lag:]
        G[lag:] -= h[: n - lag]
    else:
        L = -lag
        G[L:] = 1 - h[: n - L]
        G[: n - L] -= h[L:]
    return G

def gradient_descent_opt(h0, rng, max_steps, time_limit):

    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

        h[j] += eps

    return best_h, best_c

def sum_eq_constraint(x):

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=1500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ---- main phase (70% of time) ----
    main_deadline = start + budget_s * 0.7
    beta_list = [10, 50, 200, 500, 1000, 2000, 5000, 10000, 20000]
    while time.time() < main_deadline:
        # restart from perturbed best or fresh
        if rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)

        # gradient descent
        gd_h, gd_c = gradient_descent_opt(
            start_vec, rng, max_steps=6000, time_limit=main_deadline
        )
        if gd_c < best_c:
            best_h, best_c = gd_h, gd_c

        # SLSQP ladder
        for beta in beta_list:
            if time.time() >= main_deadline:
                break
            try:
                ref = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=1200)
            except Exception:
                continue
            rc = compute_c5(ref)
            if rc < best_c:
                best_h, best_c = ref, rc

    # ---- upsample if time permits (up to 960 points) ----
    remaining = budget_s - (time.time() - start)
    if remaining > budget_s * 0.1:
        target_n = min(960, max(720, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)
            # gradient descent on finer grid
            up_deadline = start + budget_s
            gd_h, gd_c = gradient_descent_opt(
                up_h, rng, max_steps=12000, time_limit=up_deadline
            )
            if gd_c < best_c:
                best_h, best_c, n = gd_h, gd_c, target_n
            # further SLSQP refinement
            for beta in beta_list[-5:]:
                if time.time() >= start + budget_s:
                    break
                try:
                    ref = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=1200)
                except Exception:
                    continue
                rc = compute_c5(ref)
                if rc < best_c:
                    best_h, best_c = ref, rc
                    n = target_n

    # ---- final quick hill‑climb ----
    end_deadline = start + budget_s
    step_scale = 0.3
    cur_h = best_h.copy()
    cur_c = best_c
    for _ in range(8000):
        if time.time() > end_deadline:
            break
        new_h = perturb_h(cur_h, rng, scale=step_scale)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_c, cur_h = new_c, new_h
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        step_scale *= 0.9995

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
