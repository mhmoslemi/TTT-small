def lbfgsb_optimize(x0, beta, target, nfft, max_iter):
    def obj(x):
        corr = np.fft.irfft(np.fft.rfft(x, nfft) * np.fft.rfft(1-x, nfft), nfft)
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp stable
        lse = m + np.log(np.sum(np.exp(bc - m)))
        soft = lse / beta
        penalty = 1e5 * (np.sum(x) - target)**2
        return soft + penalty
    bounds = [(0.0, 1.0)] * len(x0)
    res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly with projection
    x = project_sum(x, target, np.random.default_rng())
    return x

def perturb_h(h, rng, scale):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    # ensure distinct?
    while j == i:
        j = rng.integers(n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    # bound eps
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # adjust for slight sum error due to numeric
    # but sum remains unchanged because we transfer epsilon.
    return new_h

def hill_climb(h0, rng, max_iters=5000, init_step=0.5, temp_start=0.001, temp_end=1e-6):

def slsqp_optimize(x0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA, GLOBAL_NFFT
    CURRENT_TARGET = target
    CURRENT_BETA = beta
    nfft = GLOBAL_NFFT
    def obj(x):
        corr = np.fft.irfft(np.fft.rfft(x, nfft) * np.fft.rfft(1 - x, nfft), nfft)
        bc = CURRENT_BETA * corr
        m = np.max(bc)
        # use log-sum-exp stable
        lse = m + np.log(np.sum(np.exp(bc - m)))
        return lse / CURRENT_BETA
    bounds = [(0.0, 1.0)] * len(x0)
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(obj, x0, method="SLSQP", bounds=bounds, constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum precisely
    x = project_sum(x, target, rng)
    return x

def project_sum(h, target, rng):
    """Adjust h to satisfy sum target within tolerance, respecting 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # Try to fix using random indices
    for _ in range(5 * n):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        if diff > 0:
            # need to increase h[i]
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                continue
        else:
            # need to decrease
            add = max(diff, -h[i])
            if add >= 0:
                continue
        h[i] += add
        diff = target - h.sum()
    # If still not exact, distribute uniformly
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # last resort: adjust first element
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)

def project_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0<=h<=1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # try random fixes
    for _ in range(5 * n):
        if abs(diff) < 1e-12:

def compute_c5_fft(h):
    """Exact C5 for a discretised step function using FFT."""
    n = h.shape[0]
    nfft = 2 * n - 1
    corr = np.fft.irfft(
        np.fft.rfft(h, nfft) * np.fft.rfft(1.0 - h, nfft), nfft
    )
    return np.max(corr) * (2.0 / n)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def softmax_obj(x):
    nfft = GLOBAL_NFFT
    corr = np.fft.irfft(np.fft.rfft(x, nfft) * np.fft.rfft(1.0 - x, nfft), nfft)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA


def slsqp_optimize(x0, beta, max_iter=400):

    n = x0.size

        softmax_obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    rng = np.random.default_rng()
    x = project_sum(x, n / 2.0, rng)

                c_tmp = compute_c5_fft(h_tmp)
                if c_tmp < global_best_c:
                    global_best_h, global_best_c = h_tmp, c_tmp
            except Exception:
                pass

        if final_time > 12.0:
            h_final, c_final = pair_transfer_optimize(
                global_best_h,
                rng,
                max_iters=15000,
                time_limit=final_time * 0.9,
                init_factor=0.5,
                decay=0.999,
            )
            if c_final < global_best_c:
                global_best_h, global_best_c = h_final, c_final

    # -------------------------------------------------
    # 6. Final clean‑up and return
    # -------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = project_sum(global_best_h, target, rng)
    final_c = compute_c5_fft(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
