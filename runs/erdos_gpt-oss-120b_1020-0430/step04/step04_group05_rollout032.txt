def smooth_gradient_descent(h, target, rng, time_budget, start_time, betas, steps_per_beta, step_scale=0.5):

    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in betas:
        # compute step size schedule for this beta
        # maybe fix iteration count based on remaining time
        steps = steps_per_beta
        for it in range(steps):
            if time.time() - start_time > time_budget:
                break
            # compute corr
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            # compute weights safely
            corr_shifted = corr - max_corr
            w = np.exp(beta * corr_shifted)
            w_sum = w.sum()
            w = w / w_sum
            # convolution using FFT
            m = int(2 ** np.ceil(np.log2(len(w) + len(h) - 1)))
            # compute FFTs
            W = np.fft.rfft(w, n=m)
            H = np.fft.rfft(h, n=m)
            conv = np.fft.irfft(W * H, n=m)
            # extract central n
            start_idx = len(w) - 1
            y_backward = conv[start_idx : start_idx + len(h)]
            # forward: using reversed w
            # use same FFT but compute with reversed w
            w_rev = w[::-1]
            W_rev = np.fft.rfft(w_rev, n=m)
            conv_rev = np.fft.irfft(W_rev * H, n=m)
            y_forward = conv_rev[start_idx : start_idx + len(h)]
            # gradient
            grad = np.ones_like(h) - y_forward - y_backward
            # step size beneficial: maybe 0.5 / (1+it)
            step = step_scale / np.sqrt(it + 1)
            # update
            v = h - step * grad
            # project
            h = project_onto_sum_box(v, target)
            # evaluate C5
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    return best_h, best_c

def project_onto_sum_box(v, target):
    # Project onto set {x in [0,1]^n, sum x = target}
    # Solve for lambda using binary search
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - lam, 0.0, 1.0)
    # In rare case due to numerical rounding sum may be slightly off target; adjust
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff uniformly among non-bound elements
        free = (x > 0 + 1e-12) & (x < 1 - 1e-12)
        nfree = free.sum()
        if nfree > 0:
            x[free] += diff / nfree
            x = np.clip(x, 0.0, 1.0)

def project_onto_sum_box(v, target):
    """Euclidean projection of v onto {0<=x<=1, sum x = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        nfree = free.sum()
        if nfree:
            x[free] += diff / nfree
            x = np.clip(x, 0.0, 1.0)
    return x

    b_rev = b[::-1]
    m = int(2 ** np.ceil(np.log2(2 * n - 1)))
    A = np.fft.rfft(h, n=m)
    B = np.fft.rfft(b_rev, n=m)
    conv = np.fft.irfft(A * B, n=m)
    return conv[: 2 * n - 1]  # length 2n-1, lag -(n-1)..(n-1)


def compute_c5(h):
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Uniformly random point in [0,1]^n with sum = n/2."""
    x = rng.random(n)
    x = np.clip(x, 0.0, 1.0)
    # scale to correct sum via projection
    target = n / 2.0
    return project_onto_sum_box(x, target)


# ---------- main algorithm ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_sum_box(h0, h0.size / 2.0)
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- quick subgradient phase on the original size -------------------
    time_left = budget_s - (time.time() - t0)
    sub_budget = min(0.12 * budget_s, time_left * 0.4)
    if time_left > sub_budget:
        n = best_h.size
        target = n / 2.0
        max_iters = 2000
        for it in range(max_iters):
            if time.time() - t0 > t0 + sub_budget:
                break
            corr = compute_corr_fft(best_h)
            max_corr = corr.max()
            eps = 1e-12
            idxs = np.where(corr >= max_corr - eps)[0]
            g = np.zeros_like(best_h)
            for idx in idxs:
                L = idx - (n - 1)
                # contribution of this lag
                g += 1.0
                if L >= 0:
                    g[: n - L] -= best_h[L:]
                    g[L:] -= best_h[: n - L]
                else:
                    k = -L
                    g[k:] -= best_h[: n - k]
                    g[: n - k] -= best_h[k:]
            if idxs.size:
                g /= idxs.size  # convex combination
            step = 0.5 / np.sqrt(it + 1)
            cand = project_onto_sum_box(best_h - step * g, target)
            cand_c = compute_c5(cand)
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()

    # ---- optional up‑sampling -------------------------------------------
    time_left = budget_s - (time.time() - t0)
    if time_left > 120 and best_h.size < 720:
        n0 = best_h.size
        n_up = min(720, n0 * 2)
        x_old = (np.arange(n0) + 0.5) * (2.0 / n0)
        x_up = (np.arange(n_up) + 0.5) * (2.0 / n_up)
        h_up = np.interp(x_up, x_old, best_h)
        h_up = project_onto_sum_box(h_up, n_up / 2.0)
    else:
        h_up = best_h.copy()
        n_up = best_h.size
    # keep best on the current grid as fallback
    best_h_up = h_up.copy()
    best_c_up = compute_c5(h_up)

    # ---- smooth gradient descent with increasing beta --------------------
    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    steps_per_beta = 2500
    target_up = n_up / 2.0
    h_cur = h_up.copy()

    for beta in betas:
        if time.time() - t0 > budget_s:
            break
        for it in range(steps_per_beta):
            if time.time() - t0 > budget_s:
                break
            corr = compute_corr_fft(h_cur)
            max_corr = corr.max()
            # softmax weights (stable)
            w = np.exp(beta * (corr - max_corr))
            w /= w.sum()
            # convolution w * h (backward term)
            m = int(2 ** np.ceil(np.log2(len(w) + n_up - 1)))
            H = np.fft.rfft(h_cur, n=m)
            W = np.fft.rfft(w, n=m)
            conv = np.fft.irfft(W * H, n=m)
            start = len(w) - 1
            y_back = conv[start : start + n_up]
            # forward term uses reversed w
            W_rev = np.fft.rfft(w[::-1], n=m)
            conv_rev = np.fft.irfft(W_rev * H, n=m)
            y_fwd = conv_rev[start : start + n_up]
            grad = np.ones(n_up) - y_fwd - y_back
            step = 0.6 / np.sqrt(it + 1)
            h_cur = project_onto_sum_box(h_cur - step * grad, target_up)

            cur_c = compute_c5(h_cur)
            if cur_c < best_c_up:
                best_c_up, best_h_up = cur_c, h_cur.copy()
        # update global best if this grid beats previous one
        if best_c_up < best_c:
            best_c, best_h = best_c_up, best_h_up.copy()

    # ---- final local hill‑climb -----------------------------------------
    time_left = budget_s - (time.time() - t0)
    if time_left > 30:
        h_cur = best_h.copy()
        cur_c = best_c
        n = h_cur.size
        target = n / 2.0
        step = 0.3
        for _ in range(5000):
            if time.time() - t0 > budget_s:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h_cur[i], 1.0 - h_cur[j])
            if max_eps <= 0:
                continue
            eps = (rng.random() - 0.5) * 2.0 * max_eps * step
            new_h = h_cur.copy()
            new_h[i] -= eps
            new_h[j] += eps
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                cur_c, h_cur = new_c, new_h
                if new_c < best_c:
                    best_c, best_h = new_c, new_h.copy()
            step *= 0.9995

    # ---- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_sum_box(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
