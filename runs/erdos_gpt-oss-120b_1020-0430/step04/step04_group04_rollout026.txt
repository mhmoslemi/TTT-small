def targeted_perturb(h, grad, scale=0.1, rng):
    # grad is array of same size representing dCmax/dh_i (for current max shift)
    # pick i from indices where grad > 0 (i.e., moving mass out reduces Cmax)
    # pick j where grad < 0 (moving mass in reduces Cmax)
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0:
        return h.copy()
    eps = scale * max_transfer * rng.random()
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

CURRENT_TARGET = None
CURRENT_BETA = None


def enforce_sum(h, target, rng, max_iter=5000):
    """Force Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()

        idx = np.where(free)[0]

            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Create a random vector in [0,1]^n with Σh = n/2."""
    target = n / 2.0
    h = np.full(n, 0.5)
    # random transfers
    for _ in range(n * 4):

        j = rng.integers(0, n)
        if i == j:

        max_eps = min(h[i], 1.0 - h[j])

        h[j] += eps

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)

def compute_corr(h):
    """Full correlation conv = h * (1-h) reversed, scaled by dx."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L] * (2.0 / n)
    return conv

    """Exact C5 for a sampled step function."""
    return compute_corr(h).max()

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

def targeted_hill_climb(h0, rng, max_iters, scale_start=0.5):

    best_h = h.copy()

            s = shift
            grad[: n - s] = 1.0 - h[s:]
            grad[s:] -= h[: n - s]
        else:
            s = -shift
            grad[s:] = 1.0 - h[: n - s]
            grad[: n - s] -= h[s:]
        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer > 0:
                eps = scale * max_transfer * rng.random()
                new_h = h.copy()
                new_h[i] -= eps
                new_h[j] += eps
                new_c = compute_c5(new_h)
                if new_c <= c:
                    h = new_h
        else:
            h = perturb_h(h, rng, scale=scale)
        scale = max(scale * 0.9995, 1e-4)
        if best_c < 0.3808:
            break
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initialise vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    # upsample to a finer grid if beneficial
    target_n = 720
    if n0 < target_n:
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h0 = np.interp(new_x, old_x, h0)
    n = target_n
    h0 = enforce_sum(h0, n / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # optimisation loop
    while time.time() - start_time < budget_s * 0.85:
        # pick a start point
        if rng.random() < 0.75:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)

        # targeted hill climbing
        hc_h, hc_c = targeted_hill_climb(start_vec, rng, max_iters=3000, scale_start=0.5)
        if hc_c < best_c:
            best_c, best_h = hc_c, hc_h.copy()

        # simulated annealing refinement
        sa_h, sa_c = simulated_annealing(best_h, rng,
                                          max_iter=6000,
                                          init_temp=1e-2,
                                          final_temp=1e-5,
                                          init_scale=0.5,
                                          final_scale=0.02)
        if sa_c < best_c:
            best_c, best_h = sa_c, sa_h.copy()

        # SLSQP surrogate ladder
        for beta in (50, 200, 800, 2000, 5000):
            if time.time() - start_time > budget_s * 0.85:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

        if best_c < 0.38078:  # safe early stop
            break

    # final cleanup
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
