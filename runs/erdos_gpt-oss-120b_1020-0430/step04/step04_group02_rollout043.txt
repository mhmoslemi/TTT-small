def compute_c5_fft(h):
    n = h.size
    rev = (1 - h)[::-1]
    M = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, M)
    R = np.fft.rfft(rev, M)
    conv = np.fft.irfft(H * R, M)[:2*n-1]
    return np.max(conv) * (2.0 / n)

def targeted_move_product(h, k, rng, scale):
    n = h.size
    if k == 0:
        return perturb_h_move(h, rng, scale)
    if k > 0:
        a = h[: n - k]
        b = h[k:]
        low_mask = (a < 0.5) & (b < 0.5)
        high_mask = (a > 0.5) & (b > 0.5)
        if not low_mask.any() or not high_mask.any():
            return perturb_h_move(h, rng, scale)
        i_low = rng.choice(np.flatnonzero(low_mask))
        i_high = rng.choice(np.flatnonzero(high_mask))
        max_eps = min(1.0 - a[i_low], 1.0 - b[i_low], a[i_high], b[i_high])
        if max_eps <= 0:
            return perturb_h_move(h, rng, scale)
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[i_low] += eps
        new_h[i_low + k] += eps
        new_h[i_high] -= eps
        new_h[i_high + k] -= eps
        return new_h
    else:
        k0 = -k
        a = h[k0:]          # later index
        b = h[: n - k0]     # earlier index
        low_mask = (a < 0.5) & (b < 0.5)
        high_mask = (a > 0.5) & (b > 0.5)

            return perturb_h_move(h, rng, scale)
        i_low = rng.choice(np.flatnonzero(low_mask))
        i_high = rng.choice(np.flatnonzero(high_mask))

def perturb_h_move(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    # Determine direction
    if rng.random() < 0.5:
        # move from i to j
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h
    else:
        # move from j to i
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
        return new_h

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def correlation_full_fft(h):
    n = h.size
    rev = (1.0 - h)[::-1]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:2 * n - 1]
    return conv

def compute_c5_fast(h):
    n = h.size
    conv = correlation_full_fft(h)
    return np.max(conv) * (2.0 / n)

import time
import numpy as np

# ----- projection & sampling -----
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)


# ----- elementary moves -----
def perturb_h_move(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    if rng.random() < 0.5:
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        return nh
    else:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[j] -= eps
        nh[i] += eps
        return nh


def targeted_move_product(h, k, rng, scale):
    n = h.size
    if k == 0:
        return perturb_h_move(h, rng, scale)
    if k > 0:
        a = h[: n - k]
        b = h[k:]
        low = (a < 0.5) & (b < 0.5)
        high = (a > 0.5) & (b > 0.5)
        if not low.any() or not high.any():
            return perturb_h_move(h, rng, scale)
        i_low = rng.choice(np.flatnonzero(low))
        i_high = rng.choice(np.flatnonzero(high))
        max_eps = min(1.0 - a[i_low], 1.0 - b[i_low], a[i_high], b[i_high])
        if max_eps <= 0:
            return perturb_h_move(h, rng, scale)
        eps = rng.random() * max_eps * scale
        nh = h.copy()
        nh[i_low] += eps
        nh[i_low + k] += eps
        nh[i_high] -= eps
        nh[i_high + k] -= eps
        return nh
    # k < 0
    k0 = -k
    a = h[k0:]
    b = h[: n - k0]
    low = (a < 0.5) & (b < 0.5)
    high = (a > 0.5) & (b > 0.5)
    if not low.any() or not high.any():
        return perturb_h_move(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low))
    i_high = rng.choice(np.flatnonzero(high))
    max_eps = min(1.0 - a[i_low], 1.0 - b[i_low], a[i_high], b[i_high])
    if max_eps <= 0:
        return perturb_h_move(h, rng, scale)
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    idx_low_a = i_low + k0
    idx_high_a = i_high + k0
    nh[idx_low_a] += eps
    nh[i_low] += eps
    nh[idx_high_a] -= eps
    nh[i_high] -= eps
    return nh


# ----- FFT correlation -----
def correlation_full_fft(h):
    n = h.size
    rev = (1.0 - h)[::-1]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[: 2 * n - 1]
    return conv


def compute_c5_fast(h):
    n = h.size
    conv = correlation_full_fft(h)
    return np.max(conv) * (2.0 / n)


# ----- hill climbing -----
def hill_climb(h0, rng, max_iters, init_scale, p_target):
    h = h0.copy()
    n = h.size
    corr = correlation_full_fft(h)
    cur_c = np.max(corr) * (2.0 / n)
    best_h, best_c = h.copy(), cur_c
    scale = init_scale
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        idx = int(np.argmax(corr))
        k_star = idx - (n - 1)
        if rng.random() < p_target:
            cand = targeted_move_product(h, k_star, rng, scale)
        else:
            cand = perturb_h_move(h, rng, scale)
        cand_corr = correlation_full_fft(cand)
        cand_c = np.max(cand_corr) * (2.0 / n)
        if cand_c < cur_c - 1e-12:
            h, corr, cur_c = cand, cand_corr, cand_c
            if cur_c < best_c - 1e-12:
                best_h, best_c = h.copy(), cur_c
        if (it + 1) % decay == 0:
            scale = max(scale * 0.8, 1e-6)
    return best_h, best_c


# ----- main entry point -----
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initial resolution
    N0 = 1800
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != N0:
            old_x = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
            new_x = (np.arange(N0) + 0.5) * (2.0 / N0)
            h0 = np.interp(new_x, old_x, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, N0 / 2.0)
    else:
        h0 = random_feasible_vector(N0, rng)

    best_h, best_c = h0.copy(), compute_c5_fast(h0)

    # hill‑climbing phase on N0
    phase_time = budget_s * 0.78
    batch = 4000
    scale, p_target = 0.3, 0.5
    while time.time() - start < phase_time:
        h_new, c_new = hill_climb(best_h, rng, batch, scale, p_target)
        if c_new < best_c - 1e-12:
            best_h, best_c = h_new, c_new
            scale = max(scale * 0.9, 1e-5)
            p_target = min(p_target + 0.05, 0.9)
        else:
            best_h = perturb_h_move(best_h, rng, scale * 2.0)
            best_c = compute_c5_fast(best_h)
        if best_c < 0.3808:
            break

    # optional up‑sampling if time permits
    elapsed = time.time() - start
    if budget_s - elapsed > budget_s * 0.15:
        N1 = N0 * 2
        old_x = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
        new_x = (np.arange(N1) + 0.5) * (2.0 / N1)
        h_big = np.interp(new_x, old_x, best_h)
        h_big = np.clip(h_big, 0.0, 1.0)
        h_big = project_box_sum(h_big, N1 / 2.0)
        best_h, best_c = h_big, compute_c5_fast(h_big)

        batch2 = 6000
        scale2, p_target2 = 0.3, 0.5
        while time.time() - start < budget_s - 0.1:
            h_new, c_new = hill_climb(best_h, rng, batch2, scale2, p_target2)
            if c_new < best_c - 1e-12:
                best_h, best_c = h_new, c_new
                scale2 = max(scale2 * 0.9, 1e-5)
                p_target2 = min(p_target2 + 0.05, 0.9)
            else:
                best_h = perturb_h_move(best_h, rng, scale2 * 2.0)
                best_c = compute_c5_fast(best_h)
            if best_c < 0.3808:
                break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5_fast(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
