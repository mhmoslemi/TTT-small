def hill_climb_targeted(h_start, rng, max_iters=2000, init_step=0.5, top_m=3, p_target=0.4):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.shape[0]
    decay_interval = max_iters // 10
    no_improve = 0
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr_vals = corr * (2.0 / n)
        # find top_lags
        # top_m indices of largest values
        # Use np.argpartition for speed
        if top_m < corr_vals.size:
            top_idx = np.argpartition(-corr_vals, top_m)[:top_m]
        else:
            top_idx = np.arange(corr_vals.size)
        # Choose k_star randomly among them weighted by value
        k_star = rng.choice(top_idx) - (n - 1)
        improved = False
        # try few candidates
        for _ in range(5):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        # adapt step: if no improvement for many iter, increase step slightly?
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
            # also reset no_improve
    return h, best_c
```

def simulated_annealing(h_start, rng, max_steps=20000, init_temp=0.01, final_temp=1e-5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    temp = init_temp
    decay = (final_temp / init_temp) ** (1.0 / max_steps)
    for step in range(max_steps):
        # maybe random targeted or perturb
        k = rng.integers(-h.size+1, h.size) # random shift
        if rng.random() < 0.5:
            cand = targeted_move(h, rng, k, scale=0.1)
        else:
            cand = perturb_h(h, rng, scale=0.1)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()
        if c < compute_c5(h) or rng.random() < np.exp(-(c - compute_c5(h)) / temp):
            h = cand
        temp *= decay
    return best_h, best_c

        else:

        # hill climb with more iters
        h_climb2, c_climb2 = hill_climb_targeted(h_up, rng, max_iters=6000, init_step=0.25, top_m=5, p_target=0.5)
        if c_climb2 < best_c:

        # SLSQP refinements again

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

# ---------- basic ops ----------
def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i_low] += eps
    nh[i_low + k] += eps
    nh[i_high] -= eps
    nh[i_high + k] -= eps
    return nh


def hill_climb_targeted(h_start, rng, max_iters=2000, init_step=0.5,
                       top_m=3, p_target=0.45):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.shape[0]
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        corr_vals = corr * (2.0 / n)

        if top_m < corr_vals.size:
            top_idx = np.argpartition(-corr_vals, top_m)[:top_m]
        else:
            top_idx = np.arange(corr_vals.size)

        idx = rng.choice(top_idx)
        k = abs(int(idx - (n - 1)))
        if k == 0:
            k = rng.integers(1, n)

        improved = False
        for _ in range(5):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand

        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)

    return h, best_c


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- pattern helpers ----------
def chebyshev_candidate(n, m):
    t = (np.arange(n) + 0.5) * (2.0 / n) - 1.0
    T = np.cos(m * np.arccos(t))
    h = 0.5 + 0.5 * T
    return project_box_sum(np.clip(h, 0.0, 1.0), n / 2.0)


def gaussian_candidate(n, sigma=0.5):
    x = (np.arange(n) + 0.5) * (2.0 / n) - 1.0
    h = 0.5 + 0.5 * np.exp(-x ** 2 / (2 * sigma ** 2))
    return project_box_sum(np.clip(h, 0.0, 1.0), n / 2.0)


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise base vector
    if "initial_h_values" in globals():

        cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)

                c_ref = compute_c5(h_ref)
