def compute_gradient(h, target):
    n = h.size
    i = np.arange(n)
    conv = correlate_full(h)
    max_val = conv.max()
    ks = np.where(conv >= max_val - 1e-12)[0]  # indices achieving max near equality
    grad = np.zeros_like(h)

        s = k - (n - 1)

        plus_mask = mask & (i + s >= 0) & (i + s < n)
        grad[plus_mask] -= h[i[plus_mask] + s]
    return grad

def project_sum_bounds(h, target, eps=1e-12):
    """Clip h to [0,1] and adjust to make sum == target, preserving bounds."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) <= eps:
        return h
    # Indices of non-bound entries
    free = np.where((h > 0.0 + eps) & (h < 1.0 - eps))[0]
    # If free empty, fallback to distribute among all via scaling maybe.
    # We'll relax eps for detection.
    if free.size == 0:
        # distribute uniformly scaling
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        # may still not exact; call enforce_sum fallback
        # But diff should be tiny because we adjust equally.
        return h
    # Adjust in iterative water-filling:
    while abs(diff) > eps and free.size > 0:
        # compute amount to add per free element
        amt = diff / free.size
        # Determine where adding amt would exceed bounds
        if amt > 0:
            # limit to upper bound
            over = (h[free] + amt) - 1.0
            # over > 0 means would exceed
            over_idx = free[over > 0]
            if over_idx.size > 0:
                # set those to 1
                h[over_idx] = 1.0
                # recompute diff
                diff = target - h.sum()
                free = np.where((h > 0.0 + eps) & (h < 1.0 - eps))[0]
                continue
        else:
            # amt < 0, limit to lower bound
            under = (h[free] + amt)  # negative value <0
            under_idx = free[under < 0]
            if under_idx.size > 0:
                h[under_idx] = 0.0
                diff = target - h.sum()
                free = np.where((h > 0.0 + eps) & (h < 1.0 - eps))[0]
                continue
        # if no bound violation
        h[free] += amt
        diff = target - h.sum()
        # update free again
        free = np.where((h > 0.0 + eps) & (h < 1.0 - eps))[0]
    return h

def compute_gradient(h):
    n = h.size
    i = np.arange(n)
    conv = correlate_full(h)
    max_val = conv.max()
    ks = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    for k in ks:
        s = k - (n - 1)
        mask = (i - s >= 0) & (i - s < n)
        grad[mask] += 1.0
        grad[mask] -= h[i[mask] - s]
        plus_mask = mask & (i + s >= 0) & (i + s < n)
        grad[plus_mask] -= h[i[plus_mask] + s]
    return grad

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps

    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):

    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------------------------------------------
# Simple greedy hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# --------------------------------------------------------------
# Gradient‑descent on the max‑overlap objective
# --------------------------------------------------------------
def compute_gradient(h):
    """Sub‑gradient of maxₖ Σᵢ h[i](1‑h[i‑k]) at h."""
    n = h.size
    i = np.arange(n)
    conv = correlate_full(h)
    max_val = conv.max()
    ks = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    for k in ks:
        s = k - (n - 1)                     # shift
        mask = (i - s >= 0) & (i - s < n)   # i contributes to sum
        grad[mask] += 1.0
        grad[mask] -= h[i[mask] - s]
        plus_mask = mask & (i + s >= 0) & (i + s < n)
        grad[plus_mask] -= h[i[plus_mask] + s]
    return grad


def gradient_descent(h0, target, rng, deadline, step0=0.02, decay=0.9997):
    """Projected gradient descent preserving sum, with time stop."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    n = h.size
    i = np.arange(n)
    it = 0
    while time.time() < deadline and it < 200000:
        grad = compute_gradient(h)
        d = -grad
        d = d - d.mean()            # keep sum unchanged
        h = h + step * d
        np.clip(h, 0.0, 1.0, out=h)
        # re‑balance sum if clipping disturbed it
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) <= 1e-12:
                    break
                idx = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                else:
                    add = max(diff, -h[idx])
                h[idx] += add
                diff = target - h.sum()
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        step *= decay
        it += 1
    return best_h, best_c


# --------------------------------------------------------------
# Targeted greedy reduction of the worst shift
# --------------------------------------------------------------
def reduce_max_overlap(h0, target, rng, deadline, max_trials=2000):
    """Attempt to lower the current worst overlap by direct transfers."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    n = best_h.size
    i_arr = np.arange(n)
    trials = 0
    while time.time() < deadline and trials < max_trials:
        conv = correlate_full(best_h)
        k = np.argmax(conv)
        s = k - (n - 1)
        mask = (i_arr - s >= 0) & (i_arr - s < n)
        src_idx = np.where(mask & (best_h > 1e-12))[0]
        dst_idx = np.where(mask & (best_h < 1 - 1e-12))[0]
        if src_idx.size == 0 or dst_idx.size == 0:
            break
        behind_src = best_h[src_idx - s]
        behind_dst = best_h[dst_idx - s]
        src_order = np.argsort(behind_src)           # low behind first
        dst_order = np.argsort(-behind_dst)          # high behind first
        moved = False
        for si in src_order[:10]:
            i = src_idx[si]
            for dj in dst_order[:10]:
                j = dst_idx[dj]
                diff_val = behind_dst[dj] - behind_src[si]
                if diff_val > 1e-6:
                    eps = min(best_h[i], 1 - best_h[j])
                    eps = min(eps, diff_val * 0.5)
                    if eps <= 0:
                        continue
                    best_h[i] -= eps
                    best_h[j] += eps
                    moved = True
                    break
            if moved:
                break
        if not moved:
            break
        c = compute_c5(best_h)
        if c < best_c:
            best_c, best_h = c, best_h.copy()
        trials += 1
    return best_h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ---- 0. Initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target, rng)
    else:
        n = 720
        target = n / 2.0
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- 1. Quick diverse candidates ------------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    def maybe_update(h):
        nonlocal best_h, best_c
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # triangular tent
    tri = 1.0 - np.abs(xs - 1.0)
    maybe_update(enforce_sum(np.clip(tri, 0.0, 1.0), target, rng))

    # sin / cos families
    for f in range(1, 5):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        maybe_update(enforce_sum(np.clip(s, 0.0, 1.0), target, rng))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        maybe_update(enforce_sum(np.clip(c, 0.0, 1.0), target, rng))

    # block and alternating
    block = np.zeros(n)
    block[: n // 2] = 1.0
    maybe_update(enforce_sum(block, target, rng))

    alt = np.tile([0.0, 1.0], n // 2)
    maybe_update(enforce_sum(alt, target, rng))

    # a few random feasibles
    for _ in range(5):
        maybe_update(random_feasible_vector(n, rng))

    # ---- 2. Continuous optimisation (SLSQP + hill‑climb) ----------------
    beta_list = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
    phase1_end = start_time + 0.5 * budget_s

    while time.time() < phase1_end:
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)
        start = enforce_sum(start, target, rng)

        for beta in beta_list:
            if time.time() >= phase1_end:
                break
            h_opt = slsqp_optimize(start, beta, max_iter=400)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
            start = h_opt

        if time.time() < phase1_end:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=2000, start_step=0.3)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

    # ---- 3. Projected gradient descent ---------------------------------
    gd_deadline = deadline - 5.0  # keep a small buffer for final polishing
    if time.time() < gd_deadline:
        h_gd, c_gd = gradient_descent(best_h, target, rng, gd_deadline,
                                      step0=0.02, decay=0.9997)
        if c_gd < best_c:
            best_c, best_h = c_gd, h_gd.copy()

    # ---- 4. Targeted reduction of worst shift -------------------------
    if time.time() < deadline:
        h_red, c_red = reduce_max_overlap(best_h, target, rng, deadline, max_trials=3000)
        if c_red < best_c:
            best_c, best_h = c_red, h_red.copy()

    # ---- 5. Final polishing (hill‑climb) --------------------------------
    if time.time() < deadline:
        h_hc2, c_hc2 = hill_climb(best_h, rng, max_iters=3000, start_step=0.2)
        if c_hc2 < best_c:
            best_c, best_h = c_hc2, h_hc2.copy()

    # ---- 6. Final feasibility enforcement --------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
