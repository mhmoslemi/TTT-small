def gradient_corr_weights(h, weights):  # weights vector size 2n-1
    n = h.size
    grad = np.zeros(n)
    # s ranges from -(n-1) .. n-1 (index offset = n-1)
    for idx, w in enumerate(weights):
        s = idx - (n-1)
        if w == 0:
            continue
        if s >= 0:
            # forward part: h[j+s] if j+s < n => produce array of size n: forward = np.concatenate([np.zeros(s), h[:n-s]])
            forward = np.empty(n)
            forward[:s] = 0.0
            forward[s:] = h[:n-s]
            # backward part: h[j-s] if j-s >=0 => backward = np.concatenate([h[s:], np.zeros(s)])
            backward = np.empty(n)
            backward[:n-s] = h[s:]
            backward[n-s:] = 0.0
        else:
            t = -s
            # forward: h[j-s] where s negative => forward = np.concatenate([h[t:], np.zeros(t)])
            forward = np.empty(n)
            forward[:n-t] = h[t:]
            forward[n-t:] = 0.0
            # backward: h[j+s] where s negative => backward = np.concatenate([np.zeros(t), h[:n-t]])
            backward = np.empty(n)
            backward[:t] = 0.0
            backward[t:] = h[:n-t]
        # gradient contribution: w * ((1 - forward) - backward)
        grad += w * ((1.0 - forward) - backward)
    return grad

def project_capped_simplex(v, target):
    lo = np.min(v) - 1.0  # lower bound on lambda
    hi = np.max(v)  # upper bound on lambda
    # perform 50 iterations
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # ensure sum very close to target; adjust small residual
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        # distribute diff across non-boundary entries
        free = (proj > 0) & (proj < 1)
        nfree = free.sum()
        if nfree > 0:
            proj[free] += diff / nfree
            # clip again
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def gradient_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    exp_w = np.exp(beta * (corr - max_val))
    w = exp_w / exp_w.sum()
    # convolution with w
    conv = np.convolve(w, h, mode='full')
    conv_rev = np.convolve(w[::-1], h, mode='full')
    offset = n - 1
    forward = conv_rev[offset:offset+n]  # length n
    backward = conv[offset:offset+n]
    grad = 1.0 - forward - backward
    return grad, corr.max() * (2.0 / n)  # return objective too

def compute_c5(h):
    """Discrete C5 = max correlation * (2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_capped_simplex(v, target):
    """Euclidean projection of v onto {0≤x≤1, sum(x)=target}."""
    lo = np.min(v) - 1.0
