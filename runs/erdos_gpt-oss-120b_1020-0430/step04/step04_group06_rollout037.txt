def project_sum_cap(h, target, tol=1e-12, max_iter=100):
    x = np.clip(h, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = x < 1 - 1e-12
            cap = (1 - x[free]).sum()
            if cap <= 0:
                break
            inc = min(diff, cap)
            # distribute proportional equally
            cnt = free.sum()
            add = inc / cnt
            x[free] += add
            # clip any that exceed 1 due to rounding
            x = np.clip(x, 0.0, 1.0)
        else:
            free = x > 1e-12
            cap = x[free].sum()
            if cap <= 0:
                break
            dec = min(-diff, cap)
            cnt = free.sum()
            sub = dec / cnt
            x[free] -= sub
            x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    # final tweak using random small adjustments if need
    return x

    c = np.correlate(h, 1.0 - h, mode='full')

    # Let's iterate over idx,w where w>0
    for idx, w in enumerate(p):

import time
import numpy as np

# -------------------------------------------------
# Helper functions
# -------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max()) * (2.0 / h.size)


def project_sum_cap(h, target, tol=1e-12, max_iter=100):
    """Project vector onto the box [0,1] while fixing the sum to target."""
    x = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - x.sum()
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = x < 1.0 - 1e-12
            if not free.any():
                break
            inc = min(diff, (1.0 - x[free]).sum())
            cnt = free.sum()
            add = inc / cnt
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            free = x > 1e-12
            if not free.any():
                break
            dec = min(-diff, x[free].sum())
            cnt = free.sum()
            sub = dec / cnt
            x[free] -= sub
            x = np.clip(x, 0.0, 1.0)
    return x


def gradient_from_corr(corr, h, beta):
    """Gradient of the log‑sum‑exp surrogate w.r.t. h."""
    N = h.size
    max_c = float(corr.max())
    diff = corr - max_c
    exp_vals = np.exp(beta * diff)
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        p = np.full_like(exp_vals, 1.0 / exp_vals.size)
    else:
        p = exp_vals / sum_exp

    g = np.ones(N)
    for idx, w in enumerate(p):
        if w < 1e-16:
            continue
        shift = idx - (N - 1)
        if shift >= 0:
            if shift != 0:
                g[: N - shift] -= w * h[shift:]
                g[shift:] -= w * h[: N - shift]
            else:
                g -= 2.0 * w * h
        else:
            s = -shift
            g[s:] -= w * h[: N - s]
            g[: N - s] -= w * h[s:]
    return g, max_c


def max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Local search moving mass from i to j to lower the worst overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift

        if i_range.size == 0:
            factor = init_factor
            continue

        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue

        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
    else:
        n = 360
        h = rng.random(n)
        h = np.clip(h, 0.0, 1.0)

    target = n / 2.0
    h = project_sum_cap(h, target)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ---- gradient‑descent on smoothed max ------------------------------
    beta_schedule = [10, 50, 200, 800, 3000, 10000]
    steps_per_beta = {
        10: 400,
        50: 600,
        200: 800,
        800: 1200,
        3000: 1800,
        10000: 2500,
    }

    # initial correlation
    corr = np.correlate(h, 1.0 - h, mode="full")

    for beta in beta_schedule:
        steps = steps_per_beta[beta]
        base_step = 0.5 / np.sqrt(beta)
        for it in range(steps):
            if time.time() - start_time > budget_s * 0.75:
                break
            grad, _ = gradient_from_corr(corr, h, beta)
            step = base_step * (1.0 - it / steps)
            h = h - step * grad
            h = np.clip(h, 0.0, 1.0)
            h = project_sum_cap(h, target)
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = float(corr.max()) * (2.0 / n)
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
        if time.time() - start_time > budget_s * 0.75:
            break

    # ---- final pair‑transfer refinement --------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10.0:
        # give most of the remaining time to the pair‑transfer stage
        time_limit = remaining * 0. ninety # typo fix needed

            best_c = c_opt
            best_h = h_opt.copy()

    # ---- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_cap(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
