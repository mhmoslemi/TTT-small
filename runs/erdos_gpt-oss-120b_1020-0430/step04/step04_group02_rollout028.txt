def softmax_obj_and_grad(h, beta):
    n = h.shape[0]
    # compute c as correlation
    corr = np.correlate(h, 1 - h, mode='full')
    maxc = np.max(corr)
    expb = np.exp(beta * (corr - maxc))
    Z = np.sum(expb)
    w = expb / Z
    # weight array w length 2n-1, index shift = -(n-1)...(n-1)
    # Split w:
    w_pos = w[n-1:]  # indices 0..n-1
    w_neg = w[:n-1]  # indices -(n-1)..-1
    # Compute pos contribution via convolution with reversed h
    conv_pos = np.convolve(w_pos, h[::-1], mode='full')
    # pos_contrib[j] = conv_pos[n-1 - j]
    pos_contrib = conv_pos[n-1::-1][:n]
    # Compute neg contribution
    conv_neg = np.convolve(w_neg, h, mode='full')
    # neg_contrib[j] = conv_neg[j-1] for j>=1, 0 otherwise
    neg_contrib = np.empty(n, dtype=h.dtype)
    neg_contrib[0] = 0.0
    neg_contrib[1:] = conv_neg[:n-1]
    # w0
    w0 = w[n-1]
    grad = 1.0 - (pos_contrib + neg_contrib + w0 * h)
    # objective value
    obj = (maxc + np.log(Z)) / beta
    return obj, grad

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -1.0, 1.0
    # Because shift cannot exceed difference maybe bigger.
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low+high)/2.0, lo, hi)
    # Minor adjust
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo) & (y < hi)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) * 0.5, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo) & (y < hi)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def softmax_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")          # length 2n-1
    maxc = np.max(corr)
    expb = np.exp(beta * (corr - maxc))
    Z = expb.sum()
    w = expb / Z                                        # weights, sum=1

    # split weights
    w_pos = w[n - 1 :]          # k = 0 .. n-1
    w_neg = w[: n - 1]          # k = -(n-1) .. -1

    # contribution from k >= 0  (h_{j+k})
    conv_pos = np.convolve(w_pos, h[::-1], mode="full")
    pos_contrib = conv_pos[n - 1 :: -1][:n]

    # contribution from k < 0   (h_{j+k}=h_{j-k})
    conv_neg = np.convolve(w_neg, h, mode="full")
    neg_contrib = np.empty(n, dtype=h.dtype)
    neg_contrib[0] = 0.0
    neg_contrib[1:] = conv_neg[: n - 1]

    w0 = w[n - 1]
    grad = 1.0 - (pos_contrib + neg_contrib + w0 * h)

    obj = (maxc + np.log(Z)) / beta
    return obj, grad


def pgd_optimize(h0, beta, max_iters, step0, time_limit, start_time):
    h = h0.copy()
    target = h.size / 2.0
    step = step0
    for _ in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        obj, grad = softmax_obj_and_grad(h, beta)
        h_new = project_box_sum(h - step * grad, target)
        obj_new, _ = softmax_obj_and_grad(h_new, beta)
        if obj_new < obj:
            h = h_new
        else:
            # backtrack a few times
            improved = False
            s = step
            for _ in range(5):
                s *= 0.5
                h_try = project_box_sum(h - s * grad, target)
                obj_try, _ = softmax_obj_and_grad(h_try, beta)
                if obj_try < obj:
                    h = h_try
                    improved = True
                    break
            if not improved:
                break
        step *= 0.99
    return h


def slsqp_optimize(h0, beta, max_iter=400):
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, target)

        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=300)

            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()
        except Exception:
            pass

    # up‑sample further if time permits
    elapsed = time.time() - start
    if budget_s - elapsed > budget_s * 0.20:

        if n2 != n:

            h2 = np.interp(new_x, old_x, best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
