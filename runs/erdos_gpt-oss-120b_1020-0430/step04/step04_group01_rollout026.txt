def random_swap_mutation(h, rng, swaps=5):
    # Perform swaps between 1 and 0 positions
    h_new = h.copy()
    n = h.size
    ones = np.where(h_new == 1)[0]
    zeros = np.where(h_new == 0)[0]
    for _ in range(swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_new[i] = 0
        h_new[j] = 1
        # update arrays
        # replace swapped indices
        ones = np.where(h_new == 1)[0]
        zeros = np.where(h_new == 0)[0]
    return h_new

import time
import numpy as np

# --------------------------------------------------------------
# Utility functions
# --------------------------------------------------------------
def binary_from_continuous(vec):
    """Convert a continuous vector to a binary one with half the entries equal 1."""
    n = vec.size
    idx = np.argsort(-vec)  # descending
    bin_vec = np.zeros(n, dtype=np.int8)
    bin_vec[idx[: n // 2]] = 1
    return bin_vec


def random_binary_vector(n, rng):
    """Uniform random binary vector with exactly n/2 ones."""
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=np.int8)
    h[perm[: n // 2]] = 1
    return h


def random_swap_mutation(h, rng, swaps=5):
    """Perform a few random 1↔0 swaps (preserves the sum)."""
    n = h.size
    h_new = h.copy()
    for _ in range(swaps):
        # pick a 1
        i = rng.integers(n)
        while h_new[i] == 0:
            i = rng.integers(n)
        # pick a 0
        j = rng.integers(n)
        while h_new[j] == 1:
            j = rng.integers(n)
        h_new[i] = 0
        h_new[j] = 1
    return h_new


def compute_conv(h):
    """Full linear correlation of binary h with its complement (int16)."""
    return np.convolve(h, (1 - h)[::-1]).astype(np.int16)


# --------------------------------------------------------------
# Simulated annealing on binary vectors
# --------------------------------------------------------------
def binary_sa(start_h, rng, deadline):
    """
    Simulated annealing on a binary vector (sum = n/2).
    Returns the best vector seen and its max correlation.
    """
    h = start_h.astype(np.int8).copy()
    n = h.size
    L = 2 * n - 1
    conv = compute_conv(h)
    cur_max = int(conv.max())
    best_h = h.copy()
    best_max = cur_max

    # temperature parameters
    T0 = max(1.0, cur_max * 0.5)
    tau = 20000.0  # decay constant
    iteration = 0
    delta = np.empty(L, dtype=np.int16)

    while time.time() < deadline:
        iteration += 1
        # pick a random 1 and a random 0
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)

        # compute delta = conv_new - conv
        comp = 1 - h               # complement (int8 view)
        h_rev = h[::-1]            # reversed view
        delta.fill(0)
        delta[n - 1 - i : 2 * n - 1 - i] -= comp
        delta[n - 1 - j : 2 * n - 1 - j] += comp
        delta[i : i + n] += h_rev
        delta[j : j + n] -= h_rev

        new_conv = conv + delta
        new_max = int(new_conv.max())

        # acceptance
        if new_max <= cur_max:
            accept = True
        else:
            T = T0 * np.exp(-iteration / tau)
            prob = np.exp(-(new_max - cur_max) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            h[i] = 0
            h[j] = 1
            conv = new_conv
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()

    return best_h, best_max


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # Determine resolution and starting binary vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h_start = binary_from_continuous(np.asarray(initial_h_values, dtype=float))
    else:
        n = 720
        h_start = random_binary_vector(n, rng)

    best_h = h_start.copy()
    best_max = int(compute_conv(best_h).max())
    target_max = int(np.floor(0.3808 * n / 2))

    # ------------------------------------------------------------------
    # Repeated SA runs with short deadlines
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 0.5:
            break
        # allocate a slice of time for this SA run (up to 25 s)
        run_time = min(25.0, remaining * 0.6)
        sub_deadline = time.time() + run_time

        # start from the current best with a tiny random perturbation
        start = random_swap_mutation(best_h, rng, swaps=8)
        h_candidate, max_candidate = binary_sa(start, rng, sub_deadline)

        if max_candidate < best_max:
            best_h, best_max = h_candidate.copy(), max_candidate
            if best_max <= target_max:
                break

        # occasional full random restart
        if rng.random() < 0.12:
            start_rand = random_binary_vector(n, rng)
            h_candidate, max_candidate = binary_sa(start_rand, rng, sub_deadline)
            if max_candidate < best_max:
                best_h, best_max = h_candidate.copy(), max_candidate
                if best_max <= target_max:
                    break

    # ------------------------------------------------------------------
    # Final result
    # ------------------------------------------------------------------
    final_c5 = best_max * (2.0 / n)
    return best_h.astype(float).tolist(), float(final_c5), int(n)
