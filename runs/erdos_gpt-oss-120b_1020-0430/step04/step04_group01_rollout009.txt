import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# Global parameters for constraints (set inside the optimizer)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

# ----------------------------------------------------------------------
# Basic feasibility utilities
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with exact sum = n/2."""
    # start from a binary half‑filled vector
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    # add small noise and re‑project
    noise = rng.uniform(-0.2, 0.2, n)
    h = np.clip(h + noise, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ----------------------------------------------------------------------
# Correlation utilities (FFT based)
# ----------------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ----------------------------------------------------------------------
# Linear convolution (FFT) used for gradient computation
# ----------------------------------------------------------------------
def conv_linear(a, b):
    """Linear convolution a * b (full) via FFT."""
    n = a.shape[0] + b.shape[0] - 1
    L = 1 << (n.bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[:n]
    return conv


# ----------------------------------------------------------------------
# Optimisation helpers for smooth surrogate of max_k
# ----------------------------------------------------------------------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i‑k})."""
    corr = correlate_full(x)
    return logsumexp(CURRENT_BETA * corr) / CURRENT_BETA


def softmax_obj_grad(x):
    """Gradient of the above surrogate."""
    n = x.size
    corr = correlate_full(x)
    max_corr = corr.max()
    # stable weights
    exps = np.exp(CURRENT_BETA * (corr - max_corr))
    w = exps / exps.sum()
    # w has length 2n‑1, index shift = idx‑(n‑1)
    conv_w_h = conv_linear(w, x)[n - 1 : n - 1 + n]
    conv_w_rev_h = conv_linear(w[::-1], x)[n - 1 : n - 1 + n]
    grad = 1.0 - conv_w_h - conv_w_rev_h
    return grad


def slsqp_optimize(start, beta, rng, max_iter=250):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = start.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(

        jac=softmax_obj_grad,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # re‑project to exact sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:

    return x

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)

        if (it + 1) % decay_interval == 0:

    return best_h, best_c

def binary_sa(start_bin, rng, deadline):
    """Simulated annealing on a binary vector (sum = n/2)."""
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    # temperature schedule
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 30000

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # ensure i is 1, j is 0
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # 1. Choose resolution and initial candidate
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = init.size
        # target resolution: power of two not smaller than n0, capped at 1152
        target_n = 1024
        if n0 != target_n:
            x_old = (np.arange(n0) + 0.5) * (2.0 / n0)
            x_new = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h0 = np.interp(x_new, x_old, init)
        else:
            h0 = init.copy()
    else:
        target_n = 1024
        h0 = random_feasible_vector(target_n, rng)

    h0 = enforce_sum(h0, target_n / 2.0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Main optimisation loop (continuous SLSQP + hill‑climb)
    # ------------------------------------------------------------------
    beta_list = [2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 8000.0]
    phase_end = start_time + 0.78 * budget_s  # reserve some time for polishing

    while time.time() < phase_end:
        # start point: perturb current best or random fresh
        if rng.random() < 0.65:
            start = perturb_h(best_h, rng, scale=0.35)
        else:
            start = random_feasible_vector(target_n, rng)

        cur = start
        for beta in beta_list:
            if time.time() >= phase_end:
                break
            cur = slsqp_optimize(cur, beta, rng, max_iter=250)
            c = compute_c5(cur)
            if c < best_c:
                best_c, best_h = c, cur.copy()

        # brief hill‑climb refinement
        if time.time() < phase_end:
            hc_h, hc_c = hill_climb(best_h, rng, max_iters=2500, start_step=0.3)
            if hc_c < best_c:
                best_c, best_h = hc_c, hc_h.copy()

    # ------------------------------------------------------------------
    # 3. Binary simulated annealing (discrete exploration)
    # ------------------------------------------------------------------
    bin_deadline = start_time + 0.94 * budget_s
    # initialise binary vector from top half of current best
    idx = np.argsort(-best_h)[: target_n // 2]
    bin_start = np.zeros(target_n)
    bin_start[idx] = 1.0
    bin_h, bin_c = binary_sa(bin_start, rng, bin_deadline)
    if bin_c < best_c:
        best_c, best_h = bin_c, bin_h.copy()

    # ------------------------------------------------------------------
    # 4. Final high‑β polishing and a short hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline - 2.0
    for beta in [2000.0, 8000.0, 20000.0]:
        if time.time() >= final_deadline:
            break
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=400)
        best_c = compute_c5(best_h)

    if time.time() < deadline:
        final_h, final_c = hill_climb(best_h, rng, max_iters=3500, start_step=0.12)
        if final_c < best_c:
            best_h, best_c = final_h, final_c

    # ------------------------------------------------------------------
    # 5. Final constraint enforcement and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
