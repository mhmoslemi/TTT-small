def full_corr(a,b):
    n = a.size
    m = b.size
    size = 1 << ((n+m-1).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    # correlation a with b = conv(a, b_rev)
    fb_rev = np.conj(fb)  # But this is not reversed. Actually we need np.conj(np.fft.rfft(b[::-1], size)).
    # To avoid calculating reversed each time, we can compute via conv and then reverse indexing, but easier compute np.fft.rfft(b[::-1], size).
    fb_rev = np.fft.rfft(b[::-1], size)
    fft_corr = fa * fb_rev
    corr = np.fft.irfft(fft_corr, size)[:n+m-1]
    return corr

def pgd_optimize(h0, beta, max_iter=200, init_step=0.5):
    h = h0.copy()
    target_sum = h.size/2
    step = init_step
    # compute current objective
    A = full_corr(h,h)
    maxex = (-beta * np.min(A))
    # compute f:
    f = (np.log(np.sum(np.exp(-beta*A - maxex))) + maxex) / beta
    for it in range(max_iter):
        # compute p normalized
        # using stable log-sum-exp
        minA = np.min(A)
        # shift for exponent
        shift = -beta * minA
        exp_term = np.exp(-beta * A - shift)
        sum_exp = np.sum(exp_term)
        p = exp_term / sum_exp
        # gradient
        conv = np.convolve(p, h, mode='full')
        grad = -2.0 * conv[h.size-1:2*h.size-1]
        # line search
        for _ in range(10):
            new_h = project_box_sum(h - step * grad, target_sum)
            A_new = full_corr(new_h, new_h)
            # compute f_new
            minA_new = np.min(A_new)
            shift_new = -beta * minA_new
            f_new = (np.log(np.sum(np.exp(-beta * A_new - shift_new))) + shift_new) / beta
            if f_new < f - 1e-12:
                h = new_h
                A = A_new
                f = f_new
                step *= 1.1
                break
            else:
                step *= 0.5
        else:
            # no improvement, break
            break
    return h, f, A
```

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

import time, numpy as np
from scipy.optimize import minimize

# ---------- projection onto [0,1] with exact sum ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) * 0.5, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- FFT based full correlation ----------
def full_corr(a, b):
    n, m = a.size, b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    corr = np.fft.irfft(fa * np.conj(fb), size)[: n + m - 1]
    return corr


# ---------- C5 evaluation ----------
def compute_c5(h):
    n = h.size
    A = full_corr(h, h)
    minA = np.min(A)
    if minA < 0:
        minA = 0.0
    return 1.0 - (2.0 / n) * minA


# ---------- projected gradient descent on smoothed min autocorrelation ----------
def pgd_optimize(h0, beta, max_iter=200, init_step=0.5):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    step = init_step
    c5 = compute_c5(h)
    for _ in range(max_iter):
        A = full_corr(h, h)
        minA = np.min(A)
        if minA < 0:
            minA = 0.0
        shift = -beta * minA
        exp_term = np.exp(-beta * A - shift)          # = exp(-beta*(A-minA))
        p = exp_term / np.sum(exp_term)               # distribution over shifts
        conv = np.convolve(p, h, mode='full')
        grad = -2.0 * conv[n - 1 : 2 * n - 1]          # gradient of smooth objective
        improved = False
        for _ in range(10):
            new_h = project_box_sum(h - step * grad, target)
            new_c5 = compute_c5(new_h)
            if new_c5 < c5 - 1e-12:
                h, c5 = new_h, new_c5
                step *= 1.1
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return h, c5


# ---------- SLSQP refinement (soft‑max of max overlap) ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf

    # try several resolutions
    for n in (360, 720, 1080, 1440):
        if time.time() - start > budget_s * 0.97:
            break
        target_sum = n / 2.0

        # ---- generate a small pool of starters ----
        candidates = []
        if "initial_h_values" in globals():

            else:
                old_x = (np.arange(init.size) + 0.5) * (2.0 / init.size)
                new_x = (np.arange(n) + 0.5) * (2.0 / n)
                cand = np.interp(new_x, old_x, init)
                candidates.append(project_box_sum(cand, target_sum))

        xs = (np.arange(n) + 0.5) * (2.0 / n)
        for f in (1, 2, 3):

            cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)
            candidates.append(project_box_sum(cand, target_sum))
        for _ in range(5):
            candidates.append(random_feasible_vector(n, rng))

        for h in candidates:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()

        # ---- gradient‑based optimisation with several restarts ----
        for r in range(8):
            if time.time() - start > budget_s * 0.97:
                break
            # start point (perturbed best or fresh)

                h0 = perturb_h(best_h, rng, scale=0.3)
            else:
                h0 = random_feasible_vector(n, rng)

            for beta in (0.5, 2, 8, 32, 128, 512, 2048):

                h_opt, c_opt = pgd_optimize(h0, beta, max_iter=150, init_step=0.5)
                if c_opt < best_c:

    # ---- final SLSQP refinement with high β ----
    if best_h is not None:
        for beta in (2000, 5000, 10000):
            if time.time() - start > budget_s * 0.99:

    # ---- clean up and return ----
    if best_h is None:
        n = 360

    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
