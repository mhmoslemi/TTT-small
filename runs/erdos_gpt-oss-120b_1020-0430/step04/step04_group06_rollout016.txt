# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=4000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:      # final correction
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- correlation ----------
def compute_corr_fft(h):
    n = h.size
    size = 2 * n
    f1 = np.fft.rfft(h, size)
    f2 = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(f1 * f2, size)
    return corr[: 2 * n - 1]

def compute_c5_fft(h):
    n = h.size
    return float(np.max(compute_corr_fft(h)) * (2.0 / n))

# ---------- smooth surrogate ----------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5_fft(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            s = -shift
            i_range = np.arange(s, n)
            j_range = i_range - s
        prod = h[i_range] * (1.0 - h[j_range])
        if prod.size == 0:
            factor = init_factor
            continue
        idx = int(np.argmax(prod))
        i, j = int(i_range[idx]), int(j_range[idx])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    return best_h, best_c

# ---------- sub‑gradient descent ----------
def subgradient_descent(h0, rng, max_time=10.0, init_step=0.5, decay=0.99):
    start = time.time()
    h = h0.copy()
    n = h.size
    target = n / 2.0
    step = init_step
    best_h, best_c = h.copy(), compute_c5_fft(h)
    while time.time() - start < max_time:
        corr = compute_corr_fft(h)
        s = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros_like(h)
        if s >= 0:
            grad[: n - s] = 1.0 - h[s:]
            grad[s:] += -h[: n - s]
        else:
            s = -s
            grad[s:] = 1.0 - h[: n - s]
            grad[: n - s] += -h[s:]
        h -= step * grad
        h = np.clip(h, 0.0, 1.0)
        # enforce exact sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size:
                add = diff / free.size
                h[free] += add
                h = np.clip(h, 0.0, 1.0)
                if abs(target - h.sum()) > 1e-10:
                    h = enforce_sum(h, target, rng)
            else:
                h = enforce_sum(h, target, rng)
        c = compute_c5_fft(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        step *= decay
        if best_c <= 0.3808:
            break
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 720

    # ----- initial vector (upsampled if possible) -----
    if 'initial_h_values' in globals():
        h360 = np.array(initial_h_values, dtype=float)
        h360 = enforce_sum(h360, h360.size / 2.0, rng)
        xs360 = np.linspace(0.0, 2.0, h360.size, endpoint=False)
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(xs, xs360, h360)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        h0 = random_feasible_vector(n, rng)

    # ----- a few deterministic starts -----
    best_h, best_c = h0.copy(), compute_c5_fft(h0)
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    for f in (1, 2, 3):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, n / 2.0, rng)
            c = compute_c5_fft(h)
            if c < best_c:
                best_h, best_c = h.copy(), c
    # ----- main optimisation loop -----
    main_limit = budget_s * 0.60
    while time.time() - start < main_limit:
        # choose start point
        if rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.35)
        else:
            start_vec = random_feasible_vector(n, rng)

        # sub‑gradient descent
        sub_time = min(8.0, main_limit - (time.time() - start))
        if sub_time > 2.0:
            h_sg, c_sg = subgradient_descent(start_vec, rng, max_time=sub_time)
            if c_sg < best_c:
                best_h, best_c = h_sg.copy(), c_sg

        # hill‑climb
        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=1200,
                                      init_step=0.35, n_trials=5)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc

        # SLSQP ladder
        for beta in (10, 30, 100, 300, 1000, 3000, 6000):
            if time.time() - start >= main_limit:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=300)
                c_ref = compute_c5_fft(h_ref)
                if c_ref < best_c:
                    best_h, best_c = h_ref.copy(), c_ref
            except Exception:
                continue

    # ----- final refinement -----
    remaining = budget_s - (time.time() - start)
    if remaining > 10.0:
        h_pt, c_pt = pair_transfer_optimize(best_h, rng,
                                            max_iters=15000,
                                            time_limit=remaining * 0.7,
                                            init_factor=0.5)
        if c_pt < best_c:
            best_h, best_c = h_pt.copy(), c_pt
        for beta in (8000, 12000):
            if time.time() - start >= budget_s:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=300)
                c_ref = compute_c5_fft(h_ref)
                if c_ref < best_c:
                    best_h, best_c = h_ref.copy(), c_ref
            except Exception:
                continue

    # ----- clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(n)
