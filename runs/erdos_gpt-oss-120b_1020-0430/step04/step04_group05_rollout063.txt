```
def softmax_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    w = np.exp(beta * (corr - max_corr))
    w_sum = np.sum(w)
    w = w / w_sum
    # gradient using convolution
    # w_pos = w[n-1:]   # len n
    w_pos = w[n-1:]  # shape (n,)
    w_neg = w[:n][::-1]  # shape (n,)
    # term1: w_pos reversed convolve (1 - h)
    term1 = np.convolve(w_pos[::-1], 1 - h, mode='full')
    # term1_slice of length n: indices from n-1 to n-1 + n -1 => n-1:2n-2
    term1_slice = term1[n-1: n-1 + n]
    # term2: w_neg convolve h
    term2 = np.convolve(w_neg, h, mode='full')
    term2_slice = term2[:n]
    grad = term1_slice - term2_slice
    # obj = (max_corr + np.log(np.sum(np.exp(beta * (corr - max_corr))))) / beta
    # can also compute as (np.log(w_sum) + max_corr) / beta
    obj = (max_corr + np.log(w_sum)) / beta
    return obj, grad
```

def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Generate a random feasible vector (sum=n/2, values in [0,1])."""
    h = np.full(n, 0.5)

        j = rng.integers(0, n)
