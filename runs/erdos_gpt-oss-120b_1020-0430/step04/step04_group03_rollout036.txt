def greedy_optimize_improved(h, max_iters=2000):
    n = h.size
    dx = 2.0 / n
    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        b = corr
        idx = np.argmax(b)
        cur_c = b[idx] * dx
        s = idx - (n - 1)

        # compute neighbor sum for shift s
        # using slicing for speed
        if s >= 0:
            plus = np.concatenate((h[s:], np.zeros(s)))
            minus = np.concatenate((np.zeros(s), h[:n - s]))
        else:
            s_abs = -s
            plus = np.concatenate((np.zeros(s_abs), h[:n - s_abs]))
            minus = np.concatenate((h[s_abs:], np.zeros(s_abs)))
        neighbor_sum = plus + minus

        # choose i and j
        # i: minimal neighbor_sum among h>0
        # j: maximal neighbor_sum among h<1
        pos_mask = h > 1e-12
        neg_mask = h < 1 - 1e-12
        if not np.any(pos_mask) or not np.any(neg_mask):
            break
        i_candidates = np.where(pos_mask)[0]
        j_candidates = np.where(neg_mask)[0]
        # Find i with minimal neighbor_sum among pos candidates
        i = i_candidates[np.argmin(neighbor_sum[i_candidates])]
        j = j_candidates[np.argmax(neighbor_sum[j_candidates])]

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            break

        # compute a_s
        s_arr = np.arange(-n+1, n)
        i_plus = i + s_arr
        i_minus = i - s_arr
        j_plus = j + s_arr
        j_minus = j - s_arr

        # compute contributions safely
        i_plus_vals = np.where((i_plus >= 0) & (i_plus < n), h[i_plus], 0.0)
        i_minus_vals = np.where((i_minus >= 0) & (i_minus < n), h[i_minus], 0.0)
        j_plus_vals = np.where((j_plus >= 0) & (j_plus < n), h[j_plus], 0.0)
        j_minus_vals = np.where((j_minus >= 0) & (j_minus < n), h[j_minus], 0.0)

        a = i_plus_vals + i_minus_vals - j_plus_vals - j_minus_vals

        # ternary search for optimal epsilon
        lo = 0.0
        hi = max_eps
        # initial cur_c = b[idx]*dx
        # If max_eps is small, maybe skip search? Just compute directly.
        # We'll do ~25 iterations.
        for __ in range(25):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(b + a * m1)
            f2 = np.max(b + a * m2)
            if f1 > f2:
                lo = m1
            else:
                hi = m2

        eps_opt = (lo + hi) / 2.0
        c_opt = np.max(b + a * eps_opt) * dx

        if c_opt + 1e-12 < cur_c:
            # apply move
            h[i] -= eps_opt
            h[j] += eps_opt
            continue
        else:
            # no improvement: break
            break
    return h

def compute_corr_fft(h):
    n = h.size
    # length for convolution = 2*n - 1; pad to next power of 2 for speed.
    n_fft = 1 << ((2*n - 1).bit_length())
    # compute FFT of h and rev(1-h)
    H = np.fft.rfft(h, n=n_fft)
    rev = (1 - h)[::-1]
    R = np.fft.rfft(rev, n=n_fft)
    conv = np.fft.irfft(H * R, n=n_fft)
    # extract full length part:
    return conv[:2*n - 1]

def greedy_optimize_improved(h, max_iters=2000):
    n = h.size
    dx = 2.0 / n
    s_arr = np.arange(-n+1, n)
    b = compute_corr_fft(h)
    for _ in range(max_iters):
        idx = np.argmax(b)
        cur_c = b[idx] * dx
        s = idx - (n - 1)
        # neighbor sum patterns
        if s >= 0:
            plus = np.empty(n)
            plus[:n-s] = h[s:]
            plus[n-s:] = 0.0
            minus = np.empty(n)
            minus[:s] = 0.0
            minus[s:] = h[:n-s]
        else:
            s_abs = -s
            plus = np.empty(n)
            plus[:s_abs] = 0.0
            plus[s_abs:] = h[:n - s_abs]
            minus = np.empty(n)
            minus[:n - s_abs] = h[s_abs:]
            minus[n - s_abs:] = 0.0
        neighbor_sum = plus + minus

        # find candidate i and j
        pos_mask = (h > 1e-12)
        neg_mask = (h < 1 - 1e-12)
        if not np.any(pos_mask) or not np.any(neg_mask):
            break
        # choose i minimal neighbor sum among pos
        i_candidates = np.where(pos_mask)[0]
        i = i_candidates[np.argmin(neighbor_sum[i_candidates])]
        # choose j maximal neighbor sum among neg
        j_candidates = np.where(neg_mask)[0]
        j = j_candidates[np.argmax(neighbor_sum[j_candidates])]

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            break

        i_plus_vals = np.where((i + s_arr >= 0) & (i + s_arr < n), h[i + s_arr], 0.0)
        i_minus_vals = np.where((i - s_arr >= 0) & (i - s_arr < n), h[i - s_arr], 0.0)
        j_plus_vals = np.where((j + s_arr >= 0) & (j + s_arr < n), h[j + s_arr], 0.0)
        j_minus_vals = np.where((j - s_arr >= 0) & (j - s_arr < n), h[j - s_arr], 0.0)

        a = i_plus_vals + i_minus_vals - j_plus_vals - j_minus_vals

        # solve for eps
        lo = 0.0
        hi = max_eps
        # we could also check if a all zero -> no effect; break.
        if np.all(np.abs(a) < 1e-15):
            break
        for __ in range(25):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(b + a * m1)
            f2 = np.max(b + a * m2)
            if f1 > f2:
                lo = m1
            else:
                hi = m2
        eps_opt = (lo + hi) / 2.0
        new_c = np.max(b + a * eps_opt) * dx
        if new_c + 1e-12 < cur_c:
            # apply move
            h[i] -= eps_opt
            h[j] += eps_opt
            b = b + a * eps_opt
        else:
            # cannot improve
            break
    return h

import time, numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 5000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- correlation ----------
def compute_corr_fft(h):
    """Full correlation corr = h * (1-h) reversed using FFT."""
    n = h.size
    n_fft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    R = np.fft.rfft((1 - h)[::-1], n=n_fft)
    conv = np.fft.irfft(H * R, n=n_fft)
    return conv[: 2 * n - 1].real

def compute_c5(h):
    """Discrete C5 = max correlation * (2/n)."""
    n = h.size
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)

# ---------- SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp stabilized
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- Improved greedy ----------
def greedy_optimize_improved(h, max_iters=2000):
    """Minimax improvement via pairwise epsilon with exact line search."""
    n = h.size
    dx = 2.0 / n
    s_arr = np.arange(-n + 1, n)  # shifts for correlation indices
    b = compute_corr_fft(h)       # current full correlation
    for _ in range(max_iters):
        idx = np.argmax(b)
        cur_c = b[idx] * dx
        s = idx - (n - 1)                     # shift attaining max

        # neighbor sum for this shift
        if s >= 0:
            plus = np.empty(n)
            plus[: n - s] = h[s:]
            plus[n - s :] = 0.0
            minus = np.empty(n)
            minus[:s] = 0.0
            minus[s:] = h[: n - s]
        else:
            s_abs = -s
            plus = np.empty(n)
            plus[:s_abs] = 0.0
            plus[s_abs:] = h[: n - s_abs]
            minus = np.empty(n)
            minus[: n - s_abs] = h[s_abs:]
            minus[n - s_abs :] = 0.0
        neigh = plus + minus

        pos_mask = h > 1e-12
        neg_mask = h < 1.0 - 1e-12
        if not np.any(pos_mask) or not np.any(neg_mask):
            break

        i_candidates = np.where(pos_mask)[0]
        i = i_candidates[np.argmin(neigh[i_candidates])]
        j_candidates = np.where(neg_mask)[0]
        j = j_candidates[np.argmax(neigh[j_candidates])]

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            break

        # linear coefficients for each correlation entry
        i_plus_vals = np.where((i + s_arr >= 0) & (i + s_arr < n), h[i + s_arr], 0.0)
        i_minus_vals = np.where((i - s_arr >= 0) & (i - s_arr < n), h[i - s_arr], 0.0)
        j_plus_vals = np.where((j + s_arr >= 0) & (j + s_arr < n), h[j + s_arr], 0.0)
        j_minus_vals = np.where((j - s_arr >= 0) & (j - s_arr < n), h[j - s_arr], 0.0)
        a = i_plus_vals + i_minus_vals - j_plus_vals - j_minus_vals

        if np.all(np.abs(a) < 1e-15):
            break

        # ternary search on convex piecewise-linear function
        lo, hi = 0.0, max_eps
        for __ in range(25):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(b + a * m1)
            f2 = np.max(b + a * m2)
            if f1 > f2:
                lo = m1
            else:
                hi = m2
        eps_opt = (lo + hi) / 2.0
        new_c = np.max(b + a * eps_opt) * dx

        if new_c + 1e-12 < cur_c:
            h[i] -= eps_opt
            h[j] += eps_opt
            b = b + a * eps_opt
            continue
        else:
            break
    return h

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial vector
    if "initial_h_values" in globals():

    # multi‑restart stochastic search
    while time.time() - start_time < t_greedy:

        if rng.random() < 0.5 and best_h is not None:

        # greedy refinement
        local = greedy_optimize_improved(start, max_iters=1200)
        c_local = compute_c5(local)
        if c_local < best_c - 1e-12:
            best_h, best_c = local, c_local

        # occasional deeper greedy
        if rng.random() < 0.15:
            deep = greedy_optimize_improved(local, max_iters=2500)
            c_deep = compute_c5(deep)
            if c_deep < best_c - 1e-12:
                best_h, best_c = deep, c_deep

        # occasional SLSQP polishing

            for beta in (200, 800, 2000, 5000):
                if time.time() - start_time > t_slsqp:

    # final clean‑up

    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
