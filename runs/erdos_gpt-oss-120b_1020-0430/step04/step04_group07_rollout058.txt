def smooth_max_grad(h, beta):
    sum_h = h.sum()
    corr_self = np.correlate(h, h, mode='full')
    # O = sum_h - corr_self
    O = sum_h - corr_self
    # compute softmax weights
    # to avoid overflow, subtract max
    max_O = O.max()
    expbeta = np.exp(beta * (O - max_O))
    w = expbeta / expbeta.sum()
    # compute forward and backward sums
    Lw = O.size
    offset = Lw - 1
    forward = np.convolve(w[::-1], h, mode='full')[offset: offset + h.size]
    backward = np.convolve(w, h, mode='full')[offset: offset + h.size]
    grad = 1.0 - (forward + backward)
    # Note: for k=0, forward+backward includes 2*w_0*h_j
    # Return softmax value and gradient
    smooth = max_O + np.log(expbeta.sum()) / beta
    return smooth, grad

    # compute diff

        if diff > 0:
            # can increase up to 1

def compute_c5(h):
    corr = np.correlate(h, 1-h, mode='full')
    return npmax(corr) * (2.0 / h.size)

def softmax_c5_and_grad(h, beta):
    n = h.size
    sum_h = h.sum()
    corr_self = np.correlate(h, h, mode='full')
    O = sum_h - corr_self  # length 2n-1
    # softmax
    maxO = O.max()
    exp_beta = np.exp(beta * (O - maxO))
    Z = exp_beta.sum()
    w = exp_beta / Z
    # gradient
    L = O.size
    offset = L - 1
    forward = np.convolve(w[::-1], h, mode='full')[offset: offset+n]
    backward = np.convolve(w, h, mode='full')[offset: offset+n]
    grad = 1.0 - (forward + backward)
    smooth = maxO + np.log(Z) / beta
    return smooth, grad

def project_feasible(h, target, rng):
    # clip
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # simple iterative redistribution
    # To avoid infinite loops if diff tiny, break early.
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h

    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)

    else:

        grad[k:] -= h[: n - k]
    return grad

    j = int(np.argmin(grad))

        return h

    nh = h.copy()

    nh[j] += eps
    return nh

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

    n = h.size
    n_up = n * factor
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = project_feasible(h_up, n_up / 2.0, rng)

    h_up, _ = grad_descent_smooth(h_up, rng, time_limit, start)
    h_up, c_up = hill_climb_worst(h_up, rng, time_limit, start)
    return h_up, c_up


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    time_limit = budget_s

    best_h = None
    best_c = np.inf
    best_n = None

    # ----- parent solution -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        n0 = h0.size
        h0 = project_feasible(h0, n0 / 2.0, rng)
        c0 = compute_c5(h0)
        best_h, best_c, best_n = h0.copy(), c0, n0

    # ----- quick binary search ------------------------------------------------
    # try many random half‑ones vectors (size 256) before the main loop
    n_bin = 256
    target_bin = n_bin // 2
    for _ in range(200):
        if time.time() - start > 0.12 * time_limit:
            break
        h_bin = np.zeros(n_bin)
        h_bin[rng.choice(n_bin, target_bin, replace=False)] = 1.0
        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin.copy(), c_bin, n_bin

    # ----- main multi‑restart exploration ------------------------------------
    sizes = [128, 160, 192, 224, 256, 288, 320]
    while time.time() - start < 0.78 * time_limit:
        n = int(rng.choice(sizes))
        tgt = n / 2.0
        # random start: binary or fractional
        if rng.random() < 0.5:
            h = np.zeros(n)
            h[rng.choice(n, n // 2, replace=False)] = 1.0
        else:
            h = rng.random(n)
            h *= tgt / h.sum()
            h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, tgt, rng)

        h, c = grad_descent_smooth(h, rng, time_limit, start)
        h, c = hill_climb_worst(h, rng, time_limit, start)

        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ----- up‑sampling refinement -------------------------------------------
    if best_h is not None and time.time() - start < 0.95 * time_limit:
        for factor in (2, 3):
            if best_n * factor > 384:
                continue
            h_up, c_up = upsample_and_refine(best_h, factor, rng, time_limit, start)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, best_n * factor

    # ----- final stochastic tweaking -----------------------------------------
    while time.time() - start < time_limit * 0.999 and best_h is not None:
        g = worst_gradient(best_h)
        nh = transfer_move(best_h, g, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ----- sanity and return -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
