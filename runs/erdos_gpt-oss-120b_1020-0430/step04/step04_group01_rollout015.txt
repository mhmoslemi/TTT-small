If currently one of them is small and the other large, then moving mass from the smaller to larger may increase product term (since cross_corr term = sum_i h_i h_{i+k*}) while affecting other shifts.

Specifically, cross_corr_k* = sum_i h_i h_{i+k*}. Gradient for this shift w.r.t h_i is h_{i+k*} and w.r.t h_{i+k*} is h_i. So to increase cross_corr, we want to increase both h_i and h_{i+k*}. But sum constraint prevents increasing both without decreasing elsewhere. So candidate move: find a pair (i,j) where i is in overlap of k*, j not, and h_i is relatively low and h_{i+k*} relatively high, and move epsilon from j to i and also from some other location to i+k*. That is in theory a 2-swap.

But too complex.

Simplify: we can perform a local "hill climbing" that iteratively attempts to swap values at positions i and j if it lowers max conv, evaluating each swap thoroughly. For binary vectors, try swaps; for continuous, try moving epsilon from one to another.

But evaluating each possible swap is O(n^2). Too many.

Thus we rely on random sampling.

import time, math
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Feasibility utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# --------------------------------------------------------------
# Correlation utilities
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with reversed h (FFT based)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = h[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# --------------------------------------------------------------
# Smooth surrogate for max (softmax) used by SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP on softmax surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# --------------------------------------------------------------
# Simple binary simulated annealing (swap-based)
# --------------------------------------------------------------
def binary_sa(start_bin, rng, deadline):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = best_c * 0.5 + 1e-6
    Tf = 1e-6
    it = 0
    while time.time() < deadline:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        elapsed = time.time() - deadline + (deadline - start_time)  # rough
        frac = it / 20000.0
        T = T0 * ((Tf / T0) ** frac)
        if c < best_c or rng.random() < np.exp(-(c - best_c) / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

# --------------------------------------------------------------
# Greedy hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s

    # ---- base resolution and initial candidate ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # a modest pool of structured starts
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    pool = [best_h]

    # triangular
    tri = np.clip(1.0 - np.abs(xs - 1.0), 0.0, 1.0)
    pool.append(enforce_sum(tri, n / 2.0, rng))

    # sin / cos families
    for f in range(1, 5):
        s = np.clip(0.5 + 0.5 * np.sin(np.pi * f * xs), 0.0, 1.0)
        c = np.clip(0.5 + 0.5 * np.cos(np.pi * f * xs), 0.0, 1.0)
        pool.append(enforce_sum(s, n / 2.0, rng))
        pool.append(enforce_sum(c, n / 2.0, rng))

    # binary block and alternating
    block = np.zeros(n)
    block[: n // 2] = 1.0
    pool.append(block.copy())

    alt = np.tile([0.0, 1.0], n // 2)
    if alt.size < n:
        alt = np.append(alt, 0.0)
    pool.append(alt.copy())

    for _ in range(4):
        pool.append(random_feasible_vector(n, rng))

    for h in pool:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- Phase 1: quick SLSQP ladders ----
    beta_list = [20, 80, 200, 800, 2000]
    phase1_end = start_time + 0.45 * budget
    while time.time() < phase1_end:
        start = perturb_h(best_h, rng, scale=0.3) if rng.random() < 0.7 else random_feasible_vector(n, rng)
        for beta in beta_list:
            if time.time() >= phase1_end:
                break
            try:
                h_opt = slsqp_optimize(start, beta, max_iter=500)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_c, best_h = c_opt, h_opt.copy()
                start = h_opt
            except Exception:
                continue
        if time.time() < phase1_end:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=2000, start_step=0.25)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

    # ---- Phase 2: long continuous simulated annealing ----
    phase2_end = start_time + 0.85 * budget
    cur_h = best_h.copy()
    cur_c = best_c
    best_h2, best_c2 = cur_h.copy(), cur_c
    T0 = cur_c * 0.5 + 1e-6
    Tf = 1e-6
    step_scale = 0.2
    it = 0
    sa_start = time.time()
    while time.time() < phase2_end:
        it += 1
        if it % 5000 == 0:
            step_scale = max(step_scale * 0.8, 1e-4)

        i = rng.integers(0, n)
        if cur_h[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if cur_h[j] >= 1.0:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue

        # occasional full swap
        if rng.random() < 0.1:
            eps = max_eps
        else:
            eps = (rng.random() - 0.5) * 2.0 * max_eps * step_scale

        if eps < 0:
            i, j = j, i
            eps = -eps

        cur_h[i] -= eps
        cur_h[j] += eps
        new_c = compute_c5(cur_h)

        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c2:
                best_c2, best_h2 = new_c, cur_h.copy()
        else:
            elapsed = time.time() - sa_start
            frac = min(elapsed / (phase2_end - sa_start), 1.0)
            T = T0 * ((Tf / T0) ** frac)
            if rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
                cur_c = new_c
            else:
                cur_h[i] += eps
                cur_h[j] -= eps

        if it % 20000 == 0 and rng.random() < 0.3:
            cur_h = random_feasible_vector(n, rng)
            cur_c = compute_c5(cur_h)

    if best_c2 < best_c:
        best_h, best_c = best_h2.copy(), best_c2

    # ---- Optional binary SA after continuous SA ----
    if time.time() < phase2_end:
        idx = np.argsort(-best_h)[: n // 2]
        bin_start = np.zeros(n)
        bin_start[idx] = 1.0
        bin_h, bin_c = binary_sa(bin_start, rng, phase2_end)
        if bin_c < best_c:
            best_h, best_c = bin_h.copy(), bin_c

    # ---- Phase 3: final high‑β SLSQP polishing ----
    phase3_end = start_time + budget - 1.0
    final_beta_list = [2000, 5000, 10000]
    for beta in final_beta_list:
        if time.time() >= phase3_end:
            break
        try:
            h_opt = slsqp_optimize(best_h, beta, max_iter=800)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt.copy(), c_opt
        except Exception:
            continue

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
