import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)

# ---------- local moves ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h

# ---------- fast C5 ----------
def compute_c5(h):
    n = h.size
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[: 2 * n - 1]
    return np.max(corr) * (2.0 / n)

def compute_c5_and_idx(h):
    n = h.size
    L = 2 * n
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[: 2 * n - 1]
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / n), idx

# ---------- hill climbing ----------
def hill_climb_targeted(
    h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25
):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            cand = (
                targeted_move(h, rng, k_star, scale=step)
                if rng.random() < p_target
                else perturb_h(h, rng, scale=step)
            )
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        target_n = max(400, n0)
        if n0 != target_n:
            old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h0 = np.interp(new_x, old_x, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, target_n / 2.0)
    else:
        target_n = 400
        h0 = random_feasible_vector(target_n, rng)

    n = target_n

    # ---- population ----
    pop = [h0]
    scores = [compute_c5(h0)]

    for _ in range(5):
        h = random_feasible_vector(n, rng)
        pop.append(h)
        scores.append(compute_c5(h))

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for _ in range(5):
        f = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.7)
        cand = 0.5 + amp * np.sin(2 * np.pi * f * xs + phase)
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_box_sum(cand, n / 2.0)
        pop.append(cand)
        scores.append(compute_c5(cand))

    pop_size = 15
    idx = np.argsort(scores)[:pop_size]
    pop = [pop[i] for i in idx]
    scores = [scores[i] for i in idx]

    best_h = pop[0].copy()
    best_score = scores[0]

    # ---- Differential Evolution ----
    F = 0.8
    CR = 0.9
    while time.time() - start < budget_s * 0.85:
        for i in range(pop_size):
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            crossover = rng.random(n) < CR
            if not np.any(crossover):
                crossover[rng.integers(0, n)] = True
            trial = np.where(crossover, mutant, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_box_sum(trial, n / 2.0)
            trial_score = compute_c5(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        # occasional focused refinement
        if time.time() - start > budget_s * 0.5 and rng.random() < 0.2:
            refined_h, refined_score = hill_climb_targeted(
                best_h, rng, max_iters=250, init_step=0.2, n_trials=3, p_target=0.3
            )
            if refined_score < best_score:
                best_score = refined_score
                best_h = refined_h.copy()
                worst = np.argmax(scores)
                pop[worst] = best_h.copy()
                scores[worst] = best_score

    # ---- SLSQP refinement ----
    beta_list = [10, 50, 200, 800, 3000, 10000]
    for beta in beta_list:
        if time.time() - start > budget_s * 0.98:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=500)
        except Exception:
            continue
        refined_score = compute_c5(refined)
        if refined_score < best_score:
            best_score = refined_score
            best_h = refined.copy()

    # ---- final cleanup ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n / 2.0)
    final_score = compute_c5(best_h)

    return best_h.tolist(), float(final_score), int(n)
