import time, numpy as np
from scipy.optimize import minimize

# ---------- constraint handling ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target (water‑fill)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    # simple random redistribution
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to add mass
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:          # need to remove mass
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    # final tiny correction
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

# ---------- contribution vectors ----------
def _contrib_i(i, h):
    """Return vector c (length 2n‑1) with (h[i+s] + h[i‑s]) for each shift."""
    n = h.shape[0]
    m = 2 * n - 1
    offset = n - 1
    c = np.zeros(m, dtype=float)
    # positive shifts (including 0)
    pos_len = n - i
    c[offset:offset + pos_len] += h[i:i + pos_len]
    # negative shifts (including 0)
    t_max = min(i, n - 1 - i)
    c[offset - t_max:offset + 1] += h[i:i + t_max + 1][::-1]
    return c

def compute_g_vector(i, j, h):
    """Derivative of all correlations when epsilon moves from i to j."""
    return _contrib_i(i, h) - _contrib_i(j, h)

def optimal_transfer_eps(i, j, h, corr):
    """Best epsilon (0 ≤ eps ≤ eps_max) that minimises the maximal correlation."""
    eps_max = min(h[i], 1.0 - h[j])
    if eps_max <= 0.0:
        return 0.0
    g = compute_g_vector(i, j, h)

    idx_max = np.argmax(corr)
    v0 = corr[idx_max]

    tmp = corr.copy()
    tmp[idx_max] = -np.inf
    idx2 = np.argmax(tmp)
    v1 = corr[idx2]

    g0 = g[idx_max]
    g1 = g[idx2]
    denom = g1 - g0
    if denom > 1e-12:
        eps = (v0 - v1) / denom
    else:
        eps = eps_max
    if eps < 0.0:
        eps = 0.0
    if eps > eps_max:
        eps = eps_max
    return eps

# ---------- greedy optimisation with optimal line search ----------
def greedy_optimize_opt(h, max_steps=2000):
    n = h.size
    offset = n - 1
    for _ in range(max_steps):
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        cur_corr = corr[idx_max]
        s = idx_max - offset
        grad = gradient_shift(h, s)

        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break

        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]

        eps = optimal_transfer_eps(i, j, h, corr)
        if eps <= 1e-12:
            break

        g = compute_g_vector(i, j, h)
        new_max = (corr + eps * g).max()
        if new_max < cur_corr - 1e-12:
            h[i] -= eps
            h[j] += eps
        else:
            break
    return h

# ---------- smooth max + SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial vector (720 samples if provided)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), h0.size / 2.0, rng)
    else:

    # ensure we work with 720 points (more flexibility)
    if h0.size != 720:
        target_n = 720
        xs_old = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
        xs_new = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h = np.interp(xs_new, xs_old, h0)
        h = enforce_sum(np.clip(h, 0.0, 1.0), target_n / 2.0, rng)
    else:
        h = h0.copy()

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- stochastic multi‑restart search -----
    while time.time() - start_time < budget_s * 0.78:
        if rng.random() < 0.5 and best_h is not None:
            start = perturb_random(best_h, rng, scale=0.5)
        else:
            start = random_feasible_vector(best_h.size, rng)

        local = greedy_optimize_opt(start, max_steps=1500)
        c_local = compute_c5(local)
        if c_local < best_c:
            best_h, best_c = local, c_local

        # occasional SLSQP polishing with increasing β
        if rng.random() < 0.25:
            for beta in (500, 2000, 8000, 20000, 50000):
                if time.time() - start_time > budget_s * 0.95:
                    break
                try:

                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined, c_ref

    # ----- final local micro‑search -----
    while time.time() - start_time < budget_s * 0.99:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
