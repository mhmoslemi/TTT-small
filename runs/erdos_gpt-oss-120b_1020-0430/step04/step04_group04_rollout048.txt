def enforce_sum(h, target, rng, max_iter=1000):
    h = np.clip(h, 0., 1.)
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # add uniform adjustment (small) limited to boundaries.
        add = diff / h.size
        # but ensure we don't overshoot bounds
        # clamp:
        lower = -h  # min amount we can subtract from each (negative)
        upper = 1 - h  # max amount we can add to each (positive)
        # we need to adjust each element by a_i where a_i ∈ [lower_i, upper_i] , sum a_i = diff
        # simplest: compute a_i = np.clip(add, lower_i, upper_i)
        a = np.clip(add, lower, upper)
        # If sum a_i does not equal diff due to clipping, adjust remaining diff on random free indices
        cur = a.sum()
        h += a
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # If diff still far, adjust random index
        if abs(diff) > 0:
            # select random index with room
            idx = rng.choice(np.where((h > 1e-12) & (h < 1 - 1e-12))[0])
            # adjust as much as possible
            delta = min(diff, 1 - h[idx]) if diff>0 else max(diff, -h[idx])
            h[idx] += delta
            diff = target - h.sum()
    return h

def enforce_sum(h, target, rng, max_iter=1000):
    """Adjust h so sum equals target, while staying within [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # uniform adjustment limited by bounds
        add = diff / n
        lower = -h
        upper = 1.0 - h
        # clip add to bounds
        a = np.clip(add, lower, upper)
        s = a.sum()
        # if s is zero due to clipping (h at bounds), we need random adjustments
        if abs(s) < 1e-12:
            # find free indices not at boundaries
            free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
            if free.size == 0:
                # can't adjust, break
                break
            i = rng.choice(free)
            delta = min(diff, 1 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += delta
        else:
            h += a
        diff = target - h.sum()
    # final clip (to avoid tiny numeric overshoot)
    h = np.clip(h, 0.0, 1.0)
    # final tiny correction with random indices
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(10):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            delta = min(diff, 1 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += delta
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})

    diff = CURRENT_TARGET - x.sum()

import time
import numpy as np
from scipy.optimize import minimize

# ----------- Helper functions -----------------------------------------

def correlation_fft(x):
    """Full correlation of x with (1-x) using FFT."""
    n = x.size
    rev = (1.0 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    conv = correlation_fft(h) * (2.0 / n)
    return float(conv.max())

def enforce_sum(h, target, rng, max_iter=1000):
    """Adjust h so that sum(h)=target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        add = diff / n
        lower = -h
        upper = 1.0 - h
        a = np.clip(add, lower, upper)
        if np.allclose(a, 0):
            free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
            if free.size == 0:
                break
            i = rng.choice(free)
            delta = min(diff, 1 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += delta
        else:
            h += a
        diff = target - h.sum()
    # tiny residual fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(10):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            delta = min(diff, 1 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += delta
            diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

# ----- Surrogate for SLSQP -------------------------------------------

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = correlation_fft(x) * (2.0 / x.size)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(500):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            delta = min(diff, 1 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += delta
            diff = CURRENT_TARGET - x.sum()
    return x

# ----------- Main routine --------------------------------------------
