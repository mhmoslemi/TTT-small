1. define function compute_corr_and_max_shift(h):
    - corr = np.correlate(h, 1 - h, mode='full')
    - idx = np.argmax(corr)
    - max_val = float(corr[idx])
    - shift = idx - (len(h) - 1)
    - return max_val, shift, corr

2. define gradient_for_shift(h, shift):
    - n = len(h)
    - g = np.zeros(n)
    - # (1 - h[i+shift]) for i where i+shift < n
    - if shift >= 0:
        g[:n-shift] = 1.0 - h[shift:]
    else:
        g[-shift:] = 1.0 - h[:n+shift]  # shift negative
    - # -h[i-shift] for i where i-shift >= 0
    - if shift >= 0:
        g[shift:] -= h[:n-shift]
    else:
        g[:n+shift] -= h[-shift:]
    - return g

def subgradient_descent(h0, target, max_iters, time_limit):
    h = h0.copy()
    n = h.size
    start = time.time()
    best_h, best_val = h.copy(), compute_c5(h)
    for t in range(1, max_iters+1):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_corr_shift(h)  # compute max correlation and shift
        # compute c5 = max_val * dx
        c5 = max_val * (2.0 / n)
        if c5 < best_val:
            best_val, best_h = c5, h.copy()
        # gradient
        g = gradient_for_shift(h, shift)
        # step size
        alpha = 0.5 / np.sqrt(t)
        # gradient descent step
        h_new = h - alpha * g
        # projection:
        # Clip to [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)
        # adjust sum to target
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            # simple water-filling: add diff/n and reclip
            # iterate a few times for stability
            for _ in range(5):
                h_new += diff / n
                h_new = np.clip(h_new, 0.0, 1.0)
                diff = target - h_new.sum()
                if abs(diff) < 1e-12:
                    break
        h = h_new
    return best_h, best_val
```

def project_onto_constraints(h, target):
    # aim: 0<=h<=1 and sum = target
    # simple algorithm: clip, then adjust sum.
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # distribute diff equally
        h += diff / h.size
        # re-clip
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # after loop, might still have diff due to saturation; try redistribute only among non-saturated?
    if abs(diff) > 1e-12:
        # water filling: adjust only elements not at bound
        free = (h > 0.0) & (h < 1.0)
        n_free = free.sum()
        if n_free > 0:
            # distribute diff among free elements equally
            h[free] += diff / n_free
            h = np.clip(h, 0.0, 1.0)
    return h
```

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_corr_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # Determine index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        # transfer eps from i to j
        h[i] -= eps
        h[j] += eps
        # numerical safety
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    final_c = compute_c5(best_h)
    return best_h, final_c
```

def subgradient_descent(h0, target, max_iters, time_limit):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    for t in range(1, max_iters+1):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_corr_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # gradient
        g = gradient_for_shift(h, shift)
        # step size schedule
        alpha = 0.5 / np.sqrt(t)
        # update
        h_new = h - alpha * g
        # projection onto constraints
        h_new = np.clip(h_new, 0.0, 1.0)
        # adjust sum
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            # distribute diff equally a few times
            for _ in range(5):
                h_new += diff / n
                h_new = np.clip(h_new, 0.0, 1.0)
                diff = target - h_new.sum()
                if abs(diff) < 1e-12:
                    break
        h = h_new
    return best_h, best_c

import time
import numpy as np

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def gradient_for_shift(h, shift):
    n = h.size
    g = np.zeros(n)
    if shift >= 0:
        g[: n - shift] = 1.0 - h[shift:]
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: n - s] -= h[s:]
    return g

def max_corr_shift(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift, corr

def subgradient_descent(h0, target, max_iters, time_limit, rng):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    K = 3
    for t in range(1, max_iters + 1):
        if time.time() - start > time_limit:
            break
        max_val, shift, corr = max_corr_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # top‑K gradients
        top_idx = np.argpartition(-corr, K - 1)[:K]
        grad = np.zeros_like(h)
        for idx in top_idx:
            sh = idx - (n - 1)
            grad += gradient_for_shift(h, sh)
        grad /= K
        alpha = 0.8 / np.sqrt(t)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            for _ in range(5):
                h_new += diff / n
                h_new = np.clip(h_new, 0.0, 1.0)
                diff = target - h_new.sum()
                if abs(diff) < 1e-12:
                    break
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_eps = min(h_new[i], 1.0 - h_new[j])
                if max_eps > 0:
                    eps = (rng.random() - 0.5) * 2 * max_eps * 0.05
                    h_new[i] -= eps
                    h_new[j] += eps
                    h_new = np.clip(h_new, 0.0, 1.0)
                    diff = target - h_new.sum()
                    h_new += diff / n
                    h_new = np.clip(h_new, 0.0, 1.0)
        h = h_new
    return best_h, best_c

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift, _ = max_corr_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    return best_h, compute_c5(best_h)

def final_hill_climb(h0, rng, max_iters=2000, step=0.1):
    h = h0.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    for _ in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
        candidates = [h0]
    else:
        n = 360
        candidates = [random_feasible_vector(n, rng) for _ in range(3)]
    target = n / 2.0
    best_h = None
    best_c = float('inf')
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # -------- subgradient phase --------
    sub_time = budget_s * 0.6
    while time.time() - start < sub_time:
        if rng.random() < 0.7 and best_h is not None:
            h_start = perturb_h(best_h, rng, scale=0.3)
        else:
            h_start = random_feasible_vector(n, rng)
        remaining = sub_time - (time.time() - start)
        if remaining < 2.0:
            break
        max_iters = int(min(8000, max(200, remaining * 1500)))
        h_opt, c_opt = subgradient_descent(h_start, target, max_iters, remaining, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
    # -------- pair‑transfer phase --------
    remaining = budget_s - (time.time() - start)
    if remaining > 6.0:
        attempts = max(1, int(remaining // 30))
        per_attempt = remaining / (attempts + 1)
        for _ in range(attempts):
            start_vec = perturb_h(best_h, rng, scale=0.2) if rng.random() < 0.5 else best_h.copy()
            h_opt, c_opt = pair_transfer_optimize(start_vec, rng, max_iters=8000,
                                                 time_limit=per_attempt, init_factor=0.5)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
        h_opt, c_opt = pair_transfer_optimize(best_h, rng, max_iters=12000,
                                             time_limit=remaining * 0.9, init_factor=0.5)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
    # -------- final local hill climb --------
    h_opt, c_opt = final_hill_climb(best_h, rng, max_iters=2000, step=0.08)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt.copy()
    # clean up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
