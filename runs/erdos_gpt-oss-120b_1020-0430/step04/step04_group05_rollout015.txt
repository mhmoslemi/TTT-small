def simulated_annealing(h0, rng, max_iters, time_limit):
    h = h0.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    # define temperature schedule: exponential decay from T0 to Tf
    T0 = 0.5
    Tf = 1e-5
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        T = T0 * ((Tf/T0) ** (it / max_iters))
        # choose move:
        if rng.random() < 0.7:
            # swap move: pick i where h[i]==1, j where h[j]==0
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
            if len(ones)==0 or len(zeros)==0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            new_h = h.copy()
            new_h[i] = 0
            new_h[j] = 1
        else:
            # small epsilon move: fractional transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i==j: continue
            max_eps = min(h[i], 1-h[j])
            if max_eps<=0: continue
            eps = (rng.random()*2-1) * max_eps * 0.5
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        new_c = compute_c5(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        # occasional large block move:
        if rng.random() < 0.01:
            # choose block length L
            L = rng.integers(1, n//10+1)
            start = rng.integers(0, n-L+1)
            block = h[start:start+L]
            # reverse block
            new_h = h.copy()
            new_h[start:start+L] = block[::-1]
            # enforce sum? reversal preserves sum if block fractional values are reversed; okay.
            new_c = compute_c5(new_h)
            dE = new_c - cur_c
            if dE < 0 or rng.random() < math.exp(-dE / max(T, 1e-12)):
                h, cur_c = new_h, new_c
                if new_c < best_c:
                    best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

def sum_eq_constraint(x):

    corr = np.correlate(x, 1 - x, mode='full')

    m = np.max(bc)

def targeted_swap_move(h, rng):
    n = h.size
    max_corr = np.max(np.correlate(h, 1 - h, mode='full'))
    # But we need lag index
    corr = np.correlate(h, 1 - h, mode='full')
    lag = np.argmax(corr) - (n - 1)
    # compute contribution
    if lag >= 0:
        valid = np.arange(n - lag)
        contrib = h[valid] * (1 - h[valid + lag])
    else:
        lag_abs = -lag
        valid = np.arange(lag_abs, n)
        contrib = h[valid] * (1 - h[valid - lag_abs])
    if contrib.size == 0:
        return swap_move(h, rng)
    k = max(1, int(0.1 * len(valid)))
    top_idx = valid[np.argpartition(-contrib, k-1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k-1)[:k]]
    # choose i from top where h[i]==1 else fallback
    i_candidates = top_idx[h[top_idx] > 0.5]
    if i_candidates.size == 0:
        i = top_idx[rng.integers(len(top_idx))]
    else:
        i = rng.choice(i_candidates)
    # choose j from bot where h[j]==0
    j_candidates = bot_idx[h[bot_idx] < 0.5]
    if j_candidates.size == 0:
        zeros = np.where(h < 0.5)[0]
        if zeros.size == 0:
            j = (i + 1) % n
        else:
            j = rng.choice(zeros)
    else:
        j = rng.choice(j_candidates)
    if i == j:
        return None
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]

        sa_h, sa_c = simulated_annealing(start_vec, rng, max_sa_iters, t_end * 0.87)
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c

        for beta in beta_list:
            if time.time() >= t_end * 0.87:
                break
            refined = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=1500)
            refined_c = compute_c5(refined)
            if refined_c < best_c:
                best_h, best_c = refined, refined_c

    # ----- final local hill‑climb with diminishing step -----
    h_cur = best_h.copy()
    cur_c = best_c
    step = 0.2
    while time.time() < t_end:
        new_h = perturb_h(h_cur, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_c, h_cur = new_c, new_h
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        step *= 0.9995

    # ----- clean‑up and final evaluation -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
