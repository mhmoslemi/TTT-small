import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv.max() * (2.0 / n)

def correlation_full(x):
    n = x.shape[0]
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Use free variables if any
    free = (h > 1e-12) & (h < 1 - 1e-12)
    idx = np.where(free)[0]
    if idx.size > 0:
        per = diff / idx.size
        h[idx] += per
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # fallback iterative
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n/2.0, rng)

def upsample_h(h, n_new):
    n_old = h.shape[0]
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def slsqp_optimize(h0, beta, max_iter=800):
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    def obj(x):
        conv = correlation_full(x)
        maxc = conv.max()
        if beta == 0:
            return maxc
        t = conv - maxc
        # log-sum-exp safely
        beta_t = beta * t
        m = beta_t.max()
        # avoid overflow
        logsumexp = m + np.log(np.sum(np.exp(beta_t - m)))
        return maxc + logsumexp / beta
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # correct any tiny sum drift
    rng = np.random.default_rng()
    x = enforce_sum(x, target, rng)
    return x

def hill_climb(h, rng, max_iters=5000, init_step=0.3):
    n = h.shape[0]
    best = h.copy()
    best_c = compute_c5(best)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:

        max_eps = min(best[i], 1.0 - best[j])

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv.max() * (2.0 / n)

def correlation_full(x):
    n = x.shape[0]
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    free = (h > 1e-12) & (h < 1 - 1e-12)
    idx = np.where(free)[0]
    if idx.size:
        per = diff / idx.size
        h[idx] += per
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def upsample_h(h, n_new):
    n_old = h.shape[0]
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def slsqp_optimize(h0, beta, max_iter=800):
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def obj(x):
        conv = correlation_full(x)
        m = conv.max()
        if beta == 0:
            return m
        t = conv - m
        bt = beta * t
        M = bt.max()
        return m + (M + np.log(np.sum(np.exp(bt - M)))) / beta
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    return enforce_sum(x, target, rng)

def hill_climb(h, rng, max_iters=5000, init_step=0.3):
    n = h.shape[0]
    best = h.copy()
    best_c = compute_c5(best)
    step = init_step
    decay_int = max(1, max_iters // 10)
    for it in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_max = min(best[i], 1.0 - best[j])
        if eps_max <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * eps_max * step
        new = best.copy()
        new[i] -= eps
        new[j] += eps
        new = enforce_sum(new, n / 2.0, rng)
        c = compute_c5(new)
        if c < best_c:
            best, best_c = new, c
        if (it + 1) % decay_int == 0:
            step = max(step * 0.7, 1e-5)
    return best, best_c

def simulated_annealing(h, rng, max_iter=20000,
                        init_temp=1e-2, final_temp=1e-6,
                        init_scale=0.3, final_scale=0.02):
    n = h.shape[0]
    best = h.copy()
    best_c = compute_c5(best)
    cur = best.copy()
    cur_c = best_c
    for it in range(max_iter):
        t = it / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_max = min(cur[i], 1.0 - cur[j])
        if eps_max <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * eps_max * scale
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        new = enforce_sum(new, n / 2.0, rng)
        c = compute_c5(new)
        delta = c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur, cur_c = new, c
            if c < best_c:
                best, best_c = new, c
    return best, best_c

    # ---- SLSQP continuation (≈45% time) ----
