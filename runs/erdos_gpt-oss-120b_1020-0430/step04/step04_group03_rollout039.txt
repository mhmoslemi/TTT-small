def project_to_feasible(v, target, max_iter=100, tol=1e-9):
    lo = -np.max(v)  # if lambda very negative, all v - lambda >1, clipped to 1 => sum = n
    hi = np.max(v)  # if lambda very positive, all v - lambda <0, clipped 0 => sum = 0
    # Actually we need range that yields sum > target and sum < target.
    # At lambda = lo-1, sum = n (all 1). At lambda = hi+1, sum=0.
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    # Return final
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_to_feasible(v, target, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_to_feasible(v, n / 2.0)

def gradient_descent_step(h, beta, lr, target):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c = corr * dx
    max_c = np.max(c)
    # softmax weights
    w_raw = np.exp(beta * (c - max_c))
    w = w_raw / w_raw.sum()
    offset = 2 * n - 2
    conv1 = np.convolve(w[::-1], 1 - h, mode='full')
    conv2 = np.convolve(w, h, mode='full')
    grad = dx * (conv1[offset:offset + n] - conv2[offset:offset + n])
    h_new = h - lr * grad
    h_new = project_to_feasible(h_new, target)
    new_c = compute_c5(h_new)
    cur_c = max_c
    return h_new, new_c, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(720, rng)
    n = h0.size
    target = n / 2.0
    h0 = project_to_feasible(h0, target)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # parameters
    betas = [10, 50, 200, 800, 2000, 5000, 8000]
    shallow_iters = [500, 400, 300, 200, 150, 100, 50]
    deep_iters = [5000, 4000, 3000, 2000, 1500, 1000, 500]

    # Phase 1: many random restarts with shallow descent
    while time.time() - start < budget_s * 0.45:
        h = random_feasible_vector(n, rng)
        for beta, it in zip(betas, shallow_iters):
            lr = 2.0 / np.sqrt(beta)
            for _ in range(it):
                if time.time() - start >= budget_s * 0.45:
                    break
                h, new_c, cur_c = gradient_descent_step(h, beta, lr, target)
                # accept if improvement, else keep old (already done in step)
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
        # after shallow descent evaluate
        c_val = compute_c5(h)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()

    # Phase 2: deep refinement from best_h
    h = best_h.copy()
    for beta, it in zip(betas, deep_iters):

        for _ in range(it):

            h_new, new_c, cur_c = gradient_descent_step(h, beta, lr, target)
            # simple acceptance: keep if not worse

                    best_c = new_c
                    best_h = h_new.copy()

import time
import numpy as np

def compute_c5(h):
    """Discrete C5 for a vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(v, target, max_iter=60):
    """Project onto {0≤x≤1, sum(x)=target} using a binary search on a shift."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random point on the feasible set."""
    v = rng.random(n)
    return project_to_feasible(v, n / 2.0)


def gradient_descent_step(h, beta, lr, target):
    """One projected‑gradient step on the soft‑max approximation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c = corr * dx
    max_c = np.max(c)

    # soft‑max weights
    w_raw = np.exp(beta * (c - max_c))
    w = w_raw / w_raw.sum()

    offset = 2 * n - 2
    conv1 = np.convolve(w[::-1], 1 - h, mode="full")
    conv2 = np.convolve(w, h, mode="full")
    grad = dx * (conv1[offset : offset + n] - conv2[offset : offset + n])

    h_new = h - lr * grad
    h_new = project_to_feasible(h_new, target)

    new_c = compute_c5(h_new)
    return h_new, new_c, max_c

    # ---- parameters ------------------------------------------------------------
    betas = [10, 50, 200, 800, 2000, 5000, 8000]

    # ---- Phase 1: many random restarts with shallow descent --------------------
    while time.time() - start < budget_s * 0.45:
