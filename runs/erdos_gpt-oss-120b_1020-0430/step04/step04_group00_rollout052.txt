def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98

    best_h = None; best_c = np.inf; best_n = None

    candidate_ns = [360, 240, 180]

    # generate initial parent's interpolated vectors and random starts later.
    parent = None
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)

    for n in candidate_ns:
        if time.time() > deadline: break
        target = n/2.0
        # build start list
        start_vectors = []
        if parent is not None:
            # interpolate parent to n
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vectors.append(interp)  # fractional
            # binary version
            idx = np.argpartition(interp, -int(target))[-int(target):]
            bin_vec = np.zeros(n)

        # random binary

        start_vectors.append(bin_rand)
        # random fractional
        rand_frac = rng.random(n)
        # scale to target sum
        rand_frac = np.clip(rand_frac, 0, 1)
        rand_frac *= target / rand_frac.sum()
        # project to sum exactly just in case
        rand_frac = project_onto_box_and_sum(rand_frac, target)  # reuse function, maybe defined earlier
        start_vectors.append(rand_frac)

        # maybe more random starts as time permits

def project_onto_box_and_sum(v, target, rng, tol=1e-12, max_iter=60):
    """Project onto {0<=x<=1, sum(x)=target} via a water‑filling search."""
    lo = -np.max(v) - 1.0          # guarantees all v+lo <= 0
    hi = 1.0 - np.min(v) + 1.0     # guarantees all v+hi >= 1
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > tol:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            add = diff / free.size
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            i = rng.integers(0, v.size)
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def binary_sa_hill_climb(h_bin, rng, deadline,
                         max_iters=200000, no_improve_limit=5000):
    """Simulated‑annealing hill climb on binary vectors (swap 0↔1)."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline and it < max_iters:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > no_improve_limit:
            break
    return best_h, best_c


def gradient_descent_optimize(h0, target, deadline, rng):
    """Projected gradient descent on the smooth‑max surrogate."""
    h = np.clip(h0, 0.0, 1.0)
    h = project_onto_box_and_sum(h, target, rng)
    n = h.size
    beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000,
                     2000, 5000, 10000]
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        step = max(0.05, 0.5 / np.sqrt(beta))
        max_iter = 200
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            c = np.correlate(h, 1 - h, mode='full')
            maxc = np.max(c)
            exp_arg = beta * (c - maxc)        # maxc makes max exponent zero
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum == 0.0:                   # extreme underflow, fall back to max index
                w = np.zeros_like(w)
                w[n - 1] = 1.0
            else:
                w /= w_sum
            w_rev = w[::-1]
            conv1 = np.convolve(w, 1 - h, mode='full')
            conv2 = np.convolve(w_rev, h, mode='full')
            g = conv1[n - 1:2 * n - 1] - conv2[n - 1:2 * n - 1]
            h = project_onto_box_and_sum(h - step * g, target, rng)
        # small jitter to escape possible plateaus
        h = project_onto_box_and_sum(h + rng.normal(scale=0.005, size=n),
                                     target, rng)
    return h


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.98

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    # candidate resolutions (coarse → fine)
    candidate_ns = [360, 240, 180]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n / 2.0
        start_vectors = []

        # interpolated parent (if any)
        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vectors.append(interp)                     # fractional
            idx = np.argpartition(interp, -int(target))[-int(target):]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vectors.append(bin_vec)                    # binary

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, int(target), replace=False)] = 1.0
        start_vectors.append(bin_rand)

        # random fractional (scaled to exact sum)
        rand_frac = rng.random(n)
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        rand_frac = project_onto_box_and_sum(rand_frac, target, rng)
        start_vectors.append(rand_frac)

        # optimise each start
        for h0 in start_vectors:
            if time.time() > deadline:
                break
            h0 = project_onto_box_and_sum(np.clip(h0, 0.0, 1.0), target, rng)
            h_opt = gradient_descent_optimize(h0, target, deadline, rng)
            c = compute_c5(h_opt)
            if c < best_c:
                best_c, best_h, best_n = c, h_opt.copy(), n

    # ---------- up‑sampling for extra refinement ----------
    if (best_h is not None) and (best_n * 2 <= 720) and (time.time() < deadline):
        n_up = best_n * 2
        target_up = n_up / 2.0
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = project_onto_box_and_sum(h_up, target_up, rng)

        h_up = gradient_descent_optimize(h_up, target_up, deadline, rng)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

    # ---------- optional binary SA on the final (if time permits) ----------
    if best_h is not None and best_n <= 720 and time.time() < deadline:
        target = best_n / 2.0
        idx = np.argpartition(best_h, -int(target))[-int(target):]
        h_bin = np.zeros(best_n)
        h_bin[idx] = 1.0
        h_bin, c_bin = binary_sa_hill_climb(h_bin, rng, deadline)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.astype(float)

    # ---------- fallback ----------
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_onto_box_and_sum(best_h, n / 2.0, rng)
        best_c = compute_c5(best_h)

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_and_sum(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
