def subgradient(h):
    n = h.size
    # compute full correlation to find k_max
    conv = correlate_full(h)  # length 2n-1
    idx = conv.argmax()
    k = idx - (n - 1)  # shift offset
    # compute gradient
    grad = np.zeros_like(h)
    if k >= 0:
        # for i from 0 to n-1-k: term (1 - h[i+k])*dx
        grad[:n-k] += (1 - h[k:])  # vector of length n-k
        # term -h[i-k] for i >= k (actually i - k >=0)
        grad[k:] -= h[:n-k]
    else:
        # k negative: shift left
        kk = -k
        # for i from kk to n-1: (1 - h[i+k]) = (1 - h[i-kk])
        grad[kk:] += (1 - h[:n-kk])
        # for i from 0 to n-kk-1: -h[i-k] = -h[i+kk]
        grad[:n-kk] -= h[kk:]
    grad *= (2.0 / n)  # multiply by dx (2/n)
    return grad, k
```

    y = np.clip(x, 0.0, 1.0)

        for _ in range(5000):

                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - y[i])
            else:
                add = max(diff, -y[i])
            y[i] += add
            diff = target - y.sum()

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[:2 * n - 1]
    return conv
```

def compute_c5(h):
    return correlate_full(h).max() * (2.0 / h.size)

import time
import numpy as np

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    return correlate_full(h).max() * (2.0 / h.size)

def subgradient_and_c5(h):
    n = h.size
    conv = correlate_full(h)
    idx = conv.argmax()
    k = idx - (n - 1)
    c = conv[idx] * (2.0 / n)
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += (1 - h[k:])
        grad[k:] -= h[: n - k]
    else:
        kk = -k
        grad[kk:] += (1 - h[: n - kk])
        grad[: n - kk] -= h[kk:]
    grad *= (2.0 / n)
    return grad, c, k

def clip_and_fix_sum(x, target, rng):
    y = np.clip(x, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        n = y.size

                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - y[i]) if diff > 0 else max(diff, -y[i])
            y[i] += add
            diff = target - y.sum()
    return y
