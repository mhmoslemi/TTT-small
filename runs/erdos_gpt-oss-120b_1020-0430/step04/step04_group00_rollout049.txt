def project_onto_box_simplex(v, target, max_iter=100, tol=1e-12):
    # v: np.ndarray
    # returns projection onto {x| sum x = target, 0 <= x <= 1}
    # Use bisection on lambda.
    # Because sum clip(v - lam, 0, 1) is decreasing in lam.
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global state for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # smooth approximation of max_{k} corr_k
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def project_onto_box_simplex(v, target, max_iter=100, tol=1e-12):
    """Euclidean projection onto {x | sum x = target, 0 <= x <= 1}."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def gradient_refine(h, rng, step_initial=0.2, max_iter=200):
    n = h.size
    target = n / 2.0
    step = step_initial
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad = 1.0 - (np.roll(h, -k) + np.roll(h, k))
        h = h - step * grad
        # small random jitter every 20 steps
        if it % 20 == 0:
            h += rng.normal(scale=step * 0.05, size=n)
        h = project_onto_box_simplex(h, target)
        step *= 0.97
        if step < 1e-8:
            break
    return h

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = res.x
    x = project_onto_box_simplex(x, CURRENT_TARGET)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None
    candidate_ns = [360, 240, 180]

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        start_vecs = []

        if parent is not None:
            xs_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xs_n = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xs_n, xs_parent, parent)
            start_vecs.append(interp)                     # fractional
            bin_vec = np.zeros(n)
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)                    # binary

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        target_sum = n / 2.0
        rand_frac = rand_frac * (target_sum / rand_frac.sum())
        start_vecs.append(rand_frac)

        for base_h in start_vecs:
            if time.time() >= deadline:
                break
            h = np.array(base_h, copy=True, dtype=float)
            h = project_onto_box_simplex(h, n / 2.0)

            # early gradient refinement
            h = gradient_refine(h, rng, step_initial=0.3, max_iter=80)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # SLSQP cascade
            for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000):
                if time.time() >= deadline:
                    break
                max_iter = 300 if beta < 200 else 500
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                h = project_onto_box_simplex(h, n / 2.0)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final gradient sweep
            h = gradient_refine(h, rng, step_initial=0.15, max_iter=120)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # ---------- optional up‑sampling ----------
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = project_onto_box_simplex(h_up, n_up / 2.0)

        h_up = gradient_refine(h_up, rng, step_initial=0.2, max_iter=150)

        for beta in (10, 50, 200, 800, 3000, 10000, 25000):
            if time.time() >= deadline:
                break
            max_iter = 500 if beta >= 200 else 300
            try:
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            h_up = project_onto_box_simplex(h_up, n_up / 2.0)

        h_up = gradient_refine(h_up, rng, step_initial=0.1, max_iter=200)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ---------- fallback ----------
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_onto_box_simplex(best_h, n / 2.0)
        best_c = compute_c5(best_h)
        best_n = n

    # ---------- final safety ----------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_simplex(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
