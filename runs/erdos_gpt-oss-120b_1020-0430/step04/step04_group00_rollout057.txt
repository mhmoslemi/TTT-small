def solve_milp(n):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == n//2, t >= 0]
    # mapping y index for each pair
    y_vars = []
    for k in range(n):
        num_i = n - k
        y = cp.Variable(num_i, boolean=True)
        y_vars.append(y)
        # linearization constraints:
        # y_i <= h_i
        # y_i <= h_{i+k}
        # y_i >= h_i + h_{i+k} - 1
        constraints += [y <= h[:num_i],

        constraints.append(cp.sum(y) >= t)
    # objective max t
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.CBC, warm_start=True, max_iters=1000000)
    # retrieve solution
    h_val = h.value
    t_val = t.value
    return h_val, t_val

def continuous_sa(h, rng, time_deadline):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    it = 0
    while time.time() < time_deadline:
        i, j = rng.choice(len(cur_h), size=2, replace=False)
        # propose transfer delta from i to j (or vice versa)
        max_delta = min(cur_h[i], 1 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        cur_h[i] -= delta
        cur_h[j] += delta
        c = compute_c5(cur_h)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) * 10): # temperature scaling
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
        it += 1
    return best_h, best_c
```

def gradient_refine(h, rng, step=0.02, max_iter=20):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # compute gradient for O_k
        g = np.ones_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            # subtract h_{i+k}
            g[:n - k] -= h[k:]
            # subtract h_{i-k}
            g[k:] -= h[:n - k]
        # step size schedule
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust via random indices
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
```

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute mass so that sum(h) == n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=30):
    """Projected gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)

        g = np.ones_like(h)               # dO_k/dh = 1 - (h[i+k] if i+k<n else 0) - (h[i-k] if i-k>=0 else 0)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]

        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def softmax_obj(x):
    """Smooth max using log-sum-exp surrogate."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def upsample(h, factor):
    n = h.size
    n_up = n * factor
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
    h_up = np.interp(xt, xs, h)
    return h_up

    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None

    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    # candidate resolutions (small first for speed)

        # build start vectors
        start_vecs = []

        # down‑sampled parent if possible
        if parent is not None:

                interp = np.interp(xt, xs, parent)
            else:
                interp = parent.copy()
            start_vecs.append(interp)                         # fractional
            # binary version of the interpolated vector
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
