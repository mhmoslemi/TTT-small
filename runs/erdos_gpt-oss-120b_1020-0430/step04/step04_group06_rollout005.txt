def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine n, get initial vector
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0., 1.)
        # ensure sum exactly n/2
        h0 = enforce_sum(h0, n/2., rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # Build candidate pool
    xs = (np.arange(n) + 0.5) * (2./n)
    candidates = [h0]
    # add sin, cos transforms etc.
    for f in (1,2,3):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            h = np.clip(h, 0., 1.)
            h = enforce_sum(h, n/2., rng)
            candidates.append(h)
    # linear ramps:
    candidates.append(enforce_sum(xs / 2., n/2., rng))
    candidates.append(enforce_sum(1. - xs/2., n/2., rng))
    # binary masks:
    mask = np.zeros(n)
    mask[:n//2] = 1.
    candidates.append(enforce_sum(mask, n/2., rng))
    mask = np.zeros(n)
    start = n//4
    mask[start:start+n//2] = 1.
    candidates.append(enforce_sum(mask, n/2., rng))
    mask = np.arange(n) % 2
    candidates.append(enforce_sum(mask, n/2., rng))
    # random a few:
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # select best
    best_h = None
    best_c = float('inf')
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Simulated annealing
    cur_h = best_h.copy()
    cur_c = best_c
    best_h_sa = cur_h.copy()
    best_c_sa = cur_c
    # schedule parameters
    total_time = budget_s
    remaining_time = total_time - (time.time() - start_time)
    # allocate portion for SA, e.g., 0.6 of budget
    sa_time = remaining_time * 0.6
    end_sa = time.time() + sa_time
    # temperature schedule
    T0 = 0.02
    Tmin = 1e-5
    # We'll compute alpha based on expected number of steps.
    # Approx iterations = sa_time / avg_iter_time (maybe 0.001)
    # Let's determine a rough iteration target: 200k steps.
    # We'll compute alpha = (Tmin/T0)^(1/iterations)
    # But we don't know exact iterations; we can update T by multiplying per step.
    # Simpler: use linear cooling: T = T0 * max(0, (end_sa - time_now) / sa_time)
    # For probability: using T.
    while time.time() < end_sa:
        # choose i,j
        # sample i with prob ~ h_i (if > 0)
        # we can get non-zero entries indices via np.where?
        # For speed, we can just sample i uniformly and then ensure max_eps >0.
        i = rng.integers(0, n)
        # Ensure i is not at bound to have some epsilon
        # We'll try up to some attempts
        max_eps_i = cur_h[i]
        if max_eps_i <= 0.0:
            continue
        # sample j until different and (1 - h[j]) >0
        j = rng.integers(0, n)

def random_feasible_vector(n, rng):
    # generate random numbers uniform
    h = rng.random(n)
    # rescale to sum target
    target = n/2.0
    s = h.sum()
    if s == 0:
        h[:] = 0.5
        s = h.sum()

    # clip and adjust sum exactly

    # final enforce sum
    h = enforce_sum(h, target, rng)
    return h

import time
import numpy as np

# -------------------------------------------------
# Helper functions
# -------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that sum(h) == target while keeping 0<=h<=1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector with sum = n/2 and 0<=h<=1."""
    h = rng.random(n)
    target = n / 2.0
    s = h.sum()
    if s == 0:
        h[:] = 0.5
        s = h.sum()
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Local search moving mass from i to i+shift to lower the maximal overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c


def simulated_annealing(start_h, time_budget, rng):
    """Run SA from start_h for at most time_budget seconds; return best found."""
    n = start_h.size
    target = n / 2.0
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    # temperature schedule
    Tmin = 1e-5
    T0 = max(0.02, cur_c * 0.05)  # start a bit higher if cur_c is large
    end = time.time() + time_budget
    it = 0
    while time.time() < end:
        # pick i,j with non‑trivial epsilon
        i = rng.integers(0, n)
        if cur_h[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if j == i or cur_h[j] >= 1.0:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random()
        # propose
        cur_h[i] -= eps
        cur_h[j] += eps
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        # temperature based on remaining fraction
        frac = (end - time.time()) / time_budget
        T = max(Tmin, T0 * frac)
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += eps
            cur_h[j] -= eps
        it += 1
        if it % 5000 == 0:
            cur_h = enforce_sum(cur_h, target, rng)
            cur_c = compute_c5(cur_h)
    # final correction
    best_h = enforce_sum(best_h, target, rng)
    best_c = compute_c5(best_h)
    return best_h, best_c
