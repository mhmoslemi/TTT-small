def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0: break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

# -------------------- core utilities --------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for vector h on [0,2] (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_shift_info(h: np.ndarray):
    """Return worst shift index, its offset s and the correlation value."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (h.size - 1)
    return s, idx, corr[idx]


def gradient_shift(h: np.ndarray, s: int):
    """Gradient of the correlation at the worst shift w.r.t. h."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += 1.0 - h[s:]
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += 1.0 - h[: n - s_abs]
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0.0:  # need to increase total mass
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease total mass
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def greedy_optimize(h: np.ndarray, max_steps: int = 2000):
    """Deterministic improvement using full‑epsilon transfers."""
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h


# -------------------- SLSQP smoothing --------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final projection to exact sum
    rng_local = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng_local)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ----- initialise -----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 720
        h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    # ----- simulated annealing -----
    T0 = 0.001          # initial temperature
    Tmin = 1e-7
    iteration = 0
    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= budget_s * 0.95:
            break
        # linear cooling
        T = max(Tmin, T0 * (1.0 - elapsed / (budget_s * 0.95)))
        eps_factor = 0.5 + 0.5 * (T / T0)  # between 0.5 and 1.0
        if rng.random() < 0.7:
            # gradient based move
            s, _, _ = worst_shift_info(h)
            grad = gradient_shift(h, s)
            pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
            neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
            if pos.size > 0 and neg.size > 0:
                i = pos[np.argmax(grad[pos])]
                j = neg[np.argmin(grad[neg])]
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps < 1e-14:
                    i = j = None
                else:
                    eps = max_eps * rng.random() * eps_factor
            else:
                i = j = None
        else:
            i = j = None

        # fallback random move if needed
        if i is None or j is None:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while i == j or h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps < 1e-14:
                iteration += 1
                continue
            eps = max_eps * rng.random() * eps_factor

        # apply move
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / T):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()

        # occasional deterministic polishing
        if iteration > 0 and iteration % 50000 == 0:
            h = greedy_optimize(h, max_steps=500)
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        iteration += 1

    # ----- final polishing -----
    remaining = budget_s - (time.time() - start_time)
    # quick greedy sweep
    best_h = greedy_optimize(best_h, max_steps=2000)
    best_c = compute_c5(best_h)

    # optional SLSQP refinement if time permits
    if remaining > 30.0:
        for beta in (200, 800, 2000, 5000):
            if time.time() - start_time > budget_s * 0.98:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=600)
            except Exception:
                continue
            c_ref = compute_c5(refined)
            if c_ref < best_c:
                best_c = c_ref
                best_h = refined

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
