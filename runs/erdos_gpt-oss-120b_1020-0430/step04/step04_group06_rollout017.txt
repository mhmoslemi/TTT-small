def simulated_annealing(h0, rng, time_limit):
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    T = cur_c * 0.02  # start temp about 2% of objective
    decay = 0.9995
    iter = 0
    # precompute for speed: maybe use local variable references.
    while time.time() - start < time_limit:
        # random move
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # compute max epsilon
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # epsilon draw: random in (0, max_eps] scaled
        eps = max_eps * rng.random() * 0.5
        # apply move
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta < 0:
            accept = True
        else:
            # up-hill acceptance
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
        # update temperature
        iter += 1
        if iter % 1000 == 0:
            T *= decay
            if T < 1e-12:

    return best_h, best_c

def targeted_move(h, rng, factor=0.5):
    # find shift with max overlap
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (h.size - 1)
    max_val = corr[idx] * (2.0 / h.size)
    # choose i with max contribution
    if shift >= 0:
        i_range = np.arange(h.size - shift)
        j_range = i_range + shift
    else:
        i_range = np.arange(-shift, h.size)
        j_range = i_range + shift
    if i_range.size == 0:
        return False
    prod = h[i_range] * (1.0 - h[j_range])
    idx_max = int(np.argmax(prod))
    i = int(i_range[idx_max])
    j = int(j_range[idx_max])
    # transfer epsilon from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return False
    eps = max_eps * factor
    # apply move
    h[i] -= eps
    h[j] += eps
    # ensure bounds
    h[i] = max(0.0, min(1.0, h[i]))
    h[j] = max(0.0, min(1.0, h[j]))

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 = max_k ∫h(1‑h∘k).  dx = 2/n."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that sum(h)=target, keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random h with Σh = n/2, 0≤h≤1."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def sum_eq_constraint(x, target):
    return np.sum(x) - target


def softmax_obj(x, beta, target):
    """Log‑sum‑exp surrogate of max_k Σ_i x_i(1‑x_{i+k}) /dx."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """Refine h via SLSQP on the smooth surrogate."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint, "args": (target,)}
    res = minimize(
        softmax_obj,
        h0,
        args=(beta, target),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny drift)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng, max_iter=2000)
    return x


def simulated_annealing(h0, rng, time_limit):
    """Global search using Metropolis moves with occasional targeted transfers."""
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    T = cur_c * 0.02  # initial temperature ~2% of objective
    decay = 0.9995
    it = 0
    # pre‑bind for speed
    corr_func = np.correlate

    while time.time() - start < time_limit:
        it += 1
        # occasional targeted move
        if rng.random() < 0.07:
            # compute current worst shift
            corr = corr_func(h, 1.0 - h, mode="full")
            idx = int(np.argmax(corr))
            shift = idx - (n - 1)
            if shift >= 0:
                i_range = np.arange(n - shift)
                j_range = i_range + shift
            else:
                i_range = np.arange(-shift, n)
                j_range = i_range + shift
            if i_range.size == 0:
                continue
            prod = h[i_range] * (1.0 - h[j_range])
            idx_max = int(np.argmax(prod))
            i = int(i_range[idx_max])
            j = int(j_range[idx_max])
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * 0.4
            h[i] -= eps
            h[j] += eps
            new_c = compute_c5(h)
            delta = new_c - cur_c
            accept = delta < 0 or rng.random() < np.exp(-delta / T)
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] += eps
                h[j] -= eps
        else:
            # generic random transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random() * 0.5
            h[i] -= eps
            h[j] += eps
            new_c = compute_c5(h)
            delta = new_c - cur_c
            accept = delta < 0 or rng.random() < np.exp(-delta / T)
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] += eps
                h[j] -= eps

        if it % 1000 == 0:
            T = max(T * decay, 1e-12)

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- 1. Choose discretisation
    n = 360                     # 2‑unit interval split
    target_sum = n / 2.0

    # ---- 2. Build a diverse pool of starts
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = []

    # use parent if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_sum, rng)
        candidates.append(h0)

    # simple analytic seeds
    for f in (1, 2, 3, 4):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, target_sum, rng))
    # linear ramps
    candidates.append(enforce_sum(xs / 2.0, target_sum, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, target_sum, rng))
    # binary masks
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, target_sum, rng))
    mask = np.zeros(n)
    start = n // 4
    mask[start : start + n // 2] = 1.0
    candidates.append(enforce_sum(mask, target_sum, rng))
    mask = np.arange(n) % 2
    candidates.append(enforce_sum(mask, target_sum, rng))
    # random seeds
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # ---- 3. Pick best initial
    best_h = None
    best_c = np.inf
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # ---- 4. SLSQP polishing (moderate β list)
    beta_list = [10, 50, 200, 800, 2000, 5000]
    time_main = budget_s * 0.55
    while time.time() - start_time < time_main:
        for beta in beta_list:
            if time.time() - start_time >= time_main:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, rng, max_iter=500)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
            except Exception:
                continue

    # ---- 5. Simulated annealing on remaining time
    time_left = budget_s - (time.time() - start_time)
    if time_left > 20.0:
        # allocate most of it to SA
        sa_time = time_left * 0.85
        best_h, best_c = simulated_annealing(best_h, rng, sa_time)

    # ---- 6. Final high‑β polish
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5.0:
        for beta in [8000, 15000, 30000]:
            try:
                h_ref = slsqp_optimize(best_h, beta, rng, max_iter=400)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
            except Exception:
                continue

    # ---- 7. Clean‑up and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
