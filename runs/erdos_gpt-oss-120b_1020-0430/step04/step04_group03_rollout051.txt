def perturb_move(h, rng, scale):
    n = h.size
    # choose i, j
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    # compute max possible epsilon: decrease at i, increase at j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    # sample epsilon = max_eps * scale * rng.random()
    eps = max_eps * scale * rng.random()
    # apply
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    # sum unchanged, bounds preserved, so no need to project. However if EPS is 0, keep.
    return h_new

def anneal_optimize(h0, rng, time_limit, start_time):
    cur = h0.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    # initial temperature relative to cur_c
    T = cur_c * 0.005  # maybe 0.5% of value
    # cooling schedule: exponential, with factor = 0.9995 per iteration
    while time.time() - start_time < time_limit:
        # choose i and j distinct
        i, j = rng.integers(0, cur.size, size=2)
        if i == j:
            continue
        # compute max epsilon
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0:
            continue
        # epsilon drawn from uniform(0, max_eps) scaled by temperature factor
        # Let scale = min(1.0, max_eps * (T / cur_c) * 5)
        # Actually better: eps = max_eps * rng.random() * (T / cur_c * 10 + 0.01)
        scale = (T / cur_c) * 10 + 0.01
        if scale > 1.0:
            scale = 1.0
        eps = max_eps * rng.random() * scale
        if eps <= 0:
            continue
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        # sum unchanged
        cand_c = compute_c5(cand)
        delta = cand_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T,1e-12)):
            cur, cur_c = cand, cand_c
            if cur_c < best_c:
                best, best_c = cur.copy(), cur_c
        # cool down
        T *= 0.9998
        # occasionally reset temperature if stuck? Could add but not necessary.
    return best, best_c

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad
```

import time
import numpy as np

# ----- basic utilities -------------------------------------------------
def project_01_sum(v, target):
    """Project v onto [0,1]^n with sum equal to target."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_01_sum(v, n / 2.0)


def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)


def worst_shift_info(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]


def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1.0 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1.0 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


# ----- deterministic local improvement ---------------------------------
def greedy_optimize2(h, max_steps=800):
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break

        best_c = compute_c5(h)
        best_eps = 0.0
        for frac in (1.0, 0.75, 0.5, 0.25):
            eps = max_eps * frac
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            c = compute_c5(nh)
            if c < best_c - 1e-14:
                best_c = c
                best_eps = eps
        if best_eps > 0.0:
            h[i] -= best_eps
            h[j] += best_eps
            continue
        break
    return h


# ----- simulated annealing ---------------------------------------------
def anneal_optimize(h0, rng, deadline):
    cur = h0.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    T = max(cur_c * 0.005, 1e-6)

    while time.time() < deadline:
        n = cur.size
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0.0:
            continue
        scale = (T / cur_c) * 10.0 + 0.01
        if scale > 1.0:
            scale = 1.0
        eps = max_eps * rng.random() * scale
        if eps <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = compute_c5(cand)
        delta = cand_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur, cur_c = cand, cand_c
            if cur_c < best_c:
                best, best_c = cur.copy(), cur_c
        T *= 0.9998
    return best, best_c


# ----- main entry point ------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = float(budget_s)

    # initial vector ----------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_01_sum(h0, n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: multi‑start greedy --------------------------------------
    phase1_end = start_time + total_budget * 0.55
    while time.time() < phase1_end:
        if rng.random() < 0.5:
            start = perturb_random(best_h, rng, scale=0.5)
        else:
            start = random_feasible_vector(n, rng)
        local = greedy_optimize2(start, max_steps=800)
        c_local = compute_c5(local)
        if c_local < best_c:
            best_h, best_c = local, c_local

    # Phase 2: annealing -----------------------------------------------
    phase2_end = start_time + total_budget * 0.90
    if time.time() < phase2_end:
        cand_h, cand_c = anneal_optimize(best_h, rng, phase2_end)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c
        # a quick polish with greedy after annealing
        best_h = greedy_optimize2(best_h, max_steps=800)
        best_c = compute_c5(best_h)

    # Phase 3: final tiny random hill climbing -------------------------
    phase3_end = start_time + total_budget * 0.99
    while time.time() < phase3_end:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_01_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
