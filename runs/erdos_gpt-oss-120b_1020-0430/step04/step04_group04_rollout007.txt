def perturb_multi(h, rng, m=5, scale=0.2):
    n = h.size
    idx = rng.choice(n, size=m, replace=False)
    # generate random delta with zero sum
    delta = rng.normal(size=m)
    delta -= delta.mean()
    # max allowed scaling for each based on bounds
    max_scale = np.inf
    for i, d in zip(idx, delta):
        if d > 0:
            max_scale = min(max_scale, (1 - h[i]) / d)
        elif d < 0:
            max_scale = min(max_scale, -h[i] / d)
    # Also enforce overall scale factor
    scale_factor = min(scale, max_scale)
    delta = delta * scale_factor
    new_h = h.copy()
    new_h[idx] += delta
    return new_h

import time, numpy as np
from scipy.optimize import minimize

def enforce_sum(h, target, rng, max_iter=1000):
    # clip and iteratively adjust.
    h = np.clip(h,0.0,1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0,n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # if still off, distribute uniformly among non-boundary entries
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            inc = diff / idx.size
            h[idx] += inc
            h = np.clip(h,0.0,1.0)
            diff = target - h.sum()
    return h

def compute_c5(h):
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L-1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H*R, m)[:L]
    return np.max(conv) * (2.0/n)

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    # perform random swaps to diversify
    for _ in range(n*3):
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    # enforce exact sum
    target = n/2.0
    return enforce_sum(h, target, rng)

def perturb_multi(h, rng, m=5, scale=0.2):
    n = h.size
    idx = rng.choice(n, size=m, replace=False)
    delta = rng.normal(size=m)
    delta -= delta.mean()
    # compute max scaling to stay within [0,1]
    max_scale = np.inf
    h_idx = h[idx]
    pos = delta > 0
    if np.any(pos):
        max_scale = min(max_scale, np.min((1 - h_idx[pos]) / delta[pos]))
    neg = delta < 0
    if np.any(neg):
        max_scale = min(max_scale, np.min(-h_idx[neg] / delta[neg]))
    # include scale factor
    factor = min(scale, max_scale)
    delta = delta * factor
    new_h = h.copy()
    new_h[idx] = h_idx + delta
    return new_h

def hill_climb(h_start, rng, max_iters=2000, step=0.3):
    h = h_start.copy()
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        new_h = perturb_multi(h, rng, m=5, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, h = c, new_h
            step = min(step*1.1, 0.5)  # increase step after success
        else:
            step = max(step*0.95, 1e-4)
    return h, best_c

def simulated_annealing(h0, rng, max_iter=8000, init_temp=1e-2, final_temp=1e-5,
                        init_scale=0.5, final_scale=0.02):
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new_h = perturb_multi(cur_h, rng, m=7, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

def slsqp_surrogate_opt(h0, beta, rng, max_iter=600):
    n = h0.size
    target = n/2.0
    bounds = [(0.0,1.0)]*n
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        rev = (1.0 - x)[::-1]
        L = 2*n-1
        mlen = 1 << ((L-1).bit_length())
        H = np.fft.rfft(x, mlen)
        R = np.fft.rfft(rev, mlen)
        conv = np.fft.irfft(H*R, mlen)[:L]
        # compute log-sum-exp
        bc = beta * conv
        max_bc = np.max(bc)
        # stable log-sum-exp
        lse = max_bc + np.log(np.sum(np.exp(bc - max_bc)))
        return lse / beta
    cons = {"type":"eq","fun":sum_eq}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter":max_iter,"ftol":1e-12})
    x = np.clip(res.x,0.0,1.0)
    # correct any tiny sum drift
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        x = enforce_sum(x, target, rng)
    return x

def mix_solutions(sol1, sol2, rng, w=None):
    if w is None:
        w = rng.random()
    return np.clip(w*sol1 + (1-w)*sol2, 0.0, 1.0)

def upsample_h(h, new_n):
    n = h.shape[0]
    old_x = (np.arange(n) + 0.5) * (2.0/n)
    new_x = (np.arange(new_n) + 0.5) * (2.0/new_n)
    return np.interp(new_x, old_x, h)

import time
import numpy as np
from scipy.optimize import minimize


def enforce_sum(h, target, rng, max_iter=1000):
    """Clip to [0,1] and adjust to achieve Σh = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            inc = diff / idx.size
            h[idx] += inc
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
    return h


def compute_c5(h):
    """Exact C5 for a discrete step function on [0,2]."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 3):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    return enforce_sum(h, n / 2.0, rng)


def perturb_multi(h, rng, m=5, scale=0.2):
    """Transfer a zero‑sum random perturbation among m coordinates."""
    n = h.size
    idx = rng.choice(n, size=m, replace=False)
    delta = rng.normal(size=m)
    delta -= delta.mean()
    h_idx = h[idx]
    max_scale = np.inf
    pos = delta > 0
    if np.any(pos):
        max_scale = min(max_scale, np.min((1 - h_idx[pos]) / delta[pos]))
    neg = delta < 0
    if np.any(neg):
        max_scale = min(max_scale, np.min(-h_idx[neg] / delta[neg]))
    factor = min(scale, max_scale)
    new_h = h.copy()
    new_h[idx] = h_idx + delta * factor
    return new_h


def hill_climb(h_start, rng, max_iters=2000, step=0.3):
    h = h_start.copy()
    best_c = compute_c5(h)
    for _ in range(max_iters):
        new_h = perturb_multi(h, rng, m=5, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, h = c, new_h
            step = min(step * 1.1, 0.5)
        else:
            step = max(step * 0.95, 1e-4)
    return h, best_c


def simulated_annealing(h0, rng, max_iter=8000, init_temp=1e-2,
                        final_temp=1e-5, init_scale=0.5, final_scale=0.02):
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new_h = perturb_multi(cur_h, rng, m=7, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def slsqp_surrogate_opt(h0, beta, rng, max_iter=600):
    """SLSQP on the smooth log‑sum‑exp surrogate."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def sum_eq(x):
        return np.sum(x) - target

    def softmax_obj(x):
        rev = (1.0 - x)[::-1]
        L = 2 * n - 1
        mlen = 1 << ((L - 1).bit_length())
        H = np.fft.rfft(x, mlen)
        R = np.fft.rfft(rev, mlen)
        conv = np.fft.irfft(H * R, mlen)[:L]
        bc = beta * conv
        m = np.max(bc)
        lse = m + np.log(np.sum(np.exp(bc - m)))
        return lse / beta

    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        x = enforce_sum(x, target, rng)
    return x


def mix_solutions(a, b, rng, w=None):
    """Convex combination (preserves sum and bounds)."""
    if w is None:
        w = rng.random()
    return np.clip(w * a + (1 - w) * b, 0.0, 1.0)


def upsample_h(h, new_n):
    n = h.size
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(new_x, old_x, h)
