def enforce_sum(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    max_iter=2000
    it=0
    while abs(diff)>1e-12 and it<max_iter:
        i = rng.integers(n)
        # compute possible change
        if diff>0:
            # increase h[i] up to 1.0
            delta = min(diff, 1.0 - h[i])
        else:
            # decrease h[i] down to 0
            delta = max(diff, -h[i])
        if delta != 0:
            h[i] += delta
            diff = target - h.sum()
        it += 1
    return h

CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:

    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 2000:
        i = rng.integers(n)
        if diff > 0:

        else:
            add = max(diff, -h[i])
        if add != 0:
            h[i] += add
            diff = target - h.sum()
        it += 1
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 600) -> np.ndarray:
    """Refine h using soft‑max SLSQP for a given β."""

    step = init_step
