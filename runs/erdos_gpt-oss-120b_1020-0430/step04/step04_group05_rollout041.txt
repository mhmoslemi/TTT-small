def enforce_sum(h, target):
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    # if diff positive, need to add to some entries not at 1.
    # Add proportionally to slack.
    if abs(diff) > 1e-12:
        slack = 1 - h
        if diff > 0:
            total_slack = slack.sum()
            if total_slack > 0:
                h += diff * (slack / total_slack)
        else:
            # diff negative: need to remove from elements >0
            pos = h
            total_pos = pos.sum()
            if total_pos > 0:

        h = np.clip(h, 0, 1)
        # If still off due to rounding, fix with small adjustments.
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute small amount uniformly
            h += diff / h.size
            h = np.clip(h, 0, 1)
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    # ensure i != j
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def softmax_obj_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_corr))
    Z = np.sum(exp_vals)
    w = exp_vals / Z  # length 2n-1
    grad = np.full(n, 1.0)
    for idx, w_l in enumerate(w):
        lag = idx - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] -= w_l * h[lag:]
                grad[lag:] -= w_l * h[:n - lag]
        else:
            L = -lag
            if n - L > 0:
                grad[L:] -= w_l * h[:n - L]
                grad[:n - L] -= w_l * h[L:]
    # answer gradient of softmax_obj (divide by beta)
    obj = (max_corr + np.log(Z)) / beta
    grad = grad / beta
    return obj, grad

import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        for _ in range(max_iter):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def compute_c5_corr(h):
    corr = np.correlate(h, 1 - h, mode="full")
    c = np.max(corr) * (2.0 / h.size)
    return c, corr


def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    if lag >= n:
        return perturb_h(h, rng, scale)
    contrib = np.empty(n)
    if lag == 0:
        contrib = h * (1 - h)
    else:
        contrib[: n - lag] = h[: n - lag] * (1 - h[lag:])
        contrib[n - lag :] = h[n - lag :]
    k = max(1, int(n * top_frac))
    top_idx = np.argpartition(-contrib, k - 1)[:k]
    bot_idx = np.argpartition(contrib, k - 1)[:k]
    i = rng.choice(top_idx)
    if rng.random() < 0.7:
        j = rng.choice(bot_idx)
    else:
        j = rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.02, 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        lag = np.argmax(cur_corr) - (n - 1)
        lag_abs = abs(lag)
        if rng.random() < 0.6:
            new_h = targeted_perturb(h, rng, lag_abs, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):

            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c


def slsqp_softmax_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x


def softmax_obj_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_corr))
    Z = np.sum(exp_vals)
    w = exp_vals / Z
    grad = np.full(n, 1.0)
    for idx, w_l in enumerate(w):
        lag = idx - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[: n - lag] -= w_l * h[lag:]
                grad[lag:] -= w_l * h[: n - lag]
        else:
            L = -lag
            if n - L > 0:
                grad[L:] -= w_l * h[: n - L]
                grad[: n - L] -= w_l * h[L:]
    obj = (max_corr + np.log(Z)) / beta
    grad = grad / beta
    return obj, grad

        alpha = 0.5 / (step + 1)
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
    return h
