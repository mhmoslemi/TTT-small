def thue_morse_seq(n):
    length = 1

    while length < n:
        a = np.concatenate((a, 1 - a))

def adjust_to_half_one(arr, rng):
    n = arr.size
    target = n // 2
    current = int(arr.sum())
    diff = target - current
    if diff > 0:
        zeros = np.where(arr == 0)[0]
        if zeros.size > 0:
            idx = rng.choice(zeros, size=diff, replace=False)
            arr[idx] = 1
    elif diff < 0:
        ones = np.where(arr == 1)[0]
        if ones.size > 0:
            idx = rng.choice(ones, size=-diff, replace=False)
            arr[idx] = 0
    return arr

import time, math, numpy as np
from scipy.optimize import minimize

# globals for SLSQP soft‑max
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Log‑sum‑exp smooth max of overlaps."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """SLSQP with smooth‑max objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    # quick redistribution to repair the sum
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        if add != 0.0:
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random h ∈[0,1]^n with Σh=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project h onto {0≤h≤1, Σh=n/2}."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def thue_morse_seq(n: int) -> np.ndarray:
    a = np.array([0], dtype=int)
    while a.size < n:
        a = np.concatenate((a, 1 - a))
    return a[:n].astype(float)

def lfsr_seq_8() -> np.ndarray:
    """Maximum‑length LFSR (period 255) for m=8."""
    state = 0xFF
    seq = np.empty(255, dtype=float)
    for i in range(255):
        seq[i] = state & 1
        bit = ((state >> 7) ^ (state >> 5) ^ (state >> 4) ^ (state >> 3)) & 1
        state = ((state << 1) & 0xFF) | bit
    return seq

def is_prime(p: int) -> bool:
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(math.isqrt(p))
    for d in range(3, r + 1, 2):
        if p % d == 0:
            return False
    return True

def next_prime_ge(x: int) -> int:
    p = max(2, int(x))
    while not is_prime(p):
        p += 1
    return p

def legendre_sequence(p: int, n: int) -> np.ndarray:
    """Legendre symbol on 0..p‑1, mapped 1→1, -1→0, 0→0."""
    seq = np.empty(n, dtype=float)
    for i in range(n):
        a = i % p
        if a == 0:
            seq[i] = 0.0
        else:
            leg = pow(a, (p - 1) // 2, p)
            seq[i] = 1.0 if leg == 1 else 0.0
    return seq

def adjust_to_half_one(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force exactly n/2 ones (binary array)."""
    n = arr.size
    target = n // 2
    cur = int(arr.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(arr == 0)[0]
        if zeros.size >= diff:
            idx = rng.choice(zeros, diff, replace=False)
            arr[idx] = 1.0
    elif diff < 0:
        ones = np.where(arr == 1)[0]
        if ones.size >= -diff:
            idx = rng.choice(ones, -diff, replace=False)
            arr[idx] = 0.0
    return arr

    start = time.time()
    best_h = None
    best_c = np.inf

    # possible sample sizes (even)
    n_candidates = [128, 192, 256, 320, 384]

    # -----------------------------------------------------------------
    # Main stochastic search
    # -----------------------------------------------------------------
    while time.time() - start < budget_s * 0.78:
        n = int(rng.choice(n_candidates))
        # Choose start type
        r = rng.random()
        if "initial_h_values" in globals() and best_h is not None and r < 0.08:
            # reuse best (possibly interpolated)
            if best_n == n:
                h = best_h.copy()
            else:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
        elif r < 0.35:
            # deterministic binary patterns
            pat = rng.choice(["thue", "lfsr", "legendre"])
            if pat == "thue":

                h = adjust_to_half_one(h, rng)
            elif pat == "lfsr":
                seq = lfsr_seq_8()
                if n <= seq.size:

        elif r < 0.65:

    # -----------------------------------------------------------------
    if best_h is not None and time.time() - start < budget_s * 0.97:
