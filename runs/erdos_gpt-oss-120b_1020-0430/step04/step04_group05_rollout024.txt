def projected_gradient_descent(h0, beta, num_iters, step_alpha, rng):
    h = h0.copy()
    n = h.size
    target = n/2.0
    for it in range(num_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        # compute probabilities
        max_corr = np.max(corr)  # not needed but used for numerical stability
        #  We need p = exp(beta * corr) / sum exp
        # To avoid overflow: subtract max_beta = beta * max_corr for exponent.
        exp_term = np.exp(beta * (corr - max_corr))
        p = exp_term / exp_term.sum()
        w = p + p[::-1]  # symmetric
        conv = np.convolve(h, w, mode='full')
        g = 1.0 - conv[n-1:n-1+n]  # gradient of softmax beta objective
        # projection to zero-sum subspace: subtract mean
        g_proj = g - np.mean(g)
        # step
        h_new = h - step_alpha * g_proj
        # projection to [0,1] and maintain sum: clip + enforce_sum
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce sum quickly: use simple adjust using diff/remaining unset?
        # We'll use the earlier enforce_sum function (maybe rename).
        h_new = enforce_sum(h_new, target, rng)
        h = h_new
    return h

def gradient_optimize(h0, beta_list, steps_per_beta, step_alpha, rng):

    h = h0.copy()
    n = h.size
    target = n/2.0
    for beta, steps, alpha in zip(beta_list, steps_per_beta, step_alpha):
        # pre-calc scaling? Not needed.
        for it in range(steps):
            corr = np.correlate(h, 1 - h, mode='full')
            # compute probabilities
            maxc = np.max(corr)
            exp_term = np.exp(beta * (corr - maxc))
            p = exp_term / exp_term.sum()
            w = p + p[::-1]
            conv = np.convolve(h, w, mode='full')
            grad = 1.0 - conv[n-1:n-1+n]
            # optionally add noise
            #if alpha > 0.01: # but use small noise
            #    grad += rng.normal(scale=0.01, size=n)
            # projection to zero-sum subspace
            grad -= np.mean(grad)  # ensures sum zero
            # step
            h_new = h - alpha * grad
            # clip
            h_new = np.clip(h_new, 0.0, 1.0)
            # enforce sum exactly
            h_new = enforce_sum(h_new, target, rng)
            h = h_new
    return h

def gradient_optimize(h0, beta_list, steps_list, alpha_list, rng, momentum=0.7):
    h = h0.copy()
    n = h.size
    target = n/2.0
    for beta, steps, alpha in zip(beta_list, steps_list, alpha_list):
        v = np.zeros_like(h)
        for it in range(steps):
            corr = np.correlate(h, 1 - h, mode='full')
            maxc = np.max(corr)
            exp_term = np.exp(beta * (corr - maxc))
            p = exp_term / exp_term.sum()
            w = p + p[::-1]
            conv = np.convolve(h, w, mode='full')
            grad = 1.0 - conv[n-1:n-1+n]
            # Subtract mean to keep sum zero direction
            grad -= grad.mean()
            # momentum update
            v = momentum * v - alpha * grad
            h_new = h + v
            # clip and enforce sum
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, target, rng)
            h = h_new
    return h

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # choose an index where we can adjust
        if diff > 0:
            # need to increase sum: pick i with h[i] < 1
            candidates = np.where(h < 1.0)[0]
            if candidates.size == 0:
                break
            i = rng.choice(candidates)
            add = min(diff, 1.0 - h[i])
        else:
            # need to decrease sum: pick i with h[i] > 0
            candidates = np.where(h > 0.0)[0]
            if candidates.size == 0:
                break
            i = rng.choice(candidates)
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final corrective uniform adjustment if still off (unlikely)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-12:

                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])

                diff = target - h.sum()

def targeted_local_improvement(h, rng, steps=500, scale=0.1):
    n = h.size
    for _ in range(steps):
        corr = np.correlate(h, 1 - h, mode='full')
        lag = np.argmax(corr) - (n - 1)  # shift
        if lag < 0:
            lag = -lag
        # only consider positive shift
        if lag >= n:
            continue
        # evaluate contributions
        valid = np.arange(n - lag)
        contrib = h[valid] * (1.0 - h[valid + lag])
        # pick top and bottom indices
        i = rng.choice(valid[np.argsort(contrib)[-1:]])  # max
        j = rng.choice(valid[np.argsort(contrib)[:1]])  # min
        # attempt to move epsilon from i to j
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = scale * max_eps * rng.random()
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # Accept if objective improves
        cur_c = np.max(corr) * (2.0 / n)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            h = new_h
    return h

import time
import numpy as np

# ---------- utilities ----------
def enforce_sum(vec, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    v = np.clip(vec, 0.0, 1.0)
    diff = target - v.sum()
    n = v.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(v < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - v[i])
        else:
            cand = np.where(v > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    # tiny uniform correction if needed
    if abs(diff) > 1e-12:
        v += diff / n
        v = np.clip(v, 0.0, 1.0)
        diff = target - v.sum()
        if abs(diff) > 1e-12:
            for _ in range(10):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - v[i])
                else:
                    add = max(diff, -v[i])
                v[i] += add
                diff = target - v.sum()
    return v

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = rng.random(n)
    target = n / 2.0
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    else:
        h.fill(target / n)
    h = enforce_sum(h, target, rng)
    return h

def compute_c5(h):
    """Exact C5 given a step vector."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def targeted_local_improvement(h, rng, steps=300, scale=0.1):
    """Focus on the worst lag and move mass away from its maximal contributors."""
    n = h.size
    for _ in range(steps):
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr) - (n - 1)
        lag = abs(lag)
        if lag == 0 or lag >= n:
            continue
        valid = np.arange(n - lag)
        contrib = h[valid] * (1.0 - h[valid + lag])
        if contrib.size == 0:
            continue
        i = valid[np.argmax(contrib)]
        j = valid[np.argmin(contrib)]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = scale * max_eps * rng.random()
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        new_c = compute_c5(nh)
        cur_c = compute_c5(h)
        if new_c < cur_c:
            h = nh
    return h

# ---------- gradient optimisation ----------
def gradient_optimize(h0, beta_list, steps_list, alpha_list, rng, momentum=0.7):
    """Projected gradient descent on the soft‑max surrogate of the max‑overlap."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for beta, steps, alpha in zip(beta_list, steps_list, alpha_list):
        v = np.zeros_like(h)
        for _ in range(steps):
            corr = np.correlate(h, 1 - h, mode="full")
            maxc = np.max(corr)
            exp_term = np.exp(beta * (corr - maxc))
            p = exp_term / exp_term.sum()
            w = p + p[::-1]                 # symmetric weights, sum = 2
            conv = np.convolve(h, w, mode="full")
            grad = 1.0 - conv[n - 1 : n - 1 + n]   # gradient of soft‑max objective
            grad -= grad.mean()                     # project onto sum‑zero subspace
            v = momentum * v - alpha * grad
            h_new = h + v
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, target, rng)
            h = h_new
    return h

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # parameters for the continuation method
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    steps_list = [2500, 2500, 2500, 2000, 2000, 1500, 1200, 800]
    alpha_list = [0.30, 0.25, 0.22, 0.18, 0.15, 0.12, 0.09, 0.07]

    main_budget = budget_s * 0.85
    while time.time() - t0 < main_budget:
        # decide start point
        if rng.random() < 0.6:
            start = random_feasible_vector(n, rng)
        else:
            # perturb the current best
            start = perturb_h(best_h, rng, scale=0.10)
            start = enforce_sum(start, n / 2.0, rng)

        # gradient descent on the soft‑max surrogate
        h_opt = gradient_optimize(start, beta_list, steps_list, alpha_list, rng)

        # a quick targeted cleanup
        h_opt = targeted_local_improvement(h_opt, rng, steps=200, scale=0.07)

        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        # tiny random hill‑climb for polishing
        for _ in range(200):
            nh = perturb_h(best_h, rng, scale=0.07)
            nc = compute_c5(nh)
            if nc < best_c:
                best_c, best_h = nc, nh.copy()

    # -------- up‑sampling if time permits --------
    remaining = budget_s - (time.time() - t0)
    if remaining > budget_s * 0.1:
        target_n = min(720, n * 2)
        if target_n != n:

            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)

            # run a shorter continuation on the finer grid
            beta_up = [20, 80, 200, 500, 1000, 2000, 5000]
            steps_up = [1200, 1200, 1200, 800, 800, 600, 400]
            alpha_up = [0.18, 0.15, 0.12, 0.10, 0.08, 0.07, 0.06]
            up_h = gradient_optimize(up_h, beta_up, steps_up, alpha_up, rng)

            up_h = targeted_local_improvement(up_h, rng, steps=200, scale=0.05)

            if up_c < best_c:

        nh = perturb_h(best_h, rng, scale=0.05)

        if nc < best_c:
            best_c, best_h = nc, nh.copy()

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
