import time
import numpy as np
from scipy.optimize import minimize

# Global variables for constraints used by the smooth surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Random point in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    """Full linear correlation of h with (1-h) via FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k Σ_i h_i (1 - h_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.exp(bc - m).sum())) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

def binary_sa(start_bin, rng, deadline):
    """Swap‑based simulated annealing on a binary vector (sum = n/2)."""
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    it = 0
    max_iters = 20000
    while time.time() < deadline and it < max_iters:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            it += 1
            continue
        if h[i] < h[j]:
            i, j = j, i
        # perform swap: i becomes 0, j becomes 1
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        it += 1
    return best_h, best_c

def downsample_to(arr, target_n):
    """Downsample by averaging contiguous blocks (assumes divisor)."""
    n = arr.size
    if n == target_n:
        return arr.copy()
    factor = n // target_n
    if factor * target_n != n:
        # fallback to linear interpolation
        xs = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        xs_t = np.linspace(0, 2, target_n, endpoint=False) + 1.0 / target_n
        return np.interp(xs_t, xs, arr)
    return arr.reshape(target_n, factor).mean(axis=1)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # Base resolution (coarse) and initial vector
    # ------------------------------------------------------------------
    coarse_n = 360
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=float)
        h0 = downsample_to(h_parent, coarse_n)
    else:
        h0 = random_feasible_vector(coarse_n, rng)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, coarse_n / 2.0, rng)

    best_h = h0
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 1: coarse multi‑start SLSQP with increasing β
    # ------------------------------------------------------------------
    beta_ladder_coarse = [10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0]
    phase1_deadline = start_time + budget_s * 0.30
    while time.time() < phase1_deadline:
        if rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(coarse_n, rng)
        cur = start_vec
        for beta in beta_ladder_coarse:
            if time.time() >= phase1_deadline:
                break
            cur = slsqp_optimize(cur, beta, max_iter=250)
            cur_c = compute_c5(cur)
            if cur_c < best_c:
                best_c, best_h = cur_c, cur.copy()
        if time.time() < phase1_deadline:
            hc_h, hc_c = hill_climb(best_h, rng, max_iters=1500, start_step=0.4)
            if hc_c < best_c:
                best_c, best_h = hc_c, hc_h.copy()

    # ------------------------------------------------------------------
    # Phase 2: Upsample to fine resolution (720) and polish
    # ------------------------------------------------------------------
    fine_n = 720
    if best_h.size * 2 == fine_n:
        h_fine = np.repeat(best_h, 2)
    else:
        # generic up‑sample via interpolation
        xs_c = np.linspace(0, 2, best_h.size, endpoint=False) + 1.0 / best_h.size
        xs_f = np.linspace(0, 2, fine_n, endpoint=False) + 1.0 / fine_n
        h_fine = np.interp(xs_f, xs_c, best_h)
    h_fine = enforce_sum(h_fine, fine_n / 2.0, rng)
    best_h, best_c = h_fine, compute_c5(h_fine)

    # ------------------------------------------------------------------
    # Phase 3: Fine multi‑start continuous optimization
    # ------------------------------------------------------------------
    beta_ladder_fine = [200.0, 800.0, 2000.0, 8000.0, 20000.0]
    phase3_deadline = start_time + budget_s * 0.85
    while time.time() < phase3_deadline:
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.2)
        else:
            start_vec = random_feasible_vector(fine_n, rng)
        cur = start_vec
        for beta in beta_ladder_fine:
            if time.time() >= phase3_deadline:
                break
            cur = slsqp_optimize(cur, beta, max_iter=200)
        cur_c = compute_c5(cur)
        if cur_c < best_c:
            best_c, best_h = cur_c, cur.copy()
        if time.time() < phase3_deadline:
            hc_h, hc_c = hill_climb(best_h, rng, max_iters=2000, start_step=0.15)
            if hc_c < best_c:
                best_c, best_h = hc_c, hc_h.copy()

    # ------------------------------------------------------------------
    # Phase 4: Binary simulated annealing on the fine grid
    # ------------------------------------------------------------------
    phase4_deadline = start_time + budget_s * 0.97
    idx_top = np.argsort(-best_h)[: fine_n // 2]
    bin_start = np.zeros(fine_n)
    bin_start[idx_top] = 1.0
    bin_h, bin_c = binary_sa(bin_start, rng, phase4_deadline)
    if bin_c < best_c:
        best_c, best_h = bin_c, bin_h.copy()

    # ------------------------------------------------------------------
    # Phase 5: Optional up‑sample to 1440 for last polishing
    # ------------------------------------------------------------------
    if time.time() < start_time + budget_s * 0.99:
        extra_n = best_h.size * 2
        if extra_n <= 1440:
            h_big = np.repeat(best_h, 2)
            h_big, c_big = hill_climb(h_big, rng, max_iters=2000, start_step=0.1)
            # downsample back by averaging pairs
            h_small = h_big.reshape(best_h.size, 2).mean(axis=1)
            h_small = enforce_sum(h_small, best_h.size / 2.0, rng)
            c_small = compute_c5(h_small)
            if c_small < best_c:
                best_c, best_h = c_small, h_small.copy()

    # ------------------------------------------------------------------
    # Final cleanup
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
