def init_binary_from_parent(parent, n, rng):
    # Interpolate parent on [0,2) to n points
    xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    interp = np.interp(xt, xs, parent)
    bin_vec = (interp > 0.5).astype(float)
    cur = bin_vec.sum()
    target = n // 2
    diff = target - cur
    # diff may be positive or negative
    # adjust by flipping indices with nearest to 0.5
    idx_order = np.argsort(np.abs(interp - 0.5))
    for idx in idx_order:
        if diff == 0:
            break
        if diff > 0 and bin_vec[idx] == 0:
            bin_vec[idx] = 1
            diff -= 1
        elif diff < 0 and bin_vec[idx] == 1:
            bin_vec[idx] = 0
            diff += 1
    # ensure sum correct
    assert bin_vec.sum() == target
    return bin_vec
```

def binary_sa_optimize(h0, rng, time_deadline):
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # build ones and zeros index arrays
    ones = np.flatnonzero(cur_h == 1)

    # parameters
    T0 = max(0.001, cur_c * 0.005)  # small initial temperature
    alpha = 0.9999
    it = 0
    last_best_it = 0
    while time.time() < time_deadline:
        # pick random swap indices via ones and zeros arrays
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = int(ones[idx_i])
        j = int(zeros[idx_j])
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        # update ones and zeros arrays
        ones[idx_i] = j
        zeros[idx_j] = i
        # compute new C5
        c = compute_c5(cur_h)
        # acceptance
        T = T0 * (alpha ** it)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / max(T, 1e-12)):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_best_it = it
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
            # revert arrays
            ones[idx_i] = i
            zeros[idx_j] = j
        it += 1
        # optionally restart if no improvement for many iterations
        if it - last_best_it > 200000:
            # random restart but keep best_h as base
            cur_h = best_h.copy()
            cur_c = best_c
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
            last_best_it = it
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.shape[0]
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff randomly
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated on a uniform grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def init_binary_from_parent(parent, n, rng):
    """Interpolate a parent step function to length n and binarize."""
    xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    interp = np.interp(xt, xs, parent)
    bin_vec = (interp > 0.5).astype(float)
    target = n // 2
    cur = int(bin_vec.sum())
    diff = target - cur
    if diff != 0:
        order = np.argsort(np.abs(interp - 0.5))
        for idx in order:
            if diff == 0:
                break
            if diff > 0 and bin_vec[idx] == 0:
                bin_vec[idx] = 1.0
                diff -= 1
            elif diff < 0 and bin_vec[idx] == 1:
                bin_vec[idx] = 0.0
                diff += 1
    cur = int(bin_vec.sum())
    if cur != target:
        zeros = np.flatnonzero(bin_vec == 0)
        ones = np.flatnonzero(bin_vec == 1)
        while cur < target:
            i = rng.choice(zeros)
            bin_vec[i] = 1.0
            zeros = np.delete(zeros, np.where(zeros == i))
            cur += 1
        while cur > target:
            i = rng.choice(ones)
            bin_vec[i] = 0.0
            ones = np.delete(ones, np.where(ones == i))
            cur -= 1
    return bin_vec

# ---------- binary simulated‑annealing ----------
def binary_sa_optimize(h0, rng, time_deadline):
    """Swap a 1 with a 0, accept with Metropolis rule."""
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    T0 = max(0.001, cur_c * 0.005)
    alpha = 0.9999
    it = 0
    last_best = 0
    while time.time() < time_deadline:
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = int(ones[idx_i])
        j = int(zeros[idx_j])
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        ones[idx_i] = j
        zeros[idx_j] = i
        c = compute_c5(cur_h)
        T = T0 * (alpha ** it)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / max(T, 1e-12)):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_best = it
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            ones[idx_i] = i
            zeros[idx_j] = j
        it += 1
        if it - last_best > 300000:
            # restart from the best seen so far
            cur_h = best_h.copy()
            cur_c = best_c
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)
            last_best = it
    return best_h, best_c

# ---------- gradient refinement ----------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.shape[0]
    target = n / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if "initial_h_values" in globals() else None
    candidate_ns = [720, 540, 360]

    for n in candidate_ns:

            break

            if time.time() >= deadline:

            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.55)

                    c_opt = compute_c5(h_opt)

                if time.time() < deadline:

                    if c_opt < best_c:
