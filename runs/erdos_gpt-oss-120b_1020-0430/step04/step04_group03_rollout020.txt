CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    return enforce_sum(x, CURRENT_TARGET, rng)

def random_five_block_binary(n, rng):
    half = n // 2
    zeros_total = half
    while True:
        l2 = rng.integers(1, zeros_total)          # at least one zero block
        l4 = zeros_total - l2
        if l4 <= 0:
            continue
        if half < 3:
            # fallback to plain random binary
            h = np.zeros(n)
            idx = rng.choice(n, size=half, replace=False)
            h[idx] = 1.0
            return h
        l1 = rng.integers(1, half - 1)
        rem = half - l1
        if rem < 2:
            continue
        l3 = rng.integers(1, rem)
        l5 = rem - l3
        if l5 <= 0:
            continue
        if min(l1, l2, l3, l4, l5) <= 0:
            continue
        if l1 + l2 + l3 + l4 + l5 != n:
            continue
        return np.concatenate(
            [np.ones(l1), np.zeros(l2), np.ones(l3), np.zeros(l4), np.ones(l5)]
        )

# ------------------- main -------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total = budget_s

    # dimension
    if "initial_h_values" in globals():
        h_initial = np.array(initial_h_values, dtype=float)
        n = h_initial.size
    else:
        n = 720
        h_initial = None

    target = n / 2.0

    # start solution
    if h_initial is not None:
        best_h = enforce_sum(np.clip(h_initial, 0.0, 1.0), target, rng)
    else:
        best_h = random_feasible_vector(n, rng)
    best_c = compute_c5(best_h)

    # ---------- coarse grid search ----------
    n_coarse = 180
    if n > n_coarse:
        coarse_end = start_time + total * 0.10
        best_coarse_h = random_feasible_vector(n_coarse, rng)
        best_coarse_c = compute_c5(best_coarse_h)
        while time.time() < coarse_end:
            start = random_feasible_vector(n_coarse, rng)
            local = greedy_optimize(start, max_steps=2000)
            c_loc = compute_c5(local)
            if c_loc < best_coarse_c:
                best_coarse_h, best_coarse_c = local, c_loc
        # upscale
        xs_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
        xs_f = (np.arange(n) + 0.5) * (2.0 / n)
        h_up = np.interp(xs_f, xs_c, best_coarse_h)
        h_up = enforce_sum(np.clip(h_up, 0.0, 1.0), target, rng)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # ---------- main multi‑restart greedy ----------
    main_end = start_time + total * 0.70
    while time.time() < main_end:
        r = rng.random()
        if r < 0.3:
            start = perturb_random(best_h, rng, scale=0.5)
        elif r < 0.55:
            start = random_five_block_binary(n, rng).astype(float)
        elif r < 0.85:
            start = random_feasible_vector(n, rng)
        else:
            start = np.zeros(n)
            idx = rng.choice(n, size=n // 2, replace=False)
            start[idx] = 1.0
        local = greedy_optimize(start, max_steps=3000)
        c_loc = compute_c5(local)
        if c_loc < best_c:
            best_h, best_c = local, c_loc

        # occasional SLSQP polishing
        if rng.random() < 0.2:
            for beta in (200, 800, 2000, 5000):
                if time.time() > start_time + total * 0.95:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=600)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined, c_ref

    # ---------- simulated annealing around current best ----------
    sa_end = start_time + total * 0.90
    T0 = best_c * 0.01 + 1e-6
    while time.time() < sa_end:
        # pick a feasible transfer
        pos_idx = np.where(best_h > 1e-12)[0]
        neg_idx = np.where(best_h < 1.0 - 1e-12)[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            break
        i = rng.choice(pos_idx)
        j = rng.choice(neg_idx)
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            continue
        # epsilon linked to temperature
        eps = max_eps * rng.random() * (T0 / best_c)
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c_cand = compute_c5(cand)
        delta = c_cand - best_c
        elapsed = time.time() - start_time
        frac = elapsed / total
        T = T0 * (1 - frac)
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            best_h, best_c = cand, c_cand

    # ---------- final micro‑search ----------
    final_end = start_time + total * 0.99
    while time.time() < final_end:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # clean‑up and return
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
