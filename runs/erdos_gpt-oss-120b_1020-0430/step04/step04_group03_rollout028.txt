For those where delta_a > 0: epsilon >= delta_b / delta_a. So L = max(L, delta_b / delta_a). For delta_a < 0: epsilon <= delta_b / delta_a. So U = min(U, delta_b / delta_a). For delta_a == 0: need delta_b <= 0 else infeasible.

We must be careful: delta_b may be negative; delta_b / delta_a may be negative; L may be negative, but we keep L >= 0. Similarly U may be less than 0, but we must enforce feasible epsilon >= 0, so if U < 0 then no feasible epsilon >0; we may set U=0.

We also apply bound epsilon ≤ max_eps, epsilon ≥ 0.

Thus feasible epsilon interval is [max(L, 0), min(U, max_eps)]. If lower > upper, maybe no feasible epsilon; then we cannot improve S_s without violating other shifts; in that case we may fallback to something else like random move.

If feasible, since a_s likely positive (increase S_s), we set epsilon = min(U, max_eps). If a_s negative, maybe we should set epsilon = max(L, 0). But we only pick i/j expected to have a_s > 0; we can assert.

        # gradient_i = contribution: h[i+s] + h[i-s] (zero outside)
        grad = np.zeros(n)

        # similarly for -s

            grad[s:] += h[:n-s]   # h[i-s] = h[i - s] same as h[i-s] but careful: i-s >=0 => i>=s, so idx range s: n
        else:

            grad[k:] += h[:n-k]   # h[i+s] where i+s >=0
            grad[:n-k] += h[k:]   # h[i-s] where i-s < n

        # For decreasing, need h_i > 0; for increasing, h_j < 1.
        # We want minimize grad_i among decreasing candidates, and maximize grad_j among increasing.
        # If grad_i == 0 might be not useful; choose low non-zero.
        cand_decrease = np.where(h > 1e-12)[0]
        cand_increase = np.where(h < 1 - 1e-12)[0]
        if cand_decrease.size == 0 or cand_increase.size == 0:

        i = cand_decrease[np.argmin(grad[cand_decrease])]
        j = cand_increase[np.argmax(grad[cand_increase])]

import time
import numpy as np
from scipy.optimize import minimize

# Helper utilities

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # simple iterative correction
    for _ in range(10):
        i = rng.integers(0, h.size)
        adj = np.clip(diff, -h[i], 1.0 - h[i])
        h[i] += adj
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def random_feasible_vector(n, rng):
    """Uniform random vector, then re‑balance to exact sum n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    """Move a random epsilon from one entry to another."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    """Discrete C₅ for a vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    return enforce_sum(x, CURRENT_TARGET, rng)

        target_n = 720
        xs_old = (np.arange(h.size) + 0.5) * (2.0 / h.size)
        xs_new = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h = np.interp(xs_new, xs_old, h)
        h = enforce_sum(np.clip(h, 0.0, 1.0), target_n / 2.0, rng)

    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- main stochastic multi‑restart search -----
    while time.time() - start_time < budget_s * 0.80:
        if rng.random() < 0.6:
            start_vec = perturb_random(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)

        local = analytic_greedy_optimize(start_vec, max_steps=800)
        c_local = compute_c5(local)

        if c_local < best_c:
            best_h, best_c = local, c_local

        # occasional SLSQP polishing
        if rng.random() < 0.2:
            for beta in (500, 2000, 5000, 10000):
                if time.time() - start_time > budget_s * 0.95:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=800)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined, c_ref

    # ----- final fine‑grained micro‑search -----
    while time.time() - start_time < budget_s * 0.99:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # ----- cleanup and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
