def local_optimize(h, rng, max_iters=5000, init_step=0.5):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = init_step
    decay = max_iters // 10
    for it in range(max_iters):
        new_h = perturb_h(best_h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h
        # occasional bigger steps
        if (it+1) % decay == 0:
            step = max(step*0.7, 1e-5)

import time
import numpy as np
from scipy.optimize import differential_evolution

# globals that will be set inside run()
GLOBAL_N = None          # number of samples
GLOBAL_X = None          # sample midpoints on [0,2]
GLOBAL_RNG = None        # rng instance for deterministic tweaks


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def local_optimize(h, rng, max_iters=5000, init_step=0.5):
    """Simple stochastic hill‑climb on the high‑dimensional vector."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = init_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(best_h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

    """Symmetric two‑edge linear‑power shape."""
    n = GLOBAL_N
    x = GLOBAL_X
    # denominator from integral constraint
    denom = 1.0 - a * s / (s + 1.0)
    p = 1.0 / (2.0 * denom)
    h = np.empty(n, dtype=float)
    left = x < a
    right = x > 2.0 - a
    h[left] = p * (x[left] / a) ** s
    h[right] = p * ((2.0 - x[right]) / a) ** s
    h[~left & ~right] = p
    h = enforce_sum(h, n / 2.0, GLOBAL_RNG)
    return h

    """Symmetric two‑edge sine shape (zero at the very ends)."""
    n = GLOBAL_N

    left = x < a
    right = x > 2.0 - a
    h[left] = p * np.sin(np.pi * x[left] / (2.0 * a))
    h[right] = p * np.sin(np.pi * (2.0 - x[right]) / (2.0 * a))
    h[~left & ~right] = p
    h = enforce_sum(h, n / 2.0, GLOBAL_RNG)
    return h


def generate_h_gaussian(sigma):
    """Symmetric pair of Gaussians centred at the interval ends."""
    n = GLOBAL_N
    x = GLOBAL_X
    base = np.exp(-x ** 2 / (2.0 * sigma ** 2)) + np.exp(-(2.0 - x) ** 2 / (2.0 * sigma ** 2))
    # normalise to integral 1 (midpoint rule)
    area = base.sum() * (2.0 / n)
    p = 1.0 / area
    h = base * p
    if h.max() > 1.0:  # violate [0,1] -> reject later
        h = enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, GLOBAL_RNG)
    else:
        h = enforce_sum(h, n / 2.0, GLOBAL_RNG)
    return h


def objective_linear(params):
    a, s = params
    if not (0.0 < a < 1.0) or not (s > 0.0):
        return 10.0
    denom = 1.0 - a * s / (s + 1.0)
    if denom <= 0.5:
        return 10.0
    p = 1.0 / (2.0 * denom)
    if p > 1.0:
        return 10.0
    h = generate_h_linear(a, s)
    return compute_c5(h)


def objective_sine(params):
    a = params[0]
    if not (0.0 < a < 1.0):
        return 10.0
    h = generate_h_sine(a)
    return compute_c5(h)


def objective_gaussian(params):
    sigma = params[0]
    if sigma <= 0.0:
        return 10.0
    h = generate_h_gaussian(sigma)
    if h.max() > 1.0:
        return 10.0
    return compute_c5(h)


# -----------------------------------------------------------------------


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with a low Erdős C₅ bound."""
    global GLOBAL_N, GLOBAL_X, GLOBAL_RNG
    rng = np.random.default_rng(seed)
    GLOBAL_RNG = rng
    start_time = time.time()

    # choose a modest resolution that still gives a tight bound
    N = 720
    GLOBAL_N = N
    GLOBAL_X = (np.arange(N) + 0.5) * (2.0 / N)

    best_h = None
    best_c = np.inf

    # ------------------------------------------------------------------
    # Use parent construction if present
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != N:
            # interpolate to our grid
            old_x = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
            h0 = np.interp(GLOBAL_X, old_x, h0)
        h0 = enforce_sum(h0, N / 2.0, rng)
        c0 = compute_c5(h0)
        best_h, best_c = h0, c0

    # ------------------------------------------------------------------
    # Parametric families via differential evolution
    # ------------------------------------------------------------------
    # 1) Linear‑power family
    de_lin = differential_evolution(
        objective_linear,
        bounds=[(0.02, 0.98), (0.3, 8.0)],
        maxiter=80,
        popsize=15,
        seed=int(rng.integers(2 ** 32 - 1)),
        polish=False,
        disp=False,
    )
    a_lin, s_lin = de_lin.x
    h_lin = generate_h_linear(a_lin, s_lin)
    c_lin = compute_c5(h_lin)
    if c_lin < best_c:
        best_h, best_c = h_lin, c_lin

    # 2) Sine family
    de_sin = differential_evolution(
        objective_sine,
        bounds=[(0.02, 0.98)],
        maxiter=80,
        popsize=15,
        seed=int(rng.integers(2 ** 32 - 1)),
        polish=False,
        disp=False,
    )
    a_sin = de_sin.x[0]
    h_sin = generate_h_sine(a_sin)
    c_sin = compute_c5(h_sin)
    if c_sin < best_c:
        best_h, best_c = h_sin, c_sin

    # 3) Gaussian family
    de_gau = differential_evolution(
        objective_gaussian,
        bounds=[(0.05, 1.0)],
        maxiter=80,
        popsize=15,
        seed=int(rng.integers(2 ** 32 - 1)),
        polish=False,
        disp=False,
    )
    sigma_g = de_gau.x[0]
    h_gau = generate_h_gaussian(sigma_g)
    c_gau = compute_c5(h_gau)
    if c_gau < best_c:
        best_h, best_c = h_gau, c_gau

    # ------------------------------------------------------------------
    # Refine each promising candidate with local hill climbing
    # ------------------------------------------------------------------
    for cand_h in (h_lin, h_sin, h_gau):
        refined_h, refined_c = local_optimize(cand_h, rng, max_iters=3000, init_step=0.4)
        if refined_c < best_c:
            best_h, best_c = refined_h, refined_c

    # ------------------------------------------------------------------
    # Broad stochastic search: random binary starts + hill climbing
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.85:
        # random binary start
        start_vec = random_binary_vector(N, rng)
        h_opt, c_opt = local_optimize(start_vec, rng, max_iters=4000, init_step=0.6)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

        # perturb current best occasionally
        if best_h is not None and rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.35)
            h_opt, c_opt = local_optimize(start_vec, rng, max_iters=3000, init_step=0.3)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, N / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(N)
