def binary_local_search(x, rng, max_time, start_time):
    n = len(x)
    # compute correlation O
    corr = np.correlate(x, x, mode='full')
    O = corr[n-1:]  # lags 0..n-1
    cur_min = O[1:].min()
    cur_c5 = 1 - (2.0/n) * cur_min
    best_x, best_min = x.copy(), cur_min
    # Precompute indices of ones and zeros
    ones_idx = np.where(x == 1)[0]
    zeros_idx = np.where(x == 0)[0]

    while time.time() - start_time < max_time:
        # Sample a batch
        batch = 200
        best_impr = 0
        best_pair = None
        best_delta = None
        for _ in range(batch):
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            if i == j:
                continue
            # compute delta array
            delta = np.zeros(n, dtype=int)
            # delta for i removal
            delta[0] -= 1
            if i < n-1:
                # left contributions for l from 1 to n-1-i
                delta[1:n-i] -= x[i+1:]
            if i > 0:
                # right contributions
                # x[i-1], x[i-2], ..., x[0] correspond to l = 1..i
                delta[1:i+1] -= x[i-1::-1]
            # delta for j addition
            delta[0] += 1
            if j < n-1:
                delta[1:n-j] += x[j+1:]
            if j > 0:
                delta[1:j+1] += x[j-1::-1]

            # compute new overlaps O' = O + delta
            O_new = O + delta
            # compute new min (exclude lag 0)
            new_min = O_new[1:].min()
            if new_min > best_impr:
                best_impr = new_min
                best_pair = (i, j)
                best_delta = delta
        if best_pair is not None and best_impr > cur_min:
            # apply swap
            i, j = best_pair
            x[i] = 0
            x[j] = 1
            O += best_delta
            cur_min = best_impr
            cur_c5 = 1 - (2.0/n) * cur_min
            # update ones and zeros indices
            # could recompute arrays each time - small overhead.
            ones_idx = np.where(x == 1)[0]
            zeros_idx = np.where(x == 0)[0]
            # update best_x
            best_x = x.copy()
            best_min = cur_min
        else:
            # no improvement after batch, break or try random restart?
            break
    final_c5 = 1 - (2.0/n) * cur_min
    return best_x.copy(), float(final_c5)

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# basic helpers
# -------------------------------------------------
def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # final linear correction
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# -------------------------------------------------
# FFT based correlation / C5 computation
# -------------------------------------------------
def _next_pow2(x):
    return 1 << (x - 1).bit_length()


def compute_corr_fft(h):
    """Full correlation of h with (1-h) using FFT."""
    n = h.size
    N = _next_pow2(2 * n - 1)
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)[: 2 * n - 1]
    return corr


def compute_c5_fft(h):
    """Return (C5, full_corr) using FFT."""
    n = h.size
    corr = compute_corr_fft(h)
    c5 = np.max(corr) * (2.0 / n)
    return float(c5), corr


# -------------------------------------------------
# softmax surrogate for SLSQP
# -------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# -------------------------------------------------
def simulated_annealing_fft(h_start, rng, max_iters, time_end):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_fft(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.02, 1e-5
    for it in range(max_iters):
        if time.time() > time_end:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        # biased perturb towards worst lag
        lag = np.argmax(cur_corr) - (n - 1)
        if rng.random() < 0.6:
            new_h = perturb_h(h, rng, scale=0.4)
        else:
            new_h = perturb_h(h, rng, scale=0.4)
        new_c, new_corr = compute_c5_fft(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c


# -------------------------------------------------
# main run
# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    # initial size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5_fft(h0)[0]

    # -------------------------------------------------
    # phase 1: intensive SA + SLSQP on base grid
    # -------------------------------------------------
    main_budget = budget_s * 0.65
    while time.time() - t0 < main_budget:
        # restart choice
        if rng.random() < 0.75:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)
        # SA
        max_sa_iters = 30000
        sa_h, sa_c = simulated_annealing_fft(
            start_vec, rng, max_sa_iters, t0 + main_budget
        )
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c

        # SLSQP ladder – more aggressive β list
        beta_list = [10, 30, 100, 300, 1000, 3000, 8000, 15000]
        for beta in beta_list:
            if time.time() - t0 >= main_budget:
                break
            try:
                ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=400)
            except Exception:
                continue
            ref_c = compute_c5_fft(ref_h)[0]
            if ref_c < best_c:
                best_h, best_c = ref_h, ref_c

    # -------------------------------------------------
    # phase 2: optional up‑sampling if time permits
    # -------------------------------------------------
    remaining = budget_s - (time.time() - t0)
    if remaining > budget_s * 0.15:
        target_n = min(1440, max(720, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)
            # SA on finer grid
            max_sa_up = 20000
            up_h, up_c = simulated_annealing_fft(
                up_h, rng, max_sa_up, t0 + budget_s
            )
            # SLSQP refinement on finer grid
            for beta in beta_list:
                if time.time() - t0 >= budget_s:
                    break
                try:
                    up_h = slsqp_optimize_one_beta(up_h, beta, rng, max_iter=300)
                except Exception:
                    pass
                up_c = compute_c5_fft(up_h)[0]
                if up_c < best_c:
                    best_h, best_c, n = up_h.copy(), up_c, target_n

    # -------------------------------------------------
    # phase 3: final quick hill‑climb
    # -------------------------------------------------
    h_cur = best_h.copy()
    cur_c = best_c
    step = 0.25
    for _ in range(6000):
        if time.time() - t0 > budget_s:
            break
        new_h = perturb_h(h_cur, rng, scale=step)
        new_c = compute_c5_fft(new_h)[0]
        if new_c < cur_c:
            cur_c, h_cur = new_c, new_h
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        step *= 0.9997

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_fft(best_h)[0]

    return best_h.tolist(), float(final_c), int(best_h.size)
