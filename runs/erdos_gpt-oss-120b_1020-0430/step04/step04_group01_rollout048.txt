def project_sum_and_bounds(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need increase: indices where h <1
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            capacity = 1 - h[idx]
            total_cap = capacity.sum()
            if total_cap < diff - 1e-12:
                # cannot fill, set all to 1 and break
                h[idx] = 1.0
                diff = target - h.sum()
                break
            # distribute proportionally to capacity
            inc = diff * (capacity / total_cap)
            h[idx] += inc
        else:
            # need decrease: indices where h >0
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            capacity = h[idx]
            total_cap = capacity.sum()
            if total_cap < -diff - 1e-12:
                h[idx] = 0.0
                diff = target - h.sum()
                break
            dec = -diff * (capacity / total_cap)
            h[idx] -= dec
        diff = target - h.sum()
    # finally clip
    h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

# --------------------------------------------------------------
# FFT based convolution / correlation utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


def full_correlation(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear correlation of a with b (a * rev(b))."""
    n, m = a.size, b.size
    L = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b[::-1], L)
    conv = np.fft.irfft(fa * fb, L)[: n + m - 1]
    return conv


def full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b."""
    n, m = a.size, b.size
    L = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[: n + m - 1]
    return conv


# --------------------------------------------------------------
# Constraint handling
# --------------------------------------------------------------
def project_sum_and_bounds(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy Σh = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            cap = 1.0 - h[idx]
            tot = cap.sum()
            if tot < diff:
                h[idx] = 1.0
                diff = target - h.sum()
                continue
            inc = diff * (cap / tot)
            h[idx] += inc
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            cap = h[idx]
            tot = cap.sum()
            if tot < -diff:
                h[idx] = 0.0
                diff = target - h.sum()
                continue
            dec = -diff * (cap / tot)
            h[idx] -= dec
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in [0,1]^n with the required sum."""
    h = rng.random(n)
    return project_sum_and_bounds(h, n / 2.0)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount between two coordinates (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    out = h.copy()
    out[i] -= eps
    out[j] += eps
    return out


# --------------------------------------------------------------
# Objective evaluation
# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C₅ = max_k ∫ h(x)(1‑h(x+k))dx (dx = 2/n)."""
    n = h.size
    corr = full_correlation(h, 1.0 - h)
    return corr.max() * (2.0 / n)


# --------------------------------------------------------------
# Gradient step for maximizing min self‑correlation (i.e. minimizing C₅)
# --------------------------------------------------------------
def gradient_step(h: np.ndarray, beta: float, target: float, step_alpha: float) -> np.ndarray:
    """One ascent step on the smooth min‑corr surrogate."""
    n = h.size
    # self‑correlation
    corr_self = full_correlation(h, h)                 # length 2n‑1
    # stable soft‑max weights for min
    logw = -beta * corr_self
    logw -= np.max(logw)                               # avoid under/overflow
    w = np.exp(logw)
    w /= w.sum()
    w_rev = w[::-1]

    # gradient = conv(w,h) + conv(w_rev,h) on the central n entries
    conv1 = full_convolution(w, h)                     # length 3n‑2
    conv2 = full_convolution(w_rev, h)
    start = n - 1
    g = conv1[start:start + n] + conv2[start:start + n]

    # normalise step size (max absolute change = step_alpha)
    max_abs = np.max(np.abs(g))
    if max_abs > 0:
        g = g * (step_alpha / max_abs)

    h_new = h + g
    return project_sum_and_bounds(h_new, target)

# --------------------------------------------------------------

    """Run a multi‑β gradient ascent with occasional perturbations."""
    h = project_sum_and_bounds(h0.copy(), target)
    best_h, best_c = h.copy(), compute_c5(h)

    betas = [1.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
    rng = np.random.default_rng()

    for beta in betas:
        step_alpha = 0.4 / np.sqrt(beta)    # larger steps for small β
        for _ in range(2000):
            if time.time() >= deadline:
                return best_h, best_c
            h_new = gradient_step(h, beta, target, step_alpha)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
                h = h_new
            # optional tiny stochastic acceptance could be added here
        # small random shake before next β
        h = perturb_h(best_h, rng, scale=0.1)
        h = project_sum_and_bounds(h, target)

    # final polishing at the largest β
    beta = betas[-1]
    step_alpha = 0.1 / np.sqrt(beta)
    for _ in range(3000):
        if time.time() >= deadline:
            break
        h_new = gradient_step(h, beta, target, step_alpha)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
            h = h_new

    return best_h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    deadline = t_start + budget_s * 0.97   # small safety margin

    # ------------------------------------------------------------------
    # 1. Determine problem size and a few seeds
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        base = np.array(initial_h_values, dtype=float)
    else:
        n = 720
        base = None
    target = n / 2.0

    # ------------------------------------------------------------------
    # 2. Assemble starting candidates
    # ------------------------------------------------------------------
    candidates = []
    if base is not None:
        candidates.append(base)
    # random feasible vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))
    # simple binary half‑ones pattern
    binary = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
    candidates.append(binary)

    best_h, best_c = None, float('inf')
    for h in candidates:
        h = project_sum_and_bounds(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ------------------------------------------------------------------
    # 3. Main optimisation loop with restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.2)
        else:
            start = random_feasible_vector(n, rng)
        h_opt, c_opt = optimise_from_start(start, target, deadline)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ------------------------------------------------------------------
    # 4. Final high‑precision polishing
    # ------------------------------------------------------------------
    final_deadline = t_start + budget_s * 0.99
    h_opt, c_opt = optimise_from_start(best_h, target, final_deadline)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt.copy()

    # ------------------------------------------------------------------
    # 5. Return result
    # ------------------------------------------------------------------
    best_h = project_sum_and_bounds(np.clip(best_h, 0.0, 1.0), target)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
