def solve_mip(n, time_limit):
    import cvxpy as cp
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    constraints = []
    # sum h = n/2
    constraints.append(cp.sum(h) == n/2)
    # lower triangle zero
    for i in range(n):
        constraints.append(w[i, :i+1] == 0)  # Sets w[i,0..i] inclusive to 0
    # product constraints
    for i in range(n):
        hi = h[i]
        # For j>i
        # use vectorized maybe: constraints.extend([])
        # we can loop j
        for j in range(i+1, n):
            constraints.append(w[i, j] <= hi)
            constraints.append(w[i, j] <= h[j])
            constraints.append(w[i, j] >= hi + h[j] - 1)
    # O_k constraints
    for k in range(1, n):
        sum_h = cp.sum(h[: n - k])
        # diagonal of w[:n-k, k:]
        sum_w = cp.sum(cp.diag(w[: n - k, k:]))
        constraints.append(sum_h - sum_w <= t)
        # optional non-negative t? t >= 0
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_val = np.round(h.value).astype(float)  # round
        return h_val
    else:
        return None

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 2000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
        it += 1
    return x

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def solve_mip(n: int, time_limit: float):
    """Attempt to solve the binary min‑max problem via MIP (CBC)."""
    try:
        import cvxpy as cp
    except Exception:
        return None
    h = cp.Variable(n, boolean=True)
    w = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    cons = [cp.sum(h) == n / 2.0, t >= 0.0]
    # lower triangle zero
    for i in range(n):
        cons.append(w[i, : i + 1] == 0)
    # product linearisation
    for i in range(n):
        hi = h[i]
        for j in range(i + 1, n):
            cons.append(w[i, j] <= hi)
            cons.append(w[i, j] <= h[j])
            cons.append(w[i, j] >= hi + h[j] - 1)
    # overlap constraints
    for k in range(1, n):
        sum_h = cp.sum(h[: n - k])
        sum_w = cp.sum(cp.diag(w[: n - k, k:]))
        cons.append(sum_h - sum_w <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
        except Exception:
            return None
    if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        hv = np.rint(h.value).astype(float)  # round to 0/1
        hv = np.clip(hv, 0.0, 1.0)
        return hv
    return None

        # occasionally up‑sample
        n = best_n
        if rng.random() < 0.12 and n * 2 <= 384:
            n = n * 2
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = best_h.copy()
        # hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=1400, step_scale=0.7)
        # soft‑max SLSQP passes
        for beta in (20., 80., 200., 800., 3000., 8000.):
            if time.time() - start > budget_s * 0.90:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=600)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n
        # tiny gradient tidy‑up
        h = gradient_refine(h, rng, step=0.015, max_iter=8)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # --- final stochastic tweaks ------------------------------------
    while time.time() - start < budget_s * 0.998 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
