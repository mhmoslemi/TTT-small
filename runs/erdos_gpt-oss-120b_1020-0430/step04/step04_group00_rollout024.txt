def binary_sa(h, rng, max_iters, init_temp, final_temp):

    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    for it in range(max_iters):
        T = init_temp * ((final_temp / init_temp) ** (it / max_iters))  # exponential decay
        # pick swap i from ones, j from zeros
        i = rng.choice(ones) # arrays of indices tracked
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        # recompute c
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / T)
        if accept:
            cur_c = new_c
            # update indices
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        # optionally early stop if no improvement for many steps
    return best_h, best_c

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    # random redistribution
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:
            # no capacity, try another index
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp safely
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # fix sum exactly
    adjust_sum_exact(x, rng)
    return x

import time
import numpy as np
import scipy.optimize as opt

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-15:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def subgradient(h, k):
    n = h.size
    idx = np.arange(n)
    term1 = np.ones(n)
    idx1 = idx + k
    mask1 = (idx1 >= 0) & (idx1 < n)
    term1[mask1] = 1.0 - h[idx1[mask1]]
    term2 = np.zeros(n)
    idx2 = idx - k
    mask2 = (idx2 >= 0) & (idx2 < n)
    term2[mask2] = -h[idx2[mask2]]
    return term1 + term2


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.985

    # ----- choose resolution and initial vector -----
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n_par = parent.size
        n = max(720, n_par * 2)
        if n > 1440:
            n = 1440
        xs = np.linspace(0.0, 2.0, n_par, endpoint=False)
        xt = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xt, xs, parent)
        h = np.clip(h, 0.0, 1.0)
    else:
        n = 720
        h = rng.random(n)
        h = h * (n / 2.0) / h.sum()
        h = np.clip(h, 0.0, 1.0)

    adjust_sum_exact(h, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- subgradient descent on the worst shift -----
    step0 = 0.05
    decay = 0.9999
    it = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        cur_c = float(max_val * (2.0 / n))
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        max_idxs = np.where(corr == max_val)[0]
        k = int(max_idxs[rng.integers(0, max_idxs.size)] - (n - 1))
        g = subgradient(h, k)
        step = step0 * (decay ** it)
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        adjust_sum_exact(h, rng)

        # occasional tiny random jitter to escape plateaus
        if rng.random() < 0.001:
            i = rng.integers(0, n)
            delta = (rng.random() - 0.5) * 0.02
            h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            adjust_sum_exact(h, rng)

        it += 1
        if step < 1e-7:
            # step negligible – continue for a while but don't break prematurely
            pass

    # ----- final polishing with smooth‑max SLSQP -----
    remaining = deadline - time.time()
    if remaining > 10.0:
        for beta in (10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0, 10000.0):

                best_h = h.copy()

    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
