                best_h, best_c = ref_h, ref_c

    # Phase 4: final small‑scale hill climb
    remaining = budget_s - (time.time() - t0)
    if remaining > 2.0:
        step = 0.4
        cur_h, cur_c = best_h.copy(), best_c
        it = 0
        while time.time() - t0 < budget_s and it < 5000:
            it += 1
            new_h = perturb_h(cur_h, rng, scale=step)
            new_h = enforce_sum(np.clip(new_h, 0, 1), n / 2.0, rng)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                cur_h, cur_c = new_h, new_c
                if new_c < best_c:
                    best_h, best_c = new_h.copy(), new_c
            step *= 0.9997

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
