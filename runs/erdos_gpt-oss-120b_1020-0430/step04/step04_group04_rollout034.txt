import time, numpy as np, scipy.optimize as opt

# ---------- global placeholders populated in run ----------
_B = None          # shape (n, p) basis matrix
_B_sum = None      # shape (p,) sum of rows of _B
_n = None          # number of sample points
_CURRENT_BETA = None  # for softmax surrogate


# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 5000) -> np.ndarray:
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


# ---------- constraint helpers (used by SLSQP) ----------
def _eq_sum(a: np.ndarray) -> float:
    return float(np.dot(_B_sum, a) - _n / 2.0)


def _ineq_lower(a: np.ndarray, i: int) -> float:
    return float(np.dot(_B[i], a))


def _ineq_upper(a: np.ndarray, i: int) -> float:
    return float(1.0 - np.dot(_B[i], a))


# ---------- objective wrappers ----------
def _c5_from_a(a: np.ndarray) -> float:
    h = np.dot(_B, a)
    h = np.clip(h, 0.0, 1.0)
    return compute_c5(h)


def _softmax_obj_a(a: np.ndarray) -> float:
    h = np.clip(np.dot(_B, a), 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = _CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / _CURRENT_BETA


def _project_a_sum(a: np.ndarray) -> np.ndarray:
    """Shift a along the direction of _B_sum to satisfy the exact integral constraint."""
    delta = (_n / 2.0) - np.dot(_B_sum, a)
    if abs(delta) > 1e-12:
        factor = delta / np.dot(_B_sum, _B_sum)
        a = a + factor * _B_sum
    return a


# ---------- local refinement on full vector ----------
def _perturb_h_full(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def _hill_climb_h(start_h: np.ndarray, rng: np.random.Generator,
                  max_iters: int = 2000, init_step: float = 0.5) -> tuple:

    # ---- initial guess ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        old_x = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
        h0 = np.interp(x, old_x, h0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        h0 = np.full(n, 0.5)
    a0 = np.linalg.lstsq(_B, h0, rcond=None)[0]
    a0 = _project_a_sum(a0)

    best_a = a0.copy()
    best_h = np.clip(_B.dot(best_a), 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    best_c = compute_c5(best_h)

    # ---- random restarts with cheap surrogate ----
    beta_seq = (50, 200, 800, 2000, 5000)
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.4:
            break
        a_rand = rng.normal(scale=0.1, size=p)
        a_rand[0] = 0.5
        a_rand = _project_a_sum(a_rand)
        for beta in beta_seq[:3]:  # coarse early stage
            global _CURRENT_BETA
            _CURRENT_BETA = beta
            try:
                res = opt.minimize(_softmax_obj_a, a_rand, method="SLSQP",
                                   constraints=cons,
                                   options={"maxiter": 300, "ftol": 1e-9, "disp": False})
                a_rand = res.x
            except Exception:
                pass
        h_tmp = np.clip(_B.dot(a_rand), 0.0, 1.0)
        h_tmp = enforce_sum(h_tmp, n / 2.0, rng)
        c_tmp = compute_c5(h_tmp)
        if c_tmp < best_c:
            best_c, best_a, best_h = c_tmp, a_rand.copy(), h_tmp.copy()

    # ---- refined surrogate ladder on current best ----
    for beta in beta_seq:
        if time.time() - start_time > budget_s * 0.7:
            break
        _CURRENT_BETA = beta
        try:
            res = opt.minimize(_softmax_obj_a, best_a, method="SLSQP",
                               constraints=cons,
                               options={"maxiter": 800, "ftol": 1e-12, "disp": False})
            best_a = res.x
            h_tmp = np.clip(_B.dot(best_a), 0.0, 1.0)
            h_tmp = enforce_sum(h_tmp, n / 2.0, rng)
            c_tmp = compute_c5(h_tmp)
            if c_tmp < best_c:
                best_c, best_h = c_tmp, h_tmp.copy()
        except Exception:
            continue

    # ---- final exact SLSQP refinement ----
    if time.time() - start_time < budget_s * 0.85:
        try:
            res = opt.minimize(_c5_from_a, best_a, method="SLSQP",
                               constraints=cons,
                               options={"maxiter": 2000, "ftol": 1e-12, "disp": False})
            best_a = res.x
            best_h = np.clip(_B.dot(best_a), 0.0, 1.0)
            best_h = enforce_sum(best_h, n / 2.0, rng)
            best_c = compute_c5(best_h)
        except Exception:
            pass

    # ---- optional local hill‑climb on full vector ----
    if time.time() - start_time < budget_s * 0.95:
        hc, cc = _hill_climb_h(best_h, rng, max_iters=2500, init_step=0.4)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
