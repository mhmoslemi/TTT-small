import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Projection onto [0,1] with exact sum constraint
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

# Objective evaluation
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx

def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)

# Local perturbations (preserve sum)
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    """Move mass respecting the sign of k (only positive k are used)."""
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h

# Hill climbing that concentrates on the worst shift
def hill_climb_targeted(h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)  # signed shift
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

# Simulated annealing that swaps mass between two sites
def anneal_improve(h_start, rng, steps=20000, T0=5e-4, Tf=1e-6):
    n = h_start.size
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    for s in range(steps):
        # temperature schedule (exponential)
        T = T0 * (Tf / T0) ** (s / steps)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # decide direction of mass transfer
        if rng.random() < 0.5:
            max_eps = min(1.0 - cur_h[i], cur_h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * 0.5
            new_h = cur_h.copy()
            new_h[i] += eps
            new_h[j] -= eps
        else:
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.random() * max_eps * 0.5
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps

        new_c = compute_c5(new_h)
        d = new_c - cur_c
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

# SLSQP refinement with softmax surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

    xs = (np.arange(n) + 0.5) * (2.0 / n)

    for f in range(1, 7):
        cand = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        cand = project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0)
        candidates.append(cand)
        cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        cand = project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0)
        candidates.append(cand)

    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(project_box_sum(mask, n / 2.0))

    mask = np.zeros(n)
    start = n // 4
    mask[start : start + n // 2] = 1.0
    candidates.append(project_box_sum(mask, n / 2.0))

    for _ in range(6):
        candidates.append(random_feasible_vector(n, rng))

    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # -------------------------------------------------
    # Main optimisation loop on current resolution
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]
    max_hc_iters = 1200
    while time.time() - start_time < 0.70 * time_limit:
        # choose start point
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill climbing
        h_hc, c_hc = hill_climb_targeted(
            start_vec, rng, max_iters=max_hc_iters, init_step=0.4
        )

        # annealing refinement
        h_ann, c_ann = anneal_improve(h_hc, rng, steps=12000, T0=5e-4, Tf=1e-6)

        # pick the best of the three
        cand_h, cand_c = h_ann, c_ann
        if c_hc < cand_c:
            cand_h, cand_c = h_hc, c_hc
        if c_hc < best_c:
            cand_h, cand_c = h_hc, c_hc
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()
            # SLSQP refinement with increasing β
            for beta in beta_list:
                if time.time() - start_time > 0.70 * time_limit:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=400)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # -------------------------------------------------
    # Upsample if time permits
    elapsed = time.time() - start_time
    remaining = time_limit - elapsed
    if remaining > 0.20 * time_limit:
        target_n = min(1080, max(720, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_box_sum(h_up, target_n / 2.0)

            # intensive hill climbing + annealing on the larger grid
            h_up, c_up = hill_climb_targeted(
                h_up, rng, max_iters=3000, init_step=0.3
            )
            h_up, c_up = anneal_improve(h_up, rng, steps=30000, T0=5e-4, Tf=1e-6)

            if c_up < best_c:
                best_c, best_h, n = c_up, h_up.copy(), target_n

            # final SLSQP sweep on the enlarged grid
            for beta in beta_list:
                if time.time() - start_time > time_limit:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=400)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # -------------------------------------------------
    # Final cleanup and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
