from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

# adjustment functions as before

def adjust_sum_exact(h, rng):
    n = h.size
    target = n/2.0
    diff = target - h.sum()
    for _ in range(20000):
        if abs(diff) < 1e-10:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# gradient refinement
def gradient_refine(h, rng, step=0.04, max_iter=30):
    n = h.size
    target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        # step size decay
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # fix sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution using rng
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def guided_binary_sa(h_bin, rng, time_deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    it = 0
    last_improve = 0
    # precompute gradient placeholder
    g = np.zeros(n)
    while time.time() < time_deadline:
        # recompute gradient every 40 iterations
        if it % 40 == 0:
            corr = np.correlate(cur_h, 1 - cur_h, mode='full')
            idx = np.argmax(corr) - (n - 1)
            k = abs(idx)
            if k == 0:
                g[:] = 1.0 - 2.0 * cur_h
            else:
                upper = n - k
                g[:upper] = 1.0 - cur_h[k:]
                g[k:] = -cur_h[:upper]
        # choose promising swap
        pos_ones = [i for i in ones if g[i] > 0]
        neg_zeros = [j for j in zeros if g[j] < 0]
        if pos_ones and neg_zeros:
            i = rng.choice(pos_ones)
            j = rng.choice(neg_zeros)
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        # temperature schedule
        T = max(1e-7, 0.01 * cur_c * (0.9999 ** it))
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 30000:

    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(20000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.04, max_iter=30):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def guided_binary_sa(h_bin, rng, time_deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    it = 0
    last_improve = 0
    g = np.zeros(n)
    while time.time() < time_deadline:
        if it % 40 == 0:
            corr = np.correlate(cur_h, 1 - cur_h, mode="full")
            idx = np.argmax(corr) - (n - 1)
            k = abs(idx)
            if k == 0:
                g[:] = 1.0 - 2.0 * cur_h
            else:
                upper = n - k
                g[:upper] = 1.0 - cur_h[k:]
                g[k:] = -cur_h[:upper]
        pos_ones = [i for i in ones if g[i] > 0]
        neg_zeros = [j for j in zeros if g[j] < 0]
        if pos_ones and neg_zeros:
            i = rng.choice(pos_ones)
            j = rng.choice(neg_zeros)
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = max(1e-7, 0.01 * cur_c * (0.9999 ** it))
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 30000:
            break
    return best_h, best_c

            bin_vec[idx] = 1.0
            start_vectors.append(bin_vec)

        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vectors.append(bin_rand)

        rand_frac = rng.random(n)
        target = n / 2.0
        rand_frac *= target / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)
        start_vectors.append(rand_frac)

        for base_h in start_vectors:
            if time.time() > deadline:
                break
            h = np.clip(base_h, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            if np.all((h == 0) | (h == 1)):
                sub_dead = min(deadline,
                               time.time() + min(60.0, (deadline - time.time()) * 0.5))
                h_bin, c_bin = guided_binary_sa(h.astype(int), rng, sub_dead)
                h = h_bin.astype(float)
                c = c_bin
            else:
                c = compute_c5(h)

            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            if time.time() < deadline:
                h_ref = gradient_refine(h.copy(), rng, step=0.04, max_iter=30)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c, best_n = h_ref.copy(), c_ref, n
                for beta in betas:
                    if time.time() > deadline:
                        break
                    max_iter = 800 if beta < 5000 else 1200
                    try:
                        h_ref = slsqp_optimize(h_ref, beta, rng, max_iter=max_iter)
                    except Exception:
                        continue
                    c_ref = compute_c5(h_ref)
                    if c_ref < best_c:
                        best_h, best_c, best_n = h_ref.copy(), c_ref, n
                if time.time() < deadline:
                    h_ref = gradient_refine(best_h.copy(), rng,
                                            step=0.04, max_iter=30)
                    c_ref = compute_c5(h_ref)
                    if c_ref < best_c:
                        best_h, best_c, best_n = h_ref.copy(), c_ref, n

        # extra random binary restarts for this n
        while time.time() < deadline - 30:
            bin_rand = np.zeros(n)
            bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            sub_dead = min(deadline, time.time() + 30.0)
            h_bin, c_bin = guided_binary_sa(bin_rand.astype(int), rng, sub_dead)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin.astype(float), c_bin, n
                # quick refinement
                h_ref = gradient_refine(best_h.copy(), rng, step=0.04, max_iter=20)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c, best_n = h_ref.copy(), c_ref, n

    # up‑sample if possible
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        idx = np.argpartition(h_up, -n_up // 2)[-n_up // 2:]
        bin_up = np.zeros(n_up)
        bin_up[idx] = 1.0
        sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
        h_bin_up, c_bin_up = guided_binary_sa(bin_up.astype(int), rng, sub_dead)
        if c_bin_up < best_c:
            best_h, best_c, best_n = h_bin_up.astype(float), c_bin_up, n_up
        h = h_bin_up.astype(float)
        h = gradient_refine(h, rng, step=0.04, max_iter=30)
        for beta in betas:
            if time.time() > deadline:
                break
            max_iter = 800 if beta < 5000 else 1200
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
        h = gradient_refine(h, rng, step=0.04, max_iter=30)
        c_up = compute_c5(h)
        if c_up < best_c:
            best_h, best_c, best_n = h.copy(), c_up, n_up

    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
