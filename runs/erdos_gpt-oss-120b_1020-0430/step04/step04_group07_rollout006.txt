def pairwise_local_search(h, O, rng, trials=500, max_eps_factor=0.5):
    # returns new h and new O after best improvement.
    N = h.size
    best_h = h
    best_O = O
    best_f = best_O.max()
    for _ in range(trials):
        i = rng.integers(N)
        j = rng.integers(N)
        if i == j: continue
        # determine direction: we can move from i to j or j to i.
        # We'll treat move from i (decrease) to j (increase).
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            continue
        # a_i for each shift
        # using np.take for performance.
        idx_i = (np.arange(N) + i) % N
        idx_j = (np.arange(N) + j) % N
        a = h[idx_i] - h[idx_j]
        # Evaluate f at 0 and max_eps quickly.
        # Use ternary search to find min.
        lo = 0.0
        hi = max_eps
        # optionally early skip if a all >=0 -> f increases with eps so no improvement.
        # Actually if all a >=0, f(eps) is non-decreasing, min at eps=0, so skip.
        if np.all(a >= 0):
            continue
        # If all a <=0 => f decreasing, min at eps = max_eps.
        if np.all(a <= 0):
            eps_opt = max_eps
        else:
            for _ in range(25):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(O + m1 * a)
                f2 = np.max(O + m2 * a)
                if f1 > f2:
                    lo = m1
                else:
                    hi = m2
            eps_opt = (lo + hi) / 2.0
        f_opt = np.max(O + eps_opt * a)
        if f_opt < best_f - 1e-12:
            # apply move
            h_new = best_h.copy()
            h_new[i] -= eps_opt
            h_new[j] += eps_opt
            # recompute O efficiently? Could compute O after update: O_new = O + eps_opt*a.
            O_new = O + eps_opt * a
            best_f = f_opt
            best_h = h_new
            best_O = O_new
    return best_h, best_O

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    N = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / N)


def compute_overlap_array(h: np.ndarray) -> np.ndarray:
    """Array O_k = ∑ h_i (1‑h_{i+k}) for k=0..N‑1."""
    N = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return corr[N - 1 :] * (2.0 / N)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    N = h0.size
    CURRENT_TARGET = N / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * N
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, N)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def random_feasible_vector(N: int, rng: np.random.Generator) -> np.ndarray:
    """Random h ∈ [0,1]^N with sum = N/2."""
    h = rng.random(N)
    target = N / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, N)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    N = h.size
    target = N / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, N)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    N = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (N - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: N - k] -= h[k:]
        grad[k:] -= h[: N - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1500, step_scale: float = 0.7):
    """Hill‑climb using gradient‑directed transfers."""
    N = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 8) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    N = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (N - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: N - k] += 1.0 - h[k:]
            g[k:] -= h[: N - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = N / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, N)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def pairwise_local_search(h: np.ndarray, rng: np.random.Generator,
                          max_iters: int = 5, trials_per_iter: int = 200):
    """Exact ε‑line‑search on random index pairs."""
    N = h.size
    O = compute_overlap_array(h)
    best_val = O.max()
    for _ in range(max_iters):
        best_i = best_j = -1
        best_eps = 0.0
        best_a = None
        for _ in range(trials_per_iter):
            i = rng.integers(N)
            j = rng.integers(N)
            if i == j:
                continue
            # direction i→j
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 1e-12:
                a = h[(np.arange(N) + i) % N] - h[(np.arange(N) + j) % N]
                if not np.all(a >= 0):
                    eps_opt = _optimal_eps(a, O, max_eps)
                    val = np.max(O + eps_opt * a)
                    if val < best_val - 1e-12:
                        best_val, best_i, best_j, best_eps, best_a = val, i, j, eps_opt, a
            # direction j→i
            max_eps2 = min(h[j], 1.0 - h[i])
            if max_eps2 > 1e-12:
                a2 = -a  # opposite direction
                if not np.all(a2 >= 0):
                    eps_opt2 = _optimal_eps(a2, O, max_eps2)
                    val2 = np.max(O + eps_opt2 * a2)
                    if val2 < best_val - 1e-12:
                        best_val, best_i, best_j, best_eps, best_a = val2, j, i, eps_opt2, a2
        if best_i == -1:
            break
        # apply best move
        h[best_i] -= best_eps
        h[best_j] += best_eps
        O += best_eps * best_a
    return h, O


def _optimal_eps(a, O, eps_max):
    """Ternary search for ε∈[0,eps_max] minimizing max(O+ε a)."""
    if np.all(a <= 0):
        return eps_max
    lo, hi = 0.0, eps_max
    for _ in range(15):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(O + m1 * a)
        f2 = np.max(O + m2 * a)
        if f1 > f2:
            lo = m1
        else:
            hi = m2
    return (lo + hi) / 2.0

        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_n = h0, h0.size
        best_c = compute_c5(best_h)

    n_candidates = [96, 120, 144, 168, 192, 216, 240, 256, 288]

    # --- main multistart search --------------------------------------
    while time.time() - start < 0.74 * budget_s:
        N = int(rng.choice(n_candidates))
        fresh = rng.random() < 0.12
        if not fresh and best_h is not None:
            if best_n == N:
                h = best_h.copy()
            else:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, N, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = enforce_sum(h, rng)
        else:
            if rng.random() < 0.45:
                h = np.zeros(N)
                h[rng.choice(N, N // 2, replace=False)] = 1.0
            else:
                h = random_feasible_vector(N, rng)

        # quick hill‑climb
        h, _ = guided_hill_climb(h, rng, max_iters=1300, step_scale=0.8)

        # SLSQP soft‑max refinement with increasing β
        for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
            if time.time() - start > 0.91 * budget_s:
                break
            iters = 1000 if beta >= 200 else 600
            h = slsqp_optimize(h, beta, rng, max_iter=iters)

        # pairwise exact line‑search refinement
        h, O = pairwise_local_search(h, rng, max_iters=4,
                                     trials_per_iter=min(300, N))
        c = np.max(O) * (2.0 / N)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, N

        # final gradient tidy‑up and another hill‑climb
        h = gradient_refine(h, rng, step=0.015, max_iter=6)
        h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, N

    # --- up‑sampling refinement ---------------------------------------
    if best_h is not None and time.time() - start < 0.96 * budget_s:
        for factor in (2, 3):
            Nup = best_n * factor
            if Nup > 384:
                continue
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, Nup, endpoint=False)
            hup = np.interp(x_new, x_old, best_h)
            hup = enforce_sum(hup, rng)

            hup, _ = guided_hill_climb(hup, rng, max_iters=1800, step_scale=0.7)

            for beta in (20., 200., 2000., 8000., 20000.):
                if time.time() - start > budget_s:
                    break
                hup = slsqp_optimize(hup, beta, rng, max_iter=800)

            hup, Oup = pairwise_local_search(hup, rng, max_iters=4,
                                             trials_per_iter=min(300, Nup))
            cup = np.max(Oup) * (2.0 / Nup)
            if cup < best_c:
                best_h, best_c, best_n = hup.copy(), cup, Nup

    # --- final intensive local search until time runs out -------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        # occasional guided perturbation
        if rng.random() < 0.25:
            grad = worst_gradient(best_h)
            nh = guided_perturb(best_h, grad, rng, scale=0.02)
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
                continue
        # pairwise exact refinement (single iteration)
        best_h, O = pairwise_local_search(best_h, rng, max_iters=1,
                                          trials_per_iter=min(400, best_h.size))
        best_c = np.max(O) * (2.0 / best_h.size)

    # sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
