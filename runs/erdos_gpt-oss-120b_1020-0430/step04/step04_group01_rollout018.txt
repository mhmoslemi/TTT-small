def binary_sa(start_bin, rng, deadline, max_iters=200000):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    # Temperature schedule: start higher to allow uphill moves
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    # we will compute fraction for temperature: it/ max_iters
    for it in range(max_iters):
        if time.time() >= deadline:
             break
        # choose random i with 1, and random j with 0
        i = rng.integers(0, n)
        while h[i] != 1.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 0.0:
            j = rng.integers(0, n)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        # temperature
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # early break if best_c <= target
        if best_c <= 0.3808:
            break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Utility functions
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


# --------------------------------------------------------------
# Correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Smooth surrogate for SLSQP (log‑sum‑exp)
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """beta‑scaled log‑sum‑exp of the overlap vector."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    # stable formulation
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=350):
    """SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------------------------------------------
# Binary simulated annealing (swap based)
# --------------------------------------------------------------
def binary_sa(start_bin, rng, deadline, max_iters=200000):
    """Swap‑based simulated annealing on a binary vector (sum = n/2)."""
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue          # trivial swap

        # enforce i is 1, j is 0
        if h[i] < h[j]:
            i, j = j, i

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

        if best_c <= 0.3808:
            break

    return best_h, best_c


# --------------------------------------------------------------
# Targeted binary improvement using worst shift
# --------------------------------------------------------------
def targeted_binary_improve(h, rng, deadline, max_iters=5000):
    """Swap that directly reduces the maximal overlap shift."""
    n = h.size
    h = h.astype(float).copy()
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = correlate_full(h)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        k = max_idx - (n - 1)

        if k == 0:
            continue

        # neighbor sums for this shift
        sum_nb = np.zeros(n, dtype=int)
        if k > 0:
            sum_nb[: n - k] += h[k:].astype(int)
            sum_nb[k:] += h[: n - k].astype(int)
        else:
            kk = -k
            sum_nb[kk:] += h[: n - kk].astype(int)
            sum_nb[: n - kk] += h[kk:].astype(int)

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[np.argmin(sum_nb[ones])]
        j = zeros[np.argmax(sum_nb[zeros])]

        if sum_nb[i] >= sum_nb[j]:
            # no guaranteed improvement – make a random swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while h[i] == h[j]:
                i = rng.integers(0, n)
                j = rng.integers(0, n)

        # attempt swap
        h[i], h[j] = 0.0, 1.0
        new_max = correlate_full(h).max()
        if new_max < max_val:
            # keep improvement
            continue
        else:
            # revert
            h[i], h[j] = 1.0, 0.0

    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s
    deadline = start_time + budget

    # ---- Determine size and initial binary vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 720
        h0 = np.full(n, 0.5)

    # binary start: top half entries set to 1
    idx = np.argsort(-h0)[: n // 2]
    best_h = np.zeros(n, dtype=float)
    best_h[idx] = 1.0
    best_c = compute_c5(best_h)

    # ---- Quick targeted improvement (first 10% of budget) ----
    quick_deadline = start_time + 0.10 * budget
    if time.time() < quick_deadline:
        cand = targeted_binary_improve(best_h.copy(), rng, quick_deadline, max_iters=2000)
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_h, best_c = cand, cand_c

    # ---- Main binary simulated annealing (≈70% of budget) ----
    sa_deadline = start_time + 0.80 * budget
    while time.time() < sa_deadline:
        # run a SA segment; allocate remaining time as its deadline
        seg_deadline = min(sa_deadline, time.time() + 30.0)  # at most 30 s per segment
        h_candidate, c_candidate = binary_sa(best_h.copy(), rng, seg_deadline, max_iters=200000)
        if c_candidate < best_c:
            best_h, best_c = h_candidate, c_candidate
            if best_c <= 0.3808:
                break

        # occasional random binary restart
        if rng.random() < 0.15:
            rnd = rng.random(n)
            idx = np.argsort(-rnd)[: n // 2]
            h_rand = np.zeros(n, dtype=float)
            h_rand[idx] = 1.0
            c_rand = compute_c5(h_rand)
            if c_rand < best_c:
                best_h, best_c = h_rand, c_rand

    # ---- Fine‑grained targeted swaps (remaining time) ----
    fine_deadline = deadline - 5.0  # keep a few seconds for final polishing
    if time.time() < fine_deadline:
        best_h = targeted_binary_improve(best_h.copy(), rng, fine_deadline, max_iters=5000)
        best_c = compute_c5(best_h)

    # ---- Continuous polishing with SLSQP (high β) ----
    polish_deadline = deadline - 2.0
    beta_list = [20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
    while time.time() < polish_deadline:
        for beta in beta_list:
            if time.time() >= polish_deadline:
                break
            try:
                h_opt = slsqp_optimize(best_h.copy(), beta, max_iter=400)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt, c_opt
            except Exception:
                continue
        # if no improvement in a whole sweep, break
        break

    # ---- Final cleanup ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
