def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]
    return conv
```

Now gradient:

```
def gradient_softmax(h, beta):
    n = h.shape[0]
    conv = correlate_full(h)   # size 2n-1
    # compute softmax weights w
    bc = beta * conv
    m = bc.max()
    w = np.exp(bc - m)
    w /= w.sum()
    # conv of w with (1-h) and h
    L = 1 << ((3*n - 2).bit_length())
    fw = np.fft.rfft(w, L)
    f1mh = np.fft.rfft(1 - h, L)
    fhm = np.fft.rfft(h, L)
    conv_w_1mh = np.fft.irfft(fw * f1mh, L)
    conv_w_h = np.fft.irfft(fw * fhm, L)
    # extract G1 and G2 at indices n-1 : n-1+n
    offset = n - 1
    G1 = conv_w_1mh[offset:offset + n]
    G2 = conv_w_h[offset:offset + n]
    grad = G1 - G2
    return grad, conv
```

def project_onto_constraints(v, target, rng, max_iter=100):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # compute indices where we can adjust
        if diff > 0:
            # need to increase some entries not at 1
            free = (v < 1.0)
            if not free.any():
                break
            add = diff / free.sum()
            # limit to not exceed 1
            v[free] += add
            v = np.clip(v, 0.0, 1.0)
        else:
            free = (v > 0.0)
            if not free.any():
                break
            sub = diff / free.sum()  # diff negative, so sub negative
            v[free] += sub
            v = np.clip(v, 0.0, 1.0)
    return v
```

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i == j:
        return h.copy()
    # decide direction to transfer: from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        i, j = j, i
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def gradient_softmax(h, beta):
    n = h.shape[0]
    conv = correlate_full(h)  # size 2n-1
    bc = beta * conv
    max_bc = bc.max()
    w = np.exp(bc - max_bc)
    w /= w.sum()
    # reverse w
    w_rev = w[::-1]
    L = 1 << ((3 * n - 2).bit_length())
    fw = np.fft.rfft(w_rev, L)
    f1mh = np.fft.rfft(1 - h, L)
    fh = np.fft.rfft(h, L)
    conv_w_1mh = np.fft.irfft(fw * f1mh, L)
    conv_w_h = np.fft.irfft(fw * fh, L)
    offset = n - 1
    G1 = conv_w_1mh[offset:offset + n]
    G2 = conv_w_h[offset:offset + n]
    grad = G1 - G2
    return grad, conv
```

def project_onto_constraints(v, target, rng, max_iter=100):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (v < 1.0)
        else:
            free = (v > 0.0)
        if not free.any():
            break
        delta = diff / free.sum()
        # apply delta limited to bounds
        if diff > 0:
            add = np.minimum(delta, 1.0 - v[free])
            v[free] += add
        else:
            sub = np.maximum(delta, -v[free])
            v[free] += sub
    v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    return v
```

But note: `delta` may be negative if diff negative. For diff<0, we compute sub = np.maximum(delta, -v[free]), which ensures we don't subtract more than v (i.e., not below 0). Then add sub (negative) to v[free].

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    # decide direction: transfer from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        i, j = j, i
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

import time
import numpy as np

# --------------------------------------------------------------
# Feasibility utilities
# --------------------------------------------------------------
def project_onto_constraints(v, target, rng, max_iter=100):
    """Clip to [0,1] and adjust sum to target by water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (v < 1.0)
        else:
            free = (v > 0.0)
        if not free.any():
            break
        delta = diff / free.sum()
        if diff > 0:
            add = np.minimum(delta, 1.0 - v[free])
            v[free] += add
        else:
            sub = np.maximum(delta, -v[free])
            v[free] += sub
    v = np.clip(v, 0.0, 1.0)
    # tiny uniform correction for rounding errors
    s = v.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    return v


def random_feasible_vector(n, rng):
    """Uniform random point in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    v = project_onto_constraints(v, n / 2.0, rng)
    return v


def perturb_h(h, rng, scale=0.2):
    """Transfer ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    # try i→j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0.0:
            return h.copy()
        i, j = j, i
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Correlation utilities
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def gradient_softmax(h, beta):
    """
    Gradient of the log‑sum‑exp smoothed max of the overlap
    with respect to h. Returns (grad, conv) where conv is the raw
    correlation vector (length 2n‑1).
    """
    n = h.shape[0]
    conv = correlate_full(h)                     # length 2n‑1
    bc = beta * conv
    max_bc = bc.max()
    w = np.exp(bc - max_bc)
    w /= w.sum()                                 # soft‑max weights, length 2n‑1

    # convolution of reversed weight vector with (1‑h) and h
    w_rev = w[::-1]
    L = 1 << ((3 * n - 2).bit_length())
    fw = np.fft.rfft(w_rev, L)
    f1mh = np.fft.rfft(1 - h, L)
    fh = np.fft.rfft(h, L)

    conv_w_1mh = np.fft.irfft(fw * f1mh, L)      # length >= 3n‑2
    conv_w_h = np.fft.irfft(fw * fh, L)

    offset = n - 1
    G1 = conv_w_1mh[offset : offset + n]        # Σ w_k (1‑h[p+k])
    G2 = conv_w_h[offset : offset + n]          # Σ w_k h[p+k]
    grad = G1 - G2
    return grad, conv


# --------------------------------------------------------------
# Local hill‑climb (discrete C₅)
# --------------------------------------------------------------
def hill_climb_local(h_start, rng, max_iters, deadline):
    """Greedy hill‑climb on the true C₅ (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    target = n / 2.0
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        cand = perturb_h(h, rng, scale=0.3)
        cand = project_onto_constraints(cand, target, rng)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()
            h = cand
    return best_h, best_c

# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 720
        h = random_feasible_vector(n, rng)
    target = n / 2.0
    h = project_onto_constraints(h, target, rng)
    # small nudge to leave possible local basin
    h = perturb_h(h, rng, scale=0.02)
    h = project_onto_constraints(h, target, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Adam state for gradient descent on smoothed objective
    # ------------------------------------------------------------------
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    beta_schedule = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]

    for beta in beta_schedule:
        # initial smoothed value
        conv = correlate_full(h)
        bc = beta * conv
        max_bc = bc.max()
        obj = (np.log(np.exp(bc - max_bc).sum()) + max_bc) / beta

        base_lr = 0.5 / np.sqrt(beta)  # heuristic scaling
        beta1_pow = 1.0
        beta2_pow = 1.0

        while time.time() < deadline:
            t += 1
            beta1_pow *= beta1
            beta2_pow *= beta2

            grad, conv = gradient_softmax(h, beta)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1_pow)
            v_hat = v / (1 - beta2_pow)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            # projected step
            h_new = h - step
            h_new = project_onto_constraints(h_new, target, rng)

            # new smoothed objective
            conv_new = correlate_full(h_new)
            bc_new = beta * conv_new
            max_bc_new = bc_new.max()
            obj_new = (np.log(np.exp(bc_new - max_bc_new).sum()) + max_bc_new) / beta

            if obj_new < obj:
                h = h_new
                conv = conv_new
                obj = obj_new
                if t % 10 == 0:
                    c5 = compute_c5(h)
                    if c5 < best_c:
                        best_c = c5
                        best_h = h.copy()
            # else keep current h

            if time.time() >= deadline:
                break

        # after finishing this beta, check discrete C5
        c5 = compute_c5(h)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()

        # optional random restart if no improvement for a while
        if time.time() < deadline and rng.random() < 0.15:
            h = random_feasible_vector(n, rng)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            h = project_onto_constraints(h, target, rng)

    # ------------------------------------------------------------------
    # Final local refinement on true C5
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        iters = int(min(50000, remaining * 2000))
        best_h, best_c = hill_climb_local(best_h, rng, iters, deadline)

    # ------------------------------------------------------------------
    # Final clean up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraints(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
