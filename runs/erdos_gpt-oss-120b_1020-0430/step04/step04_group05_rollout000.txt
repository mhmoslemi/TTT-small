```python
def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # if still off, adjust uniformly
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    # final fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h
```

```python
def gradient_max_corr(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # raw corr not multiplied by dx, not needed for gradient because dx is constant factor.
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)  # lag can be negative or positive
    # Compute gradient: for each i: 1 - (h[i+lag] if in range) - (h[i-lag] if in range)
    grad = np.ones_like(h)
    if lag >= 0:
        # i+lag in range for i = 0..n-1-lag
        grad[:n - lag] -= h[lag:]
        # i-lag in range only for i >= lag i.e. i = lag..n-1
        grad[lag:] -= h[:n - lag]
    else:
        # lag negative: treat as -L where L>0
        L = -lag
        # i+lag = i-L
        grad[L:] -= h[:n - L]
        # i-lag = i+L
        grad[:n - L] -= h[L:]
    # Remove mean to keep sum direction zero
    grad -= grad.mean()
    return grad, lag, max_idx
```

Check correctness: For lag=0, correlation corresponds to sum_i h_i (1 - h_i) = sum_i h_i - sum_i h_i^2. Then gradient of corr_0 is 1 - 2*h_i. Our formula: lag=0, n-lag = n, grad[:n] -= h[0:] leads to grad = 1 - h, then also grad[0:] -= h[:n] again -> grad = 1 - 2h. Good.

```
def upscale(h, new_n):
    n = h.size
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    h_new = enforce_sum(h_new, new_n/2.0, rng)
    return h_new
```

```
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # use enforce_sum to adjust
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def compute_c5(h):
    """Return C5 for a discretised step function."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / h.size)


def gradient_max_corr(h):
    """Sub‑gradient of the worst correlation (lag with maximal overlap)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)
    grad = np.ones_like(h)
    if lag >= 0:
        grad[: n - lag] -= h[lag:]
        grad[lag:] -= h[: n - lag]
    else:
        L = -lag
        grad[L:] -= h[: n - L]
        grad[: n - L] -= h[L:]
    grad -= grad.mean()          # keep sum direction zero
    return grad, lag, max_idx


def projected_gradient_step(h, step, rng):
    """One step of projected gradient descent on the worst lag."""
    grad, _, _ = gradient_max_corr(h)
    new_h = h - step * grad
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = enforce_sum(new_h, h.size / 2.0, rng)
    return new_h


def upscale(h, new_n, rng):
    """Linear interpolation to a finer grid with sum correction."""
    n = h.size
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    return enforce_sum(h_new, new_n / 2.0, rng)


def transfer_eps(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    # initialise
    if "initial_h_values" in globals():
        h_start = np.array(initial_h_values, dtype=float)
        n0 = h_start.size
        h_start = enforce_sum(np.clip(h_start, 0.0, 1.0), n0 / 2.0, rng)
    else:
        n0 = 360
        h_start = random_feasible_vector(n0, rng)

    best_h = h_start.copy()
    best_c = compute_c5(best_h)

    # stages on increasing grid resolution
    grid_sizes = [n0, min(720, n0 * 2), min(1440, n0 * 4)]
    for n in grid_sizes:
        if h_start.size != n:
            h_start = upscale(h_start, n, rng)

        cur_h = h_start.copy()
        cur_c = compute_c5(cur_h)
        iter_no = 0
        no_improve = 0
        step0 = 0.25

        max_iters = 8000
        while time.time() < deadline and iter_no < max_iters:
            step = step0 * (0.9994 ** iter_no)
            new_h = projected_gradient_step(cur_h, step, rng)
            new_c = compute_c5(new_h)
            iter_no += 1

            if new_c < cur_c - 1e-9:
                cur_h, cur_c = new_h, new_c
                no_improve = 0
            else:
                no_improve += 1

            if new_c < best_c - 1e-9:
                best_h, best_c = new_h.copy(), new_c

            if no_improve >= 2000:
                cur_h = random_feasible_vector(n, rng)
                cur_c = compute_c5(cur_h)
                no_improve = 0

        h_start = cur_h.copy()  # seed next grid

    # ---------- final polishing ----------
    if time.time() < deadline:
        for beta in [200, 800, 3200, 12800]:
            if time.time() >= deadline:
                break
            try:
                refined = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=800)
                refined_c = compute_c5(refined)
                if refined_c < best_c:
                    best_h, best_c = refined, refined_c
            except Exception:
                continue

    # ---------- small random tweaks ----------
    while time.time() < deadline:
        new_h = transfer_eps(best_h, rng, scale=0.07)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
        else:
            # occasional larger move

                new_c = compute_c5(new_h)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c

    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
