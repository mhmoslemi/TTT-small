import time
import numpy as np

# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a vector h (dx = 2 / n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----------------------------------------------------------------------
def project_onto_box_simplex(v, target_sum, eps=1e-12):
    """
    Project vector v onto the set {0 <= x_i <= 1, sum x_i = target_sum}.
    Solves for lambda in x_i = clip(v_i + lambda, 0, 1).
    Uses binary search.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ----------------------------------------------------------------------
def corr_and_grad(h, beta):
    """
    Return (corr_vector, gradient) of the smooth max objective
    F_beta(h) = (1/beta) log sum_s exp(beta * corr_s).
    Both corr and gradient are scaled by dx = 2/n.
    """
    n = h.size
    dx = 2.0 / n
    # correlation vector (scaled)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    max_corr = np.max(corr)
    # softmax weights
    exp_w = np.exp(beta * (corr - max_corr))
    sum_w = exp_w.sum()
    probs = exp_w / sum_w  # shape 2n-1

    grad = np.zeros(n, dtype=np.float64)

    # accumulate weighted gradients for each shift
    for s in range(-n + 1, n):
        p = probs[s + n - 1]
        if p == 0.0:
            continue
        if s >= 0:
            grad[: n - s] += p * (1.0 - h[s:])
            grad[s:] -= p * h[: n - s]
        else:
            s_abs = -s
            grad[s_abs:] += p * (1.0 - h[: n - s_abs])
            grad[: n - s_abs] -= p * h[s_abs:]

    # gradient already includes dx factor because corr did
    return corr, grad


# ----------------------------------------------------------------------
def gradient_shift(h, s):
    """
    Gradient of the (unscaled) correlation at a fixed shift s.
    Used by the greedy local optimizer.
    """
    n = h.size
    grad = np.zeros(n, dtype=np.float64)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def worst_shift_info(h):
    """Return (shift, index in full correlation array, correlation value)."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]


def greedy_optimize_simple(h, max_steps=2000):
    """Deterministic greedy improvement moving mass between two points."""
    n = h.size
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) < compute_c5(h):
            h = nh
            continue
        # try half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) < compute_c5(h):
            h = nh
            continue
        break
    return h


# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initial vector
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 720
        h = rng.random(n)
    n = h.size
    target_sum = n / 2.0
    h = project_onto_box_simplex(h, target_sum)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # parameters for smooth‑max continuation
    betas = [5, 20, 80, 320, 1280, 5120]
    iters_per_beta = [250, 250, 200, 150, 120, 100]

    for beta, nit in zip(betas, iters_per_beta):
        # learning rate roughly proportional to 1/sqrt(beta)
        lr = 0.5 / np.sqrt(beta)

        for it in range(nit):
            # time guard
            if time.time() - start_time > budget_s * 0.98:
                break

            # gradient of smooth max
            _, grad = corr_and_grad(h, beta)

            # projected gradient step
            h = project_onto_box_simplex(h - lr * grad, target_sum)

            # occasional evaluation of the true objective
            if (it & 7) == 0:
                cur_c = compute_c5(h)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()

        # after each beta, do a few tiny random transfers to shake out plateaus
        for _ in range(8):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = max_eps * 0.01 * rng.random()
            h[i] -= eps
            h[j] += eps
        h = project_onto_box_simplex(h, target_sum)

    # ------------------------------------------------------------------
    # final deterministic greedy polish
    best_h = greedy_optimize_simple(best_h, max_steps=3000)
    final_c = compute_c5(best_h)

    # ensure feasibility of the returned vector
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_simplex(best_h, target_sum)

    return best_h.tolist(), float(final_c), int(n)
