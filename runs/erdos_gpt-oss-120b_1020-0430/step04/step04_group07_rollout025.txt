    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_refine(h, rng, betas=(20., 200., 2000., 8000.), max_iter=400):
    n = h.size
    target = n / 2.0
    # constraints
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for beta in betas:
        def obj(x, beta=beta):
            corr = np.correlate(x, 1 - x, mode="full")
            bc = beta * corr
            m = np.max(bc)
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta
        res = opt.minimize(
            obj,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        # enforce sum exactly via random redistribution
        x = enforce_sum(x, rng)
        c = compute_c5(x)
        if c < best_c:
            best_h, best_c = x, c
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=15):
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    for it in range(max_iter):
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        grad = np.zeros_like(best_h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * best_h
        else:
            if idx >= 0:
                # shift positive
                grad[: n - k] += 1.0 - best_h[k:]
                grad[k:] -= best_h[: n - k]
            else:
                # idx negative meaning shift left
                grad[k:] += 1.0 - best_h[: n - k]
                grad[: n - k] -= best_h[k:]
        best_h = best_h - step * grad
        best_h = np.clip(best_h, 0.0, 1.0)
        # re‑enforce sum
        diff = target - best_h.sum()
        if abs(diff) > 1e-12:
            # redistribute proportionally to free space
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
                best_h[i] += add
                diff = target - best_h.sum()
        # reduce step size
        step *= 0.7
    return best_h

import time, numpy as np, scipy.optimize as opt

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with uniform spacing."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def binary_from_continuous(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Round to 0/1 and enforce exact half‑ones."""
    n = h.size
    k = n // 2
    hb = (h >= 0.5).astype(float)
    cur = int(hb.sum())
    if cur > k:
        idx = rng.choice(np.flatnonzero(hb), cur - k, replace=False)
        hb[idx] = 0.0
    elif cur < k:
        idx = rng.choice(np.flatnonzero(1.0 - hb), k - cur, replace=False)
        hb[idx] = 1.0
    return hb


def worst_shift_info(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr) - (n - 1))
    k = abs(idx)
    return idx, k, corr.max()


def binary_hill_climb(h: np.ndarray, rng: np.random.Generator,
                     max_swaps: int = 2500, no_improve_limit: int = 300):
    """Swap a 1 with a 0 that reduces the worst overlap."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    no_improve = 0

    for _ in range(max_swaps):
        idx, k, _ = worst_shift_info(best_h)
        if k == 0:
            break

        if idx >= 0:
            mask = (best_h[: n - k] > 0.5) & (best_h[k:] < 0.5)
            cand = np.flatnonzero(mask)
            if cand.size == 0:
                no_improve += 1
                if no_improve > no_improve_limit:
                    break
                continue
            i = rng.choice(cand)
            j = i + k
        else:
            mask = (best_h[k:] > 0.5) & (best_h[: n - k] < 0.5)
            cand = np.flatnonzero(mask)
            if cand.size == 0:
                no_improve += 1
                if no_improve > no_improve_limit:
                    break
                continue
            i = rng.choice(cand) + k  # i holds the 1
            j = i - k

        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c = compute_c5(new_h)

        if new_c < best_c - 1e-12:

            no_improve = 0
        else:
            no_improve += 1
            if no_improve > no_improve_limit:
                break
    return best_h, best_c


def slsqp_refine(h: np.ndarray, rng: np.random.Generator,
                 betas=(20., 200., 2000., 8000.), max_iter=300):
    """Soft‑max SLSQP polishing."""
    n = h.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    best_h = h.copy()
    best_c = compute_c5(best_h)

    for beta in betas:
        def obj(x, beta=beta):
            corr = np.correlate(x, 1 - x, mode="full")
            bc = beta * corr
            m = np.max(bc)
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta

        res = opt.minimize(
            obj,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum(x, rng)
        c = compute_c5(x)
        if c < best_c:
            best_h, best_c = x, c
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step=0.02, max_iter=12):
    """Fine‑grained gradient steps on the worst shift."""
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    cur_step = step

    for _ in range(max_iter):
        idx, k, _ = worst_shift_info(best_h)
        grad = np.zeros_like(best_h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * best_h
        else:
            if idx >= 0:          # shift right
                grad[: n - k] += 1.0 - best_h[k:]
                grad[k:]      -= best_h[: n - k]
            else:                 # shift left
                grad[k:]      += 1.0 - best_h[: n - k]
                grad[: n - k] -= best_h[k:]
        best_h = best_h - cur_step * grad
        best_h = np.clip(best_h, 0.0, 1.0)

        # enforce sum
        diff = target - best_h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - best_h[i]) if diff > 0 else max(diff, -best_h[i])
                best_h[i] += add
                diff = target - best_h.sum()

        cur_step *= 0.7
    return best_h


def resample_to_n(h: np.ndarray, n: int, rng: np.random.Generator) -> np.ndarray:
    """Linear interpolation to a new resolution, keeping feasibility."""
    x_old = np.linspace(0.0, 2.0, h.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, rng)
    return h_new


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # start from parent if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    n_candidates = [96, 128, 160, 192, 224, 256, 320, 384, 448, 512]

    # ----- main stochastic search -----
    while time.time() - start < 0.85 * budget_s:
        n = int(rng.choice(n_candidates))
        if best_h is not None:
            h = resample_to_n(best_h, n, rng)
        else:
            h = random_binary_vector(n, rng)

        # binary hill‑climb (starts from a crisp 0/1 pattern)
        h_bin = binary_from_continuous(h, rng)
        h_bin, _ = binary_hill_climb(h_bin, rng, max_swaps=3000, no_improve_limit=500)

        # continuous polishing
        h_cont, c_cont = slsqp_refine(h_bin, rng,
                                      betas=(20., 200., 2000., 8000.),
                                      max_iter=300)
        h_cont = gradient_refine(h_cont, rng, step=0.02, max_iter=12)
        c_cont = compute_c5(h_cont)

        if c_cont < best_c:
            best_h, best_c, best_n = h_cont.copy(), c_cont, n

    # ----- up‑sample and final polish -----
    if best_h is not None and time.time() - start < 0.95 * budget_s:
        for factor in (2, 3):
            n_up = best_n * factor
            if n_up > 1024:
                continue
            h_up = resample_to_n(best_h, n_up, rng)
            h_up, c_up = slsqp_refine(h_up, rng,
                                      betas=(20., 200., 2000., 8000.),
                                      max_iter=250)
            h_up = gradient_refine(h_up, rng, step=0.015, max_iter=10)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ----- tiny random tweaks till time runs out -----
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        i, j = rng.choice(best_h.size, size=2, replace=False)
        eps = rng.random() * 0.02
        if best_h[i] >= eps and best_h[j] <= 1.0 - eps:
            h_new = best_h.copy()
            h_new[i] -= eps
            h_new[j] += eps
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_h, best_c = h_new, c_new

    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
