def gradient_h(h, lag):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if lag >= 0:
        idx_i = np.arange(0, n-lag)  # i in this range
        # +1 for these i
        g[idx_i] += 1.0
        # -h_{i+lag} for these i
        g[idx_i] -= h[idx_i + lag]
        # -h_{i} for i where i+lag in range (i from 0..n-lag-1):
        # Wait the second -h_{j-lag} term is for j in [lag,n-1], use j = i+lag, i in range as above.
        # For j = idx_i + lag:
        j = idx_i + lag
        g[j] -= h[idx_i]
    else:
        L = -lag
        idx_i = np.arange(L, n)  # i from L to n-1
        g[idx_i] += 1.0
        # -h_{i-L}
        g[idx_i] -= h[idx_i - L]
        # For j in [0, n-L-1] => j = i-L, i = j+L, we subtract h_{j+L} from g[j]
        j = np.arange(0, n - L)
        g[j] -= h[j + L]
    # Scale with dx factor = 2.0/n
    return g * (2.0 / n)

def gradient_descent_refine(h_start, rng, time_budget, time_start):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    step = 0.5  # initial step magnitude
    # We could adapt step size based on acceptance.
    while time.time() - time_start < time_budget:
        # compute gradient based on worst lag
        c, lag = compute_c5_and_lag(h)
        g = gradient_h(h, lag)
        # try step
        new_h = h - step * g  # move opposite to gradient (since gradient of C5)
        # clip
        new_h = np.clip(new_h, 0.0, 1.0)
        # enforce sum
        new_h = enforce_sum(new_h, n/2.0, rng)
        new_c = compute_c5(new_h)
        if new_c < c:
            h = new_h
            # maybe increase step
            step = min(step * 1.02, 1.0)
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        else:
            # reduce step
            step *= 0.5
            if step < 1e-6:
                # occasional random perturb to escape
                new_h = perturb_h(h, rng, scale=0.3)
                new_h = enforce_sum(new_h, n/2.0, rng)
                new_c = compute_c5(new_h)
                if new_c < best_c:
                    best_c = new_c
                    best_h = new_h.copy()
                    h = new_h
                # reset step
                step = 0.5
        # Perhaps also periodically random restart if stuck
        # Check time near end
    return best_h, best_c

def gradient_h(h, lag):
    n = h.shape[0]
    g = np.zeros_like(h, dtype=float)
    if lag >= 0:
        idx = np.arange(0, n - lag)
        g[idx] += 1.0
        g[idx] -= h[idx + lag]
        g[idx + lag] -= h[idx]
    else:
        L = -lag
        idx = np.arange(L, n)
        g[idx] += 1.0
        g[idx] -= h[idx - L]
        g[idx - L] -= h[idx]
    # scale factor
    return g * (2.0 / n)

def adjust_sum(h, target, rng):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = len(h)
    # simple add uniform and clip
    h = h + diff / n
    # clip to bounds
    h = np.clip(h, 0.0, 1.0)
    # if still off due to clipping, correct via random adjustments
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute remaining diff via random indices
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

import time

# ----- feasibility utilities -----
def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust entries to reach exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # fallback simple correction
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

# ----- objective and gradient -----
def compute_c5(h):
    """Exact C5 for vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def compute_c5_and_lag(h):
    """Return (C5, lag) where lag = index - (n-1) for the max correlation."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / n), idx - (n - 1)

def gradient_h(h, lag):
    """Gradient of C5 (scaled by dx) w.r.t. h for the worst lag."""
    n = h.size
    g = np.zeros_like(h, dtype=float)
    if lag >= 0:
        idx = np.arange(0, n - lag)
        g[idx] += 1.0
        g[idx] -= h[idx + lag]
        g[idx + lag] -= h[idx]
    else:
        L = -lag
        idx = np.arange(L, n)
        g[idx] += 1.0
        g[idx] -= h[idx - L]
        g[idx - L] -= h[idx]
    return g * (2.0 / n)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
