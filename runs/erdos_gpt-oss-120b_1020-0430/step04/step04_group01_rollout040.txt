def fft_convolve(a,b):
    n = len(a) + len(b) - 1
    L = 1 << ((n-1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa*fb, L)[:n]
    return conv

def project_to_sum(v, target, max_iter=50):
    # v assumed to be within [0,1] after clipping? Actually we can start with any v.
    lo = -1.0   # lambda lower bound; because v - lambda <=1 => lambda >= v-1 max? Actually lambda lower bound can be -1 maybe.
    hi = 1.0    # lambda upper bound
    # Actually we can set lo = min(v) - 1, hi = max(v)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    return x

import time
import numpy as np

# --------------------------------------------------------------
# Projection onto the hyperplane sum = target with box [0,1]
# --------------------------------------------------------------
def project_to_sum(v, target, max_iter=30):
    """Projection of v onto {x: 0<=x<=1, sum x = target} via bisection."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random vector projected onto sum = n/2, clipped to [0,1]."""
    v = rng.random(n)
    return project_to_sum(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j, preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# FFT convolution utilities
# --------------------------------------------------------------
def fft_convolve(a, b):
    """Linear convolution via FFT (real inputs)."""
    n = len(a) + len(b) - 1
    L = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[:n]
    return conv


def correlate_full_fft(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full_fft(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Smoothed objective and gradient (log‑sum‑exp surrogate)
# --------------------------------------------------------------
def smooth_obj_and_grad(h, beta):
    """Return smoothed objective and its gradient."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]

    max_c = conv.max()
    shifted = np.exp(beta * (conv - max_c))
    sum_shifted = shifted.sum()
    w = shifted / sum_shifted                     # normalized weights

    w_rev = w[::-1]
    # term1 = sum_s w_s * (1 - h_{n-1-(s-i)})  --> convolution w_rev * rev
    conv1 = fft_convolve(w_rev, rev)
    term1 = conv1[(2 * n - 2) - np.arange(n)]

    # term2 = sum_s w_s * h_{s+i-(n-1)}      --> convolution w_rev * h
    conv2 = fft_convolve(w_rev, h)
    term2 = conv2[(n - 1):(n - 1) + n]

    grad = (2.0 / n) * (term1 - term2)
    obj = (2.0 / n) * (max_c + np.log(sum_shifted) / beta)
    return obj, grad


def smooth_obj(h, beta):
    """Smoothed objective without gradient (used for line‑search)."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    max_c = conv.max()
    shifted = np.exp(beta * (conv - max_c))
    sum_shifted = shifted.sum()
    return (2.0 / n) * (max_c + np.log(sum_shifted) / beta)

def pgd_optimize(h_start, beta_schedule, iters_per_beta, rng, deadline):

def binary_sa(start_bin, rng, deadline):
    """Swap‑based simulated annealing on a binary vector (sum = n/2)."""
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 25000

    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i      # ensure i is 1, j is 0
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    n = h.size
    target = n / 2.0

    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

    # triangular (tent) shape
    tri = 1.0 - np.abs(xs - 1.0)
    cand = project_to_sum(np.clip(tri, 0.0, 1.0), n / 2.0)
    c_tri = compute_c5(cand)
    if c_tri < best_c:
        best_c, best_h = c_tri, cand.copy()
    # sine / cosine family
    for f in (1, 2, 3):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        cand = project_to_sum(np.clip(s, 0.0, 1.0), n / 2.0)
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        cand = project_to_sum(np.clip(c, 0.0, 1.0), n / 2.0)
        c_val = compute_c5(cand)
        if c_val < best_c:
            best_c, best_h = c_val, cand.copy()

    # ------------------------------------------------------------------
    # 2. Gradient‑descent restarts (continuous)
    # ------------------------------------------------------------------
    beta_sched = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
    iters_per_beta = 250

    while time.time() < deadline - 5.0:
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)

        h_opt = pgd_optimize(start, beta_sched, iters_per_beta, rng, deadline)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ------------------------------------------------------------------
    # 3. Binary simulated annealing from the best continuous solution
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        idx = np.argsort(-best_h)[: n // 2]
        bin_start = np.zeros(n)
        bin_start[idx] = 1.0
        h_bin, c_bin = binary_sa(bin_start, rng, deadline)
        if c_bin < best_c:
            best_c, best_h = c_bin, h_bin.copy()

    # ------------------------------------------------------------------
    # 4. Final greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        h_final, c_final = hill_climb(best_h, rng, max_iters=3000, start_step=0.2)
        if c_final < best_c:
            best_c, best_h = c_final, h_final.copy()

    # ------------------------------------------------------------------
    # 5. Cleanup and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
