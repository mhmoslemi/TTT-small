def compute_correlation_counts(h):
    n = h.size
    L = 1 << ((2*n -1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft(h[::-1], L)
    conv = np.fft.irfft(fh*fr, L)[:2*n-1]
    O = np.empty(n, dtype=np.int64)
    O[0] = np.dot(h,h)  # n/2
    O[1:] = np.rint(conv[n: 2*n-1]).astype(np.int64)  # rounding to int
    return O

def binary_local_search(h_bin, O, cur_max, rng, stop_iter, deadline):
    # h_bin is np.ndarray of float 0.0/1.0, O np.ndarray int, cur_max int.
    # pos1 list of indices where h==1, pos0 list where h==0
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    n = h_bin.size
    no_improve = 0
    for it in range(stop_iter):
        if time.time() >= deadline:
            break
        # random pick one from ones and zeros
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta as described

        delta[:] = 0

        if i > 0:

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c
```

import time, numpy as np
from scipy.optimize import minimize

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def correlation_general(a, b):
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b[::-1], L)
    conv = np.fft.irfft(fa * fb, L)[: 2 * n - 1]
    return conv

def convolve_full(a, b):
    n, m = a.size, b.size
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[: n + m - 1]
    return conv

def compute_c5(h):
    conv = correlation_general(h, 1.0 - h)
    return conv.max() * (2.0 / h.size)

def smooth_obj_and_grad(x, beta, target, lam):
    n = x.size
    sumx = x.sum()
    diff = sumx - target
    penalty = lam * diff * diff
    selfc = correlation_general(x, x)
    c = sumx - selfc                     # length 2n‑1
    bc = beta * c
    m = bc.max()
    smooth = (m + np.log(np.sum(np.exp(bc - m)))) / beta
    w = np.exp(bc - m)
    w /= w.sum()
    w_rev = w[::-1]
    start = n - 1
    term1 = convolve_full(w, x)[start : start + n]
    term2 = convolve_full(w_rev, x)[start : start + n]
    grad = 1.0 - (term1 + term2) + 2.0 * lam * diff
    return smooth + penalty, grad

def lbfgsb_optimize(x0, beta, lam=1e5, max_iter=150):
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def fun(x):
        return smooth_obj_and_grad(x, beta, target, lam)
    res = minimize(fun, x0, method='L-BFGS-B', jac=True,
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    return enforce_sum(x, target, rng)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c
