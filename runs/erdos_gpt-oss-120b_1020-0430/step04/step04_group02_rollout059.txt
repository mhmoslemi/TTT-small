def compute_corr_fft(h):
    n = h.shape[0]
    fft_len = 1 << ((2*n - 1).bit_length())
    # Real FFT
    H = np.fft.rfft(h, n=fft_len)
    B = np.fft.rfft(1.0 - h, n=fft_len)
    conv = np.fft.irfft(H * B, n=fft_len)
    corr = conv[:2*n - 1]
    return corr

def subgradient_descent(h0, n_iters, step0, rng, time_limit=None, start_time=None, budget=None):
   n = len(h0)
   target = n / 2.0
   h = h0.copy()
   best_h = h.copy()
   best_c = compute_c5(h)
   v = np.zeros_like(h)  # momentum
   for it in range(1, n_iters + 1):
       if time_limit is not None:
           if time.time() - start_time > time_limit:
               break
       corr = compute_corr_fft(h)
       idx_max = int(np.argmax(corr))
       d = idx_max - (n - 1)
       # gradient
       g = np.zeros_like(h)
       if d >= 0:
           nd = n - d
           g[:nd] += 1 - h[d:]
           g[d:]   -= h[:nd]
       else:
           dneg = -d
           nd = n - dneg
           g[dneg:] += 1 - h[:nd]
           g[:nd]   -= h[dneg:]
       # step schedule
       step = step0 / np.sqrt(it)
       v = 0.9 * v + step * g
       h = h - v
       # projection onto box and sum
       h = np.clip(h, 0.0, 1.0)
       # enforce sum exactly via project_box_sum
       h = project_box_sum(h, target)
       # occasionally random perturb to escape plateau
       if (it % 500 == 0) and (rng.random() < 0.3):
           # add small noise, then project
           noise = (rng.random(h.shape) - 0.5) * 0.02
           h = np.clip(h + noise, 0.0, 1.0)
           h = project_box_sum(h, target)
       c = compute_c5(h)
       if c < best_c:
           best_c = c
           best_h = h.copy()
   return best_h, best_c

import time
import numpy as np

# ---------- projection onto [0,1] with exact sum ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


# ---------- random feasible vector ----------
def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


# ---------- perturbation used for restarts ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- correlation via FFT (zero‑padded) ----------
def compute_corr_fft(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    FA = np.fft.rfft(h, n=L)
    FB = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(FA * np.conj(FB), n=L)
    # rearrange to shifts -(n-1)..(n-1)
    if n > 1:
        full = np.concatenate((corr[-(n - 1):], corr[:n]))
    else:
        full = corr[:1]
    return full


# ---------- compute C5 ----------
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(compute_corr_fft(h)) * dx


# ---------- sub‑gradient descent on max‑correlation ----------
def subgradient_descent(h0, n_iters, step0, rng,
                       top_k=3, momentum=0.9,
                       time_limit=None, start_time=None):
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    v = np.zeros_like(h)

    for it in range(1, n_iters + 1):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break

        corr = compute_corr_fft(h)
        # indices of top‑k correlations
        if top_k >= corr.size:
            idxs = np.arange(corr.size)
        else:
            idxs = np.argpartition(-corr, top_k)[:top_k]

        g = np.zeros_like(h)
        for idx in idxs:
            d = idx - (n - 1)          # shift corresponding to this index
            if d >= 0:
                nd = n - d
                g[:nd] += 1.0 - h[d:]
                g[d:]   -= h[:nd]
            else:
                dneg = -d
                nd = n - dneg
                g[dneg:] += 1.0 - h[:nd]
                g[:nd]   -= h[dneg:]

        g /= float(top_k)               # average gradient

        step = step0 / np.sqrt(it)
        v = momentum * v + step * g
        h = h - v

        # keep within bounds and enforce exact sum
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target)

        # occasional noise to escape flat regions
        if (it % 500 == 0) and (rng.random() < 0.3):
            noise = (rng.random(h.shape) - 0.5) * 0.02
            h = np.clip(h + noise, 0.0, 1.0)
            h = project_box_sum(h, target)

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s * 0.90  # reserve a little for final clean‑up

    # ----- initialise base grid -----
    base_n = 720
    if "initial_h_values" in globals():

        if h0.size != base_n:

            old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)

            h0 = np.interp(new_x, old_x, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, base_n / 2.0)
    else:
        h0 = random_feasible_vector(base_n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- several restarts with sub‑gradient descent -----
    restart_budget = time_limit * 0.60
    per_restart = 2500
    restarts = 6
    for r in range(restarts):
        if (time.time() - start_time) > restart_budget:

        if r == 0:
            start_vec = best_h.copy()
        else:
            start_vec = perturb_h(best_h, rng, scale=0.4)

        h_opt, c_opt = subgradient_descent(

        )

    # ----- upscale to higher resolution and refine -----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.15:
        high_n = 1080
        if best_h.size != high_n:
            old_n = best_h.size
            old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
            new_x = (np.arange(high_n) + 0.5) * (2.0 / high_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = project_box_sum(h_up, high_n / 2.0)
        else:
            h_up = best_h.copy()

        # number of iterations scaled to remaining time
        iters_high = int(min(8000, remaining / 0.0004))  # rough estimate
        h_ref, c_ref = subgradient_descent(
            h_up,
            n_iters=iters_high,
            step0=0.6,
            rng=rng,
            top_k=3,
            momentum=0.9,
            time_limit=budget_s,
            start_time=start_time,
        )
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref.copy()

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
