import time

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


# ---------- random vector ----------
def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


# ---------- perturbations ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_idx(h):

    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


def hill_climb_targeted(
    h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25
):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- softmax surrogate ----------
def grad_softmax(h, beta, dx):
    # I_k = dx * correlation of h with (1-h)
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    I = corr_raw * dx
    m = I.max()
    exp_scaled = np.exp(beta * (I - m))
    w = exp_scaled / exp_scaled.sum()  # length 2n-1
    offset = h.size - 1
    # Σ w_k h_{j+k}
    sum_fwd = np.correlate(w, h, mode="full")[offset : offset + h.size]
    # Σ w_k h_{j-k}
    sum_rev = np.correlate(w[::-1], h, mode="full")[offset : offset + h.size]
    grad = dx * (1.0 - sum_fwd - sum_rev)
    return grad, m


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    I = corr * (2.0 / x.size)
    m = np.max(I)
    exp_scaled = np.exp(CURRENT_BETA * (I - m))
    return (m + np.log(np.sum(exp_scaled)) / CURRENT_BETA)


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- gradient descent ----------
def gd_optimize(h_start, dx, target_sum, beta_schedule, deadline):
    h = h_start.copy()
    for beta in beta_schedule:
        lr = 0.5 / math.sqrt(beta)
        for _ in range(2000):
            if time.time() >= deadline:
                return h
            grad, _ = grad_softmax(h, beta, dx)
            h = project_box_sum(np.clip(h - lr * grad, 0.0, 1.0), target_sum)
    return h


# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    target_sum = n / 2.0
    dx = 2.0 / n

    # ---- candidate pool ----
    candidates = [h0]
    for _ in range(8):
        freq = rng.integers(1, 11)
        phase = rng.random() * 2 * np.pi
        cand = 0.5 + 0.5 * np.sin(np.pi * freq * xs + phase)
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_box_sum(cand, target_sum)
        candidates.append(cand)
    for _ in range(4):
        freq = rng.integers(1, 11)
        phase = rng.random() * 2 * np.pi
        cand = 0.5 + 0.5 * np.cos(np.pi * freq * xs + phase)
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_box_sum(cand, target_sum)
        candidates.append(cand)
    for _ in range(3):
        mask = np.zeros(n)
        mask[rng.permutation(n)[: n // 2]] = 1.0
        kernel = np.array([0.25, 0.5, 0.25])
        smooth = np.convolve(mask, kernel, mode="same")
        smooth = np.clip(smooth, 0.0, 1.0)
        smooth = project_box_sum(smooth, target_sum)
        candidates.append(smooth)
    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    best_h = candidates[0].copy()
    best_c = compute_c5(best_h)
    for h in candidates[1:]:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]

    # ---- refinement per candidate ----
    for h_cand in candidates:
        if time.time() - start_time > budget_s * 0.85:
            break
        deadline = start_time + budget_s * 0.85
        gd_h = gd_optimize(h_cand, dx, target_sum, beta_schedule, deadline)
        c_gd = compute_c5(gd_h)
        if c_gd < best_c:
            best_c, best_h = c_gd, gd_h.copy()
        hc_h, c_hc = hill_climb_targeted(gd_h, rng, max_iters=600, init_step=0.4)
        if c_hc < best_c:
            best_c, best_h = c_hc, hc_h.copy()

    # ---- optional up‑sampling ----
    if time.time() - start_time < budget_s * 0.95:
        target_n = min(720, max(540, n * 2))
        if target_n != n:
            old_x = xs
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            n = target_n
            xs = new_x
            dx = 2.0 / n
            target_sum = n / 2.0
            h_up = project_box_sum(h_up, target_sum)
            deadline = start_time + budget_s * 0.99
            h_up = gd_optimize(h_up, dx, target_sum, beta_schedule, deadline)
            h_up, c_up = hill_climb_targeted(h_up, rng, max_iters=800, init_step=0.3)
            if c_up < best_c:
                best_c, best_h = c_up, h_up.copy()

    # ---- final SLSQP polish if time ----
    if time.time() - start_time < budget_s * 0.995:
        try:
            h_ref = slsqp_optimize(best_h, 20000, max_iter=400)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()
        except Exception:
            pass

    # ---- final check ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
