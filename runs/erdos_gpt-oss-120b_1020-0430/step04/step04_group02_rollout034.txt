def corr_full(h):
    g = 1 - h
    L = 1 << (2*n - 1).bit_length()  # next power of 2
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(g, L)

    corr = correlation_full(h)  # length 2n-1

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


# ---------- perturbations ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


# ---------- FFT correlation ----------
def correlation_full(h):
    n = h.shape[0]
    g = 1.0 - h
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(g, L)
    corr = np.fft.irfft(H * np.conj(G), L)[: 2 * n - 1]
    return corr


def compute_c5(h):
    n = h.size
    corr = correlation_full(h)
    return np.max(corr) * (2.0 / n)


def compute_c5_and_idx(h):
    n = h.size
    corr = correlation_full(h)
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / n), idx


# ---------- hill climbing ----------
def hill_climb_targeted(h_start, rng, max_iters=2000, init_step=0.5, n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return h, best_c


# ---------- SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = correlation_full(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- smooth log‑sum‑exp descent ----------
def gradient_logsumexp(h, beta):
    n = h.size
    corr = correlation_full(h)               # length 2n‑1
    scaled = beta * corr
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # convolutions w * h and w_rev * h
    L = 1 << ((3 * n - 2).bit_length())
    W = np.fft.rfft(w, L)
    H = np.fft.rfft(h, L)
    conv_wh = np.fft.irfft(W * H, L)[: 3 * n - 2]
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, L)
    conv_wrev_h = np.fft.irfft(W_rev * H, L)[: 3 * n - 2]
    start = n - 1
    B = conv_wh[start : start + n]          # Σ w_s h_{i‑s}
    F = conv_wrev_h[start : start + n]      # Σ w_s h_{i+s}
    grad = 1.0 - (B + F)
    return grad


def smooth_descent(h, beta_list, eta=0.5, max_iter_per_beta=200):
    target = h.size / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for beta in beta_list:
        for _ in range(max_iter_per_beta):
            grad = gradient_logsumexp(best_h, beta)
            improved = False
            for factor in (1.0, 0.5, 0.2):
                cand = project_box_sum(best_h - eta * factor * grad, target)
                c = compute_c5(cand)
                if c < best_c - 1e-12:
                    best_c, best_h = c, cand.copy()
                    improved = True
                    break
            if not improved:
                eta *= 0.7
                if eta < 1e-6:
                    break
        eta = max(eta, 0.1)
    return best_h, best_c


# ---------- main entry ----------
