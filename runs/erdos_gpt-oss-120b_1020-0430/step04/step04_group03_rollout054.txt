    w = exp_terms / sum_exp
    n = h.size

import time, numpy as np

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple water-filling: distribute diff uniformly to non-saturated entries
    # repeat a few times
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:
            # all at bounds, adjust smallest distance
            if diff > 0:
                idx = np.argmin(1.0 - h)
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                idx = np.argmax(h)
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            break
        delta = diff / free.size
        # apply but respect bounds
        add = np.minimum(delta, 1.0 - h[free])
        sub = np.maximum(delta, -h[free])
        # choose sign based on delta
        adjust = np.where(delta > 0, add, sub)
        h[free] += adjust
        diff = target - h.sum()
    # if still off, fallback random adjust
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def smooth_max_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # stable log-sum-exp
    exp_terms = np.exp(beta * (corr - max_corr))
    sum_exp = exp_terms.sum()
    smooth = max_corr + np.log(sum_exp) / beta
    w = exp_terms / sum_exp   # weights over shifts
    center = n - 1
    # accumulate w_circ[t] = w_{-t}
    s_vals = np.arange(-(n-1), n)
    idx = center + s_vals
    t_vals = (-s_vals) % n
    w_circ = np.zeros(n)
    np.add.at(w_circ, t_vals, w[idx])
    # gradient = - circular convolution of h with w_circ
    grad = -np.real(np.fft.ifft(np.fft.fft(h) * np.fft.fft(w_circ))).astype(np.float64)
    return smooth, grad

def anneal_binary(best_h, rng, budget_s, start_time):
    # Convert best_h to binary (0/1) with exactly n/2 ones
    h = best_h.copy()
    n = h.size
    target = n // 2
    # Determine ordering
    order = np.argsort(-h)  # descending
    binary = np.zeros(n, dtype=np.float64)
    binary[order[:target]] = 1.0
    c_best = compute_c5(binary)
    best_vec = binary.copy()
    best_c = c_best
    # Prepare indices lists
    ones_idx = np.where(binary == 1.0)[0]
    zeros_idx = np.where(binary == 0.0)[0]
    # Annealing schedule
    max_iters = 200000
    T0 = 0.001   # initial temperature
    for it in range(max_iters):
        now = time.time()
        if now - start_time > budget_s * 0.97:
            break
        # pick random swap
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # perform swap
        binary[i] = 0.0
        binary[j] = 1.0
        c_new = compute_c5(binary)
        delta = c_new - best_c
        # temperature
        T = T0 * (1.0 - it / max_iters)
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            # update indices
            ones_idx = np.where(binary == 1.0)[0]
            zeros_idx = np.where(binary == 0.0)[0]
            if c_new < best_c:
                best_c = c_new
                best_vec = binary.copy()
        else:
            # revert swap
            binary[i] = 1.0
            binary[j] = 0.0
    return best_vec, best_c

def project_onto_box_simplex(v, target, eps=1e-12):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # If sum already target, return.
    s = v.sum()
    if abs(s - target) <= eps:
        return v
    # Use algorithm: want to find lambda such that sum(clip(v - lambda, 0,1)) = target.
    # This lambda can be found via bisection.
    # v - lambda shifted by lambda.
    lo = -1.0  # lambda lower bound (since v_i - lambda <=1 -> lambda >= v_i -1)
    hi = 1.0   # lambda upper bound (since v_i - lambda >=0 -> lambda <= v_i)
    # Actually we can set lo = -1.0, hi = 1.0; because v_i in [0,1].
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < 1e-12:
            break
    return w

import time
import numpy as np

# -------------------------------------------------------------
# utilities
# -------------------------------------------------------------
def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to make sum exactly target (deterministic water‑fill)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # simple iterative redistribution on free variables
    for _ in range(20):
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        add = np.minimum(delta, 1.0 - h[free])
        sub = np.maximum(delta, -h[free])
        adjust = np.where(delta > 0, add, sub)
        h[free] += adjust
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            return h
    # final tiny correction (rare)
    if abs(diff) > 1e-14:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a vector h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def smooth_max_and_grad(h: np.ndarray, beta: float):
    """Smooth approximation of max correlation (log‑sum‑exp) and its gradient."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    # stable log‑sum‑exp
    exp_terms = np.exp(beta * (corr - max_corr))
    sum_exp = exp_terms.sum()
    smooth = max_corr + np.log(sum_exp) / beta
    w = exp_terms / sum_exp                     # weights for each shift

    # build circular weight vector w_circ[t] = w_{-t}
    center = n - 1
    s_vals = np.arange(-(n - 1), n)             # shifts
    idx = center + s_vals
    t_vals = (-s_vals) % n
    w_circ = np.zeros(n)
    np.add.at(w_circ, t_vals, w[idx])

    # gradient = -h convolved (circularly) with w_circ
    grad = -np.real(np.fft.ifft(np.fft.fft(h) * np.fft.fft(w_circ))).astype(np.float64)
    return smooth, grad


# -------------------------------------------------------------
# optimisation phases
# -------------------------------------------------------------
def gradient_descent_phase(h0: np.ndarray, rng: np.random.Generator,
                          budget_s: float, start_time: float):
    """Smooth‑max gradient descent with increasing β."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)

    beta_schedule = [10, 50, 200, 800, 2000, 8000, 20000]
    for beta in beta_schedule:
        lr = 0.5
        max_iters = 4000
        for _ in range(max_iters):
            if time.time() - start_time > budget_s * 0.75:
                return h, best_h, best_c
            smooth, grad = smooth_max_and_grad(h, beta)

            step = lr
            # back‑tracking line search on smooth objective
            for _ in range(20):
                h_new = h - step * grad
                h_new = enforce_sum(np.clip(h_new, 0.0, 1.0), target, rng)
                smooth_new, _ = smooth_max_and_grad(h_new, beta)
                if smooth_new < smooth or step < 1e-12:
                    break
                step *= 0.5

            if smooth_new < smooth:
                h = h_new
                lr = min(lr * 1.05, 2.0)
                c5 = compute_c5(h)
                if c5 < best_c:
                    best_h, best_c = h.copy(), c5
            else:
                lr = max(lr * 0.7, 1e-4)
    return h, best_h, best_c


def anneal_binary_phase(best_h: np.ndarray, rng: np.random.Generator,
                        budget_s: float, start_time: float):
    """Simulated‑annealing on the 0/1 version of the current best vector."""
    n = best_h.size
    target = n // 2
    order = np.argsort(-best_h)                 # largest first
    binary = np.zeros(n, dtype=np.float64)
    binary[order[:target]] = 1.0
    c_best = compute_c5(binary)
    best_vec = binary.copy()

    # maintain index lists for quick swaps
    ones_idx = np.where(binary == 1.0)[0]
    zeros_idx = np.where(binary == 0.0)[0]

    max_steps = 500_000
    T0 = 0.001
    step = 0
    while step < max_steps and time.time() - start_time < budget_s * 0.97:
        step += 1
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        # propose swap
        binary[i] = 0.0
        binary[j] = 1.0
        c_new = compute_c5(binary)
        delta = c_new - c_best
        T = T0 * (1.0 - step / max_steps)

        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            # update index lists
            ones_idx = np.where(binary == 1.0)[0]
            zeros_idx = np.where(binary == 0.0)[0]
            if c_new < c_best:
                c_best = c_new
                best_vec = binary.copy()
        else:
            # revert swap
            binary[i] = 1.0
            binary[j] = 0.0

    return best_vec, c_best


# -------------------------------------------------------------
# entry point
# -------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial feasible vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 720
        h0 = rng.random(n)
        h0 = enforce_sum(h0, n / 2.0, rng)

    n = h0.size
    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n / 2.0, rng)

    # 1) smooth‑max gradient descent
    _, best_h, best_c = gradient_descent_phase(h0, rng, budget_s, start_time)

    # 2) binary simulated annealing
    best_vec, best_c = anneal_binary_phase(best_h, rng, budget_s, start_time)

    # final tidy‑up
    best_vec = enforce_sum(np.clip(best_vec, 0.0, 1.0), best_vec.size / 2.0, rng)
    final_c = compute_c5(best_vec)

    return best_vec.tolist(), float(final_c), int(best_vec.size)
