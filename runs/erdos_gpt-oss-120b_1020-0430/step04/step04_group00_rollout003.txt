def compute_full_corr(h):
    n = h.size
    # zero pad to (2n)
    H = np.fft.fft(np.concatenate([h, np.zeros(n)]))
    G = np.fft.fft(np.concatenate([1 - h, np.zeros(n)]))
    corr = np.fft.ifft(H * G).real
    # correlation output length 2n, but our full correlation length should be 2n-1, with lags from -(n-1) to (n-1) placed at indices 0..2n-2
    # Values at index n-1 + k correspond to shift k
    # We'll remove last element
    corr = corr[:2*n-1]
    return corr

def full_corr_fft(h):
    n = h.size
    H = np.fft.fft(h, n*2)
    G = np.fft.fft(1 - h, n*2)
    corr = np.fft.ifft(H * G).real
    # correlation length 2n, but last element corresponds to shift -(n) (outside range). We need first 2n-1.
    return corr[:2*n-1]

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def full_corr_fft(h):
    n = h.size
    H = np.fft.fft(h, n * 2)
    G = np.fft.fft(1.0 - h, n * 2)
    corr = np.fft.ifft(H * G).real
    return corr[: 2 * n - 1]

def compute_c5(h):
    n = h.size
    return float(np.max(full_corr_fft(h)) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def adjust_sum_exact(h, rng):

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - abs(k)
            if k > 0:
                g[:upper] += 1.0 - h[k:]
                g[k:] -= h[:upper]
            else:
                g[-k:] += 1.0 - h[:upper]
                g[:upper] -= h[-k:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                if abs(diff) < 1e-12:
                    break
    return h

def worst_shift_improve(h, rng, deadline, steps=(0.05, 0.02, 0.01), iters_per_step=400):
    n = h.size
    total_iters = sum(iters_per_step for _ in steps)
    it = 0
    step_idx = 0
    while it < total_iters and time.time() < deadline:
        if it // iters_per_step >= len(steps):
            step_idx = len(steps) - 1
        else:
            step_idx = it // iters_per_step
        step_sz = steps[step_idx]
        corr = full_corr_fft(h)
        idx = np.argmax(corr) - (n - 1)
        k = idx
        if k >= 0:
            hi = h[: n - k]
            hj = h[k:]
        else:
            hi = h[-k :]
            hj = h[: n + k]
        cap = np.minimum(1.0 - hi, 1.0 - hj)
        if cap.max() < 1e-12:
            break
        score = (hi + hj) * cap
        i_pair = np.argmax(score)
        delta = min(step_sz, cap[i_pair])
        if k >= 0:
            h[i_pair] += delta
            h[i_pair + k] += delta
        else:
            h[-k + i_pair] += delta
            h[i_pair] += delta
        need = 2.0 * delta
        while need > 1e-12:
            j = rng.integers(0, n)
            sub = min(need, h[j])
            if sub > 0:
                h[j] -= sub
                need -= sub
        np.clip(h, 0.0, 1.0, out=h)
        it += 1
    return h
