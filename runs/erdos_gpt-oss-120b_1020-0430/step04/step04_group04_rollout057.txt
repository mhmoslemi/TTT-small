Implementation details for compute_c5:

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def enforce_sum(h, target, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1 - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            # distribute evenly across all indices (may violate bounds)
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            continue
        per = diff / n_free
        # saturate
        # where per > 0 (need increase) but h + per >1, set them to 1
        if per > 0:
            # compute max increase per free index before hitting 1
            max_inc = np.min(1 - h[free])
            if per > max_inc:
                # set those indices to 1
                idx = np.where(free)[0][np.argmin(1 - h[free])]
                # Actually we want to set all free that would exceed 1? We'll just set them to 1 gradually
                # But easier: apply per, then clip, recompute diff and iterate
                h[free] += per
                h = np.clip(h, 0.0, 1.0)
                diff = target - h.sum()
                continue
        else:
            # per <0, need decrease
            max_dec = np.min(h[free])
            if per < -max_dec:
                h[free] += per
                h = np.clip(h, 0.0, 1.0)
                diff = target - h.sum()
                continue
        # If per is safe
        h[free] += per
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # ensure within bounds (should already)
    return new_h

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

def sum_eq_constraint(x):

def softmax_obj(x):
    # compute correlation
    n = x.size
    rev = (1 - x)[::-1]
    L = 2*n -1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = CURRENT_BETA * conv
    # Implement log-sum-exp in a stable way
    max_bc = np.max(bc)
    # softmax
    # Return max + (1/beta) * log(sum(exp(bc-max_bc)))
    return (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):

    n = h0.size

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())  # next power of two >= L
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 5000) -> np.ndarray:
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    h = rng.random(n)
    h = h / h.sum() * (n / 2.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, init_step: float = 0.5,
                     n_trials: int = 5) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climb."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


def simulated_annealing(h0: np.ndarray, rng: np.random.Generator,
                        max_iter: int = 8000,
                        init_temp: float = 1e-2, final_temp: float = 1e-6,
                        init_scale: float = 0.5, final_scale: float = 0.02) -> tuple[np.ndarray, float]:
    """Simulated‑annealing refinement."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c


def sum_eq_constraint(x: np.ndarray) -> float:
    """Equality constraint for SLSQP: Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth surrogate of max_k correlation using log‑sum‑exp."""
    n = x.size
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = CURRENT_BETA * conv
    max_bc = np.max(bc)
    return (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    """SLSQP refinement of the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-9:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def random_coeffs(M: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(-0.5, 0.5, M)
    s = np.sum(np.abs(a))
    if s > 0.5:
        a = a * (0.5 / s)
    return a


def surrogate_c5_from_a(a: np.ndarray, x: np.ndarray, beta: float) -> float:
    """Smooth surrogate C5 for a sinusoidal coefficient vector."""
    h = 0.5 + np.dot(np.sin(np.outer(x, np.arange(1, len(a) + 1) * np.pi)), a)
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = beta * conv
    max_bc = np.max(bc)
    return (max_bc + np.log(np.sum(np.exp(bc - max_bc)))) / beta


def get_h_from_coeffs(a: np.ndarray, x: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    h = 0.5 + np.dot(np.sin(np.outer(x, np.arange(1, len(a) + 1) * np.pi)), a)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def optimize_coeffs(M: int, x: np.ndarray, target: float, rng: np.random.Generator,
                    beta_seq: tuple = (800, 2000, 5000)) -> np.ndarray:
    """Random restarts on sinusoidal coefficients with sequential β refinement."""
    a = random_coeffs(M, rng)
    for beta in beta_seq:
        bounds = [(-0.5, 0.5)] * M

        def obj(vec):
            return surrogate_c5_from_a(vec, x, beta)

        res = minimize(obj, a, method="L-BFGS-B", bounds=bounds,
                       options={"maxiter": 200, "ftol": 1e-12})
        a = res.x
        s = np.sum(np.abs(a))
        if s > 0.5:
            a = a * (0.5 / s)
    return get_h_from_coeffs(a, x, target, rng)
