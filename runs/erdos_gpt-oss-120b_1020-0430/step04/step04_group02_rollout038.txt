import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -5.0, 5.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) * 0.5, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)

# ---------- perturbations ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low.any() or not high.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low))
    i_high = rng.choice(np.flatnonzero(high))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i_low] += eps
    new[i_low + k] += eps
    new[i_high] -= eps
    new[i_high + k] -= eps
    return new

# ---------- fast C5 ----------
def compute_c5_fft(h):
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n
    H = np.fft.rfft(h, L)
    R = np.fft.rfft(rev, L)
    corr = np.fft.irfft(H * R, L)[: 2 * n - 1]
    return corr.max() * (2.0 / n)

def compute_c5_and_idx_fft(h):
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n
    H = np.fft.rfft(h, L)
    R = np.fft.rfft(rev, L)
    corr = np.fft.irfft(H * R, L)[: 2 * n - 1]
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / n), idx

# alias for hill‑climb compatibility
compute_c5 = compute_c5_fft
compute_c5_and_idx = compute_c5_and_idx_fft

# ---------- hill climb ----------
def hill_climb_targeted(h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.size
    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

# ---------- subgradient descent ----------
def subgradient_descent(h0, max_iters=2000, init_step=1.0, decay=0.95):
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    step = init_step
    best_h, best_c = h.copy(), compute_c5(h)
    for it in range(max_iters):
        c, idx = compute_c5_and_idx(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[: n - k] += (1.0 - h[k:])
            g[k:] -= h[: n - k]
        else:
            m = -k
            g[m:] += (1.0 - h[: n - m])
            g[: n - m] -= h[m:]
        h = h - step * dx * g
        h = project_box_sum(h, n / 2.0)
        if (it + 1) % (max_iters // 10) == 0:
            step *= decay
    final_c = compute_c5(best_h)
    return best_h, final_c

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

# ---------- main ----------
