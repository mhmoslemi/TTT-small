def slsqp_optimize(h0, beta, max_iter=200):
    from scipy.optimize import minimize
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def sum_eq(x):
        return np.sum(x) - target
    def softmax_obj(x):
        c = compute_corr_and_grad(x, beta)[0]
        bc = beta * c
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # project onto sum to correct any drift
    x = project_onto_box_and_sum(x, target)
    return x

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n

    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)

def compute_corr_and_grad(h, beta):
    n = h.shape[0]
    dx = 2.0 / n
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    corr = np.fft.irfft(H * R, m)[:L] * dx   # c values (integrals)
    c = corr
    bc = beta * c
    max_bc = np.max(bc)
    w = np.exp(bc - max_bc)
    w_sum = w.sum()
    w /= w_sum
    # gradient
    L2 = L + n - 1
    m2 = 1 << ((L2 - 1).bit_length())
    # conv1 = w_rev conv (1-h)
    W_rev_fft = np.fft.rfft(w[::-1], m2)
    O_fft = np.fft.rfft(1 - h, m2)
    conv1 = np.fft.irfft(W_rev_fft * O_fft, m2)[:L2]
    # conv2 = w conv h
    W_fft = np.fft.rfft(w, m2)
    H_fft = np.fft.rfft(h, m2)
    conv2 = np.fft.irfft(W_fft * H_fft, m2)[:L2]
    start = n - 1
    grad_corr = conv1[start:start + n] - conv2[start:start + n]   # derivative of corr raw
    grad = grad_corr * dx
    return c, w, grad

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper: projection onto the box [0,1] while fixing the sum.
# --------------------------------------------------------------
def _proj_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v to [0,1]^n with Σx = target (capped simplex)."""
    low = -1.0 - np.max(v)      # forces all entries to 0 after clipping
    high = 1.0 - np.min(v)      # forces all entries to 1 after clipping
    for _ in range(30):         # bisection; 30 iterations give ~1e-9 accuracy
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)


# --------------------------------------------------------------
# Random feasible vector (box + sum)
# --------------------------------------------------------------
def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return _proj_box_sum(v, n / 2.0)


# --------------------------------------------------------------
# Exact discrete C5 (max correlation)
# --------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L] * dx
    return conv.max()


# --------------------------------------------------------------
# Correlation, soft‑max weights and gradient of the smooth objective.
# --------------------------------------------------------------
def _corr_and_grad(h: np.ndarray, beta: float):
    """Return (c, w, grad) where
    c  = full correlation array (integral values),
    w  = soft‑max weights ∝ exp(beta*c),
    grad = ∂/∂h ( (1/β) log∑exp(βc) ).
    """
    n = h.shape[0]
    dx = 2.0 / n
    rev = (1.0 - h)[::-1]

    # ----- correlation c (full, already scaled by dx) -----
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    c = np.fft.irfft(H * R, m)[:L] * dx

    # ----- soft‑max weights w -----
    bc = beta * c
    bmax = bc.max()
    w = np.exp(bc - bmax)
    w /= w.sum()                     # sum(w) = 1

    # ----- gradient of the smooth objective -----
    L2 = L + n - 1
    m2 = 1 << ((L2 - 1).bit_length())
    # w_rev * (1-h)
    W_rev_fft = np.fft.rfft(w[::-1], m2)
    O_fft = np.fft.rfft(1.0 - h, m2)
    conv1 = np.fft.irfft(W_rev_fft * O_fft, m2)[:L2]
    # w * h
    W_fft = np.fft.rfft(w, m2)
    H_fft = np.fft.rfft(h, m2)
    conv2 = np.fft.irfft(W_fft * H_fft, m2)[:L2]

    start = n - 1
    grad_corr = conv1[start:start + n] - conv2[start:start + n]   # ∂corr/∂h
    grad = grad_corr * dx                                        # ∂c/∂h
    return c, w, grad


# --------------------------------------------------------------
# SLSQP refinement of the smooth surrogate (optional final step)
# --------------------------------------------------------------
def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 150) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def _eq_sum(x):
        return np.sum(x) - target

    def _soft_obj(x):
        c, _, _ = _corr_and_grad(x, beta)   # full correlation
        bc = beta * c
        m = bc.max()
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    cons = {"type": "eq", "fun": _eq_sum}
    res = minimize(
        _soft_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return _proj_box_sum(x, target)


# --------------------------------------------------------------
# Main driver
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function on [0,2] with low C₅."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- choose discretisation size -----
    n = 960  # higher resolution than the reference 360‑point construction

    # ----- initialise from parent if present, otherwise a smooth bump -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        if n0 != n:
            old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            h = np.interp(new_x, old_x, h0)
        else:
            h = h0.copy()
    else:
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h = 0.5 + 0.3 * np.sin(np.pi * xs) + 0.2 * np.cos(2 * np.pi * xs)

    h = _proj_box_sum(h, n / 2.0)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- continuation schedule (smooth‑max beta) -----
    betas = [5, 20, 80, 320, 1280, 5120]
    # step sizes roughly scale like 1/√beta
    steps = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08]
    iters_per_beta = 600

    # ----- main stochastic optimisation loop -----
    while time.time() - start_time < budget_s * 0.9:
        # start from a perturbed best or a fresh random point
        if rng.random() < 0.6:
            cand = best_h + rng.normal(scale=0.05, size=n)
        else:
            cand = random_feasible_vector(n, rng)
        cand = _proj_box_sum(cand, n / 2.0)

        for beta, base_step in zip(betas, steps):
            step = base_step
            for _ in range(iters_per_beta):
                if time.time() - start_time > budget_s * 0.95:
                    break
                c, _, grad = _corr_and_grad(cand, beta)
                cand = _proj_box_sum(cand - step * grad, n / 2.0)
                cur_c = c.max()
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = cand.copy()
            # slightly shrink step before next β
            step *= 0.85

        # ----- short SLSQP polish at a very high β -----
        if time.time() - start_time < budget_s * 0.95:
            try:
                refined = slsqp_optimize(best_h, beta=8000, max_iter=200)
                refined_c = compute_c5(refined)
                if refined_c < best_c:
                    best_c, best_h = refined_c, refined.copy()
            except Exception:
                pass

        if time.time() - start_time > budget_s * 0.95:
            break

    # final clean‑up
    best_h = _proj_box_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
