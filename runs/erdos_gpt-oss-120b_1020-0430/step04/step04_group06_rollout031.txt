def gradient_optimize(h0, rng, time_limit, max_iters=5000, step0=0.5, step_decay=0.9995, max_beta=5000):
    h = np.clip(h0.copy(), 0.0, 1.0)
    target = h.size / 2.0
    # enforce sum to target
    h = enforce_sum(h, target, rng)
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    for it in range(1, max_iters+1):
        if time.time() - start >= time_limit:
            break
        # compute f
        f = np.correlate(h, 1.0 - h, mode='full')
        # beta schedule
        beta = max_beta * min(1.0, it / (max_iters // 2))  # ramp up, reaches max at half iters?
        # compute weights
        if beta > 0:
            max_val = np.max(beta * f)
            w = np.exp(beta * f - max_val)
            w_sum = np.sum(w)
            w /= w_sum
        else:
            w = np.full_like(f, 1.0 / f.size)
        # compute conv1 and conv2
        # conv full length = len(w) + n - 1 = (2n-1)+n-1 = 3n-2
        conv_full = np.convolve(w, h, mode='full')
        center = n - 1
        conv1 = conv_full[center:center + n]
        w_rev = w[::-1]
        conv_full_rev = np.convolve(w_rev, h, mode='full')
        conv2 = conv_full_rev[center:center + n]
        # gradient
        g = 1.0 - (conv1 + conv2)
        # step size
        step = step0 * (step_decay ** it)
        # update
        h = h - step * g
        # enforce bounds and sum
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
        # evaluate
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def enforce_sum(h, target, rng, max_iter=5000):
    """Clamp h to [0,1] and adjust to have exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Continuous feasible vector with Σh = n/2."""
    h = np.full(n, 0.5)

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Exact C5 for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def max_overlap_and_shift(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Specialised local search moving mass from i to i+shift to lower max overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    final_c = compute_c5(best_h)
    return best_h, final_c


def gradient_optimize(h0, rng, time_limit, max_iters=5000,
                      step0=0.5, step_decay=0.9995, max_beta=5000):
    """Gradient descent on smooth max objective with projection onto box and sum."""
    h = np.clip(h0.copy(), 0.0, 1.0)
    n = h.size
    target = n / 2.0
    h = enforce_sum(h, target, rng)
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    for it in range(1, max_iters + 1):
        if time.time() - start >= time_limit:
            break
        f = np.correlate(h, 1.0 - h, mode="full")
        beta = max_beta * min(1.0, it / (max_iters // 2))
        if beta > 0:
            max_val = np.max(beta * f)
            w = np.exp(beta * f - max_val)
            w /= w.sum()
        else:
            w = np.full_like(f, 1.0 / f.size)
        # convolution w * h
        conv_full = np.convolve(w, h, mode="full")
        center = n - 1
        conv1 = conv_full[center:center + n]
        w_rev = w[::-1]
        conv_full_rev = np.convolve(w_rev, h, mode="full")
        conv2 = conv_full_rev[center:center + n]
        g = 1.0 - (conv1 + conv2)
        step = step0 * (step_decay ** it)
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- Determine n and initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    target = n / 2.0

    # ---- Build diverse candidates ----
    candidates = [h0.copy()]
    for _ in range(4):
        candidates.append(random_binary_vector(n, rng))
    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in (1, 2, 3, 4):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, target, rng)
            candidates.append(h)

    # ---- Pick best initial candidate ----
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # ---- Gradient‑descent stage (≈70% of budget) ----
    grad_time = budget_s * 0.70
    while time.time() - start_time < grad_time:
        remaining = grad_time - (time.time() - start_time)
        if remaining < 5.0:
            break
        # start point: perturb current best or fresh binary
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_binary_vector(n, rng)
        # allocate a chunk (max 120 s) for this run
        chunk = min(remaining, 120.0)
        h_opt, c_opt = gradient_optimize(start_vec, rng, chunk,
                                         max_iters=8000, step0=0.5,
                                         step_decay=0.9995, max_beta=5000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # ---- Binary rounding test ----
    bin_h = np.zeros_like(best_h)
    idx = np.argsort(-best_h)[:int(target)]
    bin_h[idx] = 1.0
    c_bin = compute_c5(bin_h)
    if c_bin < best_c:
        best_c = c_bin
        best_h = bin_h.copy()

    # ---- Pair‑transfer fine‑tuning with remaining budget ----
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        h_opt, c_opt = pair_transfer_optimize(best_h, rng,
                                              max_iters=15000,
                                              time_limit=remaining * 0.95,
                                              init_factor=0.5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # ---- Final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
