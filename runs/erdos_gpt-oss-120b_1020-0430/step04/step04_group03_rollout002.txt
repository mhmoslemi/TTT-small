def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr

import time
import numpy as np
from scipy.optimize import minimize

# ----- basic utilities -------------------------------------------------
def enforce_sum(h, target, rng):
    """Clip values to [0,1] and adjust to make the sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def random_binary_vector(n, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    """Discrete C5 for vector h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

# ----- improved greedy optimizer ---------------------------------------
def greedy_optimize(h, max_steps=2000):
    """Transfer mass between two points to reduce the worst overlap."""
    n = h.size
    cur_c = compute_c5(h)
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)

        # candidates respecting bounds
        feas_i = np.where(h > 1e-12)[0]
        feas_j = np.where(h < 1.0 - 1e-12)[0]
        if feas_i.size == 0 or feas_j.size == 0:
            break

        i = feas_i[np.argmax(grad[feas_i])]
        j = feas_j[np.argmin(grad[feas_j])]
        grad_diff = grad[i] - grad[j]
        if grad_diff <= 1e-12:
            # fallback to sign‑based selection
            pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
            neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
            if pos.size == 0 or neg.size == 0:
                break
            i = pos[np.argmax(grad[pos])]
            j = neg[np.argmin(grad[neg])]
            grad_diff = grad[i] - grad[j]
            if grad_diff <= 1e-12:
                break

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break

        improved = False
        for factor in (1.0, 0.5, 0.25):
            eps = max_eps * factor
            if eps < 1e-14:
                continue
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            new_c = compute_c5(nh)
            if new_c + 1e-12 < cur_c:
                h = nh
                cur_c = new_c
                improved = True
                break
        if not improved:
            break
    return h

# ----- smooth‑max SLSQP polish -----------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-13, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_binary_vector(360, rng)
    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), h0.size / 2.0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- stochastic greedy phase -----
    while time.time() - start_time < budget_s * 0.75:
        # choose a start point
        r = rng.random()
        if r < 0.4:
            start_vec = random_binary_vector(best_h.size, rng)
        elif r < 0.7:
            start_vec = random_feasible_vector(best_h.size, rng)
        else:
            start_vec = perturb_random(best_h, rng, scale=0.4)

        local = greedy_optimize(start_vec, max_steps=4000)
        c_local = compute_c5(local)
        if c_local < best_c:
            best_h, best_c = local, c_local

    # ----- SLSQP polishing -----
    for beta in (500, 2000, 8000, 20000):
        if time.time() - start_time > budget_s * 0.95:
            break
        refined = slsqp_optimize(best_h, beta, max_iter=1200)
        c_ref = compute_c5(refined)
        if c_ref < best_c:
            best_h, best_c = refined, c_ref
            # a quick greedy after each polish
            local = greedy_optimize(best_h, max_steps=2000)
            c_local = compute_c5(local)
            if c_local < best_c:
                best_h, best_c = local, c_local

    # ----- final micro‑perturbations -----
    while time.time() - start_time < budget_s * 0.99:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
