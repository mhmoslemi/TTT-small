    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    # parameters
    max_iter = 200000  # large but break on time
    T0 = 0.1
    Tf = 1e-6
    for it in range(max_iter):
        if time.time() - start_time > time_budget:
            break
        # compute T
        T = T0 * (Tf/T0) ** (it / max_iter)
        # choose move
        # maybe with 70% targeted swap, 30% random swap
        if rng.random() < 0.7:
            # targeted swap based on worst lag
            # compute full corr and determine worst lag
            corr = np.correlate(h, 1 - h, mode='full')
            # find lag index of max - (n-1)
            lag = np.argmax(corr) - (n - 1)
            lag = lag % n # ensure positive within [0,n-1]
            # find indices where h[i]==1 and h[(i+lag)%n]==0
            idx1 = np.where((h == 1) & (np.roll(h, -lag) == 0))[0]
            if idx1.size > 0:
                i = rng.choice(idx1)
                j = (i + lag) % n
                # swap
                new_h = h.copy()
                new_h[i] = 0
                new_h[j] = 1
            else:
                # fallback random swap
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while i == j or h[i] == h[j]:
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                new_h = h.copy()
                new_h[i], new_h[j] = h[j], h[i]
        else:
            # random swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while i == j or h[i] == h[j]:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            new_h = h.copy()
            new_h[i], new_h[j] = h[j], h[i]

def targeted_binary_swap(h, rng):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    lag = np.argmax(corr) - (n - 1)
    lag = lag % n
    # locate indices i where h[i]==1 and h[(i+lag)%n]==0
    shifted = np.roll(h, -lag)  # shift left
    idx = np.where((h == 1) & (shifted == 0))[0]
    if idx.size > 0:

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=1000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            max_inc = 1.0 - h[i]
            if max_inc <= 0:
                continue
            inc = min(max_inc, diff)
        else:
            max_dec = h[i]
            if max_dec <= 0:
                continue
            inc = -min(max_dec, -diff)
        h[i] += inc
        diff = target - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    """Return max_k ∫ h(x)(1−h(x+k)) dx approximated on a uniform grid."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def random_binary_vector(n, rng):
    h = np.zeros(n, dtype=np.uint8)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1
    return h

# ---------- binary simulated annealing ----------
def binary_swap_move(h, rng):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j or h[i] == h[j]:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
    new_h = h.copy()
    new_h[i], new_h[j] = h[j], h[i]
    return new_h

def targeted_binary_swap(h, rng, cur_corr):
    n = h.size
    lag = np.argmax(cur_corr) - (n - 1)
    lag = lag % n
    shifted = np.roll(h, -lag)
    idx = np.where((h == 1) & (shifted == 0))[0]
    if idx.size:
        i = rng.choice(idx)
        j = (i + lag) % n
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        return new_h
    return binary_swap_move(h, rng)

def sa_binary_optimize(h0, rng, time_budget, start_time):
    """Binary SA that preserves exactly half ones."""
    h = h0.copy()
    n = h.size
    cur_corr = np.correlate(h, 1 - h, mode="full")
    cur_c = np.max(cur_corr) * (2.0 / n)
    best_h, best_c = h.copy(), cur_c
    while True:
        if time.time() - start_time > time_budget:
            break
        # temperature from 0.2 down to 0.001 over the allocated budget
        elapsed = time.time() - start_time
        frac = elapsed / time_budget
        T = 0.2 * (0.001 / 0.2) ** frac
        # move
        if rng.random() < 0.7:
            new_h = targeted_binary_swap(h, rng, cur_corr)
        else:
            new_h = binary_swap_move(h, rng)
        new_corr = np.correlate(new_h, 1 - new_h, mode="full")
        new_c = np.max(new_corr) * (2.0 / n)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_corr, cur_c = new_h, new_corr, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- final hill‑climb (fractional) ----------
def final_hill_climb(h, rng, time_budget, start_time):
    n = h.size
    cur_c = compute_c5(h)
    best_c, best_h = cur_c, h.copy()
    step = 0.3
    while time.time() - start_time < time_budget:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        step *= 0.9999
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # ---- initial binary vector ----
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n1 = h_init.size
        half = n1 // 2
        # force exactly half ones by taking the largest half entries
        idx = np.argpartition(-h_init, half - 1)[:half]
        h0_bin = np.zeros(n1, dtype=np.uint8)
        h0_bin[idx] = 1
    else:
        n1 = 360
        h0_bin = random_binary_vector(n1, rng)

    # ---- phase 1: binary SA on coarse grid ----
    time_budget1 = budget_s * 0.35
    best_h1, best_c1 = sa_binary_optimize(h0_bin, rng, time_budget1, t0)

    # ---- phase 2: up‑sample and binary SA on fine grid ----
    n2 = n1 * 2
    best_h2 = np.repeat(best_h1, 2).astype(np.uint8)  # still binary, half ones
    time_budget2 = budget_s * 0.55
    best_h2, best_c2 = sa_binary_optimize(best_h2, rng, time_budget2, t0)

    # ---- phase 3: SLSQP refinement (fractional) ----
    remaining = budget_s - (time.time() - t0)
    if remaining > 5.0:
        h_float = best_h2.astype(float)
        beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000]
        for beta in beta_list:
            if time.time() - t0 >= budget_s:
                break
            h_float = slsqp_optimize_one_beta(h_float, beta, rng, max_iter=500)
            c = compute_c5(h_float)
            if c < best_c2:
                best_c2 = c
                best_h2 = h_float.copy()
        # final fractional hill‑climb if time permits
        remaining = budget_s - (time.time() - t0)
        if remaining > 2.0:
            best_h2, best_c2 = final_hill_climb(best_h2, rng, remaining, t0)

    # ---- clean up and final check ----
    best_h2 = np.clip(best_h2, 0.0, 1.0)
    best_h2 = enforce_sum(best_h2, best_h2.size / 2.0, rng)
    final_c = compute_c5(best_h2)

    return best_h2.tolist(), float(final_c), int(best_h2.size)
