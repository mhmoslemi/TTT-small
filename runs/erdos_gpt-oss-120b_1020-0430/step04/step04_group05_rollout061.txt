def worst_lag_info_and_indices(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    # Compute conflict indices for this lag to use in targeted swap
    if lag >= 0:
        L = lag
        A = np.where((h[:n-L]==1) & (h[L:]==0))[0]
        B = np.where((h[:n-L]==0) & (h[L:]==1))[0]
    else:
        L = -lag
        A = np.where((h[L:]==1) & (h[:n-L]==0))[0] + L
        B = np.where((h[L:]==0) & (h[:n-L]==1))[0] + L
    return lag, A, B, corr.max() * (2.0 / n)  # also return current C5

def binary_simulated_annealing(h_start, rng, time_budget, start_time):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    # SA schedule: linear cooling from T0 to Tf over total allocated time
    T0 = 0.03
    Tf = 1e-5
    while True:
        now = time.time()
        if now - start_time > time_budget:
            break
        # compute temperature based on elapsed proportion of time_budget
        elapsed = now - start_time
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + Tf * frac
        # get worst lag info
        lag, A, B, _ = worst_lag_info_and_indices(h)
        # generate neighbor
        if rng.random() < 0.9:
            h_new = binary_targeted_swap(h, lag, A, B, rng)
        else:
            # random swap
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
        new_c = compute_c5(h_new)
        dE = new_c - cur_c
        if dE < 0 or (T > 1e-12 and rng.random() < np.exp(-dE / T)):
            h = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
    return best_h, best_c

def random_feasible_vector(n, rng):
    # start with half ones at random positions, add small noise, then project
    h = np.zeros(n)
    h[:n//2] = 1.0
    rng.shuffle(h)
    # add small random perturbation
    h = h.astype(float)
    # random small adjustments while staying within [0,1] and sum target.
    # We'll perform random swaps between entries with small epsilon.
    for _ in range(n*3):
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i==j:
            continue
        # transfer epsilon from i to j
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    # enforce sum exactly n/2
    target = n/2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # uniformly adjust
        h += diff / n
        h = np.clip(h,0.0,1.0)
        # adjust again if needed
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # iterative correction

import time
import numpy as np
from scipy.optimize import minimize

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    h = h.astype(float)
    for _ in range(n * 3):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    target = n / 2.0
    h = enforce_sum(h, target, rng)
    return h

def random_binary_vector(n, rng):
    h = np.zeros(n, dtype=int)
    h[:n // 2] = 1
    rng.shuffle(h)
    return h

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * (2.0 / n)

def worst_lag_info_and_indices(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    if lag >= 0:
        L = lag
        A = np.where((h[:n - L] == 1) & (h[L:] == 0))[0]
        B = np.where((h[:n - L] == 0) & (h[L:] == 1))[0]
    else:
        L = -lag
        A = np.where((h[L:] == 1) & (h[:n - L] == 0))[0] + L
        B = np.where((h[L:] == 0) & (h[:n - L] == 1))[0] + L
    return lag, A, B, corr.max() * (2.0 / n)

def binary_targeted_swap(h, lag, A, B, rng):
    if A.size and B.size:
        i = rng.choice(A)
        j = rng.choice(B)
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        return h_new
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if ones.size and zeros.size:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        return h_new
    return h.copy()

def binary_simulated_annealing(h_start, rng, time_budget, start_time):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.03, 1e-5
    while True:
        now = time.time()
        if now - start_time > time_budget:
            break
        elapsed = now - start_time
        frac = elapsed / time_budget
        T = T0 * (1 - frac) + Tf * frac
        lag, A, B, _ = worst_lag_info_and_indices(h)
        if rng.random() < 0.9:
            h_new = binary_targeted_swap(h, lag, A, B, rng)
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
        new_c = compute_c5(h_new)
        dE = new_c - cur_c
        if dE < 0 or (T > 0 and rng.random() < np.exp(-dE / T)):
            h, cur_c = h_new, new_c
            if new_c < best_c:
                best_c, best_h = new_c, h_new.copy()
    return best_h, best_c

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(100):
            add = diff / n
            x += add
            x = np.clip(x, 0.0, 1.0)
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def subgradient_descent(h0, target, rng, time_budget, start_time):
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    it = 0
    while True:
        now = time.time()
        if now - start_time > time_budget:
            break
        lag, _, _, cur_c = worst_lag_info_and_indices(h)
        n = h.size
        g = np.zeros(n)
        if lag >= 0:
            L = lag
            g[:n - L] += 1 - h[L:]
            g[L:] -= h[:n - L]
        else:
            L = -lag
            g[L:] += 1 - h[:n - L]
            g[:n - L] -= h[L:]
        step = 0.5 / np.sqrt(it + 1)
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            for _ in range(10):
                h_new += diff / n
                h_new = np.clip(h_new, 0.0, 1.0)
                diff = target - h_new.sum()
                if abs(diff) < 1e-12:
                    break
        new_c = compute_c5(h_new)
        if new_c < best_c - 1e-12:
            best_c, best_h = new_c, h_new.copy()
        h = h_new
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    # initial construction
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        h0 = enforce_sum(init, n / 2.0, rng)
        sorted_idx = np.argsort(-h0)
        binary0 = np.zeros(n, dtype=int)
        binary0[sorted_idx[: n // 2]] = 1
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)
        binary0 = random_binary_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)
    best_bin_h, best_bin_c = binary0.copy(), compute_c5(binary0.astype(float))

    total = budget_s
    main_time = total * 0.55
    up_time = total * 0.35
    final_time = total - (main_time + up_time)

    # ---------- main phase: SLSQP + binary SA ----------
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]
    # SLSQP ladder on continuous start
    start_main = time.time()
    h_cont, c_cont = best_h.copy(), best_c
    for beta in beta_list:
        if time.time() - t0 > main_time:
            break
        h_cont = slsqp_optimize_one_beta(h_cont, beta, rng)
        c_cont = compute_c5(h_cont)
        if c_cont < best_c:
            best_c, best_h = c_cont, h_cont.copy()
    # binary SA on best binary
    sa_time = main_time * 0.4
    if time.time() - t0 + sa_time < main_time:
        bin_h, bin_c = binary_simulated_annealing(best_bin_h, rng, sa_time, time.time())
        if bin_c < best_c:
            best_c, best_h = bin_c, bin_h.astype(float)
            best_bin_h, best_bin_c = bin_h, bin_c
    # second SLSQP pass
    for beta in beta_list:
        if time.time() - t0 > main_time:
            break
        best_h = slsqp_optimize_one_beta(best_h, beta, rng)
        best_c = compute_c5(best_h)

    # ---------- up‑sampling ----------
    if time.time() - t0 < total - final_time - 0.5:
        target_n = max(720, n * 2)
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, target_n / 2.0, rng)
        # SLSQP on finer grid
        for beta in beta_list[:5]:
            if time.time() - t0 > total - final_time:
                break
            up_h = slsqp_optimize_one_beta(up_h, beta, rng)
        best_h, best_c = up_h, compute_c5(up_h)
        n = target_n

    # ---------- final sub‑gradient refinement ----------
    remaining = total - (time.time() - t0)
    if remaining > 0.5:
        best_h, best_c = subgradient_descent(best_h, n / 2.0, rng, remaining, time.time())

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
