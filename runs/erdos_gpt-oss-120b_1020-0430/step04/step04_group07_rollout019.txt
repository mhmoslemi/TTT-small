import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def proj_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {x: 0≤x≤1, sum x = target}."""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # redistribute greedily
        idx = np.where((diff > 0) & (x < 1.0))[0]
        if diff > 0 and idx.size:
            add = min(diff, 1.0 - x[idx[0]])
            x[idx[0]] += add
            diff -= add
        elif diff < 0:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                sub = min(-diff, x[idx[0]])
                x[idx[0]] -= sub
                diff += sub
    return x


def worst_shift_and_gradient(h: np.ndarray):
    """Return index of worst shift, its correlation value, and gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr) - (n - 1)
    k = max_idx
    grad = np.zeros_like(h)
    if k >= 0:
        grad[: n - k] += 1.0 - h[k:]
        grad[k:]      -= h[: n - k]
    else:
        d = -k
        grad[d:]      += 1.0 - h[: n - d]
        grad[: n - d] -= h[d:]
    return k, corr.max() * (2.0 / n), grad


def pgd_search(h: np.ndarray, max_iters: int, step0: float, target: float):
    """Projected gradient descent on the worst‑shift objective."""
    n = h.size
    step = step0
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    for _ in range(max_iters):
        _, _, grad = worst_shift_and_gradient(h)
        h_new = h - step * grad
        h_new = proj_box_simplex(h_new, target)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if c_new <= cur_c:
            h, cur_c = h_new, c_new
            step *= 1.02
        else:
            h, cur_c = h_new, c_new
            step *= 0.7
        if step < 1e-6:
            step = 1e-6
    return best_h, best_c


def slsqp_optimize_beta(h: np.ndarray, beta: float, target: float,
                       max_iter: int, rng: np.random.Generator):

    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    def sum_eq(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * h.size
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(softmax_obj, h, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, h.size)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function giving a low C5 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # use parent construction if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = proj_box_simplex(h0, h0.size / 2.0)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    n_candidates = [128, 192, 256, 384, 512, 768]

    # main multi‑restart loop
    while time.time() - start < 0.85 * budget_s:
        n = int(rng.choice(n_candidates))
        target = n / 2.0

        # initialise a feasible start
        if best_h is not None:
            if best_n == n:
                h = best_h.copy()
            else:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = proj_box_simplex(h, target)
        else:
            if rng.random() < 0.5:
                h = np.zeros(n)
                idx = rng.choice(n, n // 2, replace=False)
                h[idx] = 1.0
            else:
                h = rng.random(n)
                h = proj_box_simplex(h, target)

        # projected gradient descent
        iters = max(200, int(2000 * 256 / n))
        h_pg, c_pg = pgd_search(h, iters, step0=0.5, target=target)
        if c_pg < best_c:
            best_h, best_c, best_n = h_pg.copy(), c_pg, n

        # short SLSQP refinement (lightweight)
        beta_schedule = (20., 80., 200., 800., 3000., 8000.)
        for beta in beta_schedule:
            if time.time() - start > 0.94 * budget_s:
                break
            maxit = 500 if beta < 200 else 300
            h_ref = slsqp_optimize_beta(best_h, beta, target, maxit, rng)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref
                best_n = n

    # final fine‑tuning with random transfers
    while time.time() - start < budget_s * 0.998 and best_h is not None:
        i, j = rng.integers(0, best_n, size=2)
        if i == j:
            continue
        nh = best_h.copy()
        eps = rng.random() * 0.04
        eps = min(eps, nh[i], 1.0 - nh[j])
        nh[i] -= eps
        nh[j] += eps
        c_new = compute_c5(nh)
        if c_new < best_c:
            best_h, best_c = nh, c_new

    # sanity enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_box_simplex(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
