def softmax_grad(h, beta):
    n = h.size
    out = np.correlate(h, 1.0 - h, mode='full')
    # compute weights
    scaled = beta * out
    # subtract max for numeric stability
    max_s = np.max(scaled)
    exp_scaled = np.exp(scaled - max_s)
    w = exp_scaled / np.sum(exp_scaled)
    # compute term1 and term2 using loops over shifts
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for idx_s, w_s in enumerate(w):
        shift = idx_s - (n - 1)
        if shift >= 0:
            # i from 0..n-1-shift
            i = slice(0, n - shift)
            term1[i] += w_s * (1.0 - h[i + shift])
            term2[i + shift] += w_s * h[i]
        else:
            s = -shift
            i = slice(s, n)
            term1[i] += w_s * (1.0 - h[i - s])
            term2[i - s] += w_s * h[i]
    grad = term1 - term2
    return grad

def project_to_feasible(h, target):
    # clip
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # free indices where we can move mass
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        free = np.count_nonzero(mask)
        if free == 0:
            break
        delta = diff / free
        # apply delta but keep within bounds
        h[mask] += delta
        # clip after addition
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

This is deterministic and quick.

def softmax_grad(h, beta):
    n = h.size
    out = np.correlate(h, 1.0 - h, mode='full')
    # compute weights
    scaled = beta * out
    max_s = np.max(scaled)
    exp_scaled = np.exp(scaled - max_s)
    w = exp_scaled / np.sum(exp_scaled)
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    # iterate over idx_s and w_s
    for idx_s, w_s in enumerate(w):
        shift = idx_s - (n - 1)
        if shift >= 0:
            # indices 0 .. n-1-shift
            if n - shift <= 0:
                continue
            i0 = 0
            i1 = n - shift
            term1[i0:i1] += w_s * (1.0 - h[i0 + shift:i1 + shift])
            term2[i0 + shift:i1 + shift] += w_s * h[i0:i1]
        else:
            s = -shift
            if n - s <= 0:
                continue
            i0 = s
            i1 = n
            term1[i0:i1] += w_s * (1.0 - h[i0 - s:i1 - s])
            term2[i0 - s:i1 - s] += w_s * h[i0:i1]
    grad = (term1 - term2) / beta
    return grad

import time
import numpy as np

# ----------------------------------------------------------------------
#  Basic utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet exact sum=target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple water‑filling
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        if diff > 0:
            inc = np.minimum(delta, 1.0 - h[free])
            h[free] += inc
        else:
            dec = np.maximum(delta, -h[free])
            h[free] += dec
        diff = target - h.sum()
    # final tiny correction if needed
    if abs(diff) > 1e-12:
        for i in range(h.size):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector projected onto the box and sum constraints."""
    h = rng.random(n)
    return project_to_feasible(h, n / 2.0)


# ----------------------------------------------------------------------
#  Smooth surrogate gradient (log‑sum‑exp of the overlap)
# ----------------------------------------------------------------------
def softmax_grad(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (log‑sum‑exp(beta*corr))/beta w.r.t. h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr
    max_s = np.max(scaled)
    exp_scaled = np.exp(scaled - max_s)
    w = exp_scaled / np.sum(exp_scaled)  # softmax weights, sum = 1

    term1 = np.zeros(n, dtype=np.float64)
    term2 = np.zeros(n, dtype=np.float64)

    # iterate over all shifts
    for idx, ws in enumerate(w):
        shift = idx - (n - 1)
        if shift >= 0:
            end = n - shift
            if end <= 0:
                continue
            term1[:end] += ws * (1.0 - h[shift:shift + end])
            term2[shift:shift + end] += ws * h[:end]
        else:
            s = -shift
            start = s
            length = n - s
            term1[start:] += ws * (1.0 - h[:length])
            term2[:length] += ws * h[start:]
    grad = (term1 - term2) / beta
    return grad


# ----------------------------------------------------------------------
#  Projected gradient descent on the smooth surrogate
# ----------------------------------------------------------------------
def gradient_descent_optimize(
    h0: np.ndarray,
    target_sum: float,
    beta_schedule,
    max_iters_per_beta: int = 2500,
    step0: float = 0.5,
    time_limit: float = None,
):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    for beta in beta_schedule:
        step = step0
        for it in range(max_iters_per_beta):
            if time_limit is not None and (time.time() - start) > time_limit:
                return best_h, best_c
            grad = softmax_grad(h, beta)
            h = h - step * grad
            h = project_to_feasible(h, target_sum)
            step *= 0.9995
            if (it & 31) == 0:  # evaluate every 32 steps
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    return best_h, best_c


# ----------------------------------------------------------------------
#  Simple pair‑transfer local search (targets the worst shift)
# ----------------------------------------------------------------------
def simple_pair_transfer_optimize(
    h0: np.ndarray, max_iters: int = 20000, time_limit: float = None
):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    for _ in range(max_iters):
        if time_limit is not None and (time.time() - start) > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        if shift >= 0:
            i = np.arange(n - shift)
            j = i + shift
        else:
            i = np.arange(-shift, n)
            j = i + shift

        if i.size == 0:
            continue
        prod = h[i] * (1.0 - h[j])
        idx_max = int(np.argmax(prod))
        ii = int(i[idx_max])
        jj = int(j[idx_max])

        max_eps = min(h[ii], 1.0 - h[jj])
        if max_eps <= 0.0:
            continue
        eps = 0.5 * max_eps
        h[ii] -= eps
        h[jj] += eps

        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- 1. Determine discretisation size --------------------------------
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
        n0 = base_h.size
        # use double resolution if it stays comfortably below 1000
        if n0 <= 500:
            n = n0 * 2
            h0 = np.repeat(base_h, 2)
        else:
            n = n0
            h0 = base_h.copy()
        target_sum = n / 2.0
        h0 = project_to_feasible(h0, target_sum)
    else:
        n = 720
        target_sum = n / 2.0
        h0 = random_feasible_vector(n, rng)

    # ---- 2. Build a modest pool of diverse candidates --------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0.copy()]

    for f in range(1, 6):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            candidates.append(project_to_feasible(h, target_sum))

    candidates.append(project_to_feasible(xs / 2.0, target_sum))
    candidates.append(project_to_feasible(1.0 - xs / 2.0, target_sum))

    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(project_to_feasible(mask, target_sum))

    mask = np.zeros(n)
    start = n // 4
    mask[start : start + n // 2] = 1.0
    candidates.append(project_to_feasible(mask, target_sum))

    mask = np.arange(n) % 2
    candidates.append(project_to_feasible(mask, target_sum))

    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    # ---- 3. Pick the best among the pool --------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # ---- 4. Gradient‑descent refinement (smooth max) --------------------
    remaining = budget_s - (time.time() - start_time)
    gd_time = remaining * 0.7
    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000]

    # main descent from the current best
    gd_h, gd_c = gradient_descent_optimize(
        best_h,
        target_sum,
        beta_schedule,
        max_iters_per_beta=3000,
        step0=0.5,
        time_limit=gd_time,
    )
    if gd_c < best_c:
        best_h, best_c = gd_h, gd_c

    # a couple of random restarts
    remaining = budget_s - (time.time() - start_time)
    restart_time = remaining * 0.5
    restarts = 2
    per_restart = restart_time / max(1, restarts)
    for _ in range(restarts):
        h_start = random_feasible_vector(n, rng)
        h_opt, c_opt = gradient_descent_optimize(
            h_start,
            target_sum,
            beta_schedule,
            max_iters_per_beta=2500,
            step0=0.5,
            time_limit=per_restart,
        )
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- 5. Final pair‑transfer fine‑tuning ------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        h_opt, c_opt = simple_pair_transfer_optimize(
            best_h, max_iters=20000, time_limit=remaining
        )
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ---- 6. Clean‑up ----------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
