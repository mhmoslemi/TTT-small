    zeros_idx = np.flatnonzero(h == 0)

    # We can maintain these arrays for quick selection; after each swap adjust.

    while time.time() < time_deadline:

        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        if shift < 0:

        if shift == 0:
            # No overlap bigger than zero; can't improve further
            break

import time
import numpy as np
from scipy.optimize import minimize

# ----- globals for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    """max overlap for periodic shift on [0,2] with n samples."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """smooth max of overlaps using log-sum-exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def adjust_sum_exact(h, rng):
    """Force Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()

        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])

    return h

def gradient_refine(h, rng, step=0.02, max_iter=20):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:                     # never happens for max, but keep safe
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]       # + (1 - h[i+k])
            g[k:] -= h[:up]            # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

            i2 = rng.choice(ones)
            j2 = rng.choice(zeros)
            d = neigh[i2] - neigh[j2]
            if d < 0:
                h[i2] = 0.0
                h[j2] = 1.0
                improved = True
                break
        if not improved:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # ----- base size -----
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
    else:
        parent = None
        n = 360

    best_h = None
    best_c = np.inf
    best_n = None

    # ----- generate start vectors -----
    starts = []

    if parent is not None:
        starts.append(parent.copy())
        starts.append(fractional_to_binary(parent, rng))

    # random binary seeds
    for _ in range(3):
        hb = np.zeros(n)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(hb)

    # random fractional seeds
    for _ in range(3):
        hf = rng.random(n)
        hf *= (n / 2) / hf.sum()
        hf = np.clip(hf, 0.0, 1.0)
        adjust_sum_exact(hf, rng)
        starts.append(hf)

    # heavy‑tailed fractional seeds
    for _ in range(2):
        hf = rng.beta(0.5, 0.5, size=n)
        hf *= (n / 2) / hf.sum()
        hf = np.clip(hf, 0.0, 1.0)
        adjust_sum_exact(hf, rng)
        starts.append(hf)

    # ----- main optimisation loop -----
    betas = (5., 20., 80., 200., 500., 1000., 2000., 5000., 10000., 20000.)
    for base in starts:
        if time.time() > deadline:
            break
        h = np.clip(base, 0.0, 1.0)
        adjust_sum_exact(h, rng)

        # SLSQP cascade
        for beta in betas:
            if time.time() > deadline:
                break
            max_iter = 1200 if beta < 200 else 1800
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # gradient polishing
        h = gradient_refine(h, rng, step=0.025, max_iter=20)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # binary greedy improvement
        hb = fractional_to_binary(h, rng)
        hb = binary_targeted_improve(hb, rng, deadline)
        c_b = compute_c5(hb)
        if c_b < best_c:
            best_h, best_c, best_n = hb.copy(), c_b, n

    # ----- optional up‑sampling refinement -----
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 720
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        for beta in betas:
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=1800)
        h_up = gradient_refine(h_up, rng, step=0.025, max_iter=30)
        h_up = binary_targeted_improve(fractional_to_binary(h_up, rng), rng, deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ----- fallback -----
    if best_h is None:
        best_n = 360
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
