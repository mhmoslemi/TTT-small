import time
import numpy as np

# --------------------------------------------------------------
# Correlation utilities (exact integer version)
# --------------------------------------------------------------
def compute_conv_full_binary(h: np.ndarray) -> np.ndarray:
    """Exact integer correlation conv[k] = Σ h[i]*(1-h[i+k]) (full mode)."""
    # using np.correlate which implements the required linear correlation
    return np.correlate(h.astype(np.int8), (1 - h).astype(np.int8), mode="full")


def conv_shift(h: np.ndarray, d: int) -> int:
    """Exact conv value for a single shift d using the current binary h."""
    n = h.size
    if d >= 0:
        return int(np.dot(h[: n - d], 1 - h[d:]))
    dp = -d
    return int(np.dot(h[dp:], 1 - h[: n - dp]))


# --------------------------------------------------------------
# Binary simulated annealing / hill climbing
# --------------------------------------------------------------
def binary_anneal(h0: np.ndarray, rng: np.random.Generator, deadline: float, target_max: int):
    """Search for a binary vector with small max overlap using fast delta updates."""
    n = h0.size
    # pad for fast neighborhood extraction
    pad = np.concatenate([np.zeros(n - 1, dtype=np.int8), h0, np.zeros(n - 1, dtype=np.int8)])
    shift_off = np.arange(2 * n - 1, dtype=np.int32)

    # initial conv and best record
    conv = compute_conv_full_binary(h0).astype(np.int32)
    center = n - 1
    conv[center] = 0
    best_h = h0.copy()
    best_max = int(conv.max())
    best_conv = conv.copy()

    # keep lists of indices for O(1) random pick & swap
    one_pos = np.full(n, -1, dtype=np.int32)
    zero_pos = np.full(n, -1, dtype=np.int32)
    one_list = []
    zero_list = []
    for i, val in enumerate(h0):
        if val:
            one_pos[i] = len(one_list)
            one_list.append(i)
        else:
            zero_pos[i] = len(zero_list)
            zero_list.append(i)

    # temperature schedule
    T = max(1.0, best_max * 0.1)
    T_min = 1e-4
    decay = 0.9999

    while time.time() < deadline:
        # pick random 1 and 0 positions
        i = one_list[rng.integers(len(one_list))]
        j = zero_list[rng.integers(len(zero_list))]

        # neighborhood slices (fast O(n) extraction)
        first_i = pad[i : i + 2 * n - 1]
        second_i = pad[i + 2 * (n - 1) - shift_off]
        first_j = pad[j : j + 2 * n - 1]
        second_j = pad[j + 2 * (n - 1) - shift_off]

        delta = (first_j + second_j) - (first_i + second_i)          # int8, range [-2,2]
        delta_i32 = delta.astype(np.int32)

        # apply swap to h and pad (temporarily)
        h0[i] = 0
        h0[j] = 1
        pad[i + n - 1] = 0
        pad[j + n - 1] = 1

        d = j - i

        # recompute the two shifts where neighborhoods overlap
        conv_d = conv_shift(h0, d)
        if d != 0:
            conv_md = conv_shift(h0, -d)
        else:
            conv_md = conv_d

        # tentative new conv
        conv_new = conv - delta_i32
        conv_new[center] = 0
        conv_new[d + center] = conv_d
        if d != 0:
            conv_new[-d + center] = conv_md

        new_max = int(conv_new.max())

        # acceptance
        accept = False
        if new_max < best_max:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_max - best_max) / T):
                accept = True

        if accept:
            # keep changes
            conv = conv_new
            if new_max < best_max:
                best_max = new_max
                best_h = h0.copy()
                best_conv = conv.copy()
                if best_max <= target_max:
                    break
            # update index lists efficiently
            # i leaves ones, joins zeros
            pos_i = one_pos[i]
            last_one = one_list[-1]
            one_list[pos_i] = last_one
            one_pos[last_one] = pos_i
            one_list.pop()
            one_pos[i] = -1

            zero_pos[i] = len(zero_list)
            zero_list.append(i)

            # j leaves zeros, joins ones
            pos_j = zero_pos[j]
            last_zero = zero_list[-1]
            zero_list[pos_j] = last_zero
            zero_pos[last_zero] = pos_j
            zero_list.pop()
            zero_pos[j] = -1

            one_pos[j] = len(one_list)
            one_list.append(j)
        else:
            # revert swap
            h0[i] = 1
            h0[j] = 0
            pad[i + n - 1] = 1
            pad[j + n - 1] = 0

        # temperature decay
        T = max(T_min, T * decay)

    return best_h, best_max

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1  # small safety margin

    # ----------------------------------------------------------
    # Initialise binary vector (exact half ones)
    # ----------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        # take the n/2 largest entries as 1s
        idx = np.argsort(-init)[: n // 2]
        h0 = np.zeros(n, dtype=np.int8)
        h0[idx] = 1
    else:
        n = 720
        h0 = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, size=n // 2, replace=False)
        h0[ones] = 1

    # target: max overlap ≤ 137 (gives C5≈0.38056 < 0.3808)
    target_max = 137

    # ----------------------------------------------------------
    # Run binary simulated annealing / hill climbing
    # ----------------------------------------------------------
    best_h, best_max = binary_anneal(h0, rng, deadline, target_max)

    # ----------------------------------------------------------
    # Final exact C5 computation
    # ----------------------------------------------------------
    final_c5 = float(best_max) * (2.0 / n)
