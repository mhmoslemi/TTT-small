def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L-1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H*R, m)[:L]
    return np.max(conv) * (2.0 / n)

def perturb_h(h, rng, scale):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    # compute max amount we can move from i to j while staying in [0,1]
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure eps in [-max_eps, max_eps]
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp, init_scale, final_scale):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    for i in range(max_iter):
        # linear schedule
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def fast_sa(h, rng, steps, init_temp, final_temp, init_scale, final_scale):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()

    for i in range(steps):

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final clipping if needed
    h = np.clip(h, 0.0, 1.0)
    # recompute diff and maybe adjust uniformly
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute uniformly among free entries
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    # generate random numbers in [0,1], then scale to target sum
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.full(n, 0.0)

        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # fix any discrepancy
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = opt.minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # fix sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple correction like enforce_sum with small iterations
        rng = np.random.default_rng()
        n = x.size
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())  # next power of two ≥ L
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final uniform correction on interior points
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.full(n, 0.0)
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def fast_hill_climb(h_start: np.ndarray, rng: np.random.Generator, steps: int, step_scale: float):
    """Plain hill‑climb: accept only improvements."""
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    for _ in range(steps):
        new_h = perturb_h(best_h, rng, scale=step_scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h
    return best_h, best_c


def fast_sa(
    h_start: np.ndarray,
    rng: np.random.Generator,
    steps: int,
    init_temp: float,
    final_temp: float,
    init_scale: float,
    final_scale: float,
):
    """Simulated‑annealing refinement."""
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    for i in range(steps):
        t = i / steps
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


# ---------- Smooth surrogate & SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- Main driver ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # ---- base resolution ----

        # SLSQP ladder
        for beta in (20, 50, 200, 800, 2000, 5000):

            try:

                c_ref = compute_c5(h_ref)
                if c_ref < best_c:

            except Exception:
                continue
        # occasional random restart
