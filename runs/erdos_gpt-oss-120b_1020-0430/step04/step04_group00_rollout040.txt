def gradient_refine(h, rng, step=0.02, max_iter=15):
    n = h.shape[0]
    target_sum = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        # Compute gradient based on worst shift only (like earlier)
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            upper = n - k
            # at positions i from 0 to upper-1 (0-index)
            # contribution from term h_i*(1 - h_{i+k})
            g[:upper] += 1.0 - h[k:]   # derivative w.r.t h_i: d/dh_i of h_i*(1-h_{i+k}) = 1 - h_{i+k}
            # derivative w.r.t h_{i+k}: -h_i
            g[k:] -= h[:upper]
        # step size decays
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # adjust sum using random reallocation similar to adjust_sum_exact
        diff = target_sum - h.sum()
        # adjust with random indices up to some loops
        if abs(diff) > 1e-12:
            # Use while loop up to maybe 200 steps
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target_sum - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def slsqp_optimize(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n/2.0
    # Define objective: smooth max using log-sum-exp; we can replicate earlier.
    def objective(x):
        # x is array
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    # constraints: sum(x) = target
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    # use scipy.optimize.minimize with method='SLSQP'
    from scipy.optimize import minimize
    res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum exactly: similar to adjust_sum_exact
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    # Use while loop
    while abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])

        diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, rng, max_iter=600):
    n = h0.size
    target = n / 2.0
