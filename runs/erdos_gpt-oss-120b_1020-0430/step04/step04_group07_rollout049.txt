    hi = np.max(y) - lower   # upper bound
    # Actually lambda range: if lambda very negative, y_i - lambda huge => clip to upper => sum = n*upper. If lambda very positive, sum = n*lower.

    hi = np.max(y) - lower

    # We'll bsearch for lambda.
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    # final x
    return np.clip(y - (lo+hi)/2.0, lower, upper)

def line_search(h, grad, target, compute_corr, alpha_init=1.0):
    cur_corr = compute_corr(h)
    cur_val = cur_corr.max()
    alpha = alpha_init
    while alpha > 1e-8:
        y = h - alpha * grad
        y_proj = proj_capped_simplex(y, target)
        val = compute_corr(y_proj).max()
        if val < cur_val - 1e-12:
            return y_proj, val, True
        alpha *= 0.5
    return h, cur_val, False

def optimize_subgradient(h0, target, time_limit):
    start = time.time()
    h = h0.copy()
    corr = full_corr(h)
    best_val = corr.max()
    best_h = h.copy()
    iter_count = 0
    while time.time() - start < time_limit:
        # compute worst shift
        idx = np.argmax(corr)
        k = idx - (h.size - 1)
        k_abs = abs(k)
        # gradient
        n = h.size
        grad = np.ones(n)
        if k_abs == 0:
            grad[:] = 1 - 2 * h
        else:
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs :] -= h[: n - k_abs]
        # line search
        h_new, val_new, improved = line_search(h, grad, target, full_corr, alpha_init=1.0)
        if improved:
            h = h_new
            corr = full_corr(h)
            if val_new < best_val - 1e-12:
                best_val = val_new
                best_h = h.copy()
            # maybe reset no-improve counter
        else:
            # no improvement: random perturb to escape plateau
            # Random small epsilon transfer: pick random i,j and transfer small amount

            y = h.copy()

            if y[i] - eps < 0: eps = y[i]
            if y[j] + eps > 1: eps = 1 - y[j]

                h = y
                best_val = val
                best_h = y.copy()
            # else maybe keep h unchanged
        iter_count += 1
        # optional break after many iterations
    return best_h, best_val

def full_corr(h):
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n-1
    # use rfft for speed
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * np.conj(G), n=size)
    # return first 2*n-1 entries
    return corr[: 2 * n - 1]

def proj_capped_simplex(y, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=50):
    n = y.shape[0]
    # trivial cases
    if target <= lower * n:
        return np.full(n, lower)
    if target >= upper * n:
        return np.full(n, upper)
    lo = np.min(y) - upper
    hi = np.max(y) - lower
    # The function f(lambda) = sum_i clip(y_i - lambda, lower, upper)
    # is decreasing in lambda.
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) / 2.0, lower, upper)

def line_search_step(h, grad, target, compute_corr, alpha_init=1.0):
    cur_val = compute_corr(h).max()
    alpha = alpha_init
    while alpha > 1e-8:
        y = h - alpha * grad
        y = proj_capped_simplex(y, target)
        val = compute_corr(y).max()
        if val < cur_val - 1e-12:
            return y, val, True
        alpha *= 0.5
    return h, cur_val, False

def subgradient_optimize(h0, target, time_limit):
    start = time.time()
    h = h0.copy()
    compute_corr = full_corr
    corr = compute_corr(h)
    best_val = corr.max()
    best_h = h.copy()
    # parameters
    max_no_improve = 20
    no_improve_counter = 0
    while time.time() - start < time_limit:
        idx = np.argmax(corr)
        k = idx - (h.size - 1)
        k_abs = abs(k)
        n = h.size
        grad = np.ones(n)
        if k_abs == 0:
            grad[:] = 1 - 2 * h
        else:
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs:] -= h[: n - k_abs]
        # try line search
        h_new, val_new, improved = line_search_step(h, grad, target, compute_corr)
        if improved:
            h = h_new
            corr = compute_corr(h)
            if val_new < best_val - 1e-12:
                best_val = val_new
                best_h = h.copy()
            no_improve_counter = 0
            continue
        else:
            no_improve_counter += 1
        # occasional random perturb
        if no_improve_counter % 5 == 0:
            # random transfer between two indices
            i, j = np.random.choice(n, 2, replace=False)
            eps = np.random.rand() * 0.02
            # adjust eps to stay within bounds
            eps = min(eps, h[i])          # cannot go below 0 at i
            eps = min(eps, 1 - h[j])      # cannot exceed 1 at j
            if eps > 1e-12:
                h[i] -= eps
                h[j] += eps
                # sum unchanged, bounds satisfied
                corr = compute_corr(h)
                val = corr.max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_h = h.copy()
        # if still no improvement for many iterations, break
        if no_improve_counter > 200:
            break
    return best_h, best_val

import time
import numpy as np

def full_corr(h: np.ndarray) -> np.ndarray:
    """Linear (full) correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * np.conj(G), n=size)
    return corr[: 2 * n - 1]

def proj_capped_simplex(y: np.ndarray, target: float,
                        lower: float = 0.0, upper: float = 1.0,
                        tol: float = 1e-9, max_iter: int = 50) -> np.ndarray:
    """Projection onto {x: lower≤x_i≤upper, Σx_i=target}."""
    n = y.size
    if target <= lower * n:
        return np.full(n, lower)
    if target >= upper * n:
        return np.full(n, upper)
    lo = np.min(y) - upper
    hi = np.max(y) - lower
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) * 0.5, lower, upper)

def subgradient_optimize(h0: np.ndarray, target: float,
                         time_limit: float, rng: np.random.Generator):
    """Sub‑gradient descent with projection and back‑tracking line search."""
    start = time.time()
    h = h0.copy()
    corr = full_corr(h)
    best_val = corr.max()
    best_h = h.copy()
    n = h.size
    no_improve = 0

    while time.time() - start < time_limit:
        idx = np.argmax(corr)
        k = idx - (n - 1)
        k_abs = abs(k)

        grad = np.ones(n)
        if k_abs == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[: n - k_abs] -= h[k_abs:]
            grad[k_abs:] -= h[: n - k_abs]

        cur_val = corr.max()
        alpha = 1.0
        improved = False
        while alpha > 1e-8:
            y = h - alpha * grad
            y = proj_capped_simplex(y, target)
            val = full_corr(y).max()
            if val < cur_val - 1e-12:
                h = y
                corr = full_corr(h)
                cur_val = val
                improved = True
                break
            alpha *= 0.5

        if improved:
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = h.copy()
            no_improve = 0
            continue

        no_improve += 1
        if no_improve % 5 == 0:
            i, j = rng.choice(n, 2, replace=False)
            eps = rng.random() * 0.02
            eps = min(eps, h[i])
            eps = min(eps, 1.0 - h[j])
            if eps > 1e-12:
                h[i] -= eps
                h[j] += eps
                corr = full_corr(h)
                val = corr.max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_h = h.copy()

        if no_improve > 300:
            break

    return best_h, best_val

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_val = np.inf
    best_n = None

    # possible sample sizes (even)
    sizes = [256, 304, 352, 384]

    # main stochastic multi‑restart loop
    while time.time() - start_time < 0.85 * budget_s:
        n = int(rng.choice(sizes))
        fresh = rng.random() < 0.2

        if not fresh and best_h is not None:
            if best_n == n:
                h0 = best_h.copy()
            else:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h0 = np.interp(x_new, x_old, best_h)
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = proj_capped_simplex(h0, n / 2.0)
        else:
            # random start
            if rng.random() < 0.5:
                h0 = np.zeros(n)
                h0[rng.choice(n, n // 2, replace=False)] = 1.0
            else:
                h0 = rng.random(n)
                h0 *= (n / 2.0) / h0.sum()
                h0 = proj_capped_simplex(h0, n / 2.0)

        # allocate a modest chunk of time for each run
        time_left = budget_s - (time.time() - start_time)
        t_alloc = min(5.0, time_left * 0.5)
        h_opt, val_opt = subgradient_optimize(h0, n / 2.0, t_alloc, rng)

        if val_opt < best_val - 1e-12:
            best_h = h_opt.copy()
            best_val = val_opt
            best_n = n

    # final up‑sampling & refinement if time permits
    while time.time() - start_time < 0.95 * budget_s:
        if best_h is None:
            break
        n_up = best_n * 2
        if n_up > 768:
            break
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = proj_capped_simplex(h_up, n_up / 2.0)

        time_left = budget_s - (time.time() - start_time)
        if time_left < 5.0:
            break
        h_up_opt, val_up_opt = subgradient_optimize(
            h_up, n_up / 2.0, time_left * 0.7, rng)

        if val_up_opt < best_val - 1e-12:
            best_h = h_up_opt.copy()
            best_val = val_up_opt
            best_n = n_up
        else:
            break

    # last stochastic tweaking phase
    while time.time() - start_time < budget_s * 0.999:
        if best_h is None:
            break
        n = best_n
        i, j = rng.choice(n, 2, replace=False)
        eps = rng.random() * 0.01
        eps = min(eps, best_h[i])
        eps = min(eps, 1.0 - best_h[j])
        if eps < 1e-12:
            continue
        h_cand = best_h.copy()
        h_cand[i] -= eps
        h_cand[j] += eps
        val = full_corr(h_cand).max()
        if val < best_val - 1e-12:
            best_h = h_cand
            best_val = val

    # final sanity
    if best_h is None:
        best_n = 256
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_capped_simplex(best_h, best_n / 2.0)

    final_c = best_val * (2.0 / best_n)
    return best_h.tolist(), float(final_c), int(best_n)
