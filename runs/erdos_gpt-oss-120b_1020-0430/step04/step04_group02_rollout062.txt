import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


# ---------- continuous hill climbing ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def hill_climb_targeted(
    h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25
):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- binary helpers ----------
def random_swap(h, rng):
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    i = rng.choice(ones)
    j = rng.choice(zeros)
    new = h.copy()
    new[i] = 0
    new[j] = 1
    return new


def targeted_move_binary(h, rng, k):
    n = h.size
    if k <= 0 or k >= n:
        return random_swap(h, rng)
    mask = (h[: n - k] == 1) & (h[k:] == 0)
    if mask.any():
        i = rng.choice(np.flatnonzero(mask))
        new = h.copy()
        new[i] = 0
        new[i + k] = 1
        return new
    return random_swap(h, rng)


def binary_anneal(start_h, rng, max_iters, init_temp, final_temp, p_target=0.3):
    n = start_h.size
    cur = start_h.copy()
    cur_c = compute_c5(cur)
    best_h = cur.copy()
    best_c = cur_c
    for it in range(max_iters):
        t = init_temp * ((final_temp / init_temp) ** (it / max_iters))
        _, idx = compute_c5_and_idx(cur)
        k_star = idx - (n - 1)
        if rng.random() < p_target:
            cand = targeted_move_binary(cur, rng, k_star)
        else:
            cand = random_swap(cur, rng)
        cand_c = compute_c5(cand)
        delta = cand_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(t, 1e-12)):
            cur, cur_c = cand, cand_c
            if cand_c < best_c:
                best_h, best_c = cand.copy(), cand_c
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial h (float)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_box_sum(h0, n / 2.0)

    # create a feasible binary start from h0
    binary_h = (h0 >= 0.5).astype(float)
    target_ones = n // 2
    current_ones = int(binary_h.sum())
    diff = current_ones - target_ones
    if diff > 0:
        # turn diff ones to zero (choose those closest to 0.5)
        ones_idx = np.where(binary_h == 1)[0]
        order = np.argsort(np.abs(h0[ones_idx] - 0.5))
        binary_h[ones_idx[order[:diff]]] = 0.0
    elif diff < 0:
        zeros_idx = np.where(binary_h == 0)[0]
        order = np.argsort(np.abs(h0[zeros_idx] - 0.5))
        binary_h[zeros_idx[order[: -diff]]] = 1.0
    binary_h = binary_h.astype(float)

    best_h = binary_h.copy()
    best_c = compute_c5(best_h)

    # ----- binary annealing -----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0.1 * budget_s:
        max_bin_iters = int(min(20000, remaining * 30))
        bin_h, bin_c = binary_anneal(
            best_h,
            rng,
            max_iters=max_bin_iters,
            init_temp=0.03,
            final_temp=1e-5,
            p_target=0.35,
        )
        if bin_c < best_c:
            best_h, best_c = bin_h.copy(), bin_c

    # ----- continuous refinement & up‑sampling -----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0.2 * budget_s:
        # possibly increase resolution
        target_n = n
        if remaining > 0.5 * budget_s:
            target_n = min(720, max(540, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = project_box_sum(up_h, target_n / 2.0)
            best_h = up_h
            n = target_n

        # hill climbing (more iterations)
        max_hc_iters = int(min(8000, remaining * 5))
        best_h, best_c = hill_climb_targeted(
            best_h, rng, max_iters=max_hc_iters, init_step=0.4
        )

        # SLSQP cascade
        beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                h_ref = slsQP_optimize = slsqp_optimize(best_h, beta, max_iter=800)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # final projection & evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
