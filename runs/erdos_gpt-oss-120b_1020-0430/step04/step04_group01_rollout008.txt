def simulated_annealing(h0, rng, deadline, start_temp_factor=0.02, min_temp_factor=1e-6, max_iters=500000):
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    n = len(h)
    # compute max possible epsilon per pair: depends on values.
    T0 = c * start_temp_factor + 1e-9
    Tf = c * min_temp_factor + 1e-12
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # schedule temperature
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # decide direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 0:
            continue
        # epsilon proportion; use rng.random()**0.5 to bias larger? maybe.
        eps = rng.random() * max_eps
        h_new = h.copy()
        h_new[src] -= eps
        h_new[dst] += eps
        c_new = compute_c5(h_new)
        delta = c_new - c
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h = h_new
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def binary_sa(start_bin, rng, deadline, max_iters=200000):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    # temperature schedule
    T0 = best_c * 0.02 + 1e-9
    Tf = 1e-8
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # ensure i is 1 and j is 0 (swap)
        if h[i] < h[j]:
            i, j = j, i
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        # temperature update
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def simulated_annealing(h0, rng, deadline):
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    n = len(h)
    start = time.time()
    T0 = c * 0.2 + 1e-9   # larger start factor
    Tf = c * 1e-8 + 1e-12
    it = 0
    while time.time() < deadline:
        it += 1
        # temperature based on elapsed fraction
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start) if deadline > start else 1.0
        T = T0 * ((Tf / T0) ** fraction)
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(h[src], 1.0 - h[dst])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps
        h_new = h.copy()
        h_new[src] -= eps
        h_new[dst] += eps
        c_new = compute_c5(h_new)
        delta = c_new - c
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            h = h_new
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

    best_c = compute_c5(h)

        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # ensure i is 1 and j is 0
        if h[i] < h[j]:

        # swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)

            h[j] = 0.0

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure after epsilon, h[i] and h[j] remain in [0,1]
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)
```

    candidates.append(alt)

    for _ in range(3):

    best_h = None
    best_c = float('inf')
    for h in candidates:

            best_h = h.copy()

    # Phase 1: Continuous simulated annealing

        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-15)):

                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    tri = 1.0 - np.abs(xs - 1.0)
    tri = np.clip(tri, 0.0, 1.0)
    candidates.append(enforce_sum(tri, n / 2.0, rng))

    for f in range(1, 5):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(s, 0.0, 1.0), n / 2.0, rng))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(np.clip(c, 0.0, 1.0), n / 2.0, rng))

    block = np.zeros(n)
    block[: n // 2] = 1.0
    candidates.append(block)

    alt = np.tile([0.0, 1.0], n // 2 + 1)[:n]
    candidates.append(alt)

    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    best_h, best_c = None, float('inf')
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 1: continuous simulated annealing
    phase1_deadline = start_time + 0.55 * budget_s
    if time.time() < phase1_deadline:
        start_h = perturb_h(best_h, rng, scale=0.3)
        h_sa, c_sa = simulated_annealing(start_h, rng, phase1_deadline)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

    # Phase 2: hill climbing
    phase2_deadline = start_time + 0.75 * budget_s
    if time.time() < phase2_deadline:
        h_hc, c_hc = hill_climb(best_h, rng, phase2_deadline, start_step=0.3)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

    # Phase 3: binary simulated annealing
    phase3_deadline = start_time + 0.95 * budget_s
    if time.time() < phase3_deadline:
        idx = np.argpartition(-best_h, n // 2)[: n // 2]
        bin_start = np.zeros(n)
        bin_start[idx] = 1.0
        h_bin, c_bin = binary_sa(bin_start, rng, phase3_deadline)
        if c_bin < best_c:
            best_c, best_h = c_bin, h_bin.copy()
        if time.time() + 0.5 < phase3_deadline:
            rand_bin = np.zeros(n)
            rand_bin[rng.choice(n, n // 2, replace=False)] = 1.0
            h_bin2, c_bin2 = binary_sa(rand_bin, rng, phase3_deadline)
            if c_bin2 < best_c:
                best_c, best_h = c_bin2, h_bin2.copy()

    # Final polishing
    final_deadline = start_time + budget_s
    if time.time() < final_deadline:
        h_fin, c_fin = hill_climb(best_h, rng, final_deadline, start_step=0.2)
        if c_fin < best_c:
            best_c, best_h = c_fin, h_fin.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
