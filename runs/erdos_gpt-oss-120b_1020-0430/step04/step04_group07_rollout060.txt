def softmax_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    # compute weights p = exp(beta*corr) / sum...
    maxv = corr.max()
    exps = np.exp(beta * (corr - maxv))
    sum_exps = exps.sum()
    p = exps / sum_exps

    # compute gradient
    g = np.zeros_like(h)
    # for each lag L (with index i in corr)
    for idx, weight in enumerate(p):
        L = idx - (n - 1)
        if L >= 0:
            # first term: for i in [0, n-1-L]
            g[:n - L] += weight * (1 - h[L:])
            # second term: for i in [L, n-1]
            g[L:] -= weight * h[:n - L]

def softmax_gradient(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    maxv = corr.max()
    exp_beta = np.exp(beta * (corr - maxv))
    sum_exp = exp_beta.sum()
    p = exp_beta / sum_exp
    g = np.zeros_like(h)
    for idx, weight in enumerate(p):
        L = idx - (n - 1)
        if L >= 0:
            # first term: i from 0 to n-1-L => h[i+L] exists
            if weight != 0:
                g[:n - L] += weight * (1 - h[L:])
                g[L:] -= weight * h[:n - L]
        else:
            d = -L
            if weight != 0:
                g[d:] += weight * (1 - h[:n - d])
                g[:n - d] -= weight * h[d:]
    return corr, g

def gradient_descent_optimize(h0, target, rng, time_limit):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    betas = [1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]
    for beta in betas:
        step = 0.5 / np.sqrt(beta)  # schedule
        # set a max iteration count but break by time
        it = 0
        while True:
            now = time.time()
            if now - start > time_limit:
                return best_h, best_c
            # compute gradient
            _, g = softmax_gradient(h, beta)
            # project to sum zero
            g = g - np.mean(g)
            # update
            h = h - step * g
            # clip
            h = np.clip(h, 0.0, 1.0)
            # enforce sum exactly
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                # fix diff by moving along random indices
                # simpler: distribute diff equally to indices not at bounds; but we will do random.
                for _ in range(1000):
                    if abs(diff) < 1e-12:
                        break
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                    if add == 0:
                        continue
                    h[i] += add
                    diff = target - h.sum()
            # evaluate periodic
            if it % 20 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            it += 1
            # limit iterations per beta maybe 5000
            if it >= 5000:
                break
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    # Define equality constraint
    def sum_constraint(x):
        return np.sum(x) - target
    # objective
    def softmax_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        maxv = corr.max()
        ex = np.exp(beta * (corr - maxv))
        return (maxv + np.log(np.sum(ex))) / beta
    # call minimize
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints={'type':'eq','fun':sum_constraint}, options={'maxiter': max_iter, 'ftol':1e-12,'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum correction if needed
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12: break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project onto Σh = n/2 while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random h ∈[0,1]^n with Σh=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (max) overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from high‑gradient index to low‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Simple hill‑climb using worst‑gradient transfers."""
    best_h, best_c = h.copy(), compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


# ---------- smoothed‑max gradient ----------
def softmax_gradient(h: np.ndarray, beta: float):
    """Return (corr, grad) of log‑sum‑exp_{β} corr."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    maxv = corr.max()
    exp_beta = np.exp(beta * (corr - maxv))
    p = exp_beta / exp_beta.sum()
    g = np.zeros_like(h)
    for idx, w in enumerate(p):
        L = idx - (n - 1)
        if L >= 0:
            if w == 0.0:
                continue
            g[: n - L] += w * (1 - h[L:])
            g[L:] -= w * h[: n - L]
        else:
            d = -L
            if w == 0.0:
                continue
            g[d:] += w * (1 - h[: n - d])
            g[: n - d] -= w * h[d:]
    return corr, g


def gradient_descent_optimize(h0: np.ndarray, rng: np.random.Generator,
                              time_limit: float):
    """Projected gradient descent on smoothed max objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    betas = [1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]
    for beta in betas:
        step = 0.5 / math.sqrt(beta)
        it = 0
        while True:
            now = time.time()
            if now - start > time_limit:
                return best_h, best_c
            _, g = softmax_gradient(h, beta)
            g -= g.mean()                     # keep Σh unchanged in the update
            h -= step * g
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                for _ in range(200):
                    if abs(diff) < 1e-12:
                        break
                    i = rng.integers(0, n)
                    add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                    if add == 0.0:
                        continue
                    h[i] += add
                    diff = target - h.sum()
            if it % 20 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
            it += 1
            if it >= 5000:
                break
    return best_h, best_c


# ---------- optional SLSQP refinement ----------
def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 300) -> np.ndarray:
    n = h0.size
    target = n / 2.0
    def sum_con(x):
        return np.sum(x) - target
    def sm_obj(x):
        c = np.correlate(x, 1 - x, mode="full")
        m = c.max()
        e = np.exp(beta * (c - m))
        return (m + np.log(e.sum())) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_con}
    res = minimize(sm_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter,
                                             "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            if add == 0.0:
                continue
            x[i] += add
            diff = target - x.sum()
    return x


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # start from parent if provided
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), rng)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    n_candidates = [192, 256, 320, 384, 512]

    # ----- main search loop -----
    while time.time() - start < budget_s * 0.85:
        n = int(rng.choice(n_candidates))
        fresh = rng.random() < 0.15

        if not fresh and best_h is not None:
            if best_n == n:
                h = best_h.copy()
            else:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = enforce_sum(np.clip(h, 0.0, 1.0), rng)
        else:
            if rng.random() < 0.5:
                h = np.zeros(n)
                h[rng.choice(n, n // 2, replace=False)] = 1.0
            else:
                h = random_feasible_vector(n, rng)

        # gradient descent phase (budgeted)
        remaining = budget_s - (time.time() - start)
        gd_time = min(remaining * 0.5, 12.0)
        h_opt, c_opt = gradient_descent_optimize(h, rng, gd_time)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, n

        # brief hill‑climb
        if time.time() - start < budget_s * 0.95:
            h_hc, c_hc = guided_hill_climb(best_h, rng, max_iters=1500, step_scale=0.5)
            if c_hc < best_c:
                best_h, best_c, best_n = h_hc.copy(), c_hc, n

    # ----- up‑sampling refinement -----
    if best_h is not None and time.time() - start < budget_s * 0.98:
        for factor in (2, 3):
            n_up = best_n * factor
            if n_up > 1024:
                continue
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = enforce_sum(np.clip(h_up, 0.0, 1.0), rng)

            # gradient descent on higher resolution
            remaining = budget_s - (time.time() - start)
            if remaining <= 0:
                break
            gd_time = min(remaining * 0.5, 20.0)
            h_up, c_up = gradient_descent_optimize(h_up, rng, gd_time)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

            # final SLSQP polish (moderate β)
            if time.time() - start < budget_s * 0.999:
                for beta in (20.0, 200.0, 2000.0, 8000.0):
                    if time.time() - start >= budget_s:
                        break
                    h_up = slsqp_optimize(best_h, beta, rng, max_iter=400)
                    c_up = compute_c5(h_up)
                    if c_up < best_c:
                        best_h, best_c, best_n = h_up.copy(), c_up, n_up

            # tiny hill‑climb
            if time.time() - start < budget_s * 0.999:
                h_up, c_up = guided_hill_climb(best_h, rng, max_iters=800, step_scale=0.4)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ----- final sanity check -----
    if best_h is None:
        # fallback arbitrary feasible vector
        n = 256
        best_h = random_feasible_vector(n, rng)
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
