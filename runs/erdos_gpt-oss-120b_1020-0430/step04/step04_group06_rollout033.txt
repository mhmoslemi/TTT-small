    return obj, grad

from scipy.optimize import minimize

def lbfgs_optimize(h0, beta, penalty_lambda, max_iter=200):
    n = h0.size
    target_sum = n / 2.0
    def fun(x):
        obj, grad = obj_and_grad(x, beta)
        sum_err = np.sum(x) - target_sum
        obj_pen = obj + penalty_lambda * sum_err * sum_err
        grad_pen = grad + 2 * penalty_lambda * sum_err
        return obj_pen, grad_pen
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    x = enforce_sum(x, target_sum, np.random.default_rng(0))
    return x

import time
import numpy as np
from scipy.optimize import minimize

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that sum(h) == target while keeping 0<=h<=1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum n/2."""
    x = rng.random(n)
    x = x / x.sum() * (n / 2.0)
    x = np.clip(x, 0.0, 1.0)
    return enforce_sum(x, n / 2.0, rng)


def compute_c5(h):
    """Exact discrete C5 for step size 2/n."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def max_overlap_and_shift(h):
    """Return (max correlation value, shift index) for h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Specialised local search moving mass from i to i+shift."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # indices participating in this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c

# Smooth surrogate (log‑sum‑exp) and its gradient
# ------------------------------------------------------------
def obj_and_grad(h, beta):
    """Soft‑max of the overlap values and its gradient."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    sum_exp = np.sum(exp_bc)
    w = exp_bc / sum_exp                     # normalized weights, length 2n‑1
    obj = (m + np.log(sum_exp)) / beta

    # gradient: 1 - (conv(w, h) + conv(w_rev, h)) at appropriate indices
    conv_minus = np.convolve(w, h, mode="full")
    conv_plus = np.convolve(w[::-1], h, mode="full")
    idx = np.arange(h.size) + (h.size - 1)
    grad = 1.0 - (conv_minus[idx] + conv_plus[idx])
    return obj, grad


def lbfgs_optimize(h0, beta, penalty_lambda, max_iter=150):
    """L‑BFGS‑B minimisation of the smooth surrogate with sum‑penalty."""
    n = h0.size
    target = n / 2.0

    def fun(x):
        obj, grad = obj_and_grad(x, beta)
        diff = np.sum(x) - target
        obj_pen = obj + penalty_lambda * diff * diff
        grad_pen = grad + 2.0 * penalty_lambda * diff
        return obj_pen, grad_pen

    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0, method="L-BFGS-B", jac=True,
                   bounds=bounds,
                   options={"maxiter": max_iter, "ftol": 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, target, np.random.default_rng())
    return x
