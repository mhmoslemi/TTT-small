def compute_c5(h):
  return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / h.size)

def softmax_obj_and_grad(h, beta):
  n = h.shape[0]
  corr = np.correlate(h, 1 - h, mode='full')
  max_corr = np.max(corr)
  exp_shift = np.exp(beta * (corr - max_corr))
  sum_exp = np.sum(exp_shift)
  obj = (max_corr + np.log(sum_exp)) / beta
  w = exp_shift / sum_exp
  # compute gradient as described
  w_pos = w[n-1:]  # size n
  w_neg = w[:n-1][::-1] # size n-1
  g = np.zeros(n)
  # s>=0
  for s in range(n):
      ws = w_pos[s]
      if ws == 0.0:
          continue
      # g[:n-s] += ws * (1 - h[s:])
      g[:n-s] += ws * (1 - h[s:])
      # g[s:] -= ws * h[:n-s]
      g[s:] -= ws * h[:n-s]
  # negative shifts t=1..n-1
  for t in range(1, n):
      ws = w_neg[t-1]
      if ws == 0.0:
          continue
      g[t:] += ws * (1 - h[:n-t])
      g[:n-t] -= ws * h[t:]
  return obj, g

def grad_descent(h0, beta, steps, step_size, rng):
  h = h0.copy()
  target = h.size / 2.0
  for i in range(steps):
    obj, g = softmax_obj_and_grad(h, beta)
    lr = step_size / np.sqrt(i+1)
    h -= lr * g
    # Clip and enforce sum
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
  return h

def project_onto_sum(h, target):
    # returns array clipped to [0,1] with sum = target.
    # Use bisection on lambda: x = clip(h + λ, 0, 1).
    lo = -2.0
    hi = 2.0
    for _ in range(30):
        mid = (lo + hi) / 2.0
        x = np.clip(h + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    # final
    x = np.clip(h + (lo+hi)/2.0, 0.0, 1.0)
    # To fix any tiny residual (due to tolerance) we can adjust last element:
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust smallest margin
        # find index where 0 < x[i] < 1
        mask = (x > 0.0) & (x < 1.0)
        if np.any(mask):
            # add diff to first such index within bounds
            i = np.where(mask)[0][0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            # all at bounds, adjust any index
            i = np.argmin(x)  # minimal value
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    # we can move mass from i to j, ensuring bounds.
    max_move = min(h[i], 1 - h[j])
    if max_move <= 0:
        return h.copy()
    eps = max_move * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import time, numpy as np

# ---------- utilities ----------
def project_onto_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        x = np.clip(h + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 0.0) & (x < 1.0)
        if np.any(mask):
            i = np.where(mask)[0][0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            i = np.argmin(x)
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return project_onto_sum(h, n / 2.0)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_move = min(h[i], 1.0 - h[j])
    if max_move <= 0.0:
        return h.copy()
    eps = max_move * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def softmax_obj_and_grad(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_shift = np.exp(beta * (corr - m))
    sum_exp = np.sum(exp_shift)
    obj = (m + np.log(sum_exp)) / beta
    w = exp_shift / sum_exp
    w_pos = w[n - 1:]          # shift s = 0..n-1
    w_neg = w[: n - 1][::-1]   # shift -t, t=1..n-1
    g = np.zeros(n)
    # s >= 0
    for s in range(n):
        ws = w_pos[s]
        if ws == 0.0:
            continue
        g[: n - s] += ws * (1.0 - h[s:])
        g[s:]       -= ws * h[: n - s]
    # s < 0  (t = 1..n-1)
    for t in range(1, n):
        ws = w_neg[t - 1]
        if ws == 0.0:
            continue
        g[t:]       += ws * (1.0 - h[: n - t])
        g[: n - t]  -= ws * h[t:]
    return obj, g

def grad_descent(h0, beta, steps, step, target, rng):
    h = h0.copy()
    for i in range(steps):
        _, g = softmax_obj_and_grad(h, beta)
        lr = step / np.sqrt(i + 1.0)
        h = project_onto_sum(h - lr * g, target)
    return h

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n_fine = 720
    target_fine = n_fine / 2.0

    # initial solution
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values, dtype=float)
        best_h = project_onto_sum(np.clip(best_h, 0.0, 1.0), target_fine)
        best_c = compute_c5(best_h)
    else:
        best_h = random_feasible_vector(n_fine, rng)
        best_c = compute_c5(best_h)

    # ---------- coarse exploration ----------
    n_coarse = 150
    target_coarse = n_coarse / 2.0
    coarse_deadline = start + budget_s * 0.18
    while time.time() < coarse_deadline:
        h = random_feasible_vector(n_coarse, rng)
        for beta, steps, step in ((200, 500, 0.5), (800, 800, 0.3), (2000, 1000, 0.2)):
            h = grad_descent(h, beta, steps, step, target_coarse, rng)
        xs_c = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
        xs_f = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
        h_f = np.interp(xs_f, xs_c, h)
        h_f = project_onto_sum(np.clip(h_f, 0.0, 1.0), target_fine)
        c = compute_c5(h_f)
        if c < best_c:
            best_c, best_h = c, h_f.copy()

    # ---------- fine‑grained optimization ----------
    fine_deadline = start + budget_s * 0.85
    beta_phases = ((2000, 2000, 0.08), (8000, 2000, 0.04), (20000, 1000, 0.02))
    while time.time() < fine_deadline:
        if rng.random() < 0.7:
            h0 = best_h.copy()
        else:
            h0 = random_feasible_vector(n_fine, rng)
        h = h0
        for beta, steps, step in beta_phases:
            if time.time() >= fine_deadline:
                break
            h = grad_descent(h, beta, steps, step, target_fine, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # occasional tiny shake
        if rng.random() < 0.25:
            cand = perturb_random(best_h, rng, scale=0.05)
            c_cand = compute_c5(cand)
            if c_cand < best_c:
                best_c, best_h = c_cand, cand.copy()

    # ---------- final micro‑search ----------
    micro_deadline = start + budget_s * 0.99
    while time.time() < micro_deadline:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_c, best_h = c_cand, cand.copy()

    # clean up and return
    best_h = project_onto_sum(np.clip(best_h, 0.0, 1.0), target_fine)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
