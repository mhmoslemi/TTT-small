Implementation of project_sum:

```
def project_sum(h, target, rng):
    diff = target - h.sum()
    # Clip each step to [0,1] while adjusting diff by random redistribution
    # Use loop with random indices.
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0,n)
        if diff > 0:
            maxadd = min(diff, 1.0 - h[i])
            if maxadd <= 0: continue
            h[i] += maxadd
            diff -= maxadd
        else:
            maxsub = min(-diff, h[i])
            if maxsub <= 0: continue
            h[i] -= maxsub
            diff += maxsub
    return h
```

def anneal_optimize(h, rng, time_limit):
    n = h.size
    dx = 2.0 / n
    # Determine initial c5
    c = compute_c5(h)
    # Determine worst_shift index? could get from corr.
    # But we just use c as measure.
    # Simulated annealing parameters:
    T0 = 0.01
    Tend = 1e-5
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # Determine temperature schedule
        t = (time.time() - start) / time_limit
        T = T0 * ((Tend/T0) ** t)  # exponential decay
        # propose move: swap two indices positions i (value hi), j (value hj) such that hi != hj
        # So we can pick a random index i and find a random index j with different value.
        i = rng.integers(n)
        hi = h[i]
        # need to pick j with opposite value
        # build list of indices of opposite? That is O(n). We can maintain arrays of positions of 1s and 0s.
        # Simpler: pick j until value different.
        j = rng.integers(n)
        while h[j] == hi:
            j = rng.integers(n)
        # swap
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        c_new = compute_c5(h_new)
        delta = c_new - c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h, c = h_new, c_new
        it += 1
    return h, c

def pgd_optimize(h, rng, betas, max_iters_per_beta, time_budget):
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    for beta in betas:
        # set initial step size relative to beta; perhaps lr = 0.5 / np.sqrt(beta)
        lr = 0.5 / np.sqrt(beta)
        # maybe also limit total iteration count by time budget
        for it in range(max_iters_per_beta):
            if time.time() - start > time_budget:
                return best_h, best_c
            corr = np.correlate(h, 1 - h, mode='full') * dx
            # compute softmax weights
            max_corr = np.max(corr)
            w = np.exp(beta * (corr - max_corr))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            # compute gradients via convolutions
            conv1 = np.convolve(w, h, mode='valid')[::-1]
            conv2 = np.convolve(w[::-1], h, mode='valid')[::-1]
            grad = -(conv1 + conv2)
            # gradient step
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, rng)
            # evaluate new C5
            c_new = compute_c5(h_new)
            # accept if improvement else keep old
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
            # we also accept if c_new < current c (monotonic descent)
            # else we revert (or maybe keep anyway)

                # shift happen
                if rng.random() < 0.01:
                    h = h_new
                # else keep old

import time, numpy as np
from scipy.optimize import minimize

# Global placeholders for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def random_binary_vector(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, rng)
    return x

def anneal_optimize_binary(h, rng, time_budget):
    n = h.size
    # ensure binary integer type
    h = h.astype(int)
    curr_h = h.copy()
    curr_c = compute_c5(curr_h)
    best_h = curr_h.copy()
    best_c = curr_c
    start = time.time()
    while time.time() - start < time_budget:
        i = rng.integers(n)
        hi = curr_h[i]
        j = rng.integers(n)
        while curr_h[j] == hi:
            j = rng.integers(n)
        # swap
        curr_h[i], curr_h[j] = curr_h[j], curr_h[i]
        new_c = compute_c5(curr_h)
        # temperature linearly decreasing
        elapsed = time.time() - start
        T = 0.01 * (1.0 - elapsed / time_budget)
        if new_c < curr_c or rng.random() < np.exp(-(new_c - curr_c) / max(T, 1e-12)):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = curr_h.copy()
        else:
            # revert swap
            curr_h[i], curr_h[j] = curr_h[j], curr_h[i]
    return best_h.astype(float), best_c

def pgd_optimize(h0, rng, betas, max_iters_per_beta, time_budget):
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    for beta in betas:
        lr = 0.5 / np.sqrt(beta)  # base step size
        for it in range(max_iters_per_beta):
            if time.time() - start > time_budget:
                return best_h, best_c
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            max_corr = np.max(corr)
            w = np.exp(beta * (corr - max_corr))
            w_sum = w.sum()
            if w_sum == 0:

            else:

            conv1 = np.convolve(w, h, mode='valid')[::-1]

            grad = -(conv1 + conv2)

                h = h_new
            else:
                # occasional uphill move
                if rng.random() < 0.001:
                    h = h_new
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    if abs(CURRENT_TARGET - x.sum()) > 1e-12:
        x = enforce_sum(x, rng)
    return x


def anneal_optimize_binary(h, rng, time_budget):
    n = h.size
    curr_h = h.copy()                     # float, values 0 or 1
    curr_c = compute_c5(curr_h)
    best_h, best_c = curr_h.copy(), curr_c
    start = time.time()
    while time.time() - start < time_budget:
        i = rng.integers(n)
        hi = curr_h[i]
        j = rng.integers(n)
        while curr_h[j] == hi:
            j = rng.integers(n)
        curr_h[i], curr_h[j] = curr_h[j], curr_h[i]
        new_c = compute_c5(curr_h)
        elapsed = time.time() - start
        T = 0.01 * (1.0 - elapsed / time_budget)
        if new_c < curr_c or rng.random() < np.exp(-(new_c - curr_c) / max(T, 1e-12)):
            curr_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, curr_h.copy()
        else:
            curr_h[i], curr_h[j] = curr_h[j], curr_h[i]
    return best_h, best_c


def pgd_optimize(h0, rng, betas, max_iters_per_beta, time_budget):
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    for beta in betas:
        lr = 0.5 / np.sqrt(beta)
        for _ in range(max_iters_per_beta):
            if time.time() - start > time_budget:
                return best_h, best_c
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            max_corr = np.max(corr)
            w = np.exp(beta * (corr - max_corr))
            w_sum = w.sum()
            w = w / w_sum if w_sum != 0 else np.ones_like(w) / w.size
            conv1 = np.convolve(w, h, mode="valid")[::-1]
            conv2 = np.convolve(w[::-1], h, mode="valid")[::-1]
            grad = -(conv1 + conv2)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, rng)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
            cur_c = compute_c5(h)
            if c_new <= cur_c:
                h = h_new
            else:
                if rng.random() < 0.001:
                    h = h_new
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0, compute_c5(h0), h0.size

    # Phase 1: many binary annealing runs (≈45% of budget)
    while time.time() - start < 0.45 * budget_s:
        n = 256
        h0 = random_binary_vector(n, rng)
        remaining = budget_s - (time.time() - start)
        ann_time = min(30.0, 0.3 * remaining)
        h_bin, c_bin = anneal_optimize_binary(h0, rng, ann_time)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin.astype(float), c_bin, n

    # Phase 2: projected gradient descent with soft‑max (majority of remaining time)
    if best_h is not None:
        remaining = budget_s - (time.time() - start)
        pgd_time = min(remaining * 0.9, remaining - 5.0)
        betas = [20., 80., 200., 800., 3000., 8000., 20000.]
        h_opt, c_opt = pgd_optimize(best_h, rng, betas, max_iters_per_beta=500, time_budget=pgd_time)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, h_opt.size

    # Phase 3: final SLSQP fine‑tuning if time permits
    if best_h is not None and time.time() - start < 0.95 * budget_s:
        for beta in (200., 800., 3000., 8000., 20000.):
            if time.time() - start > 0.99 * budget_s:
                break
            best_h = slsqp_optimize(best_h, beta, rng, max_iter=800)
            best_c = compute_c5(best_h)

    # final sanity check
    if best_h is None:
        n = 256
        best_h = np.full(n, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
