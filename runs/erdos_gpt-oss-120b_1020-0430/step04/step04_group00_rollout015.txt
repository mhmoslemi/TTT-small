def continuous_anneal(h, rng, time_deadline, init_temp=0.02, cooling=0.995):
   current = h.copy()
   cur_c = compute_c5(current)
   T = init_temp
   while time.time() < time_deadline:
        # propose two indices i,j
        i, j = rng.integers(0, len(h), size=2)
        if i==j: continue
        # max possible delta = min(current[i], 1-current[j]) if moving from i to j.
        # Also allow negative direction subtract from i to j.
        max_delta = min(current[i], 1 - current[j])
        if max_delta == 0: continue
        delta = rng.random() * max_delta  # uniform [0, max_delta]
        # apply
        new = current.copy()
        new[i] -= delta
        new[j] += delta
        new_c = compute_c5(new)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            current = new
            cur_c = new_c
        T *= cooling
   return current, cur_c

def _proj_sum_exact(h, target, rng, max_iter=2000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):

        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])

        diff = target - h.sum()
    return h

def _gradient_descent(h, rng, steps=30, step0=0.05):
    n = h.size
    target = n / 2.0
    step = step0
    for _ in range(steps):

        g -= np.mean(g)                     # keep sum constant in the infinitesimal step
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)

def _slsqp_optimize(h, beta, rng, max_iter=400):
