def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        # For i from 0 to n-s-1: term h_i * (1 - h_{i+s}) derivative w.r.t h_i is (1 - h_{i+s})
        # derivative w.r.t h_{i+s} is -h_i
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        # s < 0, let s_abs = -s
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # re-enforce sum exactly
    rng_local = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng_local)
    return x

from scipy.optimize import minimize

# -------------------- basic utilities --------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for vector h (integral approximated by dx=2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def gradient_shift(h: np.ndarray, s: int) -> np.ndarray:
    """Gradient of the correlation at shift s w.r.t. h."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += 1 - h[s:]
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += 1 - h[: n - s_abs]
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0.0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 400) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}

        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng_local = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng_local)
    return x
