    if k >= 0:

        a = h[:n - s]

def sa_binary(h0, rng, deadline):
    h = h0.copy()
    n = len(h)
    raw_corr = np.correlate(h, 1-h, mode='full')
    cur_c_raw = raw_corr.max()
    cur_c = cur_c_raw * (2.0/n)
    best_h = h.copy()
    best_c = cur_c
    # Temperature init:
    T0 = cur_c * 0.01  # proportion
    T = T0
    decay = 0.99995
    it = 0
    while time.time() < deadline:
        it += 1
        # compute worst shift
        idx = np.argmax(raw_corr)
        k = idx - (n-1)
        # get candidates
        A, B = get_swap_candidates(h, k)
        if A.size > 0 and B.size > 0:
            i = int(rng.choice(A))
            j = int(rng.choice(B))
        else:
            # random opposite-valued swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        # swap
        h[i], h[j] = h[j], h[i]
        # recompute correlation (full)
        raw_corr = np.correlate(h, 1-h, mode='full')
        new_c_raw = raw_corr.max()
        new_c = new_c_raw * (2.0/n)
        # accept
        accept = False
        if new_c <= cur_c:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            cur_c_raw = new_c_raw
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # temperature decay
        T *= decay
        # optionally early exit if T < threshold and no improvements for many it.
        if it % 1000 == 0:
            if time.time() > deadline:
                break
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            # overlapping region
            if k > 0:
                upper = n - k
                g[:upper] += 1.0 - h[k:]
                g[k:] -= h[:upper]
            else:
                s = -k
                upper = n - s
                g[s:] += 1.0 - h[:upper]
                g[:upper] -= h[s:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # adjust sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def get_swap_candidates(h, k):
    n = h.shape[0]
    if k >= 0:
        a = h[:n - k]
        b = h[k:]
        A_idx = np.where((a == 1) & (b == 0))[0]
        B_idx = np.where((a == 0) & (b == 1))[0]
        A = A_idx
        B = B_idx
    else:

        a = h[s:]
        b = h[:n - s]
        A_idx = np.where((a == 1) & (b == 0))[0]
        B_idx = np.where((a == 0) & (b == 1))[0]
        A = A_idx + s
        B = B_idx
    return A, B

    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(raw.max() * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def get_swap_candidates(h, k):
    """return positions i (type A) with h[i]=1, h[i+k]=0
       and positions j (type B) with h[j]=0, h[j+k]=1
       indices are given in the original array space."""
    n = h.shape[0]
    if k >= 0:
        a = h[: n - k]
        b = h[k:]
        A_idx = np.where((a == 1) & (b == 0))[0]
        B_idx = np.where((a == 0) & (b == 1))[0]
        A = A_idx
        B = B_idx
    else:
        s = -k
        a = h[s:]
        b = h[: n - s]
        A_idx = np.where((a == 1) & (b == 0))[0]
        B_idx = np.where((a == 0) & (b == 1))[0]
        A = A_idx + s
        B = B_idx
    return A, B

def deterministic_improve(h, max_iters=5000):
    """Greedy improvement: increase the minimum auto‑correlation C_k."""
    n = h.shape[0]
    for _ in range(max_iters):
        conv = np.correlate(h, h, mode="full")
        conv[n - 1] = np.inf  # ignore shift 0
        idx = np.argmin(conv)
        k = idx - (n - 1)
        A, B = get_swap_candidates(h, k)
        if A.size == 0 or B.size == 0:
            break
        # swap the first pair (any works)
        i = int(A[0])
        j = int(B[0])
        h[i], h[j] = h[j], h[i]
    return h

def gradient_refine(h, rng, step=0.02, max_iter=15):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g = 1.0 - 2.0 * h
        elif k > 0:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        else:  # k < 0
            s = -k
            up = n - s
            g[s:] += 1.0 - h[:up]
            g[:up] -= h[s:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def parent_binary(parent, n):
    xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    interp = np.interp(xt, xs, parent)
    bin_vec = np.zeros(n)
    idx = np.argpartition(interp, -n // 2)[-n // 2 :]
    bin_vec[idx] = 1.0
    return bin_vec

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    parent = np.array(initial_h_values, dtype=float) if "initial_h_values" in globals() else None

    best_h = None
    best_c = np.inf
    best_n = None

    # try a few resolutions, finer ones first
    for n in (720, 540, 360):
        if time.time() > deadline:
            break

        # build a small pool of start vectors
        starts = []
        if parent is not None:
            starts.append(parent_binary(parent, n))
        starts.append(random_binary(n, rng))

        for base in starts:
            if time.time() > deadline:
                break

            h = base.copy()
            # greedy deterministic improvement
            h = deterministic_improve(h, max_iters=5000)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # a quick gradient refinement (fractional values allowed)
            h_frac = gradient_refine(h.astype(float), rng, step=0.02, max_iter=12)
            c_frac = compute_c5(h_frac)
            if c_frac < best_c:
                best_h, best_c, best_n = h_frac.copy(), c_frac, n

    # fallback in the unlikely event no candidate was produced
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final safety: clip and enforce exact integral
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
