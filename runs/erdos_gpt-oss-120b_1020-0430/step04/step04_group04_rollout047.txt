def compute_corr_full(h):
    n = h.size
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

def perturb_boundary(h, rng):
    n = h.size
    i = rng.integers(0,n)
    # choose target 0 or 1 depending on current value
    if rng.random() < 0.5:
        target = 0.0
    else:
        target = 1.0
    epsilon = target - h[i]
    if epsilon == 0:
        return h.copy()
    # Find j that can absorb opposite epsilon
    # If epsilon > 0 (increase i), need j with capacity 1 - h[j] >= epsilon
    # else epsilon < 0 (decrease i), need j with h[j] >= -epsilon
    # Choose random candidate j satisfying constraint
    candidates = np.where( (epsilon>0) & (1-h >= epsilon) | (epsilon<0) & (h >= -epsilon) )[0]
    if candidates.size == 0:
        # fallback: just clip after adjusting i and skip sum fix
        new_h = h.copy()
        new_h[i] = target
        # adjust sum via enforce_sum later
        return new_h
    j = rng.choice(candidates)
    new_h = h.copy()
    new_h[i] = target
    new_h[j] -= epsilon  # because total sum unchanged
    # Clip just in case
    new_h = np.clip(new_h, 0, 1)
    return new_h

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------
def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):

            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def compute_corr_full(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

# ---------- Perturbations ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def perturb_boundary(h, rng):
    n = h.size
    i = rng.integers(0, n)
    target = 1.0 if rng.random() < 0.5 else 0.0
    eps = target - h[i]
    if eps == 0.0:
        return h.copy()
    if eps > 0:
        cand = np.where(1.0 - h >= eps)[0]
    else:
        cand = np.where(h >= -eps)[0]
    if cand.size == 0:
        new_h = h.copy()
        new_h[i] = target
        return new_h
    j = rng.choice(cand)
    new_h = h.copy()
    new_h[i] = target
    new_h[j] -= eps
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

# ---------- Local search ----------
def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            if rng.random() < 0.15:
                new_h = perturb_boundary(h, rng)
            else:
                new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < math.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = np.random.randint(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def gradient_descent_max_corr(h, rng, time_limit):
    start = time.time()
    n = h.size
    dx = 2.0 / n
    alpha = 0.5
    best_h = h.copy()
    best_c = compute_c5(h)
    while time.time() - start < time_limit:
        conv = compute_corr_full(h)
        lag_idx = int(np.argmax(conv))
        max_conv = conv[lag_idx]
        old_c = max_conv * dx
        lag = lag_idx - (n - 1)
        g = np.zeros_like(h)
        if lag >= 0:
            g[: n - lag] += (1 - h[lag:])
            g[lag:] -= h[: n - lag]
        else:
            off = -lag
            g[off:] += (1 - h[: n - off])
            g[: n - off] -= h[off:]
        pos = g > 0
        neg = g < 0
        max_alpha = np.inf
        if np.any(pos):
            max_alpha = min(max_alpha, np.min(h[pos] / g[pos]))
        if np.any(neg):
            max_alpha = min(max_alpha, np.min((1 - h[neg]) / (-g[neg])))
        step = min(alpha, max_alpha * 0.9 if max_alpha < np.inf else alpha)
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, n / 2.0, rng)
        conv_new = compute_corr_full(h_new)
        new_c = np.max(conv_new) * dx
        if new_c < old_c - 1e-12:
            h, best_h, best_c = h_new, h_new.copy(), new_c
            alpha = min(alpha * 1.1, 1.0)
        else:
            alpha = max(alpha * 0.5, 1e-4)
    return best_h, best_c
