def binary_sa(start_bin, rng, deadline, max_iters=500000):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-8
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # pick random i from ones, j from zeros
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        # temperature schedule
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c
```

    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

def hill_climb(h_start, rng, max_iters, step):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
        # optionally decay step
        if (it+1) % (max_iters//5) == 0:
            step *= 0.7
    return best_h, best_c

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    it = 0

        else:

        if add != 0.0:
            h[idx] += add
            diff = target - h.sum()
        it += 1

    """Binary vector with exactly n/2 entries equal to 1."""
    idx = rng.choice(n, size=n // 2, replace=False)

    h[idx] = 1.0

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()

    if max_eps <= 0.0:

    new_h[j] += eps

def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Smooth surrogate for SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            idx = rng.integers(0, n)
            add = min(diff, 1.0 - x[idx]) if diff > 0 else max(diff, -x[idx])
            x[idx] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------------------------------------------
# Binary simulated annealing (fast, many iterations)
# --------------------------------------------------------------
def binary_sa(start_bin, rng, deadline):
    """Swap‑based SA on a binary vector (preserves sum)."""
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    T0 = best_c * 0.02 + 1e-8
    Tf = 1e-8
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # choose indices
        i_idx = rng.integers(0, ones_arr.size)
        j_idx = rng.integers(0, zeros_arr.size)
        i = ones_arr[i_idx]
        j = zeros_arr[j_idx]
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        # temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            # update index tables
            ones_arr[i_idx] = j
            zeros_arr[j_idx] = i
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s

    # ----------------------------------------------------------
    # Initial binary solution (top half of initial_h_values if present)
    # ----------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        idx = np.argsort(-init)[: n // 2]
        best_h = np.zeros(n)
        best_h[idx] = 1.0
    else:
        n = 720
        best_h = random_binary_vector(n, rng)

    best_c = compute_c5(best_h)

    # ----------------------------------------------------------
    # Phase A: binary simulated annealing (multiple runs)
    # ----------------------------------------------------------
    phase_a_end = start_time + 0.60 * budget
    while time.time() < phase_a_end:
        start_bin = best_h.copy() if rng.random() < 0.5 else random_binary_vector(n, rng)
        h_bin, c_bin = binary_sa(start_bin, rng, phase_a_end)
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin

    # ----------------------------------------------------------
    # Phase B: continuous hill climbing on current best
    # ----------------------------------------------------------
    phase_b_end = start_time + 0.80 * budget
    if time.time() < phase_b_end:
        h_float = best_h.astype(float)
        remaining = phase_b_end - time.time()
        iters = max(2000, int(remaining / 0.00025))
        h_hc, c_hc = hill_climb(h_float, rng, max_iters=iters, step=0.1)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    # ----------------------------------------------------------
    # Phase B2: SLSQP polishing on 720‑point grid
    # ----------------------------------------------------------
    if time.time() < start_time + 0.90 * budget:
        for beta in [100, 500, 2000, 5000]:
            if time.time() >= start_time + budget:
                break
            h_opt = slsqp_optimize(best_h, beta, max_iter=300)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # ----------------------------------------------------------
    # Phase C: up‑sample to a finer grid (if room) and polish
    # ----------------------------------------------------------
    if best_h.size * 2 <= 1440 and time.time() < start_time + budget - 5:
        h_fine = np.repeat(best_h, 2)
        h_fine = enforce_sum(h_fine, h_fine.size / 2.0, rng)
        best_h_fine = h_fine.copy()
        best_c_fine = compute_c5(best_h_fine)

        # hill climbing on fine grid
        remaining = start_time + budget - time.time()
        iters_fine = max(2000, int(remaining / 0.0003))
        h_hc, c_hc = hill_climb(best_h_fine, rng, max_iters=iters_fine, step=0.05)
        if c_hc < best_c_fine:
            best_h_fine, best_c_fine = h_hc, c_hc

        # SLSQP passes on fine grid
        for beta in [200, 800, 2000, 5000]:
            if time.time() >= start_time + budget:
                break
            h_opt = slsqp_optimize(best_h_fine, beta, max_iter=300)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c_fine:
                best_h_fine, best_c_fine = h_opt, c_opt

        best_h, best_c = best_h_fine, best_c_fine

    # ----------------------------------------------------------
    # Final cleanup and return
    # ----------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
