def gradient_multi(h, top_m=3, step=0.02, max_iter=10):
    n = h.size
    target = n/2
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode='full')
        # get indices of top M values (including sign?) we need absolute values? Actually corr is positive because (1-h) >=0? Actually h ≤1 so 1-h≥0, product≥0, sum≥0. So we can use largest values.
        # we might ignore shift zero maybe avoid trivial? But it's okay.
        # get top m indices
        # If n is small, but we will compute top_m indices using argpartition
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        g = np.zeros_like(h)
        for idx in top_idx:

            if s >= 0:

                g[:n-k] += 1 - h[k:]
                g[k:] -= h[:n-k]
            else:
                k = -s
                g[k:] += 1 - h[:n-k]
                g[:n-k] -= h[k:]
        # normalize gradient maybe?
        # Step size decreasing
        step_sz = step * (0.5 ** it)

        h = np.clip(h, 0.0, 1.0)

    return h

def fractional_sa(h, rng, deadline):
    n = h.size
    target = n/2
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    it = 0
    # Temperature schedule: start T = cur_c*0.5 maybe, then decay
    T0 = max(cur_c*0.3, 1e-4)
    while time.time() < deadline:
        # propose random transfer
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # random delta within limits
        max_delta = min(cur_h[i], 1 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # apply
        cur_h[i] -= delta
        cur_h[j] += delta
        c = compute_c5(cur_h)
        # acceptance
        T = T0 * (0.99 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
        it += 1
        if it % 10000 == 0 and time.time() > deadline - 2:
            break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Global variables for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Adjust h so that sum(h) == n/2 exactly, staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=600):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def gradient_multi(h, rng, top_m=3, step=0.02, max_iter=12):
    """Projected gradient descent using the top‑M worst shifts."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        # indices of the top‑M values
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        g = np.zeros_like(h)
        for idx in top_idx:
            s = idx - (n - 1)  # shift = index - (n-1)
            if s >= 0:
                k = s
                g[: n - k] += 1.0 - h[k:]
                g[k:] -= h[: n - k]
            else:
                k = -s
                g[k:] += 1.0 - h[: n - k]
                g[: n - k] -= h[k:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def fractional_sa(h, rng, deadline):
    """Simulated‑annealing on fractional vectors (mass transfer)."""
    n = h.size
    target = n / 2.0
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    it = 0
    T0 = max(cur_c * 0.3, 1e-4)
    while time.time() < deadline and it < 200000:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        cur_h[i] -= delta
        cur_h[j] += delta
        c = compute_c5(cur_h)
        T = T0 * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            # revert move
            cur_h[i] += delta
            cur_h[j] -= delta
        it += 1
    return best_h, best_c

    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
