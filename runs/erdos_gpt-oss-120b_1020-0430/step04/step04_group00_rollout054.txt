def project_to_feasible(x, target, tol=1e-12, max_iter=60):
    low = -1.0
    high = 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(x - (low+high)/2.0, 0.0, 1.0)

def gradient_subgrad_refine(h, rng, step0=0.05, max_iter=200):
    n = h.size
    target = n/2.0
    step = step0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper: projection onto [0,1] and sum constraint ----------
def project_to_feasible(x, target, tol=1e-12, max_iter=60):
    """Project x onto the box [0,1]^n intersect {sum = target}."""
    low, high = -1.0, 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = mid
        else:
            high = mid
    y = np.clip(x - (low + high) * 0.5, 0.0, 1.0)
    # final tiny correction for numerical safety
    diff = target - y.sum()
    if abs(diff) > tol:
        # distribute the tiny residual uniformly on non‑saturated entries
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            y[free] += diff / free.sum()
    return y

# ---------- evaluation ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

# ---------- smooth‑max surrogate for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_to_feasible(x, CURRENT_TARGET)
    return x

# ---------- subgradient / gradient refinement ----------
def gradient_subgrad_refine(h, rng, step0=0.05, max_iter=200):
    n = h.size
    target = n / 2.0
    step = step0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        # decreasing step size
        step = step0 * (0.95 ** it)
        h = h - step * g
        h = project_to_feasible(h, target)
        # occasional random jitter to escape plateaus
        if (it + 1) % 30 == 0:
            i, j = rng.choice(n, 2, replace=False)
            eps = min(h[i], 0.1, 1.0 - h[j]) * rng.random()
            h[i] -= eps
            h[j] += eps
            h = project_to_feasible(h, target)
    return h

# ---------- main optimisation pipeline for a single start ----------
def optimize_h(h0, rng, deadline):
    h = h0.copy()
    n = h.size
    # initial subgradient sweep
    h = gradient_subgrad_refine(h, rng, step0=0.05, max_iter=120)
    # progressive smooth‑max optimisation
    for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000):
        if time.time() > deadline:
            break
        max_iter = 800 if beta >= 2000 else 600
        h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
        # short subgradient touch after each beta
        h = gradient_subgrad_refine(h, rng, step0=0.02, max_iter=30)
    # final fine‑grained refinement
    h = gradient_subgrad_refine(h, rng, step0=0.01, max_iter=120)
    return h

# ---------- run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.98

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate resolutions (coarse → fine)
    candidate_ns = [360, 540, 720]

    # if a parent exists, insert its size first
    parent_n = None
    if 'initial_h_values' in globals():
        parent_n = len(initial_h_values)
        if parent_n not in candidate_ns:
            candidate_ns.insert(0, parent_n)

    for n in candidate_ns:
        if time.time() > deadline:
            break

        # ---------- build start vectors ----------
        start_vecs = []

        # parent interpolation & binary version
        if parent_n is not None:
            xs = np.linspace(0.0, 2.0, parent_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, initial_h_values)
            start_vecs.append(interp)                       # fractional
            bin_vec = np.zeros(n)
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)                      # binary

        # pure binary random
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= (n / 2.0) / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        rand_frac = project_to_feasible(rand_frac, n / 2.0)
        start_vecs.append(rand_frac)

        # small perturbations of the interpolated parent (if any)
        if parent_n is not None:
            xs = np.linspace(0.0, 2.0, parent_n, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            base = np.interp(xt, xs, initial_h_values)
            for _ in range(2):
                pert = base.copy()
                for __ in range(10):
                    i, j = rng.choice(n, 2, replace=False)
                    eps = min(pert[i], 0.1, 1.0 - pert[j]) * rng.random()
                    pert[i] -= eps
                    pert[j] += eps
                pert = project_to_feasible(pert, n / 2.0)
                start_vecs.append(pert)

        # ---------- optimise each start ----------
        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h0 = project_to_feasible(np.clip(h0, 0.0, 1.0), n / 2.0)
            h_opt = optimize_h(h0, rng, deadline)
            c = compute_c5(h_opt)
            if c < best_c:
                best_c, best_h, best_n = c, h_opt.copy(), n

    # ---------- optional up‑sampling for extra refinement ----------
    if best_h is not None and time.time() < deadline and best_n * 2 <= 1080:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        h_up = project_to_feasible(h_up, n_up / 2.0)
        h_up = optimize_h(h_up, rng, deadline)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up.copy(), n_up

    # ---------- fallback ----------
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = project_to_feasible(best_h, n / 2.0)
        best_c = compute_c5(best_h)

    # final safety projection & evaluation
    best_h = project_to_feasible(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
