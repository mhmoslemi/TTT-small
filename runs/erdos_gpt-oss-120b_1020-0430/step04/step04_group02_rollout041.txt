import time, numpy as np

def project_box_sum(x, target, lo=0., hi=1., max_iter=60):
    low, high = -2., 2.
    for _ in range(max_iter):
        mid = (low+high)/2.
        y = np.clip(x+mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low+high)/2., lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo+1e-12) & (y < hi-1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y
```

Define shift_pad:

```python
def shift_pad(arr, k):
    n = arr.shape[0]
    if k > 0:
        return np.concatenate((np.zeros(k), arr[:n - k]))
    elif k < 0:
        k = -k
        return np.concatenate((arr[k:], np.zeros(k)))
    else:
        return arr.copy()
```

def subgradient_step(h, dx, step):
    # compute worst lag diff
    corr = np.correlate(h, 1.0 - h, mode='full')

    target = h.size / 2.0

def subgradient_optimize(h0, max_iters, step0, rng):
    n = h0.size
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        # find all ks achieving max
        idxs = np.where(corr == max_corr)[0]
        ks = idxs - (n - 1)
        shift_sum = np.zeros_like(h)
        for k in ks:
            shift_sum += shift_pad(h, k) + shift_pad(h, -k)
        # step size
        step = step0 / np.sqrt(it + 1)
        h = h + step * dx * shift_sum
        # project
        h = project_box_sum(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
```

def binary_hill_climb(h0, max_iters, rng):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        k = idx - (n - 1)
        # shift
        h_shift = shift_pad(h, k)
        i_cand = np.where((h == 1) & (h_shift == 0))[0]
        j_cand = np.where((h == 0) & (h_shift == 1))[0]
        if i_cand.size > 0 and j_cand.size > 0:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
        else:
            # generic random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # swap
        h[i], h[j] = 0, 1
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c, best_h = c, h.copy()
        else:
            # revert
            h[i], h[j] = 1, 0
            # occasional acceptance
            if rng.random() < 0.001:
                # keep anyway
                pass
        # optional early break if best_c <= target <? we can break if below 0.38
        if best_c < 0.3805:
            break
    return best_h, best_c

def upsample_h(h, target_n):
    n = h.size
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    h_up = np.interp(new_x, old_x, h)
    h_up = project_box_sum(h_up, target_n / 2.0)
    return h_up

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)

# ---------- evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- shift with zero padding ----------
def shift_pad(arr, k):
    n = arr.shape[0]
    if k > 0:
        return np.concatenate((np.zeros(k), arr[: n - k]))
    if k < 0:
        k = -k
        return np.concatenate((arr[k:], np.zeros(k)))
    return arr.copy()

# ---------- sub‑gradient optimizer ----------
def subgradient_optimize(h0, max_iters, step0, rng):
    n = h0.size
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        idxs = np.where(corr == max_corr)[0]
        ks = idxs - (n - 1)
        shift_sum = np.zeros_like(h)
        for k in ks:
            shift_sum += shift_pad(h, k) + shift_pad(h, -k)
        step = step0 / np.sqrt(it + 1)
        h = h + step * dx * shift_sum
        h = project_box_sum(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c

# ---------- binary hill‑climb ----------
def binary_hill_climb(h0, max_iters, rng):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)
        h_shift = shift_pad(h, k)
        i_cand = np.where((h == 1) & (h_shift == 0))[0]
        j_cand = np.where((h == 0) & (h_shift == 1))[0]
        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c, best_h = c, h.copy()
        else:
            h[i], h[j] = 1, 0
            if rng.random() < 0.001:  # tiny chance to keep bad move
                pass
        if best_c < 0.3805:
            break
    return best_h, best_c

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

# ---------- up‑sampling ----------
def upsample_h(h, target_n):
    n = h.size
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    h_up = np.interp(new_x, old_x, h)
    return project_box_sum(h_up, target_n / 2.0)
