def compute_c5_fft(h):
    n = len(h)
    hs = h.astype(np.float64)
    r = 1 - hs
    L = 1 << ((2*n-1).bit_length())
    fh = np.fft.rfft(hs, n=L)
    fr = np.fft.rfft(r[::-1], n=L)
    conv = np.fft.irfft(fh * fr, n=L)
    corr = conv[:2*n-1]
    return np.max(corr) * (2.0 / n), corr
```

def lbfgsb_optimize_one_beta(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA

    if abs(diff) > 1e-12:

    return x

import time, math, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def compute_corr_fft(x):
    n = x.shape[0]
    h = x.astype(np.float64)
    r = 1.0 - h
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, n=L)
    fr = np.fft.rfft(r[::-1], n=L)
    conv = np.fft.irfft(fh * fr, n=L)
    return conv[: 2 * n - 1]

def compute_c5_fft(h):
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size), corr

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- simulated annealing ----------
def simulated_annealing(start_h, rng, max_iters, time_limit):

    h = start_h.copy()
    cur_c, cur_corr = compute_c5_fft(h)
    best_h, best_c = h.copy(), cur_c

            break

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

                    n = up_h.size

    # ---------- final local hill‑climb ----------
    cur_h, cur_c = best_h.copy(), best_c
    step = 0.3
    for _ in range(3000):
        if time.time() - t0 > budget_s:
            break
        nh = perturb_h(cur_h, rng, scale=step)
        nc = compute_c5_fft(nh)[0]
        if nc < cur_c:
            cur_c, cur_h = nc, nh
            if nc < best_c:
                best_c, best_h = nc, nh.copy()
        step *= 0.9995

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_fft(best_h)[0]
    return best_h.tolist(), float(final_c), int(best_h.size)
