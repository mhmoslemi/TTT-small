def project_onto_box_simplex(v, S):
    # Find lambda such that sum_i clip(v_i - lambda, 0, 1) = S.
    # Use binary search on lambda.
    # Since clip is monotone decreasing in lambda.
    # lower bound: lambda_low = min(v_i - 1) (i.e., v_i-1)
    # upper bound: lambda_high = max(v_i) (i.e., v_i)
    # Actually we need clip = min(1, max(0, v_i - lambda)).
    # As lambda -> -inf, clip -> 1 (since v_i - lambda becomes large positive) => sum = n.
    # As lambda -> +inf, clip -> 0 => sum = 0.
    # So we can binary search between low = -max(v_i) (to guarantee all >1) and high = max(v_i) (maybe more). Simpler: set low = -1e5, high = 1e5.

def project_box_simplex(v, S, tol=1e-9):

    n = len(h)

    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    alpha = init_alpha
    while time.time() - start < time_limit and alpha > min_alpha:
        max_val, shift = max_overlap_and_shift(h)
        c = max_val * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # compute gradient
        g = np.ones(n)
        if shift >= 0:
            g[0:n-shift] -= h[shift:]  # h_{i+shift}
            g[shift:n] -= h[0:n-shift]  # h_{i-shift}
        else:
            k = -shift
            g[k:n] -= h[0:n-k]
            g[0:n-k] -= h[k:n]
        # step
        h_new = h - alpha * g
        # project
        h = project_box_simplex(h_new, S)
        alpha *= decay
    return best_h, best_c

def max_corr_and_shift_fft(h):
    n = h.size
    N = 2 * n
    a = np.fft.rfft(h, N)  # real FFT
    b = np.fft.rfft(1 - h, N)
    # correlation: ifft( a * conj(b) )
    corr = np.fft.irfft(a * np.conj(b), N)
    # trunc to length 2*n-1: ignore last element (since zero padding at end maybe)
    corr = corr[:2 * n - 1]
    idx = np.argmax(corr)
    max_val = corr[idx]
    shift = idx - (n - 1)
    return max_val, shift

def pair_transfer_optimize(h0, rng, max_iters=20000, time_limit=60.0, init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    while time.time() - start < time_limit and max_iters > 0:
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # get indices for shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            factor = init_factor
            continue
        eps = factor * max_eps
        h[i] -= eps
        h[j] += eps
        # clip
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
        max_iters -= 1
    final_c = compute_c5(best_h)
    return best_h, final_c

import time
import numpy as np

# ---------- projection onto [0,1]^n with a fixed sum ----------
def project_box_simplex(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0  # ensures sum > target
    hi = np.max(v) + 1.0   # ensures sum < target
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > tol:
        mask = (w > 0) & (w < 1)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    return project_box_simplex(h, target)

# ---------- random feasible vector ----------
def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_box_simplex(v, n / 2.0)

# ---------- exact C5 evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def max_overlap_and_shift(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)

# ---------- gradient descent on a smoothed max ----------
def gradient_descent_optimize(h0, time_limit, top_k=2, init_alpha=0.2,
                             decay=0.9995, mu=0.9):
    start = time.time()
    n = h0.size
    S = n / 2.0
    h = h0.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    # initial C5
    best_c = compute_c5(h)
    alpha = init_alpha
    while time.time() - start < time_limit:
        # correlation via FFT
        N = 2 * n
        fa = np.fft.rfft(h, N)
        fb = np.fft.rfft(1.0 - h, N)
        corr = np.fft.irfft(fa * np.conj(fb), N)[:2 * n - 1]
        # top shifts
        idxs = np.argpartition(-corr, top_k)[:top_k]
        # gradient
        g = np.full(n, top_k, dtype=float)
        for idx in idxs:
            shift = idx - (n - 1)
            if shift >= 0:
                g[:n - shift] -= h[shift:]
                g[shift:] -= h[:n - shift]
            else:
                k = -shift
                g[k:] -= h[:n - k]
                g[:n - k] -= h[k:]
        # momentum & projection
        v = mu * v - alpha * g
        h = project_box_simplex(h + v, S)
        # evaluate new C5
        fa2 = np.fft.rfft(h, N)
        fb2 = np.fft.rfft(1.0 - h, N)
        corr2 = np.fft.irfft(fa2 * np.conj(fb2), N)[:2 * n - 1]
        cur_c = np.max(corr2) * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        alpha *= decay
        if alpha < 1e-6:
            break
    return best_h, best_c

# ---------- pair‑transfer local search ----------
def pair_transfer_optimize(h0, rng, max_iters=20000, time_limit=60.0,
                          init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    it = 0
    while time.time() - start < time_limit and it < max_iters:
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            k = -shift
            i_range = np.arange(k, n)
            j_range = i_range - k
        if i_range.size == 0:
            factor = init_factor
            it += 1
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            it += 1
            continue
        eps = factor * max_eps
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
        it += 1
    return best_h, best_c

# ---------- binary local search ----------
def binary_local_search(mask, rng, time_limit=30.0):
    h = mask.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0
        h[j] = 1
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        if rng.random() < 0.001:
            h[:] = 0
            h[rng.choice(n, n // 2, replace=False)] = 1.0
    return best_h, best_c

    start = time.time()

                xs_new = np.linspace(0, 2, n, endpoint=False)

            h0 = enforce_sum(h0, target_sum, rng)

    candidates.append(h)

    best_h = None

    # ----- main optimisation phase -----
    elapsed = time.time() - start

        elapsed = time.time() - start

        h_pt, c_pt = pair_transfer_optimize(best_h, rng,
                                            max_iters=30000,
                                            time_limit=pt_time,
                                            init_factor=0.5, decay=0.999)
        if c_pt < best_c:

    bin_h = np.zeros_like(best_h)

    bin_h[idx] = 1.0

    best_h = enforce_sum(best_h, target_sum, rng)

    return best_h.tolist(), float(final_c), int(n)
