def aggressive_hill_climb(h, rng, max_iters=3000, initial_scale=0.9):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    scale = initial_scale
    for it in range(max_iters):
        grad = worst_gradient(best_h)
        pos = np.argmax(grad)
        neg = np.argmin(grad)
        if grad[pos] <= -grad[neg] + 1e-12:
            # no improvement possible
            break
        max_eps = min(best_h[pos], 1.0 - best_h[neg])
        eps = max_eps * scale
        # optionally random small perturb if max_eps is zero
        if eps <= 0:
            pos = rng.integers(0,n)
            neg = rng.integers(0,n)
            max_eps = min(best_h[pos], 1.0 - best_h[neg])
            eps = max_eps * scale
        nh = best_h.copy()
        nh[pos] -= eps
        nh[neg] += eps
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c = nh, c
            # occasionally increase scale to accelerate
            scale = min(scale*1.05, 0.99)
        else:
            # reduce scale
            scale = max(scale*0.7, 0.01)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders used by the soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1] with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def aggressive_hill_climb(h: np.ndarray, rng: np.random.Generator,
                         max_iters: int = 2000, initial_scale: float = 0.9) -> tuple:
    """Rapid transfer‑based hill climbing targeting the worst shift."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    scale = initial_scale
    for it in range(max_iters):
        grad = worst_gradient(best_h)
        pos = int(np.argmax(grad))
        neg = int(np.argmin(grad))
        if grad[pos] <= -grad[neg] + 1e-12:
            break
        max_eps = min(best_h[pos], 1.0 - best_h[neg])
        if max_eps <= 0.0:
            # fallback to random pair
            pos = rng.integers(0, n)
            neg = rng.integers(0, n)
            max_eps = min(best_h[pos], 1.0 - best_h[neg])
            if max_eps <= 0.0:
                continue
        eps = max_eps * scale
        nh = best_h.copy()
        nh[pos] -= eps
        nh[neg] += eps
        c = compute_c5(nh)
        if c < best_c:
            best_h, best_c = nh, c
            scale = min(scale * 1.05, 0.99)
        else:
            scale = max(scale * 0.7, 0.01)
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function giving a low C5 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # Initialise best solution (use parent if present)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        best_h = enforce_sum(h0, rng)
        best_n = best_h.size
        best_c = compute_c5(best_h)
    else:
        # fallback random binary of size 256
        n0 = 256
        best_h = np.zeros(n0)
        best_h[rng.choice(n0, n0 // 2, replace=False)] = 1.0
        best_h = enforce_sum(best_h, rng)
        best_n = n0
        best_c = compute_c5(best_h)

    # Candidate sizes (keep under 1024)
    n_candidates = [128, 160, 192, 224, 256, 320, 384, 448, 512]

    # Primary multi‑restart search
    while time.time() - start < 0.85 * budget_s:
        n = int(rng.choice(n_candidates))
        use_best = rng.random() < 0.7
        if use_best:
            if best_n == n:
                h = best_h.copy()
            else:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
        else:
            if rng.random() < 0.5:
                h = np.zeros(n)
                h[rng.choice(n, n // 2, replace=False)] = 1.0
            else:
                h = random_feasible_vector(n, rng)

        # Aggressive hill climbing
        max_iters = int(2500 * 256 / n)  # keep effort roughly constant
        h, c = aggressive_hill_climb(h, rng, max_iters=max_iters, initial_scale=0.9)

        # Soft‑max SLSQP refinement (increasing β)
        for beta in (50., 200., 800., 3000., 10000.):
            if time.time() - start > 0.94 * budget_s:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=1200)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # Fine gradient tidy‑up
        h = gradient_refine(h, rng, step=0.015, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # Upsampling and high‑resolution polish
    if best_h is not None and time.time() - start < 0.97 * budget_s:
        for factor in (2, 3):
            n_up = best_n * factor
            if n_up > 1024:
                continue
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = enforce_sum(h_up, rng)

            max_iters = int(3000 * 256 / n_up)
            h_up, c_up = aggressive_hill_climb(
                h_up, rng, max_iters=max_iters, initial_scale=0.9
            )

            for beta in (100., 800., 3000., 12000.):
                if time.time() - start > budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

            h_up = gradient_refine(h_up, rng, step=0.01, max_iter=8)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # Final stochastic nudging
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
