def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # adjust uniformly?
    # We'll distribute diff proportionally to remaining capacity.
    # Use simple loop with random picks as earlier.
    max_iter = 2000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    # After attempts, if still diff, adjust small epsilon uniformly
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h0, target, rng, max_iters, time_start, time_budget):
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = 0.5
    decay = 0.9995
    tol_corr = 1e-12
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        # find worst shifts
        worst_idxs = np.where(np.abs(corr - max_corr) <= tol_corr)[0]
        g = np.zeros_like(h)
        for idx in worst_idxs:
            shift = idx - (n - 1)
            if shift >= 0:
                # forward
                if n - shift > 0:
                    f_i = np.arange(0, n - shift, dtype=int)
                    g[f_i] += (1 - h[f_i + shift])
                # backward
                if n - shift > 0:
                    b_i = np.arange(shift, n, dtype=int)
                    g[b_i] -= h[b_i - shift]
            else:
                s = -shift
                # forward for negative shift
                if n - s > 0:
                    f_i = np.arange(s, n, dtype=int)
                    g[f_i] += (1 - h[f_i - s])
                # backward
                if n - s > 0:
                    b_i = np.arange(0, n - s, dtype=int)
                    g[b_i] -= h[b_i + s]
        # gradient step
        step = lr * 2.0 / n
        h = h - step * g
        # clip and enforce sum
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        lr *= decay
    return best_h, best_c
```

def enforce_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    h_best, best_c = optimize_on_grid(h0, n0, n0 / 2.0, mid_time, rng)

    # upsample to a finer grid if time permits
    if time.time() < t0 + budget_s * 0.9:
        n_up = max(720, n0 * 2)
        h_up = upsample(h_best, n_up, rng)
        end_time = t0 + budget_s
        h_up_best, c_up_best = optimize_on_grid(
            h_up, n_up, n_up / 2.0, end_time, rng)
        if c_up_best < best_c:
            h_best, best_c = h_up_best, c_up_best

    # final local refinement
    remaining = t0 + budget_s - time.time()
    if remaining > 0:
        h_best, best_c = final_hill_climb(
            h_best, rng, max_iters=5000,
            time_start=time.time(),
            time_budget=t0 + budget_s,
            cur_c=best_c)

    # final clean‑up
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = enforce_sum(h_best, h_best.size / 2.0, rng)
    final_c = compute_c5(h_best)
    return h_best.tolist(), float(final_c), int(h_best.size)
