def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    cons = {'type':'eq','fun':sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,

    x = np.clip(res.x, 0.0, 1.0)

    return project_box_sum(x, CURRENT_TARGET)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


# ---------- local moves ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


# ---------- evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise base grid -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ----- build initial population -----
    pop_size = 15
    candidates = [h0]

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in range(1, 6):
        cand = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0))
        cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0))

    # block masks with random offsets
    for _ in range(4):
        mask = np.zeros(n)
        start_idx = rng.integers(0, n)
        length = n // 2
        mask[start_idx : start_idx + length] = 1.0
        candidates.append(project_box_sum(mask, n / 2.0))

    # extra random feasible vectors
    for _ in range(6):
        candidates.append(random_feasible_vector(n, rng))

    # evaluate and keep top pop_size
    cand_c = [compute_c5(c) for c in candidates]
    idx_sort = np.argsort(cand_c)[:pop_size]
    population = [candidates[i].copy() for i in idx_sort]
    pop_c5 = [cand_c[i] for i in idx_sort]

    best_idx = int(np.argmin(pop_c5))
    best_h = population[best_idx].copy()
    best_c = pop_c5[best_idx]

    # ----- evolutionary search (population) -----
    while time.time() - start < budget_s * 0.7:
        elapsed = time.time() - start
        progress = elapsed / (budget_s * 0.7)
        scale = 0.4 * (1 - progress) + 0.02 * progress

        # tournament selection
        t_idx = rng.choice(len(population), size=3, replace=False)
        parent_idx = t_idx[np.argmin([pop_c5[i] for i in t_idx])]
        parent = population[parent_idx]

        # decide offspring generation
        if rng.random() < 0.2 and len(population) >= 2:
            # crossover
            other_idx = parent_idx
            while other_idx == parent_idx:
                other_idx = rng.integers(len(population))
            other = population[other_idx]
            child = (parent + other) * 0.5
            child = project_box_sum(child, n / 2.0)
        else:
            # mutation
            _, idx = compute_c5_and_idx(parent)
            k_star = idx - (n - 1)
            if rng.random() < 0.5:
                child = targeted_move(parent, rng, k_star, scale=scale)
            else:
                child = perturb_h(parent, rng, scale=scale)
            child = np.clip(child, 0.0, 1.0)
            child = project_box_sum(child, n / 2.0)

        child_c = compute_c5(child)

        worst_idx = int(np.argmax(pop_c5))
        if child_c < pop_c5[worst_idx]:
            population[worst_idx] = child
            pop_c5[worst_idx] = child_c
            if child_c < best_c:
                best_c, best_h = child_c, child.copy()

        # early stop if already very good
        if best_c < 0.3808:
            break

    # ----- up‑sample to higher resolution -----
    target_n = min(1080, max(540, n * 2))
    if target_n != n:
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = project_box_sum(h_up, target_n / 2.0)
        best_h = h_up
        n = target_n

    # ----- hill‑climb on high resolution -----
    max_hc_iters = 6000
    step = 0.3
    decay_interval = max(1, max_hc_iters // 10)
    for it in range(max_hc_iters):
        if time.time() - start >= budget_s * 0.95:
            break
        _, idx = compute_c5_and_idx(best_h)
        k_star = idx - (n - 1)
        if rng.random() < 0.7:
            cand = targeted_move(best_h, rng, k_star, scale=step)
        else:
            cand = perturb_h(best_h, rng, scale=step)
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_box_sum(cand, n / 2.0)
        cand_c = compute_c5(cand)
        if cand_c < best_c - 1e-12:
            best_c, best_h = cand_c, cand.copy()
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)

    # ----- SLSQP refinement with increasing beta -----
    for beta in [2000, 5000, 10000, 20000]:
        if time.time() - start >= budget_s:
            break
        try:
            refined = slsqp_optimize(best_h, beta, max_iter=300)
        except Exception:
            continue
        refined_c = compute_c5(refined)
        if refined_c < best_c:
            best_c, best_h = refined_c, refined.copy()

    # ----- final projection and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
