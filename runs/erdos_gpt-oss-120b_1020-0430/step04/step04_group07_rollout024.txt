import time, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(v: np.ndarray) -> np.ndarray:
    n = v.size
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        if x.sum() > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h, dtype=float)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    h = h / h.sum() * (n / 2.0)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h)

def upsample(h: np.ndarray, factor: int) -> np.ndarray:

    x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)

def gradient_descent_optimize(h: np.ndarray, rng: np.random.Generator,
                             max_iters=2000, init_lr=0.5,
                             min_lr=1e-5, decay=0.99):
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    lr = init_lr
    for _ in range(max_iters):
        grad = worst_gradient(h)
        step = lr * grad
        step_adj = np.where(step > 0,
                            np.minimum(step, h),
                            np.maximum(step, -(1.0 - h)))
        h_new = h - step_adj
        h_new = enforce_sum(h_new)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
            lr = min(lr * 1.02, init_lr)
        else:
            lr = max(lr * decay, min_lr)
        h = h_new
    return best_h, best_c

CURRENT_BETA = None
