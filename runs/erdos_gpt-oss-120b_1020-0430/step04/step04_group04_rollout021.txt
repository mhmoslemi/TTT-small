def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # set n_small = 360, n_large = 720
    n_large = 720
    # get initial h at n_large either from upsampled initial_h_values or random.
    if 'initial_h_values' in globals():
        h_small = np.array(initial_h_values, dtype=float)
        n_small = h_small.size
        # Enforce sum
        h_small = enforce_sum(h_small, n_small/2.0, rng)
        # upsample to n_large using linear interpolation
        x_small = (np.arange(n_small) + 0.5) * (2.0 / n_small)
        x_large = (np.arange(n_large) + 0.5) * (2.0 / n_large)
        h_large = np.interp(x_large, x_small, h_small)
        h_large = enforce_sum(h_large, n_large/2.0, rng)
    else:
        # no parent, start random.
        h_large = random_feasible_vector(n_large, rng)
    best_h = h_large
    best_c = compute_c5(best_h)

    # Helper function to maybe update best
    def try_update(h):
        nonlocal best_h, best_c
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Main loop with time check
    while time.time() - start_time < budget_s * 0.6:
        # Random restart 30% probability
        if rng.random() < 0.3:
            start_vec = random_feasible_vector(n_large, rng)
        else:
            start_vec = perturb_h(best_h, rng, scale=0.4)

        # hill climb
        hc_h, hc_c = hill_climb_multi(start_vec, rng, max_iters=3000, init_step=0.5, n_trials=8)
        try_update(hc_h)

        # simulated annealing
        sa_h, sa_c = simulated_annealing(best_h, rng,
                                         max_iter=12000,
                                         init_temp=2e-2,
                                         final_temp=1e-5,
                                         init_scale=0.5,
                                         final_scale=0.02)
        try_update(sa_h)

        # SLSQP ladder
        for beta in (20, 100, 500, 2000, 8000, 20000):
            if time.time() - start_time > budget_s * 0.6:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=800)
            try_update(h_ref)

    # After main loop, try some extra random restarts with SLSQP only
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        h0 = random_feasible_vector(n_large, rng)
        for beta in (100, 2000, 20000):
            if time.time() - start_time > budget_s * 0.9:
                break
            h_opt = slsqp_optimize(h0, beta, max_iter=500)
            try_update(h_opt)

    # final local refinement on best_h with hill climb and SA again
    if time.time() - start_time < budget_s * 0.95:
        # hill climb
        best_h, _ = hill_climb_multi(best_h, rng, max_iters=5000, init_step=0.4, n_trials=10)
        # SA
        best_h, _ = simulated_annealing(best_h, rng,
                                        max_iter=15000,
                                        init_temp=1e-2,
                                        final_temp=1e-6,
                                        init_scale=0.4,
                                        final_scale=0.01)

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n_large/2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n_large)

```
def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)
```

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # Use log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def correlation_fft(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

def softmax_obj(x):
    corr = correlation_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    conv = correlation_fft(h)
    return np.max(conv) * (2.0 / h.shape[0])

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global vars used by the smooth surrogate for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality: sum(x) == CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def correlation_fft(h):
    """Correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv


def softmax_obj(x):
    """Smooth approximation of max_k correlation."""
    corr = correlation_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact C₅ for a sampled step function."""
    conv = correlation_fft(h)
    return np.max(conv) * (2.0 / h.shape[0])


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one entry to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple stochastic hill‑climb."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    """Simulated‑annealing refinement."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h, cur_c = best_h.copy(), best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP refinement of the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with a low Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 720  # final grid size

    # -------------------------------------------------- initialise
    if "initial_h_values" in globals():
        h_small = np.array(initial_h_values, dtype=float)
        n_small = h_small.size
        h_small = enforce_sum(h_small, n_small / 2.0, rng)
        xs_small = (np.arange(n_small) + 0.5) * (2.0 / n_small)
        xs_large = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.interp(xs_large, xs_small, h_small)
        h = enforce_sum(h, n / 2.0, rng)
    else:
        h = random_feasible_vector(n, rng)

    best_h = h
    best_c = compute_c5(best_h)

    def try_update(candidate):
        nonlocal best_h, best_c
        c = compute_c5(candidate)
        if c < best_c:
            best_c, best_h = c, candidate.copy()

    # -------------------------------------------------- three refinement cycles
    for _ in range(3):
        if time.time() - start > budget_s * 0.55:
            break
        start_vec = random_feasible_vector(n, rng) if rng.random() < 0.5 else perturb_h(best_h, rng, scale=0.4)

        hc_h, _ = hill_climb_multi(start_vec, rng,
                                   max_iters=2500,
                                   init_step=0.5,
                                   n_trials=8)
        try_update(hc_h)

        sa_h, _ = simulated_annealing(best_h, rng,
                                      max_iter=10000,
                                      init_temp=2e-2,
                                      final_temp=1e-5,
                                      init_scale=0.5,
                                      final_scale=0.02)
        try_update(sa_h)

        for beta in (20, 100, 500, 2000, 8000, 20000):
            if time.time() - start > budget_s * 0.55:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=600)
            try_update(h_ref)

    # -------------------------------------------------- random SLSQP restarts
    while time.time() - start < budget_s * 0.85:
        h0 = random_feasible_vector(n, rng)
        for beta in (100, 2000, 8000, 20000):
            if time.time() - start > budget_s * 0.85:
                break
            h_opt = slsqp_optimize(h0, beta, max_iter=500)
            try_update(h_opt)

    # -------------------------------------------------- final fine‑grained search
    if time.time() - start < budget_s * 0.95:
        best_h, _ = hill_climb_multi(best_h, rng,
                                     max_iters=6000,
                                     init_step=0.3,
                                     n_trials=12)
        best_h, _ = simulated_annealing(best_h, rng,
                                        max_iter=15000,
                                        init_temp=5e-3,
                                        final_temp=1e-6,
                                        init_scale=0.3,
                                        final_scale=0.01)

        # one last high‑β SLSQP polish
        if time.time() - start < budget_s * 0.99:
            h_final = slsqp_optimize(best_h, beta=50000, max_iter=800)
            try_update(h_final)

    # -------------------------------------------------- clean‑up and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
