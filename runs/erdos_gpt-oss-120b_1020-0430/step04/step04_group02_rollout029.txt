def subgradient_descent(h0, max_iters, step0, rng):
    h = h0.copy()
    n = h.size
    delta = 2.0 / n
    for it in range(max_iters):
        autocorr = np.correlate(h, h, mode='full')
        total = h.sum()
        g = delta * (total - autocorr)
        max_g = np.max(g)
        # indices within tolerance as active set
        active = np.where(g >= max_g - 1e-12)[0]
        # choose random active shift
        k_idx = rng.choice(active)
        k = k_idx - (n - 1)   # shift value
        # compute gradient
        # gradient = -delta * (shift(h, k) + shift(h, -k))
        grad = np.zeros_like(h)
        if k >= 0:
            grad[:n-k] -= delta * h[k:]   # shift h by -k (i.e., h_{i+k})
            grad[k:]   -= delta * h[:n-k] # shift h by +k (i.e., h_{i-k})
        else:
            k2 = -k
            grad[k2:] -= delta * h[:n-k2]
            grad[:n-k2] -= delta * h[k2:]
        # step size schedule
        step = step0 / (1.0 + it / (max_iters * 0.5))
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h = project_box_sum(h_new, n / 2.0)
    return h

import time, numpy as np
from scipy.optimize import minimize

# ---- basic utilities -------------------------------------------------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) * 0.5, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)

def compute_c5(h):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low.any() or not high.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low))
    i_high = rng.choice(np.flatnonzero(high))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i_low] += eps
    new[i_low + k] += eps
    new[i_high] -= eps
    new[i_high + k] -= eps
    return new

# ---- SLSQP refinement (softmax surrogate) ---------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj, h0, method="SLSQP", bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

def smooth_max_descent(h0, beta_list, steps_per_beta, step0, rng):

    delta = 2.0 / n
    for beta in beta_list:

        if k >= 0:
            grad[: n - k] -= delta * h[k:]
            grad[k:]    -= delta * h[: n - k]
        else:
            km = -k
            grad[km:]   -= delta * h[: n - km]
            grad[: n - km] -= delta * h[km:]
        step = step0 / (1.0 + it / (max_iters * 0.5))
        h_new = h - step * grad
        h = project_box_sum(np.clip(h_new, 0.0, 1.0), n / 2.0)
    return h

# ---- main run -------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = project_box_sum(np.clip(h0, 0.0, 1.0), n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # quick diversifiers
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]
    for f in (1, 2, 3):
        cand = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0))
        cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0))
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # -----------------------------------------------------------------
    # Phase 1: hill‑climbing + occasional SLSQP (first half of time)
    while time.time() - start < budget_s * 0.5:
        start_vec = perturb_h(best_h, rng, scale=0.4) if rng.random() < 0.8 else random_feasible_vector(n, rng)
        h_hc, c_hc = hill_climb_targeted(start_vec, rng,
                                          max_iters=1200, init_step=0.5,
                                          n_trials=6, p_target=0.3)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
        # gentle SLSQP refinement
        if best_c < 0.382:
            for beta in (20, 80, 200, 500):
                if time.time() - start >= budget_s * 0.5:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=300)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # -----------------------------------------------------------------
    # Phase 2: smooth‑max gradient descent (up to ~80% time)
    beta_sched = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]
    while time.time() - start < budget_s * 0.8:
        h = smooth_max_descent(best_h, beta_sched, steps_per_beta=120,
                               step0=0.2, rng=rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # -----------------------------------------------------------------
    # Phase 3: subgradient descent (till ~95% time)
    while time.time() - start < budget_s * 0.95:
        h = subgradient_descent(best_h, max_iters=500, step0=0.3, rng=rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # -----------------------------------------------------------------
    # Final local refinement
    while time.time() - start < budget_s - 2.0:
        h_hc, c_hc = hill_climb_targeted(best_h, rng,
                                          max_iters=800, init_step=0.3)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

    # final projection and return
    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
