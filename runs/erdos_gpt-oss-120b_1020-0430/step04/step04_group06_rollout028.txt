import time
import numpy as np

def _max_corr(arr: np.ndarray) -> int:
    """Maximum of the overlap sum for binary array `arr`."""
    # arr should be int 0/1
    comp = 1 - arr
    # np.correlate works with int arrays; result is int
    return int(np.max(np.correlate(arr, comp, mode="full")))


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise a balanced binary vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_cont = np.clip(np.array(initial_h_values, dtype=float), 0.0, 1.0)
        n = h_cont.size
        target = n // 2
        h_bin = np.rint(h_cont).astype(np.int32)          # 0/1 rounding
        diff = target - h_bin.sum()
        if diff > 0:   # need extra ones
            zero_idx = np.where(h_bin == 0)[0]
            if zero_idx.size:
                order = np.argsort(-h_cont[zero_idx])[:diff]
                h_bin[zero_idx[order]] = 1
        elif diff < 0:  # too many ones
            one_idx = np.where(h_bin == 1)[0]
            if one_idx.size:
                order = np.argsort(h_cont[one_idx])[: -diff]
                h_bin[one_idx[order]] = 0
        h = h_bin
    else:
        n = 360
        target = n // 2
        h = np.zeros(n, dtype=np.int32)
        ones = rng.choice(n, target, replace=False)
        h[ones] = 1

    # ------------------------------------------------------------------
    # Helper to compute full correlation and its max (used many times)
    # ------------------------------------------------------------------
    def _corr_and_max(v):
        c = np.correlate(v, 1 - v, mode="full")
        return c, int(c.max())

    # ------------------------------------------------------------------
    # 2. Initial evaluation
    # ------------------------------------------------------------------
    cur_corr, cur_max = _corr_and_max(h)
    best_h = h.copy()
    best_max = cur_max

    # ------------------------------------------------------------------
    # 3. Hill‑climbing / simulated annealing on binary vectors
    # ------------------------------------------------------------------
    max_T, min_T = 1.0, 1e-4
    target_max = int(np.floor(0.3808 * n / 2))

    while True:
        now = time.time()
        if now - start_time >= budget_s * 0.95:
            break

        # temperature schedule (exponential cooling)
        frac = (now - start_time) / budget_s
        T = max_T * (min_T / max_T) ** frac

        # recompute correlation for the current state (already have cur_corr)
        # decide move type
        if rng.random() < 0.3:  # targeted move
            idx_max = int(np.argmax(cur_corr))
            shift = idx_max - (n - 1)

            # indices that actually contribute to the current worst shift
            if shift >= 0:
                i_range = np.arange(n - shift)
                j_range = i_range + shift
            else:
                i_range = np.arange(-shift, n)
                j_range = i_range + shift
            contrib = (h[i_range] == 1) & (h[j_range] == 0)
            i_candidates = i_range[contrib]
            if i_candidates.size == 0:
                # fall back to random swap
                ones_idx = np.where(h == 1)[0]
                zeros_idx = np.where(h == 0)[0]
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
            else:
                i = rng.choice(i_candidates)

                # prefer zeros that do NOT create a new (1,0) pair for this shift
                zeros_idx = np.where(h == 0)[0]
                if shift >= 0:
                    safe = (zeros_idx + shift >= n) | (h[zeros_idx + shift] == 1)
                else:
                    safe = (zeros_idx + shift < 0) | (h[zeros_idx + shift] == 1)
                safe_zeros = zeros_idx[safe]
                if safe_zeros.size:
                    j = rng.choice(safe_zeros)
                else:
                    j = rng.choice(zeros_idx)
        else:  # pure random swap
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)

        # perform swap
        h[i], h[j] = 0, 1
        new_corr, new_max = _corr_and_max(h)
        delta = new_max - cur_max

        # Metropolis acceptance
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_corr, cur_max = new_corr, new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                if best_max <= target_max:
                    # already at or below the target – keep searching a bit more
                    pass
        else:
            # revert swap
            h[i], h[j] = 1, 0

    # ------------------------------------------------------------------
    # 4. Final clean‑up and output
    # ------------------------------------------------------------------
    final_max = _max_corr(best_h)
    final_c5 = final_max * (2.0 / n)

    return best_h.astype(float).tolist(), float(final_c5), int(n)
