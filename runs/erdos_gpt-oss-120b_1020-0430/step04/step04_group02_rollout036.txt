def compute_c5_fft(h):

    H = np.fft.rfft(h, Lc)
    B = np.fft.rfft((1 - h)[::-1], Lc)
    corr = np.fft.irfft(H * B, Lc)[:2*n - 1]
    return np.max(corr) * dx

def hill_climb_simple(h, rng, max_iters, step):
    best_h = h.copy()
    best_c = compute_c5_fft(best_h)
    n = h.size
    for _ in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # move epsilon from i to j within bounds
        eps_max = min(best_h[i], 1.0 - best_h[j])
        if eps_max <= 0: continue
        eps = (rng.random() - 0.5) * 2 * eps_max * step
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_box_sum(cand, n/2.0)
        c = compute_c5_fft(cand)
        if c < best_c:
            best_c, best_h = c, cand
    return best_h, best_c

import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length() if x > 1 else 1

def _project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def _random_feasible_vector(n, rng):
    return _project_box_sum(rng.random(n), n / 2.0)

def _compute_c5_fft(h):
    n = h.size
    dx = 2.0 / n
    L = _next_pow2(2 * n - 1)
    H = np.fft.rfft(h, L)
    B = np.fft.rfft((1.0 - h)[::-1], L)
    corr = np.fft.irfft(H * B, L)[: 2 * n - 1]
    return np.max(corr) * dx

def _fast_step(h, beta, step, Lc, target_sum):
    n = h.size
    H = np.fft.rfft(h, Lc)
    H_rev = np.fft.rfft(h[::-1], Lc)
    # autocorrelation → f_k values
    conv_auto = np.fft.irfft(H * H_rev, Lc)[: 2 * n - 1]
    auto_shifts = conv_auto[::-1]                     # shift k = i-(n-1)
    f = h.sum() - auto_shifts                         # f_k = sum_h - autocorr_k
    m = f.max()
    w = np.exp(beta * (f - m))
    w /= w.sum()
    w_rev = w[::-1]
    w_sum = w + w_rev
    W_sum = np.fft.rfft(w_sum, Lc)
    conv_sum = np.fft.irfft(H * W_sum, Lc)[:n]       # A_i + B_i
    grad = 1.0 - conv_sum
    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _project_box_sum(h_new, target_sum)
    c5_new = _compute_c5_fft(h_new)
    return h_new, c5_new

def _hill_climb_simple(h, rng, max_iters, step):
    n = h.size
    best_h = h.copy()
    best_c = _compute_c5_fft(best_h)
    for _ in range(max_iters):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_max = min(best_h[i], 1.0 - best_h[j])
        if eps_max <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * eps_max * step
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_box_sum(cand, n / 2.0)
        c = _compute_c5_fft(cand)
        if c < best_c:
            best_c, best_h = c, cand
    return best_h, best_c

def _upsample(h, n_new):
    n_old = h.size
    old_x = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    new_x = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_up = np.interp(new_x, old_x, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    h_up = _project_box_sum(h_up, n_new / 2.0)
    return h_up

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project_box_sum(h0, n / 2.0)
    else:
        n = 360
        h0 = _random_feasible_vector(n, rng)

    cur_h = h0.copy()
    cur_c = _compute_c5_fft(cur_h)
    best_h, best_c = cur_h.copy(), cur_c

    # main gradient‐descent phase (low‑n)
    Lc = _next_pow2(3 * n - 2)
    step = 0.5
    min_step = 1e-5
    step_decay = 0.9995
    restart_prob = 0.001

    while True:
        now = time.time()
        if now - start > budget_s * 0.92:
            break
        prog = (now - start) / budget_s
        beta = 0.5 + (2000.0 - 0.5) * (prog ** 1.5)
        cur_h, cur_c = _fast_step(cur_h, beta, step, Lc, n / 2.0)
        if cur_c < best_c:
            best_c, best_h = cur_c, cur_h.copy()
        step = max(step * step_decay, min_step)
        if rng.random() < restart_prob:
            if rng.random() < 0.5:
                cur_h = _random_feasible_vector(n, rng)
            else:
                pert = cur_h + rng.normal(scale=0.05, size=n)
                cur_h = np.clip(pert, 0.0, 1.0)
                cur_h = _project_box_sum(cur_h, n / 2.0)
            cur_c = _compute_c5_fft(cur_h)

    # optional up‑sampling if time permits
    remaining = budget_s - (time.time() - start)
    if remaining > budget_s * 0.15 and n < 720:
        n2 = min(720, max(540, n * 2))
        best_h = _upsample(best_h, n2)
        n = n2
        Lc = _next_pow2(3 * n - 2)
        cur_h = best_h.copy()
        step = 0.5
        min_step = 1e-5
        step_decay = 0.9995

            now = time.time()
            if now - start > budget_s * 0.96:
                break
            prog = (now - start) / budget_s
            beta = 0.5 + (2000.0 - 0.5) * (prog ** 1.5)
            cur_h, cur_c = _fast_step(cur_h, beta, step, Lc, n / 2.0)
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
            step = max(step * step_decay, min_step)

    if cur_c < best_c:
        best_c, best_h = cur_c, cur_h.copy()

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, best_h.size / 2.0)
    final_c = _compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
