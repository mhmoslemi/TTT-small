def slsqp_optimize(h0, beta, max_iter=2000, ftol=1e-12):
    n = h0.size
    target = n/2.0

    def obj(x):
        corr = np.correlate(x, 1-x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp
        lse = m + np.log(np.sum(np.exp(bc - m)))
        return lse / beta

    def grad(x):
        corr = np.correlate(x, 1-x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        exp_bc = np.exp(bc - m)
        w = exp_bc / np.sum(exp_bc)
        # compute weighted sum of gradient_shift
        n_ = x.size
        g = np.zeros_like(x)
        # loop over all shifts
        for idx, weight in enumerate(w):
            s = idx - (n_ - 1)
            # gradient shift
            # compute efficiently? gradient_shift uses slices; okay
            g += weight * gradient_shift(x, s)
        return g

    bounds = [(0.0,1.0)] * n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target}
    # Provide Jacob for constraint automatically via SciPy? Could also provide constraint jac, but not needed.

    res = minimize(obj, h0, method='SLSQP', jac=grad, bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol, 'disp': False})
    x = np.clip(res.x,0.0,1.0)
    rng = np.random.default_rng()
    # enforce sum exactly due to numeric
    x = enforce_sum(x, target, rng)
    return x

    g = conv1[offset:offset+n] - conv2[offset:offset+n]

def slsqp_optimize(h0, beta, max_iter=2000):
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        max_bc = np.max(bc)
        lse = max_bc + np.log(np.sum(np.exp(bc - max_bc)))
        return lse / beta

    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        max_bc = np.max(bc)
        exp_bc = np.exp(bc - max_bc)
        sum_exp = np.sum(exp_bc)
        w = exp_bc / sum_exp
        # compute weighted gradient using convolution
        rev = w[::-1]
        conv1 = np.convolve(rev, 1 - x, mode='full')
        conv2 = np.convolve(w, x, mode='full')
        offset = len(w) - 1
        g = conv1[offset:offset+n] - conv2[offset:offset+n]
        return g

    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    res = minimize(obj, h0, method='SLSQP', jac=grad, bounds=bounds,
                    constraints=cons,
                    options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x_opt = np.clip(res.x, 0.0, 1.0)
    x_opt = project_to_sum(x_opt, target)
    return x_opt

import time
import numpy as np
from scipy.optimize import minimize

# ------------------- utilities -------------------
def project_to_sum(v, target, rng=None):
    """Project v onto the cube [0,1]^n intersect {sum = target}."""
    y = np.clip(v, 0.0, 1.0)
    s = y.sum()
    if abs(s - target) < 1e-12:
        return y
    # simple water‑filling iteration
    for _ in range(100):
        diff = target - s
        free = (y > 0) & (y < 1)
        if not free.any():
            break
        delta = diff / free.sum()
        y[free] += delta
        y = np.clip(y, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) < 1e-12:
            break
    # tiny fix if still off
    if abs(s - target) > 1e-9:
        if rng is None:
            rng = np.random.default_rng()
        i = rng.integers(0, y.size)
        y[i] = np.clip(y[i] + target - s, 0.0, 1.0)
    return y


def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_to_sum(v, n / 2.0, rng)


def compute_c5(h):
    """Discrete C5 for vector h on [0,2] with uniform spacing."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_random(h, rng, scale=0.2):
    """Transfer a small amount of mass from one index to another."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def gradient_shift(h, s):
    """Gradient of overlap for a single shift s."""
    n = h.size
    g = np.zeros(n)
    if s >= 0:
        g[: n - s] += (1 - h[s:])
        g[s:] -= h[: n - s]
    else:
        s = -s
        g[s:] += (1 - h[: n - s])
        g[: n - s] -= h[s:]
    return g


def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]


def greedy_optimize(h, max_steps=2000):
    """Deterministic pairwise transfer improving the worst shift."""
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)

        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break

        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break

        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue

        # try half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h


# ------------------- SLSQP with analytic gradient -------------------
def slsqp_optimize(h0, beta, max_iter=1200):
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        lse = m + np.log(np.sum(np.exp(bc - m)))
        return lse / beta

    def grad(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        exp_bc = np.exp(bc - m)
        sum_exp = np.sum(exp_bc)
        w = exp_bc / sum_exp  # softmax weights, length 2n‑1

        rev = w[::-1]
        conv1 = np.convolve(rev, 1 - x, mode="full")
        conv2 = np.convolve(w, x, mode="full")
        off = len(w) - 1
        return conv1[off:off + n] - conv2[off:off + n]

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        jac=grad,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_to_sum(x, target)
