def project_onto_sum_box(x, target, max_iter=50):
    lo = -1e3
    hi = 1e3
    for _ in range(max_iter):
        mid = (lo + hi)/2
        y = np.clip(x + mid, 0.0, 1.0)  # shift by λ
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + lo, 0.0, 1.0)
    # ensure sum close; adjust residual
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # add diff to smallest positive slack maybe
        # to keep within bounds, we can distribute to non-boundary values
        mask = (y > 0) & (y < 1)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def project_onto_hyperplane_box(x, target, max_iter=60):
    lo = -1.0
    hi = 1.0
    # Because shift λ could be between -1 and 1; but w/ extreme values maybe bigger.
    # We'll instead compute bounds as min(-x) <= λ <= 1 - max(x)
    lo = -np.max(x)  # to make x+λ >=0
    hi = 1.0 - np.min(x)  # to make x+λ <=1
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + lo, 0.0, 1.0)
    # occasionally adjust residual
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # distribute to free variables
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

import time
import numpy as np

# -------------------------------------------------
# Core utilities

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_hyperplane_box(x, target, max_iter=60):
    """Project x onto {0<=y<=1, sum(y)=target} by binary search on a shift λ."""
    lo = -np.max(x)                # ensures x+λ >= 0 after clipping
    hi = 1.0 - np.min(x)           # ensures x+λ <= 1 after clipping
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + lo, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def max_shift_gradient(h):
    """Gradient of the overlap for the shift achieving the maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(n - shift)
        grad[i] += 1.0 - h[i + shift]
        grad[i + shift] -= h[i]
    else:
        k = -shift
        i = np.arange(k, n)
        grad[i] += 1.0 - h[i - k]
        grad[i - k] -= h[i]
    return grad, shift


def gradient_descent_optimize(h0, target, time_limit, step=0.03,
                              decay=0.999, min_step=1e-5):
    """Projected gradient descent on max‑overlap objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        grad, _ = max_shift_gradient(h)
        new_h = h - step * grad
        new_h = project_onto_hyperplane_box(new_h, target)
        c = compute_c5(new_h)
        if c < best_c - 1e-12:
            best_c, best_h = c, new_h.copy()
        else:
            step = max(step * decay, min_step)
        h = new_h
    return best_h, best_c


def max_overlap_and_shift(h):
    """Return (max correlation value, shift) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Move mass from i to j along the worst‑overlap shift to lower the max."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        # index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            k = -shift
            i_range = np.arange(k, n)
            j_range = i_range - k
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    # final check
    final_c = compute_c5(best_h)
    return best_h, final_c


def upscale(h, n_new):
    """Linear interpolation of h to a finer grid of size n_new."""
    n_old = h.size
    xs_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xs_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(xs_new, xs_old, h)


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    x = rng.random(n)
    target = n / 2.0
    return project_onto_hyperplane_box(x, target)


# -------------------------------------------------
# Main routine
# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- Stage 1: moderate resolution ---------------------------------
    n1 = 540
    target1 = n1 / 2.0

    if "initial_h_values" in globals():
        h0_raw = np.array(initial_h_values, dtype=float)
        if h0_raw.size != n1:
            h0 = upscale(h0_raw, n1)
        else:
            h0 = h0_raw.copy()
    else:
        h0 = random_feasible_vector(n1, rng)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_hyperplane_box(h0, target1)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # allocate time for stage 1 (≈60% of budget)
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    stage1_time = min(remaining * 0.6, 500.0)

    # gradient descent
    h_gd, c_gd = gradient_descent_optimize(best_h, target1, stage1_time,
                                           step=0.03, decay=0.999, min_step=1e-5)
    if c_gd < best_c:
        best_h, best_c = h_gd, c_gd

    # pair‑transfer on remaining time of stage 1
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 20.0:
        h_pt, c_pt = pair_transfer_optimize(best_h, rng,
                                            max_iters=20000,
                                            time_limit=remaining,
                                            init_factor=0.5,
                                            decay=0.999)
        if c_pt < best_c:
            best_h, best_c = h_pt, c_pt

    # ---- Stage 2: higher resolution -----------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30.0:
        n2 = 720
        target2 = n2 / 2.0
        h2 = upscale(best_h, n2)
        h2 = np.clip(h2, 0.0, 1.0)
        h2 = project_onto_hyperplane_box(h2, target2)

        # split remaining time between GD and pair‑transfer
        stage2_time = remaining * 0.5
        h_gd2, c_gd2 = gradient_descent_optimize(h2, target2, stage2_time,
                                                step=0.04, decay=0.999, min_step=1e-5)
        if c_gd2 < best_c:
            best_h, best_c = h_gd2, c_gd2

        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining > 10.0:
            h_pt2, c_pt2 = pair_transfer_optimize(best_h, rng,
                                                  max_iters=25000,
                                                  time_limit=remaining,
                                                  init_factor=0.5,
                                                  decay=0.999)
            if c_pt2 < best_c:
                best_h, best_c = h_pt2, c_pt2

    # ---- Final clean‑up ------------------------------------------------
    n_final = best_h.size
    target_final = n_final / 2.0
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_hyperplane_box(best_h, target_final)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n_final)
