            grad[: n_cur - k] += best_h[k:]
            grad[k:] += best_h[: n_cur - k]
        i = np.argmax(grad)
        j = np.argmin(grad)
        max_eps = min(1.0 - best_h[i], best_h[j])
        if max_eps <= 0:
            break
        eps = max_eps * 0.02 * rng.random()
        nh = best_h.copy()
        nh[i] += eps
        nh[j] -= eps
        nh = np.clip(nh, 0.0, 1.0)
        nh = enforce_sum(nh)
        c_new = compute_c5(nh)
        if c_new < best_c:
            best_c, best_h = c_new, nh.copy()

    # -------------------------------------------------------------
    # Final sanity check and return
    # -------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
