def project_bounds_sum(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # compute diff
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Determine indices where we can increase (if diff>0) or decrease (diff<0)
    # For diff>0: capacity_i = 1 - x[i]; for diff<0: capacity_i = x[i]
    # We can distribute diff uniformly across active set, but need to respect capacities.
    # Implement iterative water-filling:
    idx = np.arange(len(x))
    if diff > 0:
        caps = 1 - x
        active = caps > 1e-12
        while diff > 1e-12 and np.any(active):
            # number of active
            n_active = active.sum()
            # amount to add per active
            add = diff / n_active
            # limit add to capacity
            limited = np.minimum(add, caps[active])
            # add limited to x[active]
            x[active] += limited
            diff = target - x.sum()
            # update active set
            caps = 1 - x
            active = caps > 1e-12
    else:
        caps = x  # amount we can decrease
        active = caps > 1e-12
        while diff < -1e-12 and np.any(active):
            n_active = active.sum()
            sub = (-diff) / n_active
            limited = np.minimum(sub, caps[active])
            x[active] -= limited
            diff = target - x.sum()
            caps = x
            active = caps > 1e-12
    return x

def anneal(h, rng, target, total_time, start_time):
    T0 = 0.1
    while time.time() - start_time < total_time:
        # compute remaining fraction
        elapsed = time.time() - start_time
        fraction = elapsed / total_time
        T = T0 * (1 - fraction)  # linear cooling
        # propose move
        i = rng.integers(0, len(h))
        j = rng.integers(0, len(h))
        if i == j:
            continue
        # max allowed transfer limited by capacity
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * (T/T0+0.01) # scale with temperature maybe
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        # ensure sum still target? Transfer keeps sum constant.
        c = compute_c5(nh)
        delta = c - compute_c5(h) # compute current c maybe precompute
        # Actually compute c_h once per outer loop
        # Accept if improve or prob.
        if c < best_c or rng.random() < np.exp(-delta / max(T, 1e-9)):
            h = nh
            if c < best_c:
                best_c = c
                best_h = h.copy()
        # stop when time triggers
    return best_h, best_c

def project_bounds_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Determine sign
    if diff > 0:
        # need to add mass
        # capacity to increase each coordinate
        caps = 1 - x
        # keep active set where caps>0
        while diff > 1e-12:
            active = caps > 1e-12
            if not np.any(active):
                break
            n_active = active.sum()
            add = diff / n_active
            # limit
            add_limited = np.minimum(add, caps[active])
            x[active] += add_limited
            diff = target - x.sum()
            caps = 1 - x
    else:
        # need to remove mass
        caps = x  # amount we can remove
        while diff < -1e-12:
            active = caps > 1e-12
            if not np.any(active):
                break
            n_active = active.sum()
            sub = (-diff) / n_active
            sub_limited = np.minimum(sub, caps[active])
            x[active] -= sub_limited
            diff = target - x.sum()
            caps = x
    # final clip in case of numeric inaccuracies
    return np.clip(x, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------
# Helper functions
# ----------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# deterministic projection onto [0,1] and sum = target
def project_bounds_sum(x: np.ndarray, target: float) -> np.ndarray:
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # add mass
    if diff > 0:
        caps = 1.0 - x
        while diff > 1e-12:
            active = caps > 1e-12
            if not np.any(active):
                break
            n_active = active.sum()
            add = diff / n_active
            add_limited = np.minimum(add, caps[active])
            x[active] += add_limited
            diff = target - x.sum()
            caps = 1.0 - x
    else:  # remove mass
        caps = x
        while diff < -1e-12:
            active = caps > 1e-12
            if not np.any(active):
                break
            n_active = active.sum()
            sub = (-diff) / n_active
            sub_limited = np.minimum(sub, caps[active])
            x[active] -= sub_limited
            diff = target - x.sum()
            caps = x
    return np.clip(x, 0.0, 1.0)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1] (deterministic)."""
    return project_bounds_sum(h, len(h) / 2.0)


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector on [0,1] with the required sum."""
    v = rng.random(n)
    return enforce_sum(v, rng)


# ----------------------------------------------
# Gradient and hill‑climbing utilities
# ----------------------------------------------

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


# ----------------------------------------------
# Soft‑max SLSQP optimisation
# ----------------------------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (deterministic)
    x = project_bounds_sum(x, CURRENT_TARGET)
    return x
