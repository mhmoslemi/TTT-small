import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp (smooth max)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def adjust_sum_exact(h, rng):
    """Redistribute to hit Σh = n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def slsqp_optimize(h0, beta, rng, max_iter=600):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def anneal_optimize(h, rng, deadline):
    """Directed simulated‑annealing on the worst shift."""
    n = h.size
    adjust_sum_exact(h, rng)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = corr.max()
    best_h = h.copy()
    best_raw = cur_max
    offsets = np.arange(-(n - 1), n)               # length 2n‑1
    h_pad = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
    init_T = max(0.1, best_raw * 0.02)
    it = 0
    while time.time() < deadline:
        # worst shift
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)
        # gradient for this shift
        if k >= 0:
            h_plus = np.empty(n)
            h_plus[:n - k] = h[k:]
            h_plus[n - k:] = 0.0
            h_minus = np.empty(n)
            h_minus[k:] = h[:n - k]
            h_minus[:k] = 0.0
        else:
            kp = -k
            h_plus = np.empty(n)
            h_plus[kp:] = h[: n - kp]
            h_plus[:kp] = 0.0
            h_minus = np.empty(n)
            h_minus[: n - kp] = h[kp:]
            h_minus[n - kp :] = 0.0
        grad = (1.0 - h_plus) - h_minus
        src = np.where((grad > 1e-12) & (h > 1e-12))[0]
        trg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if src.size == 0 or trg.size == 0:
            # fallback random move
            i = rng.integers(0, n)
            if h[i] <= 1e-12:
                continue
            j = rng.integers(0, n)
            if i == j or h[j] >= 1.0 - 1e-12:
                continue
            max_delta = min(h[i], 1.0 - h[j])
        else:
            i = rng.choice(src)
            j = rng.choice(trg)
            max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        # step size decays
        step = 0.02 * (0.995 ** it)
        delta = min(max_delta, step)
        # grad vectors for i and j (full shift space)
        base_i = i + (n - 1)
        g_i = (1.0 - h_pad[base_i + offsets]) - h_pad[base_i - offsets]
        base_j = j + (n - 1)
        g_j = (1.0 - h_pad[base_j + offsets]) - h_pad[base_j - offsets]
        new_corr = corr + delta * (g_j - g_i)
        new_max = new_corr.max()
        T = init_T * (0.9997 ** it)
        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / (T + 1e-12)):
                accept = True
        if accept:
            h[i] -= delta
            h[j] += delta
            h_pad[n - 1 + i] = h[i]
            h_pad[n - 1 + j] = h[j]
            corr = new_corr
            cur_max = new_max
            if new_max < best_raw:
                best_raw = new_max
                best_h = h.copy()
        it += 1
    dx = 2.0 / n
    return best_h, best_raw * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    # initialise
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 360
        h0 = np.zeros(n)
        h0[: n // 2] = 1.0
        adjust_sum_exact(h0, rng)

    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    best_n = n

    # SLSQP cascade
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0)
    for beta in betas:
        if time.time() > deadline:
            break
        try:
            h0 = slsqp_optimize(h0, beta, rng, max_iter=800)
        except Exception:
            continue
        c = compute_c5(h0)
        if c < best_c:
            best_c, best_h = c, h0.copy()
            best_n = n

    # Annealing refinement
    if time.time() < deadline:
        h_opt, c_opt = anneal_optimize(best_h.copy(), rng, deadline)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
            best_n = n

    # optional up‑sampling (one level) if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        # single SLSQP pass at high beta
        if time.time() < deadline:
            h_up = slsqp_optimize(h_up, 5000.0, rng, max_iter=800)
        # final anneal
        if time.time() < deadline:
            h_up, c_up = anneal_optimize(h_up, rng, deadline)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up, n_up

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
