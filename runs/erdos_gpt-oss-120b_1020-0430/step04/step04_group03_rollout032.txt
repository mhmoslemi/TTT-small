import time, numpy as np

def proj_cap_simplex(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return w

def random_feasible_vector(n, rng):
    # Sample uniformly random vector then project
    v = rng.random(n)
    return proj_cap_simplex(v, n / 2.0)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s = -s
        grad[s:] += (1 - h[:n - s])
        grad[:n - s] -= h[s:]
    return grad

def gradient_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    beta_corr = beta * corr
    m = np.max(beta_corr)
    exp_shift = np.exp(beta_corr - m)
    sum_exp = exp_shift.sum()
    weights = exp_shift / sum_exp
    grad = np.zeros(n)
    # loops over shift indices
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        s = idx - (n - 1)
        grad += w * gradient_shift(h, s)
    return grad * (2.0 / n)

def softmax_objective(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    beta_corr = beta * corr
    m = np.max(beta_corr)
    # log-sum-exp stable
    lse = (m + np.log(np.sum(np.exp(beta_corr - m)))) / beta
    return lse * (2.0 / n)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        # compute worst shift s
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        s = idx - (h.size - 1)
        # gradient of worst shift
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full move
        h_new = h.copy()
        h_new[i] -= max_eps
        h_new[j] += max_eps
        if compute_c5(h_new) + 1e-12 < compute_c5(h):
            h = h_new
            continue
        # half move
        eps = max_eps * 0.5
        h_new[i] -= eps
        h_new[j] += eps
        if compute_c5(h_new) + 1e-12 < compute_c5(h):
            h = h_new
            continue
        break
    return h

def pgd_optimize(h0, beta, max_iter=200, init_step=0.2):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_obj = softmax_objective(h, beta)
    step = init_step
    for _ in range(max_iter):
        grad = gradient_softmax(h, beta)
        # take step
        h_new = proj_cap_simplex(h - step * grad, target)
        obj_new = softmax_objective(h_new, beta)
        if obj_new < best_obj - 1e-12:
            best_h = h_new.copy()
            best_obj = obj_new
            step *= 1.05
        else:
            step *= 0.5
        h = h_new
        if step < 1e-8:
            break
    return best_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine dimension
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 720
        h = random_feasible_vector(n, rng)
    # Ensure feasibility
    h = proj_cap_simplex(np.clip(h, 0.0, 1.0), n / 2.0)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Multi-stage optimization
    # Stage 1: PGD with increasing beta
    betas = [10, 100, 500, 2000, 5000]
    for beta in betas:
        if time.time() - start > budget_s * 0.80:
            break
        # PGD
        h_pg = pgd_optimize(best_h, beta, max_iter=300, init_step=0.3)
        c_pg = compute_c5(h_pg)
        if c_pg < best_c:
            best_h, best_c = h_pg, c_pg
        # Greedy refinement
        if time.time() - start > budget_s * 0.80:
            break
        h_g = greedy_optimize(best_h, max_steps=2000)
        c_g = compute_c5(h_g)
        if c_g < best_c:
            best_h, best_c = h_g, c_g

        # Random perturbation and local greedy
        for _ in range(5):
            if time.time() - start > budget_s * 0.85:
                break
            cand = perturb_random(best_h, rng, scale=0.1)
            cand = greedy_optimize(cand, max_steps=500)
            c_cand = compute_c5(cand)
            if c_cand < best_c:
                best_h, best_c = cand, c_cand

    # Final fine perturbations
    while time.time() - start < budget_s * 0.98:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_cap_simplex(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

import time, numpy as np

# ---------- projection onto [0,1] with fixed sum ----------
def proj_cap_simplex(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) * 0.5
    return np.clip(v - mid, 0.0, 1.0)

def random_feasible_vector(n, rng):
    v = rng.random(n)
    return proj_cap_simplex(v, n / 2.0)

# ---------- evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

# ---------- gradient utilities ----------
def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s = -s
        grad[s:] += (1 - h[: n - s])
        grad[: n - s] -= h[s:]
    return grad

def gradient_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    m = np.max(bc)
    exp_shift = np.exp(bc - m)
    w = exp_shift / exp_shift.sum()
    g = np.zeros(n)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - (n - 1)
        g += weight * gradient_shift(h, s)
    return g * (2.0 / n)

def softmax_objective(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    m = np.max(bc)
    lse = (m + np.log(np.sum(np.exp(bc - m)))) / beta
    return lse * (2.0 / n)

# ---------- random perturbation ----------
def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- greedy improvement ----------
def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        s = idx - (h.size - 1)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

# ---------- projected gradient descent on smoothed max ----------
def pgd_optimize(h0, beta, max_iter=300, init_step=0.2):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_obj = softmax_objective(h, beta)
    step = init_step
    for _ in range(max_iter):
        grad = gradient_softmax(h, beta)
        h_new = proj_cap_simplex(h - step * grad, target)
        obj_new = softmax_objective(h_new, beta)
        if obj_new < best_obj - 1e-12:
            best_h = h_new.copy()
            best_obj = obj_new
            step *= 1.05
        else:
            step *= 0.5
        h = h_new
        if step < 1e-8:
            break
    return best_h

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initial vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 720
        h = random_feasible_vector(n, rng)

    h = proj_cap_simplex(np.clip(h, 0.0, 1.0), n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # ----- staged PGD + greedy refinement -----
    betas = [10, 100, 500, 2000, 5000]
    for beta in betas:
        if time.time() - start_time > budget_s * 0.80:
            break
        # PGD
        h_pg = pgd_optimize(best_h, beta, max_iter=300, init_step=0.3)
        c_pg = compute_c5(h_pg)
        if c_pg < best_c:
            best_h, best_c = h_pg, c_pg

        # Greedy polishing
        if time.time() - start_time > budget_s * 0.80:
            break
        h_g = greedy_optimize(best_h, max_steps=2000)
        c_g = compute_c5(h_g)
        if c_g < best_c:
            best_h, best_c = h_g, c_g

        # Small guided random moves
        for _ in range(5):
            if time.time() - start_time > budget_s * 0.85:
                break
            cand = perturb_random(best_h, rng, scale=0.1)
            cand = greedy_optimize(cand, max_steps=500)
            c_cand = compute_c5(cand)
            if c_cand < best_c:
                best_h, best_c = cand, c_cand

    # ----- final fine‑grained search -----
    while time.time() - start_time < budget_s * 0.98:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # final greedy pass if time permits
    if time.time() - start_time < budget_s * 0.99:
        best_h = greedy_optimize(best_h, max_steps=2000)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_cap_simplex(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
