import time
import numpy as np
from scipy.optimize import minimize

# ---------- core evaluation ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def autocorr_bin(h: np.ndarray) -> np.ndarray:
    """Circular autocorrelation of a binary (0/1) vector using FFT."""
    f = np.fft.rfft(h)
    corr = np.fft.irfft(np.abs(f) ** 2, n=h.size)
    # rounding eliminates tiny numerical noise
    return np.rint(corr).astype(int)


def compute_c5_binary(h: np.ndarray) -> float:
    """C5 for a binary pattern via the autocorrelation shortcut."""
    n = h.size
    min_corr = autocorr_bin(h)[1:].min()
    return 1.0 - (2.0 * min_corr) / n


# ---------- simulated annealing for binary patterns ----------
def anneal_binary(n: int, rng: np.random.Generator, time_limit: float):
    """Start from a random binary pattern and perform SA."""
    h = np.zeros(n, dtype=np.int8)
    h[rng.choice(n, n // 2, replace=False)] = 1
    corr = autocorr_bin(h)
    min_corr = corr[1:].min()
    best_h = h.copy()
    best_min_corr = min_corr

    start = time.time()
    T0, T_end = 0.02, 1e-6
    while time.time() - start < time_limit:
        # temperature schedule (exponential)
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * ((T_end / T0) ** frac)

        # propose a swap (1 -> 0 and 0 -> 1)
        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)

        # apply swap
        h[i] = 0
        h[j] = 1
        corr_cand = autocorr_bin(h)
        min_corr_cand = corr_cand[1:].min()
        delta = min_corr_cand - min_corr
        accept = delta > 0 or rng.random() < np.exp(delta / T)
        if accept:
            corr = corr_cand
            min_corr = min_corr_cand
            if min_corr > best_min_corr:
                best_min_corr = min_corr
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0

    return best_h, 1.0 - (2.0 * best_min_corr) / n


def anneal_binary_starting(n: int, rng: np.random.Generator, time_limit: float, h0: np.ndarray):
    """SA starting from a given binary pattern."""
    h = h0.copy()
    corr = autocorr_bin(h)
    min_corr = corr[1:].min()
    best_h = h.copy()
    best_min_corr = min_corr

    start = time.time()
    T0, T_end = 0.02, 1e-6
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * ((T_end / T0) ** frac)

        i = rng.integers(n)
        while h[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] == 1:
            j = rng.integers(n)

        h[i] = 0
        h[j] = 1
        corr_cand = autocorr_bin(h)
        min_corr_cand = corr_cand[1:].min()
        delta = min_corr_cand - min_corr
        accept = delta > 0 or rng.random() < np.exp(delta / T)
        if accept:

# ---------- SLSQP refinement (soft‑max) ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Single SLSQP pass on a (possibly fractional) step function."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(_softmax_obj, h, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
