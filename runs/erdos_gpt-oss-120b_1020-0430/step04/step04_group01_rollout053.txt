# --------------------------------------------------------------
# Basic feasibility tools
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

def proj_box_sum(y, target):
    """Euclidean projection of y onto {0≤z≤1, Σz=target}."""
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(y - mid, 0.0, 1.0)
        s = z.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

# --------------------------------------------------------------
def smooth_descent(h0, betas, iters_per_beta, rng, deadline):
    """Projected gradient descent on log‑sum‑exp smooth of max Corr."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    L = 1 << ((3 * n - 2).bit_length())          # size for conv with w (len 2n-1)
    for beta in betas:
        for it in range(iters_per_beta):
            if time.time() >= deadline:
                return h
            # correlation (full) and soft‑max weights
            corr = correlate_full(h)               # length 2n-1
            bc = beta * corr
            m = bc.max()
            exp_bc = np.exp(bc - m)
            w = exp_bc / exp_bc.sum()               # weights, sum to 1

            # convolution w * h (backward) and w_rev * h (forward)
            Fh = np.fft.rfft(h, L)
            Fw = np.fft.rfft(w, L)
            conv_back = np.fft.irfft(Fw * Fh, L)[:n]          # Σ_k w_k h_{i‑k}
            Fw_rev = np.fft.rfft(w[::-1], L)
            conv_fwd = np.fft.irfft(Fw_rev * Fh, L)[:n]      # Σ_k w_k h_{i+k}
            grad = 1.0 - conv_back - conv_fwd                 # sub‑gradient of smoothed max

            step = 0.18 / np.sqrt(it + 1)                    # decreasing step
            h = proj_box_sum(h - step * grad, target)
    return h

# SLSQP polishing (same as the previous solution)
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # 1. Base size & initial candidates
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        base = np.array(initial_h_values, dtype=float)
        base = np.clip(base, 0.0, 1.0)
        base = enforce_sum(base, n / 2.0, rng)
    else:
        n = 720
        base = random_feasible_vector(n, rng)

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [base]

    # triangular
    tri = np.clip(1.0 - np.abs(xs - 1.0), 0.0, 1.0)
    candidates.append(enforce_sum(tri, n / 2.0, rng))

    # sin / cos patterns
    for f in range(1, 4):
        s = np.clip(0.5 + 0.5 * np.sin(np.pi * f * xs), 0.0, 1.0)
        candidates.append(enforce_sum(s, n / 2.0, rng))
        c = np.clip(0.5 + 0.5 * np.cos(np.pi * f * xs), 0.0, 1.0)
        candidates.append(enforce_sum(c, n / 2.0, rng))

    # block and alternating
    block = np.zeros(n)
    block[: n // 2] = 1.0
    candidates.append(enforce_sum(block, n / 2.0, rng))
    alt = np.tile([0.0, 1.0], n // 2)
    candidates.append(enforce_sum(alt, n / 2.0, rng))

    # a few random feasible vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # ------------------------------------------------------------------
    # 2. Pick best among the pool
    # ------------------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ------------------------------------------------------------------
    # 3. Continuous optimisation via smooth continuation
    # ------------------------------------------------------------------
    betas = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]
    iters_per_beta = 300                     # moderate number per β

    # run many restarts until time runs out (leaving a safety margin)
    while time.time() + 12 < deadline:
        if rng.random() < 0.6:
            start = perturb_h(best_h, rng, scale=0.35)
        else:
            start = random_feasible_vector(n, rng)
        h_opt = smooth_descent(start, betas, iters_per_beta, rng, deadline)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ------------------------------------------------------------------
    # 4. Final polishing with high‑β SLSQP (few iterations)
    # ------------------------------------------------------------------
    try:
        h_opt = slsqp_optimize(best_h, beta=8000.0, max_iter=800)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 5. Ensure exact feasibility and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
