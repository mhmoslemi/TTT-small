def compute_c5_fft(h):
    n = h.size
    n2 = 2*n - 1
    # rfft length = n2//2 + 1
    fh = np.fft.rfft(h, n2)
    fg = np.fft.rfft(1 - h, n2)
    corr = np.fft.irfft(fh * fg, n2)
    return np.max(corr) * (2.0 / n)

def perturb_h(h, rng, scale=0.2):

    n = h.size

    i, j = rng.choice(n, size=2, replace=False)

    max_eps = min(h[i], 1.0 - h[j])

    # scaling factor and random sign?

    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale

    # Clip epsilon within [-max_eps, max_eps]
    eps = max(-max_eps, min(eps, max_eps))

    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def max_corr_and_grad_fft(h):
    n = h.size
    n2 = 2 * n - 1
    fh = np.fft.rfft(h, n2)
    fb = np.fft.rfft(1 - h, n2)
    corr = np.fft.irfft(fh * np.conj(fb), n2)
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    # gradient
    g = np.zeros_like(h)
    if k >= 0:
        # part from j>=k
        g[k:] += 1 - h[:n - k]
        # part from j <= n-1-k
        g[:n - k] -= h[k:]
    else:
        kk = -k
        # part from j <= n-1-kk
        g[:n - kk] += 1 - h[kk:]
        # part from j >= kk
        g[kk:] -= h[:n - kk]
    max_corr = corr[idx] * (2.0 / n)
    return max_corr, k, g

    best_h = best_h_global
    best_c = best_c_global
    n = h.size

        d -= np.mean(d)

import time
import numpy as np
from scipy.optimize import minimize

# ----- projection onto the box with exact sum -----
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


# ----- random feasible vector -----
def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


# ----- small perturbation preserving the sum -----
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ----- correlation via FFT (matches np.correlate) -----
def _corr_fft(h):
    n = h.size
    n2 = 2 * n - 1
    fh = np.fft.rfft(h, n2)
    fb = np.fft.rfft(1.0 - h, n2)
    corr = np.fft.irfft(fh * np.conj(fb), n2)
    return corr


def compute_c5_fft(h):
    n = h.size
    corr = _corr_fft(h)
    return np.max(corr) * (2.0 / n)


def max_corr_and_grad_fft(h):
    n = h.size
    corr = _corr_fft(h)
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    g = np.zeros_like(h)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        kk = -k
        g[:n - kk] += 1.0 - h[kk:]
        g[kk:] -= h[:n - kk]
    max_corr = corr[idx] * (2.0 / n)
    return max_corr, k, g


# ----- SLSQP refinement (softmax surrogate) -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ----- gradient‑descent loop on the maximal correlation -----
def descent_until(h_start, rng, time_limit, global_best_h, global_best_c):
    h = h_start.copy()
    best_h = global_best_h
    best_c = global_best_c
    n = h.size
    stagnation = 0
    while time.time() < time_limit:
        c, k, g = max_corr_and_grad_fft(h)
        if c < best_c - 1e-12:
            best_c, best_h = c, h.copy()
            stagnation = 0
        else:
            stagnation += 1

        d = -g
        d -= np.mean(d)                # keep zero‑sum direction

        pos = d > 1e-12
        neg = d < -1e-12
        if not (pos.any() or neg.any()):
            break

        alphas = []
        if pos.any():
            alphas.append(np.min((1.0 - h[pos]) / d[pos]))
        if neg.any():
            alphas.append(np.min((0.0 - h[neg]) / d[neg]))
        alpha_max = min(alphas) if alphas else 0.0
        if alpha_max <= 0.0:
            h = perturb_h(h, rng, scale=0.2)
            h = project_box_sum(h, n / 2.0)
            continue

        step = min(alpha_max * 0.5, 0.5)
        improved = False
        while step > 1e-10:
            nh = h + step * d
            nh = np.clip(nh, 0.0, 1.0)
            nh = project_box_sum(nh, n / 2.0)
            nc = compute_c5_fft(nh)
            if nc < c - 1e-13:
                h, c = nh, nc
                improved = True
                break
            step *= 0.5

        if not improved:
            h = perturb_h(h, rng, scale=0.1)
            h = project_box_sum(h, n / 2.0)

        # occasional SLSQP refinement on the current global best
        if stagnation % 200 == 0 and best_c < c:
            try:
                h_ref = slsqp_optimize(best_h, beta=2000, max_iter=300)
                c_ref = compute_c5_fft(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
                    h = h_ref.copy()
            except Exception:
                pass

    return best_h, best_c, h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    base_n = 720
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        if n0 != base_n:
            old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
            new_x = (np.arange(base_n) + 0.5) * (2.0 / base_n)
            h0 = np.interp(new_x, old_x, h0)
        else:
            h0 = h0.copy()
    else:
        h0 = random_feasible_vector(base_n, rng)

    h0 = project_box_sum(h0, base_n / 2.0)
    best_h = h0.copy()
    best_c = compute_c5_fft(best_h)

    # ---- descent at base resolution ----
    deadline1 = start_time + budget_s * 0.75
    best_h, best_c, cur_h = descent_until(
        h0, rng, deadline1, best_h, best_c
    )

    # ---- optional up‑sampling if time permits ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.2:
        up_n = min(1440, base_n * 2)
        old_x = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
        new_x = (np.arange(up_n) + 0.5) * (2.0 / up_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = project_box_sum(h_up, up_n / 2.0)

        deadline2 = start_time + budget_s * 0.98
        best_h_up, best_c_up, _ = descent_until(
            h_up, rng, deadline2, best_h, best_c
        )
        if best_c_up < best_c:
            best_h, best_c = best_h_up, best_c_up

    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
