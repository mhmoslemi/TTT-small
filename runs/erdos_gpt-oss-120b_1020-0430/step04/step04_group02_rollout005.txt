def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=50):
    # find offset to shift and clip
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low+high)/2
        y = np.clip(x+mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x+(low+high)/2, lo, hi)
    # small correction if sum slightly off due to precision
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y>lo+1e-12) & (y<hi-1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def anneal(h_start, rng, time_limit, p_target=0.3, init_temp=0.01, alpha=0.9999, scale=0.2):
    n = h_start.size
    cur_h = h_start.copy()
    cur_c, idx = compute_c5_and_idx(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    T = init_temp
    start = time.time()
    it = 0
    # For tracking k_star recompute maybe every few iters
    k_star = idx - (n - 1)
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        # decide move
        if rng.random() < p_target:
            cand_h = targeted_move(cur_h, k_star, rng, scale=scale)
        else:
            cand_h = pair_move(cur_h, rng, scale=scale)
        cand_c, cand_idx = compute_c5_and_idx(cand_h)
        delta = cand_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c, idx = cand_h, cand_c, cand_idx
            k_star = idx - (n - 1)
            # update best
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        # temperature cooling
        T *= alpha
        it += 1
        # occasionally adapt scale? we can reduce scale gradually
        if it % 10000 == 0:
            scale = max(scale * 0.9, 0.02)
    return best_h, best_c

def slsqp_refine(h0, beta_list, rng, time_limit):
    # returns best refined h and c
    n = h0.size
    target = n / 2.0
    best_h, best_c = h0.copy(), compute_c5(h0)
    start = time.time()
    for beta in beta_list:
        if time.time() - start >= time_limit:
            break
        # define objective
        def obj(x, beta=beta):
            corr = np.correlate(x, 1.0 - x, mode='full')
            bc = beta * corr
            m = np.max(bc)
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta
        # constraints
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
        bounds = [(0.0, 1.0)] * n
        # Use minimize
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
        except Exception:
            continue
        if not res.success:
            # still try
            pass
        x = np.clip(res.x, 0.0, 1.0)
        x = project_box_sum(x, target)
        c = compute_c5(x)
        if c < best_c:
            best_c, best_h = c, x.copy()
    return best_h, best_c

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) * 0.5, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx

# ---------- move operators ----------
def pair_move(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = min(h[j], 1.0 - h[i])
    if max_pos <= 0 and max_neg <= 0:
        return h.copy()
    delta = rng.uniform(-max_neg, max_pos) * scale
    delta = max(-max_neg, min(delta, max_pos))
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def targeted_move(h, k, rng, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return pair_move(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return pair_move(h, rng, scale)
    low_idx = rng.choice(np.flatnonzero(low_mask))
    high_idx = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[low_idx],
        0.5 - h[low_idx + k],
        h[high_idx] - 0.5,
        h[high_idx + k] - 0.5,
    )
    if max_eps <= 0:
        return pair_move(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[low_idx] += eps
    new_h[low_idx + k] += eps
    new_h[high_idx] -= eps
    new_h[high_idx + k] -= eps
    return new_h

# ---------- annealing ----------
def anneal(h_start, rng, time_limit, p_target=0.3,
           init_temp=0.01, alpha=0.9999, scale=0.2):
    n = h_start.size
    cur_h = h_start.copy()
    cur_c, idx = compute_c5_and_idx(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    T = init_temp
    start = time.time()
    it = 0
    k_star = idx - (n - 1)
    while True:
        if time.time() - start >= time_limit:
            break
        if rng.random() < p_target:
            cand_h = targeted_move(cur_h, k_star, rng, scale=scale)
        else:
            cand_h = pair_move(cur_h, rng, scale=scale)
        cand_c, cand_idx = compute_c5_and_idx(cand_h)
        delta = cand_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c, idx = cand_h, cand_c, cand_idx
            k_star = idx - (n - 1)
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        T *= alpha
        it += 1
        if it % 10000 == 0:
            scale = max(scale * 0.9, 0.02)
    return best_h, best_c

# ---------- SLSQP refinement ----------
def slsqp_refine(h0, beta_list, rng, time_limit):
    n = h0.size
    target = n / 2.0
    best_h, best_c = h0.copy(), compute_c5(h0)
    start = time.time()
    for beta in beta_list:
        if time.time() - start >= time_limit:
            break

        def obj(x, beta=beta):
            corr = np.correlate(x, 1.0 - x, mode="full")
            bc = beta * corr
            m = np.max(bc)
            return (m + np.log(np.sum(np.exp(bc - m)))) / beta

        cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                obj,
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 500, "ftol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        x = np.clip(res.x, 0.0, 1.0)
        x = project_box_sum(x, target)
        c = compute_c5(x)
        if c < best_c:
            best_c, best_h = c, x.copy()
    return best_h, best_c

# ---------- up‑sampling ----------
def upsample_h(h):
    n = h.size
    target_n = min(720, max(540, n * 2))
    if target_n == n:
        return h.copy()
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    new_h = np.interp(new_x, old_x, h)
    new_h = project_box_sum(new_h, target_n / 2.0)
    return new_h

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # initialise
    if "initial_h_values" in globals():

        n = h0.size

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ----- Phase 1: annealing with restarts -----
    remaining = budget_s - (time.time() - start_time)
    phase1_time = remaining * 0.6
    phase1_end = start_time + phase1_time
    while time.time() < phase1_end:
        chunk = min(30.0, phase1_end - time.time())
        if chunk <= 0.5:

        start_vec = best_h.copy() if rng.random() < 0.5 else random_feasible_vector(n, rng)
        cand_h, cand_c = anneal(start_vec, rng, chunk,
                                p_target=0.3, init_temp=0.01,
                                alpha=0.9999, scale=0.2)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()

    # ----- Phase 2: SLSQP refinement -----
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        beta_list = [20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]
        best_h, best_c = slsqp_refine(best_h, beta_list, rng, remaining * 0.3)

    # ----- Phase 3: up‑sampling and final polish -----
    remaining = budget_s - (time.time() - start_time)
    if remaining > budget_s * 0.07 and n < 720:
        h_up = upsample_h(best_h)
        best_h = h_up
        n = best_h.size
        # a short anneal on the finer grid

            cand_h, cand_c = anneal(best_h, rng, remaining * 0.4,

            if cand_c < best_c:

        # final SLSQP on fine grid
        remaining = budget_s - (time.time() - start_time)
        if remaining > 5.0:
            best_h, best_c = slsqp_refine(best_h, beta_list, rng, remaining)

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
