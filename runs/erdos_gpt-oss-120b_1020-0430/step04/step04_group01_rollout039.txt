def soft_obj_and_grad(h, beta):
    n = h.size
    total_sum = n/2.0   (since sum constraint)
    # compute cross_corr via FFT
    L = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    cross = np.fft.irfft(fh * np.conj(fh), L)[:2*n-1]  # linear autocorrelation
    w = np.exp(-beta * cross)
    sum_w = w.sum()
    # objective:
    obj = total_sum + (1.0/beta) * np.log(sum_w)
    # gradient via convolution:
    # compute conv of w with h
    # ensure length for convolution: len(w)=2n-1, len(h)=n, need L2 = 1 << ((len(w)+n-1).bit_length())
    L2 = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, L2)
    fh2 = np.fft.rfft(h, L2)
    conv1 = np.fft.irfft(fw * fh2, L2)[:len(w)+n-1]  # length 3n-2
    # extract middle n entries
    start = n - 1
    conv1_mid = conv1[start:start+n]
    # w_rev:
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, L2)
    conv2 = np.fft.irfft(fw_rev * fh2, L2)[:len(w)+n-1]
    conv2_mid = conv2[start:start+n]
    grad = -(conv1_mid + conv2_mid) / sum_w
    return obj, grad

def soft_obj_and_grad(h, beta):
    n = h.size
    # total sum target (should be n/2)
    total_sum = n / 2.0
    # compute autocorrelation
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    cross = np.fft.irfft(fh * np.conj(fh), L)[:2 * n - 1]
    # scaling for stability
    c_min = cross.min()
    shift = cross - c_min
    w = np.exp(-beta * shift)
    sum_w = w.sum()
    obj = total_sum - c_min + (1.0 / beta) * np.log(sum_w)
    # compute gradient via convolution w * h
    len_w = w.size
    L2 = 1 << ((len_w + n - 1).bit_length())
    fw = np.fft.rfft(w, L2)
    fh2 = np.fft.rfft(h, L2)
    conv_full = np.fft.irfft(fw * fh2, L2)[:len_w + n - 1]
    start = n - 1
    conv1_mid = conv_full[start:start + n]
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, L2)
    conv_full_rev = np.fft.irfft(fw_rev * fh2, L2)[:len_w + n - 1]
    conv2_mid = conv_full_rev[start:start + n]
    grad = -(conv1_mid + conv2_mid) / sum_w
    return obj, grad

CURRENT_BETA = None
CURRENT_TARGET = None

# ----------------------------------------------------------------------
# Projection onto the capped simplex (sum = target, 0 <= x <= 1)
# ----------------------------------------------------------------------
def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / x.size
            x = np.clip(x, 0.0, 1.0)
    return x


def enforce_sum(h: np.ndarray, target: float) -> np.ndarray:
    return _project_capped_simplex(h, target)

def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    h = rng.random(n)
    return enforce_sum(h, n / 2.0)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    # move mass from i to j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ----------------------------------------------------------------------
# Correlation utilities (FFT based)
# ----------------------------------------------------------------------
def correlate_full(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h: np.ndarray) -> float:
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _soft_obj_and_grad(h: np.ndarray, beta: float):
    n = h.size
    total_sum = n / 2.0

    # ----- autocorrelation of h -----
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    cross = np.fft.irfft(fh * np.conj(fh), L)[: 2 * n - 1]

    # ----- stabilize exponentials -----
    c_min = cross.min()
    shift = cross - c_min
    w = np.exp(-beta * shift)
    sum_w = w.sum()

    # ----- smoothed objective -----
    obj = total_sum - c_min + (1.0 / beta) * np.log(sum_w)

    # ----- gradient via convolution w * h -----
    m = w.size
    L2 = 1 << ((m + n - 1).bit_length())
    fw = np.fft.rfft(w, L2)
    fh2 = np.fft.rfft(h, L2)

    conv_full = np.fft.irfft(fw * fh2, L2)[: m + n - 1]
    start = n - 1
    part1 = conv_full[start:start + n]

    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, L2)
    conv_rev = np.fft.irfft(fw_rev * fh2, L2)[: m + n - 1]
    part2 = conv_rev[start:start + n]

    grad = -(part1 + part2) / sum_w
    return obj, grad


def _soft_obj_fun(x: np.ndarray) -> float:
    obj, _ = _soft_obj_and_grad(x, CURRENT_BETA)
    return obj


def _soft_obj_jac(x: np.ndarray) -> np.ndarray:
    _, grad = _soft_obj_and_grad(x, CURRENT_BETA)
    return grad


def _sum_eq_constraint(x: np.ndarray) -> float:

def _sum_eq_jac(x: np.ndarray) -> np.ndarray:
    return np.ones_like(x)

    n = h0.size
    CURRENT_BETA = beta

              deadline: float):

    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 50000

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        # swap a 1 (at i) with a 0 (at j)
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c
