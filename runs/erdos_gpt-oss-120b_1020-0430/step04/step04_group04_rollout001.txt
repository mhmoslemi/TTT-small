def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)
```

def enforce_sum(h, target, rng=None, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    if rng is None:
        rng = np.random.default_rng()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Choose random index where we can adjust
        if diff > 0:
            # need to increase sum: pick index where h < 1
            idxs = np.where(h < 1 - 1e-12)[0]
        else:
            idxs = np.where(h > 1e-12)[0]
        if idxs.size == 0:
            break
        i = rng.choice(idxs)
        max_delta = (1 - h[i]) if diff > 0 else -h[i]
        delta = np.clip(diff, -abs(max_delta), abs(max_delta))
        h[i] += delta
        diff = target - h.sum()
    # final small adjust using any free indices
    if abs(diff) > 1e-12:
        # distribute diff equally among all indices with slack
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size:
            per = diff / free.size
            h[free] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h
```

def hill_climb(start_h, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = start_h.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            # occasional larger step
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

def simulated_annealing(h0, rng, max_iter=5000,
                       init_temp=1e-2, final_temp=1e-6,
                       init_scale=0.5, final_scale=0.02):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * ((final_temp / init_temp) ** t)
        scale = init_scale * ((final_scale / init_scale) ** t)
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c
```

                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)

    diff = CURRENT_TARGET - x.sum()

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (sum constraint, surrogate, etc.)
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint: sum(x) == CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k correlation using log‑sum‑exp."""
    n = x.shape[0]
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = CURRENT_BETA * conv
    m_bc = np.max(bc)
    # log‑sum‑exp
    return (m_bc + np.log(np.sum(np.exp(bc - m_bc)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng=None, max_iter=2000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    if rng is None:
        rng = np.random.default_rng()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idxs = np.where(h < 1 - 1e-12)[0]
        else:
            idxs = np.where(h > 1e-12)[0]
        if idxs.size == 0:
            break
        i = rng.choice(idxs)
        max_delta = (1 - h[i]) if diff > 0 else -h[i]
        delta = np.clip(diff, -abs(max_delta), abs(max_delta))
        h[i] += delta
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size:
            per = diff / free.size
            h[free] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(start_h, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple stochastic hill‑climb."""
    h = start_h.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


def simulated_annealing(h0, rng, max_iter=5000,
                       init_temp=1e-2, final_temp=1e-6,
                       init_scale=0.5, final_scale=0.02):
    """Simulated‑annealing refinement."""
    best_h, best_c = h0.copy(), compute_c5(h0)
    cur_h, cur_c = best_h.copy(), best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * ((final_temp / init_temp) ** t)
        scale = init_scale * ((final_scale / init_scale) ** t)
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP refinement of the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-9:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
