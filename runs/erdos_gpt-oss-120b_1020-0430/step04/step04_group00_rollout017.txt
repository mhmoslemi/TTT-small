def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        max_change = 1.0 - h[i] if diff > 0 else -h[i]
        change = min(diff, max_change) if diff > 0 else max(diff, max_change)
        h[i] += change
        diff = target - h.sum()
    return h
```

def gradient_refine(h, rng, max_iter=15, step_factor=0.02):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step = step_factor * (0.5 ** it)
        h = h - step * g
        # clip 
        h = np.clip(h, 0.0, 1.0)
        # fix sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = adjust_sum_exact(h, rng)
    return h
```

def sum_eq_constraint(x):
    return np.sum(x) - SL_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = SL_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp in stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / SL_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global SL_TARGET, SL_BETA
    n = h0.size
    SL_TARGET = n / 2.0
    SL_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-5)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < math.exp(-(new_c - cur_c) / T):
                accept = True
        if accept:
            cur_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c

def continuous_sa(h, rng, time_deadline):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = h.size
    target = n / 2.0
    # parameters
    init_T = max(0.05 * cur_c, 1e-5)
    init_step = 0.05  # max delta change
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        # ensure i != j
        delta = (rng.random() - 0.5) * 2 * init_step * (0.999 ** it)
        # delta to add to i, subtract from j
        # clip to [0,1] and preserve sum
        hi = cur_h[i]
        hj = cur_h[j]
        # compute feasible delta range
        lo = max(-hi, -delta)  # cannot reduce below 0, also not exceed delta negative bound
        hi_delta = min(1.0 - hi, -delta) if delta < 0 else min(1.0 - hi, delta)
        # Actually simpler: compute new hi = hi + delta, new hj = hj - delta;

import time, math
import numpy as np
from scipy.optimize import minimize

# ----- globals for SLSQP -----
SL_TARGET = None
SL_BETA = None


def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Make sum(h) == len(h)/2 while keeping 0<=h<=1."""
    target = h.size / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        if diff > 0:
            if h[i] >= 1.0:
                continue
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
        else:
            if h[i] <= 0.0:
                continue
            dec = min(-diff, h[i])
            h[i] -= dec
        diff = target - h.sum()
    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    max_iter: int = 12, step_factor: float = 0.02) -> np.ndarray:
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step = step_factor * (0.5 ** it)
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h = adjust_sum_exact(h, rng)
    return h


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - SL_TARGET


def softmax_obj(x: np.ndarray) -> float:
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = SL_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / SL_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 600) -> np.ndarray:
    global SL_TARGET, SL_BETA
    n = h0.size
    SL_TARGET = n / 2.0
    SL_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = SL_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = adjust_sum_exact(x, rng)
    return x


def binary_sa_hill_climb(h_bin: np.ndarray, rng: np.random.Generator,
                         time_deadline: float):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-5)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c


def continuous_sa(h: np.ndarray, rng: np.random.Generator,
                  time_deadline: float):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = h.size
    init_T = max(0.05 * cur_c, 1e-5)
    init_step = 0.05
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = cur_h[i], cur_h[j]
        delta = (rng.random() - 0.5) * 2.0 * init_step * (0.999 ** it)
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        if delta_min > delta_max:
            it += 1
            continue
        delta = np.clip(delta, delta_min, delta_max)
        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        new_c = compute_c5(new_h)
        T = init_T * (0.9995 ** it) + 1e-12
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
                last_improve = it
        it += 1
        if it - last_improve > 20000:
            break
    return best_h, best_c

    best_h = adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
