import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Global helpers for the smooth surrogate (used by SLSQP)
# --------------------------------------------------------------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) using log‑sum‑exp."""
    beta = CURRENT_BETA
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)          # stabilized
    obj = (max_bc + np.log(np.sum(exp_bc))) / beta
    return obj


def softmax_obj_grad(x):
    """Analytic gradient of softmax_obj w.r.t. x."""
    beta = CURRENT_BETA
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    Z = np.sum(exp_bc)
    w = exp_bc / Z                       # weights for each shift
    # term1 = Σ_k w_k * (1 - x_{p+k})
    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, (1.0 - x), mode="full")
    start = len(w) - 1
    term1 = conv1[start:start + n]
    # term2 = Σ_k w_k * x_{p-k}
    conv2 = np.convolve(w, x, mode="full")
    term2 = conv2[start:start + n]
    grad = term1 - term2
    return grad


def project_sum_box(h, target, max_iter=10):
    """Project h onto the intersection of [0,1]^n and Σh = target."""
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        delta = (target - s) / h.size
        h = h + delta
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h0, beta, max_iter=400):
    """Refine h using SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint,
            "jac": lambda x: np.ones_like(x)}
    res = minimize(lambda x: softmax_obj(x),
                   h0,
                   jac=lambda x: softmax_obj_grad(x),
                   method="SLSQP",
                   bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = project_sum_box(x, CURRENT_TARGET, max_iter=10)
    return x


# --------------------------------------------------------------
# Discrete evaluation utilities
# --------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def max_corr_and_pair(h):
    """Return (max_corr, shift, i, j) where i->j gives the worst product."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (n - 1)
    # indices contributing to this shift
    if shift >= 0:
        i_range = np.arange(n - shift)
        j_range = i_range + shift
    else:
        i_range = np.arange(-shift, n)
        j_range = i_range + shift
    prod = h[i_range] * (1.0 - h[j_range])
    max_idx = int(np.argmax(prod))
    i = int(i_range[max_idx])
    j = int(j_range[max_idx])
    return max_val, shift, i, j


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.8, decay=0.999):
    """Specialised local search moving mass from i to j to lower max overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift, i, j = max_corr_and_pair(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c


def random_feasible_vector(n, rng):
    """Generate a random feasible h with Σh = n/2 and 0≤h≤1."""
    h = rng.random(n)
    h = project_sum_box(h, n / 2.0, max_iter=10)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def generate_candidates(n, rng):
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    target = n / 2.0
    cand = []

    # use parent construction if available (resampled)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # resample to current grid
        n0 = h0.size
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_interp = np.interp(xs, xs0, h0, left=h0[0], right=h0[-1])
        h_interp = np.clip(h_interp, 0.0, 1.0)
        h_interp = project_sum_box(h_interp, target, max_iter=10)
        cand.append(h_interp)

    # sinusoidal families
    for f in (1, 2, 3, 4, 5):
        for fn in (np.sin, np.cos):
            h = 0.5 + 0.5 * fn(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            h = project_sum_box(h, target, max_iter=10)
            cand.append(h)

    # linear ramps (ascending / descending)
    ramp = xs / 2.0
    cand.append(project_sum_box(ramp, target, max_iter=10))
    cand.append(project_sum_box(1.0 - ramp, target, max_iter=10))

    # binary masks with few blocks
    mask = np.zeros(n)
    mask[: n // 3] = 1.0
    mask[n // 2 : n // 2 + n // 3] = 1.0
    cand.append(project_sum_box(mask, target, max_iter=10))

    mask = np.zeros(n)
    start = n // 4
    mask[start:start + n // 2] = 1.0
    cand.append(project_sum_box(mask, target, max_iter=10))

    # alternating mask (period 2) – not good but included
    mask = np.arange(n) % 2
    cand.append(project_sum_box(mask, target, max_iter=10))

    # random vectors
    for _ in range(5):
        cand.append(random_feasible_vector(n, rng))

    return cand

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 360                     # keep size moderate for speed
    target = n / 2.0

    # ---- 1. build initial pool -------------------------------------------------
    candidates = generate_candidates(n, rng)

    # ---- 2. pick best from pool ------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    global_best_h, global_best_c = best_h, best_c

    # ---- 3. main optimisation (SLSQP ladder + pair transfer) ------------------
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    main_limit = budget_s * 0.70

    while time.time() - start_time < main_limit:
        # start point selection
        if rng.random() < 0.7:
            start_vec = perturb_h(global_best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        # a short SLSQP ladder on this start point
        h_cur = start_vec.copy()
        for beta in beta_list:
            if time.time() - start_time >= main_limit:
                break
            try:
                h_cur = slsqp_optimize(h_cur, beta, max_iter=250)
            except Exception:
                continue

        c_cur = compute_c5(h_cur)
        if c_cur < global_best_c:
            global_best_h, global_best_c = h_cur.copy(), c_cur

        # dedicated pair‑transfer phase
        remaining = main_limit - (time.time() - start_time)
        if remaining > 10.0:
            h_opt, c_opt = pair_transfer_optimize(
                h_cur, rng,
                max_iters=4000,
                time_limit=remaining,
                init_factor=0.8,
                decay=0.999
            )
            if c_opt < global_best_c:
                global_best_h, global_best_c = h_opt.copy(), c_opt

    # ---- 4. final intensive pair‑transfer --------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 15.0:
        # multiple attempts proportionally to remaining time
        attempts = max(1, int(remaining // 30))
        per_attempt = remaining / (attempts + 1)
        for _ in range(attempts):
            h_start = perturb_h(global_best_h, rng, scale=0.2) if rng.random() < 0.5 else global_best_h.copy()
            h_opt, c_opt = pair_transfer_optimize(
                h_start, rng,
                max_iters=8000,
                time_limit=per_attempt,
                init_factor=0.9,
                decay=0.999
            )
            if c_opt < global_best_c:
                global_best_h, global_best_c = h_opt.copy(), c_opt
        # final fine‑tune
        h_opt, c_opt = pair_transfer_optimize(
            global_best_h, rng,

            time_limit=remaining * 0.9,
            init_factor=0.9,
            decay=0.999

    # ---- 5. final clean‑up ------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = project_sum_box(global_best_h, target, max_iter=10)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
