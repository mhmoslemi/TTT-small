def hill_climb_h(h, rng, steps=5000, step_scale=0.5):
    best = h.copy()
    best_c = compute_c5(best)
    n = len(h)
    target_sum = n/2.0
    for s in range(steps):
        i,j = rng.integers(0,n,size=2)
        if i==j: continue
        max_eps = min(best[i], 1-best[j])
        if max_eps <= 0: continue
        eps = (rng.random()-0.5) * 2 * max_eps * step_scale
        new = best.copy()
        new[i] -= eps
        new[j] += eps
        c = compute_c5(new)
        if c < best_c:
            best = new
            best_c = c
    # ensure sum exact
    best = enforce_sum(best, target_sum, rng)
    return best, best_c

from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders – will be filled inside `run`
P_GRID = None       # distances |x-1| on the current grid
P_NODES = None      # parameter nodes on [0,1]
W = None            # integration weights for the parameters
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())          # next power of two ≥ L
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return float(np.max(conv) * (2.0 / n))


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator,
                max_iter: int = 5000) -> np.ndarray:
    """Adjust h so that Σh = target while staying inside [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:                      # tiny residual – spread uniformly
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):

                    add = min(diff, 1.0 - h[i])

    return h


def softmax_obj(y: np.ndarray) -> float:
    """Smooth surrogate for max‑correlation using log‑sum‑exp."""
    h = np.interp(P_GRID, P_NODES, y)
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = CURRENT_BETA * conv
    bc_max = np.max(bc)
    return float((bc_max + np.log(np.sum(np.exp(bc - bc_max)))) / CURRENT_BETA)


def integral_eq_constraint(y: np.ndarray) -> float:
    """Weighted integral of the parametrised shape must equal ½."""
    return float(np.dot(W, y) - 0.5)


def monotonic_ineq_constraint(y: np.ndarray) -> np.ndarray:
    """Enforce y₀ ≥ y₁ ≥ … ≥ y_{m‑1}."""
    return y[:-1] - y[1:]


def hill_climb_h(h: np.ndarray, rng: np.random.Generator,
                 steps: int = 2000, step_scale: float = 0.5):
    """Simple stochastic hill‑climb on the full grid."""
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for _ in range(steps):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step_scale
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_h, best_c = new_h, c
    best_h = enforce_sum(best_h, target, rng)
    return best_h, best_c


def simulated_annealing(h0: np.ndarray, rng: np.random.Generator,
                        max_iter: int, init_temp: float, final_temp: float,
                        init_scale: float = 0.5, final_scale: float = 0.05):
    """Temperature‑controlled random walk."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    n = h0.shape[0]
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        i1 = rng.integers(0, n)
        i2 = rng.integers(0, n)
        if i1 == i2:
            continue
        max_eps = min(cur_h[i1], 1.0 - cur_h[i2])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
        new_h = cur_h.copy()
        new_h[i1] -= eps
        new_h[i2] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c
