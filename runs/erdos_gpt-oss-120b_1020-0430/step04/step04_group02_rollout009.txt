def gradient_move(h, rng, k, scale=0.2):
    n = h.size
    # compute forward shift: h_fwd[i] = h[i+k] if i+k < n else 0
    # backward shift: h_bwd[i] = h[i-k] if i-k >=0 else 0
    fwd = np.zeros_like(h)
    bwd = np.zeros_like(h)
    if k >= 0:
        fwd[:n-k] = h[k:]
        bwd[k:] = h[:n-k]
    else:
        # negative k handled similarly
        kneg = -k
        fwd[kneg:] = h[:n-kneg]
        bwd[:n-kneg] = h[kneg:]
    # gradient of C(k) wrt h = (1 - fwd) - bwd
    grad = (1.0 - fwd) - bwd
    # Want to reduce C(k), so move mass from indices where grad>0 (increase C) to where grad<0.
    pos_idx = np.where(grad > 1e-12)[0]
    neg_idx = np.where(grad < -1e-12)[0]
    if pos_idx.size == 0 or neg_idx.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(pos_idx)
    j = rng.choice(neg_idx)
    # epsilon limited by bounds and step
    max_eps = min(h[i], 1.0 - h[j])
    # Also limit by magnitude of gradient perhaps
    eps = scale * max_eps * (rng.random() - 0.5) * 2
    if eps > max_eps: eps = max_eps
    if eps < -max_eps: eps = -max_eps
    # Actually we want to move from i (positive grad) to j (negative grad), so epsilon should be positive up to max_eps.
    # Ensure direction: subtract from i, add to j.
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_targeted(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5, p_target=0.2, p_grad=0.3):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 20)
    n = h.shape[0]
    for it in range(max_iters):
        c_curr, k_star = compute_c5_and_idx(h)
        improved = False
        for _ in range(n_trials):
            r = rng.random()
            if r < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            elif r < p_target + p_grad:
                cand = gradient_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c_new = compute_c5(cand)
            if c_new < best_c - 1e-12:
                best_c, h = c_new, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c_new = compute_c5(cand)
            if c_new < best_c:
                best_c, h = c_new, cand
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.8, 1e-5)
    return h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)

# ---------- basic moves ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def gradient_move(h, rng, k, scale=0.2):
    n = h.size
    fwd = np.zeros(n)
    bwd = np.zeros(n)
    if k >= 0:
        if k < n:
            fwd[: n - k] = h[k:]
            bwd[k:] = h[: n - k]
    else:
        kn = -k
        if kn < n:
            fwd[kn:] = h[: n - kn]
            bwd[: n - kn] = h[kn:]
    grad = (1.0 - fwd) - bwd
    pos = np.where(grad > 1e-12)[0]
    neg = np.where(grad < -1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return perturb_h(h, rng, scale)
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ---------- evaluation ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def compute_c5_and_idx(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / n), idx - (n - 1)

# ---------- SLSQP (softmax) ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- hill climbing ----------
def hill_climb_targeted(
    h_start,
    rng,
    max_iters=2000,
    init_step=0.5,
    n_trials=5,
    p_target=0.25,
    p_grad=0.25,
):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 20)
    for it in range(max_iters):
        c_curr, k_star = compute_c5_and_idx(h)
        improved = False
        for _ in range(n_trials):
            r = rng.random()
            if r < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            elif r < p_target + p_grad:
                cand = gradient_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c_new = compute_c5(cand)
            if c_new < best_c - 1e-12:
                best_c, h = c_new, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c_new = compute_c5(cand)
            if c_new < best_c:
                best_c, h = c_new, cand
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.8, 1e-5)
    return h, best_c


# ---------- simulated annealing ----------
def anneal(h_start, rng, max_iter, init_temp=1e-4, step_scale=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    c_curr = best_c
    T = init_temp
    for it in range(max_iter):
        T *= 0.9995
        c_curr, k_star = compute_c5_and_idx(h)
        r = rng.random()
        if r < 0.33:
            cand = targeted_move(h, rng, k_star, scale=step_scale)
        elif r < 0.66:
            cand = gradient_move(h, rng, k_star, scale=step_scale)
        else:
            cand = perturb_h(h, rng, scale=step_scale)

        delta = c_new - c_curr
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = cand
            c_curr = c_new
            if c_new < best_c:
                best_c, best_h = c_new, cand.copy()
        if (it + 1) % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-5)
    return best_h, best_c

                h_ann, c_ann = anneal(best_h, rng, max_iter=max_ann, init_temp=1e-4, step_scale=0.2)
                if c_ann < best_c:
                    best_c, best_h = c_ann, h_ann.copy()
            # SLSQP refinement with growing β
            for beta in (10, 50, 200, 800, 2000):
                if time.time() - start_time > budget_s * 0.60:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=300)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # ---- up‑sampling stage (remaining budget) ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.15:
        target_n = n * 2
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = project_box_sum(h_up, target_n / 2.0)

        max_up_iter = int(min(5000, remaining * 8))
        h_up, c_up = hill_climb_targeted(
            h_up,
            rng,
            max_iters=max_up_iter,
            init_step=0.3,
            n_trials=6,
            p_target=0.25,
            p_grad=0.25,
        )
        best_c, best_h, n = c_up, h_up.copy(), target_n

        for beta in (20, 200, 2000, 10000):
            if time.time() - start_time > budget_s:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # ---- final tidy up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
