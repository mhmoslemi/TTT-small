def binary_sa_fast(start_bin, rng, deadline):
    n = start_bin.size
    half = n // 2
    h = start_bin.astype(np.int8)
    ones_arr = np.flatnonzero(h).astype(np.int32)
    zeros_arr = np.flatnonzero(1 - h).astype(np.int32)
    # Initialize O
    O = np.empty(n, dtype=np.int32)
    O[0] = half
    for d in range(1, n):
        O[d] = (h[:n-d] * h[d:]).sum()
    # Best
    O_min = O.min()
    best_c = 1.0 - O_min / half
    best_h = h.copy()
    best_ones_arr = ones_arr.copy()
    start_time = time.time()
    # temperature schedule parameters
    init_temp = max(best_c - 0.33, 0.001)
    target_overlap = half * (1 - 0.38055)  # compute target overlap >=223
    target_overlap = int(np.ceil(target_overlap))

    while time.time() < deadline:
        # propose swap
        i_idx = rng.integers(0, half)
        j_idx = rng.integers(0, half)
        i = ones_arr[i_idx]
        j = zeros_arr[j_idx]

import time, numpy as np

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# --------------------------------------------------------------
# Correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

def binary_sa_fast(start_bin, rng, deadline):
    """Swap‑based SA that works directly on the O‑array (pair counts)."""
    n = start_bin.size
    half = n // 2
    h = start_bin.astype(np.int8)
    ones = np.flatnonzero(h).astype(np.int32)          # length = half
    zeros = np.flatnonzero(1 - h).astype(np.int32)     # length = half

    # initial pair‑count array O[d] = #pairs of ones at distance d
    O = auto_corr_bruteforce(h)

    O_min = O.min()
    best_c = 1.0 - O_min / half
    best_h = h.astype(float)

    start_time = time.time()
    init_temp = max(best_c - 0.33, 0.001)               # ≈0.05 at start
    target_overlap = int(np.ceil(half * (1.0 - 0.38055)))  # 223 for n=720

    while time.time() < deadline:
        # ----- propose a single 1↔0 swap -----
        i_idx = rng.integers(0, half)
        j_idx = rng.integers(0, half)
        i = ones[i_idx]
        j = zeros[j_idx]

        # remove contributions of i
        dist_i = np.abs(ones - i)          # includes 0
        O[dist_i] -= 1

        # add contributions of j (i has already been removed)
        ones_wo_i = np.delete(ones, i_idx)
        if ones_wo_i.size:

            O[dist_j] += 1

        # perform the swap in the index lists
        ones[i_idx] = j
        zeros[j_idx] = i

        # evaluate new objective
        O_min = O.min()

        elapsed = time.time() - start_time

        if accept:
            if c < best_c:
                best_c = c
                best_h = np.zeros(n, dtype=float)
                best_h[ones] = 1.0
                if O_min >= target_overlap:
                    break
        else:
            # revert O
            O[dist_i] += 1
            if ones_wo_i.size:
                O[dist_j] -= 1
            # revert index lists
            ones[i_idx] = i
            zeros[j_idx] = j

    return best_h, best_c


# --------------------------------------------------------------
# Simple greedy hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb_continuous(h_start, rng, max_iters, deadline):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        new_h = perturb_h(h, rng, scale=0.2)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
    return best_h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ---- construct initial binary candidate ----
    if "initial_h_values" in globals():
        init_vals = np.array(initial_h_values, dtype=float)
        n = init_vals.size
        half = n // 2
        idx = np.argpartition(-init_vals, half)[:half]
        start_bin = np.zeros(n, dtype=float)
        start_bin[idx] = 1.0
    else:
        n = 720
        half = n // 2
        start_bin = np.concatenate([np.ones(half), np.zeros(half)], dtype=float)

    # ---- binary simulated annealing (core) ----
    best_h, best_c = binary_sa_fast(start_bin, rng, deadline)

    # ---- leftover time: continuous polishing ----
    if time.time() < deadline:
        h_pol, c_pol = hill_climb_continuous(best_h, rng, max_iters=3000, deadline=deadline)
        if c_pol < best_c:
            best_h, best_c = h_pol, c_pol

    # final safety enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
