def compute_gradient_for_shift(h, k):

    if k >= 0:
        # forward term

        # backward term
        backward = np.empty(n, dtype=float)

import time, numpy as np
from scipy.optimize import minimize

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n/2.0)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h.copy()
    # move epsilon from i to j
    eps = (rng.random() - 0.5) * 2 * scale
    # bound epsilon by feasible box:
    max_down = min(eps, h[i]) if eps > 0 else max(eps, - (1 - h[i]))
    # Actually re-evaluate: we want to subtract from i and add to j.
    # So if eps>0 we subtract from i, add to j; else opposite.

    if eps > 0:

        eps = max_eps

    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

def gradient_shift(h, k):
    n = h.size
    if k >= 0:
        forward = np.empty(n, dtype=float)
        forward[:n - k] = 1.0 - h[k:]
        forward[n - k:] = 1.0
        backward = np.empty(n, dtype=float)
        backward[:k] = 0.0
        backward[k:] = h[:n - k]
    else:
        kk = -k
        forward = np.empty(n, dtype=float)
        forward[kk:] = 1.0 - h[:n - kk]
        forward[:kk] = 1.0
        backward = np.empty(n, dtype=float)
        backward[:n - kk] = h[kk:]
        backward[n - kk:] = 0.0
    return forward - backward

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


# ---------- random feasible ----------
def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)


# ---------- simple perturb ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- correlation via FFT ----------
def _full_corr_fft(h):
    n = h.size
    size = 2 * n
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    G = np.fft.rfft(np.concatenate([1.0 - h, np.zeros(n)]))
    conv = np.fft.irfft(H * G, n=size)
    return conv[: 2 * n - 1]


def compute_c5(h):
    return np.max(_full_corr_fft(h)) * (2.0 / h.size)


def compute_c5_and_idx(h):
    corr = _full_corr_fft(h)
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


# ---------- SLSQP helpers ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- sub‑gradient for exact max ----------
def gradient_shift(h, k):
    n = h.size
    if k >= 0:
        forward = np.empty(n, dtype=float)
        forward[: n - k] = 1.0 - h[k:]
        forward[n - k :] = 1.0
        backward = np.empty(n, dtype=float)
        backward[:k] = 0.0
        backward[k:] = h[: n - k]
    else:
        kk = -k
        forward = np.empty(n, dtype=float)
        forward[kk:] = 1.0 - h[: n - kk]
        forward[:kk] = 1.0
        backward = np.empty(n, dtype=float)
        backward[: n - kk] = h[kk:]
        backward[n - kk :] = 0.0
    return forward - backward


def subgradient_descent(h, deadline, rng):
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    while time.time() < deadline and it < 20000:
        c5, idx = compute_c5_and_idx(h)
        if c5 < best_c:
            best_c, best_h = c5, h.copy()
        k = idx - (n - 1)
        g = gradient_shift(h, k)
        alpha = 0.05 / np.sqrt(it + 1)
        h = project_box_sum(h - alpha * g, target)
        # occasional shake
        if it % 500 == 0 and rng.random() < 0.3:
            h = perturb_h(h, rng, scale=0.1)
        it += 1
    return best_h, best_c


# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ----- initial vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n0 / 2.0)
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    # ----- upscale to finer grid -----
    n_target = 720
    if n_target != n0:
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(n_target) + 0.5) * (2.0 / n_target)
        h_up = np.interp(new_x, old_x, h0)
        h0 = project_box_sum(h_up, n_target / 2.0)
        n = n_target
    else:
        n = n0

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- a few random starts -----
    for _ in range(5):
        if time.time() > deadline - 30:
            break
        h = random_feasible_vector(n, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- SLSQP refinement with increasing smoothness -----
    betas = [10, 50, 200, 800, 3000, 10000]
    h_cur = best_h.copy()
    for beta in betas:
        if time.time() > deadline - 20:
            break
        try:
            h_opt = slsqp_optimize(h_cur, beta, max_iter=400)
        except Exception:
            continue
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        h_cur = h_opt

    # ----- sub‑gradient phase on remaining time -----
    if time.time() < deadline - 1.0:
        h_sg, c_sg = subgradient_descent(best_h, deadline - 0.5, rng)
        if c_sg < best_c:
            best_c, best_h = c_sg, h_sg.copy()

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
