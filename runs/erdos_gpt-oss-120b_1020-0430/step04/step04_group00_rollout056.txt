def gradient_refine(h, rng, max_iter=20, init_step=0.05):

    n = h.size; target = n/2.0

    step = init_step

    for it in range(max_iter):

        corr = compute_corr(h)

        idx = np.argmax(corr) - (n - 1)

        k = idx

        # compute gradient g
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            if k > 0:
                upper = n - k
                g[:upper] += 1.0 - h[k:]
                g[k:] -= h[:upper]
            else:
                k2 = -k
                upper = n - k2
                # shift left
                g[:upper] -= h[k2:]
                g[k2:] += 1.0 - h[:upper]
        # scaled step decaying
        step = init_step * (0.5 ** it)
        # update
        h = h - step * g
        # clip and adjust sum
        h = np.clip(h, 0.0, 1.0)
        # adjust sum precisely
        diff = target - h.sum()
        # adjust using random picks
        if abs(diff) > 1e-12:
            # get capacities
            # loop up to say 2000 iterations or until diff small
            for _ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                if add == 0.0:
                    continue
                h[i] += add
                diff = target - h.sum()
        # optionally early break if step already tiny?
    return h

def fractional_sa(h, rng, time_deadline, init_T=0.001, decay=0.9995):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T = init_T
    it = 0
    while time.time() < time_deadline:
        # pick i, j distinct
        i, j = rng.integers(0, len(cur_h), size=2)
        if i == j:
            continue
        # propose epsilon uniformly in [-eps_range, eps_range]
        # choose eps_range maybe 0.05 or shrink with iteration
        eps_range = 0.02 * (0.99 ** it)  # decaying
        eps = rng.uniform(-eps_range, eps_range)
        # ensure feasibility of move:
        hi = cur_h[i] - eps
        hj = cur_h[j] + eps
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        new_h = cur_h.copy()
        new_h[i] = hi
        new_h[j] = hj
        # sum unchanged
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        it += 1
        T *= decay
        # early exit if not improving and T small? Not needed
    return best_h, best_c

    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # Determine candidates
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        n0 = parent.size
        n_candidates = [n0]
        if n0 * 2 <= 720:
            n_candidates.append(n0*2)
        if n0 * 3 // 2 <= 720 and (n0*3)%2==0:
            n_candidates.append(n0*3//2)
    else:
        n_candidates = [240, 360]

    # Main loop over n
    for n in n_candidates:
        if time.time() > deadline:
            break
        # Build start vectors list
        start_vecs = []
        # 1. interpolated parent if available
        if 'parent' in locals():
            xs_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs_parent, parent)
            start_vecs.append(interp.copy())
            # binary from top half
            idx = np.argpartition(interp, -n//2)[-n//2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)
            # parent perturbed

            adjust_sum_exact(pert, rng)
            start_vecs.append(pert)

        # 3. random fractional

        rand_frac *= (n/2) / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)
        start_vecs.append(rand_frac)

        for base_h in start_vecs:
            if time.time() > deadline:
                break

            h = np.clip(h, 0.0, 1.0)

            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)
            h_up = gradient_refine(h_up, rng, max_iter=10, init_step=0.02)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    if best_h is None:
        best_n = 180
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
