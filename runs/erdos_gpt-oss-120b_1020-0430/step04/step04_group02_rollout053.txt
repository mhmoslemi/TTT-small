def gradient_max_corr(h):
    n = h.shape[0]
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)
    k = idx - (n - 1)

    g = np.zeros_like(h)
    # positive shift k>0:
    if k >= 0:
        # i from 0 to n-1-k
        i = np.arange(n - k)
        g[i] += (1 - h[i + k])
        # i from k to n-1
        j = np.arange(k, n)
        g[j] -= h[j - k]
    else:
        k2 = -k
        # i from k2 to n-1
        i = np.arange(k2, n)
        g[i] += (1 - h[i - k2])
        # i from 0 to n-1-k2
        j = np.arange(n - k2)
        g[j] -= h[j + k2]
    # Multiply by dx factor
    g = g * (2.0 / n)
    return g, corr[idx] * (2.0 / n), k

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

# ---------- projection onto box with exact sum ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)

# ---------- C5 evaluation ----------
def compute_c5(h):
    n = h.shape[0]
    corr = correlate(h, 1.0 - h, mode="full", method="fft")
    return np.max(corr) * (2.0 / n)

# ---------- gradient of the active shift ----------
def gradient_max_corr(h):
    n = h.shape[0]
    corr = correlate(h, 1.0 - h, mode="full", method="fft")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    cur_c = corr[idx] * (2.0 / n)
    g = np.zeros_like(h)
    if k >= 0:
        i = np.arange(n - k)
        g[i] = 1.0 - h[i + k]
        j = np.arange(k, n)
        g[j] -= h[j - k]
    else:
        k2 = -k
        i = np.arange(k2, n)
        g[i] = 1.0 - h[i - k2]
        j = np.arange(n - k2)
        g[j] -= h[j + k2]
    g *= (2.0 / n)
    return g, cur_c, k

# ---------- simple hill‑climbing moves ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h

def hill_climb_targeted(h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            cand = targeted_move(h, rng, k_star, scale=step) if rng.random() < p_target else perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

def compute_c5_and_idx(h):
    n = h.shape[0]
    corr = correlate(h, 1.0 - h, mode="full", method="fft")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / n), idx

# ---------- SLSQP with soft‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = correlate(x, 1.0 - x, mode="full", method="fft")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-13, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf

    # initialise helper for a given resolution
    def init_h(n):
        if "initial_h_values" in globals():
            old = np.array(initial_h_values, dtype=float)
            old_n = old.size
            old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            h = np.interp(new_x, old_x, old)
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, n / 2.0)
            return h
        else:
            return random_feasible_vector(n, rng)

    # phases: (resolution, fraction of total time)
    phases = [(540, 0.45), (1080, 0.45)]
    for phase_idx, (n, frac) in enumerate(phases):
        phase_budget = budget_s * frac
        phase_start = time.time()

        if phase_idx == 0:
            h = init_h(n)
        else:
            # upscale previous best
            old_n = best_h.size
            old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            h = np.interp(new_x, old_x, best_h)
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, n / 2.0)

        # gradient‑descent loop
        step0 = 0.6
        it = 0
        no_improve = 0
        while time.time() - phase_start < phase_budget:
            it += 1
            g, cur_c, _ = gradient_max_corr(h)
            # decaying step size
            step = step0 / (1.0 + it * 1e-3)
            improved = False
            for _ in range(10):
                h_new = project_box_sum(h - step * g, n / 2.0)
                new_c = compute_c5(h_new)
                if new_c < cur_c - 1e-12:
                    h, cur_c = h_new, new_c
                    improved = True
                    no_improve = 0
                    break
                step *= 0.5
            if not improved:
                no_improve += 1
                if no_improve > 300:
                    h = random_feasible_vector(n, rng)
                    no_improve = 0
                elif rng.random() < 0.001:
                    h = perturb_h(h, rng, scale=0.5)

            if cur_c < best_c - 1e-12:
                best_c, best_h = cur_c, h.copy()

        # keep best for next phase
        if phase_idx < len(phases) - 1:
            # upscale now for the next iteration
            continue

    # optional Hill‑Climb fine‑tuning
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 20.0 and best_h is not None:
        hc_iters = min(2500, int(remaining * 5))
        best_h, best_c = hill_climb_targeted(best_h, rng, max_iters=hc_iters, init_step=0.3)

    # SLSQP refinement with increasing beta
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 10.0 and best_h is not None:
        beta_list = [50, 200, 800, 2000, 5000, 10000, 20000, 50000]
        for beta in beta_list:
            if time.time() - start_time > budget_s * 0.97:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
            except Exception:
                continue

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
