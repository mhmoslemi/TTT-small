        s = -s
        out = np.empty_like(arr)

def pgd_max_corr(h, target, iters, step0):
    for k in range(iters):
        # compute shift with max correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        s = idx - (h.size - 1)
        # compute gradient
        h_plus = shift_arr(h, s)
        h_minus = shift_arr(h, -s)
        grad = 1.0 - (h_plus + h_minus)
        # step size schedule
        step = step0 * (0.99 ** k)
        h = project_to_sum_box(h - step * grad, target)
    return h

import time
import numpy as np
from scipy.optimize import minimize

# Global variables for softmax surrogate
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def shift_arr(arr, s):
    """Return arr shifted by s (out-of-range entries become 0)."""
    n = arr.shape[0]
    out = np.empty_like(arr)
    if s >= 0:
        out[: n - s] = arr[s:]
        out[n - s :] = 0.0
    else:
        s = -s
        out[s:] = arr[: n - s]
        out[:s] = 0.0
    return out


def project_to_sum_box(x, target):
    """Project x onto [0,1]^n with sum == target using bisection on offset λ."""
    lo = -np.max(x)
    hi = 1.0 - np.min(x)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random vector projected onto the feasible set Σh = n/2, 0≤h≤1."""
    x = rng.random(n)
    return project_to_sum_box(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def pgd_max_corr(h, target, iters, step0):
    """Projected gradient descent on the max correlation shift."""
    for k in range(iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        s = idx - (h.size - 1)
        h_plus = shift_arr(h, s)
        h_minus = shift_arr(h, -s)
        grad = 1.0 - (h_plus + h_minus)
        step = step0 * (0.99 ** k)
        h = project_to_sum_box(h - step * grad, target)
    return h


def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum projection
    x = project_to_sum_box(x, CURRENT_TARGET)
    return x


def max_overlap_and_shift(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift


def pair_transfer_optimize(
    h0,
    rng,
    max_iters=8000,
    time_limit=30.0,
    init_factor=0.5,
    decay=0.999,
):
    """Local search moving mass along the worst‑overlap shift."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # index ranges where shift is applicable
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c


def generate_periodic_mask(n, block_len, gap_len):
    """0/1 mask consisting of alternating blocks of length block_len and gaps of length gap_len."""
    mask = np.zeros(n, dtype=float)
    pos = 0
    while pos < n:
        end = min(pos + block_len, n)
        mask[pos:end] = 1.0
        pos = end + gap_len
    return mask
