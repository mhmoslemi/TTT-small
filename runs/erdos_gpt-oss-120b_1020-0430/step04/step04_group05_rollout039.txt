def project_sum_and_bounds(h, target):
    h = np.clip(h, 0., 1.)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # free indices
    mask = (diff > 0) & (h < 1.0) | (diff < 0) & (h > 0.0)
    # but we need to incorporate dynamic updates as some become saturated
    # We'll use a while loop.
    while abs(diff) > 1e-12:
        free_idx = np.where(mask)[0]
        if free_idx.size == 0:
            # no free indices, break
            break
        add = diff / free_idx.size
        # clip the addition to avoid crossing bounds
        # compute new values
        new_vals = h[free_idx] + add
        # identify which become out of bounds
        over = (add > 0) & (new_vals > 1.0)
        under = (add < 0) & (new_vals < 0.0)
        # set new_vals to bound where needed
        new_vals[over] = 1.0
        new_vals[under] = 0.0
        # update h
        h[free_idx] = new_vals
        diff = target - h.sum()
        # update mask
        mask = (diff > 0) & (h < 1.0) | (diff < 0) & (h > 0.0)
    return h

def project_onto_box_simplex(h, target, tol=1e-12, max_iter=50):
    # binary search on lambda
    low = -1.0 # lower bound for lambda: could be -1 to ensure all become 0 after shift.
    high = 1.0 # high bound such that all become 1 after shift.
    # But we may need more extreme bounds; we can compute min and max needed.
    # Since h_i in [0,1], shift lambda in [-1,1] ensures h_i+lambda in [-1,2] ; after clipping sum range from 0 to n.
    # Should be enough.
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(h + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
        if high - low < tol:
            break
    x = np.clip(h + (low+high)/2.0, 0.0, 1.0)
    return x

def compute_S_k_fft(h):
    n = h.size
    L = next_pow2(2 * n - 1)
    H = np.fft.rfft(h, n=L)
    S_full = np.fft.irfft(H * np.conj(H), n=L)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_onto_box_simplex(v, target, tol=1e-9, max_iter=100):
    low, high = -1.0, 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
        if high - low < tol:
            break
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)


# ---------- FFT correlation ----------
def next_pow2(x):
    return 1 << ((x - 1).bit_length())


def compute_S_k_fft(h):
    """S_k = sum_i h_i * h_{i+k} for k=0..n-1 (zero‑padded)."""
    n = h.size
    L = next_pow2(2 * n - 1)
    H = np.fft.rfft(h, n=L)
    S_full = np.fft.irfft(H * np.conj(H), n=L)
    return S_full[:n]


def compute_c5_fast(h):
    """Exact C5 for a discrete step function."""
    n = h.size
    target = n / 2.0
    S = compute_S_k_fft(h)
    min_S = S.min()
    c = (target - min_S) * (2.0 / n)
    return float(c), S

    return project_onto_box_simplex(v, n / 2.0)
