def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99
    best_h = None
    best_c = np.inf
    best_n = None

    for n in [720, 960]:
        if time.time() > deadline:
            break
        # generate initial candidate binary from initial_h_values if not None
        base_candidates = []
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            # interpolate to current n
            xs = np.linspace(0., 2., parent.size, endpoint=False)
            xt = np.linspace(0., 2., n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            # binary via top n/2 entries
            idx = np.argpartition(interp, -n//2)[-n//2:]
            bin_vec = np.zeros(n, dtype=np.float64)
            bin_vec[idx] = 1.0
            base_candidates.append(bin_vec.copy())
        # also random binary candidate
        for _ in range(2):
            bin_rand = np.zeros(n, dtype=np.float64)
            bin_rand[rng.choice(n, n//2, replace=False)] = 1.0
            base_candidates.append(bin_rand.copy())

        for init_h in base_candidates:
            # evaluate and maybe skip if not enough time
            if time.time() > deadline:
                break
            # compute initial correlation
            corr = np.correlate(init_h, 1.0 - init_h, mode='full')
            cur_c = float(corr.max() * (2.0 / n))
            # keep best if initial improvement
            if cur_c < best_c:
                best_h, best_c, best_n = init_h.copy(), cur_c, n

            # Simulated annealing: allocate some time portion
            # For each init, run SA with maybe time per run = (deadline - time.time()) * 0.3
            run_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.3)
            # SA
            h_sa, c_sa = simulated_annealing_with_corr(
                h=init_h.copy(),
                corr=corr,
                cur_c=cur_c,
                rng=rng,
                time_deadline=run_deadline,
            )
            if c_sa < best_c:
                best_h, best_c, best_n = h_sa.copy(), c_sa, n

            # After SA, apply gradient refinement and SLSQP cascade
            h_opt = gradient_refine(h_sa.copy(), rng, step=0.02, max_iter=12)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n

            # SLSQP cascade
            for beta in (5., 20., 80., 200., 500., 1000., 2000., 5000.):

                    break
                try:
                    h_opt = slsqp_optimize(h_opt, beta, rng, max_iter=600)

                c_opt = compute_c5(h_opt)

def simulated_annealing_with_corr(h, corr, cur_c, rng, time_deadline):
    n = h.size
    offset = n - 1
    ks = np.arange(-n+1, n)
    # sets as described
    # maintain position maps for i and j for fast updates as sets
    # initialize maps
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    pos_one = np.empty(n, dtype=int)
    pos_zero = np.empty(n, dtype=int)
    pos_one[ones] = np.arange(len(ones))
    pos_zero[zeros] = np.arange(len(zeros))

    best_h = h.copy()
    best_c = cur_c

    it = 0
    # initial temperature
    T0 = max(0.01 * cur_c, 1e-5)

    while time.time() < time_deadline:
        # random swap
        idx_one = rng.integers(0, len(ones))
        idx_zero = rng.integers(0, len(zeros))
        i = ones[idx_one]
        j = zeros[idx_zero]

        # compute delta
        # define function delta_corr_swap_inline
        # compute mask arrays
        # note using old h values
        # compute new_corr = corr + delta
        # compute new_c = new_corr.max() * (2.0 / n)

        # compute temperature
        T = T0 * (0.9995 ** it)

        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            # accept
            cur_c = new_c
            corr = new_corr
            # update h
            h[i] = 0.0
            h[j] = 1.0
            # update ones/zeros arrays and pos maps
            # swap elements
            # for ones: replace i with j
            ones[idx_one] = j

            # no longer i

            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        # else reject: no changes

        it += 1
    return best_h, best_c

    n = len(h)
    # we need correlation of h with (1-h). Compute convolution via FFT:
    a = h.astype(np.float64)
    b = (1.0 - h).astype(np.float64)

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def adjust_sum_exact(h, rng):
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]       # + (1 - h[i+k])
            g[k:] -= h[:upper]               # - h[i]

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):

        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])

            h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c = h_up, c_up

    if best_h is None:
        n = 360
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
