def binary_sa_restart(h_bin, rng, iter_limit, T0, Tf):
    h = h_bin.copy()
    best_h = h.copy()
    best_c = compute_c5_bin(h, L, dx)
    n = h.size
    for it in range(iter_limit):
        # temperature schedule:
        t = T0 * (Tf/T0)**(it/iter_limit)
        # propose swap:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # ensure i is 1 and j is 0 (swap)
        if h[i] == 0:
            i, j = j, i
        # perform swap:
        h[i] = 0
        h[j] = 1
        c = compute_c5_bin(h, L, dx)
        delta = c - best_c
        if delta < 0 or rng.random() < np.exp(-delta/t):
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def _enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def _random_feasible(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return _enforce_sum(h, n / 2.0, rng)


def _perturb(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def _correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def _compute_c5(h):
    """C5 = 1 - min_autocorr * dx, using FFT autocorrelation."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    conv = np.fft.irfft(fh * np.conj(fh), L)[: 2 * n - 1]
    minB = conv.min()
    return 1.0 - minB * (2.0 / n)


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    conv = _correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def _slsqp_optimize(h0, beta, rng, max_iter=250):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.shape[0]
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = _enforce_sum(x, CURRENT_TARGET, rng)
    return x


def _binary_sa(start_h, rng, deadline):
    """Swap‑based simulated annealing on a binary vector."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _compute_c5(h)
    T0 = max(best_c * 0.02, 1e-4)
    Tf = 1e-6
    est = int((deadline - time.time()) / 0.00012)
    it_max = max(2000, min(est, 200000))
    for it in range(it_max):
        if time.time() >= deadline:
            break
        t = T0 * (Tf / T0) ** (it / it_max)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] == 0:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = _compute_c5(h)
        delta = c - best_c
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


def _hill_climb(start_h, rng, max_iters, step):
    h = start_h.copy()
    best_h = h.copy()
    best_c = _compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        new_h = _perturb(h, rng, scale=step)
        c = _compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c
