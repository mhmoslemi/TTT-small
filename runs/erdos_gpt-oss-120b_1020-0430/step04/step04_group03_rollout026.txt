def project_box_sum(v, target):
    # find lambda such that sum(clip(v - lam, 0, 1)) = target
    lo = -1.0

def gradient_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        # indices 0..n-s-1 for h_i interacting with h_{i+s}
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad
```

    current = h.copy()
    best = current.copy()
    best_val = compute_c5(current)
    cur_val = best_val
    # set initial step scale based on average allowed epsilon?
    step_scale = 0.1   # maximum epsilon as fraction of max possible
    # temperature schedule
    t0 = 0.0005
    decay = 0.9995
    temp = t0
    while time.time() - start_time < time_limit:

        if i == j:
            continue

import time, numpy as np
from scipy.optimize import minimize

# ---------- constraint projection ----------
def project_box_sum(v, target):
    """Project onto {0<=x<=1, sum(x)=target} via bisection on a Lagrange multiplier."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def random_feasible(n, rng):
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)

# ---------- evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (h.size - 1)
    return shift, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    g = np.zeros(n)
    if s >= 0:
        g[: n - s] += (1 - h[s:])
        g[s:] -= h[: n - s]
    else:
        a = -s
        g[a:] += (1 - h[: n - a])
        g[: n - a] -= h[a:]
    return g

# ---------- local search ----------
def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def greedy_optimize(h, rng, max_steps=2000):
    for _ in range(max_steps):
        s, _, cur = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break

        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue

        # try half
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue

        # try a small random fraction
        eps = max_eps * 0.2 * rng.random()
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue

        break
    return h

def simulated_annealing(h, rng, time_limit, start_time):
    cur = h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    temp = 5e-4
    decay = 0.9995
    step_scale = 0.1

    while time.time() - start_time < time_limit:
        n = cur.size
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if cur[i] <= 1e-12 or cur[j] >= 1.0 - 1e-12:
            continue
        max_eps = min(cur[i], 1.0 - cur[j])
        eps = max_eps * step_scale * rng.random()
        nxt = cur.copy()
        nxt[i] -= eps
        nxt[j] += eps
        nxt_val = compute_c5(nxt)
        delta = nxt_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur, cur_val = nxt, nxt_val
            if nxt_val < best_val:
                best, best_val = nxt, nxt_val
        temp *= decay
        if rng.random() < 0.01:
            step_scale = max(1e-4, step_scale * 0.99)
    return best, best_val

def slsqp_optimize(h0, beta, max_iter, rng):
    n = h0.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_box_sum(x, target)
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible(720, rng)
    n = h0.size
    h0 = project_box_sum(np.clip(h0, 0.0, 1.0), n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- multi‑restart greedy phase -----
    greedy_deadline = start + budget_s * 0.45
    while time.time() < greedy_deadline:
        # starting point: random perturb of current best or fresh random
        if rng.random() < 0.6:
            start_vec = random_feasible(n, rng)
        else:
            start_vec = perturb_random(best_h, rng, scale=0.3)
        local = greedy_optimize(start_vec, rng, max_steps=2500)
        local_c = compute_c5(local)
        if local_c < best_c:
            best_h, best_c = local, local_c

    # ----- simulated annealing fine‑tuning -----
    sa_deadline = start + budget_s * 0.90
    if time.time() < sa_deadline:
        sa_h, sa_c = simulated_annealing(best_h, rng, sa_deadline, start)
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c

    # ----- occasional SLSQP polishing if time permits -----
    polish_deadline = start + budget_s * 0.99
    if time.time() < polish_deadline:
        for beta in (200, 800, 2000, 5000):
            if time.time() > polish_deadline:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=700, rng=rng)
            except Exception:
                continue
            refined_c = compute_c5(refined)
            if refined_c < best_c:
                best_h, best_c = refined, refined_c

    # final projection and evaluation
    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
