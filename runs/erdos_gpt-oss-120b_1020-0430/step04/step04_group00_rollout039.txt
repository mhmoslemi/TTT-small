def binary_sa_search(n, rng, time_deadline):
    h = np.zeros(n, dtype=float)
    # random positions for ones
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # positions lists
    ones = list(idx)
    zeros = [i for i in range(n) if h[i]==0]

    # temperature schedule
    T = cur_c * 0.2 + 1e-8
    cooling = 0.99996
    it = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            # update lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T *= cooling
        it += 1
        # optional restart if temperature too low or no improvement for many steps
        if it % 20000 == 0:
            # maybe reheat
            T = cur_c * 0.2 + 1e-8
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp scaled
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth max (log‑sum‑exp) surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # + (1 - h[i+k])
            g[k:] -= h[:upper]               # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_search(h_init, cur_c, rng, deadline):
    """SA on binary vectors with exact weight."""
    h = h_init.copy()
    n = h.shape[0]
    # build ones/zeros lists and index maps
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] <= 0.5]
    idx_one = np.full(n, -1, dtype=int)
    idx_zero = np.full(n, -1, dtype=int)
    for p, i in enumerate(ones):
        idx_one[i] = p
    for p, i in enumerate(zeros):
        idx_zero[i] = p

    best_h = h.copy()
    best_c = cur_c
    cur_c_val = cur_c

    T = cur_c * 0.2 + 1e-12
    cooling = 0.99994
    it = 0
    while time.time() < deadline:
        # pick random swap
        pos_i = rng.integers(len(ones))
        i = ones[pos_i]
        pos_j = rng.integers(len(zeros))
        j = zeros[pos_j]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c_val
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c_val = new_c
            # update lists via index maps
            # i moves to zeros at position pos_j
            zeros[pos_j] = i
            idx_zero[i] = pos_j
            # j moves to ones at position pos_i
            ones[pos_i] = j
            idx_one[j] = pos_i
            # clean old entries
            idx_one[i] = -1
            idx_zero[j] = -1
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T *= cooling
        it += 1
        if it % 30000 == 0:
            # reheating if temperature too low
            if T < 1e-12:
                T = cur_c_val * 0.2 + 1e-12
    return best_h, best_c

        if cur_c_opt < best_c:
            best_c = cur_c_opt
            best_h = h_opt.copy()
            best_n = n

    # optional up‑sampling + refinement if time left
    if best_h is not None and time.time() < deadline - 5.0 and best_n * 2 <= 720:
        n_up = best_n * 2

        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:

            best_h = h_up
            best_n = n_up

            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_search(h_init, cur_c, rng, deadline):
    """Simulated annealing on a binary vector with exact weight."""
    h = h_init.copy()
    n = h.shape[0]
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] <= 0.5]
    idx_one = np.full(n, -1, dtype=int)
    idx_zero = np.full(n, -1, dtype=int)
    for p, i in enumerate(ones):
        idx_one[i] = p
    for p, i in enumerate(zeros):
        idx_zero[i] = p

    best_h = h.copy()
    best_c = cur_c
    cur_c_val = cur_c

    T = cur_c * 3.0 + 1e-12   # initial temperature
    cooling = 0.99995
    it = 0
    while time.time() < deadline:
        pos_i = rng.integers(len(ones))
        i = ones[pos_i]
        pos_j = rng.integers(len(zeros))
        j = zeros[pos_j]

        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c_val
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c_val = new_c
            # update data structures
            zeros[pos_j] = i
            idx_zero[i] = pos_j
            ones[pos_i] = j
            idx_one[j] = pos_i
            idx_one[i] = -1
            idx_zero[j] = -1
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cooling
        it += 1
        if it % 50000 == 0 and T < 1e-12:
            T = cur_c_val * 3.0 + 1e-12
    return best_h, best_c

    start = time.time()

            break

                bin_parent = (parent >= 0.5).astype(float)
                diff = half - int(bin_parent.sum())
                if diff > 0:
                    zeros = np.flatnonzero(bin_parent == 0)
                    if len(zeros) >= diff:
                        turn = rng.choice(zeros, diff, replace=False)
                        bin_parent[turn] = 1.0
                elif diff < 0:
                    ones = np.flatnonzero(bin_parent == 1)
                    turn = rng.choice(ones, -diff, replace=False)
                    bin_parent[turn] = 0.0
                h0 = bin_parent.copy()
        cur_c0 = compute_c5(h0)

        remaining = deadline - time.time()

        sub_dead = min(deadline, time.time() + slice_time)
        h_sa, c_sa = binary_sa_search(h0, cur_c0, rng, sub_dead)

        # gradient refinement
        h_grad = gradient_refine(h_sa.copy(), rng, step=0.02, max_iter=12)
        c_grad = compute_c5(h_grad)

        # SLSQP cascade
        h_opt = h_grad
        cur_c_opt = c_grad
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() > deadline:
                break
            max_iter = 500 if beta < 200 else 800

            except Exception:
                continue
            cur_c_opt = compute_c5(h_opt)
        # final gradient sweep
        h_opt = gradient_refine(h_opt, rng, step=0.02, max_iter=12)
        cur_c_opt = compute_c5(h_opt)

        if cur_c_opt < best_c:
            best_c = cur_c_opt
            best_h = h_opt.copy()
            best_n = n

    if best_h is not None and time.time() < deadline - 5.0 and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)

        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:

            best_h = h_up
            best_n = n_up

        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.shape[0])
