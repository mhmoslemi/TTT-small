def compute_corr_fft(h):
    n = h.size
    N = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, N)
    rev_a = (1 - h)[::-1]  # reversed
    A = np.fft.rfft(rev_a, N)
    conv = np.fft.irfft(H * A, N)
    # conv is length N. The result for full correlation of length 2n-1 is conv[:2*n-1]
    return conv[:2*n-1]

# ---------- projection onto [0,1] box with exact sum ----------
def project_to_sum_box(x, target):
    """Project x onto the intersection of the box [0,1]^n and sum(x)=target."""
    # binary search for lambda such that sum(clamp(x+lambda,0,1)) = target
    low, high = -1.0, 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            high = mid
        else:
            low = mid
    lam = (low + high) * 0.5
    proj = np.clip(x + lam, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = np.where((proj > 0.0) & (proj < 1.0))[0]
        if free.size:
            add = diff / free.size
            proj[free] += add
            proj = np.clip(proj, 0.0, 1.0)
    return proj

# ---------- correlation via FFT ----------
def compute_corr_fft(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    A = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(H * A, N)[: 2 * n - 1]
    return conv[::-1]  # reverse to match np.correlate full mode

def compute_c5_fft(h):
    n = h.shape[0]
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)

# ---------- gradient descent on the max‑lag objective ----------
def gradient_descent(h_start, target_sum, time_end, rng,
                    step0=0.05, decay=0.99997):
    n = h_start.size
    h_cur = h_start.copy()
    cur_c = compute_c5_fft(h_cur)
    best_h, best_c = h_cur.copy(), cur_c
    step = step0
    while time.time() < time_end:
        corr = compute_corr_fft(h_cur)
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        L = abs(lag)
        if L == 0:
            shift_r = h_cur
            shift_l = h_cur
        else:
            shift_r = np.concatenate([h_cur[L:], np.zeros(L, dtype=h_cur.dtype)])
            shift_l = np.concatenate([np.zeros(L, dtype=h_cur.dtype), h_cur[: n - L]])
        grad = 1.0 - shift_r - shift_l
        h_new = h_cur - step * grad
        h_new = project_to_sum_box(h_new, target_sum)
        new_c = compute_c5_fft(h_new)
        if new_c < cur_c:
            h_cur, cur_c = h_new, new_c
            if new_c < best_c:
                best_h, best_c = h_new.copy(), new_c
            step = min(step0, step / decay)
        else:
            step *= decay
        # occasional random perturbation
        if rng.random() < 0.01:
            noise = (rng.random(n) - 0.5) * 0.02
            cand = np.clip(h_cur + noise, 0.0, 1.0)
            cand = project_to_sum_box(cand, target_sum)
            cand_c = compute_c5_fft(cand)
            if cand_c < cur_c:
                h_cur, cur_c = cand, cand_c
                if cand_c < best_c:
                    best_h, best_c = cand.copy(), cand_c
        # restart if step becomes too tiny
        if step < 1e-7:
            h_cur = best_h.copy()
            noise = (rng.random(n) - 0.5) * 0.1
            h_cur = np.clip(h_cur + noise, 0.0, 1.0)
            h_cur = project_to_sum_box(h_cur, target_sum)
            cur_c = compute_c5_fft(h_cur)
            step = step0
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # ---- initial vector ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(360, rng)
    n = h0.size
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_sum_box(h0, n / 2.0)

    best_h, best_c = h0.copy(), compute_c5_fft(h0)

    # ---- main optimization (coarse grid) ----
    main_deadline = t0 + budget_s * 0.85
    while time.time() < main_deadline:
        # pick a start point
        if rng.random() < 0.6:
            start = best_h.copy()
            noise = (rng.random(n) - 0.5) * 0.05
            start = np.clip(start + noise, 0.0, 1.0)
            start = project_to_sum_box(start, n / 2.0)
        else:
            start = random_feasible_vector(n, rng)

        time_left = main_deadline - time.time()
        if time_left < 1.0:
            break
        alloc = min(time_left * 0.5, 35.0)
        sub_deadline = time.time() + alloc

        local_h, local_c = gradient_descent(start, n / 2.0, sub_deadline, rng)
        if local_c < best_c:
            best_h, best_c = local_h, local_c

    # ---- optional refinement on a finer grid ----
    remaining = budget_s - (time.time() - t0)
    if remaining > budget_s * 0.15:
        new_n = n * 2
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = project_to_sum_box(h_up, new_n / 2.0)

        up_deadline = time.time() + remaining * 0.9
        local_up_h, local_up_c = gradient_descent(h_up, new_n / 2.0,
                                                  up_deadline, rng,
                                                  step0=0.03)
        if local_up_c < best_c:
            best_h, best_c, n = local_up_h, local_up_c, new_n

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum_box(best_h, best_h.size / 2.0)
    final_c = compute_c5_fft(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
