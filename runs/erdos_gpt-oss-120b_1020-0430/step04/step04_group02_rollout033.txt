def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h

import time
import numpy as np

# ---------- projection onto box with exact sum ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)


# ---------- FFT convolution ----------
def conv_fft(a, b):
    n = a.size + b.size - 1
    L = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, n=L)[:n]


# ---------- correlation and C5 ----------
def compute_bc(h):
    return conv_fft(h, (1 - h)[::-1])


def compute_c5(h):
    n = h.size
    bc = compute_bc(h)
    return np.max(bc) * (2.0 / n)


def compute_c5_and_idx(h):
    n = h.size
    bc = compute_bc(h)
    idx = int(np.argmax(bc))
    return bc[idx] * (2.0 / n), idx


# ---------- smooth max (log‑sum‑exp) ----------
def softmax_objective(h, beta):
    bc = compute_bc(h)
    exp_arg = beta * bc
    m = np.max(exp_arg)
    sum_exp = np.sum(np.exp(exp_arg - m))
    return (m + np.log(sum_exp)) / beta


def softmax_obj_grad(h, beta):
    n = h.size
    bc = compute_bc(h)
    exp_arg = beta * bc
    m = np.max(exp_arg)
    shift = np.exp(exp_arg - m)
    p = shift / np.sum(shift)                     # softmax weights
    # gradient components via convolutions
    p_rev = p[::-1]
    A_full = conv_fft(p_rev, 1 - h)               # size 3n-2
    B_full = conv_fft(p, h)
    start = n - 1
    A = A_full[start:start + n]
    B = B_full[start:start + n]
    grad = A - B
    f = (m + np.log(np.sum(shift))) / beta
    return f, grad


# ---------- local moves ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low.any() or not high.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low))
    i_high = rng.choice(np.flatnonzero(high))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i_low] += eps
    nh[i_low + k] += eps
    nh[i_high] -= eps
    nh[i_high + k] -= eps
    return nh


def hill_climb_targeted(h_start, rng, max_iters=500, init_step=0.3):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (h.size - 1)
        improved = False
        for _ in range(4):
            if rng.random() < 0.25:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- gradient descent with smoothing ----------
def gradient_descent_optimize(
    h0, target_sum, beta_schedule, iters_per_beta, step0, rng, time_limit
):
    h = h0.copy()
    for beta in beta_schedule:
        step = step0
        for _ in range(iters_per_beta):
            if time.time() > time_limit:
                return h
            f, grad = softmax_obj_grad(h, beta)
            # backtracking line search (objective only)
            success = False
            for _ in range(8):
                h_new = project_box_sum(h - step * grad, target_sum)
                f_new = softmax_objective(h_new, beta)
                if f_new < f:
                    h = h_new
                    f = f_new
                    step *= 1.1
                    success = True
                    break
                step *= 0.5
            if not success:
                if step < 1e-9:
                    break
    return h


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ---- base vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    # ---- upsample to a finer grid ----
    target_n = max(720, n0 * 2)
    if target_n != n0:
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_initial = np.interp(new_x, old_x, h0)
        h_initial = project_box_sum(h_initial, target_n / 2.0)
    else:
        h_initial = h0.copy()
    n = target_n
    target_sum = n / 2.0

    best_h = h_initial.copy()
    best_c = compute_c5(best_h)

    # optimisation parameters
    beta_schedule = [200, 800, 3200, 12800, 51200]
    iters_per_beta = 1500
    step0 = 0.5
    restart = 0

    while time.time() - start_time < budget_s * 0.85:
        if restart == 0:
            h_start = h_initial.copy()
        else:
            h_start = perturb_h(best_h, rng, scale=0.15)

        # gradient descent phase
        h_opt = gradient_descent_optimize(
            h_start,
            target_sum,
            beta_schedule,
            iters_per_beta,
            step0,
            rng,
            start_time + budget_s * 0.75,
        )

        # final hill‑climb fine tune
        h_opt, c_opt = hill_climb_targeted(h_opt, rng, max_iters=800, init_step=0.25)

        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        restart += 1
        if time.time() - start_time > budget_s * 0.95:
            break

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
