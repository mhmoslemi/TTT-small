def compute_c5_binary(h):

    # conv = np.correlate(h, h, mode='full') - but we can compute via FFT

def run(seed=42, budget_s=1100, **kwargs):

    T0 = 0.02
    Tf = 1e-5
    max_iters = 300000
    it = 0
    while time.time() - start < budget_s * 0.9 and it < max_iters:

        # compute worst lag

def softmax_obj(x):
    # compute correlation as vector
    n = x.size
    n_fft = 1 << ((2*n-1).bit_length())
    # compute fft of x and (1 - x) each iter; heavy but okay for refinement few times.
    fft_x = np.fft.fft(x, n_fft)
    fft_one_minus_x = np.fft.fft(1 - x, n_fft)
    corr = np.fft.ifft(fft_x * np.conj(fft_one_minus_x)).real[:2*n-1]
    # compute softmax approx of max (smooth max)
    # use scaling beta
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_refine(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global parameters for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    n = x.size
    n_fft = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.fft(x, n_fft)
    f1mx = np.fft.fft(1.0 - x, n_fft)
    corr = np.fft.ifft(fx * np.conj(f1mx)).real[: 2 * n - 1]
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h0, beta, rng, max_iter=300):
    """Refine a vector with SLSQP on the smoothed max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (tiny adjustments)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def compute_c5_fft(h):
    """Exact C5 using FFT‑based linear correlation."""
    n = h.size
    n_fft = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.fft(h, n_fft)
    f1mh = np.fft.fft(1.0 - h, n_fft)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real[: 2 * n - 1]
    dx = 2.0 / n
    return float(corr.max() * dx), corr


def select_swap_for_lag(h, lag, rng):
    """Find a 1↔0 swap that improves overlap at the given lag."""
    n = h.size
    if lag >= 0:
        limit = n - lag
        a = np.where((h[:limit] == 1) & (h[lag:] == 0))[0]
        b = np.where((h[:limit] == 0) & (h[lag:] == 1))[0]
        if a.size and b.size:
            i = rng.choice(a)
            j = rng.choice(b)
            return int(i), int(j)
    else:
        d = -lag
        limit = n - d
        a = np.where((h[d:] == 1) & (h[:limit] == 0))[0]
        b = np.where((h[d:] == 0) & (h[:limit] == 1))[0]
        if a.size and b.size:
            i = d + rng.choice(a)
            j = rng.choice(b)
            return int(i), int(j)
    return None


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ---- parameters ----
    n = 720                     # grid size
    dx = 2.0 / n
    n_fft = 1 << ((2 * n - 1).bit_length())
    # ---- initial binary vector ----
    h = np.zeros(n, dtype=np.uint8)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1
    # ---- simulated annealing on the binary space ----
    T0, Tf = 0.02, 1e-5
    max_iters = 600_000        # generous upper bound, will be cut by time
    it = 0
    # evaluate initial state
    cur_c5, _ = compute_c5_fft(h)
    best_h = h.copy()
    best_c5 = cur_c5

    while it < max_iters and (time.time() - start_time) < budget_s * 0.85:
        # temperature
        T = T0 * (Tf / T0) ** (it / max_iters)

        # correlation to locate worst lag
        _, corr = compute_c5_fft(h)
        worst_idx = int(corr.argmax())
        lag = worst_idx - (n - 1)

        swap = select_swap_for_lag(h, lag, rng)
        if swap is None:
            # random 1↔0 swap
            ones_pos = np.flatnonzero(h)
            zeros_pos = np.flatnonzero(1 - h)
            i = rng.choice(ones_pos)
            j = rng.choice(zeros_pos)
        else:
            i, j = swap

        # perform swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1

        new_c5, _ = compute_c5_fft(h_new)
        dE = new_c5 - cur_c5
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        it += 1

    # ---- optional continuous refinement with SLSQP ----
    # allocate remaining time
    remaining = budget_s - (time.time() - start_time)
    if remaining > budget_s * 0.05:
        h_float = best_h.astype(np.float64)
        beta_list = [50, 200, 800, 3000, 8000]
        for beta in beta_list:
            if (time.time() - start_time) >= budget_s:
                break
            refined = slsqp_refine(h_float, beta, rng, max_iter=250)
            refined_c5, _ = compute_c5_fft(refined)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined.copy()
                h_float = refined
    # final clean‑up
    final_c5, _ = compute_c5_fft(best_h)
    # guarantee exact sum (should already hold)
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h.tolist(), float(final_c5), int(n)
