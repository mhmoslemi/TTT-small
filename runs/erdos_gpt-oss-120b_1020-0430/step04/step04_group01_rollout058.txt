def gradient_descent(h0, beta_schedule, steps_per_beta, lr_initial, target, deadline):
    h = h0.copy()
    n = h.size
    target_sum = target
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    m = np.zeros_like(h)  # Adam m (first moment)
    v = np.zeros_like(h)  # Adam v (second moment)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() >= deadline:
                return best_h, best_c
            t += 1
            # compute gradient
            g = grad_logsumexp(h, beta)  # returns gradient of f w.r.t h
            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr_initial * m_hat / (np.sqrt(v_hat) + eps)
            # take step
            h_new = h - step
            # project
            h_new = project_capped_simplex(h_new, target_sum)
            h = h_new
            # optional evaluate current C5 occasionally (every 100 iter)
            if (t % 100) == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
    # final evaluation
    c = compute_c5(h)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2*n - 1]
    return conv

def grad_logsumexp(h, beta):
    n = h.shape[0]
    dx = 2.0 / n
    conv = correlate_full(h)  # length 2n-1
    # compute softmax weights
    z = beta * conv * dx
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    w = exp_z / exp_z.sum()
    # compute conv1 = convolution of w with (1 - h)
    # compute conv2 = convolution of w_rev with h
    L = 1 << ((3*n - 2).bit_length())
    w_pad = np.fft.rfft(w, L)
    omh = np.fft.rfft(1 - h, L)
    conv1_full = np.fft.irfft(w_pad * omh, L)[: 3*n - 2]
    w_rev = w[::-1]
    w_rev_pad = np.fft.rfft(w_rev, L)
    h_pad = np.fft.rfft(h, L)
    conv2_full = np.fft.irfft(w_rev_pad * h_pad, L)[: 3*n - 2]
    start = n - 1
    conv1 = conv1_full[start:start + n]
    conv2 = conv2_full[start:start + n]
    grad = dx * (conv1 - conv2)
    return grad

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def binary_sa_limited(start_bin, rng, deadline, max_iters=20000):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    # temperature schedule: we can compute initial T based on best
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # pick i with h[i]=1, j with h[j]=0
        # we can randomly pick i and j until difference
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # ensure i is 1, j is 0
        if h[i] < h[j]:
            i, j = j, i
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def max_gradient_descent(h, steps, lr, target, deadline):
    n = h.size
    dx = 2.0 / n
    for it in range(steps):
        if time.time() >= deadline:
            break
        conv = correlate_full(h) * dx
        s_idx = np.argmax(conv)  # index of max
        s = s_idx - (n - 1)  # shift
        # compute gradient
        # term1: (1 - h_{j - s}) if j - s in range else 0
        # term2: h_{j + s} if j + s in range else 0
        # gradient = dx * ((1 - h_{j - s}) - h_{j + s})
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        # compute with slicing
        if s >= 0:
            # j - s >= 0 => j >= s
            term1[s:] = 1 - h[:-s] if s != 0 else 1 - h
            # j + s <= n-1 => j <= n-1-s
            term2[:n - s] = h[s:]
        else:
            # s negative, let t = -s >0
            t = -s

import time
import numpy as np
from scipy.special import logsumexp

# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------
def project_capped_simplex(v, target):
    """Projection of v onto the set {0<=x<=1, sum x = target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        if proj.sum() > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def correlate_full(h):
    """Linear correlation of h with (1-h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def grad_logsumexp(h, beta):
    """Gradient of (1/β) log‑sum‑exp[β·conv·dx] w.r.t. h."""
    n = h.size
    dx = 2.0 / n
    conv = correlate_full(h)                # length 2n‑1
    z = beta * conv * dx
    maxz = np.max(z)
    w = np.exp(z - maxz)
    w /= w.sum()                            # soft‑max weights, sum=1

    # conv1 = w * (1‑h) , conv2 = w_rev * h
    L = 1 << ((3 * n - 2).bit_length())
    w_pad = np.fft.rfft(w, L)
    omh_pad = np.fft.rfft(1 - h, L)
    conv1_full = np.fft.irfft(w_pad * omh_pad, L)[: 3 * n - 2]

    w_rev = w[::-1]
    w_rev_pad = np.fft.rfft(w_rev, L)
    h_pad = np.fft.rfft(h, L)
    conv2_full = np.fft.irfft(w_rev_pad * h_pad, L)[: 3 * n - 2]

    start = n - 1
    conv1 = conv1_full[start:start + n]
    conv2 = conv2_full[start:start + n]
    return dx * (conv1 - conv2)


def random_feasible_vector(n, rng):
    """Uniform random point in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    return project_capped_simplex(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j, preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# --------------------------------------------------------------
# Simple hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    step = start_step
    decay_int = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay_int == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# --------------------------------------------------------------
# Binary simulated annealing (limited by deadline)
# --------------------------------------------------------------
def binary_sa_limited(start_bin, rng, deadline, max_iters=20000):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# --------------------------------------------------------------
# Gradient descent on smooth surrogate (Adam + projection)
# --------------------------------------------------------------
def gradient_descent(h0, beta_sched, steps_per_beta, lr0, target, deadline):
    h = h0.copy()
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    for beta in beta_sched:
        for _ in range(steps_per_beta):
            if time.time() >= deadline:
                return best_h, best_c
            t += 1
            g = grad_logsumexp(h, beta)
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
            h = project_capped_simplex(h - step, target)
            if t % 100 == 0:
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
    c = compute_c5(h)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c


# --------------------------------------------------------------
# Hard‑max gradient descent (focus on worst shift)
# --------------------------------------------------------------
def max_gradient_descent(h, steps, lr, target, deadline):
    n = h.size
    dx = 2.0 / n
    for _ in range(steps):
        if time.time() >= deadline:
            break
        conv = correlate_full(h) * dx
        idx = np.argmax(conv)
        s = idx - (n - 1)                     # integer shift
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if s > 0:
            term1[s:] = 1 - h[:-s]
            term2[:n - s] = h[s:]
        elif s < 0:
            t = -s
            term1[:n - t] = 1 - h[t:]
            term2[t:] = h[:n - t]
        else:
            term1 = 1 - h
            term2 = h
        g = dx * (term1 - term2)
        h = project_capped_simplex(h - lr * g, target)
    return h
