def gradient_multi_shift(h, shifts):
    n = h.size
    grad = np.zeros(n)
    for s in shifts:
        if s >= 0:
            grad[:n-s] += (1 - h[s:])
            grad[s:] -= h[:n-s]
        else:
            s_abs = -s
            grad[s_abs:] += (1 - h[:n-s_abs])
            grad[:n-s_abs] -= h[s_abs:]
    return grad

def multi_greedy_optimize(h0, max_steps=2000, tol_corr=1e-5):
    h = h0.copy()
    best_c = compute_c5(h)
    n = h.size
    for step in range(max_steps):
        # compute correlation across all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        mx = corr.max()
        # compute active shifts
        # Set tolerance as max(1e-2, mx*1e-5) // adjusting for scale
        tol = max(1e-2, mx * 1e-5)
        act_idx = np.where(corr >= mx - tol)[0]
        # Convert to shifts
        shifts = act_idx - (n - 1)
        # compute combined gradient
        grad = np.zeros(n)
        for s in shifts:
            if s >= 0:
                grad[:n - s] += (1 - h[s:])
                grad[s:] -= h[:n - s]
            else:
                s_abs = -s
                grad[s_abs:] += (1 - h[:n - s_abs])
                grad[:n - s_abs] -= h[s_abs:]
        # select i (positive grad) and j (negative grad)
        pos_mask = (grad > 1e-12) & (h > 1e-12)
        neg_mask = (grad < -1e-12) & (h < 1 - 1e-12)
        if not np.any(pos_mask) or not np.any(neg_mask):
            break
        i = np.argmax(grad * pos_mask)  # but argmax of array includes zeros from mask; we could do i = pos_mask.nonzero()[0][np.argmax(grad[pos_mask])]
        # compute candidates
        pos_inds = np.where(pos_mask)[0]
        i = pos_inds[np.argmax(grad[pos_inds])]
        neg_inds = np.where(neg_mask)[0]
        j = neg_inds[np.argmin(grad[neg_inds])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            break
        # try full epsilon then halving
        improved = False
        eps = max_eps
        for _ in range(5):
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            c_new = compute_c5(nh)
            if c_new + 1e-12 < best_c:
                h = nh
                best_c = c_new
                improved = True
                break
            eps *= 0.5
        if not improved:
            # try alternative direction? Move from j to i?
            # Actually i has positive grad, moving mass from i to j reduces C(s) for active shifts.
            # Maybe no improvement; break.
            break
    return h, best_c

import time, numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def enforce_sum(h, target, rng):
    """Clip h to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase total mass
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:      # need to decrease total mass
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Uniform random point on the box with sum=n/2 (approximately)."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def compute_c5(h):
    """Discrete C5 for vector h (dx factor included)."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def gradient_multi_shift(h, shifts):
    """Sum of gradients of correlation for a list of integer shifts."""
    n = h.size
    grad = np.zeros(n)
    for s in shifts:
        if s >= 0:
            grad[:n - s] += (1 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] += (1 - h[:n - s_abs])
            grad[:n - s_abs] -= h[s_abs:]
    return grad

def multi_greedy_optimize(h0, max_steps=2000, rng=None):
    """Greedy descent using the combined gradient of all near‑max shifts."""
    h = h0.copy()
    n = h.size
    best_c = compute_c5(h)
    for _ in range(max_steps):
        corr = np.correlate(h, 1 - h, mode='full')
        mx = corr.max()
        # active shifts: within a small absolute or relative tolerance
        tol = max(1e-2, mx * 1e-5)
        act_idx = np.where(corr >= mx - tol)[0]
        shifts = act_idx - (n - 1)

        grad = gradient_multi_shift(h, shifts)

        pos_mask = (grad > 1e-12) & (h > 1e-12)
        neg_mask = (grad < -1e-12) & (h < 1.0 - 1e-12)
        if not np.any(pos_mask) or not np.any(neg_mask):
            break

        pos_inds = np.where(pos_mask)[0]
        i = pos_inds[np.argmax(grad[pos_inds])]
        neg_inds = np.where(neg_mask)[0]
        j = neg_inds[np.argmin(grad[neg_inds])]

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            break

        improved = False
        eps = max_eps
        for _ in range(5):          # try full epsilon then halve a few times
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            c_new = compute_c5(nh)
            if c_new + 1e-12 < best_c:
                h = nh
                best_c = c_new
                improved = True
                break
            eps *= 0.5
        if not improved:
            break
    return h, best_c

# ---------- SLSQP polishing (soft‑max smoothing) ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    """Smooth max via soft‑max and run SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x
