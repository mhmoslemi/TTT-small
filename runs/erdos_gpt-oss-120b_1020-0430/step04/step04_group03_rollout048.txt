def project_sum(v, target, tol=1e-10):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo+hi)/2
        s = np.sum(np.clip(v+mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v+hi,0.0,1.0)

from scipy.optimize import differential_evolution

def de_optimize(n_blocks, block_size, target, rng):
    bounds = [(0.0, 1.0)] * n_blocks
    # objective takes param vector p
    def obj(p):
        h = np.repeat(p, block_size)
        h = project_sum(h, target)
        # compute c5
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / h.size)
    result = differential_evolution(obj, bounds, seed=int(rng.integers(2**31)), maxiter=200, popsize=15, polish=False, updating='deferred')
    best_p = result.x
    best_h = np.repeat(best_p, block_size)
    best_h = project_sum(best_h, target)
    return best_h

import time, numpy as np
from scipy.optimize import minimize, differential_evolution

# ---------- utilities ----------
def compute_c5(h):
    """Discrete C5 for vector h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)


def project_to_sum(v, target, iters=60):
    """Shift v by a constant and clip to [0,1] so that sum == target."""
    lo, hi = -1.0, 1.0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + hi, 0.0, 1.0)


def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_to_sum(v, n / 2.0)


def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def worst_shift_info(h):
    """Return (shift, idx, value) for the worst correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (h.size - 1)
    return shift, idx, corr[idx]


def gradient_shift(h, s):
    """Gradient of C_s w.r.t. h (for a fixed shift s)."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def greedy_optimize(h, max_steps=2000):
    """Deterministic improvement by moving mass between two points."""
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        improved = False
        for fac in (1.0, 0.5, 0.33, 0.25):
            eps = max_eps * fac
            nh = h.copy()
            nh[i] -= eps
            nh[j] += eps
            if compute_c5(nh) + 1e-12 < compute_c5(h):
                h = nh
                improved = True
                break
        if not improved:
            break
    return h


# ---------- SLSQP polishing ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = project_to_sum(x, CURRENT_TARGET, iters=60)
    return x
