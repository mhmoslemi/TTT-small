def proj_box_sum(v, target, tol=1e-12, max_iter=100):
    lo = -1.0  # tau lower bound (maybe negative, increase sum)
    hi = 1.0   # tau upper bound
    # Actually we need tau such that after clipping, sum(x) = target.
    # For large negative tau, x = min(v - tau, 1) -> values may be >1, clipped to 1, sum = n (max).
    # For large positive tau, x = max(v - tau, 0) -> sum 0.
    # So search tau in range [-1, 1] perhaps more: v elements in [0,1] maybe after gradient step values can go beyond these bounds. We'll allow a bigger range e.g., [-2,2].
    lo = -2.0
    hi = 2.0
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
        if hi - lo < tol:
            break
    x = np.clip(v - tau, 0.0, 1.0)
    # If sum slightly off due to tolerance, adjust a single element.
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # Add diff to a not-full or subtract from a non-zero element.
        idx = np.where((diff > 0) & (x < 1.0))[0]
        if idx.size > 0:
            i = idx[0]
            x[i] += diff
        else:
            idx = np.where((diff < 0) & (x > 0.0))[0]
            if idx.size > 0:
                i = idx[0]
                x[i] += diff
    # Clip final
    return np.clip(x, 0.0, 1.0)

import time, numpy as np

# ---------- basic utilities ----------
def compute_c5(h):
    """Exact discrete C5 for vector h."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def proj_box_sum(v, target, tol=1e-12, max_iter=80):
    """Project v onto [0,1]^n while forcing sum(v)=target via bisection on a shift."""
    lo, hi = -10.0, 10.0
    for _ in range(max_iter):
        tau = (lo + hi) * 0.5
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
        if hi - lo < tol:
            break
    tau = (lo + hi) * 0.5
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-12)[0]
        else:
            idx = np.where(x > 1e-12)[0]
        if idx.size:
            i = idx[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def gradient_shift(h, s):
    n = h.size
    g = np.zeros(n)
    if s >= 0:
        g[: n - s] += (1 - h[s:])
        g[s:] -= h[: n - s]
    else:
        s = -s
        g[s:] += (1 - h[: n - s])
        g[: n - s] -= h[s:]
    return g

    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]


def greedy_optimize(h, max_steps=800):
    """Deterministic transfer move based on current worst shift."""
    for _ in range(max_steps):
        s, _, cur = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:

        max_eps = min(h[i], 1.0 - h[j])

def random_move(h, rng, scale=0.02):

    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])

    nh[j] += eps

# ---------- smooth objective ----------
def smooth_obj_grad(h, beta, dx):
    """Log‑sum‑exp smoothed max and its gradient."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = np.max(corr)
    exp_terms = np.exp(beta * (corr - m))
    sum_exp = np.sum(exp_terms)
    w = exp_terms / sum_exp                     # length 2n‑1
    w_rev = w[::-1]

    conv_rev = np.convolve(1.0 - h, w_rev, mode='full')
    conv = np.convolve(h, w, mode='full')
    off = n - 1
    grad_corr = conv_rev[off:off + n] - conv[off:off + n]
    grad = dx * grad_corr
    obj = (m + np.log(sum_exp)) / beta
    return obj, grad


def projected_gradient_descent(h0, beta, target, dx, max_iter, time_limit):
    """Simple back‑tracking PGD on the smoothed objective."""
    h = h0.copy()
    obj, grad = smooth_obj_grad(h, beta, dx)

    for _ in range(max_iter):
        if time.time() > time_limit:

            obj_new, _ = smooth_obj_grad(h_new, beta, dx)

            h = h_new
            obj = obj_new
            obj, grad = smooth_obj_grad(h, beta, dx)
            step = min(step * 1.2, 1.0)

            break

    betas = [10, 50, 200, 1000, 5000, 20000]
    for beta in betas:
        if time.time() > time_limit:
            break
        # PGD
        h, _ = projected_gradient_descent(h, beta, target_sum, dx,
                                          max_iter=300, time_limit=time_limit)
        h = proj_box_sum(h, target_sum)
        # extra greedy polishing for high beta
        if beta >= 1000:
            h = greedy_optimize(h, max_steps=1500)
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_h, best_c = h.copy(), cur_c

    # final tiny stochastic tweaks
    while time.time() < time_limit:
        cand = random_move(best_h, rng, scale=0.02)
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_h, best_c = cand, cand_c

    # final clean‑up
    best_h = proj_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
