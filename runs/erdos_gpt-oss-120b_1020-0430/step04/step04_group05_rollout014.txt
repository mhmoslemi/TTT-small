```python
def compute_c5_fft(h):
    n = h.size
    # Compute convolution via FFT: conv = ifft(fft(h) * fft(flip(1-h)))

    max_corr = np.max(conv)
```

```python
def targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5):
    n = h.size
    if lag >= 0:
        valid = np.arange(n - lag)
        contrib = h[valid] * (1 - h[valid + lag])
        i_candidates = valid
        j_candidates = valid + lag  # Actually we want to move from i to j (different)
    else:
        lag = -lag
        valid = np.arange(lag, n)
        # contributions for negative lag: h[valid] * (1 - h[valid - lag])
        contrib = h[valid] * (1 - h[valid - lag])
        i_candidates = valid
        # j candidates are valid - lag
    # select top fraction
    k = max(1, int(len(i_candidates) * top_frac))
    top_idx = i_candidates[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = i_candidates[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    # choose j from bottom indices (or random)
    if rng.random() < 0.8:
        j = rng.choice(bot_idx)
    else:
        j = rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # fallback to simple perturb
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def simulated_annealing(h, rng, max_iters, start_time, time_limit):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    n = h.size

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # compute temperature schedule
        # We can use T = T0 * (Tf/T0)**(it / max_iters)
        # set T0 = 0.05, Tf = 1e-5
        T = 0.05 * (1e-5 / 0.05) ** (it / max_iters)
        # find worst lag
        lag, _, _ = max_lag_and_corr(cur_h)
        # pick perturb type
        if rng.random() < 0.7:
            new_h = targeted_perturb(cur_h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(cur_h, rng, scale=0.5)
        new_c = compute_c5(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = correlation_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n

    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,

            diff = CURRENT_TARGET - x.sum()

            new_h = targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c = compute_c5(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c


# ---------- SLSQP with softmax ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = correlation_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


# ---------- stage runner ----------
def run_stage(h, n_target, budget, rng):
    if h.size != n_target:
        x_old = (np.arange(h.size) + 0.5) * (2.0 / h.size)
        x_new = (np.arange(n_target) + 0.5) * (2.0 / n_target)
        h = np.interp(x_new, x_old, h)
        h = enforce_sum(h, n_target / 2.0, rng)
    start = time.time()
    deadline = start + budget
    max_iters = int(20000 * (720.0 / n_target))
    if max_iters < 5000:
        max_iters = 5000
    h_sa, c_sa = simulated_annealing(h, rng, max_iters, start, deadline)
    best_h, best_c = h_sa, c_sa
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    for beta in beta_list:
        if time.time() > deadline:
            break
        try:
            ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=300)
        except Exception:
            continue
        ref_c = compute_c5(ref_h)
        if ref_c < best_c:
            best_h, best_c = ref_h, ref_c
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # --- initial vector ---
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    best_h, best_c = h0, compute_c5(h0)

    total_budget = budget_s
    remaining = total_budget - 10  # keep a few seconds for final clean‑up
    stage0_budget = max(5.0, remaining * 0.20)
    stage1_budget = max(5.0, remaining * 0.35)
    stage2_budget = max(5.0, remaining - stage0_budget - stage1_budget)

    # --- stage 0 (360) ---
    h_opt, c_opt = run_stage(best_h, 360, stage0_budget, rng)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt

    # --- stage 1 (720) ---
    if time.time() - t0 < total_budget - 5:
        h_opt, c_opt = run_stage(best_h, 720, stage1_budget, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # --- stage 2 (1440) ---
    if time.time() - t0 < total_budget - 5:
        h_opt, c_opt = run_stage(best_h, 1440, stage2_budget, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # --- final quick hill‑climb ---
    cur_h, cur_c = best_h.copy(), best_c
    step = 0.3
    for _ in range(5000):
        if time.time() - t0 > total_budget:
            break
        new_h = perturb_h(cur_h, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        step *= 0.9995

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
