def targeted_perturb_worst_lag(h, rng, max_scale=0.5):
    n = h.size
    # compute full correlation to get worst lag
    _, corr = compute_corr_and_max(h)
    # find index for max
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    # valid start indices i
    if lag >= 0:
        valid_i = np.arange(n - lag)
        partner = valid_i + lag
    else:
        lag_abs = -lag
        valid_i = np.arange(lag_abs, n)
        partner = valid_i - lag_abs
    # compute contributions
    contrib = h[valid_i] * (1 - h[partner])
    # choose i with max contribution
    i = valid_i[np.argmax(contrib)]
    j = partner[np.argmax(contrib)]  # partner index
    # we will transfer epsilon from i to j (or maybe from j to i depending)
    # If we want to increase h[j] (currently small maybe), transfer from i
    # Ensure allowed: max_transfer = min(h[i], 1 - h[j])
    max_transfer = min(h[i], 1 - h[j])
    if max_transfer <= 0:
        # fallback to generic perturb
        return perturb_h(h, rng, scale=max_scale)
    eps = rng.random() * max_transfer * max_scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing_fast(start_h, rng, time_limit, max_iters=1e9, init_temp=0.05, final_temp=1e-5):

    h = start_h.copy()
    cur_c, _ = compute_corr_and_max(h)
    cur_c *= 2.0 / n
    best_h = h.copy()
    best_c = cur_c
    # precompute scaling schedule
    T0 = init_temp
    Tf = final_temp
    it = 0
    while time.time() < time_limit and it < max_iters:

                best_h = new_h.copy()

def enforce_sum(h, target, rng, max_iter=10000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # If diff small, fine.
    if abs(diff) < 1e-12:
        return h
    # Use simple approach: Add diff/n to all entries, then clip, repeat.
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Distribute diff across entries uniformly but limited by bounds
        add = diff / n
        # Choose entries that can absorb addition without violating bounds
        if add > 0:
            # find indices where h < 1
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def compute_corr_full(h):
    """Full linear correlation of h with (1-h) using FFT (no scaling)."""
    n = h.size
    N = 2 * n
    fft_h = np.fft.rfft(h, n=N)
    fft_g = np.fft.rfft(1.0 - h, n=N)
    prod = fft_h * np.conj(fft_g)
    corr = np.fft.irfft(prod, n=N)[: 2 * n - 1]
    return corr

def compute_corr_and_max(h):
    """Return max correlation value and the full correlation array."""
    corr = compute_corr_full(h)
    return np.max(corr), corr

def compute_c5_fft(h):
    """C5 value for step function represented by h (scaled by dx)."""
    n = h.size
    max_corr = np.max(compute_corr_full(h))
    return max_corr * (2.0 / n)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb_worst_lag(h, rng, max_scale=0.5):
    """Bias transfer toward indices that dominate the worst lag."""
    n = h.size
    max_val, corr = compute_corr_and_max(h)
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)
    if lag >= 0:
        start_idx = np.arange(n - lag)
        partner_idx = start_idx + lag
    else:
        lag_abs = -lag
        start_idx = np.arange(lag_abs, n)
        partner_idx = start_idx - lag_abs
    contrib = h[start_idx] * (1.0 - h[partner_idx])
    if contrib.size == 0:
        return perturb_h(h, rng, scale=max_scale)
    ix = np.argmax(contrib)
    i = start_idx[ix]
    j = partner_idx[ix]
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale=max_scale)
    eps = rng.random() * max_eps * max_scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ---------- SLSQP with soft‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = compute_corr_full(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- simulated annealing ----------
def simulated_annealing_fast(start_h, rng, time_limit):
    n = start_h.size
    h = start_h.copy()
    cur_c = compute_c5_fft(h)
    best_h, best_c = h.copy(), cur_c
    it = 0
    # parameters
    T0, Tf = 0.04, 1e-5
    max_iters = int(1e9)  # effectively unlimited, stopped by time
    while time.time() < time_limit and it < max_iters:
        # cooling schedule
        T = T0 * (Tf / T0) ** (it / (max_iters if max_iters else 1))
        if rng.random() < 0.6:
            new_h = targeted_perturb_worst_lag(h, rng, max_scale=0.6)
        else:
            new_h = perturb_h(h, rng, scale=0.6)
        new_c = compute_c5_fft(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / (T + 1e-12)):
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    total = budget_s

    # ----- choose grid size -----
    n_target = 720  # reasonably fine while staying fast

    # ----- initialise h -----
    if "initial_h_values" in globals():

    else:

    h0 = enforce_sum(h0, n_target / 2.0, rng)
    best_h, best_c = h0.copy(), compute_c5_fft(h0)

    # ----- main stochastic search (≈85% of budget) -----
    main_until = t0 + total * 0.85
    while time.time() < main_until:
        # restart from a perturbed best or a fresh random vector
        if rng.random() < 0.8:
            start = perturb_h(best_h, rng, scale=0.8)
        else:
            start = random_feasible_vector(n_target, rng)
        # run SA until main_until
        h_sa, c_sa = simulated_annealing_fast(start, rng, main_until)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa
        # occasional SLSQP refinement

            for beta in [10, 50, 200, 800, 3000, 8000]:

                except Exception:
                    continue
                c_ref = compute_c5_fft(h_ref)
                if c_ref < best_c:

    # ----- final local hill‑climb (≈10% of budget) -----
    hill_until = t0 + total * 0.95
    cur_h, cur_c = best_h.copy(), best_c
    step = 0.6
    while time.time() < hill_until:
        new_h = perturb_h(cur_h, rng, scale=step)

        if new_c < cur_c:

            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        step *= 0.9999

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
