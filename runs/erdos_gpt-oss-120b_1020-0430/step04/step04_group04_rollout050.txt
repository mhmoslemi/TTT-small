def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # scale to have proper sum: linear scaling plus clipping
    total = h.sum()
    if total == 0:
        h[:] = target / n
    else:
        # scale
        h = h * (target / total)
    h = enforce_sum(h, target, rng)
    return h

def perturb_h(h, rng, scale=0.5):
    n = h.size
    # choose i (source) where h[i] > 0
    # choose j (dest) where h[j] < 1
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    # Determine max possible epsilon to move from i to j:
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # Due to floating point, sum unchanged, but small drift possible
    return new_h

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp, init_scale=0.5, final_scale=0.01):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x, beta), h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # make sure sum constraint satisfied exactly:
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        # simple correction distribute diff uniformly
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        # recompute diff
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            # distribute via free indices
            free = (x > 1e-12) & (x < 1 - 1e-12)
            idx = np.where(free)[0]
            if idx.size:
                per = diff / idx.size
                x[idx] += per
                x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------------------------------------------------------
# Helper functions
# ---------------------------------------------------------

def compute_c5(h):
    """Exact C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng, tol=1e-12, max_iter=5000):
    """Adjust h so that sum(h) == target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    it = 0
    while abs(diff) > tol and it < max_iter:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size == 0:
            per = diff / n
            h += per
        else:
            per = diff / idx.size
            h[idx] += per
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        it += 1
    # fallback deterministic correction
    if abs(diff) > tol:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = enforce_sum(h, target, rng)
    return h


def random_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    h[:half] = 1.0
    rng.shuffle(h)
    return h


def perturb_h(h, rng, scale=0.5):
    """Transfer a random amount ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h0, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Stochastic hill‑climbing."""
    h = h0.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            nh = perturb_h(h, rng, scale=step)
            c = compute_c5(nh)
            if c + 1e-12 < best_c:
                h, best_c = nh, c
                improved = True
                break
        if not improved and rng.random() < 0.02:
            nh = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(nh)
            if c + 1e-12 < best_c:
                h, best_c = nh, c
                improved = True
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.01):
    """Simulated‑annealing refinement."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        nh = perturb_h(cur_h, rng, scale=scale)
        nc = compute_c5(nh)
        delta = nc - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = nh, nc
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c


def softmax_obj(x, beta):
    """Smooth approximation of max via log‑sum‑exp."""
    n = x.size
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    maxc = np.max(conv)
    shifted = conv - maxc
    lse = maxc + np.log(np.sum(np.exp(beta * shifted))) / beta
    return lse


def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP refinement on the smooth surrogate."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x, beta), h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            free = (x > 1e-12) & (x < 1 - 1e-12)
            idx = np.where(free)[0]
            if idx.size:
                per = diff / idx.size
                x[idx] += per
                x = np.clip(x, 0.0, 1.0)
    return x
