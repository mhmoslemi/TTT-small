def project_onto_bounds_and_sum(v, target, max_iter=60):
    n = v.shape[0]
    # Clip initial maybe to bounds, but we will incorporate via lambda.
    lo = -np.max(v) - 1.0   # ensure v + lo <= 0
    hi = 1.0 - np.min(v) + 1.0  # ensure v + hi >= 1
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi)/2.0, 0.0, 1.0)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----- global vars for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # smooth max of the correlation vector
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

# ----- core utilities -----
def compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)

def worst_shift_and_corr(h):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (h.shape[0] - 1)
    return s, corr[idx]

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (h.shape[0] - 1)
    return s, idx, corr[idx]

def gradient_of_corr_shift(h, s):
    """∇_h Corr_s where Corr_s = Σ h[i]·(1-h[i+s]) (zero outside)."""
    n = h.shape[0]
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += 1.0 - h[s:]
        grad[s:] -= h[: n - s]
    else:
        t = -s
        grad[t:] += 1.0 - h[: n - t]
        grad[: n - t] -= h[t:]
    return grad

def project_onto_bounds_and_sum(v, target, max_iter=60):
    """Find x = clip(v+λ,0,1) with Σx = target (binary search on λ)."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            delta = residual / free.size
            x[free] += delta
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.argmax(x)
            x[idx] = np.clip(x[idx] + residual, 0.0, 1.0)
    return x

def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_onto_bounds_and_sum(v, n / 2.0)

def perturb_random(h, rng, scale=0.2):
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def subgradient_optimize(start, max_iters, step0, decay, time_limit, start_time):
    h = start.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        s, corr_val = worst_shift_and_corr(h)
        grad = gradient_of_corr_shift(h, s) * (2.0 / n)
        h_new = h - step * grad
        h_new = project_onto_bounds_and_sum(h_new, target)
        c_new = compute_c5(h_new)
        if c_new < best_c:

        h = h_new
        step *= decay
    return best_h, best_c
