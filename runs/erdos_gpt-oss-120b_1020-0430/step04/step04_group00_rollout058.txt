Implementation:

```
def compute_O(h):
    n = len(h)
    O = np.zeros(n, dtype=int)  # O[0] for shift 0, but we rarely use.
    for k in range(1, n):
        O[k] = np.dot(h[:n-k], h[k:])
    return O
```

Now for swap:

```
def delta_O_for_swap(i, j, h):
    n = len(h)
    delta = np.zeros(n, dtype=int)
    # i originally 1, become 0
    # contributions from i with higher index
    for k in range(1, n - i):
        delta[k] -= h[i + k]
    # contributions from i with lower index
    for k in range(1, i + 1):
        delta[k] -= h[i - k]
    # j originally 0, become 1
    for k in range(1, n - j):
        delta[k] += h[j + k]
    for k in range(1, j + 1):
        delta[k] += h[j - k]
    return delta

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def compute_c5(h):
    """Return the overlap constant for a step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(x, rng, max_iter=2000, eps=1e-12):
    """Exact projection onto the hyper‑plane sum(x)=target with box bounds."""
    target = x.size / 2.0
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        i = rng.integers(0, x.size)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x


# ----------------------------------------------------------------------
# Gradient refinement (smooth max surrogate already handled by SLSQP)
# ----------------------------------------------------------------------
def gradient_refine_full(h, rng, step=0.02, max_iter=30):
    """Projected gradient descent targeting the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        shift = np.argmax(corr) - (n - 1)
        k = abs(shift)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:up]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            adjust_sum_exact(h, rng)
    return h


# ----------------------------------------------------------------------
# SLSQP using a smooth log‑sum‑exp surrogate for the max
# ----------------------------------------------------------------------
def softmax_obj(x, beta):
    """Log‑sum‑exp surrogate of max_k corr_k / beta."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def sum_eq_constraint(x, target):
    return np.sum(x) - target


def slsqp_optimize(h0, beta, rng, max_iter=500):
    """SLSQP on the smooth surrogate."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint, "args": (target,)}
    res = minimize(
        softmax_obj,
        h0,
        args=(beta,),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x


# ----------------------------------------------------------------------
# Differential evolution on a low‑dimensional representation
# ----------------------------------------------------------------------
def de_and_refine(n, rng, time_deadline):
    """DE on 30‑segment representation, followed by local refinements."""
    m = 30
    block = n // m
    target_s = m / 2.0                     # sum of segment values
    bounds = [(0.0, 1.0)] * m

    def obj(seg):
        # penalty for violating the sum constraint
        pen = 0.0
        s = seg.sum()
        if abs(s - target_s) > 1e-6:
            pen = 10.0 * abs(s - target_s)
        h_full = np.repeat(seg, block)
        return compute_c5(h_full) + pen

    # allocate iterations proportionally to remaining time
    remaining = time_deadline - time.time()
    # rough estimate: 90 s per 200 generations for this size
    maxgen = max(20, int(remaining // 0.9))
    popsize = 12
    result = differential_evolution(
        obj,
        bounds,
        maxiter=maxgen,
        popsize=popsize,
        polish=False,
        seed=int(rng.integers(1 << 30)),
        updating="deferred",
        disp=False,
    )
    seg_opt = result.x
    adjust_sum_exact(seg_opt, rng)
    h = np.repeat(seg_opt, block)

    # local refinements
    h = gradient_refine_full(h, rng, step=0.02, max_iter=12)
    for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
        if time.time() > time_deadline:
            break
        h = slsqp_optimize(h, beta, rng, max_iter=300)
    h = gradient_refine_full(h, rng, step=0.02, max_iter=12)
    final_c = compute_c5(h)
    return h, final_c
