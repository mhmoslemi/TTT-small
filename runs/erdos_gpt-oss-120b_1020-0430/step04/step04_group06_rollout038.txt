def compute_c5_fft(h):
    n = h.size
    N = 2*n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)

    # The maximum correlation occurs somewhere possibly at shift - (n-1) to n-1.
    # We'll simply take max over all values.
    max_val = np.max(corr)

import time
import numpy as np

# -------------------------------------------------
# Helper utilities
# -------------------------------------------------
def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 5000) -> np.ndarray:
    """Adjust h so that sum(h) == target while keeping 0 <= h <= 1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final safety correction
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5_fft(h: np.ndarray) -> float:
    """Exact C5 using FFT (full correlation of h with 1‑h)."""
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    max_val = np.max(corr.real)
    return max_val * (2.0 / n)


def compute_dot(h: np.ndarray) -> np.ndarray:
    """Autocorrelation dot_k = Σ_i h_i h_{i+k} for k=0..n‑1 (linear, zero‑padded)."""
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    corr = np.fft.irfft(H * np.conj(H), n=N)
    return corr[:n].real  # length n


# -------------------------------------------------
# Simulated annealing with fast incremental updates
# -------------------------------------------------
def simulated_annealing(
    start_h: np.ndarray,
    start_dot: np.ndarray,
    target_sum: float,
    rng: np.random.Generator,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """Run SA for the given time_limit and return the best (h, c5) found."""
    n = start_h.size
    best_h = start_h.copy()
    best_dot = start_dot.copy()
    best_c = (target_sum - best_dot.min()) * 2.0 / n

    cur_h = start_h.copy()
    cur_dot = start_dot.copy()
    cur_c = best_c
    start_time = time.time()

    while time.time() - start_time < time_limit:
        # ---- propose a move -------------------------------------------------
        if rng.random() < 0.3:  # binary‑swap style
            # pick a 1 and a 0
            ones = np.where(cur_h > 0.999)[0]
            zeros = np.where(cur_h < 0.001)[0]
            if ones.size == 0 or zeros.size == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            eps = 1.0
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps < 1e-12:
                continue
            eps = max_eps * rng.random()

        # ---- incremental dot update -----------------------------------------
        delta = np.zeros(n, dtype=float)

        # contribution of i (decrease)
        delta[: n - i] -= eps * cur_h[i:]
        delta[: i + 1] -= eps * cur_h[i::-1]

        # contribution of j (increase)
        delta[: n - j] += eps * cur_h[j:]
        delta[: j + 1] += eps * cur_h[j::-1]

        new_dot = cur_dot + delta
        new_c = (target_sum - new_dot.min()) * 2.0 / n

        # ---- temperature schedule --------------------------------------------
        elapsed = time.time() - start_time
        t_frac = elapsed / time_limit
        T0, Tf = 5e-3, 1e-7
        T = T0 * (Tf / T0) ** t_frac

        # ---- accept / reject --------------------------------------------------
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / T):
            # accept
            cur_h[i] -= eps
            cur_h[j] += eps
            cur_dot = new_dot
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
                best_dot = cur_dot.copy()
    return best_h, best_c
