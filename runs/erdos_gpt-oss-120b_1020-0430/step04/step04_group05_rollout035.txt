def get_contributions(h, lag):
    n = h.size
    if lag >= 0:
        valid = np.arange(n - lag)
        contrib = h[valid] * (1.0 - h[valid + lag])
    else:
        lag = -lag
        valid = np.arange(lag, n)
        contrib = h[valid] * (1.0 - h[valid - lag])
    return valid, contrib

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    valid, contrib = get_contributions(h, lag)
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    # Indices with largest contributions, i.e., worst.
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    # Indices with smallest contributions (good)
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    # Choose i from top, j from bot
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.7 else rng.integers(0, n)
    # Ensure i != j
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def softmax_obj_grad(x, beta, lam, target):
    n = x.size
    corr = np.correlate(x, 1 - x, mode='full')
    bc = beta * corr
    max_bc = np.max(bc)
    exp_terms = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_terms)
    w = exp_terms / sum_exp  # weights
    # gradient of softmax part
    grad = np.zeros_like(x)
    # Loop over lag indices
    # Index in corr array: idx from 0 to 2*n-2, corresponding lag = idx - (n-1)
    for idx in range(2*n - 1):
        d = idx - (n - 1)
        w_d = w[idx]
        if d >= 0:
            # range 0 to n-1-d
            i_end = n - d
            if i_end > 0:
                grad[:i_end] += w_d * (1 - x[d:])
                grad[d:] -= w_d * x[:i_end]
        else:
            d_pos = -d
            i_start = d_pos
            i_len = n - d_pos
            if i_len > 0:
                grad[i_start:] += w_d * (1 - x[:i_len])
                grad[:i_len] -= w_d * x[i_start:]
    # softmax objective value:
    obj = (max_bc + np.log(sum_exp)) / beta
    # penalty
    diff = np.sum(x) - target
    obj += lam * diff * diff
    grad += 2.0 * lam * diff * np.ones_like(x)
    return obj, grad

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def enforce_sum(h, target, rng, max_iter=5000):
    """Clip h to [0,1] and adjust to have sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:                     # fallback deterministic correction
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def compute_c5_corr(h):
    """Return C5 and full correlation array."""
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * (2.0 / h.size)
    return c5, corr

def compute_c5(h):
    c, _ = compute_c5_corr(h)
    return c

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def get_contributions(h, lag):
    """Contribution h[i]*(1-h[i+lag]) for the lag (positive or negative)."""
    n = h.size
    if lag >= 0:
        valid = np.arange(n - lag)
        contrib = h[valid] * (1.0 - h[valid + lag])
    else:
        lag = -lag
        valid = np.arange(lag, n)
        contrib = h[valid] * (1.0 - h[valid - lag])
    return valid, contrib

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    """Move mass from indices with large contribution to those with small."""
    n = h.size
    valid, contrib = get_contributions(h, lag)
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def softmax_obj_grad(x, beta, lam, target):
    """Log‑sum‑exp surrogate of max correlation + quadratic penalty."""
    n = x.size
    corr = np.correlate(x, 1 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    exp_terms = np.exp(bc - m)
    sum_exp = np.sum(exp_terms)
    w = exp_terms / sum_exp
    grad = np.zeros_like(x)
    # gradient of each lag contribution
    for idx in range(2 * n - 1):
        d = idx - (n - 1)
        wd = w[idx]
        if d >= 0:
            i_end = n - d
            if i_end > 0:
                grad[:i_end] += wd * (1 - x[d:])
                grad[d:]     -= wd * x[:i_end]
        else:
            dpos = -d
            i_len = n - dpos
            if i_len > 0:
                grad[dpos:]  += wd * (1 - x[:i_len])
                grad[:i_len] -= wd * x[dpos:]
    obj = (m + np.log(sum_exp)) / beta
    diff = np.sum(x) - target
    obj += lam * diff * diff
    grad += 2.0 * lam * diff * np.ones_like(x)
    return obj, grad

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    # ---- initial vector (n=360) ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = random_feasible_vector(360, rng)
    # ---- upscale to n=720 ----
    target_n = 720
    if h0.size != target_n:
        old_x = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h = np.interp(new_x, old_x, h0)
    else:
        h = h0.copy()
    h = enforce_sum(h, target_n / 2.0, rng)
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c, _ = compute_c5_corr(best_h)

    # ---- simulated annealing (80% of budget) ----
    sa_budget = budget_s * 0.78
    n = target_n
    cur_h = best_h.copy()
    cur_c, cur_corr = compute_c5_corr(cur_h)
    T0, Tf = 0.1, 1e-5
    while time.time() - t0 < sa_budget:
        # occasional random restart
        if rng.random() < 0.001:
            cur_h = random_feasible_vector(n, rng)
            cur_c, cur_corr = compute_c5_corr(cur_h)

        elapsed = time.time() - t0
        t_frac = (elapsed) / sa_budget
        T = T0 * (Tf / T0) ** t_frac
        worst_idx = np.argmax(cur_corr)
        lag = worst_idx - (n - 1)

        scale = max(0.5 * (1 - t_frac), 0.01)
        if rng.random() < 0.7:
            new_h = targeted_perturb(cur_h, rng, lag, top_frac=0.2, scale=scale)
        else:
            new_h = perturb_h(cur_h, rng, scale=scale)

        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            cur_h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()

    # ---- L‑BFGS‑B refinement (remaining budget) ----
    ref_deadline = t0 + budget_s
    target_sum = n / 2.0
    lam = 1000.0
    for beta in (100, 500, 2000, 8000):
        if time.time() >= ref_deadline:
            break
        res = minimize(
            fun=lambda x, beta=beta: softmax_obj_grad(x, beta, lam, target_sum),
            x0=best_h,
            method="L-BFGS-B",
            jac=True,
            bounds=[(0.0, 1.0)] * n,
            options={"maxiter": 300, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum(x, target_sum, rng)
        c = compute_c5(x)
        if c < best_c:
            best_c, best_h = c, x.copy()

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
