def project_onto_capped_simplex(v, target, eps=1e-12):
    n = v.shape[0]
    # Clip initial: but we will handle within projection.
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(100):
        tau = (lo + hi) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) <= eps:
            return h
        if s > target:
            # need larger tau to reduce sum
            lo = tau
        else:
            hi = tau
    # after loop
    h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # adjust small diff
    diff = target - h.sum()
    if abs(diff) > eps:
        # distribute diff to non-saturated entries
        idx = np.where((h > 0 + eps) & (h < 1 - eps))[0]
        if idx.size > 0:
            # add equally
            add = diff / idx.size
            h[idx] += add
            h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_capped_simplex(v, target, eps=1e-12, max_iter=100):
    # Project onto {0 <= x_i <= 1, sum x_i = target}
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) <= eps:
            return h
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # final small adjustment if needed
    diff = target - h.sum()
    if abs(diff) > eps:
        idx = np.where((h > eps) & (h < 1 - eps))[0]
        if idx.size > 0:
            add = diff / idx.size
            h[idx] += add
            h = np.clip(h, 0.0, 1.0)
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    # ensure i is source (decrease), j is destination (increase)
    # we may swap if not enough capacity.
    # Determine max epsilon that respects bounds.
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def gradient_descent_softmax(h0, beta, max_iter=200, step0=1.0, target=None):
    if target is None:
        target = h0.size / 2.0
    h = h0.copy()
    n = h.size
    for it in range(max_iter):
        # correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        # softmax weights
        # subtract max_corr for numerical stability
        exp_beta = np.exp(beta * (corr - max_corr))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp
        # gradient (same shape as h)
        grad = np.zeros_like(h)
        # accumulate weighted gradients
        # using loop over shifts
        # Note offset index: offset = s + (n-1)
        for offset, weight in enumerate(w):
            if weight == 0.0:
                continue
            s = offset - (n - 1)
            if s >= 0:
                grad[:n - s] += weight * (1.0 - h[s:])
                grad[s:] -= weight * h[:n - s]
            else:
                s_abs = -s
                grad[s_abs:] += weight * (1.0 - h[:n - s_abs])
                grad[:n - s_abs] -= weight * h[s_abs:]
        # objective value
        f = max_corr + np.log(sum_exp) / beta
        # line search
        step = step0
        for _ in range(20):
            h_new = h - step * grad
            h_new = project_onto_capped_simplex(h_new, target)
            # compute new objective
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            max_corr_new = np.max(corr_new)
            exp_beta_new = np.exp(beta * (corr_new - max_corr_new))
            sum_exp_new = exp_beta_new.sum()
            f_new = max_corr_new + np.log(sum_exp_new) / beta
            if f_new < f - 1e-12:
                h = h_new
                break
            step *= 0.5
        # optional early stop if gradient small
        if np.linalg.norm(grad) * step < 1e-10:
            break
    return h

import time

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 from vector h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_capped_simplex(v: np.ndarray, target: float, eps: float = 1e-12, max_iter: int = 50) -> np.ndarray:
    """Projection onto {0<=x_i<=1, sum x_i = target}."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(max_iter):
        tau = (lo + hi) * 0.5
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) <= eps:
            return h
        if s > target:
            lo = tau
        else:
            hi = tau
    # final correction for tiny residual
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > eps:
        idx = np.where((h > eps) & (h < 1.0 - eps))[0]
        if idx.size:
            h[idx] += diff / idx.size
            h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in the capped simplex."""
    v = rng.random(n)
    return project_onto_capped_simplex(v, n / 2.0)


def perturb_random(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Move a random amount of mass from one coordinate to another."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ---------- gradient helpers ----------
def gradient_shift(h: np.ndarray, s: int) -> np.ndarray:
    """Gradient of a single correlation term with shift s."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1.0 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1.0 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def worst_shift_info(h: np.ndarray):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]


def greedy_optimize(h: np.ndarray, max_steps: int = 2000) -> np.ndarray:
    """Deterministic improvement by full‑epsilon transfers along the worst shift."""
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full epsilon
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half epsilon
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h


# ---------- soft‑max gradient descent ----------
def gradient_descent_softmax(
    h0: np.ndarray,
    beta: float,
    max_iter: int = 200,
    step0: float = 1.0,
    target: float = None,
) -> np.ndarray:
    """Projected gradient descent on the smooth max (soft‑max) objective."""
    h = h0.copy()
    n = h.size
    if target is None:
        target = n / 2.0
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        # soft‑max weights
        exp_beta = np.exp(beta * (corr - max_corr))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp

        # gradient = sum_s w_s * grad_shift(s)
        grad = np.zeros_like(h)
        for offset, weight in enumerate(w):
            if weight == 0.0:
                continue
            s = offset - (n - 1)
            if s >= 0:
                grad[: n - s] += weight * (1.0 - h[s:])
                grad[s:] -= weight * h[: n - s]
            else:
                s_abs = -s
                grad[s_abs:] += weight * (1.0 - h[: n - s_abs])
                grad[: n - s_abs] -= weight * h[s_abs:]

        # current smooth objective value
        f = max_corr + np.log(sum_exp) / beta

        # back‑tracking line search with projection
        step = step0
        for _ in range(20):
            h_new = h - step * grad
            h_new = project_onto_capped_simplex(h_new, target)
            corr_new = np.correlate(h_new, 1 - h_new, mode="full")
            max_corr_new = np.max(corr_new)
            exp_beta_new = np.exp(beta * (corr_new - max_corr_new))
            sum_exp_new = exp_beta_new.sum()
            f_new = max_corr_new + np.log(sum_exp_new) / beta
            if f_new < f - 1e-12:
                h = h_new
                break
            step *= 0.5

        # stop if step is tiny
        if np.linalg.norm(grad) * step < 1e-10:
            break
    return h


def gradient_descent_sequence(
    h0: np.ndarray, betas, iters_per_beta, target: float
) -> np.ndarray:
    """Run a cascade of soft‑max descents with increasing beta."""
    h = h0.copy()
    for beta, iters in zip(betas, iters_per_beta):
        h = gradient_descent_softmax(h, beta, max_iter=iters, step0=1.0, target=target)
    return h


# ---------- main entry ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial vector -----
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h_initial = np.array(initial_h_values, dtype=float)
        h_initial = np.clip(h_initial, 0.0, 1.0)
        target = n / 2.0
        h_initial = project_onto_capped_simplex(h_initial, target)
    else:
        n = 720
        target = n / 2.0
        h_initial = random_feasible_vector(n, rng)

    best_h = h_initial.copy()
    best_c = compute_c5(best_h)

    # ----- multi‑start optimisation -----
    betas = [10, 50, 200, 800, 2000, 8000]
    iters_per_beta = [60, 60, 80, 80, 80, 80]

    while time.time() - start_time < budget_s * 0.85:
        # choose start point
        if best_h is not None and rng.random() < 0.6:
            start = perturb_random(best_h, rng, scale=0.6)
        else:
            start = random_feasible_vector(n, rng)

        # smooth‑max descent cascade
        h = gradient_descent_sequence(start, betas, iters_per_beta, target)

        # deterministic greedy polishing
        h = greedy_optimize(h, max_steps=1500)

        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h, c

        # occasional high‑beta polishing
        if rng.random() < 0.2 and time.time() - start_time < budget_s * 0.90:
            h_ref = gradient_descent_sequence(
                best_h, [20000, 50000], [80, 80], target
            )
            h_ref = greedy_optimize(h_ref, max_steps=800)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref

    # ----- fine‑grained random search -----
    while time.time() - start_time < budget_s * 0.99:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_capped_simplex(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
