def full_pair_transfer_optimize(h, rng, time_limit):
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / h.size)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # compute overlapping indices
        if shift >= 0:
            i_range = np.arange(h.size - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, h.size)
            j_range = i_range + shift
        if i_range.size == 0:
            # shift = +/- n, but not possible.
            break
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        # compute max epsilon to transfer
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            # if cannot transfer, try a random fill
            # pick random i where h[i] > 1e-12, pick random j where h[j] < 1-1e-12
            idx1 = rng.choice(np.where(h > 1e-12)[0])
            idx2 = rng.choice(np.where(h < 1-1e-12)[0])
            max_eps = min(h[idx1], 1.0 - h[idx2])
            i, j = idx1, idx2
        # Transfer full epsilon
        eps = max_eps
        h[i] -= eps
        h[j] += eps
        # clamp
        # Actually after transfer both within [0,1], no need clamp.

    # final check
    final_c = compute_c5(best_h)
    return best_h, final_c

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def max_overlap_and_shift(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (h.size - 1)
    return float(corr[idx]), shift

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 10):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    return enforce_sum(h, n/2.0, rng)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_multi(h_start, rng, max_iters=3000, init_step=0.5, n_trials=7):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.03:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def slsqp_optimize(h0, beta, max_iter=800):
    n = h0.size
    target = n / 2.0
    # smooth objective
    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    def sum_eq(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def full_pair_transfer_optimize(h_start, rng, time_limit):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    n = h.size
    while time.time() - start < time_limit:
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # determine overlapping indices
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            # fallback random move
            pos = np.where(h > 1e-12)[0]
            neg = np.where(h < 1 - 1e-12)[0]
            if pos.size == 0 or neg.size == 0:
                break
            i = int(rng.choice(pos))
            j = int(rng.choice(neg))
            max_eps = min(h[i], 1.0 - h[j])
        # transfer full possible amount
        h[i] -= max_eps
        h[j] += max_eps
    final_c = compute_c5(best_h)
    return best_h, final_c

def simulated_annealing_optimize(h_start, rng, time_limit):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # temperature schedule: start 1e-3, halve each 5000 iters
        T = 1e-3 * (0.5 ** (it // 5000))
        # select random i from >0, j from <1
        pos = np.where(h > 1e-12)[0]
        neg = np.where(h < 1 - 1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = int(rng.choice(pos))
        j = int(rng.choice(neg))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * rng.random()  # random fraction
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # choose n
    n = 500
    target = n / 2.0
    # build candidate pool
    candidates = []
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    # linear ramp (xs/2)
    candidates.append(enforce_sum(xs / 2.0, target, rng))
    # sine/cosine patterns
    for f in (1, 2, 3, 4, 5, 6):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            candidates.append(enforce_sum(h, target, rng))
    # random masks
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, target, rng))
    # random vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))
    # evaluate initial candidates
    best_h = None
    best_c = float('inf')
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_best_h = best_h
    global_best_c = best_c
    # Main loop: hill climb + SLSQP + pair transfer
    main_deadline = start_time + budget_s * 0.75
    while time.time() < main_deadline:
        # pick start
        if rng.random() < 0.7:
            start_vec = perturb_h(global_best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)
        # hill climb
        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=2500, init_step=0.5, n_trials=8)
        if c_hc < global_best_c:
            global_best_h, global_best_c = h_hc.copy(), c_hc
        # SLSQP ladder
        for beta in (10, 30, 100, 300, 800, 1500, 3000):
            if time.time() >= main_deadline:
                break
            h_ref = slsqp_optimize(global_best_h, beta, max_iter=600)
            c_ref = compute_c5(h_ref)
            if c_ref < global_best_c:
                global_best_h, global_best_c = h_ref.copy(), c_ref
    # Pair transfer phase
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10.0:
        h_opt, c_opt = full_pair_transfer_optimize(global_best_h.copy(), rng, time_limit=remaining * 0.6)
        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt
        # final simulated annealing
        remaining2 = budget_s - (time.time() - start_time)
        if remaining2 > 5.0:
            h_opt2, c_opt2 = simulated_annealing_optimize(global_best_h.copy(), rng, time_limit=remaining2 * 0.9)
            if c_opt2 < global_best_c:
                global_best_h, global_best_c = h_opt2.copy(), c_opt2
    # final cleanup
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, target, rng)
    final_c = compute_c5(global_best_h)
    return global_best_h.tolist(), float(final_c), int(n)

    cur_c = best_c
    n = h.size
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        T = 1e-3 * (0.5 ** (it // 5000))  # temperature schedule
        pos = np.where(h > 1e-12)[0]
        neg = np.where(h < 1 - 1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = int(rng.choice(pos))
        j = int(rng.choice(neg))
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random()
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    n = 500
    target = n / 2.0

    # ----- build a pool of start vectors -----
    candidates = []

    # use parent construction if present (interpolated to n points)
    if "initial_h_values" in globals():
        h0_coarse = np.array(initial_h_values, dtype=float)
        n0 = h0_coarse.size
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h_interp = np.interp(xs, xs0, h0_coarse)
        candidates.append(enforce_sum(h_interp, target, rng))

    xs = (np.arange(n) + 0.5) * (2.0 / n)

    # linear ramp
    candidates.append(enforce_sum(xs / 2.0, target, rng))

    # sinusoidal families
    for f in (1, 2, 3, 4, 5, 6):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            candidates.append(enforce_sum(h, target, rng))

    # binary mask (first half =1)
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, target, rng))

    # a few random feasible vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # ----- pick the best initial candidate -----
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_best_h, global_best_c = best_h, best_c

    # ----- main optimisation phase (hill‑climb + SLSQP) -----
    main_deadline = start_time + budget_s * 0.75
    while time.time() < main_deadline:
        # start from a perturbed copy or a fresh random vector
        if rng.random() < 0.7:
            start_vec = perturb_h(global_best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        # aggressive hill‑climb
        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=2500,
                                      init_step=0.5, n_trials=8)
        if c_hc < global_best_c:
            global_best_h, global_best_c = h_hc.copy(), c_hc

        # a short SLSQP ladder
        for beta in (10, 100, 800, 2000):
            if time.time() >= main_deadline:
                break
            h_ref = slsqp_optimize(global_best_h, beta, max_iter=600)
            c_ref = compute_c5(h_ref)
            if c_ref < global_best_c:
                global_best_h, global_best_c = h_ref.copy(), c_ref

    # ----- pair‑transfer optimisation (majority of remaining time) -----
    remaining = budget_s - (time.time() - start_time)
    if remaining > 12.0:
        h_pt, c_pt = full_pair_transfer_optimize(
            global_best_h.copy(), rng, time_limit=remaining * 0.6
        )
        if c_pt < global_best_c:
            global_best_h, global_best_c = h_pt.copy(), c_pt

        # ----- final simulated‑annealing refinement -----
        remaining2 = budget_s - (time.time() - start_time)
        if remaining2 > 5.0:
            h_sa, c_sa = simulated_annealing_optimize(
                global_best_h.copy(), rng, time_limit=remaining2 * 0.9
            )
            if c_sa < global_best_c:
                global_best_h, global_best_c = h_sa.copy(), c_sa

    # ----- final clean‑up -----
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, target, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
