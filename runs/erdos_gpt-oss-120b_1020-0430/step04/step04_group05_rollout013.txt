def anneal_binary(h, rng, max_iters, time_start, time_budget):
    n = h.size
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    T0 = cur_c * 0.5 # starting temperature somewhat fraction of C5
    Tf = cur_c * 1e-4   # final small.
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf/T0) ** (it / max_iters)
        # compute worst lag
        lag_idx = np.argmax(cur_corr)
        lag = lag_idx - (n - 1)  # lag from -(n-1) to n-1
        # choose targeted or random
        if rng.random() < 0.7 and lag != 0:
            # get candidate i where h[i]=1 and (i+lag) within bounds and h[i+lag]=0
            if lag > 0:
                valid_i = np.where((h[:n-lag] == 1) & (h[lag:] == 0))[0]
            else:
                l = -lag
                valid_i = np.where((h[l:] == 1) & (h[:n-l] == 0))[0] + l
            if valid_i.size > 0:
                i = rng.choice(valid_i)
                # choose j from zeros not overlapping index i (any zero)
                zero_idx = np.where(h == 0)[0]
                # ensure j != i
                if zero_idx.size > 1:
                    j = rng.choice(zero_idx[zero_idx != i])
                else:
                    continue
            else:
                # fallback to random
                one_idx = np.where(h == 1)[0]
                zero_idx = np.where(h == 0)[0]
                i = rng.choice(one_idx)
                j = rng.choice(zero_idx)
        else:
            # random swap
            one_idx = np.where(h == 1)[0]
            zero_idx = np.where(h == 0)[0]
            i = rng.choice(one_idx)
            j = rng.choice(zero_idx)
        # swap
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h = new_h
            cur_c, cur_corr = new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:  # tiny residual correction
        for i in range(n):
            if 0 <= h[i] + diff <= 1:
                h[i] += diff
                break
    return h

def compute_c5_corr(h):
    """Return C5 and raw full correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = np.max(corr) * (2.0 / h.size)
    return c, corr

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- binary simulated annealing ----------
def anneal_binary(start_h, rng, max_iters, time_start, time_budget):
    """SA on binary {0,1} vectors (stored as float)."""
    n = start_h.size
    h = start_h.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    T0 = 0.02
    Tf = 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = max(T0 * (1.0 - it / max_iters), Tf)
        # worst lag
        lag_idx = np.argmax(cur_corr)
        lag = lag_idx - (n - 1)
        # try targeted swap with probability 0.7
        if rng.random() < 0.7 and lag != 0:
            if lag > 0:
                cand = np.where((h[:n - lag] == 1) & (h[lag:] == 0))[0]
                if cand.size:
                    i = rng.choice(cand)
                    j = i + lag
                else:
                    i = rng.choice(np.where(h == 1)[0])
                    j = rng.choice(np.where(h == 0)[0])
            else:
                l = -lag
                cand = np.where((h[l:] == 1) & (h[:n - l] == 0))[0] + l
                if cand.size:
                    i = rng.choice(cand)
                    j = i + lag  # lag negative
                else:
                    i = rng.choice(np.where(h == 1)[0])
                    j = rng.choice(np.where(h == 0)[0])
        else:
            i = rng.choice(np.where(h == 1)[0])
            j = rng.choice(np.where(h == 0)[0])
        if i == j:
            continue
        new_h = h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    n = 720

    # initial binary vector (half ones)
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        idx = np.argsort(-init)[: n // 2]
        h_bin = np.zeros(n, dtype=int)
        h_bin[idx] = 1
    else:
        h_bin = np.zeros(n, dtype=int)
        h_bin[: n // 2] = 1
        rng.shuffle(h_bin)

    best_h = h_bin.astype(float)
    best_c = compute_c5(best_h)

    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    target = 0.38079

    # main optimisation loop
    while time.time() - t0 < budget_s * 0.95 and best_c > target:
        remaining = budget_s - (time.time() - t0)
        # allocate time for SA
        sa_time = min(40.0, remaining * 0.6)
        max_iters = int(sa_time * 1500) + 2000  # ensure enough iterations
        # start point for SA
        if rng.random() < 0.6:
            start_bin = (best_h > 0.5).astype(int)
        else:
            start_bin = np.zeros(n, dtype=int)
            start_bin[: n // 2] = 1
            rng.shuffle(start_bin)
        sa_h, sa_c = anneal_binary(start_bin.astype(float), rng, max_iters,
                                   t0, t0 + sa_time)
        if sa_c < best_c:
            best_h, best_c = sa_h.astype(float), sa_c

        # SLSQP refinements
        for beta in beta_list:
            if time.time() - t0 >= budget_s * 0.95:
                break
            refined = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=500)
            refined_c = compute_c5(refined)
            if refined_c < best_c:
                best_h, best_c = refined, refined_c

    # final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
