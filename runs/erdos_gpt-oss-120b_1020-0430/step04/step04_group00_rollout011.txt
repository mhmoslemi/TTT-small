import time
import numpy as np
from scipy.optimize import minimize

# global vars for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def binary_sa_hill_climb(h_bin, rng, time_deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, float(best_c)


def tournament_select(pop, fitness, rng, tsize=3):
    idx = rng.integers(0, len(pop), size=tsize)
    best = idx[np.argmin(fitness[idx])]
    return pop[best]


def adjust_binary_to_target(arr, target_ones, rng):
    cur = int(arr.sum())
    diff = target_ones - cur
    if diff > 0:
        zeros = np.flatnonzero(arr == 0)
        if zeros.size >= diff:
            choose = rng.choice(zeros, size=diff, replace=False)
            arr[choose] = 1
    elif diff < 0:
        ones = np.flatnonzero(arr == 1)
        if ones.size >= -diff:
            choose = rng.choice(ones, size=-diff, replace=False)
            arr[choose] = 0
    return arr


def make_random_binary(n, target_ones, rng):
    arr = np.zeros(n, dtype=np.uint8)
    arr[rng.choice(n, target_ones, replace=False)] = 1
    return arr


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    n = 720
    target_ones = n // 2

    # ---------- initial population ----------
    popsize = 12
    population = []

    # use parent if provided
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size * 2 == n:
            parent_rep = np.repeat(parent, 2)
        else:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            parent_rep = np.interp(xt, xs, parent)
        # binary version from parent_rep
        idx = np.argpartition(parent_rep, -target_ones)[-target_ones:]
        bin_parent = np.zeros(n, dtype=np.uint8)
        bin_parent[idx] = 1
        population.append(bin_parent)

    while len(population) < popsize:
        population.append(make_random_binary(n, target_ones, rng))

    # evaluate initial fitness
    fitness = np.array([compute_c5(ind) for ind in population])
    best_overall_h = population[int(np.argmin(fitness))].copy()
    best_overall_c = float(np.min(fitness))

    # ---------- genetic algorithm ----------
    elite_cnt = 2
    crossover_prob = 0.9
    mutation_prob = 0.2
    max_ga_iters = 100000  # effectively limited by time

    for _ in range(max_ga_iters):
        now = time.time()
        if now > deadline - 5:
            break

        # select elites
        elite_idxs = np.argsort(fitness)[:elite_cnt]
        elites = [population[i] for i in elite_idxs]

        # generate new population
        new_pop = elites.copy()
        while len(new_pop) < popsize:
            p1 = tournament_select(population, fitness, rng)
            p2 = tournament_select(population, fitness, rng)
            child = p1.copy()
            if rng.random() < crossover_prob:
                cut = rng.integers(1, n)
                child = np.concatenate([p1[:cut], p2[cut:]]).astype(np.uint8)
                child = adjust_binary_to_target(child, target_ones, rng)
            if rng.random() < mutation_prob:
                ones = np.flatnonzero(child == 1)
                zeros = np.flatnonzero(child == 0)
                if ones.size and zeros.size:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    child[i], child[j] = child[j], child[i]
            new_pop.append(child)

        population = new_pop
        fitness = np.array([compute_c5(ind) for ind in population])
        cur_best_idx = int(np.argmin(fitness))
        cur_best_c = float(fitness[cur_best_idx])
        if cur_best_c < best_overall_c:
            best_overall_c = cur_best_c
            best_overall_h = population[cur_best_idx].copy()

    # ---------- binary hill climbing ----------
    remain = deadline - time.time()
    if remain > 30.0:
        sa_time = min(remain * 0.25, 30.0)
        sa_dead = time.time() + sa_time
        best_bin, best_c_bin = binary_sa_hill_climb(best_overall_h.copy(), rng, sa_dead)
        best_overall_h = best_bin
        best_overall_c = best_c_bin

    # ---------- continuous refinement ----------
    h_float = best_overall_h.astype(float)

    h_float = gradient_refine(h_float, rng, step=0.02, max_iter=12)

    betas = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
    for beta in betas:
        if time.time() > deadline - 5:
            break
        max_iter = 300 if beta < 200 else 600
        if beta >= 2000:
            max_iter = 800
        h_float = slsqp_optimize(h_float, beta, rng, max_iter=max_iter)
        c_val = compute_c5(h_float)
        if c_val < best_overall_c:
            best_overall_c = c_val
            best_overall_h = h_float.copy()

    # ---------- random perturb + short SLSQP ----------
    while time.time() + 5 < deadline:
        pert = best_overall_h + rng.normal(scale=0.02, size=n)
        np.clip(pert, 0.0, 1.0, out=pert)
        adjust_sum_exact(pert, rng)
        pert = slsqp_optimize(pert, beta=2000, rng=rng, max_iter=200)
        c_pert = compute_c5(pert)
        if c_pert < best_overall_c:
            best_overall_c = c_pert
            best_overall_h = pert.copy()

    # final safety
    best_overall_h = np.clip(best_overall_h, 0.0, 1.0)
    adjust_sum_exact(best_overall_h, rng)
    final_c = compute_c5(best_overall_h)

    return best_overall_h.tolist(), float(final_c), int(n)
