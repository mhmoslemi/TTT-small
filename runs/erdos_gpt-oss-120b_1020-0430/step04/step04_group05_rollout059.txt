def enforce_sum(h, target):
    diff = target - h.sum()
    h += diff / h.size
    # Now maybe some entries out of bounds; we clip and then re-adjust.
    # Iterate until all within bounds or diff small.
    for _ in range(10):
        over = h > 1
        under = h < 0
        if not (over.any() or under.any()):
            break
        # Clip
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # Distribute diff proportionally to free variables (not at bound)
        free = (~over) & (~under)
        n_free = free.sum()
        if n_free == 0:   # shouldn't happen; fallback equally
            h += diff / h.size
        else:
            h[free] += diff / n_free
    # final clip
    h = np.clip(h, 0, 1)
    # final correction to meet sum exactly (maybe just small rounding)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # adjust a random free index
        i = np.random.randint(0, h.size)
        h[i] = min(1, max(0, h[i] + diff))
    return h

def gradient_descent_refine(h, target_sum, rng, max_iters=2000, step0=0.1):
    n = h.size

        # compute correlation to get worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        worst_lag = np.argmax(corr) - (n - 1)
        # compute gradient for that lag
        if worst_lag >= 0:
            h_plus = np.concatenate([h[worst_lag:], np.zeros(worst_lag)])
            h_minus = np.concatenate([np.zeros(worst_lag), h[:n-worst_lag]])

            L = -worst_lag
            h_plus = np.concatenate([np.zeros(L), h[:n-L]])

def simulated_annealing_fast(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    n = h.size
    cur_corr = np.correlate(h, 1 - h, mode='full')
    cur_c = cur_corr.max() * (2.0 / n)
    best_h, best_c = h.copy(), cur_c
    T0 = 0.02 * (n / 360)**0.5  # scaling
    Tf = 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        # schedule
        T = T0 * (Tf / T0) ** (it / max_iters)
        # choose perturb
        if rng.random() < 0.8:
            # targeted perturb
            lag = np.argmax(cur_corr) - (n - 1)
            lag_abs = abs(lag)
            # compute contributions for this lag
            if lag >= 0:
                contrib = h[:n - lag] * (1.0 - h[lag:])
            else:
                L = -lag
                contrib = h[L:] * (1.0 - h[:n - L])
            # top and bottom indices
            k = max(1, int(len(contrib) * 0.1))
            top_idx = np.argpartition(-contrib, k - 1)[:k]
            bot_idx = np.argpartition(contrib, k - 1)[:k]
            # map back to original indices
            if lag >= 0:
                i = top_idx
                j = bot_idx + lag
            else:
                i = top_idx + lag
                j = bot_idx
            # ensure distinct
            if rng.random() < 0.5:
                i = rng.choice(i)
                j = rng.choice(j)
            else:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * (0.5 + 0.5 * T/T0)  # scale with temperature
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # Ensure still within bounds already
        # compute new C5
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c = new_corr.max() * (2.0 / n)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c

def gradient_descent_refine(h, target_sum, rng, max_iters=2000, step0=0.1):
    n = h.size
    # precompute factor
    factor = 2.0 / n
    for it in range(max_iters):
        # correlation to get worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c = corr.max() * factor
        lag = np.argmax(corr) - (n - 1)
        # compute gradient for worst lag (subgradient)
        if lag >= 0:
            h_plus = np.concatenate([h[lag:], np.zeros(lag)])
            h_minus = np.concatenate([np.zeros(lag), h[:n - lag]])
        else:
            L = -lag
            h_plus = np.concatenate([np.zeros(L), h[:n - L]])
            h_minus = np.concatenate([h[L:], np.zeros(L)])
        grad = 1.0 - (h_plus + h_minus)
        step = step0 * (0.99 ** it)
        new_h = h - step * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = enforce_sum(new_h, target_sum, rng)
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c = new_corr.max() * factor
        if new_c < cur_c:
            h = new_h
        # else maybe keep same; we do not accept worse
    return h

def simulated_annealing_fast(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    n = h.size
    factor = 2.0 / n
    cur_corr = np.correlate(h, 1 - h, mode='full')
    cur_c = cur_corr.max() * factor
    best_h, best_c = h.copy(), cur_c
    T0 = 0.02 * (n / 360.0) ** 0.5
    Tf = 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        # pick move
        if rng.random() < 0.8:
            lag = np.argmax(cur_corr) - (n - 1)
            lag_abs = abs(lag)
            # compute contributions for this lag
            if lag >= 0:
                contrib = h[:n - lag] * (1.0 - h[lag:])
            else:
                L = -lag
                contrib = h[L:] * (1.0 - h[:n - L])
            k = max(1, int(len(contrib) * 0.1))
            top_idx = np.argpartition(-contrib, k - 1)[:k]
            bot_idx = np.argpartition(contrib, k - 1)[:k]
            # map to original indices
            if lag >= 0:
                i = top_idx
                j = bot_idx + lag
            else:
                i = top_idx + lag
                j = bot_idx
            i = int(rng.choice(i))
            j = int(rng.choice(j))
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * (0.5 + 0.5 * T / T0)
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # compute new corr & c
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c = new_corr.max() * factor
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):

            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c

            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ----- main coarse phase (≈70% of budget) -----
    main_budget = budget_s * 0.7
    while True:
        elapsed = time.time() - t0
        if elapsed >= main_budget:
            break
        if budget_s - (time.time() - t0) < 2.0:
            break
        # start point
        if rng.random() < 0.75:
            start_vec = perturb_h(best_h, rng, scale=0.6)
        else:
            start_vec = random_feasible_vector(n, rng)
        # SA
        sa_h, sa_c = simulated_annealing_fast(
            start_vec, rng, max_iters=50000, time_start=t0, time_budget=t0 + main_budget
        )
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c
        # gradient descent
        gd_h = gradient_descent_refine(best_h.copy(), n / 2.0, rng, max_iters=2000, step0=0.2)
        gd_c = compute_c5(gd_h)
        if gd_c < best_c:
            best_h, best_c = gd_h, gd_c
        # focused improvement
        imp_h = improve_worst_lag(best_h.copy(), n / 2.0, rng, max_iters=2000)
        imp_c = compute_c5(imp_h)
        if imp_c < best_c:
            best_h, best_c = imp_h, imp_c
        # SLSQP ladder
        for beta in (200, 800, 2000, 5000):
            if time.time() - t0 >= main_budget:
                break
            try:
                ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=250)
            except Exception:
                continue
            ref_c = compute_c5(ref_h)
            if ref_c < best_c:
                best_h, best_c = ref_h, ref_c

    # ----- fine phase on higher resolution if time permits -----
    remaining = budget_s - (time.time() - t0)
    if remaining > budget_s * 0.1:
        n_high = min(720, max(480, n * 2))
        if n_high != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(n_high) + 0.5) * (2.0 / n_high)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, n_high / 2.0, rng)
        else:
            up_h = best_h.copy()
        # SA on fine grid
        sa_h, sa_c = simulated_annealing_fast(
            up_h,
            rng,
            max_iters=50000,
            time_start=t0,
            time_budget=t0 + budget_s,
        )
        if sa_c < best_c:
            best_h, best_c, n = sa_h, sa_c, n_high
        # gradient descent fine
        gd_h = gradient_descent_refine(best_h.copy(), n / 2.0, rng, max_iters=3000, step0=0.1)
        gd_c = compute_c5(gd_h)
        if gd_c < best_c:
            best_h, best_c = gd_h, gd_c
        # focused improvement fine
        imp_h = improve_worst_lag(best_h.copy(), n / 2.0, rng, max_iters=2000)
        imp_c = compute_c5(imp_h)
        if imp_c < best_c:
            best_h, best_c = imp_h, imp_c
        # SLSQP fine
        for beta in (500, 2000, 5000):
            if time.time() - t0 >= budget_s:
                break
            try:
                ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=200)
            except Exception:
                continue
            ref_c = compute_c5(ref_h)
            if ref_c < best_c:
                best_h, best_c = ref_h, ref_c
                n = n_high

    # ----- final quick hill‑climb -----
    cur_h, cur_c = best_h.copy(), best_c
    step = 0.2
    for _ in range(5000):
        if time.time() - t0 > budget_s:
            break
        new_h = perturb_h(cur_h, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        step *= 0.9995

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
