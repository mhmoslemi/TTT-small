```
def compute_overlap_one_one(h):
    # convolution of h with reversed h
    n = h.size
    L = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft(h[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]
    # conv[k] = sum_i h[i] * h[n-1-(k-i)]
    # For forward shift s = n-1 - k
    # So overlap_one_one[s] = conv[n-1 - s]
    # For s =0..n-1:
    return conv[n-1: -1:-1]   # maybe simpler: conv[n-1::-1][:n]
```

def correlate_full(h):
    n = h.shape[0]

```
During loop: swap on cur_h; compute new_c. Then decide acceptance based on delta = new_c - cur_c, not best_c. If new_c < cur_c: accept (set cur_c = new_c; cur_h already swapped). Also if new_c better than best_c, update best. If new_c >= cur_c, accept with probability exp(-delta/T) maybe, then cur_c = new_c (if accepted) else revert.

Thus need both cur_h and cur_c.
```

import time, math
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic feasibility tools
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Smooth surrogate for SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------------------------------------------
# Binary simulated annealing (fast)
# --------------------------------------------------------------
def binary_sa_fast(start_h, rng, deadline):
    """Swap‑based SA on a binary vector (exact half ones)."""
    h = start_h.astype(int)
    n = h.size
    # current state
    corr = np.correlate(h, h, mode="full")
    ov = corr[n - 1 :]
    cur_c = 1.0 - (2.0 / n) * ov.min()
    best_h = h.copy()
    best_c = cur_c

    # temperature schedule
    T0 = cur_c * 0.5 + 1e-6
    Tf = 1e-9
    max_steps = 2000000
    step = 0
    while time.time() < deadline and step < max_steps:
        step += 1
        frac = step / max_steps
        T = T0 * ((Tf / T0) ** frac)

        # pick a random 1 and a random 0
        i = rng.integers(0, n)
        while h[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] == 1:
            j = rng.integers(0, n)

        # perform swap
        h[i] = 0
        h[j] = 1

        # evaluate
        corr = np.correlate(h, h, mode="full")
        ov = corr[n - 1 :]
        new_c = 1.0 - (2.0 / n) * ov.min()
        delta = new_c - cur_c

        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            if T > 0 and rng.random() < math.exp(-delta / T):
                cur_c = new_c
            else:
                # revert swap
                h[i] = 1
                h[j] = 0
    return best_h.astype(float), best_c


# --------------------------------------------------------------
# Continuous simulated annealing
# --------------------------------------------------------------
def continuous_sa(h_start, rng, deadline):
    """SA on continuous values (preserves sum)."""
    h = np.clip(h_start.copy(), 0.0, 1.0)
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)

    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

    T0 = cur_c * 0.5 + 1e-6
    Tf = 1e-9
    max_steps = 500000
    step = 0
    while time.time() < deadline and step < max_steps:
        step += 1
        frac = step / max_steps
        T = T0 * ((Tf / T0) ** frac)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        scale = 0.5 * (1 - frac) + 0.01
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale

        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5(h_new)
        delta = new_c - cur_c

        if new_c < cur_c:
            h = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
        else:
            if T > 0 and rng.random() < math.exp(-delta / T):
                h = h_new
                cur_c = new_c
    return best_h, best_c


# --------------------------------------------------------------
# Simple greedy hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max_iters // 10 or 1
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ----------------------------------------------------------
    # Initialise base resolution and candidate
    # ----------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        best_h = np.array(initial_h_values, dtype=float)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, n / 2.0, rng)
    else:
        n = 720
        best_h = random_feasible_vector(n, rng)

    best_c = compute_c5(best_h)

    # ----------------------------------------------------------
    # Binary SA phase (≈55% of time)
    # ----------------------------------------------------------
    phase1_deadline = start_time + 0.55 * budget_s
    # start binary from top half of best_h
    idx = np.argsort(-best_h)[: n // 2]
    bin_start = np.zeros(n, dtype=float)
    bin_start[idx] = 1.0

    while time.time() < phase1_deadline:
        run_deadline = min(phase1_deadline, time.time() + 30.0)
        if rng.random() < 0.7:
            start_bin = bin_start.copy()
        else:
            pos = rng.choice(n, size=n // 2, replace=False)
            start_bin = np.zeros(n, dtype=float)
            start_bin[pos] = 1.0
        bin_h, bin_c = binary_sa_fast(start_bin, rng, run_deadline)
        if bin_c < best_c:
            best_c = bin_c
            best_h = bin_h.copy()
            idx = np.argsort(-best_h)[: n // 2]
            bin_start = np.zeros(n, dtype=float)
            bin_start[idx] = 1.0
        if time.time() < phase1_deadline:
            h_hc, c_hc = hill_climb(bin_h, rng, max_iters=2000, start_step=0.3)
            if c_hc < best_c:
                best_c = c_hc
                best_h = h_hc.copy()

    # ----------------------------------------------------------
    # Continuous refinement phase (≈30% of time)
    # ----------------------------------------------------------
    phase2_deadline = start_time + 0.85 * budget_s
    beta_list = [10.0, 40.0, 150.0, 500.0, 2000.0, 8000.0]
    cur_start = best_h.copy()
    while time.time() < phase2_deadline:
        for beta in beta_list:
            if time.time() >= phase2_deadline:
                break
            try:
                h_opt = slsqp_optimize(cur_start, beta, max_iter=600)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()
                    cur_start = h_opt
            except Exception:
                continue
        if time.time() < phase2_deadline:
            h_sa, c_sa = continuous_sa(best_h, rng, phase2_deadline)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa.copy()
                cur_start = h_sa
        if time.time() < phase2_deadline:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=3000, start_step=0.2)
            if c_hc < best_c:
                best_c = c_hc
                best_h = h_hc.copy()
                cur_start = h_hc

    # ----------------------------------------------------------
    # Upsampling and final polish (remaining time)
    # ----------------------------------------------------------
    if time.time() < deadline:
        if best_h.size * 2 <= 1440:
            factor = 2
            h_fine = np.repeat(best_h, factor)
            n_fine = h_fine.size
            h_fine = enforce_sum(h_fine, n_fine / 2.0, rng)

            h_fine, c_fine = hill_climb(h_fine, rng, max_iters=5000, start_step=0.2)
            if c_fine < best_c:
                best_c = c_fine
                best_h = h_fine.copy()

            for beta in [2000.0, 8000.0, 20000.0]:
                if time.time() >= deadline:
                    break
                try:
                    h_opt = slsqp_optimize(best_h, beta, max_iter=300)
                    c_opt = compute_c5(h_opt)
                    if c_opt < best_c:
                        best_c = c_opt
                        best_h = h_opt.copy()
                except Exception:
                    continue

    # Final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
