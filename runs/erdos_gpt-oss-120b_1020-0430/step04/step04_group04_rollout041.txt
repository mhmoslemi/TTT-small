def pgd_optimize(h0, beta, lambda_eq=1.0, max_iter=5000, step0=0.2):
    h = h0.copy()
    n = len(h)
    target = n/2.0
    beta = float(beta)
    offset_len = 2*n-1
    for it in range(max_iter):
        # compute C
        C = np.correlate(h, 1 - h, mode='full')
        # for numerics, subtract max for stability
        maxC = C.max()
        expC = np.exp(beta * (C - maxC))
        sumExp = expC.sum()
        w = expC / sumExp
        # conv1
        corr = np.correlate(w, h, mode='full')
        offset = len(w) - 1
        conv1 = corr[offset:offset+n]
        # conv2
        conv2 = np.convolve(w, h, mode='full')[:n]
        grad = 1.0 - (conv1 + conv2)
        # equality penalty gradient
        sum_h = h.sum()
        grad += 2.0 * lambda_eq * (sum_h - target)
        # step size schedule
        step = step0 / np.sqrt(it + 1)
        # gradient descent step
        h_new = h - step * grad
        # project onto [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce sum using enforce_sum
        h_new = enforce_sum(h_new, target, np.random.default_rng())

Implementation details:

def conv_corr(w, h, nfft):
    # returns (conv1, conv2)
    # conv1 = correlation of w with h: sum_i w_i * h_{i+lag}
    # compute rev_h = h[::-1]
    W = np.fft.rfft(w, nfft)
    H_rev = np.fft.rfft(h[::-1], nfft)
    corr_full = np.fft.irfft(W * H_rev, nfft)  # length nfft
    # The correlation length = len(w) + len(h) - 1 = L = 3n-2
    L = len(w) + len(h) - 1
    corr_full = corr_full[:L]  # truncate
    offset = len(w) - 1
    conv1 = corr_full[offset:offset+len(h)]  # length n
    # conv2 = convolution (w * h)
    H = np.fft.rfft(h, nfft)
    conv_full = np.fft.irfft(W * H, nfft)[:L]
    conv2 = conv_full[:len(h)]
    return conv1, conv2

def pgd_optimize(h0, betas, max_iter_per_beta=1500, step0=0.5, lambda_eq=10.0):
    h = h0.copy()
    n = len(h)
    target = n/2.0
    for beta in betas:
        for it in range(max_iter_per_beta):
            # compute C
            C = np.correlate(h, 1 - h, mode='full')
            # softmax
            maxC = C.max()
            expC = np.exp(beta * (C - maxC))
            w = expC / expC.sum()
            # compute convs via FFT
            nfft = 1 << ((3*n - 2 - 1).bit_length())
            conv1, conv2 = conv_corr(w, h, nfft)
            grad = 1.0 - (conv1 + conv2)
            sum_h = h.sum()
            grad += 2.0 * lambda_eq * (sum_h - target)
            step = step0 / np.sqrt(it + 1)
            h = h - step * grad
            # project onto [0,1]
            h = np.clip(h, 0.0, 1.0)
            # sum projection
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                free = (h > 1e-12) & (h < 1.0 - 1e-12)
                nfree = int(free.sum())
                if nfree > 0:
                    h[free] += diff / nfree
                else:
                    h += diff / n
                h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv.max() * (2.0 / n)

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    # perform random transfers while maintaining sum
    for _ in range(n * 5):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0: continue
        eps = (rng.random() - 0.5) * max_eps * 2 * 0.3  # scale 30%
        h[i] -= eps
        h[j] += eps
    # enforce sum exactly
    target = n/2.0
    h = enforce_sum(h, target, rng)
    return h

def conv_corr(w, h, nfft):
    # w length Lw = 2n-1, h length n
    W = np.fft.rfft(w, nfft)
    H_rev = np.fft.rfft(h[::-1], nfft)

    conv_full = np.fft.irfft(W * H, nfft)[:L]
    conv2 = conv_full[:len(h)]
    return conv1, conv2

def pgd_optimize(h0, betas, max_iter=1500, step0=0.5, lambda_eq=10.0, rng=None):
    n = h0.size
    target = n/2.0
    h = h0.copy()
    Lw = 2*n - 1
    L = 3*n - 2
    nfft = 1 << ((L - 1).bit_length())
    for beta in betas:
        beta_f = float(beta)
        for it in range(max_iter):
            C = np.correlate(h, 1 - h, mode='full')
            maxC = C.max()
            expC = np.exp(beta_f * (C - maxC))
            w = expC / expC.sum()
            conv1, conv2 = conv_corr(w, h, nfft)
            grad = 1.0 - (conv1 + conv2)
            # equality penalty
            sum_h = h.sum()
            grad += 2.0 * lambda_eq * (sum_h - target)
            # step size schedule: simple diminishing
            step = step0 / np.sqrt(it + 1)
            # update
            h = h - step * grad
            # clip and enforce sum
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                free = (h > 1e-12) & (h < 1 - 1e-12)
                nfree = int(free.sum())
                if nfree > 0:
                    h[free] += diff / nfree
                else:
                    h += diff / n
                h = np.clip(h, 0.0, 1.0)
    return h

def enforce_sum(h, target, rng):
    """Project h onto the box [0,1] and adjust to reach the given sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1 - 1e-12)
        nfree = int(free.sum())
        if nfree:
            inc = diff / nfree
            h[free] += inc
        else:
            h += diff / n

        diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if diff > 0 and h[i] < 1 - 1e-12:

                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 1e-12:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def compute_c5(h):

    n = h.shape[0]
    rev = (1 - h)[::-1]

    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv.max() * (2.0 / n)

    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
