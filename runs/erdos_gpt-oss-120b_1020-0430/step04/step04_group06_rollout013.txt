import time

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random h with Σh = n/2 and 0≤h≤1."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(2000):
            if abs(diff) < 1e-14:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple hill‑climb with multiple perturbations per step."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Move mass from i to j that participate in the worst shift."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        max_val = float(corr[idx])
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        shift = idx - (n - 1)
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i, j = int(i_range[idx_max]), int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c

    start = time.time()
    h = h0.copy()

    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    beta_vals = [5, 20, 80, 200, 500, 1000, 2000, 5000]
    step = 0.2
    while time.time() - start < max_time:
        for beta in beta_vals:
            for _ in range(40):
                if time.time() - start >= max_time:
                    return best_h, best_c
                corr = np.correlate(h, 1.0 - h, mode="full")
                maxb = beta * np.max(corr)
                exp_terms = np.exp(beta * corr - maxb)
                p = exp_terms / np.sum(exp_terms)
                a = h
                b = 1.0 - h
                conv_pb = np.convolve(p, b[::-1], mode="full")
                conv_pa = np.convolve(p, a[::-1], mode="full")
                offset = len(p) // 2
                g = conv_pb[offset:offset + n] - conv_pa[offset:offset + n]
                gn = np.linalg.norm(g)
                h_new = h - step * g / gn if gn > 0 else h.copy()
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = enforce_sum(h_new, target, rng)
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    best_c, best_h = c_new, h_new.copy()
                if c_new <= cur_c:
                    h, cur_c = h_new, c_new
                step *= 0.995
            step = max(step, 0.001)
    return best_h, best_c


# -------------------------------------------------
# Main entry point
# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    total_start = time.time()

    # choose the discretisation size (more points give a tighter bound)
    n = 480 if budget_s > 900 else 360
    target = n / 2.0

    # ----- Build a pool of starting vectors -----
    candidates = []
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target, rng)
        candidates.append(h0)

    for _ in range(4):
        candidates.append(random_feasible_vector(n, rng))

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in range(1, 7):
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.cos(2 * np.pi * f * xs / 2 + phase)
        h = np.clip(h, 0.0, 1.0)
        candidates.append(enforce_sum(h, target, rng))

    for _ in range(4):
        terms = rng.integers(3, 7)
        coeffs = rng.normal(scale=0.2, size=terms)
        freqs = rng.integers(1, 9, size=terms)
        h = np.full(n, 0.5)
        for a, f in zip(coeffs, freqs):
            h += a * np.cos(2 * np.pi * f * xs / 2)
        h = np.clip(h, 0.0, 1.0)
        candidates.append(enforce_sum(h, target, rng))

    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, target, rng))
    mask2 = np.zeros(n)
    s = n // 4
    mask2[s : s + n // 2] = 1.0
    candidates.append(enforce_sum(mask2, target, rng))

    # ----- Initial best -----
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- Main optimisation loop (gradient + hill‑climb) -----
    while time.time() - total_start < budget_s * 0.70:
        start_vec = (
            perturb_h(best_h, rng, scale=0.3)
            if rng.random() < 0.7
            else random_feasible_vector(n, rng)
        )
        remaining = budget_s - (time.time() - total_start)
        if remaining < 30.0:
            break
        # gradient descent with a modest time budget
        gd_h, gd_c = gradient_descent_opt(
            start_vec, rng, max_time=min(30.0, remaining * 0.5)
        )
        if gd_c < best_c:
            best_c, best_h = gd_c, gd_h.copy()
        # short hill‑climb for quick tweaks
        h_hc, c_hc = hill_climb_multi(
            best_h, rng, max_iters=800, init_step=0.3, n_trials=4
        )
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

    # ----- Final refinement with pair‑transfer -----
    remaining = budget_s - (time.time() - total_start)
    if remaining > 5.0:
        attempts = max(1, int(remaining // 30))
        per_attempt = remaining / (attempts + 1)
        for _ in range(attempts):
            h_start = best_h.copy()
            if rng.random() < 0.5:
                h_start = perturb_h(h_start, rng, scale=0.2)
            h_opt, c_opt = pair_transfer_optimize(
                h_start, rng, max_iters=15000, time_limit=per_attempt, init_factor=0.5
            )
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
        # one last, longer run
        h_opt, c_opt = pair_transfer_optimize(
            best_h, rng, max_iters=25000, time_limit=remaining * 0.9, init_factor=0.5
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- Clean up and return -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
