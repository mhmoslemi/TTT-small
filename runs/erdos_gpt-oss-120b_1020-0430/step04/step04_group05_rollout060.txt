def project_to_feasible(h, target):
    # ensure within bounds
    h_proj = np.clip(h, 0.0, 1.0)
    # Use water-filling
    # iterate few times
    for _ in range(10):
        s = h_proj.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # identify free indices
        free = (h_proj > 0 + 1e-12) & (h_proj < 1 - 1e-12)
        num_free = free.sum()
        if num_free == 0:
            # adjust by distributing equally (some will exceed bounds)
            # fallback: add diff/len to all then clip
            h_proj += diff / len(h_proj)
            h_proj = np.clip(h_proj, 0.0, 1.0)
            continue
        add = diff / num_free
        h_proj[free] += add
        # clip again
        h_proj = np.clip(h_proj, 0.0, 1.0)
    # final rounding adjustment
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        # adjust small random index within bounds
        for i in range(len(h_proj)):
            if diff > 0 and h_proj[i] < 1.0:
                inc = min(diff, 1.0 - h_proj[i])
                h_proj[i] += inc
                diff -= inc
            elif diff < 0 and h_proj[i] > 0.0:
                dec = min(-diff, h_proj[i])
                h_proj[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return h_proj

def gradient_step_worst_lag(h, target, step):
    n = len(h)
    # compute full correlation (conv) to get sums for each lag
    # correlation yields indices from -(n-1) to (n-1)
    corr = np.correlate(h, 1 - h, mode='full')

    # find lag index
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)  # negative or positive
    # compute gradient g_i = (1 - h_{i+lag}) - h_{i-lag}
    # we need to treat out-of-range indices as 0? Actually correlation considers wrap? No, np.correlate treats zero padding outside bounds. So h_{i+lag} is 0 when i+lag is out of [0,n-1], similarly h_{i-lag}=0.

    g = np.zeros(n)
    if lag >= 0:
        # i+lag within for i in 0..n-1-lag
        g[:n-lag] += (1 - h[lag:])  # (1 - h_{i+lag})
        # i-lag within for i in lag..n-1  => h_{i-lag}

        g[:n-L] -= h[L:]
    # step direction opposite to gradient for minimization of C5_k
    h_new = h - step * g
    # project back
    h_new = project_to_feasible(h_new, target)
    return h_new

def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    T0 = 0.5 # initial temperature
    Tf = 1e-4
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        # propose neighbor: transfer epsilon between random indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.4  # scale factor
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # maybe also accept small random neighbor that doesn't change sum? Already ok.
        new_c = compute_c5(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c

def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    T0 = 0.5
    Tf = 1e-4
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf/T0) ** (it / max_iters)
        # compute current correlation to find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        lag = np.argmax(corr) - (n - 1)
        # with probability 0.7 use targeted perturb
        if rng.random() < 0.7:

        else:
            new_h = perturb_h(h, rng, scale=0.5)

        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()

def gradient_descent_improve(h, target, time_start, time_budget):
    n = h.size
    step = 0.1
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c

        # gradient
        g = np.zeros(n)

            g[:n-L] += (1 - h[L:])

            L = -lag
            g[L:] += (1 - h[:n-L])
            g[:n-L] -= h[L:]

                best_c = new_c
                best_h = h.copy()

        # bound step
        if step < 1e-5:
            step = 1e-5

def project_to_feasible(h, target):
    # Clip first
    h = np.clip(h, 0.0, 1.0)
    # water-filling iterative adjustment
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # free indices
        free = (h > 0.0) & (h < 1.0)
        num_free = free.sum()
        if num_free == 0:
            # all at bounds, distribute uniformly then clip
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        add = diff / num_free
        h[free] += add
        h = np.clip(h, 0.0, 1.0)
    # final correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute to first indices with room
        for i in range(h.size):
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return h

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx (discrete version)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-12:

        h[free] += add

        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.4
        h[i] -= eps
        h[j] += eps
    # enforce exact sum
    target = n / 2.0
    return project_to_feasible(h, target)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small ε from one index to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def targeted_perturb(h: np.ndarray, rng: np.random.Generator, lag: int,
                    scale: float = 0.5, top_frac: float = 0.1) -> np.ndarray:
    """Bias transfer toward indices that dominate the worst lag."""
    n = h.size
    if lag >= n or lag <= -n:
        return perturb_h(h, rng, scale)
    if lag >= 0:
        contrib = h[: n - lag] * (1.0 - h[lag:])
        idx = np.arange(n - lag)
    else:
        L = -lag
        contrib = h[L:] * (1.0 - h[: n - L])
        idx = np.arange(L, n)
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(contrib) * top_frac))
    top_idx = idx[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = idx[np.argpartition(contrib, k - 1)[:k]]
    i = top_idx[rng.integers(0, k)]
    if rng.random() < 0.7:
        j = bot_idx[rng.integers(0, k)]
    else:
        j = rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def simulated_annealing(h_start: np.ndarray, rng: np.random.Generator,
                        max_iters: int, time_start: float,
                        time_budget: float) -> tuple[np.ndarray, float]:
    """Simple SA using targeted perturbations."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.5, 1e-4
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        corr = np.correlate(h, 1.0 - h, mode="full")
        lag = np.argmax(corr) - (n - 1)
        if rng.random() < 0.7:
            new_h = targeted_perturb(h, rng, lag, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c = compute_c5(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c


def gradient_descent_improve(h: np.ndarray, target: float,
                             time_start: float, time_budget: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent focusing on the current worst lag."""
    n = h.size
    step = 0.1
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    while time.time() - time_start < time_budget:
        corr = np.correlate(h, 1.0 - h, mode="full")
        lag = np.argmax(corr) - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            L = lag
            g[: n - L] += (1.0 - h[L:])
            g[L:] -= h[: n - L]
        else:
            L = -lag
            g[L:] += (1.0 - h[: n - L])
            g[: n - L] -= h[L:]
        h_new = h - step * g
        h_new = project_to_feasible(h_new, target)
        new_c = compute_c5(h_new)
        if new_c < cur_c:
            h, cur_c = h_new, new_c
            step *= 1.07
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            step *= 0.70
        if step < 1e-5:
            step = 1e-5
    return best_h, best_c


# ---------- SLSQP refinement ----------
def sum_eq_constraint(x: np.ndarray, target: float) -> float:
    return np.sum(x) - target


def softmax_obj(x: np.ndarray, beta: float) -> float:
    corr = np.correlate(x, 1.0 - x, mode="full")
    v = beta * corr
    m = np.max(v)
    return (m + np.log(np.sum(np.exp(v - m)))) / beta


def slsqp_refine(h: np.ndarray, beta: float,
                 max_iter: int = 150) -> np.ndarray:
    """SLSQP on the smooth max (softmax) objective."""
    n = h.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint, "args": (target,)}
    res = minimize(softmax_obj, h, args=(beta,), method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-9, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_to_feasible(x, target)


# ---------- main entry ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    # --- initial vector ---
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_feasible(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # --- main stochastic phase (≈70% of budget) ---
    main_budget = budget_s * 0.70
    main_limit = t0 + main_budget
    while time.time() < main_limit:
        # restart
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.2)
        else:
            start = random_feasible_vector(n, rng)

        # simulated annealing
        sa_h, sa_c = simulated_annealing(
            start, rng, max_iters=8000,
            time_start=t0, time_budget=main_limit)
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c

        # gradient descent refinement
        gd_h, gd_c = gradient_descent_improve(
            sa_h, n / 2.0, time_start=t0, time_budget=main_limit)
        if gd_c < best_c:
            best_h, best_c = gd_h, gd_c

        # occasional SLSQP polishing
        if time.time() < main_limit - 2.0:
            try:
                refined = slsqp_refine(best_h, beta=50, max_iter=150)
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined, c_ref
            except Exception:
                pass

    # --- up-sampling & fine tuning if time permits ---
    remaining = budget_s - (time.time() - t0)
    if remaining > budget_s * 0.15:
        new_n = min(720, max(480, best_h.size * 2))
        if new_n != best_h.size:
            old_n = best_h.size
            old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
            new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = project_to_feasible(up_h, new_n / 2.0)

            # SA on finer grid
            up_h, up_c = simulated_annealing(
                up_h, rng, max_iters=8000,
                time_start=time.time(),
                time_budget=time.time() + remaining * 0.5)

            # gradient descent on finer grid
            up_h, up_c = gradient_descent_improve(
                up_h, new_n / 2.0,
                time_start=time.time(),
                time_budget=time.time() + remaining * 0.5)

            if up_c < best_c:
                best_h, best_c = up_h, up_c
                n = new_n

    # --- final clean‑up ---
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
