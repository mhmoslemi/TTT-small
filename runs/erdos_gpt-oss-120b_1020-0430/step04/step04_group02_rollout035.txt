import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)

# ---------- evaluation ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def compute_corr_raw_fft(h):
    n = h.size
    n_fft = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(h, n_fft)
    b_fft = np.fft.rfft((1.0 - h)[::-1], n_fft)
    conv = np.fft.irfft(a_fft * b_fft, n_fft)
    return conv[: 2 * n - 1]  # raw correlation (no dx)

# ---------- SLSQP softmax ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

# ---------- gradient descent on softmax ----------
def gradient_step(h, step, beta):
    n = h.size
    corr_raw = compute_corr_raw_fft(h)
    max_corr = corr_raw.max()
    w = np.exp(beta * (corr_raw - max_corr))
    w /= w.sum()
    n_fft = 1 << ((3 * n - 2).bit_length())
    w_fft = np.fft.rfft(w, n_fft)
    h_fft = np.fft.rfft(h, n_fft)
    conv1_full = np.fft.irfft(w_fft * h_fft, n_fft)
    conv1 = conv1_full[n - 1 : 2 * n - 1]
    w_rev_fft = np.fft.rfft(w[::-1], n_fft)
    conv2_full = np.fft.irfft(w_rev_fft * h_fft, n_fft)
    conv2 = conv2_full[n - 1 : 2 * n - 1]
    grad = 1.0 - (conv1 + conv2)
    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_box_sum(h_new, n / 2.0)

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- phase 1: gradient descent on base grid ----
    step = 0.25
    beta = 0.5
    step_decay = 0.995
    beta_factor = 1.4
    beta_max = 5000.0

    while True:
        if time.time() - start_time > budget_s * 0.65:
            break
        # occasional random restart
        if rng.random() < 0.02:
            cur_h = random_feasible_vector(n, rng)
        else:
            cur_h = best_h.copy()
        cur_h = gradient_step(cur_h, step, beta)
        cur_c = compute_c5(cur_h)
        if cur_c < best_c:
            best_c, best_h = cur_c, cur_h.copy()
        step = max(step * step_decay, 1e-5)
        beta = min(beta * beta_factor, beta_max)

    # ---- phase 2: upsample and fine‑tune ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.15:
        target_n = n * 2
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = project_box_sum(h_up, target_n / 2.0)
        n = target_n
        best_h = h_up
        best_c = compute_c5(best_h)

        step = 0.2
        beta = 0.5
        step_decay = 0.998
        beta_factor = 1.3
        beta_max = 10000.0

        while True:
            if time.time() - start_time > budget_s:
                break
            cur_h = gradient_step(best_h, step, beta)
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
            step = max(step * step_decay, 1e-6)
            beta = min(beta * beta_factor, beta_max)

    # ---- phase 3: SLSQP refinement ----
    beta_list = [800, 2000, 5000, 10000]
    for b in beta_list:
        if time.time() - start_time > budget_s:
            break
        try:
            refined = slsqp_optimize(best_h, b, max_iter=400)
        except Exception:
            continue
        refined_c = compute_c5(refined)
        if refined_c < best_c:
            best_c, best_h = refined_c, refined.copy()

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
