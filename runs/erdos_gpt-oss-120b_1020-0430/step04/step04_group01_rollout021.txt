import time, numpy as np
from scipy.optimize import minimize

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]
    return conv

def compute_c5(h):
    return correlate_full(h).max() * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
            h = nh
        if (it+1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# Smooth max via log-sum-exp for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, max_iter=350):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options={"maxiter": max_iter, "ftol":1e-12, "disp":False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff>0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def binary_sa_extended(start_h, rng, deadline):
    h = start_h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    state_h = h.copy()
    state_c = best_c
    T0 = best_c * 0.02 + 1e-8
    Tf = 1e-8
    n = h.size
    while time.time() < deadline:
        frac = (time.time() - start_h_time) / (deadline - start_h_time)
        T = T0 * ((Tf / T0) ** frac)
        if rng.random() < 0.2:
            ones = np.where(state_h == 1.0)[0]
            zeros = np.where(state_h == 0.0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(ones, size=2, replace=False)
            j1, j2 = rng.choice(zeros, size=2, replace=False)
            state_h[i1] = 0.0; state_h[i2] = 0.0
            state_h[j1] = 1.0; state_h[j2] = 1.0
            new_c = compute_c5(state_h)
            delta = new_c - state_c
            if delta < 0 or rng.random() < np.exp(-delta / max(T,1e-12)):
                state_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = state_h.copy()
            else:
                state_h[i1] = 1.0; state_h[i2] = 1.0
                state_h[j1] = 0.0; state_h[j2] = 0.0
        else:
            # simple swap
            # find a 1 and a 0
            i = rng.choice(np.where(state_h == 1.0)[0])
            j = rng.choice(np.where(state_h == 0.0)[0])
            state_h[i] = 0.0; state_h[j] = 1.0
            new_c = compute_c5(state_h)
            delta = new_c - state_c
            if delta < 0 or rng.random() < np.exp(-delta / max(T,1e-12)):
                state_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = state_h.copy()
            else:
                state_h[i] = 1.0; state_h[j] = 0.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.96
    # Initialize
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        idx = np.argsort(-h0)[:n//2]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
    else:
        n = 720
        h_bin = np.zeros(n, dtype=float)
        h_bin[:n//2] = 1.0
        rng.shuffle(h_bin)
    best_h = h_bin.copy()
    best_c = compute_c5(best_h)
    # Simulated annealing on binary
    global start_h_time
    start_h_time = time.time()
    bin_deadline = start_h_time + (deadline - start_time) * 0.7
    bin_h, bin_c = binary_sa_extended(best_h, rng, bin_deadline)
    if bin_c < best_c:
        best_h, best_c = bin_h, bin_c
    # Continuous hill climb
    remaining = deadline - time.time()
    if remaining > 5.0:
        hc_h, hc_c = hill_climb(best_h, rng, max_iters=8000, start_step=0.2)
        if hc_c < best_c:
            best_h, best_c = hc_h, hc_c
    # SLSQP polishing
    remaining = deadline - time.time()
    if remaining > 20.0:
        start = best_h.copy()
        for beta in [5.0,20.0,80.0,200.0,800.0,2000.0,5000.0]:
            if time.time() >= deadline:
                break
            try:
                refined = slsqp_optimize(start, beta, max_iter=400)
            except Exception:
                continue
            c = compute_c5(refined)
            if c < best_c:
                best_c = c
                best_h = refined.copy()
            start = refined
    # final enforce and compute
    best_h = np.clip(best_h,0.0,1.0)
    best_h = enforce_sum(best_h, best_h.size/2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np
from scipy.optimize import minimize

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    return correlate_full(h).max() * (2.0 / h.size)

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def hill_climb(h_start, rng, max_iters=20000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        nh = perturb_h(h, rng, scale=step)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
            h = nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# ------- smooth‑max SLSQP helpers -------
CURRENT_TARGET = None
CURRENT_BETA = None
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def slsqp_optimize(h0, beta, max_iter=350):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
