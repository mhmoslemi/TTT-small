    n = h.size
    dx = 2.0 / n
    # reverse 1-h
    rh = (1 - h)[::-1]
    size = 1 << ((2*n-1).bit_length())

    conv = np.fft.irfft(fft_h * fft_rh, n=size)[:2*n-1]

def simulated_annealing(h, rng, budget_s, start_time, end_time):
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # schedule parameters
    T0 = 0.02
    T_end = 1e-5
    # loop
    while time.time() < end_time:
        # temperature based on elapsed fraction of remaining interval
        frac = (time.time() - start_time) / (end_time - start_time)
        T = max(T_end, T0 * (1 - frac))
        i, j = rng.choice(n, 2, replace=False)
        max_eps = min(cur_h[i], 1 - cur_h[j])
        if max_eps == 0:
            continue
        eps = rng.random() * max_eps
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5_fft(nh)
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / T):
            cur_h, cur_c = nh, nc
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global placeholders for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None
FFT_SIZE = None

# ---------- utility functions ----------
def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in-place so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h

def compute_corr_fft(h: np.ndarray, size: int) -> np.ndarray:
    """Full correlation of h and (1-h) via FFT, length 2*len(h)-1."""
    rh = (1 - h)[::-1]
    fft_h = np.fft.fft(h, n=size)
    fft_rh = np.fft.fft(rh, n=size)
    conv = np.fft.ifft(fft_h * fft_rh).real[: 2 * h.size - 1]
    return conv

def compute_c5_fft(h: np.ndarray) -> float:
    """Discrete C5 for a step function using FFT."""
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    conv = compute_corr_fft(h, size)
    return conv.max() * (2.0 / n)

def sum_eq_constraint_slsqp(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET

def softmax_obj_slsqp(x: np.ndarray) -> float:
    """Smooth approximation of max_k overlap used by SLSQP."""
    conv = compute_corr_fft(x, FFT_SIZE)
    bc = CURRENT_BETA * conv
    m = np.max(bc)
    # stable log‑sum‑exp, clipping to avoid overflow
    exp_vals = np.exp(np.clip(bc - m, -700, None))
    return (m + np.log(np.sum(exp_vals))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Refine h via SLSQP on the soft‑max objective."""
    global CURRENT_TARGET, CURRENT_BETA, FFT_SIZE
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    FFT_SIZE = 1 << ((2 * n - 1).bit_length())
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint_slsqp}
    res = minimize(
        softmax_obj_slsqp,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    it = 0

        x[i] += add
        diff = CURRENT_TARGET - x.sum()
        it += 1
    return x

def hill_climb_greedy(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.7):
    """Simple greedy hill‑climb using transfers that preserve the sum."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5_fft(cur)
    best_h = cur.copy()
    best_c = cur_c
    for _ in range(max_iters):
        i, j = rng.choice(n, 2, replace=False)
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * step_scale
        nh = cur.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5_fft(nh)
        if nc < cur_c:
            cur, cur_c = nh, nc
            if nc < best_c:
                best_c, best_h = nc, nh.copy()

def simulated_annealing(h: np.ndarray, rng: np.random.Generator,
                        start_time: float, end_time: float):
    """SA with temperature linearly decreasing from T0 to T_end."""
    n = h.size
    cur = h.copy()
    cur_c = compute_c5_fft(cur)
    best_h = cur.copy()
    best_c = cur_c

    T_end = 1e-5

        frac = (time.time() - start_time) / (end_time - start_time)
        T = max(T_end, T0 * (1.0 - frac))
        i, j = rng.choice(n, 2, replace=False)
        max_eps = min(cur[i], 1.0 - cur[j])
        if max_eps <= 0.0:

        eps = rng.random() * max_eps
        nh = cur.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5_fft(nh)
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / T):
            cur, cur_c = nh, nc
            if nc < best_c:
                best_c, best_h = nc, nh.copy()
    return best_h, best_c
