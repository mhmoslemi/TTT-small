def compute_c5_fft(h):
    n = h.shape[0]
    g = 1.0 - h
    # reverse g for convolution
    grev = g[::-1]
    size = 1 << ((2*n - 1).bit_length())  # next power of two for speed
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(grev, size)
    conv = np.fft.irfft(H * G, size)
    # conv length = size, but the "valid" region for full convolution length = 2*n -1, located at indices 0..2n-2
    conv = conv[:2*n-1]
    max_corr = np.max(conv)
    return float(max_corr * (2.0 / n))
```

    cur_h = h.copy()

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5_naive(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def compute_c5_fft(h):
    n = h.shape[0]
    g = 1.0 - h
    grev = g[::-1]
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(grev, size)
    conv = np.fft.irfft(H * G, size)
    conv = conv[: 2 * n - 1]
    return float(conv.max() * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact redistribution to achieve Σh = n/2 for binary vectors."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- gradient based refinement (fractional) ----------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # contribution of (1-h_{i+k})
            g[k:] -= h[:upper]                # contribution of -h_i
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):

        slice_deadline = time.time() + remaining * 0.85

        # ---- simulated annealing (binary) ----
        h_sa, c_sa = simulated_annealing(h0, rng, slice_deadline)
        if c_sa < best_c:
            best_c, best_h, best_n = c_sa, h_sa.copy(), n

        # ---- refinement (gradient + optional SLSQP) ----
        if time.time() < deadline:
            h_ref, c_ref = refine_solution(best_h, rng, deadline)
            if c_ref < best_c:
                best_c, best_h, best_n = c_ref, h_ref.copy(), n

    # fallback safety
    if best_h is None:
        n = 360
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = adjust_sum_exact(best_h, rng)
        best_c = compute_c5_naive(best_h)
        best_n = n

    # final safety clip and exact evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h, rng)
    final_c = compute_c5_naive(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
