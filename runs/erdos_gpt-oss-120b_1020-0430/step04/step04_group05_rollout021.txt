def subgradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    max_idx = np.where(corr >= max_val - 1e-12)[0]  # indices
    g = np.zeros_like(h)
    for idx in max_idx:
        k = idx - (n - 1)
        if k >= 0:
            k_int = int(k)
            term1 = np.ones(n)
            if k_int < n:
                term1[:n - k_int] = 1 - h[k_int:]
            term2 = np.zeros(n)
            if k_int < n:
                term2[k_int:] = h[:n - k_int]
        else:
            k_int = int(-k)
            term1 = np.ones(n)
            if k_int < n:
                term1[k_int:] = 1 - h[:n - k_int]
            term2 = np.zeros(n)
            if k_int < n:
                term2[:n - k_int] = h[k_int:]
        g += term1 - term2
    g /= len(max_idx)
    return g

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------
# Helper utilities

def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # simple equal redistribution with clipping
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        add = diff / n
        h += add
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        # fallback random adjustments
        for _ in range(max_iter):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def compute_c5(h):
    """C5 value for a discrete h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small epsilon from one entry to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def subgradient_and_corr(h):
    """Return subgradient of max lag and the full correlation array."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    max_idx = np.where(corr >= max_val - 1e-12)[0]
    g = np.zeros_like(h)

    for idx in max_idx:
        k = idx - (n - 1)
        if k >= 0:
            kk = int(k)
            term1 = np.ones(n)
            if kk < n:
                term1[: n - kk] = 1 - h[kk:]
            term2 = np.zeros(n)
            if kk < n:
                term2[kk:] = h[: n - kk]
        else:
            kk = int(-k)
            term1 = np.ones(n)
            if kk < n:
                term1[kk:] = 1 - h[: n - kk]
            term2 = np.zeros(n)
            if kk < n:
                term2[: n - kk] = h[kk:]
        g += term1 - term2

    g /= len(max_idx)
    return g, corr


def gradient_descent(h0, rng, deadline, max_iters=20000):
    """Projected subgradient descent with simple step‑size schedule."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)
    it = 0
    while time.time() < deadline and it < max_iters:
        g, corr = subgradient_and_corr(h)
        # step size: constant magnitude 0.2 (half of eta0) decaying slowly
        norm = np.linalg.norm(g) + 1e-12
        step_base = 0.2 / np.sqrt(1.0 + it * 0.001)  # magnitude ~0.2 initially
        h_new = h - step_base * g / norm

        # occasional jitter to avoid stagnation

    return best_h, best_c

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
