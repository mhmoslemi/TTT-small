def smooth_obj_and_jac(x, beta):
    r = np.correlate(x, 1 - x, mode='full')
    max_bc = beta * np.max(r)  # is okay.
    bc = beta * r
    max_bc = np.max(bc)
    expb = np.exp(bc - max_bc)
    sumexp = np.sum(expb)
    lse = max_bc + np.log(sumexp)
    f = lse / beta
    w = expb / sumexp
    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, 1 - x, mode='full')
    conv2 = np.convolve(w, x, mode='full')
    n = len(x)
    start = n - 1
    g = conv1[start:start + n] - conv2[start:start + n]
    return f, g

def worst_shift_gradient(h):
    n = h.shape[0]
    # compute correlation via np.correlate
    r = np.correlate(h, 1 - h, mode='full')
    k_max = np.argmax(r)
    # shift s = k_max - (n - 1). (because index offset)
    s = k_max - (n - 1)
    # compute gradient g
    g = np.zeros_like(h)
    # term1: (1 - h[i + s]) for i where i + s in [0, n-1]
    if s >= 0:
        g[:n-s] += 1 - h[s:]
        # term2: -h[i - s] for i where i - s >=0
        g[s:] -= h[:n-s]
    else:
        # s negative
        t = -s
        g[t:] += 1 - h[:n-t]
        g[:n-t] -= h[t:]
    return g, r[k_max]

    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, 1 - x, mode='full')
    conv2 = np.convolve(w, x, mode='full')
    n = x.size
    start = n - 1
    g = conv1[start:start + n] - conv2[start:start + n]
    return f, g

def slsqp_smooth_optimize(h0, beta, max_iter=300):
    """SLSQP on the smooth surrogate using analytic gradient."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {
        "type": "eq",
        "fun": lambda x: np.sum(x) - target,
        "jac": lambda x: np.ones_like(x),
    }

    def fun_jac(x):
        return smooth_obj_and_jac(x, beta)

    res = minimize(fun_jac, h0, method="SLSQP", jac=True,
                   bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, np.random.default_rng())
    return x

# ------------------------------------------------------------
# Min‑max subgradient descent (exact worst shift)
# ------------------------------------------------------------
def worst_shift_gradient(h):
    n = h.size
    r = np.correlate(h, 1 - h, mode='full')
    k_max = np.argmax(r)
    s = k_max - (n - 1)                     # shift (can be negative)
    grad = np.zeros_like(h)
    if s >= 0:
        grad[:n - s] += 1 - h[s:]
        grad[s:]   -= h[:n - s]
    else:
        t = -s
        grad[t:]   += 1 - h[:n - t]
        grad[:n - t] -= h[t:]
    return grad, r[k_max]

def minmax_descent(h, rng, max_iters=800, init_alpha=0.5):
    n = h.size
    target = n / 2.0
    cur = h.copy()
    for it in range(max_iters):
        grad, worst = worst_shift_gradient(cur)
        alpha = init_alpha / np.sqrt(it + 1)
        nxt = cur - alpha * grad
        nxt = np.clip(nxt, 0.0, 1.0)
        nxt = enforce_sum(nxt, target, rng, max_iter=200)
        new_worst = np.max(np.correlate(nxt, 1 - nxt, mode='full'))
        if new_worst < worst - 1e-12:
            cur = nxt
        elif rng.random() < 0.01:
            cur = nxt
    return cur

# ------------------------------------------------------------
# Simple stochastic hill‑climb and SA (reuse from parent)
# ------------------------------------------------------------
def hill_climb_multi(h_start, rng, max_iters=1500, init_step=0.5, n_trials=6):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        cand = perturb_h(cur_h, rng, scale=scale)
        c = compute_c5(cand)
        delta = c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = cand, c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

# ------------------------------------------------------------
# Main driver
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ----- candidate pool -------------------------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    tri = np.where(xs <= 1, xs, 2 - xs)
    candidates.append(enforce_sum(tri, n / 2.0, rng))

    cosine = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    candidates.append(enforce_sum(cosine, n / 2.0, rng))

    gauss = np.exp(-((xs - 1) ** 2) / (0.2 ** 2))
    candidates.append(enforce_sum(gauss, n / 2.0, rng))

    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ----- Phase A: smooth SLSQP with increasing beta --------------------
    for beta in (30, 100, 300, 1000):
        if time.time() - start_time > budget_s * 0.5:
            break
        h_ref = slsqp_smooth_optimize(best_h, beta, max_iter=300)
        c_ref = compute_c5(h_ref)
        if c_ref < best_c:
            best_c, best_h = c_ref, h_ref.copy()

    # ----- Phase B: exact min‑max descent ---------------------------------
    if time.time() - start_time < budget_s * 0.8:
        h_cur = minmax_descent(best_h, rng, max_iters=800, init_alpha=0.5)
        c_cur = compute_c5(h_cur)
        if c_cur < best_c:
            best_c, best_h = c_cur, h_cur.copy()

    # ----- Phase C: hill‑climb / SA for remaining budget ------------------
    while time.time() - start_time < budget_s * 0.95:
        h_hc, c_hc = hill_climb_multi(best_h, rng, max_iters=1000,
                                      init_step=0.4, n_trials=5)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
        h_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iter=2500,
                                         init_temp=5e-3, final_temp=1e-6,
                                         init_scale=0.3, final_scale=0.01)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        if rng.random() < 0.1:
            start_vec = random_feasible_vector(n, rng)
            best_h, best_c = hill_climb_multi(start_vec, rng,
                                              max_iters=600,
                                              init_step=0.5, n_trials=4)
        if time.time() - start_time > budget_s * 0.95:
            break

    # ----- Optional up‑sampling -------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.08:
        target_n = min(720, max(n * 2, n + 120))
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, target_n / 2.0, rng)
        for beta in (100, 300, 1000):
            if time.time() - start_time > budget_s:
                break
            up_h = slsqp_smooth_optimize(up_h, beta, max_iter=200)
        if time.time() - start_time < budget_s:
            up_h = minmax_descent(up_h, rng, max_iters=500, init_alpha=0.4)
        c_up = compute_c5(up_h)
        if c_up < best_c:
            best_c, best_h, n = c_up, up_h.copy(), target_n

    # ----- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
