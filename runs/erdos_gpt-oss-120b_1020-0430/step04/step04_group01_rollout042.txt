    n = h.shape[0]
    offset = n - 1
    size = 2 * n - 1
    delta = np.zeros(size, dtype=np.int16)
    # contribution for i (remove)
    delta_i = -1
    start1 = offset - i
    delta[start1:start1 + n] += delta_i * h
    start2 = i
    delta[start2:start2 + n] += delta_i * h[::-1]
    # contribution for j (add)
    delta_j = 1
    start1 = offset - j
    delta[start1:start1 + n] += delta_j * h
    start2 = j

    # squares
    delta[offset] += 2
    # cross terms
    shift = j - i
    delta[offset + shift] += -1
    delta[offset - shift] += -1
    return delta

import time, math, numpy as np
from scipy.optimize import minimize

# -------------------- helpers for continuous optimization --------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k}) via log‑sum‑exp."""
    n = x.size
    # full correlation of x with (1‑x)
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def compute_c5_float(h):
    """C5 for a (possibly non‑binary) step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * dx)


def c5_from_min_conv(min_conv, n):
    """C5 expressed via minimal self‑correlation."""
    return (2.0 / n) * ((n / 2.0) - min_conv)


# -------------------- binary utilities --------------------
def delta_swap(h, i, j):
    """Δ for swapping a 1 at i with a 0 at j (h binary)."""
    n = h.size
    off = n - 1
    size = 2 * n - 1
    delta = np.zeros(size, dtype=np.int16)

    # contribution of index i (removal)
    di = -1
    s1 = off - i
    delta[s1 : s1 + n] += di * h
    s2 = i
    delta[s2 : s2 + n] += di * h[::-1]

    # contribution of index j (addition)
    dj = 1
    s1 = off - j
    delta[s1 : s1 + n] += dj * h
    s2 = j
    delta[s2 : s2 + n] += dj * h[::-1]

    # squares at zero shift
    delta[off] += 2

    # cross terms
    shift = j - i
    delta[off + shift] += -1
    delta[off - shift] += -1
    return delta


def binary_simulated_annealing(h0, rng, deadline):
    """SA on binary vectors (exact half the entries are 1)."""
    n = h0.size
    off = n - 1
    h = h0.astype(np.int8).copy()
    conv = np.correlate(h, h, mode="full").astype(np.int16)
    cur_min = min(conv[:off].min(), conv[off + 1 :].min())
    best_min = cur_min
    best_h = h.copy()

    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))

    total_steps = 800_000
    T0, Tf = 3.0, 0.001
    step = 0
    while time.time() < deadline and step < total_steps:
        step += 1
        pos_i = rng.integers(len(ones))
        i = ones[pos_i]
        pos_j = rng.integers(len(zeros))
        j = zeros[pos_j]

        d = delta_swap(h, i, j)
        new_conv = conv + d
        left = new_conv[:off].min()
        right = new_conv[off + 1 :].min()
        new_min = left if left < right else right
        delta_min = new_min - cur_min

        frac = step / total_steps
        T = T0 * (Tf / T0) ** frac
        accept = delta_min > 0 or (T > 0 and rng.random() < math.exp(delta_min / T))

        if accept:
            # perform swap
            h[i] = 0
            h[j] = 1
            conv = new_conv
            cur_min = new_min

            # update lists in O(1)
            ones[pos_i] = ones[-1]
            ones.pop()
            zeros.append(i)

            zeros[pos_j] = zeros[-1]
            zeros.pop()
            ones.append(j)

            if cur_min > best_min:
                best_min = cur_min
                best_h = h.copy()
    return best_h, best_min


def perturb_h_continuous(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    budget = budget_s

    target_n = 360                     # even and reasonably small
    half = target_n // 2

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = init.size
        if n0 != target_n:
            if n0 % target_n == 0:
                factor = n0 // target_n
                init = init.reshape(-1, factor).mean(axis=1)
            else:
                xs0 = np.linspace(0, 2, n0, endpoint=False) + (2.0 / n0) / 2
                xs = np.linspace(0, 2, target_n, endpoint=False) + (2.0 / target_n) / 2
                init = np.interp(xs, xs0, init)
        h0 = (init > 0.5).astype(float)
    else:
        h0 = np.zeros(target_n, dtype=float)
        idx = rng.choice(target_n, half, replace=False)
        h0[idx] = 1.0

    # enforce exact half‑ones
    cur = int(h0.sum())
    if cur > half:
        excess = cur - half
        ones = np.flatnonzero(h0).tolist()
        for _ in range(excess):
            pos = rng.choice(ones)
            h0[pos] = 0.0
            ones.remove(pos)
    elif cur < half:
        deficit = half - cur
        zeros = np.flatnonzero(1 - h0).tolist()
        for _ in range(deficit):
            pos = rng.choice(zeros)
            h0[pos] = 1.0
            zeros.remove(pos)

    # ---- phase 1: binary simulated annealing ---------------------------
    deadline_sa = start + budget * 0.7
    bin_start = (h0 > 0.5).astype(np.int8)
    best_bin, best_min = binary_simulated_annealing(bin_start, rng, deadline_sa)
    best_c = c5_from_min_conv(best_min, target_n)

    # ---- phase 2: continuous polishing via SLSQP ------------------------
    h_cont = best_bin.astype(float)
    beta_seq = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0, 10000.0]
    for beta in beta_seq:
        if time.time() >= start + budget * 0.95:
            break
        try:
            h_opt = slsqp_optimize(h_cont, beta, max_iter=400)
            c_opt = compute_c5_float(h_opt)
            if c_opt < best_c:
                best_c = c_opt
                h_cont = h_opt.copy()
        except Exception:
            continue

    # ---- phase 3: simple hill‑climb on the continuous function ---------
    deadline_hc = start + budget
    best_h_cont = h_cont.copy()
    best_c_cont = compute_c5_float(best_h_cont)
    scale = 0.2
    while time.time() < deadline_hc:

        diff = (target_n / 2.0) - new_h.sum()

            new_h[idx] += add

        c_new = compute_c5_float(new_h)
        if c_new < best_c_cont:
            best_c_cont = c_new
            best_h_cont = new_h.copy()

                best_c = c_new

    # final cleanup
    final_h = np.clip(best_h_cont, 0.0, 1.0)
    final_c = float(best_c)
    return final_h.tolist(), final_c, target_n
