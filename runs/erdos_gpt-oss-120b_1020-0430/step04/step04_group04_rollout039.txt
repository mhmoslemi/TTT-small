def build_basis(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    basis = []
    # constant 0.5
    basis.append(np.full(n, 0.5))
    # triangle centered at c
    for c in (1.0, 0.5, 1.5):
        tri = np.maximum(1 - np.abs(x - c), 0.0)
        basis.append(tri)
    # raised cosine centered at c with support width 2
    for c in (1.0, 0.5, 1.5):
        d = np.abs(x - c)
        cos = np.where(d <= 1, (1 + np.cos(np.pi * d)) / 2.0, 0.0)
        basis.append(cos)
    # linear ramps
    basis.append(x / 2.0)        # ramp up
    basis.append(1 - x / 2.0)    # ramp down
    # sine series
    for f in (1, 2, 3):
        s = 0.5 + 0.5 * np.sin(np.pi * f * x)
        c = 0.5 + 0.5 * np.cos(np.pi * f * x)
        basis.append(s)
        basis.append(c)
    # Ensure each basis is within [0,1] (already)
    return basis

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())  # next power of two >= L
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 5000) -> np.ndarray:
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:

            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start: np.ndarray, rng: np.random.Generator,
                     max_iters: int = 2000, init_step: float = 0.5, n_trials: int = 5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def simulated_annealing(h0: np.ndarray, rng: np.random.Generator,
                        max_iter: int, init_temp: float, final_temp: float,
                        init_scale: float = 0.5, final_scale: float = 0.05):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h, cur_c = best_h.copy(), best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

    # linear ramps
    B.append(x / 2.0)          # up
    B.append(1.0 - x / 2.0)    # down

    # sine & cosine series (f = 1,2,3)
    for f in (1, 2, 3):
        B.append(0.5 + 0.5 * np.sin(np.pi * f * x))
        B.append(0.5 + 0.5 * np.cos(np.pi * f * x))

    return B


# ---------- main run ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # Phase 1: DE on a modest grid
    n0 = 180
    target0 = n0 / 2.0
    basis = build_basis(n0)
    dim = len(basis)
    Bmat = np.stack(basis, axis=0)          # (dim, n0)

    # DE hyper‑parameters
    pop_size = max(20, dim * 8)
    F = 0.8
    CR = 0.9

    # initialise population on simplex
    pop = rng.random((pop_size, dim))
    pop = np.clip(pop, 0.0, 1.0)
    pop = pop / pop.sum(axis=1, keepdims=True)

    def fitness_vec(pop_vec):
        h_all = np.dot(pop_vec, Bmat)      # (pop, n0)
        h_all = np.clip(h_all, 0.0, 1.0)
        cs = np.empty(pop_vec.shape[0])
        for i in range(pop_vec.shape[0]):
            cs[i] = compute_c5(h_all[i])
        err = np.abs(h_all.sum(axis=1) - target0)
        return cs + 1e3 * err

    fitness = fitness_vec(pop)
    best_idx = int(np.argmin(fitness))
    best_w = pop[best_idx].copy()
    best_c = fitness[best_idx]

    # DE main loop (time‑bounded)
    de_budget = total_budget * 0.20
    while time.time() - start_time < de_budget:
        for i in range(pop_size):
            if time.time() - start_time >= de_budget:
                break
            # select a,b,c distinct from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            s = mutant.sum()
            if s == 0:
                mutant = pop[i].copy()
            else:
                mutant = mutant / s
            # crossover
            cross = rng.random(dim) < CR
            if not np.any(cross):
                cross[rng.integers(dim)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            s = trial.sum()
            if s == 0:
                trial = pop[i].copy()
            else:
                trial = trial / s
            # evaluate trial
            h_trial = np.dot(trial, Bmat)
            h_trial = np.clip(h_trial, 0.0, 1.0)
            c_trial = compute_c5(h_trial)
            err_trial = abs(h_trial.sum() - target0)
            f_trial = c_trial + 1e3 * err_trial
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_c:
                    best_c = f_trial
                    best_w = trial.copy()
        # optional early exit if no improvement for many generations could be added

    # Build best vector from DE
    best_h = np.dot(best_w, Bmat)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target0, rng)

    # Phase 2: local refinement on n0 grid
    phase2_budget = total_budget * 0.35
    while time.time() - start_time < de_budget + phase2_budget:
        # start from a perturbed best or random feasible vector
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n0, rng)
        # hill climb
        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=3500, init_step=0.5, n_trials=8)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
        # simulated annealing
        h_sa, c_sa = simulated_annealing(
            best_h, rng, max_iter=9000,
            init_temp=1e-2, final_temp=1e-5,
            init_scale=0.5, final_scale=0.02
        )
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        # SLSQP ladder (stop if budget exhausted)
        for beta in (50, 200, 800, 2000, 5000):
            if time.time() - start_time >= de_budget + phase2_budget:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=600)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # Phase 3: upscale and final refinement
    elapsed = time.time() - start_time
    remaining = total_budget - elapsed
    if remaining > total_budget * 0.10:
        # upscale to n=360 (or higher if time permits)
        n_up = 360
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(n_up) + 0.5) * (2.0 / n_up)
        up_h = np.interp(new_x, old_x, best_h)
        target_up = n_up / 2.0
        up_h = enforce_sum(up_h, target_up, rng)

        # modest hill climb + SA on finer grid
        up_h, _ = hill_climb_multi(up_h, rng, max_iters=2500, init_step=0.4, n_trials=6)
        up_h, _ = simulated_annealing(
            up_h, rng,
            max_iter=6000,
            init_temp=5e-3, final_temp=1e-5,
            init_scale=0.4, final_scale=0.02,
        )
        # SLSQP refinements
        for beta in (200, 800, 2000, 5000):
            if time.time() - start_time >= total_budget:
                break
            up_h = slsqp_optimize(up_h, beta, max_iter=500)
        # evaluate and possibly adopt
        c_up = compute_c5(up_h)
        if c_up < best_c:
            best_c, best_h, n0 = c_up, up_h.copy(), n_up

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n0 / 2.0, rng)
    final_c = compute_c5(best_h)

    # incorporate provided initial_h_values if it turns out better
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)
        init_n = init_h.size
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = enforce_sum(init_h, init_n / 2.0, rng)
        init_c = compute_c5(init_h)
        if init_c < final_c:
            best_h, final_c, n0 = init_h, init_c, init_n

    return best_h.tolist(), float(final_c), int(n0)
