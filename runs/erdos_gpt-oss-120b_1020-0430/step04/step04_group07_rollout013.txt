    lo = v.min() - upper  # or lower?

        s = w.sum()

def differential_evolution_optimize(n, rng, max_evals, popsize=20, F=0.8, CR=0.9):
    # Initialize population
    target = n/2
    population = [random_feasible_vector(n, rng) for _ in range(popsize)]
    fitness = [compute_c5(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best = population[best_idx].copy()
    best_f = fitness[best_idx]

    evals = popsize
    while evals < max_evals:
        for i in range(popsize):
            # generate a,b,c distinct from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            # Clip or project to feasible region
            mutant = project_simplex_bounds(mutant, target, 0., 1.)
            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, population[i])
            # Ensure feasibility (project again)
            trial = project_simplex_bounds(trial, target, 0., 1.)
            f = compute_c5(trial)
            evals += 1
            if f < fitness[i]:
                population[i] = trial
                fitness[i] = f
                if f < best_f:
                    best_f = f
                    best = trial.copy()
            if evals >= max_evals:
                break
        # optionally shuffle
    return best, best_f

import time, numpy as np, math
from scipy.optimize import minimize

# ----- utility functions -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (entries in [0,1], sum=n/2)."""
    h = rng.random(n)
    return enforce_sum(h, rng)


# ----- gradient / hill‑climb helpers ------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


# ----- soft‑max SLSQP refinement ----------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:

            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

        n = 256
        best_h = random_feasible_vector(n, rng)
    best_c = compute_c5(best_h)
    target = n / 2.0

    # ----- differential evolution ----------------------------------------
    popsize = 30
    F, CR = 0.8, 0.9
    # initial population (include current best)
    population = [best_h.copy()]
    while len(population) < popsize:
        population.append(random_feasible_vector(n, rng))
    population = np.array(population)
    fitness = np.array([compute_c5(ind) for ind in population])
    idx = int(np.argmin(fitness))
    best_h, best_c = population[idx].copy(), fitness[idx]

    de_time = 0.6 * budget_s
    while time.time() - start < de_time:
        for i in range(popsize):
            if time.time() - start >= de_time:
                break
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = enforce_sum(trial, rng)
            f = compute_c5(trial)
            if f < fitness[i]:
                population[i] = trial
                fitness[i] = f
                if f < best_c:
                    best_c, best_h = f, trial.copy()

    # ----- guided hill‑climb ---------------------------------------------
    hill_time = 0.15 * budget_s
    while time.time() - start < hill_time:
        h_new, c_new = guided_hill_climb(best_h, rng, max_iters=1200, step_scale=0.7)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if time.time() - start >= hill_time:
            break

    # ----- SLSQP refinement -----------------------------------------------
    slsqp_time = 0.15 * budget_s
    for beta in (20., 80., 200., 800., 3000., 8000.):
        if time.time() - start >= slsqp_time:
            break
        maxit = 500 if beta < 200 else 300
        best_h = slsqp_optimize(best_h, beta, rng, max_iter=maxit)
        c = compute_c5(best_h)
        if c < best_c:
            best_c = c

    # ----- final gradient tidy‑up -----------------------------------------
    if time.time() - start < budget_s:
        best_h = gradient_refine(best_h, rng, step=0.01, max_iter=12)
        final_c = compute_c5(best_h)
        if final_c < best_c:
            best_c = final_c

    # ----- optional up‑sampling if time permits ---------------------------
    if time.time() - start < 0.95 * budget_s:
        for factor in (2, 3):
            n_up = n * factor
            if n_up > 384:
                continue
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = enforce_sum(h_up, rng)

            # a quick hill‑climb at higher resolution
            h_up, c_up = guided_hill_climb(h_up, rng, max_iters=1500, step_scale=0.6)
            if c_up < best_c:
                best_c, best_h, n = c_up, h_up.copy(), n_up

            # brief SLSQP passes
            for beta in (20., 200., 2000., 8000.):
                if time.time() - start >= budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=400)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_c, best_h, n = c_up, h_up.copy(), n_up

            # final tidy‑up
            h_up = gradient_refine(h_up, rng, step=0.01, max_iter=8)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_c, best_h, n = c_up, h_up.copy(), n_up

    # ----- final sanity check ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
