def compute_corr_fft(h):
    n = h.size
    b = 1.0 - h
    b_rev = b[::-1]
    N = 1 << ((2*n - 1).bit_length())
    # pad h and b_rev
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(b_rev, N)
    conv = np.fft.irfft(fa * fb, N)[:2*n-1]
    return conv

def compute_c5(h):
    conv = compute_corr_fft(h)
    return np.max(conv) * (2.0 / h.size)

def targeted_move(h, rng, k_star, scale):
    n = h.size
    if k_star >= 0:
        i_range = np.arange(n - k_star)
        i2 = i_range + k_star
    else:
        i_range = np.arange(-k_star, n)
        i2 = i_range + k_star  # negative shift
    # masks
    low = (h[i_range] < 0.5) & (h[i2] < 0.5)
    high = (h[i_range] > 0.5) & (h[i2] > 0.5)
    if not low.any() or not high.any():
        return None
    i_low = rng.choice(i_range[low])
    i_high = rng.choice(i_range[high])
    # compute max epsilon to move towards 0.5 (both low increase, both high decrease)
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k_star],
        h[i_high] - 0.5,
        h[i_high + k_star] - 0.5,
    )
    if max_eps <= 0:
        return None
    eps = (rng.random() - 0.5) * 2 * max_eps * scale

    # Actually we add eps to low pair (both positions) and subtract eps from high pair (both positions)

def targeted_move(h, rng, k_star, scale):
    n = h.size
    if k_star >= 0:
        max_i = n - k_star
        i_vals = np.arange(max_i)
        i2_vals = i_vals + k_star
    else:
        max_i = n + k_star  # k_star negative
        i_vals = np.arange(-k_star, n)
        i2_vals = i_vals + k_star
    # now low/high masks
    low = (h[i_vals] < 0.5) & (h[i2_vals] < 0.5)
    high = (h[i_vals] > 0.5) & (h[i2_vals] > 0.5)
    if not low.any() or not high.any():
        return None
    i_low = rng.choice(i_vals[low])
    i_high = rng.choice(i_vals[high])
    # compute max epsilon
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k_star],
        h[i_high] - 0.5,
        h[i_high + k_star] - 0.5,
    )
    if max_eps <= 0:
        return None
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    return i_low, i_low + k_star, i_high, i_high + k_star, eps

def anneal_optimize(h0, rng, time_limit, p_target=0.2):
    h = h0.copy()
    n = h.size
    c_cur, shift_cur = compute_c5_and_shift(h)
    best_h = h.copy()
    best_c = c_cur
    t0 = time.time()
    temp = 0.01
    final_temp = 1e-5
    scale = 0.2
    min_scale = 1e-4
    decay = 0.9995  # for both temp and scale
    while time.time() - t0 < time_limit:
        if rng.random() < p_target:
            move = targeted_move(h, rng, shift_cur, scale)
            if move is None:
                # fallback to pair
                move = "pair"
        else:
            move = "pair"

        if move == "pair":
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
            eps = (rng.random() - 0.5) * 2 * max_eps * scale
            if eps == 0:
                continue
            h[i] -= eps
            h[j] += eps
            c_new, shift_new = compute_c5_and_shift(h)
            dE = c_new - c_cur
            if dE < 0 or rng.random() < np.exp(-dE / temp):
                c_cur, shift_cur = c_new, shift_new
                if c_cur < best_c:
                    best_c = c_cur; best_h = h.copy()
            else:
                # revert
                h[i] += eps
                h[j] -= eps
        else:
            i1, i2, i3, i4, eps = move
            # apply
            h[i1] += eps
            h[i2] += eps
            h[i3] -= eps
            h[i4] -= eps
            c_new, shift_new = compute_c5_and_shift(h)
            dE = c_new - c_cur
            if dE < 0 or rng.random() < np.exp(-dE / temp):
                c_cur, shift_cur = c_new, shift_new
                if c_cur < best_c:

            else:

                h[i1] -= eps
                h[i2] -= eps
                h[i3] += eps
                h[i4] += eps
        # update temperature & scale
        temp = max(final_temp, temp * decay)
        scale = max(min_scale, scale * decay)

import time

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    h = np.clip(x, 0.0, 1.0)
    corr = compute_corr_fft(h)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

        if time.time() - start > budget_s * 0.9:
            break
        try:
            h_ref = slsqp_optimize(h_best, beta, max_iter=400)
        except Exception:
            continue
        c_ref = compute_c5_and_shift(h_ref)[0]
        if c_ref < c_best:
            c_best, h_best = c_ref, h_ref.copy()

    # ---- phase 3: up‑sample and fine‑tune (if time permits) ----
    if time.time() - start < budget_s * 0.95:
        new_n = min(2400, n * 2)
        if new_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
            h_up = np.interp(new_x, old_x, h_best)
            h_up = project_box_sum(h_up, new_n / 2.0)

            remaining = budget_s - (time.time() - start)
            up_time = max(0.0, remaining * 0.4)
            h_up_best, c_up_best = anneal_optimize(h_up, rng, up_time, p_target=0.2)

            if c_up_best < c_best:
                c_best, h_best = c_up_best, h_up_best.copy()
            for beta in beta_list:
                if time.time() - start > budget_s:
                    break
                try:
                    h_ref = slsqp_optimize(h_best, beta, max_iter=300)
                except Exception:
                    continue
                c_ref = compute_c5_and_shift(h_ref)[0]
                if c_ref < c_best:
                    c_best, h_best = c_ref, h_ref.copy()

    # ---- final cleanup ----
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_box_sum(h_best, h_best.size / 2.0)
    final_c = compute_c5_and_shift(h_best)[0]
    return h_best.tolist(), float(final_c), int(h_best.size)
