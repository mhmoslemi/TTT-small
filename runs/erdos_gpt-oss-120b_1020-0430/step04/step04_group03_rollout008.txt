def improved_greedy_optimize(h, max_steps=2000, tol=1e-12):
    n = len(h)
    for step in range(max_steps):
        # compute corr and worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        s_max = np.argmax(corr) - (n - 1)
        cur_val = corr.max()
        # compute gradient for worst shift
        grad = gradient_shift(h, s_max)
        # pos indices: grad>0 and h>0 (able to decrease)
        # neg indices: grad<0 and h<1 (able to increase)
        pos_idx = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg_idx = np.where((grad < -1e-12) & (h < 1 - 1e-12))[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            break
        # choose best i and j? Use max grad for pos, min grad for neg.
        i = pos_idx[np.argmax(grad[pos_idx])]
        j = neg_idx[np.argmin(grad[neg_idx])]
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 1e-15:
            break
        # compute a = corr (array length 2n-1)
        a = corr
        # compute b_s = grad_s_j - grad_s_i
        # Need grad_s for each shift; compute gradient_shift for i and j individually.
        grad_i = gradient_shift_one(h, i) # custom: effect of h_i on corr_s
        grad_j = gradient_shift_one(h, j)
        b = grad_j - grad_i
        # binary search for optimal eps in [0, max_eps]
        lo, hi = 0.0, max_eps
        for _ in range(30):
            mid = (lo + hi) / 2.0
            vals = a + b * mid
            max_val = vals.max()
            max_idx = vals.argmax()
            slope = b[max_idx]
            if slope > 0:
                hi = mid
            else:
                lo = mid
        eps_opt = (lo + hi) / 2.0
        # if small improvement: ensure new max < cur_val - 1e-12
        new_max = (a + b * eps_opt).max()
        if new_max < cur_val - 1e-12:
            # Update h
            h[i] -= eps_opt
            h[j] += eps_opt
            # Continue
        else:
            # No improvement with line search; maybe try have other pos/neg combos? Could break.
            # Could try random pos/neg pair.
            # For simplicity break.
            break
    return h

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):

    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    """Exact discrete C5 for a vector h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def gradient_shift(h, s):
    """Gradient of correlation at shift s w.r.t each component."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

def grad_vec_for_index(h, idx):
    """Gradient of correlation for *all* shifts w.r.t h[idx]."""
    n = h.size
    s = np.arange(-(n - 1), n)  # shifts
    grad = np.zeros_like(s, dtype=float)
    plus = idx + s
    mask = (plus >= 0) & (plus < n)
    grad[mask] += 1.0 - h[plus[mask]]
    minus = idx - s
    mask = (minus >= 0) & (minus < n)
    grad[mask] -= h[minus[mask]]
    return grad

        best_eps = 0.0

        for i in top_pos:
            grad_i = grad_vec_for_index(h, i)
            for j in top_neg:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 1e-15:
                    continue
                grad_j = grad_vec_for_index(h, j)
                delta = grad_j - grad_i

                # binary search for optimal epsilon
                lo, hi = 0.0, max_eps
                for _ in range(22):
                    mid = (lo + hi) * 0.5
                    vals = corr + delta * mid
                    max_val = vals.max()
                    slope = delta[vals.argmax()]
                    if slope > 0:
                        hi = mid
                    else:
                        lo = mid
                eps_opt = (lo + hi) * 0.5
                new_max = (corr + delta * eps_opt).max()
                if new_max < best_new - 1e-12:
                    best_new = new_max
                    best_i, best_j = i, j
                    best_eps = eps_opt

        if best_i is not None:
            h[best_i] -= best_eps
            h[best_j] += best_eps
            continue
        # no improving pair among top‑k → try a random pair
        i, j = pos[rng.integers(0, pos.size)], neg[rng.integers(0, neg.size)]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-15:
            break
        grad_i = grad_vec_for_index(h, i)
        grad_j = grad_vec_for_index(h, j)
        delta = grad_j - grad_i
        lo, hi = 0.0, max_eps
        for _ in range(22):
            mid = (lo + hi) * 0.5
            vals = corr + delta * mid
            max_val = vals.max()
            slope = delta[vals.argmax()]
            if slope > 0:
                hi = mid
            else:
                lo = mid
        eps_opt = (lo + hi) * 0.5
        new_max = (corr + delta * eps_opt).max()
        if new_max < cur_max - 1e-12:
            h[i] -= eps_opt
            h[j] += eps_opt
            continue
        break
    return h

# ---------- SLSQP polishing ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    exp_term = np.exp(bc - m)
    sm = np.log(exp_term.sum()) + m
    return sm / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

# ---------- up‑sampling ----------
def upscale_linear(coarse, n_full):
    """Linear interpolation of a coarse vector to n_full points on [0,2]."""
    n_coarse = coarse.size
    xs_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xs_full = (np.arange(n_full) + 0.5) * (2.0 / n_full)
    up = np.interp(xs_full, xs_coarse, coarse)
    up = np.clip(up, 0.0, 1.0)
    target = n_full / 2.0
    up = enforce_sum(up, target, np.random.default_rng())
    return up

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- establish coarse grid -----
    if "initial_h_values" in globals():
        h_full = np.array(initial_h_values, dtype=float)
        n_full = h_full.size
        # choose a factor so that coarse size ≤ 200
        factor = max(2, n_full // 200)
        while n_full % factor != 0:
            factor += 1
        n_coarse = n_full // factor
        h_coarse = h_full.reshape(n_coarse, factor).mean(axis=1)
    else:
        n_full = 180
        factor = 1
        n_coarse = n_full
        h_coarse = random_feasible_vector(n_coarse, rng)

    # ensure coarse feasibility
    h_coarse = enforce_sum(np.clip(h_coarse, 0.0, 1.0), n_coarse / 2.0, rng)

    best_coarse = h_coarse.copy()
    best_c_coarse = compute_c5(best_coarse)

    # ----- coarse‑grid search -----
    while time.time() - start < budget_s * 0.55:
        # generate start
        if rng.random() < 0.5:
            start_vec = perturb_random(best_coarse, rng, scale=0.4)
        else:
            start_vec = random_feasible_vector(n_coarse, rng)
        # local improvement
        local = greedy_optimize_enhanced(start_vec, max_steps=1200, k=2)
        c_local = compute_c5(local)
        if c_local < best_c_coarse - 1e-12:
            best_coarse, best_c_coarse = local, c_local

        # occasional SLSQP polishing
        if rng.random() < 0.25:
            for beta in (200, 800, 2000, 5000):
                if time.time() - start > budget_s * 0.95:
                    break
                try:
                    refined = slsqp_optimize(best_coarse, beta, max_iter=800)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c_coarse - 1e-12:
                    best_coarse, best_c_coarse = refined, c_ref

    # ----- upscale to full resolution -----
    h_full = upscale_linear(best_coarse, n_full)
    best_h = h_full.copy()
    best_c = compute_c5(best_h)

    # ----- fine‑grid search -----
    while time.time() - start < budget_s * 0.90:
        if rng.random() < 0.6:
            start_vec = perturb_random(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n_full, rng)
        local = greedy_optimize_enhanced(start_vec, max_steps=800, k=2)
        c_local = compute_c5(local)
        if c_local < best_c - 1e-12:
            best_h, best_c = local, c_local

        if rng.random() < 0.2:
            for beta in (500, 2000, 5000):
                if time.time() - start > budget_s * 0.98:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=800)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c - 1e-12:
                    best_h, best_c = refined, c_ref

    # ----- final micro‑search -----
    while time.time() - start < budget_s * 0.99:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
