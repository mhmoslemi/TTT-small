def proj_bounds_simplex(v, target, max_iter=1000, eps=1e-10):
    # clip to [0,1]
    v = np.clip(v, 0, 1)
    # iterate water-filling
    for _ in range(max_iter):
        cur_sum = v.sum()
        diff = target - cur_sum
        if abs(diff) <= eps:
            break
        # active indices where 0 < v < 1
        act = np.where((v > 0) & (v < 1))[0]
        if act.size == 0:
            # all at bounds, just adjust by distributing diff equally but may violate bounds, fallback to simple scaling
            v = np.clip(v + diff / len(v), 0, 1)
        else:
            add = diff / act.size
            v[act] += add
            # clip again
            v = np.clip(v, 0, 1)
    return v

def gradient_descent_smooth(h, beta_schedule, step0=0.2, max_iter_per_beta=200, target_sum=None, time_deadline=None):
    n = h.size
    target = target_sum if target_sum is not None else n / 2.0
    step = step0
    for beta in beta_schedule:
        for _ in range(max_iter_per_beta):
            if time_deadline is not None and time.time() > time_deadline:
                return h
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            diff = corr - max_corr
            # compute weights stable
            exp_vals = np.exp(beta * diff)  # diff <=0
            sum_exp = exp_vals.sum()
            if sum_exp == 0:
                weights = np.zeros_like(exp_vals)
            else:
                weights = exp_vals / sum_exp
            # compute gradient
            grad = np.zeros_like(h)
            # mask to reduce loops
            mask = weights > 1e-12
            idxs = np.where(mask)[0] - (n - 1)
            for s in idxs:
                w = weights[s + n - 1]
                if s >= 0:
                    grad[:n-s] += w * (1 - h[s:])
                    grad[s:] -= w * h[:n-s]
                else:
                    t = -s
                    grad[t:] += w * (1 - h[:n-t])
                    grad[:n-t] -= w * h[t:]
            # gradient descent step
            h -= step * grad
            # project to [0,1] and correct sum
            np.clip(h, 0.0, 1.0, out=h)
            h = adjust_sum(h, target)
            # optionally adapt step size
        step *= 0.7  # reduce after each beta
    return h

def binary_hill_climb(h_bin, max_iter, deadline):
    n = h_bin.size
    cur_h = h_bin.astype(float).copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    it = 0
    while it < max_iter and time.time() < deadline:
        i = random choice from ones
        j = random choice from zeros
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        c = compute_c5(cur_h)
        if c < cur_c:
            cur_c = c
            # update ones/zeros
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
    return best_h, best_c

import time
import numpy as np

# -------------------------------------------------
# Helper utilities
# -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector onto the box [0,1] and enforce Σv = target."""
    diff = target - v.sum()
    n = v.size
    # simple water‑filling loop
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = np.random.randint(0, n)
        if diff > 0:
            add = min(diff, 1.0 - v[i])
        else:
            add = max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v


def smooth_max_and_grad(h: np.ndarray, beta: float):
    """Smooth maximum (log‑sum‑exp) of the correlation and its gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    diff = corr - max_corr                     # ≤ 0
    # stable exponentials
    exp_vals = np.exp(beta * diff)
    sum_exp = exp_vals.sum()
    weights = exp_vals / sum_exp

    # gradient via slice updates (only for weights that matter)
    grad = np.zeros_like(h)
    mask = weights > 1e-12
    shifts = np.where(mask)[0] - (n - 1)       # actual shift values
    for s in shifts:
        w = weights[s + n - 1]
        if s >= 0:
            grad[: n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[: n - s]
        else:
            t = -s
            grad[t:] += w * (1.0 - h[: n - t])
            grad[: n - t] -= w * h[t:]
    smooth = max_corr + np.log(sum_exp) / beta
    return smooth, grad


def gradient_descent_smooth(
    h: np.ndarray,
    beta_schedule,
    step0: float = 0.3,
    max_iter_per_beta: int = 100,
    target_sum: float = None,
    deadline: float = None,
):
    """Projected gradient descent on the smooth‑max surrogate."""
    n = h.size
    target = target_sum if target_sum is not None else n / 2.0
    step = step0
    for beta in beta_schedule:
        for _ in range(max_iter_per_beta):
            if deadline is not None and time.time() > deadline:
                return h
            _, grad = smooth_max_and_grad(h, beta)
            h -= step * grad
            np.clip(h, 0.0, 1.0, out=h)
            adjust_sum(h, target)
        step *= 0.7                      # shrink step as beta grows
    return h


def binary_hill_climb(h_bin: np.ndarray, max_iter: int, deadline: float):
    """Simple swap‑based hill climbing on a binary vector."""
    n = h_bin.size
    cur = h_bin.astype(float)
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    it = 0
    while it < max_iter and time.time() < deadline:
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        # perform swap
        cur[i] = 0.0
        cur[j] = 1.0
        c = compute_c5(cur)
        if c < cur_c:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best = cur.copy()
        else:
            # revert
            cur[i] = 1.0
            cur[j] = 0.0
        it += 1
    return best, best_c
