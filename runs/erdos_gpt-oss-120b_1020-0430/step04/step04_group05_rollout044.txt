import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def project_h(h, target, iters=60):
    """Project h onto the box [0,1] with sum equal to target via a simple bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        proj = np.clip(h + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    proj = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if needed
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with exact sum = n/2 via projection."""
    h = rng.random(n)
    return project_h(h, n / 2.0)

def compute_c5_and_corr(h):
    """Return (C5, full correlation array) for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = corr.max() * (2.0 / h.size)
    return c5, corr

def compute_c5(h):
    """Return only C5."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)

def transfer_eps(h, i, j, eps):
    """Transfer eps from i to j (preserving sum). Assumes eps respects bounds."""
    h[i] -= eps
    h[j] += eps
    return h

def perturb_transfer(h, rng, scale=0.2):
    """Randomly move a small epsilon from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, lag, top_frac=0.15, scale=0.5):
    """Bias transfer towards indices that contribute most to the worst lag."""
    n = h.size
    if lag >= n or lag <= -n:
        return perturb_transfer(h, rng, scale)
    if lag >= 0:
        valid = np.arange(n - lag)
        contrib = h[valid] * (1.0 - h[valid + lag])
    else:
        lag = -lag
        valid = np.arange(lag, n)
        contrib = h[valid] * (1.0 - h[valid - lag])
    if contrib.size == 0:
        return perturb_transfer(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    # with high prob move from a high to a low contributor
    if rng.random() < 0.8:
        j = rng.choice(bot_idx)
    else:
        j = rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_transfer(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def subgradient_step(h, target, rng, step0, decay, max_iters, time_end):
    """Projected subgradient descent aimed at reducing the worst lag."""
    n = h.size
    step = step0
    for it in range(max_iters):
        if time.time() > time_end:
            break
        # correlation and worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        # gradient for this lag
        grad = np.zeros_like(h)
        if lag >= 0:
            i = np.arange(n - lag)
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] -= h[i]
        else:
            lag_abs = -lag
            i = np.arange(lag_abs, n)
            grad[i] += (1.0 - h[i - lag_abs])
            grad[i - lag_abs] -= h[i]
        # update
        step = step0 * (decay ** it)
        new_h = h - step * grad
        new_h = project_h(new_h, target)
        # simple acceptance (always accept if improves)
        new_c5 = compute_c5(new_h)
        cur_c5 = corr.max() * (2.0 / n)
        if new_c5 < cur_c5 or rng.random() < 0.02:
            h = new_h
    return h

# ---------- SLSQP with softmax ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run SLSQP on the smooth max objective with given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = project_h(x, CURRENT_TARGET)
    return x

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    t_end = t_start + budget_s

    # ----- initial vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_h(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- Phase 1: coarse stochastic search (70% of time) -----
    phase1_end = t_start + 0.70 * budget_s
    beta_sequence = [10, 30, 100, 300, 800, 2000, 5000]
    while time.time() < phase1_end:
        # random restart or perturb best
        if rng.random() < 0.6:
            start = perturb_transfer(best_h, rng, scale=0.4)
        else:
            start = random_feasible_vector(n, rng)

        # Simulated annealing style loop
        cur_h = start
        cur_c, cur_corr = compute_c5_and_corr(cur_h)
        iters = 0
        max_sa_iters = 20000
        while iters < max_sa_iters and time.time() < phase1_end:
            iters += 1
            # temperature schedule
            T = 0.02 * (0.5 ** (iters / max_sa_iters))
            # decide perturbation
            lag = np.argmax(cur_corr) - (n - 1)
            if rng.random() < 0.7:
                new_h = targeted_perturb(cur_h, rng, lag, top_frac=0.2, scale=0.6)
            else:
                new_h = perturb_transfer(cur_h, rng, scale=0.6)
            new_c, new_corr = compute_c5_and_corr(new_h)
            dE = new_c - cur_c
            if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
                cur_h, cur_c, cur_corr = new_h, new_c, new_corr
                if new_c < best_c:
                    best_h, best_c = new_h.copy(), new_c
        # SLSQP refinement ladder
        for beta in beta_sequence:
            if time.time() >= phase1_end:
                break
            cand = slsqp_optimize(best_h, beta, rng, max_iter=400)
            cand_c = compute_c5(cand)
            if cand_c < best_c:
                best_h, best_c = cand, cand_c

    # ----- Phase 2: upscale to finer grid (if time permits) -----
    if time.time() < t_end - 30.0:  # keep some buffer
        n_fine = 720
        # linear interpolation onto finer grid
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        fine_x = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
        fine_h = np.interp(fine_x, old_x, best_h)
        fine_h = project_h(fine_h, n_fine / 2.0)

        # Subgradient descent on fine grid
        fine_h = subgradient_step(
            fine_h,
            n_fine / 2.0,
            rng,
            step0=0.12,
            decay=0.9998,
            max_iters=20000,
            time_end=t_end - 15.0,
        )
        fine_c = compute_c5(fine_h)
        if fine_c < best_c:
            best_h, best_c, n = fine_h, fine_c, n_fine

        # SLSQP refinement on fine grid
        beta_fine = [30, 150, 600, 2500, 8000]
        for beta in beta_fine:
            if time.time() >= t_end - 10.0:
                break
            cand = slsqp_optimize(best_h, beta, rng, max_iter=350)
            cand_c = compute_c5(cand)
            if cand_c < best_c:
                best_h, best_c = cand, cand_c

    # ----- Final brief local improvement -----
    # simple hill‑climb with decreasing step size
    cur_h = best_h.copy()
    cur_c = best_c
    step = 0.25
    for _ in range(3000):
        if time.time() >= t_end:
            break
        new_h = perturb_transfer(cur_h, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        step *= 0.9996

    # final projection safeguard
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_h(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
