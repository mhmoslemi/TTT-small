import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Shift and clip x so that each entry lies in [lo,hi] and sum equals target."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def hill_climb_targeted(h_start, rng, max_iters=300, init_step=0.4, n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.shape[0]
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        # shift that currently attains the max
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k_star = idx - (n - 1)

        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- smooth‑max gradient optimisation ----------
def gradient_step(h, beta, dx):
    """One projected gradient step on the β‑softmax of the overlap."""
    n = h.size
    y = 1.0 - h
    corr = np.correlate(h, y, mode="full")        # length 2n‑1
    max_corr = corr.max()
    w = np.exp(beta * (corr - max_corr))
    w /= w.sum()
    # convolution with same mode aligns the centre (shift 0) with each index
    conv_y = np.convolve(y, w, mode="same")
    conv_h_rev = np.convolve(h, w[::-1], mode="same")
    grad = dx * (conv_y - conv_h_rev)
    return grad, corr.max()


def sgd_optimize(h0, betas, iters_per_beta, step0, rng, time_limit):
    """Projected stochastic gradient descent with increasing β."""
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()

    for beta, n_iter in zip(betas, iters_per_beta):
        step = step0
        for it in range(n_iter):
            if time.time() - start > time_limit:
                return best_h, best_c
            grad, cur_max = gradient_step(h, beta, dx)
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_box_sum(h_new, n / 2.0)
            c_val = cur_max * dx
            if c_val < best_c - 1e-12:
                best_c, best_h = c_val, h_new.copy()
            h = h_new
            step *= 0.998  # gentle decay
    return best_h, best_c


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)
