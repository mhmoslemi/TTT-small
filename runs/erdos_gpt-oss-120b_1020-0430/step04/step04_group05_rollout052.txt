def binary_sa(h_start, rng, max_iters, time_start, time_budget):

    best_h, best_c = h.copy(), cur_c

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j: return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0: return h.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):

    if lag >= n: return perturb_h(h, rng, scale)

    if lag >= 0:

        contrib = h[valid] * (1.0 - h[valid + lag])
    else:
        lag_abs = -lag
        valid = np.arange(lag_abs, n)
        contrib = h[valid] * (1.0 - h[valid + lag])

    k = max(1, int(len(valid) * top_frac))

def random_balanced_binary(n, rng):
    ones = np.full(n//2, 1, dtype=float)
    zeros = np.full(n - n//2, 0, dtype=float)
    arr = np.concatenate([ones, zeros])
    rng.shuffle(arr)
    return arr

                best_c, best_h = c_ref, h_ref.copy()

    # ---- Stage 3 : Upsample + continuous SA + SLSQP ----
    remaining = budget_s - (time.time() - t0)
    if remaining > budget_s * 0.15:
        n_fine = n_base * 2
        old_x = (np.arange(n_base) + 0.5) * (2.0 / n_base)
        new_x = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
        h_fine = np.interp(new_x, old_x, best_h)
        h_fine = enforce_sum(h_fine, n_fine / 2.0, rng)

        t_start = time.time()
        sa_time = remaining * 0.5
        cur_h_f, cur_c_f = continuous_sa(
            h_fine, rng, max_iters=12000, time_start=t_start, time_budget=t_start + sa_time
        )
        if cur_c_f < best_c:
            best_c, best_h = cur_c_f, cur_h_f.copy()
            n_base = n_fine

        beta_fine = [20, 100, 500, 2000, 8000]
        for beta in beta_fine:
            if time.time() - t0 >= budget_s:
                break
            h_ref_f = slsqp_optimize_one_beta(cur_h_f, beta, rng)
            c_ref_f = compute_c5(h_ref_f)
            if c_ref_f < best_c:
                best_c, best_h = c_ref_f, h_ref_f.copy()
                n_base = n_fine

    # ---- Stage 4 : final hill‑climb ----
    cur_h = best_h.copy()
    cur_c = best_c
    step = 0.3
    for _ in range(2000):
        if time.time() - t0 > budget_s:
            break
        nh = perturb_h(cur_h, rng, scale=step)
        nc = compute_c5(nh)
        if nc < cur_c:
            cur_c, cur_h = nc, nh
            if nc < best_c:
                best_c, best_h = nc, nh.copy()
        step *= 0.9995

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
