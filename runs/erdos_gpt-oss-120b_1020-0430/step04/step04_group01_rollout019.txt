def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        total = cap.sum()
        if total > 0:
            h += diff * cap / total
    else:
        cap = h
        total = cap.sum()
        if total > 0:
            h += diff * cap / total
    # clip again (might go slightly out due to numerics)
    h = np.clip(h, 0.0, 1.0)
    # final small correction using loop (few steps)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # simple loop 
        for _ in range(10):
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                idx = np.argmax(1 - h)
                add = min(diff, 1 - h[idx])
                h[idx] += add
            else:
                idx = np.argmax(h)
                sub = min(-diff, h[idx])
                h[idx] -= sub
            diff = target - h.sum()
    return h
```

def gradient_step(h, beta, step_size, target, dx):
    corr = correlate_full(h)
    # compute w
    bc = beta * corr
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()
    # compute conv w * h
    L = 1 << ((len(w) + len(h) - 1).bit_length())
    fw = np.fft.rfft(w, L)
    fh = np.fft.rfft(h, L)
    conv = np.fft.irfft(fw * fh, L)
    # indices for i
    start = len(w)//2  # (n-1)
    conv_i = conv[start:start+len(h)]
    # conv_rev: w reversed
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, L)
    conv_rev = np.fft.irfft(fw_rev * fh, L)
    conv_rev_i = conv_rev[start:start+len(h)]
    grad = dx * (1.0 - conv_i - conv_rev_i)
    h_new = h - step_size * grad
    h_new = project_to_feasible(h_new, target)
    return h_new
```

Note: computing FFT of w and w_rev each step; can reuse w_rev = w[::-1] and compute rfft. L same size.

def smooth_descent(h, target, dx, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = len(h)
    betas = [1, 5, 20, 80, 300, 1000, 3000, 8000, 20000]
    for beta in betas:
        step = 1.0
        # compute L for convolutions (size for w+h)
        L = 1 << ((3*n - 2).bit_length())
        while time.time() < deadline:
            # compute corr and w
            corr = correlate_full(best_h)  # length 2n-1
            bc = beta * corr
            max_bc = bc.max()
            exp_bc = np.exp(bc - max_bc)
            w = exp_bc / exp_bc.sum()
            # FFT
            fw = np.fft.rfft(w, L)
            fh = np.fft.rfft(best_h, L)
            conv = np.fft.irfft(fw * fh, L)
            start = n - 1
            conv_i = conv[start:start+n]
            # rev
            w_rev = w[::-1]
            fw_rev = np.fft.rfft(w_rev, L)
            conv_rev = np.fft.irfft(fw_rev * fh, L)
            conv_rev_i = conv_rev[start:start+n]
            grad = dx * (1.0 - conv_i - conv_rev_i)
            h_new = best_h - step * grad
            h_new = project_to_feasible(h_new, target)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_h, best_c = h_new, c_new
                # increase step slightly
                step = min(step * 1.02, 2.0)
            else:
                step = max(step * 0.98, 1e-4)
            # break if step too small
            if step < 1e-5:
                break
        # after each beta, maybe do a few random perturb steps
        for _ in range(10):
            h_rand = perturb_h(best_h, rng, scale=0.3)
            c_rand = compute_c5(h_rand)
            if c_rand < best_c:
                best_h, best_c = h_rand, c_rand
        # early exit if time almost up
        if time.time() + 5.0 >= deadline:
            break
    return best_h, best_c

def binary_sa(start_bin, rng, deadline):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 50000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]

    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]

import time
import numpy as np

# --------------------------------------------------------------
# Feasibility utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Correlation / objective utilities
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Projection onto hyperplane + box
# --------------------------------------------------------------
def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        total = cap.sum()
        if total > 0:
            h += diff * cap / total
    else:
        cap = h
        total = cap.sum()
        if total > 0:
            h += diff * cap / total
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(5):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argmax(1 - h)
            add = min(diff, 1 - h[idx])
            h[idx] += add
        else:
            idx = np.argmax(h)
            sub = min(-diff, h[idx])
            h[idx] -= sub
        diff = target - h.sum()
    return h


# --------------------------------------------------------------
# Smoothed gradient descent (soft‑max surrogate)
# --------------------------------------------------------------
def smooth_descent(h, target, dx, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = len(best_h)
    # FFT length for convolutions of w (len 2n-1) with h (len n)
    L = 1 << ((3 * n - 2).bit_length())
    betas = [1, 5, 20, 80, 300, 1000, 3000, 8000, 20000]
    for beta in betas:
        step = 0.5
        while time.time() < deadline:
            corr = correlate_full(best_h)                # length 2n-1
            bc = beta * corr
            max_bc = bc.max()
            exp_bc = np.exp(bc - max_bc)
            w = exp_bc / exp_bc.sum()                    # soft‑max weights
            # convolution w * h  (gives Σ_k w_k h_{i‑k})
            fw = np.fft.rfft(w, L)
            fh = np.fft.rfft(best_h, L)
            conv = np.fft.irfft(fw * fh, L)
            start = n - 1
            conv_i = conv[start:start + n]
            # convolution with reversed w (gives Σ_k w_k h_{i+k})
            w_rev = w[::-1]
            fw_rev = np.fft.rfft(w_rev, L)
            conv_rev = np.fft.irfft(fw_rev * fh, L)
            conv_rev_i = conv_rev[start:start + n]
            grad = dx * (1.0 - conv_i - conv_rev_i)
            h_new = best_h - step * grad
            h_new = project_to_feasible(h_new, target)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_h, best_c = h_new, c_new
                step = min(step * 1.02, 2.0)
            else:
                step = max(step * 0.98, 1e-5)
            if step < 1e-5:
                break
        # few random nudges after each β
        for _ in range(10):
            h_try = perturb_h(best_h, rng, scale=0.3)
            c_try = compute_c5(h_try)
            if c_try < best_c:
                best_h, best_c = h_try, c_try
        if time.time() + 5.0 >= deadline:
            break
    return best_h, best_c


# --------------------------------------------------------------
# Simple hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h, target, dx, deadline, rng, max_iters=20000, step0=0.3):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = step0
    n = best_h.size
    it = 0
    while time.time() < deadline and it < max_iters:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_h, best_c = new_h, c
        it += 1
        if it % 2000 == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c


# --------------------------------------------------------------
# Binary simulated annealing (swap version)
# --------------------------------------------------------------
def binary_sa(start_bin, rng, deadline):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 50000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

        bin_start = np.zeros(n)
        bin_start[idx] = 1.0
        bin_h, bin_c = binary_sa(bin_start, rng, deadline)
        if bin_c < best_c:
            best_h, best_c = bin_h, bin_c
        if time.time() < deadline:
            best_h, best_c = hill_climb(best_h, target, dx, deadline, rng,

    # Final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)
