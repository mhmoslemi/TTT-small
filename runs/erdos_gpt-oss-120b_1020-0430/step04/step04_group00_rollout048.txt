def simulated_annealing(h, rng, deadline):
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    cur_h = h.copy()
    cur_c = best_c
    n = len(h)
    total_time = deadline - time.time()
    start_time = time.time()
    # initial temperature scaling factor: set T0 = 0.02 maybe.
    T0 = 0.02
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linear cooling:
        T = T0 * max(0.0, 1.0 - elapsed / total_time)
        # step size = T * 0.5 maybe.
        max_step = T * 0.7  # adjust
        # choose two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # compute allowed delta range
        hi, hj = cur_h[i], cur_h[j]
        delta_min = max(-max_step, -hi, hj - 1)  # ensure hi+delta >=0, hi+delta <=1, hj-delta >=0, hj-delta <=1
        delta_max = min(max_step, 1 - hi, hj)  # ensure hi+delta <=1, hj-delta >=0
        if delta_min >= delta_max:
            continue
        delta = rng.uniform(delta_min, delta_max)
        # apply move
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta
        # compute new C5
        new_c = compute_c5_fft(cur_h)
        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = hi
            cur_h[j] = hj
    return best_h, best_c

def compute_c5(h):
    n = h.size
    m = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * G, m)[:2*n-1]
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng, max_iter=2000):

    for _ in range(max_iter):
        if abs(diff) < 1e-12:

            diff -= add

        # small correction adjust by distributing uniformly

        # clamp
        np.clip(h, 0.0, 1.0, out=h)

import time

# ---------- utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """max overlap using FFT (linear correlation)."""
    n = h.size
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * np.conj(G), m)[: 2 * n - 1]
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000) -> np.ndarray:
    """Redistribute to achieve Σh = n/2 exactly (preserving bounds)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # tiny residual fix
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

        best_n = n

    # final safety adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
