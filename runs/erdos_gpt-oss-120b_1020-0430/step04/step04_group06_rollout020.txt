    # h may be float; enforce sum exactly target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])

def enforce_sum(h, target, rng, max_iter=5000):
    """Project h onto the simplex Σh = target with 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def compute_c5(h):
    """Exact C5 for a step function h (dx = 2 / n)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)


def max_overlap_and_shift(h):
    """Return (max_val, shift) where shift = index-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift


def pair_transfer_optimize(
    h0,
    rng,
    max_iters=5000,
    time_limit=30.0,
    init_factor=0.5,
    decay=0.999,
):
    """Local search moving mass from i to i+shift to lower max overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c

def binary_anneal(n, rng, time_limit, start_time, start_h=None, max_iters=300000):
    """Simulated annealing on binary vectors with sum = n/2."""
    dx = 2.0 / n
    target_ones = n // 2

    # initialise h (float 0.0/1.0) and auxiliary structures
    h = np.zeros(n, dtype=np.float64)

    if start_h is not None:
        # binary rounding of start_h
        ones_idx = np.where(start_h >= 0.5)[0]
        if ones_idx.size > target_ones:
            ones_idx = ones_idx[:target_ones]
        elif ones_idx.size < target_ones:
            missing = target_ones - ones_idx.size
            cand = np.setdiff1d(np.arange(n), ones_idx, assume_unique=True)
            add = rng.choice(cand, size=missing, replace=False)
            ones_idx = np.concatenate([ones_idx, add])
        h[ones_idx] = 1.0
    else:
        perm = rng.permutation(n)
        ones_idx = perm[:target_ones]
        h[ones_idx] = 1.0

    # build fast lookup structures
    ones = np.empty(target_ones, dtype=np.int64)
    zeros = np.empty(target_ones, dtype=np.int64)
    pos_one = np.full(n, -1, dtype=np.int64)
    pos_zero = np.full(n, -1, dtype=np.int64)

    for p, idx in enumerate(ones_idx):
        ones[p] = idx
        pos_one[idx] = p
    zeros_idx = np.setdiff1d(np.arange(n), ones_idx, assume_unique=True)
    for p, idx in enumerate(zeros_idx):
        zeros[p] = idx
        pos_zero[idx] = p

    # initial statistics
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    cur_max_val = float(corr[max_idx])
    cur_shift = max_idx - (n - 1)
    cur_c5 = cur_max_val * dx
    best_h = h.copy()
    best_c5 = cur_c5

    # annealing schedule
    initial_T = max(0.6, cur_c5)
    T_end = 1e-5

    it = 0
    while (time.time() - start_time < time_limit) and it < max_iters:
        it += 1

        # ----- choose i (1) and j (0) -----
        shift = cur_shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift

        mask = (h[i_range] == 1.0) & (h[j_range] == 0.0)
        cand_i = i_range[mask]
        if cand_i.size > 0:
            i = int(cand_i[rng.integers(cand_i.size)])
        else:
            i = int(ones[rng.integers(ones.size)])
        j = int(zeros[rng.integers(zeros.size)])

        # ----- perform swap (keep structures consistent) -----
        pi = pos_one[i]
        pj = pos_zero[j]

        # update arrays and positions
        ones[pi] = j
        zeros[pj] = i
        pos_one[j] = pi
        pos_one[i] = -1
        pos_zero[i] = pj
        pos_zero[j] = -1

        # swap values
        h[i] = 0.0
        h[j] = 1.0

        # ----- evaluate new state -----
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        max_idx_new = int(np.argmax(corr_new))
        max_val_new = float(corr_new[max_idx_new])
        shift_new = max_idx_new - (n - 1)

        delta = max_val_new - cur_max_val  # raw change in max overlap (before scaling)

        # temperature based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / time_limit
        T = initial_T * ((T_end / initial_T) ** frac)

        accept = False
        if delta <= 0:
            accept = True
        else:
            prob = math.exp(-delta * dx / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_max_val = max_val_new
            cur_shift = shift_new
            cur_c5 = cur_max_val * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            ones[pi] = i
            zeros[pj] = j
            pos_one[i] = pi
            pos_one[j] = -1
            pos_zero[j] = pj
            pos_zero[i] = -1

    return best_h, best_c5

            best_h, best_c5 = h_ref, c_ref

    # Final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
