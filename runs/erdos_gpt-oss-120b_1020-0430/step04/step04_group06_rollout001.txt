def max_corr_and_shift(h):
    n = h.shape[0]
    L = 2*n - 1
    A = np.fft.rfft(h, n=L)
    B = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(A * np.conj(B), n=L)
    idx = int(np.argmax(corr))
    # shift mapping: lag = idx - (n-1)
    shift = idx - (n - 1)
    return corr[idx], shift

def pair_transfer_multi_optimize(h, max_iters, time_limit):
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    cur_h = h.copy()
    cur_c = best_c
    factor = 0.5
    for iter in range(max_iters):
        if time.time() - start > time_limit:
            break
        # find shift with max corr
        max_val, shift = max_corr_and_shift(cur_h)
        cur_c_val = max_val * (2.0 / cur_h.size)
        # early update best
        if cur_c_val < best_c:
            best_c = cur_c_val
            best_h = cur_h.copy()
        # compute product for this shift
        n = cur_h.size
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = 0.5
            continue
        prod = cur_h[i_range] * (1.0 - cur_h[j_range])

        m = min(5, i_range.size)
        top_idx = np.argpartition(-prod, m-1)[:m]  # indices of top m
        # transfer from each selected i to j by eps = factor * max_eps (scaled)
        for idx in top_idx:

    # final best
    return best_h, best_c

def max_corr_and_shift(h):
    n = h.shape[0]
    L = 2 * n - 1
    A = np.fft.rfft(h, n=L)
