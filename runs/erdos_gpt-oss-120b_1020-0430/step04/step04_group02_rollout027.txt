def softmax_max_and_grad(h, beta):
    n = h.size
    # compute overlaps c
    c = compute_overlap_vector(h) # length 2n-1
    # compute normalized weights
    # compute log-sum-exp
    # using stable approach
    bc = beta * c
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    Z = np.sum(exp_bc)
    w = exp_bc / Z
    # compute conv1 = convolution of w with h
    conv1 = np.convolve(w, h, mode='full')
    # conv2 uses reversed w
    w_rev = w[::-1]

import time
import numpy as np

# ---------- basic utilities ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Project x onto [lo,hi] while fixing the sum to target."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def compute_overlap_vector(h):
    """Full vector of overlaps C_k = ∫ h(x)(1-h(x+k))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * (2.0 / h.size)


def compute_c5(h):
    return np.max(compute_overlap_vector(h))


# ---------- targeted hill‑climb (same as previous version) ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


def hill_climb_targeted(
    h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25
):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)  # shift corresponding to the worst overlap
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- smooth‑max gradient descent ----------
def softmax_max_and_grad(h, beta):
    """Return smooth max of overlaps and its gradient w.r.t. h."""
    n = h.size
    delta = 2.0 / n
    c = compute_overlap_vector(h)                # length 2n‑1
    bc = beta * c
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    Z = np.sum(exp_bc)
    w = exp_bc / Z                               # weights, sum to 1

    # convolution terms for Σ_k w_k h_{i±k}
    conv1 = np.convolve(w, h, mode="full")      # Σ_k w_k h_{i‑k}
    conv2 = np.convolve(w[::-1], h, mode="full")  # Σ_k w_k h_{i+k}
    conv_sum = conv1[n - 1 : 2 * n - 1] + conv2[n - 1 : 2 * n - 1]

    # linear part Σ_k w_k I(i+k∈[0,n‑1])
    cum_w = np.cumsum(w)
    linear_sum = np.empty(n, dtype=np.float64)
    for i in range(n):
        start = n - 1 - i
        end = 2 * n - 2 - i
        if start == 0:
            linear_sum[i] = cum_w[end]
        else:
            linear_sum[i] = cum_w[end] - cum_w[start - 1]

    grad = delta * (linear_sum - conv_sum)

    smooth_val = (np.log(Z) + m) / beta
    return smooth_val, grad


def gradient_descent_smooth_max(h0, rng, betas, iters_per_beta, step0=0.5):
    """Run projected gradient descent on the smooth‑max objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0

    for beta in betas:
        for _ in range(iters_per_beta):
            _, g = softmax_max_and_grad(h, beta)
            new_h = h - step * g
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = project_box_sum(new_h, target)
            new_c = compute_c5(new_h)
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = new_h.copy()
                # keep step size on a healthy scale
                step = max(step, step0)
            else:
                step *= 0.5
                if step < 1e-6:
                    step = 1e-6
            h = new_h
    return best_h, best_c

    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise base vector ----
    try:
        h0 = np.array(initial_h_values, dtype=float)  # may not exist
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    except Exception:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ---- generate a small diverse pool ----
    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    for f in (1, 2, 3):
        cand = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        cand = project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0)
        candidates.append(cand)
        cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        cand = project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0)
        candidates.append(cand)

    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(project_box_sum(mask, n / 2.0))

    mask2 = np.zeros(n)
    start = n // 4
    mask2[start : start + n // 2] = 1.0
    candidates.append(project_box_sum(mask2, n / 2.0))

    for _ in range(4):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = np.inf
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # ---- allocate time for the main optimisation ----
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed)

    # smooth‑max schedule
    betas = [0.5, 2, 5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]

    # conservative iteration budget (≈500 iters / s)
    approx_iters = int(remaining * 500)
    if approx_iters < len(betas):
        iters_per_beta = 1
    else:
        iters_per_beta = max(1, approx_iters // len(betas))
        if iters_per_beta > 500:
            iters_per_beta = 500

    # gradient descent on smooth max
    best_h, best_c = gradient_descent_smooth_max(best_h, rng, betas, iters_per_beta, step0=0.7)

    # ---- local refinement with targeted hill‑climb ----
    best_h, best_c = hill_climb_targeted(
        best_h, rng, max_iters=1200, init_step=0.3, n_trials=6, p_target=0.3
    )

    # ---- optional up‑sampling if time permits ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.12:
        target_n = min(720, max(540, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_box_sum(h_up, target_n / 2.0)

            # a shorter gradient phase on the finer grid
            iters_up = max(1, iters_per_beta // 2)
            best_h, best_c = gradient_descent_smooth_max(
                h_up, rng, betas, iters_up, step0=0.5
            )

            # final local hill‑climb
            best_h, best_c = hill_climb_targeted(
                best_h, rng, max_iters=800, init_step=0.3, n_trials=6, p_target=0.3
            )
            n = target_n

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
