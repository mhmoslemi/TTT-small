def subgradient_descent(h0, deadline):
    h = h0.copy()
    n = h.size
    target = n/2
    best_h = h.copy()
    best_c = compute_c5(h)
    iteration = 0
    initial_step = 0.05   # maybe adjust after experiments.
    while time.time() < deadline:
        # compute correlation
        conv = correlate_full(h)
        # find index of max
        k_idx = np.argmax(conv)
        k = k_idx - (n - 1)  # shift corresponding to index in full correlation.
        # compute gradient
        grad = gradient_of_corr(k, h)
        # step size schedule: step = initial_step / np.sqrt(iteration+1)
        step = initial_step / np.sqrt(iteration + 1)
        # update
        h_new = h - step * grad
        # clip
        np.clip(h_new, 0.0, 1.0, out=h_new)
        # enforce sum
        h_new = enforce_sum(h_new, target, rng)  # need rng; can pass globally.
        h = h_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    return best_h, best_c

def project_sum_box(h, target):
    diff = target - h.sum()
    # iterate adjusting until diff small
    max_iter = 10
    for _ in range(max_iter):
        # find indices where h can be moved in direction of diff
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # can increase entries <1
            idx = np.where(h < 1 - 1e-12)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            # limit inc to not exceed 1
            inc = np.minimum(inc, 1 - h[idx])
            h[idx] += inc
            diff = target - h.sum()
        else:
            # diff < 0: decrease entries >0
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            dec = diff / idx.size
            dec = np.maximum(dec, -h[idx])  # ensure not go negative
            h[idx] += dec

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def adjust_sum(h, target, rng, tol=1e-12):
    """Fast small‑adjustment of sum; falls back to enforce_sum if needed."""
    diff = target - h.sum()
    if abs(diff) < tol:
        return h
    for _ in range(200):
        i = rng.integers(0, h.size)
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = max(diff, -h[i])
            h[i] += sub
            diff -= sub
        if abs(diff) < tol:
            break
    if abs(diff) >= tol:
        h = enforce_sum(h, target, rng)
    return h

def hill_climb(h_start, rng, deadline):
    """Greedy hill‑climbing accepting only improvements."""
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    it = 0
    step = 0.3
    decay = 0.99
    while time.time() < deadline:
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        it += 1
        if it % 2000 == 0:
            step = max(step * decay, 1e-5)
    return best_h, best_c


def upsample_and_refine(h, factor, deadline, rng):
    """Upsample by integer factor and finish with hill‑climbing."""
    h_fine = np.repeat(h, factor)
    n_fine = h_fine.size
    target = n_fine / 2.0
    np.clip(h_fine, 0.0, 1.0, out=h_fine)
    h_fine = enforce_sum(h_fine, target, rng)
    return hill_climb(h_fine, rng, deadline)

# Main entry point

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    end = start + budget_s

    # ------------------------------------------------------------------
    # Initial feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        if abs(h0.sum() - target) > 1e-12:
            h0 = enforce_sum(h0, target, rng)
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ------------------------------------------------------------------
    # Phase 1: long sub‑gradient descent
    # ------------------------------------------------------------------
    phase1_end = start + 0.5 * budget_s
    best_h, best_c = subgradient_descent(best_h, phase1_end, rng)

    # ------------------------------------------------------------------
    # Phase 1b: short random restarts (if time permits)
    # ------------------------------------------------------------------
    while time.time() + 15 < end:
        h_rand = random_feasible_vector(n, rng)
        h_loc, c_loc = subgradient_descent(h_rand, time.time() + 15, rng)
        if c_loc < best_c:
            best_h, best_c = h_loc, c_loc

    # ------------------------------------------------------------------
    # Phase 2: hill climbing refinement
    # ------------------------------------------------------------------
    phase2_end = start + 0.8 * budget_s
    best_h, best_c = hill_climb(best_h, rng, phase2_end)

    # ------------------------------------------------------------------
    # Phase 3: optional up‑sampling to finer grid
    # ------------------------------------------------------------------
    if time.time() < end:
        # try a single factor‑2 refinement if size permits
        if best_h.size * 2 <= 1440:
            best_h, best_c = upsample_and_refine(
                best_h, 2, end, rng
            )

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
