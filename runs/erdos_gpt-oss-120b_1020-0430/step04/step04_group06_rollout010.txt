def gradient_descent(h0, rng, max_total_iters=1000, beta_start=10.0, beta_end=2000.0, beta_steps=5, lr_start=0.2, lr_decay=0.9):
    h = h0.copy()
    n = h.size
    target = n/2.0
    # schedule: create list of betas increasing geometrically
    betas = np.geomspace(beta_start, beta_end, num=beta_steps)
    iters_per = max_total_iters // beta_steps
    lr = lr_start
    for beta in betas:
        for _ in range(iters_per):
            # compute gradient
            grad = grad_smooth_objective(h, beta)
            # gradient descent step
            h_new = h - lr * grad
            # enforce bounds
            h_new = np.clip(h_new, 0.0, 1.0)
            # fix sum
            diff = target - h_new.sum()
            if abs(diff) > 1e-12:
                # simple correction
                h_new += diff / n
                h_new = np.clip(h_new, 0.0, 1.0)
                # If still off due to clipping, adjust with enforce_sum small loops
                if abs(h_new.sum() - target) > 1e-10:
                    h_new = enforce_sum(h_new, target, rng, max_iter=200)
            # accept if objective improved
            cur_val = smooth_objective(h, beta)
            new_val = smooth_objective(h_new, beta)
            if new_val < cur_val:
                h = h_new
        lr = max(lr * lr_decay, 1e-4)
    return h

def grad_smooth_objective(h, beta):
    n = h.size
    sum_h = h.sum()
    conv = np.correlate(h, h, mode='full')
    c = sum_h - conv
    bc = beta * c
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / np.sum(exp_bc)   # shape (2n-1)
    # compute convs
    conv1 = np.convolve(h, w, mode='full')
    conv2 = np.convolve(h, w[::-1], mode='full')
    start = n-1
    t1 = conv1[start:start+n]
    t2 = conv2[start:start+n]
    grad = 1.0 - (t1 + t2)
    return grad

import time
import numpy as np

# -------------------------------------------------
# Helper functions
# -------------------------------------------------
def enforce_sum(h, target, rng, max_iter=5000):
    """Project h onto {0≤h≤1, sum(h)=target} by simple random adjustments."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        # final uniform correction with clipping
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Uniform random feasible vector with sum=n/2 and 0≤h≤1."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(2000):
            if abs(diff) < 1e-14:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Local search moving mass from i to i+shift to lower max overlap."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    # final verification
    final_c = compute_c5(best_h)
    return best_h, final_c

# Smooth surrogate objective and its gradient
# -------------------------------------------------
def smooth_objective(h, beta):
    """Log-sum-exp surrogate of max overlap (scaled by 1/beta)."""
    sum_h = h.sum()
    conv = np.correlate(h, h, mode="full")
    c = sum_h - conv                     # length 2n-1
    bc = beta * c
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def grad_smooth_objective(h, beta):
    """Gradient of the smooth surrogate."""
    n = h.size
    sum_h = h.sum()
    conv = np.correlate(h, h, mode="full")
    c = sum_h - conv
    bc = beta * c
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / np.sum(exp_bc)          # weights, sum to 1, length 2n-1

    # two convolutions with w and reversed w
    conv1 = np.convolve(h, w, mode="full")
    conv2 = np.convolve(h, w[::-1], mode="full")
    start = n - 1
    t1 = conv1[start:start + n]
    t2 = conv2[start:start + n]
    grad = 1.0 - (t1 + t2)                # because Σ w_s = 1
    return grad


def gradient_descent(h0, rng, max_total_iters=3000,
                    beta_start=10.0, beta_end=2000.0,
                    beta_steps=5, lr_start=0.5, lr_decay=0.9):
    """Smoothed gradient descent with gradually increasing β."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    betas = np.geomspace(beta_start, beta_end, num=beta_steps)
    iters_per = max(1, max_total_iters // beta_steps)
    lr = lr_start
    for beta in betas:
        for _ in range(iters_per):
            grad = grad_smooth_objective(h, beta)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            # simple uniform correction for sum, then clip again
            diff = target - h_new.sum()
            if abs(diff) > 1e-12:
                h_new += diff / n
                h_new = np.clip(h_new, 0.0, 1.0)
                # fall back to enforce_sum if still off
                if abs(h_new.sum() - target) > 1e-10:
                    h_new = enforce_sum(h_new, target, rng, max_iter=200)
            h = h_new
        lr = max(lr * lr_decay, 1e-4)
    return h


# -------------------------------------------------
# Main entry point
# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ----- build a modest pool of diverse starts ------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    # structured seeds
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n / 2.0, rng))
    for f in (1, 2, 3):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            candidates.append(enforce_sum(h, n / 2.0, rng))

    # binary masks
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))
    mask = np.zeros(n)
    start = n // 4
    mask[start : start + n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))
    mask = np.arange(n) % 2
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    # random vectors
    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    # ----- pick best among the pool ------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- iterative optimisation loop ----------------------------------
    while time.time() < deadline * 0.85:
        # choose a start point
        if rng.random() < 0.7:
            start_h = perturb_h(best_h, rng, scale=0.15)
        else:
            start_h = random_feasible_vector(n, rng)

        # smooth gradient descent
        h_opt = gradient_descent(
            start_h,
            rng,
            max_total_iters=3000,
            beta_start=10.0,
            beta_end=2000.0,
            beta_steps=5,
            lr_start=0.5,
            lr_decay=0.9,
        )

        # pair‑transfer fine‑tuning with remaining time slice
        rem = deadline - time.time()
        if rem < 5.0:
            break
        pt_time = min(30.0, rem * 0.5)
        h_opt, c_opt = pair_transfer_optimize(
            h_opt, rng, max_iters=8000, time_limit=pt_time, init_factor=0.5
        )

        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- final clean‑up ------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
