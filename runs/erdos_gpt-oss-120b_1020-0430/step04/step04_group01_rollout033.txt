def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

            i = rng.integers(0, h.size)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# --------------------------------------------------------------
# Correlation utilities
# --------------------------------------------------------------
def correlate_full(h):
    n = h.shape[0]
    L = 2 * n - 1
    M = 1 << ((L).bit_length())
    fh = np.fft.rfft(h, M)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, M)
    conv = np.fft.irfft(fh * fr, M)[:L]
    return conv

def compute_c5(h):
    n = h.size
    L = 2 * n - 1
    M = 1 << ((L).bit_length())
    fh = np.fft.rfft(h, M)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, M)
    conv = np.fft.irfft(fh * fr, M)[:L]
    return conv.max() * (2.0 / n)

# --------------------------------------------------------------
# SLSQP (smooth max) utilities
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# --------------------------------------------------------------
# Simulated annealing on binary vectors
# --------------------------------------------------------------
def binary_sa(start_bin, rng, deadline):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 20000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

# --------------------------------------------------------------
# Simple greedy hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    L = 2 * n - 1
    M_all = 1 << ((3 * n - 2).bit_length())  # >= n+L-1

    betas = [1.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]

    # ---- repeated gradient‑descent restarts -----------------------------
    while time.time() - start_time < budget * 0.75:
        # start point for this run
        if rng.random() < 0.6:
            h = perturb_h(best_h, rng, scale=0.2)
        else:
            h = random_feasible_vector(n, rng)

        for beta in betas:
            step = 0.5 / np.sqrt(beta)
            remaining = budget * 0.75 - (time.time() - start_time)
            if remaining <= 0:
                break
            max_iters = int(min(2000, remaining * 30))
            for it in range(max_iters):
                FH = np.fft.rfft(h, M_all)
                FR = np.fft.rfft((1.0 - h)[::-1], M_all)
                corr = np.fft.irfft(FH * FR, M_all)[:L]

                maxc = beta * corr.max()
                w = np.exp(beta * corr - maxc)
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.ones_like(w) / w.size

                fw = np.fft.rfft(w, M_all)
                conv_full1 = np.fft.irfft(np.conj(fw) * FH, M_all)[: n + L - 1]
                conv1 = conv_full1[L - 1 : L - 1 + n]

                w_rev = w[::-1]
                fw_rev = np.fft.rfft(w_rev, M_all)
                conv_full2 = np.fft.irfft(np.conj(fw_rev) * FH, M_all)[: n + L - 1]
                conv2 = conv_full2[L - 1 : L - 1 + n]

                grad = 1.0 - conv1 - conv2
                h = h - step * grad
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, n / 2.0, rng)

                if (it & 15) == 0:
                    c_cur = compute_c5(h)
                    if c_cur < best_c:
                        best_c, best_h = c_cur, h.copy()
                step *= 0.9995

            if time.time() - start_time >= budget * 0.75:
                break

        # short hill‑climb on the current best
        if time.time() - start_time < budget * 0.8:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=1500, start_step=0.15)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

        # binary simulated annealing
        if time.time() - start_time < budget * 0.85:
            idx = np.argsort(-best_h)[: n // 2]
            bin_start = np.zeros(n)
            bin_start[idx] = 1.0
            deadline = start_time + budget * 0.85
            bin_h, bin_c = binary_sa(bin_start, rng, deadline)
            if bin_c < best_c:
                best_c, best_h = bin_c, bin_h.copy()

        if time.time() - start_time < budget * 0.9:
            h_hc2, c_hc2 = hill_climb(best_h, rng, max_iters=2000, start_step=0.1)
            if c_hc2 < best_c:
                best_c, best_h = c_hc2, h_hc2.copy()

    # ---- final polishing (optional) ------------------------------------
    if time.time() - start_time < budget * 0.97:
        try:
            h_opt = slsqp_optimize(best_h, beta=8000.0, max_iter=200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
        except Exception:
            pass

    # ---- final projection and return ------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
