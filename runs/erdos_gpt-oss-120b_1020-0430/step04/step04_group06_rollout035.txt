    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)


def enforce_sum(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust so that Σvec = len(vec)/2."""
    vec = np.clip(vec, 0.0, 1.0)
    target = vec.size / 2.0
    diff = target - vec.sum()
    if abs(diff) < 1e-14:
        return vec
    # simple iterative correction
    for _ in range(2000):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, vec.size)
        add = min(diff, 1.0 - vec[i]) if diff > 0 else max(diff, -vec[i])
        vec[i] += add
        diff = target - vec.sum()
    return vec


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h with Σh = n/2 and 0≤h≤1."""
    h = np.full(n, 0.5)
    # random pairwise swaps to introduce variance
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def max_overlap_shift(h: np.ndarray):
    """Return (idx_of_max, shift) for correlation of h with (1-h)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (h.size - 1)
    return idx, shift

# Main optimisation routine
# -------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # -------------------- initialise h --------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    else:
        n = 360
        # try a few random starts and keep the best
        best_h = None
        best_c = float("inf")
        for _ in range(6):
            cand = random_feasible_vector(n, rng)
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand.copy()
        h = best_h
    # keep track of the overall best
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # -------------------- parameters --------------------
    end_time = start + budget_s - 1.0          # 1 s safety margin
    factor_init = 0.9
    decay = 0.9995
    min_factor = 1e-6

    factor = factor_init
    no_improve = 0
    iteration = 0
    # pre‑compute the first correlation
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_c = corr.max() * (2.0 / n)

    # -------------------- main pair‑transfer loop --------------------
    while time.time() < end_time:
        # locate the shift that gives the current maximum overlap
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        # indices participating in that shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift

        # pick the pair that contributes most to the overlap
        prod = h[i_range] * (1.0 - h[j_range])
        if prod.size == 0:
            # shouldn't happen, but reset if it does
            factor = factor_init
            no_improve = 0
            corr = np.correlate(h, 1.0 - h, mode="full")
            cur_c = corr.max() * (2.0 / n)
            continue
        pair_idx = int(np.argmax(prod))
        i = int(i_range[pair_idx])
        j = int(j_range[pair_idx])

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = factor_init
            no_improve = 0
            corr = np.correlate(h, 1.0 - h, mode="full")
            cur_c = corr.max() * (2.0 / n)
            continue

        eps = max_eps * factor
        h[i] -= eps
        h[j] += eps

        # recompute correlation after the move
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_c = corr.max() * (2.0 / n)

        # record best found
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # schedule factor
        factor = max(min_factor, factor * decay)

        iteration += 1

        # occasional small random disturbance to escape flat regions
        if iteration % 50000 == 0:
            # reset to best and add a tiny jitter
            h = best_h.copy()
            for _ in range(5):
                i2, j2 = rng.integers(0, n, size=2)
                if i2 == j2:
                    continue
                max_eps2 = min(h[i2], 1.0 - h[j2])
                if max_eps2 <= 0.0:
                    continue
                eps2 = max_eps2 * rng.random() * 0.05
                h[i2] -= eps2
                h[j2] += eps2
            corr = np.correlate(h, 1.0 - h, mode="full")
            cur_c = corr.max() * (2.0 / n)
            factor = factor_init
            no_improve = 0

        # reset factor if stagnation persists
        if no_improve > 8000:
            factor = factor_init
            no_improve = 0

        # occasional time check
        if iteration % 1000 == 0 and time.time() >= end_time:
            break

    # -------------------- final clean‑up --------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
