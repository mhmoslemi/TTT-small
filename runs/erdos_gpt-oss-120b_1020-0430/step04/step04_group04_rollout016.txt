    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    # Actually we can set lo = -np.max(v) - 1, hi = 1 - np.min(v) + 1 ensures larger bounds.
    for _ in range(max_iter):
        mid = (lo+hi)/2
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    # After loop, choose hi (or lo)

        if free.any():
            step = diff / free.sum()
            proj[free] += step
            proj = np.clip(proj, 0.0, 1.0)

def smooth_obj_and_grad(h, beta):
    n = h.shape[0]
    dx = 2.0 / n
    S = compute_S(h, dx)  # length L=2n-1
    S_min = S.min()
    exp_shift = np.exp(-beta * (S - S_min))
    sum_exp = exp_shift.sum()
    w = exp_shift / sum_exp
    # objective
    f = S_min - (1.0 / beta) * np.log(sum_exp)
    # gradient
    grad = grad_from_weights(w, h, dx)

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step*2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
        if (it+1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

import time
import numpy as np
import scipy.optimize as opt

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def compute_c5(h):
    """Exact C5 for step function h sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = _next_pow2(L)
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def compute_S(h):
    """Correlation Σ h_i h_{i+ℓ} * dx for all shifts."""
    n = h.shape[0]
    dx = 2.0 / n
    rev = h[::-1]
    L = 2 * n - 1
    m = _next_pow2(L)
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv * dx

def grad_from_weights(w, h):
    """Gradient of smooth surrogate given weights w (length 2n‑1)."""
    n = h.shape[0]
    dx = 2.0 / n
    sym_w = w + w[::-1]
    L = sym_w.shape[0]          # =2n‑1
    m = _next_pow2(L + n - 1)
    fw = np.fft.rfft(sym_w, m)
    fh = np.fft.rfft(h, m)
    conv = np.fft.irfft(fw * fh, m)
    start = n - 1
    return -dx * conv[start:start + n]

def smooth_obj_and_grad(h, beta):
    """Smooth approximation of min_{ℓ} Σ h_i h_{i+ℓ}dx and its gradient."""
    S = compute_S(h)
    S_min = S.min()
    exp_shift = np.exp(-beta * (S - S_min))
    sum_exp = exp_shift.sum()
    w = exp_shift / sum_exp                      # probabilities over shifts
    f = S_min - (1.0 / beta) * np.log(sum_exp)   # smooth min
    g = grad_from_weights(w, h)
    return f, g

def project_to_sum_and_bounds(v, target, max_iter=60):
    """Euclidean projection of v onto {0≤x≤1, Σx=target}."""
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v + hi, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1 - 1e-12)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj += diff / proj.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def gradient_ascent(h, target, budget, start):
    """Continuation gradient ascent on the smooth surrogate."""
    beta_sched = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]
    for beta in beta_sched:
        step = 0.5 / np.sqrt(beta)
        for _ in range(4000):
            if time.time() - start > budget * 0.95:
                return h
            f, g = smooth_obj_and_grad(h, beta)
            if np.linalg.norm(g) < 1e-12:
                break
            hnew = project_to_sum_and_bounds(h + step * g, target)
            fnew, _ = smooth_obj_and_grad(hnew, beta)
            if fnew > f:
                h = hnew
                step *= 1.05
            else:
                step *= 0.5
                if step < 1e-8:
                    break
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_multi(h0, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h0.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            nh = perturb_h(h, rng, scale=step)
            c = compute_c5(nh)
            if c < best_c:
                best_c, h = c, nh
                improved = True
                break
        if not improved and rng.random() < 0.02:
            nh = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(nh)
            if c < best_c:
                best_c, h = c, nh
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP on smooth surrogate (maximising the smooth min)."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    def obj(x):
        S = compute_S(x)
        S_min = S.min()
        exp_shift = np.exp(-beta * (S - S_min))
        sum_exp = exp_shift.sum()
        return -(S_min - (1.0 / beta) * np.log(sum_exp))
    res = opt.minimize(obj, h0, method="SLSQP", bounds=bounds,
                       constraints=cons,
                       options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    return project_to_sum_and_bounds(x, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 720
    target = n / 2.0

    # initialise h
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)

    h = gradient_ascent(h, target, budget_s, start)

    if c < best_c:

    while time.time() - start < budget_s * 0.9:

        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=3000,
                                      init_step=0.5, n_trials=6)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
        if time.time() - start > budget_s * 0.9:

    if time.time() - start < budget_s * 0.95:
        try:
            best_h = slsqp_optimize(best_h, beta=5000, max_iter=300)
            best_c = compute_c5(best_h)
        except Exception:
            pass
