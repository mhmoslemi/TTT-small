def perturb_transfer(h, rng, scale):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    if rng.random() < 0.5:
        # move epsilon from i to j
        max_eps = min(h[i], 1.0 - h[j])
        eps = rng.random() * max_eps * scale  # epsilon in [0, max_eps*scale]
        new = h.copy()
        new[i] -= eps
        new[j] += eps
    else:
        # move from j to i
        max_eps = min(h[j], 1.0 - h[i])
        eps = rng.random() * max_eps * scale
        new = h.copy()
        new[j] -= eps
        new[i] += eps
    return new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine initial
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # enforce sum exactly
        target = n/2.0
        h0 = project_to_sum(np.clip(h0,0,1), target)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # Baseline evaluation
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    
    # Add simple deterministic patterned candidates
    xs = (np.arange(n) + 0.5)*(2.0/n)
    # triangular
    tri = np.where(xs <= 1, xs, 2 - xs)
    best_c_tri = compute_c5(tri)
    if best_c_tri < best_c:
        best_c = best_c_tri
        best_h = tri.copy()
    # cosine bump
    cos = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    best_c_cos = compute_c5(cos)
    if best_c_cos < best_c:
        best_c = best_c_cos
        best_h = cos.copy()
    # random seeds
    for _ in range(5):
        cand = random_feasible_vector(n, rng)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()

def project_to_sum(v, target):
    # ensure sum(v) close to target via water-filling
    # We solve w_i = min(max(v_i - t, 0), 1) with sum w_i = target.
    # Use binary search on t.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        t = (lo + hi) / 2.0
        w = np.clip(v - t, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = t
        else:
            hi = t
    w = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # adjust any numerical drift
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly over free variables (not at bounds)
        free = (w > 1e-12) & (w < 1 - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # adjust all
            w += diff / len(w)
            w = np.clip(w, 0.0, 1.0)
    return w

def softmax_obj(x):
    n = x.size
    rev = (1.0 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    X = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(X * R, m)[:L] * (2.0 / n)
    max_val = conv.max()
    # compute log-sum-exp
    shifted = conv - max_val
    # using CURRENT_BETA global
    # Avoid overflow: we can compute exp(beta * shifted) safely
    exp_vals = np.exp(CURRENT_BETA * shifted)
    logsum = np.log(exp_vals.sum())
    return max_val + logsum / CURRENT_BETA

    target = CURRENT_TARGET

    x = project_to_sum(x, target)
    return x

import time

import scipy.optimize as opt

# ---------- Global placeholders for the smooth surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


# ---------- Core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv.max() * (2.0 / n)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto the box [0,1]^n intersect {sum = target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        t = (lo + hi) * 0.5
        w = np.clip(v - t, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = (lo + hi) * 0.5
    w = np.clip(v - t, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 1e-12) & (w < 1.0 - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w += diff / w.size
            w = np.clip(w, 0.0, 1.0)
    return w


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in [0,1]^n with prescribed sum."""
    v = rng.random(n)
    return project_to_sum(v, n / 2.0)


def perturb_transfer(h: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer a small amount of mass between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    if rng.random() < 0.5:
        max_eps = min(h[i], 1.0 - h[j])
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
    else:
        max_eps = min(h[j], 1.0 - h[i])
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[j] -= eps
        new_h[i] += eps
    return new_h


def simulated_annealing(
    h0: np.ndarray,
    rng: np.random.Generator,
    max_iter: int,
    init_temp: float,
    final_temp: float,
    init_scale: float,
    final_scale: float,
) -> tuple[np.ndarray, float]:
    """Simple SA that respects the sum constraint by construction."""
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    for it in range(max_iter):
        t = it / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_transfer(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0.0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


# ---------- Smooth surrogate for SLSQP ----------
def softmax_obj(x: np.ndarray) -> float:
    n = x.size
    rev = (1.0 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    X = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(X * R, m)[:L] * (2.0 / n)
    mval = conv.max()
    shifted = conv - mval
    # log‑sum‑exp, safely scaled
    exp_vals = np.exp(CURRENT_BETA * shifted)
    logsum = np.log(exp_vals.sum())
    return mval + logsum / CURRENT_BETA


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_to_sum(x, CURRENT_TARGET)
