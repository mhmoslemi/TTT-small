def max_corr_and_shifts(h):

    max_val = np.max(corr)

def upsample_binary(h_bin, new_n, rng):
    n = h_bin.size
    xs = np.linspace(0, 2, n, endpoint=False)
    xt = np.linspace(0, 2, new_n, endpoint=False)
    h_cont = np.interp(xt, xs, h_bin.astype(float))
    # threshold
    h_up = (h_cont > 0.5).astype(np.int8)
    # adjust sum to new_n//2
    target = new_n // 2
    cur = int(h_up.sum())
    diff = target - cur
    # while diff !=0 adjust
    while diff !=0:
        if diff > 0:
            # add 1 to a zero position
            zeros = np.flatnonzero(h_up == 0)
            if zeros.size == 0:
                break
            idx = rng.choice(zeros)
            h_up[idx] = 1
            diff -= 1
        else:
            ones = np.flatnonzero(h_up == 1)
            if ones.size == 0:
                break
            idx = rng.choice(ones)
            h_up[idx] = 0
            diff += 1
    return h_up.astype(float)

def binary_sa_optimize(h0, rng, deadline):
    h = h0.copy()
    n = h.shape[0]
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    # sets for ones and zeros
    ones_set = set(np.flatnonzero(h == 1).tolist())
    zeros_set = set(np.flatnonzero(h == 0).tolist())
    it = 0
    # initial temperature
    T0 = max(cur_c * 0.02, 1e-5)
    alpha = 0.9997
    while time.time() < deadline:
        it += 1
        T = T0 * (alpha ** it) + 1e-12
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        d = max_idx - (n - 1)
        # identify contributing indices
        if d >= 0:
            i_vals = np.arange(0, n - d)
            i2_vals = i_vals + d
        else:
            i_vals = np.arange(-d, n)
            i2_vals = i_vals + d
        contrib_mask = (h[i_vals] == 1) & (h[i2_vals] == 0)
        contrib_i = i_vals[contrib_mask]
        # choose i
        if contrib_i.size > 0:
            i = rng.choice(contrib_i)
            # choose j not adding contribution
            attempts = 0
            j = None
            while attempts < 10:
                j_candidate = rng.choice(list(zeros_set))
                # check if after swapping, contribution not increased
                # after swap, j becomes 1, check if contributes to worst shift
                # Contribution if h[j] becomes 1 and (j+d) in range and h[j+d]==0
                if not (0 <= j_candidate + d < n and h[j_candidate + d] == 0):
                    j = j_candidate
                    break
                attempts += 1
            if j is None:
                # fallback to random zero
                j = rng.choice(list(zeros_set))
        else:
            # random 1 and 0
            i = rng.choice(list(ones_set))
            j = rng.choice(list(zeros_set))
        # perform swap
        h[i] = 0
        h[j] = 1
        # update sets
        ones_set.remove(i); zeros_set.add(i)
        zeros_set.remove(j); ones_set.add(j)
        # compute new C5
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            # revert sets
            ones_set.add(i); zeros_set.remove(i)
            zeros_set.add(j); ones_set.remove(j)
            # cur_c unchanged
        # optional early exit if best_c already < target threshold (0.3808)?
        # We'll keep searching.
    return best_h.astype(float), best_c

            i_vals = np.arange(-d, n)
            i2_vals = i_vals + d
        contrib_mask = (h[i_vals] == 1) & (h[i2_vals] == 0)
        contrib_i = i_vals[contrib_mask]
        if contrib_i.size > 0:
            i = rng.choice(contrib_i)
            # try to pick a zero that does NOT create a new contribution
            j = None
            for _ in range(10):
                cand = rng.choice(list(zeros))
                if not (0 <= cand + d < n and h[cand + d] == 0):
                    j = cand
                    break
            if j is None:
                j = rng.choice(list(zeros))
        else:
            i = rng.choice(list(ones))
            j = rng.choice(list(zeros))
        # perform swap
        h[i] = 0
        h[j] = 1
        ones.remove(i); zeros.add(i)
        zeros.remove(j); ones.add(j)
        new_c = compute_c5(h)
        delta = new_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
            ones.add(i); zeros.discard(i)
            zeros.add(j); ones.discard(j)
    return best_h.astype(float), best_c

def upsample_binary(h_bin: np.ndarray, new_n: int,
                    rng: np.random.Generator) -> np.ndarray:
    """Upsample a binary pattern via linear interpolation and re‑binarize."""
    n = h_bin.size
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xt = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_cont = np.interp(xt, xs, h_bin.astype(float))
    h_up = (h_cont > 0.5).astype(np.int8)
    target = new_n // 2
    cur = int(h_up.sum())
    diff = target - cur
    while diff != 0:
        if diff > 0:
            zeros = np.flatnonzero(h_up == 0)
            if zeros.size == 0:
                break
            idx = rng.choice(zeros)
            h_up[idx] = 1
            diff -= 1
        else:
            ones = np.flatnonzero(h_up == 1)
            if ones.size == 0:
                break
            idx = rng.choice(ones)
            h_up[idx] = 0
            diff += 1
    return h_up.astype(float)

def continuous_refine(h: np.ndarray, rng: np.random.Generator,
                     deadline: float) -> np.ndarray:
    """Gradient + SLSQP cascade on a continuous vector."""
    h = gradient_refine(h, rng, step=0.02, max_iter=12)
    for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000):
        if time.time() > deadline:
            break
        try:
            h = slsqp_optimize(h, beta, rng, max_iter=800)
        except Exception:
            pass
        h = gradient_refine(h, rng, step=0.02, max_iter=4)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.98   # a little safety margin

    # initial parent if supplied
    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None
    base_n = len(parent) if parent is not None else 360
    if base_n % 2:
        base_n += 1

    best_h = None
    best_c = np.inf
    best_n = None

    # ------------------- Phase 1 : binary SA at base resolution -------------------
    while time.time() + 5 < deadline:
        if best_h is not None and rng.random() < 0.3:
            # perturb current best
            h0 = (best_h > 0.5).astype(np.int8)
            for _ in range(3):
                one_idx = rng.choice(np.flatnonzero(h0 == 1))
                zero_idx = rng.choice(np.flatnonzero(h0 == 0))
                h0[one_idx] = 0
                h0[zero_idx] = 1
        else:
            # fresh random binary
            h0 = np.zeros(base_n, dtype=np.int8)
            h0[rng.choice(base_n, base_n // 2, replace=False)] = 1
        sub_dead = min(deadline, time.time() + 5.0)
        h_opt, c_opt = binary_sa_optimize(h0, rng, sub_dead)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, base_n
        # occasional continuous refinement of the binary best
        if best_h is not None and rng.random() < 0.1:
            h_cont = best_h.astype(float)
            h_ref = continuous_refine(h_cont, rng, deadline)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c, best_n = h_ref.copy(), c_ref, base_n

    # ------------------- Phase 2 : upscale and refine at higher resolution -------------------
    high_n = 720
    if best_h is not None and high_n > base_n and time.time() + 10 < deadline:
        h_up = upsample_binary(best_h, high_n, rng)
        while time.time() + 5 < deadline:
            sub_dead = min(deadline, time.time() + 5.0)
            h_opt, c_opt = binary_sa_optimize(h_up.astype(np.int8), rng, sub_dead)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, high_n
                h_up = h_opt.copy()
            if best_h is not None and rng.random() < 0.1:
                h_ref = continuous_refine(best_h.astype(float), rng, deadline)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c, best_n = h_ref.copy(), c_ref, high_n
                    h_up = h_ref.copy()
            if time.time() > deadline - 5:
                break

    # ------------------- Final refinement if time remains -------------------
    if best_h is not None and time.time() < deadline:
        h_final = best_h.astype(float)
        h_final = continuous_refine(h_final, rng, deadline)
        # final safety adjustment
        np.clip(h_final, 0.0, 1.0, out=h_final)
        adjust_sum_exact(h_final, rng)
        final_c = compute_c5(h_final)
        if final_c < best_c:
            best_h, best_c, best_n = h_final.copy(), final_c, h_final.size

    # ------------------- Return result -------------------
    if best_h is None:
        # fallback simple half‑interval pattern
        n = base_n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
