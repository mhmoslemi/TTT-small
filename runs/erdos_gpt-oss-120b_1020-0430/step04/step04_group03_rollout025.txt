import time
import numpy as np
from scipy.optimize import minimize

# ----- basic helpers -------------------------------------------------
def enforce_sum(h, target, rng):
    """Clip h to [0,1] and adjust (by random transfers) to hit exact sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    eps = 1e-12
    while abs(diff) > eps and it < max_iter:
        if diff > 0:  # need to increase total
            idx = np.where(h < 1.0 - eps)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease total
            idx = np.where(h > eps)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > eps:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1]^n with exact sum = n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    """Move a random amount (scaled) from one coordinate to another."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    # random direction
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_eps = min(h[src], 1.0 - h[dst])
    if max_eps <= 1e-14:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[src] -= eps
    nh[dst] += eps
    return nh

def compute_c5(h):
    """Discrete C5 for vector h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)

def worst_shift_info(h):
    """Return (lag s, index in full corr, value) where C5 is attained."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (n - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    """Gradient of C_s = sum_i h_i(1-h_{i+s}) w.r.t. h."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1.0 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1.0 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

# ----- refined greedy with line search ------------------------------
def refined_greedy(start_h, max_iters=15, time_limit=None):
    """Greedy improvement focusing on the current worst shift.
    Moves mass between a positive‑gradient and a negative‑gradient index,
    choosing epsilon from a small set that yields the best C5 reduction."""
    h = start_h.copy()
    n = h.size
    best_c = compute_c5(h)
    start_time = time.time()
    alphas = (1.0, 0.6, 0.3, 0.1)
    for _ in range(max_iters):
        if time_limit is not None and (time.time() - start_time) >= time_limit:
            break
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-10) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-10) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        # keep only the most promising candidates
        K = 7
        pos_idx = pos[np.argsort(-grad[pos])[:K]]
        neg_idx = neg[np.argsort(grad[neg])[:K]]
        improved = False
        best_local_c = best_c
        best_move = None
        for i in pos_idx:
            for j in neg_idx:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps <= 1e-14:
                    continue
                for a in alphas:
                    eps = max_eps * a
                    if eps <= 1e-14:
                        continue
                    nh = h.copy()
                    nh[i] -= eps
                    nh[j] += eps
                    c = compute_c5(nh)
                    if c < best_local_c - 1e-12:
                        best_local_c = c
                        best_move = (i, j, eps)
                        improved = True
        if not improved:
            break
        i, j, eps = best_move
        h[i] -= eps
        h[j] += eps
        best_c = best_local_c
    return h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    """Smooth approximation of max via softmax, optimised with SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
