def worst_shift_info(h):
    # compute correlation using fft as above, but also need index of max.
    n = len(h)
    N = 2*n - 1
    a = np.fft.rfft(h, n=N)
    b = np.fft.rfft((1 - h)[::-1], n=N)
    conv = np.fft.irfft(a * b, n=N)
    idx = np.argmax(conv)
    s = idx - (n - 1)  # shift offset
    max_val = conv[idx]
    return s, idx, max_val

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, idx, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]

    return h

def perturb_random(h, rng, scale=0.2):
    n = h.shape[0]
    # choose two indices i!=j
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    # compute max epsilon to keep within [0,1]
    max_eps = min(h[i], 1.0 - h[j])

        return h.copy()
    eps = max_eps * scale * rng.random()
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps

def enforce_sum(vec, target):

    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec

    # if diff >0, need to increase some entries not at 1
    # if diff <0, need to decrease some entries not at 0
    # We'll use linear scaling: add diff to entries proportionally to free capacity.
    # But we need to ensure no bounds violations.
    # We'll iterate.
    for _ in range(1000):
        if diff == 0:
            break
        if diff > 0:
            free = np.where(vec < 1 - 1e-12)[0]
            if free.size == 0:
                break
            add = diff / free.size
            add = np.minimum(add, 1 - vec[free])

    return vec

def enforce_sum(vec, target):
    """Clip to [0,1] and adjust a single entry to hit the exact sum."""
    vec = np.clip(vec, 0.0, 1.0)
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    # try to adjust a free entry
    if diff > 0:
        cand = np.where(vec < 1.0 - 1e-12)[0]
    else:
        cand = np.where(vec > 1e-12)[0]
    if cand.size == 0:
        # fallback: adjust a random entry
        i = np.random.randint(0, vec.size)
    else:
        i = np.random.choice(cand)
    vec[i] = np.clip(vec[i] + diff, 0.0, 1.0)
    return vec


def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0)


def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def compute_c5_fast(h):
    """Discrete C5 using FFT based convolution."""
    n = h.shape[0]
    N = 2 * n - 1
    A = np.fft.rfft(h, n=N)
    B = np.fft.rfft((1.0 - h)[::-1], n=N)
    conv = np.fft.irfft(A * B, n=N)
    return np.max(conv) * (2.0 / n)


def worst_shift_info(h):
    n = h.shape[0]
    N = 2 * n - 1
    A = np.fft.rfft(h, n=N)
    B = np.fft.rfft((1.0 - h)[::-1], n=N)
    conv = np.fft.irfft(A * B, n=N)
    idx = np.argmax(conv)
    s = idx - (n - 1)
    return s, idx, conv[idx]


def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1.0 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1.0 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def greedy_optimize(h, max_steps=2000):
    cur_c = compute_c5_fast(h)
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full epsilon
        h_new = h.copy()
        h_new[i] -= max_eps
        h_new[j] += max_eps
        new_c = compute_c5_fast(h_new)
        if new_c + 1e-12 < cur_c:
            h, cur_c = h_new, new_c
            continue
        # try half epsilon
        eps = max_eps * 0.5
        h_new[i] = h[i] - eps
        h_new[j] = h[j] + eps
        new_c = compute_c5_fast(h_new)
        if new_c + 1e-12 < cur_c:
            h, cur_c = h_new, new_c
            continue
        break
    return h

    zeros = list(np.where(h == 0)[0])
    t0 = time.time()
    while time.time() - t0 < time_limit:
        if not ones or not zeros:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5_fast(h)
        if c < best_c - 1e-12:
            best_c = c
            ones[i_idx] = j
            zeros[j_idx] = i
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # initial vector
    if 'initial_h_values' in globals():
        h_frac = np.array(initial_h_values, dtype=np.float64)
        n = h_frac.size
    else:
        n = 720
        h_frac = random_feasible_vector(n, rng)

    h_frac = enforce_sum(np.clip(h_frac, 0.0, 1.0), n / 2.0)
    best_h = h_frac.copy()
    best_c = compute_c5_fast(best_h)

    # -------- Phase 1: fractional greedy with restarts --------
    while time.time() - start_time < total_budget * 0.55:
        if rng.random() < 0.5:
            start = perturb_random(best_h, rng, scale=0.4)
        else:
            start = random_feasible_vector(n, rng)
        refined = greedy_optimize(start, max_steps=3000)
        c_ref = compute_c5_fast(refined)
        if c_ref < best_c - 1e-12:
            best_h, best_c = refined, c_ref

    # -------- Phase 2: binary search --------
    # convert to binary
    binary_h = (best_h >= 0.5).astype(np.float64)
    target_sum = n / 2.0
    diff = target_sum - binary_h.sum()
    if diff > 0:
        zero_idxs = np.where(binary_h == 0)[0]
        chosen = rng.choice(zero_idxs, int(diff), replace=False)
        binary_h[chosen] = 1.0
    elif diff < 0:
        one_idxs = np.where(binary_h == 1)[0]
        chosen = rng.choice(one_idxs, int(-diff), replace=False)
        binary_h[chosen] = 0.0
    binary_h = enforce_sum(binary_h, target_sum)

    bin_best_h = binary_h.copy()
    bin_best_c = compute_c5_fast(bin_best_h)

    binary_time_budget = total_budget * 0.35
    bin_start = time.time()
    while time.time() - bin_start < binary_time_budget:
        # random binary start
        h_start = random_binary_vector(n, rng)
        remaining = binary_time_budget - (time.time() - bin_start)
        per_restart = min(2.0, max(0.2, remaining / 2.0))
        h_loc, c_loc = binary_local_search(h_start, rng, per_restart)
        if c_loc < bin_best_c - 1e-12:
            bin_best_h, bin_best_c = h_loc, c_loc

    # keep the better of fractional and binary
    if bin_best_c < best_c:
        best_h, best_c = bin_best_h, bin_best_c

    # -------- Phase 3: final fractional refinement --------
    while time.time() - start_time < total_budget * 0.95:
        start = perturb_random(best_h, rng, scale=0.2)
        refined = greedy_optimize(start, max_steps=2000)
        c_ref = compute_c5_fast(refined)
        if c_ref < best_c - 1e-12:
            best_h, best_c = refined, c_ref

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0)
    final_c = compute_c5_fast(best_h)

    return best_h.tolist(), float(final_c), int(n)
