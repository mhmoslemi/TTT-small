def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.02, 1e-5
    for it in range(max_iters):
        if time.time() >= time_budget:
            break
        T = T0 * (Tf/T0) ** (it / max_iters)
        # compute lag with worst correlation
        lag_idx = np.argmax(cur_corr)
        lag = lag_idx - (n - 1)
        lag = abs(lag)
        # choose move
        if rng.random() < 0.6:
            new_h = targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

def enforce_sum(h, target, rng, max_iter=2000):
    # ensure exact sum by adjusting random positions
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break

def softmax_obj(x):
    # uses global CURRENT_BETA
    # compute correlation via FFT
    n = x.size
    # compute convolution via FFT of h and rev(1-h)
    # allocate length m as next power of two >= 2n
    m = 1 << ((2 * n - 1).bit_length())
    a = np.zeros(m)
    b = np.zeros(m)
    a[:n] = x
    b[:n] = (1 - x)[::-1]
    A = np.fft.rfft(a)
    B = np.fft.rfft(b)
    conv = np.fft.irfft(A * B, n=m)[:2 * n - 1]
    bc = CURRENT_BETA * conv
    mval = np.max(bc)
    # log-sum-exp
    return (mval + np.log(np.sum(np.exp(bc - mval)))) / CURRENT_BETA

def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h in [0,1] so that sum(h)=target (exact)."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # final even correction
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
