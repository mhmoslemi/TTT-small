import time, numpy as np
from scipy.optimize import minimize

# ---------- constraint handling ----------
def enforce_sum(v, target, rng):
    """Clip v to [0,1] and adjust to make sum exactly target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    max_iter = 20000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(v < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - v[i]) * rng.random()
            v[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(v > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, v[i]) * rng.random()
            v[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, v.size)
        v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v

# ---------- core evaluation ----------
def compute_c5_full(h):
    """Discrete C5 for full vector h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info_full(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift_full(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

# ---------- symmetry utilities ----------
def reflect_half(h_half):
    """Return full symmetric vector from half."""
    return np.concatenate([h_half, h_half[::-1]])

def grad_half_from_full(h_half, s):
    """Gradient for half variables (average of mirrored positions)."""
    h_full = reflect_half(h_half)
    g_full = gradient_shift_full(h_full, s)
    m = h_half.size
    return 0.5 * (g_full[:m] + g_full[-1:-m-1:-1])

# ---------- local deterministic search ----------
def greedy_optimize_half(h_half, max_steps=2000, time_limit=None, start_time=None):
    """Transfer mass between half indices using worst‑shift gradient."""
    m = h_half.size
    steps = 0
    while steps < max_steps:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        s, _, _ = worst_shift_info_full(reflect_half(h_half))
        grad = grad_half_from_full(h_half, s)
        pos = np.where((grad > 1e-12) & (h_half > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h_half < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h_half[i], 1.0 - h_half[j])
        if max_eps < 1e-12:
            break
        # try full epsilon
        cand = h_half.copy()
        cand[i] -= max_eps
        cand[j] += max_eps
        c_new = compute_c5_full(reflect_half(cand))
        c_cur = compute_c5_full(reflect_half(h_half))
        if c_new + 1e-12 < c_cur:
            h_half = cand
            steps += 1
            continue
        # try half epsilon
        eps = max_eps * 0.5
        cand[i] += max_eps - eps   # revert previous subtraction
        cand[j] -= max_eps - eps
        cand[i] -= eps
        cand[j] += eps
        c_new = compute_c5_full(reflect_half(cand))
        if c_new + 1e-12 < c_cur:
            h_half = cand
            steps += 1
            continue
        break
    return h_half

def perturb_random_half(h_half, rng, scale=0.2):
    """Random transfer preserving sum."""
    m = h_half.size
    i, j = rng.integers(0, m, size=2)
    if i == j:
        return h_half.copy()
    max_eps = min(h_half[i], 1.0 - h_half[j])
    if max_eps <= 0.0:
        return h_half.copy()
    eps = max_eps * scale * rng.random()
    nh = h_half.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_full(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- determine problem size -----
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0_full = np.array(initial_h_values, dtype=float)
    else:
        n = 720
        h0_full = None
    if n % 2 != 0:
        n += 1  # enforce even size for symmetry
    half_n = n // 2
    target_half = n / 4.0

    # ----- initialise half vector -----
    if h0_full is not None:
        # symmetrize
        h_half = 0.5 * (h0_full[:half_n] + h0_full[-1:-half_n-1:-1])
        h_half = enforce_sum(h_half, target_half, rng)
    else:
        h_half = rng.random(half_n)
        h_half = enforce_sum(h_half, target_half, rng)

    # ----- track best -----
    best_h_half = h_half.copy()
    best_c = compute_c5_full(reflect_half(best_h_half))

    # ----- simulated‑annealing loop -----
    elapsed = lambda: time.time() - start
    time_limit_sa = budget_s * 0.90
    iteration = 0
    current_h_half = h_half.copy()
    current_c = best_c

    while elapsed() < time_limit_sa:
        iteration += 1
        prog = elapsed() / time_limit_sa
        # temperature descends linearly
        T = max(1e-6, 0.001 * (1.0 - prog))
        # perturbation scale: large early, small late
        scale = max(0.5 * (1.0 - prog), 0.02)
        cand_half = perturb_random_half(current_h_half, rng, scale=scale)
        cand_c = compute_c5_full(reflect_half(cand_half))
        # acceptance
        if cand_c < current_c or rng.random() < np.exp(-(cand_c - current_c) / T):
            current_h_half, current_c = cand_half, cand_c
            if cand_c < best_c:
                best_h_half, best_c = cand_half.copy(), cand_c
        # occasional deterministic refinement
        if iteration % 100 == 0:
            remaining = time_limit_sa - elapsed()
            if remaining > 0.5:
                max_steps = int(min(500, max(10, remaining * 20)))
                current_h_half = greedy_optimize_half(
                    current_h_half,
                    max_steps=max_steps,
                    time_limit=remaining,
                    start_time=time.time(),
                )
                current_c = compute_c5_full(reflect_half(current_h_half))
                if current_c < best_c:
                    best_h_half, best_c = current_h_half.copy(), current_c

    # ----- final local search -----
    remaining = budget_s - elapsed()
    if remaining > 5.0:
        max_steps = int(min(5000, remaining * 200))
        best_h_half = greedy_optimize_half(
            best_h_half,
            max_steps=max_steps,
            time_limit=remaining,
            start_time=time.time(),
        )
        best_c = compute_c5_full(reflect_half(best_h_half))

    # ----- SLSQP polishing -----
    remaining = budget_s - elapsed()
    if remaining > 2.0:
        try:
            refined = slsqp_optimize_full(reflect_half(best_h_half), beta=5000, max_iter=800)
            refined_c = compute_c5_full(refined)
            if refined_c < best_c:
                best_h_half = 0.5 * (refined[:half_n] + refined[-1:-half_n-1:-1])
                best_c = refined_c
        except Exception:
            pass

    # ----- final clean‑up -----
    best_full = reflect_half(best_h_half)
    best_full = np.clip(best_full, 0.0, 1.0)
    best_full = enforce_sum(best_full, n / 2.0, rng)
    final_c = compute_c5_full(best_full)

    return best_full.tolist(), float(final_c), int(n)
