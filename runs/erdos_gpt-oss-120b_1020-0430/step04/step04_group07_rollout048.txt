def greedy_step(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    a = np.zeros(n)
    if shift >= 0:
        if shift < n:
            a[:n-shift] = 1.0 - h[shift:]
    else:
        s = -shift
        if s < n:
            a[s:] = 1.0 - h[:n-s]
    # sort indices by a ascending (lower penalty first)
    order = np.argsort(a)
    t = n // 2
    new_h = np.zeros(n)
    new_h[order[:t]] = 1.0
    # any remaining fractional? Not needed.
    return new_h

def slsqp_refine(h, beta, max_iter, rng):
    n = h.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    # use SLSQP
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum precisely
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, rng)
    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] with uniform grid."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Project h onto the simplex Σh = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def greedy_step(h: np.ndarray) -> np.ndarray:
    """One greedy reallocation step w.r.t. the current worst shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # could be negative
    a = np.zeros(n)
    if shift >= 0:
        if shift < n:
            a[: n - shift] = 1.0 - h[shift:]
    else:
        s = -shift
        if s < n:
            a[s:] = 1.0 - h[: n - s]
    order = np.argsort(a)          # smallest penalty first
    t = n // 2
    new_h = np.zeros(n)
    new_h[order[:t]] = 1.0
    return new_h


def perturb_step(h: np.ndarray, rng: np.random.Generator, scale: float = 0.05) -> np.ndarray:
    """Small random transfer of mass while respecting bounds."""
    n = h.size
    # pick a source with noticeable mass and a sink with little mass
    src = rng.choice(np.where(h > 1e-9)[0]) if np.any(h > 1e-9) else rng.integers(n)
    dst = rng.choice(np.where(h < 1 - 1e-9)[0]) if np.any(h < 1 - 1e-9) else rng.integers(n)
    max_eps = min(h[src], 1.0 - h[dst])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[src] -= eps
    nh[dst] += eps
    return nh


def slsqp_refine(h: np.ndarray, beta: float, max_iter: int, rng: np.random.Generator) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h.size
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # log‑sum‑exp stabilized
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return enforce_sum(x, rng)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf

    def update_candidate(h):
        nonlocal best_h, best_c
        c = compute_c5(h)
        if c < best_c:
            best_h = h.copy()
            best_c = c

    # --- initialise from possible parent construction -----------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        update_candidate(h0)

    # --- parameters --------------------------------------------------
    n_candidates = [192, 224, 240, 256, 272, 288]
    outer_time_limit = 0.94 * budget_s

    # --- main search loop --------------------------------------------
    while time.time() - start < outer_time_limit:
        n = int(rng.choice(n_candidates))
        # initialise a start vector
        if best_h is not None and best_h.size == n and rng.random() < 0.7:
            h = best_h.copy()
            h = perturb_step(h, rng, scale=0.07)
            h = enforce_sum(h, rng)
        else:
            h = np.zeros(n)
            h[rng.choice(n, n // 2, replace=False)] = 1.0

        # inner greedy/perturb iterations
        for it in range(50):
            h = greedy_step(h)
            update_candidate(h)

            # occasional random perturbation
            if rng.random() < 0.2:
                h = perturb_step(h, rng, scale=0.12)
                h = enforce_sum(h, rng)

            # occasional modest SLSQP refinement
            if rng.random() < 0.05:
                h = slsqp_refine(h, beta=120.0, max_iter=250, rng=rng)
                h = greedy_step(h)
                update_candidate(h)

            if time.time() - start > 0.96 * budget_s:
                break

        # occasional global refinement of the current best
        if best_h is not None and rng.random() < 0.1 and time.time() - start < 0.96 * budget_s:
            hb = best_h.copy()
            hb = slsqp_refine(hb, beta=250.0, max_iter=400, rng=rng)
            update_candidate(hb)

    # --- optional up‑sampling (if size permits) ----------------------
    if best_h is not None and time.time() - start < 0.99 * budget_s:
        up_factor = 2
        if best_h.size * up_factor <= 384:
            n_up = best_h.size * up_factor
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = enforce_sum(h_up, rng)

            for _ in range(30):
                h_up = greedy_step(h_up)
                update_candidate(h_up)
                if rng.random() < 0.2:
                    h_up = perturb_step(h_up, rng, scale=0.09)
                    h_up = enforce_sum(h_up, rng)

            if time.time() - start < budget_s * 0.995:
                h_up = slsqp_refine(h_up, beta=800.0, max_iter=500, rng=rng)
                update_candidate(h_up)

    # --- final high‑β refinement on the best found solution ----------
    if best_h is not None and time.time() - start < budget_s * 0.998:
        best_h = slsqp_refine(best_h, beta=2000.0, max_iter=800, rng=rng)

    # --- sanity and return -------------------------------------------
    if best_h is None:
        # fallback uniform half‑mass vector
        n = 256
        best_h = np.full(n, 0.5)
        best_h = enforce_sum(best_h, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = float(compute_c5(best_h))
    return best_h.tolist(), final_c, int(best_h.size)
