    lo=0.0; hi=max_eps
    for _ in range(30): 
        m1=lo+(hi-lo)/3; m2=hi-(hi-lo)/3
        f1 = np.max(a + delta*m1)

        if f1 < f2:

        else:
            lo = m1
    eps_opt = (lo+hi)/2

    for _ in range(max_iters):

        idx = np.argmax(a)

# enforce_sum (similar to previous code)
def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple iterative approach as before
    max_iter = 10000
    it = 0
    # use while loop; for efficiency use while but break early
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    # fallback
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n/2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i,j = rng.integers(0,n,size=2)
    if i==j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

import time, numpy as np

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def greedy_optimize_line_search(h, max_iters, rng):
    n = h.size
    s_arr = np.arange(-n + 1, n)
    a = np.correlate(h, 1 - h, mode='full')
    for _ in range(max_iters):
        idx = np.argmax(a)
        s = idx - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            grad[: n - s] += (1 - h[s:])
            grad[s:] -= h[: n - s]
        else:
            s_abs = -s
            grad[s_abs:] += (1 - h[: n - s_abs])
            grad[: n - s_abs] -= h[s_abs:]
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-14:
            break
        pad = np.concatenate([np.zeros(n - 1), h, np.zeros(n - 1)])
        base = n - 1
        idx_i_plus = base + i + s_arr
        idx_i_minus = base + i - s_arr
        idx_j_plus = base + j + s_arr
        idx_j_minus = base + j - s_arr
        delta = pad[idx_i_plus] + pad[idx_i_minus] - pad[idx_j_plus] - pad[idx_j_minus]
        if np.all(delta >= -1e-15):
            eps_opt = 0.0
        elif np.all(delta <= 1e-15):
            eps_opt = max_eps
        else:
            lo, hi = 0.0, max_eps
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(a + delta * m1)
                f2 = np.max(a + delta * m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            eps_mid = (lo + hi) / 2.0
            f0 = np.max(a)
            fmax = np.max(a + delta * max_eps)
            fmid = np.max(a + delta * eps_mid)
            eps_opt = 0.0
            best_f = f0
            if fmax < best_f:
                best_f = fmax
                eps_opt = max_eps
            if fmid < best_f:
                best_f = fmid
                eps_opt = eps_mid
        if eps_opt <= 1e-14:
            break
        h[i] -= eps_opt
        h[j] += eps_opt
        a = a + delta * eps_opt
    return h

def run(seed=42, budget_s=1100, **kwargs):
