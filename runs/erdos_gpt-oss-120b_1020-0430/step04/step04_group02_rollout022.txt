def compute_c5(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    fft_h = np.fft.rfft(np.pad(h, (0, L - n)))
    autocorr = np.fft.irfft(fft_h * np.conj(fft_h), n=L)[:2*n-1]
    corr = n/2 - autocorr
    max_corr = np.max(corr)
    return max_corr * (2.0/n)
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection onto [0,1] with exact sum ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h

    corr = n / 2.0 - autocorr
    return np.max(corr) * (2.0 / n)

# ---------- hill climbing ----------
def hill_climb_targeted(
    h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25
):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)  # convert to shift
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- softmax based gradient step ----------
def gradient_step_one(h, beta, lr):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    fft_h = np.fft.rfft(np.pad(h, (0, L - n)))
    autocorr = np.fft.irfft(fft_h * np.conj(fft_h), n=L)[: 2 * n - 1]
    corr = n / 2.0 - autocorr
    max_corr = np.max(corr)
    w = np.exp(beta * (corr - max_corr))
    w /= w.sum()
    conv_wh = np.convolve(w, h, mode="full")
    conv_wrevh = np.convolve(w[::-1], h, mode="full")
    offset = n - 1
    grad = 1.0 - (conv_wh[offset : offset + n] + conv_wrevh[offset : offset + n])
    h_new = h - lr * grad
    h_new = project_box_sum(h_new, n / 2.0)
    c_new = compute_c5(h_new)
    return h_new, c_new


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- diverse pool of candidates ----
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    for f in range(1, 7):
        cand = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        cand = project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0)
        candidates.append(cand)
        cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        cand = project_box_sum(np.clip(cand, 0.0, 1.0), n / 2.0)
        candidates.append(cand)

    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(project_box_sum(mask, n / 2.0))

    mask = np.zeros(n)
    start = n // 4
    mask[start : start + n // 2] = 1.0
    candidates.append(project_box_sum(mask, n / 2.0))

    for _ in range(6):
        candidates.append(random_feasible_vector(n, rng))

    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- gradient descent refinement (softmax) ----
    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]
    steps_schedule = [200, 200, 300, 300, 400, 400, 500, 500, 600]
    lr_schedule = [0.5, 0.4, 0.3, 0.3, 0.2, 0.2, 0.15, 0.15, 0.1]

    for beta, steps, lr in zip(beta_schedule, steps_schedule, lr_schedule):
        for _ in range(steps):
            if time_left() < 0.02 * budget_s:
                break
            h_new, c_new = gradient_step_one(best_h, beta, lr)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
            else:
                best_h = h_new
        if time_left() < 0.02 * budget_s:
            break

    # ---- localized hill climbing with targeted moves ----
    if time_left() > 0.1 * budget_s:
        h_hc, c_hc = hill_climb_targeted(
            best_h, rng, max_iters=1500, init_step=0.3, n_trials=6, p_target=0.3
        )
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

    # ---- SLSQP fine‑tuning if time permits ----
    for beta in [5000, 10000]:
        if time_left() < 0.02 * budget_s:
            break
        try:
            h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()
        except Exception:
            pass

    # ---- optional up‑sampling and refinement ----
    if time_left() > 0.15 * budget_s:
        target_n = min(720, max(540, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_box_sum(h_up, target_n / 2.0)

            # quick gradient descent on up‑sampled version
            beta_up = [5, 20, 80, 200, 500, 1000]
            steps_up = [100, 100, 150, 150, 200, 200]
            lr_up = [0.5, 0.4, 0.3, 0.3, 0.2, 0.2]
            for beta, steps, lr in zip(beta_up, steps_up, lr_up):
                for _ in range(steps):
                    if time_left() < 0.01 * budget_s:
                        break
                    h_up, _ = gradient_step_one(h_up, beta, lr)
                if time_left() < 0.01 * budget_s:
                    break

            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_c, best_h, n = c_up, h_up.copy(), target_n

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
