def slsqp_optimize_one_beta(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # same correction as before
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x
```

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_corr_fft(h):
    n = len(h)
    size = 1 << ((2*n - 1).bit_length())
    a = np.fft.rfft(np.concatenate([h, np.zeros(size - n)]))
    b_rev = (1.0 - h)[::-1]
    b = np.fft.rfft(np.concatenate([b_rev, np.zeros(size - n)]))
    corr = np.fft.irfft(a * b)[:2*n-1]
    return corr

def compute_c5_and_corr(h):
    corr = compute_corr_fft(h)
    n = h.size
    c5 = np.max(corr) * (2.0 / n)
    return c5, corr

import time
import numpy as np

# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=5000):
    """Clip h to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tiny correction if needed
    if abs(diff) > 1e-12:
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    """Bias ε‑transfer toward indices that dominate the worst lag."""
    n = h.size
    if abs(lag) >= n:
        return perturb_h(h, rng, scale)
    if lag >= 0:
        idx = np.arange(n - lag)
        contrib = h[idx] * (1.0 - h[idx + lag])
    else:
        lag = -lag
        idx = np.arange(lag, n)
        contrib = h[idx] * (1.0 - h[idx - lag])
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(idx) * top_frac))
    top = idx[np.argpartition(-contrib, k - 1)[:k]]
    bot = idx[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top)
    j = rng.choice(bot) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ---------- correlation ----------
def compute_corr_fft(h):
    """Return raw full correlation of h with 1−h using FFT."""
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    a = np.fft.rfft(np.concatenate([h, np.zeros(size - n)]))
    b_rev = (1.0 - h)[::-1]
    b = np.fft.rfft(np.concatenate([b_rev, np.zeros(size - n)]))
    corr = np.fft.irfft(a * b)[: 2 * n - 1]
    return corr

def compute_c5_and_corr(h):
    """C5 = max(corr) * (2/n) and raw correlation array."""
    corr = compute_corr_fft(h)
    n = h.size
    c5 = np.max(corr) * (2.0 / n)
    return c5, corr

def compute_c5(h):
    c5, _ = compute_c5_and_corr(h)
    return c5

# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    lse = m + np.log(np.sum(np.exp(bc - m)))
    return lse / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- simulated annealing ----------
def simulated_annealing(h_start, rng, max_iters, time_start, time_limit):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_and_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.05, 1e-6
    for it in range(max_iters):
        if time.time() - time_start > time_limit:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        lag = np.argmax(cur_corr) - (n - 1)
        if rng.random() < 0.7:
            new_h = targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_and_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ---- low‑resolution stage ----
    n_low = max(480, min(720, n0 * 2))
    if n_low != n0:
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(n_low) + 0.5) * (2.0 / n_low)
        h = np.interp(new_x, old_x, h0)
    else:
        h = h0.copy()
    h = enforce_sum(h, n_low / 2.0, rng)
    best_h, best_c = h.copy(), compute_c5(h)

    total_budget = budget_s
    # allocate ~60 % to low‑res
    low_end = t0 + total_budget * 0.6

    beta_sched = [10, 30, 100, 300, 800, 1500, 3000, 6000]
    for beta in beta_sched:
        if time.time() >= low_end:
            break
        max_it = 200 if beta < 200 else 400 if beta < 1000 else 800
        h = slsqp_optimize_one_beta(h, beta, rng, max_iter=max_it)
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()

    # SA on low‑res
    if time.time() < low_end:
        max_sa = 200000
        h, cur_c = simulated_annealing(best_h, rng, max_sa, t0, low_end)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()

    # ---- high‑resolution if time permits ----
    time_now = time.time()
    remaining = total_budget - (time_now - t0)
    if remaining > total_budget * 0.2:
        n_high = min(1440, n_low * 2)
        old_x = (np.arange(n_low) + 0.5) * (2.0 / n_low)
        new_x = (np.arange(n_high) + 0.5) * (2.0 / n_high)
        h = np.interp(new_x, old_x, best_h)
        h = enforce_sum(h, n_high / 2.0, rng)
        best_h, best_c = h.copy(), compute_c5(h)

        high_end = t0 + total_budget * 0.95
        for beta in beta_sched:
            if time.time() >= high_end:
                break
            max_it = 300 if beta < 200 else 600 if beta < 1000 else 1000
            h = slsqp_optimize_one_beta(h, beta, rng, max_iter=max_it)
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()

        if time.time() < high_end:
            max_sa_h = 200000
            h, cur_c = simulated_annealing(best_h, rng, max_sa_h, t0, high_end)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()

    # ---- final quick local hill‑climb ----
    step = 0.3
    h_cur = best_h.copy()
    cur_c = best_c
    for _ in range(8000):
        if time.time() - t0 > total_budget:
            break
        new_h = perturb_h(h_cur, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_c, h_cur = new_c, new_h
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        step *= 0.9995

    # ---- final cleanup ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
