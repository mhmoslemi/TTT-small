def l_bfgs_optimize(h0, beta, target, rng, maxiter=200, lam=1e3):
    n = h0.size
    bounds = [(0.0, 1.0)] * n

    def obj_grad(x):
        # compute smooth max plus penalty
        # correlation of x with x
        corr = np.correlate(x, x, mode='full')
        v = target - corr[n-1:]  # size n
        maxv = np.max(v)
        # smooth max
        exp_beta = np.exp(beta * (v - maxv))
        sumexp = np.sum(exp_beta)
        f = maxv + (np.log(sumexp) / beta)
        # penalty
        diff = np.sum(x) - target
        f += lam * diff * diff
        # gradient of smooth max: weighted sum of grad v_i
        w = exp_beta / sumexp  # shape n
        # compute part1 and part2 using correlation
        part1 = np.correlate(w, 1 - x, mode='full')[n-1:n-1+n]
        part2 = np.correlate(w, x[::-1], mode='full')[n-1 - np.arange(n)]
        grad = part1 - part2
        # penalty gradient
        grad += 2 * lam * diff
        return f, grad.astype(float)

    res = minimize(obj_grad, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # final sum adjust
    adjust_sum_exact(x, rng)
    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for discrete h on [0,2] with spacing 2/n."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute a tiny amount to make Σh = n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.05, max_iter: int = 30) -> np.ndarray:
    """Projected gradient descent on the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            up = n - k
            g[:up] += 1.0 - h[k:]          # (1 - h[i+k])
            g[k:]  -= h[:up]               # -h[i-k]
        step_sz = step * (0.5 ** it)
        h -= step_sz * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def smooth_obj_and_grad(x: np.ndarray, beta: float, target: float,
                        lam: float) -> tuple:
    """Smooth‑max of v_k = target - corr_hh[k] plus quadratic sum penalty."""
    n = x.size
    corr = np.correlate(x, x, mode="full")
    v = target - corr[n - 1 :]           # v_k for k=0..n-1
    maxv = np.max(v)
    # smooth max
    exp_beta = np.exp(beta * (v - maxv))
    sumexp = np.sum(exp_beta)
    f = maxv + np.log(sumexp) / beta

    # penalty for sum constraint
    diff = np.sum(x) - target
    f += lam * diff * diff

    # gradient of smooth max: weighted sum of ∂v_k/∂x
    w = exp_beta / sumexp                # softmax weights
    part1 = np.correlate(w, 1 - x, mode="full")[n - 1 : n - 1 + n]
    part2 = np.correlate(w, x[::-1], mode="full")[n - 1 - np.arange(n)]
    grad = part1 - part2
    grad += 2.0 * lam * diff
    return f, grad.astype(float)

def l_bfgs_optimize(h0: np.ndarray, beta: float, target: float,
                    rng: np.random.Generator,
                    maxiter: int = 200, lam: float = 1e3) -> np.ndarray:
    """L‑BFGS‑B on the smooth‑max surrogate."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        f, g = smooth_obj_and_grad(x, beta, target, lam)
        return f, g

    res = minimize(fun, h0, method="L-BFGS-B", jac=True,
                   bounds=bounds, options={"maxiter": maxiter,
                                           "ftol": 1e-12,
                                           "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def generate_starts(n: int, rng: np.random.Generator):
    """Yield a few diverse initial vectors for size n."""
    # 1. interpolated parent if available
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
        else:
            interp = parent.copy()
        interp = np.clip(interp, 0.0, 1.0)
        adjust_sum_exact(interp, rng)
        yield interp
        # binary version of parent
        idx = np.argpartition(interp, -n // 2)[-n // 2:]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        adjust_sum_exact(bin_vec, rng)
        yield bin_vec

    # 2. deterministic alternating pattern
    alt = np.zeros(n)
    alt[::2] = 1.0
    adjust_sum_exact(alt, rng)
    yield alt

    # 3. random binary
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    adjust_sum_exact(bin_rand, rng)
    yield bin_rand

    # 4. random fractional uniform
    frac = rng.random(n)
    frac *= (n / 2.0) / frac.sum()
    np.clip(frac, 0.0, 1.0, out=frac)
    adjust_sum_exact(frac, rng)
    yield frac

    # 5. random fractional beta distribution (more spread)
    frac2 = rng.beta(0.5, 0.5, size=n)   # U-shaped
    frac2 *= (n / 2.0) / frac2.sum()
    np.clip(frac2, 0.0, 1.0, out=frac2)
    adjust_sum_exact(frac2, rng)
    yield frac2

def refine_solution(h: np.ndarray, n: int, rng: np.random.Generator,
                    deadline: float) -> np.ndarray:
    """Sequence of gradient & L‑BFGS refinements."""
    target = n / 2.0
    h = gradient_refine(h, rng, step=0.05, max_iter=30)
    beta_list = [1.0, 5.0, 10.0, 20.0, 50.0,
                 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
    lam = 1e3
    for beta in beta_list:
        if time.time() > deadline:
            break
        h = l_bfgs_optimize(h, beta, target, rng,
                             maxiter=250, lam=lam)
        h = gradient_refine(h, rng, step=0.02, max_iter=12)
    return h

def maybe_upsample(best_h: np.ndarray, rng: np.random.Generator,
                   deadline: float) -> tuple:
    """If time permits, double the resolution and re‑optimize."""
    n = best_h.size
    if n * 2 > 720:
        return best_h, compute_c5(best_h), n
    if time.time() > deadline:
        return best_h, compute_c5(best_h), n
    n2 = n * 2
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xt = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(xt, xs, best_h)
    adjust_sum_exact(h2, rng)
    h2 = refine_solution(h2, n2, rng, deadline)
    c2 = compute_c5(h2)
    if c2 < compute_c5(best_h):
        return h2, c2, n2
    return best_h, compute_c5(best_h), n

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97  # safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    candidate_ns = [360, 240, 180]  # try finer grids first
    for n in candidate_ns:
        if time.time() > deadline:
            break
        for h0 in generate_starts(n, rng):
            if time.time() > deadline:
                break
            h = h0.copy()
            h = adjust_sum_exact(h, rng)
            h = refine_solution(h, n, rng, deadline)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # optional up‑sampling for extra refinement
    if best_h is not None and time.time() < deadline:
        best_h, best_c, best_n = maybe_upsample(best_h, rng, deadline)

    # final safety guarantees
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    np.clip(best_h, 0.0, 1.0, out=best_h)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
