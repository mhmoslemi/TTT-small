def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def enforce_sum(h, target, rng):
    # clip
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        # adjust
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # If still not satisfied (rare), do proportional adjust
    if abs(diff) > 1e-9:
        # find free indices
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback distribute across all
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        # final fix if needed
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved:
            # occasional larger step
            if rng.random() < 0.02:
                new_h = perturb_h(h, rng, scale=step * 2.0)
                c = compute_c5(new_h)
                if c < best_c:
                    best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def simulated_annealing(h0, rng, max_iter=8000, init_temp=1e-2, final_temp=1e-5,
                        init_scale=0.5, final_scale=0.02):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def compute_corr(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

def softmax_obj(x):
    global CURRENT_TARGET, CURRENT_BETA
    corr = compute_corr(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:

                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved:
            if rng.random() < 0.02:
                new_h = perturb_h(h, rng, scale=step * 2.0)
                c = compute_c5(new_h)
                if c < best_c:
                    best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def simulated_annealing(h0, rng, max_iter=8000,
                        init_temp=1e-2, final_temp=1e-5,
                        init_scale=0.5, final_scale=0.02):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # small drift correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def local_refine(h, rng, max_iters=5000, step=0.2):
    c = compute_c5(h)
    n = h.size
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:

        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < c:
            h, c = new_h, new_c
    return h, c

def local_refine(h, rng, max_iters=5000, step=0.2):
    c = compute_c5(h)
    n = h.size
    for _ in range(max_iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        if new_c < c:
            h, c = new_h, new_c
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:
        n = 360
        target = n / 2.0
        h0 = random_feasible_vector(n, rng)

    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    tri = np.where(xs <= 1, xs, 2 - xs)
    candidates.append(enforce_sum(tri, target, rng))

    cosine = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    candidates.append(enforce_sum(cosine, target, rng))

    candidates.append(enforce_sum(xs / 2.0, target, rng))
    candidates.append(enforce_sum(1 - xs / 2.0, target, rng))

    for f in (1, 2, 3):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(s, target, rng))
        candidates.append(enforce_sum(c, target, rng))

    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    main_limit = budget_s * 0.6
    while time.time() - start_time < main_limit:
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)

        h_hc, c_hc = hill_climb_multi(start_vec, rng,
                                      max_iters=3000, init_step=0.5, n_trials=8)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        if time.time() - start_time < main_limit * 0.95:
            h_sa, c_sa = simulated_annealing(best_h, rng,
                                             max_iter=8000,
                                             init_temp=1e-2, final_temp=1e-5,
                                             init_scale=0.5, final_scale=0.02)
            if c_sa < best_c:
                best_c, best_h = c_sa, h_sa.copy()

        for beta in (50, 200, 800, 2000, 5000):
            if time.time() - start_time > main_limit:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.1:
        target_n = max(n * 2, n + 120)
        target_n = min(target_n, 960)
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            n = target_n
            target = n / 2.0
            up_h = enforce_sum(up_h, target, rng)

            up_h, _ = hill_climb_multi(up_h, rng,
                                       max_iters=4000, init_step=0.4, n_trials=6)
            up_h, _ = simulated_annealing(up_h, rng,
                                          max_iter=15000,
                                          init_temp=5e-3, final_temp=1e-5,
                                          init_scale=0.4, final_scale=0.02)

            for beta in (200, 800, 2000, 5000, 10000):
                if time.time() - start_time > budget_s:
                    break
                up_h = slsqp_optimize(up_h, beta, max_iter=400)

            if time.time() - start_time < budget_s * 0.98:
                up_h, _ = local_refine(up_h, rng, max_iters=10000, step=0.1)

            c_up = compute_c5(up_h)
            if c_up < best_c:
                best_c, best_h = c_up, up_h.copy()

            for _ in range(3):
                if time.time() - start_time > budget_s:
                    break
                rand_h = random_feasible_vector(n, rng)
                rand_h, c_rand = hill_climb_multi(rand_h, rng,
                                                  max_iters=2000,
                                                  init_step=0.3, n_trials=6)
                if c_rand < best_c:
                    best_c, best_h = c_rand, rand_h.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_h.size / 2.0
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
