def worst_shift_and_gradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.zeros_like(h)
        upper = n - k
        # derivative wrt h[i] for i<upper: term 1-h[i+k] (increase h[i] increases overlap)
        grad[:upper] += 1.0 - h[k:]
        # derivative wrt h[i+k]: decreasing h[i+k] reduces overlap => -h[i]
        grad[k:] -= h[:upper]
    return k, grad

def gradient_refine(h, rng, step=0.03, max_iter=12):
    n = h.size; target = n/2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr) - (n-1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # adjust sum by repeatedly moving mass, up to some loops
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def simulated_annealing(h0, deadline, rng):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    target = n / 2.0
    # initial temperature based on cur_c
    T = cur_c * 0.5 + 1e-9
    # cooling factor: compute number of iterations expected? We'll use exponential
    cooling = 0.9995
    # Counters for improvement
    last_improve_iter = 0
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        # decide move type
        if rng.random() < 0.6:
            # gradient guided
            k, grad = compute_worst_gradient(h)
            pos_idx = np.where(grad > 1e-12)[0]
            neg_idx = np.where(grad < -1e-12)[0]
            if pos_idx.size > 0 and neg_idx.size > 0:
                i = rng.choice(pos_idx)
                j = rng.choice(neg_idx)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta > 0:
                    delta = max_delta * rng.random() * 0.5
                else:
                    # fallback to random
                    i, j = rng.integers(0, n, size=2)
                    max_delta = min(h[i], 1.0 - h[j])
                    if max_delta <= 0:
                        continue
                    delta = max_delta * rng.random() * 0.5
            else:
                i, j = rng.integers(0, n, size=2)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0:
                    continue
                delta = max_delta * rng.random() * 0.5
        else:
            # random move
            i, j = rng.integers(0, n, size=2)
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            # sometimes larger moves
            if rng.random() < 0.05:
                delta = max_delta * rng.random()
            else:
                delta = max_delta * rng.random() * 0.3
        # propose
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c = compute_c5(new_h)
        # acceptance
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_h = h.copy()
                best_c = cur_c
                last_improve_iter = iter_count
        # cooling
        T *= cooling
        # occasional reheating if stuck
        if iter_count - last_improve_iter > 20000:
            T = cur_c * 0.5 + 1e-9
            last_improve_iter = iter_count
    return best_h, best_c

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    target = h.size / 2.0
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.03, max_iter: int = 12) -> np.ndarray:
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                if add != 0.0:
                    h[i] += add
                    diff = target - h.sum()
                    if abs(diff) < 1e-12:
                        break
    return h

def compute_worst_gradient(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.zeros_like(h)
        upper = n - k
        grad[:upper] += 1.0 - h[k:]
        grad[k:] -= h[:upper]
    return k, grad

def simulated_annealing(h0: np.ndarray, deadline: float,
                        rng: np.random.Generator) -> (np.ndarray, float):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    T = cur_c * 0.5 + 1e-12
    cooling = 0.9995
    last_improve = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if rng.random() < 0.6:                     # gradient‑guided move
            _, grad = compute_worst_gradient(h)
            pos = np.where(grad > 1e-12)[0]
            neg = np.where(grad < -1e-12)[0]
            if pos.size and neg.size:
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0.0:
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                    max_delta = min(h[i], 1.0 - h[j])
                delta = max_delta * rng.random() * 0.5
            else:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 0.0:
                    continue
                delta = max_delta * rng.random() * 0.5
        else:                                      # random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            delta = max_delta * rng.random() * (0.3 if rng.random() > 0.05 else 1.0)

        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c = compute_c5(new_h)

        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_h = h.copy()
                best_c = cur_c
                last_improve = it

        T *= cooling
        if it - last_improve > 20000:
            T = cur_c * 0.5 + 1e-12
            last_improve = it
    return best_h, best_c

def binary_swap_optimize(h_bin: np.ndarray, deadline: float,
                         rng: np.random.Generator) -> (np.ndarray, float):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    it = 0
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c = compute_c5(cur)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best = cur.copy()
                best_c = new_c
        else:
            cur[i], cur[j] = cur[j], cur[i]
        it += 1
        if it >= 200000:
            break
    return best, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    levels = [180, 360, 720]

    for n in levels:
        if time.time() > deadline:
            break
        # initialise h at this resolution
        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(xt, xs, parent)
        else:
            h0 = np.zeros(n)
            h0[rng.choice(n, n // 2, replace=False)] = 1.0
        adjust_sum_exact(h0, rng)

        # allocate time for SA (≈40% of remaining)
        rem = deadline - time.time()
        if rem < 5.0:
            break
        sa_dead = time.time() + min(rem * 0.4, 30.0)
        h_sa, c_sa = simulated_annealing(h0, sa_dead, rng)

        # gradient refinement (use whatever time is left, but at most 5 s)
        rem = deadline - time.time()
        if rem > 0.5:
            ref_dead = time.time() + min(rem * 0.2, 5.0)
            h_ref = h_sa.copy()
            while time.time() < ref_dead:
                h_ref = gradient_refine(h_ref, rng, step=0.03, max_iter=8)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c, best_n = h_ref.copy(), c_ref, n

        # optional binary rounding + local swaps if time permits
        rem = deadline - time.time()
        if rem > 4.0 and best_h is not None and best_h.size == n:
            idx = np.argpartition(best_h, -n // 2)[-n // 2:]
            h_bin = np.zeros(n)
            h_bin[idx] = 1.0
            bin_dead = deadline - 0.5
            h_bin_opt, c_bin_opt = binary_swap_optimize(h_bin, bin_dead, rng)
            if c_bin_opt < best_c:
                best_h, best_c, best_n = h_bin_opt.astype(float).copy(), c_bin_opt, n

        # set parent for next finer level
        if best_h is not None:
            parent = best_h.copy()

    # final safety pass
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = best_h.size

    return best_h.tolist(), float(best_c), int(best_n)
