import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Return a random h∈[0,1]^n with Σh = n/2."""
    target = n / 2.0
    h = rng.random(n)
    h *= target / h.sum()
    h = enforce_sum(h, target, rng)
    return h

def max_overlap_and_shift(h):
    """Return (max_correlation, shift) for h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (h.size - 1)
    return max_val, shift

def pair_transfer_optimize(h0, rng, max_iters=20000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Local search moving mass from i to i+shift to lower max overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift == 0:
            factor = init_factor
            continue
        if shift > 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx = int(np.argmax(prod))
        i = int(i_range[idx])
        j = int(j_range[idx])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    final_c = compute_c5(best_h)
    return best_h, final_c

            j += 1
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            max_eps = min(cur_h[j], 1.0 - cur_h[i])
            if max_eps <= 0.0:
                continue
            i, j = j, i
        eps = max_eps * rng.random() * 0.5
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()

    # ----- final polishing with pair‑transfer -----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 5.0:
        attempts = min(3, int(remaining // 20))
        per = remaining / (attempts + 1)
        for _ in range(attempts):
            h_opt, c_opt = pair_transfer_optimize(best_h.copy(), rng,
                                                  max_iters=40000,
                                                  time_limit=per,
                                                  init_factor=0.5)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
        # one final longer run
        h_opt, c_opt = pair_transfer_optimize(best_h.copy(), rng,
                                              max_iters=80000,
                                              time_limit=remaining * 0.9,
                                              init_factor=0.5)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
