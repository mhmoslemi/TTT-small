def sa_optimize(h0, max_seconds, rng):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = max_corr_fast(h)
    cur_c = best_c
    # temperature schedule
    T0 = 0.1 * cur_c if cur_c > 0 else 0.01
    T = T0
    it = 0
    start = time.time()
    while time.time() - start < max_seconds:
        # pick src, dst
        src = rng.integers(n)
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1
        direction = rng.random() < 0.5  # True: src->dst
        if direction:
            delta_max = min(h[src], 1.0 - h[dst])
        else:
            delta_max = min(h[dst], 1.0 - h[src])
        if delta_max <= 0.0:
            continue
        # random delta up to delta_max
        delta = rng.random() * delta_max
        if direction:
            h[src] -= delta
            h[dst] += delta
        else:
            h[dst] -= delta
            h[src] += delta
        new_c = max_corr_fast(h)
        dC = new_c - cur_c
        accept = False
        if dC < 0:
            accept = True
        else:
            if T > 1e-12 and rng.random() < np.exp(-dC / T):
                accept = True
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            if direction:
                h[src] += delta
                h[dst] -= delta
            else:
                h[dst] += delta
                h[src] -= delta
        it += 1
        # update temperature
        # For example, exponential decay: T = T0 * (0.9995 ** it)
        T = T0 * (0.9995 ** it)
        # Periodically check time
        if it % 1000 == 0 and time.time() - start >= max_seconds:
            break
    # Return best_h (copy) and best_c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    # candidate n list
    candidate_ns = [360]  # focusing on 360
    for n in candidate_ns:
        if time.time() > deadline:
            break
        # build start vectors
        start_vectors = []
        if parent is not None and parent.size != n:
            # interpolate parent to n
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vectors.append(interp)  # fractional

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # + (1 - h[i+k])
            g[k:] -= h[:upper]               # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()

        if abs(diff) < 1e-12:

        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])

def max_corr_fast(h):
    n = h.size
    g = 1.0 - h
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(g[::-1], n=N)
    conv = np.fft.irfft(H * G, n=N)
    c = conv[:2 * n - 1]

import time

from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def max_corr_fast(h):
    n = h.size
    g = 1.0 - h
    N = 2 * n
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(g[::-1], n=N)
    conv = np.fft.irfft(H * G, n=N)
    c = conv[: 2 * n - 1]
    return float(c.max() * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]
            g[k:] -= h[:up]
        h = h - step * (0.5 ** it) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def sa_optimize(h0, max_seconds, rng):
    if max_seconds <= 0:
        return h0.copy(), max_corr_fast(h0)
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = max_corr_fast(h)
    cur_c = best_c
    T0 = 0.1 * cur_c if cur_c > 0 else 0.01
    it = 0
    start = time.time()
    while time.time() - start < max_seconds:
        src = rng.integers(n)
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        direction = rng.random() < 0.5
        if direction:
            delta_max = min(h[src], 1.0 - h[dst])
        else:
            delta_max = min(h[dst], 1.0 - h[src])
        if delta_max <= 0.0:
            continue
        delta = rng.random() * delta_max
        if direction:
            h[src] -= delta
            h[dst] += delta
        else:
            h[dst] -= delta
            h[src] += delta
        new_c = max_corr_fast(h)
        dC = new_c - cur_c
        accept = dC < 0.0 or (T0 * (0.9995 ** it) > 1e-12 and rng.random() < np.exp(-dC / (T0 * (0.9995 ** it))))
        if accept:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            if direction:
                h[src] += delta
                h[dst] -= delta
            else:
                h[dst] += delta
                h[src] -= delta
        it += 1
    return best_h, best_c
