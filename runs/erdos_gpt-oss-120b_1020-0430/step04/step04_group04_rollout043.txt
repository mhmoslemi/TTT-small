def conv_counts(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    # round to nearest integer
    return np.rint(conv).astype(int)

import time, numpy as np
from scipy.optimize import minimize

# ---------- global placeholders for the smooth surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

# ---------- core evaluation ----------
def compute_c5(h):
    """Exact discrete C5 for a sampled step function."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def conv_counts(h):
    """Integer convolution counts of h with (1‑h) (no scaling)."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.rint(conv).astype(int)

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- binary helpers ----------
def random_binary(n, rng):
    half = n // 2
    h = np.zeros(n, dtype=np.int8)
    h[:half] = 1
    rng.shuffle(h)
    return h

# ---------- deterministic reduction ----------
def deterministic_reduce(h0, rng, time_limit):
    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        conv = conv_counts(h)
        prod = conv[n - 1:]                 # shift k = 0..n‑1
        max_val = int(prod.max())
        cur_c = max_val * dx
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        # try to lower the current maximum
        k = int(prod.argmax())
        # collect i candidates (1→0) that do not create new product via i‑k)
        i_cand = np.where((h[:n - k] == 1) & (h[k:] == 0))[0]
        if i_cand.size:
            if k:
                ok = (i_cand < k) | (h[i_cand - k] == 0)
                i_cand = i_cand[ok]
        if i_cand.size == 0:
            # fallback random swap
            i = rng.integers(n)
            while h[i] != 1:
                i = rng.integers(n)
            j = rng.integers(n)
            while h[j] != 0:
                j = rng.integers(n)
            h[i], h[j] = 0, 1
            continue
        i = rng.choice(i_cand)
        # collect j candidates (0→1) that keep (j+k) safe
        zeros = np.where(h == 0)[0]
        if k:
            j_ok = (zeros + k >= n) | (h[zeros + k] == 1)
            zeros = zeros[j_ok]
        if zeros.size == 0:
            j = rng.integers(n)
            while h[j] != 0:
                j = rng.integers(n)
        else:
            j = rng.choice(zeros)
        h[i], h[j] = 0, 1
    return best_h, best_c

# ---------- simulated annealing on binary vectors ----------
def anneal_binary(h0, rng, time_limit):
    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    max_temp = 0.03
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = max_temp * (1 - elapsed / time_limit)
        i = rng.integers(n)
        while h[i] != 1:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 0:
            j = rng.integers(n)
        h[i], h[j] = 0, 1
        c = compute_c5(h)
        delta = c - cur_c
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            cur_c = c
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i], h[j] = 1, 0
    return best_h, best_c

# ---------- SLSQP refinement ----------
def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj, h0, method="SLSQP", bounds=bounds,
        constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- main driver ----------
