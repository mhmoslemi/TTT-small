def compute_corr_fft(h):
    n = h.size
    b = 1 - h
    n_fft = 1 << (2*n-1).bit_length()
    H = np.fft.rfft(h, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    corr = np.fft.irfft(H * np.conj(B), n=n_fft)
    return corr[:2*n-1]

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global state for soft‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact C₅ for a step function on [0,2] (linear correlation)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    """Full linear correlation using FFT (real input)."""
    n = h.size
    n_fft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=n_fft)
    B = np.fft.rfft(1 - h, n=n_fft)
    corr = np.fft.irfft(H * np.conj(B), n=n_fft)
    return corr[: 2 * n - 1]


def compute_c5_fft(h: np.ndarray) -> float:
    """C₅ using the FFT correlation (faster, same result)."""
    n = h.size
    return np.max(compute_corr_fft(h)) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of maxₖ Oₖ."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def annealing_refine(h: np.ndarray, rng: np.random.Generator,
                     time_limit: float) -> (np.ndarray, float):
    """Simulated‑annealing style refinement using gradient‑guided transfers."""
    n = h.size
    cur_h = h.copy()
    cur_corr = compute_corr_fft(cur_h)
    cur_c = np.max(cur_corr) * (2.0 / n)

    best_h = cur_h.copy()
    best_c = cur_c

    start = time.time()
    temp0 = 0.01          # initial temperature for uphill moves
    step_scale = 0.5

    while time.time() - start < time_limit:
        # gradient of the current worst shift
        idx = np.argmax(cur_corr) - (n - 1)
        k = abs(idx)
        grad = np.ones_like(cur_h)
        if k == 0:
            grad[:] = 1.0 - 2.0 * cur_h
        else:
            grad[: n - k] -= cur_h[k:]
            grad[k:] -= cur_h[: n - k]

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        max_eps = min(cur_h[i_max], 1.0 - cur_h[i_min])
        if max_eps <= 0:
            # fall back to a random pair
            i_max = rng.integers(0, n)
            i_min = rng.integers(0, n)
            max_eps = min(cur_h[i_max], 1.0 - cur_h[i_min])
            if max_eps <= 0:
                continue

        eps = rng.random() * max_eps * step_scale
        cur_h[i_max] -= eps
        cur_h[i_min] += eps

        new_corr = compute_corr_fft(cur_h)
        new_c = np.max(new_corr) * (2.0 / n)
        delta = new_c - cur_c

        elapsed = time.time() - start
        temp = temp0 * max(0.0, 1.0 - elapsed / time_limit)

        accept = False
        if delta < 0:
            accept = True
        elif temp > 0 and rng.random() < np.exp(-delta / temp):
            accept = True

        if accept:
            cur_c = new_c
            cur_corr = new_corr
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        else:
            cur_h[i_max] += eps
            cur_h[i_min] -= eps

    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_end = start + budget_s * 0.995  # small safety margin

    # ----- initialise -----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)

    else:

        h = enforce_sum(h, rng)

    best_h = h.copy()

    # ----- soft‑max SLSQP passes (quick global polishing) -----
    for beta in (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0, 20000.0):
        if time.time() >= total_end:

        max_iter = 500 if beta < 200 else 800
        h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # ----- simulated‑annealing refinement on current resolution -----
    remaining = total_end - time.time()
    if remaining > 2.0:
        # keep a small buffer for possible up‑sampling later
        anneal_time = max(0.0, remaining - 2.0)
        h_refined, c_refined = annealing_refine(best_h, rng, anneal_time)
        if c_refined < best_c:
            best_h, best_c = h_refined, c_refined

    if time.time() < total_end - 1.0:

                if c_up_ref < best_c:
                    best_h, best_c = h_up_ref, c_up_ref

    # ----- final sanity check -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
