def multi_shift_descent(h, rng, max_iters=2000, delta=0.01, step=0.1):
    n = h.size
    best_c = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        O = np.correlate(h, 1-h, mode='full')
        O_max = O.max()
        # which indices have near max
        threshold = O_max - delta*O_max
        # get shift indices where O >= threshold
        shift_idxs = np.where(O >= threshold)[0]
        # shift offset = n-1, so actual shifts = shift_idxs - (n-1)
        # compute gradient per index i: sum over selected shifts of (1 - h[i+shift]) where i+shift index within bounds else 0.
        grad = np.zeros(n)
        for s_idx in shift_idxs:
            shift = s_idx - (n-1)
            # we need (1 - h[i+shift])
            if shift >= 0:
                grad[:n-shift] += 1 - h[shift:]
                grad[n-shift:] += 0  # out-of-range contributions zero
            else:
                # shift negative: i+shift = i-shift_neg, so for i >= -shift
                grad[-shift:] += 1 - h[:n+shift]  # shift negative => -shift >0
                # rest zero
        # gradient for decreasing mass: high grad -> decreasing reduces O. For increasing mass: low grad.
        # Choose source i with grad largest, target j with grad smallest.
        source = int(np.argmax(grad))
        target = int(np.argmin(grad))
        if source == target:
            # No move
            step *= 0.9
            continue
        max_eps = min(h[source], 1-h[target])
        if max_eps <= 0:
            step *= 0.9
            continue
        eps = min(step, max_eps)
        h[source] -= eps
        h[target] += eps
        # Clip
        # compute new C5
        new_c = compute_c5(h)
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()
            # maybe increase step
            step = min(step*1.05, 0.5)
        else:
            # revert?
            # Usually simulated annealing would accept with prob. For simplicity, revert:
            h[source] += eps
            h[target] -= eps
            step *= 0.9
    return best_h, best_c

    best_h = h.copy()

    for it in range(max_iters):

            best_c = c
            best_h = h.copy()
        # find shift(s) achieving max O
        max_shift_idx = int(np.argmax(O))
        shift = max_shift_idx - (n - 1)
        # compute partner indices for each i
        if shift >= 0:
            # i range 0..n-1-shift, partner i+shift
            # compute arrays for h_i and h_{i+shift}

            # target index j0 with h_j0<1 and partner low (h_{j0+shift}).
        else:

            h_i = h[shift_neg:]
            h_j = h[:n-shift_neg]

        # We'll compute arrays for partner_val (for each i, partner = h[i+shift] if defined else 0)
        partner = np.zeros(n)
        if shift >= 0:
            partner[:n-shift] = h[shift:]
        else:
            partner[-shift:] = h[:n+shift]
        # source candidates: indices where h>0 and partner high.
        source_candidates = np.where(h > 1e-12)[0]
        target_candidates = np.where(h < 1 - 1e-12)[0]
        if source_candidates.size == 0 or target_candidates.size == 0:
            break

        source_idx = source_candidates[np.argmax(partner[source_candidates])]
        # Choose target as argmin of partner value among targets
        target_idx = target_candidates[np.argmin(partner[target_candidates])]

    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()

    while time.time() - start < time_limit:
        O = compute_overlap_fft(h)
        O_max = O.max()

            partner[-shift:] = h[:n+shift]

        source_cands = np.where(h > 1e-12)[0]
        target_cands = np.where(h < 1 - 1e-12)[0]

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def compute_overlap_fft(h):
    n = h.size
    L = 2 * n
    fh = np.fft.rfft(h, n=L)
    gh = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(fh * gh, n=L)
    return conv[: 2 * n - 1]

# -------------------------------------------------
# Greedy worst‑shift transfer
# -------------------------------------------------
def greedy_search(h0, rng, time_limit):
    h = h0.copy()
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    eps_factor = 0.5

    while time.time() - start < time_limit:
        O = compute_overlap_fft(h)
        O_max = O.max()
        c = O_max * (2.0 / n)

        if c < best_c:
            best_c, best_h = c, h.copy()

        shift_idx = int(np.argmax(O))
        shift = shift_idx - (n - 1)

        # partner values (h shifted by `shift`)
        partner = np.zeros(n)
        if shift >= 0:
            partner[: n - shift] = h[shift:]
        else:
            partner[-shift :] = h[: n + shift]

        src_cand = np.where(h > 1e-12)[0]
        tgt_cand = np.where(h < 1 - 1e-12)[0]
        if src_cand.size == 0 or tgt_cand.size == 0:
            break

        src = src_cand[np.argmin(partner[src_cand])]  # low partner -> decrease
        tgt = tgt_cand[np.argmax(partner[tgt_cand])]  # high partner -> increase
        if src == tgt:
            src, tgt = src_cand[0], tgt_cand[0]

        max_eps = min(h[src], 1.0 - h[tgt])
        if max_eps <= 0:
            continue
        eps = max_eps * eps_factor
        if eps < 1e-9:
            eps = max_eps * 0.1

        h[src] -= eps
        h[tgt] += eps

        # evaluate improvement
        O2 = compute_overlap_fft(h)
        c2 = O2.max() * (2.0 / n)
        if c2 < c:
            eps_factor = min(0.9, eps_factor * 1.05)
        else:
            # revert
            h[src] += eps
            h[tgt] -= eps
            eps_factor = max(0.1, eps_factor * 0.7)

    return best_h, best_c

# -------------------------------------------------
# Pair‑transfer local search (from previous solution)
# -------------------------------------------------
def max_overlap_and_shift(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        # index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i, j = int(i_range[idx_max]), int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    final_c = compute_c5(best_h)
    return best_h, final_c

# -------------------------------------------------
# Smooth surrogate + SLSQP
# -------------------------------------------------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# -------------------------------------------------
# Upscaling of an existing vector
# -------------------------------------------------
def upscale_h(h_small, n_new, rng):
    n_small = h_small.size
    xs_small = (np.arange(n_small) + 0.5) * (2.0 / n_small)
    xs_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(xs_new, xs_small, h_small, left=h_small[0], right=h_small[-1])
    h_new = np.clip(h_new, 0.0, 1.0)
    return enforce_sum(h_new, n_new / 2.0, rng)

# -------------------------------------------------
# Main entry point
# -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # -----------------------------------------------------------------
    # 1. Choose resolution and initial vector
    # -----------------------------------------------------------------
    n = 720  # higher resolution than the 360‑point record
    if "initial_h_values" in globals():
        h_small = np.array(initial_h_values, dtype=float)
        if h_small.size != n:
            h0 = upscale_h(h_small, n, rng)
        else:
            h0 = enforce_sum(np.clip(h_small, 0.0, 1.0), n / 2.0, rng)
    else:
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # -----------------------------------------------------------------
    # 2. Greedy worst‑shift search (major portion of budget)
    # -----------------------------------------------------------------
    greedy_budget = budget_s * 0.55
    while time.time() - start_time < greedy_budget:
        # diverse start points
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)
        # allocate a small slice ( ≤ 12 s) for each greedy run
        slice_time = min(12.0, greedy_budget - (time.time() - start_time))
        h_g, c_g = greedy_search(start_vec, rng, slice_time)
        if c_g < best_c:
            best_h, best_c = h_g.copy(), c_g

    # -----------------------------------------------------------------
    # 3. SLSQP refinement with increasing β
    # -----------------------------------------------------------------
    beta_list = [50, 200, 800, 2000, 5000, 10000]
    slsqp_budget = budget_s * 0.30
    while time.time() - start_time < greedy_budget + slsqp_budget:
        for beta in beta_list:
            if time.time() - start_time >= greedy_budget + slsqp_budget:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=800)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_h, best_c = h_ref.copy(), c_ref
            except Exception:
                continue
        # occasional random perturb between cycles
        best_h = perturb_h(best_h, rng, scale=0.15)

    # -----------------------------------------------------------------
    # 4. Pair‑transfer fine tuning (remaining time)
    # -----------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10.0:
        h_pt, c_pt = pair_transfer_optimize(best_h, rng,
                                            max_iters=12000,
                                            time_limit=remaining * 0.9,
                                            init_factor=0.5)
        if c_pt < best_c:
            best_h, best_c = h_pt.copy(), c_pt

    # -----------------------------------------------------------------
    # 5. Final clean‑up and return
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
