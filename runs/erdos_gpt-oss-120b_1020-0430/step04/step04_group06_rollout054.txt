def gradient_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_beta = np.exp(beta * (corr - max_corr))  # shape 2n-1
    sum_exp = np.sum(exp_beta)
    w = exp_beta / sum_exp  # normalized weights
    grad = np.zeros_like(h)
    # shift index mapping: shift k from -(n-1) to (n-1) -> idx = k + n - 1
    for idx, wk in enumerate(w):
        k = idx - (n - 1)
        if wk == 0:
            continue
        if k >= 0:
            i = slice(0, n - k)
            grad[i] += wk * (1 - h[i + k])
            i2 = slice(k, n)
            grad[i2] -= wk * h[i2 - k]
        else:
            kk = -k
            i = slice(kk, n)
            grad[i] += wk * (1 - h[i - kk])
            i2 = slice(0, n - kk)
            grad[i2] -= wk * h[i2 + kk]
    return grad

def project_onto_box_and_sum(x, target):
    # ensure 0 <= x <= 1 and sum = target.
    # First clip
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Use simple iterative scaling:
    # While diff not zero:
    #   if diff > 0: increase variables not at 1 by min(diff, total_capacity)
    #   else: decrease variables not at 0.
    # We'll do proportional to remaining capacity.
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # capacity = sum(1 - x_i)
            cap = np.sum(1.0 - x)
            if cap <= 0:
                break
            # amount to add proportionally
            add = diff * (1.0 - x) / cap
            x = x + add
            # clip again (should be within)
            x = np.clip(x, 0.0, 1.0)
        else:
            # can reduce
            cap = np.sum(x)
            if cap <= 0:
                break
            sub = diff * x / cap  # diff negative, so sub negative
            x = x + sub
            x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    # If still not exact due to rounding, adjust a few entries
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust by small epsilon on first non-boundary index
        if diff > 0:
            # find index with x < 1
            idx = np.where(x < 1 - 1e-12)[0]
            if idx.size > 0:
                i = idx[0]
                x[i] += diff
        else:
            idx = np.where(x > 1e-12)[0]
            if idx.size > 0:
                i = idx[0]
                x[i] += diff
        x = np.clip(x, 0.0, 1.0)
    # final correction for numerical error
    # Ensure sum close to target
    # Small epsilon distribution
    epsilon = target - x.sum()
    if abs(epsilon) > 1e-12:
        # distribute epsilon uniformly among non-boundary variables
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if np.any(mask):
            x[mask] += epsilon / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback adjust first variable
            x[0] = np.clip(x[0] + epsilon, 0.0, 1.0)
    return x

def subgrad_step(h, lr):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)

    grad = np.zeros_like(h)

        i = slice(0, n - k)

        i2 = slice(k, n)
        grad[i2] -= h[i2 - k]

def subgrad_step(h, lr, rng):
    n = h.size
    target = n / 2.0
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    grad = np.zeros_like(h)
    if k >= 0:
        i = slice(0, n-k)
        grad[i] += (1 - h[i + k])
        i2 = slice(k, n)
        grad[i2] -= h[i2 - k]
    else:
        kk = -k
        i = slice(kk, n)
        grad[i] += (1 - h[i - kk])
        i2 = slice(0, n-kk)
        grad[i2] -= h[i2 + kk]
    new_h = h - lr * grad
    # clip
    new_h = np.clip(new_h, 0.0, 1.0)
    # adjust sum
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        # distribute diff proportionally to capacity
        # choose random index with room
        if diff > 0:
            # increase up to 1
            # compute capacity
            cap = 1.0 - new_h
            total_cap = cap.sum()
            if total_cap > 0:
                # add proportionally
                new_h += diff * cap / total_cap
            else:
                # fallback: set random
                idx = rng.integers(0, n)
                new_h[idx] = min(1.0, new_h[idx] + diff)
        else:
            # diff negative: decrease down to 0
            cap = new_h
            total_cap = cap.sum()
            if total_cap > 0:
                new_h += diff * cap / total_cap
            else:
                idx = rng.integers(0, n)
                new_h[idx] = max(0.0, new_h[idx] + diff)
        new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def anneal_optimize(h_start, rng, total_time, init_temp=0.01, final_temp=1e-5):
    # total_time: time budget
    # perform move proposals until time runs out
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    start = time.time()
    # schedule temperature exponential decay
    # compute steps based on some guess of moves per second; but we just compute T each iteration using ratio of elapsed/total.
    while True:
        elapsed = time.time() - start
        if elapsed >= total_time:
            break
        # temperature linearly (or exponentially) decaying
        # Using exponential: T = init_temp * (final_temp / init_temp) ** (elapsed / total_time)
        T = init_temp * (final_temp / init_temp) ** (elapsed / total_time)
        # propose move:
        # choose i and j random (distinct)
        i = rng.integers(0, h.size)
        j = rng.integers(0, h.size)
        if i == j:
            continue
        # compute max possible epsilon based on bounds
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps  # uniform transfer
        # try moving
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # because sum unchanged, no need to adjust.
        c_new = compute_c5(h_new)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def gradient_softmax(h, beta):
    n = h.size
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    exp_beta = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_beta)
    w = exp_beta / sum_exp
    grad = np.zeros_like(h)

import time
import numpy as np

# ----------------------------------------------------------
# Utility functions
# ----------------------------------------------------------
def compute_corr_fft(h):
    """Full correlation of h with (1-h) using FFT."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    corr = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    return corr


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:  # final correction if needed
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h with Σh = n/2 and 0≤h≤1."""
    h = np.full(n, 0.5)
    for _ in range(n * 4):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ----------------------------------------------------------
# Gradient of smooth max (softmax) objective
# ----------------------------------------------------------
def gradient_softmax(h, beta):
    """Gradient of (log‑sum‑exp β·corr)/β w.r.t. h."""
    n = h.size
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    exp_beta = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_beta)
    if sum_exp == 0.0:
        w = np.zeros_like(exp_beta)
    else:
        w = exp_beta / sum_exp
    grad = np.zeros_like(h)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            i = slice(0, n - k)
            grad[i] += wk * (1.0 - h[i + k])
            i2 = slice(k, n)
            grad[i2] -= wk * h[i2 - k]
        else:
            kk = -k
            i = slice(kk, n)
            grad[i] += wk * (1.0 - h[i - kk])
            i2 = slice(0, n - kk)
            grad[i2] -= wk * h[i2 + kk]
    return grad


def softmax_descend(h0, beta, lr, steps, rng):
    """Gradient descent on the smooth max objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for _ in range(steps):
        g = gradient_softmax(h, beta)
        h = h - lr * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
    return h

# ----------------------------------------------------------
def max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    corr = compute_corr_fft(h)
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)


def pair_transfer_optimize(
    h0, rng, max_iters=5000, time_limit=30.0, init_factor=0.5, decay=0.999
):
    """Local search moving mass from i to i+shift to lower max overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c

def anneal_optimize(h_start, rng, total_time, init_temp=0.001, final_temp=1e-6):
    """Annealing on C5 = max_k ∫h(x)(1‑h(x+k))dx."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    n = h.size
    start = time.time()
    while True:
        elapsed = time.time() - start
        if elapsed >= total_time:
            break
        # exponential temperature schedule
        T = init_temp * (final_temp / init_temp) ** (elapsed / total_time)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = compute_c5(h_new)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c


# ----------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    total_budget = budget_s

    # ------------------------------------------------------
    # 1. Choose a modest discretisation (even n)
    # ------------------------------------------------------
    n = 360
    target = n / 2.0

    # ------------------------------------------------------
    # 2. Build an initial pool of diverse candidates
    # ------------------------------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = []

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target, rng)
        candidates.append(h0)

    # binary masks
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, target, rng))

    mask2 = np.zeros(n)
    start_idx = n // 4
    mask2[start_idx : start_idx + n // 2] = 1.0
    candidates.append(enforce_sum(mask2, target, rng))

    mask3 = (np.arange(n) % 2).astype(float)
    candidates.append(enforce_sum(mask3, target, rng))

    # sinusoidal patterns
    for f in (1, 2, 3):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, target, rng))

    # a few random vectors
    for _ in range(4):
        candidates.append(random_feasible_vector(n, rng))

    # ------------------------------------------------------
    # 3. Pick the best among the pool
    # ------------------------------------------------------
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ------------------------------------------------------
    # 4. Smooth‑max gradient descent (softmax) schedule
    # ------------------------------------------------------
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]
    lr = 0.15
    for beta in beta_list:
        steps = 500 if beta <= 2000 else 250
        best_h = softmax_descend(best_h, beta, lr, steps, rng)
        c = compute_c5(best_h)
        if c < best_c:
            best_c, best_h = c, best_h.copy()
        lr *= 0.9  # gradual decay

    # ------------------------------------------------------
    # 5. Heavy pair‑transfer optimisation (most of the budget)
    # ------------------------------------------------------
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    # reserve a modest chunk for final annealing
    anneal_reserve = min(30.0, total_budget * 0.07)
    pair_time = max(0.0, remaining - anneal_reserve)

    pair_start = time.time()
    while time.time() - pair_start < pair_time:
        time_left = pair_time - (time.time() - pair_start)
        if time_left < 6.0:
            break
        # allocate ≈15 s per attempt, keeping a small buffer
        attempt_time = min(15.0, time_left - 2.0)
        start_vec = (
            perturb_h(best_h, rng, scale=0.3)
            if rng.random() < 0.6
            else random_feasible_vector(n, rng)
        )
        h_opt, c_opt = pair_transfer_optimize(
            start_vec, rng, max_iters=20000, time_limit=attempt_time, init_factor=0.5
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ------------------------------------------------------
    # 6. Fine‑tune with a final high‑beta softmax step
    # ------------------------------------------------------
    elapsed = time.time() - start
    if total_budget - elapsed > 8.0:
        best_h = softmax_descend(best_h, beta=50000, lr=0.05, steps=120, rng=rng)
        c = compute_c5(best_h)
        if c < best_c:
            best_c, best_h = c, best_h.copy()

    # ------------------------------------------------------
    # 7. Simulated annealing on the exact C5 (remaining time)
    # ------------------------------------------------------
    elapsed = time.time() - start
    anneal_time = max(0.0, total_budget - elapsed)
    if anneal_time >= 5.0:
        h_ann, c_ann = anneal_optimize(best_h, rng, total_time=anneal_time)
        if c_ann < best_c:
            best_c, best_h = c_ann, h_ann.copy()

    # ------------------------------------------------------
    # 8. Final clean‑up
    # ------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
