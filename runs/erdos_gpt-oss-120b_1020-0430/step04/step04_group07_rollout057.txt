def simulated_annealing_binary(h, rng, max_iter, init_temp=0.001, decay=0.9995):
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iter):
        # pick i where h[i]==1
        i = rng.integers(0, n)
        while h[i] == 0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] == 1:
            j = rng.integers(0, n)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0 or rng.random() < np.exp(-delta / temp):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        temp *= decay
        # early break if temp very low (approx zero)
        if temp < 1e-12:
            temp = 1e-12
    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

# globals used by the soft‑max SLSQP optimisation
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] given samples h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σ h = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of the max‑overlap."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """SLSQP optimisation of the soft‑max objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing_binary(h: np.ndarray, rng: np.random.Generator,
                               max_iter: int = 120000,
                               init_temp: float = 0.001,
                               decay: float = 0.9999):
    """Binary SA that swaps a 1 and a 0 while respecting the sum constraint."""
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for _ in range(max_iter):
        # pick a 1 and a 0
        i = rng.integers(0, n)
        while h[i] == 0.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] == 1.0:
            j = rng.integers(0, n)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        temp *= decay
        if temp < 1e-12:
            temp = 1e-12
    return best_h, best_c

            best_h, best_c, best_n = h0.copy(), c0, h0.size

    # ------------------------------------------------------------------
    # 2.   Multi‑size stochastic search (binary SA + smooth refinement)
    # ------------------------------------------------------------------
    n_candidates = [128, 144, 168, 192, 224, 256, 288]
    beta_schedule = (20.0, 80.0, 200.0, 2000.0, 8000.0)

    outer_break = False
    for n in n_candidates:
        if time.time() - start > 0.90 * budget_s or outer_break:
            break
        # a few independent restarts per size
        for _ in range(3):
            if time.time() - start > 0.90 * budget_s:
                outer_break = True
                break
            # binary start
            h_bin = np.zeros(n)
            ones = rng.choice(n, n // 2, replace=False)
            h_bin[ones] = 1.0
            # binary SA
            max_sa = min(150000, int((budget_s - (time.time() - start)) * 150))
            h_sa, c_sa = simulated_annealing_binary(
                h_bin, rng, max_iter=max_sa, init_temp=0.001, decay=0.9999
            )
            # smooth refinement (soft‑max SLSQP)
            h_ref = enforce_sum(h_sa.astype(float), rng)
            for beta in beta_schedule:
                if time.time() - start > 0.94 * budget_s:
                    break
                h_ref = slsqp_optimize(h_ref, beta, rng, max_iter=800)
            # final guided hill‑climb
            h_ref, c_ref = guided_hill_climb(
                h_ref, rng, max_iters=2500, step_scale=0.6
            )
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c, best_n = h_ref.copy(), c_ref, n
                # early exit if we already beat the target record
                if best_c < 0.3808:
                    outer_break = True
                    break
        if outer_break:
            break

    # ------------------------------------------------------------------
    # 3.   Optional up‑sampling refinement (higher resolution)
    # ------------------------------------------------------------------
    if best_h is not None and time.time() - start < 0.97 * budget_s:
        for factor in (2,):
            n_up = best_n * factor
            if n_up > 512:
                continue
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_up, x_old, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = enforce_sum(h_up, rng)
            for beta in beta_schedule:
                if time.time() - start > budget_s * 0.99:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
            h_up, _ = guided_hill_climb(
                h_up, rng, max_iters=2000, step_scale=0.5
            )
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ------------------------------------------------------------------
    # 4.   Final tidy‑up (tiny stochastic tweaks)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.998 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # sanity
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        final_c = compute_c5(best_h)
    else:
        final_c = np.inf
        best_h = np.zeros(256)
        best_n = 256

    return best_h.tolist(), float(final_c), int(best_h.size)
