def project_capped_simplex(v, target, cap=1.0):
    # Clip to [0, cap]
    v = np.clip(v, 0, cap)
    # Sum after clipping
    s = v.sum()
    if np.isclose(s, target):
        return v
    # iterative: while not converged and iteration < max_iter
    # Use while loop:
    max_iter = 10 * len(v)
    for _ in range(max_iter):
        if s > target:
            # Need to reduce sum
            # Identify indices where v > 0 (i.e., can reduce)
            mask = v > 0
            # Compute reduction per element
            excess = s - target
            # Determine amount to subtract
            # Let cnt = mask.sum()
            cnt = mask.sum()
            if cnt == 0:
                break
            delta = excess / cnt
            v[mask] -= delta
            # Clip negatives to zero
            v = np.clip(v, 0, cap)
        else:
            # Need to increase sum
            mask = v < cap
            cnt = mask.sum()
            if cnt == 0:
                break
            deficit = target - s
            delta = deficit / cnt
            v[mask] += delta
            v = np.clip(v, 0, cap)
        s = v.sum()
        if abs(s - target) <= 1e-12:
            break
    # After loop, due to rounding, we may adjust final small diff:
    diff = target - s
    if abs(diff) > 1e-12:
        # distribute diff arbitrarily among indices not at bound
        for i in range(len(v)):
            if diff == 0:
                break
            if diff > 0 and v[i] < cap:
                inc = min(diff, cap - v[i])
                v[i] += inc
                diff -= inc
            elif diff < 0 and v[i] > 0:
                dec = min(-diff, v[i])
                v[i] -= dec
                diff += dec
    return v

def correlate_full(h):
    # compute full correlation of h with (1-h) and return array length 2n-1
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

from scipy.special import logsumexp

def softmax_weights(corr, beta):
    # returns weights w (same length as corr) sum to 1; stable.
    # corr is a numpy array.
    scaled = beta * corr
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    weights = exp_scaled / sum_exp
    return weights

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def project_capped_simplex(v, target, cap=1.0):
    """Clip to [0,cap] and iteratively adjust to hit exact sum=target."""
    v = np.clip(v, 0.0, cap)
    s = v.sum()
    if abs(s - target) <= 1e-12:
        return v
    # simple water‑filling iteration
    for _ in range(10 * len(v)):
        if s > target:
            mask = v > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            excess = s - target
            delta = excess / cnt
            v[mask] -= delta
            v = np.clip(v, 0.0, cap)
        else:
            mask = v < cap
            cnt = mask.sum()
            if cnt == 0:
                break
            deficit = target - s
            delta = deficit / cnt
            v[mask] += delta
            v = np.clip(v, 0.0, cap)
        s = v.sum()
        if abs(s - target) <= 1e-12:
            break
    # final tiny correction
    diff = target - s
    if abs(diff) > 1e-12:
        for i in range(len(v)):
            if diff == 0.0:
                break
            if diff > 0.0 and v[i] < cap:
                inc = min(diff, cap - v[i])
                v[i] += inc
                diff -= inc
            elif diff < 0.0 and v[i] > 0.0:
                dec = min(-diff, v[i])
                v[i] -= dec
                diff += dec
    return v


def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_capped_simplex(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    return correlate_full(h).max() * (2.0 / h.size)


# --------------------------------------------------------------
# Smooth objective and gradient (softmax approximation)
# --------------------------------------------------------------
def soft_obj_grad(x, beta, lam, target):
    """Return (objective, gradient) for smoothed max overlap + sum penalty."""
    n = x.size
    corr = correlate_full(x)                     # length 2n‑1
    lse = logsumexp(beta * corr)
    obj = lse / beta + lam * (x.sum() - target) ** 2
    w = np.exp(beta * corr - lse)                # weights, sum to 1
    m = w.size                                   # =2n‑1
    L = 1 << ((m + n - 1).bit_length())
    w_fft = np.fft.rfft(w, L)
    w_rev_fft = np.fft.rfft(w[::-1], L)
    h_fft = np.fft.rfft(x, L)
    om_h_fft = np.fft.rfft(1.0 - x, L)
    conv_rev = np.fft.irfft(w_rev_fft * om_h_fft, L)[: m + n - 1]
    conv_wh = np.fft.irfft(w_fft * h_fft, L)[: m + n - 1]
    start = n - 1
    A = conv_rev[start:start + n]   # Σ w_ℓ (1‑h[i+ℓ])
    B = conv_wh[start:start + n]    # Σ w_ℓ h[i‑ℓ]
    grad = A - B + 2.0 * lam * (x.sum() - target)
    return obj, grad


def l_bfgs_optimize(h0, beta, lam, max_iter, rng):
    """L‑BFGS‑B with softmax objective."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return soft_obj_grad(x, beta, lam, target)

    res = minimize(fun, h0, method="L-BFGS-B", jac=True,
                   bounds=bounds, options={"maxiter": max_iter,
                                           "ftol": 1e-12, "disp": False})
    h = np.clip(res.x, 0.0, 1.0)
    h = project_capped_simplex(h, target)
    return h

def greedy_refine(h, rng, max_iters, start_step=0.2):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    step = start_step
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        new_h = perturb_h(best_h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_h, best_c = new_h, c
        if (it + 1) % decay == 0:
            step = max(step * 0.6, 1e-5)
    return best_h, best_c

            start = h_opt

        # short greedy polish
        if time.time() < main_end:
            h_g, c_g = greedy_refine(best_h, rng, max_iters=2000, start_step=0.25)
            if c_g < best_c:
                best_h, best_c = h_g, c_g

    # ------------------------------------------------------------------
    # Final refinement with limited budget
    # ------------------------------------------------------------------
    final_end = start_time + budget
    while time.time() < final_end:
        h_g, c_g = greedy_refine(best_h, rng, max_iters=3000, start_step=0.15)
        if c_g < best_c:
            best_h, best_c = h_g, c_g
        # attempt a final high‑beta L‑BFGS‑B pass
        if time.time() >= final_end:
            break
        try:
            h_opt = l_bfgs_optimize(best_h, 4000.0, 3000.0, 150, rng)
        except Exception:
            break
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ------------------------------------------------------------------
    # Clean and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_capped_simplex(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
