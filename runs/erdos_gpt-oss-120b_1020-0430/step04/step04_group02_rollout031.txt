def correlated_swap_move(h, rng):
    n = h.size
    # compute current worst lag
    _, idx = compute_c5_and_idx(h)
    k_star = idx - (n - 1)
    if k_star <= 0 or k_star >= n:
        # fallback to random perturb
        return perturb_h(h, rng, scale=0.2)
    # diff on overlapping region
    diff = h[:n - k_star] - h[k_star:]
    pos_idx = np.flatnonzero(diff > 1e-12)
    neg_idx = np.flatnonzero(diff < -1e-12)
    if pos_idx.size == 0 or neg_idx.size == 0:
        return perturb_h(h, rng, scale=0.2)
    i = rng.choice(pos_idx)
    j = rng.choice(neg_idx)
    # compute max epsilon
    eps_max = min(h[i], 1.0 - h[i + k_star], 1.0 - h[j], h[j + k_star])
    # also limit to diff values to not overshoot sign change
    eps_max = min(eps_max, diff[i] / 2.0, -diff[j] / 2.0)
    if eps_max <= 0:
        return perturb_h(h, rng, scale=0.2)
    # random epsilon fraction
    eps = rng.random() * eps_max  # uniform in [0, eps_max]
    new_h = h.copy()
    new_h[i] -= eps
    new_h[i + k_star] += eps
    new_h[j] += eps
    new_h[j + k_star] -= eps
    return new_h
```

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx

# ---------- hill climbing ----------
def hill_climb(
    h_start,
    rng,
    max_iters=800,
    init_step=0.5,
    n_trials=4,
    p_target=0.2,
    p_corr=0.2,
    p_avg=0.2,
):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        c_curr, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            r = rng.random()
            if r < p_corr:
                cand = correlated_swap_move(h, rng, k_star)
            elif r < p_corr + p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            elif r < p_corr + p_target + p_avg:
                cand = averaging_move(h, rng, k_star)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

# ---------- SLSQP smoothing ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- diverse starters ----
    candidates = [h0]
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    for f in range(1, 5):
        cand = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        candidates.append(project_box_sum(np.clip(cand, 0, 1), n / 2.0))
        cand = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(project_box_sum(np.clip(cand, 0, 1), n / 2.0))

    # Chebyshev seed
    m = int(np.sqrt(n))
    t = np.linspace(-1, 1, n)
    coeff = np.zeros(m + 1)
    coeff[-1] = 1.0
    cheb = chebval(t, coeff)  # T_m(t)
    cand = 0.5 + 0.5 * cheb
    candidates.append(project_box_sum(np.clip(cand, 0, 1), n / 2.0))

    # random seeds
    for _ in range(6):
        candidates.append(random_feasible_vector(n, rng))

    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]

    # ---- main optimisation loop ----
    while time.time() - start_time < budget_s * 0.7:
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        h_hc, c_hc = hill_climb(
            start_vec,
            rng,
            max_iters=1200,
            init_step=0.4,
            n_trials=6,
            p_target=0.2,
            p_corr=0.25,
            p_avg=0.25,
        )
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
            for beta in beta_list:
                if time.time() - start_time > budget_s * 0.7:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=500)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # ---- optional up‑sampling ----
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.15:
        target_n = min(720, max(540, n * 2))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_box_sum(h_up, target_n / 2.0)

            h_up, c_up = hill_climb(
                h_up,
                rng,
                max_iters=3000,
                init_step=0.3,
                n_trials=6,
                p_target=0.2,
                p_corr=0.25,
                p_avg=0.25,
            )
            if c_up < best_c:
                best_c, best_h, n = c_up, h_up.copy(), target_n

            for beta in beta_list:
                if time.time() - start_time > budget_s:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=500)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
