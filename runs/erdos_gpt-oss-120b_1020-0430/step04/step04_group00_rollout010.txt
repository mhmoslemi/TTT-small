def min_overlap_and_shifts(h):
   n = h.size
   corr = np.correlate(h, h, mode='full')
   # zero shift index
   zero = n-1
   # exclude zero
   # extract all values except corr[zero]
   # we want min value and indices where it occurs.
   # Since corr integer array, we can use np.argmin.
   # Use mask for indices != zero.
   # Convert to list of ks: for idx < zero: k = idx - zero; for idx > zero: k = idx - zero (positive)
   # We'll produce both signs? Actually shift k corresponds to idx - zero; we can keep positive for simplicity.
   # We'll return list of ks (excluding 0) with min value.
   # Note we might treat both negative and positive same; we can restrict to positive ks (1..n-1)
   # For shift k>0, corr[zero + k] = O(k) = O(-k). So we can use only positive ks.
   # So min over k=1..n-1.
   corr_pos = corr[zero+1:]  # length n-1 for k=1..n-1
   min_val = corr_pos.min()
   ks = np.where(corr_pos == min_val)[0] + 1   # shift values
   return int(min_val), ks

def improve_binary_with_random_restarts(h_initial, rng, max_total_iters=20000, patience=2000):
    best_h = h_initial.copy()
    best_min, _ = min_overlap_and_shifts(best_h)
    cur_h = h_initial.copy()
    cur_min = best_min
    it_no_improve = 0
    while it_no_improve < patience and max_total_iters > 0:
        # perform deterministic swaps
        cur_h = binary_swaps_max_min_overlap(cur_h, rng, max_iters=1000)
        cur_min, _ = min_overlap_and_shifts(cur_h)
        if cur_min > best_min:
            best_h = cur_h.copy()
            best_min = cur_min
            it_no_improve = 0
        else:
            it_no_improve += 1
        # random perturb: do a random swap of two bits (ensuring sum unchanged) maybe a few times
        for _ in range(10):
            # pick random 1 and 0 to swap (swap any two positions)
            ones = np.flatnonzero(cur_h == 1)

            j = rng.choice(zeros)
            cur_h[i], cur_h[j] = 0, 1
        max_total_iters -= 1000 + 10

def upsample(parent, n):
    xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    interp = np.interp(xt, xs, parent)
    return interp

import time
import numpy as np

def min_overlap_and_shifts(h):
    """Return (min_overlap, array_of_shifts) for binary h."""
    n = h.shape[0]
    corr = np.correlate(h, h, mode="full")
    zero = n - 1
    pos = corr[zero + 1 :]          # overlaps for k = 1..n-1
    min_val = int(pos.min())
    ks = np.where(pos == min_val)[0] + 1
    return min_val, ks


def compute_c5_binary(h):
    """C5 for a binary vector using the overlap relation."""
    n = h.size
    min_val, _ = min_overlap_and_shifts(h)
    return 1.0 - 2.0 * min_val / n


def binary_swaps_max_min_overlap(h, rng, max_iters=5000):
    """Greedy swap routine that tries to raise the minimal overlap."""
    n = h.shape[0]
    for _ in range(max_iters):
        min_val, ks = min_overlap_and_shifts(h)
        if min_val >= n // 2:          # perfect – cannot improve further
            break
        improved = False
        ks_shuffled = ks.copy()
        rng.shuffle(ks_shuffled)
        for k in ks_shuffled:
            rolled = np.roll(h, -k)
            A = np.where((h == 1) & (rolled == 0))[0]
            B = np.where((h == 0) & (rolled == 1))[0]
            if A.size and B.size:
                i = rng.choice(A)
                j = rng.choice(B)
                h[i] = 0
                h[j] = 1
                improved = True
                break
        if not improved:
            break
    return h


def improve_binary_random_restart(h0, rng, deadline):
    """
    Repeatedly apply deterministic swaps, and occasionally
    random perturbations, until the time budget expires.
    Returns the best binary vector seen.
    """
    h = h0.copy()
    best_h = h.copy()
    best_min, _ = min_overlap_and_shifts(best_h)

    while time.time() < deadline:
        # deterministic phase
        h = binary_swaps_max_min_overlap(h, rng, max_iters=2000)
        cur_min, _ = min_overlap_and_shifts(h)

        if cur_min > best_min:
            best_h = h.copy()
            best_min = cur_min
            # continue from this improved state
        else:
            # random perturbation to escape plateaus
            for _ in range(5):
                ones = np.flatnonzero(h == 1)
                zeros = np.flatnonzero(h == 0)
                if ones.size == 0 or zeros.size == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i], h[j] = 0, 1

        if best_min >= h.size // 2:
            break
    return best_h


def upsample_vector(v, new_n):
    """Linear interpolation of a vector from its length to new_n."""
    old_n = v.shape[0]
    xs = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xt = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xt, xs, v)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate grid sizes (all even)
    candidates = [360, 480, 540, 600, 720]

    # parent construction, if supplied
    parent = None
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)

    for n in candidates:
        if time.time() > deadline:
            break
        # --- generate starting binary vectors for this n ---
        starts = []

        if parent is not None:
            if parent.size != n:
                p_interp = upsample_vector(parent, n)
            else:
                p_interp = parent.copy()
            bin_from_parent = np.zeros(n, dtype=float)
            idx = np.argpartition(p_interp, -n // 2)[-n // 2 :]
            bin_from_parent[idx] = 1.0
            starts.append(bin_from_parent)

        # a few random binary seeds
        for _ in range(2):
