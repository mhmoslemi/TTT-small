def compute_c5_fft(h):
    n = h.size
    m = 1 << (2*n-1).bit_length()
    H = np.fft.rfft(h, n=m)

def hill_climb_sa(h_start, rng, max_iters, init_step, temp0, temp_decay, p_target, time_limit):
    cur_h = h_start.copy()
    cur_c, _ = compute_c5_and_idx(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    step = init_step
    T = temp0
    n = cur_h.size
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        # compute max lag
        # maybe compute idx only once; need correlation anyway
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx = int(np.argmax(corr))
        k_star = idx - (n - 1)

        if rng.random() < p_target:
            cand = targeted_move(cur_h, rng, k_star, scale=step)
        else:
            cand = perturb_h(cur_h, rng, scale=step)

        cand_c, _ = compute_c5_and_idx(cand)

import time
import numpy as np
from scipy.optimize import minimize

# -------------------- utilities --------------------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def max_overlap(h):
    """return C5 value for sample vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# -------------------- simulated‑annealing hill climb --------------------
def hill_climb_sa(
    h_start,
    rng,
    max_iters=10000,
    init_step=0.5,
    temp0=5e-4,
    temp_decay=0.997,
    p_target=0.3,
    time_limit=None,
):
    cur_h = h_start.copy()
    n = cur_h.size
    cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
    cur_c = np.max(cur_corr) * (2.0 / n)
    best_h = cur_h.copy()
    best_c = cur_c
    step = init_step
    T = temp0
    it = 0
    while it < max_iters:
        if time_limit is not None and time.time() > time_limit:
            break
        idx = int(np.argmax(cur_corr))
        k_star = idx - (n - 1)

        if rng.random() < p_target:
            cand = targeted_move(cur_h, rng, k_star, scale=step)
        else:
            cand = perturb_h(cur_h, rng, scale=step)

        cand_c = max_overlap(cand)
        improve = False
        if cand_c < cur_c - 1e-12:
            improve = True
        else:
            # simulated‑annealing acceptance
            prob = np.exp(-(cand_c - cur_c) / max(T, 1e-12))
            if rng.random() < prob:
                improve = True
        if improve:
            cur_h = cand
            cur_c = cand_c
            cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
        # schedule
        T = max(T * temp_decay, 1e-12)
        step = max(step * 0.99, 1e-5)
        it += 1
    return best_h, best_c


# -------------------- SLSQP refinement (soft‑max surrogate) --------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# -------------------- main driver --------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    candidates.append(project_box_sum(mask2, n / 2.0))
    for _ in range(5):

    best_h = candidates[0].copy()
