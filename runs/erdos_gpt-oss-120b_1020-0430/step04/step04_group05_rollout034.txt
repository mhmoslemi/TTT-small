import time
import numpy as np
from scipy.optimize import minimize

# ----- Global variables for SLSQP -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def enforce_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        for _ in range(n):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1]^n with exact sum n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def perturb_move(h, rng, scale=0.5):
    """Transfer a positive ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, lag, scale=0.5, top_frac=0.2):
    """Biased transfer based on contributions for the worst lag."""
    n = h.size
    lag_abs = abs(lag)
    if lag_abs >= n:
        return perturb_move(h, rng, scale)
    valid_len = n - lag_abs
    if valid_len <= 0:
        return perturb_move(h, rng, scale)
    contrib = h[:valid_len] * (1.0 - h[lag_abs:lag_abs + valid_len])
    k = max(1, int(valid_len * top_frac))
    top_idx = np.argpartition(-contrib, k - 1)[:k]
    bot_idx = np.argpartition(contrib, k - 1)[:k]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx)
    if i == j:
        j = (j + 1) % valid_len
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_move(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    """SA with cooling; returns best (h, raw_max_corr)."""
    h = h_start.copy()
    n = h.size
    cur_corr = np.correlate(h, 1.0 - h, mode="full")
    cur_raw = np.max(cur_corr)
    best_h, best_raw = h.copy(), cur_raw
    T0, Tf = 0.02 * (n / 360.0), 1e-5
    for it in range(max_iters):
        if time.time() > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        lag_idx = np.argmax(cur_corr)
        lag = lag_idx - (n - 1)
        if rng.random() < 0.6:
            new_h = targeted_perturb(h, rng, lag, scale=0.5, top_frac=0.2)
        else:
            new_h = perturb_move(h, rng, scale=0.5)
        new_corr = np.correlate(new_h, 1.0 - new_h, mode="full")
        new_raw = np.max(new_corr)
        dE = new_raw - cur_raw
        if dE < 0.0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_corr, cur_raw = new_h, new_corr, new_raw
            if new_raw < best_raw:
                best_raw, best_h = new_raw, new_h.copy()
    return best_h, best_raw

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):
    """SLSQP refinement using softmax surrogate with given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        n = x.size
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    best_h = h0.copy()
    best_raw = np.max(np.correlate(best_h, 1.0 - best_h, mode="full"))

    # ---- main stochastic search (coarse grid) ----
    main_deadline = t0 + 0.65 * budget_s
    BETA_LIST = [50, 200, 800, 3000, 10000]

    while time.time() < main_deadline:
        # random restart
        if rng.random() < 0.8:
            start_h = perturb_move(best_h, rng, scale=0.5)
        else:
            start_h = random_feasible_vector(n0, rng)

        # simulated annealing
        sa_h, sa_raw = simulated_annealing(
            start_h, rng, max_iters=20000, time_start=t0, time_budget=main_deadline
        )
        if sa_raw < best_raw:
            best_h, best_raw = sa_h.copy(), sa_raw

        # SLSQP refinement ladder
        for beta in BETA_LIST:
            if time.time() >= main_deadline:
                break
            try:
                ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=200)
                ref_raw = np.max(np.correlate(ref_h, 1.0 - ref_h, mode="full"))
                if ref_raw < best_raw:
                    best_h, best_raw = ref_h.copy(), ref_raw
            except Exception:
                continue

    # ---- optional up‑sampling to finer grid ----
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 0.15 * budget_s:
            n_target = n0 * 3  # 1080 when n0=360
            old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
            new_x = (np.arange(n_target) + 0.5) * (2.0 / n_target)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, n_target / 2.0, rng)

            # SA on finer grid
            up_h, up_raw = simulated_annealing(
                up_h,
                rng,
                max_iters=50000,
                time_start=time.time(),
                time_budget=deadline,
            )
            if up_raw < best_raw:
                best_h, best_raw = up_h.copy(), up_raw

            # SLSQP refinement on finer grid
            for beta in BETA_LIST:
                if time.time() >= deadline:
                    break
                try:
                    up_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=200)
                    up_raw = np.max(np.correlate(up_h, 1.0 - up_h, mode="full"))
                    if up_raw < best_raw:
                        best_h, best_raw = up_h.copy(), up_raw
                except Exception:
                    continue

    # ---- final quick hill‑climb ----
    cur_h = best_h.copy()
    cur_raw = best_raw
    step = 0.2
    for _ in range(8000):
        if time.time() >= deadline:
            break
        new_h = perturb_move(cur_h, rng, scale=step)
        new_raw = np.max(np.correlate(new_h, 1.0 - new_h, mode="full"))
        if new_raw < cur_raw:
            cur_h, cur_raw = new_h, new_raw
            if new_raw < best_raw:
                best_h, best_raw = new_h.copy(), new_raw
        step *= 0.9998

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(best_h.size)
