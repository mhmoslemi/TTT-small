def compute_c5(h):
    # h is float array length n
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    max_val = conv.max()
    return max_val * (2.0 / n)

import time, numpy as np, scipy.optimize as opt

def enforce_sum(h, target, rng):
    # adjust as before
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12: break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def correlate_full(x):
    n = x.shape[0]; L = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, L)
    rev = (1.0 - x)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fx * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    n = h.shape[0]
    conv = correlate_full(h)
    return conv.max() * (2.0 / n)

# Global vars for SLSQP

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # small adjustment
        x = enforce_sum(x, CURRENT_TARGET, np.random.default_rng())
    return x

def perturb_h(h, rng, scale=0.2):
    # transfer epsilon from i to j
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def correlate_full(x):
    """Full linear correlation of x with (1‑x) using FFT."""
    n = x.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, L)
    rev = (1.0 - x)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fx * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.shape[0])


# --------------------------------------------------------------
# SLSQP smoothing (optional polishing)
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1‑x_{i+k})."""
    conv = correlate_full(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=250):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # quick correction
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x


# --------------------------------------------------------------
# Small continuous perturbation (used by hill‑climb)
# --------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=3000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# --------------------------------------------------------------
# Binary helpers
# --------------------------------------------------------------
def random_binary_vector(n, rng):
    """Generate a binary vector with exactly n/2 ones."""
    perm = rng.permutation(n)
    h = np.zeros(n, dtype=float)
    h[perm[: n // 2]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = budget_s

    # -----------------------------------------------------------
    # Determine resolution and initial candidate
    # -----------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h_cont = np.clip(np.array(initial_h_values, dtype=float), 0.0, 1.0)
        h_cont = enforce_sum(h_cont, n / 2.0, rng)
        # binary version: keep the n/2 largest entries as 1
        idx_top = np.argsort(-h_cont)[: n // 2]
        best_h = np.zeros(n, dtype=float)
        best_h[idx_top] = 1.0
        best_c = compute_c5(best_h)
    else:
        n = 720
        best_h = random_binary_vector(n, rng)
        best_c = compute_c5(best_h)

    # a few extra random starts
    for _ in range(3):
        h_try = random_binary_vector(n, rng)
        c_try = compute_c5(h_try)
        if c_try < best_c:
            best_h, best_c = h_try, c_try

    # -----------------------------------------------------------
    # Simulated annealing on binary vectors
    # -----------------------------------------------------------
    sa_fraction = 0.90
    sa_deadline = start_time + total_budget * sa_fraction
    T0 = 0.2            # initial temperature
    Tf = 1e-6           # final temperature
    restart_limit = 50000

    cur_h = best_h.copy()
    cur_c = best_c
    no_improve = 0

    while time.time() < sa_deadline:
        # temperature according to elapsed portion of SA phase
        elapsed = time.time() - (start_time + total_budget * (1 - sa_fraction))
        t_frac = min(elapsed / (total_budget * sa_fraction), 1.0)
        T = T0 * ((Tf / T0) ** t_frac)

        # pick random 1 and 0 to swap
        ones = np.flatnonzero(cur_h)
        zeros = np.flatnonzero(1.0 - cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = ones[rng.integers(ones.size)]
        j = zeros[rng.integers(zeros.size)]

        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)

        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c
            no_improve = 0
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h.copy(), new_c
        else:
            no_improve += 1

        # restart if stuck for a long period
        if no_improve >= restart_limit:
            # restart near the current global best
            cur_h = best_h.copy()
            # perform a few random swaps to escape the basin
            for _ in range(5):
                ones = np.flatnonzero(cur_h)
                zeros = np.flatnonzero(1.0 - cur_h)
                if ones.size == 0 or zeros.size == 0:
                    break
                i = ones[rng.integers(ones.size)]
                j = zeros[rng.integers(zeros.size)]
                cur_h[i], cur_h[j] = 0.0, 1.0
            cur_c = compute_c5(cur_h)
            no_improve = 0

    # -----------------------------------------------------------
    # Optional continuous polishing (hill‑climb + SLSQP)
    # -----------------------------------------------------------
    remaining = total_budget - (time.time() - start_time)
    if remaining > 12.0:
        # hill‑climb (continuous perturbations)
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=5000, start_step=0.3)
        if c_hc < best_c:
            best_h, best_c = h_hc, c_hc

    remaining = total_budget - (time.time() - start_time)
    if remaining > 20.0:
        # SLSQP polishing with increasing β
        for beta in (200.0, 800.0, 2000.0, 5000.0):
            if time.time() - start_time >= total_budget:
                break
            h_opt = slsqp_optimize(best_h, beta, max_iter=250)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # -----------------------------------------------------------
    # Final cleanup
    # -----------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
