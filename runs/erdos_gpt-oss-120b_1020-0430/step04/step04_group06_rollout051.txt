def random_fourier_h(rng, n, J=5, amp=0.5):
    xs = (np.arange(n)+0.5)*(2.0/n)
    coeff_sin = rng.uniform(-amp, amp, size=J)
    coeff_cos = rng.uniform(-amp, amp, size=J)
    base = np.full(n, 0.5)
    for j in range(1,J+1):
        base += coeff_sin[j-1] * np.sin(j * np.pi * xs) + coeff_cos[j-1] * np.cos(j * np.pi * xs)
    base = np.clip(base, 0.0, 1.0)
    # enforce sum
    target = n/2
    return enforce_sum(base, target, rng)

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that sum(h) = target while keeping 0<=h<=1."""
    h = np.clip(h, 0.0, 1.0)

    n = h.size

            break

        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
