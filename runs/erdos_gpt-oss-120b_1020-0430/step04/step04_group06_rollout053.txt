def compute_c5(h):
    corr = np.convolve(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def full_conv_fft(a,b):
    n = a.size + b.size - 1
    size = 1 << ((n-1).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:n]
    return conv

def project_capped_simplex(y, target):
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo+hi)/2
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(y - (lo+hi)/2, 0.0, 1.0)
    # small correction
    diff = proj.sum() - target
    if abs(diff) > 1e-10:
        mask = (proj > 0) & (proj < 1)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] -= diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def gradient_step(h, beta, step):
    # compute corr
    v = 1.0 - h
    corr = full_conv_fft(h, v)  # length 2n-1
    max_c = np.max(corr)
    # compute w softmax
    # avoid overflow: subtract max_c
    exp_terms = np.exp(beta * (corr - max_c))
    w = exp_terms / exp_terms.sum()
    # compute convolution of w with h and w_rev with h
    conv_w = full_conv_fft(w, h)  # length 2n-1 + n -1 = 3n-2, but we only need slice
    conv_w_rev = full_conv_fft(w[::-1], h)
    n = h.size
    # slice indices n-1 to 2n-2 inclusive
    start = n-1
    end = start + n
    F = conv_w_rev[start:end]
    B = conv_w[start:end]
    g = 1.0 - F - B
    h_new = h - step * g
    return h_new

import time
import numpy as np

# ---------- helpers ---------------------------------------------------------

def full_conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real inputs)."""
    n = a.size + b.size - 1
    size = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:n]
    return conv


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.convolve(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_capped_simplex(y: np.ndarray, target: float) -> np.ndarray:
    """Projection onto {0≤x≤1, sum x = target} via bisection on λ."""
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = x.sum() - target
    if abs(diff) > 1e-12:
        mask = (x > 0.0) & (x < 1.0)
        cnt = mask.sum()
        if cnt:
            x[mask] -= diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random point in the capped simplex {0≤h_i≤1, Σh_i=n/2}."""
    y = rng.random(n)
    return project_capped_simplex(y, n / 2.0)


def get_initial_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Initial vector: up‑sample if a parent construction exists."""
    if "initial_h_values" in globals():
        src = np.array(initial_h_values, dtype=float)
        ns = src.size
        xs_src = (np.arange(ns) + 0.5) * (2.0 / ns)
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.interp(xs, xs_src, src)
        h = project_capped_simplex(h, n / 2.0)
    else:
        h = random_feasible_vector(n, rng)
    return h


def max_overlap_and_shift(h: np.ndarray):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)


def pair_transfer_optimize(
    h0: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 8000,
    time_limit: float = 30.0,
    init_factor: float = 0.5,
    decay: float = 0.999,
):
    """Move mass from i to i+shift to reduce the maximal overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    return best_h, best_c


# ---------- gradient descent on smoothed objective -------------------------

def gradient_step(h: np.ndarray, beta: float, step: float) -> np.ndarray:
    """One gradient descent step on the log‑sum‑exp surrogate."""
    v = 1.0 - h
    corr = full_conv_fft(h, v)                # length 2n‑1
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    # conv with w and w reversed
    conv_w = full_conv_fft(w, h)              # length 3n‑2
    conv_wr = full_conv_fft(w[::-1], h)       # length 3n‑2
    n = h.size
    start = n - 1
    F = conv_wr[start : start + n]
    B = conv_w[start : start + n]
    g = 1.0 - F - B
    return h - step * g


def gradient_descent(
    h0: np.ndarray,
    beta: float,
    step: float,
    max_iters: int,
    target_sum: float,
    time_limit: float,
):
    """Projected gradient descent with the capped‑simplex projection."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    n = h.size
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        h = gradient_step(h, beta, step)
        h = project_capped_simplex(h, target_sum)
        if (it & 15) == 0:
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
        # simple step decay
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    c_final = compute_c5(h)
    if c_final < best_c:
        best_c, best_h = c_final, h.copy()
    return best_h, best_c

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    start_time = time.time()

    # problem size – 480 gives a fine discretisation while staying fast
    n = 480

    # continuation schedule on beta (smoothness)

    # main loop – projected gradient descent with occasional random restarts
    for beta in beta_schedule:
        if time.time() - start_time >= budget_s * 0.78:
            break
        remaining = budget_s - (time.time() - start_time)
        # allocate iterations roughly proportional to remaining time
        max_iters = int(min(2000, max(10, remaining * 120)))
        step = max(1e-4, 0.5 / np.sqrt(beta))

        # start from current best, with a mild random perturbation
        h_start = best_h.copy()
        if rng.random() < 0.7:
            noise = rng.normal(scale=0.02, size=n)
            h_start = project_capped_simplex(h_start + noise, target_sum)

        h_new, c_new = gradient_descent(
            h_start, beta, step, max_iters, target_sum, time_limit=remaining * 0.90
        )
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new

        # a quick random restart with half the budget for this beta
        if rng.random() < 0.2:
            rem2 = budget_s - (time.time() - start_time)
            it2 = max(5, int(max_iters * 0.5))
            h_rand = random_feasible_vector(n, rng)
            h_new2, c_new2 = gradient_descent(
                h_rand, beta, step, it2, target_sum, time_limit=rem2 * 0.80
            )
            if c_new2 < best_c:
                best_h, best_c = h_new2.copy(), c_new2

        # early exit if we already beat the target
        if best_c < 0.3808:
            break

    # ---------- fine‑tuning with the dedicated pair‑transfer routine ----------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 12.0:
        attempts = max(1, int(remaining // 30))
        per_attempt = remaining / (attempts + 1)
        for _ in range(attempts):
            h_start = best_h.copy()
            if rng.random() < 0.5:
                h_start = project_capped_simplex(
                    h_start + rng.normal(scale=0.02, size=n), target_sum
                )
            h_opt, c_opt = pair_transfer_optimize(
                h_start,
                rng,
                max_iters=8000,
                time_limit=per_attempt,
                init_factor=0.5,
            )
            if c_opt < best_c:
                best_h, best_c = h_opt.copy(), c_opt
        # final polish
        h_opt, c_opt = pair_transfer_optimize(
            best_h,
            rng,
            max_iters=12000,
            time_limit=remaining * 0.90,
            init_factor=0.5,
        )
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

    # final safety projection and exact evaluation
    best_h = project_capped_simplex(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
