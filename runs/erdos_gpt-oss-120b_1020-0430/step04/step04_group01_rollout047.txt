def project_onto_hyperplane_box(h, target):
    # Find lambda such that sum clip(h_i + lambda, 0, 1) = target.
    # We'll binary search lambda in range [-1, 1] maybe.
    lo, hi = -1.0, 1.0
    for _ in range(60): # enough for double precision
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo+hi)/2, 0.0, 1.0)

    # beta possibly large, so compute w = exp(beta*(conv - Mmax))
    # Use dtype float64
    beta = CURRENT_BETA
    # Exp
    w = np.exp(beta * (conv - Mmax))
    Z = w.sum()
    # normalized weights
    w = w / Z
    # Compute A = conv of w with (1-h)
    fw = np.fft.rfft(w, M)
    f_one_minus_h = np.fft.rfft(1 - h, M)
    convA_full = np.fft.irfft(fw * f_one_minus_h, M)[:(L + n - 1)]
    # A slice
    A = convA_full[n-1:n-1+n]
    # Compute B = conv of reversed w with h
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, M)
    fh = np.fft.rfft(h, M)
    convB_full = np.fft.irfft(fw_rev * fh, M)[:(L + n - 1)]
    B = convB_full[n-1:n-1+n]
    grad = A - B
    # objective
    obj = Mmax + np.log(Z) / beta
    return obj, grad

def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    M = 1 << ((L).bit_length())  # next power of two
    fh = np.fft.rfft(h, M)
    fr = np.fft.rfft((1.0 - h)[::-1], M)
    conv = np.fft.irfft(fh * fr, M)[:L]
    return conv


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


def project_sum(h, target):
    """Project vector onto {0≤x≤1, sum(x)=target} via binary search on a shift."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:

    while time.time() < phase1_end:
        # start from a perturbed version of the current best
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.3)
        else:
            start = random_feasible_vector(n, rng)
        start = project_sum(start, n / 2.0)

        for beta in beta_schedule:
            if time.time() >= phase1_end:
                break
            try:
                h_opt = slsqp_optimize_grad(start, beta, max_iter=250)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_c, best_h = c_opt, h_opt.copy()
                start = h_opt
            except Exception:
                continue

        # final high‑β projected gradient descent
        if time.time() < phase1_end:
            CURRENT_BETA = 10000.0
            CURRENT_TARGET = n / 2.0
            h_pg = best_h.copy()
            step = 0.5
            for it in range(1000):
                if time.time() >= phase1_end:
                    break
                obj, grad = softmax_obj_and_grad(h_pg)
                h_new = h_pg - step * grad
                h_new = project_sum(np.clip(h_new, 0.0, 1.0), n / 2.0)
                new_obj, _ = softmax_obj_and_grad(h_new)
                if new_obj < obj:
                    h_pg = h_new
                    step = min(step * 1.1, 1.0)
                else:
                    step *= 0.7
            c_pg = compute_c5(h_pg)
            if c_pg < best_c:
                best_c, best_h = c_pg, h_pg.copy()

    # ----- Phase 2: binary simulated annealing ---------------------------
    phase2_end = start_time + 0.90 * budget_s
    # binary start: take top n/2 entries of best_h
    idx = np.argsort(-best_h)[: n // 2]
    bin_start = np.zeros(n)
    bin_start[idx] = 1.0
    bin_h, bin_c = binary_sa(bin_start, rng, phase2_end)
    if bin_c < best_c:
        best_h, best_c = bin_h.copy(), bin_c

    # short SLSQP polish on binary result
    if time.time() < phase2_end:
        try:
            h_opt = slsqp_optimize_grad(best_h, 2000.0, max_iter=200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_h, best_c = h_opt.copy(), c_opt
        except Exception:
            pass

    # ----- final enforce and return ---------------------------------------
    best_h = project_sum(np.clip(best_h, 0.0, 1.0), n / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
