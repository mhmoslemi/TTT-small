def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s * 0.98

    best_h = None
    best_c = np.inf
    best_n = None

    # Determine candidate n (360 if parent else maybe 180)
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        n_parent = parent.size
        candidate_ns = [n_parent]  # just use same size for now
    else:
        parent = None
        candidate_ns = [180, 240, 360]  # could try several

    # We'll iterate over candidate ns
    for n in candidate_ns:
        if time.time() > deadline:
            break

        # Build start vectors
        start_vecs = []

        # Interpolated parent if exists and size differs
        if parent is not None and parent.size != n:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp)  # fractional
            # binary version
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)
        elif parent is not None and parent.size == n:
            # direct parent fractional and binary
            start_vecs.append(parent.copy())
            idx = np.argpartition(parent, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)

        # Random binary start
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)

        # Random fractional start
        rand_frac = rng.random(n)
        target = n / 2.0
        rand_frac *= target / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)

        # Loop over each start
        for base_h in start_vecs:

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_opt(h_init, rng, time_deadline):
    h = h_init.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = max(0.1 * cur_c, 1e-4)
    decay = 0.99992  # slower decay
    while time.time() < time_deadline:
        # pick random 1 and 0
        i = rng.integers(0, n)
        while h[i] != 1.0:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        while h[j] != 0.0:
            j = rng.integers(0, n)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        T *= decay
        if T < 1e-6:
            T = 1e-6
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def binary_sa_opt(h_init, rng, time_deadline):
    h = h_init.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = max(0.1 * cur_c, 1e-4)
    decay = 0.99993
    while time.time() < time_deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        moved = False
        if k > 0:
            left = h[: n - k]
            right = h[k:]
            A = np.where((left == 1) & (right == 0))[0]
            B = np.where((left == 0) & (right == 1))[0]
            if A.size and B.size:
                i = A[rng.integers(0, A.size)]
                j = B[rng.integers(0, B.size)]
                h[i], h[j] = 0.0, 1.0
                new_c = compute_c5(h)
                delta = new_c - cur_c
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c = new_c
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                else:
                    h[i], h[j] = 1.0, 0.0
                moved = True
        if not moved:
            i = rng.integers(0, n)
            while h[i] != 1.0:
                i = rng.integers(0, n)
            j = rng.integers(0, n)
            while h[j] != 0.0:
                j = rng.integers(0, n)
            h[i], h[j] = 0.0, 1.0
            new_c = compute_c5(h)
            delta = new_c - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c = new_c
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
        T *= decay
        if T < 1e-6:
            T = 1e-6
    return best_h, best_c
