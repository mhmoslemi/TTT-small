CURRENT_TARGET = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_refine(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

            h = np.clip(h, 0.0, 1.0)

    return h

import time
import numpy as np
from scipy.optimize import minimize

# ----------- basic utilities ---------------------------------
def enforce_sum(h, target):
    """Fast projection onto [0,1] box and affine sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h


def enforce_sum_precise(h, target):
    """Deterministic exact projection onto the simplex intersected with [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0
            if not free.any():
                break
            cap = np.sum(1.0 - h[free])
            add = min(diff, cap)
            inc = add / np.sum(free)
            h[free] += inc
        else:
            free = h > 0.0
            if not free.any():
                break
            cap = np.sum(h[free])
            sub = min(-diff, cap)
            dec = sub / np.sum(free)
            h[free] -= dec
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h


def compute_c5(h):
    """Exact C5 value for a given discretisation."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def worst_lag_gradient(h):
    """Gradient of the worst (largest) overlap integral."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)          # lag in [-n+1, n-1]
    g = np.zeros_like(h)
    if lag > 0:
        if lag < n:
            g[: n - lag] += h[lag:]          # h[i+lag]
            g[lag:] += h[: n - lag]          # h[i-lag]
    elif lag < 0:
        l = -lag
        if l < n:
            g[l:] += h[: n - l]              # h[i+lag] with lag<0
            g[: n - l] += h[l:]              # h[i-lag] with lag<0
    # for lag==0 gradient is 2*h (will be ignored in the descent)
    return g, lag


# ----------- gradient descent on the worst lag ---------------
def gradient_descent(start_h, time_budget, rng, max_iters=60000, alpha0=1.5):
    cur_h = start_h.copy()
    n = cur_h.size
    target = n / 2.0
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        g, lag = worst_lag_gradient(cur_h)
        if lag == 0:
            # simple random transfer to keep motion when worst lag is zero
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                if max_eps > 0:
                    eps = (rng.random() - 0.5) * 2 * max_eps * 0.1
                    cur_h[i] -= eps
                    cur_h[j] += eps
        else:
            alpha = alpha0 * (1 - it / max_iters)
            dx = 2.0 / n
            step = alpha * dx * (g - 1.0)   # g has mean 1, so sum(step)=0
            cur_h = cur_h + step
        cur_h = enforce_sum(cur_h, target)
        cur_c = compute_c5(cur_h)
        if cur_c < best_c:
            best_c, best_h = cur_c, cur_h.copy()
        # occasional random injection to escape plateaus
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                if max_eps > 0:
                    eps = (rng.random() - 0.5) * 2 * max_eps * 0.05
                    cur_h[i] -= eps
                    cur_h[j] += eps
                    cur_h = enforce_sum(cur_h, target)
    return best_h, best_c


# ----------- SLSQP refinement with smooth max ---------------
CURRENT_TARGET = None
CURRENT_BETA = None


def _eq_sum(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h0, beta, rng, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _eq_sum}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-10, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x


# ----------- final polishing ---------------------------------
def final_polish(h, time_budget, rng):
    start = time.time()
    best_h, best_c = h.copy(), compute_c5(h)

    # SLSQP ladder (half of the remaining budget)
    beta_list = [20, 100, 500, 2000, 5000]
    for beta in beta_list:
        if time.time() - start > time_budget * 0.5:
            break
        try:
            cand = slsqp_refine(best_h, beta, rng, max_iter=200)
            cand_c = compute_c5(cand)
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
        except Exception:
            continue

    # Simple hill‑climb with random transfers (rest of the budget)
    cur_h, cur_c = best_h.copy(), best_c
    n = cur_h.size
    target = n / 2.0
    while time.time() - start < time_budget:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j:
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps > 0:
                eps = (rng.random() - 0.5) * 2 * max_eps * 0.08
                cur_h[i] -= eps
                cur_h[j] += eps
                cur_h = enforce_sum(cur_h, target)
                new_c = compute_c5(cur_h)
                if new_c < cur_c:
                    cur_c = new_c
                    if new_c < best_c:
                        best_c, best_h = new_c, cur_h.copy()
    # final exact projection
    best_h = enforce_sum_precise(best_h, best_h.size / 2.0)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
