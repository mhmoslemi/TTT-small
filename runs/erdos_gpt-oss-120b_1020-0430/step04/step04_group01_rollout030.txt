def binary_sa_local(start_bin, rng, deadline):
    current = start_bin.copy()
    current_c = compute_c5(current)
    best = current.copy()
    best_c = current_c
    n = len(current)
    # schedule:
    T0 = 0.02 * best_c
    Tf = 1e-6
    max_iters = int(1e6) # high
    it = 0
    while time.time() < deadline and it < max_iters:
        # propose a random swap
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if current[i] == current[j]: continue
        # enforce i is 1, j is 0 (swap 1->0 and 0->1)
        if current[i] < current[j]:
            i, j = j, i
        # swap
        current[i] = 0.0
        current[j] = 1.0
        new_c = compute_c5(current)
        delta = new_c - current_c
        # compute temperature
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = current.copy()
        else:
            # revert
            current[i] = 1.0
            current[j] = 0.0
        it += 1
    return best, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic feasibility utilities ----------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary_vector(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


# ---------- correlation utilities ----------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# ---------- convolution helper for gradients ----------
def conv_full(a, b):
    """Linear convolution (full) via FFT."""
    n = a.size + b.size - 1
    L = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    C = np.fft.irfft(A * B, L)[:n]
    return C


def conv_slices(w, x):
    """Return (conv_fwd_slice, conv_rev_slice) needed for gradient."""
    n = x.size
    conv_fwd = conv_full(w, x)          # length 3n-2
    conv_rev = conv_full(w[::-1], x)    # length 3n-2
    start = n - 1
    end = start + n
    return conv_fwd[start:end], conv_rev[start:end]


# ---------- smooth surrogate (log‑sum‑exp) ----------
def softmax_obj_grad(x, beta, target):
    """Smooth max of C5 and its gradient (with sum‑penalty)."""
    corr = correlate_full(x)                     # length 2n‑1
    m = corr.max()
    shifted = beta * (corr - m)
    exp_shift = np.exp(shifted)
    Z = exp_shift.sum()
    w = exp_shift / Z                            # softmax weights, sum=1
    f = m + np.log(Z) / beta                     # smooth max

    # gradient of the smooth part
    conv_fwd, conv_rev = conv_slices(w, x)
    grad = 1.0 - (conv_fwd + conv_rev)

    # quadratic penalty for the sum constraint
    diff = x.sum() - target
    pen = 1e3
    f += pen * diff * diff
    grad += 2.0 * pen * diff
    return f, grad


# ---------- continuous refinement ----------
def continuous_optimize(x0, beta_list, target, deadline):
    best_x = x0.copy()
    best_c = compute_c5(best_x)
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    for beta in beta_list:
        if time.time() >= deadline:
            break

        def fun(v):
            f, g = softmax_obj_grad(v, beta, target)
            return f, g

        res = minimize(fun, best_x, method="L-BFGS-B", jac=True,
                       bounds=bounds,
                       options={"maxfun": 300, "ftol": 1e-12, "disp": False})
        x = np.clip(res.x, 0.0, 1.0)
        c = compute_c5(x)
        if c < best_c:
            best_c, best_x = c, x.copy()
    return best_x, best_c


# ---------- binary simulated annealing ----------
def binary_sa_local(start_bin, rng, deadline):
    h = start_bin.astype(float).copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    max_iters = 300_000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        # swap 1 from i to 0 at j
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        frac = it / max_iters
        T = max(cur_c * 0.02 * (1 - frac), 1e-9)
        if delta < 0 or rng.random() < np.exp(-delta / T):

                best_c, best_h = new_c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


def binary_from_h(h, rng):
    """Binarize h by taking the top n/2 entries."""
    n = h.size
    idx = np.argsort(-h)[: n // 2]
    bin_h = np.zeros(n)
    bin_h[idx] = 1.0
    return bin_h


# ---------- deterministic patterns ----------
def thue_morse_bits(n):
    bits = np.empty(n, dtype=float)
    for i in range(n):
        bits[i] = bin(i).count("1") & 1
    return bits


def legendre_binary(p):
    # p must be prime
    bits = np.empty(p, dtype=float)
    for i in range(1, p):
        leg = pow(i, (p - 1) // 2, p)
        bits[i] = 1.0 if leg == 1 else 0.0
    bits[0] = 0.0
    return bits

    if time.time() < deadline and best_h.size * 2 <= 1440:
        h_fine = np.repeat(best_h, 2)
        h_fine = enforce_sum(h_fine, h_fine.size / 2.0, rng)
        fine_deadline = deadline
        beta_fine = [8000, 20000, 50000]
        h_fine, c_fine = continuous_optimize(h_fine, beta_fine,
                                             h_fine.size / 2.0, fine_deadline)
        if c_fine < best_c:
            best_h, best_c = h_fine, c_fine

    # ---- final polishing ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
