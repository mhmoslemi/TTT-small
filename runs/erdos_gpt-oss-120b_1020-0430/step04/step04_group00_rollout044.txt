def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = np.flatnonzero(cur_h==1).tolist()
    zeros = np.flatnonzero(cur_h==0).tolist()
    # initial temperature
    T0 = max(0.01*cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        # temperature schedule exponential decay
        T = T0 * (0.9999 ** it)
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / max(T, 1e-12)):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        # optional early exit on stagnation?
        if it % 10000 == 0 and best_c == cur_c:
            # no improvement in last 10k steps
            break
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# globals for the smooth‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Redistribute mass to satisfy Σh = n/2 exactly."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth max via log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=600):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # Exact sum correction
    adjust_sum_exact(x, rng)
    return x


def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]      # + (1 - h[i+k])
            g[k:] -= h[:up]           # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # fix sum with small random adjustments
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def binary_sa_hill_climb(h_bin, rng, deadline):
    """Simulated‑annealing hill climb on binary vectors (swap 0↔1)."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = np.flatnonzero(cur_h == 1).tolist()
    zeros = np.flatnonzero(cur_h == 0).tolist()
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = T0 * (0.9999 ** it)
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / max(T, 1e-12)):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it % 10000 == 0 and best_c == cur_c:
            break
    return best_h, best_c


def random_perturb_refine(h, rng, deadline, max_trials=30):
    """Random local perturbations + brief gradient refinement."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    trials = max_trials
    while time.time() < deadline and trials > 0:
        trials -= 1
        h2 = best_h.copy()
        m = max(1, n // 100)
        idxs = rng.choice(n, m, replace=False)
        h2[idxs] = rng.random(m)
        adjust_sum_exact(h2, rng)
        h2 = gradient_refine(h2, rng, step=0.02, max_iter=5)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_h, best_c = h2, c2
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # candidate resolutions (high → low)
    candidate_ns = [360, 300, 240, 200, 180]

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        # ---- build start vectors ---------------------------------
        starts = []

        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            starts.append(interp)                               # fractional
            bin_vec = np.zeros(n)
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec[idx] = 1.0
            starts.append(bin_vec)                              # binary

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        # random fractional (scaled to meet sum)
        rand_frac = rng.random(n)
        target = n / 2.0
        rand_frac *= target / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        starts.append(rand_frac)

        # ---- process each start ---------------------------------
        for h0 in starts:
            if time.time() >= deadline:
                break
            h = np.clip(h0, 0.0, 1.0).astype(float)
            adjust_sum_exact(h, rng)

            # brief gradient refinement
            h = gradient_refine(h, rng, step=0.02, max_iter=10)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # if binary → SA
            if np.all((h == 0) | (h == 1)):
                h_bin, c_bin = binary_sa_hill_climb(h.copy(), rng, deadline)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin.astype(float)

            # cascade of SLSQP with increasing β
            for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000):
                if time.time() >= deadline:
                    break
                max_iter = 600
                if beta >= 2000:
                    max_iter = 800
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final gradient sweep
            h = gradient_refine(h, rng, step=0.01, max_iter=12)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # binarize, evaluate, and run SA on the binary pattern
            h_bin = np.zeros_like(h)
            idx = np.argpartition(h, -n // 2)[-n // 2 :]
            h_bin[idx] = 1.0
            c_bin = compute_c5(h_bin)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin.copy(), c_bin, n
            if time.time() < deadline:
                h_bin_sa, c_bin_sa = binary_sa_hill_climb(h_bin.copy(), rng, deadline)
                if c_bin_sa < best_c:
                    best_h, best_c, best_n = h_bin_sa.copy(), c_bin_sa, n

            # random local perturbations on the current best
            if time.time() < deadline:
                h_pert, c_pert = random_perturb_refine(best_h.copy(), rng, deadline, max_trials=15)
                if c_pert < best_c:
                    best_h, best_c, best_n = h_pert.copy(), c_pert, n

    # ---- optional up‑sampling for extra refinement ------------
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 720
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000):
            if time.time() >= deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)
        h_up = gradient_refine(h_up, rng, step=0.01, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ---- fallback ------------------------------------------------
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
```

This implementation expands the search space with additional resolutions, longer β cascades, explicit binarization followed by simulated‑annealing, and occasional random perturbations, all while respecting the time budget.
