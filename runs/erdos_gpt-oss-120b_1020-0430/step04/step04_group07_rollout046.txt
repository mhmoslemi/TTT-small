def worst_gradient_multi(h, K=5):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    O = corr  # unscaled
    # indices sorted descending
    idx = np.argpartition(O, -K)[-K:]
    grad = np.zeros_like(h)
    for ind in idx:
        k = ind - (n - 1)
        # gradient for shift k
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[: n - k] -= h[k:]
            grad[k:]       -= h[: n - k]
    return grad

def worst_gradient_multi(h, K=5):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    # O values simply corr
    # We need indices of top K values
    # Use np.argpartition to get K largest
    # Ensure K <= len(corr)
    K = min(K, corr.size)
    top_idx = np.argpartition(corr, -K)[-K:]
    grad = np.zeros_like(h)

import time, numpy as np
from scipy.optimize import minimize

# global placeholders for SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    """Smooth approximation of max via log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """SLSQP on the smooth max objective."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def worst_gradient_multi(h: np.ndarray, K: int = 5) -> np.ndarray:
    """Gradient of the average of the K largest overlap terms."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    K = min(K, corr.size)
    idxs = np.argpartition(corr, -K)[-K:]
    grad = np.zeros_like(h)
    for ind in idxs:
        k = ind - (n - 1)
        kk = abs(k)
        if kk == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[: n - kk] -= h[kk:]
            grad[kk:] -= h[: n - kk]
    return grad

def guided_hill_climb_multi(h: np.ndarray, rng: np.random.Generator,
                            max_iters: int = 800, step_scale: float = 0.8,
                            K: int = 5, random_prob: float = 0.1):
    """Steepest‑descent hill climb using the multi‑gradient."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        grad = worst_gradient_multi(h, K=K)
        if rng.random() < random_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while i == j:
                j = rng.integers(0, n)

            j = int(np.argmin(grad))

                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while i == j:
                    j = rng.integers(0, n)
        max_eps = min(h[i], 1.0 - h[j])

        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % (max_iters // 5 + 1) == 0:
            step_scale = max(step_scale * 0.85, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 5) -> np.ndarray:
    """Fine‑tune using the exact worst‑shift gradient."""

            diff = target - h.sum()
    return h
