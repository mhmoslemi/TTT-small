def gradient_descent_step(h, beta, step):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_c = corr.max()
    # compute softmax weights
    # Subtract max for stability
    w = np.exp(beta * (corr - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # compute gradient
    grad = np.zeros_like(h)
    # vectorized loop
    # We'll loop over idx where w > 0 threshold
    thresh = 1e-12
    idxs = np.where(w > thresh)[0]
    for idx in idxs:
        weight = w[idx]
        s = idx - (n - 1)
        if s >= 0:
            # shift left for 1-h

import time
import numpy as np

# ---------- projection onto box + sum ----------
def project_box_sum(v, target, tol=1e-12, max_iter=80):
    """Euclidean projection of v onto {x: 0<=x<=1, sum(x)=target}."""
    lo = -np.max(v)               # makes v+lo <= 0
    hi = 1.0 - np.min(v)          # makes v+hi >= 1
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= tol:
            return proj
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ---------- random feasible vector ----------
def random_feasible(n, rng):
    v = rng.random(n)
    return project_box_sum(v, n / 2.0)


# ---------- small perturbation ----------
def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# ---------- correlation utilities ----------
def raw_max_corr(h):
    """Maximum of the raw (unscaled) correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode="full").max()


def compute_c5(h):
    """Scaled C5 for a vector h."""
    return raw_max_corr(h) * (2.0 / h.size)


# ---------- gradient for a given shift ----------
def gradient_shift(h, s):
    """Gradient of C_s = sum h_i (1-h_{i+s}) w.r.t. h."""
    n = h.size
    grad = np.zeros_like(h)
    if s >= 0:
        grad[: n - s] += (1.0 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1.0 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


# ---------- greedy local improvement ----------
def greedy_optimize(h, max_steps=2000):
    """Deterministic pairwise transfer improving the worst shift."""
    n = h.size
    for _ in range(max_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        s = idx - (n - 1)
        grad = gradient_shift(h, s)

        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break

        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break

        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if raw_max_corr(nh) < corr[idx] - 1e-12:
            h = nh
            continue

        # try half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if raw_max_corr(nh) < corr[idx] - 1e-12:
            h = nh
            continue

        break
    return h


# ---------- softmax‑weighted gradient descent ----------
def descent_step(h, beta, step):
    """One back‑tracking step of projected gradient descent on max C_s."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = corr.max()
    # softmax weights (focus on large C_s)
    w = np.exp(beta * (corr - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # weighted gradient of C_s
    grad = np.zeros_like(h)
    idxs = np.where(w > 1e-12)[0]
    for idx in idxs:
        weight = w[idx]
        s = idx - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[: n - s] -= weight * (1.0 - h[s:])
                grad[s:] -= weight * h[: n - s]
        else:
            s_abs = -s
            if n - s_abs > 0:
                grad[s_abs:] -= weight * (1.0 - h[: n - s_abs])
                grad[: n - s_abs] -= weight * h[s_abs:]

    v = h - step * grad
    h_new = project_box_sum(v, n / 2.0)
    new_max = raw_max_corr(h_new)
    return h_new, new_max, max_c


def attempt_descent(h, beta, init_step, trials=5):
    """Try decreasing step sizes until an improvement is found."""
    step = init_step
    for _ in range(trials):
        h_new, new_max, old_max = descent_step(h, beta, step)
        if new_max < old_max - 1e-12:
            return h_new, new_max, step
        step *= 0.5
    return h, old_max, None
