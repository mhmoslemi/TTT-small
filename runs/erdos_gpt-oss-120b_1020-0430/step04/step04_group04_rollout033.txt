```python
def compute_corr_full(h):
    n = h.size
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H*R, m)[:L]
    return conv * (2.0 / n)
```

def gradient_max_corr(h):
    n = h.size
    conv = compute_corr_full(h) # returns array length L, each already multiplied by dx.
    p_max = np.argmax(conv)
    k = p_max - (n - 1)
    # compute grad
    g = np.zeros_like(h)
    if k >= 0:
        # i + k range: i from 0 to n-1-k
        g[:n-k] = 1 - h[k:]
        # i - k range: i from k to n-1 => i - k from 0 to n-1-k
        g[k:] -= h[:n-k]
    else:
        k2 = -k
        # i + k range: i from k2 to n-1 => i + k = i - k2
        g[k2:] = 1 - h[:n-k2]
        # i - k range: i from 0 to n-1 - k2 => i - k = i + k2
        g[:n-k2] -= h[k2:]
    return g, conv[p_max]

def gradients_top_k(h, top=3):
    conv = compute_corr_full(h)
    idxs = np.argpartition(-conv, top)[:top] # top indices for max
    g = np.zeros_like(h)

def project_sum(v, target, max_iter=10):
    v = np.clip(v, 0, 1)
    diff = target - v.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # free indices that are not at bounds after potential addition
        # For diff>0, free are those where v < 1
        # For diff<0, free are those where v > 0
        if diff > 0:
            free = np.where(v < 1 - 1e-12)[0]
        else:
            free = np.where(v > 1e-12)[0]
        if free.size == 0:
            # cannot adjust further, break
            break
        add = diff / free.size
        # Clip add to not exceed bounds
        # For each free index, max add possible = 1-v[i] if diff>0, else -v[i] if diff<0
        if diff > 0:
            max_add = 1 - v[free]

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------

def compute_corr_full(h):
    """Full correlation array of h with (1-h) using FFT.
    Returns array of length 2*len(h)-1, already multiplied by dx = 2/n."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())          # next power of two >= L
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv * (2.0 / n)


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    return np.max(compute_corr_full(h))


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    """Simulated‑annealing refinement."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def gradients_from_conv(conv, h, top=3):
    """Average gradient of the top‐k correlation values."""
    n = h.size
    idxs = np.argpartition(-conv, top)[:top]
    g = np.zeros_like(h)
    for p in idxs:
        k = p - (n - 1)
        if k >= 0:
            g[:n - k] += 1 - h[k:]
            g[k:] -= h[:n - k]
        else:
            k2 = -k
            g[k2:] += 1 - h[:n - k2]
            g[:n - k2] -= h[k2:]
    g /= top
    return g, conv[idxs].max()

        remaining = budget_s - (time.time() - start)
        if remaining > 2.0:
            max_iter = int(min(8000, remaining * 12))
            h_sa, c_sa = simulated_annealing(
                best_h,
                rng,
                max_iter=max_iter,
                init_temp=5e-3,
                final_temp=1e-6,
                init_scale=0.3,
                final_scale=0.01,
            )
            if c_sa < best_c:
                best_c, best_h = c_sa, h_sa.copy()
                n = target_n

    # ------------------------------------------------------------------
    # Final clean‑up and return
    # ------------------------------------------------------------------
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
