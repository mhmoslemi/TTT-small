def softmax_obj_and_grad(h, beta, target, lam):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)  # for numeric stability
    bc = beta * (corr - max_corr)
    exp_bc = np.exp(bc)
    sum_exp = exp_bc.sum()
    lse = max_corr + np.log(sum_exp) / beta
    obj = lse + lam * (h.sum() - target) ** 2

    # gradient of lse
    w = exp_bc / sum_exp  # shape (2n-1)
    n = h.size
    grad_O = np.zeros_like(h)
    # loop over k indices
    # using vectorized loops for each k:
    # For k from -(n-1) to n-1, offset idx = k + n - 1
    for idx_k, w_k in enumerate(w):
        if w_k == 0.0:
            continue
        k = idx_k - (n - 1)
        if k >= 0:
            # left indices 0..n-1-k
            i_left = slice(0, n - k)
            # right indices k..n-1
            i_right = slice(k, n)
            # compute contributions
            # left: (1 - h[i_right])
            grad_O[i_left] += w_k * (1 - h[i_right])
            # right: -h[i_left]
            grad_O[i_right] += w_k * (-h[i_left])
        else:
            kp = -k
            # idx = range 0..n-1-kp
            idx = np.arange(n - kp)
            # left part: h[i] is second factor -> -h[i+kp]
            grad_O[idx] += w_k * (-h[idx + kp])
            # right part: h[i+kp] as first factor -> (1 - h[i])
            grad_O[idx + kp] += w_k * (1 - h[idx])
    # gradient of penalty
    grad = grad_O + 2 * lam * (h.sum() - target)
    return obj, grad

def lbfgsb_optimize(h0, beta, lam, max_iter=400):

    target = n / 2.0

        obj, grad = softmax_obj_and_grad(x, beta, target, lam)
        return obj, grad
    res = minimize(fun, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum
    diff = target - x.sum()
    # use random redistribution like earlier
    rng = np.random.default_rng(0) # maybe reuse global rng
    n = x.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C₅ for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random h in [0,1]^n with Σh = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the currently worst shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr) - (n - 1)
    grad = np.zeros_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1200, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def softmax_obj_and_grad(h: np.ndarray, beta: float,
                         target: float, lam_: float):
    """Soft‑max objective (≈ max O_k) with quadratic sum‑penalty."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    # log‑sum‑exp (stable)
    bc = beta * (corr - max_corr)
    exp_bc = np.exp(bc)
    sum_exp = exp_bc.sum()
    lse = max_corr + np.log(sum_exp) / beta
    obj = lse + lam_ * (h.sum() - target) ** 2

    # weights for each shift
    w = exp_bc / sum_exp
    grad_O = np.zeros_like(h)

    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - (n - 1)
        if k >= 0:
            left_len = n - k
            if left_len:
                i_left = slice(0, left_len)
                i_right = slice(k, n)
                grad_O[i_left] += wk * (1 - h[i_right])
                grad_O[i_right] += wk * (-h[i_left])
        else:
            kp = -k
            L = n - kp
            if L:
                idxs = np.arange(L)
                grad_O[idxs] += wk * (-h[idxs + kp])
                grad_O[idxs + kp] += wk * (1 - h[idxs])
    grad = grad_O + 2.0 * lam_ * (h.sum() - target)
    return obj, grad


def lbfgsb_optimize(h0: np.ndarray, beta: float, lam_: float,
                    rng: np.random.Generator, max_iter: int = 300) -> np.ndarray:
    """L‑BFGS‑B minimisation of soft‑max C₅ with sum‑penalty."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        obj, grad = softmax_obj_and_grad(x, beta, target, lam_)
        return obj, grad

    res = minimize(fun, h0, method="L-BFGS-B", jac=True,
                   bounds=bounds, options={"maxiter": max_iter,
                                           "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, rng)
    return x
