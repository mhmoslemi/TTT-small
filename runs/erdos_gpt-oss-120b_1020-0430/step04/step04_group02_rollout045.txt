def grad_softmax(x, beta):
    # compute correlation g (length 2n-1)
    g = np.correlate(x, 1 - x, mode='full')
    # scaling? Not needed maybe multiply by dx not needed for direction.
    # compute w = exp(beta * g)
    # To avoid overflow, subtract max before exponent.
    maxg = np.max(beta * g)
    w = np.exp(beta * g - maxg)
    w_sum = np.sum(w)
    w = w / w_sum

    # compute convolution with w and w_rev
    w_arr = w  # length 2n-1
    # w_rev = w_arr[::-1]
    # Compute convolution via FFT
    n = x.size
    size = 1 << ((3*n - 1).bit_length())
    X = np.fft.rfft(x, size)
    W = np.fft.rfft(w_arr, size)
    W_rev = np.fft.rfft(w_arr[::-1], size)
    conv = np.fft.irfft(W * X, size)  # length size
    conv_rev = np.fft.irfft(W_rev * X, size)

    # slice:
    start = n - 1
    sum_ws_x_shift = conv[start:start+n]
    sum_ws_x_shift_rev = conv_rev[start:start+n]

    grad = 1.0 - (sum_ws_x_shift + sum_ws_x_shift_rev)
    # multiply dx if needed for consistent scaling.
    grad *= (2.0 / n)  # dx
    return grad

import time

# --------------------------------------------------------------
# Helper: projection onto [0,1] while enforcing exact sum constraint
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)
