def soft_weights(h, beta):
    c = np.correlate(h, 1 - h, mode='full')
    maxc = c.max()
    w = np.exp(beta * (c - maxc))
    w /= w.sum()
    return w, c

def softmax_grad(h, beta):
    N = h.size
    w, c = soft_weights(h, beta)
    grad = np.ones(N)
    # lag0
    w0 = w[N-1]
    grad -= 2 * w0 * h
    # positive lags

        if wk == 0:

        grad[:N-lag] -= wk * h[lag:]
        grad[lag:]   -= wk * h[:N-lag]
    return grad

def softmax_grad(h, beta):
    N = h.size
    c = np.correlate(h, 1 - h, mode='full')
    maxc = c.max()
    w = np.exp(beta * (c - maxc))
    w_sum = w.sum()
    w /= w_sum
    # Build w_total
    w_total = np.empty(N)
    w_total[0] = 2 * w[N-1]  # lag 0 contribution double
    if N > 1:
        pos = w[N: ]  # after index N-1
        neg = w[N-2::-1]  # before index N-1 reversed
        w_total[1:] = pos[:N-1] + neg[:N-1]
    # compute convolution
    conv = np.convolve(h, w_total, mode='full')
    grad = np.ones(N) - conv[N-1: 2*N-1]
    return grad

import time, numpy as np

def compute_c5(h: np.ndarray) -> float:
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    N = h.size
    target = N / 2.0
    diff = target - h.sum()
    it = 0
    while abs(diff) > 1e-12 and it < 5000:
        i = rng.integers(0, N)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    return h

def softmax_grad(h: np.ndarray, beta: float) -> np.ndarray:
    N = h.size
    c = np.correlate(h, 1 - h, mode="full")
    maxc = c.max()
    w = np.exp(beta * (c - maxc))
    w /= w.sum()
    w_total = np.empty(N)
    w_total[0] = 2.0 * w[N - 1]
    if N > 1:
        pos = w[N:]          # k = 1..N-1
        neg = w[N - 2::-1]   # k = -1..-(N-1)
        w_total[1:] = pos[: N - 1] + neg[: N - 1]
    conv = np.convolve(h, w_total, mode="full")
    grad = np.ones(N) - conv[N - 1 : 2 * N - 1]
    return grad

def softmax_descent(h0: np.ndarray, beta: float, rng: np.random.Generator,
                    iters: int = 800, step0: float = 0.3) -> np.ndarray:
    h = h0.copy()
    N = h.size
    target = N / 2.0
    for t in range(iters):
        g = softmax_grad(h, beta)
        eta = step0 * (0.99 ** t)
        h -= eta * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def worst_gradient(h: np.ndarray) -> np.ndarray:
    N = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (N - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.ones(N)
    g[: N - k] -= h[k:]
    g[k:] -= h[: N - k]
    return g

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1500, step_scale: float = 0.8):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        g = worst_gradient(h)
        nh = guided_perturb(h, g, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c
