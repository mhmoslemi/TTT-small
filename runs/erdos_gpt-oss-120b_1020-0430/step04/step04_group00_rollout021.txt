def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.shape[0]
    target = n/2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_step(h, rng, step):
    n = h.shape[0]
    # compute worst shift
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    g = np.zeros_like(h)
    if k == 0:
        # derivative of sum h_i (1 - h_i) = 1 - 2 h_i
        g[:] = 1.0 - 2.0 * h
    else:
        upper = n - k
        g[:upper] += 1.0 - h[k:]
        g[k:] -= h[:upper]
    # perform line search
    # compute candidate step sizes
    # ensure we don't exceed bounds: compute max feasible α for each i
    # The feasible α must keep h_i - α*g_i in [0,1].
    # Since we want h' = h - α g, we compute α_max_i = (h_i) / g_i if g_i>0 else (1 - h_i) / -g_i if g_i<0.
    # Actually we need α such that 0 <= h_i - α g_i <= 1.

    # if g_i > 0: h_i - α g_i >= 0 -> α <= h_i / g_i; also h_i - α g_i <= 1 -> α >= (h_i - 1) / g_i (negative)
    # if g_i < 0: h_i - α g_i >= 0 -> α >= h_i / g_i (negative); and <= 1 -> α <= (h_i - 1) / g_i (positive)
    # So we can compute α_max positive = min over i where g_i>0 of h_i/g_i, and over i where g_i<0 of (1 - h_i)/(-g_i). This is max step that stays in bounds.
    # We'll compute bound = min positive bound
    pos = g > 1e-12

    # to avoid division by zero

    max_alpha = min(bound_pos, bound_neg)

        direction = rng.choice([-1, 1])
        if direction == 1:
            # move from i to j
            max_delta = min(0.1, cur_h[i], 1.0 - cur_h[j])
        else:
            max_delta = min(0.1, cur_h[j], 1.0 - cur_h[i])
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta
        if direction == -1:
            delta = -delta
        # propose new h
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # ensure bounds (should be satisfied)
        # compute new C5
        c_new = compute_c5(h_new)
        accept = False
        if c_new < cur_c:
            accept = True
        else:
            # acceptance probability
            if rng.random() < np.exp(-(c_new - cur_c) / T):
                accept = True
        if accept:
            cur_h, cur_c = h_new, c_new
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
        # update temperature
        T *= 0.9995
        iter_count += 1
        if iter_count % 5000 == 0:
            # occasional reset of T based on current best
            T = base_T_factor * best_c + 1e-6
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=600):
    # similar to earlier but maybe we add a small penalty for sum deviation
    n = h0.size
    target = n / 2.0
    # define objective
    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        # compute log-sum-exp
        # use stable version
        lse = m + np.log(np.sum(np.exp(bc - m)))
        return lse / beta
    # constraints: equality sum
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    # use SLSQP
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    # adjust sum exactly
    adjust_sum_exact(x, rng)
    return x

import time
import numpy as np
import scipy.optimize as opt

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute mass so that Σh = n/2 exactly, keeping 0≤h≤1."""
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_step(h: np.ndarray, rng: np.random.Generator, step: float) -> np.ndarray:
    """Projected gradient step against the worst shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr) - (n - 1))
    k = abs(idx)

    g = np.zeros_like(h)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        upper = n - k
        g[:upper] += 1.0 - h[k:]
        g[k:] -= h[:upper]

    # feasible step size (keep within [0,1])
    pos = g > 1e-12
    neg = g < -1e-12
    bound_pos = np.min(h[pos] / g[pos]) if np.any(pos) else np.inf
    bound_neg = np.min((1.0 - h[neg]) / (-g[neg])) if np.any(neg) else np.inf
    max_alpha = min(bound_pos, bound_neg)
    if not np.isfinite(max_alpha) or max_alpha <= 0:
        max_alpha = 1.0
    alpha = min(step * max_alpha, max_alpha)

    h_new = h - alpha * g
    np.clip(h_new, 0.0, 1.0, out=h_new)
    adjust_sum_exact(h_new, rng)
    return h_new

def stochastic_anneal(h: np.ndarray, rng: np.random.Generator,
                     deadline: float, base_T_factor: float = 0.02):
    """Continuous simulated‑annealing on the mass distribution."""
    n = h.shape[0]
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    T = base_T_factor * cur_c + 1e-12
    it = 0
    while time.time() < deadline:
        i, j = rng.choice(n, 2, replace=False)
        # direction: 1 = i→j, -1 = j→i
        direction = rng.choice([-1, 1])
        if direction == 1:
            max_delta = min(0.1, cur_h[i], 1.0 - cur_h[j])
        else:
            max_delta = min(0.1, cur_h[j], 1.0 - cur_h[i])
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta
        if direction == -1:
            delta = -delta
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c_new = compute_c5(h_new)
        if c_new < cur_c or rng.random() < np.exp(-(c_new - cur_c) / T):
            cur_h, cur_c = h_new, c_new
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
        T *= 0.9995
        it += 1
        if it % 5000 == 0:
            T = base_T_factor * best_c + 1e-12
    return best_h, best_c

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 600) -> np.ndarray:
    """SLSQP on the smooth log‑sum‑exp surrogate of the max."""
    n = h0.size
    target = n / 2.0

    def softmax_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        lse = m + np.log(np.sum(np.exp(bc - m)))
        return lse / beta

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(softmax_obj, h0, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def upsample_and_refine(h: np.ndarray, rng: np.random.Generator,
                        deadline: float):
    """Doubling the resolution and polishing."""
    n = h.shape[0]
    if n * 2 > 720:
        return h, compute_c5(h)
    n_up = n * 2
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
    h_up = np.interp(xt, xs, h)
    adjust_sum_exact(h_up, rng)

    for _ in range(15):
        if time.time() > deadline:
            break
        h_up = gradient_step(h_up, rng, step=0.02)

    for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000):
        if time.time() > deadline:
            break
        h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)

    for _ in range(15):
        if time.time() > deadline:
            break
        h_up = gradient_step(h_up, rng, step=0.02)

    return h_up, compute_c5(h_up)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None

    # try a couple of resolutions; shuffle to vary order
    n_candidates = [360, 480, 240, 180]
    rng.shuffle(n_candidates)

    for n in n_candidates:
        if time.time() > deadline:
            break

        starts = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            starts.append(interp)                     # fractional
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            starts.append(bin_vec)                    # binary

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= (n / 2) / rand_frac.sum()
        adjust_sum_exact(rand_frac, rng)
        starts.append(rand_frac)

        for base_h in starts:
            if time.time() > deadline:
                break
            h = base_h.copy()
            np.clip(h, 0.0, 1.0, out=h)
            adjust_sum_exact(h, rng)

            # a few gradient refinements
            for _ in range(20):
                if time.time() > deadline:
                    break
                h = gradient_step(h, rng, step=0.04)

            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # stochastic annealing on the current vector
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
            h_sa, c_sa = stochastic_anneal(h.copy(), rng, sub_dead)
            if c_sa < best_c:
                best_h, best_c, best_n = h_sa.copy(), c_sa, n
            h = h_sa  # continue from annealed state

            # cascade of SLSQP optimizations (smooth‑max surrogate)
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
                if time.time() > deadline:
                    break
                max_iter = 800 if beta >= 200 else 400
                h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final moderate gradient sweep
            for _ in range(20):
                if time.time() > deadline:
                    break
                h = gradient_step(h, rng, step=0.02)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # optional up‑sampling if time permits
    if best_h is not None and time.time() < deadline:
        h_up, c_up = upsample_and_refine(best_h, rng, deadline)
        if c_up < best_c:
            best_h, best_c = h_up, c_up
            best_n = best_h.shape[0]

    # safety fallback
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.shape[0])
