def softmax_and_grad(x, beta):
    n = x.size
    c = correlate_full(x)    # length 2n-1
    # compute w
    bcz = beta * c
    max_bc = bcz.max()
    exp_bc = np.exp(bcz - max_bc)
    sum_exp = exp_bc.sum()
    w = exp_bc / sum_exp
    # compute conv2 = (w * x) convolution (linear) using FFT
    L = 1 << ((2*n - 1 + n - 1).bit_length())  # next power of 2 >= 3n-2
    w_arr = w
    x_arr = x
    fw = np.fft.rfft(w_arr, L)
    fx = np.fft.rfft(x_arr, L)
    conv2_full = np.fft.irfft(fw * fx, L)
    conv2 = conv2_full[n-1:n-1+n]
    # conv1 = (w_rev * x)
    w_rev = w_arr[::-1]
    fw_rev = np.fft.rfft(w_rev, L)
    conv1_full = np.fft.irfft(fw_rev * fx, L)
    conv1 = conv1_full[n-1:n-1+n]
    grad = 1.0 - (conv1 + conv2)
    f = (max_bc + np.log(sum_exp)) / beta
    return f, grad
```

def lbfgs_optimize(x0, beta, penalty, max_iter=500):
    n = x0.size
    target = n/2.0
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        f, grad = softmax_and_grad(x, beta)
        diff = x.sum() - target
        f_total = f + penalty * diff * diff
        grad_total = grad + 2 * penalty * diff
        return f_total, grad_total
    res = minimize(fun, x0, method='L-BFGS-B', jac=True, bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12})
    x_opt = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = target - x_opt.sum()
    if abs(diff) > 1e-12:
        # adjust via simple scaling
        x_opt = enforce_sum(x_opt, target, np.random.default_rng())
    return x_opt

import time

from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def random_binary(n, rng):
    """Random binary vector with exactly n/2 ones."""
    x = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, n // 2, replace=False)
    x[idx] = 1.0
    return x


# --------------------------------------------------------------
# Correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Softmax (log‑sum‑exp) and its gradient for continuous phase
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    global CURRENT_BETA
    c = correlate_full(x)                     # c_k = Σ x_i(1‑x_{i+k})
    bc = CURRENT_BETA * c
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def softmax_and_grad(x, beta):
    """Return (value, gradient) of the smooth max for a given β."""
    n = x.size
    # correlation c = Σ x_i(1‑x_{i+k}) = sum_h - autocorr
    c = correlate_full(x)                     # length 2n‑1
    bc = beta * c
    m = bc.max()
    exp_bc = np.exp(bc - m)
    sum_exp = exp_bc.sum()
    w = exp_bc / sum_exp                     # softmax weights, length 2n‑1

    # Convolution of w with x (linear) → conv2
    L = 1 << ((3 * n - 2).bit_length())       # ≥ 3n‑2
    fw = np.fft.rfft(w, L)
    fx = np.fft.rfft(x, L)
    conv2_full = np.fft.irfft(fw * fx, L)
    conv2 = conv2_full[n - 1:n - 1 + n]

    # Convolution of reversed w with x → conv1
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, L)
    conv1_full = np.fft.irfft(fw_rev * fx, L)
    conv1 = conv1_full[n - 1:n - 1 + n]

    grad = 1.0 - (conv1 + conv2)              # ∇ softmax
    f = (m + np.log(sum_exp)) / beta
    return f, grad

# L‑BFGS‑B optimizer using the analytic gradient
# --------------------------------------------------------------
def lbfgs_optimize(x0, beta, penalty, max_iter=400):
    """L‑BFGS‑B on the smoothed objective with quadratic sum‑penalty."""
    n = x0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        f, g = softmax_and_grad(x, beta)
        diff = x.sum() - target
        f_tot = f + penalty * diff * diff
        g_tot = g + 2.0 * penalty * diff
        return f_tot, g_tot

    res = minimize(fun, x0, method="L-BFGS-B", jac=True,
                   bounds=bounds, options={"maxiter": max_iter, "ftol": 1e-12})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny adjustment)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, target, rng)
    return x
