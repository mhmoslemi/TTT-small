def compute_c5(h):
    n = h.size
    L = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    conv = np.fft.irfft(fh * np.conj(fh), L)[:2*n-1]
    min_corr = conv.min()
    return 1.0 - (2.0 / n) * min_corr

def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(10_000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Start from a binary half‑filled vector and add small random noise."""
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    # add a few random transfers to break symmetry
    for _ in range(n * 3):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or h[i] == h[j]:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.4
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    conv = np.fft.irfft(fh * np.conj(fh), L)[: 2 * n - 1]
    min_corr = conv.min()
    if min_corr < 0:
        min_corr = 0.0
    return 1.0 - (2.0 / n) * min_corr

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

            else:
                s = min_idx - (2 * n - 1)
            if s == 0:
                break
            if s > 0:
                mism1 = np.where((best_h[: n - s] == 1) & (best_h[s:] == 0))[0]
                mism0 = np.where((best_h[: n - s] == 0) & (best_h[s:] == 1))[0]
                if mism1.size == 0 or mism0.size == 0:
                    break
                i = mism1[rng.integers(0, mism1.size)]
                j = mism0[rng.integers(0, mism0.size)]
            else:
                ss = -s
                mism1 = np.where((best_h[ss:] == 1) & (best_h[: n - ss] == 0))[0]
                mism0 = np.where((best_h[ss:] == 0) & (best_h[: n - ss] == 1))[0]
                if mism1.size == 0 or mism0.size == 0:
                    break
                i = mism1[rng.integers(0, mism1.size)] + ss
                j = mism0[rng.integers(0, mism0.size)] + ss
            # swap
            best_h[i], best_h[j] = best_h[j], best_h[i]
            new_c = compute_c5(best_h)
            if new_c < best_c:
                best_c = new_c
            else:
                # revert if not beneficial
                best_h[i], best_h[j] = best_h[j], best_h[i]

    # ----------------------------------------------------------
    # Phase 4 – final high‑beta polishing
    # ----------------------------------------------------------
    phase4_end = start_time + budget_s
    for beta in [15000.0, 30000.0]:
        if time.time() >= phase4_end:
            break
        try:
            opt_h = slsqp_optimize(best_h, beta, max_iter=600)
        except Exception:
            continue
        c_val = compute_c5(opt_h)
        if c_val < best_c:
            best_c, best_h = c_val, opt_h.copy()

    # ----------------------------------------------------------
    # Final cleanup
    # ----------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
