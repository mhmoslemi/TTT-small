def compute_c5(h):
    n = h.shape[0]

    min_conv = conv[:n-1].min()

    pos_min = conv[n:].min()
    min_conv = min(min_conv, pos_min)
    c = 1.0 - min_conv * (2.0 / n)
    return float(c)

def fractional_move(h, step_max):
  i = rng.integers(n)
  j = rng.integers(n)
  while j == i:
    j = rng.integers(n)
  # delta limited such that after moving delta from i to j, both remain [0,1]

  delta_min = -min(step_max, 1 - h[i], h[j]) # negative delta would move from j to i
  # we pick delta uniformly between delta_min and delta_max
  if delta_max < delta_min:

  h_new[i] -= delta

            add = min(diff, 1.0 - h[i])

    right = conv[n:]

    # Determine range of i such that i and i+k are valid indices

    # find candidates
    hi = h[i_vals]
    hk = h[i_vals + k]
    cand_i = i_vals[(hi > 0.999) & (hk < 1e-6)]

    if new_c <= cur_c:
        return h, new_c
    else:
        # revert
        h[i], h[j] = h[j], h[i]
        return h, cur_c

import time, math
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    # smooth max of the overlap (log-sum-exp) scaled by beta
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def adjust_sum_exact(h, rng):
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def compute_c5(h):
    n = h.shape[0]
    conv = np.correlate(h, h, mode="full")
    # exclude zero shift at position n-1
    min_left = conv[: n - 1].min()
    min_right = conv[n:].min()
    min_conv = min(min_left, min_right)
    return 1.0 - min_conv * (2.0 / n)


def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            up = n - k
            g[:up] += 1.0 - h[k:]          # (1 - h[i+k])
            g[k:] -= h[:up]                # -h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def attempt_worst_shift_improve(h, cur_c, rng):
    n = h.shape[0]
    conv = np.correlate(h, h, mode="full")
    left = conv[: n - 1]
    right = conv[n:]
    if left.min() < right.min():
        k = np.argmin(left) - (n - 1)      # negative shift
    else:
        k = np.argmin(right) + 1          # positive shift
    if k > 0:
        idxs = np.arange(0, n - k)
    else:
        idxs = np.arange(-k, n)
    hi = h[idxs]
    hk = h[idxs + k]
    cand_i = idxs[(hi > 0.999) & (hk < 1e-6)]
    cand_j = idxs[(hi < 1e-6) & (hk > 0.999)]
    if cand_i.size == 0 or cand_j.size == 0:
        return h, cur_c
    i = rng.choice(cand_i)
    j = rng.choice(cand_j)
    h[i], h[j] = h[j], h[i]
    new_c = compute_c5(h)
    if new_c <= cur_c:
        return h, new_c
    h[i], h[j] = h[j], h[i]
    return h, cur_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    # decide problem size
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        n = parent.size
    else:
        parent = None
        n = 360

    best_h = None
    best_c = np.inf
    best_n = n

    restarts = 3
    for r in range(restarts):
        if time.time() > deadline:
            break

        if r == 0 and parent is not None:
            h = parent.copy()
        else:
            h = np.zeros(n)
            ones = rng.choice(n, n // 2, replace=False)
            h[ones] = 1.0

        h = np.clip(h, 0.0, 1.0)
        adjust_sum_exact(h, rng)

        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c, best_h, best_n = cur_c, h.copy(), n

        # Simulated annealing
        T = 0.02
        decay = 0.9995
        while time.time() < deadline:
            if rng.random() < 0.7:
                # binary swap
                ones_idx = np.flatnonzero(h > 0.999)
                zeros_idx = np.flatnonzero(h < 1e-6)
                if ones_idx.size == 0 or zeros_idx.size == 0:
                    break
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                h[i], h[j] = h[j], h[i]          # swap
                new_c = compute_c5(h)
                accept = False
                if new_c < cur_c:
                    accept = True
                else:
                    if T > 0 and rng.random() < math.exp(-(new_c - cur_c) / T):
                        accept = True
                if accept:
                    cur_c = new_c
                    if cur_c < best_c:
                        best_c, best_h = cur_c, h.copy()
                else:
                    h[i], h[j] = h[j], h[i]      # revert
            else:
                # fractional move
                i = rng.integers(n)
                j = rng.integers(n)
                while j == i:
                    j = rng.integers(n)
                max_delta = min(T, h[i], 1.0 - h[j])
                min_delta = -min(T, 1.0 - h[i], h[j])
                if max_delta <= min_delta:
                    continue
                delta = rng.uniform(min_delta, max_delta)
                h[i] -= delta
                h[j] += delta
                new_c = compute_c5(h)
                accept = False
                if new_c < cur_c:
                    accept = True
                else:
                    if T > 0 and rng.random() < math.exp(-(new_c - cur_c) / T):
                        accept = True
                if accept:
                    cur_c = new_c
                    if cur_c < best_c:
                        best_c, best_h = cur_c, h.copy()
                else:
                    h[i] += delta
                    h[j] -= delta

            T = max(T * decay, 1e-9)

            # occasional deterministic improvement
            if rng.random() < 0.02:
                h, cur_c = attempt_worst_shift_improve(h, cur_c, rng)
                if cur_c < best_c:
                    best_c, best_h = cur_c, h.copy()

        # deterministic gradient refinement
        h = gradient_refine(h, rng, step=0.02, max_iter=12)
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()

        # cascade of smooth‑max SLSQP
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() > deadline:
                break
            try:
                h = slsqp_optimize(h, beta, rng, max_iter=500)
            except Exception:
                continue
            cur_c = compute_c5(h)
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()

        # final gradient sweep
        h = gradient_refine(h, rng, step=0.02, max_iter=12)
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()

    # optional up‑sampling for a final polish if time permits
    if best_h is not None and time.time() < deadline and best_h.shape[0] * 2 <= 720:
        n_up = best_h.shape[0] * 2
        xs = np.linspace(0.0, 2.0, best_h.shape[0], endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=500)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_c, best_h, best_n = c_up, h_up, n_up

    # fallback safety
    if best_h is None:
        best_n = n
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.shape[0])
