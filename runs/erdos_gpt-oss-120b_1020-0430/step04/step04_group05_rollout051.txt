def slsqp_optimize(h0, beta, rng, max_iter=400):
    n = h0.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exact as before
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    if lag < 0:
        lag = -lag
    if lag >= n:
        return perturb_random(h, rng, scale)
    valid = np.arange(n - lag)
    contrib = h[valid] * (1.0 - h[valid + lag])
    if contrib.size == 0:
        return perturb_random(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.8 else rng.integers(0, n)
    # ensure source i has enough mass, target j space
    if h[i] < 1e-12:
        # fallback
        return perturb_random(h, rng, scale)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # fallback
        return perturb_random(h, rng, scale)
    eps = max_eps * rng.random() * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    target = n / 2.0
    h = rng.random(n)
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5_corr(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    scaled = corr * (2.0 / n)
    return float(np.max(scaled)), scaled

def compute_c5(h):
    c, _ = compute_c5_corr(h)
    return c

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * rng.random() * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    if lag < 0:
        lag = -lag
    if lag >= n or lag == 0:
        return perturb_random(h, rng, scale)
    valid = np.arange(n - lag)
    contrib = h[valid] * (1.0 - h[valid + lag])
    if contrib.size == 0:
        return perturb_random(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.8 else rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_random(h, rng, scale)
    eps = max_eps * rng.random() * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def simulated_annealing(start_h, rng, max_iters, deadline, T0=0.05, Tf=1e-6):
    h = start_h.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        if rng.random() < 0.7:
            lag = np.argmax(cur_corr) - (n - 1)
            new_h = targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.6)
        else:
            new_h = perturb_random(h, rng, scale=0.6)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c

def slsqp_optimize(h0, beta, rng, max_iter=300):
    n = h0.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full") * (2.0 / n)
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.97  # leave a small margin

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # Phase 1: long simulated annealing
    max_iters = int(5e6)  # effectively limited by deadline
    start_vec = perturb_random(best_h, rng, scale=0.4)
    sa_h, sa_c = simulated_annealing(start_vec, rng, max_iters, deadline)
    if sa_c < best_c:
        best_h, best_c = sa_h, sa_c

    # Phase 2: SLSQP refinement ladder
    beta_list = [20, 80, 200, 800, 2000, 5000]
    for beta in beta_list:
        if time.time() > deadline:
            break
        ref_h = slsqp_optimize(best_h, beta, rng, max_iter=400)
        ref_c = compute_c5(ref_h)
        if ref_c < best_c:
            best_h, best_c = ref_h, ref_c

    # Phase 3: second annealing on the refined solution (if time)
    if time.time() < deadline:
        remaining_iters = int(2e6)
        sa_h2, sa_c2 = simulated_annealing(best_h, rng, remaining_iters, deadline)
        if sa_c2 < best_c:
            best_h, best_c = sa_h2, sa_c2

    # Final local hill‑climb (zero temperature)
    cur_h, cur_c = best_h.copy(), best_c
    step = 0.2
    for _ in range(4000):
        if time.time() > deadline:
            break
        nh = perturb_random(cur_h, rng, scale=step)
        nc = compute_c5(nh)
        if nc < cur_c:
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        step *= 0.9995

    # Clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
