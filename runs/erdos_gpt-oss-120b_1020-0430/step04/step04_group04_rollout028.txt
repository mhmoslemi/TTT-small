def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)
```

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final small correction
    if abs(diff) > 1e-12:
        # distribute uniformly to free indices
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb(h_start, rng, max_iters=2000, init_step=0.3, n_trials=4):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = 0.9
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved:
            # occasional random bigger move
            new_h = perturb_h(h, rng, scale=step * 2)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * decay, 1e-4)
    return h, best_c

import time
import numpy as np
from scipy.optimize import differential_evolution

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------

def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, init_step=0.3, n_trials=4):
    """Simple stochastic hill‑climb."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = 0.9
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * decay, 1e-4)
    return h, best_c


def shape_with_plateau(w, n):
    """Symmetric piecewise‑linear shape with a flat top width w (0≤w≤2)."""
    s = (2.0 - w) / 2.0                # ramp length on each side
    h0 = 2.0 / (2.0 + w)               # plateau height, guarantees integral = 1
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.empty_like(xs)
    left = xs <= s
    h[left] = h0 * xs[left] / s
    mid = (xs > s) & (xs <= s + w)
    h[mid] = h0
    right = xs > s + w
    h[right] = h0 * (2.0 - xs[right]) / s
    return h


def search_plateau(n, rng, time_budget):
    """Coarse‑to‑fine grid search over w for the plateau family."""
    best_h = None
    best_c = np.inf
    # coarse grid
    ws = np.linspace(0.0, 1.8, 81)   # w beyond ~1.8 gives poor C5
    for w in ws:
        h = shape_with_plateau(w, n)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, best_w = c, h.copy(), w
    # fine refinement around best_w
    step = 0.2
    while step > 1e-4 and time.time() < time_budget:
        candidates = [best_w - step, best_w, best_w + step]
        for w in candidates:
            if w < 0.0 or w > 2.0:
                continue
            h = shape_with_plateau(w, n)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h, best_w = c, h.copy(), w
        step *= 0.5
    return best_h, best_c

# Main entry point
# --------------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # final resolution (≤ 1000 points)
    n = 720
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 1) Parametric DE on a low‑dimensional monotone spline
    # ------------------------------------------------------------------
    m = 13                         # number of knots for symmetric spline
    t_knots = np.linspace(0.0, 1.0, m)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    t = np.abs(xs - 1.0)

    # initial population: random + a few structured members
    popsize = 12
    init_pop = rng.uniform(0.0, 1.0, size=(popsize, m))
    # triangular shape
    init_pop[0] = 1.0 - t_knots
    # quadratic shapes (parameter a∈[0.75,1])
    for j, a in enumerate(np.linspace(0.75, 1.0, 3)):
        b = a**3 / 0.5625
        quad = np.maximum(0.0, a - b * t_knots**2)
        if 1 + j < popsize:
            init_pop[1 + j] = quad
    # sinusoidal shape
    init_pop[4] = 0.5 + 0.5 * np.cos(np.pi * t_knots)
    # constant (average) shape
    if popsize > 5:
        init_pop[5] = 0.5

    bounds = [(0.0, 1.0)] * m

    def de_obj(z):
        p = np.sort(z)[::-1]                  # enforce monotone decreasing
        raw_h = np.interp(t, t_knots, p)
        s_raw = raw_h.sum()
        if s_raw < target_sum:
            # heavy penalty to keep average ≥0.5
            return 10.0 + (target_sum - s_raw) * 100.0
        scale = target_sum / s_raw
        h = raw_h * scale
        return compute_c5(h)

    max_de_iter = 180
    # Respect remaining budget for DE
    de_time_limit = start_time + budget_s * 0.5
    while time.time() < de_time_limit:
        result = differential_evolution(
            de_obj,
            bounds,
            init=init_pop,
            maxiter=max_de_iter,
            popsize=popsize,
            polish=False,
            seed=seed,
            updating='deferred',
            workers=1,
        )
        break   # single run is sufficient; loop only for safety

    best_z = result.x
    best_p = np.sort(best_z)[::-1]
    raw_h = np.interp(t, t_knots, best_p)
    h_de = enforce_sum(raw_h, target_sum, rng)
    c_de = compute_c5(h_de)

    # ------------------------------------------------------------------
    # 2) Independent search in the plateau family
    # ------------------------------------------------------------------
    plate_time_limit = start_time + budget_s * 0.75
    h_plate, c_plate = search_plateau(n, rng, plate_time_limit)

    # ------------------------------------------------------------------
    # 3) Compare and pick the better of the two constructions
    # ------------------------------------------------------------------
    if c_plate < c_de:
        best_h, best_c = h_plate, c_plate
    else:
        best_h, best_c = h_de, c_de

    # ------------------------------------------------------------------
    # 4) If an initial construction is provided, give it a chance
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            # up‑sample / down‑sample to required size
            old_x = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
            new_x = xs
            h0 = np.interp(new_x, old_x, h0)
        h0 = enforce_sum(h0, target_sum, rng)
        c0 = compute_c5(h0)
        if c0 < best_c:
            best_h, best_c = h0, c0
        # local refinement of the provided vector
        if time.time() < start_time + budget_s * 0.9:
            h0_ref, c0_ref = hill_climb(best_h, rng,
                                         max_iters=3000, init_step=0.4, n_trials=8)
            if c0_ref < best_c:
                best_h, best_c = h0_ref, c0_ref

    # ------------------------------------------------------------------
    # 5) Final short hill‑climb to polish the chosen solution
    # ------------------------------------------------------------------
    time_left = budget_s - (time.time() - start_time)
    if time_left > 5.0:
        iters = int(min(6000, time_left * 10))
        best_h, best_c = hill_climb(best_h, rng,
                                    max_iters=iters, init_step=0.3, n_trials=6)

    # ------------------------------------------------------------------
    # 6) Clean‑up and final return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
