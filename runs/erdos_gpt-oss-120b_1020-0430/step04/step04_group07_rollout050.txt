def batch_exchange(h, grad, rng):
    pos_inds = np.where(grad > 1e-8)[0]
    neg_inds = np.where(grad < -1e-8)[0]
    if len(pos_inds) == 0 or len(neg_inds) == 0:
        return h
    # sort
    pos_sorted = pos_inds[np.argsort(-grad[pos_inds])]  # descending
    neg_sorted = neg_inds[np.argsort(grad[neg_inds])]   # ascending (most negative)
    nh = h.copy()
    i_ptr = 0
    j_ptr = 0
    while i_ptr < len(pos_sorted) and j_ptr < len(neg_sorted):
        i = pos_sorted[i_ptr]
        j = neg_sorted[j_ptr]
        # max epsilon we can move from i to j
        max_eps = min(nh[i], 1.0 - nh[j])
        if max_eps <= 1e-12:
            # cannot move; advance pointer with zero capacity
            if nh[i] <= 1e-12:
                i_ptr += 1
            if nh[j] >= 1 - 1e-12:
                j_ptr += 1
            continue
        # move some fraction, maybe full
        eps = max_eps * rng.random()  # random fraction
        # Could also "full transfer" to accelerate
        nh[i] -= eps
        nh[j] += eps
        # update pointers
        if nh[i] <= 1e-12:
            i_ptr += 1
        if nh[j] >= 1 - 1e-12:
            j_ptr += 1
    return nh
```

        grad = worst_gradient(h)

import time, numpy as np
from scipy.optimize import minimize

# globals for SLSQP soft‑max objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h)=len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random h ∈[0,1]^n with Σh = n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def reduce_worst_shift(h: np.ndarray, rng: np.random.Generator,
                       max_iters: int = 200) -> np.ndarray:
    """Transfer mass from positive‑gradient to negative‑gradient sites."""
    n = h.size
    for _ in range(max_iters):
        grad = worst_gradient(h)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        eps = min(h[i], 1.0 - h[j])
        if eps <= 1e-14:
            break
        # move full feasible amount (or random fraction for diversity)
        eps *= rng.random()
        h[i] -= eps
        h[j] += eps
    return h


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1200, step_scale: float = 0.7):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * step_scale
                h[i] -= eps
                h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h, h = c, h.copy(), h.copy()
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 6) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf

    # start from parent if available

        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    n_candidates = [120, 144, 168, 192, 216, 240, 256, 288, 320, 352, 384]

    # main search
    while time.time() - start < 0.75 * budget_s:
        n = int(rng.choice(n_candidates))
        fresh = rng.random() < 0.2

        if not fresh and best_h is not None:
