    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    w_raw = np.exp(beta * (corr - max_corr))

def enforce_sum(h, target, rng):

    diff = target - h.sum()

    # if diff < 0, reduce from random indices with >0

            i = rng.choice(idx)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility helpers ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def compute_c5(h):
    """C5 using FFT for speed."""
    N = h.shape[0]
    L = 2 * N - 1
    f1 = np.fft.rfft(h, n=L)
    rev = (1 - h)[::-1]
    f2 = np.fft.rfft(rev, n=L)
    corr = np.fft.irfft(f1 * f2, n=L)
    return np.max(corr) * (2.0 / N)

def compute_softmax_c5_and_grad(h, beta):
    """Soft‑max approximation of max C5 and its gradient."""
    N = h.shape[0]
    L = 2 * N - 1
    f1 = np.fft.rfft(h, n=L)
    rev = (1 - h)[::-1]
    f2 = np.fft.rfft(rev, n=L)
    corr = np.fft.irfft(f1 * f2, n=L)
    max_corr = np.max(corr)
    w_raw = np.exp(beta * (corr - max_corr))
    w_sum = np.sum(w_raw)
    w = w_raw / w_sum
    grad = np.zeros(N)
    for idx, ws in enumerate(w):
        if ws == 0.0:
            continue
        s = idx - (N - 1)
        if s >= 0:
            grad[:N - s] += ws * (1 - h[s:])
            grad[s:] -= ws * h[:N - s]
        else:
            s_abs = -s
            grad[s_abs:] += ws * (1 - h[:N - s_abs])
            grad[:N - s_abs] -= ws * h[s_abs:]
    soft_c = max_corr + np.log(w_sum) / beta
    return soft_c, grad

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    N = x.shape[0]
    L = 2 * N - 1
    f1 = np.fft.rfft(x, n=L)
    rev = (1 - x)[::-1]
    f2 = np.fft.rfft(rev, n=L)
    corr = np.fft.irfft(f1 * f2, n=L)
    max_corr = np.max(corr)
    w_raw = np.exp(CURRENT_BETA * (corr - max_corr))
