import time, numpy as np
from scipy.optimize import minimize

# ---------- global placeholders for constraints ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with N samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h in-place to satisfy Σh = N/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple iterative redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1]^n with sum n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)

def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

# ---------- soft‑max SLSQP ----------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, rng)
    return x

# ---------- gradient‑based hill climbing ----------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the current worst‑overlap shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:]      -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def round_to_binary(h: np.ndarray) -> np.ndarray:
    """Return binary vector with exactly half the entries =1 (largest values)."""
    n = h.size
    m = n // 2
    idx = np.argpartition(-h, m - 1)[:m]
    hb = np.zeros_like(h)
    hb[idx] = 1.0
    return hb

# ---------- main driver ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # ---- start from parent if available ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    # candidate sample sizes (even numbers)
    n_candidates = [96, 120, 144, 168, 192, 216, 240, 256, 288, 320, 352, 384]

    # ----- main multi‑restart search -----
    while time.time() - start < 0.80 * budget_s:
        n = int(rng.choice(n_candidates))
        fresh = rng.random() < 0.12  # occasional fresh start

        if not fresh and best_h is not None:
            if best_n == n:
                h = best_h.copy()
            else:
                # linear interpolation to new size
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
        else:
            # random start: half binary, half fractional
            if rng.random() < 0.55:
                h = random_binary_vector(n, rng)
            else:
                h = random_feasible_vector(n, rng)

        # quick hill climb
        h, c = guided_hill_climb(h, rng, max_iters=1800, step_scale=0.7)

        # SLSQP soft‑max refinement with increasing β
        for beta in (20., 80., 200., 800., 3000., 8000., 20000.):
            if time.time() - start > 0.94 * budget_s:
                break
            maxit = 800 if beta >= 200 else 500
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # fine gradient tidy‑up
        h = gradient_refine(h, rng, step=0.015, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # another hill‑climb pass
        h, c = guided_hill_climb(h, rng, max_iters=2000, step_scale=0.55)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ----- up‑sampling refinement (up to 512 points) -----
    if best_h is not None and time.time() - start < 0.97 * budget_s:
        while best_n < 512 and time.time() - start < budget_s * 0.96:
            n_up = min(512, best_n * 2)
            if n_up == best_n:
                break
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = enforce_sum(h_up, rng)

            # hill‑climb at higher resolution
            h_up, c_up = guided_hill_climb(h_up, rng,
                                           max_iters=2500, step_scale=0.65)

            # SLSQP passes (moderate β)
            for beta in (20., 200., 2000., 8000.):
                if time.time() - start > budget_s * 0.97:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

            # try binary rounding of the refined continuous solution
            h_bin = round_to_binary(h_up)
            c_bin = compute_c5(h_bin)
            if c_bin < best_c:
                best_h, best_c, best_n = h_bin.copy(), c_bin, n_up

            # tiny gradient tidy‑up
            h_up = gradient_refine(h_up, rng, step=0.01, max_iter=8)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

            # prepare for possible further up‑sampling
            best_h, best_n = best_h.copy(), best_n

    # ----- final stochastic tweaking until time runs out -----
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity: clip and enforce sum one last time
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
