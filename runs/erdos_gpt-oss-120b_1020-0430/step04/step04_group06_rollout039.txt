def simulated_annealing(h0, rng, time_limit):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    current_h = h0.copy()
    current_c = best_c
    start = time.time()
    T0 = 0.01  # initial temperature? Need scale: difference in C is maybe ~0.001. So set T0 ~0.0005?
    # Using schedule: T = T0 * (1 - elapsed/limit)**alpha, or exponential.
    while time.time() - start < time_limit:
        # choose epsilon magnitude: small random up to maybe 0.2 * max_eps
        i = rng.integers(0, n)
        # choose j maybe random or i+shift? Could choose random j.
        j = rng.integers(0, n)
        if i == j: continue
        max_eps = min(current_h[i], 1 - current_h[j])
        if max_eps <= 0: continue
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.5  # uniform within bound
        # propose new_h:
        new_h = current_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # compute new c
        new_c = compute_c5(new_h)
        # acceptance
        # if new_c <= current_c: accept
        # else accept with prob exp(-(new_c - current_c)/temp)
        # compute temp based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_limit
        # Temperature schedule: T = T0 * (1 - frac) # linear cooling
        temp = T0 * (1 - frac)  # at end temp=0
        if new_c < current_c or rng.random() < np.exp(-(new_c - current_c) / (temp + 1e-12)):
            current_h = new_h
            current_c = new_c
            # update best if improved
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
    return best_h, best_c

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        # final uniform correction
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            # fallback loops
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h
```

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    # adjust sum just in case
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h = enforce_sum(h, target, rng)
    return h

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.4, n_trials=6):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        # clip for safety
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    # final check
    final_c = compute_c5(best_h)
    return best_h, final_c

def simulated_annealing_optimize(h0, rng, time_limit=10.0):
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    start = time.time()
    # initial temperature based on typical C5 variation
    T0 = 0.004  # adjust as needed
    while time.time() - start < time_limit:
        # propose transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.5
        if eps == 0.0:
            continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5(new_h)
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = T0 * (1.0 - frac)  # linear cooling
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / (temp + 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        # occasional targetted pair move
        if rng.random() < 0.05:
            # do a pair transfer move with larger factor
            max_val, shift = max_overlap_and_shift(cur_h)
            # compute product for shift
            if shift >= 0:
                i_range = np.arange(n - shift)
                j_range = i_range + shift
            else:
                i_range = np.arange(-shift, n)
                j_range = i_range + shift
            if i_range.size > 0:
                prod = cur_h[i_range] * (1.0 - cur_h[j_range])
                idx_max = int(np.argmax(prod))
                i_pt = int(i_range[idx_max])
                j_pt = int(j_range[idx_max])
                max_eps_pt = min(cur_h[i_pt], 1.0 - cur_h[j_pt])
                if max_eps_pt > 0:
                    eps_pt = 0.7 * max_eps_pt
                    new_h2 = cur_h.copy()
                    new_h2[i_pt] -= eps_pt
                    new_h2[j_pt] += eps_pt
                    new_c2 = compute_c5(new_h2)
                    # accept based on current temperature as well
                    if new_c2 < cur_c or rng.random() < np.exp(-(new_c2 - cur_c) / (temp + 1e-12)):
                        cur_h = new_h2
                        cur_c = new_c2
                        if new_c2 < best_c:
                            best_c = new_c2
                            best_h = new_h2.copy()
    return best_h, best_c

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=2000):
    """Adjust h so that Σ h_i = target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        # final uniform correction, then clean up any leftovers
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible vector h with ∑h = n/2 and 0≤h≤1."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):

        j = rng.integers(0, n)

        h[j] += eps

def max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx - (n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)


def pair_transfer_optimize(

    max_iters=5000,
    time_limit=30.0,
    init_factor=0.5,
    decay=0.999,
):

    h = h0.copy()

    best_h = h.copy()

    start = time.time()

        cur_c = max_val * (2.0 / n)

        factor = max(1e-4, factor * decay)

CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=250):
    """Refine h using SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x
