def enforce_sum(h, target, rng, max_iter=10000):
    h = np.clip(h,0,1)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # choose idx where we can add/subtract
        if diff > 0:
            # need to increase sum: select idx where h[i] < 1
            eligible = np.where(h < 1 - 1e-12)[0]
        else:
            eligible = np.where(h > 1e-12)[0]
        if eligible.size == 0:
            break
        i = rng.choice(eligible)
        max_add = (1 - h[i]) if diff > 0 else -h[i]
        add = min(abs(diff), abs(max_add)) * (1 if diff > 0 else -1)
        h[i] += add
        diff = target - h.sum()
    # final iteration
    # adjust by uniform small shift
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h,0,1)
        # ensure sum
        h = enforce_sum(h, target, rng, max_iter=1000) # may recurs but limited
    return h

    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (h.size - 1)
    return corr[idx], shift

def enforce_sum(h, target, rng, max_iter=10000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # Simple proportional scaling to correct large diff
    if abs(diff) > 1e-12:
        # distribute diff uniformly first
        h += diff / n
        # clip again
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

    for _ in range(max_iter):
        if abs(diff) < 1e-12:

        diff = target - h.sum()
    # final clip

CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp approximation of max_k Σ_i x_i(1‑x_{i+k})."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def enforce_sum(h, target, rng, max_iter=10000):
    """Clip to [0,1] then adjust to hit the exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = rng.choice(np.where(h < 1.0)[0])
            add = min(diff, 1.0 - h[idx])
        else:
            idx = rng.choice(np.where(h > 0.0)[0])
            add = max(diff, -h[idx])
        h[idx] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)

    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def hill_climb_random_transfer(start_h, rng, max_iters=1200, init_step=0.4, n_trials=5):
    """Simple stochastic hill‑climb using pairwise transfers."""
    h = start_h.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                h = cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c = c
                h = cand
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """Refine `h0` using SLSQP on the log‑sum‑exp surrogate with β = `beta`."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        diff2 = CURRENT_TARGET - x.sum()
        if abs(diff2) > 1e-12:
            rng = np.random.default_rng()
            for _ in range(1000):
                if abs(diff2) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff2, 1.0 - x[i]) if diff2 > 0 else max(diff2, -x[i])
                x[i] += add
                diff2 = CURRENT_TARGET - x.sum()
    return x


def worst_shift_corr(h):
    """Return max correlation value and its shift (relative to centre)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (h.size - 1)
    return float(corr[idx]), shift


def pair_transfer_worst_shift_optimize(h0, rng, max_iters=5000, time_limit=30.0):
    """Targeted transfer using the current worst shift."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    eps_factor = 0.5
    while time.time() - start < time_limit and max_iters > 0:
        max_corr, shift = worst_shift_corr(h)
        cur_c = max_corr * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # indices participating in this shift
        if shift >= 0:
            i_start, i_end = 0, n - shift
        else:
            i_start, i_end = -shift, n
        length = i_end - i_start
        if length <= 0:
            break
        # partner slice
        j_start, j_end = i_start + shift, i_end + shift
        part = 1.0 - h[j_start:j_end]          # (1‑h[i+shift]) for each i
        src_rel = int(np.argmax(part))
        src_i = i_start + src_rel
        # try to pick a target outside the range to avoid increasing this shift
        outside = np.concatenate((np.arange(0, i_start), np.arange(i_end, n))) if (i_start > 0 or i_end < n) else np.empty(0, dtype=int)
        tgt_i = None
        if outside.size > 0:
            elig = outside[h[outside] < 1.0]
            if elig.size > 0:
                tgt_i = rng.choice(elig)
        if tgt_i is None:
            # fall back to the index with smallest (1‑h[i+shift]) inside the range
            tgt_rel = int(np.argmin(part))
            tgt_i = i_start + tgt_rel
        max_eps = min(h[src_i], 1.0 - h[tgt_i])
        if max_eps <= 0.0:
            # final fallback: random pair
            src_i = rng.integers(i_start, i_end)
            tgt_i = rng.integers(0, n)
            max_eps = min(h[src_i], 1.0 - h[tgt_i])
        if max_eps <= 0.0:
            break
        eps = eps_factor * max_eps
        if eps <= 0.0:
            eps = max_eps * 0.5
        h[src_i] -= eps
        h[tgt_i] += eps
        h[src_i] = max(0.0, min(1.0, h[src_i]))
        h[tgt_i] = max(0.0, min(1.0, h[tgt_i]))
        eps_factor = max(1e-4, eps_factor * 0.9995)
        max_iters -= 1
    final_c = compute_c5(best_h)
    return best_h, final_c


def swap_optimize(h0, rng, max_iters=5000, time_limit=20.0):
    """Swap two coordinates if it improves the objective."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # -------------------------------------------------
    # Phase 0 – small discretisation
    # -------------------------------------------------
    n_small = 200
    target_small = n_small / 2.0
    candidates = []

    # random seeds
    for _ in range(5):
        candidates.append(random_feasible_vector(n_small, rng, target_small))

    # Fourier‑type seeds
    xs = (np.arange(n_small) + 0.5) * (2.0 / n_small)
    for f in (1, 2, 3):
        for a in (0.2, 0.4):
            h = 0.5 + a * np.sin(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, target_small, rng))
            h = 0.5 + a * np.cos(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, target_small, rng))

    # binary mask
    mask = np.zeros(n_small)
    mask[: n_small // 2] = 1.0
    candidates.append(enforce_sum(mask, target_small, rng))

    # incorporate parent solution if present
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n_init = init.size
        init = enforce_sum(init, n_init / 2.0, rng)
        xs_init = (np.arange(n_init) + 0.5) * (2.0 / n_init)
        init_small = np.interp(xs, xs_init, init)
        init_small = np.clip(init_small, 0.0, 1.0)
        init_small = enforce_sum(init_small, target_small, rng)
        candidates.append(init_small)

    # pick best initial candidate
    best_h = None
    best_c = np.inf
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # -------------------------------------------------
    # Phase 1 – intensive local search on small grid
    # -------------------------------------------------
    phase1_limit = budget_s * 0.45
    while elapsed() < phase1_limit:
        # start point
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n_small, rng, target_small)

        # hill‑climb
        h_hc, c_hc = hill_climb_random_transfer(start_vec, rng,
                                                max_iters=1200,
                                                init_step=0.4,
                                                n_trials=5)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # SLSQP ladder
        for beta in (5, 20, 80, 200, 500, 1000, 2000):
            if elapsed() >= phase1_limit:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
            except Exception:
                continue

        # worst‑shift pair transfer
        remaining = phase1_limit - elapsed()
        if remaining > 5.0:
            h_pt, c_pt = pair_transfer_worst_shift_optimize(
                best_h, rng, max_iters=4000, time_limit=min(remaining, 10.0)
            )
            if c_pt < best_c:
                best_c, best_h = c_pt, h_pt.copy()

    # -------------------------------------------------
    # Phase 2 – upsample and refine on a larger grid
    # -------------------------------------------------
    remaining = budget_s - elapsed()
    if remaining > 30.0:
        n_large = 400
        xs_small = (np.arange(n_small) + 0.5) * (2.0 / n_small)
        xs_large = (np.arange(n_large) + 0.5) * (2.0 / n_large)
        h_large = np.interp(xs_large, xs_small, best_h)
        h_large = np.clip(h_large, 0.0, 1.0)
        h_large = enforce_sum(h_large, n_large / 2.0, rng)
        best_h, best_c = h_large, compute_c5(h_large)

        phase2_end = start_time + budget_s * 0.95
        while time.time() < phase2_end:
            # stochastic start
            if rng.random() < 0.6:
                start_vec = perturb_h(best_h, rng, scale=0.25)
            else:
                start_vec = random_feasible_vector(n_large, rng, n_large / 2.0)

            # hill‑climb (fewer iterations on the larger grid)
            h_hc, c_hc = hill_climb_random_transfer(start_vec, rng,
                                                    max_iters=800,
                                                    init_step=0.35,
                                                    n_trials=4)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

            # SLSQP ladder (coarser)
            for beta in (10, 30, 100, 300, 600, 1200):
                if time.time() >= phase2_end:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=300)
                    c_ref = compute_c5(h_ref)
                    if c_ref < best_c:
                        best_c, best_h = c_ref, h_ref.copy()
                except Exception:
                    continue

            # worst‑shift pair transfer
            time_left = phase2_end - time.time()
            if time_left > 4.0:
                h_pt, c_pt = pair_transfer_worst_shift_optimize(
                    best_h, rng, max_iters=2000, time_limit=min(time_left, 6.0)
                )
                if c_pt < best_c:
                    best_c, best_h = c_pt, h_pt.copy()

    # -------------------------------------------------
    # Phase 3 – final polishing (swap optimisation)
    # -------------------------------------------------
    time_left = budget_s - elapsed()
    if time_left > 2.0:
        h_sw, c_sw = swap_optimize(best_h, rng, max_iters=5000, time_limit=time_left)
        if c_sw < best_c:
            best_c, best_h = c_sw, h_sw.copy()

    # final clean‑up
    n_final = best_h.size
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n_final / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n_final)
