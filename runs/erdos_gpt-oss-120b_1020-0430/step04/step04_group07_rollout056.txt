def simulated_annealing_binary(n, rng, time_budget):
    # initialize with random binary
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n//2, replace=False)
    h[ones]=1.0
    zeros = np.setdiff1d(np.arange(n), ones)
    # compute c5
    c_curr = compute_c5(h)
    best = h.copy(); best_c = c_curr
    start = time.time()
    T0 = 0.1  # maybe adapt based on c_curr
    T_end = 1e-4
    while time.time()-start < time_budget:
        elapsed = time.time()-start
        # linear or exponential cooling
        T = T0 * (T_end/T0) ** (elapsed / time_budget)
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        # swap
        h[i]=0.0; h[j]=1.0
        # compute new c5
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            c_curr = c_new
            # update lists
            ones[ones == i] = j  # replace i with j in ones
            zeros[zeros == j] = i
            if c_new < best_c:
                best_c = c_new
                best = h.copy()
        else:
            # reject, revert swap
            h[i]=1.0; h[j]=0.0
        # maybe occasional random restart if stuck? maybe not needed.
    return best, best_c

def binary_sa(n, rng, time_budget, init_h=None):
    # returns best binary vector and its C5
    start = time.time()
    best_h = None
    best_c = np.inf
    # initialize best with init_h if provided and binary?
    if init_h is not None and (init_h.shape[0]==n):
        # round to binary preserving sum? Create binary from init_h: pick top n/2 values
        idx = np.argpartition(-init_h, n//2)[:n//2]
        h = np.zeros(n, dtype=float)
        h[idx]=1.0
        c = compute_c5(h)
        best_h = h.copy()
        best_c = c
    # If no initial, start random
    # We'll perform multiple restarts; each restart runs until time budget fraction or improvement stalls.
    while time.time() - start < time_budget:
        # start a new random binary vector
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n//2, replace=False)
        h[ones]=1.0
        c_curr = compute_c5(h)
        # store best of this restart
        if c_curr < best_c:
            best_c = c_curr
            best_h = h.copy()
        # maintain index mappings
        mask = np.zeros(n, dtype=bool)
        mask[ones]=True
        ones_arr = np.where(mask)[0]
        zeros_arr = np.where(~mask)[0]
        pos_one = np.full(n, -1, dtype=int)
        pos_zero = np.full(n, -1, dtype=int)
        pos_one[ones_arr] = np.arange(len(ones_arr))
        pos_zero[zeros_arr] = np.arange(len(zeros_arr))
        # SA loop for this restart
        # allocate portion of remaining time to this restart? We'll just run until time left < small threshold
        while time.time() - start < time_budget:
            elapsed = time.time() - start
            # cooling schedule: exponential
            # T0 set based on current best maybe 0.05?
            T0 = 0.05
            T_end = 1e-4
            T = T0 * (T_end/T0) ** (elapsed / time_budget)
            # choose random swap
            i_idx = rng.integers(len(ones_arr))
            j_idx = rng.integers(len(zeros_arr))
            i = ones_arr[i_idx]
            j = zeros_arr[j_idx]
            # Perform swap
            h[i] = 0.0
            h[j] = 1.0
            # compute new cost

# Global variables for the soft‑max SLSQP optimizer
CURRENT_TARGET = None
CURRENT_BETA = None

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ---- stochastic tweaking until the budget closes ----
    while time.time() - start < total_budget * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # ---- final sanitisation ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
