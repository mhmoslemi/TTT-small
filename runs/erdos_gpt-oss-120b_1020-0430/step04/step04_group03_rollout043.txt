    e = np.exp(beta * corr - np.max(beta*corr)) # to avoid overflow
    w = e / np.sum(e)

    # we can compute sum_over_shifts w_k * grad_shift(s_k) ; but maybe we can compute efficiently: compute gradient as w convolved?

    n = h.size

        if s >= 0:

                continue

import time, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(v, target, rng):
    """Clip to [0,1] and adjust to hit exact sum `target`."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    it = 0
    max_iter = 20000
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(v < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - v[i]) * rng.random()
            v[i] += add
            diff -= add
        else:
            idx = np.where(v > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, v[i]) * rng.random()
            v[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, v.size)
        v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v

def random_feasible_vector(n, rng):
    v = rng.random(n)
    return enforce_sum(v, n / 2.0, rng)

def perturb_random(v, rng, scale=0.2):
    n = v.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return v.copy()
    max_eps = min(v[i], 1.0 - v[j])
    if max_eps <= 0.0:
        return v.copy()
    eps = max_eps * scale * rng.random()
    nv = v.copy()
    nv[i] -= eps
    nv[j] += eps
    return nv

def compute_c5(v):
    corr = np.correlate(v, 1 - v, mode='full')
    return np.max(corr) * (2.0 / v.size)

def gradient_shift(v, s):
    n = v.size
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += (1 - v[s:])
        g[s:] -= v[:n - s]
    else:
        s = -s
        g[s:] += (1 - v[:n - s])
        g[:n - s] -= v[s:]
    return g

def worst_shift_info(v):
    corr = np.correlate(v, 1 - v, mode='full')
    idx = np.argmax(corr)
    s = idx - (v.size - 1)
    return s, idx, corr[idx]

def greedy_optimize(v, max_steps=2000):
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(v)
        g = gradient_shift(v, s)
        pos = np.where((g > 1e-12) & (v > 1e-12))[0]
        neg = np.where((g < -1e-12) & (v < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(g[pos])]
        j = neg[np.argmin(g[neg])]
        max_eps = min(v[i], 1.0 - v[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nv = v.copy()
        nv[i] -= max_eps
        nv[j] += max_eps
        if compute_c5(nv) + 1e-12 < compute_c5(v):
            v = nv
            continue
        # try half transfer
        eps = max_eps * 0.5
        nv[i] -= eps
        nv[j] += eps
        if compute_c5(nv) + 1e-12 < compute_c5(v):
            v = nv
            continue
        break
    return v

# ---------- SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(v0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = v0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        v0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- Simulated annealing ----------
def simulated_annealing(v, best_v, best_c, rng, time_limit):
    n = v.size
    cur_v = v.copy()
    cur_c = compute_c5(cur_v)
    start = time.time()
    T0 = max(0.01, best_c * 0.05)  # initial temperature
    while time.time() - start < time_limit:
        frac = (time.time() - start) / time_limit
        T = T0 * (1 - frac)  # linear cooling
        # move scale: larger early, smaller later
        scale = 0.2 if frac < 0.3 else 0.03
        # pick i with room to give, j with room to receive
        for _ in range(10):
            i = rng.integers(0, n)
            if cur_v[i] > 1e-12:
                break
        else:
            continue
        for _ in range(10):
            j = rng.integers(0, n)
            if cur_v[j] < 1.0 - 1e-12:
                break
        else:
            continue
        max_eps = min(cur_v[i], 1.0 - cur_v[j])
        if max_eps <= 1e-15:
            continue
        eps = max_eps * scale * rng.random()
        # apply move
        old_i, old_j = cur_v[i], cur_v[j]
        cur_v[i] = old_i - eps
        cur_v[j] = old_j + eps
        new_c = compute_c5(cur_v)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_v = new_c, cur_v.copy()
        else:
            cur_v[i] = old_i
            cur_v[j] = old_j
    return best_v, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initialise
    if "initial_h_values" in globals():

    h = greedy_optimize(h, max_steps=5000)
    best_h = h.copy()
    best_c = compute_c5(best_h)

    # simulated annealing (≈60% of remaining time)
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.1:
        sa_time = remaining * 0.6
        best_h, best_c = simulated_annealing(best_h, best_h, best_c, rng, sa_time)

    # occasional SLSQP polishing (≈30% of remaining)
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > budget_s * 0.05:
        for beta in (500, 2000, 5000, 10000):
            if time.time() - start > budget_s * 0.98:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=1000)
            except Exception:
                continue
            c_ref = compute_c5(refined)
            if c_ref < best_c:
                best_h, best_c = refined, c_ref

    # final tiny perturb search
    while time.time() - start < budget_s * 0.99:
        cand = perturb_random(best_h, rng, scale=0.01)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # clean up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
