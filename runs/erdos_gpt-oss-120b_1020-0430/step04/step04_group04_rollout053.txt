def compute_conv(h, dx):
    # linear convolution correlate h with (1-h)
    # use FFT for speed
    n = h.size
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv * dx

def perturb_h(h, rng, scale=0.1):

    i = rng.integers(0, n)
    j = rng.integers(0, n)

        return h.copy()

        return h.copy()
    eps = (rng.random() * 2 - 1) * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # iterative correction
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # pick random index
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # after loop, if still diff residual, distribute uniformly among free entries
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        # final small adjustments
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def softmax_grad(x, beta, dx):
    conv = compute_conv(x, dx)
    # compute softmax weights
    lse = logsumexp(beta * conv)
    p = np.exp(beta * conv - lse)  # size L
    # compute term1 and term2 using fftconvolve
    term1 = fftconvolve(p, (1 - x)[::-1], mode='valid')[::-1]
    term2 = fftconvolve(p, x, mode='valid')
    grad = dx * (term1 - term2)
    return grad

def sum_eq_constraint(x):
    """Equality: sum(x) == CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET

def compute_conv(h, dx):
    """Linear correlation h * (1-h) multiplied by dx."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv * dx

    """Exact discrete C5 for a given step function."""
    return np.max(compute_conv(h, dx))

        diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


# ----------------------------------------------------------------------
# Simple perturbation / local moves
def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (sum preserved)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ----------------------------------------------------------------------
# Hill climbing
def hill_climb(h_start, rng, n, max_iters=1500, step=0.5):
    dx = 2.0 / n
    h = h_start.copy()
    best_c = compute_c5(h, dx)
    decay_int = max(1, max_iters // 10)
    cur_step = step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=cur_step)
        c = compute_c5(new_h, dx)
        if c < best_c:
            best_c = c
            h = new_h
        if (it + 1) % decay_int == 0:
            cur_step = max(cur_step * 0.8, 1e-5)
    return h, best_c


# ----------------------------------------------------------------------
# Simulated annealing
def simulated_annealing(h0, rng, n, max_iter=5000,
                        init_temp=1e-2, final_temp=1e-5,
                        init_scale=0.5, final_scale=0.02):
    dx = 2.0 / n
    cur_h = h0.copy()
    best_h = cur_h.copy()
    cur_c = compute_c5(cur_h, dx)
    best_c = cur_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h, dx)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


# ----------------------------------------------------------------------
# Smooth surrogate for max (log-sum-exp) and its gradient
def softmax_obj(x, beta, dx):
    conv = compute_conv(x, dx)
    return logsumexp(beta * conv) / beta


def softmax_grad(x, beta, dx):
    conv = compute_conv(x, dx)
    lse = logsumexp(beta * conv)
    p = np.exp(beta * conv - lse)  # softmax weights, sum to 1
    # term1 = Σ_s p_s * (1 - h[i+s])  (correlation)
    term1 = fftconvolve(p, (1 - x)[::-1], mode='valid')[::-1]
    # term2 = Σ_s p_s * h[i-s]   (convolution)
    term2 = fftconvolve(p, x, mode='valid')
    grad = dx * (term1 - term2)
    return grad


# ----------------------------------------------------------------------
# SLSQP schedule with increasing β
def slsqp_schedule(start_h, n, rng, budget_s, start_time):
    global CURRENT_TARGET
    target = n / 2.0
    dx = 2.0 / n
    h = start_h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, dx)

    beta_vals = [1, 4, 16, 64, 256, 1024, 4096]
    for beta in beta_vals:
        if time.time() - start_time > budget_s * 0.9:
            break
        CURRENT_TARGET = target
        bounds = [(0.0, 1.0)] * n
        cons = {"type": "eq", "fun": sum_eq_constraint}
        res = minimize(
            softmax_obj,
            h,
            args=(beta, dx),
            method="SLSQP",
            jac=softmax_grad,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 150, "ftol": 1e-12, "disp": False},
        )
        x = np.clip(res.x, 0.0, 1.0)
        x = enforce_sum(x, target, rng)
        c_val = compute_c5(x, dx)
        if c_val < best_c:
            best_c, best_h = c_val, x.copy()
        h = x
    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Initial feasible vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h, 2.0 / n)

    # --------------------------------------------------------------
    # Main multi‑restart optimisation loop
    while time.time() - start_time < budget_s * 0.75:
        # start point for this round
        if rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        # SLSQP ladder
        h_slsqp, c_slsqp = slsqp_schedule(start_vec, n, rng, budget_s, start_time)
        if c_slsqp < best_c:
            best_c, best_h = c_slsqp, h_slsqp.copy()

        # Hill‑climb refinement
        h_hc, c_hc = hill_climb(best_h, rng, n, max_iters=1200, step=0.4)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # Simulated‑annealing refinement
        h_sa, c_sa = simulated_annealing(best_h, rng, n,
                                         max_iter=4000,
                                         init_temp=1e-2, final_temp=1e-5,
                                         init_scale=0.4, final_scale=0.02)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        # safety break
        if time.time() - start_time > budget_s * 0.75:
            break

    # --------------------------------------------------------------
    # Optional up‑sampling to finer grid
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.1:
        target_n = min(720, max(n * 2, n + 200))
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)

            # optimise on finer grid
            h_up, c_up = slsqp_schedule(up_h, target_n, rng, budget_s, start_time)

            h_up_hc, c_up_hc = hill_climb(h_up, rng, target_n, max_iters=1500, step=0.3)
            if c_up_hc < c_up:
                c_up, h_up = c_up_hc, h_up_hc

            h_up_sa, c_up_sa = simulated_annealing(h_up, rng, target_n,
                                                   max_iter=5000,
                                                   init_temp=8e-3, final_temp=1e-5,
                                                   init_scale=0.3, final_scale=0.02)
            if c_up_sa < c_up:
                c_up, h_up = c_up_sa, h_up_sa

            if c_up < best_c:
                best_c, best_h, n = c_up, h_up.copy(), target_n

    # --------------------------------------------------------------
    # Final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h, 2.0 / n)

    return best_h.tolist(), float(final_c), int(n)
