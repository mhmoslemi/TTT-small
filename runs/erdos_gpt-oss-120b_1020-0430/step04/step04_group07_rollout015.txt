def simulated_annealing(h, rng, time_end):
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    T = 0.01  # initial temperature
    # schedule: geometric cooling each 1000 iterations
    iter = 0
    while time.time() < time_end:
        # choose i,j
        i, j = rng.integers(0, cur_h.size, size=2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random() * 0.5  # step scale
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / T):
            cur_h, cur_c = nh, nc
        iter += 1
        if iter % 500 == 0:
            T *= 0.95  # cool
    return cur_h, cur_c
```

import time, numpy as np
from scipy.optimize import minimize

# globals for the soft‑max SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with n points."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while keeping 0≤h≤1."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the currently worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1500, step_scale: float = 0.7):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector with entries in [0,1] and sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def make_symmetric(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force pointwise symmetry h(x)=h(2‑x)."""
    hs = (h + h[::-1]) * 0.5
    return enforce_sum(hs, rng)


def simulated_annealing(h: np.ndarray, rng: np.random.Generator,
                        time_end: float, start_temp: float = 0.02,
                        cool_rate: float = 0.995, step_frac: float = 0.5):
    """Simple SA that moves ε between two coordinates."""
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    T = start_temp
    n = cur_h.size
    it = 0
    while time.time() < time_end:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = max_eps * rng.random() * step_frac
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        if nc < cur_c or rng.random() < np.exp(-(nc - cur_c) / T):
            cur_h, cur_c = nh, nc
        it += 1
        if it % 400 == 0:
            T = max(T * cool_rate, 1e-6)
    return cur_h, cur_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h, best_c, best_n = None, np.inf, None

    # use parent solution if available
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    n_choices = [128, 160, 192, 224, 256, 288, 320, 352, 384]

    # ---- multi‑restart search ---------------------------------
    while time.time() - start < 0.78 * budget_s:
        n = int(rng.choice(n_choices))
        sym = rng.random() < 0.27  # sometimes enforce symmetry

        # initialise h
        if best_h is not None and best_n == n:
            h = best_h.copy()
        elif best_h is not None:
            # linear interpolation of best known
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
        else:
            # random start
            if rng.random() < 0.5:
                h = np.zeros(n)
                h[rng.choice(n, n // 2, replace=False)] = 1.0
            else:
                h = random_feasible_vector(n, rng)

        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
        if sym:
            h = make_symmetric(h, rng)

        # quick gradient‑directed hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=1300, step_scale=0.7)

        # soft‑max SLSQP passes
        for beta in (20., 80., 200., 800., 2000.):
            if time.time() - start > 0.90 * budget_s:
                break
            maxit = 500 if beta < 200 else 800
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # another gradient climb
        h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.6)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ---- up‑sampling refinement ---------------------------------
    if best_h is not None and time.time() - start < 0.95 * budget_s:
        for factor in (2, 3):
            n_up = best_n * factor
            if n_up > 384:
                continue
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = enforce_sum(h_up, rng)

            # local refinement at higher resolution
            h_up, c_up = guided_hill_climb(h_up, rng,
                                           max_iters=1800, step_scale=0.6)
            for beta in (20., 200., 2000., 8000.):
                if time.time() - start > budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

            h_up = guided_hill_climb(h_up, rng,
                                      max_iters=1600, step_scale=0.5)[0]
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # ---- final simulated annealing on the best solution -------
    if best_h is not None:
        time_end = start + budget_s * 0.995
        best_h, best_c = simulated_annealing(best_h, rng, time_end)

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
