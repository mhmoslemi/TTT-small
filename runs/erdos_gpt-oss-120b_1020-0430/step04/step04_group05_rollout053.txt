def improve_worst_lag(h, rng, steps=5, eps_scale=0.1):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    if lag < 0: lag = -lag
    # contributions
    if lag >= n:
        return h
    idx = np.arange(n - lag)
    contrib = h[idx] * (1 - h[idx + lag])
    # top indices
    k = max(1, int(len(idx) * 0.02))  # top 2%
    top = idx[np.argpartition(-contrib, k-1)[:k]]
    # for each top idx, transfer epsilon from h[top] to h[top+lag]
    new_h = h.copy()
    for i in top:
        max_eps = min(new_h[i], 1.0 - new_h[i+lag])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * eps_scale
        new_h[i] -= eps
        new_h[i+lag] += eps
    # repair sum if needed? but transfers preserve sum.
    return new_h

def improve_worst_lag_step(h, rng):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    # treat lag positive (absolute)
    if lag < 0:
        lag = -lag
    # if lag ==0: nothing to improve
    if lag == 0 or lag >= n:
        return h

def improve_worst_lag(h, rng, top_frac=0.05, eps_scale=0.3):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    lag = abs(lag)
    if lag == 0 or lag >= n:
        return h
    idx = np.arange(n - lag)
    contrib = h[idx] * (1 - h[idx + lag])
    k = max(1, int(len(idx) * top_frac))
    top_idx = idx[np.argpartition(-contrib, k - 1)[:k]]
    new_h = h.copy()

        max_eps = min(new_h[i], 1.0 - new_h[i + lag])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * eps_scale
        new_h[i] -= eps
        new_h[i + lag] += eps
    return new_h

        return perturb_h(h, rng, scale)
    valid = np.arange(n - lag)
    contrib = h[valid] * (1.0 - h[valid + lag])
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def improve_worst_lag(h, rng, top_frac=0.05, eps_scale=0.3):
    """Deterministic move that reduces the contribution of the worst lag."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    lag = abs(max_idx - (n - 1))
    if lag == 0 or lag >= n:
        return h
    idx = np.arange(n - lag)
    contrib = h[idx] * (1 - h[idx + lag])
    k = max(1, int(len(idx) * top_frac))
    top_idx = idx[np.argpartition(-contrib, k - 1)[:k]]
    new_h = h.copy()
    for i in top_idx:
        max_eps = min(new_h[i], 1.0 - new_h[i + lag])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * eps_scale
        new_h[i] -= eps
        new_h[i + lag] += eps
    return new_h


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_one_beta(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair possible tiny sum error
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x


# ---------- simulated annealing ----------
def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.1, 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        lag = abs(np.argmax(cur_corr) - (n - 1))
        if rng.random() < 0.6:
            new_h = targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        if rng.random() < 0.1:
            new_h = improve_worst_lag(new_h, rng)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c


# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    n = 720
    target_sum = n / 2.0

    # ----- initial vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            old_n = h0.size
            old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            h0 = np.interp(new_x, old_x, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_sum, rng)
    else:
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000]

    # ----- main stochastic exploration -----
    while time.time() - t0 < budget_s * 0.9:
        # random restart: perturb best or fresh random
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)

        # SA phase
        max_sa_iters = 25000
        sa_h, sa_c = simulated_annealing(
            start_vec, rng, max_sa_iters, t0, t0 + budget_s * 0.7
        )
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c

        # SLSQP refinement ladder
        for beta in beta_list:
            if time.time() - t0 >= budget_s * 0.9:
                break
            max_iter = 400 if beta < 5000 else 800
            try:
                ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=max_iter)
            except Exception:
                continue
            ref_c = compute_c5(ref_h)
            if ref_c < best_c:
                best_h, best_c = ref_h, ref_c

        # deterministic worst‑lag cleanup (few passes)
        for _ in range(3):
            if time.time() - t0 >= budget_s * 0.9:
                break
            new_h = improve_worst_lag(best_h, rng)
            new_c = compute_c5(new_h)
            if new_c < best_c:
                best_h, best_c = new_h, new_c

    # ----- final quick hill‑climb -----
    h_cur = best_h.copy()
    cur_c = best_c
    step = 0.3
    while time.time() - t0 < budget_s:
        new_h = perturb_h(h_cur, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_c, h_cur = new_c, new_h
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        step *= 0.9995

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
