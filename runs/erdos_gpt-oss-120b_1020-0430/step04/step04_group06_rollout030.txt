def apply_swap(h, corr, i, j):
    # h array of length n
    n = len(h)
    # Determine delta values
    old_i = h[i]
    old_j = h[j]
    if old_i == old_j:
        return
    # delta_i = new_i - old_i = old_j - old_i
    delta_i = old_j - old_i
    delta_j = old_i - old_j  # negative of delta_i
    # Apply delta_i updates:
    # 1) contributions from h[i] as first factor:
    #    for shift k where i + k in [0, n-1]:
    #       idx = k + n - 1
    #       corr[idx] += delta_i * (1 - h[i + k])
    # Use loop over k = -i .. n-1-i:
    # That is i + k = t; t runs from 0.. n-1, k = t - i
    for t in range(0, n):
        k = t - i
        idx = k + n - 1
        # (1 - h[t]) is based on current h[t]; note this uses old h[t] (including perhaps h[j] before swap)
        # We'll use current h[t] (still old values for i,j). This is fine for delta_i.
        corr[idx] += delta_i * (1.0 - h[t])
    # 2) contributions from h[i] as second factor (i appears as part of (1 - h) for other i'):
    # For i' in [0, n-1] where i' + k = i => k = i - i'
    # So for each i', the shift k = i - i':
    # idx = k + n - 1 = i - i' + n - 1
    # contribution originally: h[i'] * (1 - h[i]) ; change because (1 - h[i]) changes by -delta_i
    # So corr[idx] += -h[i'] * delta_i
    for i_prime in range(0, n):
        k = i - i_prime
        idx = k + n - 1
        corr[idx] += -h[i_prime] * delta_i
    # Now set h[i] = old_j (new)
    h[i] = old_j
    # Similarly apply delta_j updates using updated h (i now swapped)
    # Make sure to recompute using updated h values for (1 - h[t]) etc.
    # delta_j = -delta_i
    # apply contributions of new h[i]? Actually we have already accounted for delta_i fully, and now we need to apply delta_j for index j.
    # Determine old_j (value we moved from i) but now at i; we need similar loops for j.
    # For delta_j now: new_h_j - old_h_j = old_i - old_j = delta_j

    # For loops similar.

    # Finally set h[j] = old_i (new)

```python
def adjust_binary_sum(h, target, rng):
    # h is binary (0/1) array of floats
    cur = int(h.sum())
    while cur != target:
        if cur < target:
            # flip a 0 to 1
            zero_inds = np.where(h == 0)[0]
            i = rng.choice(zero_inds)
            h[i] = 1.0
            cur += 1
        else:
            one_inds = np.where(h == 1)[0]
            i = rng.choice(one_inds)
            h[i] = 0.0
            cur -= 1
    return h
```

```python
def simulated_annealing_binary(h0, rng, time_limit):
    n = h0.size
    target = n // 2
    h = h0.copy()
    current_c = compute_c5_fft(h)
    best_h = h.copy()
    best_c = current_c
    # temperature schedule
    T0 = current_c * 0.5  # initial temperature proportional
    T_min = 1e-7
    decay = 0.99995  # per iteration
    T = T0
    it = 0
    start = time.time()
    while time.time() - start < time_limit:
        # choose random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # propose swap
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        c_new = compute_c5_fft(h_new)
        delta = c_new - current_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            current_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        T = max(T_min, T * decay)
        it += 1
    return best_h, best_c
```

```python
def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple correction by distributing diff uniformly then clipping
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        # final correction if still off
        diff2 = target - x.sum()
        if abs(diff2) > 1e-12:
            # adjust individual entries
            for idx in range(n):
                if abs(diff2) < 1e-12:
                    break
                add = min(diff2, 1.0 - x[idx]) if diff2 > 0 else max(diff2, -x[idx])
                x[idx] += add
                diff2 = target - x.sum()
    return x
```

```python
def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0, init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_exact(h)
    start = time.time()
    factor = init_factor
    while time.time() - start < time_limit and max_iters > 0:
        max_corr_val, shift = max_overlap_and_shift(h)  # need this function
        cur_c = max_corr_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # get indices i and j that realize max product
        # compute product for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        # index of max product
        idx_max = np.argmax(prod)
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        # clip
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
        max_iters -= 1
    final_c = compute_c5_exact(best_h)
    return best_h, final_c
```

```python
def max_overlap_and_shift(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (n - 1)
    return max_val, shift
```

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers for smooth surrogate ----------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ---------- exact C5 computation ----------
def compute_c5_exact(h):
    """C5 using np.correlate (exact for the discrete grid)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


# ---------- fast C5 (FFT) ----------
def compute_c5_fft(h):
    """Fast C5 using FFT (same result as compute_c5_exact)."""
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(H * G, N)[: 2 * n - 1]
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)


# ---------- binary sum adjustment ----------
def adjust_binary_sum(h, target, rng):
    """Flip bits until sum(h) == target (both are integers)."""
    cur = int(h.sum())
    while cur != target:
        if cur < target:
            zero_idx = np.where(h == 0)[0]
            i = rng.choice(zero_idx)
            h[i] = 1.0
            cur += 1
        else:
            one_idx = np.where(h == 1)[0]
            i = rng.choice(one_idx)
            h[i] = 0.0
            cur -= 1
    return h


# ---------- simulated annealing on binary vectors ----------
def simulated_annealing_binary(h0, rng, time_limit):
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5_fft(h)
    best_h, best_c = h.copy(), cur_c

    T = cur_c * 0.5 + 1e-8          # initial temperature
    T_min = 1e-9
    decay = 0.99995                # per‑iteration cooling

    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]

        c_new = compute_c5_fft(h_new)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
        T = max(T_min, T * decay)
    return best_h, best_c


# ---------- SLSQP refinement ----------
def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # small exact‑sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            for k in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - x[k]) if diff > 0 else max(diff, -x[k])
                x[k] += add
                diff = target - x.sum()
    return x


# ---------- pair‑transfer local search ----------
def max_overlap_and_shift(h):
    """Return (max_corr, shift) for current h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (n - 1)


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    h = h0.copy()
    n = h.size
    best_h, best_c = h.copy(), compute_c5_exact(h)
    start = time.time()
    factor = init_factor

    while time.time() - start < time_limit and max_iters > 0:
        max_corr, shift = max_overlap_and_shift(h)
        cur_c = max_corr * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()

        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift

        if i_range.size:
            prod = h[i_range] * (1.0 - h[j_range])
            idx = int(np.argmax(prod))
            i, j = int(i_range[idx]), int(j_range[idx])
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps > 0:
                eps = factor * max_eps
                if eps > 0:
                    h[i] -= eps
                    h[j] += eps
                    h[i] = max(0.0, min(1.0, h[i]))
                    h[j] = max(0.0, min(1.0, h[j]))
                    factor = max(1e-4, factor * decay)
        max_iters -= 1
    return best_h, best_c
