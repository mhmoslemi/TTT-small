def softmax_gradient(h, beta):
    n = h.size
    total_sum = np.sum(h)
    # overlap array
    overlap = np.correlate(h, h, mode='full')[n-1:]
    O = total_sum - overlap
    # compute softmax weights
    # use stable version
    m = np.max(beta * O)
    exps = np.exp(beta * O - m)
    w = exps / np.sum(exps)
    # gradient start as sum w (since derivative of total_sum): each O_k includes total_sum term derivative 1
    grad = np.full(n, np.sum(w))
    # subtract contributions
    for k in range(n):
        wk = w[k]
        if wk == 0:
            continue
        if k == 0:
            grad -= 2 * wk * h
        else:
            # subtract wk * h[i+k] for i< n-k
            grad[:n - k] -= wk * h[k:]
            # subtract wk * h[i-k] for i>=k
            grad[k:] -= wk * h[:n - k]
    return grad

def perturb_h(h, rng, epsilon_scale=0.1):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    # compute max possible epsilon
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = rng.random() * max_eps * epsilon_scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import time, math, numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # random redistribution; cheap and converges quickly
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if abs(add) < 1e-15:
            break
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector with entries in [0,1] and sum=n/2."""
    h = rng.random(n)
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def softmax_weights(O: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max weights for a vector O."""
    m = np.max(beta * O)
    exp_shifted = np.exp(beta * O - m)
    return exp_shifted / exp_shifted.sum()


def softmax_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the soft‑max of the overlap function."""
    n = h.size
    total = h.sum()
    # overlap of h with itself (linear, not circular)
    overlap = np.correlate(h, h, mode="full")[n - 1 :]
    O = total - overlap                     # O_k = Σh – ⟨h,shift^k h⟩
    w = softmax_weights(O, beta)            # soft‑max weights
    grad = np.full(n, w.sum())              # derivative of Σh part
    for k, wk in enumerate(w):
        if wk == 0.0:
            continue
        if k == 0:
            grad -= 2.0 * wk * h
        else:
            grad[: n - k] -= wk * h[k:]
            grad[k:] -= wk * h[: n - k]
    return grad


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.1) -> np.ndarray:
    """Transfer a random ε from one entry to another (keeps the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    # maximal admissible ε
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the currently worst (largest) overlap O_k."""
    n = h.size
    corr_full = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr_full)
    shift = idx - (n - 1)
    k = abs(shift)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # --- initial solution from parent, if any -----------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    # candidate sample sizes (keep them modest for speed)
    n_candidates = [128, 160, 192, 224, 256, 288, 320, 352, 384]

    # ----------------------------------------------------------------
    # main stochastic search (simulated annealing + gradient steps)
    # ----------------------------------------------------------------
    while time.time() - start < 0.80 * budget_s:
        n = int(rng.choice(n_candidates))
        # start from best known (interpolated) or fresh random
        if best_h is not None and best_n == n:
            h = best_h.copy()
        elif best_h is not None:

            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)

            h = random_feasible_vector(n, rng)

        c = compute_c5(h)

        # -------- simulated annealing --------------------------------
        iters = 40000

        for it in range(iters):

            nh = perturb_h(h, rng, scale=0.5)
            nc = compute_c5(nh)
            delta = nc - c
            if delta < 0 or rng.random() < math.exp(-delta / t):
                h, c = nh, nc
                if c < best_c:

            if time.time() - start > 0.80 * budget_s:

        # -------- soft‑max gradient descent ---------------------------

                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
                c = compute_c5(h)
                if c < best_c:

        # -------- targeted hill‑climb on the worst shift -------------
        for _ in range(1800):

            j = int(np.argmin(grad))
            if i != j:
                max_eps = min(h[i], 1.0 - h[j])
                if max_eps > 0:
                    eps = rng.random() * max_eps * 0.4
                    h[i] -= eps
                    h[j] += eps
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ----------------------------------------------------------------
    # final fine‑tuning until the time runs out
    # ----------------------------------------------------------------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i != j:
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps > 0:
                eps = rng.random() * max_eps * 0.03
                best_h[i] -= eps
                best_h[j] += eps
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        # occasional acceptance check
        cur_c = compute_c5(best_h)
        if cur_c < best_c:
            best_c = cur_c

    # final sanity check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
