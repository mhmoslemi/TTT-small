def compute_c5_corr_fast(h):
    n = h.size
    m = 2*n
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * G, n=m)
    conv = conv[:2*n-1]
    c = np.max(conv) * (2.0 / n)
    return c, conv

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    if lag >= n:
        return perturb_h(h, rng, scale)
    valid = np.arange(n - lag)
    # compute contributions to overlap for this lag (h[i]*(1 - h[i+lag]))
    contrib = h[valid] * (1.0 - h[valid + lag])
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    # pick j as a low contribution index or random
    if rng.random() < 0.7:
        j = rng.choice(bot_idx)
    else:
        j = rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    # ensure we transfer from i to j without violating bounds.
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    # epsilon magnitude scaled
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    # ensure within bounds
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

    n = h.size

    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)

            up_h = enforce_sum(up_h, n_up / 2.0, rng)

            up_h, up_c = simulated_annealing_fast(
                up_h, rng, max_iters=15000, start_time=t0, max_time=t0 + remaining
            )

            for beta in beta_list:
                if time.time() - t0 >= budget_s:
                    break
                try:
                    up_h = slsqp_optimize_one_beta(up_h, beta, rng, max_iter=500)
                except Exception:
                    pass
                up_c = compute_c5_fast(up_h)
                if up_c < best_c:
                    best_h, best_c = up_h.copy(), up_c
                    n0 = n_up

    # ----- final local hill‑climb -----
    cur_h, cur_c = best_h.copy(), best_c
    step_scale = 0.3
    for _ in range(8000):
        if time.time() - t0 > budget_s:
            break
        new_h = perturb_h(cur_h, rng, scale=step_scale)
        new_c = compute_c5_fast(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        step_scale *= 0.9997

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_fast(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
