    start = time.time()
    best_h = None; best_c = np.inf; best_n = None

    if 'initial_h_values' in globals():

        # ensure sum constraint

        best_h = h0.copy()
        best_n = h0.size
        best_c = compute_c5(best_h)

    # list of target n values ascending
    n_vals = [128, 176, 224, 256, 320, 384]

    for n in n_vals:
        if time.time() - start > budget_s * 0.95:
            break

        # get starting point for this n
        if best_h is not None and best_n == n:
            h = best_h.copy()
        elif best_h is not None:
            # interpolate from best_n -> n
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)

            h = np.clip(h, 0, 1)
            h = enforce_sum(h, rng)

            h[idx] = 1.0

        T0 = 0.05

            T = T0 * ((T1 / T0) ** (iter / max_iter))
            # decide move type
            if rng.random() < 0.7:

                grad = worst_gradient(h)
                i = int(np.argmax(grad))
                j = int(np.argmin(grad))

                    i = rng.integers(n)

import time, numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()

        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.zeros_like(h)

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))

        if abs(diff) > 1e-12:
            for _ in range(200):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector with entries in [0,1] and sum=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def sa_optimize(h: np.ndarray, rng: np.random.Generator,
                max_iter: int, end_time: float,
                start_temp: float = 0.04, end_temp: float = 1e-5):
    """Simulated‑annealing refinement (transfer epsilon between two indices)."""
    n = h.size
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    for it in range(max_iter):
        if time.time() > end_time:
            break
        # temperature schedule (geometric)
        T = start_temp * ((end_temp / start_temp) ** (it / max_iter))
        # pick move type
        if rng.random() < 0.7:
            grad = worst_gradient(h)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            # fallback random pair
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
            max_eps = min(h[i], 1.0 - h[j])
            if max_eps <= 0:
                continue
        eps = rng.random() * max_eps * (T / start_temp)
        if eps <= 0:
            continue
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        c_new = compute_c5(nh)
        diff = c_new - c_curr
        if diff <= 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            h = nh
            c_curr = c_new
            if c_curr < best_c:
                best_h = h.copy()
                best_c = c_curr
    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function giving a low C₅ bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    end_time = start + budget_s * 0.995  # reserve thin safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    # --- start from parent construction if available --------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0.copy()
        best_n = h0.size
        best_c = compute_c5(best_h)

    # Candidate resolutions (higher gives finer control)
    n_vals = [96, 128, 160, 192, 224, 256, 320, 384]

    for n in n_vals:
        if time.time() > end_time:
            break

        # --- initialise h for this resolution -------------------------
        if best_h is not None and best_n == n:
            h = best_h.copy()
        elif best_h is not None:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, rng)
        else:
            h = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            h[idx] = 1.0

        # --- simulated annealing ---------------------------------------
        # iterations scale roughly with 256/n to keep total work bounded
        max_iter = int(200000 * (256.0 / n))
        h_sa, c_sa = sa_optimize(h, rng, max_iter, end_time)
        if c_sa < best_c:
            best_h, best_c, best_n = h_sa.copy(), c_sa, n

        # optional binary rounding of the SA result (sometimes helpful)
        h_bin = np.zeros_like(h_sa)
        top = np.argsort(h_sa)[-n // 2 :]
        h_bin[top] = 1.0
        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin.copy(), c_bin, n

        # --- fine gradient refinement ----------------------------------
        h_gr = gradient_refine(h_sa, rng, step=0.015, max_iter=12)
        c_gr = compute_c5(h_gr)
        if c_gr < best_c:
            best_h, best_c, best_n = h_gr.copy(), c_gr, n

        # --- guided hill climb -----------------------------------------
        h_hc, c_hc = guided_hill_climb(h_gr, rng, max_iters=2000, step_scale=0.6)
        if c_hc < best_c:
            best_h, best_c, best_n = h_hc.copy(), c_hc, n

    # --- final stochastic tweaks until time runs out --------------------
    while time.time() < end_time and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # --- final sanity check --------------------------------------------
    if best_h is None:
        # fallback deterministic construction
        n = 256
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_h = enforce_sum(best_h, rng)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
