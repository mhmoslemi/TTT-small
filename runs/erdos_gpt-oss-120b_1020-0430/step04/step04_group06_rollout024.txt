    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        # final uniform correction (rare)
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Binary feasible vector with exactly n/2 ones (n assumed even)."""
    target = n // 2
    idx = rng.permutation(n)[:target]
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple stochastic hill‑climb."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_int = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
                improved = True
        if (it + 1) % decay_int == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


# ---------- pair‑transfer local search ----------
def max_overlap_and_shift(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)


def pair_transfer_optimize(h0, rng, max_iters=8000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Move mass from i to i+shift to reduce the current worst overlap."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    factor = init_factor

    while time.time() - start < time_limit and max_iters > 0:
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        # indices involved in this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            max_iters -= 1
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i, j = int(i_range[idx_max]), int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            max_iters -= 1
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            max_iters -= 1
            continue
        h[i] -= eps
        h[j] += eps
        # enforce bounds
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-6, factor * decay)
        max_iters -= 1

    # final clean‑up of sum (numeric drift)
    best_h = enforce_sum(best_h, target, rng)
    return best_h, best_c


# ---------- smooth‑max gradient refinement ----------
def gradient_refine(h0, rng, beta, time_limit, alpha0=0.2):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    alpha = alpha0
    offset = n - 1

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        bc = beta * corr
        m = np.max(bc)
        w = np.exp(bc - m)          # scaled weights, max=1
        Z = w.sum()
        # convolution gradients
        conv1 = np.convolve(w[::-1], 1.0 - h, mode="full")
        conv2 = np.convolve(w, h, mode="full")
        grad = (conv1[offset:offset + n] - conv2[offset:offset + n]) / Z
        # step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce exact sum via uniform correction
        diff = h_new.sum() - target
        if abs(diff) > 1e-12:
            h_new -= diff / n
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, target, rng)
        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
        h = h_new
        alpha *= 0.9995  # gentle decay
    return best_h, best_c


# ---------- SLSQP surrogate (kept for occasional use) ----------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    # deterministic seeds
    for f in (1, 2, 3, 4, 5):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))

    # ramps
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n / 2.0, rng))

    # binary blocks
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    mask = np.zeros(n)
    start = n // 4
    mask[start : start + n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    mask = np.arange(n) % 2
    candidates.append(enforce_sum(mask, n / 2.0, rng))

    # more random binary seeds
    for _ in range(8):
        candidates.append(random_feasible_vector(n, rng))

    # pick best
    best_h, best_c = None, float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    global_best_h, global_best_c = best_h, best_c

    # ---- main stochastic search ----
    main_time_limit = budget_s * 0.78
    beta_schedule = [200, 800, 2000, 5000, 12000]
    while time.time() - start_time < main_time_limit:
        # pick start
        if rng.random() < 0.6:
            start_vec = perturb_h(global_best_h, rng, scale=0.35)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill‑climb
        iters = 2000
        hc_h, hc_c = hill_climb_multi(start_vec, rng, max_iters=iters,
                                      init_step=0.45, n_trials=6)
        if hc_c < global_best_c:
            global_best_h, global_best_c = hc_h.copy(), hc_c

        # occasional SLSQP refinement
        if rng.random() < 0.35:
            for beta in beta_schedule:
                if time.time() - start_time >= main_time_limit:
                    break
                try:
                    h_ref = slsqp_optimize(global_best_h, beta, max_iter=400)
                    c_ref = compute_c5(h_ref)
                    if c_ref < global_best_c:
                        global_best_h, global_best_c = h_ref.copy(), c_ref
                except Exception:
                    continue

        # occasional gradient refinement (high‑beta)
        if rng.random() < 0.2:
            grad_time = min(6.0, main_time_limit - (time.time() - start_time))
            if grad_time > 1.0:
                h_ref, c_ref = gradient_refine(global_best_h, rng,
                                               beta=12000, time_limit=grad_time,
                                               alpha0=0.15)
                if c_ref < global_best_c:
                    global_best_h, global_best_c = h_ref.copy(), c_ref

    # ---- intensive pair‑transfer on the remaining budget ----
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20.0:
        # split remaining into a few focused attempts
        attempts = max(1, int(remaining // 30))
        per = remaining / (attempts + 1)
        best_h, best_c = global_best_h, global_best_c
        for _ in range(attempts):
            h_start = perturb_h(best_h, rng, scale=0.22) if rng.random() < 0.5 else best_h.copy()
            h_opt, c_opt = pair_transfer_optimize(h_start, rng,
                                                  max_iters=12000,
                                                  time_limit=per,
                                                  init_factor=0.6,
                                                  decay=0.999)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
                global_best_h, global_best_c = best_h.copy(), best_c

        # final polish with gradient descent
        final_time = remaining * 0.7
        h_opt, c_opt = gradient_refine(global_best_h, rng,
                                       beta=18000, time_limit=final_time,
                                       alpha0=0.12)
        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt

    # ---- clean up and return ----
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
