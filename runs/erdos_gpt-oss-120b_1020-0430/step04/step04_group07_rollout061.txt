def binary_hill_climb(h, rng, max_iters, temp_start, temp_end):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    current_h = h.copy()
    current_c = best_c

    # Precompute correlation once
    corr = correlation_fft(current_h)
    # Max correlation index
    for it in range(max_iters):
        temperature = temp_start * ((temp_end / temp_start) ** (it / max_iters))
        # Determine worst shift
        idx = np.argmax(corr)
        k = idx - (n - 1)  # shift
        # Determine mismatched indices
        if k >= 0:
            i_range = np.arange(n - k)
            i1 = i_range[(current_h[i_range] == 1) & (current_h[i_range + k] == 0)]
            i0 = i_range[(current_h[i_range] == 0) & (current_h[i_range + k] == 1)]
        else:
            kk = -k
            i_range = np.arange(kk, n)
            i1 = i_range[(current_h[i_range] == 1) & (current_h[i_range - kk] == 0)]
            i0 = i_range[(current_h[i_range] == 0) & (current_h[i_range - kk] == 1)]

        if i1.size > 0 and i0.size > 0:
            i = int(rng.choice(i1))
            j = int(rng.choice(i0))
            # Swap
            new_h = current_h.copy()
            new_h[i] = 0
            new_h[j] = 1
        else:
            # Random swap of any 1 and 0
            ones = np.flatnonzero(current_h == 1)
            zeros = np.flatnonzero(current_h == 0)
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            new_h = current_h.copy()
            new_h[i] = 0
            new_h[j] = 1

        # Compute correlation for new_h
        new_corr = correlation_fft(new_h)
        new_c = new_corr.max() * (2.0 / n)  # more efficient: new_corr.max() * dx
        # acceptance
        if new_c < current_c or rng.random() < np.exp((current_c - new_c) / temperature):
            current_h = new_h
            corr = new_corr
            current_c = new_c
            if current_c < best_c:
                best_c = current_c
                best_h = current_h.copy()
        # optionally early exit if current_c <= target threshold
    return best_h, best_c

Implementation of correlation_fft:

def correlation_fft(h):
    n = h.shape[0]
    # Cast to float
    hf = h.astype(np.float64)
    # Reverse h
    rev = hf[::-1]
    L = 1 << ((2*n - 1).bit_length())
    # Real FFT
    Hr = np.fft.rfft(rev, n=L)
    G = np.fft.rfft(1.0 - hf, n=L)
    corr_full = np.fft.irfft(Hr * G, n=L)
    # Truncate to length 2*n-1
    return corr_full[:2*n-1]

def binary_hill_climb_timed(h, rng, time_limit):
    start = time.time()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    curr_h = h.copy()
    corr = correlation_fft(curr_h)
    curr_c = corr.max() * (2.0 / n)

    # temperature schedule: start at temp_start = best_c * 0.05? Let's set start_temp maybe 0.01.
    # We'll implement simple simulated annealing.
    # But could just accept only improving moves. Might still find improvements, but could stuck.
    # We'll include occasional random swaps to escape.
    iter = 0
    # define temperatures decreasing
    max_iters = 1000000  # large
    while time.time() - start < time_limit and iter < max_iters:
        iter += 1
        # schedule temperature based on iter and max_iters
        temperature = 0.01 * (1 - iter / max_iters) + 0.0001
        # find worst shift
        idx = int(np.argmax(corr))
        k = idx - (n - 1)
        # find mismatched indices
        if k >= 0:
            i_range = np.arange(n - k)
            mask1 = (curr_h[i_range] == 1) & (curr_h[i_range + k] == 0)
            mask0 = (curr_h[i_range] == 0) & (curr_h[i_range + k] == 1)
        else:
            kk = -k
            i_range = np.arange(kk, n)
            mask1 = (curr_h[i_range] == 1) & (curr_h[i_range - kk] == 0)
            mask0 = (curr_h[i_range] == 0) & (curr_h[i_range - kk] == 1)

        i1 = i_range[mask1]
        i0 = i_range[mask0]
        if i1.size > 0 and i0.size > 0:
            i = int(rng.choice(i1))
            j = int(rng.choice(i0))
        else:
            # random swap
            ones = np.flatnonzero(curr_h == 1)
            zeros = np.flatnonzero(curr_h == 0)
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))

        new_h = curr_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0

        new_corr = correlation_fft(new_h)
        new_c = new_corr.max() * (2.0 / n)

        # acceptance
        if new_c < curr_c or rng.random() < np.exp((curr_c - new_c) / temperature):
            curr_h = new_h
            corr = new_corr
            curr_c = new_c
            if curr_c < best_c:
                best_c = curr_c
                best_h = curr_h.copy()
        # optional break if best_c very low (< 0.3805)
        if best_c < 0.3805:
            break
    return best_h, best_c

def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    target = h.size / 2.0
    # first clip to [0,1] (should already be)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # We will adjust iteratively
    while abs(diff) > 1e-12:
        # choose index i where adjustment possible
        if diff > 0:
            # need to increase
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            i = int(rng.choice(idxs))
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff = target - h.sum()
        else:
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            i = int(rng.choice(idxs))
            sub = min(-diff, h[i])
            h[i] -= sub
            diff = target - h.sum()
    return h

def correlation_fft(h):
    n = h.shape[0]
    hf = h.astype(np.float64)
    rev = hf[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    # Real FFT of reversed h
    Hr = np.fft.rfft(rev, n=L)
    G = np.fft.rfft(1.0 - hf, n=L)
    corr_full = np.fft.irfft(Hr * G, n=L)
    return corr_full[:2 * n - 1]

import time

CURRENT_BETA = None


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 10000:
        if diff > 0:
            idxs = np.where(h < 1.0)[0]
            if idxs.size == 0:
                break
            i = int(rng.choice(idxs))
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            idxs = np.where(h > 0.0)[0]
            if idxs.size == 0:
                break
            i = int(rng.choice(idxs))
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
        attempts += 1
    return h


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * (2.0 / n)


def correlation_fft(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1‑h) via FFT (full length)."""
    n = h.shape[0]
    hf = h.astype(np.float64)
    rev = hf[::-1]
    L = 1 << ((2 * n - 1).bit_length())
    Hr = np.fft.rfft(rev, n=L)
    G = np.fft.rfft(1.0 - hf, n=L)
    full = np.fft.irfft(Hr * G, n=L)
    return full[: 2 * n - 1]


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin.copy(), c_bin, n0

    # --- parameters -----------------------------------------------------------
    n_candidates = [96, 120, 144, 168, 192, 224, 256, 320, 384, 512]

    # --- binary hill‑climbing search -----------------------------------------
    while time.time() - start < 0.85 * budget:
        remaining = budget - (time.time() - start)
        per_round = min(0.03 * budget, remaining * 0.5)

        n = int(rng.choice(n_candidates))
        # start from the current best when sizes match, else random
        if best_h is not None and best_n == n and rng.random() < 0.6:
            h_start = best_h.copy()
        else:
            h_start = random_binary_vector(n, rng)

        h_opt, c_opt = binary_hill_climb_timed(h_start, rng, per_round)

        if c_opt < best_c:
            best_h, best_c, best_n = h_opt.copy(), c_opt, n

    # --- fractional refinement with SLSQP (soft‑max) -------------------------
    if best_h is not None:
        rem = budget - (time.time() - start)
        if rem > 0.07 * budget:
            h_float = best_h.astype(np.float64)
            for beta in (20.0, 80.0, 200.0, 800.0, 3000.0, 8000.0):
                if time.time() - start > 0.99 * budget:
                    break
                maxit = 500 if beta < 200 else 800
                h_float = slsqp_optimize(h_float, beta, rng, max_iter=maxit)
                c_now = compute_c5(h_float)
                if c_now < best_c:
                    best_h, best_c = h_float.copy(), c_now
                    best_n = best_h.size

    # --- final sanity check ---------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
