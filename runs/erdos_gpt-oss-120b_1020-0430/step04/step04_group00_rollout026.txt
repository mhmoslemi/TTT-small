import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper globals ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- core utilities ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Redistribute to satisfy Σh = n/2 exactly (in‑place)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stabilization
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    """Projected gradient descent against the current worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]        # + (1 - h[i+k])
            g[k:] -= h[:upper]               # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_hill_climb(h_bin, rng, time_deadline):
    """Simulated‑annealing hill climb on binary vectors (swap 0↔1)."""
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur_h == 1))
    zeros = list(np.flatnonzero(cur_h == 0))
    init_T = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < time_deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0
        cur_h[j] = 1
        c = compute_c5(cur_h)
        T = init_T * (0.9995 ** it) + 1e-12
        accept = False
        if c < cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(c - cur_c) / T):
                accept = True
        if accept:
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = cur_h.copy()
                last_improve = it
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best_h, best_c

        max_samples = 500
        if total_pairs <= max_samples:
            pairs = [(i, j) for i in S1 for j in S0]
        else:
            pairs = [(rng.choice(S1), rng.choice(S0)) for _ in range(max_samples)]
        best_pair = None
        best_c = max_c
        for i, j in pairs:
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
    return h

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else None
    if parent is not None:
        candidate_ns = [parent.size, max(240, parent.size // 2), 180]
    else:
        candidate_ns = [360, 240, 180]
    candidate_ns = [n for n in candidate_ns if n % 2 == 0]

    for n in candidate_ns:
        if time.time() > deadline:
            break

        start_vecs = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp)  # fractional
            # binary version
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        target = n / 2.0
        rand_frac *= target / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)
        start_vecs.append(rand_frac)

        for base_h in start_vecs:
            if time.time() > deadline:
                break
            h = base_h.copy()
            adjust_sum_exact(h, rng)
            h = np.clip(h, 0.0, 1.0)

            # ----- binary SA (if applicable) -----
            if np.all((h == 0) | (h == 1)):
                sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.3)
                h_bin, c_bin = binary_sa_hill_climb(h.copy(), rng, sub_dead)
                if c_bin < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c_bin, n
                h = h_bin.astype(float)

            # ----- gradient refinement -----
            h = gradient_refine(h, rng, step=0.02, max_iter=20)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # ----- SLSQP cascade -----
            betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
                     2000.0, 5000.0, 10000.0, 20000.0)
            for beta in betas:
                if time.time() > deadline:
                    break
                max_iter = 300 if beta < 200 else 600
                if beta >= 2000:
                    max_iter = 800
                try:
                    h = slsqp_optimize(h, beta, rng, max_iter=max_iter)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # ----- final gradient refinement -----
            h = gradient_refine(h, rng, step=0.02, max_iter=30)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # ----- greedy local improvement for binary -----
            if np.all((np.abs(h - 0) < 1e-12) | (np.abs(h - 1) < 1e-12)):
                h_bin = np.rint(h).astype(float)
                adjust_sum_exact(h_bin, rng)
                h_bin = greedy_improve(h_bin, rng, max_iter=50, deadline=deadline)
                c = compute_c5(h_bin)
                if c < best_c:
                    best_h, best_c, best_n = h_bin.copy(), c, n
                h = h_bin

    # ----- optional up‑sampling & extra refinement -----
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        h_up = np.repeat(best_h, 2)  # duplicate each entry
        adjust_sum_exact(h_up, rng)

        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=30)
        for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=30)

        if np.all((np.abs(h_up - 0) < 1e-12) | (np.abs(h_up - 1) < 1e-12)):
            h_up = greedy_improve(h_up, rng, max_iter=50, deadline=deadline)

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ----- fallback -----
    if best_h is None:
        n = 180
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # ----- final safety handling -----
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
