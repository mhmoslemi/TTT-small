def transfer_hill_climb(h, rng, max_iters=2000):
    n = h.size
    best_c = compute_c5(h)
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] - grad[j] < 1e-12:
            break
        # compute max epsilon
        eps = min(h[i], 1.0 - h[j])
        if eps <= 0:
            # maybe randomly select other indices
            # fallback to guided_perturb
            nh = guided_perturb(h, grad, rng, scale=0.5)
            c = compute_c5(nh)
            if c < best_c:
                h, best_c = nh, c
            continue
        # try full transfer
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5(nh)
        if c < best_c - 1e-12:
            h, best_c = nh, c
        else:
            # try half epsilon
            eps2 = eps * 0.5
            nh2 = h.copy()
            nh2[i] -= eps2
            nh2[j] += eps2
            c2 = compute_c5(nh2)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    # load initial if present
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h = h0
        best_c = compute_c5(best_h)
        best_n = h0.size
    else:
        best_n = None

    # define list of n sizes to try
    n_candidates = [256, 320, 384, 448, 512]

    # main loop over sizes
    for n in n_candidates:
        # stop if remaining time small
        if time.time() - start > budget_s * 0.94:
            break
        # generate start h
        if best_h is not None:
            if best_n == n:
                h = best_h.copy()
            else:
                # interpolate
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = enforce_sum(h, rng)
        else:
            # random binary or fractional
            if rng.random() < 0.5:
                h = np.zeros(n)
                idx = rng.choice(n, n // 2, replace=False)
                h[idx] = 1.0
            else:
                h = random_feasible_vector(n, rng)

        # hill climb deterministic
        max_iters = 2500 if n <= 384 else 3000
        h = transfer_hill_climb(h, rng, max_iters=max_iters)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # SLSQP refinement with beta schedule
        for beta in (20., 80., 200., 800., 3000., 8000.):
            if time.time() - start > budget_s * 0.96:
                break
            maxit = 600 if beta >= 200 else 400
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # small gradient refinement
        h = gradient_refine(h, rng, step=0.01, max_iter=8)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # final stochastic tweaks
    while time.time() - start < budget_s * 0.998 and best_h is not None:
        # occasional bigger random transfer
        if rng.random() < 0.1:
            # random pair
            i = rng.integers(0, best_h.size)
            j = rng.integers(0, best_h.size)
            if i != j:
                eps = min(best_h[i], 1 - best_h[j])
                if eps > 0:
                    eps = eps * rng.random() * 0.5
                    nh = best_h.copy()
                    nh[i] -= eps
                    nh[j] += eps
                    nc = compute_c5(nh)
                    if nc < best_c:
                        best_h, best_c = nh, nc
        else:
            grad = worst_gradient(best_h)
            nh = guided_perturb(best_h, grad, rng, scale=0.02)  # we have guided_perturb defined earlier or we can implement similar small transfer
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh, nc

    # final clip and enforce sum
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

import time, numpy as np
from scipy.optimize import minimize

# globals for the soft‑max SLSQP objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h to satisfy Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n * 0.5
    diff = target - h.sum()
    # simple random redistribution
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add != 0.0:
            h[i] += add
            diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1] with sum=n/2."""
    h = rng.random(n)
    h *= (n * 0.5) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of the max overlap."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n * 0.5
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj, h0, method="SLSQP", bounds=bounds,
        constraints=cons, options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum repair (fast)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Small ε transfer from max‑grad to min‑grad index."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def transfer_hill_climb(h: np.ndarray, rng: np.random.Generator,
                        max_iters: int = 2000) -> np.ndarray:
    """Deterministic transfer hill‑climb using the worst‑shift gradient."""
    best_c = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        grad = worst_gradient(h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] - grad[j] < 1e-12:
            break
        eps = min(h[i], 1.0 - h[j])
        if eps <= 0.0:
            # fallback to a tiny random move
            nh = guided_perturb(h, grad, rng, scale=0.2)
            c = compute_c5(nh)
            if c < best_c:
                h, best_c = nh, c
            continue
        # try full transfer
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        c = compute_c5(nh)
        if c < best_c - 1e-12:
            h, best_c = nh, c
        else:
            # try half transfer
            eps2 = eps * 0.5
            nh2 = h.copy()
            nh2[i] -= eps2
            nh2[j] += eps2
            c2 = compute_c5(nh2)
            if c2 < best_c:
                h, best_c = nh2, c2
        # occasional random shake to escape plateaus
        if (it + 1) % 250 == 0 and rng.random() < 0.2:
            a, b = rng.integers(0, n, size=2)
            if a != b:
                eps = min(h[a], 1.0 - h[b])
                if eps > 0:
                    eps *= rng.random()
                    nh = h.copy()
                    nh[a] -= eps
                    nh[b] += eps
                    c = compute_c5(nh)
                    if c < best_c:
                        h, best_c = nh, c
    return h


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 8) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h
