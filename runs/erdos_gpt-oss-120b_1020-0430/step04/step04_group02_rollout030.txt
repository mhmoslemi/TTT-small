def grad_step(h, step):
    # compute worst shift k_bad
    idx = np.argmax(np.correlate(h, 1 - h, mode="full"))
    k = idx - (len(h)-1)
    if k <= 0 or k >= len(h):
        # if zero shift or out-of-range, use random
        grad = np.zeros_like(h)
    else:
        # gradient of C5 w.r.t h: -dx * ((P_k + P_k^T) @ h)
        # combine forward shift and backward shift contributions
        grad = np.zeros_like(h)
        # forward part: for i from 0 to n-1-k, grad[i] -= dx * h[i+k]
        grad[:len(h)-k] -= h[k:]
        # backward part: for i from k to n-1, grad[i] -= dx * h[i-k]
        grad[k:] -= h[:len(h)-k]
        # index shift though constant factor dx will be absorbed via step size
    # take step opposite gradient to reduce C5 (since C5 gradient is positive direction of increase)
    new_h = h - step * grad
    # project onto box sum
    new_h = project_box_sum(new_h, len(h)/2.0)
    return new_h

def hill_climb(h_start, rng, max_iters=3000, step0=0.5, p_target=0.5, decay=0.95, decay_interval=200):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = step0
    n = h.shape[0]
    no_improve_counter = 0

    for it in range(max_iters):
        # compute worst shift
        _, idx = compute_c5_and_idx(h)
        k_bad = idx - (n - 1)
        # choose move
        r = rng.random()
        if r < p_target*0.5:
            cand = targeted_move(h, rng, k_bad, scale=step)
        elif r < p_target:
            cand = gradient_move(h, k_bad, step)
        else:
            cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c - 1e-12:
            best_c = c
            h = cand
            no_improve_counter = 0
        else:
            no_improve_counter += 1
            # occasional random restart if stuck?
            if no_improve_counter > 200 and rng.random() < 0.1:
                # perturb heavily
                cand = perturb_h(h, rng, scale=step*2)
                c = compute_c5(cand)
                if c < best_c:
                    best_c = c
                    h = cand
                    no_improve_counter = 0

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def targeted_move(h, rng, k, scale=0.2):

def hill_climb(start_h, rng, max_iters=2500, step0=0.4, p_target=0.5, decay=0.7, decay_interval=200):
    h = start_h.copy()
    best_c = compute_c5(h)
    step = step0
    n = h.size
    no_improve = 0
    for it in range(max_iters):
        # worst shift
        _, idx = compute_c5_and_idx(h)
        k_bad = idx - (n - 1)
        r = rng.random()
        if r < p_target * 0.5:
            cand = targeted_move(h, rng, k_bad, scale=step)
        elif r < p_target:
            cand = gradient_move(h, rng, k_bad, step)
        else:
            cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c - 1e-12:
            best_c, h = c, cand
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300 and rng.random() < 0.2:
                cand = perturb_h(h, rng, scale=step * 2.0)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, h = c, cand
                    no_improve = 0
        if (it + 1) % decay_interval == 0:
            step = max(step * decay, 1e-6)
    return h, best_c

Define global variables:

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    from scipy.optimize import minimize
    res = minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def gradient_move(h, rng, k, step):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale=step)
    g = np.zeros_like(h)
    g[: n - k] = h[k:]
    g[k:] += h[: n - k]
    new_h = h + step * g
    return project_box_sum(new_h, n / 2.0)


def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


# ---------- hill climbing ----------
def hill_climb(start_h, rng, max_iters=2500, step0=0.4, p_target=0.5,
               decay=0.7, decay_interval=200):
    h = start_h.copy()
    best_c = compute_c5(h)
    step = step0
    n = h.size
    no_imp = 0
    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_bad = idx - (n - 1)
        r = rng.random()
        if r < p_target * 0.5:
            cand = targeted_move(h, rng, k_bad, scale=step)
        elif r < p_target:
            cand = gradient_move(h, rng, k_bad, step)
        else:
            cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c - 1e-12:
            best_c, h = c, cand
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 300 and rng.random() < 0.2:
                cand = perturb_h(h, rng, scale=step * 2.0)
                c = compute_c5(cand)
                if c < best_c:
                    best_c, h = c, cand
                    no_imp = 0
        if (it + 1) % decay_interval == 0:
            step = max(step * decay, 1e-6)
    return h, best_c


# ---------- smooth SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)
