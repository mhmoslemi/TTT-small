def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size

            break

        h[i] += add
        diff = target - h.sum()
    # if still off, distribute uniformly
    if abs(diff) > 1e-12:
        per = diff / n
        h += per
        h = np.clip(h, 0.0, 1.0)

        # final adjust

        c = compute_c5(new_h)

            h = new_h

        h_new = slsqp_optimize(h_best, beta, max_iter=max_iter)
        c_new = compute_c5(h_new)
        if c_new < c_best:
            h_best, c_best = h_new, c_new

    # ------------------------------------------------------------------
    # Upsampling and final refinement (if time permits)
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.9:
        n_fine = min(720, n * 2)
        h_fine = upsample_h(h_best, n_fine, rng)

        h_fine, c_fine = hill_climb(h_fine, rng, max_iters=2500, step=0.4)
        if c_fine < c_best:
            h_best, c_best, n = h_fine, c_fine, n_fine

        h_fine, c_fine = simulated_annealing(
            h_fine,
            rng,
            max_iter=6000,
            init_temp=5e-3,
            final_temp=1e-6,
            init_scale=0.4,
            final_scale=0.02,
        )
        if c_fine < c_best:
            h_best, c_best, n = h_fine, c_fine, n_fine

        # a second, shorter SLSQP pass on the fine grid
        for beta in [200, 800, 2000, 8000, 20000]:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.99:
                break
            remaining = budget_s - elapsed
            max_iter = int(min(1500, max(300, remaining * 0.04)))
            h_new = slsqp_optimize(h_best, beta, max_iter=max_iter)
            c_new = compute_c5(h_new)
            if c_new < c_best:
                h_best, c_best = h_new, c_new
                n = n_fine

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = enforce_sum(h_best, n / 2.0, rng)
    final_c = compute_c5(h_best)

    return h_best.tolist(), float(final_c), int(n)
