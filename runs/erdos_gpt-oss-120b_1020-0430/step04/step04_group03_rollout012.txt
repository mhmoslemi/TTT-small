def project_box_sum(v, target, max_iter=100, tol=1e-9):
    # clip v to [0,1], compute initial sum; if sum == target, return clipped
    # else find lambda such that sum_i clip(v_i - lambda, 0, 1) = target.
    # Since sum of clipped is decreasing function of lambda, find lambda using binary search.
    # Determine lambda_low and lambda_high bounds:
    # lambda_low = min(v) - 1 (ensuring all clipped to <=1)
    # lambda_high = max(v) (ensuring all clipped to >=0). Actually we can set low = -1, high = 1.
    low = -1.0
    high = 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < tol:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    # after loop, return with mid
    return np.clip(v - (low+high)/2.0, 0.0, 1.0)

import time
import numpy as np

# ---------- core utilities ----------
def compute_c5(h):
    """Discrete C5 for vector h (integral constraint already satisfied)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def gradient_shift(h, s):
    """Gradient of the overlap for a single shift s (s can be negative)."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def project_onto_box_sum(v, target, iters=30, tol=1e-9):
    """
    Project v onto {x | sum(x)=target , 0<=x<=1 } by binary searching a shift λ
    such that x = clip(v-λ,0,1) has the required sum.
    """
    lo, hi = -1.0, 1.0  # bounds guarantee sum ranges from n to 0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def greedy_optimize(h0, max_steps=2000):
    """Deterministic improvement: move mass from the most harmful to the most helpful positions."""
    h = h0.copy()
    n = h.size
    for _ in range(max_steps):
        # worst shift and its gradient
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        s = idx - (n - 1)
        grad = gradient_shift(h, s)

        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break

        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break

        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue

        # try half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

        B = np.fft.rfft(1.0 - h, m)
