def binary_pattern(n, p):
    """Return binary vector of length n with exactly n/2 ones using step p."""
    pattern = ((np.arange(n) * p) % n) < (n // 2)
    # Convert to float, can be 0 or 1.
    return pattern.astype(float)

def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) using log-sum-exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
