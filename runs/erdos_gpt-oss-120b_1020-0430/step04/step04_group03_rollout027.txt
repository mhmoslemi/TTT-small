    # If sum already target:
    s = v_clipped.sum()
    if abs(s - target) < 1e-12:

    # Binary search lambda
    lo = -1.0
    hi = 1.0
    for _ in range(50):

        s = proj.sum()
        if s > target:

        else:
            hi = lam
    return proj

def softmax_obj_and_grad(x, beta):
    n = x.size
    a = 1.0 - x
    corr = np.correlate(x, a, mode='full')
    c_max = corr.max()
    # compute weights
    exp_term = np.exp(beta * (corr - c_max))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp

    # f = (c_max + np.log(sum_exp)) / beta  # this is stable expression
    f = (c_max + np.log(sum_exp)) / beta

    # gradient: sum_i w_i * grad_c_i
    grad = np.zeros_like(x)
    # s range
    for idx, weight in enumerate(w):
        s = idx - (n - 1)
        if weight == 0:
            continue
        if s >= 0:
            # forward term
            grad[:n-s] += weight * a[s:]
            # backward term
            grad[s:] -= weight * x[:n-s]
        else:
            s_abs = -s
            grad[s_abs:] += weight * a[:n-s_abs]
            grad[:n-s_abs] -= weight * x[s_abs:]
    return f, grad
```

    n = h0.size
    target_sum = n / 2.0

    def fun(x):
        f, _ = softmax_obj_and_grad(x, beta)

        _, g = softmax_obj_and_grad(x, beta)
        return g

    bounds = [(0.0,1.0)]*n
    cons = {'type':'eq','fun': lambda x: np.sum(x)-target_sum}
    res = minimize(fun, h0, method='SLSQP', jac=jac, bounds=bounds, constraints=cons,

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# projection onto the simplex with box constraints [0,1]
def _project_box_simplex(v, target):
    """Project v onto {0<=z<=1, sum(z)=target} using a simple bisection on the shift."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    lo, hi = -1.0, 1.0  # enough for [0,1] vectors and target=n/2
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - ((lo + hi) * 0.5), 0.0, 1.0)

def random_feasible_vector(n, rng):
    """Uniform random point on the box‑simplex."""
    v = rng.random(n)
    return _project_box_simplex(v, n / 2.0)

# ------------------------------------------------------------
# objective and gradient (smoothed max via log‑sum‑exp)
def _softmax_obj_and_grad(x, beta):
    """Return (f, grad) where f = logsumexp(beta*c)/beta, c = corr(x,1-x)."""
    n = x.size
    a = 1.0 - x
    corr = np.correlate(x, a, mode='full')
    c_max = corr.max()
    # stable weights
    exp_term = np.exp(beta * (corr - c_max))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp
    f = (c_max + np.log(sum_exp)) / beta

    # gradient = sum_i w_i * grad_c_i
    grad = np.zeros_like(x)
    # loop over shifts
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - (n - 1)  # shift value
        if s >= 0:
            # forward contribution: a[j+s]
            grad[: n - s] += weight * a[s:]
            # backward contribution: -x[j-s]
            grad[s:] -= weight * x[: n - s]
        else:
            s_abs = -s
            grad[s_abs:] += weight * a[: n - s_abs]
            grad[: n - s_abs] -= weight * x[s_abs:]
    return f, grad

# ------------------------------------------------------------
# discrete C5 evaluation (scaled by dx = 2/n)
def compute_c5(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * (2.0 / x.size)

# ------------------------------------------------------------
# greedy improvement based on the worst shift
def _worst_shift_info(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def _gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1.0 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1.0 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

def _greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, _, _ = _worst_shift_info(h)
        grad = _gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # full move
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # half move
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

# ------------------------------------------------------------
# SLSQP with analytic gradient on the smoothed objective
def _slsqp_optimize(h0, beta, max_iter=800):
    n = h0.size
    target = n / 2.0

    def fun(x):
        f, _ = _softmax_obj_and_grad(x, beta)
        return f

    def jac(x):
        _, g = _softmax_obj_and_grad(x, beta)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        fun,
        h0,
        method="SLSQP",
        jac=jac,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = _project_box_simplex(x, target)
    return x
