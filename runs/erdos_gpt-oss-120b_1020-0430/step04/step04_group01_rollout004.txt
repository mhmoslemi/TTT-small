def gradient_softmax(h, beta):

    # compute C_k = correlation of h with (1 - h)
    # using correlate_full implemented earlier (FFT)

import time, numpy as np

def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh*fr, L)[:2*n-1]
    return conv

def compute_c5(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

def proj_cap_simplex(v, total, cap=1.0, max_iter=20):
    v = np.clip(v, 0.0, cap)
    s = v.sum()
    diff = total - s
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        mask = (v > 0) & (v < cap)
        m = mask.sum()
        if m == 0:
            # distribute diff equally among non-cap values (maybe all at 0 or cap)
            # if diff>0: add to those < cap; else subtract from those >0.
            # For simplicity break.
            break
        adjust = diff / m
        v[mask] += adjust
        v = np.clip(v, 0.0, cap)
        diff = total - v.sum()
    return v

def gradient_softmax(h, beta, L):
    n = h.size
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    corr = np.fft.irfft(fh*fr, L)[:2*n-1]
    bc = beta * corr
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()
    fw = np.fft.rfft(w, L)
    conv_wh = np.fft.irfft(fw*fh, L)[:len(w)+n-1]
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, L)
    conv_wrev_h = np.fft.irfft(fw_rev*fh, L)[:len(w_rev)+n-1]
    offset = n - 1
    grad = 1.0 - conv_wh[offset:offset+n] - conv_wrev_h[offset:offset+n]
    return grad

def gradient_descent(h_start, beta, step, iters, deadline):
    h = h_start.copy()
    n = h.size
    L = 1 << ((3*n-2).bit_length())
    total = n / 2.0
    for i in range(iters):
        if time.time() >= deadline:
            break
        g = gradient_softmax(h, beta, L)
        h_new = h - step * g
        h = proj_cap_simplex(h_new, total)
        # decay step occasionally
        if (i+1) % (max(1, iters // 5)) == 0:
            step *= 0.7
    return h

def random_perturb(h, scale, rng):
    n = h.size
    h_new = h + rng.normal(0, scale, size=n)
    return proj_cap_simplex(h_new, n/2.0)

def binary_sa(start_bin, rng, deadline):
    n = start_bin.size
    h = start_bin.astype(float).copy()
    total = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 20000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # enforce i is 1, j is 0
        if h[i] < h[j]:
            i, j = j, i
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def random_feasible(n, rng):
    # generate random vector in [0,1] with sum n/2
    # simple approach: start with zeros, fill n/2 positions with 1, then random swap small amounts
    h = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    # optionally randomize fractions
    for _ in range(n*2):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] > 0 and h[j] < 1:
            eps = rng.random() * min(h[i], 1 - h[j])
            h[i] -= eps
            h[j] += eps
    return h

import time
import numpy as np

# --------------------------------------------------------------
# Correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Projection onto the capped simplex {0≤x≤1, Σx=n/2}
# --------------------------------------------------------------
def proj_cap_simplex(v, total, cap=1.0, max_iter=20):
    """Clip to [0,cap] then adjust to meet the exact sum (water‑filling)."""
    v = np.clip(v, 0.0, cap)
    diff = total - v.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        mask = (v > 0) & (v < cap)
        m = mask.sum()
        if m == 0:
            break
        adjust = diff / m
        v[mask] += adjust
        v = np.clip(v, 0.0, cap)
        diff = total - v.sum()
    return v


# --------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) surrogate
# --------------------------------------------------------------
def gradient_softmax(h, beta, L):
    n = h.size
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    corr = np.fft.irfft(fh * fr, L)[: 2 * n - 1]

    bc = beta * corr
    max_bc = bc.max()
    w = np.exp(bc - max_bc)
    w /= w.sum()

    fw = np.fft.rfft(w, L)
    conv_wh = np.fft.irfft(fw * fh, L)[: len(w) + n - 1]

    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, L)
    conv_wrev_h = np.fft.irfft(fw_rev * fh, L)[: len(w_rev) + n - 1]

    offset = n - 1
    grad = 1.0 - conv_wh[offset : offset + n] - conv_wrev_h[offset : offset + n]
    return grad


# --------------------------------------------------------------
# Simple projected gradient descent on the smooth surrogate
# --------------------------------------------------------------
def gradient_descent(h_start, beta, step, iters, deadline):
    h = h_start.copy()
    n = h.size
    L = 1 << ((3 * n - 2).bit_length())
    total = n / 2.0
    for i in range(iters):
        if time.time() >= deadline:
            break
        g = gradient_softmax(h, beta, L)
        h = proj_cap_simplex(h - step * g, total)
        if (i + 1) % max(1, iters // 5) == 0:
            step *= 0.7
    return h


def random_perturb(h, scale, rng):
    return proj_cap_simplex(h + rng.normal(0.0, scale, size=h.size), h.size / 2.0)


# --------------------------------------------------------------
# Binary simulated annealing (explores {0,1} vectors)
# --------------------------------------------------------------
def binary_sa(start_bin, rng, deadline):
    n = start_bin.size
    h = start_bin.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 20000

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


def random_feasible(n, rng):
    """Quick random feasible vector (sum = n/2)."""
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    for _ in range(n * 2):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] > 0 and h[j] < 1:
            eps = rng.random() * min(h[i], 1 - h[j])
            h[i] -= eps
            h[j] += eps
    return h


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # 1. Initialise
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = proj_cap_simplex(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible(n, rng)

    best_h = h0
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Continuous optimisation (smooth surrogate)
    # ------------------------------------------------------------------
    betas = [10.0, 100.0, 500.0, 2000.0]
    step0 = 0.30
    stage_deadline = start_time + 0.70 * budget_s

    while time.time() < stage_deadline:
        beta = rng.choice(betas)
        if rng.random() < 0.7:
            start = random_perturb(best_h, scale=0.05, rng=rng)
        else:
            start = random_feasible(n, rng)
        iters = 2000
        step = step0 * (0.5 ** betas.index(beta))
        h_opt = gradient_descent(start, beta, step, iters, deadline)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # ------------------------------------------------------------------
    # 3. Binary simulated annealing refinement
    # ------------------------------------------------------------------
    bin_deadline = start_time + 0.90 * budget_s
    idx = np.argsort(-best_h)[: n // 2]
    bin_start = np.zeros(n)
    bin_start[idx] = 1.0
    bin_h, bin_c = binary_sa(bin_start, rng, bin_deadline)
    if bin_c < best_c:
        best_h, best_c = bin_h, bin_c

    # optional second SA run if time remains
    if time.time() < bin_deadline:
        other_start = np.zeros(n)
        other_start[rng.choice(n, size=n // 2, replace=False)] = 1.0
        bin_h2, bin_c2 = binary_sa(other_start, rng, bin_deadline)
        if bin_c2 < best_c:
            best_h, best_c = bin_h2, bin_c2

    # ------------------------------------------------------------------
    # 4. Fine‑grained gradient polishing (high β)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        h_fine = gradient_descent(best_h, beta=5000.0, step=0.01,
                                  iters=2000, deadline=deadline)
        c_fine = compute_c5(h_fine)
        if c_fine < best_c:
            best_h, best_c = h_fine, c_fine

    # ------------------------------------------------------------------
    # 5. Final clean‑up and return
    # ------------------------------------------------------------------
    best_h = proj_cap_simplex(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
