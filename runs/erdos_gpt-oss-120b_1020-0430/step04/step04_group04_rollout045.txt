def compute_c5(h):
    """Exact discrete C5 for step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that sum(h)=target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # Final small correction if needed
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # fallback greedy
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, init_step=0.5, n_trials=6):
    """Simple stochastic hill‑climb."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def simulated_annealing(h0, rng, max_iter=6000,
                        init_temp=1e-2, final_temp=1e-5,
                        init_scale=0.4, final_scale=0.02):
    """Simulated‑annealing refinement."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp surrogate of max correlation."""
    n = x.size
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = CURRENT_BETA * conv
    mval = np.max(bc)
    return (mval + np.log(np.sum(np.exp(bc - mval)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=600):
    """SLSQP on smooth surrogate for given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

        candidates.append(random_feasible_vector(n0, rng))

    best_h = None
    best_c = np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # -----------------------------------------------------------------
    # Coarse optimisation (≈50% time budget)
    # -----------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.5:
        # starting point
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.4)
        else:
            start_vec = random_feasible_vector(n0, rng)

        # hill‑climb
        h_hc, c_hc = hill_climb(start_vec, rng, max_iters=1500,
                                 init_step=0.5, n_trials=6)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # simulated annealing
        h_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iter=5000,
                                         init_temp=1e-2, final_temp=1e-5,
                                         init_scale=0.5, final_scale=0.02)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        # a few SLSQP passes with modest β
        for beta in (30, 100, 300, 1000):
            if time.time() - start_time >= budget_s * 0.5:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # -----------------------------------------------------------------
    # Upsample to finer grid if time permits
    # -----------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.15:
        n1 = 720
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(n1) + 0.5) * (2.0 / n1)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, n1 / 2.0, rng)

        # modest hill‑climb + SA on finer grid
        up_h, _ = hill_climb(up_h, rng, max_iters=2000,
                             init_step=0.4, n_trials=6)
        up_h, _ = simulated_annealing(up_h, rng,
                                      max_iter=6000,
                                      init_temp=5e-3, final_temp=1e-5,
                                      init_scale=0.4, final_scale=0.02)

        # refined SLSQP ladder
        for beta in (50, 200, 800, 2000, 5000, 12000, 25000):
            if time.time() - start_time >= budget_s:
                break
            try:
                up_h = slsqp_optimize(up_h, beta, max_iter=800)
            except Exception:
                continue
            c_up = compute_c5(up_h)
            if c_up < best_c:
                best_c, best_h, n0 = c_up, up_h.copy(), n1
        # keep the finer version as current reference
        best_h, n0 = up_h, n1

    # -----------------------------------------------------------------
    # Fine‑grained optimisation on the final grid (remaining time)
    # -----------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.97:
        # hill‑climb
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=2500,
                                 init_step=0.3, n_trials=8)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # simulated annealing
        h_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iter=8000,
                                         init_temp=1e-2, final_temp=1e-6,
                                         init_scale=0.3, final_scale=0.01)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        # SLSQP with a broader β schedule
        for beta in (20, 60, 180, 540, 1620, 4860, 14580, 43740):
            if time.time() - start_time >= budget_s * 0.97:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=1000)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # -----------------------------------------------------------------
    # Final cleanup and return
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n0 / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n0)
