def compute_c5(h):
    """Return the C5 value for a step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def adjust_sum_exact(h, rng):
    """Exact sum correction to make Σh == n/2 while staying in [0,1]."""
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def binary_local_opt(h, rng, deadline, max_iters=20000):

def gradient_optimize(h, rng, deadline, max_iter=200):
    """Projected gradient descent against the current worst shift."""
    n = h.shape[0]
    target = n / 2.0
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    step0 = 0.05

    for it in range(max_iter):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)

        # gradient of the worst shift
        g = np.zeros_like(h)
        if k >= 0:
            g[k:] += 1.0 - h[: n - k]
            g[: n - k] -= h[k:]
        else:
            kk = -k
            g[: n - kk] += 1.0 - h[kk:]
            g[kk:] -= h[: n - kk]

        # step with decay
        α = step0 * (0.8 ** it)
        h = h - α * g
        h = np.clip(h, 0.0, 1.0)

        # exact sum correction
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break

        cur_c = compute_c5(h)
        if cur_c < best_c - 1e-12:
            best_c, best_h = cur_c, h.copy()

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.97  # safety margin

    # --------------------------------------------------------
    # Choose discretisation size
    # --------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 360
    if n % 2 != 0:
        n += 1

    # --------------------------------------------------------
    # Build starter vectors
    # --------------------------------------------------------
    starters = []

    # binary version of the parent (if any)
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=float)
        # enforce exact sum
        parent = adjust_sum_exact(parent, rng)
        idx = np.argpartition(parent, -n // 2)[-n // 2:]
        bin_vec = np.zeros(n, dtype=float)
        bin_vec[idx] = 1.0
        starters.append(bin_vec)

        # keep the fractional parent as a starter as well
        starters.append(parent)

    # several random binary vectors
    for _ in range(6):
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        starters.append(h)

    best_h = None
    best_c = np.inf

    # --------------------------------------------------------
    # Search loop over starters
    # --------------------------------------------------------
    for h0 in starters:
        if time.time() > deadline:
            break
        h_opt, c_opt = improve_until_deadline(h0, rng, deadline)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # One final refinement on the global best (if time permits)
    if best_h is not None and time.time() < deadline:
        best_h, best_c = improve_until_deadline(best_h, rng, deadline)

    # --------------------------------------------------------
    # Fallback (should never happen)
    # --------------------------------------------------------
    if best_h is None:
        best_h = np.zeros(n, dtype=float)
        best_h[: n // 2] = 1.0
        best_c = compute_c5(best_h)

    # Ensure feasibility one last time
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_exact(best_h, rng)

    return best_h.tolist(), float(best_c), int(n)
