def generate_symmetric_h(a, n):
    xs = (np.arange(n)+0.5)*(2.0/n)
    t = np.where(xs <= 1, xs, 2 - xs)
    # piecewise linear with plateau a
    h = np.empty_like(t)
    # compute threshold points:
    # plateau: t <= a => h=1
    # tail: t >= 1 - a => h=0
    # middle: linear decreasing
    mask1 = t <= a
    mask2 = t >= 1 - a
    mask_mid = (~mask1) & (~mask2)
    h[mask1] = 1.0

    # adjust sum
    h = enforce_sum(h, n/2.0, rng)   # use rng
    return h

def enforce_sum_det(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # find indices where 0 < h < 1
    free = (h > 1e-12) & (h < 1 - 1e-12)
    idx = np.where(free)[0]
    if idx.size > 0:
        per = diff / idx.size
        h[idx] += per
        # clip possibly
        h = np.clip(h, 0.0, 1.0)
        # recalc diff and repeat if needed
        diff = target - h.sum()
    # if still leftover, distribute to all entries
    if abs(diff) > 1e-12:
        # adjust smallest/largest?
        # We'll just distribute equally to all elements
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final fix (rare)
    diff = target - h.sum()
    # if still diff, adjust first entry
    if abs(diff) > 1e-12:
        for i in range(h.size):
            max_add = (1.0 - h[i]) if diff > 0 else -h[i]
            add = max(min(diff, max_add), -max_add)
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def perturb_h_preserve(h, rng, scale):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)  # ensure within bounds
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

    if abs(diff) < 1e-12:
        return h

    free = (h > 1e-12) & (h < 1 - 1e-12)
    idx = np.where(free)[0]
    if idx.size > 0:
        per = diff / idx.size
        h[idx] += per
        h[idx] = np.clip(h[idx], 0.0, 1.0)
        diff = target - h.sum()
    # if still diff, adjust all indices equally
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tweak: adjust first few free indices
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(h.size):
            max_add = 1.0 - h[i] if diff > 0 else -h[i]
            add = max(min(diff, max_add), -max_add)
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def correlation_raw(x):
    n = x.shape[0]
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

def compute_c5(h):
    """Exact discrete C5 (max correlation) for a sampled h."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

# ---------- deterministic sum enforcement ----------
def enforce_sum_det(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    free = (h > 1e-12) & (h < 1 - 1e-12)
    idx = np.where(free)[0]
    if idx.size:
        per = diff / idx.size
        h[idx] += per
        h[idx] = np.clip(h[idx], 0.0, 1.0)
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(h.size):

            add = max(min(diff, max_add), -max_add)
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def generate_h_from_a(a, n):
    """Piecewise linear symmetric h on [0,2] defined by plateau length a (0≤a≤0.5)."""
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    t = np.where(xs <= 1.0, xs, 2.0 - xs)  # mirror to [0,1]
    h = np.empty_like(t)
    eps = 1e-12
    if a >= 0.5 - eps:
        h = (t <= 0.5).astype(float)
    else:
        denom = 1.0 - 2.0 * a
        mask1 = t <= a
        mask2 = t >= 1.0 - a
        mask_mid = (~mask1) & (~mask2)
        h[mask1] = 1.0
        h[mask2] = 0.0
        h[mask_mid] = 1.0 - (t[mask_mid] - a) / denom
    return np.clip(h, 0.0, 1.0)

def perturb_h_preserve(h, rng, scale):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()

    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps

        if (it + 1) % decay == 0:

    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        nh = perturb_h_preserve(cur_h, rng, scale)
        nc = compute_c5(nh)
        delta = nc - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = nh, nc
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

# ---------- smooth surrogate & SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = correlation_raw(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

# ---------- main driver ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 720                     # base sample size (< 1000)
    target = n / 2.0

    # ---- grid search over plateau length a ----
    best_h = None
    best_c = np.inf
    a_vals = np.linspace(0.0, 0.5, 1001)
    for a in a_vals:
        h = generate_h_from_a(a, n)
        h = enforce_sum_det(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            best_a = a

    # ---- include parent construction if present ----
    if "initial_h_values" in globals():
        ih = np.array(initial_h_values, dtype=float)
        if ih.size != n:
            old_x = (np.arange(ih.size) + 0.5) * (2.0 / ih.size)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            ih = np.interp(new_x, old_x, ih)
        ih = np.clip(ih, 0.0, 1.0)
        ih = enforce_sum_det(ih, target)
        c = compute_c5(ih)
        if c < best_c:
            best_c, best_h = c, ih.copy()

    # ---- a few random starts ----
    for _ in range(5):
        h = random_feasible_vector(n, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ---- intensive hill‑climbing loop ----
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.75:
            break
        # main hill‑climb from current best
        h_new, c_new = hill_climb(best_h, rng,
                                  max_iters=4000,
                                  init_scale=0.5)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        # occasional random‑restart climb
        if rng.random() < 0.25:
            h_rand = random_feasible_vector(n, rng)
            h_new, c_new = hill_climb(h_rand, rng,
                                      max_iters=3000,
                                      init_scale=0.5)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()

    # ---- simulated annealing (if time remains) ----
    if time.time() - start < budget_s * 0.9:
        h_sa, c_sa = simulated_annealing(
            best_h, rng,
            max_iter=8000,
            init_temp=1e-2,
            final_temp=1e-5,
            init_scale=0.5,
            final_scale=0.02,
        )
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

    # ---- SLSQP surrogate refinement (last stage) ----
    if time.time() - start < budget_s:
        for beta in (50, 200, 800, 2000, 5000):
            if time.time() - start > budget_s:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
            except Exception:
                continue

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_det(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
