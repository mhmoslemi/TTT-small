def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initial
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0, 1)
        h0 = enforce_sum(h0, n/2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)
    # candidate pool
    candidates = [h0]
    # maybe add triangular, cosine etc...
    # compute best
    best_h = None; best_c=inf
    for cand in candidates:
        c=compute_c5(cand)
        if c<best_c: best_c=c; best_h=cand.copy()
    # Main loop
    while time.time() - start < 0.55*budget_s:
        # random restart with certain prob
        if rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.4)
        else:
            start_vec = random_feasible_vector(n, rng)
        # hill climbing
        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=4000, init_step=0.5, n_trials=6)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
        # SA
        h_sa, c_sa = simulated_annealing(best_h, rng,
                                        max_iter=12000,
                                        init_temp=5e-2,
                                        final_temp=1e-6,
                                        init_scale=0.5,
                                        final_scale=0.02)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        # SLSQP ladder
        beta_seq = [20, 100, 500, 2000, 8000, 20000]
        for beta in beta_seq:
            if time.time() - start > 0.55*budget_s:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=800)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * ((final_temp / init_temp) ** t) if init_temp > 0 else 0.0
        scale = init_scale * ((final_scale / init_scale) ** t) if init_scale > 0 else 0.0

            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

def softmax_obj(x):
    # Global CURRENT_BETA, CURRENT_TARGET not used directly except within compute constraints satisfied later.
    rev = (1 - x)[::-1]
    n = x.shape[0]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    corr = np.fft.irfft(H * R, m)[:L]
    bc = CURRENT_BETA * corr
    m_val = np.max(bc)
    # LSE stable
    lse = np.log(np.sum(np.exp(bc - m_val)))
    # return smoothed max
    return (m_val + lse) / CURRENT_BETA

                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * ((final_temp / init_temp) ** t) if init_temp > 0 else 0.0
        scale = init_scale * ((final_scale / init_scale) ** t) if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def softmax_obj(x):
    """Smooth surrogate of max using log-sum-exp, using globals."""
    rev = (1 - x)[::-1]
    n = x.shape[0]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    corr = np.fft.irfft(H * R, m)[:L]
    bc = CURRENT_BETA * corr
    m_val = np.max(bc)
    lse = np.log(np.sum(np.exp(bc - m_val)))
    return (m_val + lse) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - CURRENT_TARGET}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # Initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ------------------------------------------------------------------
    # Build a modest candidate pool
    # ------------------------------------------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    # triangular bump
    tri = np.where(xs <= 1, xs, 2 - xs)
    candidates.append(enforce_sum(tri, n / 2.0, rng))

    # cosine bump
    cos = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    candidates.append(enforce_sum(cos, n / 2.0, rng))

    # linear ramp up
    ramp_up = xs / 2.0
    candidates.append(enforce_sum(ramp_up, n / 2.0, rng))

    # linear ramp down
    ramp_down = 1 - xs / 2.0
    candidates.append(enforce_sum(ramp_down, n / 2.0, rng))

    # binary patterns
    for _ in range(3):
        candidates.append(enforce_sum(binary_feasible_vector(n, rng), n / 2.0, rng))

    # random vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # ------------------------------------------------------------------
    # Pick the best initial candidate
    # ------------------------------------------------------------------
    best_h = None
    best_c = np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ------------------------------------------------------------------
    # Phase 1: intensive search on current grid size
    # ------------------------------------------------------------------
    phase1_end = start + budget_s * 0.55
    while time.time() < phase1_end:
        # starting point for this round
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill climbing
        h_hc, c_hc = hill_climb_multi(start_vec, rng,
                                      max_iters=4000,
                                      init_step=0.5,
                                      n_trials=8)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # simulated annealing
        h_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iter=15000,
                                         init_temp=5e-2,
                                         final_temp=1e-6,
                                         init_scale=0.5,
                                         final_scale=0.02)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        # SLSQP refinement ladder
        for beta in (20, 100, 500, 2000, 8000, 20000):
            if time.time() >= phase1_end:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=700)
            except Exception:
                continue
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # ------------------------------------------------------------------
    # Phase 2: up‑sample and fine‑tune on finer grid
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > budget_s * 0.2:
        n_up = min(1080, max(720, n * 3))
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(n_up) + 0.5) * (2.0 / n_up)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, n_up / 2.0, rng)

        # quick local improvement
        up_h, c_up = hill_climb_multi(up_h, rng,
                                      max_iters=2500,
                                      init_step=0.4,
                                      n_trials=6)
        if c_up < best_c:
            best_c, best_h, n = c_up, up_h.copy(), n_up

        up_h, c_up = simulated_annealing(up_h, rng,
                                          max_iter=8000,
                                          init_temp=3e-2,
                                          final_temp=1e-6,
                                          init_scale=0.4,
                                          final_scale=0.02)
        if c_up < best_c:
            best_c, best_h, n = c_up, up_h.copy(), n_up

        # SLSQP ladder on fine grid (fewer iterations)
        for beta in (50, 200, 800, 2000, 8000):
            if time.time() - start > budget_s:
                break
            try:
                up_h = slsqp_optimize(up_h, beta, max_iter=500)
            except Exception:
                continue
            c_up = compute_c5(up_h)
            if c_up < best_c:
                best_c, best_h, n = c_up, up_h.copy(), n_up

        # final fine hill‑climb if time permits
        while time.time() - start < budget_s - 5:
            up_h, c_up = hill_climb_multi(up_h, rng,
                                          max_iters=1500,
                                          init_step=0.3,
                                          n_trials=5)
            if c_up < best_c:
                best_c, best_h, n = c_up, up_h.copy(), n_up
            else:
                break

    # ------------------------------------------------------------------
    # Final cleanup
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
