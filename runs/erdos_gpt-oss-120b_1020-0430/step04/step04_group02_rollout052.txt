def hill_climb(h, rng, max_iters=2000, init_step=0.5, n_trials=5, p_target=0.3, top_k=3):
    h_cur = h.copy()
    best_c = compute_c5_fft(h_cur)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h_cur.size

    for it in range(max_iters):
        # compute corr and top shifts
        corr = np.fft.rfft(h_cur, n*2)
        rev = (1 - h_cur)[::-1]
        rev_fft = np.fft.rfft(rev, n*2)
        conv_full = np.fft.irfft(corr * rev_fft, n*2)[:2*n-1]
        max_idx = int(np.argmax(conv_full))
        # top idx
        top_indices = np.argpartition(-conv_full, top_k-1)[:top_k]
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                # pick random top shift
                idx = rng.choice(top_indices)
                k = idx - (n - 1)
                cand = targeted_move(h_cur, rng, k, scale=step)
            else:
                cand = perturb_h(h_cur, rng, scale=step)
            c = compute_c5_fft(cand)
            if c < best_c - 1e-12:
                best_c, h_cur = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h_cur, rng, scale=step * 2.0)
            c = compute_c5_fft(cand)
            if c < best_c:
                best_c, h_cur = c, cand
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return h_cur, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i_low] += eps
    nh[i_low + k] += eps
    nh[i_high] -= eps
    nh[i_high + k] -= eps
    return nh


# ---------- correlation ----------
def compute_corr_fft(h):
    n = h.size
    a = np.fft.rfft(h, n * 2)
    b = np.fft.rfft((1 - h)[::-1], n * 2)
    conv = np.fft.irfft(a * b, n * 2)[: 2 * n - 1]
    return conv


def compute_c5_fft(h):
    n = h.size
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)


# ---------- softmax and gradient ----------
def softmax_weights(corr, beta):
    m = np.max(corr)
    ex = np.exp(beta * (corr - m))
    s = np.sum(ex)
    return ex / s


def grad_softmax_max(h, beta):
    n = h.size
    corr = compute_corr_fft(h)
    w = softmax_weights(corr, beta)
    w_pad = np.zeros(2 * n)
    w_pad[: 2 * n - 1] = w
    fw = np.fft.rfft(w_pad, 2 * n)
    fh = np.fft.rfft(h, 2 * n)
    fa = np.fft.rfft(1.0 - h, 2 * n)
    conv1 = np.fft.irfft(fw * fa, 2 * n)
    conv2 = np.fft.irfft(fw * fh, 2 * n)
    term1 = conv1[n - 1 : 2 * n - 1]
    term2 = conv2[n - 1 : 2 * n - 1]
    return term1 - term2


# ---------- hill climbing ----------
def hill_climb(h_start, rng, max_iters=2000, init_step=0.5,
               n_trials=5, p_target=0.3, top_k=3, deadline=None):
    h = h_start.copy()
    best_c = compute_c5_fft(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.size
    for it in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        corr = compute_corr_fft(h)
        max_idx = int(np.argmax(corr))
        top_idx = np.argpartition(-corr, top_k - 1)[:top_k]
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                idx = rng.choice(top_idx)
                k = idx - (n - 1)
                cand = targeted_move(h, rng, k, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5_fft(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5_fft(cand)
            if c < best_c:
                best_c, h = c, cand
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return h, best_c


# ---------- gradient descent on smoothed max ----------
def gradient_descent(h_start, rng, beta, steps, lr, deadline=None):
    h = h_start.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fft(h)
    for _ in range(steps):
        if deadline is not None and time.time() >= deadline:
            break
        grad = grad_softmax_max(h, beta)
        nh = h - lr * grad
        nh = project_box_sum(np.clip(nh, 0.0, 1.0), n / 2.0)
        h = nh
        c = compute_c5_fft(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c


# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq(x):
    return np.sum(x) - CURRENT_TARGET


def _soft_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq}
    res = minimize(
        _soft_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)
