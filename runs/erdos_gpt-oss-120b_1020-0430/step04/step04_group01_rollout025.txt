import time, numpy as np, scipy.fft (or np.fft)

def correlate_full(h):
    n = len(h)
    L = 1 << ((2*n-1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]
    return conv

def compute_c5(h):
    conv = correlate_full(h)
    return conv.max() * (2.0 / len(h))

def grad_softmax(h, beta):
    n = len(h)
    conv = correlate_full(h)
    # convert to w
    # To avoid overflow, subtract max before exp
    bc = beta * conv
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / exp_bc.sum()
    # compute w_rev
    w_rev = w[::-1]
    # convolution lengths
    L = 1 << ((3*n-2).bit_length())
    # compute conv_rev = w_rev * (1-h)
    fw = np.fft.rfft(w_rev, L)
    f_one_minus = np.fft.rfft(1 - h, L)
    conv_rev = np.fft.irfft(fw * f_one_minus, L)
    # conv2 = w * h
    fw2 = np.fft.rfft(w, L)
    fh = np.fft.rfft(h, L)
    conv2 = np.fft.irfft(fw2 * fh, L)
    start = n - 1
    grad = conv_rev[start:start + n] - conv2[start:start + n]
    return grad

def project_feasible(h, target, rng):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # Ensure sum
    diff = target - h.sum()
    # If diff small, return
    if abs(diff) < 1e-12:
        return h
    # Randomly adjust
    n = len(h)
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # pick index that can accept diff
        if diff > 0:
            # need increase some entries not at 1
            idx = rng.integers(0, n)
            if h[idx] >= 1.0:
                continue
            add = min(diff, 1.0 - h[idx])
        else:
            idx = rng.integers(0, n)
            if h[idx] <= 0.0:
                continue
            add = max(diff, -h[idx])
        h[idx] += add
        diff = target - h.sum()
    # If still diff, adjust uniformly
    if abs(diff) > 1e-12:
        # distribute diff equally among non-bound entries
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            avg = diff / free.size
            h[free] += avg
            # clip again
            h = np.clip(h, 0.0, 1.0)
    # final adjust if any residual
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        # small final adjustment on a random index
        idx = rng.integers(0, n)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def adam_optimize(h0, beta, steps, lr, target, rng, deadline):
    h = h0.copy()
    n = len(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    for i in range(steps):
        if time.time() >= deadline:
            break
        t += 1
        grad = grad_softmax(h, beta)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, target, rng)
    return h

def binary_sa(start_bin, rng, deadline):
    # copy previous code but adjust for possibly larger n (720). Use compute_c5.
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()

    Tf = 1e-6
    max_iters = 20000
    for it in range(max_iters):
        if time.time() >= deadline:

        j = rng.integers(0, n)
        if h[i] == h[j]:

        delta = c - best_c

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):

    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):

        c = compute_c5(new_h)

            h = new_h

import time
import numpy as np

# --------------------------------------------------------------
# Core correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_full(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) (full length)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h: np.ndarray) -> float:
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Feasibility helpers
# --------------------------------------------------------------
def project_feasible(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the required sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return h
    # random iterative fix
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    # small uniform correction if still off
    if abs(diff) > 1e-9:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            avg = diff / free.size
            h[free] += avg
            h = np.clip(h, 0.0, 1.0)
        else:
            i = rng.integers(0, n)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    return project_feasible(h, target, rng)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) objective
# --------------------------------------------------------------
def grad_softmax(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of (1/β)·log‑sum‑exp(β·conv) w.r.t. h."""
    n = h.size
    conv = correlate_full(h)

    # stabilized softmax weights
    bc = beta * conv
    m = bc.max()
    w = np.exp(bc - m)
    w /= w.sum()

    # convolution with w and w reversed
    L = 1 << ((3 * n - 2).bit_length())
    w_rev = w[::-1]

    fw_rev = np.fft.rfft(w_rev, L)
    f_one_minus = np.fft.rfft(1 - h, L)
    conv_rev = np.fft.irfft(fw_rev * f_one_minus, L)

    fw = np.fft.rfft(w, L)
    fh = np.fft.rfft(h, L)
    conv2 = np.fft.irfft(fw * fh, L)

    start = n - 1
    grad = conv_rev[start : start + n] - conv2[start : start + n]
    return grad


# --------------------------------------------------------------
# Adam optimizer with projection
# --------------------------------------------------------------
def adam_optimize(
    h0: np.ndarray,
    beta: float,
    steps: int,
    lr: float,
    target: float,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    h = h0.copy()
    n = h.size
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    for _ in range(steps):
        if time.time() >= deadline:
            break
        t += 1
        g = grad_softmax(h, beta)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, target, rng)
    return h


# --------------------------------------------------------------
# Binary simulated annealing (discrete refinement)
# --------------------------------------------------------------
def binary_sa(start_bin: np.ndarray, rng: np.random.Generator, deadline: float):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 20000

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# --------------------------------------------------------------
# Simple greedy hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h_start: np.ndarray, rng: np.random.Generator, max_iters: int = 2000, start_step: float = 0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c
