import time, numpy as np

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_sum_bounds(h, target, iters=5):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h + diff / h.size
    h = np.clip(h, 0.0, 1.0)
    for _ in range(iters):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.any():
                adjust = diff / mask.sum()
                h[mask] += adjust
        else:
            mask = h > 0.0
            if mask.any():
                adjust = (-diff) / mask.sum()
                h[mask] -= adjust
        h = np.clip(h, 0.0, 1.0)
    return h

def build_shape(c, e, n):
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    d = np.minimum(xs, 2.0 - xs)
    if e == c:
        h = np.full(n, c, dtype=float)
        return h
    a = (1.0 - 2.0 * c) / (e - c)
    a = max(0.0, min(1.0, a))
    h = np.empty(n, dtype=float)
    if a > 1e-12:
        linear = d <= a
        h[linear] = e - (e - c) * (d[linear] / a)
        h[~linear] = c
    else:
        # no linear region, everything is c (which also implies e=c)
        h[:] = c
    return h

# gradient for shift
def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 720

    target_sum = n / 2.0
    best_h = None

    # --- parametric search ---

    for c in c_vals:

        if e_min > 1.0:
            continue
        e_steps = int(round((1.0 - e_min) / 0.01))
        e_vals = np.linspace(e_min, 1.0, max(1, e_steps + 1))
        for e in e_vals:

            h = project_sum_bounds(h, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_params = (c, e)

    # fine search around best_params
    c0, e0 = best_params
    c_fine = np.linspace(max(0.0, c0 - 0.02), min(0.5, c0 + 0.02), 41)
    for c in c_fine:
        e_min = max(c, 1.0 - c)
        e_low = max(e_min, e0 - 0.02)
        e_high = min(1.0, e0 + 0.02)
        if e_low > e_high:
            continue
        e_fine = np.linspace(e_low, e_high, 41)
        for e in e_fine:
            h = build_shape(c, e, n)
            h = project_sum_bounds(h, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_params = (c, e)

    # polish with greedy local search
    best_h = greedy_optimize(best_h, max_steps=4000)

    # further random perturbation loop until time nearly done
    while time.time() - start_time < budget_s * 0.95:
        cand = perturb_random(best_h, rng, scale=0.05)
        cand = greedy_optimize(cand, max_steps=2000)
        c5cand = compute_c5(cand)
        if c5cand < best_c5:
            best_c5 = c5cand
            best_h = cand

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.size)

import time, numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def project_sum_bounds(h, target, iters=5):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h + diff / h.size
    h = np.clip(h, 0.0, 1.0)
    for _ in range(iters):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.any():
                adjust = diff / mask.sum()
                h[mask] += adjust
        else:
            mask = h > 0.0
            if mask.any():
                adjust = (-diff) / mask.sum()
                h[mask] -= adjust
        h = np.clip(h, 0.0, 1.0)
    return h

def build_shape(c, e, n):
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    d = np.minimum(xs, 2.0 - xs)
    if abs(e - c) < 1e-12:
        return np.full(n, c, dtype=float)
    a = (1.0 - 2.0 * c) / (e - c)
    a = max(0.0, min(1.0, a))
    h = np.empty(n, dtype=float)
    if a > 1e-12:
        linear = d <= a
        h[linear] = e - (e - c) * (d[linear] / a)
        h[~linear] = c
    else:
        h[:] = c
    return h

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# SLSQP helper globals
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum projection
    x = project_sum_bounds(x, CURRENT_TARGET)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 720
    target_sum = n / 2.0
    best_h = None
    best_c5 = float('inf')
    # ---- coarse grid search ----
    for c in np.linspace(0.0, 0.5, 51):
        e_min = max(c, 1.0 - c)
        if e_min > 1.0:
            continue
        steps = max(1, int(round((1.0 - e_min) / 0.01)))
        for e in np.linspace(e_min, 1.0, steps + 1):
            h = build_shape(c, e, n)
            h = project_sum_bounds(h, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_params = (c, e)
    # ---- fine grid around best ----
    c0, e0 = best_params
    for c in np.linspace(max(0.0, c0-0.02), min(0.5, c0+0.02), 41):
        e_min = max(c, 1.0 - c)
        e_low = max(e_min, e0-0.02)
        e_high = min(1.0, e0+0.02)
        if e_low > e_high:
            continue
        for e in np.linspace(e_low, e_high, 41):
            h = build_shape(c, e, n)
            h = project_sum_bounds(h, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_params = (c, e)
    # ---- greedy refinement ----
    best_h = greedy_optimize(best_h, max_steps=5000)
    best_c5 = compute_c5(best_h)

    # ---- SLSQP polishing with increasing beta ----
    for beta in (200, 800, 2000, 5000, 10000):
        if time.time() - start > budget_s * 0.90:
            break
        refined = slsqp_optimize(best_h, beta, max_iter=1000)
        c5 = compute_c5(refined)
        if c5 < best_c5:
            best_h = refined
            best_c5 = c5

    # ---- random perturb local search ----
    while time.time() - start < budget_s * 0.98:
        cand = perturb_random(best_h, rng, scale=0.05)
        cand = greedy_optimize(cand, max_steps=2000)
        c5cand = compute_c5(cand)
        if c5cand < best_c5:
            best_h = cand
            best_c5 = c5cand

    # final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.size)

import time, numpy as np
from scipy.optimize import minimize

# -------------------- core utilities --------------------
def compute_c5(h):
    """Maximum overlap integral for discrete h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_bounds(h, target, iters=5):
    """Adjust h to satisfy sum(target) while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h + diff / h.size
    h = np.clip(h, 0.0, 1.0)
    for _ in range(iters):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.any():
                adj = diff / mask.sum()
                h[mask] += adj
        else:
            mask = h > 0.0
            if mask.any():
                adj = (-diff) / mask.sum()
                h[mask] -= adj
        h = np.clip(h, 0.0, 1.0)
    return h


def build_shape(c, e, n):
    """Symmetric piecewise‑linear shape with edge value e, centre plateau c."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    d = np.minimum(xs, 2.0 - xs)          # distance to nearest endpoint
    if abs(e - c) < 1e-12:
        return np.full(n, c, dtype=float)
    a = (1.0 - 2.0 * c) / (e - c)        # half‑width of linear edge region
    a = max(0.0, min(1.0, a))
    h = np.empty(n, dtype=float)
    if a > 1e-12:
        linear = d <= a
        h[linear] = e - (e - c) * (d[linear] / a)
        h[~linear] = c
    else:
        h[:] = c
    return h


# -------------------- greedy local search --------------------
def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]


def greedy_optimize(h, max_steps=2000):
    """Deterministic pairwise transfers that never increase C5."""
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h


def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


# -------------------- smooth SLSQP refinement --------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_sum_bounds(x, CURRENT_TARGET)
    return x


# -------------------- main entry point --------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 720
    target_sum = n / 2.0

    best_h = None
    best_c5 = float("inf")
    best_params = (0.0, 1.0)

    # ---- coarse grid over (c, e) ----
    for c in np.linspace(0.0, 0.5, 51):
        e_min = max(c, 1.0 - c)
        steps = max(1, int(round((1.0 - e_min) / 0.01)))
        for e in np.linspace(e_min, 1.0, steps + 1):
            h = build_shape(c, e, n)
            h = project_sum_bounds(h, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h, best_params = c5, h, (c, e)

    # ---- fine grid around current best ----
    c0, e0 = best_params
    for c in np.linspace(max(0.0, c0 - 0.02), min(0.5, c0 + 0.02), 41):
        e_min = max(c, 1.0 - c)
        e_low = max(e_min, e0 - 0.02)
        e_high = min(1.0, e0 + 0.02)
        if e_low > e_high:
            continue
        for e in np.linspace(e_low, e_high, 41):
            h = build_shape(c, e, n)
            h = project_sum_bounds(h, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h, best_params = c5, h, (c, e)

    # ---- deterministic greedy polishing ----
    best_h = greedy_optimize(best_h, max_steps=5000)
    best_c5 = compute_c5(best_h)

    # ---- SLSQP smoothing with increasing beta ----
    for beta in (200, 800, 2000, 5000, 10000):
        if time.time() - start > budget_s * 0.90:
            break
        refined = slsqp_optimize(best_h, beta, max_iter=1000)
        c5 = compute_c5(refined)
        if c5 < best_c5:
            best_h, best_c5 = refined, c5

    # ---- stochastic local search until time limit ----
    while time.time() - start < budget_s * 0.98:
        cand = perturb_random(best_h, rng, scale=0.05)
        cand = greedy_optimize(cand, max_steps=2000)
        c5cand = compute_c5(cand)
        if c5cand < best_c5:
            best_h, best_c5 = cand, c5cand

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.size)
