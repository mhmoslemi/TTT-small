            if c_ref < best_c:

    # final ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size/2.0, rng)

def subgradient_descent(h, max_iters, factor_start=0.5, factor_end=0.02, time_limit=None):
    start = time.time()
    n = h.size
    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break
        s, _, cur = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos_idx = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg_idx = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            break
        i = pos_idx[np.argmax(grad[pos_idx])]
        j = neg_idx[np.argmin(grad[neg_idx])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            break
        factor = factor_start * ((factor_end/factor_start) ** (it / max_iters))
        eps = max_eps * factor
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        # we need to ensure sum unchanged, it is.
        # compute C5 improvement
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            # optionally continue; else could go to next iteration
        else:
            # If didn't improve, maybe try smaller eps?
            # Try eps = max_eps * factor * 0.5, else break
            eps2 = eps * 0.5
            if eps2 > 1e-12:
                nh[i] = h[i] - eps2
                nh[j] = h[j] + eps2
                if compute_c5(nh) + 1e-12 < compute_c5(h):
                    h = nh
                    continue
            break
    return h

def search_on_grid(n, rng, start_time, time_budget, init=None):
    best_h = None
    best_c = np.inf
    # start with init if exists
    if init is not None:
        h0 = init.copy()
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n/2.0, rng)
        best_h = h0
        best_c = compute_c5(h0)
    else:
        # random start
        h0 = random_feasible_vector(n, rng)
        best_h = h0
        best_c = compute_c5(h0)

    # main loop
    while time.time() - start_time < start_time + time_budget:
        # generate start
        if best_h is not None and rng.random() < 0.6:
            start = perturb_random(best_h, rng, scale=0.4)
        else:
            start = random_feasible_vector(n, rng)

        # local improvement: greedy
        local = greedy_optimize(start, max_steps=2000)
        # small descent
        local = subgradient_descent(local, max_iters=2000, factor_start=0.3, factor_end=0.02,
                                   time_limit=time_budget - (time.time() - start_time))
        c_local = compute_c5(local)
        if c_local < best_c:
            best_h, best_c = local, c_local

        # occasional SLSQP polish
        if rng.random() < 0.2:
            for beta in (2000, 5000, 8000):
                if time.time() - start_time > start_time + time_budget * 0.95:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=800)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined, c_ref
        # break condition if time low
        if time.time() - start_time > time_budget * 0.95:
            break

    return best_h, best_c

import time, numpy as np
from scipy.optimize import minimize

# ---------- global vars for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- basic utilities ----------
def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

def subgradient_descent(h, max_iters, factor_start, factor_end, end_time):
    it = 0
    n = h.size
    while it < max_iters and time.time() < end_time:
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            break
        factor = factor_start * ((factor_end / factor_start) ** (it / max_iters))
        eps = max_eps * factor
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
        else:
            # try a smaller step
            eps2 = eps * 0.5
            if eps2 > 1e-12:
                nh[i] = h[i] - eps2
                nh[j] = h[j] + eps2
                if compute_c5(nh) + 1e-12 < compute_c5(h):
                    h = nh
        it += 1
    return h

# ---------- SLSQP with softmax ----------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- interpolation ----------
def upscale(h, old_n, new_n):
    xs_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    xs_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(xs_new, xs_old, h)

# ---------- core search on a fixed grid ----------
def search_on_grid(n, rng, global_start, time_budget, init=None):
    end_time = global_start + time_budget
    best_h = None
    best_c = np.inf

    if init is not None:
        cand = enforce_sum(np.clip(init, 0.0, 1.0), n / 2.0, rng)
        best_h = cand
        best_c = compute_c5(cand)

    # a few quick random seeds to possibly improve start
    for _ in range(5):
        if time.time() > end_time:
            break
        cand = random_feasible_vector(n, rng)
        c = compute_c5(cand)
        if c < best_c:
            best_h, best_c = cand, c

    while time.time() < end_time:
        if best_h is not None and rng.random() < 0.6:
            start = perturb_random(best_h, rng, scale=0.4)
        else:
            start = random_feasible_vector(n, rng)

        local = greedy_optimize(start, max_steps=1500)
        local = subgradient_descent(
            local,
            max_iters=1500,
            factor_start=0.3,
            factor_end=0.02,
            end_time=end_time,
        )
        c_local = compute_c5(local)
        if c_local < best_c:
            best_h, best_c = local, c_local

        if rng.random() < 0.15:
            for beta in (2000, 5000, 8000):
                if time.time() > end_time * 0.95:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=600)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined, c_ref
        # safety break if time low
        if time.time() > end_time - 0.5:
            break
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    remaining = budget_s

    # optional parent solution
    init_parent = None
    if "initial_h_values" in globals():

    # ----- Stage 1: coarse grid 240 -----

    h_up = upscale(best_h, 240, 480)
    best_h, best_c = search_on_grid(480, rng, start_time, stage_time, init=h_up)
    remaining -= stage_time

    # ----- Stage 3: grid 720 (use parent if available) -----
    stage_time = remaining * 0.30
    h_up = upscale(best_h, 480, 720)
    init_candidates = [h_up]
    if init_parent is not None and init_parent.size == 720:
        init_candidates.append(init_parent)
    # pick best among candidates quickly
    best_init = h_up
    best_init_c = compute_c5(h_up)
    for cand in init_candidates[1:]:
        c = compute_c5(cand)
        if c < best_init_c:
            best_init, best_init_c = cand, c
    best_h, best_c = search_on_grid(
        720, rng, start_time, stage_time, init=best_init

    remaining -= stage_time

    # ----- Stage 4: grid 960 (if time permits) -----

        best_h, best_c = search_on_grid(960, rng, start_time, stage_time, init=h_up)
        remaining -= stage_time

    # ----- Final polishing with high‑beta SLSQP -----

        c_ref = compute_c5(refined)
        if c_ref < best_c:
            best_h, best_c = refined, c_ref

    # final feasibility fix
    best_h = np.clip(best_h, 0.0, 1.0)
