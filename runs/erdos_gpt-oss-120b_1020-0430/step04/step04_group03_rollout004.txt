def pair_optimize(h, rng, time_limit, start_time):
    n = h.shape[0]
    L = 2*n - 1
    best_h = h.copy()
    best_corr = compute_corr(h)
    best_max = best_corr.max()
    # Pre-allocate s_vals? Not needed inside.
    while True:
        # Check time
        if time.time() - start_time > time_limit:
            break
        # recompute corr and grads each iteration
        corr = compute_corr(h)
        max_corr = corr.max()
        s0_idx = np.argmax(corr)
        # compute gradient matrix
        grads = gradient_matrix(h)  # shape (n, L)
        # sample pairs
        B = 4000   # batch size
        i_arr = rng.integers(0, n, size=B)
        j_arr = rng.integers(0, n, size=B)
        mask_neq = i_arr != j_arr
        i_arr = i_arr[mask_neq]
        j_arr = j_arr[mask_neq]
        if i_arr.size == 0:
            break
        eps_max_arr = np.minimum(h[i_arr], 1.0 - h[j_arr])
        valid_mask = eps_max_arr > 1e-12
        if not np.any(valid_mask):
            break
        i_arr = i_arr[valid_mask]
        j_arr = j_arr[valid_mask]
        eps_max_arr = eps_max_arr[valid_mask]

        # compute delta rows
        delta = grads[j_arr] - grads[i_arr]   # shape (m, L)
        delta0 = delta[:, s0_idx]  # shape (m,)

        # need delta0 < 0
        mask_delta0 = delta0 < -1e-12
        if not np.any(mask_delta0):
            # No pair reduces worst shift directly, may need other approach
            # break to avoid infinite loop
            break
        i_arr = i_arr[mask_delta0]
        j_arr = j_arr[mask_delta0]
        eps_max_arr = eps_max_arr[mask_delta0]
        delta = delta[mask_delta0]
        delta0 = delta0[mask_delta0]

        # compute epsilon candidates
        diff = max_corr - corr  # shape L
        # denom = delta - delta0[:, None]
        denom = delta - delta0[:, None]  # broadcast; shape (m, L)
        # eps_candidates where denom > 0
        pos_mask = denom > 1e-12
        # set large value for non-positive denom
        eps_cand = np.full_like(denom, np.inf)
        eps_cand[pos_mask] = diff[None, :] / denom[pos_mask]
        # min positive eps per row
        min_eps = np.min(eps_cand, axis=1)  # shape (m,)
        # limit by eps_max
        eps_limit = np.minimum(eps_max_arr, min_eps)
        # compute new max for each candidate
        new_max = max_corr + eps_limit * delta0  # delta0 negative
        improvement = max_corr - new_max
        # choose best improvement
        best_idx = np.argmax(improvement)
        if improvement[best_idx] <= 1e-12:
            # no improving pair found
            break
        # apply move
        i_best = i_arr[best_idx]
        j_best = j_arr[best_idx]
        eps_best = eps_limit[best_idx]
        h[i_best] -= eps_best
        h[j_best] += eps_best
        # ensure clipping due to numerical errors
        # but should stay within [0,1]
        h[i_best] = max(0.0, min(1.0, h[i_best]))
        h[j_best] = max(0.0, min(1.0, h[j_best]))
        # continue loop
    return h

import time
import numpy as np
from scipy.optimize import minimize

# -------------------- feasibility utilities --------------------
def enforce_sum(h, target, rng):
    """Clip h to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# -------------------- objective --------------------
def compute_c5(h):
    """Discrete C5 for a vector h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

# -------------------- greedy improvement --------------------
def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full epsilon
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half epsilon

        nh[j] += eps

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# -------------------- gradient matrix (full correlation) --------------------
def gradient_matrix(h):
    """Return array (n, 2n-1) where grads[i,s] = d/d h_i corr(s)."""
    n = h.size
    L = 2 * n - 1
    s_vals = np.arange(-(n - 1), n, dtype=int)  # shape (L,)
    grads = np.empty((n, L), dtype=float)
    for i in range(n):
        grad = np.zeros(L, dtype=float)
        pos_idx = i + s_vals
        mask = (pos_idx >= 0) & (pos_idx < n)
        grad[mask] = 1.0 - h[pos_idx[mask]]
        neg_idx = i - s_vals
        mask2 = (neg_idx >= 0) & (neg_idx < n)
        grad[mask2] -= h[neg_idx[mask2]]
        grads[i] = grad
    return grads

def pair_optimize(h, rng, end_time, start_time):
    n = h.size
    L = 2 * n - 1
    while True:
        if time.time() > end_time:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        s0_idx = np.argmax(corr)
        grads = gradient_matrix(h)  # (n, L)

        B = 4000
        i_arr = rng.integers(0, n, size=B)
        j_arr = rng.integers(0, n, size=B)
        mask = i_arr != j_arr
        i_arr = i_arr[mask]
        j_arr = j_arr[mask]
        if i_arr.size == 0:

        valid = eps_max_arr > 1e-12
        if not np.any(valid):

        j_arr = j_arr[valid]
        eps_max_arr = eps_max_arr[valid]

        delta = grads[j_arr] - grads[i_arr]          # (m, L)
        delta0 = delta[:, s0_idx]                    # (m,)

        # Need a direction that reduces the worst shift
        good = delta0 < -1e-12
        if not np.any(good):
            break
        i_arr = i_arr[good]
        j_arr = j_arr[good]
        eps_max_arr = eps_max_arr[good]
        delta = delta[good]
        delta0 = delta0[good]

        diff = max_corr - corr                       # (L,)
        denom = delta - delta0[:, None]              # (m, L)
        eps_cand = np.where(denom > 1e-12, diff / denom, np.inf)  # (m, L)
        min_eps = eps_cand.min(axis=1)               # (m,)

        eps_limit = np.minimum(eps_max_arr, min_eps)
        new_max = max_corr + eps_limit * delta0      # delta0<0

        h[j_best] = max(0.0, min(1.0, h[j_best]))
    return h
