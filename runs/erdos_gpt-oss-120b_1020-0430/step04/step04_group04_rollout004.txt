    best_c = compute_c5(h)

import time, numpy as np
from scipy.optimize import minimize

# ----- constraints for SLSQP surrogate -----
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    rev = (1 - x)[::-1]
    n = x.size
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = CURRENT_BETA * conv
    mval = np.max(bc)
    return (mval + np.log(np.sum(np.exp(bc - mval)))) / CURRENT_BETA

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def conv_fft(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    return np.fft.irfft(H * R, m)[:L]

def enforce_sum(h, target, rng, max_iter=5000):

    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:

            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)

                    break

        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    best_h, best_c = h0.copy(), compute_c5(h0)
    cur_h, cur_c = best_h.copy(), best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    rng = np.random.default_rng()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def project_simple(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            continue
        delta = diff / cnt
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

def subgradient_descent(h, target, max_iters, step0, time_limit, start_time, rng):
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        conv = conv_fft(h)
        idx = np.argmax(conv)
        c = conv[idx] * (2.0 / n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        shift = idx - (n - 1)
        g = np.ones(n)
        if shift == 0:
            g -= 2 * h
        elif shift > 0:
            k = shift
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
        else:
            k = -shift
            g[k:] -= h[:n - k]
            g[:n - k] -= h[k:]
        step = step0 / (1 + it * 0.01)
        h = h - step * g
        h = project_simple(h, target)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n0 = 360
    target0 = n0 / 2.0

    # ----- initial pool -----
    cand = []
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = enforce_sum(h0, target0, rng)
        cand.append(h0)
    else:
        cand.append(random_feasible_vector(n0, rng))

    xs = (np.arange(n0) + 0.5) * (2.0 / n0)
    tri = np.where(xs <= 1, xs, 2 - xs)
    cand.append(enforce_sum(tri, target0, rng))
    cos = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    cand.append(enforce_sum(cos, target0, rng))
    cand.append(enforce_sum(xs / 2.0, target0, rng))
    cand.append(enforce_sum(1 - xs / 2.0, target0, rng))
    for f in (1, 2, 3):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        cand.append(enforce_sum(s, target0, rng))
        cand.append(enforce_sum(c, target0, rng))
    for _ in range(1200):
        cand.append(random_feasible_vector(n0, rng))

    best_h, best_c = None, np.inf
    for h in cand:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- local search on n0 -----
    while time.time() - start < budget_s * 0.5:
        start_vec = perturb_h(best_h, rng, scale=0.4) if rng.random() < 0.8 else random_feasible_vector(n0, rng)
        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=2500, init_step=0.5, n_trials=8)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
        h_sa, c_sa = simulated_annealing(best_h, rng, max_iter=12000,
                                         init_temp=1e-2, final_temp=1e-5,
                                         init_scale=0.5, final_scale=0.02)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        for beta in (50, 200, 800, 2000, 5000, 8000):
            if time.time() - start > budget_s * 0.5:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=600)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # ----- upsample to n=720 -----
    if time.time() - start < budget_s * 0.8:
        n1 = 720
        target1 = n1 / 2.0
        old_x = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
        new_x = (np.arange(n1) + 0.5) * (2.0 / n1)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = project_simple(up_h, target1)
        time_left = budget_s - (time.time() - start)
        up_h, up_c = subgradient_descent(up_h, target1, max_iters=20000,
                                         step0=0.3, time_limit=time_left,
                                         start_time=start, rng=rng)
        if up_c < best_c:
            best_c, best_h = up_c, up_h.copy()
        h_hc, c_hc = hill_climb_multi(best_h, rng, max_iters=2000, init_step=0.4, n_trials=6)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
        h_sa, c_sa = simulated_annealing(best_h, rng, max_iter=8000,
                                         init_temp=5e-3, final_temp=1e-5,
                                         init_scale=0.4, final_scale=0.02)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        for beta in (200, 800, 2000, 5000, 8000, 12000):
            if time.time() - start > budget_s:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=500)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

        # optional second upsample to n=960
        if time.time() - start < budget_s * 0.95:
            n2 = 960
            target2 = n2 / 2.0
            cur_n = best_h.size
            old_x = (np.arange(cur_n) + 0.5) * (2.0 / cur_n)
            new_x = (np.arange(n2) + 0.5) * (2.0 / n2)
            up_h2 = np.interp(new_x, old_x, best_h)
            up_h2 = project_simple(up_h2, target2)
            time_left = budget_s - (time.time() - start)
            up_h2, up_c2 = subgradient_descent(up_h2, target2, max_iters=15000,
                                               step0=0.25, time_limit=time_left,
                                               start_time=start, rng=rng)
            if up_c2 < best_c:
                best_c, best_h = up_c2, up_h2.copy()
            for beta in (500, 2000, 8000, 20000):
                if time.time() - start > budget_s:
                    break
                h_ref = slsqp_optimize(best_h, beta, max_iter=400)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # ----- final cleanup -----
    final_n = best_h.size
    final_target = final_n / 2.0
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, final_target, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(final_n)
