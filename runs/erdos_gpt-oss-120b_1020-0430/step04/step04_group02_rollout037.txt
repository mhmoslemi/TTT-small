        # compute autocorrelation
        corr = np.correlate(h, h, mode='full')
        # find shift index with minimal autocorrelation (excluding trivial shift maybe? but shift 0 is max indeed, but we need minimal)

        k = idx - (n - 1)
        # compute gradient
        g = np.zeros_like(h)
        if k >= 0:
            g[:n-k] += h[k:]  # h[i+k] for i=0..n-k-1
            g[k:] += h[:n-k]   # h[i-k] for i=k..n-1

            k_abs = -k

            g[:n-k_abs] += h[k_abs:]  # h[i-k]
        # step size schedule: step = step0 * (1 - t / iters) maybe.
        step = step0 * (1 - t / iters)
        if step <= 0:
            break
        h_new = h + step * g
        # project
        h = project_box_sum(h_new, n/2.0)
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def hill_climb_targeted(h_start, rng, max_iters=800, init_step=0.5, n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.shape[0]
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# --------- gradient descent on worst shift ----------
def gradient_descent_worst(h0, rng, max_iters=300, init_step=0.3):
    h = h0.copy()
    n = h.size
    best_c = compute_c5(h)
    step = init_step
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        k = idx - (n - 1)

        if k >= 0:
            fwd = np.concatenate([h[k:], np.zeros(k)])
            bwd = np.concatenate([np.zeros(k), h[: n - k]])
        else:
            kabs = -k
            fwd = np.concatenate([np.zeros(kabs), h[: n - kabs]])
            bwd = np.concatenate([h[kabs:], np.zeros(kabs)])
        grad = 1.0 - (fwd + bwd)

        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_box_sum(h_new, n / 2.0)

        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            best_c, h = c_new, h_new
            step = min(step * 1.05, init_step)
        else:
            step *= 0.7
        if step < 1e-6:
            break
    return h, best_c


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

            h_up = project_box_sum(np.clip(h_up, 0.0, 1.0), up_n / 2.0)

            h_gd, c_gd = gradient_descent_worst(h_up, rng, max_iters=400, init_step=0.3)
            h_hc, c_hc = hill_climb_targeted(h_gd, rng, max_iters=800, init_step=0.3)

            best_c, best_h, n = c_hc, h_hc.copy(), up_n

            for beta in beta_list:
                if time.time() - start_time > budget_s:
                    break
                try:
                    h_ref = slsqp_optimize(best_h, beta, max_iter=400)
                except Exception:
                    continue
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
