def shift_array(arr, shift):
    if shift > 0:
        # arr[i - shift] = arr shifted right by shift, where out-of-bounds become 0
        return np.concatenate([np.zeros(shift), arr[:n-shift]])
    elif shift < 0:

        return np.concatenate([arr[s:], np.zeros(s)])

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Uniformly random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):

def perturb_h(h, rng, scale=0.2):

def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
