def subgradient_descent(h0, rng, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    target_sum = h.size / 2
    start = time.time()
    initial_alpha = 0.5
    alpha = initial_alpha
    decay = 0.9995
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        max_idx = int(np.argmax(corr))
        shift = max_idx - (h.size - 1)
        # compute subgradient for this shift
        grad = np.zeros_like(h)
        # shift positive
        if shift >= 0:
            # for i in 0..n-shift-1: contribution
            # g_i = 1 - h_{i+shift} - h_{i-shift} (the latter out-of-range if i-shift<0)
            # we can compute with vectorized approach:
            # first part: 1 - h_{i+shift}
            grad[:h.size - shift] = 1.0 - h[shift:]
            # second part: - h_{i-shift} where i-shift >=0
            grad[shift:] -= h[:h.size - shift]
        else:
            # shift negative: similar using -shift
            s = -shift
            grad[s:] = 1.0 - h[:h.size - s]
            grad[:h.size - s] -= h[s:]
        # Because subgradient for max shift maybe repeated if multiple indices achieve max; but we just pick one.
        # line search
        improved = False
        trial_alpha = alpha
        for _ in range(10):
            h_new = h - trial_alpha * grad
            # project to [0,1]
            h_new = np.clip(h_new, 0.0, 1.0)
            # enforce sum
            h_new = enforce_sum(h_new, target_sum, rng)
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12: # improvement
                h = h_new
                best_h = h.copy()
                best_c = c_new
                improved = True
                break
            else:
                trial_alpha *= 0.5
        if not improved:
            # no improvement, reduce alpha
            alpha *= decay
        else:
            # maybe increase alpha slightly
            alpha = min(alpha*1.05, 1.0)
    return best_h, best_c

Define enforce_sum(h, target, rng, max_iter=1000). Use vectorized algorithm:

def enforce_sum(h, target, rng, max_iter=500):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # capacity = 1 - h
            cap = 1.0 - h
        else:
            cap = -h
        total_cap = cap.sum()
        if total_cap == 0:
            # cannot adjust further, break
            break
        # distribute proportionally
        delta = diff * (cap / total_cap)
        h += delta
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # final small random fix if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(10):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def enforce_sum(h, target, rng, max_iter=200):
    """Project h onto [0,1]^n with Σh = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            cap = 1.0 - h
        else:
            cap = -h
        total = cap.sum()
        if total == 0:
            break
        delta = diff * (cap / total)
        h += delta
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # final tiny fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(10):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-14:
                break
    return h


def random_feasible_vector(n, rng):
    """Generate a random h with Σh = n/2, 0≤h≤1."""
    target = n / 2.0
    h = rng.random(n)
    h = h / h.sum() * target
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a random amount ε from one coordinate to another."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def softmax_obj(x):

    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def pair_transfer_optimize(
    h0, rng, max_iters=5000, time_limit=30.0, init_factor=0.5, decay=0.999
):
    """Iteratively move mass from i to i+shift for the worst shift."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        # index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)
    final_c = compute_c5(best_h)
    return best_h, final_c

def aggregated_subgradient(h, corr, max_shifts=3):
    """Sum sub‑gradients for the top few shifts (averaged)."""
    n = h.size
    idxs = np.argsort(-corr)[:max_shifts]
    grad = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)
        if shift >= 0:
            s = shift
            grad[: n - s] += 1.0 - h[s:]
            grad[s:] -= h[: n - s]
        else:
            s = -shift
            grad[s:] += 1.0 - h[: n - s]
            grad[: n - s] -= h[s:]
    if len(idxs) > 0:
        grad /= len(idxs)
    return grad

    min_alpha = 1e-6
    max_shifts = 3

        corr = np.correlate(h, 1.0 - h, mode="full")

        improved = False
        for _ in range(10):
            h_new = h - trial_alpha * grad
            h_new = np.clip(h_new, 0.0, 1.0)

            if c_new < best_c - 1e-12:

                improved = True
                break
            trial_alpha *= 0.5
        if not improved:
            alpha *= decay
        else:
            alpha = min(alpha * 1.05, 0.5)
    return best_h, best_c

    # ----- initial h -----
    if "initial_h_values" in globals():
        init_vals = np.array(initial_h_values, dtype=float)
        n0 = init_vals.size
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs_new = (np.arange(n) + 0.5) * (2.0 / n)
        h0 = np.interp(xs_new, xs0, init_vals)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, target_sum, rng)
    else:
        h0 = random_feasible_vector(n, rng)

    # ----- candidate pool -----
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    for f in range(1, 6):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            h = np.clip(h, 0.0, 1.0)
            candidates.append(enforce_sum(h, target_sum, rng))

    candidates.append(enforce_sum(xs / 2.0, target_sum, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, target_sum, rng))

    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, target_sum, rng))

    mask = np.zeros(n)
    start_idx = n // 4
    mask[start_idx : start_idx + n // 2] = 1.0
    candidates.append(enforce_sum(mask, target_sum, rng))

    mask = np.arange(n) % 2
    candidates.append(enforce_sum(mask, target_sum, rng))

    for _ in range(4):
        coeff_sin = rng.normal(size=3)
        coeff_cos = rng.normal(size=3)
        freqs = np.array([1, 2, 3])
        h = 0.5 + np.sum(
            coeff_sin[:, None] * np.sin(2 * np.pi * freqs[:, None] * xs), axis=0
        ) + np.sum(
            coeff_cos[:, None] * np.cos(2 * np.pi * freqs[:, None] * xs), axis=0
        )
        h = np.clip(h, 0.0, 1.0)
        candidates.append(enforce_sum(h, target_sum, rng))

    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    # ----- pick best start -----
    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    global_best_h, global_best_c = best_h, best_c

    # ----- main optimisation phase -----
    main_end = start_time + budget_s * 0.55
    while time.time() < main_end:
        # choose start vector
        if rng.random() < 0.7:
            start_vec = perturb_h(global_best_h, rng, scale=0.35)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill climbing
        try:
            h_hc, c_hc = hill_climb_multi(
                start_vec,
                rng,
                max_iters=2500,
                init_step=0.45,
                n_trials=8,
            )
            if c_hc < global_best_c:
                global_best_h, global_best_c = h_hc, c_hc
        except Exception:
            pass

        # pair‑transfer refinement
        try:
            h_pt, c_pt = pair_transfer_optimize(
                global_best_h,
                rng,
                max_iters=8000,
                time_limit=12.0,
                init_factor=0.5,
            )
            if c_pt < global_best_c:
                global_best_h, global_best_c = h_pt, c_pt
        except Exception:
            pass

        # SLSQP ladder
        for beta in [20, 80, 200, 500, 1000, 2000]:
            if time.time() >= main_end:
                break
            try:
                h_ref = slsqp_optimize(global_best_h, beta, max_iter=300)
                c_ref = compute_c5(h_ref)
                if c_ref < global_best_c:
                    global_best_h, global_best_c = h_ref, c_ref
            except Exception:
                continue

    # ----- final sub‑gradient phase -----
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        try:
            h_sub, c_sub = subgradient_optimize(
                global_best_h, rng, time_limit=remaining
            )
            if c_sub < global_best_c:
                global_best_h, global_best_c = h_sub, c_sub
        except Exception:
            pass

    # ----- clean‑up and return -----
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, target_sum, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
