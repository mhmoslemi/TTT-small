def project_to_capped_simplex(v, target):
    # returns w = clamp(v+lam,0,1) sum = target
    # uses binary search.
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = v + lam
        np.clip(w, 0.0, 1.0, out=w)
        s = w.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    w = v + lam
    np.clip(w, 0.0, 1.0, out=w)
    return w

def gradient_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    # compute softmax weights
    # To avoid overflow: subtract max.
    max_val = np.max(beta * corr)
    exps = np.exp(beta * corr - max_val)
    Z = exps.sum()
    p = exps / Z  # shape (2n-1,)
    # compute convolution of p and (1 - h)
    g = 1 - h
    # note: correlate(p, g) yields length 2n-1 + n -1 = 3n-2
    corr_pg = np.correlate(p, g, mode='full')
    # get central n entries (shifts 0..n-1)
    grad = corr_pg[n-1:2*n-1] - p[n-1] * h
    return grad

def pgd_smooth(h0, beta, max_iter, step0, time_limit, start_time):
    h = h0.copy()
    n = h.size
    target = n / 2.0

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection onto [0,1] with a fixed sum ----------
def project_to_capped_simplex(v, target):
    """Euclidean projection of v onto the set {0≤x≤1, sum x = target}."""
    lo = -np.max(v) - 1.0      # all entries become 0 -> sum < target
    hi = 1.0 - np.min(v) + 1.0  # all entries become 1 -> sum > target
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = v + lam
        np.clip(w, 0.0, 1.0, out=w)
        s = w.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    w = v + lam
    np.clip(w, 0.0, 1.0, out=w)
    return w

# ---------- basic utilities ----------
def random_feasible_vector(n, rng):
    return project_to_capped_simplex(rng.random(n), n / 2.0)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    """Move full epsilon from a positive‑gradient index to a negative‑gradient index."""
    for _ in range(max_steps):
        s, _, cur_corr = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half transfer
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- smooth‑max gradient ----------
def gradient_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    # stable softmax weights
    exps = np.exp(beta * (corr - max_corr))
    Z = exps.sum()
    p = exps / Z                     # length 2*n-1, shift s = idx-(n-1)
    g = 1 - h
    corr_pg = np.correlate(p, g, mode="full")   # length 3*n-2
    grad = corr_pg[n - 1 : 2 * n - 1] - p[n - 1] * h
    return grad

# ---------- projected gradient descent with continuation ----------
def pgd_multi_beta(h0, betas, iters, step0, time_limit, start_time):
    """Run projected GD for a sequence of betas (continuation)."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 1
    for beta, it in zip(betas, iters):
        for _ in range(it):
            if time.time() > time_limit:
                return best_h, best_c
            grad = gradient_softmax(h, beta)
            step = step0 / np.sqrt(t)
            h = h - step * grad
            h = project_to_capped_simplex(h, target)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            t += 1
    return best_h, best_c

# ---------- SLSQP polishing ----------
def slsqp_optimize(h0, beta, max_iter=2000):
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def soft_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        max_corr = np.max(corr)
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        soft_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_to_capped_simplex(x, target)

        h_loc, c_loc = pgd_multi_beta(
            start_vec,
            betas,
            iters_per_beta,
            step0=0.8,
            time_limit=t1_end,
            start_time=start,
        )
        if c_loc < best_c:
            best_h, best_c = h_loc, c_loc

    # ----- stage 2: greedy refinement -----
    t2_end = start + budget_s * 0.75
    if time.time() < t2_end:
        h_g = greedy_optimize(best_h, max_steps=4000)
        c_g = compute_c5(h_g)
        if c_g < best_c:
            best_h, best_c = h_g, c_g

    # ----- stage 3: SLSQP polishing with larger betas -----
    t3_end = start + budget_s * 0.95
    if time.time() < t3_end:
        for beta in (8000, 15000, 25000):
            if time.time() > t3_end:
                break
            try:
                refined = slsqp_optimize(best_h, beta, max_iter=2500)
            except Exception:
                continue
            c_ref = compute_c5(refined)
            if c_ref < best_c:
                best_h, best_c = refined, c_ref

    # ----- stage 4: final micro‑search -----
    t4_end = start + budget_s * 0.99
    while time.time() < t4_end:
        cand = perturb_random(best_h, rng, scale=0.01)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # ----- clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_capped_simplex(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
