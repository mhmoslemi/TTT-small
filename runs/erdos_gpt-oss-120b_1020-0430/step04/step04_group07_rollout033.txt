def gradient_refine(h, rng, step=0.015, max_iter=10):
    n = len(h)
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode='full')
        idx = np.argmax(corr) - (n-1)
        k = abs(idx)
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros_like(h)
            g[: n - k] = 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        # enforce sum by random redistribution
        diff = n/2.0 - h.sum()

                    break

                add = min(diff, 1.0 - h[i]) if diff>0 else max(diff, -h[i])
                h[i] += add
                diff = n/2.0 - h.sum()
    return h

def upsample_vector(v, new_n):
    old_n = len(v)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, v)
    return h_new

def random_perturb_small(h, rng, scale=0.02):
    # choose i,j random and move small delta
    n = len(h)
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i == j:
        return h
    max_delta = min(h[i], 1.0 - h[j])
    delta = rng.random() * max_delta * scale
    nh = h.copy()
    nh[i] -= delta
    nh[j] += delta
    return nh

            h_up = slsqp_optimize(h_up, beta, rng, max_iter=500)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, up_n
        h_up = gradient_refine(h_up, rng, step=0.008, max_iter=8)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, up_n

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
