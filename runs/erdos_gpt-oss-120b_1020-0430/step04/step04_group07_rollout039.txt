import time
import numpy as np
from scipy.optimize import minimize

# globals for SLSQP soft‑max
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Force Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    # simple redistribution loop (few hundred iterations are enough)
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            add = diff / idx.size
            max_add = 1.0 - h[idx]
            actual = np.minimum(add, max_add)
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            add = diff / idx.size
            max_sub = -h[idx]
            actual = np.maximum(add, max_sub)
        h[idx] += actual
        diff = target - h.sum()
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k with a penalty on the sum."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr * (2.0 / x.size)
    m = np.max(bc)
    # log‑sum‑exp (stabilised)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET)
    return x


def gradient_descent_softmax(h: np.ndarray, target: float, dx: float,
                            rng: np.random.Generator,
                            start_time: float, budget_s: float) -> np.ndarray:
    """Projected gradient descent on the soft‑max of the overlap."""
    n = h.size
    beta_sched = [10.0, 50.0, 200.0, 800.0, 3000.0, 8000.0, 20000.0]
    for beta in beta_sched:
        step = 0.5 / np.sqrt(beta)
        for _ in range(200):
            if time.time() - start_time > budget_s * 0.98:
                return h
            corr = np.correlate(h, 1 - h, mode="full")
            O = corr * dx
            bc = beta * O
            max_bc = np.max(bc)
            exp_bc = np.exp(bc - max_bc)
            w = exp_bc / np.sum(exp_bc)          # soft‑max weights, length 2n‑1
            grad = np.zeros(n)
            # loop over all shifts
            for idx_shift, weight in enumerate(w):
                if weight == 0.0:
                    continue
                s = idx_shift - (n - 1)          # real shift value
                if s >= 0:
                    i_start, i_end = s, n
                else:
                    i_start, i_end = 0, n + s
                if i_end <= i_start:
                    continue
                i = np.arange(i_start, i_end)
                j = i - s
                wdx = weight * dx
                grad[i] += wdx * (1.0 - h[j])
                grad[j] += wdx * (-h[i])
            h = h - step * grad
            h = np.clip(h, 0.0, 1.0)
            h = enforce_sum(h, target)

def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Balanced binary vector (0/1) of length n."""
    h = np.zeros(n)

    h[ones] = 1.0
    return h

        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nh = enforce_sum(nh, best_n / 2.0)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh, nc

    # ----- final sanity check -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
