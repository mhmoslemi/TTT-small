def grad_softmax(h, beta):
    n = h.size
    N = 2 * n  # power of 2 maybe but fine.
    # compute correlation C = h ⋆ (1-h)
    fh = np.fft.rfft(h, N)
    fg = np.fft.rfft(1 - h, N)
    C = np.fft.irfft(fh * np.conj(fg), N)[:2*n-1]  # length 2n-1
    # compute softmax weights
    # Use offset so that shift k corresponds to index idx=k + n-1
    maxC = np.max(C)
    expC = np.exp(beta * (C - maxC))
    p = expC / np.sum(expC)
    # p array length 2n-1; split
    p_full = p
    # p_pos for k>=0:
    p_pos = p_full[n-1:]  # length n
    # p_neg for k<0:
    p_neg = p_full[:n-1]   # length n-1; corresponds to k = -(n-1)..-1, but we need q_k = p_{-k} for k=1..n-1: q = p_neg reversed?
    # For k>0, p_{-k} corresponds to p_full[n-1 - k] = p_neg[n-1 - k]? Actually p_neg[ (n-1) - k ] = p_{-k}. Since p_neg is in order from -(n-1) to -1. So q_k = p_neg[n-1 - k] = p_neg_rev[k-1]? Simpler: q = p_neg[::-1] gives array q indexed 0..n-2 where q[0]=p_{-1}, q[1]=p_{-2}, etc. But we need p_{-k} with k=1..n-1. So q = p_neg[::-1] satisfies q[k-1] = p_{-k}. So q = p_neg[::-1].
    q = p_neg[::-1]  # length n-1
    # compute term1 via correlation of p_pos and (1 - h)
    # we can compute via FFT to get length n + n -1; we need first n values (i from 0..n-1)
    f1 = np.fft.rfft(p_pos, N)
    f2 = np.fft.rfft(1 - h, N)
    term1_full = np.fft.irfft(f1 * np.conj(f2), N)[:2*n-1]
    term1 = term1_full[:n]  # first n entries correspond to i shift
    # compute term2 via convolution of h with q
    f_h = np.fft.rfft(h, N)
    f_q = np.fft.rfft(q, N)
    conv_full = np.fft.irfft(f_h * f_q, N)[:2*n-1]
    # conv index i-1 corresponds to term2_i for i>=1, term2_0 = 0
    term2 = np.empty(n, dtype=float)
    term2[0] = 0.0
    term2[1:] = conv_full[:n-1]  # conv_full[i-1] for i>=1
    grad = term1 - term2
    # note gradient of softmax obj = grad
    return grad

def gradient_descent_softmax(h0, beta, lr=0.5, max_iter=2000, tol=1e-12):
    h = h0.copy()
    n = h.size
    target_sum = n/2
    for it in range(max_iter):
        grad = grad_softmax(h, beta)
        # gradient descent step
        h_new = h - lr * grad
        # clip to [0,1]
        h_new = np.clip(h_new, 0.0, 1.0)
        # enforce sum
        # compute current sum and diff
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-12:
            # distribute diff uniformly to non-saturated entries
            # find indices where not at bounds
            mask = (h_new > 0.0) & (h_new < 1.0)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                adjust = diff / cnt
                # apply adjust, then clip again and maybe repeat
                h_new[mask] += adjust
                h_new = np.clip(h_new, 0.0, 1.0)
            else:
                # all at bounds, just adjust globally then clip
                h_new += diff / n
                h_new = np.clip(h_new, 0.0, 1.0)
        # Check convergence
        change = np.max(np.abs(h_new - h))
        h = h_new
        if change < tol:
            break
    # final enforce sum
    h = enforce_sum(h, target_sum, np.random.default_rng())
    return h

def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] then adjust to hit exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with exactly sum=n/2."""

    j = rng.integers(0, n)

    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Exact C5 for a discretised step function using FFT."""
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)[: 2 * n - 1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)


def max_overlap_and_shift(h):
    """Return (max_corr, shift) for h (full correlation with 1-h)."""
    n = h.size
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)[: 2 * n - 1]
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (n - 1)
    return max_val, shift

def pair_transfer_optimize(
    h0,
    rng,
    max_iters=5000,
    time_limit=30.0,
    init_factor=0.5,
    decay=0.999,
):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c

def grad_softmax(h, beta):
    """Gradient of (1/β) log‑sum‑exp of the overlap values."""
    n = h.size
    N = 2 * n
    # correlation C = h ⋆ (1‑h)
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    C = np.fft.irfft(H * np.conj(G), N)[: 2 * n - 1]

    maxC = np.max(C)
    expC = np.exp(beta * (C - maxC))
    p = expC / np.sum(expC)  # weights for each shift

    # split weights
    p_pos = p[n - 1 :]          # k = 0 … n‑1
    p_neg = p[: n - 1]          # k = -(n‑1) … -1
    q = p_neg[::-1]             # q[k‑1] = p_{‑k}, length n‑1

    # term1 = Σ_k p_k (1‑h_{i+k})  for k≥0
    f1 = np.fft.rfft(p_pos, N)
    f2 = np.fft.rfft(1.0 - h, N)
    term1_full = np.fft.irfft(f1 * np.conj(f2), N)[: 2 * n - 1]
    term1 = term1_full[:n]

    # term2 = Σ_{k=1}^{i} p_{‑k} h_{i‑k}
    f_h = np.fft.rfft(h, N)
    f_q = np.fft.rfft(q, N)
    conv_full = np.fft.irfft(f_h * f_q, N)[: 2 * n - 1]
    term2 = np.empty(n, dtype=h.dtype)
    term2[0] = 0.0
    term2[1:] = conv_full[: n - 1]

    return term1 - term2


def gradient_descent_softmax(
    h0, beta, lr, max_iter, target_sum, rng, tol=1e-12, max_iter_allow=5000
):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    n = h.size
    for it in range(min(max_iter, max_iter_allow)):
        grad = grad_softmax(h, beta)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)

        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-12:
            mask = (h_new > 0.0) & (h_new < 1.0)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_new[mask] += diff / cnt
                h_new = np.clip(h_new, 0.0, 1.0)
            else:
                h_new += diff / n
                h_new = np.clip(h_new, 0.0, 1.0)

        if np.max(np.abs(h_new - h)) < tol:
            h = h_new
            break
        h = h_new

    h = enforce_sum(h, target_sum, rng)
    return h
