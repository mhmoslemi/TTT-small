```python
def perturb_h(h, rng, scale=0.2):
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def softmax_obj(x, beta):
    # compute correlation using FFT for speed
    n = x.shape[0]
    # compute h * rev convolution
    rev = (1 - x)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    # compute smooth max: (1/beta) * log(sum(exp(beta*conv)))
    # subtract max for stability
    maxc = np.max(conv)
    # To avoid overflow
    z = np.exp(beta * (conv - maxc))
    return (maxc + np.log(z.sum()) / beta)

def directed_perturb(h, rng, scale=0.2):
    n = h.size
    # find worst shift
    corr = np.correlate(h, 1 - h, mode='full')
    k = np.argmax(corr)
    s = k - (n - 1)
    # compute contributions
    idx = np.arange(n)
    contrib = np.zeros(n)
    # (1 - h[i+s]) term
    j = idx + s
    mask = (j >= 0) & (j < n)
    contrib[mask] += (1 - h[j[mask]])
    # -h[i - s] term
    j2 = idx - s
    mask2 = (j2 >= 0) & (j2 < n)
    contrib[mask2] -= h[j2[mask2]]
    # Choose i where contrib > 0 to decrease, j where contrib < 0 to increase
    pos = np.where(contrib > 0)[0]
    neg = np.where(contrib < 0)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = min(max_eps, scale * max_eps)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def directed_perturb(h, rng, scale=0.2):
    """Perturb guided by the currently worst shift."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    k = np.argmax(corr)
    s = k - (n - 1)                     # shift corresponding to that index
    idx = np.arange(n)
    contrib = np.zeros(n)
    # (1 - h[i+s]) term
    ip = idx + s
    mask = (ip >= 0) & (ip < n)
    contrib[mask] += (1.0 - h[ip[mask]])
    # -h[i-s] term
    im = idx - s
    mask2 = (im >= 0) & (im < n)
    contrib[mask2] -= h[im[mask2]]
    pos = np.where(contrib > 0)[0]
    neg = np.where(contrib < 0)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = rng.choice(pos)
    j = rng.choice(neg)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = min(max_eps, scale * max_eps)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_multi(h_start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    """Simple stochastic hill‑climb with occasional directed moves."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            if rng.random() < 0.1:
                new_h = directed_perturb(h, rng, scale=step)
            else:
                new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def simulated_annealing(h0, rng, max_iter, init_temp, final_temp,
                        init_scale=0.5, final_scale=0.05):
    """Simulated‑annealing refinement."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t if init_temp > 0 else 0.0
        scale = init_scale * (final_scale / init_scale) ** t if init_scale > 0 else 0.0
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def softmax_obj(x, beta):
    """Smooth approximation of max using log‑sum‑exp."""
    n = x.shape[0]
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return logsumexp(beta * conv) / beta


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP refinement of the smooth surrogate."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        softmax_obj,
        h0,
        args=(beta,),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, np.random.default_rng())
    return x

# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with a low Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Base dimension and initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ------------------------------------------------------------------
    # Build a modest pool of candidates
    # ------------------------------------------------------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    # Triangular bump
    tri = np.where(xs <= 1, xs, 2 - xs)
    candidates.append(enforce_sum(tri, n / 2.0, rng))

    # Cosine bump
    cosine = (1 + np.cos(np.pi * (xs - 1))) / 2.0
    candidates.append(enforce_sum(cosine, n / 2.0, rng))

    # Linear ramps
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1 - xs / 2.0, n / 2.0, rng))

    # Sine / cosine phases
    for f in (1, 2, 3):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        candidates.append(enforce_sum(s, n / 2.0, rng))
        candidates.append(enforce_sum(c, n / 2.0, rng))

    # Random vectors
    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # ------------------------------------------------------------------
    # Pick the best of the pool
    # ------------------------------------------------------------------
    best_h = None
    best_c = np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop (hill‑climb → SA → SLSQP)
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 15.0:
            break

        # Starting point
        if rng.random() < 0.6:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)

        # Hill‑climb
        h_hc, c_hc = hill_climb_multi(
            start_vec, rng, max_iters=3000, init_step=0.5, n_trials=8
        )
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

        # Simulated annealing
        h_sa, c_sa = simulated_annealing(
            best_h,
            rng,
            max_iter=10000,
            init_temp=1e-2,
            final_temp=1e-5,
            init_scale=0.5,
            final_scale=0.02,
        )
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()

        # SLSQP ladder (smooth surrogate)
        for beta in (20, 80, 320, 1280, 5120):
            if time.time() - start_time > budget_s * 0.85:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
            except Exception:
                continue

    # ------------------------------------------------------------------
    # Optional up‑sampling if time permits
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > budget_s * 0.1:
        target_n = max(720, n * 2)
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)

            # Refine on finer grid
            up_h, _ = hill_climb_multi(
                up_h, rng, max_iters=2500, init_step=0.4, n_trials=6
            )
            up_h, _ = simulated_annealing(
                up_h,
                rng,
                max_iter=8000,
                init_temp=5e-3,
                final_temp=1e-5,
                init_scale=0.4,
                final_scale=0.02,
            )
            for beta in (200, 800, 3200, 12800):
                if time.time() - start_time > budget_s:
                    break
                up_h = slsqp_optimize(up_h, beta, max_iter=300)
                c_up = compute_c5(up_h)
                if c_up < best_c:
                    best_c, best_h, n = c_up, up_h.copy(), target_n

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
