import time
import numpy as np
from scipy.optimize import minimize

# Global placeholders for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate for max_k correlation."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp in a numerically stable way
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 (max overlap) for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]                     # reversed (1‑h)
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())        # next power of two >= L
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)          # multiply by dx


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target while preserving 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # simple coordinate‑wise fixes
    for _ in range(max_iter):
        if abs(diff) < 1e-13:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tiny correction (if needed)
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h0, beta, max_iter=300):
    """Perform SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-13:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def guide_refine(h, steps, rng):
    """Local adjustment guided by the current worst shift."""
    n = h.size
    best_c = compute_c5(h)
    for _ in range(steps):
        # full correlation (linear, zero‑padded)
        corr_full = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr_full)
        k = idx - (n - 1)                     # shift corresponding to max overlap

        # gradient magnitude for this k (ignoring dx factor)
        grad = np.zeros(n)
        if k >= 0:
            end = n - k
            grad[:end] += h[k:]               # contribution from h[i+k]
            grad[k:] += h[:end]               # contribution from h[i‑k]
        else:  # negative shift, mirror the logic
            kp = -k
            end = n - kp
            grad[:end] += h[kp:]
            grad[kp:] += h[:end]

        inc = np.where(h < 1 - 1e-12)[0]
        dec = np.where(h > 1e-12)[0]
        if inc.size == 0 or dec.size == 0:
            break
        i_max = inc[np.argmax(grad[inc])]
        i_min = dec[np.argmin(grad[dec])]
        eps = min(0.01, 1 - h[i_max], h[i_min])
        if eps <= 0:
            break
        new_h = h.copy()
        new_h[i_max] += eps
        new_h[i_min] -= eps
        c_new = compute_c5(new_h)
        if c_new < best_c - 1e-12:
            h, best_c = new_h, c_new
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h on [0,2] with a low Erdős C5 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Base discretisation (midpoints of n intervals)
    # --------------------------------------------------------------
    n = 360                                    # < 1k samples → fast evaluation
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.sin(np.pi * xs / 2.0) ** 2          # analytic candidate (integrates to 1)
    h = enforce_sum(h, n / 2.0, rng)           # enforce the exact integral constraint

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # --------------------------------------------------------------
    # SLSQP continuation (smooth surrogate) – accept only improvements
    # --------------------------------------------------------------
    betas = (50, 200, 800, 2000, 5000, 10000)
    for beta in betas:
        if time.time() - start > budget_s * 0.65:
            break
        cand = slsqp_optimize(best_h, beta, max_iter=500)
        cand_c = compute_c5(cand)
        if cand_c < best_c - 1e-10:
            best_h, best_c = cand, cand_c

    # --------------------------------------------------------------
    # Guided local refinement (focus on the worst shift)
    # --------------------------------------------------------------
    if time.time() - start < budget_s * 0.75:
        best_h, best_c = guide_refine(best_h, steps=2000, rng=rng)

    # --------------------------------------------------------------
    # Optional up‑sampling for a finer discretisation (if time permits)
    # --------------------------------------------------------------
    elapsed = time.time() - start
    if budget_s - elapsed > budget_s * 0.15:
        # double the resolution
        n2 = n * 2
        xs2 = (np.arange(n2) + 0.5) * (2.0 / n2)
        up_h = np.interp(xs2, xs, best_h)
        up_h = enforce_sum(up_h, n2 / 2.0, rng)

        # brief SLSQP ladder on the finer grid
        for beta in (200, 800, 2000, 5000, 10000):
            if time.time() - start > budget_s:
                break
            up_h = slsqp_optimize(up_h, beta, max_iter=400)
        up_c = compute_c5(up_h)

        if up_c < best_c:
            best_h, best_c, n = up_h, up_c, n2

    # --------------------------------------------------------------
    # Final clean‑up (ensure feasibility)
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
