def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * dx
```

def enforce_sum(h, target, rng, max_iter=1000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # Find indices that can accommodate change
        if diff > 0:
            # need to add, find those <1
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            per = diff / idx.size
            # Clip per to max possible per index
            # compute max add per index
            max_add = np.min(1.0 - h[idx])
            per = min(per, max_add)
            h[idx] += per
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            per = diff / idx.size
            max_sub = np.min(h[idx])
            per = max(per, -max_sub)
            h[idx] += per
        diff = target - h.sum()
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # Might still have small diff; adjust by a single element
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # add diff to a random index that can accommodate
        for i in range(n):
            i = rng.integers(0, n)
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff = target - h.sum()
            elif diff < 0 and h[i] > 0.0:
                sub = max(diff, -h[i])
                h[i] += sub
                diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

### perturb_h

```
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def worst_shift_and_grad(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full') * dx
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)  # shift value (could be negative)
    # gradient g = 1 - h[x+shift] - h[x-shift]; treat out-of-bounds as 0
    g = np.ones(n)
    if shift >= 0:
        # x+shift within bounds for x in [0, n-1-shift]
        g[:n-shift] -= h[shift:]
        # x-shift within bounds for x in [shift, n-1]
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        # x+shift = x - s, valid for x >= s
        g[s:] -= h[:n-s]
        # x-shift = x + s, valid for x < n-s
        g[:n-s] -= h[s:]
    return shift, g, conv.max()
```

### softmax_obj

```
def softmax_obj(x, beta):
    # x is array
    n = x.shape[0]
    # compute convolution via FFT
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    # max and log-sum-exp
    bc = beta * conv
    mmax = np.max(bc)
    return (mmax + np.log(np.sum(np.exp(bc - mmax)))) / beta

def slsqp_optimize(h0, beta, max_iter=500):
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x, beta), h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum again
    x = enforce_sum(x, target, np.random.default_rng())
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * dx


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size

    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            per = diff / idx.size
            max_add = np.min(1.0 - h[idx])
            per = min(per, max_add)
            h[idx] += per
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            per = diff / idx.size
            max_sub = np.min(h[idx])
            per = max(per, -max_sub)
            h[idx] += per
        diff = target - h.sum()

    # tiny residual fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(200):
            i = rng.integers(0, n)
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff = target - h.sum()
            elif diff < 0 and h[i] > 0.0:
                sub = max(diff, -h[i])
                h[i] += sub
                diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with Σh = n/2."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def worst_shift_and_grad(h: np.ndarray):
    """Return worst shift index, its gradient, and the corresponding overlap value."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full") * dx
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)            # signed shift
    a = abs(shift)
    if a == 0:
        g = 1.0 - 2.0 * h
    else:
        g = np.ones(n)
        g[: n - a] -= h[a:]
        g[a:] -= h[: n - a]
    return shift, g, conv.max()


def projected_gradient_descent(
    h0: np.ndarray,
    max_iter: int,
    step0: float,
    rng: np.random.Generator,
    target: float,
    best_c: float,
):
    """Gradient descent on the worst‑shift overlap, with projection onto constraints."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_val = best_c

    for it in range(max_iter):
        shift, g, cur_c = worst_shift_and_grad(h)
        if cur_c < best_val:
            best_val = cur_c
            best_h = h.copy()
        # decaying step size
        t = it / max_iter
        step = step0 * (1.0 - t) ** 2 + 1e-5
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
    return best_h, best_val


def softmax_obj(x: np.ndarray, beta: float) -> float:
    """Smooth approximation of max_k correlation using log‑sum‑exp."""
    n = x.shape[0]
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    bc = beta * conv
    mmax = np.max(bc)
    return (mmax + np.log(np.sum(np.exp(bc - mmax)))) / beta


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 600) -> np.ndarray:
    """SLSQP refinement of the smooth surrogate for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        lambda x: softmax_obj(x, beta),
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # small drift correction
    x = enforce_sum(x, target, np.random.default_rng())
    return x
