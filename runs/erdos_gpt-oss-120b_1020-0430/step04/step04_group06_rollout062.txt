import time
import numpy as np

# ---------------------------------------------------------
# Helper functions
# ---------------------------------------------------------

def enforce_sum_simple(h, target, max_iter=10):
    """Project h onto the hyperplane sum(h)=target with 0<=h<=1."""
    diff = target - h.sum()
    n = h.shape[0]
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        delta = diff / n
        h += delta
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
    return h


def compute_c5(h):
    """Exact discrete C5 using FFT based correlation."""
    n = h.shape[0]
    N = 2 * n - 1
    fh = np.fft.fft(h, N)
    f1mh = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(fh * np.conj(f1mh)).real
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)


def gradient_and_corr(h, beta):
    """
    Compute gradient of the smooth surrogate
        L(h) = (1/beta) * log sum exp(beta * f_k(h)),
    where f_k are the overlap values, together with the full correlation f_k.
    Returns (grad, corr) where corr is the array of f_k (length 2n-1).
    """
    n = h.shape[0]
    N = 2 * n - 1

    fh = np.fft.fft(h, N)
    f1mh = np.fft.fft(1.0 - h, N)

    # correlation f_k = Σ h_i (1-h_{i+k})
    corr = np.fft.ifft(fh * np.conj(f1mh)).real

    # softmax weights w_k
    bc = beta * corr
    m = np.max(bc)
    exp_bc = np.exp(bc - m)
    w = exp_bc / exp_bc.sum()

    fw = np.fft.fft(w, N)

    # convolution h * w
    conv_hw = np.fft.ifft(fh * fw).real
    # correlation h with w
    corr_hw = np.fft.ifft(fh * np.conj(fw)).real

    start = n - 1
    # gradient of L: 1 - (h * w) - (h ⋆ w)
    grad = 1.0 - conv_hw[start:start + n] - corr_hw[start:start + n]
    return grad, corr


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1] with sum=n/2."""
    h = rng.random(n)
    h = h / h.sum() * (n / 2.0)
    # clip and fix any small violations
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum_simple(h, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0,
                          init_factor=0.5, decay=0.999):
    """Local search moving mass from i to i+shift to reduce the max overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    while True:
        if time.time() - start > time_limit:
            break
        # current max overlap
        corr = np.fft.ifft(np.fft.fft(h, 2 * n - 1) *
                           np.conj(np.fft.fft(1.0 - h, 2 * n - 1))).real
        idx = int(np.argmax(corr))
        max_val = corr[idx]
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        shift = idx - (n - 1)

        # index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue

        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    # final check
    final_c = compute_c5(best_h)
    return best_h, final_c

# ---------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # -------------------------------------------------
    # Choose discretisation size
    # -------------------------------------------------
    n = 720  # even number, gives target sum n/2 = 360
    target = n / 2.0
    dx = 2.0 / n

    # -------------------------------------------------
    # Initialise h (use provided initial_h_values if possible)
    # -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n:
            xs_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(xs_new, xs_old, h0, period=2.0)
    else:
        h0 = np.full(n, 0.5) + rng.uniform(-0.05, 0.05, n)
        h0 = np.clip(h0, 0.0, 1.0)

    h0 = enforce_sum_simple(h0, target)

    # -------------------------------------------------
    # Gradient‑descent on the smooth surrogate
    # -------------------------------------------------
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    beta_start = 5.0
    beta_max = 2000.0
    beta = beta_start
    beta_inc = 0.01          # linear increase per iteration
    step0 = 0.5
    it = 0

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.85:
            break

        # gradient and current correlation
        grad, corr = gradient_and_corr(h, beta)

        cur_c = np.max(corr) * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        # update beta (linear schedule capped)
        beta = min(beta_max, beta + beta_inc)

        # projected gradient step (preserve sum)
        grad -= grad.mean()
        step = step0 / np.sqrt(it + 1)
        h = h - step * grad
        np.clip(h, 0.0, 1.0, out=h)
        h = enforce_sum_simple(h, target)

        # occasional random jitter to escape flat regions

            h = perturb_h(h, rng, scale=0.1)
            h = enforce_sum_simple(h, target)

    if remaining > 5.0:

            time_limit=remaining,

            best_h = h_opt

    # -------------------------------------------------
    # Clean‑up and return
    # -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum_simple(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
