def propose_move(h, epsilon, rng):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    # choose direction
    delta = rng.random()*epsilon
    if rng.random() < 0.5:
        # i + delta, j - delta
        maxadd = min(delta, 1 - h[i])
        maxsub = min(delta, h[j])
        delta = min(maxadd, maxsub)
        if delta <= 0: return None
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    else:
        # i - delta, j + delta
        maxsub = min(delta, h[i])
        maxadd = min(delta, 1 - h[j])
        delta = min(maxsub, maxadd)
        if delta <= 0: return None
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    return new_h

def simulated_annealing_fractional(h0, rng, deadline):
    n = h0.size
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = init_T
    epsilon = epsilon_init
    it = 0
    while time.time() < deadline:
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        # random delta
        delta = rng.random() * epsilon
        # direction random
        if rng.random() < 0.5:

            d = min(delta, maxadd, maxsub)
            if d <= 0:

            h_new[i] += d
            h_new[j] -= d
        else:
            maxsub = h[i]
            maxadd = 1 - h[j]
            d = min(delta, maxsub, maxadd)
            if d <= 0:

            h_new[j] += d
        new_c = compute_c5(h_new)

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]  # (1 - h[i+k])
            g[k:] -= h[:upper]       # - h[i]
        # step decreasing
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        np.clip(h, 0.0, 1.0, out=h)
        # adjust sum exactly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(100):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def sa_optimize(h, rng, deadline, init_T=0.01, epsilon=0.05, T_decay=0.9995, epsilon_decay=0.9995, max_iter=5000):
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = init_T
    eps = epsilon
    it = 0
    while it < max_iter and time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # choose direction
        if rng.random() < 0.5:
            # i added, j sub
            maxadd = 1 - h[i]
            maxsub = h[j]
            delta = min(eps, maxadd, maxsub)
            if delta <= 0:
                it += 1
                continue
            # apply
            h[i] += delta
            h[j] -= delta
            new_c = compute_c5(h)
            delta_c = new_c - cur_c
            accept = delta_c < 0 or rng.random() < math.exp(-delta_c / T)
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                # revert
                h[i] -= delta
                h[j] += delta
        else:
            # i sub, j add
            maxsub = h[i]
            maxadd = 1 - h[j]
            delta = min(eps, maxsub, maxadd)
            if delta <= 0:
                it += 1
                continue
            h[i] -= delta
            h[j] += delta
            new_c = compute_c5(h)
            delta_c = new_c - cur_c
            accept = delta_c < 0 or rng.random() < math.exp(-delta_c / T)
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] += delta
                h[j] -= delta
        it += 1
        if it % 200 == 0:
            T *= T_decay
            eps *= epsilon_decay
            eps = max(eps, 1e-6)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    # define candidate ns
    candidate_ns = [360, 300, 240, 180]
    # Maybe include 540? But bigger n increase evaluation cost; we can set upsample later.

    for n in candidate_ns:
        if time.time() >= deadline:
            break
        # build start vectors
        start_vecs = []
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            if parent.size != n:
                # interpolate parent to n
                xs = np.linspace(0, 2, parent.size, endpoint=False)
                xt = np.linspace(0, 2, n, endpoint=False)
                interp = np.interp(xt, xs, parent)
            else:
                interp = parent.copy()
            # fractional init
            start_vecs.append(interp)
            # binary init: choose top n/2 values
            idx = np.argpartition(interp, -n//2)[-n//2:]

            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)
        # random binary

        bin_rand[rng.choice(n, n//2, replace=False)] = 1.0
        start_vecs.append(bin_rand)

        np.clip(rand_frac, 0.0, 1.0, out=rand_frac)

import time, math
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- gradient refinement ----------
def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]  # (1 - h[i+k])
            g[k:] -= h[:upper]       # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(100):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- Simulated annealing for fractional h ----------
def sa_optimize(h, rng, deadline, init_T=0.01, epsilon=0.05,
                T_decay=0.9995, eps_decay=0.9995, max_iter=2000):
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = init_T
    eps = epsilon
    it = 0
    while it < max_iter and time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            maxadd = 1.0 - h[i]
            maxsub = h[j]
            delta = min(eps, maxadd, maxsub)
            if delta <= 0:
                it += 1
                continue
            h[i] += delta
            h[j] -= delta
            new_c = compute_c5(h)
            dC = new_c - cur_c
            accept = dC < 0 or rng.random() < math.exp(-dC / T)
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] -= delta
                h[j] += delta
        else:
            maxsub = h[i]
            maxadd = 1.0 - h[j]
            delta = min(eps, maxsub, maxadd)
            if delta <= 0:
                it += 1
                continue
            h[i] -= delta
            h[j] += delta
            new_c = compute_c5(h)
            dC = new_c - cur_c
            accept = dC < 0 or rng.random() < math.exp(-dC / T)
            if accept:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] += delta
                h[j] -= delta
        it += 1
        if it % 200 == 0:
            T *= T_decay
            eps *= eps_decay
            eps = max(eps, 1e-6)
    return best_h, best_c

# ---------- main optimization entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    candidate_ns = [360, 300, 240, 180]

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        starts = []

        # parent based starts
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            if parent.size != n:
                xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                interp = np.interp(xt, xs, parent)
            else:
                interp = parent.copy()
            starts.append(interp)                         # fractional
            idx = np.argpartition(interp, -n // 2)[-n // 2:]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            starts.append(bin_vec)                        # binary

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        rand_frac *= (n / 2.0) / rand_frac.sum()
        np.clip(rand_frac, 0.0, 1.0, out=rand_frac)
        adjust_sum_exact(rand_frac, rng)
        starts.append(rand_frac)

        for base_h in starts:
            if time.time() >= deadline:
                break
            h = np.clip(base_h, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # modest gradient refinement
            h = gradient_refine(h, rng, step=0.02, max_iter=6)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

            # SLSQP cascade
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
                if time.time() >= deadline:
                    break
                try:

                                       max_iter=400 if beta < 200 else 600)
                except Exception:
                    continue
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # short SA refinement (≈2 s max)
            per_dead = min(deadline, time.time() + 2.0)
            if time.time() < deadline:
                h, _ = sa_optimize(h, rng, per_dead,
                                   init_T=0.01, epsilon=0.05,
                                   max_iter=2000)
                c = compute_c5(h)
                if c < best_c:
                    best_h, best_c, best_n = h.copy(), c, n

            # final gradient fine‑tune
            h = gradient_refine(h, rng, step=0.02, max_iter=8)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # ---------- optional up‑sampling ----------
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)

        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=8)
        for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0):
            if time.time() >= deadline:
                break
            try:
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
            except Exception:
                continue
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=8)

        # SA on up‑sampled version if time permits
        if time.time() < deadline:
            per_dead = min(deadline, time.time() + 2.0)
            h_up, _ = sa_optimize(h_up, rng, per_dead,
                                   init_T=0.01, epsilon=0.04,
                                   max_iter=2000)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ---------- fallback ----------
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)

        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
