def anneal_h(h_start, rng, max_iters, init_step, final_step,
             init_temp, final_temp, p_target, start_time, budget_s):
    n = h_start.size
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    curr_c = best_c
    for it in range(max_iters):
        if time.time() - start_time > budget_s:
            break
        t = it / max_iters
        step = init_step * (1 - t) + final_step * t
        # exponential temperature decay: T = init_temp * (final_temp/init_temp) ** t
        if init_temp == 0:
            temp = 0.0
        else:
            temp = init_temp * ((final_temp / init_temp) ** t)
        # choose move
        if rng.random() < p_target:
            # compute worst shift index
            _, idx = compute_c5_and_idx(h)
            k_star = idx - (n - 1)
            cand = targeted_move(h, rng, k_star, scale=step)
        else:

            h = cand
            curr_c = cand_c
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand.copy()
        else:
            # accept with probability exp(-Δ/T)
            if temp > 0 and rng.random() < np.exp(-(cand_c - curr_c) / temp):
                h = cand
                curr_c = cand_c
        # optional periodic re-evaluation of best if we haven't updated for many steps? Not needed.
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    # Use SLSQP
    res = opt.minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

import time
import numpy as np
import scipy.optimize as opt

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) * 0.5, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)

# ---------- evaluation ----------
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx

# ---------- move operators ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)

    low_low = (h[:n - k] < 0.5) & (h[k:] < 0.5)
    high_high = (h[:n - k] > 0.5) & (h[k:] > 0.5)

    if low_low.any() and high_high.any():
        i_low = rng.choice(np.flatnonzero(low_low))
        i_high = rng.choice(np.flatnonzero(high_high))
        max_eps = min(
            0.5 - h[i_low],
            0.5 - h[i_low + k],
            h[i_high] - 0.5,
            h[i_high + k] - 0.5,
        )
        if max_eps > 0:
            eps = rng.random() * max_eps * scale
            new_h = h.copy()
            new_h[i_low] += eps
            new_h[i_low + k] += eps
            new_h[i_high] -= eps
            new_h[i_high + k] -= eps
            return new_h

    high_low = (h[:n - k] > 0.5) & (h[k:] < 0.5)
    if high_low.any():
        i = rng.choice(np.flatnonzero(high_low))
        j = i + k
        max_eps = min(h[i] - 0.5, 0.5 - h[j])
        if max_eps > 0:
            eps = rng.random() * max_eps * scale
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
            return new_h

    low_high = (h[:n - k] < 0.5) & (h[k:] > 0.5)
    if low_high.any():
        i = rng.choice(np.flatnonzero(low_high))
        j = i + k
        max_eps = min(0.5 - h[i], h[j] - 0.5)
        if max_eps > 0:
            eps = rng.random() * max_eps * scale
            new_h = h.copy()
            new_h[i] += eps
            new_h[j] -= eps
            return new_h

    return perturb_h(h, rng, scale)

# ---------- annealing ----------
def anneal_h(h_start, rng, max_iters, init_step, final_step,
             init_temp, final_temp, p_target, start_time, budget_s):
    n = h_start.size
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    for it in range(max_iters):
        if time.time() - start_time > budget_s:
            break
        t = it / max_iters
        step = init_step * (1 - t) + final_step * t
        temp = init_temp * ((final_temp / init_temp) ** t) if init_temp > 0 else 0.0
        if rng.random() < p_target:
            _, idx = compute_c5_and_idx(h)
            k_star = idx - (n - 1)
            cand = targeted_move(h, rng, k_star, scale=step)
        else:
            cand = perturb_h(h, rng, scale=step)
        cand_c = compute_c5(cand)
        if cand_c < cur_c:
            h, cur_c = cand, cand_c
            if cand_c < best_c:
                best_c, best_h = cand_c, cand.copy()
        elif temp > 0 and rng.random() < np.exp(-(cand_c - cur_c) / temp):
            h, cur_c = cand, cand_c
    return best_h, best_c

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n0 / 2.0)
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    best_h, best_c = h0, compute_c5(h0)

    # ---- phase 1 : annealing on base resolution ----
    phase1_budget = budget_s * 0.55
    best_h, best_c = anneal_h(
        best_h,
        rng,
        max_iters=500000,
        init_step=0.5,
        final_step=0.01,
        init_temp=0.02,
        final_temp=1e-6,
        p_target=0.5,
        start_time=start,
        budget_s=phase1_budget,
    )

    # ---- optional up‑sampling ----
    elapsed = time.time() - start
    if elapsed < budget_s * 0.80 and best_h.size < 720:
        n1 = min(720, best_h.size * 2)
        old_x = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
        new_x = (np.arange(n1) + 0.5) * (2.0 / n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = project_box_sum(h_up, n1 / 2.0)
        best_h, best_c = anneal_h(
            h_up,
            rng,
            max_iters=400000,
            init_step=0.3,
            final_step=0.005,
            init_temp=0.01,
            final_temp=1e-6,
            p_target=0.5,
            start_time=start,
            budget_s=budget_s * 0.95,
        )

    # ---- final SLSQP refinement ----
    beta_list = [100, 500, 2000, 10000]
    final_h = best_h.copy()
    final_c = best_c
    for beta in beta_list:
        if time.time() - start > budget_s * 0.99:

        if rc < final_c:

    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = project_box_sum(final_h, final_h.size / 2.0)
    final_c = compute_c5(final_h)

    return final_h.tolist(), float(final_c), int(final_h.size)
