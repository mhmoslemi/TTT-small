def project_onto_capped_simplex(v, target):
    # to numpy array
    h = np.clip(v, 0.0, 1.0)
    # compute current sum and adjust
    # If sum equals target within tol, done
    # else need to adjust with water-filling
    if abs(h.sum() - target) < 1e-12:
        return h
    # Determine direction
    # We'll use while loop to adjust
    # We maintain mask of not at bound
    mask = np.ones_like(h, dtype=bool)
    # initialization: treat h as current point after clipping; maybe some at bound.
    # We'll treat two cases:
    # If sum > target: need to reduce some non-zero >0 values.
    # If sum < target: need to increase some non-1 values.
    # We'll use iterative approach:
    while True:
        current_sum = h.sum()
        if abs(current_sum - target) < 1e-12:
            break
        if current_sum > target:
            # need to reduce total by delta = current_sum - target
            # consider variables >0 (not at lower bound)
            free = (h > 0) & mask
            # compute lambda reduction
            delta = current_sum - target
            # we will subtract lambda from each free variable equally; lambda = delta / count_free
            count_free = free.sum()
            if count_free == 0:
                break
            lam = delta / count_free
            # subtract lam from free entries
            # ensure not drop below 0
            # find those that would drop below 0
            new_vals = h[free] - lam
            # any negative?
            below = new_vals < 0
            if below.any():
                # for those, set to 0, and recompute with updated target
                # compute amount actually subtracted from those: h_i
                subted = h[free][below]
                # set to zero
                idx_free = np.where(free)[0]
                idx_below = idx_free[below]
                h[idx_below] = 0.0
                # update mask: these now fixed at lower bound
                mask[idx_below] = False
                # continue loop
                continue
            else:
                h[free] -= lam
                break
        else:
            # current_sum < target: need to increase
            free = (h < 1) & mask
            delta = target - current_sum
            count_free = free.sum()
            if count_free == 0:
                break
            lam = delta / count_free
            new_vals = h[free] + lam
            above = new_vals > 1
            if above.any():
                idx_free = np.where(free)[0]
                idx_above = idx_free[above]
                # set to 1
                h[idx_above] = 1.0
                mask[idx_above] = False
                continue
            else:
                h[free] += lam
                break
    # After loop, maybe small numeric errors: clip again
    h = np.clip(h, 0.0, 1.0)
    # If sum off small, correct by distributing difference across free entries
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # adjust randomly
        free = (h > 0) & (h < 1)
        if free.any():
            # distribute diff equally among free
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_capped_simplex(v, target):
    v = np.asarray(v, dtype=float)
    # initial lower and upper bounds for lambda
    # Because h_i = clamp(v_i - lambda, 0, 1).
    # For lambda very negative (i.e., -inf), v_i - lambda is huge => clamp to 1 => sum = n.
    # For lambda very positive (inf), clamp to 0 => sum = 0.
    # So we can set lo = -1e6, hi = 1e6.
    lo = -1.0
    hi = 1.0
    # adapt bounds: increase until sum( clamp(v - lo,0,1) ) >= target and sum( clamp(v - hi,0,1) ) <= target
    def sum_clamp(lam):
        # compute sum of clamp(v - lam,0,1)
        # use np.clip(np.subtract(v, lam), 0, 1)
        return np.clip(v - lam, 0.0, 1.0).sum()
    # Expand bounds if needed
    while sum_clamp(lo) < target:
        lo *= 2
    while sum_clamp(hi) > target:
        hi *= 2
    # bisection
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = sum_clamp(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    # final lambda = (lo+hi)/2
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    # due to numeric, adjust diff
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # distribute diff over entries not at bounds
        free = (h > 0) & (h < 1)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

def compute_c5(h):
    """Discrete C5 for vector h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_capped_simplex(v, target):
    """Projection of v onto {0 ≤ x ≤ 1, sum x = target}."""
    v = np.asarray(v, dtype=float)

    def sum_clipped(lam):
        return np.clip(v - lam, 0.0, 1.0).sum()

    # bounds for lambda: lo may be negative (increase sum), hi positive (decrease sum)
    lo, hi = -1.0, 1.0
    while sum_clipped(lo) < target:
        lo *= 2.0
    while sum_clipped(hi) > target:
        hi *= 2.0
    # bisection
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = sum_clipped(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def subgradient_step(h, step, target):
    """One subgradient ascent step for maximising the minimal autocorrelation."""
    n = h.size
    # autocorrelation of h with itself
    corr = np.correlate(h, h, mode="full")
    zero_idx = n - 1
    corr[zero_idx] = np.inf          # ignore shift 0
    min_idx = np.argmin(corr)
    s = min_idx - zero_idx           # worst shift (most negative corr)
    grad = np.zeros_like(h)
    if s >= 0:
        grad[: n - s] += h[s:]
        grad[s:]      += h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:]      += h[: n - s_abs]
        grad[: n - s_abs] += h[s_abs:]
    h_new = h + step * grad
    return project_onto_capped_simplex(h_new, target)


def gradient_shift(h, s):
    """Gradient of the overlap integral for a given shift s."""
    n = h.size
    grad = np.zeros_like(h)
    if s >= 0:
        grad[: n - s] += (1.0 - h[s:])
        grad[s:]      -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:]      += (1.0 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad


def worst_shift_info(h):
    """Return (shift, index, correlation) for the worst (max overlap) shift."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]


def greedy_optimize(h, max_steps=2000):
    """Deterministic improvement via full‑epsilon transfers (as in the reference)."""
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full epsilon
        h_new = h.copy()
        h_new[i] -= max_eps
        h_new[j] += max_eps
        if compute_c5(h_new) + 1e-12 < compute_c5(h):
            h = h_new
            continue
        # try half epsilon

        h_new[j] += eps
        if compute_c5(h_new) + 1e-12 < compute_c5(h):
            h = h_new
            continue
        break
    return h
