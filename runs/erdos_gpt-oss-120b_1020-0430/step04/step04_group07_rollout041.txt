```python
def anneal_binary(h, rng, max_iters, start_temp, end_temp):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    # gather indices of 1s and 0s
    for it in range(max_iters):
        t = start_temp + (end_temp - start_temp) * (it / max_iters)
        # propose a swap
        i1 = rng.choice(np.flatnonzero(cur_h == 1))
        i0 = rng.choice(np.flatnonzero(cur_h == 0))
        # swap
        cur_h[i1] = 0
        cur_h[i0] = 1
        new_c = compute_c5(cur_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            # accept
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i1] = 1
            cur_h[i0] = 0
    return best_h, best_c
```

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def guided_perturb(h, grad, rng, scale=0.5):

    j = int(np.argmin(grad))

    max_eps = min(h[i], 1.0 - h[j])

    nh = h.copy()

    nh[j] += eps

CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact C₅ for a discretised step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x: np.ndarray) -> float:
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector with entries in [0,1] and sum=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1200, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def optimize_h(h_start: np.ndarray, n: int, rng: np.random.Generator,
               beta_sched, start, budget):
    """Run the full local optimisation pipeline on a single start."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    # quick hill‑climb
    h, c = guided_hill_climb(h, rng, max_iters=800, step_scale=0.7)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # soft‑max SLSQP passes
    for beta in beta_sched:
        if time.time() - start > 0.95 * budget:
            break
        max_it = 800 if beta >= 200 else 400
        h = slsqp_optimize(h, beta, rng, max_iter=max_it)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    # gradient tidy‑up
    h = gradient_refine(h, rng, step=0.015, max_iter=8)
    c = compute_c5(h)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # final hill‑climb
    h, c = guided_hill_climb(h, rng, max_iters=1200, step_scale=0.5)
    if c < best_c:
        best_h, best_c = h.copy(), c

    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function with a low C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # start from parent if available
    base_h = None
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
        base_h = np.clip(base_h, 0.0, 1.0)
        base_h = enforce_sum(base_h, rng)

    # problem sizes to explore
    n_candidates = [256, 384, 512, 640]
    # β schedule for SLSQP
    beta_schedule = [5., 20., 80., 300., 1200., 4800., 20000.]

    for n in n_candidates:
        if time.time() - start_time > 0.90 * budget_s:
            break

        # number of random restarts per size (more for coarser grids)
        restarts = 5 if n == 256 else 4 if n == 384 else 3 if n == 512 else 2

        # a deterministic start via interpolation of the parent solution
        if base_h is not None:
            x_old = np.linspace(0.0, 2.0, base_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, base_h)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = enforce_sum(h0, rng)
            h_opt, c_opt = optimize_h(h0, n, rng, beta_schedule, start_time, budget_s)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n

        # random restarts
        for _ in range(restarts):
            if time.time() - start_time > 0.90 * budget_s:
                break
            h0 = random_feasible_vector(n, rng)
            h_opt, c_opt = optimize_h(h0, n, rng, beta_schedule, start_time, budget_s)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt.copy(), c_opt, n

    # final polishing on the overall best solution
    if best_h is not None:
        while time.time() - start_time < 0.98 * budget_s:
            # tiny gradient‑guided perturbation
            nh = guided_perturb(best_h, worst_gradient(best_h), rng, scale=0.02)
            # a few moderate‑β SLSQP passes
            for beta in (20., 200., 2000.):
                if time.time() - start_time > 0.98 * budget_s:
                    break
                nh = slsqp_optimize(nh, beta, rng, max_iter=300)
            nc = compute_c5(nh)
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
    else:
        # fallback – uniform 0.5 function
        best_h = np.full(256, 0.5)
        best_h = enforce_sum(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = 256

    return best_h.tolist(), float(best_c), int(best_h.size)
