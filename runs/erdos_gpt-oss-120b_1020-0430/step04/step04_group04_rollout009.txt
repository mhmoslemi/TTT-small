```python
    lo, hi = -1.0, 1.0
```

```python
def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    m = 1 << ((2*n-1 - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H*R, m)[:2*n-1]
    return np.max(conv) * (2.0 / n)
```

```python
def slsqp_refine(h0, beta, max_iter=400, rng=None):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)] * n
    cons = {'type':'eq', 'fun': sum_eq_constraint}
    res = minimize(smooth_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x, 0.0, 1.0)
    # correct sum error if any
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12 and rng is not None:
        # adjust via projection
        x = project_to_simplex_bounds(x, CURRENT_TARGET)
    return x
```

```python
def perturb_transfer(h, rng, scale):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new
```

import time
import numpy as np
from scipy.optimize import minimize

# ----- global holders for the smooth surrogate -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def full_corr_fft(x):
    """Full correlation of x with (1-x) using FFT."""
    n = x.shape[0]
    rev = (1 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())          # next power of two
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv


def compute_c5(h):
    """Exact discrete C5 for step function h sampled on [0,2]."""
    n = h.shape[0]
    conv = full_corr_fft(h)
    return np.max(conv) * (2.0 / n)


def smooth_obj(x):
    """Smooth approximation of max correlation using log‑sum‑exp."""
    conv = full_corr_fft(x)
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.exp(bc - m).sum())) / CURRENT_BETA


def project_to_simplex_bounds(v, target):
    """Project v onto {0 ≤ x ≤ 1, Σx = target} by a binary search on λ."""
    lo, hi = -1.0, 1.0
    # expand bounds if needed
    while np.sum(np.clip(v + lo, 0.0, 1.0)) > target:
        lo -= 1.0
    while np.sum(np.clip(v + hi, 0.0, 1.0)) < target:
        hi += 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def slsqp_refine(h0, beta, max_iter=400, rng=None):
    """Refine h0 via SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n * 0.5
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        smooth_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final sum correction if needed
    if rng is not None and abs(x.sum() - CURRENT_TARGET) > 1e-12:
        x = project_to_simplex_bounds(x, CURRENT_TARGET)
    return x


def random_feasible(n, rng):
    """Uniform random vector in [0,1]^n with exact sum n/2."""
    v = rng.random(n)
    return project_to_simplex_bounds(v, n * 0.5)


def perturb_transfer(h, rng, scale):
    """Transfer a small ε between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def hill_climb(h, rng, max_iters=1500, step=0.3):
    """Simple hill‑climb with transfer perturbations."""
    best = h.copy()
    best_c = compute_c5(best)
    for it in range(max_iters):
        cur_step = step * (0.99 ** (it // 100))
        cand = perturb_transfer(best, rng, cur_step)
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best, best_c = cand, cand_c
    return best


def sine_candidate(n, rng, J=4):
    """Random combination of sine/cosine basis functions, then projected."""
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    coeffs = rng.uniform(-0.4, 0.4, size=2 * J)
    vals = np.full(n, 0.5)
    for k in range(1, J + 1):
        a = coeffs[2 * (k - 1)]
        b = coeffs[2 * (k - 1) + 1]
        vals += a * np.sin(k * np.pi * xs / 2) + b * np.cos(k * np.pi * xs / 2)
    vals = np.clip(vals, 0.0, 1.0)
    return project_to_simplex_bounds(vals, n * 0.5)


def random_bump_candidate(n, rng, n_bumps=2):
    """Sum of a few triangular bumps with random parameters."""
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.zeros(n)
    for _ in range(n_bumps):
        ctr = rng.random() * 2.0
        width = rng.random() * 0.6 + 0.2
        amp = rng.random() * 0.6 + 0.2
        bump = np.maximum(1 - np.abs(xs - ctr) / width, 0.0) * amp
        h += bump
    h = np.clip(h, 0.0, 1.0)
    return project_to_simplex_bounds(h, n * 0.5)

    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial dimension and vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)

    else:

        h0 = random_feasible(n, rng)

    cosb = (1.0 + np.cos(np.pi * (xs - 1.0))) * 0.5

    max_pop = 30

    # evaluate initial population
    scores = [compute_c5(v) for v in pop]
    order = np.argsort(scores)
    pop = [pop[i] for i in order[:max_pop]]
    scores = [scores[i] for i in order[:max_pop]]
    best_h = pop[0].copy()
    best_c = scores[0]

    generation = 0
    # ---- evolutionary loop (main work) ----
    while time.time() - start < budget_s * 0.75:
        generation += 1
        n_best = min(8, len(pop))
        n_off = max_pop - n_best
        parents = pop[:n_best]
        offspring = []
        # step size decays slowly with generations
        step_scale = max(0.3 * (0.99 ** (generation // 50)), 0.01)

            alpha = rng.random()

        off_scores = [compute_c5(v) for v in offspring]
        # merge and keep the best max_pop individuals

        scores.extend(off_scores)
        order = np.argsort(scores)
        pop = [pop[i] for i in order[:max_pop]]
        scores = [scores[i] for i in order[:max_pop]]
        if scores[0] < best_c:

        best_h = slsqp_refine(best_h, beta, max_iter=300, rng=rng)
        best_c = compute_c5(best_h)

    # ---- final clean‑up ----
    best_h = project_to_simplex_bounds(best_h, n * 0.5)
    best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(n)
