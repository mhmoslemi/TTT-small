def gradient_softmax(h, beta):
    n = len(h)
    full_corr = np.correlate(h, 1 - h, mode='full')  # shape (2n-1)
    # compute softmax weights
    scaled = beta * full_corr
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()
    # w corresponds to lags index k = 0..2n-2, with lag = k - (n-1)
    g = np.zeros_like(h)
    for idx, weight in enumerate(w):
        lag = idx - (n-1)
        if lag >= 0:
            # shift plus
            # for j from 0 to n-1-lag
            g[:n-lag] += weight * (1 - h[lag:])  # (1 - h[j+lag])
            # shift minus term
            if lag > 0:
                g[lag:] -= weight * h[:n-lag]   # h[j-lag] where j>=lag
        else:
            # lag negative => shift minus side reversed
            lag_abs = -lag
            # for j from lag_abs to n-1: (lag negative, 1-h_{j+lag} is 1-h_{j-lag_abs})
            g[lag_abs:] += weight * (1 - h[:n-lag_abs])
            # shift minus term: h_{j-lag} = h_{j+lag_abs}
            g[:n-lag_abs] -= weight * h[lag_abs:]
    return g

def gradient_softmax(h, beta):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = beta * corr
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    grad = np.zeros_like(h)
    # loop
    for idx, weight in enumerate(w):
        lag = idx - (n - 1)
        if lag >= 0:
            # (1 - h_{i+lag}) term: for i from 0 to n-1-lag
            grad[:n-lag] += weight * (1 - h[lag:])
            # -h_{i-lag} term: for i from lag to n-1
            if lag > 0:
                grad[lag:] -= weight * h[:n-lag]
        else:
            lag_abs = -lag
            # (1 - h_{i - lag_abs}) for i >= lag_abs
            grad[lag_abs:] += weight * (1 - h[:n-lag_abs])
            # -h_{i + lag_abs} term for i from 0 to n-lag_abs-1
            grad[:n-lag_abs] -= weight * h[lag_abs:]
    return grad

def slsqp_optimize(h0, beta):

    target = n/2.0

        # we can compute softmax with beta

    return x

def gradient_softmax(h,beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = beta * corr
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()
    grad = np.zeros_like(h)
    for idx, weight in enumerate(w):
        lag = idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += weight * (1 - h[lag:])
            if lag > 0:
                grad[lag:] -= weight * h[:n - lag]
        else:
            lag_abs = -lag
            grad[lag_abs:] += weight * (1 - h[:n - lag_abs])
            grad[:n - lag_abs] -= weight * h[lag_abs:]
    max_corr = np.max(corr)  # unscaled
    return grad, max_corr

def perturb_h(h, rng, scale=0.1):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # Ensure sum unchanged; also within bounds due to eps selection.
    return new_h

def gradient_softmax(h, beta):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = beta * corr
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    grad = np.zeros_like(h)
    for idx, weight in enumerate(w):
        lag = idx - (n - 1)
        if lag >= 0:
            # first term
            grad[:n - lag] += weight * (1 - h[lag:])
            # second term
            grad[lag:] -= weight * h[:n - lag]
        else:
            lag_abs = -lag
            grad[lag_abs:] += weight * (1 - h[:n - lag_abs])
            grad[:n - lag_abs] -= weight * h[lag_abs:]
    max_corr = np.max(corr)
    return grad, max_corr

            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h.copy()
    return best_h, best_c5


def gradient_descent_softmax(start_h, betas, target, rng, time_budget, t0):
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in betas:
        step = 0.5 / np.sqrt(beta)
        for _ in range(300):
            if time.time() - t0 > time_budget:
                break
            grad, _ = gradient_softmax(h, beta)
            new_h = project_onto_box_and_sum(h - step * grad, target)
            if rng.random() < 0.05:
                new_h = project_onto_box_and_sum(
                    new_h + rng.normal(scale=step * 0.1, size=n), target
                )
            c5_new = compute_c5(new_h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = new_h.copy()
            h = new_h
            step *= 0.9995
            if step < 1e-6:
                break
        if time.time() - t0 > time_budget:
            break
    return best_h, best_c5


def slsqp_optimize_one_beta(h0, beta, target):
    global CONSTRAINT_TARGET, CONSTRAINT_BETA
    CONSTRAINT_TARGET = target
    CONSTRAINT_BETA = beta
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 200, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    if abs(target - x.sum()) > 1e-12:
        x = project_onto_box_and_sum(x, target)
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    # ----- initial vector -----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_and_sum(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    betas = [5, 20, 80, 200, 500, 1000, 2000, 5000]

    main_budget = budget_s * 0.85
    while time.time() - t_start < main_budget:
        # start point
        if rng.random() < 0.7:
            start = perturb_h(best_h, rng, scale=0.4)
        else:
            start = random_feasible_vector(n, rng)

        # gradient descent on softmax surrogate
        h_opt, c5_opt = gradient_descent_softmax(
            start, betas, target, rng, main_budget, t_start
        )
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

        # SLSQP refinements
        for beta in betas:
            if time.time() - t_start > main_budget:
                break
            try:
                h_ref = slsqp_optimize_one_beta(best_h, beta, target)
                c5_ref = compute_c5(h_ref)
                if c5_ref < best_c5:
                    best_c5, best_h = c5_ref, h_ref.copy()
            except Exception:
                continue

    # ----- optional up‑sampling -----
    remaining = budget_s - (time.time() - t_start)
    if remaining > budget_s * 0.07:
        new_n = min(720, max(480, n * 2))
        if new_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = project_onto_box_and_sum(up_h, new_n / 2.0)

            fine_budget = remaining
            h_fine, c5_fine = gradient_descent_softmax(
                up_h, betas[:4], new_n / 2.0, rng, fine_budget, time.time()
            )
            if c5_fine < best_c5:
                best_c5, best_h, n = c5_fine, h_fine.copy(), new_n

            # quick SLSQP on fine grid
            for beta in betas[:4]:
                if time.time() - t_start > budget_s:
                    break
                try:
                    h_ref = slsqp_optimize_one_beta(best_h, beta, n / 2.0)
                    c5_ref = compute_c5(h_ref)
                    if c5_ref < best_c5:
                        best_c5, best_h = c5_ref, h_ref.copy()
                except Exception:
                    continue

    # ----- final local hill‑climb -----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    step = 0.2
    for _ in range(3000):
        if time.time() - t_start > budget_s:
            break
        new_h = perturb_h(cur_h, rng, scale=step)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_c5, cur_h = new_c5, new_h
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h.copy()
        step *= 0.9995

    # ----- final clean‑up -----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_and_sum(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(n)
