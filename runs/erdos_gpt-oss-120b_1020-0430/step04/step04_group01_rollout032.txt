def proj_box_sum(u, lower, upper, target):
    lo = np.min(u) - upper
    hi = np.max(u) - lower
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(u - mid, lower, upper)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(u - (lo+hi)/2.0, lower, upper)

import time, numpy as np

# --------------------------------------------------------------
# Core utilities

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def proj_box_sum(u, lower, upper, target):
    lo = np.min(u) - upper
    hi = np.max(u) - lower
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(u - mid, lower, upper)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(u - (lo + hi) * 0.5, lower, upper)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# Correlation (FFT based)
# --------------------------------------------------------------
def correlate_full_fft(a, b):
    n, m = a.shape[0], b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    corr = np.fft.irfft(fa * np.conj(fb), L)[: n + m - 1]
    return corr


def compute_c5(h):
    n = h.shape[0]
    conv = correlate_full_fft(h, 1 - h)
    return np.max(conv) * (2.0 / n)


def smooth_obj(h, beta):
    g = correlate_full_fft(h, 1 - h)
    bc = beta * g
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def grad_descent(h0, beta, max_iters, deadline):
    h = h0.copy()
    f_val = smooth_obj(h, beta)
    step = 1.0
    n = h.size
    target = n / 2.0
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # weights for each shift
        g = correlate_full_fft(h, 1 - h)
        bc = beta * g
        m = bc.max()
        w = np.exp(bc - m)
        w /= w.sum()
        # A_i = sum_k w_k h_{i+k}
        A = correlate_full_fft(w, h)[n - 1 : n - 1 + n]
        # B_i = sum_k w_k h_{i-k}
        B = correlate_full_fft(w[::-1], h)[n - 1 : n - 1 + n]
        grad = 1.0 - A - B
        # backtrack line search
        s = step
        for _ in range(20):
            h_new = proj_box_sum(h - s * grad, 0.0, 1.0, target)
            f_new = smooth_obj(h_new, beta)
            if f_new <= f_val:
                break
            s *= 0.5
        else:
            break
        h = h_new
        f_val = f_new
        step = max(step, s)
    return h

def hill_climb(h_start, rng, max_iters=1500, start_step=0.4):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step

            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c

    Tf = 1e-6
    max_iters = 30000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.clip(np.array(initial_h_values, dtype=float), 0.0, 1.0)
        n = h0.size
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 720
        h0 = np.zeros(n)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- Phase 1: binary SA + hill‑climb -------------------------------
    phase1_end = start + 0.35 * budget_s
    while time.time() < phase1_end:
        # binary SA
        bin_start = np.zeros(n)
        idx = np.argsort(-best_h)[: n // 2]
        bin_start[idx] = 1.0
        h_bin, c_bin = binary_sa(bin_start, rng, phase1_end)
        if c_bin < best_c:
            best_h, best_c = h_bin.copy(), c_bin
        # hill‑climb refinement
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=1200, start_step=0.5)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc
        # occasional random restart
        if rng.random() < 0.15:
            rnd = np.zeros(n)
            rnd[: n // 2] = 1.0
            rng.shuffle(rnd)
            rnd = enforce_sum(rnd.astype(float), n / 2.0, rng)
            c_r = compute_c5(rnd)
            if c_r < best_c:
                best_h, best_c = rnd.copy(), c_r

    # ----- Phase 2: smooth gradient descent (continuation) ----------------
    beta_seq = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
    phase2_end = start + 0.85 * budget_s
    while time.time() < phase2_end:
        # start from a perturbed copy of current best
        h_start = perturb_h(best_h, rng, scale=0.3)
        for beta in beta_seq:
            if time.time() >= phase2_end:
                break
            max_iters = 400
            h_start = grad_descent(h_start, beta, max_iters, phase2_end)
            c_now = compute_c5(h_start)
            if c_now < best_c:
                best_h, best_c = h_start.copy(), c_now

    # ----- Phase 3: final high‑beta polishing ----------------------------
    for beta in (2000.0, 5000.0):
        if time.time() >= deadline:
            break
        best_h = grad_descent(best_h, beta, max_iters=1200, deadline=deadline)
        best_c = compute_c5(best_h)

    # final hill‑climb as safety net
    if time.time() < deadline:
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=2000, start_step=0.3)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc

    # enforce feasibility one last time
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
