def adam_logsumexp_optimize(h0, target, budget_deadline):
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    # allocate schedule
    betas = np.geomspace(5.0, 5000.0, num=30)
    # compute L once for this n
    L = 1 << ((3*n - 2).bit_length())
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    for beta in betas:
        for _ in range(200):  # 200 steps per beta
            if time.time() > budget_deadline:
                return best_h, best_c
            # correlation
            f = np.correlate(h, 1.0 - h, mode='full') * dx
            # smooth max weights
            # use stability
            max_fb = np.max(beta * f)
            w = np.exp(beta * f - max_fb)
            w_sum = np.sum(w)
            w = w / w_sum
            # forward & backward via FFT convolution
            # w_rev
            w_rev = w[::-1]
            # compute conv_w_h
            w_fft = np.fft.rfft(w, n=L)
            h_fft = np.fft.rfft(h, n=L)
            conv_bw = np.fft.irfft(w_fft * h_fft, n=L)
            # forward uses w_rev
            w_rev_fft = np.fft.rfft(w_rev, n=L)
            conv_fw = np.fft.irfft(w_rev_fft * h_fft, n=L)
            start = n - 1
            backward = conv_bw[start:start + n]
            forward = conv_fw[start:start + n]
            # gradient
            g = dx * (1.0 - forward - backward)
            t += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            # clip to [0,1]
            np.clip(h, 0.0, 1.0, out=h)
            # adjust sum exactly
            diff = target - h.sum()
            # simple loop to adjust via random indices
            for __ in range(2000):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                    if add <= 0:
                        continue
                else:
                    add = max(diff, -h[i])
                    if add >= 0:
                        continue
                h[i] += add
                diff = target - h.sum()
            # evaluate c5
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def adam_optimize(h0, target, rng, deadline):
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    # ensure symmetry and monotonic at start
    h = symmetrize(h)
    h = project_monotonic_decreasing(h)
    # adjust sum to target exactly
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # adjust using adjust_sum_exact (which may break monotonic)
        h = adjust_sum_exact(h, rng)
    # after adjust, apply monotonic again
    h = project_monotonic_decreasing(h)
    # Initialize Adam parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    # precompute FFT length
    L = 1 << ((3*n - 2).bit_length())
    # schedule betas
    betas = np.geomspace(5.0, 5000.0, num=30)
    for beta in betas:
        # steps per beta
        steps = 150  # can be adjusted based on time
        for _ in range(steps):
            if time.time() > deadline:
                return best_h, best_c
            # compute f
            f = np.correlate(h, 1.0 - h, mode="full") * dx
            # compute softmax weights w
            max_beta_f = np.max(beta * f)  # stability
            w = np.exp(beta * f - max_beta_f)
            w_sum = w.sum()
            w /= w_sum
            # compute forward, backward via FFT
            w_fft = np.fft.rfft(w, n=L)
            h_fft = np.fft.rfft(h, n=L)
            conv_bw = np.fft.irfft(w_fft * h_fft, n=L)
            w_rev_fft = np.fft.rfft(w[::-1], n=L)
            conv_fw = np.fft.irfft(w_rev_fft * h_fft, n=L)
            start = n - 1
            backward = conv_bw[start:start + n]
            forward = conv_fw[start:start + n]
            g = dx * (1.0 - forward - backward)
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            # clip
            np.clip(h, 0.0, 1.0, out=h)
            # symmetrize
            h = (h + h[::-1]) / 2
            # monotonic decreasing
            h = project_monotonic_decreasing(h)
            # adjust sum
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                # adjust using simple approach: add diff to first element if diff>0, else subtract from last
                # ensure within bounds
                if diff > 0:
                    # add to first element (largest)
                    add = min(diff, 1.0 - h[0])
                    h[0] += add
                    diff -= add
                else:
                    # subtract from last element (smallest)
                    sub = max(diff, -h[-1])
                    h[-1] += sub
                    diff -= sub
                # if still diff, fallback random adjustment
                if abs(diff) > 1e-12:
                    h = adjust_sum_exact(h, rng)
                    h = project_monotonic_decreasing(h)
            # evaluate c5 occasionally (every iteration)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def local_search(h, target, rng, dead):
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    half = n // 2
    eps = 0.01
    while time.time() < dead:
        i, j = rng.choice(half, 2, replace=False)
        # ensure we can adjust by eps both ways respecting bounds
        if best_h[i] + eps > 1.0 or best_h[j] - eps < 0.0:
            continue
        h_new = best_h.copy()
        h_new[i] += eps
        h_new[j] -= eps
        # enforce symmetry
        h_new[-i-1] = h_new[i]
        h_new[-j-1] = h_new[j]

        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new

def adam_optimize(h0, target, rng, deadline):

    dx = 2.0 / n
    h = h0.copy()

    diff = target - h.sum()

    v = np.zeros_like(h)

    beta2 = 0.999

    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    # FFT length
    L = 1 << ((3*n - 2).bit_length())
    betas = np.geomspace(5.0, 5000.0, num=30)
    for beta in betas:
        steps = 150
        for _ in range(steps):
            if time.time() > deadline:
                return best_h, best_c
            # correlation
            f = np.correlate(h, 1.0 - h, mode="full") * dx
            max_beta_f = np.max(beta * f)
            w = np.exp(beta * f - max_beta_f)
            w_sum = w.sum()
            w /= w_sum
            # FFT conv
            w_fft = np.fft.rfft(w, n=L)
            h_fft = np.fft.rfft(h, n=L)
            conv_bw = np.fft.irfft(w_fft * h_fft, n=L)
            w_rev_fft = np.fft.rfft(w[::-1], n=L)
            conv_fw = np.fft.irfft(w_rev_fft * h_fft, n=L)
            start = n - 1
            backward = conv_bw[start:start + n]
            forward = conv_fw[start:start + n]
            g = dx * (1.0 - forward - backward)
            t += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            np.clip(h, 0.0, 1.0, out=h)
            # symmetrize and monotonic
            h = (h + h[::-1]) / 2
            h = project_monotonic_decreasing(h)
            # sum adjust
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                # simple adjustment to extremes
                if diff > 0:
                    # add to first element
                    add = min(diff, 1.0 - h[0])
                    h[0] += add
                else:
                    sub = max(diff, -h[-1])
                    h[-1] += sub
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    h = adjust_sum_exact(h, rng)
                    h = project_monotonic_decreasing(h)
            # evaluate
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()

import time
import numpy as np

# ----------------- basic utilities -----------------
def compute_c5(h: np.ndarray) -> float:
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = n/2 exactly (modifies in place)."""
    target = h.size / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            if add <= 0:
                continue
        else:
            add = max(diff, -h[i])
            if add >= 0:
                continue
        h[i] += add
        diff = target - h.sum()
    return h


def project_monotonic_decreasing(h: np.ndarray) -> np.ndarray:
    """Force h to be symmetric and non‑increasing on the left half."""
    n = h.size
    half = n // 2
    left = h[:half]
    # decreasing: left[i] >= left[i+1]
    neg = -left
    nd = np.maximum.accumulate(neg)  # non‑decreasing on -left
    left_mon = -nd
    h[:half] = left_mon
    h[-half:] = left_mon[::-1]
    return h


def symmetrize(h: np.ndarray) -> np.ndarray:
    return (h + h[::-1]) / 2


# ----------------- start vector generators -----------------
def random_symmetric_start(n: int, rng: np.random.Generator) -> np.ndarray:
    half = n // 2
    left = rng.random(half)
    left = np.sort(left)[::-1]  # decreasing
    left_target = n / 4.0
    left = left * (left_target / left.sum())
    left = np.clip(left, 0.0, 1.0)
    h = np.concatenate([left, left[::-1]])
    h = adjust_sum_exact(h, rng)
    h = symmetrize(h)
    h = project_monotonic_decreasing(h)
    return h


def random_binary_symmetric_start(n: int, rng: np.random.Generator) -> np.ndarray:
    half = n // 2
    left = np.zeros(half)
    left[rng.choice(half, half // 2, replace=False)] = 1.0
    h = np.concatenate([left, left[::-1]])
    h = project_monotonic_decreasing(h)
    return h


# ----------------- core optimizer (Adam on smooth max) -----------------
def adam_optimize(h0: np.ndarray, target: float, rng: np.random.Generator,
                  deadline: float) -> tuple[np.ndarray, float]:
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    # ensure exact sum before starting
    h = adjust_sum_exact(h, rng)

    # Adam parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_h = h.copy()
    best_c = compute_c5(h)

    # FFT length (next power of two >= 3n-2)
    L = 1 << ((3 * n - 2).bit_length())

    betas = np.geomspace(5.0, 5000.0, num=30)
    steps_per_beta = 250  # enough iterations per beta

    for beta in betas:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_h, best_c

            # smooth‑max weights
            f = np.correlate(h, 1.0 - h, mode="full") * dx
            max_beta_f = np.max(beta * f)
            w = np.exp(beta * f - max_beta_f)
            w /= w.sum()

            # forward / backward convolutions via FFT
            w_fft = np.fft.rfft(w, n=L)
            h_fft = np.fft.rfft(h, n=L)
            conv_bw = np.fft.irfft(w_fft * h_fft, n=L)
            conv_fw = np.fft.irfft(np.fft.rfft(w[::-1], n=L) * h_fft, n=L)

            start = n - 1
            backward = conv_bw[start:start + n]
            forward = conv_fw[start:start + n]

            # gradient of smooth max
            g = dx * (1.0 - forward - backward)

            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)
            h -= step

            # box constraints
            np.clip(h, 0.0, 1.0, out=h)

            # keep symmetry & monotonicity
            h = symmetrize(h)
            h = project_monotonic_decreasing(h)

            # enforce exact sum (cheap tweak first, fallback to random if needed)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                if diff > 0:
                    add = min(diff, 1.0 - h[0])
                    h[0] += add
                else:
                    sub = max(diff, -h[-1])
                    h[-1] += sub
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    h = adjust_sum_exact(h, rng)
                    h = project_monotonic_decreasing(h)

            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    return best_h, best_c
