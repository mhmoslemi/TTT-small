def grad_i_vector(i, h):
    n = h.size
    s = np.arange(-n+1, n)
    grad = np.zeros(2*n-1)
    # Part where i+s in range: s from -i to n-1-i
    s1_start = -i
    s1_end = n-1-i
    idx1 = np.where((s >= s1_start) & (s <= s1_end))[0]
    grad[idx1] = 1 - h[i + s[idx1]]
    # Part where i-s in range: s from i-(n-1) to i
    s2_start = i-(n-1)
    s2_end = i
    idx2 = np.where((s >= s2_start) & (s <= s2_end))[0]
    grad[idx2] -= h[i - s[idx2]]
    return grad

import time, numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:  # need to increase
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:  # need to decrease
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def compute_c5(h):
    """Discrete C5 for a vector h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * (2.0 / h.size)

def gradient_shift(s, h):
    """gradient of a single shift s w.r.t. every h[i] (size n)."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[: n - s_abs])
        grad[: n - s_abs] -= h[s_abs:]
    return grad

def grad_vector_for_index(i, h):
    """gradient of all shifts w.r.t. h[i]; length 2n-1."""
    n = h.size
    s = np.arange(-n + 1, n)
    grad = np.zeros(2 * n - 1)
    # part where i+s in range
    mask1 = (s >= -i) & (s <= n - 1 - i)
    indices1 = np.where(mask1)[0]
    grad[indices1] = 1.0 - h[i + s[indices1]]
    # part where i-s in range
    mask2 = (s >= i - (n - 1)) & (s <= i)
    indices2 = np.where(mask2)[0]
    grad[indices2] -= h[i - s[indices2]]
    return grad

# ---------- SLSQP polishing ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

    start_time = time.time()
    # ----- initial feasible vector -----

    h = enforce_sum(np.clip(h, 0.0, 1.0), h.size / 2.0, rng)
    n = h.size
    dx = 2.0 / n

    # ---- initial evaluation ----
    corr = np.correlate(h, 1 - h, mode='full')
    best_c = corr.max() * dx
    best_h = h.copy()

    # ---- main optimization loop ----
    while time.time() - start_time < budget_s * 0.92:
        # occasional full restart
        if rng.random() < 0.001:
            h = random_feasible_vector(n, rng)
            corr = np.correlate(h, 1 - h, mode='full')
            continue

        # worst shift
        idx0 = corr.argmax()
        s0 = idx0 - (n - 1)
        # second worst shift (needed only for tie‑breaking)
        sec_corr = corr.copy()
        sec_corr[idx0] = -np.inf
        idx1 = sec_corr.argmax()
        s1 = idx1 - (n - 1)

        # gradient of worst shift
        grad_worst = gradient_shift(s0, h)

        # choose i (decrease) and j (increase)

            j = neg[np.argmin(grad_worst[neg])]
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)

        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            # nothing to move
            continue

        # full gradient vectors for i and j
        gi = grad_vector_for_index(i, h)
        gj = grad_vector_for_index(j, h)
        d = gj - gi  # effect of epsilon on each shift (raw correlation)

        # ternary search for epsilon that minimizes max(raw_corr + eps*d)
        lo, hi = 0.0, max_eps
        raw_corr = corr  # already raw (no dx)
        for _ in range(12):
            m1 = lo + (hi - lo) / 3.0

                lo = m1

        # apply move

            if a != b:

                h[b] = max(0.0, min(1.0, h[b] + eps2))

        corr = np.correlate(h, 1 - h, mode='full')
