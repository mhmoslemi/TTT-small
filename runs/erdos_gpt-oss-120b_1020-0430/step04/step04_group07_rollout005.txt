import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def symmetrize(h):
    return (h + h[::-1]) * 0.5

def random_feasible_vector(n, rng):
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # adjust random indices
        while abs(diff) > 1e-12:
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stabilized log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        nh = symmetrize(nh)
        nh = enforce_sum(nh, rng)
        c = compute_c5(nh)
        if c < best_c:
            best_c = c
            best_h = nh.copy()
            h = nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.015, max_iter=10):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:]       -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h

def smooth_vector(h, weight=0.5):
    # simple three-point smoothing with given weight for neighbors
    kernel = np.array([weight / 2.0, 1.0 - weight, weight / 2.0])
    # use mode='same' (zero padding) then symmetrize to reduce edge effect
    h_sm = np.convolve(h, kernel, mode='same')
    h_sm = symmetrize(h_sm)
    return h_sm

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c = np.inf
    best_n = None

    # initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = symmetrize(h0)
        h0 = enforce_sum(h0, rng)
        best_h = h0
        best_n = h0.size
        best_c = compute_c5(best_h)

    # sample sizes
    n_candidates = [96, 128, 160, 192, 224, 256, 288, 320, 352, 384]

    # main search
    while time.time() - start < 0.80 * budget_s:
        n = int(rng.choice(n_candidates))
        fresh = rng.random() < 0.12
        if not fresh and best_h is not None:
            if best_n == n:
                h = best_h.copy()
            else:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0.0, 1.0)
                h = symmetrize(h)
                h = enforce_sum(h, rng)
        else:
            if rng.random() < 0.5:
                h = np.zeros(n)
                h[rng.choice(n, n // 2, replace=False)] = 1.0
            else:
                h = random_feasible_vector(n, rng)
            h = symmetrize(h)
            h = enforce_sum(h, rng)

        # quick hill climb
        h, c = guided_hill_climb(h, rng, max_iters=1800, step_scale=0.65)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # SLSQP annealing
        for beta in (30., 100., 400., 1500., 6000., 20000.):
            if time.time() - start > 0.94 * budget_s:
                break
            maxit = 800 if beta >= 1500 else 500
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            # small smoothing after high-beta steps
            if beta >= 6000:

                h = enforce_sum(h, rng)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # fine gradient refinement
        h = gradient_refine(h, rng, step=0.012, max_iter=12)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # final hill climb on this run
        h, c = guided_hill_climb(h, rng, max_iters=2000, step_scale=0.55)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # upsampling refinement if time permits
    if best_h is not None and time.time() - start < 0.96 * budget_s:
        # allow up to 512 points
        for factor in (2, 3):
            n_up = best_n * factor
            if n_up > 512:
                continue
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = symmetrize(h_up)
            h_up = enforce_sum(h_up, rng)

            # hill climb at higher resolution
            h_up, c_up = guided_hill_climb(h_up, rng, max_iters=2500, step_scale=0.6)

            # SLSQP passes with moderate β
            for beta in (40., 200., 2000., 10000.):
                if time.time() - start > budget_s:
                    break
                h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
                if beta >= 2000:
                    h_up = smooth_vector(h_up, weight=0.25)
                    h_up = enforce_sum(h_up, rng)
                c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, best_n = h_up.copy(), c_up, n_up

            # tiny gradient tidy‑up
            h_up = gradient_refine(h_up, rng, step=0.008, max_iter=8)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # final stochastic tweaking
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        # occasionally do random perturbations
        if rng.random() < 0.03:
            # random small noise
            nh = best_h + rng.uniform(-0.02, 0.02, size=best_h.shape)
            nh = np.clip(nh, 0.0, 1.0)
            nh = symmetrize(nh)
        else:
            grad = worst_gradient(best_h)
            nh = guided_perturb(best_h, grad, rng, scale=0.02)
            nh = symmetrize(nh)
        nh = enforce_sum(nh, rng)
        nc = compute_c5(nh)
        if nc < best_c:

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = symmetrize(best_h)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)

import time
import numpy as np
from scipy.optimize import minimize

# globals used by the SLSQP soft‑max objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def symmetrize(h: np.ndarray) -> np.ndarray:
    """Make the vector symmetric around the centre."""
    return (h + h[::-1]) * 0.5


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector with entries in [0,1] and sum=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    scale = step_scale
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=scale)
        nh = symmetrize(nh)
        nh = enforce_sum(nh, rng)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            scale = max(scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:]       -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, rng)
    return h


def smooth_vector(h: np.ndarray, weight: float = 0.5) -> np.ndarray:
    """Simple three‑point smoothing."""
    kernel = np.array([weight / 2.0, 1.0 - weight, weight / 2.0])
    h = np.convolve(h, kernel, mode="same")
    h = symmetrize(h)
    return h
