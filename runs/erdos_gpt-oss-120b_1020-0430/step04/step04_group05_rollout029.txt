def project_box_sum(x, target, eps=1e-12, max_iter=100):
    lo = -1.0  # sufficiently low λ -> x_i - λ large >>1 so clip to 1 => sum= n
    hi = 2.0   # sufficiently high λ -> x_i - λ <<0 => clip to 0 => sum=0
    for _ in range(max_iter):

        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= eps:
            return y
        if s > target:
            lo = mid
        else:
            hi = mid

    n = h.size

    corr = np.fft.irfft(H * np.conj(B), n=L_corr)[:2*n-1]
    max_f = corr.max()

def project_box_sum(x, target, eps=1e-9, max_iter=100):
    # clip to box [0,1]
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) <= eps:
        return x
    # binary search λ such that sum(clip(x - λ, 0, 1)) = target
    lo = -1.0
    hi = 1.0
    # ensure lo yields sum >= target, hi yields <= target.
    # Expand bounds if needed.
    while np.clip(x - lo, 0.0, 1.0).sum() < target:
        lo *= 2.0
    while np.clip(x - hi, 0.0, 1.0).sum() > target:
        hi *= 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= eps:
            return y
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

    h = h_start.copy()

        if now >= time_deadline:
            break
        t += 1
        # compute FFTs

        Bc = np.fft.rfft(1 - h, n=Lc)
        corr_full = np.fft.irfft(Hc * np.conj(Bc), n=Lc)
        corr = corr_full[:2*n-1]
        max_f = corr.max()
        # weights
        w = np.exp(beta * (corr - max_f))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # compute gradient via convolution
        w_rev = w[::-1]
        # convolution with h
        conv_plus_full = np.fft.irfft(Hc * np.fft.rfft(w_rev, n=Lc), n=Lc)
        conv_minus_full = np.fft.irfft(Hc * np.fft.rfft(w, n=Lc), n=Lc)

        S_minus = conv_minus_full[start:start+n]
        grad = 1.0 - S_plus - S_minus
        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

def project_box_sum(x, target, rng, eps=1e-12, max_iter=100):
    # Clip to [0,1] first
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) <= eps:
        return x
    # We need to find lambda such that sum(clip(x - lam, 0, 1)) = target
    # Determine bounds
    # low lambda (maybe negative) yields sum >= target
    lam_low = -1.0
    while np.clip(x - lam_low, 0.0, 1.0).sum() < target:
        lam_low *= 2.0
    lam_high = 1.0
    while np.clip(x - lam_high, 0.0, 1.0).sum() > target:
        lam_high *= 2.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        y = np.clip(x - lam_mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= eps:
            return y
        if s > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(x - (lam_low + lam_high) / 2.0, 0.0, 1.0)

import time
import numpy as np

# ---------- projection onto box and sum ----------
def project_box_sum(x, target, rng, eps=1e-12, max_iter=100):
    """Project x onto [0,1]^n with exact sum `target` using a binary search on λ."""
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) <= eps:
        return x
    # find λ low (sum >= target) and λ high (sum <= target)
    lam_low = -1.0
    while np.clip(x - lam_low, 0.0, 1.0).sum() < target:
        lam_low *= 2.0
    lam_high = 1.0
    while np.clip(x - lam_high, 0.0, 1.0).sum() > target:
        lam_high *= 2.0
    for _ in range(max_iter):
        lam_mid = (lam_low + lam_high) / 2.0
        y = np.clip(x - lam_mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= eps:
            return y
        if s > target:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(x - (lam_low + lam_high) / 2.0, 0.0, 1.0)


# ---------- random feasible vector ----------
def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0, rng)


# ---------- small transfer perturbation ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- FFT‑based correlation ----------
def compute_corr_fft(h):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft(1.0 - h, n=L)
    corr_full = np.fft.irfft(H * np.conj(B), n=L)
    return corr_full[: 2 * n - 1]


def compute_c5_from_corr(corr, n):
    return corr.max() * (2.0 / n)


def compute_c5(h):
    return compute_c5_from_corr(compute_corr_fft(h), h.size)


# ---------- Adam optimisation on smooth max ----------
def adam_optimize(start_h, beta, target_sum, rng, deadline):
    n = start_h.size
    Lc = 1 << ((3 * n - 2).bit_length())
    h = start_h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h)
    while True:
        if time.time() >= deadline:
            break
        t += 1
        # FFT of h (size Lc) and of 1‑h
        Hc = np.fft.rfft(h, n=Lc)
        Bc = np.fft.rfft(1.0 - h, n=Lc)
        # linear correlation (first 2n‑1 values)
        corr_full = np.fft.irfft(Hc * np.conj(Bc), n=Lc)
        corr = corr_full[: 2 * n - 1]
        max_f = corr.max()
        # soft‑max weights (stable)
        w = np.exp(beta * (corr - max_f))
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum
        # gradients via convolutions
        w_rev = w[::-1]
        conv_plus = np.fft.irfft(Hc * np.fft.rfft(w_rev, n=Lc), n=Lc)
        conv_minus = np.fft.irfft(Hc * np.fft.rfft(w, n=Lc), n=Lc)
        start = n - 1
        S_plus = conv_plus[start : start + n]
        S_minus = conv_minus[start : start + n]
        grad = 1.0 - S_plus - S_minus
        # Adam update
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_box_sum(h_new, target_sum, rng)
        h = h_new
        cur_c = max_f * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
