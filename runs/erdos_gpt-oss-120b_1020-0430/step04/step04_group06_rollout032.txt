    while time.time() - start < time_limit:

        cur_max = float(cur_corr[idx_max])
        shift = idx_max - (n - 1)
        # compute i_range, j_range
        if shift >= 0:

            j_rng = i_rng + shift
        else:

            j_rng = i_rng + shift

            break

# Global for SLSQP surrogate
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(2000):
            if abs(diff) < 1e-14:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def hill_climb_multi(h_start, rng, max_iters=1500, init_step=0.4, n_trials=5):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def slsqp_optimize(h0, beta, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def optimal_pair_transfer(h0, corr0, rng, time_limit):
    n = h0.size
    cur_h = h0.copy()
    cur_corr = corr0.copy()
    best_h = cur_h.copy()
    best_val = cur_corr.max()
    start = time.time()
    no_improve = 0
    s_arr = np.arange(-(n - 1), n, dtype=int)
    while time.time() - start < time_limit:
        idx_max = int(np.argmax(cur_corr))
        cur_max = float(cur_corr[idx_max])
        shift = idx_max - (n - 1)
        if shift >= 0:
            i_rng = np.arange(n - shift)
            j_rng = i_rng + shift
        else:
            i_rng = np.arange(-shift, n)
            j_rng = i_rng + shift
        if i_rng.size == 0:
            break
        prod = cur_h[i_rng] * (1.0 - cur_h[j_rng])
        local_idx = int(np.argmax(prod))
        i = int(i_rng[local_idx])
        j = int(j_rng[local_idx])
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 1e-15:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 1e-15:
                continue
        # build v
        v = np.zeros_like(cur_corr)
        i_ps = i + s_arr
        mask = (i_ps >= 0) & (i_ps < n)
        v[mask] -= (1.0 - cur_h[i_ps[mask]])
        i_ms = i - s_arr
        mask = (i_ms >= 0) & (i_ms < n)
        v[mask] += cur_h[i_ms[mask]]
        j_ps = j + s_arr
        mask = (j_ps >= 0) & (j_ps < n)
        v[mask] += (1.0 - cur_h[j_ps[mask]])
        j_ms = j - s_arr
        mask = (j_ms >= 0) & (j_ms < n)
        v[mask] -= cur_h[j_ms[mask]]
        v_max = v[idx_max]
        denom = v - v_max
        pos = denom > 1e-15
        eps_opt = max_eps
        if np.any(pos):
            eps_cand = (cur_max - cur_corr[pos]) / denom[pos]
            eps_cand = eps_cand[(eps_cand > 1e-15) & (eps_cand < max_eps)]
            if eps_cand.size:
                eps_opt = eps_cand.min()
        if eps_opt <= 1e-15:
            no_improve += 1
            if no_improve > 2000:
                cur_h = perturb_h(cur_h, rng, scale=0.05)
                cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
                no_improve = 0
            continue
        new_corr = cur_corr + eps_opt * v
        new_max = new_corr.max()
        if new_max < cur_max - 1e-14:
            cur_h[i] -= eps_opt
            cur_h[j] += eps_opt
            cur_corr = new_corr
            no_improve = 0
            if new_max < best_val - 1e-14:
                best_val, best_h = new_max, cur_h.copy()
        else:
            no_improve += 1
            if no_improve > 500:
                cur_h = perturb_h(cur_h, rng, scale=0.05)
                cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
                no_improve = 0
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    n = 360
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        h0 = random_feasible_vector(n, rng)

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]
    for f in (1, 2, 3, 4, 5):
        for func in (np.sin, np.cos):
            h = 0.5 + 0.5 * func(np.pi * f * xs)
            candidates.append(enforce_sum(np.clip(h, 0.0, 1.0), n / 2.0, rng))
    candidates.append(enforce_sum(xs / 2.0, n / 2.0, rng))
    candidates.append(enforce_sum(1.0 - xs / 2.0, n / 2.0, rng))
    mask = np.zeros(n)
    mask[: n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))
    mask = np.zeros(n)
    start_mid = n // 4
    mask[start_mid : start_mid + n // 2] = 1.0
    candidates.append(enforce_sum(mask, n / 2.0, rng))
    candidates.append(enforce_sum(np.arange(n) % 2, n / 2.0, rng))
    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    global_best_h, global_best_c = best_h, best_c

    main_limit = budget_s * 0.70
    while time.time() - start < main_limit:
        start_vec = (
            perturb_h(global_best_h, rng, scale=0.35)
            if rng.random() < 0.65
            else random_feasible_vector(n, rng)
        )
        h_hc, c_hc = hill_climb_multi(start_vec, rng, max_iters=1500, init_step=0.4, n_trials=5)
        if c_hc < global_best_c:
            global_best_h, global_best_c = h_hc, c_hc
        for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000):
            if time.time() - start >= main_limit:
                break
            try:
                h_ref = slsqp_optimize(global_best_h, beta, max_iter=400)
                c_ref = compute_c5(h_ref)
                if c_ref < global_best_c:
                    global_best_h, global_best_c = h_ref, c_ref
            except Exception:
                continue

    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        corr = np.correlate(global_best_h, 1.0 - global_best_h, mode="full")
        best_h2, _ = optimal_pair_transfer(
            global_best_h, corr, rng, time_limit=remaining * 0.95
        )
        best_h2 = np.clip(best_h2, 0.0, 1.0)
        best_h2 = enforce_sum(best_h2, n / 2.0, rng)
        global_best_h = best_h2
        global_best_c = compute_c5(global_best_h)

    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
