def gradient_descent(h0, beta_sched, target, max_iters_per_beta, time_deadline, rng):

    best_h = h.copy()
    best_c = compute_c5(h)

    for beta in beta_sched:
        step = 0.5 / np.sqrt(beta)  # maybe dynamic
        for it in range(max_iters_per_beta):
            now = time.time()
            if now >= time_deadline:
                return best_h, best_c
            # compute correlation
            corr = correlate_full(h)
            max_corr = corr.max()
            exp_vals = np.exp(beta * (corr - max_corr))
            w = exp_vals / exp_vals.sum()
            # compute gradient via FFT
            n = h.size
            Lconv = 1 << ((3 * n - 2).bit_length())
            # compute rfft transforms
            fh = np.fft.rfft(h, Lconv)
            fw = np.fft.rfft(w, Lconv)
            fw_rev = np.fft.rfft(w[::-1], Lconv)
            # convolution: conv_bwd = irfft(fw * fh, Lconv)
            conv_bwd = np.fft.irfft(fw * fh, Lconv)
            conv_fwd = np.fft.irfft(fw_rev * fh, Lconv)
            offset = n - 1
            grad = 1.0 - (conv_fwd[offset:offset+n] + conv_bwd[offset:offset+n])
            # update
            h_new = h - step * grad
            # clip and enforce sum
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_sum(h_new, target)
            h = h_new
            # compute c5
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
            # optionally reduce step if no improvement over few iters? Could incorporate but may not be needed.

    return best_h, best_c

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # if diff > 0, need to increase some entries not at 1
    # if diff < 0, decrease some entries not at 0
    if diff > 0:
        while diff > 1e-12:
            free = np.where(h < 1 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            # compute max inc before hitting bound
            max_inc = np.min(1 - h[free])
            if inc > max_inc:
                inc = max_inc
            h[free] += inc
            diff = target - h.sum()
    else:
        while diff < -1e-12:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size  # negative
            max_dec = -np.min(h[free])  # most negative possible (down to 0)
            if dec < max_dec:
                dec = max_dec
            h[free] += dec
            diff = target - h.sum()
    # final clip to avoid tiny out-of-bound numerical errors
    h = np.clip(h, 0.0, 1.0)
    return h

# ----------------------------------------------------------------------
# Utility: project a vector onto the box [0,1] with exact prescribed sum
# ----------------------------------------------------------------------
def project_sum(h, target):
    """Clip to [0,1] and adjust to achieve exactly `target` sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # increase entries (diff>0) that are not at the upper bound
    it = 0
    while abs(diff) > 1e-12 and it < 200:
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            max_inc = np.min(1.0 - h[free])
            if inc > max_inc:
                inc = max_inc
            h[free] += inc
        else:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size          # diff is negative
            max_dec = -np.min(h[free])
            if dec < max_dec:
                dec = max_dec
            h[free] += dec
        diff = target - h.sum()
        it += 1
    return np.clip(h, 0.0, 1.0)

# Random feasible vector (entries in [0,1], sum = n/2)
# ----------------------------------------------------------------------
def random_feasible_vector(n, rng):
    target = n / 2.0
    h = rng.random(n)
    h = h / h.sum() * target
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, target)
    return h

def perturb_h(h, rng, scale=0.2):

    i = rng.integers(0, n)
    j = rng.integers(0, n)

    new_h[j] += eps
    return new_h

# Correlation utilities (FFT based, exact for the discrete grid)

def correlate_full(h):
    """Full linear correlation of h with (1‑h)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# Gradient descent on the smooth soft‑max surrogate of the max‑overlap
# ----------------------------------------------------------------------
def gradient_descent(h0, beta_sched, target, max_iters_per_beta, deadline, rng):
    """
    Projected gradient descent on
        f_beta(h) = (1/β) log Σ_k exp(β * corr_k(h))
    where corr_k are the overlap values.
    """
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    n = h.size
    # pre‑computed FFT length for convolutions (covers w * h)
    Lconv = 1 << ((3 * n - 2).bit_length())

    for beta in beta_sched:
        # step size decays with sqrt(beta); tuned experimentally
        step = 0.5 / np.sqrt(beta)
        for _ in range(max_iters_per_beta):
            if time.time() >= deadline:
                return best_h, best_c

            # correlation vector (overlap values)
            corr = correlate_full(h)

            # soft‑max weights
            m = corr.max()
            w = np.exp(beta * (corr - m))
            w /= w.sum()

            # FFT of h and w (and reversed w)
            fh = np.fft.rfft(h, Lconv)
            fw = np.fft.rfft(w, Lconv)
            fw_rev = np.fft.rfft(w[::-1], Lconv)

            # convolutions w * h and w_rev * h
            conv_bwd = np.fft.irfft(fw * fh, Lconv)
            conv_fwd = np.fft.irfft(fw_rev * fh, Lconv)

            off = n - 1
            grad = 1.0 - (conv_fwd[off:off + n] + conv_bwd[off:off + n])

            # projected gradient step
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_sum(h_new, target)

            h = h_new
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()

    return best_h, best_c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99  # safety margin

    # ------------------------------------------------------------------
    # 1. Base resolution and initial candidates
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_sum(h0, n / 2.0)
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    target = n / 2.0
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # hand‑crafted patterns
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    patterns = [h0]

    tri = 1.0 - np.abs(xs - 1.0)          # triangular bump
    patterns.append(project_sum(np.clip(tri, 0.0, 1.0), target))

    for f in (1, 2, 3):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        patterns.append(project_sum(np.clip(s, 0.0, 1.0), target))
        c = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        patterns.append(project_sum(np.clip(c, 0.0, 1.0), target))

    for _ in range(3):
        patterns.append(random_feasible_vector(n, rng))

    # keep the best of the pool
    for h in patterns:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ------------------------------------------------------------------
    # 2. Main optimisation loop – many gradient‑descent restarts
    # ------------------------------------------------------------------
    beta_sched = [1, 2, 5, 10, 20, 40, 80, 160, 320, 640, 1280]
    max_iters_per_beta = 250  # ~250 * 11 ≈ 2750 gradient steps per restart

    restart = 0
    while time.time() < deadline:
        # select start point: occasionally fresh random, otherwise perturb current best
        if restart % 4 == 0:
            start = random_feasible_vector(n, rng)
        else:
            start = perturb_h(best_h, rng, scale=0.3)

        h_opt, c_opt = gradient_descent(
            start,
            beta_sched,
            target,
            max_iters_per_beta,
            deadline,
            rng,
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        restart += 1
        # keep a small guard against endless looping as deadline approaches
        if time.time() + 5.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # 3. Optional fine‑graining (double the resolution if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline and n * 2 <= 1440:
        h_fine = np.repeat(best_h, 2)
        n_fine = h_fine.size
        target_fine = n_fine / 2.0
        h_fine = project_sum(h_fine, target_fine)

        # refined gradient descent on the finer grid
        beta_fine = [20, 40, 80, 160, 320, 640, 1280]
        h_opt_f, c_opt_f = gradient_descent(
            h_fine,
            beta_fine,
            target_fine,
            max_iters_per_beta=200,
            deadline=deadline,
            rng=rng,
        )
        if c_opt_f < best_c:
            best_h, best_c = h_opt_f, c_opt_f

    # ------------------------------------------------------------------
    # 4. Final clean‑up and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
