def compute_corr_full(x):
    n = x.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fx = np.fft.rfft(x, L)
    fr = np.fft.rfft(x[::-1], L)
    conv = np.fft.irfft(fx * fr, L)[: 2*n-1]
    return conv

def enforce_sum(x, target, rng):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # compute diff
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Perform simple water-filling iteration
    # We'll move mass among free slots in small increments.
    # Use while loop for up to 100 iterations
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(x < 1.0)[0]
            if free.size == 0:
                break
            # capacity
            cap = 1.0 - x[free]
            total_cap = cap.sum()
            add = min(diff, total_cap)
            # distribute proportionally
            x[free] += add * (cap / total_cap)
        else:
            free = np.where(x > 0.0)[0]
            if free.size == 0:
                break
            cap = x[free]
            total_cap = cap.sum()
            sub = min(-diff, total_cap)
            x[free] -= sub * (cap / total_cap)
        diff = target - x.sum()
    # final tiny adjustment
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute evenly
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        # if still off, use fallback random
        diff2 = target - x.sum()
        if abs(diff2) > 1e-8:
            # random adjustment
            for _ in range(1000):
                i = rng.integers(0, len(x))
                if diff2 > 0:
                    add = min(diff2, 1.0 - x[i])
                    x[i] += add
                    diff2 = target - x.sum()
                else:
                    sub = min(-diff2, x[i])
                    x[i] -= sub
                    diff2 = target - x.sum()
                if abs(diff2) < 1e-12:
                    break
    return x

def enforce_sum(x, target, rng):
    """Project x onto [0,1] and Σx = target (water‑filling)."""
    x = np.clip(x, 0.0, 1.0)

        diff = target - x.sum()
        if abs(diff) < 1e-12:

        if diff > 0:
            free = np.where(x < 1.0)[0]
            if free.size == 0:
                break
            cap = 1.0 - x[free]
            tot = cap.sum()
            inc = min(diff, tot)
            x[free] += inc * (cap / tot)

            free = np.where(x > 0.0)[0]
            if free.size == 0:
                break
            cap = x[free]
            tot = cap.sum()

            x[free] -= dec * (cap / tot)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # tiny uniform tweak + random fix

                x[i] += add
                diff -= add
            else:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x


def random_feasible_vector(n, rng):
    """Uniform random point in [0,1]^n with Σ = n/2."""
    x = rng.random(n)
    return enforce_sum(x, n / 2.0, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserves Σ)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- evaluation ----------
def compute_c5(h):
    """Exact discrete C5 for step values h (dx = 2/len)."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / h.size)


def corr_full(x):
    """Full linear correlation of x with itself."""
    n = x.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, L)
    conv = np.fft.irfft(fx * np.conj(fx), L)[: 2 * n - 1]
    return conv


# ---------- gradient based optimisation ----------
def gradient_optimize(start, beta_sched, max_iters, deadline, rng):
    """Projected gradient descent on L = log‑sum‑exp(‑β·corr)."""
    n = start.size
    target = n / 2.0
    x = enforce_sum(start, target, rng)
    best_h, best_c = x.copy(), compute_c5(x)

    for beta in beta_sched:
        base_step = 0.25 / np.sqrt(beta)  # heuristic scaling
        for it in range(max_iters):
            if time.time() >= deadline:
                return best_h, best_c
            r = corr_full(x)                     # correlation vector
            logw = -beta * r
            m = logw.max()
            w = np.exp(logw - m)
            w_sum = w.sum()
            a = w / w_sum
            a = (a + a[::-1]) * 0.5               # enforce symmetry
            L = a.size
            Lc = 1 << ((L + n - 1).bit_length())
            Fa = np.fft.rfft(a, Lc)
            Fx = np.fft.rfft(x, Lc)
            conv = np.fft.irfft(Fa * Fx, Lc)
            g = -2.0 * beta * conv[n - 1 : n - 1 + n]   # gradient of L
            gn = np.linalg.norm(g) + 1e-12
            step = - (base_step / gn) * g
            x_new = enforce_sum(np.clip(x + step, 0.0, 1.0), target, rng)
            # accept if L decreases
            r2 = corr_full(x_new)
            logw2 = -beta * r2
            m2 = logw2.max()
            L_new = m2 + np.log(np.exp(logw2 - m2).sum())
            L_cur = m + np.log(w_sum)
            if L_new <= L_cur:
                x = x_new
                c = compute_c5(x)
                if c < best_c:
                    best_c, best_h = c, x.copy()
            # occasional jitter
            if (it & 127) == 0 and rng.random() < 0.08:
                x = perturb_h(best_h, rng, scale=0.04)
        # end of beta
    return best_h, best_c


# ---------- binary simulated annealing ----------
def binary_sa(start, rng, deadline):
    n = start.size
    h = start.astype(float).copy()
    best_h, best_c = h.copy(), compute_c5(h)
    total_iters = 20000
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    for it in range(total_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / total_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n / 2.0, rng)
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # --------- phase 1: gradient descent ----------
    beta_schedule = [5, 20, 80, 200, 800, 2000, 5000]
    it_per_beta = 1200
    best_h, best_c = gradient_optimize(best_h, beta_schedule, it_per_beta, deadline, rng)

    # try a second random start if time permits
    if time.time() < deadline - 30:
        rnd_start = random_feasible_vector(n, rng)
        h2, c2 = gradient_optimize(rnd_start, beta_schedule, it_per_beta, deadline, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # --------- phase 2: continuous hill‑climb ----------
    hill_iters = 3000
    step = 0.2
    h_cur = best_h.copy()
    cur_c = best_c
    for it in range(hill_iters):
        if time.time() >= deadline:
            break
        h_new = perturb_h(h_cur, rng, scale=step)
        c = compute_c5(h_new)
        if c < cur_c:
            cur_c, h_cur = c, h_new
        if (it + 1) % (hill_iters // 10) == 0:
            step = max(step * 0.7, 1e-5)
    if cur_c < best_c:
        best_h, best_c = h_cur, cur_c

    # --------- phase 3: binary simulated annealing ----------
    bin_start = np.zeros(n)
    idx = np.argsort(-best_h)[: n // 2]
    bin_start[idx] = 1.0
    bin_h, bin_c = binary_sa(bin_start, rng, deadline)
    if bin_c < best_c:
        best_h, best_c = bin_h, bin_c

    # --------- optional up‑sampling (factor 2) ----------
    if time.time() < deadline and best_h.size * 2 <= 1440:
        h_fine = np.repeat(best_h, 2)
        n_fine = h_fine.size
        h_fine = enforce_sum(h_fine, n_fine / 2.0, rng)
        # brief gradient pass on fine grid
        beta_fine = [20, 80, 200]
        it_fine = 400
        h_fine, c_fine = gradient_optimize(
            h_fine, beta_fine, it_fine, deadline, rng
        )
        if c_fine < best_c:
            best_h, best_c = h_fine, c_fine

    # final clean‑up
    best_h = enforce_sum(np.clip(best_h, 0.0, 1.0), best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
