def hill_climb_targeted(h_start, rng, max_iters=2000, init_step=0.5, n_trials=4, p_target=0.25, decay_factor=0.7, min_step=1e-6):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        c, idx = compute_c5_and_idx(h)
        # idx shift receives current maximum location
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            r = rng.random()
            if r < p_target and 0 < k_star < n - 1:
                cand = targeted_move(h, rng, k_star, scale=step)
            elif r < p_target + 0.2:
                cand = swap_extreme(h, rng)
            else:
                cand = perturb_h(h, rng, scale=step)
            c_cand = compute_c5(cand)
            if c_cand < best_c - 1e-14:
                best_c, h = c_cand, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c_cand = compute_c5(cand)
            if c_cand < best_c:
                best_c, h = c_cand, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * decay_factor, min_step)
    return h, best_c

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        # fallback
        return perturb_h(h, rng, scale)
    # low_mask: both positions are below 0.5
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        # fallback
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    # Determine max epsilon for moving low positions up and high positions down.
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h

import time
import math
import numpy as np
from scipy.optimize import minimize

# ---------- projection ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) * 0.5, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y

# ---------- random vectors ----------
def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)

def random_binary_vector(n, rng):
    idx = rng.permutation(n)[: n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

# ---------- moves ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def swap_extreme(h, rng):
    low = np.where(h < 0.05)[0]
    high = np.where(h > 0.95)[0]
    if low.size == 0 or high.size == 0:
        return perturb_h(h, rng, scale=0.1)
    i = rng.choice(low)
    j = rng.choice(high)
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h

# ---------- correlation via FFT ----------
def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def compute_corr_fft(h):
    n = h.size
    nfft = _next_pow2(2 * n - 1)
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * G, nfft)[: 2 * n - 1]
    return conv.real

def compute_c5(h):
    conv = compute_corr_fft(h)
    return np.max(conv) * (2.0 / h.size)

def compute_c5_and_idx(h):
    conv = compute_corr_fft(h)
    idx = int(np.argmax(conv))
    return conv[idx] * (2.0 / h.size), idx

# ---------- hill climbing ----------
def hill_climb_targeted(
    h_start,
    rng,
    max_iters=2000,
    init_step=0.5,
    n_trials=4,
    p_target=0.25,
    decay_factor=0.7,
    min_step=1e-6,
):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    n = h.size
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        c, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        improved = False
        for _ in range(n_trials):
            r = rng.random()
            if r < p_target and 0 < k_star < n - 1:
                cand = targeted_move(h, rng, k_star, scale=step)
            elif r < p_target + 0.2:
                cand = swap_extreme(h, rng)
            else:
                cand = perturb_h(h, rng, scale=step)
            c_cand = compute_c5(cand)
            if c_cand < best_c - 1e-14:
                best_c, h = c_cand, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c_cand = compute_c5(cand)
            if c_cand < best_c:
                best_c, h = c_cand, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * decay_factor, min_step)
    return h, best_c

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)
