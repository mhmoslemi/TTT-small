import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the soft‑max SLSQP objective
CURRENT_TARGET = None
CURRENT_BETA = None

# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp (stabilised)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    eps = 1e-12
    it = 0
    while abs(diff) > eps and it < 10000:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
        it += 1
    if abs(diff) > eps:      # very unlikely, fix uniformly
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 12) -> np.ndarray:
    """Fine‑tune using the explicit worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


def simulated_annealing_binary(h_start: np.ndarray,
                               rng: np.random.Generator,
                               max_time: float):
    """Binary simulated‑annealing (swap 0/1) to minimise the max overlap."""
    n = h_start.size
    h = h_start.copy()
    cur_O = np.max(np.correlate(h, 1 - h, mode="full"))
    best_h = h.copy()
    best_O = cur_O

    start = time.time()
    it = 0
    # simple exponential cooling
    T_start = 5.0
    T_end = 0.01
    while time.time() - start < max_time:
        frac = (time.time() - start) / max_time
        T = T_start * ((T_end / T_start) ** frac)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap the bits
        h[i], h[j] = h[j], h[i]
        new_O = np.max(np.correlate(h, 1 - h, mode="full"))
        delta = new_O - cur_O
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_O = new_O
            if new_O < best_O:
                best_O = new_O
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        it += 1
    return best_h, best_O

    # allocate roughly half the time budget to SA
    sa_time = budget_s * 0.55
    h_best_bin, O_best = simulated_annealing_binary(h_bin, rng, sa_time)
    c_best_bin = (2.0 / n) * O_best
    if c_best_bin < best_c:
        best_c = c_best_bin
        best_h = h_best_bin.copy()
        best_n = n

    # --------------------------------------------------------------
    # Fractional refinement (soft‑max SLSQP + gradient tidy‑up)
    time_left = budget_s - (time.time() - start)
    if time_left > budget_s * 0.25 and best_h is not None:
        h_float = best_h.astype(float)
        h_float = enforce_sum(h_float, rng)

        betas = [5., 20., 80., 200., 800., 2000., 8000.]
        for beta in betas:
            if time.time() - start > budget_s * 0.95:
                break
            maxit = 400 if beta < 200 else 800
            h_float = slsqr_optimize := slsqp_optimize(h_float, beta, rng, max_iter=maxit)
            c = compute_c5(h_float)
            if c < best_c:
                best_c, best_h, best_n = c, h_float.copy(), n

        # final gradient tidy‑up
        h_float = gradient_refine(best_h, rng, step=0.01, max_iter=12)
        c = compute_c5(h_float)
        if c < best_c:
            best_c, best_h = c, h_float.copy()

    # --------------------------------------------------------------
    # Final sanity enforcement and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
