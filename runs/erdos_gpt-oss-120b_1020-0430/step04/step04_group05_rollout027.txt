def enforce_sum(h, target):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Determine free indices that are not at bounds
    # We'll apply iterative redistribution
    max_iter = 100
    for _ in range(max_iter):
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:
            # No free entries, fallback: distribute to all and clip again
            delta = diff / h.size
            h += delta
            h = np.clip(h, 0, 1)
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
            continue
        delta = diff / n_free
        # Add delta to free entries
        h[free] += delta
        # Clip again
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    # Final adjust if small diff remains: add to first free entry
    if abs(diff) > 1e-12:
        # distribute residual random
        for i in range(h.size):
            if diff == 0:
                break
            max_add = (1.0 - h[i]) if diff > 0 else -h[i]
            add = min(max_add, diff) if diff > 0 else max(max_add, diff)
            h[i] += add
            diff = target - h.sum()
    return h
```

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    if lag >= n:
        return perturb_h(h, rng, scale)
    # compute contributions h[i]*(1 - h[i+lag])
    i_vals = np.arange(n - lag)
    contrib = h[i_vals] * (1.0 - h[i_vals + lag])
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(contrib) * top_frac))
    # get indices of top k contributions
    top_idxs = i_vals[np.argpartition(-contrib, k-1)[:k]]
    # bottom k
    bot_idxs = i_vals[np.argpartition(contrib, k-1)[:k]]
    i = rng.choice(top_idxs)
    # choose j from bottom or random
    if rng.random() < 0.7:
        j = rng.choice(bot_idxs)
    else:
        j = rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    T0 = 0.05 * (n / 360) ** 0.5
    Tf = 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        # worst lag
        idx = np.argmax(cur_corr)
        lag = idx - (n - 1)
        lag = abs(lag)
        if rng.random() < 0.65:
            new_h = targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def enforce_sum(h, target):
    """Project h onto [0,1] and enforce sum == target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # distribute uniformly as fallback
            delta = diff / h.size
            h += delta
            h = np.clip(h, 0.0, 1.0)
        else:
            delta = diff / n_free
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # tiny residual correction
    if abs(diff) > 1e-12:
        for i in range(h.size):
            if abs(diff) < 1e-12:
                break
            max_add = (1.0 - h[i]) if diff > 0 else -h[i]
            add = max_add if diff > 0 else max_add
            add = min(add, diff) if diff > 0 else max(add, diff)
            h[i] += add
            diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0)

def compute_c5_corr(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = np.max(corr) * (2.0 / h.size)
    return c, corr

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    if lag >= n or lag == 0:
        return perturb_h(h, rng, scale)
    idx = np.arange(n - lag)
    contrib = h[idx] * (1.0 - h[idx + lag])
    k = max(1, int(len(contrib) * top_frac))
    top = idx[np.argpartition(-contrib, k - 1)[:k]]
    bot = idx[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top)
    j = rng.choice(bot) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0 = 0.05 * (n / 360.0) ** 0.5
    Tf = 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        lag = abs(np.argmax(cur_corr) - (n - 1))
        if rng.random() < 0.65:
            new_h = targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):

    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta

    cons = {"type": "eq", "fun": sum_eq_constraint}

        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET)
    return x
