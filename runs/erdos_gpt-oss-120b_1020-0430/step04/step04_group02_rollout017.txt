def swap_move(h, rng, eps_scale=0.2):
    n = h.size
    low_idx = np.where(h < 0.5)[0]
    high_idx = np.where(h > 0.5)[0]
    if low_idx.size == 0 or high_idx.size == 0:
        return h.copy()
    i = rng.choice(low_idx)
    j = rng.choice(high_idx)
    max_eps = min(0.5 - h[i], h[j] - 0.5)  # ensure after move still within [0,1]
    max_eps = min(max_eps, eps_scale)  # bound
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps
    new_h = h.copy()
    new_h[i] += eps
    new_h[j] -= eps
    return new_h

def hill_climb(h_start, rng, max_iters=5000, init_step=0.4, target_moves_prob=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    T = 1e-4
    decay = 0.995
    n = h.shape[0]
    for it in range(max_iters):
        # compute worst shift
        corr = np.convolve(h, (1 - h)[::-1], mode='full')
        k_star = np.argmax(corr) - (n - 1)
        # propose move
        if rng.random() < target_moves_prob:
            cand = targeted_move(h, rng, k_star, scale=step)
        else:
            cand = swap_move(h, rng, eps_scale=step)
        # Ensure cand stays within bounds (targeted_move takes care)
        c = compute_c5(cand)
        # accept if better or with probability
        if c < best_c - 1e-12:
            best_c, best_h = c, cand.copy()
            h = cand
        else:
            # SA acceptance
            if T > 1e-12 and rng.random() < math.exp(-(c - best_c) / T):
                h = cand
        # update temperature and step
        T *= decay
        if (it + 1) % 200 == 0:
            step = max(step * 0.9, 1e-5)
    return best_h, best_c

def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale=scale)
    # compute masks
    low_mask = (h[:n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[:n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale=scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        1.0 - h[i_low],
        1.0 - h[i_low + k],
        h[i_high],
        h[i_high + k],
    )
    if max_eps <= 0:
        return perturb_h(h, rng, scale=scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    # Clip for safety
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h
```

def perturb_h(h, rng, scale=0.2):

def slsqp_optimize(h0, beta, max_iter=300):
    n = h0.size
    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target
    def obj(x):
        corr = np.convolve(x, (1 - x)[::-1], mode='full')
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq}
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # Ensure sum exactly
    return project_box_sum(x, target)
```

import time
import math

import numpy as np
from scipy.optimize import minimize

# -----------------------------------------------------------
# Helper functions – must be top‑level
# -----------------------------------------------------------

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Project x onto [lo,hi] while enforcing sum(x)=target (water‑filling)."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1] with sum = n/2."""
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Move a random amount from one coordinate to another."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    """Increase two low values at distance k and decrease two high values at the same distance."""
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale=scale)

    # work with positive shift
    k_abs = k
    # low–low and high–high masks
    low_mask = (h[: n - k_abs] < 0.5) & (h[k_abs:] < 0.5)
    high_mask = (h[: n - k_abs] > 0.5) & (h[k_abs:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale=scale)

    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))

    max_eps = min(
        1.0 - h[i_low],
        1.0 - h[i_low + k_abs],
        h[i_high],
        h[i_high + k_abs],
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale=scale)

    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k_abs] += eps
    new_h[i_high] -= eps
    new_h[i_high + k_abs] -= eps
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h


def swap_move(h, rng, eps_scale=0.2):
    """Swap a small amount between a low (<0.5) and a high (>0.5) entry."""
    low_idx = np.where(h < 0.5)[0]
    high_idx = np.where(h > 0.5)[0]
    if low_idx.size == 0 or high_idx.size == 0:
        return h.copy()
    i = rng.choice(low_idx)
    j = rng.choice(high_idx)
    max_eps = min(0.5 - h[i], h[j] - 0.5, eps_scale)
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps
    new_h = h.copy()
    new_h[i] += eps
    new_h[j] -= eps
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h


def compute_c5(h):
    """Maximum overlap integral for the discretised function h."""
    corr = np.convolve(h, (1.0 - h)[::-1], mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_idx(h):
    """Return (max_overlap, index_of_max) for the full correlation."""
    corr = np.convolve(h, (1.0 - h)[::-1], mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


def hill_climb(
    h_start,
    rng,
    max_iters=2000,
    init_step=0.4,
    target_moves_prob=0.4,
):
    """Simulated‑annealing hill climbing with targeted moves."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    T = 1e-4
    decay_T = 0.995
    n = h.size

    for it in range(max_iters):
        # worst shift
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)
        k_abs = abs(k_star)

        # propose a move
        if rng.random() < target_moves_prob:
            cand = targeted_move(h, rng, k_abs, scale=step)
        else:
            cand = swap_move(h, rng, eps_scale=step)

        c = compute_c5(cand)

        if c < best_c - 1e-12:
            best_c, best_h = c, cand.copy()
            h = cand
        else:
            # occasional uphill acceptance
            if T > 1e-12 and rng.random() < math.exp(-(c - best_c) / T):
                h = cand

        # schedule decay
        T *= decay_T
        if (it + 1) % 200 == 0:
            step = max(step * 0.9, 1e-5)

    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """Smooth max‑overlap optimisation via SLSQP (softmax approximation)."""
    n = h0.size
    target = n / 2.0

    def sum_eq(x):
        return np.sum(x) - target

    def obj(x):
        corr = np.convolve(x, (1.0 - x)[::-1], mode="full")
        bc = beta * corr
        m = np.max(bc)
        # log‑sum‑exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq}
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, target)


def upsample_h(h, new_n):
    """Linear interpolation to a finer uniform grid."""
    old_n = h.size
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_box_sum(h_new, new_n / 2.0)
