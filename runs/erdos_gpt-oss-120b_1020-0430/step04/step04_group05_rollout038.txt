def compute_corr_fft(h):
    n = h.shape[0]
    size = 2 * n
    # Use real FFT for speed
    H = np.fft.rfft(np.concatenate([h, np.zeros(n)]))
    Hc = np.fft.rfft(np.concatenate([1 - h, np.zeros(n)]))
    corr = np.fft.irfft(H * Hc, n=size)
    # The correlation length is 2n-1, last element is zero due to padding.
    return corr[:2 * n - 1]

def compute_c5_corr_fft(h):
    corr = compute_corr_fft(h)
    n = h.shape[0]
    c5 = np.max(corr) * (2.0 / n)
    return c5, corr

def compute_c5_fft(h):
    # compute only max

    # For speed, we can compute correlation via FFT and then max; reuse compute_corr_fft
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / n)

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    if lag <= 0 or lag >= n:
        return perturb_h(h, rng, scale)
    if lag > 0:
        valid = np.arange(n - lag)
        a = h[valid]
        b = 1 - h[valid + lag]
    else:
        # negative lag: convert to positive
        lag = -lag
        valid = np.arange(lag, n)
        a = h[valid]
        b = 1 - h[valid - lag]
    contrib = a * b
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    # indices within original h: top contributions correspond to positions in 'valid'
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    # i from top_idx, j from bot_idx
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def sum_eq(x):
        return np.sum(x) - target
    cons = {"type": "eq", "fun": sum_eq}
    def softmax_obj(x):
        # compute correlation via FFT
        corr = compute_corr_fft(x)
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust using enforce_sum
        x = enforce_sum(x, target, rng)
    return x

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=100):
    n = h0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def sum_eq(x):
        return np.sum(x) - target
    cons = {"type": "eq", "fun": sum_eq}
    def softmax_obj(x):
        corr = compute_corr_fft(x)
        bc = beta * corr
        m = np.max(bc)
        # logsumexp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target, rng)
    return x

def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:

        h += uniform
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):

                break
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            if add == 0:
                continue
            h[i] += add
            diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    if lag <= 0 or lag >= n:
        return perturb_h(h, rng, scale)
    if lag > 0:
        valid = np.arange(n - lag)
        a = h[valid]
        b = 1 - h[valid + lag]
    else:  # negative lag
        lag = -lag
        valid = np.arange(lag, n)
        a = h[valid]
        b = 1 - h[valid - lag]
    contrib = a * b
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def compute_corr_fft(h):
    n = h.shape[0]
    size = 2 * n
    h_pad = np.concatenate([h, np.zeros(n, dtype=h.dtype)])
    rev = np.concatenate([(1 - h)[::-1], np.zeros(n, dtype=h.dtype)])
    H = np.fft.rfft(h_pad)
    R = np.fft.rfft(rev)
    conv = np.fft.irfft(H * R, n=size)
    return conv[:2 * n - 1]

def compute_c5_corr_fft(h):
    corr = compute_corr_fft(h)
    n = h.shape[0]
    return np.max(corr) * (2.0 / n), corr

def compute_c5_fft(h):

    return np.max(compute_corr_fft(h)) * (2.0 / n)
