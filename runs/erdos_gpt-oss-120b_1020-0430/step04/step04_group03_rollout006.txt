def optimal_transfer(h, i, j, O_s, max_iter=30):
    max_eps = min(h[i], 1 - h[j])
    if max_eps < 1e-12:
        return None, None
    n = h.size
    m = 2*n - 1
    a = np.empty(m, dtype=float)
    # compute a by iterating over shift index k
    # shift s = k - (n-1)
    # use local alias h for speed
    for k in range(m):
        s = k - (n-1)
        val = 0.0
        # contribution for j
        idx = j + s
        if 0 <= idx < n:
            val += 1 - h[idx]
        idx2 = j - s
        if 0 <= idx2 < n:
            val -= h[idx2]
        # contribution for i subtract
        idx = i + s
        if 0 <= idx < n:
            val -= 1 - h[idx]
        idx2 = i - s
        if 0 <= idx2 < n:
            val += h[idx2]
        a[k] = val
    # ternary search
    lo, hi = 0.0, max_eps
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(O_s + a * m1)
        f2 = np.max(O_s + a * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    eps_opt = (lo + hi) / 2.0
    new_max = np.max(O_s + a * eps_opt)
    if new_max + 1e-12 < np.max(O_s):
        return eps_opt, new_max
    else:
        return None, None

def enforce_sum(h, target, rng):
    """Clip to [0,1] and adjust to make sum exactly target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:

        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
