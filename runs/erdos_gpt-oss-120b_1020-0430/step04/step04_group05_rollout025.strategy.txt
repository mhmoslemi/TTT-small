   def enforce_sum(h, target):
        # Clip and adjust sum
        clip_min = np.min(h)
        clip_max = np.max(h)
        if clip_min < 0 or clip_max > 1:
            raise ValueError("Values out of bounds")
        s = np.sum(h)
        diff = target - s
        eps = 0.1 if diff > 0 else -0.1
        if 0 < eps < diff:
            h += eps
        elif diff < 0 and eps < diff:
            h = h + eps
        return h

   def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        n = len(h)
        max_corr_val = np.max(corr)
        scale_factor = (2.0 / n) * max_corr_val
        return scale_factor

   def global_search(h0, n_grid, rng, T0, Tf, max_iters):
        h = h0.copy()
        curr_c5 = compute_c5(h)
        for i in range(max_iters):

        return h, curr_c5

        x = h0.copy()

        res = minimize(compute_c5, x, method="SLSQP", bounds=bounds, constraints=cons)
        return res.x if res.success else x

   def interpolate_solution(old_h, new_n):

        old_grid = np.linspace(0, 2, old_n)
        new_h = np.interp(new_grid, old_grid, old_h)
        new_h = enforce_sum(new_h, new_n/2)
        return new_h

   - Cooldown schedule: `Tf = 0.8^stages` ensures slow cooling in the simmer annealing phase.  
   **Variants**:  
   - **Gradient Boosting Variant**: Replace SLSQP with gradient descent + line search to handle higher dimensions (though not implemented).  
   - **Multi-start Variant**: If a parent is provided, use multiple interpolated copies for initial search.
