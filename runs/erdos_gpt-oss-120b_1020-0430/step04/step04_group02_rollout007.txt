import time
import numpy as np
from scipy.optimize import minimize

# ----- basic helpers ---------------------------------------------------------

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Clip to [lo,hi] after adding a constant so that sum == target."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    # final tiny fix if needed
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Move a small amount of mass between two random positions."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_c5(h):
    """Return C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ----- SLSQP surrogate -------------------------------------------------------

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    """Smooth max minimisation via SLSQP."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ----- hill‑climbing with worst‑shift targeting ------------------------------

def hill_climb_wtarget(h_start, rng, max_iters=2000, step=0.15):
    """Local search that transfers mass from low‑weight to high‑weight sites."""
    h = h_start.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = 0.9995
    for _ in range(max_iters):
        # worst shift
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr_full))
        k_star = idx - (n - 1)

        # weight w[i] = h[i+k] + h[i-k] (zero outside)
        w = np.zeros(n)
        if k_star > 0:
            w[k_star:] += h[: n - k_star]
            w[: n - k_star] += h[k_star:]
        elif k_star < 0:
            kk = -k_star
            w[: n - kk] += h[kk:]
            w[kk:] += h[: n - kk]
        else:
            w = 2.0 * h

        # pick donors and receivers
        low_thr = np.percentile(w, 30)
        high_thr = np.percentile(w, 70)
        low_idxs = np.where(w <= low_thr)[0]
        high_idxs = np.where(w >= high_thr)[0]

        if rng.random() < 0.2 or low_idxs.size == 0 or high_idxs.size == 0:
            cand = perturb_h(h, rng, scale=step)
        else:
            i_from = int(rng.choice(low_idxs))
            i_to = int(rng.choice(high_idxs))
            max_eps = min(h[i_from], 1.0 - h[i_to])
            if max_eps <= 0.0:
                cand = perturb_h(h, rng, scale=step)
            else:
                eps = step * max_eps * rng.random()
                if eps <= 0.0:
                    cand = perturb_h(h, rng, scale=step)
                else:
                    cand = h.copy()
                    cand[i_from] -= eps
                    cand[i_to] += eps

        c = compute_c5(cand)
        if c < best_c - 1e-12:
            best_c = c
            h = cand
        step = max(step * decay, 1e-5)
    return h, best_c
