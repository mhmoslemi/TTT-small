def compute_overlap_fft(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())  # next power of two for speed
    H = np.fft.rfft(h, L)
    conv = np.fft.irfft(H * H, L)  # real convolution
    # conv length L, first 2n-1 entries are convolution.
    # overlap array for shift k = conv[n-1 + k] (k=0..n-1)
    overlap = conv[n-1 : n-1 + n]
    return overlap[:n] # shape (n,)

def optimize_h(h0, rng, max_iter=2000, init_step=0.5, step_decay=0.998, stop_patience=200):
    n = h0.size
    target = n/2
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    patience = 0
    for it in range(max_iter):
        overlap = compute_overlap_fft(h)
        min_overlap = overlap.min()
        k = int(overlap.argmin())
        c = 1 - min_overlap * (2.0 / n)
        # update best
        if c < best_c - 1e-9:
            best_c = c
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        # check early stop
        if patience >= stop_patience:
            break
        # compute gradient
        g = grad_for_shift(h, k)
        # step
        h = h - step * g
        # clip
        np.clip(h, 0.0, 1.0, out=h)
        # adjust sum
        diff = target - h.sum()
        # while loop adjust
        for _ in range(500):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                # need to increase some entry not at 1
                candidates = np.where(h < 1 - 1e-12)[0]
                if candidates.size == 0:
                    break
                i = rng.choice(candidates)
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                candidates = np.where(h > 1e-12)[0]
                if candidates.size == 0:
                    break
                i = rng.choice(candidates)
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
        step *= step_decay
    # final evaluate best_h
    final_c = compute_c5(best_h)
    return best_h, final_c

def adjust_sum_exact(h, target, rng):
    diff = target - h.sum()
    n = len(h)
    # limit iterations
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # find indices with h < 1
            idx = rng.integers(0, n)
            # compute capacity
            cap = 1.0 - h[idx]
            if cap <= 0:
                continue
            add = min(diff, cap)
            h[idx] += add
            diff -= add
        else:
            idx = rng.integers(0, n)
            if h[idx] <= 0:
                continue
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    # if still diff, distribute uniformly small amounts
    if abs(diff) > 1e-9:
        # distribute proportionally to slack
        slack_up = 1 - h
        slack_down = h
        if diff > 0:
            total = slack_up.sum()
            if total > 0:
                h += diff * slack_up / total
        else:
            total = slack_down.sum()
            if total > 0:
                h += diff * (-slack_down) / total
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_overlap_fft(h):
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    conv = np.fft.irfft(H * H, L)
    overlap = conv[n - 1 : n - 1 + n]
    return np.clip(overlap, 0.0, None)


def grad_for_shift(h, k):
    n = h.size
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.ones(n)
    upper = n - k
    g[:upper] -= h[k:]
    g[k:] -= h[:upper]
    return g


def adjust_sum_exact(h, target, rng):
    diff = target - h.sum()
    n = h.size
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            i = rng.integers(0, n)
            cap = 1.0 - h[i]
            if cap <= 0:
                continue
            add = min(diff, cap)
            h[i] += add
            diff -= add
        else:
            i = rng.integers(0, n)
            if h[i] <= 0:
                continue
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    if abs(diff) > 1e-9:
        slack_up = 1.0 - h
        slack_dn = h
        if diff > 0 and slack_up.sum() > 0:
            h += diff * slack_up / slack_up.sum()
        elif diff < 0 and slack_dn.sum() > 0:
            h -= (-diff) * slack_dn / slack_dn.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def optimize_h(h0, rng, max_iter=2000, init_step=0.8, step_decay=0.999, stop_patience=300):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = init_step
    patience = 0
    for it in range(max_iter):
        overlap = compute_overlap_fft(h)
        k = int(np.argmin(overlap))
        c = 1.0 - overlap[k] * (2.0 / n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience >= stop_patience:

        g = grad_for_shift(h, k)
        h -= step * g
        np.clip(h, 0.0, 1.0, out=h)
        h = adjust_sum_exact(h, target, rng)
        step *= step_decay
    final_c = compute_c5(best_h)
    return best_h, final_c
