import time, math
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------
def enforce_sum(v, target, rng):
    """Adjust v so that sum(v)=target while keeping 0<=v<=1."""
    diff = target - v.sum()
    n = v.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - v[i]) if diff > 0 else max(diff, -v[i])
        v[i] += add
        diff = target - v.sum()
    return v


def random_feasible_vector(n, rng):
    """Continuous random feasible vector (sum=n/2, 0<=h<=1)."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)


def random_binary(n, rng):
    """Binary vector with exactly n/2 ones."""
    b = np.zeros(n, dtype=float)
    b[rng.choice(n, n // 2, replace=False)] = 1.0
    return b


def binary_from_continuous(h):
    """Return binary vector (top half entries set to 1)."""
    n = h.size
    b = np.zeros(n, dtype=float)
    b[np.argsort(-h)[: n // 2]] = 1.0
    return b


# --------------------------------------------------------------
# Correlation helpers (FFT based)
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Smooth surrogate for SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --------------------------------------------------------------
# Simple perturbation / hill climbing (continuous)
# --------------------------------------------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# --------------------------------------------------------------
# Binary simulated annealing (long run)
# --------------------------------------------------------------
def binary_sa_long(start_bin, rng, deadline):
    """SA on binary vectors with swap moves, runs until deadline."""
    b = start_bin.astype(float).copy()
    n = b.size
    cur_c = compute_c5(b)
    best_b = b.copy()
    best_c = cur_c
    t0 = time.time()
    total = max(deadline - t0, 1e-9)
    T0 = max(0.5, cur_c)          # generous initial temperature
    Tf = 1e-8
    while True:
        now = time.time()
        if now >= deadline:
            break
        frac = (now - t0) / total
        T = T0 * (Tf / T0) ** frac
        i, j = rng.integers(0, n, size=2)
        if i == j or b[i] == b[j]:
            continue
        # ensure i is a 1, j a 0
        if b[i] < b[j]:
            i, j = j, i
        b[i] = 0.0
        b[j] = 1.0
        new_c = compute_c5(b)
        delta = new_c - cur_c
        if new_c < best_c:
            best_c, best_b = new_c, b.copy()
        if delta <= 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_c
