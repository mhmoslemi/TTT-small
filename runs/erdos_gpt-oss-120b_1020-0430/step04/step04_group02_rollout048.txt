import time

# ------------------------------------------------------------
# Projection onto the box [0,1] with exact sum constraint
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)

# Evaluation of C5 for a discrete vector
def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

# Simple random perturbation (swap epsilon between two coordinates)
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# Targeted move that tries to raise the product for the worst shift
def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)

    # compute products for shift k
    prod = h[: n - k] * h[k:]
    low_idx = np.argmin(prod)                     # smallest product
    high_idx = np.argmax(prod)                    # largest product

    # ensure we can move both pairs without violating bounds
    max_eps_low = min(1.0 - h[low_idx], 1.0 - h[low_idx + k])
    max_eps_high = min(h[high_idx], h[high_idx + k])
    max_eps = min(max_eps_low, max_eps_high)
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)

    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = np.clip(eps, -max_eps, max_eps)

    nh = h.copy()
    nh[low_idx] += eps
    nh[low_idx + k] += eps
    nh[high_idx] -= eps
    nh[high_idx + k] -= eps
    return nh

                        n_trials=4, p_target=0.25):

    n = h.shape[0]

    for it in range(max_iters):

        idx = int(np.argmax(corr))

    return h, best_c

# Smoothed objective (log‑sum‑exp) and its gradient
def smooth_obj_and_grad(h, beta):
    # correlation h * (1-h) with zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    z = beta * corr
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    sum_exp = exp_z.sum()
    w = exp_z / sum_exp                      # soft‑max weights, length 2n‑1

    # gradient = conv(w_rev, 1-h) - conv(w, h)
    w_rev = w[::-1]
    conv1 = fftconvolve(w_rev, (1.0 - h), mode="full")
    conv2 = fftconvolve(w, h, mode="full")
    off = h.size - 1
    grad = conv1[off:off + h.size] - conv2[off:off + h.size]

    # smooth objective value
    f = (max_z + np.log(sum_exp)) / beta
    return f, grad, corr, w

    for _ in range(max_iters):
        f, g, _, _ = smooth_obj_and_grad(h, beta)
        if f < best_f:
            best_f, best_h = f, h.copy()
        # simple projected step
        h_new = h - step * g
        h_new = project_box_sum(h_new, target)
        h = h_new
        step *= decay
        if step < 1e-9:
            break
    return best_h, best_f


# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_t = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n / 2.0)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    # ----- generate a small pool of diverse starters ------------------
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    starters = [h0]
    for f in (1, 2, 3):
        s = 0.5 + 0.5 * np.sin(np.pi * f * xs)
        starters.append(project_box_sum(np.clip(s, 0.0, 1.0), n / 2.0))
        s = 0.5 + 0.5 * np.cos(np.pi * f * xs)
        starters.append(project_box_sum(np.clip(s, 0.0, 1.0), n / 2.0))
    for _ in range(5):
        starters.append(random_feasible_vector(n, rng))

    best_h = None
    best_c = np.inf
    for h in starters:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- core refinement (smooth GD + short hill‑climb) ----------
    beta_schedule = [10, 50, 200, 500, 1000, 2000, 5000, 10000]
    def refine(vec):
        nonlocal best_h, best_c, start_t
        cur = vec.copy()
        for beta in beta_schedule:
            if time.time() - start_t > budget_s * 0.75:
                break
            # gradient descent on smooth proxy
            max_iter = 250 if beta <= 500 else 150
            step0 = 0.6 / np.sqrt(beta)
            cur, _ = gradient_descent_smooth(cur, beta, max_iter,
                                             step0, 0.98, rng)

            # evaluate true C5
            c = compute_c5(cur)
            if c < best_c:
                best_c, best_h = c, cur.copy()

            # brief hill‑climb to polish
            if time.time() - start_t < budget_s * 0.75:
                cur, c2 = hill_climb_targeted(cur, rng,
                                              max_iters=200,
                                              init_step=0.3,
                                              n_trials=2,
                                              p_target=0.3)
                if c2 < best_c:
                    best_c, best_h = c2, cur.copy()
        return cur

    best_h = refine(best_h)

    # ----- optional up‑sampling ---------------------------------------
    elapsed = time.time() - start_t
    if budget_s - elapsed > budget_s * 0.2:
        n_up = min(720, max(540, best_h.size * 2))
        if n_up != best_h.size:
            old_x = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
            new_x = (np.arange(n_up) + 0.5) * (2.0 / n_up)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_box_sum(h_up, n_up / 2.0)
            best_h = refine(h_up)

    # ----- final clean‑up ---------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
