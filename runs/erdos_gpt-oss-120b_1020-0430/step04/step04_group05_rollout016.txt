def compute_corr_fft(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(np.flip(1-h), n=L)
    corr = np.fft.irfft(H*G, n=L)[:2*n-1]
    return corr

def compute_c5_fft(h):
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / h.size)

def enforce_sum(h, target):
    # ensure bounds 0-1 then adjust exactly.
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:

        total_slack = slack.sum()
        if total_slack > 0:

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(h[j], 1 - h[i])
        if max_eps <= 0:
            return h.copy()
        # swap i and j
        i, j = j, i
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # ensure bounds maybe via clip, though eps ensures within bounds
    return new_h

def simulated_annealing(h0, rng, max_iters, time_start, time_budget):

    T0, Tf = 0.05, 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        # exponential cooling
        T = T0 * (Tf / T0) ** (it / max_iters)
        # choose operation: maybe targeted 60% else random

import time
import numpy as np
from scipy.optimize import minimize

# ---------- global for SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

# ---------- utility functions ----------
def enforce_sum(h, target, rng=None):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(5):
        if diff > 0:
            slack = 1.0 - h
            s = slack.sum()
            if s > 0:
                h += diff * slack / s
        else:
            avail = h
            s = avail.sum()
            if s > 0:
                h -= (-diff) * avail / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    # fallback random walk if still off
    if rng is not None and abs(diff) > 1e-12:
        n = h.size
        for _ in range(1000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def compute_corr_fft(h):
    """Full correlation of h with (1‑h) using FFT."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(np.flip(1.0 - h), n=L)
    corr = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    return corr

def compute_c5_fft(h):
    """C5 value for step vector h."""
    return np.max(compute_corr_fft(h)) * (2.0 / h.size)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        i, j = j, i
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    lag = int(abs(lag))
    if lag >= n:
        return perturb_h(h, rng, scale)
    valid = np.arange(n - lag)
    contrib = h[valid] * (1.0 - h[valid + lag])
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    n = h.size
    corr = compute_corr_fft(h)
    cur_c = np.max(corr) * (2.0 / n)
    best_h, best_c = h.copy(), cur_c
    T0, Tf = 0.05, 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        lag = np.argmax(corr) - (n - 1)
        if rng.random() < 0.6:
            new_h = targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_corr = compute_corr_fft(new_h)
        new_c = np.max(new_corr) * (2.0 / n)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = compute_corr_fft(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

def upsample(old_h, new_n, rng):
    old_n = old_h.size
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    new_h = np.interp(x_new, x_old, old_h)
    return enforce_sum(new_h, new_n / 2.0, rng)

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, h0.size / 2.0, rng)
    else:
        h0 = random_feasible_vector(360, rng)

    best_h = h0
    best_c = compute_c5_fft(best_h)

    # grid progression
    grid_sizes = [best_h.size, best_h.size * 2, best_h.size * 4]
    grid_sizes = [n for n in grid_sizes if n <= 2000]

    for n_target in grid_sizes:
        if time.time() - t0 > budget_s * 0.95:
            break
        if best_h.size != n_target:
            best_h = upsample(best_h, n_target, rng)
            best_c = compute_c5_fft(best_h)

        # Simulated annealing
        max_iters = max(4000, int(20000 * 360 / n_target))
        remaining = budget_s - (time.time() - t0)
        sa_budget = min(remaining * 0.5, remaining - 5)
        h_sa, c_sa = simulated_annealing(best_h, rng, max_iters, t0, t0 + sa_budget)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

        # SLSQP refinement over a ladder of betas
        for beta in [5, 20, 80, 200, 500, 1000, 2000, 5000]:
            if time.time() - t0 > budget_s * 0.97:
                break
            try:
                ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=300)
                rc = compute_c5_fft(ref_h)
                if rc < best_c:
                    best_h, best_c = ref_h, rc
            except Exception:
                continue

    # final local hill‑climb
    cur_h = best_h.copy()
    cur_c = best_c
    step = 0.3
    for _ in range(6000):
        if time.time() - t0 > budget_s:
            break
        new_h = perturb_h(cur_h, rng, scale=step)
        new_c = compute_c5_fft(new_h)
        if new_c < cur_c:
            cur_c, cur_h = new_c, new_h
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
        step *= 0.9997

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_fft(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
