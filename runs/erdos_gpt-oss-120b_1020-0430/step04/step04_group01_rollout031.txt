        term[mask] += h[idx[mask]]

        idx = j + s_range
        mask = (idx >= 0) & (idx < n)
        term[mask] -= h[idx[mask]]

        term[offset] -= 2
        diff = i - j
        term[diff + offset] += 1
        term[-diff + offset] += 1

        new_max = int(np.max(conv + term))
        delta = new_max - curr_max

        # temperature schedule
        t_frac = (time.time() - start) / (deadline - start)
        T = T0 * ((Tf / T0) ** t_frac)

        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept swap
            h[i] = 0
            h[j] = 1
            A[i] = 1
            A[j] = 0
            conv += term
            curr_max = new_max
            if curr_max < best_max:
                best_max = curr_max
                best_h = h.copy()

    # ----- final recomputation of bound --------------------------------
    final_conv = np.rint(correlate_full(best_h.astype(float))).astype(np.int32)
    final_max = int(final_conv.max())
    final_c = final_max * (2.0 / n)

    return best_h.astype(float).tolist(), float(final_c), int(n)
