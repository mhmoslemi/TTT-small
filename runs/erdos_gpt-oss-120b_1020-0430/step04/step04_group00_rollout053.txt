def _softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = _CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / _CURRENT_BETA

def _slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max (log‑sum‑exp) surrogate."""
    global _CURRENT_TARGET, _CURRENT_BETA
    n = h0.size
    _CURRENT_TARGET = n / 2.0
    _CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},

                if abs(diff) < 1e-12:
                    break
    return h

# ---- linear‑programming refinement -----------------------------------------

def _lp_refine(h0, target, rng, deadline, max_iters=15):
    """Successive linear‑programming based upper‑bound tightening."""
    if not _HAS_CVX:
        return h0, compute_c5(h0)

    n = h0.size
    h = np.clip(h0, 0.0, 1.0)
    adjust_sum_exact(h, rng)
    best_h = h.copy()
    best_c = compute_c5(h)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # compute overlap O_k and the linearisation constants
        O = np.empty(2 * n - 1)
        G = []                     # list of (gk, const) for each shift
        S = target                 # exact sum (n/2)

        for idx, k in enumerate(range(-n + 1, n)):
            if k >= 0:
                valid = n - k
                overlapping = np.dot(h[:valid], h[k:])
                gk = np.ones(n)
                if valid > 0:
                    gk[:valid] -= h[k:]
                    gk[k:]    -= h[:valid]
            else:
                k_abs = -k
                valid = n - k_abs
                overlapping = np.dot(h[k_abs:], h[:valid])
                gk = np.ones(n)
                if valid > 0:
                    gk[k_abs:] -= h[:valid]
                    gk[:valid] -= h[k_abs:]
            O_k = S - overlapping
            O[idx] = O_k
            const = np.dot(gk, h) - O_k   # RHS becomes t + const
            G.append((gk, const))

        # build LP
        x = cp.Variable(n)
        t = cp.Variable()
        cons = [x >= 0, x <= 1, cp.sum(x) == target]
        for gk, const in G:
            cons.append(gk @ x <= t + const)
        prob = cp.Problem(cp.Minimize(t), cons)

        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=5000, verbose=False)
        except Exception:  # pragma: no cover
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        new_h = np.array(x.value).flatten()
        new_h = np.clip(new_h, 0.0, 1.0)
        adjust_sum_exact(new_h, rng)

        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c, best_h = new_c, new_h.copy()

        # convergence test (tiny improvement)
        if abs(best_c - new_c) < 1e-7:
            break
        h = new_h

    return best_h, best_c

# ---- driver for a specific resolution ---------------------------------------

def _solve_at_resolution(n, rng, deadline):
    target = n / 2.0
    start_vectors = []

    # parent interpolation (if any)
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        if parent.size != n:
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
        else:
            interp = parent.copy()
        start_vectors.append(interp)
        # binary version of the interpolant
        idx = np.argpartition(interp, -n // 2)[-n // 2:]
        bin_vec = np.zeros(n)
        bin_vec[idx] = 1.0
        start_vectors.append(bin_vec)

    # random binary
    bin_rand = np.zeros(n)
    bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    start_vectors.append(bin_rand)

    # random fractional
    rand_frac = rng.random(n)
    rand_frac *= target / rand_frac.sum()
    rand_frac = np.clip(rand_frac, 0.0, 1.0)
    adjust_sum_exact(rand_frac, rng)
    start_vectors.append(rand_frac)

    best_h = None
    best_c = np.inf

    for h0 in start_vectors:
        if time.time() > deadline:
            break
        h = np.clip(h0, 0.0, 1.0)
        adjust_sum_exact(h, rng)

        # ---- LP refinement ----
        h, c = _lp_refine(h, target, rng, deadline, max_iters=15)
        if c < best_c:
            best_h, best_c = h.copy(), c

        # ---- SLSQP cascade (smooth‑max) ----
        for beta in (20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() > deadline:
                break
            try:
                h = _slsqp_optimize(h, beta, rng, max_iter=500)
            except Exception:  # pragma: no cover
                continue
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c = h.copy(), c

        # ---- final gradient polishing ----
        h = gradient_refine(h, rng, step=0.015, max_iter=20)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c = h.copy(), c

    return best_h, best_c

# ---- top‑level entry point ---------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    # try higher resolutions first
    for n in (360, 240, 180):
        if time.time() > deadline:
            break
        h, c = _solve_at_resolution(n, rng, deadline)
        if h is not None and c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # optional up‑sampling if time permits
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        up_h = np.interp(xt, xs, best_h)
        adjust_sum_exact(up_h, rng)

        up_h, up_c = _lp_refine(up_h, n_up / 2.0, rng, deadline, max_iters=20)
        # a light SLSQP pass
        for beta in (80.0, 200.0, 500.0, 1000.0):
            if time.time() > deadline:
                break
            try:
                up_h = _slsqp_optimize(up_h, beta, rng, max_iter=400)
            except Exception:
                continue
        up_h = gradient_refine(up_h, rng, step=0.015, max_iter=15)
        up_c = compute_c5(up_h)
        if up_c < best_c:
            best_h, best_c, best_n = up_h, up_c, n_up

    # fallback simple construction
    if best_h is None:
        best_n = 180
        best_h = np.zeros(best_n)
        best_h[: best_n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final safety clip & exact sum
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
