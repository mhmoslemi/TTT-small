**Projection algorithm**:

Goal: find x ∈ [0,1]^n such that sum x = target and x is as close as possible (Euclidean distance) to v.

One approach: start with x = np.clip(v, 0, 1); compute deficit = target - sum(x). If deficit > 0, we need to add mass to elements not at upper bound (x < 1). Distribute deficit uniformly across those; but after adding might go above 1, so we need iterative water-filling.

Simpler scheme: we can use a linear scaling approach: adjust by adding delta to all elements that are not at bounds, then reclip, iterate until sum matches target within tolerance. Since n small and tolerance can be e.g., 1e-12, this will converge quickly.

def project_to_simplex_box(v, target, lower=0.0, upper=1.0):
    x = np.clip(v, lower, upper)
    # compute current sum
    cur = x.sum()
    if abs(cur - target) < 1e-12:
        return x
    # Use iterative water-filling: while deficit nonzero
    # We'll maintain a mask of free variables (0< x < upper)
    for _ in range(100):
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            # No free variable, adjust by distributing to all?
            # fallback: adjust entire vector proportionally then clip
            x = np.clip(v, lower, upper)
            break
        delta = (target - x.sum()) / n_free
        x[free] += delta
        # reclip
        x = np.clip(x, lower, upper)
        if abs(x.sum() - target) < 1e-12:
            break
    # final correction if needed due to numeric
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff among non-bound entries arbitrarily
        i = np.argmax(x < upper)  # choose one non-full index
        if i >= 0:
            add = min(diff, upper - x[i])
            x[i] += add
            diff -= add
        if abs(diff) > 1e-10:
            # adjust from full entries
            i = np.argmax(x > lower)
            if i >=0:
                sub = min(-diff, x[i] - lower)
                x[i] -= sub
    return x
```

def autocorr_fft(h):
    n = h.size
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=L)
    B = np.fft.irfft(H * np.conj(H), n=L)[:2*n-1]
    return B.real
```

def convolve_fft(a, b):
    L = 1 << ((len(a) + len(b) - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = np.fft.irfft(A*B, n=L)[:len(a) + len(b) - 1]
    return C.real

def projected_adam(h0, beta_schedule, total_iters_per_beta, step_alpha, rng):
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    target = h.size / 2.0
    for beta in beta_schedule:
        for t in range(total_iters_per_beta):
            # compute B via autocorr_fft
            B = autocorr_fft(h)
            total_mass = target
            o = total_mass - B
            # compute softmax
            max_o = np.max(beta * o)
            exp_beta_o = np.exp(beta * o - max_o)
            w = exp_beta_o / np.sum(exp_beta_o)
            # compute convs
            conv1 = convolve_fft(w, h)  # length 3n-2
            conv2 = convolve_fft(w[::-1], h)
            # slice indices
            start = h.size - 1
            S1 = conv1[start:start + h.size]
            S2 = conv2[start:start + h.size]
            grad = 1.0 - (S1 + S2)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1**(t+1))
            v_hat = v / (1 - beta2**(t+1))
            h = h - step_alpha * m_hat / (np.sqrt(v_hat) + eps)
            # project
            h = project_to_simplex_box(h, target, lower=0.0, upper=1.0)
    return h

def project_to_simplex_box(v, target, lower=0.0, upper=1.0):
    x = np.clip(v, lower, upper)
    # Compute current sum
    cur = x.sum()
    if abs(cur - target) < 1e-12:
        return x
    for _ in range(100):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        # free indices where we can add (if diff>0) or subtract (if diff<0)
        if diff > 0:
            free = x < upper - 1e-12
        else:
            free = x > lower + 1e-12
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # cannot adjust further
            break
        delta = diff / n_free
        # Apply delta with clipping
        x[free] += delta
        x = np.clip(x, lower, upper)
    # final correction
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute greedily
        for i in range(len(x)):
            if diff > 0 and x[i] < upper:
                add = min(diff, upper - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > lower:
                sub = min(-diff, x[i] - lower)
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return x

def projected_gradient_descent(h0, beta_schedule, iters_per_beta, rng):
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_global = 0
    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            t_global += 1
            # autocorr
            B = autocorr_fft(h)
            total_mass = target
            o = total_mass - B  # shape 2n-1
            # softmax
            log_exp = beta * o
            max_log = np.max(log_exp)
            w = np.exp(log_exp - max_log)
            w_sum = w.sum()
            w /= w_sum
            # convs
            conv1 = convolve_fft(w, h)
            conv2 = convolve_fft(w[::-1], h)
            start = n - 1
            S1 = conv1[start:start + n]
            S2 = conv2[start:start + n]
            grad = 1.0 - (S1 + S2)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t_global)
            v_hat = v / (1 - beta2**t_global)
            step = 0.5 / (np.sqrt(v_hat) + eps)  # base step 0.5 scaled
            h = h - step * m_hat
            # project
            h = project_to_simplex_box(h, target, 0.0, 1.0)
    return h

def pair_transfer_optimize(h0, rng, max_iters=5000, time_limit=30.0, init_factor=0.5, decay=0.999):
    n = h0.size
    target = n/2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        # compute max overlap and shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n-1)
        # compute product for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            factor = init_factor
            continue
        eps = factor * max_eps
        h[i] -= eps
        h[j] += eps
        # clip
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        # update best
        cur_c = np.max(corr) * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        factor = max(1e-4, factor * decay)
    return best_h, best_c

    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            factor = init_factor
            continue
        eps = factor * max_eps
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        cur_c = np.max(corr) * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        factor = max(1e-4, factor * decay)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    n = 720  # moderate resolution, still < 1000
    if "initial_h_values" in globals():
        h0 = upsample_initial(np.array(initial_h_values, dtype=float), n)
    else:
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    beta_schedule = [5, 20, 100, 500, 2000, 8000]
    iters_per_beta = 1500

    # Main stochastic optimisation loop
    while time.time() - start_time < 0.80 * budget_s:
        # pick start point
        if rng.random() < 0.7:
            h_start = best_h + rng.normal(scale=0.05, size=n)
        else:
            h_start = random_feasible_vector(n, rng)
        h_start = project_to_simplex_box(h_start, n / 2.0, 0.0, 1.0)

        # gradient descent phase
        h_opt = projected_gradient_descent(h_start, beta_schedule, iters_per_beta, rng)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        # brief pair‑transfer refinement if time permits
        if time.time() - start_time < 0.85 * budget_s:
            rem = budget_s - (time.time() - start_time)
            tl = min(10.0, rem * 0.05)
            h_ref, c_ref = pair_transfer_optimize(best_h, rng,
                                                   max_iters=3000,
                                                   time_limit=tl,
                                                   init_factor=0.5)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    # final longer refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        h_final, c_final = pair_transfer_optimize(best_h, rng,
                                                  max_iters=8000,
                                                  time_limit=remaining,
                                                  init_factor=0.5)
        if c_final < best_c:
            best_c, best_h = c_final, h_final.copy()

    # final feasibility enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_simplex_box(best_h, n / 2.0, 0.0, 1.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
