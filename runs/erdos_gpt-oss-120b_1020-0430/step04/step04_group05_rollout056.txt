def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # choose random index with capacity to adjust
        if diff > 0:
            # need to add mass; pick index where h[i] < 1
            idx = rng.integers(0, n)
            cap = 1.0 - h[idx]
            if cap <= 0:
                continue
            add = min(diff, cap)
            h[idx] += add
        else:
            idx = rng.integers(0, n)
            cap = h[idx]
            if cap <= 0:
                continue
            add = max(diff, -cap)
            h[idx] += add
        diff = target - h.sum()
    # final uniform correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        # second correction
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute to interior
            interior = (h > 0) & (h < 1)
            if interior.any():
                cnt = interior.sum()
                h[interior] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h
```

import time
import numpy as np
from scipy.optimize import minimize

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            i = rng.integers(0, n)
            cap = 1.0 - h[i]
            if cap <= 0:
                continue
            add = min(diff, cap)
            h[i] += add
        else:
            i = rng.integers(0, n)
            cap = h[i]
            if cap <= 0:
                continue
            add = max(diff, -cap)
            h[i] += add
        diff = target - h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            interior = (h > 0) & (h < 1)
            if interior.any():
                cnt = interior.sum()
                h[interior] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def binary_feasible_vector(n, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_lag_and_value(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = h.size
    idx = np.argmax(corr)
    k = idx - (n - 1)
    return k, corr[idx]

def gradient_worst_lag(h, k):
    n = h.size
    if k >= 0:
        a = np.empty(n)
        a[:n - k] = 1.0 - h[k:]
        a[n - k:] = 1.0
        b = np.empty(n)
        b[k:] = -h[:n - k]
        b[:k] = 0.0
    else:
        kk = -k
        a = np.empty(n)
        a[kk:] = 1.0 - h[:n - kk]
        a[:kk] = 1.0
        b = np.empty(n)
        b[:n - kk] = -h[kk:]
        b[n - kk:] = 0.0
    return a + b

def projected_gradient_step(h, target, lr, rng):
    k, _ = worst_lag_and_value(h)
    grad = gradient_worst_lag(h, k)
    new_h = h - lr * grad
    new_h = enforce_sum(new_h, target, rng)
    return new_h

def gradient_descent(h, target, rng, iters, lr0):
    lr = lr0
    for _ in range(iters):
        h = projected_gradient_step(h, target, lr, rng)
        lr *= 0.9995
    return h

    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()

    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return np.clip(new_h, 0.0, 1.0)

def neighbor_targeted(h, target, rng, scale=0.3):
    k, _ = worst_lag_and_value(h)
    n = h.size
    if k >= 0:
        a = np.empty(n)
        a[:n - k] = 1.0 - h[k:]
        a[n - k:] = 1.0
    else:
        kk = -k
        a = np.empty(n)
        a[kk:] = 1.0 - h[:n - kk]
        a[:kk] = 1.0
    top = max(1, int(0.1 * n))
    top_idx = np.argpartition(-a, top - 1)[:top]
    bot_idx = np.argpartition(a, top - 1)[:top]
    i = int(rng.choice(top_idx))
    j = int(rng.choice(bot_idx))
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()

    new_h[j] += eps

    diff = target - new_h.sum()

    return new_h

def simulated_annealing(h0, target, rng, max_iters, time_limit):
    h = h0.copy()
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.02, 1e-5
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        if rng.random() < 0.7:
            new_h = neighbor_targeted(h, target, rng, scale=0.3)
        else:
            new_h = perturb_h(h, rng, scale=0.3)
        new_c = compute_c5(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
    return best_h, best_c

def upsample(h, new_n):
    n = h.size
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(new_x, old_x, h)

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):
    n = h0.size
    target = n / 2.0
    def sum_eq(x):
        return np.sum(x) - target
    def soft_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_eq}
    res = minimize(soft_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter,
                                             'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, target, rng)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 360
        h0 = binary_feasible_vector(n0, rng)
    target0 = n0 / 2.0
    best_h, best_c = h0.copy(), compute_c5(h0)

    # Phase 1: low resolution exploration
    while time.time() - t0 < budget_s * 0.45:
        start = perturb_h(best_h, rng, scale=0.4) if rng.random() < 0.7 else binary_feasible_vector(n0, rng)
        limit = t0 + budget_s * 0.45
        sa_h, sa_c = simulated_annealing(start, target0, rng, max_iters=4000, time_limit=limit)
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c
        gd_h = gradient_descent(best_h, target0, rng, iters=1500, lr0=0.12)
        gd_c = compute_c5(gd_h)
        if gd_c < best_c:
            best_h, best_c = gd_h, gd_c
        for beta in (5, 20, 80, 200, 500, 1000):
            if time.time() - t0 >= budget_s * 0.45:
                break
            try:
                ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=150)
                ref_c = compute_c5(ref_h)
                if ref_c < best_c:
                    best_h, best_c = ref_h, ref_c
            except Exception:
                pass

    # Phase 2: higher resolution if time permits
    if time.time() - t0 < budget_s * 0.9:
        n_high = max(720, n0 * 2)
        up_h = upsample(best_h, n_high)
        up_h = enforce_sum(up_h, n_high / 2.0, rng)
        target_high = n_high / 2.0
        limit = t0 + budget_s * 0.9
        sa_h, sa_c = simulated_annealing(up_h, target_high, rng, max_iters=8000, time_limit=limit)
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c
        gd_h = gradient_descent(best_h, target_high, rng, iters=3000, lr0=0.07)
        gd_c = compute_c5(gd_h)
        if gd_c < best_c:
            best_h, best_c = gd_h, gd_c
        for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000):
            if time.time() - t0 >= budget_s:
                break
            try:
                ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=150)
                ref_c = compute_c5(ref_h)
                if ref_c < best_c:
                    best_h, best_c = ref_h, ref_c
            except Exception:
                pass

    # Phase 3: final local tweaks
    cur_h, cur_c = best_h.copy(), best_c
    for _ in range(2500):
        if time.time() - t0 > budget_s:
            break
        new_h = perturb_h(cur_h, rng, scale=0.15)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
