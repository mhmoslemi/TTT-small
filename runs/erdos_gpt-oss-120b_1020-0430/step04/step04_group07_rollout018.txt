import time, numpy as np
from scipy.optimize import minimize

# globals for SLSQP soft‑max objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2] using correlation."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k overlap."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stabilization
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """SLSQP optimisation of soft‑max objective for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction by redistribution
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:

            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add

    return x

def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator = None) -> np.ndarray:
    """Adjust h in‑place to achieve exact sum = target while staying in [0,1]."""
    if rng is None:
        rng = np.random.default_rng()
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the shift with maximal overlap."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6):
    """Strict hill‑climb using gradient‑directed mass transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    """Fine‑tune using the explicit gradient of the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    # initialise from parent if provided
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, h0.size / 2.0, rng)
        best_h = h0
        best_n = h0.size
        best_c = compute_c5(best_h)

    n_candidates = [96, 128, 160, 192, 224, 256, 288, 320]

    # ---------- main multi‑restart search ----------
    while time.time() - start < 0.8 * budget_s:
        n = int(rng.choice(n_candidates))
        # generate start point
        if best_h is not None and rng.random() < 0.7:
            # interpolate current best
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h)
        else:
            # random start, occasionally binary
            if rng.random() < 0.5:
                h = np.zeros(n)
                idx = rng.choice(n, n // 2, replace=False)
                h[idx] = 1.0
            else:
                h = random_feasible_vector(n, rng)
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, n / 2.0, rng)

        # hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.7)

        # SLSQP soft‑max passes
        for beta in (20., 80., 200., 800., 3000., 8000.):
            if time.time() - start > 0.94 * budget_s:
                break
            maxit = 800 if beta >= 200 else 500
            h = slsqp_optimize(h, beta, rng, max_iter=maxit)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

        # fine gradient refinement
        h = gradient_refine(h, rng, step=0.015, max_iter=8)
        c = compute_c5(h)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

        # final hill‑climb on this run
        h, c = guided_hill_climb(h, rng, max_iters=1500, step_scale=0.5)
        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # ---------- up‑sampling refinement ----------
    while time.time() - start < 0.97 * budget_s:
        if best_h is None:
            break
        factor = int(rng.choice([2, 3, 4]))
        n_up = best_n * factor
        if n_up > 1024:
            break
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_up = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_up, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, n_up / 2.0, rng)

        # aggressive hill‑climb
        h_up, c_up = guided_hill_climb(h_up, rng, max_iters=3000, step_scale=0.5)

        # moderate SLSQP passes
        for beta in (50., 200., 1000., 5000., 15000.):
            if time.time() - start > 0.99 * budget_s:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=800)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_h, best_c, best_n = h_up.copy(), c_up, n_up

        # tiny gradient tidy‑up
        h_up = gradient_refine(h_up, rng, step=0.01, max_iter=6)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

        best_h, best_n = h_up, n_up

    # ---------- final stochastic tweaks ----------
    while time.time() - start < budget_s * 0.999 and best_h is not None:
        grad = worst_gradient(best_h)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0:
            i, j = rng.integers(0, best_h.size, size=2)
            max_eps = min(best_h[i], 1.0 - best_h[j])
            if max_eps <= 0:
                continue
        eps = rng.random() * max_eps * 0.02
        nh = best_h.copy()
        nh[i] -= eps
        nh[j] += eps
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
