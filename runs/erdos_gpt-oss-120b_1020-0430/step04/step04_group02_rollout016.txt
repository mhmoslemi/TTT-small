def binary_greedy_optimize(h, rng, max_iters=2000):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iters):
        corr = np.correlate(h, 1-h, mode='full')
        max_idx = int(np.argmax(corr))
        cur_c = corr[max_idx] * (2.0/n)
        if cur_c < best_c - 1e-12:
            best_c, best_h = cur_c, h.copy()
        # if no improvement possible? We'll try to reduce this corr
        k = max_idx - (n-1)
        # find swap indices
        if k >= 0:
            i_candidates = np.where((h[:n-k] == 1) & (h[k:] == 0))[0]
            j_candidates = np.where((h[:n-k] == 0) & (h[k:] == 1))[0]
        else:
            p = -k
            i_candidates = np.where((h[p:] == 1) & (h[:n-p] == 0))[0] + p
            j_candidates = np.where((h[p:] == 0) & (h[:n-p] == 1))[0] + p
        if i_candidates.size > 0 and j_candidates.size > 0:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
            # swap
            h[i], h[j] = h[j], h[i]
            continue
        # else if cannot reduce L_k via direct swap, do random swap or break
        # random swap of any 1 and 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = h[j], h[i]
    # after loop, final evaluate
    final_c = compute_c5(h)
    if final_c < best_c:
        best_c, best_h = final_c, h.copy()
    return best_h, best_c

def binary_greedy_optimize(h, rng, max_iters=2000, patience=100):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    no_improve = 0
    for it in range(max_iters):
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = int(np.argmax(corr))
        cur_max = corr[max_idx]
        cur_c = cur_max * (2.0 / n)
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # determine shift
        k = max_idx - (n - 1)
        # find swap candidates
        if k >= 0:
            i_cand = np.where((h[:n-k] == 1) & (h[k:] == 0))[0]
            j_cand = np.where((h[:n-k] == 0) & (h[k:] == 1))[0]
        else:
            p = -k
            # i index where h[i]=1 and neighbor h[i-p]=0 => i>=p
            i_cand = np.where((h[p:] == 1) & (h[:n-p] == 0))[0] + p
            j_cand = np.where((h[p:] == 0) & (h[:n-p] == 1))[0] + p
        if i_cand.size > 0 and j_cand.size > 0:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
        else:
            # random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        h[i], h[j] = h[j], h[i]
        if no_improve >= patience:
            break
    # final evaluate
    final_c = compute_c5(h)
    if final_c < best_c:
        best_c, best_h = final_c, h.copy()
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    """Project x onto [lo,hi] while enforcing sum(x)=target."""
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def compute_c5(h):
    """Return the C5 value for a step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def compute_c5_and_idx(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * (2.0 / h.size), idx


# ---------- binary greedy optimizer ----------
def binary_greedy_optimize(h, rng, max_iters=3000, patience=200):
    """
    h is a float array with values 0 or 1 and sum(h)=n/2.
    Repeatedly swaps a (1,0) pair for a (0,1) pair at the worst shift.
    """
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(best_h)
    no_improve = 0

    for _ in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = int(np.argmax(corr))
        cur_c = corr[max_idx] * (2.0 / n)

        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k = max_idx - (n - 1)

        # find a (1,0) pair and a (0,1) pair at this shift
        if k >= 0:
            i_cand = np.where((h[: n - k] == 1) & (h[k:] == 0))[0]
            j_cand = np.where((h[: n - k] == 0) & (h[k:] == 1))[0]
        else:
            p = -k
            i_cand = np.where((h[p:] == 1) & (h[: n - p] == 0))[0] + p
            j_cand = np.where((h[p:] == 0) & (h[: n - p] == 1))[0] + p

        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
        else:
            # fallback random swap keeping the sum
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

        h[i], h[j] = h[j], h[i]

        if no_improve >= patience:
            break

    final_c = compute_c5(h)
    if final_c < best_c:
        best_c, best_h = final_c, h.copy()
    return best_h, best_c


# ---------- continuous hill‑climb (targeted moves) ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_move(h, rng, k, scale=0.2):
    n = h.size
    if k <= 0 or k >= n:
        return perturb_h(h, rng, scale)
    low_mask = (h[: n - k] < 0.5) & (h[k:] < 0.5)
    high_mask = (h[: n - k] > 0.5) & (h[k:] > 0.5)
    if not low_mask.any() or not high_mask.any():
        return perturb_h(h, rng, scale)
    i_low = rng.choice(np.flatnonzero(low_mask))
    i_high = rng.choice(np.flatnonzero(high_mask))
    max_eps = min(
        0.5 - h[i_low],
        0.5 - h[i_low + k],
        h[i_high] - 0.5,
        h[i_high + k] - 0.5,
    )
    if max_eps <= 0.0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i_low] += eps
    new_h[i_low + k] += eps
    new_h[i_high] -= eps
    new_h[i_high + k] -= eps
    return new_h


def hill_climb_targeted(h_start, rng, max_iters=1200, init_step=0.5, n_trials=4, p_target=0.25):
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    n = h.shape[0]

    for it in range(max_iters):
        _, idx = compute_c5_and_idx(h)
        k_star = idx - (n - 1)  # shift corresponding to maximum overlap
        improved = False
        for _ in range(n_trials):
            if rng.random() < p_target:
                cand = targeted_move(h, rng, k_star, scale=step)
            else:
                cand = perturb_h(h, rng, scale=step)
            c = compute_c5(cand)
            if c < best_c - 1e-12:
                best_c, h = c, cand
                improved = True
                break
        if not improved and rng.random() < 0.02:
            cand = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(cand)
            if c < best_c:
                best_c, h = c, cand
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


# ---------- SLSQP refinement with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """Smooth‑max SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)

                bin_vec[idx] = 1.0
        elif diff < 0:

            if ones.size >= -diff:

                bin_vec[idx] = 0.0
