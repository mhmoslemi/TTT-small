def gradient_refine(h, rng, step=0.02, max_iter=20):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = full_corr(h)
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]      # for i where i+k < n
            g[k:] -= h[:upper]            # for i where i-k >= 0
        # step size schedule: geometric decay
        step_sz = step * (0.7 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # re-balance sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution: add diff to random index respecting bounds
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time, numpy as np, scipy.optimize as opt

# ---------- core utilities ----------
def compute_c5(h):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    hf = np.fft.rfft(h, n=L)
    gf = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(hf * np.conj(gf), n=L)[: 2 * n - 1]
    return float(np.max(corr) * (2.0 / n))


def _full_corr(h):
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    hf = np.fft.rfft(h, n=L)
    gf = np.fft.rfft(1.0 - h, n=L)
    return np.fft.irfft(hf * np.conj(gf), n=L)[: 2 * n - 1]


def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()


# ---------- gradient refinement ----------
def gradient_refine(h, rng, step=0.03, max_iter=15):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = _full_corr(h)
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            u = n - k
            g[:u] += 1.0 - h[k:]
            g[k:] -= h[:u]
        step_sz = step * (0.7 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

CURRENT_TARGET = None
CURRENT_BETA = None


def _sum_eq(x):
    return np.sum(x) - CURRENT_TARGET


def _softmax_obj(x):
    corr = _full_corr(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, rng, max_iter=400):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq}
    res = opt.minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x


# ---------- binary simulated annealing ----------
def binary_sa_hill_climb(h_bin, rng, deadline):
    n = h_bin.size
    cur = h_bin.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    ones = list(np.flatnonzero(cur == 1))
    zeros = list(np.flatnonzero(cur == 0))
    T0 = max(0.01 * cur_c, 1e-4)
    it = 0
    last_improve = 0
    while time.time() < deadline:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0
        cur[j] = 1
        c = compute_c5(cur)
        T = T0 * (0.9995 ** it) + 1e-12
        if c < cur_c or rng.random() < np.exp(-(c - cur_c) / T):
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c, best, last_improve = c, cur.copy(), it
        else:
            cur[i] = 1
            cur[j] = 0
        it += 1
        if it - last_improve > 5000:
            break
    return best, best_c
