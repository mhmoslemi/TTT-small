def binary_sa(h_bin, rng, max_iters, time_start, time_budget):
    n = h_bin.size
    cur_h = h_bin.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    # Precompute number of ones = n//2
    ones = int(round(n/2))
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        # temperature schedule:
        T = 0.2 * (1 - it / max_iters) + 0.001
        # pick random 1 and random 0 to swap
        ones_idx = np.flatnonzero(cur_h > 0.5)  # vector
        zeros_idx = np.flatnonzero(cur_h < 0.5)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # swap
        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c = compute_c5(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / T):
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

import time

# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        # final correction
        corr = diff / n
        h += corr
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):

        h[i] -= eps

def compute_c5(h):
    """Return C5 for a vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    """Run SLSQP on the soft‑max surrogate for a single beta."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()

            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:

    return x

def binary_sa_loop(h_start, rng, t_start, t_end):
    """Swap‑based SA on a binary (0/1) vector."""
    n = h_start.size
    cur_h = h_start.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    while time.time() < t_end:

        frac = (time.time() - t_start) / (t_end - t_start)
        T = max(0.001, (1.0 - frac) * 0.2)

                best_h = cur_h.copy()

        if int(time.time() - t_start) % 50 == 0 and cur_c > best_c:

            cur_c = best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    # decide resolution
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
    else:
        n0 = 360
    target_n = 720 if budget_s >= 600 else n0
    target_n = max(target_n, n0)  # never go below

    # initial continuous h
    if "initial_h_values" in globals():
        h_cont = np.array(initial_h_values, dtype=float)
        h_cont = np.clip(h_cont, 0.0, 1.0)
        if h_cont.size != target_n:
            h_cont = upsample_h(h_cont, target_n, rng)
    else:
        h_cont = random_feasible_vector(target_n, rng)

    h_cont = enforce_sum(h_cont, target_n / 2.0, rng)

    # binary projection of the continuous vector
    k = target_n // 2
    idx = np.argpartition(-h_cont, k - 1)[:k]
    h_bin = np.zeros(target_n, dtype=float)
    h_bin[idx] = 1.0

    # binary simulated annealing (use most of the budget)
    sa_budget = (t_end - time.time()) * 0.85
    h_bin_best, bin_best_c = binary_sa_loop(
        h_bin, rng, time.time(), time.time() + sa_budget
    )
    best_h = h_bin_best.copy()
    best_c = bin_best_c

    # final continuous refinement if time permits
    remaining = t_end - time.time()
    if remaining > 15.0:
        # a tiny random jitter before SLSQP
        h_cont = best_h.copy()
        for _ in range(5):
            i = rng.integers(0, target_n)
            j = rng.integers(0, target_n)
            if i == j:
                continue
            max_eps = min(h_cont[i], 1.0 - h_cont[j])
            if max_eps <= 0:
                continue
            eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.05
            h_cont[i] -= eps
            h_cont[j] += eps
        h_cont = enforce_sum(h_cont, target_n / 2.0, rng)

        beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000]
        per_beta = remaining / len(beta_list)
        for beta in beta_list:
            if time.time() >= t_end:
                break
            start = time.time()
            try:
                refined = slsqp_optimize_one_beta(h_cont, beta, rng, max_iter=250)
            except Exception:
                continue
            refined_c = compute_c5(refined)
            if refined_c < best_c:
                best_c = refined_c
                best_h = refined.copy()
            # allow a few extra inner passes with the same beta
            while time.time() - start < per_beta and time.time() < t_end:
                try:
                    refined = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=120)
                except Exception:
                    break
                refined_c = compute_c5(refined)
                if refined_c < best_c:
                    best_c = refined_c
                    best_h = refined.copy()

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
