import time
import numpy as np
from scipy.optimize import minimize

# ---- basic projection -------------------------------------------------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:                         # rare correction
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    x = rng.random(n)
    return project_box_sum(x, n / 2.0)


# ---- small random perturbation (keeps sum) ---------------------------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---- aggressive move using the worst shift ---------------------------
def aggressive_move(h, delta, target):
    """Return a binary vector that minimises overlap for the given shift."""
    n = h.size
    if delta >= 0:
        start = 0
        end = n - delta
        w = 1.0 - h[start + delta : end + delta]
    else:
        start = -delta
        end = n
        w = 1.0 - h[start + delta : end + delta]

    # outside indices (do not participate in this shift)
    if start == 0 and end == n:
        out_idx = np.empty(0, dtype=int)
    else:
        out_idx = np.concatenate((np.arange(start, dtype=int),
                                  np.arange(end, n, dtype=int)))
    out_len = out_idx.size
    need_inside = target - out_len

    h_new = np.zeros(n, dtype=float)
    if need_inside <= 0:
        # we can satisfy the weight using only outside positions
        h_new[out_idx[:target]] = 1.0
    else:
        if out_len:
            h_new[out_idx] = 1.0
        # choose positions with smallest w inside the overlapping region
        need = int(need_inside)
        idx_small = np.argpartition(w, need - 1)[:need]
        inside_idx = start + idx_small
        h_new[inside_idx] = 1.0
    return h_new


# ---- correlation (FFT based) -----------------------------------------
def make_corr_fun(n):
    N = 1 << ((2 * n - 1).bit_length())

    def corr(h):
        H = np.fft.rfft(h, N)
        G = np.fft.rfft((1.0 - h)[::-1], N)
        conv = np.fft.irfft(H * G, N)
        c = conv[: 2 * n - 1][::-1]                 # matches np.correlate
        return c
    return corr


# ---- softmax surrogate for final refinement -------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_refine(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)


# ---- main entry point ------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        n_target = 720
        if n0 != n_target:
            old_x = np.linspace(0.0, 2.0, n0, endpoint=False) + 1.0 / n0
            new_x = np.linspace(0.0, 2.0, n_target, endpoint=False) + 1.0 / n_target
            h0 = np.interp(new_x, old_x, h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_sum(h0, n_target / 2.0)
        n = n_target
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    corr_fun = make_corr_fun(n)

    def compute_c5(h):
        c = corr_fun(h)
        return (2.0 / n) * np.max(c)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    h = best_h.copy()
    cur_c = best_c
    target = n // 2

    # ----- hill‑climbing ------------------------------------------------
    prob_aggr = 0.12                # frequency of aggressive moves
    temp_start = 5e-3
    temp_end = 1e-7
    max_iters = 200000

    for it in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.92:
            break

        # current correlation and worst shift
        corr = corr_fun(h)
        max_corr = np.max(corr)
        idx = int(np.argmax(corr))
        delta = idx - (n - 1)          # shift in index units

        # candidate generation
        if rng.random() < prob_aggr:
            cand = aggressive_move(h, delta, target)
        else:
            cand = perturb_h(h, rng, scale=0.18)

        cand_c = (2.0 / n) * np.max(corr_fun(cand))

        # simulated‑annealing acceptance
        temp = temp_start * (temp_end / temp_start) ** (it / max_iters)
        if cand_c < cur_c - 1e-12 or rng.random() < np.exp((cur_c - cand_c) / max(temp, 1e-12)):
            h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()

        # occasional cooling of aggressive probability
        if (it + 1) % 4000 == 0:
            prob_aggr = max(0.02, prob_aggr * 0.95)

    # ----- final smooth refinement --------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > budget_s * 0.05:
        for beta in [10, 50, 200, 800, 3000, 10000]:
            if time.time() - start_time > budget_s * 0.97:
                break
            try:
                refined = slsqp_refine(best_h, beta, max_iter=500)
            except Exception:
                continue
            refined_c = compute_c5(refined)
            if refined_c < best_c:
                best_c, best_h = refined_c, refined.copy()

    # ----- clean final result -------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
