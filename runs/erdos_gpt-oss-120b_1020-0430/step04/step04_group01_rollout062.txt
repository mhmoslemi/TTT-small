def project_sum(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # identify indices not at bound (0< h<1)
    mask = (h > 1e-12) & (h < 1-1e-12)
    slack = np.where(diff > 0, 1 - h, h)  # capacity to increase or decrease
    total_slack = slack[mask].sum()
    if total_slack > 0:
        # distribute proportionally
        h[mask] += diff * slack[mask] / total_slack
        # may overshoot slight; clip
        h = np.clip(h, 0, 1)
    # If still diff left (due to bounds), use random adjustments
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # fallback to iterative adjust
        h = enforce_sum(h, target, rng)
    return h
```

def softmax_grad(h, beta):
    n = h.size
    target = n / 2.0
    # correlation self
    conv = correlate_self(h)
    g = target - conv  # shape 2n-1
    # subtract max for numerical stability
    g_max = g.max()
    w = np.exp(beta * (g - g_max))
    w_sum = w.sum()
    w /= w_sum
    # compute convolution w * h
    L = 1 << ((3*n - 2).bit_length())
    fw = np.fft.rfft(w, L)
    fh = np.fft.rfft(h, L)
    conv_wh_full = np.fft.irfft(fw * fh, L)
    start = n - 1
    conv_wh = conv_wh_full[start:start + n]
    # reversed w
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, L)
    conv_rev_full = np.fft.irfft(fw_rev * fh, L)
    conv_rev = conv_rev_full[start:start + n]
    grad = - (conv_wh + conv_rev)
    return grad
```

def compute_c5(h):
    n = h.size
    # compute correlation of h with rev(1 - h)
    L = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]
    max_val = conv.max()
    C5 = max_val * (2.0 / n)
    return float(C5)
```

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
```

import time
import numpy as np

# --------------------------------------------------------------
# Feasibility utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = rng.random(n)
    h = h / h.sum() * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_self(h):
    """Full linear correlation of h with itself (i.e. h * rev(h))."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft(h[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return float(conv.max() * (2.0 / n))


# --------------------------------------------------------------
# Smoothed objective gradient
# --------------------------------------------------------------
def softmax_grad(h, beta):
    """Gradient of (1/β) log Σ exp(β * g_k) where g_k = Σ_i h_i(1-h_{i+k})."""
    n = h.size
    target = n / 2.0
    # self‑correlation gives Σ_i h_i h_{i+k}
    conv = correlate_self(h)
    g = target - conv                     # shape 2n‑1
    g_max = g.max()
    w = np.exp(beta * (g - g_max))
    w /= w.sum()                          # normalized weights

    L = 1 << ((3 * n - 2).bit_length())   # length for convolutions w*h
    fw = np.fft.rfft(w, L)
    fh = np.fft.rfft(h, L)

    # w * h
    conv_wh_full = np.fft.irfft(fw * fh, L)
    start = n - 1
    conv_wh = conv_wh_full[start : start + n]

    # w_rev * h  (w_rev[i] = w[-i])
    fw_rev = np.fft.rfft(w[::-1], L)
    conv_rev_full = np.fft.irfft(fw_rev * fh, L)
    conv_rev = conv_rev_full[start : start + n]

    grad = - (conv_wh + conv_rev)
    return grad


# --------------------------------------------------------------
# Gradient descent with continuation on β
# --------------------------------------------------------------
def gradient_descent_optimize(h, rng, deadline):
    """Multi‑β gradient descent respecting bounds and sum constraint."""
    n = h.size
    target = n / 2.0
    betas = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
    for beta in betas:
        # estimated remaining time determines iteration budget
        rem = max(0.0, deadline - time.time())
        max_iters = int(rem * 250) // len(betas)
        max_iters = max(1, min(max_iters, 2500))
        lr = 2.0 / np.sqrt(beta)          # heuristic step size
        for _ in range(max_iters):
            if time.time() > deadline:
                return h
            grad = softmax_grad(h, beta)
            h = h - lr * grad
            # box projection
            h = np.clip(h, 0.0, 1.0)
            # simple sum correction
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                free = (h > 1e-12) & (h < 1 - 1e-12)
                n_free = free.sum()
                if n_free > 0:
                    h[free] += diff / n_free
                    h = np.clip(h, 0.0, 1.0)
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    h = enforce_sum(h, target, rng)
    return h


# --------------------------------------------------------------
# Simple greedy hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = float(budget_s)
    deadline = start_time + total_budget

    # ------------------------------------------------------------------
    # Initial feasible vector (use parent if available)
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:

    n = h0.size
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, n / 2.0, rng)

    # ------------------------------------------------------------------
    # Coarse optimisation (down‑sample if possible)
    # ------------------------------------------------------------------
    if n > 360:
        n_coarse = n // 2
        h_coarse = 0.5 * (h0[0::2] + h0[1::2])
        h_coarse = enforce_sum(h_coarse, n_coarse / 2.0, rng)
        coarse_deadline = start_time + 0.4 * total_budget
        h_coarse = gradient_descent_optimize(h_coarse, rng, coarse_deadline)
        # up‑sample by repeating each entry twice
        h_fine = np.repeat(h_coarse, 2)
        h_fine = np.clip(h_fine, 0.0, 1.0)
        h_fine = enforce_sum(h_fine, n / 2.0, rng)
    else:
        h_fine = h0

    # ------------------------------------------------------------------
    # Fine optimisation
    # ------------------------------------------------------------------
    fine_deadline = start_time + 0.85 * total_budget
    h_fine = gradient_descent_optimize(h_fine, rng, fine_deadline)

    # ------------------------------------------------------------------
    # Local hill‑climb for polishing
    # ------------------------------------------------------------------
    if time.time() < start_time + 0.95 * total_budget:
        h_fine, _ = hill_climb(h_fine, rng, max_iters=3000, start_step=0.2)

    # ------------------------------------------------------------------
    # Keep the best solution found so far
    # ------------------------------------------------------------------
    best_h = h_fine
    best_c = compute_c5(best_h)

    # Random restarts while time permits
    # ------------------------------------------------------------------
    while time.time() + 0.5 < deadline:
        h_rand = random_feasible_vector(n, rng)
        h_rand = gradient_descent_optimize(h_rand, rng, deadline)
        c_rand = compute_c5(h_rand)
        if c_rand < best_c:
            best_h, best_c = h_rand, c_rand

    # ------------------------------------------------------------------
    # Final sanity projection
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
