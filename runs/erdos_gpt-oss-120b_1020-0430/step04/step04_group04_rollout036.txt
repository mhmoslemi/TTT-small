Define compute_c5:

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L-1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H*R, m)[:L]
    return np.max(conv) * (2.0 / n)

def hill_climb(start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 8)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2)
            c = compute_c5(new_h)
            if c < best_c:
                best_c = c
                h = new_h
                improved = True
        if (it+1) % decay_interval == 0:
            step = max(step * 0.75, 1e-5)
    return h, best_c

def simulated_annealing(start, rng, max_iter=10000,
                       init_temp=1e-2, final_temp=1e-6,
                       init_scale=0.5, final_scale=0.02):
    cur = start.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new = perturb_h(cur, rng, scale=scale)
        new_c = compute_c5(new)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur = new
            cur_c = new_c
            if cur_c < best_c:
                best = cur.copy()
                best_c = cur_c
    return best, best_c

def slsqp_optimize(h0, beta, max_iter=500):
    n = h0.size
    target = n/2.0
    bounds = [(0.,1.)]*n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x)-target}
    # softmax surrogate
    def surrogate(x):
        corr = np.correlate(x, 1-x, mode='full')
        # scale by beta
        b_corr = beta * corr
        # log-sum-exp approx max
        m = np.max(b_corr)
        return (m + np.log(np.sum(np.exp(b_corr - m)))) / beta
    # Use minimize with SLSQP
    res = minimize(surrogate, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0.,1.)
    # correct sum
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        # adjust proportional to free variables
        free = (x>1e-12) & (x<1-1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff/idx.size
            x[idx] += per
            x = np.clip(x,0.,1.)
        else:
            x += diff/n
            x = np.clip(x,0.,1.)
        # re-check diff (likely small)
        diff2 = target - np.sum(x)
        if abs(diff2) > 1e-12:
            # fallback simple correction
            for i in range(n):
                if abs(diff2) < 1e-12:
                    break
                add = min(diff2, 1.-x[i]) if diff2>0 else max(diff2, -x[i])
                x[i] += add
                diff2 = target - np.sum(x)
    return x

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h,0.,1.)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.-h[i]) if diff>0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

Define random_feasible_vector(n, rng):
    # Start with 0.5+small random noise
    h = np.full(n, 0.5)
    # Add small random perturbations preserving sum: generate random direction vectors with sum zero.
    for _ in range(n*3):
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i==j: continue
        max_eps = min(h[i], 1.-h[j])
        if max_eps <=0: continue
        eps = (rng.random()*2 - 1)* max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h,0.,1.)
    diff = target = n/2.0 - h.sum()
    if abs(diff)>1e-12:
        h = enforce_sum(h, target, rng)
    return h

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]

def random_feasible_vector(n, rng):
    target = n / 2.0
    h = np.full(n, 0.5)
    # perturb with random swaps multiple times
    for _ in range(n * 4):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h

perturb_h:

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

hill_climb:

def hill_climb(start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 8)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c

simulated_annealing:

def simulated_annealing(start, rng, max_iter=10000,
                       init_temp=1e-2, final_temp=1e-6,
                       init_scale=0.5, final_scale=0.02):
    cur = start.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new = perturb_h(cur, rng, scale=scale)
        new_c = compute_c5(new)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur, cur_c = new, new_c
            if cur_c < best_c:
                best, best_c = cur.copy(), cur_c
    return best, best_c

slsqp_optimize:

def slsqp_optimize(h0, beta, max_iter=500):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t}
    def surrogate(x):
        corr = np.correlate(x, 1 - x, mode='full')
        b_corr = beta * corr
        m = np.max(b_corr)
        # avoid overflow
        return (m + np.log(np.sum(np.exp(b_corr - m)))) / beta
    res = minimize(surrogate, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        # adjust free variables
        free = (x > 1e-12) & (x < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            x[idx] += per
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
        # second correction if needed
        diff2 = target - np.sum(x)
        if abs(diff2) > 1e-12:
            for i in range(n):
                if abs(diff2) < 1e-12:
                    break
                add = min(diff2, 1.0 - x[i]) if diff2 > 0 else max(diff2, -x[i])
                x[i] += add
                diff2 = target - np.sum(x)
    return x

        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(start, rng, max_iters=2000, init_step=0.5, n_trials=5):
    h = start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 8)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return h, best_c


def simulated_annealing(start, rng, max_iter=10000,
                       init_temp=1e-2, final_temp=1e-6,
                       init_scale=0.5, final_scale=0.02):
    cur = start.copy()
    cur_c = compute_c5(cur)
    best = cur.copy()
    best_c = cur_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new = perturb_h(cur, rng, scale=scale)
        new_c = compute_c5(new)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur, cur_c = new, new_c
            if cur_c < best_c:
                best, best_c = cur.copy(), cur_c
    return best, best_c


def slsqp_optimize(h0, beta, max_iter=500):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t}

    def surrogate(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta

    res = minimize(surrogate, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            x[idx] += per
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
        diff2 = target - np.sum(x)
        if abs(diff2) > 1e-12:
            for i in range(n):
                if abs(diff2) < 1e-12:
                    break
                add = min(diff2, 1.0 - x[i]) if diff2 > 0 else max(diff2, -x[i])
                x[i] += add
                diff2 = target - np.sum(x)
    return x


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, n / 2.0, rng)
    else:
        n = 180
        h = random_feasible_vector(n, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    phase0_end = t0 + budget_s * 0.4
    while time.time() < phase0_end:
        start_vec = perturb_h(best_h, rng, scale=0.6) if rng.random() < 0.7 else random_feasible_vector(n, rng)

        h_hc, c_hc = hill_climb(start_vec, rng, max_iters=2000,
                                 init_step=0.6, n_trials=6)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc

        h_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iter=8000,
                                         init_temp=2e-2, final_temp=1e-6,
                                         init_scale=0.5, final_scale=0.02)
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa

        for beta in (50, 200, 800, 2000, 5000):
            if time.time() >= phase0_end:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=600)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # Upsample if enough time remains
    elapsed = time.time() - t0
    if budget_s - elapsed > budget_s * 0.2:
        new_n = 720
        old_x = (np.arange(n) + 0.5) * (2.0 / n)
        new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
        up_h = np.interp(new_x, old_x, best_h)
        up_h = enforce_sum(up_h, new_n / 2.0, rng)
        best_h = up_h
        n = new_n
        best_c = compute_c5(best_h)

    fine_end = t0 + budget_s * 0.95
    while time.time() < fine_end:
        start_vec = perturb_h(best_h, rng, scale=0.4) if rng.random() < 0.6 else random_feasible_vector(n, rng)

        h_hc, c_hc = hill_climb(start_vec, rng, max_iters=1500,
                                 init_step=0.4, n_trials=6)
        if c_hc < best_c:
            best_h, best_c = h_hc.copy(), c_hc

        h_sa, c_sa = simulated_annealing(best_h, rng,
                                         max_iter=12000,
                                         init_temp=1e-2, final_temp=1e-6,
                                         init_scale=0.35, final_scale=0.015)
        if c_sa < best_c:
            best_h, best_c = h_sa.copy(), c_sa

        for beta in (200, 800, 2000, 5000, 10000, 20000):
            if time.time() >= fine_end:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=800)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
