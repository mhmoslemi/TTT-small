def slsqp_optimize(x0, beta, max_iter=800):
    n = x0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(lambda x: softmax_obj(x, beta),
                   x0,
                   method="SLSQP",
                   bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    # Clip
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple adjust distribution among interior points
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
        # re-correct if needed
        diff = target - x.sum()
        # final adjustments with random
        rng = np.random.default_rng()
        for _ in range(10):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x
```

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)
```

                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def upsample_h(h, target_n, rng):
    n = h.shape[0]
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
    up_h = np.interp(new_x, old_x, h)
    return enforce_sum(up_h, target_n / 2.0, rng)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    cand = [h0]
    for _ in range(3):
        cand.append(random_bump_vector(n, rng, nbumps=4,
                                       width=0.15 + rng.random() * 0.2))
    for _ in range(4):
        cand.append(random_feasible_vector(n, rng))

    best_h, best_c = None, np.inf
    for v in cand:
        c = compute_c5(v)
        if c < best_c:
            best_c, best_h = c, v.copy()

    while time.time() - start < budget_s * 0.6:
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.5)
        else:
            start_vec = random_feasible_vector(n, rng)
        h_hc, c_hc = hill_climb_multi(start_vec, rng,
                                      max_iters=2000, init_step=0.5, n_trials=8)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()
        h_sa, c_sa = simulated_annealing(best_h, rng, max_iter=4000,
                                         init_temp=1e-2, final_temp=1e-5,
                                         init_scale=0.5, final_scale=0.02)
        if c_sa < best_c:
            best_c, best_h = c_sa, h_sa.copy()
        for beta in (10, 50, 200, 800, 2000, 5000, 10000):
            if time.time() - start > budget_s * 0.6:
                break
            h_ref = slsqp_optimize(best_h, beta, max_iter=400)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c:
                best_c, best_h = c_ref, h_ref.copy()

    elapsed = time.time() - start
    if budget_s - elapsed > budget_s * 0.1:
        target_n = min(1080, max(n * 2, n + 240))
        if target_n != n:
            best_h = upsample_h(best_h, target_n, rng)
            n = target_n
            h_hc, c_hc = hill_climb_multi(best_h, rng,
                                          max_iters=2000, init_step=0.4, n_trials=6)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()
            h_sa, c_sa = simulated_annealing(best_h, rng, max_iter=4000,
                                             init_temp=5e-3, final_temp=1e-5,
                                             init_scale=0.4, final_scale=0.02)
            if c_sa < best_c:
                best_c, best_h = c_sa, h_sa.copy()
            for beta in (200, 800, 2000, 5000, 10000):
                if time.time() - start > budget_s:
                    break
                best_h = slsqp_optimize(best_h, beta, max_iter=400)
                c_tmp = compute_c5(best_h)
                if c_tmp < best_c:
                    best_c = c_tmp

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(n)
