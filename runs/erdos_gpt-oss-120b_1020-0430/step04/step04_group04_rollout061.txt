def binary_sa(N, time_limit, rng):
    # initialize
    h = np.zeros(N, dtype=np.int8)
    # Randomly choose weight positions
    indices = rng.choice(N, size=N//2, replace=False)
    h[indices] = 1

    # compute convolution via FFT
    m = 1 << ((2*N-1 - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(h[::-1], m)
    conv_full = np.fft.irfft(H * R, m)[:2*N-1]
    conv = np.rint(conv_full[N-1:]).astype(np.int32) # round to int

    weight = N//2
    cur_min = conv.min()
    cur_max_overlap = weight - cur_min
    best_h = h.copy()
    best_min = cur_min
    best_overlap = cur_max_overlap

    # lists
    ones = list(indices)
    zeros = [i for i in range(N) if h[i]==0]

    max_iter = int(1e8) # large placeholder
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # schedule temperature
        t = it / 1e7  # not known; adapt using iteration count maybe.
        # use exponential schedule using time proportion
        elapsed = time.time() - start
        progress = elapsed / time_limit
        temp = 0.02 * (1e-6/0.02) ** progress  # start 0.02, end 1e-6

        # pick swap
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # compute conv_new via delta (in-place on copy)
        conv_new = conv.copy()
        # delete contributions of i:
        if i+1 < N:
            conv_new[1:N-i] -= h[i+1:N]
        if i > 0:
            conv_new[1:i+1] -= h[i-1::-1]
        # add contributions of j:
        if j+1 < N:
            conv_new[1:N-j] += h[j+1:N]
        if j > 0:
            conv_new[1:j+1] += h[j-1::-1]

        new_min = conv_new.min()
        new_overlap = weight - new_min

        delta = new_overlap - cur_max_overlap
        if delta <= 0 or rng.random() < np.exp(-delta / temp):
            # accept
            h[i] = 0
            h[j] = 1
            # update ones/zeros list: replace i with j
            ones[i_idx] = j
            zeros[j_idx] = i
            # update conv and min/overlap
            conv = conv_new
            cur_min = new_min
            cur_max_overlap = new_overlap
            if cur_max_overlap < best_overlap:
                best_overlap = cur_max_overlap
                best_min = cur_min
                best_h = h.copy()
        it += 1
    return best_h, best_overlap, conv

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers for continuous optimisation ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σx = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max correlation via log‑sum‑exp."""
    # correlation of x with 1‑x (linear, non‑periodic)
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, scaled back
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=3000):
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tiny correction
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP refinement of the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


# ---------- binary simulated annealing ----------
def binary_sa(N, time_limit, rng):
    """Binary (0/1) simulated‑annealing optimisation for C₅."""
    weight = N // 2
    h = np.zeros(N, dtype=np.int8)
    ones_idx = rng.choice(N, size=weight, replace=False)
    h[ones_idx] = 1

    # initial convolution via FFT (exact integer values)
    m = 1 << ((2 * N - 2).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(h[::-1], m)
    full = np.fft.irfft(H * R, m)[: 2 * N - 1]
    conv = np.rint(full[N - 1 :]).astype(np.int16)  # conv[k] for k≥0

    cur_min = int(conv.min())
    cur_overlap = weight - cur_min

    best_h = h.copy()
    best_overlap = cur_overlap
    best_min = cur_min

    # maintain explicit lists of positions
    ones = list(ones_idx)
    zeros = [i for i in range(N) if h[i] == 0]

    start = time.time()
    while time.time() - start < time_limit:
        # temperature schedule (0.02 → 1e-6)
        elapsed = time.time() - start
        progress = elapsed / time_limit
        temp = 0.02 * ((1e-6 / 0.02) ** progress)

        # pick a random 1 and a random 0
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]

        # compute new convolution (copy + delta)
        conv_new = conv.copy()
        # remove contributions of i (which will become 0)
        if i + 1 < N:
            conv_new[1 : N - i] -= h[i + 1 : N]
        if i > 0:
            conv_new[1 : i + 1] -= h[i - 1 :: -1]
        # add contributions of j (which will become 1)
        if j + 1 < N:
            conv_new[1 : N - j] += h[j + 1 : N]
        if j > 0:
            conv_new[1 : j + 1] += h[j - 1 :: -1]

        new_min = int(conv_new.min())
        new_overlap = weight - new_min
        delta = new_overlap - cur_overlap

        if delta <= 0 or rng.random() < np.exp(-delta / temp):
            # accept move
            h[i] = 0
            h[j] = 1
            # update index lists
            ones[i_pos] = j
            zeros[j_pos] = i
            conv = conv_new
            cur_min = new_min
            cur_overlap = new_overlap
            if cur_overlap < best_overlap:
                best_overlap = cur_overlap
                best_min = cur_min
                best_h = h.copy()
    return best_h, best_overlap, best_min


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # target resolution
    N = 720  # use a fine grid
    # ----- binary SA (≈35 % of time) -----
    time_binary = budget_s * 0.35
    best_bin, best_overlap, _ = binary_sa(N, time_binary, rng)
    best_c5 = best_overlap * (2.0 / N)

    # ----- continuous refinement on the same grid -----
    time_elapsed = time.time() - start_time
    time_left = budget_s - time_elapsed
    if time_left > budget_s * 0.1:
        h_cont = best_bin.astype(float)  # start from the binary pattern
        # a few SLSQP passes with increasing β
        for beta in (50, 200, 800, 2000, 5000):
            if time.time() - start_time > budget_s * 0.95:
                break
            h_cont = slsqp_optimize(h_cont, beta, max_iter=400)
        final_c = compute_c5(h_cont)
        if final_c < best_c5:
            best_c5 = final_c
            best_h = h_cont.copy()
        else:
            best_h = h_cont.copy()
    else:
        best_h = best_bin.astype(float)

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, N / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(N)
