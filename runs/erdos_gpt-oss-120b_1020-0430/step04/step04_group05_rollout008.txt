def simulated_annealing(h_start, rng, max_iters, time_start, time_limit):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.02, 1e-5
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        # determine lag of current max corr
        idx = np.argmax(cur_corr)
        lag = idx - (n - 1)
        # choose move
        if rng.random() < 0.7:
            new_h = targeted_perturb(h, rng, abs(lag), top_frac=0.25, scale=0.6)
        else:
            new_h = perturb_h(h, rng, scale=0.6)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=800):
    n = h0.size
    target = n/2.
    # define global variables as closure? we can avoid global using lambda capturing
    # but we need to pass them to optimizer. simpler: define objective function that uses captured beta and target.
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        bc = beta * corr
        m = np.max(bc)
        # log-sum-exp
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    bounds = [(0., 1.)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0., 1.)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust using simple loop
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1. - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def improve_worst_lag(h, rng, frac=0.5):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    if lag == 0:
        return h.copy()
    # contributions indices
    if lag > 0:
        i_vals = np.arange(n - lag)
        contrib = h[i_vals] * (1 - h[i_vals + lag])
    else:
        lag_abs = -lag
        i_vals = np.arange(lag_abs, n)
        contrib = h[i_vals] * (1 - h[i_vals - lag_abs])
    # find max and min indices
    max_idx = i_vals[np.argmax(contrib)]
    min_idx = i_vals[np.argmin(contrib)]
    # ensure we can move mass from max_idx to min_idx respecting bounds
    max_eps = min(h[max_idx], 1. - h[min_idx])
    if max_eps <= 0:
        # fallback random perturb
        return perturb_h(h, rng, scale=0.1)
    eps = frac * max_eps
    new_h = h.copy()
    new_h[max_idx] -= eps
    new_h[min_idx] += eps
    return new_h

def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break

            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def compute_c5_corr(h):
    """Return C5 and full correlation array."""
    corr = np.correlate(h, 1 - h, mode="full")
    c = np.max(corr) * (2.0 / h.size)
    return c, corr

def compute_c5(h):
    c, _ = compute_c5_corr(h)
    return c

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    """Bias the transfer toward indices that dominate the worst lag."""
    n = h.size
    if lag >= n:
        return perturb_h(h, rng, scale)
    valid = np.arange(n - lag)
    contrib = h[valid] * (1.0 - h[valid + lag])
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing(h_start, rng, max_iters, time_start, time_limit):
    """SA on the true max-overlap objective."""
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.02, 1e-5
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        idx = np.argmax(cur_corr)
        lag = idx - (n - 1)
        lag_abs = abs(lag)
        if rng.random() < 0.7:
            new_h = targeted_perturb(h, rng, lag_abs, top_frac=0.25, scale=0.6)
        else:
            new_h = perturb_h(h, rng, scale=0.6)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=800):
    """Smooth max via softmax and SLSQP."""
    n = h0.size
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        bc = beta * corr
        m = np.max(bc)
        return (m + np.log(np.sum(np.exp(bc - m)))) / beta
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

def improve_worst_lag(h, rng, frac=0.5):
    """Move mass from a high‑contribution index to a low‑contribution one for the worst lag."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    if lag == 0:
        return h.copy()
    if lag > 0:
        i_vals = np.arange(n - lag)
        contrib = h[i_vals] * (1.0 - h[i_vals + lag])
    else:
        lag_abs = -lag
        i_vals = np.arange(lag_abs, n)
        contrib = h[i_vals] * (1.0 - h[i_vals - lag_abs])
    max_idx = i_vals[np.argmax(contrib)]
    min_idx = i_vals[np.argmin(contrib)]
    max_eps = min(h[max_idx], 1.0 - h[min_idx])
    if max_eps <= 0:
        return perturb_h(h, rng, scale=0.1)
    eps = frac * max_eps
    new_h = h.copy()
    new_h[max_idx] -= eps
    new_h[min_idx] += eps
    return new_h
