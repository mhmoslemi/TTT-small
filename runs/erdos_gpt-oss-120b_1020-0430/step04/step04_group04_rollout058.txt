def compute_c5(h):
    n = h.size
    # conv = np.correlate(h, 1-h, mode='full')
    # but using FFT is faster.
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i==j:
        return h.copy()
    # max epsilon we can transfer from i (decrease) to j (increase)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random()*2 - 1) * max_eps * scale
    # Clip eps to [-max_eps, max_eps] maybe not needed cause product already limited.
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    # Ensure within [0,1] (should be)
    # small floating errors
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def hill_climb(h0, rng, max_iters=2000, init_step=0.5, patience=0):
    h = h0.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max_iters // 10 if max_iters>=10 else 1
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            h = new_h
        # decay step periodically
        if (it+1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c

def simulated_annealing(h0, rng, max_iter=5000, init_temp=1e-2, final_temp=1e-6, init_scale=0.5, final_scale=0.02):
    n = h0.size
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    for t in range(max_iter):
        progress = t / max_iter
        temp = init_temp * (final_temp / init_temp) ** progress
        scale = init_scale * (final_scale / init_scale) ** progress
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / temp):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c

def softmax_obj(x, beta):
    # x must be numpy array
    n = x.size
    rev = (1 - x)[::-1]
    L = 2*n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    # compute max plus log-sum-exp
    max_c = np.max(conv)
    # log-sum-exp stable
    aligned = conv - max_c
    return max_c + (1.0/beta) * np.log(np.sum(np.exp(beta * aligned)))

def slsqp_optimize(h0, beta, max_iter=300):

    target = n/2.0

def enforce_sum(h, target, rng):
    # Ensure sum of h equals target (sum of entries)
    # While adjusting within [0,1]
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return h
    # Simple iterative adjustments
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # if still off, distribute equally among free indices
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def enforce_sum(h, target, rng, max_iter=5000):
    """Scale/adjust h so that sum(h) == target while 0<=h<=1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.size
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def random_feasible_vector(n, rng):
    """Generate a random feasible h with sum=n/2."""
    h = np.full(n, 0.5)
    target = n / 2.0
    for _ in range(n * 3):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() * 2.0 - 1.0) * max_eps * 0.4
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() * 2.0 - 1.0) * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h0, rng, max_iters=2000, init_step=0.5):
    """Simple stochastic hill‑climb."""
    h = h0.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            h = new_h
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def simulated_annealing(h0, rng, max_iter=5000,
                        init_temp=1e-2, final_temp=1e-6,
                        init_scale=0.5, final_scale=0.02):

def softmax_obj(x, beta):
    """Smooth surrogate of the max‑overlap using log‑sum‑exp."""
    n = x.size
    rev = (1.0 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())

    aligned = conv - max_c
    return max_c + (1.0 / beta) * np.log(np.sum(np.exp(beta * aligned) + 1e-300))

                up_h = slsqp_optimize(up_h, beta, max_iter=300)

            c_up = compute_c5(up_h)
            if c_up < best_c:
                best_c, best_h, n = c_up, up_h.copy(), target_n

    # ---- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
