def compute_c5_and_corr(h):
    n = h.size
    N = 1 << (2*n-1).bit_length()  # next pow of 2 >= 2n-1.
    H = np.fft.fft(h, n=N)
    M = np.fft.fft(1-h, n=N)
    corr = np.fft.ifft(H * np.conj(M)).real
    # correlation "full" corresponds to indices 0..N-1, but actual full corr length = 2*n-1.
    # The standard np.correlate (full) returns shifted sequence: index offset = n-1.
    # In fft, circular correlation emerges; we need to shift.
    # However, using zero-padded sequences of length N >= 2n-1 yields linear convolution.
    # Indeed, linear convolution length = len(a)+len(b)-1 = 2n-1. The result stored in first 2n-1 entries.
    # Use corr[:2*n-1]
    corr = corr[:2*n-1]
    max_corr = corr.max()
    c5 = max_corr * (2.0 / n)
    return c5, corr

def enforce_sum(h, target, rng, tol=1e-12, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tiny correction if needed
    if abs(diff) > tol:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)

    if lag >= 0:
        valid = np.arange(n - lag)
        contrib = h[valid] * (1.0 - h[valid + lag])
    else:
        L = -lag
        valid = np.arange(n - L)
        contrib = h[valid + L] * (1.0 - h[valid])
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def deterministic_improve(h, max_iter, rng):
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        _, corr = compute_c5_corr_fft(h)
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        if lag == 0:
            break
        v = np.zeros(n)
        if lag > 0:
            v[: n - lag] += h[lag:]
            v[lag:] += h[: n - lag]
        else:
            L = -lag
            v[L:] += h[: n - L]
            v[: n - L] += h[L:]
        asc = np.argsort(v)
        desc = asc[::-1]
        i_low = i_high = 0
        changed = False
        while i_low < n and i_high < n:
            low = asc[i_low]
            high = desc[i_high]
            if v[low] >= v[high] - 1e-12:
                break
            transfer = min(h[low], 1.0 - h[high])
            if transfer <= 1e-12:
                if h[low] <= 1e-12:
                    i_low += 1
                if h[high] >= 1.0 - 1e-12:
                    i_high += 1
                continue
            h[low] -= transfer
            h[high] += transfer
            changed = True
            if h[low] <= 1e-12:
                i_low += 1
            if h[high] >= 1.0 - 1e-12:
                i_high += 1
        if not changed:
            break
    return enforce_sum(h, target, rng)

def simulated_annealing(h_start, rng, max_iters, time_start, time_budget):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr_fft(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.02, 1e-5
    for it in range(max_iters):
        if time.time() - time_start > time_budget:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        lag = np.argmax(cur_corr) - (n - 1)
        if rng.random() < 0.7:
            new_h = targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr_fft(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

# ---------- SLSQP ladder ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

# ---------- main ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # initial vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # deterministic improvement on coarse grid
    h_det = deterministic_improve(best_h.copy(), max_iter=200, rng=rng)
    c_det = compute_c5(h_det)
    if c_det < best_c:
        best_h, best_c = h_det, c_det

    # simulated annealing on coarse grid (half budget)
    max_sa_iters = 50000
    sa_h, sa_c = simulated_annealing(best_h, rng, max_sa_iters, t0, t0 + budget_s * 0.5)
    if sa_c < best_c:
        best_h, best_c = sa_h, sa_c

    # optional up‑sampling if time permits
    remaining = budget_s - (time.time() - t0)
    if remaining > budget_s * 0.2:
        target_n = 720 if best_h.size <= 360 else 1080
        if target_n > best_h.size:
            old_x = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)

            # deterministic improvement on finer grid
            up_h = deterministic_improve(up_h, max_iter=200, rng=rng)
            up_c = compute_c5(up_h)
            if up_c < best_c:
                best_h, best_c, n = up_h.copy(), up_c, target_n

            # SA on finer grid (use remaining budget)
            max_sa_iters_up = int(6e4)
            sa_h_up, sa_c_up = simulated_annealing(best_h, rng, max_sa_iters_up,
                                                   t0, t0 + budget_s)
            if sa_c_up < best_c:
                best_h, best_c, n = sa_h_up.copy(), sa_c_up, target_n

    # SLSQP refinement ladder (use remaining time)
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]
    for beta in beta_list:
        if time.time() - t0 >= budget_s:
            break
        try:
            ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=500)
        except Exception:
            continue
        ref_c = compute_c5(ref_h)
        if ref_c < best_c:
            best_h, best_c = ref_h, ref_c

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
