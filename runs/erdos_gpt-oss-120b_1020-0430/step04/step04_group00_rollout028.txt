def improve_pair(h, O, i, j):
    # compute feasible interval
    lower = max(-h[i], h[j] - 1.0)
    upper = min(1.0 - h[i], h[j])
    if lower >= upper - 1e-12:
        return False, None
    # compute derivative diff
    n = h.size
    der_i = np.empty(n)
    der_i[:n-i] = 1.0 - h[i:n]
    der_i[i:] -= h[:i]
    der_j = np.empty(n)
    der_j[:n-j] = 1.0 - h[j:n]
    der_j[j:] -= h[:j]
    diff = der_i - der_j
    # define f(delta)
    def f(delta):
        return np.max(O + delta * diff)
    # ternary search
    lo, hi = lower, upper
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = f(m1)
        f2 = f(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    best_delta = (lo+hi)/2.0
    best_val = f(best_delta)
    # compare to current max
    cur_max = np.max(O)
    if best_val < cur_max - 1e-12:
        # apply update
        h[i] += best_delta
        h[j] -= best_delta
        # recompute O
        # O(k) = sum_{p=0}^{n-1-k} h[p]*(1-h[p+k])
        # We can recompute full O via convolution
        O_new = np.correlate(h, 1.0 - h, mode='full')[n-1:]
        return True, O_new
    else:
        return False, None

def slsqp_optimize(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': sum_eq_constraint, 'args': (target,)}
    res = minimize(lambda x: softmax_obj(x, beta), h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    diff = target - x.sum()
    # Use water-filling random fix
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        O = corr[n-1:]  # length n
        max_val = O.max()
        # indices where max
        worst_idxs = np.where(O == max_val)[0]
        # compute gradient as sum over worst indices
        g = np.zeros_like(h)
        for k in worst_idxs:
            # derivative of O(k) w.r.t h_i:
            # for i such that i <= n-1-k: derivative = 1 - h[i+k]
            # for i >= k: derivative -= h[i-k]
            # combine
            g[:n-k] += 1.0 - h[k:]
            g[k:] -= h[:n-k]
        # note O is sum; we want gradient of max but we use sum of worst gradients
        # step size schedule
        alpha = step * (0.5 ** it)
        h = h - alpha * g
        h = np.clip(h, 0.0, 1.0)
        # fix sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff via random moves
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h
```

    # log‑sum‑exp in a numerically stable way
    return (max_val + np.log(np.sum(np.exp(beta * corr - max_val)))) / beta

def slsqp_optimize(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint, "args": (target,)}
    res = minimize(
        lambda x: softmax_obj(x, beta),
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    return x

def gradient_refine(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        O = corr[n - 1 :]
        max_val = O.max()
        worst = np.where(O == max_val)[0]
        g = np.zeros_like(h)
        for k in worst:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        alpha = step * (0.5 ** it)
        h = h - alpha * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def _derivative_vector(h, idx):
    n = h.size
    der = np.empty(n)
    der[: n - idx] = 1.0 - h[idx:]
    der[idx:] -= h[:idx]
    return der

def pairwise_improve(h, O, i, j, rng):
    n = h.size
    lower = max(-h[i], h[j] - 1.0)
    upper = min(1.0 - h[i], h[j])
    if lower >= upper - 1e-12:
        return False, None
    der_i = _derivative_vector(h, i)
    der_j = _derivative_vector(h, j)
    diff = der_i - der_j

    def f(delta):
        return np.max(O + delta * diff)

    lo, hi = lower, upper
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    best_delta = (lo + hi) / 2.0
    best_val = f(best_delta)
    if best_val < O.max() - 1e-12:
        h[i] += best_delta
        h[j] -= best_delta
        O_new = np.correlate(h, 1.0 - h, mode="full")[n - 1 :]
        return True, O_new
    return False, None

def pairwise_local_search(h, O, rng, deadline, max_attempts=2000):
    n = h.size
    attempts = 0
    no_improve = 0
    while time.time() < deadline and attempts < max_attempts:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        improved, O_new = pairwise_improve(h, O, i, j, rng)
        attempts += 1
        if improved:
            O = O_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 500:
            break
    return h, O

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.99

    best_h = None
    best_c = np.inf
    best_n = None

    parent = globals().get("initial_h_values", None)

    ns = [720, 480, 360, 240]  # larger resolutions first

    for n in ns:
        if time.time() > deadline:
            break

        start_vecs = []

        if parent is not None:
            xs = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            start_vecs.append(interp.copy())
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            bin_vec = np.zeros(n)
            bin_vec[idx] = 1.0
            start_vecs.append(bin_vec)

        # random binary
        bin_rand = np.zeros(n)
        bin_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        start_vecs.append(bin_rand)

        # random fractional
        rand_frac = rng.random(n)
        target = n / 2.0
        rand_frac *= target / rand_frac.sum()
        rand_frac = np.clip(rand_frac, 0.0, 1.0)
        adjust_sum_exact(rand_frac, rng)
        start_vecs.append(rand_frac)

        # smoothed binary
        kernel = np.array([0.2, 0.6, 0.2])
        smooth = np.convolve(bin_rand.astype(float), kernel, mode="same")
        smooth = np.clip(smooth, 0.0, 1.0)
        adjust_sum_exact(smooth, rng)
        start_vecs.append(smooth)

        for h0 in start_vecs:
            if time.time() > deadline:
                break
            h = np.clip(h0, 0.0, 1.0)
            adjust_sum_exact(h, rng)

            # quick gradient polishing
            h = gradient_refine(h, rng, step=0.02, max_iter=8)

            # cascade of smooth‑max SLSQP
            for beta in (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
                if time.time() > deadline:
                    break
                h = slsqp_optimize(h, beta, rng, max_iter=500)
                h = gradient_refine(h, rng, step=0.015, max_iter=4)

            # overlap array
            O = np.correlate(h, 1.0 - h, mode="full")[n - 1 :]

            # pairwise sum‑preserving local search
            h, O = pairwise_local_search(h, O, rng, deadline)

            h = gradient_refine(h, rng, step=0.02, max_iter=8)
            c = compute_c5(h)
            if c < best_c:
                best_h, best_c, best_n = h.copy(), c, n

    # optional up‑sampling of the best candidate
    if (
        best_h is not None
        and time.time() < deadline
        and best_n * 2 <= 1080
    ):
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        for beta in (20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
            h_up = gradient_refine(h_up, rng, step=0.015, max_iter=4)
        O_up = np.correlate(h_up, 1.0 - h_up, mode="full")[n_up - 1 :]
        h_up, O_up = pairwise_local_search(h_up, O_up, rng, deadline, max_attempts=3000)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=12)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    if best_h is None:
        n = 240
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)
        best_n = n

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
