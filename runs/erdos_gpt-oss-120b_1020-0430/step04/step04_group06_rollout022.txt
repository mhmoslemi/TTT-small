def perturb_h(h, rng, scale=0.2):
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)

def compute_c5_fast(h):
    n = h.size
    # compute linear convolution via FFT
    length = 2*n
    fh = np.fft.rfft(h, length)
    gh = np.fft.rfft(1.0 - h, length)
    conv = np.fft.irfft(fh * gh, length)[:2*n-1]
    max_val = conv.max()
    return max_val * (2.0 / n)

def max_overlap_shift_fast(h):
    n = h.size
    length = 2*n
    conv = np.fft.irfft(np.fft.rfft(h, length) * np.fft.rfft(1.0 - h, length), length)[:2*n-1]
    idx = int(np.argmax(conv))
    max_val = float(conv[idx])
    shift = idx - (n - 1)
    return max_val, shift

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth surrogate (softmax) for SLSQP
# ----------------------------------------------------------------------
CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) using log-sum-exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so that Σh = target (0≤h≤1)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            for i in range(n):
                if abs(diff) < 1e-14:
                    break
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h with Σh = n/2 and 0≤h≤1."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(2000):
            if abs(diff) < 1e-14:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps1 = min(h[i], 1.0 - h[j])
    max_eps2 = min(h[j], 1.0 - h[i])
    if max_eps1 <= 0 and max_eps2 <= 0:
        return h.copy()
    if rng.random() < 0.5 and max_eps1 > 0:
        eps = rng.random() * max_eps1 * scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
    elif max_eps2 > 0:
        eps = rng.random() * max_eps2 * scale
        new_h = h.copy()
        new_h[i] += eps
        new_h[j] -= eps
    else:
        new_h = h.copy()
    return new_h


def compute_c5_fast(h):
    """Fast C5 using FFT (same scaling as exact)."""
    n = h.size
    length = 2 * n
    fh = np.fft.rfft(h, length)
    gh = np.fft.rfft(1.0 - h, length)
    conv = np.fft.irfft(fh * gh, length)[: 2 * n - 1]
    return float(conv.max()) * (2.0 / n)


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def max_overlap_shift_fast(h):
    """Return (max_val, shift) for the current h using FFT."""
    n = h.size
    length = 2 * n
    conv = np.fft.irfft(np.fft.rfft(h, length) * np.fft.rfft(1.0 - h, length), length)[
        : 2 * n - 1
    ]
    idx = int(np.argmax(conv))
    max_val = float(conv[idx])
    shift = idx - (n - 1)
    return max_val, shift


def hill_climb(h0, rng, max_iters=2000):
    """Simple hill‑climb using random perturbations."""
    h = h0.copy()
    best_c = compute_c5_fast(h)
    best_h = h.copy()
    step = 0.5
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5_fast(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=400):
    """Refine h using SLSQP on the smooth surrogate with given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def reduce_worst_shift(
    h,
    rng,
    max_iters=5000,
    time_limit=30.0,
    init_factor=0.8,
    decay=0.995,
):
    """
    Targeted mass‑transfer to lower the current worst overlap.
    Moves mass from the source index (largest contribution) to a
    destination with a larger forward neighbour.
    """
    n = h.size
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break

        max_val, shift = max_overlap_shift_fast(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()

        # overlapping index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue

        # source: index with largest product contribution
        prod = h[i_range] * (1.0 - h[j_range])
        src = i_range[int(np.argmax(prod))]

        # forward neighbour values for all indices
        forward_h = np.empty(n)
        if shift >= 0:
            forward_h[: n - shift] = h[shift:]
            forward_h[n - shift :] = 0.0
        else:
            forward_h[-shift :] = h[: n + shift]
            forward_h[: -shift] = 0.0

        # candidate destinations: have capacity and a forward neighbour
        cap_mask = (h < 1.0 - 1e-12) & (forward_h > 1e-12)
        cap_mask[src] = False
        if not np.any(cap_mask):
            cap_mask = (h < 1.0 - 1e-12)
            cap_mask[src] = False
            if not np.any(cap_mask):
                factor = init_factor
                continue

        candidates = np.where(cap_mask)[0]
        scores = forward_h[candidates] * (1.0 - h[candidates])
        tgt = candidates[int(np.argmax(scores))]

        max_eps = min(h[src], 1.0 - h[tgt])
        if max_eps <= 0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0:
            factor = init_factor
            continue

        h[src] -= eps
        h[tgt] += eps
        h[src] = max(0.0, min(1.0, h[src]))
        h[tgt] = max(0.0, min(1.0, h[tgt]))

        factor = max(1e-4, factor * decay)

    final_c = compute_c5_fast(best_h)
    return best_h, final_c

        # Targeted worst‑shift reduction
        remaining = budget_s - (time.time() - start_time)
        if remaining < 15.0:
            break
        # allocate a slice of the remaining time
        time_slice = min(remaining * 0.4, 40.0)
        h_red, c_red = reduce_worst_shift(
            global_best_h,
            rng,
            max_iters=8000,
            time_limit=time_slice,
            init_factor=0.8,
        )
        if c_red < global_best_c:
            global_best_h, global_best_c = h_red.copy(), c_red

    # ------------------------------------------------------------------
    # 5. SLSQP polishing with increasing β
    # ------------------------------------------------------------------
    beta_list = [500, 2000, 8000, 20000]
    for beta in beta_list:
        if time.time() - start_time >= budget_s * 0.95:
            break
        try:
            h_ref = slsqp_optimize(global_best_h, beta, max_iter=500)
            c_ref = compute_c5_fast(h_ref)
            if c_ref < global_best_c:
                global_best_h, global_best_c = h_ref.copy(), c_ref
        except Exception:
            continue

    # ------------------------------------------------------------------
    # 6. Final clean‑up
    # ------------------------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0, rng)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
