import time, numpy as np
from scipy.optimize import minimize

# globals for SLSQP (similar as previous)
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                    constraints=cons,
                    options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def compute_Ovals_fft(h):
    n = h.size
    m = 2 * n
    ft_h = np.fft.fft(h, m)
    ft_1mh = np.fft.fft(1 - h, m)
    conv = np.fft.ifft(ft_h * np.conj(ft_1mh)).real
    O = conv[n-1:n-1+n]  # length n
    return O

def neighbor_at_shift(h, k):
    n = h.shape[0]
    if k >= 0:
        if k >= n:
            return np.zeros(n, dtype=h.dtype)
        return np.concatenate((h[k:], np.zeros(k, dtype=h.dtype)))
    else:
        k = -k
        if k >= n:
            return np.zeros(n, dtype=h.dtype)
        return np.concatenate((np.zeros(k, dtype=h.dtype), h[:n - k]))

def random_binary_vector(n, rng):
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def binary_hillclimb(h, rng, max_iters=3000):
    n = h.size
    O = compute_Ovals_fft(h)
    max_O = O.max()
    T = 0.01  # temperature
    for it in range(max_iters):
        idx_max = O.argmax()
        signed_k = idx_max - (n - 1)
        neighbor = neighbor_at_shift(h, signed_k)
        i_cand = np.where((h == 1) & (neighbor == 0))[0]
        j_cand = np.where((h == 0) & (neighbor == 1))[0]
        made_swap = False
        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
            h[i], h[j] = h[j], h[i]
            O = compute_Ovals_fft(h)
            max_O = O.max()
            made_swap = True
        else:
            # random swap with annealing
            i_ones = np.where(h == 1)[0]
            j_zeros = np.where(h == 0)[0]
            if i_ones.size == 0 or j_zeros.size == 0:
                break
            i = rng.choice(i_ones)
            j = rng.choice(j_zeros)
            h[i], h[j] = h[j], h[i]
            O_new = compute_Ovals_fft(h)
            max_O_new = O_new.max()
            if max_O_new < max_O or rng.random() < np.exp(-(max_O_new - max_O) / max(T, 1e-12)):
                O = O_new
                max_O = max_O_new
                made_swap = True
            else:
                # revert
                h[i], h[j] = h[j], h[i]
        # reduce temperature slowly
        T *= 0.9995
        # optional early break if max_O already low near best known? Not needed.
    return h, max_O * (2.0 / n)

def worst_gradient(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def gradient_refine(h, rng, step=0.015, max_iter=10):
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h

def enforce_sum(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

import time, numpy as np
from scipy.optimize import minimize

# globals for the soft‑max SLSQP objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# --- SLSQP utilities -------------------------------------------------
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)

    # exact sum correction (quick redistribution)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# --- binary‑vector utilities -----------------------------------------
def compute_Ovals_fft(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1‑h) using FFT."""
    n = h.size
    m = 2 * n
    ft_h = np.fft.fft(h, m)
    ft_1mh = np.fft.fft(1 - h, m)
    conv = np.fft.ifft(ft_h * np.conj(ft_1mh)).real
    return conv[: 2 * n - 1]  # length 2n‑1, shifts –(n‑1)…(n‑1)


def neighbor_at_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Values of h at distance k (linear, zero outside)."""
    n = h.shape[0]
    if k >= 0:
        if k >= n:
            return np.zeros(n, dtype=h.dtype)
        return np.concatenate((h[k:], np.zeros(k, dtype=h.dtype)))
    else:
        k = -k
        if k >= n:
            return np.zeros(n, dtype=h.dtype)
        return np.concatenate((np.zeros(k, dtype=h.dtype), h[: n - k]))


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def binary_hillclimb(h: np.ndarray, rng: np.random.Generator,
                    max_iters: int = 2500) -> tuple[np.ndarray, float]:
    """Hill‑climb on binary vectors to reduce max overlap."""
    n = h.size
    O = compute_Ovals_fft(h)
    max_O = O.max()
    T = 0.01  # temperature for occasional annealing moves
    for _ in range(max_iters):
        idx_max = O.argmax()
        signed_k = idx_max - (n - 1)

        neighbor = neighbor_at_shift(h, signed_k)
        i_cand = np.where((h == 1) & (neighbor == 0))[0]
        j_cand = np.where((h == 0) & (neighbor == 1))[0]

        if i_cand.size and j_cand.size:
            i = rng.choice(i_cand)
            j = rng.choice(j_cand)
            h[i], h[j] = h[j], h[i]
            O = compute_Ovals_fft(h)
            max_O = O.max()
        else:
            # random swap with simulated‑annealing acceptance
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = h[j], h[i]
            O_new = compute_Ovals_fft(h)
            max_O_new = O_new.max()
            if max_O_new < max_O or rng.random() < np.exp(-(max_O_new - max_O) / max(T, 1e-12)):
                O, max_O = O_new, max_O_new
            else:
                h[i], h[j] = h[j], h[i]  # revert

        T *= 0.9995
    return h, max_O * (2.0 / n)


# --- continuous‑vector utilities ------------------------------------
def worst_gradient(h: np.ndarray) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 2000, step_scale: float = 0.6) -> tuple[np.ndarray, float]:
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                    step: float = 0.015, max_iter: int = 10) -> np.ndarray:
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[: n - k] += 1.0 - h[k:]
            g[k:] -= h[: n - k]
        h = h - (step * (0.5 ** it)) * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h
