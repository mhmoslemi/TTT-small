    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# ---------- simulated annealing ----------
def simulated_annealing(h_start, rng, max_time, start_time):
    """Time‑bounded SA using full self‑correlation."""
    deadline = start_time + max_time
    h = h_start.copy()
    n = h.size
    corr = np.correlate(h, h, mode='full')
    cur_c = 1.0 - (2.0 / n) * np.min(corr)
    best_h, best_c = h.copy(), cur_c
    max_iters = 50000
    T0, Tf = 0.03, 1e-5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t_frac = it / max_iters
        T = T0 * (Tf / T0) ** t_frac
        lag_idx = np.argmin(corr)
        lag = lag_idx - (n - 1)
        if rng.random() < 0.7:
            new_h = targeted_perturb(h, rng, lag, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_corr = np.correlate(new_h, new_h, mode='full')
        new_c = 1.0 - (2.0 / n) * np.min(new_corr)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initial vector -----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n0 / 2.0, rng)
    else:
        n0 = 360
        h0 = random_feasible_vector(n0, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ----- stage progression (coarse → fine) -----
    stage_ns = [360, 720, 1080, 1440]
    for n in stage_ns:
        # up‑sample if needed
        if best_h.size != n:
            old_n = best_h.size
            old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            best_h = np.interp(new_x, old_x, best_h)
            best_h = enforce_sum(best_h, n / 2.0, rng)
        best_c = compute_c5(best_h)

        # time budgeting for this stage
        now = time.time()
        remaining = budget_s - (now - start)
        if remaining < 12:
            break
        stage_time = min(remaining * 0.5, max(30.0, remaining * 0.2))
        stage_start = time.time()

        # run several SA attempts within the allocated stage time
        while time.time() - stage_start < stage_time and time.time() - start < budget_s - 5:
            if rng.random() < 0.75:
                start_vec = perturb_h(best_h, rng, scale=0.5)
            else:
                start_vec = random_feasible_vector(n, rng)

            # allocate a modest slice for a single SA run
            run_deadline = time.time() + min(20.0, stage_time - (time.time() - stage_start))
            sa_h, sa_c = simulated_annealing(start_vec, rng, max_time=run_deadline - time.time(),
                                             start_time=time.time())
            if sa_c < best_c:
                best_h, best_c = sa_h, sa_c

            # quick local hill‑climb after SA
            cur_h, cur_c = best_h.copy(), best_c
            for _ in range(200):
                if time.time() - start > budget_s:
                    break
                nh = perturb_h(cur_h, rng, scale=0.3)
                nc = compute_c5(nh)
                if nc < cur_c:
                    cur_h, cur_c = nh, nc
                    if nc < best_c:
                        best_h, best_c = nh, nc
                    break
            best_h, best_c = cur_h, cur_c

    # ----- final polishing hill‑climb -----
    cur_h, cur_c = best_h.copy(), best_c
    step = 0.3
    for _ in range(5000):
        if time.time() - start > budget_s:
            break
        nh = perturb_h(cur_h, rng, scale=step)
        nc = compute_c5(nh)
        if nc < cur_c:
            cur_h, cur_c = nh, nc
            if nc < best_c:
                best_h, best_c = nh, nc
        step *= 0.9995

    # final feasibility clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
