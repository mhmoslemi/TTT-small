def compute_derivative_matrix(h):
    n = h.size
    shifts = np.arange(-(n-1), n)
    # shape (n, m)
    i = np.arange(n)[:, None]  # (n,1)
    s = shifts[None, :]        # (1,m)
    plus_idx = i + s
    valid_plus = (plus_idx >= 0) & (plus_idx < n)
    h_plus = np.zeros_like(plus_idx, dtype=float)
    h_plus[valid_plus] = h[plus_idx[valid_plus]]
    minus_idx = i - s
    valid_minus = (minus_idx >= 0) & (minus_idx < n)
    h_minus = np.zeros_like(minus_idx, dtype=float)
    h_minus[valid_minus] = h[minus_idx[valid_minus]]
    d = 1.0 - h_plus - h_minus
    return d.T  # shape (m,n)

def solve_lp_step(h, step):
    n = h.size
    c = np.correlate(h, 1.0 - h, mode='full')
    # Not scaling by dx yet; scaling matters only for comparison with best_c (which uses dx). But t returned from LP corresponds to correlation (not scaled). We'll compute scaled later.

    # Build derivative matrix
    dd_T = compute_dd_T(h)  # shape (m,n)
    # Use CVXPY
    Δ = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        dd_T @ Δ + c <= t,
        cp.sum(Δ) == 0,
        Δ >= -h,
        Δ <= 1 - h,
        Δ >= -step,
        Δ <= step
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=200, abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception as e:
        return None, None
    if Δ.value is None:
        return None, None
    Δ_val = Δ.value
    t_val = t.value
    return Δ_val, t_val

def compute_dd_T(h):

    d[mask_minus] -= h[minus_idx[mask_minus]]
    return d.T

def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0

        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid

    minus_idx = i - s
    d = np.ones((n, shifts.size), dtype=float)
    mask_plus = (plus_idx >= 0) & (plus_idx < n)
    d[mask_plus] -= h[plus_idx[mask_plus]]
    mask_minus = (minus_idx >= 0) & (minus_idx < n)
    d[mask_minus] -= h[minus_idx[mask_minus]]
    return d.T  # shape (m,n)

def solve_lp_step(h, c, dd_T, step):
    n = h.size
    Δ = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        dd_T @ Δ + c <= t,
        cp.sum(Δ) == 0,
        Δ >= -h,
        Δ <= 1 - h,
        Δ >= -step,
        Δ <= step,
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=200, abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:
        return None, None
    if Δ.value is None:
        return None, None
    return Δ.value, t.value

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    target_n = 720

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = upsample_h(h0, target_n)
    else:
        h0 = random_feasible_vector(target_n, rng)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_box_sum(h0, target_n / 2.0)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    h = h0.copy()
    step = 0.30
    min_step = 1e-5
    max_outer = 200
    improve_tol = 1e-12
    it = 0

    while time.time() - start < budget_s * 0.80 and step > min_step and it < max_outer:
        it += 1
        c = np.correlate(h, 1.0 - h, mode="full")
        dd_T = compute_dd_T(h)
        delta, t_val = solve_lp_step(h, c, dd_T, step)
        if delta is None:
            step *= 0.5
            continue

        new_h = h + delta
        new_h = np.clip(new_h, 0.0, 1.0)
        if abs(new_h.sum() - target_n / 2.0) > 1e-8:
            new_h = project_box_sum(new_h, target_n / 2.0)

        new_c = compute_c5(new_h)
        old_c = compute_c5(h)

        if new_c < old_c - improve_tol:
            h = new_h
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
            step = min(step * 1.05, 0.40)
        else:
            step *= 0.7

        if best_c < 0.3808:
            break

    # Small random hill‑climb for final tweaking
    hc_iters = 1500
    hc_step = 0.10
    for _ in range(hc_iters):
        if time.time() - start >= budget_s * 0.95:
            break
        i = rng.integers(target_n)
        j = rng.integers(target_n)
        if i == j:
            continue
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_transfer * hc_step
        eps = max(-max_transfer, min(eps, max_transfer))
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = compute_c5(cand)
        if cand_c < best_c:
            best_h = cand
            best_c = cand_c

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, target_n / 2.0)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(target_n)
