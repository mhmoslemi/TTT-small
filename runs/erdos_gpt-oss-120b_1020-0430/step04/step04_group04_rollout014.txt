def gradient_f(h, beta, M):
    n = h.shape[0]
    sum_h = h.sum()
    # autocorr via rfft
    H = np.fft.rfft(h, M)
    rev = h[::-1]
    R = np.fft.rfft(rev, M)
    autocorr = np.fft.irfft(H * R, M)[:2*n-1]
    conv = sum_h - autocorr
    # weights
    bc = beta * conv
    max_bc = bc.max()
    w = np.exp(bc - max_bc)
    w /= w.sum()
    # Compute conv_w_h = convolution of w and h (full)
    W = np.fft.rfft(w, M)
    conv_w_h = np.fft.irfft(W * H, M)[:(2*n-1)+n-1]  # length 3n-2
    # conv_w_rev = convolution of w and rev
    conv_w_rev = np.fft.irfft(W * R, M)[:(2*n-1)+n-1]
    # G = reverse of conv_w_h first n elements
    G = conv_w_h[::-1][:n]
    # H_slice
    H_slice = conv_w_rev[n-1 : 2*n-1]
    grad = 1.0 - G - H_slice
    return grad

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------ helper functions ------------------------

def compute_c5(h, n=None):
    """Exact C5 for a step function sampled on [0,2] (full correlation)."""
    if n is None:
        n = h.shape[0]
    L = 2 * n - 1
    M = 1 << ((L - 1).bit_length())          # next power of two >= L
    H = np.fft.rfft(h, M)
    R = np.fft.rfft(h[::-1], M)
    autocorr = np.fft.irfft(H * R, M)[:L]
    conv = h.sum() - autocorr
    return conv.max() * (2.0 / n)


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1]^n with Σh = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 0.5
        # move delta from i to j respecting bounds
        delta = max(-h[i], min(delta, 1 - h[i]))
        delta = max(-(1 - h[j]), min(delta, h[j]))
        h[i] -= delta
        h[j] += delta
    # final correction (water‑filling)
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def project_onto_box_sum(x, target, max_iter=100):
    """Project onto {0≤x≤1, Σx=target} using simple water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        x[mask] += delta
        x = np.clip(x, 0.0, 1.0)
    return x


# ----- gradient of the smooth surrogate (log‑sum‑exp) -----
def gradient_f(h, beta, M):
    """Gradient of (1/β)·log∑exp(β·conv) where conv = Σh – autocorr."""
    n = h.shape[0]
    sum_h = h.sum()
    L = 2 * n - 1

    # FFT of h and its reverse
    H = np.fft.rfft(h, M)
    R = np.fft.rfft(h[::-1], M)

    # autocorrelation and conv
    autocorr = np.fft.irfft(H * R, M)[:L]
    conv = sum_h - autocorr

    # softmax weights
    bc = beta * conv
    max_bc = bc.max()
    w = np.exp(bc - max_bc)
    w /= w.sum()

    # FFT of weights
    W = np.fft.rfft(w, M)

    # convolutions w * h  and  w * h[::-1]
    conv_w_h = np.fft.irfft(W * H, M)[:L + n - 1]          # length 3n‑2
    conv_w_rev = np.fft.irfft(W * R, M)[:L + n - 1]

    # G_k = Σ_i w_i·h_rev_{i‑k} = conv_w_h[n‑1 : 2n‑1]
    G = conv_w_h[n - 1 : 2 * n - 1]

    # H_k = Σ_i w_i·h_{i+k-(n‑1)} = reversed(conv_w_rev)[:n]
    H_slice = conv_w_rev[::-1][:n]

    # gradient (sum of components is zero)
    return 1.0 - G - H_slice


# ----- SLSQP on the smooth surrogate -----
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    n = x.shape[0]
    sum_h = x.sum()
    L = 2 * n - 1
    M = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, M)
    R = np.fft.rfft(x[::-1], M)
    autocorr = np.fft.irfft(H * R, M)[:L]
    conv = sum_h - autocorr
    bc = CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = project_onto_box_sum(x, CURRENT_TARGET)
    return x

        tri = np.where(xs <= 1, xs, 2 - xs)
        starts.append(project_onto_box_sum(tri, target_sum))
        cosine = (1 + np.cos(np.pi * (xs - 1))) / 2.0
        starts.append(project_onto_box_sum(cosine, target_sum))

        # a few random vectors

        # ------------------------------------------------------------------
        betas = [10, 50, 200, 800, 3200, 12800]

            h = h_start.copy()
