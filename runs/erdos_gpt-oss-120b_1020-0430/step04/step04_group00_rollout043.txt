def hill_climb(h, max_O, n, rng, deadline):
    best_h = h.copy()
    best_max = max_O
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]; j = zeros[j_idx]
        # perform swap
        h[i] = 0
        h[j] = 1
        # compute new max O
        O_new = np.max(np.correlate(h, 1 - h, mode='full'))
        if O_new < best_max:
            best_max = O_new
            best_h = h.copy()
            # update ones and zeros: replace indices
            ones[i_idx] = j
            zeros[j_idx] = i
        else:
            # revert
            h[i] = 1
            h[j] = 0
        # (optional) break when no improvement for many iterations
    return best_h, best_max

import time
import numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    """Exact sum correction to enforce Σh = n/2."""
    target = h.size / 2.0
    diff = target - np.sum(h)
    n = h.size
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - np.sum(h)
    return h

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=400):
    """SLSQP on the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    adjust_sum_exact(x, rng)
    return x

def greedy_binary(n, rng):
    """Construct a binary sequence minimizing max_k Σ h_i(1‑h_{i+k})."""
    L = n // 2
    h = np.zeros(n, dtype=np.int8)
    O = np.zeros(2 * n - 1, dtype=np.int32)  # correlation of all‑zero vector
    for _ in range(L):
        O_adj = O + 1
        zeros = np.flatnonzero(h == 0)
        best_w = None
        best_ps = []
        for p in zeros:
            start = n - 1 - p
            window = O_adj[start : start + n] - h
            w = window.max()
            if best_w is None or w < best_w:
                best_w = w
                best_ps = [p]
            elif w == best_w:
                best_ps.append(p)
        p = rng.choice(best_ps)
        h[p] = 1
        O = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode="full")
        O[n - 1] = 0  # enforce O_0 = 0
    return h, O

def hill_climb(h_bin, max_O, n, rng, deadline):
    """Simple hill‑climb on binary vectors (swap 1↔0)."""
    best_h = h_bin.copy()
    best_max = max_O
    cur_h = h_bin.copy()
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        O_new = np.max(
            np.correlate(cur_h.astype(np.int32), (1 - cur_h).astype(np.int32), mode="full")
        )
        if O_new < best_max:
            best_max = O_new
            best_h = cur_h.copy()
            # update index lists
            ones[i_idx] = j
            zeros[j_idx] = i
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_max
