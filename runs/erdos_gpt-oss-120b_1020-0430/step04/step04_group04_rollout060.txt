def project_onto_capped_simplex(v, target):
    # target is sum
    # Ensure feasible: target between 0 and n (n= len(v)), and target <= n (since each ≤1)
    lo = np.max(v - 1)  # λ lower bound
    hi = np.min(v)      # λ upper bound
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0, 1)
    # due to numeric, adjust to meet exact target via small correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # adjust within free variables (0 < x < 1)
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            # distribute equally
            x[free] += diff / free.sum()
            x = np.clip(x, 0, 1)
        else:
            # fallback: adjust all
            x += diff / len(v)
            x = np.clip(x, 0, 1)
        # Might still not achieve exactly due to clipping; iterate a couple times
    return x

def compute_c5_and_subgrad(h):
    n = h.size
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L-1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    C = conv.max()
    # find indices of max within tolerance
    tol = 1e-12
    idxs = np.where(conv >= C - tol)[0]
    # compute gradient average
    grad = np.zeros_like(h)
    for idx in idxs:
        s = idx - (n - 1)  # shift offset
        if s >= 0:
            # term1: 1 - h[s:] added to grad[0:n-s]
            grad[0:n - s] += (1 - h[s:])
            # term2: -h[:n-s] subtracted from grad[s:]
            grad[s:] -= h[:n - s]
        else:
            t = -s
            # term1: 1 - h[:n - t] added to grad[t:]
            grad[t:] += (1 - h[:n - t])
            # term2: -h[t:] subtracted from grad[:n - t]
            grad[:n - t] -= h[t:]
    grad /= len(idxs)
    return C * (2.0 / n), grad
```

def c5_and_subgrad(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    max_val = conv.max()
    idxs = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        s = idx - (n - 1)
        if s >= 0:
            grad[:n - s] += (1 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            t = -s
            grad[t:] += (1 - h[:n - t])
            grad[:n - t] -= h[t:]
    grad /= len(idxs)
    # scale gradient for scaled objective
    grad *= (2.0 / n)
    return max_val * (2.0 / n), grad
```

import time
import numpy as np

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x | 0<=x_i<=1, sum x_i = target}."""
    n = v.shape[0]
    lo = np.max(v - 1.0)          # λ lower bound
    hi = np.min(v)                # λ upper bound
    # binary search for λ
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x


def _random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return _project_capped_simplex(v, n / 2.0)


def _compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv.max() * (2.0 / n)


def _c5_and_subgrad(h: np.ndarray):
    """Return (C5 value, gradient of C5) with respect to h."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    max_val = conv.max()
    idxs = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        s = idx - (n - 1)          # shift offset
        if s >= 0:
            grad[: n - s] += (1.0 - h[s:])
            grad[s:]      -= h[: n - s]
        else:
            t = -s
            grad[t:]      += (1.0 - h[: n - t])
            grad[: n - t] -= h[t:]
    grad /= len(idxs)
    grad *= (2.0 / n)              # scale for C5 = conv_max * (2/n)
    return max_val * (2.0 / n), grad

    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 360
        h = np.full(n, 0.5, dtype=float)

    h = np.clip(h, 0.0, 1.0)
    h = _project_capped_simplex(h, n / 2.0)

    best_h = h.copy()
    best_c = _compute_c5(best_h)
    coarse_h, coarse_c, coarse_n = best_h.copy(), best_c, n

    # ------------------------------------------------------------
    # Subgradient descent with simulated annealing on coarse grid
    # ------------------------------------------------------------
    max_iters = 200_000
    step_init = 30.0                     # large w.r.t. scaled gradient
    decay = 0.9999
    temp0 = 0.01
    patience = 8_000
    no_improve = 0

    h_cur = best_h.copy()
    c_cur = best_c

    for it in range(max_iters):
        if time.time() - start > budget_s * 0.85:
            break

        c_cur, grad = _c5_and_subgrad(h_cur)
        step = step_init * (decay ** it)
        h_new = h_cur - step * grad
        h_new = _project_capped_simplex(h_new, n / 2.0)

        c_new = _compute_c5(h_new)

        # SA acceptance
        temp = temp0 * (1.0 - it / max_iters)
        if c_new < c_cur or (temp > 0.0 and rng.random() < np.exp(-(c_new - c_cur) / temp)):
            h_cur, c_cur = h_new, c_new

        if c_new < best_c - 1e-9:
            best_c, best_h = c_new, h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            h_cur = _random_feasible_vector(n, rng)
            c_cur = _compute_c5(h_cur)
            no_improve = 0

    # ------------------------------------------------------------
    # Optional up‑sampling and refinement
    # ------------------------------------------------------------
    elapsed = time.time() - start
    if budget_s - elapsed > budget_s * 0.12:
        up_n = min(1440, n * 2)
        if up_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(up_n) + 0.5) * (2.0 / up_n)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = _project_capped_simplex(h_up, up_n / 2.0)

            # refinement on the finer grid
            max_iters_up = 80_000
            step_init_up = 30.0
            decay_up = 0.9999
            temp0_up = 0.01
            patience_up = 6_000

            h_cur = h_up.copy()
            c_cur = _compute_c5(h_cur)
            best_h_fine = h_cur.copy()
            best_c_fine = c_cur
            no_improve = 0

            for it in range(max_iters_up):
                if time.time() - start > budget_s:
                    break
                c_cur, grad = _c5_and_subgrad(h_cur)
                step = step_init_up * (decay_up ** it)
                h_new = h_cur - step * grad
                h_new = _project_capped_simplex(h_new, up_n / 2.0)
                c_new = _compute_c5(h_new)

                temp = temp0_up * (1.0 - it / max_iters_up)
                if c_new < c_cur or (temp > 0.0 and rng.random() < np.exp(-(c_new - c_cur) / temp)):
                    h_cur, c_cur = h_new, c_new

                if c_new < best_c_fine - 1e-9:
                    best_c_fine, best_h_fine = c_new, h_new.copy()
                    no_improve = 0
                else:
                    no_improve += 1

                if no_improve >= patience_up:
                    h_cur = _random_feasible_vector(up_n, rng)
                    c_cur = _compute_c5(h_cur)
                    no_improve = 0

            # compare fine solution with coarse one
            if best_c_fine < best_c:
                best_c, best_h, n = best_c_fine, best_h_fine.copy(), up_n
            else:
                n = coarse_n
                best_h = coarse_h.copy()
                best_c = coarse_c
        else:
            n = coarse_n
            best_h = coarse_h.copy()
            best_c = coarse_c
    else:
        n = coarse_n
        best_h = coarse_h.copy()
        best_c = coarse_c

    # ------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------
    best_h = _project_capped_simplex(best_h, n / 2.0)
    final_c = _compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
