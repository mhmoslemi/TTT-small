def gradient_step(h, lag, step):
    n = h.size
    g = np.zeros_like(h)
    if lag > 0:
        # forward shift
        g[:n - lag] += h[lag:]  # h_{i+lag}
        g[lag:] += h[:n - lag]  # h_{i-lag}
    elif lag < 0:
        lag2 = -lag
        g[lag2:] += h[:n - lag2]
        g[:n - lag2] += h[lag2:]
    else:
        # lag 0: product sum = sum h_i^2 ; gradient = 2 h_i
        g = 2 * h
    # Step
    new_h = h + step * g
    # Project
    new_h = np.clip(new_h,0,1)
    # Enforce sum
    new_h = enforce_sum(new_h, n/2., rng)
    return new_h

def gradient_step(h, lag, step, rng):
    n = h.size
    g = np.zeros_like(h)
    if lag > 0:
        g[:n - lag] += h[lag:]
        g[lag:] += h[:n - lag]
    elif lag < 0:
        L = -lag
        g[L:] += h[:n - L]
        g[:n - L] += h[L:]
    else:
        g = 2 * h
    new_h = h + step * g
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = enforce_sum(new_h, n / 2.0, rng)
    return new_h

def softmax_obj(x):

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):

# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):

    x = x / x.sum() * (n / 2.0)

            nc, nlag, _ = compute_c5_and_info(nh)
            if nc < cur_c - 1e-12:
                cur_h, cur_c, cur_lag = nh, nc, nlag
                step = min(step * 1.02, 0.1)
                if nc < best_c:
                    best_h, best_c = nh.copy(), nc
            else:
                step *= 0.5
                if step < 1e-5:
                    step = 1e-5
                    cur_h = perturb_h(cur_h, rng, scale=0.2)
                    cur_c, cur_lag, _ = compute_c5_and_info(cur_h)
        if time.time() < main_end:
            sa_h, sa_c = simple_sa(cur_h, rng, max_iters=4000, deadline=main_end)
            if sa_c < best_c:
                best_h, best_c = sa_h.copy(), sa_c

    remaining = budget_s - (time.time() - t0)
    if remaining > budget_s * 0.2:
        target_n = min(1080, max(n, int(best_h.size * 1.5)))
        if target_n != best_h.size:
            old_x = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)
            cur_h = up_h
            cur_c, cur_lag, _ = compute_c5_and_info(cur_h)
            step = 0.02
            up_end = t0 + budget_s * 0.95
            for it in range(15000):
                if time.time() > up_end:
                    break
                nh = gradient_step(cur_h, cur_lag, step, rng)
                nc, nlag, _ = compute_c5_and_info(nh)
                if nc < cur_c - 1e-12:
                    cur_h, cur_c, cur_lag = nh, nc, nlag
                    step = min(step * 1.02, 0.1)
                else:
                    step *= 0.5
                    if step < 1e-5:
                        step = 1e-5
                        cur_h = perturb_h(cur_h, rng, scale=0.2)
                        cur_c, cur_lag, _ = compute_c5_and_info(cur_h)
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
            if time.time() < t0 + budget_s * 0.95:
                sa_h, sa_c = simple_sa(best_h, rng, max_iters=6000, deadline=t0 + budget_s * 0.95)
                if sa_c < best_c:
                    best_h, best_c = sa_h.copy(), sa_c

    final_deadline = t0 + budget_s * 0.99
    for beta in [10, 50, 200, 1000, 5000]:
        if time.time() > final_deadline:
            break
        try:
            refined = slsqp_optimize_one_beta(best_h, beta, rng)
        except Exception:
            continue
        refined = enforce_sum(refined, best_h.size / 2.0, rng)
        rc = compute_c5_and_info(refined)[0]
        if rc < best_c:
            best_h, best_c = refined, rc

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5_and_info(best_h)[0]
    return best_h.tolist(), float(final_c), int(best_h.size)
