def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    # simple uniform correction first
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # random walk fix for any remaining tiny error
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    return enforce_sum(h, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# Correlation utilities (FFT based)

def correlate_full(h):
    """Full linear correlation of h with (1‑h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    n = h.size
    corr = correlate_full(h)
    return corr.max() * (2.0 / n)


# --------------------------------------------------------------
# Gradient of smooth max (log‑sum‑exp) of the overlap
# --------------------------------------------------------------
def compute_gradient(h, w):
    """
    Gradient of smooth max of corr = h ⋆ (1‑h) with respect to h.
    w are the softmax weights (sum to 1) of the same length as corr.
    """
    n = h.size
    g = np.zeros_like(h)
    # lag runs from -(n-1) … (n-1)
    for lag in range(-(n - 1), n):
        wk = w[lag + n - 1]
        if wk == 0.0:
            continue
        if lag >= 0:
            l = n - lag
            if l > 0:
                # (1 - h_{j+lag}) term
                g[:l] += wk * (1.0 - h[lag:])
                # -h_{j‑lag} term
                g[lag:] -= wk * h[:l]
        else:
            l = -lag
            m = n - l
            if m > 0:
                # (1 - h_{j+lag}) term (here j+lag = j‑l)
                g[l:] += wk * (1.0 - h[:m])
                # -h_{j‑lag} term (j‑lag = j+l)
                g[:m] -= wk * h[l:]
    return g

def gradient_descent(h0, betas, steps_per_beta, lr0, deadline, rng):
    """Multi‑beta gradient descent on the smooth max of the overlap."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    it_total = 0
    n = h.size
    target = n / 2.0

    for beta in betas:
        lr = lr0 / np.sqrt(beta)
        for _ in range(steps_per_beta):

                return best_h, best_c

            w = np.exp(bc - log_sum)            # softmax weights
            grad = compute_gradient(h, w)
            # simple gradient step

def binary_sa(start_bin, rng, deadline):
    """Swap‑based simulated annealing on a binary vector (sum = n/2)."""
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)

    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 30000

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # ensure i is the 1 and j the 0 for a swap
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            # revert the swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

            best_c, best_h = c_opt, h_opt.copy()

        # occasional greedy hill‑climb
        if time.time() < phase1_end:
            h_hc, c_hc = hill_climb(best_h, rng, max_iters=2000, start_step=0.2)
            if c_hc < best_c:
                best_c, best_h = c_hc, h_hc.copy()

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to a finer grid and continue optimisation
    # ------------------------------------------------------------------
    # up‑sample by factor 2 (max 1440 points)
    if best_h.size * 2 <= 1440:
        factor = 2
        n_fine = best_h.size * factor
        # linear interpolation on the interval [0,2)
        xs_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, n_fine, endpoint=False)
        h_fine = np.interp(xs_new, xs_old, best_h, left=best_h[0], right=best_h[-1])
        h_fine = enforce_sum(h_fine, n_fine / 2.0, rng)

        phase2_end = start_time + budget_s * 0.75
        betas_fine = [50.0, 200.0, 800.0, 2000.0, 8000.0, 20000.0]
        steps_per_beta_fine = 500

        h_opt_f, c_opt_f = gradient_descent(
            h_fine, betas_fine, steps_per_beta_fine, lr0=0.5, deadline=phase2_end, rng=rng
        )
        if c_opt_f < best_c:
            best_c, best_h = c_opt_f, h_opt_f.copy()

        # extra hill‑climb on the fine grid
        if time.time() < phase2_end:
            h_hc_f, c_hc_f = hill_climb(best_h, rng, max_iters=3000, start_step=0.15)
            if c_hc_f < best_c:
                best_c, best_h = c_hc_f, h_hc_f.copy()

    # ------------------------------------------------------------------
    # Phase 3 – binary simulated annealing and final polishing
    # ------------------------------------------------------------------
    phase3_end = start_time + budget_s * 0.95
    if time.time() < phase3_end:
        n_final = best_h.size
        idx_top = np.argsort(-best_h)[: n_final // 2]
        bin_start = np.zeros(n_final)
        bin_start[idx_top] = 1.0
        bin_h, bin_c = binary_sa(bin_start.astype(float), rng, phase3_end)
        if bin_c < best_c:
            best_c, best_h = bin_c, bin_h.copy()

        if time.time() < phase3_end:
            h_pol, c_pol = hill_climb(best_h, rng, max_iters=3000, start_step=0.1)
            if c_pol < best_c:
                best_c, best_h = c_pol, h_pol.copy()

    # ------------------------------------------------------------------
    # Optional short SLSQP polish with a very high β
    # ------------------------------------------------------------------
    if time.time() < deadline:
        try:
            h_slsqp = slsqp_optimize(best_h, beta=20000.0, max_iter=200)
            c_slsqp = compute_c5(h_slsqp)
            if c_slsqp < best_c:
                best_c, best_h = c_slsqp, h_slsqp.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Final cleanup and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
