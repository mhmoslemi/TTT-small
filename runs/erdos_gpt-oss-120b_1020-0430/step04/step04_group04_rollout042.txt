def project_onto_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    # water-filling algorithm
    # while sum not target, adjust free entries
    for _ in range(10):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        free = (x > 1e-12) & (x < 1-1e-12)
        if free.any():
            per = diff / free.sum()
            x[free] += per
            x = np.clip(x, 0.0, 1.0)
        else:
            # all at bounds, adjust all small amount then clip
            per = diff / x.size
            x += per
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)
```

def enforce_sum(h, target, rng, max_iter=5000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # pick index among free (not at bound)
        # if diff positive, need to increase some entries
        # choose random index with h[i] < 1
        if diff > 0:
            idx = rng.integers(0, n)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = rng.integers(0, n)
            add = max(diff, -h[idx])
            h[idx] += add
        diff = target - h.sum()
    # final small correction using free entries
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.any():
            per = diff / free.sum()
            h[free] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h
```

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    # assume CURRENT_BETA set
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # fix sum if drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def enforce_sum(h, target, rng, max_iter=2000):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:

        else:

            h += diff / n

    return h

def autocorr_fft(h):
    n = h.shape[0]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    A = np.fft.irfft(H * np.conj(H), m)[:L]
    return A.real  # ignore tiny imag

def shift_with_pad(v, shift):
    n = v.size
    out = np.zeros_like(v)
    if shift > 0:
        out[shift:] = v[:n-shift]
    elif shift < 0:
        s = -shift
        out[:n-s] = v[s:]
    else:
        out[:] = v
    return out

def improve_h(h, rng, time_limit, start_step=0.5):
    n = h.size
    target = n / 2.0
    step = start_step
    best_h = h.copy()
    A = autocorr_fft(h)
    minA = A.min()
    best_c = 1.0 - minA * (2.0 / n)
    it = 0
    while time.time() < time_limit:
        it += 1

        minA = A[idx]

        if it % 1000 == 0 and time.time() > time_limit - 5.0:

def compute_c5(h):
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)

        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def autocorr_fft(h):
    """Linear autocorrelation via FFT (no scaling)."""
    n = h.shape[0]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    A = np.fft.irfft(H * np.conj(H), m)[:L]
    return A.real


def shift_with_pad(v, shift):
    """Shift with zero padding: positive shift → right."""
    n = v.size
    out = np.zeros_like(v)
    if shift > 0:
        out[shift:] = v[: n - shift]
    elif shift < 0:
        s = -shift
        out[: n - s] = v[s:]
    else:
        out[:] = v
    return out


def improve_h(h, rng, time_limit, start_step=0.5):
    """Local optimisation: maximise the minimum autocorrelation."""
    n = h.size
    target = n / 2.0
    step = start_step
    # current C5 via autocorrelation
    A = autocorr_fft(h)
    cur_c = 1.0 - A.min() * (2.0 / n)
    best_h_local = h.copy()
    best_c_local = cur_c
    it = 0
    while time.time() < time_limit:
        it += 1
        A = autocorr_fft(h)
        idx = int(np.argmin(A))
        minA = A[idx]
        cur_c = 1.0 - minA * (2.0 / n)
        k = idx - (n - 1)
        # gradient direction for the worst lag
        d = shift_with_pad(h, -k) + shift_with_pad(h, k)
        cur_step = step * (0.9995 ** it)
        h_new = h + cur_step * d
        h_new = enforce_sum(h_new, target, rng)
        A_new = autocorr_fft(h_new)
        c_new = 1.0 - A_new.min() * (2.0 / n)
        # accept improvements
        if c_new < cur_c - 1e-12:
            h = h_new
            cur_c = c_new
        else:
            if rng.random() < 0.005:
                h = h_new
        if cur_c < best_c_local:
            best_c_local = cur_c
            best_h_local = h.copy()
        if it % 500 == 0 and time.time() > time_limit - 2.0:
            break
    return best_h_local


def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def random_feasible_vector(n, rng):
    """Draw a random vector in [0,1]^n with Σh = n/2."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c = np.inf

    # grid sizes to explore (increasing resolution)
    n_grid = [256, 512, 720, 1024]

    for n in n_grid:
        if time.time() - start_time > budget_s * 0.95:
            break
        target = n / 2.0
        xs = (np.arange(n) + 0.5) * (2.0 / n)

        # base candidate: symmetric triangular bump (integrates to 1)
        h0 = np.where(xs <= 1.0, xs, 2.0 - xs)
        h0 = enforce_sum(h0, target, rng)

        # if we already have a decent solution, up‑sample it
        if best_h is not None and best_h.size != n:
            old_x = (np.arange(best_h.size) + 0.5) * (2.0 / best_h.size)
            h0 = np.interp(xs, old_x, best_h)
            h0 = enforce_sum(h0, target, rng)

        # time budget for this resolution
        remaining = budget_s - (time.time() - start_time)
        grad_time = min(remaining * 0.5, 240.0)  # at most 4 min per level
        t_end = time.time() + grad_time

        # local optimisation from the current start
        h_opt = improve_h(h0, rng, t_end, start_step=0.5)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

        # a few random restarts (cheap, help escape bad basins)
        for _ in range(3):
            if time.time() - start_time > budget_s * 0.95:
                break
            h_rand = random_feasible_vector(n, rng)
            t_local = time.time() + 20.0  # short burst
            h_local = improve_h(h_rand, rng, t_local, start_step=0.5)
            c_local = compute_c5(h_local)
            if c_local < best_c:
                best_c, best_h = c_local, h_local.copy()

        # SLSQP refinement ladder (smooth surrogate)
        for beta in (50, 200, 800, 2000, 5000):
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                h_ref = slsqp_optimize(best_h, beta, max_iter=500)
                c_ref = compute_c5(h_ref)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref.copy()
            except Exception:
                continue

        # final random sampling at this resolution
        for _ in range(200):
            if time.time() - start_time > budget_s * 0.95:
                break
            h_samp = random_feasible_vector(n, rng)
            c_samp = compute_c5(h_samp)
            if c_samp < best_c:
                best_c, best_h = c_samp, h_samp.copy()

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    if best_h is None:
        n = 360
        best_h = np.full(n, 0.5)
    n = best_h.size
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
