def binary_sa(N, rng, time_limit):
    import time, numpy as np
    start = time.time()
    # Setup start vectors: use initial if exists else random
    best_h = None
    best_min_A = -1
    best_c = np.inf

    # Pre-allocate arrays for speed
    # Main loop: repeat restarts
    while time.time() - start < time_limit:
        # choose start method:
        if "initial_h_values" in globals() and N == len(initial_h_values):
            h = np.array(initial_h_values, dtype=float)
            # convert to binary
            h_bin = (h > 0.5).astype(np.int8)
            # adjust to exact sum
            diff = N//2 - h_bin.sum()
            if diff > 0:
                zeros = np.where(h_bin == 0)[0]
                add_idx = rng.choice(zeros, size=diff, replace=False)
                h_bin[add_idx] = 1
            elif diff < 0:
                ones = np.where(h_bin == 1)[0]
                rem_idx = rng.choice(ones, size=-diff, replace=False)
                h_bin[rem_idx] = 0
        else:
            # random binary vector
            h_bin = np.zeros(N, dtype=np.int8)
            ones_idx = rng.choice(N, size=N//2, replace=False)
            h_bin[ones_idx] = 1

        # compute A
        corr = np.correlate(h_bin, h_bin, mode='full')
        A = corr[N-1:].astype(np.int32)  # length N
        min_A = A[1:].min()
        max_O = N//2 - min_A
        c = max_O * (2.0/N)

        # keep best
        if c < best_c:
            best_c = c
            best_h = h_bin.copy()
            best_min_A = min_A

        # Prepare indices list
        ones_idx = list(np.where(h_bin==1)[0])
        zeros_idx = list(np.where(h_bin==0)[0])
        current_A = A
        current_min_A = min_A
        current_max_O = max_O
        current_h = h_bin

        # SA parameters
        T0 = N/2.0  # initial temperature
        # set max_iter based on time limit: approximate iteration time 0.0003 sec? Not sure.
        # We'll compute as while loop using time check.
        iter_count = 0
        # Accept zero-temperature for final greed improvements?
        while time.time() - start < time_limit:
            # temperature linear decay
            elapsed = time.time() - start
            # compute T based on proportion of allocated time used
            T = T0 * max(0.001, 1 - (elapsed / time_limit))
            # choose random swap
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)

            # compute delta
            delta = np.zeros(N, dtype=np.int32)
            if i+1 < N:
                delta[1:N-i] -= current_h[i+1:]
            if i > 0:
                delta[1:i+1] -= current_h[:i][::-1]
            if j+1 < N:
                delta[1:N-j] += current_h[j+1:]
            if j > 0:
                delta[1:j+1] += current_h[:j][::-1]
            d = abs(i - j)
            if d > 0:
                delta[d] -= 1

            new_A = current_A + delta
            new_min_A = new_A[1:].min()
            new_max_O = N//2 - new_min_A
            new_c = new_max_O * (2.0/N)

            accept = False
            if new_c < current_max_O * (2.0/N):
                accept = True
            else:
                # accept with probability exp(-(new_c - current_c)/T)
                # current_c = current_max_O * (2/N)
                # delta_c = new_c - current_c
                delta_c = new_c - (current_max_O * (2.0/N))
                if T > 0 and rng.random() < np.exp(-delta_c / T):
                    accept = True

            if accept:
                # perform swap
                current_h[i] = 0
                current_h[j] = 1
                # update indices lists
                ones_idx.remove(i)
                zeros_idx.append(i)
                zeros_idx.remove(j)
                ones_idx.append(j)
                # update A and min/max
                current_A = new_A
                current_min_A = new_min_A
                current_max_O = new_max_O
                # update best if improved
                if new_c < best_c:
                    best_c = new_c
                    best_h = current_h.copy()
                    best_min_A = new_min_A
            # else reject, do nothing

            iter_count += 1
            # possible early break if time low
            if time.time() - start > time_limit * 0.99:
                break

    # done, return best_h as float array (0/1) and best_c
    return best_h.astype(float), best_c

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:]       -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      max_iters: int = 1500, step_scale: float = 0.6):
    """Hill‑climb using gradient‑directed transfers."""
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h = c, nh.copy()
            h = nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def binary_sa(N: int, rng: np.random.Generator, time_limit: float):
    """Simulated annealing on binary vectors (weight N/2). Returns best float h."""
    start = time.time()
    # initialise with a reasonable start
    if "initial_h_values" in globals() and len(initial_h_values) == N:
        h = (np.array(initial_h_values, dtype=float) > 0.5).astype(np.int8)
        diff = N // 2 - h.sum()
        if diff > 0:
            zeros = np.where(h == 0)[0]
            add = rng.choice(zeros, diff, replace=False)
            h[add] = 1
        elif diff < 0:
            ones = np.where(h == 1)[0]
            rem = rng.choice(ones, -diff, replace=False)
            h[rem] = 0
    else:
        h = np.zeros(N, dtype=np.int8)
        h[rng.choice(N, N // 2, replace=False)] = 1

    corr = np.correlate(h, h, mode="full")
    A = corr[N - 1 :].astype(np.int32)          # autocorrelation for shifts 0..N-1
    min_A = A[1:].min()
    cur_c = (N // 2 - min_A) * (2.0 / N)

    best_h = h.astype(float).copy()
    best_c = cur_c

    # bookkeeping
    ones_idx = list(np.where(h == 1)[0])
    zeros_idx = list(np.where(h == 0)[0])
    cur_A = A
    cur_min_A = min_A
    cur_c = cur_c
    cur_h = h

    delta = np.empty(N, dtype=np.int32)
    last_improve = 0
    restart_limit = 200_000

    while time.time() - start < time_limit:
        # temperature linear decay to a tiny value
        elapsed = time.time() - start
        T = max(1e-8, (N / 2.0) * (1.0 - elapsed / time_limit))

        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        delta.fill(0)
        if i + 1 < N:
            delta[1 : N - i] -= cur_h[i + 1 :]
        if i > 0:
            delta[1 : i + 1] -= cur_h[:i][::-1]
        if j + 1 < N:
            delta[1 : N - j] += cur_h[j + 1 :]
        if j > 0:
            delta[1 : j + 1] += cur_h[:j][::-1]
        d = abs(i - j)
        if d > 0:
            delta[d] -= 1  # correct the i‑j double‑count

        new_A = cur_A + delta
        new_min_A = new_A[1:].min()
        new_c = (N // 2 - new_min_A) * (2.0 / N)

        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_c - cur_c) / T):
                accept = True

        if accept:
            # perform the swap
            cur_h[i] = 0
            cur_h[j] = 1
            ones_idx.remove(i)
            zeros_idx.append(i)
            zeros_idx.remove(j)
            ones_idx.append(j)

            cur_A = new_A
            cur_min_A = new_min_A
            cur_c = new_c
            last_improve = 0

            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.astype(float).copy()
        else:
            last_improve += 1

        if last_improve > restart_limit:
            # restart with a fresh random configuration
            cur_h = np.zeros(N, dtype=np.int8)
            cur_h[rng.choice(N, N // 2, replace=False)] = 1
            corr = np.correlate(cur_h, cur_h, mode="full")
            cur_A = corr[N - 1 :].astype(np.int32)
            cur_min_A = cur_A[1:].min()
            cur_c = (N // 2 - cur_min_A) * (2.0 / N)
            ones_idx = list(np.where(cur_h == 1)[0])
            zeros_idx = list(np.where(cur_h == 0)[0])
            last_improve = 0

    return best_h, best_c
