        cur_c = compute_c5(h)
        best_local_h = h.copy()
        best_local_c = cur_c

def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def perturb_h(h, rng, scale=0.2):

    n = h.size

def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Log‑sum‑exp smoothing of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def hill_climb(h_start, rng, max_iters=20000, start_step=0.5):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()

            h = new_h

    cand = []

    tri = 1.0 - np.abs(xs - 1.0)
    cand.append(enforce_sum(np.clip(tri, 0.0, 1.0), target_sum, rng))

    s1 = 0.5 + 0.5 * np.sin(np.pi * xs)
    cand.append(enforce_sum(np.clip(s1, 0.0, 1.0), target_sum, rng))

    c1 = 0.5 + 0.5 * np.cos(np.pi * xs)
    cand.append(enforce_sum(np.clip(c1, 0.0, 1.0), target_sum, rng))

    for _ in range(2):
        cand.append(random_feasible_vector(n, rng))

    for h in cand:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- Simulated annealing (continuous) -------------------------------
    sa_deadline = start_time + 0.7 * budget_s
    h = best_h.copy()
    cur_c = best_c
    best_local_h = h.copy()
    best_local_c = cur_c

    T0 = cur_c * 0.1 + 1e-9
    Tf = 1e-8
    start_sa = time.time()
    while time.time() < sa_deadline:
        # temperature based on elapsed SA time
        tfrac = (time.time() - start_sa) / max(sa_deadline - start_sa, 1e-12)
        T = T0 * ((Tf / T0) ** tfrac)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        # adapt step size with tfrac
        eps_factor = 0.5 * (1.0 - tfrac) + 0.05
        eps = (rng.random() - 0.5) * 2.0 * max_eps * eps_factor
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        new_c = compute_c5(h_new)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            h = h_new
            cur_c = new_c
            if cur_c < best_local_c:
                best_local_c = cur_c
                best_local_h = h.copy()
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()

    # ----- targeted gradient descent on worst shift -----------------------
    if time.time() < deadline:
        h = gradient_descent_on_max(best_local_h, rng, target_sum,
                                    steps=40, alpha=0.015)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # ----- fine hill climb ------------------------------------------------
    if time.time() < deadline:
        h_hc, c_hc = hill_climb(best_h, rng, max_iters=40000, start_step=0.25)
        if c_hc < best_c:
            best_c, best_h = c_hc, h_hc.copy()

    # ----- final polishing with SLSQP (high β) ---------------------------
    beta_series = [800.0, 2000.0, 5000.0, 10000.0, 20000.0]
    for beta in beta_series:
        if time.time() >= deadline:
            break
        try:
            best_h = slsqp_optimize(best_h, beta, max_iter=600)
            best_c = compute_c5(best_h)
        except Exception:
            continue

    # ----- final feasibility enforcement ----------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_sum, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
