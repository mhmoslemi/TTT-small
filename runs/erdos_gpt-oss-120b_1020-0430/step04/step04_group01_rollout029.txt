import time, numpy as np, scipy.optimize as opt

def enforce_sum(h, target, rng, max_iter=5000):
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    # start binary half half
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    # perturb a few times
    for _ in range(n):
        h = perturb_h(h, rng, scale=0.4)
    # adjust sum exact
    return enforce_sum(h, n/2, rng)

def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1 - h[j])
    if max_eps <= 0:
        return h
    eps = (rng.random()*2-1)*scale*max_eps
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new

# correlation and compute c5
def correlate_full(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1-h)[::-1], L)
    conv = np.fft.irfft(fh*fr, L)[:2*n-1]
    return conv

def compute_c5(h):
    n = h.shape[0]
    conv = correlate_full(h)
    return np.max(conv)*(2.0/n)

# smoothed objective
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def soft_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m))))/CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    cons = {"type":"eq","fun":sum_eq_constraint}
    res = opt.minimize(soft_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x,0.0,1.0)
    # ensure sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff)>1e-12:
        # adjust via enforce_sum
        # using rng from outer context? we can pass rng global.
        # define a global rng variable
        x = enforce_sum(x, CURRENT_TARGET, GLOBAL_RNG)
    return x

# hill climb
def hill_climb(h_start, rng, max_iters=3000, step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters//10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it+1)%decay == 0:
            step *= 0.7
            if step < 1e-5:
                step = 1e-5
    return best_h, best_c

# continuous SA
def continuous_sa(start_h, rng, deadline):
    h = start_h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    T0 = best_c*0.02 + 1e-6
    Tf = 1e-6
    max_iters = 20000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # epsilon magnitude proportional to temperature perhaps
        # Use random uniform in [-max_eps, max_eps] * (temp factor)
        eps = (rng.random()*2 -1) * max_eps * 0.5
        # ensure in bounds
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf/T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T,1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
    return best_h, best_c

# binary SA
def binary_sa(start_bin, rng, deadline):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    # initial T
    T0 = best_c*0.02 + 1e-6
    Tf = 1e-6
    max_iters = 20000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # ensure i is 1, j is 0
        if h[i] < h[j]:
            i,j = j,i
        # swap
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf/T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T,1e-12)):
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

import time
import numpy as np
import scipy.optimize as opt

# global RNG for functions that need it
GLOBAL_RNG = None


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h so Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def random_feasible_vector(n, rng):
    """Binary start + a few perturbations."""
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    for _ in range(2 * n):
        h = perturb_h(h, rng, scale=0.4)
    return enforce_sum(h, n / 2.0, rng)


def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1.0 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    n = h.shape[0]
    conv = correlate_full(h)
    return np.max(conv) * (2.0 / n)


# --------------------------------------------------------------
# Smoothed objective for SLSQP
# --------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def soft_obj(x):
    corr = correlate_full(x)
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=400):
    """SLSQP on smooth surrogate (preserves sum via equality constraint)."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = opt.minimize(
        soft_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, GLOBAL_RNG)
    return x


# --------------------------------------------------------------
# Greedy hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h_start, rng, max_iters=3000, step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


# --------------------------------------------------------------
# Simulated annealing on continuous vectors
# --------------------------------------------------------------
def continuous_sa(start_h, rng, deadline):
    h = start_h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 15000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() * 2.0 - 1.0) * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] += eps
            h[j] -= eps
    return best_h, best_c


# --------------------------------------------------------------
# Simulated annealing on binary vectors
# --------------------------------------------------------------
def binary_sa(start_bin, rng, deadline):
    h = start_bin.astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 20000
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    global GLOBAL_RNG
    GLOBAL_RNG = np.random.default_rng(seed)
    rng = GLOBAL_RNG
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # Initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Small pool of random starts
    # ------------------------------------------------------------------
    top = [(best_c, best_h.copy())]
    for _ in range(4):
        cand = random_feasible_vector(n, rng)
        c = compute_c5(cand)
        top.append((c, cand.copy()))
    top.sort(key=lambda x: x[0])
    best_c, best_h = top[0]

    # ------------------------------------------------------------------
    # Phase 1: smooth SLSQP + hill‑climb + continuous SA
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # start point
        start = perturb_h(best_h, rng, scale=0.3) if rng.random() < 0.7 else random_feasible_vector(n, rng)

        # ladder of betas
        for beta in (5, 20, 80, 200, 800, 2000, 5000):
            if time.time() >= deadline - 30.0:
                break
            try:
                start = slsqp_optimize(start, beta, max_iter=400)
            except Exception:
                continue
            c = compute_c5(start)
            if c < best_c:
                best_c, best_h = c, start.copy()
                top.append((c, start.copy()))
                top.sort(key=lambda x: x[0])
                top = top[:5]

        if time.time() >= deadline - 30.0:
            break

        # greedy hill‑climb
        cand, c = hill_climb(best_h, rng, max_iters=2500, step=0.3)
        if c < best_c:
            best_c, best_h = c, cand.copy()
            top.append((c, cand.copy()))
            top.sort(key=lambda x: x[0])
            top = top[:5]

        if time.time() >= deadline - 30.0:
            break

        # occasional continuous SA
        if rng.random() < 0.3:
            cand, c = continuous_sa(best_h, rng, deadline - 5.0)
            if c < best_c:
                best_c, best_h = c, cand.copy()
                top.append((c, cand.copy()))
                top.sort(key=lambda x: x[0])
                top = top[:5]

    # ------------------------------------------------------------------
    # Phase 2: binary SA from current best
    # ------------------------------------------------------------------
    idx = np.argsort(-best_h)[: n // 2]
    bin_start = np.zeros(n)
    bin_start[idx] = 1.0
    cand, c = binary_sa(bin_start, rng, deadline - 5.0)
    if c < best_c:
        best_c, best_h = c, cand.copy()

    # ------------------------------------------------------------------
    # Phase 3: up‑sampling and final refinement
    # ------------------------------------------------------------------
    max_n = 1440
    if best_h.size * 2 <= max_n and time.time() < deadline - 5.0:
        factor = 2
        h_fine = np.repeat(best_h, factor)
        n_fine = h_fine.size
        h_fine = enforce_sum(h_fine, n_fine / 2.0, rng)

        # hill‑climb on fine grid
        cand, c = hill_climb(h_fine, rng, max_iters=5000, step=0.2)
        if c < best_c:
            best_c, best_h = c, cand.copy()

        # a few SLSQP passes with larger β
        for beta in (2000, 5000, 8000):
            if time.time() >= deadline:
                break
            try:
                cand = slsqp_optimize(best_h, beta, max_iter=300)
            except Exception:
                continue
            c = compute_c5(cand)
            if c < best_c:
                best_c, best_h = c, cand.copy()

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
