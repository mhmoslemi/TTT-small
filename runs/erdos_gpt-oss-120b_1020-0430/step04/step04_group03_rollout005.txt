def random_interval_configuration(K, rng):
    # generate lengths sum to 1
    lengths = rng.exponential(1.0, size=K)
    lengths = lengths / lengths.sum()
    # gaps: K+1
    gaps = rng.exponential(1.0, size=K+1)
    gaps = gaps / gaps.sum()
    return lengths, gaps

def build_binary_h_from_intervals(n, lengths, gaps):
    # compute start positions
    starts = np.empty(len(lengths))
    ends = np.empty_like(starts)
    pos = gaps[0]  # start of first interval
    for i in range(len(lengths)):
        starts[i] = pos
        ends[i] = pos + lengths[i]
        pos = ends[i] + gaps[i+1]
    # ensure ends[-1] <= 2
    # sample points
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.zeros(n, dtype=float)
    for s,e in zip(starts, ends):
        # mask for xs in [s, e)
        mask = (xs >= s) & (xs < e)
        h[mask] = 1.0
    # due to rounding may not sum to exactly n/2, enforce sum
    h = enforce_sum(h, n/2.0, rng)
    return h
```

def random_interval_search(n, K, iterations, rng):

import time, numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def enforce_sum(v, target, rng):
    """Clip to [0,1] and adjust to meet exact sum (simple projection)."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # distribute the deficit/excess uniformly respecting bounds
    while abs(diff) > 1e-12:
        if diff > 0:
            free = np.where(v < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            add = min(diff / free.size, 1.0 - v[free]).min()
            v[free] += add
            diff -= add * free.size
        else:
            free = np.where(v > 1e-12)[0]
            if free.size == 0:
                break
            sub = min(-diff / free.size, v[free]).min()
            v[free] -= sub
            diff += sub * free.size
        # safety break
        if diff != 0 and (free.size == 0):
            break
    # final fix on a random coordinate if needed
    if abs(diff) > 1e-12:
        i = rng.integers(0, v.size)
        v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v

def compute_c5(h):
    """Discrete C5, scale by dx = 2/n."""
    n = h.size
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s = -s
        grad[s:] += (1 - h[: n - s])
        grad[: n - s] -= h[s:]
    return grad

def greedy_optimize(h, max_steps=2000):
    """Move mass from a positive to a negative gradient coordinate."""
    for _ in range(max_steps):
        # worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        s = idx - (h.size - 1)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        eps = min(h[i], 1.0 - h[j])
        if eps < 1e-12:
            break
        # try full move
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half move
        eps *= 0.5
        nh[i] = h[i] - eps
        nh[j] = h[j] + eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

def perturb_random(h, rng, scale=0.2):
    """Transfer a random amount of mass between two coordinates."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = max_eps * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

# ---------- interval based construction ----------
def random_interval_configuration(K, rng):
    """Return lengths (K) and gaps (K+1) that sum to 1 each."""
    lengths = rng.exponential(1.0, size=K)
    lengths = lengths / lengths.sum()
    gaps = rng.exponential(1.0, size=K + 1)
    gaps = gaps / gaps.sum()
    return lengths, gaps

def build_binary_h_from_intervals(n, lengths, gaps, rng):
    """Binary step function on a grid of n points."""
    starts = []
    pos = gaps[0]
    for l, g in zip(lengths, gaps[1:]):
        starts.append(pos)
        pos += l + g
    # actual interval ends
    ends = [s + l for s, l in zip(starts, lengths)]
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.zeros(n, dtype=float)
    for s, e in zip(starts, ends):
        mask = (xs >= s) & (xs < e)
        h[mask] = 1.0
    h = enforce_sum(h, n / 2.0, rng)
    return h

# ---------- SLSQP refinement ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- main entry ----------

    start_time = time.time()

    # baseline from parent if available

    else:
        # simple binary start with 3 intervals
        lengths, gaps = random_interval_configuration(3, rng)
        best_h = build_binary_h_from_intervals(target_n, lengths, gaps, rng)
    best_c = compute_c5(best_h)

    # --------- coarse interval search (20% budget) ----------
    n_small = 240
    K_options = [3, 4]
    while time.time() - start_time < budget_s * 0.20:

        # quickly reject bad candidates

        h_big = build_binary_h_from_intervals(target_n, lengths, gaps, rng)

    # --------- stochastic local refinement (up to 85% budget) ----------
    while time.time() - start_time < budget_s * 0.85:
        start = perturb_random(best_h, rng, scale=0.5)
        local = greedy_optimize(start, max_steps=4000)
        c_loc = compute_c5(local)
        if c_loc < best_c:

        # occasional SLSQP polishing

                except Exception:

                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined, c_ref

    # --------- final tiny perturbations (up to 99% budget) ----------
    while time.time() - start_time < budget_s * 0.99:

        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # final greedy sweep
    best_h = greedy_optimize(best_h, max_steps=5000)
    best_c = compute_c5(best_h)

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target_n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(target_n)
