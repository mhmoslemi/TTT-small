CURRENT_TARGET = None
CURRENT_BETA = None

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA
def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET
def slsqp_optimize(h, beta, rng, max_iter=800):
    global CURRENT_TARGET, CURRENT_BETA
    n = h.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h, method="SLSQP", bounds=bounds,
                   constraints=cons,
                   options={"maxiter": max_iter, "ftol": 1e-12, "disp": False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time, numpy as np
from scipy.optimize import minimize

# ------------------- basic utilities ------------------- #
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, rng)


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad


# ------------------- local optimisation ------------------- #
def local_optimize(h0: np.ndarray, max_iters: int, step0: float,
                   rng: np.random.Generator, time_limit: float = None):
    """Projected gradient descent on the worst‑shift objective."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    c_curr = compute_c5(h)
    best_h, best_c = h.copy(), c_curr
    step = step0
    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break
        grad = worst_gradient(h)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = enforce_sum(h_new, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        if c_new <= c_curr:
            h, c_curr = h_new, c_new
        step *= 0.9997
        if step < 1e-6:
            step = step0
    return best_h, best_c


# ------------------- SLSQP refinement ------------------- #
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 800) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # final exact sum repair
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ------------------- tiny random move ------------------- #
def small_random_move(h: np.ndarray, rng: np.random.Generator, scale: float = 0.02):
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    max_eps = min(h[i], 1.0 - h[j]) if h[i] > 0 else 0.0
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    h2 = h.copy()
    h2[i] -= eps
    h2[j] += eps
    return h2
