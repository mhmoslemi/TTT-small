def compute_corr(h):
    n = len(h)
    L = 1 << ((2 * n - 1).bit_length())
    # use real FFT for speed
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]
    return conv

def compute_c5(h):
    conv = compute_corr(h)
    return conv.max() * (2.0 / len(h))

def project_onto_box_simplex(y, target):
    # solve min_{x∈[0,1]^n, sum x = target} ||x - y||^2
    # can use bisection for lambda.
    # The solution is x_i = min(1, max(0, y_i - λ)) where λ chosen such that sum x_i = target.
    # Since function sum_i min(1, max(0, y_i - λ)) is monotonic decreasing in λ, we can bisection.
    lo = -np.max(y) - 1.0  # low enough
    hi = np.max(y) + 1.0   # high enough
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
    # due to numeric issues, may need to adjust small difference
    diff = target - x.sum()

        mask = (x > 1e-12) & (x < 1 - 1e-12)
        nfree = mask.sum()
        if nfree > 0:
            x[mask] += diff / nfree
            x = np.clip(x, 0.0, 1.0)
            # recompute diff if still off, but likely fine.
    return x

def project_onto_box_simplex(y, target):
    n = y.size
    lo = -np.max(np.abs(y)) - 10.0
    hi = np.max(np.abs(y)) + 10.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust diff if needed (rare)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff evenly among free entries
        free = (x > 1e-12) & (x < 1 - 1e-12)
        nfree = free.sum()
        if nfree > 0:
            x[free] += diff / nfree
            x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------
def compute_corr(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h: np.ndarray) -> float:
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = compute_corr(h)
    return conv.max() * (2.0 / h.size)


def gradient_at_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the overlap term g_k at the shift achieving the current max."""
    n = h.size
    g = np.zeros(n, dtype=float)
    if shift >= 0:
        s = shift
        g[: n - s] = 1.0 - h[s:]
        g[s:] -= h[: n - s]
    else:
        s = -shift
        g[s:] = 1.0 - h[: n - s]
        g[: n - s] -= h[s:]
    return g


def project_onto_box_simplex(y: np.ndarray, target: float) -> np.ndarray:
    """Projection of y onto {x∈[0,1]^n : sum x = target} (Euclidean)."""
    lo = -np.max(np.abs(y)) - 10.0
    hi = np.max(np.abs(y)) + 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        nfree = int(free.sum())
        if nfree:
            x[free] += diff / nfree
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point in [0,1]^n with prescribed sum = n/2."""
    y = rng.uniform(0.0, 1.0, n)
    return project_onto_box_simplex(y, n * 0.5)


# --------------------------------------------------------------
# Gradient descent with projection
# --------------------------------------------------------------
def gradient_descent(
    h_start: np.ndarray,
    target_sum: float,
    max_iters: int,
    step0: float,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max‑overlap objective."""
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    cur_h = h_start.copy()
    cur_c = best_c

    n = h_start.size
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = compute_corr(cur_h)
        r = int(np.argmax(conv))
        shift = r - (n - 1)
        grad = gradient_at_shift(cur_h, shift)
        step = step0 / np.sqrt(it + 1.0)
        y = cur_h - step * grad
        new_h = project_onto_box_simplex(y, target_sum)
        new_c = compute_c5(new_h)

        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        else:
            if rng.random() < 5e-4:
                cur_h, cur_c = new_h, new_c

        # occasional random jitter to escape plateaus
        if (it + 1) % 500 == 0 and rng.random() < 0.1:
            jitter = rng.normal(scale=0.04, size=n)
            cur_h = project_onto_box_simplex(cur_h + jitter, target_sum)
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c

    return best_h, best_c


# --------------------------------------------------------------
# Simple hill‑climb refinement
# --------------------------------------------------------------
def hill_climb_refine(
    h_start: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 5000,
    step_scale: float = 0.2,
) -> tuple[np.ndarray, float]:
    """Greedy transfer of mass between two coordinates."""
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    n = best_h.size

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * step_scale
        new_h = best_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        c = compute_c5(new_h)
        if c < best_c:
            best_h, best_c = new_h, c
    return best_h, best_c

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ----- initial feasible vector ---------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_box_simplex(h0, n * 0.5)
    best_h, best_c = h0.copy(), compute_c5(h0)

    # ----- Phase 1: projected gradient descent with many restarts ---
    phase1_deadline = start_time + 0.55 * budget_s
    max_iters_per_restart = 1800
    step0 = 0.6
    restart = 0
    while time.time() < phase1_deadline:
        if restart >= 40:
            break
        # start point: perturb current best or fresh random
        if rng.random() < 0.6:
            pert = rng.normal(scale=0.07, size=n)
            start = project_onto_box_simplex(best_h + pert, n * 0.5)
        else:
            start = random_feasible_vector(n, rng)
        cur_h, cur_c = gradient_descent(
            start,
            n * 0.5,
            max_iters_per_restart,
            step0,
            rng,
            phase1_deadline,
        )
        if cur_c < best_c:
            best_h, best_c = cur_h.copy(), cur_c
        restart += 1

    # ----- Phase 2: optional up‑sampling and refinement ------------
    remaining = deadline - time.time()
    if remaining > 0.2 * budget_s and best_h.size * 2 <= 1440:
        # up‑sample by simple repetition
        factor = 2
        fine_h = np.repeat(best_h, factor)
        n_fine = fine_h.size
        fine_target = n_fine * 0.5
        fine_h = project_onto_box_simplex(fine_h, fine_target)

        fine_deadline = time.time() + remaining * 0.7
        fine_h, fine_c = gradient_descent(
            fine_h,
            fine_target,
            max_iters=2500,
            step0=0.4,
            rng=rng,
            deadline=fine_deadline,
        )
        if fine_c < best_c:
            best_h, best_c = fine_h, fine_c

    # ----- Phase 3: final greedy polishing -------------------------
    final_deadline = deadline
    best_h, best_c = hill_climb_refine(best_h, rng, final_deadline, max_iters=4000)

    # ----- Ensure constraints and return ---------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_simplex(best_h, best_h.size * 0.5)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
