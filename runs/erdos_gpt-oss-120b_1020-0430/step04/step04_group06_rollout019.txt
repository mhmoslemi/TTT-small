def enforce_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = (1.0 - h).sum()
        if cap <= 0:
            return h
        add = diff * (1.0 - h) / cap
        h = h + add
    else:
        cap = h.sum()
        if cap <= 0:
            return h
        sub = diff * h / cap  # diff negative, sub negative
        h = h + sub
    # Clip again due to numeric errors
    h = np.clip(h, 0.0, 1.0)
    # May still have tiny diff; adjust via final correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

# Global variables for the smooth surrogate used by SLSQP

CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0


def enforce_sum(h, target):
    """Project h onto the simplex {0≤h≤1, sum(h)=target}."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = (1.0 - h).sum()
        if cap > 0:
            h += diff * (1.0 - h) / cap
    else:
        cap = h.sum()
        if cap > 0:
            h += diff * h / cap
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    """Exact C5 for a discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Uniform random vector with sum=n/2 and 0≤h≤1."""
    h = rng.random(n)
    return enforce_sum(h, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer ε from one index to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new = h.copy()
    new[i] -= eps
    new[j] += eps
    return new


def hill_climb_multi(h_start, rng, max_iters=1500, init_step=0.4, n_trials=6, time_limit=None):
    """Simple hill‑climb with multiple random perturbations per step."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay = max(1, max_iters // 10)
    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
        if (it + 1) % decay == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine h using SLSQP on the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = enforce_sum(x, CURRENT_TARGET)
    return x


def max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx - (n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)


def pair_transfer_optimize(
    h0,
    rng,
    max_iters=12000,
    time_limit=30.0,
    init_factor=0.5,
    decay=0.999,
):
    """Specialised local search moving mass along the worst shift."""
    h = h0.copy()
    n = h.size
    best_h, best_c = h.copy(), compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c


def legendre_sequence(p):
    """Return a 0/1 Legendre symbol sequence for prime p (length p)."""
    seq = np.empty(p, dtype=float)
    seq[0] = 0.0
    for i in range(1, p):
        ls = pow(i, (p - 1) // 2, p)
        seq[i] = 1.0 if ls == 1 else 0.0
    return seq


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -------------------------------------------------
    # 1. Choose discretisation size and initial vector
    # -------------------------------------------------
    n_target = 720
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
        if n_target % base.size == 0:
            repeat = n_target // base.size
            h0 = np.repeat(base, repeat)
        else:
            xs = np.linspace(0, 2, base.size, endpoint=False)
            xnew = np.linspace(0, 2, n_target, endpoint=False)
            h0 = np.interp(xnew, xs, base)
        h0 = enforce_sum(h0, n_target / 2.0)
        n = n_target
    else:
        n = n_target
        h0 = random_feasible_vector(n, rng)

    # -------------------------------------------------
    # 2. Build a modest pool of diverse starters
    # -------------------------------------------------
    candidates = [h0]

    # random starters
    for _ in range(3):
        candidates.append(random_feasible_vector(n, rng))

    # Legendre‑based starter
    p = 719
    leg = legendre_sequence(p)
    if n % p == 0:
        leg_up = np.repeat(leg, n // p)
    else:
        xs = np.linspace(0, 2, p, endpoint=False)
        xnew = np.linspace(0, 2, n, endpoint=False)
        leg_up = np.interp(xnew, xs, leg)
    leg_up = enforce_sum(leg_up, n / 2.0)
    candidates.append(leg_up)

    # pick the best among the pool
    best_h, best_c = None, float("inf")
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    global_best_h, global_best_c = best_h, best_c

    # -------------------------------------------------
    # 3. Main optimisation phase (hill‑climb + SLSQP ladder)
    # -------------------------------------------------
    main_limit = budget_s * 0.70
    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]

    while time.time() - start_time < main_limit:
        # starting point
        if rng.random() < 0.6:
            start_vec = perturb_h(global_best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        # hill climb
        h_hc, c_hc = hill_climb_multi(
            start_vec,
            rng,
            max_iters=1500,
            init_step=0.4,
            n_trials=6,
            time_limit=main_limit - (time.time() - start_time),
        )
        if c_hc < global_best_c:
            global_best_h, global_best_c = h_hc.copy(), c_hc

        # SLSQP refinement ladder
        for beta in beta_list:
            if time.time() - start_time >= main_limit:
                break
            try:
                h_ref = slsqp_optimize(global_best_h, beta, max_iter=300)
                c_ref = compute_c5(h_ref)
                if c_ref < global_best_c:
                    global_best_h, global_best_c = h_ref.copy(), c_ref
            except Exception:
                continue

    # -------------------------------------------------
    # 4. Pair‑transfer fine tuning with the remaining time
    # -------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20.0:
        attempts = max(1, int(remaining // 30))
        per_attempt = remaining / (attempts + 1)
        best_h, best_c = global_best_h, global_best_c
        for _ in range(attempts):
            h_start = (
                perturb_h(best_h, rng, scale=0.2)
                if rng.random() < 0.5
                else best_h.copy()
            )
            h_opt, c_opt = pair_transfer_optimize(
                h_start,
                rng,
                max_iters=8000,
                time_limit=per_attempt,
                init_factor=0.5,
            )
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
                global_best_h, global_best_c = best_h.copy(), best_c
        # final long run
        h_opt, c_opt = pair_transfer_optimize(
            global_best_h,
            rng,
            max_iters=15000,
            time_limit=remaining * 0.9,
            init_factor=0.5,
        )
        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt.copy(), c_opt

    # -------------------------------------------------
    # 5. Final clean‑up and return
    # -------------------------------------------------
    global_best_h = np.clip(global_best_h, 0.0, 1.0)
    global_best_h = enforce_sum(global_best_h, n / 2.0)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
