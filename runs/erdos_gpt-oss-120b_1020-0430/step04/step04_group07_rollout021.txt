def worst_shift_and_gradient(h):

def project_sum_box(v, target):
    lo = -np.max(v)
    hi = 1 - np.min(v)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + lo, 0.0, 1.0)
    # adjust for any residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff among non-bound entries
        # find indices not at bound
        idxs = np.where((x > 0) & (x < 1))[0]
        if idxs.size == 0:
            # if all at bounds, just adjust globally (unlikely)
            x[0] = np.clip(x[0] + diff, 0, 1)
        else:
            # distribute equally
            delta = diff / idxs.size
            for i in idxs:
                x[i] = np.clip(x[i] + delta, 0, 1)
            # correct residual
            diff2 = target - x.sum()
            if abs(diff2) > 1e-12:
                # apply greedy
                for i in idxs:
                    add = min(diff2, 1.0 - x[i]) if diff2 > 0 else max(diff2, -x[i])
                    x[i] += add
                    diff2 = target - x.sum()
                    if abs(diff2) < 1e-12:
                        break
    return x

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution loop
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

import time, numpy as np
from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v+λ onto [0,1]^n with Σ = target (λ found by bisection)."""
    lo = -np.max(v)
    hi = 1 - np.min(v)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + lo, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        idxs = np.where((x > 0) & (x < 1))[0]
        if idxs.size == 0:
            x[0] = np.clip(x[0] + diff, 0, 1)
        else:
            delta = diff / idxs.size
            for i in idxs:
                x[i] = np.clip(x[i] + delta, 0.0, 1.0)
            diff2 = target - x.sum()
            if abs(diff2) > 1e-12:
                for i in idxs:
                    add = min(diff2, 1.0 - x[i]) if diff2 > 0 else max(diff2, -x[i])
                    x[i] += add
                    diff2 = target - x.sum()
                    if abs(diff2) < 1e-12:
                        break
    return x


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force Σh = len(h)/2 while staying in [0,1]."""
    target = h.size / 2.0
    if abs(h.sum() - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    return project_sum_box(h, target)


def worst_shift_and_gradient(h: np.ndarray):
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        grad = 1 - 2 * h
    else:
        grad = 1 - np.roll(h, -k) - np.roll(h, k)
    return k, grad, corr.max()


def worst_gradient(h: np.ndarray) -> np.ndarray:
    _, grad, _ = worst_shift_and_gradient(h)
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(h: np.ndarray, rng: np.random.Generator, max_iters: int = 2000, step_scale: float = 0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def gradient_descent(h: np.ndarray, rng: np.random.Generator, max_iter: int = 2000, step0: float = 0.5):
    target = h.size / 2.0
    for t in range(max_iter):
        grad = worst_gradient(h)
        eta = step0 / np.sqrt(t + 1)
        h = project_sum_box(h - eta * grad, target)
        if (t + 1) % 500 == 0:
            jitter = (rng.random(h.size) - 0.5) * 0.01
            h = project_sum_box(h + jitter, target)
    return h


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    best_h = None
    best_c = np.inf
    best_n = None

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, rng)
        best_h, best_c, best_n = h0.copy(), compute_c5(h0), h0.size

    if best_h is None:
        n = 256
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
        best_h, best_c, best_n = h0, compute_c5(h0), n

    n_candidates = [96, 120, 144, 168, 192, 216, 240, 256, 288, 320, 352, 384]

    # ------------------- main search -------------------
    while time.time() - start < 0.85 * budget_s:
        n = int(rng.choice(n_candidates))
        fresh = rng.random() < 0.2

        if not fresh and best_h is not None:
            if best_n == n:
                h = best_h.copy()
            else:
                x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                h = np.interp(x_new, x_old, best_h)
                h = np.clip(h, 0, 1)
                h = enforce_sum(h, rng)
        else:
            h = np.zeros(n)
            h[rng.choice(n, n // 2, replace=False)] = 1.0

        # phase A: guided hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=2500, step_scale=0.8)

        # phase B: gradient descent
        h = gradient_descent(h, rng, max_iter=1500, step0=0.5)
        c = compute_c5(h)

        # phase C: SLSQP polish
        for beta in (50.0, 200.0, 2000.0, 8000.0):
            if time.time() - start > 0.93 * budget_s:
                break
            h = slsqp_optimize(h, beta, rng, max_iter=400)
            c = compute_c5(h)

        # phase D: short hill‑climb
        h, c = guided_hill_climb(h, rng, max_iters=1200, step_scale=0.6)

        if c < best_c:
            best_h, best_c, best_n = h.copy(), c, n

    # -------------- final refinement on best --------------
    if best_h is not None:
        for _ in range(3):
            if time.time() - start > 0.96 * budget_s:
                break
            best_h = gradient_descent(best_h, rng, max_iter=800, step0=0.3)
            best_c = compute_c5(best_h)

    # -------------- stochastic tweaks till time limit --------------
    while time.time() - start < budget_s * 0.998 and best_h is not None:
        grad = worst_gradient(best_h)
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
