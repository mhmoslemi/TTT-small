def simulated_annealing(h_start, rng, max_iters, time_limit, t0):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h = h.copy()
    best_c = cur_c
    n = len(h)
    T0, Tf = 0.02, 1e-5
    for it in range(max_iters):
        # time check
        if time.time() - t0 > time_limit:
            break
        # schedule
        T = T0 * (Tf / T0) ** (it / max_iters)
        # decide move type
        r = rng.random()
        lag = np.argmax(cur_corr) - (n - 1)
        if r < 0.5:
            # targeted epsilon move
            new_h = targeted_perturb(h, rng, abs(lag), top_frac=0.2, scale=0.5)
        elif r < 0.8:
            # swap move
            # choose top vs bottom
            if abs(lag) > 0:
                # use contributions
                contrib = get_contributions(h, lag)
                k = max(1, int(len(contrib) * 0.1))
                top_idx = np.argpartition(-contrib, k-1)[:k]
                bot_idx = np.argpartition(contrib, k-1)[:k]
                # map to full indices
                if lag >= 0:
                    i_candidates = top_idx
                    j_candidates = bot_idx + lag
                else:
                    lagpos = -lag
                    i_candidates = top_idx + lagpos
                    j_candidates = bot_idx
                i = rng.choice(i_candidates)
                j = rng.choice(j_candidates)
                new_h = h.copy()
                # swap values
                new_h[i], new_h[j] = new_h[j], new_h[i]
            else:
                # random swap
                i, j = rng.choice(n, size=2, replace=False)
                new_h = h.copy()
                new_h[i], new_h[j] = new_h[j], new_h[i]
        else:
            # random epsilon transfer
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = len(h0)
    CURRENT_TARGET = n/2.0
    CURRENT_BETA = beta
    bounds = [(0.0,1.0)]*n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method="SLSQP", bounds=bounds, constraints=cons, options={"maxiter": max_iter, "ftol":1e-12, "disp": False})
    # Clip and enforce sum
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # Distribute diff
        # similar to enforce_sum but inline to avoid extra import.
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12: break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

def enforce_sum(h, target, rng, max_iter=5000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0.0:

        h[i] += add

        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)

def compute_c5(h):

    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    """Bias the transfer toward indices that dominate the worst lag."""
    n = h.size
    if lag >= n:
        return perturb_h(h, rng, scale)
    valid = np.arange(n - lag)
    contrib = h[valid] * (1.0 - h[valid + lag])
    if contrib.size == 0:
        return perturb_h(h, rng, scale)
    k = max(1, int(len(valid) * top_frac))
    top_idx = valid[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = valid[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx) if rng.random() < 0.7 else rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def swap_move(h, rng):
    """Swap two entries (preserves sum)."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize_one_beta(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


# ---------- Simulated annealing ----------
def simulated_annealing(h_start, rng, max_iters, time_limit):
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.02, 1e-5
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        r = rng.random()
        lag_abs = abs(np.argmax(cur_corr) - (n - 1))
        if r < 0.5:
            new_h = targeted_perturb(h, rng, lag_abs, top_frac=0.2, scale=0.5)
        elif r < 0.8:
            new_h = swap_move(h, rng)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c


# ---------- Deterministic local improvement ----------
def local_improve(h, rng, max_steps=500):
    cur_c = compute_c5(h)
    n = h.size
    for _ in range(max_steps):
        corr = np.correlate(h, 1 - h, mode="full")
        lag_abs = abs(np.argmax(corr) - (n - 1))
        new_h = targeted_perturb(h, rng, lag_abs, top_frac=0.2, scale=0.3)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            h, cur_c = new_h, new_c
        else:
            new_h = perturb_h(h, rng, scale=0.1)
            new_c = compute_c5(new_h)
            if new_c < cur_c:
                h, cur_c = new_h, new_c
    return h, cur_c


# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    total_budget = budget_s

    # ---- initial vector ----
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ---- time allocation ----
    main_budget = total_budget * 0.70
    upsample_budget = total_budget * 0.25
    final_budget = total_budget - main_budget - upsample_budget
    main_end = t0 + main_budget
    upsample_end = main_end + upsample_budget

    beta_list = [5, 20, 80, 200, 500, 1000, 2000, 5000, 10000]

    # ---- main stochastic search (n = current) ----
    while time.time() < main_end:
        # random restart
        if rng.random() < 0.7:
            start_h = perturb_h(best_h, rng, scale=0.5)
        else:
            start_h = random_feasible_vector(n, rng)

        # simulated annealing
        sa_h, sa_c = simulated_annealing(start_h, rng, max_iters=12000, time_limit=main_end)
        if sa_c < best_c:
            best_h, best_c = sa_h, sa_c

        # deterministic tweak
        li_h, li_c = local_improve(best_h, rng, max_steps=500)
        if li_c < best_c:
            best_h, best_c = li_h, li_c

        # SLSQP ladder
        for beta in beta_list:
            if time.time() >= main_end:
                break
            ref_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=500)
            ref_c = compute_c5(ref_h)
            if ref_c < best_c:
                best_h, best_c = ref_h, ref_c

    # ---- up‑sample phase (if time permits) ----
    if time.time() < upsample_end:
        target_n = min(720, n * 2)
        if target_n != n:
            old_x = (np.arange(n) + 0.5) * (2.0 / n)
            new_x = (np.arange(target_n) + 0.5) * (2.0 / target_n)
            up_h = np.interp(new_x, old_x, best_h)
            up_h = enforce_sum(up_h, target_n / 2.0, rng)

            # SA on finer grid
            up_h, up_c = simulated_annealing(up_h, rng, max_iters=12000, time_limit=upsample_end)

            # deterministic tweak
            up_h, up_c = local_improve(up_h, rng, max_steps=800)

            # SLSQP ladder on finer grid
            for beta in beta_list:
                if time.time() >= upsample_end:
                    break
                up_h = slsqp_optimize_one_beta(up_h, beta, rng, max_iter=500)
                up_c = compute_c5(up_h)
                if up_c < best_c:
                    best_h, best_c, n = up_h.copy(), up_c, target_n

    # ---- final refinement ----
    final_deadline = t0 + total_budget
    step = 0.2
    cur_h, cur_c = best_h.copy(), best_c
    for _ in range(5000):
        if time.time() > final_deadline:
            break
        new_h = perturb_h(cur_h, rng, scale=step)
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        step *= 0.9995

    # final high‑beta SLSQP (if time)
    for beta in (20000, 50000):
        if time.time() > final_deadline:
            break
        best_h = slsqp_optimize_one_beta(best_h, beta, rng, max_iter=500)
        best_c = compute_c5(best_h)

    # ---- clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
