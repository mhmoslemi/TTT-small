def fix_worst_shift(h, rng):
    n = h.size
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr) - (n - 1)
    k = idx
    # Determine overlapping region indices where i and i+k in bounds.
    if k >= 0:
        i_range = np.arange(0, n - k)
        # For each i, h[i]=1 and h[i+k]=0 is bad
        bad = i_range[(h[i_range] == 1) & (h[i_range + k] == 0)]
        # Good positions where h[i]=0 and h[i+k]=1
        good = i_range[(h[i_range] == 0) & (h[i_range + k] == 1)]
    else:
        k_abs = -k
        i_range = np.arange(k_abs, n)
        # For negative shift, overlap region indices i where i+k >=0
        # That's same as h[i]=1, h[i+k]=0 => i_range, with i+k >=0
        bad = i_range[(h[i_range] == 1) & (h[i_range + k] == 0)]
        good = i_range[(h[i_range] == 0) & (h[i_range + k] == 1)]
    if len(bad) > 0 and len(good) > 0:
        i = rng.choice(bad)
        j = rng.choice(good)
        # Swap
        h[i] = 0
        h[j] = 1
        return True
    return False
```

def random_swap_indices(h, rng):
    n = h.shape[0]
    while True:
        i = rng.integers(n)
        if h[i] == 1:
            break
    while True:
        j = rng.integers(n)
        if h[j] == 0:
            break
    return i, j

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

import time
import numpy as np
from scipy.optimize import minimize

# ---------- core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for a vector h (any dtype, values in [0,1])."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def random_swap_indices(h: np.ndarray, rng: np.random.Generator) -> tuple:
    """Return indices i (h[i]==1) and j (h[j]==0) for a binary vector."""
    n = h.shape[0]
    i = rng.integers(n)
    while h[i] == 0:
        i = rng.integers(n)
    j = rng.integers(n)
    while h[j] == 1:
        j = rng.integers(n)
    return i, j


def fix_worst_shift(h: np.ndarray, rng: np.random.Generator) -> bool:
    """Try to swap a bad 1 with a good 0 for the currently worst shift.
    Returns True if a swap was made."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)          # shift index (may be negative)
    k = idx
    if k >= 0:
        i_range = np.arange(0, n - k)
    else:
        i_range = np.arange(-k, n)
    # bad: ones that see a zero after shift
    bad = i_range[(h[i_range] == 1) & (h[i_range + k] == 0)]
    # good: zeros that would see a one after shift
    good = i_range[(h[i_range] == 0) & (h[i_range + k] == 1)]
    if bad.size and good.size:
        i = rng.choice(bad)
        j = rng.choice(good)
        h[i] = 0
        h[j] = 1
        return True
    return False


def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> None:
    """Exact correction to enforce Σh = n/2 for binary or fractional vectors."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    # simple iterative redistribution; works for binary and fractional alike
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()


# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max over shifts via log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # (m + log Σ exp(bc - m)) / β   = smooth max
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 600) -> np.ndarray:
    """SLSQP minimisation of the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum repair
    adjust_sum_exact(x, rng)
    return x


# ---------- projected gradient refinement ----------
def gradient_refine(h: np.ndarray, rng: np.random.Generator,
                   step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Projected gradient descent against the worst shift."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = idx
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - abs(k)
            if k > 0:
                g[:upper] += 1.0 - h[k:]      # + (1 - h[i+k])
                g[k:] -= h[:upper]            # - h[i]
            else:
                kk = -k
                g[kk:] += 1.0 - h[:upper]     # + (1 - h[i-k])
                g[:upper] -= h[kk:]           # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # simple redistribution to fix the sum
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h


# ---------- binary simulated annealing ----------
def binary_sa_optimize(start_h: np.ndarray, rng: np.random.Generator,
                       deadline: float) -> tuple:
    """SA on binary vectors (swap 1↔0) respecting the exact half‑sum constraint."""
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = cur_h.size
    T0 = max(cur_c * 0.5, 1e-4)
    it = 0
    while time.time() < deadline:
        i, j = random_swap_indices(cur_h, rng)
        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_c = compute_c5(cur_h)
        T = T0 * (0.9999 ** it) + 1e-12
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
            # occasional deterministic fix for the worst shift
            if rng.random() < 0.07:
                if fix_worst_shift(cur_h, rng):
                    cur_c = compute_c5(cur_h)
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
        it += 1
    return best_h, best_c


# ---------- candidate generation ----------
def generate_candidates(n: int, rng: np.random.Generator) -> list:
    """Construct a diverse set of binary starting vectors of length n."""
    cand = []

    # Sturmian‑type sequences with several irrationals
    alphas = [
        (np.sqrt(5) - 1) / 2,    # 1/phi ≈ 0.618
        np.sqrt(2) - 1,          # ≈0.414
        np.sqrt(3) - 1,          # ≈0.732
        np.pi % 1,               # ≈0.14159
        np.e % 1,                # ≈0.71828
    ]
    for alpha in alphas:
        f = (np.arange(n) * alpha) % 1.0
        h = (f < 0.5).astype(np.int8)
        target = n // 2
        cur = int(h.sum())
        if cur != target:
            diff = target - cur
            dist = np.abs(f - 0.5)
            order = np.argsort(dist)
            if diff > 0:
                zeros = order[h[order] == 0][:diff]
                h[zeros] = 1
            else:
                ones = order[h[order] == 1][: -diff]
                h[ones] = 0
        cand.append(h)

    # a few pure random binary vectors
    for _ in range(3):
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, n // 2, replace=False)] = 1
        cand.append(h)

    # parent if available
    if "initial_h_values" in globals():
        parent = np.array(initial_h_values, dtype=np.int8)
        if parent.size != n:
            # interpolate to required length
            xs = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(xt, xs, parent)
            idx = np.argpartition(interp, -n // 2)[-n // 2 :]
            h = np.zeros(n, dtype=np.int8)
            h[idx] = 1
            cand.append(h)
        else:
            if np.all(np.logical_or(parent == 0, parent == 1)):
                cand.append(parent.copy())
            else:
                idx = np.argpartition(parent, -n // 2)[-n // 2 :]
                h = np.zeros(n, dtype=np.int8)
                h[idx] = 1
                cand.append(h)
    return cand

def run(seed: int = 42, budget_s: int = 1100, **kwargs) -> tuple:
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.94  # safety margin

    base_n = 360                     # primary resolution
    candidates = generate_candidates(base_n, rng)

    # ---- pick the best among simple candidates ----
    best_h = None
    best_c = np.inf
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if best_h is None:
        # fallback deterministic pattern
        best_h = np.zeros(base_n, dtype=np.int8)
        best_h[: base_n // 2] = 1
        best_c = compute_c5(best_h)

    # ---- binary simulated annealing on the chosen pattern ----
    if time.time() < deadline:
        best_h, best_c = binary_sa_optimize(best_h, rng, deadline)

    # ---- continuous gradient / SLSQP refinements (fractional) ----
    if time.time() < deadline:
        h_float = best_h.astype(float)
        # a modest gradient sweep before entering SLSQP
        h_float = gradient_refine(h_float, rng, step=0.02, max_iter=12)
        cur_c = compute_c5(h_float)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_float.copy()
        # soft‑max SLSQP cascade
        for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000, 20000):
            if time.time() >= deadline:
                break
            max_iter = 800 if beta >= 2000 else 500
            h_float = slsqp_optimize(h_float, beta, rng, max_iter=max_iter)

            # brief gradient polishing after each β

            if cur_c < best_c:
                best_c = cur_c
                best_h = h_float.copy()

    # ---- up‑sample to double resolution if time permits ----
    if time.time() < deadline and best_h.size * 2 <= 720:
        # binarise current best (if fractional) before repetition
        if not np.all(np.logical_or(best_h == 0, best_h == 1)):
            # keep the top half as ones

            idx = np.argpartition(flat, -best_h.size // 2)[-best_h.size // 2 :]

        h_up = np.repeat(best_h, 2)

        # binary SA on the up‑sampled vector

            if c_up < best_c:
                best_c = c_up
                best_h = h_up.copy()

        # optional fractional polishing on the up‑sampled set
        if time.time() < deadline:
            h_float = best_h.astype(float)
            h_float = gradient_refine(h_float, rng, step=0.02, max_iter=12)
            cur_c = compute_c5(h_float)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_float.copy()
            for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000):
                if time.time() >= deadline:
                    break
                h_float = slsqp_optimize(h_float, beta, rng, max_iter=600)
                cur_c = compute_c5(h_float)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h_float.copy()
                h_float = gradient_refine(h_float, rng, step=0.02, max_iter=8)

    adjust_sum_exact(best_h, rng)

    return best_h.tolist(), float(final_c), int(best_h.size)
