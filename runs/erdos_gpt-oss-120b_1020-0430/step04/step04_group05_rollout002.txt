def slsqp_optimize_one_beta(h0, beta, rng, max_iter=200):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': lambda x: np.sum(x) - n/2.0}
    # define objective and gradient
    def obj(x):
        # compute corr
        corr = np.correlate(x, 1-x, mode='full')
        # softmax with beta
        # vectorized: use log-sum-exp
        maxv = np.max(beta * corr)
        # Actually we can compute f = (maxv + np.log(np.sum(np.exp(beta*corr - maxv)))) / beta
        f = (maxv + np.log(np.sum(np.exp(beta*corr - maxv)))) / beta
        return f
    def jac(x):
        corr = np.correlate(x, 1-x, mode='full')
        # weights p_k = exp(beta*corr_k) / sum exp
        maxv = np.max(beta * corr)
        exp_terms = np.exp(beta*corr - maxv)
        Z = np.sum(exp_terms)
        p = exp_terms / Z
        # gradient: sum_k p_k * d corr_k/dx
        n = x.size
        grad = np.zeros(n)
        # loop over lag indices
        # use vectorized loops
        # We'll compute by vectorized slicing for each L.
        # Efficiency: perform for L>=0 and L<0 using broadcasting
        # This can be done using cumulative but loops fine.
        for idx_l, w in enumerate(p):
            L = idx_l - (n - 1)
            if w == 0:
                continue
            if L >= 0:
                # add term (1 - x[i+L]) for i up to n-1-L
                grad[:n-L] += w * (1 - x[L:])
                # subtract term x[i-L] for i from L to n-1
                grad[L:] -= w * x[:n-L]
            else:
                K = -L
                grad[K:] += w * (1 - x[:n-K])
                grad[:n-K] -= w * x[K:]
        return grad
    res = minimize(obj, h0, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x,0,1)
    # adjust sum if needed
    target = n/2.0
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple adjustment: distribute diff evenly, then clip
        x += diff / n
        x = np.clip(x,0,1)
        # adjust again if needed using loop
        diff = target - x.sum()
        for _ in range(10):
            if abs(diff) < 1e-12:
                break
            # add diff to a random index that can accommodate
            i = rng.integers(0,n)
            add = min(diff, 1 - x[i]) if diff >0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

import time

from scipy.optimize import minimize

def enforce_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to meet the exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    if abs(diff) > 1e-12:
        for _ in range(max_iter):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            if add == 0.0:
                continue
            h[i] += add
            diff = target - h.sum()
    # tiny residual correction
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Start from 0.5 everywhere and shuffle mass randomly."""
    h = np.full(n, 0.5, dtype=float)
    for _ in range(n * 3):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = enforce_sum(h, n / 2.0, rng)
    return h

def compute_c5(h):
    """Return C5 for the step vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def perturb_h(h, rng, scale=0.2):
    """Move a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    """Bias transfers toward indices that dominate the worst lag."""
    n = h.size
    L = abs(lag)
    if L >= n:
        return perturb_h(h, rng, scale)
    valid_len = n - L
    if valid_len <= 0:
        return perturb_h(h, rng, scale)
    idx = np.arange(valid_len)
    contrib = h[idx] * (1.0 - h[idx + L])
    k = max(1, int(valid_len * top_frac))
    top_idx = idx[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = idx[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    j = rng.choice(bot_idx)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    """SLSQP smoothing with analytical gradient for a given beta."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_cons = {
        "type": "eq",
        "fun": lambda x, t=target: np.sum(x) - t,
        "jac": lambda x, t=target: np.ones_like(x),
    }

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        maxv = beta * np.max(corr)
        # log‑sum‑exp
        f = (maxv + np.log(np.sum(np.exp(beta * corr - maxv)))) / beta
        return f

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        maxv = beta * np.max(corr)
        exp_terms = np.exp(beta * corr - maxv)
        Z = np.sum(exp_terms)
        p = exp_terms / Z  # weights summing to 1
        grad_vec = np.zeros_like(x)
        # contributions for each lag
        for idx, w in enumerate(p):
            if w == 0.0:
                continue
            L = idx - (n - 1)  # lag value
            if L >= 0:
                grad_vec[: n - L] += w * (1.0 - x[L:])
                grad_vec[L:] -= w * x[: n - L]
            else:
                K = -L
                grad_vec[K:] += w * (1.0 - x[: n - K])
                grad_vec[: n - K] -= w * x[K:]
        return grad_vec

    res = minimize(
        obj,
        h0,
        method="SLSQP",
        jac=grad,
        bounds=bounds,
        constraints=eq_cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # simple redistribution
        for _ in range(10):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

    h = h_start.copy()

            if new_c < best_c:

    return best_h, best_c

        step *= 0.9995

    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_h.size)
