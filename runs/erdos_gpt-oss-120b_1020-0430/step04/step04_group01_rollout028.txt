def greedy_reduce(h, rng, deadline, max_iters=20000):
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    while time.time() < deadline and max_iters>0:
        conv = correlate_full(best_h)
        max_val = conv.max() * (2.0 / n)
        if max_val <= best_c - 1e-8:  # maybe improved already
            best_c = max_val
        # find indices of top shifts:
        # we need index of max conv array
        max_idx = int(np.argmax(conv))
        k = max_idx - (n - 1)
        # compute delta_i = h_{i+k} - h_{i-k}
        # compute shifted arrays
        if k >= 0:
            h_plus = np.concatenate([h[k:], np.zeros(k)])
            h_minus = np.concatenate([np.zeros(k), h[:n-k]])
        else:
            kp = -k
            h_plus = np.concatenate([np.zeros(kp), h[:n-kp]])
            h_minus = np.concatenate([h[kp:], np.zeros(kp)])
        delta = h_plus - h_minus
        # find source i (max delta) with h>0, dest j (min delta) with h<1
        source_candidates = np.where(best_h > 1e-12)[0]
        dest_candidates = np.where(best_h < 1 - 1e-12)[0]
        if source_candidates.size == 0 or dest_candidates.size == 0:
            break
        i = source_candidates[np.argmax(delta[source_candidates])]
        j = dest_candidates[np.argmin(delta[dest_candidates])]
        # compute epsilon
        eps = min(best_h[i], 1 - best_h[j], 0.02)  # step size
        if eps <= 0:
            break
        # apply
        best_h[i] -= eps
        best_h[j] += eps
        # recompute max after update to decide acceptance
        new_c = compute_c5(best_h)
        if new_c < best_c - 1e-9:
            best_c = new_c
            # continue
        else:
            # revert; maybe try random small pert
            best_h[i] += eps
            best_h[j] -= eps
            # random perturb fallback
            best_h = perturb_h(best_h, rng, scale=0.1)
            best_c = compute_c5(best_h)
        max_iters -= 1
    return best_h, best_c

def pgd_optimize(h0, betas, max_iters_per_beta, step0, rng, deadline):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in betas:
        step = step0
        for it in range(max_iters_per_beta):
            if time.time() >= deadline:
                return best_h, best_c
            # compute correlation conv
            conv = correlate_full(h)
            # compute weights
            bc = beta * conv
            max_bc = bc.max()
            exp_bc = np.exp(bc - max_bc)
            w = exp_bc / exp_bc.sum()
            # compute convolutions
            # length for FFT
            L = 1 << ((3*n - 2).bit_length())
            fh = np.fft.rfft(h, L)
            fw = np.fft.rfft(w, L)
            fw_rev = np.fft.rfft(w[::-1], L)
            conv2 = np.fft.irfft(fh * fw, L)[:3*n-2]   # w * h
            conv1 = np.fft.irfft(fh * fw_rev, L)[:3*n-2]  # w_rev * h
            term2 = conv2[:n]
            term1 = conv1[:n]
            grad = 1.0 - (term1 + term2)
            # gradient descent step
            h_new = h - step * grad
            # clip
            h_new = np.clip(h_new, 0.0, 1.0)
            # enforce sum
            h_new = enforce_sum(h_new, target, rng)
            # evaluate
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
                h = h_new
                # possibly increase step?
                step = min(step * 1.05, 0.5)
            else:
                # reduce step
                step = step * 0.7
                # optionally keep h unchanged
        # after each beta, maybe reset step
        step = step0
    return best_h, best_c

def greedy_reduce(h_start, rng, deadline, max_iters=10000):
    h = h_start.copy()
    n = h.size
    target = n/2.0
    best_c = compute_c5(h)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = correlate_full(h)
        idx_max = int(np.argmax(conv))
        k = idx_max - (n - 1)
        # compute delta_i = h_{i+k} - h_{i-k}
        # Get shifted arrays
        if k >= 0:
            h_plus = np.concatenate([h[k:], np.zeros(k)])
            h_minus = np.concatenate([np.zeros(k), h[:n-k]])
        else:
            kp = -k
            h_plus = np.concatenate([np.zeros(kp), h[:n-kp]])
            h_minus = np.concatenate([h[kp:], np.zeros(kp)])
        delta = h_plus - h_minus
        # source: max delta among indices where h > 0
        source_mask = h > 1e-12
        if not np.any(source_mask):
            break
        i = np.argmax(delta * source_mask)  # But need to ensure source_mask; use masked argmax
        # dest: min delta among indices where h < 1
        dest_mask = h < 1 - 1e-12
        if not np.any(dest_mask):
            break
        j = np.argmin(delta + 1e6 * (~dest_mask))  # or masked
        # compute epsilon
        eps = min(h[i], 1 - h[j], 0.02)  # step max
        if eps <= 0:
            # try random fallback
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            eps = min(h[i], 1 - h[j], 0.02)
        # apply
        h[i] -= eps
        h[j] += eps
        # ensure sum via epsilon exactly preserving sum; we didn't modify sum (since we moved mass)
        # but clipping might cause small errors: no need enforce sum.
        # compute new c
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            # continue
        else:
            # revert move
            h[i] += eps
            h[j] -= eps
            # small random perturb
            h = perturb_h(h, rng, scale=0.05)
            # enforce sum after perturb
            h = enforce_sum(h, target, rng)
    return h, best_c

    h = h0.copy()
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in betas:
        step = 0.1
        for it in range(max_iters_per_beta):
            if time.time() >= deadline:
                return best_h, best_c
            # compute conv for weights
            conv = correlate_full(h)  # length 2n-1
            bc = beta * conv
            m = bc.max()
            exp_bc = np.exp(bc - m)
            w = exp_bc / exp_bc.sum()
            # compute term1 and term2 via FFT
            L = 1 << ((3*n - 2).bit_length())
            fh = np.fft.rfft(h, L)
            fw = np.fft.rfft(w, L)
            fw_rev = np.fft.rfft(w[::-1], L)
            conv2_full = np.fft.irfft(fh * fw, L)[:3*n-2]
            conv1_full = np.fft.irfft(fh * fw_rev, L)[:3*n-2]
            offset = n - 1
            term2 = conv2_full[offset:offset + n]
            term1 = conv1_full[offset:offset + n]
            grad = 1.0 - (term1 + term2)

import time
import numpy as np

# --------------------------------------------------------------
# Basic feasibility utilities
# --------------------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0, rng)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

# --------------------------------------------------------------
# Correlation utilities (FFT based)
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1-h) using FFT (zero‑padding)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv

def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)

# --------------------------------------------------------------
# Projected Gradient Descent on smooth surrogate
# --------------------------------------------------------------
def pgd_optimize(h0, betas, max_iters_per_beta, rng, deadline):
    """Gradient descent on log‑sum‑exp surrogate, with projection."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)

    for beta in betas:
        step = 0.1
        for _ in range(max_iters_per_beta):
            if time.time() >= deadline:
                return best_h, best_c
            # weights from smoothed max
            conv = correlate_full(h)
            bc = beta * conv
            m = bc.max()
            exp_bc = np.exp(bc - m)
            w = exp_bc / exp_bc.sum()
            # compute term1 = Σ_k w_{-k} h_{i-k}, term2 = Σ_k w_k h_{i-k}
            L = 1 << ((3 * n - 2).bit_length())
            fh = np.fft.rfft(h, L)
            fw = np.fft.rfft(w, L)
            fw_rev = np.fft.rfft(w[::-1], L)
            conv2_full = np.fft.irfft(fh * fw, L)[: 3 * n - 2]
            conv1_full = np.fft.irfft(fh * fw_rev, L)[: 3 * n - 2]
            off = n - 1
            term2 = conv2_full[off:off + n]
            term1 = conv1_full[off:off + n]
            grad = 1.0 - (term1 + term2)          # Σ w = 1 guarantees zero‑sum gradient
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = enforce_sum(h_new, target, rng)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
                h = h_new
                step = min(step * 1.2, 0.5)
            else:
                step = step * 0.7
                if rng.random() < 0.01:          # occasional escape
                    h = perturb_h(h, rng, scale=0.05)
    return best_h, best_c

# --------------------------------------------------------------
# Greedy pairwise mass transfer targeting worst shift
# --------------------------------------------------------------
def greedy_reduce(h_start, rng, deadline, max_iters=8000):
    h = h_start.copy()
    n = h.size
    target = n / 2.0
    best_c = compute_c5(h)
    eps_cap = 0.04
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        conv = correlate_full(h)
        idx = int(np.argmax(conv))
        k = idx - (n - 1)                     # shift giving the maximum overlap
        # compute delta_i = h_{i+k} - h_{i-k}
        if k >= 0:
            h_plus = np.concatenate([h[k:], np.zeros(k)])
            h_minus = np.concatenate([np.zeros(k), h[:n - k]])
        else:
            kp = -k
            h_plus = np.concatenate([np.zeros(kp), h[:n - kp]])
            h_minus = np.concatenate([h[kp:], np.zeros(kp)])
        delta = h_plus - h_minus
        src_candidates = np.where(h > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(delta[src_candidates])]
        dst_candidates = np.where(h < 1 - 1e-12)[0]
        if dst_candidates.size == 0:
            break
        j = dst_candidates[np.argmin(delta[dst_candidates])]
        eps = min(h[i], 1 - h[j], eps_cap)
        if eps <= 0:
            # fallback random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            eps = min(h[i], 1 - h[j], eps_cap)
            if eps <= 0:
                break
        h[i] -= eps
        h[j] += eps
        # sum unchanged; potential tiny drift from floating‑point
        c = compute_c5(h)
        if c < best_c:
            best_c = c
        else:
            # revert and try a small random perturb
            h[i] += eps
            h[j] -= eps
            h = perturb_h(h, rng, scale=0.03)
            h = enforce_sum(h, target, rng)
    return h, best_c
