def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay = 0.95
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
            # optionally increase step slightly
            step = min(step * 1.05, start_step)
        else:
            # maybe accept occasional non-improving to escape
            if rng.random() < 0.001:
                h = new_h
        # decay step every some iterations
        if (it+1) % (max_iters // 10 + 1) == 0:
            step = max(step * decay, 1e-5)
    return best_h, best_c

import time
import numpy as np

# ---------- Utility functions ----------
def enforce_sum(h, target, rng):
    """Adjust h so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    # simple iterative correction
    for _ in range(10000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # fallback if still off
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            scale = target / h.sum()
            h = np.clip(h * scale, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    return enforce_sum(h, target, rng)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from i to j (preserves the sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1.0 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv.max() * (2.0 / n)

# ---------- Simulated annealing (continuous) ----------
def continuous_sa(h0, rng, deadline):
    """SA that moves a small ε between two coordinates, preserving sum."""
    h = h0.copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    T0 = cur_c * 0.05 + 1e-8
    T_end = 1e-9
    init_scale = 0.3
    while time.time() < deadline:
        # progress fraction
        t = (time.time() - start) / max(deadline - start, 1e-12)
        T = T0 * ((T_end / T0) ** t)
        scale = init_scale * (1 - t) + 1e-6
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if new_c < best_c:
            best_c, best_h = new_c, h.copy()
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
        else:
            # revert
            h[i] += eps
            h[j] -= eps
    return best_h, best_c

# ---------- Simulated annealing (binary) ----------
def binary_sa(start_bin, rng, deadline):
    """Swap‑based SA on a binary vector (sum = n/2)."""
    h = start_bin.astype(float).copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h, best_c = h.copy(), cur_c
    start = time.time()
    T0 = cur_c * 0.02 + 1e-8
    T_end = 1e-9
    while time.time() < deadline:
        t = (time.time() - start) / max(deadline - start, 1e-12)
        T = T0 * ((T_end / T0) ** t)
        # pick a random 1 and a random 0
        i = rng.integers(0, n)
        if h[i] < 0.5:
            continue
        j = rng.integers(0, n)
        if h[j] > 0.5:
            continue
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if new_c < best_c:
            best_c, best_h = new_c, h.copy()
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

# ---------- Greedy hill climbing ----------
def hill_climb(h_start, rng, max_iters=3000, start_step=0.3):
    """Greedy hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 15)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-6)
    return best_h, best_c

# ---------- Main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ---- Initial vector ----
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = h0.copy(), compute_c5(h0)

    # ---- Phase 1: Continuous SA ----
    phase1_end = start_time + 0.55 * budget_s
    while time.time() < phase1_end:
        # start from best with a small random perturbation
        h_start = perturb_h(best_h, rng, scale=0.3)
        if rng.random() < 0.1:  # occasional full random restart
            h_start = random_feasible_vector(n, rng)
        cand_h, cand_c = continuous_sa(h_start, rng, phase1_end)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

    # ---- Phase 2: Binary SA ----
    phase2_end = start_time + 0.80 * budget_s
    # binary start: top n/2 entries of current best
    thresh = np.partition(best_h, -n // 2)[-n // 2]
    bin_start = (best_h >= thresh).astype(float)
    bin_start = enforce_sum(bin_start, n / 2.0, rng)

    while time.time() < phase2_end:
        cand_h, cand_c = binary_sa(bin_start, rng, phase2_end)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c
        # update binary start from new best
        thresh = np.partition(best_h, -n // 2)[-n // 2]
        bin_start = (best_h >= thresh).astype(float)
        bin_start = enforce_sum(bin_start, n / 2.0, rng)

    # ---- Phase 3: Hill‑climb refinement ----
    while time.time() < deadline:
        cand_h, cand_c = hill_climb(best_h, rng, max_iters=5000, start_step=0.2)
        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c
        # small random perturb before next climb if time permits
        if time.time() + 0.1 >= deadline:
            break
        best_h = perturb_h(best_h, rng, scale=0.05)

    # ---- Optional up‑sampling to 1440 points ----
    if time.time() + 8 < deadline and best_h.size * 2 <= 1440:
        factor = 2
        h_fine = np.repeat(best_h, factor)
        n_fine = h_fine.size
        h_fine = enforce_sum(h_fine, n_fine / 2.0, rng)
        # polish fine grid
        h_fine, c_fine = hill_climb(h_fine, rng, max_iters=8000, start_step=0.15)
        if c_fine < best_c:
            best_h, best_c = h_fine, c_fine

    # ---- Final cleanup ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
