def project_sum_to_target(x, target, rng):
    # Clip into [0,1], then adjust sum to target using water-filling algorithm.
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Determine free variables not at bounds.
    free = (x > 1e-12) & (x < 1.0 - 1e-12)
    # If free exist, distribute diff evenly.
    # But diff may exceed capacity of free variables; in that case, fallback.
    # Use simple iterative approach:
    for _ in range(10):
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        per = diff / free.sum()
        # Clip per addition to avoid hitting bounds
        add = np.where(free, per, 0.0)
        # Apply but cap each variable within [0,1]
        x_new = x + add
        # Clip 
        x_new = np.clip(x_new, 0.0, 1.0)
        diff = target - x_new.sum()
        x = x_new
        if abs(diff) < 1e-12:
            break
    # If still diff, adjust randomly.
    if abs(diff) > 1e-10:
        # Randomly add/subtract small eps to any variable not at bound.
        for _ in range(1000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, len(x))
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x
```

def correlation_full(h):
    n = len(h)
    rev = (1 - h)[::-1]
    L = 2*n - 1
    m = 1 << ((L-1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())  # next power of two >= L
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return np.max(conv) * (2.0 / n)


def correlation_full(h):
    """Full correlation of h with (1-h). Returns array of length 2n-1."""
    n = h.shape[0]
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv


def subgradient(h):
    """Subgradient of the worst‑case correlation term."""
    n = h.shape[0]
    corr = correlation_full(h)
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)               # shift giving the maximal overlap loss
    grad = np.ones(n)
    if k >= 0:
        grad[:n - k] -= h[k:]          # term h[i+k]
        grad[k:] -= h[:n - k]          # term h[i-k]
    else:
        kk = -k
        grad[kk:] -= h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad, corr[max_idx]


def enforce_sum(x, target, rng, max_iter=5000):
    """Adjust x so that Σx = target while keeping 0≤x≤1."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    n = x.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - x[i])
        else:
            add = max(diff, -x[i])
        x[i] += add
        diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            x[idx] += per
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            for i in range(n):
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - x[i])
                else:
                    add = max(diff, -x[i])
                x[i] += add
                diff = target - x.sum()
    return x


def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with Σ = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.5
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def generate_binary_vector(n, rng):
    """Binary vector with exactly n/2 ones (n must be even)."""
    half = n // 2
    idx = rng.permutation(n)[:half]
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    return h


def projected_gradient_descent(h0, rng, iters=300, step0=0.5):
    """Simple projected gradient descent using a subgradient of max‑correlation."""
    n = h0.size
    target = n / 2.0
    h = enforce_sum(h0.copy(), target, rng)
    best_h = h.copy()
    best_c = compute_c5(h)
    v = np.zeros_like(h)  # momentum
    for t in range(iters):
        grad, _ = subgradient(h)
        step = step0 * (0.95 ** t)
        v = 0.9 * v - step * grad          # momentum update
        h_new = h + v
        h_new = enforce_sum(h_new, target, rng)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        h = h_new
    return best_h, best_c


def sum_eq_constraint(x):
    """Equality constraint needed for SLSQP."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth surrogate for max_k correlation using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine solution using SLSQP on the smooth surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
