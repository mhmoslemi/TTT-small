    total = np.sum(h)
    corr = np.correlate(h, h, mode='full')
    g = total - corr  # length 2n-1

    obj = (max_g + np.log(np.sum(exp_beta))) / beta

def project_onto_simplex_cap(v, target, cap=1.0):
    # clip to [0, cap]
    v = np.clip(v, 0, cap)
    # if sum(v) == target, done
    s = v.sum()
    if s == target:
        return v
    # need to adjust downward if s > target else upward if s < target
    # Use algorithm to shrink or expand.
    # We'll use iterative approach:
    for _ in range(10):
        if abs(s - target) < 1e-12:
            break
        if s > target:
            # reduce proportionally on entries that are > 0
            # compute reduction factor
            # allocate reduction only to entries > 0
            # we can compute delta = (s - target) / np.count_nonzero(v)
            # but must not go below zero
            # We'll compute a lambda such that sum max(v - lam, 0) = target
            # This is standard projection onto simplex without upper bound.
            # It can be solved by sorting.
            # For simplicity we use sorting solution:
            # compute u = sorted(v, reverse=True)
            # find rho s.t. u_i - (sum_{j=1}^i u_j - target) / i > 0.
            # Not need upper cap after first clipping.
            # Let's implement standard simplex projection:
            u = np.sort(v)[::-1]
            cssv = np.cumsum(u) - target
            rho = np.where(u - cssv / np.arange(1, len(u)+1) > 0)[0]
            if len(rho)==0:
                theta = 0
            else:
                rho = rho[-1]
                theta = cssv[rho] / (rho+1)
            v = np.maximum(v - theta, 0)
        else:
            # s < target, need to push up, can only push up to cap.
            # Consider flipped: minimize distance to v such that sum = target and v_i <= cap.
            # Equivalent to project onto simplex with upper bound using symmetry: 
            # define w = cap - v, target' = n*cap - target. Project w onto simplex without cap (lower bound 0), then set v = cap - w.
            w = cap - v
            target_w = n*cap - target
            # project w onto simplex (lower bound 0)
            u = np.sort(w)[::-1]
            cssv = np.cumsum(u) - target_w
            rho = np.where(u - cssv / np.arange(1, len(u)+1) > 0)[0]
            if len(rho)==0:
                theta = 0
            else:
                rho = rho[-1]
                theta = cssv[rho] / (rho+1)
            w = np.maximum(w - theta, 0)
            v = cap - w
        s = v.sum()
    return v

def adam_optimize(h0, betas, time_limit):
    start = time.time()
    n = h0.size
    target = n/2
    h = project_onto_simplex_cap(h0, target)  # ensure feasible
    best_h = h.copy()
    best_c = compute_c5(h)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    for beta in betas:
        # compute number of steps allowed for this beta given remaining time
        remaining = time_limit - (time.time() - start)
        if remaining <= 0:
            break
        # decide steps count based on maybe 0.5 sec per step? 
        # Let's set step_budget = min(5000, int(remaining * 20))  # 20 steps per second approx
        step_budget = max(1, int(remaining * 15))

        step_budget = min(step_budget, 5000)
        lr = 0.5 / np.sqrt(beta)  # shrink step as beta grows to be stable
        for _ in range(step_budget):
            t += 1
            obj, grad = softmax_obj_and_grad(h, beta)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h - step
            # project onto feasible set
            h_new = project_onto_simplex_cap(h_new, target)
            # re-evaluate exact C5 less frequently? We'll compute C5 each step maybe heavy but still small.
            c_val = compute_c5(h_new)
            if c_val < best_c:
                best_c = c_val
                best_h = h_new.copy()
            h = h_new
            if time.time() - start >= time_limit - 0.5:
                break
        if time.time() - start >= time_limit - 0.5:
            break
    return best_h, best_c

def softmax_obj_and_grad(h, beta):
    total = h.sum()
    corr = np.correlate(h, h, mode='full')
    g = total - corr
    max_g = np.max(g)
    exp_beta = np.exp(beta * (g - max_g))
    w = exp_beta / exp_beta.sum()
    conv1 = np.convolve(w, h, mode='same')
    conv2 = np.convolve(w[::-1], h, mode='same')
    grad = 1.0 - (conv1 + conv2)
    obj = (max_g + np.log(exp_beta.sum())) / beta
    return obj, grad

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_simplex_cap(v, target, cap=1.0):
    """Projection of v onto {0≤x_i≤cap, Σx_i = target}."""
    n = v.shape[0]
    v = np.clip(v, 0.0, cap)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    if s > target:
        # reduce sum: standard simplex projection (lower bound 0)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - target
        rho = np.where(u > cssv / np.arange(1, n + 1))[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)
    else:
        # increase sum: use symmetry with upper bound
        w = cap - v
        target_w = n * cap - target
        u = np.sort(w)[::-1]
        cssv = np.cumsum(u) - target_w
        rho = np.where(u > cssv / np.arange(1, n + 1))[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w_proj = np.maximum(w - theta, 0.0)
        return cap - w_proj


def softmax_obj_and_grad(h, beta):
    """Smooth max objective (log‑sum‑exp) and its gradient."""
    total = h.sum()
    corr = np.correlate(h, h, mode="full")
    g = total - corr                     # length 2n‑1
    max_g = np.max(g)
    exp_beta = np.exp(beta * (g - max_g))
    w = exp_beta / exp_beta.sum()        # probabilities over shifts
    # gradient = 1 - (conv(w,h) + conv(w_rev,h))
    conv1 = np.convolve(w, h, mode="same")
    conv2 = np.convolve(w[::-1], h, mode="same")
    grad = 1.0 - (conv1 + conv2)
    obj = (max_g + np.log(exp_beta.sum())) / beta
    return obj, grad
