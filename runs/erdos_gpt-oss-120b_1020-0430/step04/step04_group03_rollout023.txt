import time, numpy as np

def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_simplex(v, target):
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_softmax(h, beta):
    n = h.size
    corr = np.convolve(h, 1 - h, mode='full')
    max_c = corr.max()
    w = np.exp(beta * (corr - max_c))
    w_sum = w.sum()
    w = w / w_sum
    # compute gradient
    w_rev = w[::-1]
    g1 = np.convolve(w_rev, 1 - h, mode='full')
    g2 = np.convolve(w, h, mode='full')
    start = len(w) - 1
    grad = g1[start:start+n] - g2[start:start+n]
    return grad

def adam_optimize(h0, betas, iters_per_beta, lr, rng):
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for beta in betas:
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        for t in range(1, iters_per_beta+1):
            grad = gradient_softmax(h, beta)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            # random noise occasionally to escape (optional):
            if t % 200 == 0:
                # small perturb
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j:
                    # transfer small mass
                    eps_transfer = 0.01 * rng.random()
                    eps_transfer = min(eps_transfer, h[i], 1 - h[j])
                    h[i] -= eps_transfer
                    h[j] += eps_transfer
            # projection
            h = project_onto_box_simplex(h, target)
    return h

def perturb_small(h, rng, scale=0.05):
    n = h.size
    # pick random pair of indices
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h.copy()
    max_transfer = min(h[i], 1.0 - h[j])
    eps = max_transfer * scale * rng.random()
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

import time, numpy as np

# ----------------------- core utilities -----------------------

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a vector h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_onto_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x : 0 <= x_i <= 1, sum x_i = target}.
    Uses a simple binary search on the Lagrange multiplier.
    """
    lo, hi = -1.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        idx = np.where((x > 0.0) & (x < 1.0))[0]
        if idx.size:
            x[idx[0]] = np.clip(x[idx[0]] + diff, 0.0, 1.0)
        else:  # all at bounds – adjust any entry
            i = np.argmin(x) if diff > 0 else np.argmax(x)
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def gradient_softmax(h: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the softmax–approximated max_{k} ∫h(1‑h(·+k))
    with respect to h.  Uses convolution for O(n log n) cost.
    """
    n = h.size
    corr = np.convolve(h, 1.0 - h, mode="full")
    max_c = corr.max()
    w = np.exp(beta * (corr - max_c))
    w /= w.sum()
    w_rev = w[::-1]
    g1 = np.convolve(w_rev, 1.0 - h, mode="full")
    g2 = np.convolve(w, h, mode="full")
    start = len(w) - 1          # = 2n‑2
    return g1[start:start + n] - g2[start:start + n]


def adam_optimize(h0: np.ndarray,
                  betas,
                  iters_per_beta: int,
                  lr: float,
                  rng: np.random.Generator) -> np.ndarray:
    """
    Projected Adam optimizer that follows a schedule of increasing beta
    (smoothness of the max).  After each step the iterate is projected onto
    the feasible box–simplex.
    """
    h = h0.copy()
    n = h.size
    target = n / 2.0
    for beta in betas:
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        b1, b2 = 0.9, 0.999
        eps = 1e-8
        for t in range(1, iters_per_beta + 1):
            grad = gradient_softmax(h, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * (grad * grad)

            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h -= step

            # occasional tiny random transfer to avoid stagnation
            if t % 200 == 0:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    max_tr = min(h[i], 1.0 - h[j])
                    tr = max_tr * 0.01 * rng.random()
                    h[i] -= tr
                    h[j] += tr

            h = project_onto_box_simplex(h, target)
    return h


def gradient_shift_one(h: np.ndarray, s: int) -> np.ndarray:
    """Gradient of a single correlation term with shift s."""
    n = h.size
    g = np.zeros(n)
    if s >= 0:
        g[: n - s] += (1.0 - h[s:])
        g[s:] -= h[: n - s]
    else:
        s = -s
        g[s:] += (1.0 - h[: n - s])
        g[: n - s] -= h[s:]
    return g


def greedy_optimize_local(h: np.ndarray, max_steps: int = 500) -> np.ndarray:
    """
    Deterministic local improvement by moving mass from a positive‑gradient
    coordinate to a negative‑gradient one for the currently worst shift.
    """
    for _ in range(max_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        s = idx - (h.size - 1)

        grad = gradient_shift_one(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break

        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        eps = min(h[i], 1.0 - h[j])
        if eps < 1e-12:
            break

        # try full transfer
        h2 = h.copy()
        h2[i] -= eps
        h2[j] += eps
        if compute_c5(h2) + 1e-12 < compute_c5(h):
            h = h2
            continue

        # try half transfer
        eps *= 0.5
        h2[i] -= eps
        h2[j] += eps
        if compute_c5(h2) + 1e-12 < compute_c5(h):
            h = h2
            continue
        break
    return h


def small_perturb(h: np.ndarray, rng: np.random.Generator, scale: float = 0.05) -> np.ndarray:
    """Move a random amount of mass from one coordinate to another."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_tr = min(h[i], 1.0 - h[j])
    tr = max_tr * scale * rng.random()
    nh = h.copy()
    nh[i] -= tr
    nh[j] += tr
    return nh

    # ---- initial feasible vector --------------------------------

        # Adam optimisation

            h_opt = adam_optimize(h_start, betas, iters_per_beta, lr, rng)
        except Exception:
            continue

        # final deterministic polishing
        h_opt = greedy_optimize_local(h_opt, max_steps=800)

                if c_cand < best_c:
                    best_h, best_c = cand, c_cand

    # final feasibility enforcement
