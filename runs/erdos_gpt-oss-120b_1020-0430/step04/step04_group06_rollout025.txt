def project_capped_simplex(v, s):
    # v: np.ndarray
    n = v.shape[0]
    # if already within bounds and sum close to s, just return clipped
    # else binary search for λ
    # compute λ bounds:
    # λ_low such that v+λ_low <= 0 for all => λ_low = -max(v)
    # λ_high such that v+λ_high >= 1 for all => λ_high = 1 - min(v)
    lo = -np.max(v) - 1.0  # ensure < needed
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        total = x.sum()
        if total > s:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # Occasionally due to numeric issues diff
    diff = s - x.sum()
    if abs(diff) > 1e-9:
        # adjust few free indices
        free = (x > 1e-8) & (x < 1 - 1e-8)
        cnt = free.sum()
        if cnt > 0:
            # distribute diff equally
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        # If still off, fallback:
        # perform simple correction: adjust first element
        # Not crucial
    return x

def compute_c5(h: np.ndarray) -> float:
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_capped_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {0≤x≤1, sum(x)=s} using binary search on the shift λ."""
    n = v.size
    # bounds for λ
    lo = -np.max(v) - 1.0   # guarantees all v+λ ≤ 0 → sum=0
    hi = 1.0 - np.min(v) + 1.0   # guarantees all v+λ ≥ 1 → sum=n
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        total = x.sum()
        if total > s:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = s - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-8) & (x < 1.0 - 1e-8)
        cnt = free.sum()
        if cnt > 0:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector projected onto the capped simplex."""
    v = rng.random(n)
    return project_capped_simplex(v, n / 2.0)


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_gradient(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of (1/β)·log‑sum‑exp(β·r) w.r.t. h, where w ∝ exp(‑β·p)."""
    n = h.size
    # convolution with w (backward)   Σ_k w_k·h_{i‑k}
    conv_full = np.convolve(w, h, mode="full")
    start = n - 1
    conv_bwd = conv_full[start:start + n]
    # convolution with reversed w (forward) Σ_k w_k·h_{i+k}
    conv_fwd = np.convolve(w[::-1], h, mode="full")[start:start + n]
    return conv_fwd + conv_bwd


def pgd_optimize(
    h0: np.ndarray,
    beta_schedule,
    step_start: float = 0.2,
    step_decay: float = 0.995,
    max_iters_per_beta: int = 150,
    time_limit: float = 120.0,
) -> np.ndarray:
    """Projected gradient descent on the smoothed max‑overlap objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    start = time.time()
    for beta in beta_schedule:
        step = step_start
        for _ in range(max_iters_per_beta):
            if time.time() - start > time_limit:
                return h
            # p = autocorrelation of h
            p = np.correlate(h, h, mode="full")
            # w_k ∝ exp(-β·p_k)
            shift = -beta * np.min(p)          # makes exponent ≤ 0
            w = np.exp(-beta * p - shift)
            w /= w.sum()
            # gradient of the smooth objective
            grad = compute_gradient(h, w)
            # gradient descent step
            h = h - step * grad
            # enforce feasibility
            h = project_capped_simplex(h, target)
            step *= step_decay
    return h


def max_overlap_and_shift(h: np.ndarray):
    """Return (max_corr, shift) where shift = index-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)


def pair_transfer_optimize(
    h0: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 5000,
    time_limit: float = 30.0,
    init_factor: float = 0.5,
    decay: float = 0.999,
):
    """Local search moving mass from i to i+shift to lower the maximum overlap."""
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
        # index ranges affected by this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------
    # 1. Choose number of points and initial vector
    # ----------------------------------------------------------
    n = 720                                   # < 1k, fast enough
    target_sum = n / 2.0

    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        repeats = n // init.size
        remainder = n % init.size
        h0 = np.concatenate([np.tile(init, repeats), init[:remainder]]) if remainder else np.tile(init, repeats)
        h0 = project_capped_simplex(h0, target_sum)
    else:
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----------------------------------------------------------
    # 2. Optimisation schedule
    # ----------------------------------------------------------
    beta_schedule = [5, 20, 80, 200, 500, 1000, 2000, 5000]

    # main loop – keep trying until most of the budget is spent
    while time.time() - start_time < budget_s * 0.80:
        # pick a start point
        if rng.random() < 0.7:
            start_vec = perturb_h(best_h, rng, scale=0.3)
        else:
            start_vec = random_feasible_vector(n, rng)

        # PGD phase
        remaining = budget_s - (time.time() - start_time)
        pgd_time = min(120.0, remaining * 0.5)
        h_opt = pgd_optimize(
            start_vec,
            beta_schedule,
            step_start=0.2,
            step_decay=0.995,
            max_iters_per_beta=150,
            time_limit=pgd_time,
        )
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt

        # short pair‑transfer refinement if time permits
        remaining = budget_s - (time.time() - start_time)
        if remaining > 30.0:
            h_ref, c_ref = pair_transfer_optimize(
                best_h,
                rng,
                max_iters=5000,
                time_limit=min(30.0, remaining * 0.5),
                init_factor=0.5,
                decay=0.999,
            )
            if c_ref < best_c:
                best_h, best_c = h_ref.copy(), c_ref

    # ----------------------------------------------------------
    # 3. Final polishing
    # ----------------------------------------------------------
    best_h = project_capped_simplex(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
