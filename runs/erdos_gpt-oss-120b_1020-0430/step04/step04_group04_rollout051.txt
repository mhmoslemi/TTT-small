def reduce_max_shift(h, rng, iters=200, step=0.05):
    for _ in range(iters):
        conv = compute_full_conv(h)
        s_max = int(np.argmax(conv))
        max_val = conv[s_max]
        # gradient component g_j = 1 - h_{j+s} - h_{j-s}
        N = len(h)
        g = np.ones(N)
        # subtract h[j+s_max] where j+s_max < N
        j_end = N - s_max
        if s_max < N:
            g[:j_end] -= h[s_max:]  # h[j+s_max] for j < N-s_max
        # subtract h[j-s_max] where j-s_max >= 0
        if s_max > 0:
            g[s_max:] -= h[:N - s_max]
        # find indices with positive and negative g
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        max_eps = min(g[i], -g[j], h[i], 1.0 - h[j])
        eps = max_eps * step
        if eps <= 0:
            continue
        h[i] -= eps
        h[j] += eps
        # ensure bounds and sum, but we maintain sum via transfer.
        # Clip just in case epsilon may cause small overshoot
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
    return h
```

def enforce_sum(h, target, rng, max_iter=2000):
    """Clip to [0,1] and adjust to meet exact sum 'target'."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # final tiny correction if needed
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Uniform random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    target = n / 2.0
    cur = h.sum()
    if cur > 0:
        h *= target / cur
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def compute_full_conv(h):
    """Full convolution h * (1-h) reversed, scaled by dx."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv * (2.0 / n)  # include Δx


def compute_c5(h):
    """Exact C5 = max full convolution."""
    return np.max(compute_full_conv(h))


def softmax_surrogate(x, beta):
    """Smooth approximation of max using log‑sum‑exp."""
    n = x.size
    rev = (1.0 - x)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(x, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L] * (2.0 / n)
    maxc = np.max(conv)
    # stable log‑sum‑exp
    bc = beta * (conv - maxc)
    logsum = np.log(np.sum(np.exp(bc))) / beta
    return maxc + logsum


def slsqp_surrogate_opt(h0, beta, rng, max_iter=500):
    """SLSQP optimizer on the smooth surrogate for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = minimize(
        softmax_surrogate,
        h0,
        args=(beta,),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x


def hill_climb(h_start, rng, max_iters=3000, init_step=0.5, n_trials=5):
    """Simple stochastic hill‑climb."""
    h = h_start.copy()
    best_c = compute_c5(h)
    step = init_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        improved = False
        for _ in range(n_trials):
            new_h = perturb_h(h, rng, scale=step)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
                improved = True
                break
        if not improved and rng.random() < 0.02:
            new_h = perturb_h(h, rng, scale=step * 2.0)
            c = compute_c5(new_h)
            if c < best_c:
                best_c, h = c, new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return h, best_c


def simulated_annealing(h0, rng, max_iter=8000,
                        init_temp=1e-2, final_temp=1e-5,
                        init_scale=0.5, final_scale=0.02):
    """Simulated‑annealing refinement."""
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c = best_c
    for i in range(max_iter):
        t = i / max_iter
        temp = init_temp * (final_temp / init_temp) ** t
        scale = init_scale * (final_scale / init_scale) ** t
        new_h = perturb_h(cur_h, rng, scale=scale)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or (temp > 0 and rng.random() < np.exp(-delta / temp)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


def reduce_max_shift(h, rng, iters=200, step=0.5):
    """Targeted reduction of the worst shift using gradient‑like moves."""
    n = h.size
    for _ in range(iters):
        conv = compute_full_conv(h)
        idx = int(np.argmax(conv))
        s_shift = idx - (n - 1)
        s = abs(s_shift)
        if s == 0:
            # gradient becomes 1‑2h, handle similarly
            g = 1.0 - 2.0 * h
        else:
            g = np.ones(n)
            # subtract h[j+s] where valid
            limit = n - s
            g[:limit] -= h[s:]
            # subtract h[j‑s] where valid
            g[s:] -= h[:limit]
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = rng.choice(pos)
        j = rng.choice(neg)
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = max_eps * step * rng.random()
        if eps <= 0:
            continue
        h[i] -= eps
        h[j] += eps
    return h


def upsample_h(old_h, new_n, rng):
    """Linear interpolation to a finer grid and sum correction."""
    old_n = old_h.size
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    new_h = np.interp(new_x, old_x, old_h)
    new_h = enforce_sum(new_h, new_n / 2.0, rng)
    return new_h
