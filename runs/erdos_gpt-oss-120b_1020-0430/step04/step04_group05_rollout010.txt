def enforce_sum(h, target, rng, max_iter=5000):
    # Clip and adjust to exact sum
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # Simple loop distributing diff randomly
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        # compute allowable change
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    # if still diff: small uniform correction
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    # final correction for residual diff
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute remaining diff, ensure not infinite loop
        for i in range(n):
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            if add != 0:
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def targeted_transfer(h, L, rng, scale=0.5):
    n = h.size
    if L >= n:
        # fallback to random transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            return h.copy()
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            return h.copy()
        eps = rng.random() * max_eps * scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        return new_h
    valid = np.arange(n - L)  # indices where i and i+L both valid
    shifted_vals = h[valid + L]
    if L == 0:
        # shift is itself: choose source low h, target high h
        src_idx = valid[np.argmin(shifted_vals)]
        tgt_idx = valid[np.argmax(shifted_vals)]
    else:
        # source where shifted is small, target where shifted is large
        src_idx = valid[np.argmin(shifted_vals)]
        tgt_idx = valid[np.argmax(shifted_vals)]
    max_eps = min(h[src_idx], 1.0 - h[tgt_idx])
    if max_eps <= 0:
        # fallback to random perturb
        return h.copy()
    eps = scale * max_eps * rng.random()  # positive epsilon
    new_h = h.copy()
    new_h[src_idx] -= eps
    new_h[tgt_idx] += eps
    return new_h

def enforce_sum(h, target, rng, max_iter=5000):
    """Clip h to [0,1] and adjust to hit exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-12:
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            if add == 0:
                continue
            h[i] += add
            diff = target - h.sum()
    return h

def random_feasible_vector(n, rng):
    """Create a random vector in [0,1]^n with sum = n/2."""
    h = rng.random(n)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def compute_c5_corr(h):
    """Full correlation array (without dx scaling)."""
    return np.correlate(h, 1 - h, mode="full")

def compute_c5(h):
    """C5 value for vector h."""
    return np.max(compute_c5_corr(h)) * (2.0 / h.size)

def swap_h(h, rng):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def random_transfer(h, rng, scale=0.5):
    """Move a positive ε from i to j (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    if rng.random() < 0.5:
        i, j = j, i
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def targeted_transfer(h, L, rng, scale=0.5):
    """Transfer ε from a source where h[i+L] is small to a target where it is large."""
    n = h.size
    if L >= n:
        return random_transfer(h, rng, scale)
    valid = np.arange(n - L)
    shifted = h[valid + L]
    if L == 0:
        src = valid[np.argmin(shifted)]
        tgt = valid[np.argmax(shifted)]
    else:
        src = valid[np.argmin(shifted)]
        tgt = valid[np.argmax(shifted)]
    max_eps = min(h[src], 1.0 - h[tgt])
    if max_eps <= 0:
        return h.copy()
    eps = scale * max_eps * rng.random()
    h_new = h.copy()
    h_new[src] -= eps
    h_new[tgt] += eps
    return h_new

def gradient_step(h, L, alpha):
    """One projected gradient step for the worst lag L."""
    n = h.size
    g = np.zeros(n)
    if L < n:
        idx = np.arange(n - L)
        g[idx] = 1.0 - h[idx + L]
        g[idx + L] = -h[idx]
    h_new = h - alpha * g
    return np.clip(h_new, 0.0, 1.0)
