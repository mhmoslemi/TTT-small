def softmax_obj(x, beta, target):
    corr = np.correlate(x, 1-x, mode='full')
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def enforce_sum(h, target, rng):
    diff = target - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(n)
        # adjust h[i] by amount limited to [0,1]
        if diff > 0:
            add = min(diff, 1 - h[i])
        else:
            add = max(diff, -h[i])
        if abs(add) < 1e-12:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2] with uniform grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that Σh == target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    attempts = 0
    max_attempts = n * 10
    while abs(diff) > 1e-12 and attempts < max_attempts:
        i = rng.integers(n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if abs(add) > 1e-12:
            h[i] += add
            diff = target - h.sum()
        attempts += 1
    if abs(diff) > 1e-12:  # final safety correction
        h[0] += diff
    return h

def softmax_obj(x: np.ndarray, beta: float, target: float) -> float:
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta

def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst (largest) overlap term."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr) - (n - 1)
    k = abs(idx)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h: np.ndarray, grad: np.ndarray,
                   rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Move a little mass from a positive‑gradient index to a negative one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def simple_hill_climb(h: np.ndarray, rng: np.random.Generator,
                      steps: int = 2000, scale: float = 0.5):
    """A few gradient‑guided moves after SLSQP."""
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c

            if c < best_c:

        scale = max(scale * 0.999, 1e-5)
    return best_h, best_c
