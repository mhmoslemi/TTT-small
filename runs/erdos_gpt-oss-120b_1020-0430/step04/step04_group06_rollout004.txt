def gradient_move(h, shift, step):
   n = len(h)
   if shift >=0:
        grad = np.zeros(n)
        # left part
        grad[:n-shift] += 1 - h[shift:]
        # right part
        grad[shift:] -= h[:n-shift]
   else:
        k = -shift
        grad = np.zeros(n)
        grad[:n-k] -= h[k:]
        grad[k:] += 1 - h[:n-k]
   # We want to move opposite to gradient.
   # Identify indices where grad > 0 (decrease) and grad < 0 (increase)
   dec_idx = np.where((grad > 0) & (h > 1e-12))[0]
   inc_idx = np.where((grad < 0) & (h < 1 - 1e-12))[0]
   if dec_idx.size == 0 or inc_idx.size == 0:
       return h.copy()
   # Choose idx with max grad magnitude
   i_dec = dec_idx[np.argmax(grad[dec_idx])]
   i_inc = inc_idx[np.argmin(grad[inc_idx])]  # most negative (largest magnitude)
   max_eps = min(h[i_dec], 1 - h[i_inc])
   if max_eps <= 0:
       return h.copy()
   eps = step * max_eps
   new_h = h.copy()
   new_h[i_dec] -= eps
   new_h[i_inc] += eps
   return new_h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utility functions ----------
def enforce_sum(h, target, rng, max_iter=3000):
    """Clip to [0,1] and adjust to hit exact sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    if abs(diff) > 1e-14:  # final tiny correction
        correction = diff / n
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Uniform-ish feasible vector with sum=n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 4):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-14:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        if add == 0.0:
            continue
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)

def local_search(h0, rng, max_iters, time_limit):
    """SA‑style search using three move types."""
    n = h0.size
    best_h, best_c = h0.copy(), compute_c5(h0)
    cur_h, cur_c = best_h.copy(), best_c
    step = 0.5
    T0, T1 = 0.015, 1e-5
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        T = T0 * ((max_iters - it) / max_iters) + T1 * (it / max_iters)
        r = rng.random()
        if r < 0.5:                               # simple perturb
            new_h = perturb_h(cur_h, rng, scale=step)
        elif r < 0.8:                             # pair‑transfer on current worst shift
            _, shift = max_overlap_and_shift(cur_h)
            if shift >= 0:
                i_range = np.arange(n - shift)
                j_range = i_range + shift
            else:
                k = -shift
                i_range = np.arange(k, n)
                j_range = i_range - k
            if i_range.size:
                prod = cur_h[i_range] * (1.0 - cur_h[j_range])
                idx = int(np.argmax(prod))
                i, j = int(i_range[idx]), int(j_range[idx])
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                eps = step * max_eps
                new_h = cur_h.copy()
                new_h[i] -= eps
                new_h[j] += eps
                new_h[i] = max(0.0, min(1.0, new_h[i]))
                new_h[j] = max(0.0, min(1.0, new_h[j]))
            else:
                new_h = cur_h.copy()
        else:                                      # gradient‑guided move
            _, shift = max_overlap_and_shift(cur_h)
            if shift >= 0:
                grad = np.zeros(n)
                grad[:n - shift] += 1.0 - cur_h[shift:]
                grad[shift:] -= cur_h[:n - shift]
            else:
                k = -shift
                grad = np.zeros(n)
                grad[:n - k] -= cur_h[k:]
                grad[k:] += 1.0 - cur_h[:n - k]

CURRENT_TARGET = 0.0
CURRENT_BETA = 0.0

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny residuals)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break

            if add == 0.0:
                continue
            x[i] += add
            diff = target - x.sum()
    return x
