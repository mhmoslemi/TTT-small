import time
import numpy as np

# -------------------------------------------------
# Utility functions
# -------------------------------------------------
def enforce_sum(h, target, rng):
    """Adjust h ∈ [0,1]^n to have exact sum `target`."""
    diff = target - h.sum()
    n = h.size
    if abs(diff) < 1e-12:
        return h
    # distribute diff uniformly among non‑saturated entries
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        if idx.size == 0:
            break
        add = diff / idx.size
        if diff > 0:
            add = np.minimum(add, 1 - h[idx])
        else:
            add = np.maximum(add, -h[idx])
        h[idx] += add
        diff = target - h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # greedy finish
        for _ in range(5000):
            i = rng.integers(0, n)
            step = min(diff, 1 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += step
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def compute_c5(h):
    """Exact C5 from full correlation."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)

# -------------------------------------------------
# Core optimizer for a fixed grid size n
# -------------------------------------------------
def optimize_n(n, rng, time_budget, init_h=None):
    """Hill‑climbing / SA on the min‑overlap formulation."""
    target = n / 2.0

    # ----- initialise h -------------------------------------------------
    if init_h is not None:
        if init_h.size != n:
            old_x = (np.arange(init_h.size) + 0.5) * (2.0 / init_h.size)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            h = np.interp(new_x, old_x, init_h)
        else:
            h = init_h.copy()
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)
    else:
        # start from the uniform vector and perturb a bit
        h = np.full(n, 0.5)
        for _ in range(n * 3):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_eps = min(h[i], 1 - h[j])
            if max_eps <= 0:
                continue
            eps = (rng.random() - 0.5) * 2 * max_eps * 0.4
            h[i] -= eps
            h[j] += eps
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target, rng)

    # ----- initial autocorrelation sums S_L = Σ h_i h_{i+L} ----------
    fx = np.fft.fft(h)
    S = np.real(np.fft.ifft(fx * fx.conj()))
    S = np.maximum(S, 0.0)

    cur_c = 1.0 - (2.0 / n) * S.min()
    best_h = h.copy()
    best_c = cur_c

    # ----- SA / hill‑climb parameters -------------------------------
    T0, Tf = 0.02, 1e-5
    step_max = 0.4 * target   # absolute maximal transfer per move
    start = time.time()
    it = 0

    while True:
        now = time.time()
        if now - start > time_budget:
            break

        # temperature (exponential decay)
        frac = (now - start) / time_budget
        T = T0 * (Tf / T0) ** frac

        # lag with smallest overlap (worst for C5)
        lag_w = int(np.argmin(S))
        if lag_w == 0:
            g = 2.0 * h
        else:
            g = np.roll(h, -lag_w) + np.roll(h, lag_w)

        # choose source (decrease) and sink (increase) indices
        mask_low = h > 1e-12
        mask_high = h < 1 - 1e-12
        if mask_low.any() and mask_high.any():
            low_idx = np.where(mask_low)[0]
            high_idx = np.where(mask_high)[0]
            i_low = low_idx[np.argmin(g[low_idx])]
            i_high = high_idx[np.argmax(g[high_idx])]
        else:
            i_low = rng.integers(0, n)
            i_high = rng.integers(0, n)
            while i_high == i_low:
                i_high = rng.integers(0, n)

        max_eps = min(h[i_low], 1 - h[i_high])
        if max_eps <= 0:
            it += 1
            continue
        eps = min(step_max, max_eps) * rng.random()
        if eps <= 0:
            it += 1
            continue

        # rolls for update
        hp_i = np.roll(h, -i_low)
        hm_i = np.roll(h, i_low)
        hp_j = np.roll(h, -i_high)
        hm_j = np.roll(h, i_high)

        # provisional update
        S += eps * (-hp_i - hm_i + hp_j + hm_j)
        h[i_low] -= eps
        h[i_high] += eps

        # periodic drift correction
        if (it & 255) == 0:
            diff = target - h.sum()
            if abs(diff) > 1e-10:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
                fx = np.fft.fft(h)
                S = np.real(np.fft.ifft(fx * fx.conj()))
                S = np.maximum(S, 0.0)

        new_c = 1.0 - (2.0 / n) * S.min()
        dE = new_c - cur_c
        accept = False
        if dE < 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert
            S -= eps * (-hp_i - hm_i + hp_j + hm_j)
            h[i_low] += eps
            h[i_high] -= eps

        step_max *= 0.9999
        it += 1

    # ----- binary projection (exact n/2 ones) -------------------------
    if n % 2 == 0:
        half = n // 2
        idx_top = np.argpartition(-h, half - 1)[:half]
        binary_h = np.zeros(n)
        binary_h[idx_top] = 1.0
        c_bin = compute_c5(binary_h)
        if c_bin < best_c:
            best_c, best_h = c_bin, binary_h.copy()

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, target, rng)
    final_c = compute_c5(best_h)
    return best_h, final_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    # base size (use parent if available)
    base_n = 360
    if "initial_h_values" in globals():
        base_n = len(initial_h_values)

    # candidate grid sizes (all ≤ 960)
    candidates = [base_n]
    for f in (1.33, 1.5, 2.0):
        n = int(base_n * f)
        if 256 <= n <= 960:
            candidates.append(n)
    candidates = sorted(set(candidates))

    best_h, best_c, best_n = None, np.inf, None

    for idx, n in enumerate(candidates):
        time_elapsed = time.time() - t0
        time_left = budget_s - time_elapsed
        if time_left < 5:
            break
        # allocate a share of the remaining budget
        remaining = len(candidates) - idx
        allot = max(5.0, time_left / remaining)

        # initialise for this n
        init_h = None
        if n == base_n and "initial_h_values" in globals():
            init_h = np.array(initial_h_values, dtype=float)
        elif best_h is not None and best_n is not None:
            # upscale the current best solution
            old_x = (np.arange(best_n) + 0.5) * (2.0 / best_n)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            init_h = np.interp(new_x, old_x, best_h)

        h_n, c_n = optimize_n(n, rng, allot, init_h)

        if c_n < best_c:
            best_c, best_h, best_n = c_n, h_n.copy(), n

    # final verification and return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_n / 2.0, rng)
    final_c = compute_c5(best_h)
    return best_h.tolist(), float(final_c), int(best_n)
