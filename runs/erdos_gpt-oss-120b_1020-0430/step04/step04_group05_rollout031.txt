def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    n = h.size
    if lag >= n:
        return perturb_h(h, rng, scale)
    length = n - lag

def simulated_annealing(h_start, rng, max_iters, max_duration):
    t_start = time.time()
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0 = 0.1
    Tf = 1e-5
    for it in range(max_iters):
        if time.time() - t_start > max_duration:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        # compute worst lag
        lag = np.argmax(cur_corr) - (n - 1)
        lag_abs = abs(lag)
        if rng.random() < 0.6:
            new_h = targeted_perturb(h, rng, lag_abs, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / (T + 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # compute log-sum-exp = m + log(sum(exp(bc-m)))
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum exactly if needed
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

import time
import numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng, max_iter=10000):
    """Clip to [0,1] and adjust to meet exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(h < 1 - 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            cand = np.where(h > 1e-12)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            sub = min(-diff, h[i])
            h[i] -= sub
    # tiny residual correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum=n/2."""
    h = rng.random(n)
    # scale to average 0.5 then clip
    h = h / h.mean() * 0.5
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, n / 2.0, rng)
    return h

def compute_c5_corr(h):
    """Return (C5, full correlation array)."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    c = np.max(corr) * (2.0 / n)
    return c, corr

def compute_c5(h):
    c, _ = compute_c5_corr(h)
    return c

def perturb_h(h, rng, scale=0.2):
    """Transfer a small ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = (rng.random() * 2 - 1) * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def targeted_perturb(h, rng, lag, top_frac=0.1, scale=0.5):
    """Bias the transfer toward indices that dominate the worst lag."""
    n = h.size
    if lag >= n:
        return perturb_h(h, rng, scale)
    length = n - lag
    if length <= 0:
        return perturb_h(h, rng, scale)
    idx = np.arange(length)
    contrib = h[idx] * (1.0 - h[idx + lag])
    k = max(1, int(len(idx) * top_frac))
    top_idx = idx[np.argpartition(-contrib, k - 1)[:k]]
    bot_idx = idx[np.argpartition(contrib, k - 1)[:k]]
    i = rng.choice(top_idx)
    if rng.random() < 0.7:
        j = rng.choice(bot_idx)
    else:
        j = rng.integers(0, n)
    if i == j:
        j = (j + 1) % n
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return perturb_h(h, rng, scale)
    eps = (rng.random() * 2 - 1) * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def simulated_annealing(h_start, rng, max_iters, max_duration):
    """SA with temperature schedule and targeted moves."""
    t_start = time.time()
    h = h_start.copy()
    cur_c, cur_corr = compute_c5_corr(h)
    best_h, best_c = h.copy(), cur_c
    n = h.size
    T0, Tf = 0.1, 1e-5
    for it in range(max_iters):
        if time.time() - t_start > max_duration:
            break
        T = T0 * (Tf / T0) ** (it / max_iters)
        lag = np.argmax(cur_corr) - (n - 1)
        lag_abs = abs(lag)
        if rng.random() < 0.6:
            new_h = targeted_perturb(h, rng, lag_abs, top_frac=0.2, scale=0.5)
        else:
            new_h = perturb_h(h, rng, scale=0.5)
        new_c, new_corr = compute_c5_corr(new_h)
        dE = new_c - cur_c
        if dE < 0 or rng.random() < np.exp(-dE / (T + 1e-12)):
            h, cur_c, cur_corr = new_h, new_c, new_corr
            if new_c < best_c:
                best_c, best_h = new_c, new_h.copy()
    return best_h, best_c

# ---------- SLSQP with smooth max ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize_one_beta(h0, beta, rng, max_iter=300):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-10, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, CURRENT_TARGET, rng)
    return x
