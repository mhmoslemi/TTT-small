import time, numpy as np
from scipy.optimize import minimize

# ---------- utilities ----------
def enforce_sum(h, target, rng):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    max_iter = 10000
    it = 0
    while abs(diff) > 1e-12 and it < max_iter:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            add = min(diff, 1.0 - h[i]) * rng.random()
            h[i] += add
            diff -= add
        else:
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = rng.choice(idx)
            sub = min(-diff, h[i]) * rng.random()
            h[i] -= sub
            diff += sub
        it += 1
    if abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    return enforce_sum(h, n / 2.0, rng)

def perturb_random(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    # move epsilon from i to j
    if h[i] > 0 and h[j] < 1:
        eps = min(h[i], 1.0 - h[j]) * scale * rng.random()
        nh = h.copy()
        nh[i] -= eps
        nh[j] += eps
        return nh
    return h.copy()

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def worst_shift_info(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (h.size - 1)
    return s, idx, corr[idx]

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[: n - s] += (1 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        ss = -s
        grad[ss:] += (1 - h[: n - ss])
        grad[: n - ss] -= h[ss:]
    return grad

# ---------- greedy ----------
def greedy_optimize(h, max_steps=5000):
    for _ in range(max_steps):
        s, _, cur = worst_shift_info(h)
        grad = gradient_shift(h, s)
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-12:
            break
        # try full transfer
        nh = h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        # try half
        eps = max_eps * 0.5
        nh[i] -= eps
        nh[j] += eps
        if compute_c5(nh) + 1e-12 < compute_c5(h):
            h = nh
            continue
        break
    return h

# ---------- SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log-sum-exp safely
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=1000):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    rng = np.random.default_rng()
    x = enforce_sum(x, CURRENT_TARGET, rng)
    return x

# ---------- gradient descent on worst shift ----------
def gradient_descent_reduce(h, rng, time_limit):
    start = time.time()
    alpha = 0.02
    while time.time() - start < time_limit:
        s, _, cur = worst_shift_info(h)
        grad = gradient_shift(h, s)
        proj = grad - grad.mean()
        for _ in range(5):
            step = -alpha * proj
            nh = h + step
            nh = np.clip(nh, 0.0, 1.0)
            nh = enforce_sum(nh, h.size / 2.0, rng)
            if compute_c5(nh) + 1e-12 < compute_c5(h):
                h = nh
                alpha *= 1.1
                break
            alpha *= 0.5
        else:
            break
    return h

# ---------- main entry ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 720
        h0 = random_feasible_vector(n, rng)

    h0 = enforce_sum(np.clip(h0, 0.0, 1.0), n / 2.0, rng)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # main optimisation phase (≈80% of budget)
    main_limit = start_time + budget_s * 0.80
    while time.time() < main_limit:
        # generate start point
        if rng.random() < 0.6 and best_h is not None:
            start = perturb_random(best_h, rng, scale=0.6)
        else:
            start = random_feasible_vector(n, rng)
        # greedy refinement
        local = greedy_optimize(start, max_steps=6000)
        c_local = compute_c5(local)
        if c_local < best_c:
            best_h, best_c = local, c_local

        # occasional SLSQP polishing
        if rng.random() < 0.25:
            betas = (200, 800, 2000, 5000, 10000, 20000)
            for beta in betas:
                if time.time() > main_limit:
                    break
                try:
                    refined = slsqp_optimize(best_h, beta, max_iter=1200)
                except Exception:
                    continue
                c_ref = compute_c5(refined)
                if c_ref < best_c:
                    best_h, best_c = refined, c_ref

        # gradient‑descent on worst shift
        if rng.random() < 0.2:
            gd_time = min(0.5, main_limit - time.time())
            if gd_time > 0:
                best_h = gradient_descent_reduce(best_h, rng, gd_time)
                best_c = compute_c5(best_h)

    # final micro‑search (≈remaining 20% of budget)
    final_limit = start_time + budget_s * 0.99
    while time.time() < final_limit:
        cand = perturb_random(best_h, rng, scale=0.02)
        c_cand = compute_c5(cand)
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
