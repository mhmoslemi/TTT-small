import time, numpy as np, random, math
from scipy.optimize import minimize

def compute_c5(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)
    corr = np.fft.irfft(H * G, L)[:2*n-1]
    return float(corr.max() * (2.0 / n))

def enforce_sum(h, target_sum=None, rng=None):
    if target_sum is None:
        target_sum = h.size / 2.0
    diff = target_sum - h.sum()
    n = h.size
    # simple loop
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target_sum - h.sum()
    return h

def random_feasible_vector(n, rng):
    h = rng.random(n)
    target = n/2.0
    h *= target / h.sum()
    h = np.clip(h, 0, 1)
    return enforce_sum(h, target_sum=target, rng=rng)

def binary_feasible_vector(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def worst_gradient(h):
    n = h.shape[0]
    # compute corr using FFT (fast)
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1 - h, L)
    corr = np.fft.irfft(H * G, L)[:2*n-1]
    idx = np.argmax(corr)
    k = idx - (n - 1)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros_like(h)
    grad[:n-k] += 1.0
    grad[:n-k] -= h[k:]
    grad[k:] -= h[:n-k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

def guided_hill_climb(h, rng, max_iters=2000, step_scale=0.6):
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        # gradually reduce scale
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c

def softmax_obj(x, beta):
    # compute O_k = correlation of x and 1-x
    n = len(x)
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(x, L)
    G = np.fft.rfft(1 - x, L)
    corr = np.fft.irfft(H * G, L)[:2*n-1]
    bc = beta * corr
    m = bc.max()
    # stable log-sum-exp
    return (m + np.log(np.exp(bc - m).sum())) / beta

def slsqp_optimize(h0, beta, rng, max_iter=500):
    n = h0.size
    target = n/2.0
    bounds = [(0.0,1.0)]*n
    constraints = [{'type':'eq','fun':lambda x: np.sum(x)-target}]
    # Use lambda but cannot capture target? It does capture target from outer scope which is fine.
    # However constraints need function that accepts array; ok.

    # Define objective closure
    def obj(x):
        return softmax_obj(x, beta)

    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    x = np.clip(res.x,0,1)
    # enforce exact sum via adjustment
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # add diff to random indices respecting bounds
        n = x.size
        while abs(diff) > 1e-12:
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

def simulated_annealing(h, rng, time_limit, start_time):
    n = len(h)
    target = n/2.0
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    # initial temperature as fraction of cur_c perhaps
    T0 = cur_c * 0.02
    T = T0
    decay = 0.9995
    steps = 0
    while time.time() - start_time < time_limit:
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # propose epsilon within feasible
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0:
            continue
        eps = rng.random() * max_eps * 0.5
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        # adjust sum might be off due to rounding? Should be exact.
        # compute new c
        new_c = compute_c5(nh)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < math.exp(-delta / T):

        T *= decay
        steps += 1
        if steps % 5000 == 0:
            # slowly reduce temperature faster if stagnating
            if T < 1e-10:
                T = T0 * 0.1
    return cur_h, cur_c

import time, math, numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """C5 for a step function on [0,2]. Uses FFT for speed."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * G, L)[: 2 * n - 1]
    return float(corr.max() * (2.0 / n))


def enforce_sum(h: np.ndarray, target_sum: float = None, rng: np.random.Generator = None) -> np.ndarray:
    """Adjust h in‑place so that Σh = target_sum while staying in [0,1]."""
    if target_sum is None:
        target_sum = h.size / 2.0
    if rng is None:
        rng = np.random.default_rng()
    diff = target_sum - h.sum()
    n = h.size
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target_sum - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with Σ= n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target_sum=target, rng=rng)


def binary_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Half ones, half zeros."""
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def worst_gradient(h: np.ndarray) -> np.ndarray:
    """Gradient of the worst‑shift overlap O_k."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(H * G, L)[: 2 * n - 1]
    idx = np.argmax(corr)
    k = idx - (n - 1)
    if k == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros_like(h)
    grad[: n - k] += 1.0
    grad[: n - k] -= h[k:]
    grad[k:] -= h[: n - k]
    return grad


def guided_perturb(h: np.ndarray, grad: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    """Transfer ε from a positive‑gradient index to a negative‑gradient one."""
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh


def guided_hill_climb(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    step_scale: float = 0.6,
) -> tuple[np.ndarray, float]:
    """Hill‑climb using the worst‑shift gradient."""
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    decay = max(1, max_iters // 5)
    for it in range(max_iters):
        grad = worst_gradient(h)
        nh = guided_perturb(h, grad, rng, scale=step_scale)
        c = compute_c5(nh)
        if c < best_c:
            best_c, best_h, h = c, nh.copy(), nh
        if (it + 1) % decay == 0:
            step_scale = max(step_scale * 0.7, 1e-4)
    return best_h, best_c


def softmax_obj(x: np.ndarray, beta: float) -> float:
    """Smooth max via log‑sum‑exp of O_k."""
    n = x.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(x, L)
    G = np.fft.rfft(1.0 - x, L)
    corr = np.fft.irfft(H * G, L)[: 2 * n - 1]
    b = beta * corr
    m = b.max()
    return (m + np.log(np.exp(b - m).sum())) / beta


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP refinement for a given β."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    obj = lambda x: softmax_obj(x, beta)
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        x = enforce_sum(x, target_sum=target, rng=rng)
    return x


def gradient_refine(h: np.ndarray, rng: np.random.Generator, step: float = 0.015, max_iter: int = 5) -> np.ndarray:
    """Fine‑tune using the worst‑shift gradient."""
    n = h.size
    target = n / 2.0
    for _ in range(max_iter):
        grad = worst_gradient(h)
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, target_sum=target, rng=rng)
    return h


def simulated_annealing(
    h: np.ndarray,
    rng: np.random.Generator,
    time_limit: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Metropolis annealing allowing occasional uphill moves."""
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    T = cur_c * 0.02 + 1e-6
    decay = 0.9995
    while time.time() - start_time < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(cur_h[i], 1.0 - cur_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * 0.5
        nh = cur_h.copy()
        nh[i] -= eps
        nh[j] += eps
        new_c = compute_c5(nh)
        d = new_c - cur_c
        if d < 0.0 or rng.random() < math.exp(-d / T):
            cur_h, cur_c = nh, new_c
        T *= decay
    return cur_h, cur_c
