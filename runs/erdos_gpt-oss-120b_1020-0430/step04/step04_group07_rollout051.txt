def binary_sa_optimize(h_bin, rng, time_budget):
    n = h_bin.size
    # compute initial c
    c_cur = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c = c_cur
    # indices sets
    ones = np.flatnonzero(h_bin == 1)
    zeros = np.flatnonzero(h_bin == 0)
    start = time.time()
    it = 0
    no_improve = 0
    while time.time() - start < time_budget:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h_bin[i], h_bin[j] = 0, 1
        c_new = compute_c5(h_bin)
        # acceptance (hill climb)
        if c_new < c_cur or rng.random() < 0.001: # small probability
            c_cur = c_new
            # update sets
            # ones: replace i with j
            # zeros: replace j with i
            # can use swap in arrays for speed
            # but simpler: recompute set each time? Might cost O(n); fine.
            ones = np.flatnonzero(h_bin == 1)
            zeros = np.flatnonzero(h_bin == 0)
        else:
            # revert swap
            h_bin[i], h_bin[j] = 1, 0
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1
        it += 1
        if no_improve > 2000:
            # random shuffle of ones maybe
            # reset to random binary of same sum
            h_bin = random_binary(n, n//2, rng)
            c_cur = compute_c5(h_bin)
            ones = np.flatnonzero(h_bin == 1)
            zeros = np.flatnonzero(h_bin == 0)
            no_improve = 0
    return best_h, best_c

def random_binary(n, m, rng):
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, m, replace=False)] = 1.0

import time, math, numpy as np
from scipy.optimize import minimize

# globals for SLSQP soft‑max objective
CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k O_k."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator,
                   max_iter: int = 500) -> np.ndarray:
    """Soft‑max SLSQP optimisation for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    constraints = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random vector in [0,1] with sum=n/2."""
    h = rng.random(n)
    target = n / 2.0
    h *= target / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def enforce_sum(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale vector to satisfy Σh = n/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly m ones."""
    h = np.zeros(n, dtype=float)
    h[rng.choice(n, m, replace=False)] = 1.0
    return h


def binary_sa_optimize(h_bin: np.ndarray, rng: np.random.Generator,
                       time_budget: float):
    """Simulated‑annealing hill‑climb on binary vectors."""
    n = h_bin.size
    best_h = h_bin.copy()
    best_c = compute_c5(best_h)
    cur_h = h_bin.copy()
    cur_c = best_c
    start = time.time()
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = 0.01 * (1.0 - frac) + 1e-6
        ones = np.flatnonzero(cur_h == 1.0)
        zeros = np.flatnonzero(cur_h == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c = compute_c5(cur_h)
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            if rng.random() < math.exp(-(new_c - cur_c) / T):
                accept = True
        if not accept:
            cur_h[i], cur_h[j] = 1.0, 0.0
        else:
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c
