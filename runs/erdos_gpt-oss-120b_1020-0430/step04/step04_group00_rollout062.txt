def binary_sa_optimize(h_init, time_deadline, rng, max_iters=200000):
    n = len(h_init)
    m = 2*n - 1
    dx = 2.0 / n
    h = h_init.astype(np.int8)
    corr = np.correlate(h, 1 - h, mode='full')
    cur_max = corr.max()
    cur_c = cur_max * dx
    best_c = cur_c
    best_h = h.copy()
    # arrays for speed
    arr = 1 - 2*h   # int8
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    delta = np.empty(m, dtype=int)  # reuse
    # temperature schedule
    init_T = cur_c * 0.05 + 1e-9  # small factor
    # schedule parameters
    for it in range(max_iters):
        if time.time() > time_deadline:
            break
        # optional early exit if no improvement for many steps? we can skip.
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # compute delta
        delta.fill(0)
        delta[i:i+n] -= arr
        delta[j:j+n] += arr
        idx1 = i - j + n - 1
        delta[idx1] += 1
        idx2 = j - i + n - 1
        if idx2 != idx1:
            delta[idx2] += 1
        # compute new max
        # new_corr = corr + delta  # we just compute max directly
        new_max = (corr + delta).max()
        new_c = new_max * dx
        # temperature
        T = init_T * (1 - it / max_iters)
        # acceptance
        if new_c <= cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            # accept move
            # update h
            h[i] = 0
            h[j] = 1

def binary_sa_optimize(h_init, deadline, rng, max_iters=200000):
    n = len(h_init)
    m = 2*n - 1
    dx = 2.0 / n
    h = h_init.astype(np.int8)
    not_h = 1 - h
    corr = np.correlate(h, not_h, mode='full')
    cur_max = corr.max()
    cur_c = cur_max * dx
    best_c = cur_c
    best_h = h.copy()
    # pre-allocate delta_h array
    delta_h = np.zeros(n, dtype=np.int8)
    # precompute indexed zeros and ones
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(not_h))
    # temperature schedule parameters
    init_T = cur_c * 0.2 + 1e-12  # starting temperature
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # choose random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # reset delta_h
        delta_h.fill(0)
        delta_h[i] = -1
        delta_h[j] = 1
        # compute correlations
        d_not_h = np.correlate(delta_h, not_h, mode='full')
        h_delta = np.correlate(h, delta_h, mode='full')
        d_d = np.correlate(delta_h, delta_h, mode='full')
        delta_corr = d_not_h - h_delta - d_d
        new_corr = corr + delta_corr
        new_max = new_corr.max()
        new_c = new_max * dx
        # temperature
        T = init_T * (1 - it / max_iters)
        accept = False
        if new_c <= cur_c:
            accept = True
        else:
            if rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
                accept = True
        if accept:
            # commit swap
            h[i] = 0
            h[j] = 1
            not_h[i] = 1
            not_h[j] = 0
            # update lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            # update corr and cur values
            corr = new_corr
            cur_max = new_max
            cur_c = new_c
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def gradient_refine(h, rng, step=0.02, max_iter=6):
    n = h.shape[0]
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)  # shift index as negative maybe
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            # contributions for shift k in corr index: we need to compute gradient w.r.t h.
            # earlier code used slices: g[:upper] += 1 - h[k:] ; g[k:] -= h[:upper]
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        # adjust sum to target
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # fix sum via random adjustments as before
            for _ in range(200):
                idx_adj = rng.integers(0, n)
                add = min(diff, 1.0 - h[idx_adj]) if diff > 0 else max(diff, -h[idx_adj])
                h[idx_adj] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def gradient_refine(h, rng, step=0.02, max_iter=6):
    n = h.shape[0]
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def binary_sa_optimize(h_init, deadline, rng, max_iters=200000):
    n = h_init.shape[0]
    m = 2 * n - 1
    dx = 2.0 / n
    h = h_init.astype(np.int8)
    not_h = 1 - h
    corr = np.correlate(h, not_h, mode='full')
    cur_max = corr.max()
    cur_c = cur_max * dx
    best_c = cur_c
    best_h = h.copy()
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(not_h))
    delta_h = np.zeros(n, dtype=np.int8)
    init_T = cur_c * 0.2 + 1e-12
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        delta_h.fill(0)
        delta_h[i] = -1
        delta_h[j] = 1
        d_not_h = np.correlate(delta_h, not_h, mode='full')
        h_delta = np.correlate(h, delta_h, mode='full')
        d_d = np.correlate(delta_h, delta_h, mode='full')
        delta_corr = d_not_h - h_delta - d_d
        new_corr = corr + delta_corr
        new_max = new_corr.max()
        new_c = new_max * dx
        T = init_T * (1 - it / max_iters)
        if new_c <= cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            # accept
            h[i] = 0
            h[j] = 1
            not_h[i] = 1
            not_h[j] = 0
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            corr = new_corr
            cur_max = new_max
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
