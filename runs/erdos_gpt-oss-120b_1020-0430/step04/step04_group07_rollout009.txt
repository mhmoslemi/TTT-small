def binary_anneal(h_bin, rng, time_limit):
    start = time.time()
    n = h_bin.size
    best_h = h_bin.copy()
    best_c = compute_c5(best_h)
    current_h = h_bin.copy()
    current_c = best_c
    T = best_c * 0.5 + 1e-6
    it = 0
    while time.time() - start < time_limit:
        ones = np.flatnonzero(current_h == 1)
        zeros = np.flatnonzero(current_h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        new_h[j] = 1

            current_h = new_h

            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        it += 1
        if it % 200 == 0:

def slsqp_optimize(h0, beta, rng, max_iter=600):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(softmax_obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    # enforce sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        # adjust using random steps
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

from scipy.optimize import minimize

CURRENT_TARGET = None
CURRENT_BETA = None


def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h: np.ndarray, rng: np.random.Generator, target: float = None) -> np.ndarray:
    """Adjust h so that sum(h)=target while staying in [0,1]."""
    n = h.size
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    h[rng.choice(n, n // 2, replace=False)] = 1
    return h


def binary_anneal(h_bin: np.ndarray, rng: np.random.Generator, time_limit: float) -> tuple:
    """Simulated‑annealing on binary vectors to reduce max overlap."""
    start = time.time()
    n = h_bin.size
    best_h = h_bin.copy()

    cur_h = h_bin.copy()

        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)

def sum_eq_constraint(x: np.ndarray) -> float:

def softmax_obj(x: np.ndarray) -> float:
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, rng: np.random.Generator, max_iter: int = 600) -> np.ndarray:
    """Soft‑max SLSQP refinement."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:

            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x
