import time, math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_bounds(h, target):
    """Clip h to [0,1] and project onto the affine hyperplane sum(h)=target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # add proportionally to available capacity
    if diff > 0:
        cap = (1.0 - h).sum()
        if cap > 0:
            h += (diff / cap) * (1.0 - h)
    else:
        cap = h.sum()
        if cap > 0:
            h += (diff / cap) * h
    # final numeric clean‑up
    return np.clip(h, 0.0, 1.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε between two random coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def gd_optimize(start_h, target, time_limit):
    """Projected gradient descent focusing on the worst shift."""
    h = start_h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step0 = 0.25
    decay = 0.9996
    it = 0
    start = time.time()
    n = h.size
    while time.time() - start < time_limit:
        # correlation and current worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        cur_c = corr[idx] * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # gradient of overlap for this shift (unscaled by dx)
        if shift >= 0:
            term1 = np.empty(n)
            term1[: n - shift] = 1.0 - h[shift:]
            term1[n - shift :] = 0.0
            term2 = np.empty(n)
            term2[shift:] = h[: n - shift]
            term2[: shift] = 0.0
        else:
            s = -shift
            term1 = np.empty(n)
            term1[s:] = 1.0 - h[: n - s]
            term1[:s] = 0.0
            term2 = np.empty(n)
            term2[: n - s] = h[s:]
            term2[n - s :] = 0.0
        grad = term1 - term2
        # step size schedule
        alpha = step0 * (decay ** it)
        # gradient step, then projection
        h = h - alpha * grad
        h = project_sum_bounds(h, target)
        it += 1
        # occasional random jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.15:
            h = perturb_h(h, rng, scale=0.3)
    return best_h, best_c


def enforce_sum_exact(h, target):
    """Final tiny adjustment to hit the sum constraint without breaking bounds."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to adjust a single entry
    if diff > 0:
        cand = np.where(h < 1.0 - 1e-12)[0]
    else:
        cand = np.where(h > 1e-12)[0]
    if cand.size:
        i = int(cand[0])
        h[i] += diff
        h[i] = max(0.0, min(1.0, h[i]))
    # if still off, fall back to projection
    return project_sum_bounds(h, target)

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    # -------------------------------------------------
    # 0. Choose discretisation size (larger n gives more flexibility)
    # -------------------------------------------------

        h_interp = project_sum_bounds(h_interp, target_sum)

    # deterministic shapes

    # linear ramps

    candidates.append(project_sum_bounds(1.0 - xs / 2.0, target_sum))

    candidates.append(project_sum_bounds(mask, target_sum))

    # random draws
    for _ in range(4):
        rand = rng.random(n)
        candidates.append(project_sum_bounds(rand, target_sum))

    # -------------------------------------------------
    # 2. Pick the best among the pool
    # -------------------------------------------------
    best_h = None
    best_c = math.inf
    for h in candidates:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    global_best_h, global_best_c = best_h, best_c

    # -------------------------------------------------

    # -------------------------------------------------
    main_time = budget_s * 0.90
    while time.time() - start_time < main_time:
        # choose start point
        if rng.random() < 0.6:
            start_vec = perturb_h(global_best_h, rng, scale=0.3)
        else:
            start_vec = rng.random(n)
            start_vec = project_sum_bounds(start_vec, target_sum)
        # allocate a small chunk of time for this run
        remaining = budget_s - (time.time() - start_time)
        run_time = min(35.0, remaining * 0.6)
        h_opt, c_opt = gd_optimize(start_vec, target_sum, run_time)
        if c_opt < global_best_c:
            global_best_h, global_best_c = h_opt, c_opt

    # -------------------------------------------------
    # 4. Final clean‑up
    # -------------------------------------------------
    global_best_h = enforce_sum_exact(np.clip(global_best_h, 0.0, 1.0), target_sum)
    final_c = compute_c5(global_best_h)

    return global_best_h.tolist(), float(final_c), int(n)
