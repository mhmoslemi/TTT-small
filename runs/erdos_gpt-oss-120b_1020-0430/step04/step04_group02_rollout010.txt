import time
import numpy as np
from scipy.optimize import minimize
from numpy.polynomial.chebyshev import chebval

# ---------- feasibility helpers ----------
def project_box_sum(x, target, lo=0.0, hi=1.0, max_iter=60):
    low, high = -2.0, 2.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, lo, hi)
        s = y.sum()
        if s > target:
            high = mid
        else:
            low = mid
    y = np.clip(x + (low + high) / 2.0, lo, hi)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > lo + 1e-12) & (y < hi - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, lo, hi)
    return y


def random_feasible_vector(n, rng):
    return project_box_sum(rng.random(n), n / 2.0)


def perturb_h(h, rng, scale=0.2):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ---------- objective ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def compute_worst_shift(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    val = corr[idx] * (2.0 / n)
    return shift, val


def gradient_for_shift(h, k):
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        if k < n:
            g[: n - k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[: n - k]
    else:
        kp = -k
        if kp < n:
            g[kp:] += (1.0 - h[: n - kp])
        if kp > 0:
            g[: n - kp] -= h[kp:]
    return g


# ---------- gradient descent ----------
def gradient_descent(h0, rng, max_iters, time_limit):
    start = time.time()
    n = h0.size
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    scales = (0.02, 0.05, 0.1, 0.2, 0.4)
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        k_star, _ = compute_worst_shift(cur_h)
        grad = gradient_for_shift(cur_h, k_star)
        improved = False
        for s in scales:
            cand = cur_h - s * grad
            cand = project_box_sum(cand, n / 2.0)
            cand_c = compute_c5(cand)
            if cand_c < cur_c - 1e-12:
                cur_h, cur_c = cand, cand_c
                improved = True
                if cand_c < best_c:
                    best_c, best_h = cand_c, cand.copy()
                break
        if not improved:
            cur_h = perturb_h(cur_h, rng, scale=0.05)
            cur_c = compute_c5(cur_h)
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
    return best_h, best_c


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_box_sum(x, CURRENT_TARGET)
