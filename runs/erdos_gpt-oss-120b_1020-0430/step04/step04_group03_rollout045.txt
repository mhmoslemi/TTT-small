def corr_and_grad_softmax(h, beta, lambda_sum):
    # compute correlation array (size 2*n-1)
    corr = np.correlate(h, 1 - h, mode='full')
    # objective = softmax (max) + lambda sum penalty
    # Compute max for numeric stability:
    m = np.max(beta * corr)
    exp_terms = np.exp(beta * corr - m)
    sum_exp = np.sum(exp_terms)
    soft = (m + np.log(sum_exp)) / beta
    # gradient weighting:
    p = exp_terms / sum_exp # this is p_k = exp(beta*corr_k)/sum_exp
    # compute gradient contributions
    # gradient_shift function returns gradient for shift s: gradient of correlation wrt h.
    n = h.size
    grad = np.zeros_like(h)
    # loop over all s with weight p_k
    # vectorized? Could precompute something. Let's just loop.
    for idx, weight in enumerate(p):
        s = idx - (n - 1)
        # compute grad_s
        if s >= 0:
            # term: grad[:n-s] += (1 - h[s:])
            # grad[s:] -= h[:n-s]
            # Multiply by weight
            grad[:n - s] += weight * (1 - h[s:])
            grad[s:] -= weight * h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] += weight * (1 - h[:n - s_abs])
            grad[:n - s_abs] -= weight * h[s_abs:]
    # add sum constraint penalty gradient
    sum_diff = h.sum() - n/2.0
    grad += 2 * lambda_sum * sum_diff
    # objective also includes sum penalty term:
    soft_obj = soft + lambda_sum * sum_diff * sum_diff
    return soft_obj, grad

def balance_worst_shift(h):
    while True:
        # find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        s = idx - (h.size - 1)
        # compute grad array for shift s
        n = h.size
        grad = np.zeros(n)
        if s >= 0:
            grad[:n - s] = (1 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] = (1 - h[:n - s_abs])
            grad[:n - s_abs] -= h[s_abs:]
        # Identify pos and neg indices
        pos = np.where((grad > 1e-12) & (h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (h < 1 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        # sort pos descending by grad, neg ascending by grad
        pos_order = pos[np.argsort(-grad[pos])]
        neg_order = neg[np.argsort(grad[neg])]  # ascending (most negative first)
        # supplies and caps
        supply = h[pos_order].copy()
        cap = 1 - h[neg_order]
        i = 0
        j = 0
        moved = False
        while i < len(pos_order) and j < len(neg_order):
            move = min(supply[i], cap[j])
            if move > 1e-12:
                h[pos_order[i]] -= move
                h[neg_order[j]] += move
                moved = True
                supply[i] -= move
                cap[j] -= move
            if supply[i] < 1e-12:
                i += 1
            if cap[j] < 1e-12:
                j += 1
        # recompute correlation after balancing
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = new_corr.max()
        # if not improved break
        if new_max >= corr[idx] - 1e-12:
            break
        # else continue loop (again find worst shift)
    return h
