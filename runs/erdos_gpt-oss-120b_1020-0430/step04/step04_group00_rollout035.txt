def delta_corr_swap(h, i, j):
    n = len(h)
    g = 1 - h
    L = 2*n - 1
    delta = np.zeros(L)
    for k in range(-n+1, n):
        idx = k + n - 1
        val = 0
        # term1
        if 0 <= j + k < n:
            val += g[j + k]
        if 0 <= i + k < n:
            val -= g[i + k]
        # term2
        if 0 <= i - k < n:
            val += h[i - k]
        if 0 <= j - k < n:
            val -= h[j - k]
        # term3
        if k == 0:
            val += -2
        if k == j - i:
            val += 1
        if k == i - j:
            val += 1
        delta[idx] = val
    return delta
```

def adjust_sum_exact(h, rng, max_iter=2000):
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:

    delta[n-1 + (i - j)] += 1.0

        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # ----- fallback if nothing succeeded -----
    if best_h is None:
        n = 360
        best_n = n
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        adjust_sum_exact(best_h, rng)
        best_c = compute_c5(best_h)

    # final safety
    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
