def gradient_refine_multi(h, rng, step=0.02, max_iter=12):
    n = h.size
    target = n/2
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
        ks = idxs - (n - 1)
        g = np.zeros_like(h)
        for k in ks:
            if k == 0:
                g += 1 - 2*h
            elif k > 0:
                upper = n - k
                g[:upper] += 1 - h[k:]
                g[k:] -= h[:upper]
            else: # k <0
                m = -k
                upper = n - m
                g[m:] += 1 - h[:upper]
                g[:upper] -= h[m:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        # clip
        h = np.clip(h, 0.0, 1.0)
        # adjust sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

import time
import numpy as np
from scipy.optimize import minimize

# global variables for SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

def compute_c5(h: np.ndarray) -> float:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force sum(h) == n/2 by moving mass between random coordinates."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x: np.ndarray) -> float:
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0: np.ndarray, beta: float,
                   rng: np.random.Generator, max_iter: int = 400) -> np.ndarray:
    """SLSQP on the log‑sum‑exp surrogate of the max overlap."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

def gradient_refine_multi(h: np.ndarray, rng: np.random.Generator,
                         step: float = 0.02, max_iter: int = 12) -> np.ndarray:
    """Projected sub‑gradient descent using all worst shifts."""
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = corr.max()
        idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
        ks = idxs - (n - 1)
        g = np.zeros_like(h)
        for k in ks:
            if k == 0:
                g += 1 - 2 * h
            elif k > 0:
                upper = n - k
                g[:upper] += 1 - h[k:]
                g[k:] -= h[:upper]
            else:  # k < 0
                m = -k
                upper = n - m
                g[m:] += 1 - h[:upper]
                g[:upper] -= h[m:]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def simulated_annealing_continuous(h: np.ndarray,
                                   rng: np.random.Generator,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Simple SA that swaps small mass between two coordinates."""
    n = h.size
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    T0 = cur_c * 0.2 + 1e-8
    it = 0
    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_h[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if cur_h[j] >= 1.0:
            continue
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        if delta <= 0.0:
            continue
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c = compute_c5(new_h)
        if new_c < cur_c:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
        else:

            if rng.random() < np.exp(-(new_c - cur_c) / T):
