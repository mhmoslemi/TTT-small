def simulated_annealing(h, rng, time_limit, start_time):
    n = h.size
    target = n / 2.0
    cur_h = h.copy()
    cur_c, cur_shift, _ = compute_c5_and_shift(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    # temperature schedule
    T0 = 0.001
    T = T0
    cooling = 0.9995
    step_scale = 0.5
    while time.time() - start_time < time_limit:
        grad = worst_gradient(cur_h, cur_shift)
        nh = guided_perturb(cur_h, grad, rng, scale=step_scale)
        nc, nshift, _ = compute_c5_and_shift(nh)
        if nc < cur_c or rng.random() < math.exp(-(nc - cur_c) / max(T, 1e-12)):
            cur_h, cur_c, cur_shift = nh, nc, nshift
            if nc < best_c:
                best_h, best_c = nh.copy(), nc
        # temperature decay
        T = max(T * cooling, 1e-12)
        # adjust step_scale softly
        step_scale = max(step_scale * 0.995, 0.01)
        # occasional random restart
        if rng.random() < 0.001:
            # random feasible init
            nh = np.zeros(n)
            nh[rng.choice(n, n//2, replace=False)] = 1.0
            cur_h, cur_c, cur_shift = nh, *compute_c5_and_shift(nh)
    return best_h, best_c

import time, math, numpy as np

# ----- utilities -------------------------------------------------
def enforce_sum(h, target, rng):
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        i = rng.integers(0, h.size)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

def compute_c5_details(h):
    n = h.size
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, size)
    G = np.fft.rfft((1 - h)[::-1], size)
    conv = np.fft.irfft(H * G, size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    max_corr = conv[idx]
    shift = idx - (n - 1)
    c5 = max_corr * (2.0 / n)
    return c5, shift, conv

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def worst_gradient(h, shift):
    n = h.size
    k = abs(shift)
    grad = np.ones_like(h)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    return grad

def guided_perturb(h, grad, rng, scale=0.5):
    i = int(np.argmax(grad))
    j = int(np.argmin(grad))
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = rng.random() * max_eps * scale
    nh = h.copy()
    nh[i] -= eps
    nh[j] += eps
    return nh

    # optional up‑sampling -------------------------------------------
    if time.time() - start < budget_s * 0.95 and best_n * 2 <= 512:
        factor = 2
        n_up = best_n * factor
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0.0, 1.0)
        h_up = enforce_sum(h_up, n_up / 2.0, rng)
        h_up, _ = hill_climb(h_up, rng, max_iters=3000, step_scale=0.6)
        for beta in (20., 200., 2000., 8000.):
            if time.time() - start > budget_s * 0.99:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up.copy(), c_up, n_up

    # final stochastic tweaks ----------------------------------------
    while time.time() - start < budget_s * 0.999:
        grad = worst_gradient(best_h, compute_c5_details(best_h)[1])
        nh = guided_perturb(best_h, grad, rng, scale=0.02)
        nc = compute_c5(nh)
        if nc < best_c:
            best_h, best_c = nh.copy(), nc

    # final sanity check --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
