    h = np.clip(h, 0.0, 1.0)
    # ensure sum = target via water-filling:
    diff = target - h.sum()
    # We'll adjust by adding diff uniformly to non-bound entries
    # Repeat until diff small.
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # indices not at bounds
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # all at bounds, cannot adjust; just distribute equally then clip
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            continue
        add = diff / free.size
        h[free] += add
        # clip
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

def compute_gradient(h, beta):
    n = h.size
    target = n/2.0
    # compute auto_corr via convolution
    L = 1 << ((3*n - 2).bit_length())
    # FFT of h and h_rev
    f_h = np.fft.rfft(h, n=L)
    f_h_rev = np.fft.rfft(h[::-1], n=L)
    auto_corr = np.fft.irfft(f_h_rev * f_h, n=L)[:2*n-1]  # correlation h with h
    S = h.sum()
    c = S - auto_corr  # shape 2n-1
    # softmax weights w
    max_c = np.max(c)
    exp_vals = np.exp(beta * (c - max_c))
    w = exp_vals / exp_vals.sum()
    # conv of w with h and w_rev with h
    f_w = np.fft.rfft(w, n=L)
    f_w_rev = np.fft.rfft(w[::-1], n=L)
    conv_wh = np.fft.irfft(f_w * f_h, n=L)[:n]
    conv_w_rev_h = np.fft.irfft(f_w_rev * f_h, n=L)[:n]
    grad = 1.0 - (conv_wh + conv_w_rev_h)
    return grad

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _project_feasible(h, target):
    """Clip to [0,1] and enforce Σh = target (bounded simplex)."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple water‑filling on free variables
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            continue
        add = diff / free.size
        h[free] += add
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h


def _random_feasible(n, rng):
    """Create a random vector in [0,1]^n with Σ = n/2."""
    target = n / 2.0
    h = rng.random(n)
    h = _project_feasible(h, target)
    return h


def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _max_overlap_and_shift(h):
    """Return (max_corr, shift) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1)


def pair_transfer_optimize(h0, rng, max_iters=8000, time_limit=30.0,
                           init_factor=0.5, decay=0.999):
    """Local search moving mass from i to i+shift to reduce the worst overlap."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    factor = init_factor

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        max_val, shift = _max_overlap_and_shift(h)
        cur_c = max_val * (2.0 / n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # determine index ranges for this shift
        if shift >= 0:
            i_range = np.arange(n - shift)
            j_range = i_range + shift
        else:
            i_range = np.arange(-shift, n)
            j_range = i_range + shift
        if i_range.size == 0:
            factor = init_factor
            continue
        prod = h[i_range] * (1.0 - h[j_range])
        idx_max = int(np.argmax(prod))
        i = int(i_range[idx_max])
        j = int(j_range[idx_max])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            factor = init_factor
            continue
        eps = factor * max_eps
        if eps <= 0.0:
            factor = init_factor
            continue
        h[i] -= eps
        h[j] += eps
        h[i] = max(0.0, min(1.0, h[i]))
        h[j] = max(0.0, min(1.0, h[j]))
        factor = max(1e-4, factor * decay)

    final_c = compute_c5(best_h)
    return best_h, final_c


def _next_pow2(x):
    return 1 << (x - 1).bit_length()


def _gradient(h, beta, L):
    """Gradient of smooth max_{k} Σ h_i (1 - h_{i+k}) with respect to h."""
    n = h.size
    # correlation h with itself (linear correlation)
    f_h = np.fft.rfft(h, n=L)
    f_h_rev = np.fft.rfft(h[::-1], n=L)
    auto_corr = np.fft.irfft(f_h_rev * f_h, n=L)[: 2 * n - 1]
    S = h.sum()
    c = S - auto_corr                       # shape 2n-1
    # softmax weights
    max_c = np.max(c)
    exp_vals = np.exp(beta * (c - max_c))
    w = exp_vals / exp_vals.sum()
    # convolutions w*h and w_rev*h
    f_w = np.fft.rfft(w, n=L)
    f_w_rev = np.fft.rfft(w[::-1], n=L)
    conv_wh = np.fft.irfft(f_w * f_h, n=L)[:n]
    conv_w_rev_h = np.fft.irfft(f_w_rev * f_h, n=L)[:n]
    grad = 1.0 - (conv_wh + conv_w_rev_h)
    return grad

# Main optimisation driver
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise ------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project_feasible(h0, target)
    else:
        n = 360
        target = n / 2.0
        h0 = _random_feasible(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- parameters ------------------------------------------------
    L = _next_pow2(3 * n - 2)
    beta_schedule = [1, 5, 20, 80, 200, 500, 1500, 4000]
    total_iters = 0
    max_total_iters = 25000

    # ---- gradient‑descent phase ------------------------------------
    for beta in beta_schedule:
        lr0 = 0.5 / np.sqrt(beta)          # base learning rate
        iters = 2500
        for it in range(iters):
            if time.time() - start_time > budget_s * 0.85:
                break
            grad = _gradient(best_h, beta, L)
            lr = lr0 / (1.0 + it * 0.001)   # mild decay
            h_new = best_h - lr * grad
            h_new = _project_feasible(h_new, target)
            best_h = h_new
            total_iters += 1
            if (total_iters % 30) == 0:
                c = compute_c5(best_h)
                if c < best_c:
                    best_c, best_h = c, best_h.copy()
        # quick check after each beta
        c = compute_c5(best_h)
        if c < best_c:
            best_c, best_h = c, best_h.copy()
        if time.time() - start_time > budget_s * 0.85:
            break

    # ---- occasional random restarts --------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30.0:
        restarts = 4
        per_restart = remaining / (restarts + 1)
        for _ in range(restarts):
            h = _random_feasible(n, rng)
            # short gradient run with moderate beta
            for beta in [5, 80, 200]:
                iters = 1200
                for it in range(iters):
                    if time.time() - start_time > budget_s * 0.97:
                        break
                    grad = _gradient(h, beta, L)
                    lr = 0.3 / np.sqrt(beta) / (1.0 + it * 0.001)
                    h = _project_feasible(h - lr * grad, target)
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
            if time.time() - start_time > budget_s * 0.97:
                break

    # ---- final pair‑transfer polishing -----------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        h_opt, c_opt = pair_transfer_optimize(
            best_h, rng, max_iters=20000, time_limit=remaining, init_factor=0.5
        )
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()

    # ---- final clean‑up --------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_feasible(best_h, target)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
