def conv_full(x, y):
    L = len(x) + len(y) - 1
    nfft = 1 << (L-1).bit_length()
    X = np.fft.rfft(x, nfft)
    Y = np.fft.rfft(y, nfft)
    res = np.fft.irfft(X*Y, nfft)
    return res[:L]

    h = h0.copy()
    n = len(h)
    S = n/2.0
    for beta in betas:
        for it in range(max_iters_per_beta):
            # time check

def project_onto_capped_simplex(v, S):
    # Ensure sum(v) = S after projection onto [0,1] caps.
    # Binary search for lambda such that sum(clamp(v - lambda, 0, 1)) = S.
    lo = -1.0
    hi = 2.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # ensure sum exact via small correction if needed
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        # adjust a non-bound element
        mask = (x > 1e-12) & (x < 1-1e-12)
        idx = np.where(mask)[0]
        if idx.size == 0:
            idx = np.arange(len(x))
        i = idx[0]
        x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def gradient_shift(h, s):
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for _ in range(max_steps):
        s, _, cur = worst_shift_info(best_h)
        grad = gradient_shift(best_h, s)
        pos = np.where((grad > 1e-12) & (best_h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (best_h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps < 1e-12:
            break
        # try full
        nh = best_h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        c_new = compute_c5(nh)
        if c_new < best_c - 1e-12:
            best_h, best_c = nh, c_new
            continue
        # try half
        eps = max_eps * 0.5
        nh[i] = best_h[i] - eps
        nh[j] = best_h[j] + eps
        c_new = compute_c5(nh)
        if c_new < best_c - 1e-12:
            best_h, best_c = nh, c_new
            continue
        break
    return best_h

import time
import numpy as np

# ---------- basic operations ----------
def conv_full(x, y):
    """Full convolution using FFT (real inputs)."""
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    res = np.fft.irfft(X * Y, N)
    return res[:n]

def project_onto_capped_simplex(v, S):
    """Projection of v onto {x: 0<=x<=1, sum x = S}."""
    lo, hi = -1.0, 2.0            # safe bounds for lambda
    for _ in range(40):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        idx = np.where(mask)[0]
        if idx.size == 0:
            idx = np.arange(len(x))
        i = idx[0]
        x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def compute_c5(h):
    """Exact discrete C5 for a vector h."""
    n = h.size
    a = 1.0 - h
    c = conv_full(h, a)
    return c.max() * (2.0 / n)

def worst_shift_info(h):
    """Return worst shift (s), its index and correlation value."""
    n = h.size
    c = conv_full(h, 1.0 - h)
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    return s, idx, c[idx]

def gradient_shift(h, s):
    """Gradient of the worst‑shift correlation w.r.t. h."""
    n = h.size
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] += (1 - h[:n - s_abs])
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def greedy_optimize(h, max_steps=2000):
    """Deterministic greedy improvement using the worst shift."""
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for _ in range(max_steps):
        s, _, _ = worst_shift_info(best_h)
        grad = gradient_shift(best_h, s)
        pos = np.where((grad > 1e-12) & (best_h > 1e-12))[0]
        neg = np.where((grad < -1e-12) & (best_h < 1.0 - 1e-12))[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps < 1e-12:
            break
        # full transfer
        nh = best_h.copy()
        nh[i] -= max_eps
        nh[j] += max_eps
        c_new = compute_c5(nh)
        if c_new < best_c - 1e-12:
            best_h, best_c = nh, c_new
            continue
        # half transfer
        eps = max_eps * 0.5
        nh[i] = best_h[i] - eps
        nh[j] = best_h[j] + eps
        c_new = compute_c5(nh)
        if c_new < best_c - 1e-12:
            best_h, best_c = nh, c_new
            continue
        break
    return best_h

def random_feasible_vector(n, rng):
    v = rng.random(n)
    return project_onto_capped_simplex(v, n / 2.0)

def upscale_h(h, new_n):
    """Linear interpolation to a finer grid."""
    old_n = h.size
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(x_new, x_old, h)
