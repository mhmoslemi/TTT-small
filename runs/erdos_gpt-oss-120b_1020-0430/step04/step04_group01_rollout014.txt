def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


def enforce_sum(h: np.ndarray, target: float, rng: np.random.Generator) -> np.ndarray:
    """Adjust h so that its sum equals target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1]^n with sum = n/2."""
    h = np.full(n, 0.5)
    for _ in range(n * 5):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = (rng.random() - 0.5) * 2.0 * max_eps * 0.3
        h[i] -= eps
        h[j] += eps
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = enforce_sum(h, target, rng)
    return h

# Correlation and C5 evaluation
# --------------------------------------------------------------
def correlate_full(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1‑h) using FFT (zero‑padded)."""
    n = h.shape[0]
    L = next_pow2(2 * n - 1)
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft((1 - h)[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h: np.ndarray) -> float:
    """Discrete C₅ for step function h (dx = 2/n)."""
    return correlate_full(h).max() * (2.0 / h.size)

# Smooth surrogate for SLSQP
# --------------------------------------------------------------
_CURRENT_TARGET = None
_CURRENT_BETA = None


def _sum_eq_constraint(x):
    return np.sum(x) - _CURRENT_TARGET


def _softmax_obj(x):
    conv = correlate_full(x)
    bc = _CURRENT_BETA * conv
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m)))) / _CURRENT_BETA


def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    """SLSQP on the smooth surrogate for a given β."""
    global _CURRENT_TARGET, _CURRENT_BETA
    n = h0.size
    _CURRENT_TARGET = n / 2.0
    _CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": _sum_eq_constraint}
    res = minimize(
        _softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = _CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
            x[i] += add
            diff = _CURRENT_TARGET - x.sum()
    return x

            i, j = j, i
        eps_max = min(h[i], 1.0 - h[j])
        if eps_max <= 0.0:
            it += 1
            continue
        eps = eps_max * factor
        if eps <= 0.0:
            it += 1
            continue
        h[i] -= eps
        h[j] += eps
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        else:
            h[i] += eps
            h[j] -= eps
        it += 1
    return best_h, best_c


# --------------------------------------------------------------
# Simulated annealing (continuous)
# --------------------------------------------------------------
def simulated_annealing(h_start, rng, deadline, max_iters=300000):
    n = h_start.size
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T0 = cur_c * 0.2 + 1e-12
    Tf = 1e-8
    it = 0
    while time.time() < deadline and it < max_iters:
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] < h[j]:
            i, j = j, i
        eps_max = min(h[i], 1.0 - h[j])
        if eps_max <= 0.0:
            it += 1
            continue
        eps = eps_max * rng.random() * 0.5
        if eps <= 0.0:
            it += 1
            continue
        h[i] -= eps
        h[j] += eps
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i] += eps
            h[j] -= eps
        it += 1
    return best_h, best_c


# --------------------------------------------------------------
# Binary simulated annealing (swap‑based)
# --------------------------------------------------------------
def binary_sa(start_bin, rng, deadline, max_iters=500000):
    h = start_bin.astype(float).copy()
    n = h.size
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T0 = cur_c * 0.2 + 1e-12
    Tf = 1e-8
    it = 0
    while time.time() < deadline and it < max_iters:
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        i = rng.integers(n)
        j = rng.integers(n)
        if h[i] == h[j]:
            it += 1
            continue
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c, best_h = new_c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        it += 1
    return best_h, best_c


# --------------------------------------------------------------
# Upsampling + fine hill‑climb
# --------------------------------------------------------------
def upsample_and_opt(h_coarse, rng, deadline):
    h_fine = np.repeat(h_coarse, 2)
    n_fine = h_fine.size
    target = n_fine / 2.0
    diff = target - h_fine.sum()
    if abs(diff) > 1e-12:
        h_fine = enforce_sum(h_fine, target, rng)
    # remaining time driven hill‑climb
    h_opt, c_opt = hill_climb_improve(
        h_fine,
        rng,
        deadline,
        total_iters=200000,
        eps_start=0.3,
        eps_end=1e-6,
    )
    return h_opt, c_opt


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s

    # ----- Initial candidate -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 360
        h0 = random_feasible_vector(n, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ----- Phase 1: deterministic hill‑climb --------------------------------
    phase1_dead = start_time + 0.35 * budget
    h1, c1 = hill_climb_improve(best_h, rng, phase1_dead,
                                total_iters=300000, eps_start=0.4, eps_end=1e-6)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # ----- Phase 2: simulated annealing (continuous) ------------------------
    phase2_dead = start_time + 0.70 * budget
    h2, c2 = simulated_annealing(best_h, rng, phase2_dead, max_iters=400000)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # ----- Phase 3: binary simulated annealing -------------------------------
    phase3_dead = start_time + 0.90 * budget
    idx = np.argsort(-best_h)[: best_h.size // 2]
    bin_start = np.zeros(best_h.size)
    bin_start[idx] = 1.0
    h3, c3 = binary_sa(bin_start, rng, phase3_dead, max_iters=500000)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # ----- Phase 4: optional up‑sampling -------------------------------------
    if time.time() < start_time + budget and best_h.size * 2 <= 1440:
        up_dead = start_time + budget
        h_up, c_up = upsample_and_opt(best_h, rng, up_dead)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # ----- Phase 5: final high‑β SLSQP refinements ---------------------------
    while time.time() < start_time + budget:
        for beta in (8000.0, 20000.0, 50000.0):
            if time.time() >= start_time + budget:
                break
            try:
                h_opt = slsqp_optimize(best_h, beta, max_iter=300)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c:
                    best_h, best_c = h_opt, c_opt
            except Exception:
                continue
        # if still time, attempt a few random restarts with brief hill‑climb
        if time.time() >= start_time + budget:
            break
        h_rand = random_feasible_vector(best_h.size, rng)
        h_tmp, c_tmp = hill_climb_improve(
            h_rand, rng, start_time + budget,
            total_iters=50000, eps_start=0.3, eps_end=1e-6,
        )
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # ----- Final cleaning ----------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
