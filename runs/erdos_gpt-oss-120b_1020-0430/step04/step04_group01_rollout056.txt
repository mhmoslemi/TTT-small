def gradient_h(h, beta, lam):
    n = h.size
    # compute overlap convolution: autocorr
    L = 1 << ((2*n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    fr = np.fft.rfft(h[::-1], L)
    conv = np.fft.irfft(fh * fr, L)[:2*n-1]  # overlap_k
    sum_h = h.sum()
    corr = sum_h - conv  # length 2n-1, k from -(n-1)..(n-1)
    # smooth weights
    m = np.max(beta * corr)
    exp_terms = np.exp(beta * corr - m)
    w = exp_terms / exp_terms.sum()
    # compute weighted neighbor sum
    # full convolution w * h
    L2 = 1 << ((3*n - 2).bit_length())
    fw = np.fft.rfft(w, L2)
    fh2 = np.fft.rfft(h, L2)
    conv_fwd = np.fft.irfft(fw * fh2, L2)[: (2*n-1) + n -1]  # len 3n-2
    g_fwd = conv_fwd[n-1 : n-1 + n]
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, L2)
    conv_rev = np.fft.irfft(fw_rev * fh2, L2)[: (2*n-1) + n -1]
    g_rev = conv_rev[n-1 : n-1 + n]
    g = g_fwd + g_rev
    # gradient of smooth part = 1 - g
    grad = 1.0 - g
    # add penalty gradient
    sum_diff = sum_h - (n/2.0)
    grad += 2.0 * lam * sum_diff
    return grad

import time
import numpy as np

# ---------- feasibility utilities ----------
def enforce_sum(h, target, rng):
    """Adjust h so that Σh == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    n = h.size
    # simple iterative redistribution; usually finishes in a few steps
    for _ in range(5000):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = rng.integers(0, n)
            if h[idx] >= 1.0:
                continue
            add = min(diff, 1.0 - h[idx])
        else:
            idx = rng.integers(0, n)
            if h[idx] <= 0.0:
                continue
            add = max(diff, -h[idx])
        h[idx] += add
        diff = target - h.sum()
    return h

def adjust_sum(h, target, rng):
    """Lightweight correction of the sum, falling back to enforce_sum."""
    diff = target - h.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        add = diff / cnt
        h[mask] += add
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
    if abs(target - h.sum()) > 1e-10:
        h = enforce_sum(h, target, rng)
    return h

# ---------- core correlation ----------
def corr_full(h):
    """Linear autocorrelation of h (overlap values) using FFT."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    conv = np.fft.irfft(fh * np.conj(fh), L)[: 2 * n - 1]
    return conv  # length 2n-1, index shift = -(n-1)..(n-1)

def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/n)."""
    n = h.size
    overlap = corr_full(h)
    sum_h = h.sum()
    corr = sum_h - overlap
    return np.max(corr) * (2.0 / n)

# ---------- gradient for smoothed max ----------
def gradient_and_corr(h, beta, L, L2):
    """Return gradient of (log‑sum‑exp)/β and the raw corr array."""
    n = h.size
    # overlap via autocorrelation
    fh = np.fft.rfft(h, L)
    overlap = np.fft.irfft(fh * np.conj(fh), L)[: 2 * n - 1]
    sum_h = h.sum()
    corr = sum_h - overlap                     # length 2n-1

    # softmax weights
    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    w = exp_bc / np.sum(exp_bc)                # length 2n-1

    # weighted neighbor sum g = Σ_k w_k (h_{i+k}+h_{i-k})
    fh2 = np.fft.rfft(h, L2)
    fw = np.fft.rfft(w, L2)
    conv_fwd = np.fft.irfft(fw * fh2, L2)[: (2 * n - 1) + n - 1]
    g_fwd = conv_fwd[n - 1 : n - 1 + n]

    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, L2)
    conv_rev = np.fft.irfft(fw_rev * fh2, L2)[: (2 * n - 1) + n - 1]
    g_rev = conv_rev[n - 1 : n - 1 + n]

    g = g_fwd + g_rev
    grad = 1.0 - g
    return grad, corr

# ---------- main optimisation ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = budget_s

    # ---- initialise h ----
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum(h, n / 2.0, rng)
    else:
        n = 720
        h = np.full(n, 0.5)
        h = enforce_sum(h, n / 2.0, rng)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    target_sum = n / 2.0
    L = 1 << ((2 * n - 1).bit_length())
    L2 = 1 << ((3 * n - 2).bit_length())

    # ---- Adam hyper‑parameters ----
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    # ---- beta schedule ----
    beta_list = [5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0]

    for beta in beta_list:
        # time check
        if time.time() - start_time > budget * 0.85:
            break

        # learning rate scaled with beta
        lr = 0.4 / np.sqrt(beta)

        # momentum buffers
        m = np.zeros_like(best_h)
        v = np.zeros_like(best_h)
        t = 0

        # iterations per beta (fixed)
        iters = 500
        for _ in range(iters):
            if time.time() - start_time > budget * 0.85:
                break
            t += 1
            grad, _ = gradient_and_corr(best_h, beta, L, L2)
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            m_hat = m / (1 - b1 ** t)
            v_hat = v / (1 - b2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            best_h = best_h - step
            # box projection
            np.clip(best_h, 0.0, 1.0, out=best_h)
            # light sum correction
            best_h = adjust_sum(best_h, target_sum, rng)

            # occasional evaluation
            if t % 50 == 0:
                c = compute_c5(best_h)
                if c < best_c:
                    best_c, best_h = c, best_h.copy()

        # final projection for this beta
        best_h = enforce_sum(best_h, target_sum, rng)
        c = compute_c5(best_h)
        if c < best_c:
            best_c, best_h = c, best_h.copy()

    # ---- optional up‑sampling ----
    time_now = time.time()
    if (time_now - start_time) < budget * 0.9 and best_h.size * 2 <= 1440:
        # up‑sample by factor 2
        n_fine = best_h.size * 2
        h_fine = np.repeat(best_h, 2)
        target_fine = n_fine / 2.0
        h_fine = enforce_sum(h_fine, target_fine, rng)

        Lf = 1 << ((2 * n_fine - 1).bit_length())
        L2f = 1 << ((3 * n_fine - 2).bit_length())
        beta = 5000.0
        lr = 0.2 / np.sqrt(beta)

        mf = np.zeros_like(h_fine)
        vf = np.zeros_like(h_fine)
        tf = 0
        iters_fine = 1200
        for _ in range(iters_fine):
            if time.time() - start_time > budget:
                break
            tf += 1
            grad, _ = gradient_and_corr(h_fine, beta, Lf, L2f)
            mf = b1 * mf + (1 - b1) * grad
            vf = b2 * vf + (1 - b2) * (grad * grad)
            mf_hat = mf / (1 - b1 ** tf)
            vf_hat = vf / (1 - b2 ** tf)
            step = lr * mf_hat / (np.sqrt(vf_hat) + eps)
            h_fine = h_fine - step
            np.clip(h_fine, 0.0, 1.0, out=h_fine)
            h_fine = adjust_sum(h_fine, target_fine, rng)

        h_fine = enforce_sum(h_fine, target_fine, rng)
        c_fine = compute_c5(h_fine)
        if c_fine < best_c:
            best_c, best_h = c_fine, h_fine.copy()

    # ---- final clean‑up ----
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, best_h.size / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
