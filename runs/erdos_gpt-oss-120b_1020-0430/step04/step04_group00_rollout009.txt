def correlate_fft(h):
    n = h.shape[0]
    L = 2*n - 1
    nfft = 1 << (L-1).bit_length()
    H = np.fft.fft(h, nfft)
    G = np.fft.fft(1.0 - h, nfft)
    c = np.fft.ifft(H * np.conj(G)).real
    # c[0:L] corresponds to shift indices -(n-1) .. (n-1)
    return c[:L]

import time
import numpy as np

# ---------- core utilities ----------
def _correlate_fft(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1-h) using FFT (real inputs)."""
    n = h.shape[0]
    L = 2 * n - 1
    nfft = 1 << ((L - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(1.0 - h, n=nfft)
    c = np.fft.irfft(H * np.conj(G), n=nfft)[:L]
    return c

def _c5_from_corr(corr: np.ndarray, n: int) -> float:
    return float(corr.max() * (2.0 / n))

def compute_c5_fft(h: np.ndarray) -> float:
    """C5 value for a discrete h using the FFT correlation."""
    return _c5_from_corr(_correlate_fft(h), h.shape[0])

def _adjust_sum_exact(h: np.ndarray, rng: np.random.Generator) -> None:
    """Modify h in-place so that sum(h) == len(h)/2 while staying in [0,1]."""
    n = h.shape[0]
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return
    # simple stochastic repair
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            cap = 1.0 - h[i]
            if cap <= 0.0:
                continue
            add = min(diff, cap)
        else:
            cap = h[i]
            if cap <= 0.0:
                continue
            add = max(diff, -cap)
        h[i] += add
        diff = target - h.sum()

def _worst_shift_and_corr(h: np.ndarray):
    """Return (shift, correlation array) where shift gives the maximal correlation."""
    n = h.shape[0]
    corr = _correlate_fft(h)
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, corr

def gradient_descent(
    h: np.ndarray,
    rng: np.random.Generator,
    max_iter: int = 300,
    step0: float = 0.2,
    decay: float = 0.99,
    line_search: bool = False,
):
    """Projected gradient descent on the worst shift."""
    n = h.shape[0]
    target = n * 0.5
    for it in range(max_iter):
        shift, _ = _worst_shift_and_corr(h)

        if shift == 0:
            g = 1.0 - 2.0 * h
        else:
            k = abs(shift)
            g = np.zeros_like(h)
            if shift > 0:
                g[: n - k] += 1.0 - h[k:]
                g[k:] -= h[: n - k]
            else:  # shift < 0
                g[k:] += 1.0 - h[: n - k]
                g[: n - k] -= h[k:]

        step = step0 * (decay ** it)

        if line_search:
            # try three candidates and keep the best
            best_h = h
            best_val = compute_c5_fft(h)
            for factor in (1.0, 0.5, 2.0):
                trial = h - step * factor * g
                trial = np.clip(trial, 0.0, 1.0)
                _adjust_sum_exact(trial, rng)
                val = compute_c5_fft(trial)
                if val < best_val:
                    best_val = val
                    best_h = trial
            h = best_h
        else:
            h = h - step * g
            h = np.clip(h, 0.0, 1.0)
            _adjust_sum_exact(h, rng)

    return h

def local_random_search(
    h: np.ndarray,
    rng: np.random.Generator,
    iters: int = 300,
    delta: float = 0.01,
):
    """Small random mass transfers to possibly lower C5."""
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5_fft(best_h)
    for _ in range(iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # we will move up to `delta` from j to i
        max_take = min(delta, best_h[j])
        max_give = min(delta, 1.0 - best_h[i])
        move = min(max_take, max_give)
        if move <= 0.0:
            continue
        h_new = best_h.copy()
        h_new[i] += move
        h_new[j] -= move
        c_new = compute_c5_fft(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h
