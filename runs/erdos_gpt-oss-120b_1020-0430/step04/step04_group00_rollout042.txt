def greedy_binary_optimize(h, deadline):
    n = h.size
    # h is binary 0/1 uint8
    # ensure h sums to n/2
    target = n // 2
    # loop
    while True:
        # time check
        if time.time() > deadline:
            break
        # compute O (max overlaps) via convolution
        conv = np.correlate(h, h, mode='full')
        O = target - conv[n-1:]  # length n, O[0]=0
        current_max = O.max()
        # compute Hmat
        # allocate Hmat as float32 for speed
        H = np.zeros((n, n), dtype=np.float32)
        for i in range(n):
            # forward part
            H[i, :n-i] += h[i:]
            # backward part
            # h[i::-1] length i+1
            H[i, :i+1] += h[i::-1]
        # find best swap using H
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        best_improve = 0
        best_i = -1
        best_j = -1
        # iterate over ones
        for i in ones:
            O_plus = O + H[i]  # shape n
            # compute max for each zero
            # broadcasting: O_plus[None,:] - H[zeros]  shape (len(zeros), n)
            cand_max = np.max(O_plus[None, :] - H[zeros], axis=1)
            improve_arr = current_max - cand_max
            # find max improvement
            j_idx = np.argmax(improve_arr)
            improve = improve_arr[j_idx]
            if improve > best_improve:
                best_improve = improve
                best_i = i
                best_j = zeros[j_idx]
        if best_improve > 0:
            # perform swap
            h[best_i] = 0
            h[best_j] = 1
        else:
            break
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helpers ----------
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def adjust_sum_exact(h, rng):
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

# ---------- gradient refinement ----------
def gradient_refine(h, rng, step=0.02, max_iter=20):
    n = h.size
    target = n / 2.0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]
            g[k:] -= h[:upper]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(200):
                i = rng.integers(0, n)
                add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

# ---------- smooth‑max SLSQP ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=500):
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    diff = CURRENT_TARGET - x.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        add = min(diff, 1.0 - x[i]) if diff > 0 else max(diff, -x[i])
        x[i] += add
        diff = CURRENT_TARGET - x.sum()
    return x

# ---------- greedy binary optimizer ----------
def greedy_binary_optimize(h0, deadline):
    h = h0.copy()
    n = h.size
    target = n // 2
    while True:
        if time.time() > deadline:
            break
        # overlap O for shifts 0..n-1
        conv = np.correlate(h, h, mode="full")
        O = target - conv[n - 1 :]
        cur_max = O.max()
        # build H matrix: H[i,k] = h[i+k] + h[i-k] (k>0), H[i,0]=1
        H = np.zeros((n, n), dtype=np.float32)
        for i in range(n):
            H[i, : n - i] += h[i:]
            H[i, : i + 1] += h[i::-1]
        H[:, 0] = 1.0
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        best_imp = 0
        best_i = -1
        best_j = -1
        for i in ones:
            O_plus = O + H[i]  # shape (n,)
            cand = np.max(O_plus[None, :] - H[zeros], axis=1)  # (len(zeros),)
            imp = cur_max - cand
            j_idx = np.argmax(imp)
            if imp[j_idx] > best_imp:
                best_imp = imp[j_idx]
                best_i = i
                best_j = zeros[j_idx]
        if best_imp > 0:
            h[best_i] = 0
            h[best_j] = 1
        else:
            break
    return h

# ---------- main run ----------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.97

    best_h = None
    best_c = np.inf
    best_n = None

    parent = globals().get("initial_h_values", None)
    candidate_ns = [360, 300, 240, 180]

    for n in candidate_ns:
        if time.time() > deadline:
            break
        target = n // 2
        max_restarts = 3
        for _ in range(max_restarts):
            if time.time() > deadline:
                break
            # initialise binary vector
            if parent is not None and parent.size == n:
                h0 = (np.array(parent) > 0.5).astype(np.uint8)
                # force exact half‑ones
                cur = int(h0.sum())
                if cur != target:
                    diff = target - cur
                    if diff > 0:
                        zeros = np.flatnonzero(h0 == 0)
                        h0[rng.choice(zeros, diff, replace=False)] = 1
                    else:
                        ones = np.flatnonzero(h0 == 1)
                        h0[rng.choice(ones, -diff, replace=False)] = 0
            elif parent is not None:
                m = parent.size
                xs = np.linspace(0.0, 2.0, m, endpoint=False)
                xt = np.linspace(0.0, 2.0, n, endpoint=False)
                interp = np.interp(xt, xs, parent)
                idx = np.argpartition(interp, -target)[-target:]
                h0 = np.zeros(n, dtype=np.uint8)
                h0[idx] = 1
            else:
                h0 = np.zeros(n, dtype=np.uint8)
                h0[rng.choice(n, target, replace=False)] = 1

            # greedy binary optimisation
            sub_dead = min(deadline, time.time() + (deadline - time.time()) * 0.4)
            h_bin = greedy_binary_optimize(h0, sub_dead)

            # evaluate
            c = compute_c5(h_bin.astype(float))
            if c < best_c:
                best_h, best_c, best_n = h_bin.astype(float).copy(), c, n

            # gradient refinement (fractional)
            h_frac = gradient_refine(h_bin.astype(float), rng, step=0.04, max_iter=30)
            c = compute_c5(h_frac)
            if c < best_c:
                best_h, best_c, best_n = h_frac.copy(), c, n

            # SLSQP cascade
            for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000, 10000):
                if time.time() > deadline:
                    break
                try:
                    h_frac = slsqp_optimize(h_frac, beta, rng, max_iter=600)
                except Exception:
                    continue
                c = compute_c5(h_frac)
                if c < best_c:
                    best_h, best_c, best_n = h_frac.copy(), c, n

            # final gradient sweep
            h_frac = gradient_refine(h_frac, rng, step=0.02, max_iter=20)
            c = compute_c5(h_frac)
            if c < best_c:
                best_h, best_c, best_n = h_frac.copy(), c, n

    # optional up‑sampling for extra refinement
    if best_h is not None and time.time() < deadline and best_n * 2 <= 720:
        n_up = best_n * 2
        xs = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xt = np.linspace(0.0, 2.0, n_up, endpoint=False)
        h_up = np.interp(xt, xs, best_h)
        adjust_sum_exact(h_up, rng)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=30)
        for beta in (5, 20, 80, 200, 500, 1000, 2000, 5000):
            if time.time() > deadline:
                break
            h_up = slsqp_optimize(h_up, beta, rng, max_iter=600)
        h_up = gradient_refine(h_up, rng, step=0.02, max_iter=20)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            best_h, best_c, best_n = h_up, c_up, n_up

    # fallback
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_n = n
        best_c = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    adjust_sum_exact(best_h, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_n)
