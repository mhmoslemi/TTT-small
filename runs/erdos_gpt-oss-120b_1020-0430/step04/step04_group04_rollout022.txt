def compute_c5(h):
    n = h.shape[0]
    # convolution via FFT
    rev = (1 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv.max() * (2.0 / n)

def perturb_pairwise(h, rng, scale=0.5):
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the SLSQP surrogate
CURRENT_TARGET = None
CURRENT_BETA = None

# ----------------------------------------------------------------------
# Projection onto the box [0,1] and the hyperplane sum = target
def project_onto_box_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        xm = np.clip(x + mid, 0.0, 1.0)
        s = xm.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


# ----------------------------------------------------------------------
# Simple shift of an array with zero padding outside the domain
def shift_array(arr: np.ndarray, s: int) -> np.ndarray:
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if s >= 0:
        out[: n - s] = arr[s:]
    else:
        s = -s
        out[s:] = arr[: n - s]
    return out


# ----------------------------------------------------------------------
# Correlation (full) via FFT; returns the raw convolution vector
def compute_corr_fft(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact discrete C5 from the correlation
def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    conv = compute_corr_fft(h)
    return conv.max() * (2.0 / n)


# ----------------------------------------------------------------------
# Random feasible vector using the projection routine
def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    x = rng.random(n)
    return project_onto_box_and_sum(x, n / 2.0)


# ----------------------------------------------------------------------
# Small pairwise transfer that preserves the sum
def perturb_pairwise(h: np.ndarray, rng: np.random.Generator, scale: float = 0.5) -> np.ndarray:
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# ----------------------------------------------------------------------
# Equality constraint for SLSQP
def sum_eq_constraint(x: np.ndarray) -> float:
    return np.sum(x) - CURRENT_TARGET


# ----------------------------------------------------------------------
# Smooth max (log-sum-exp) objective for SLSQP
def softmax_obj(x: np.ndarray) -> float:
    conv = compute_corr_fft(x)
    bc = CURRENT_BETA * conv
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


# ----------------------------------------------------------------------
# SLSQP optimisation of the smooth surrogate
def slsqp_optimize(h0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = project_onto_box_and_sum(x, CURRENT_TARGET)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] with a low Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Target resolution
    n = 720
    target_sum = n / 2.0
    dx = 2.0 / n

    # --------------------------------------------------------------
    # Initialise from parent if available, otherwise random
    if "initial_h_values" in globals():
        old_h = np.array(initial_h_values, dtype=float)
        old_n = old_h.size
        old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
        new_x = (np.arange(n) + 0.5) * (2.0 / n)
        h0 = np.interp(new_x, old_x, old_h)
        h0 = project_onto_box_and_sum(h0, target_sum)
    else:
        h0 = random_feasible_vector(n, rng)

    # --------------------------------------------------------------
    # Small pool of structured candidates
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = [h0]

    tri = np.where(xs <= 1.0, xs, 2.0 - xs)                     # triangular bump
    candidates.append(project_onto_box_and_sum(tri, target_sum))

    cosb = (1.0 + np.cos(np.pi * (xs - 1.0))) / 2.0           # cosine bump
    candidates.append(project_onto_box_and_sum(cosb, target_sum))

    gauss = np.exp(-((xs - 1.0) ** 2) / (2 * 0.2 ** 2))       # Gaussian‑like
    candidates.append(project_onto_box_and_sum(gauss, target_sum))

    sinb = 0.5 + 0.5 * np.sin(np.pi * (xs - 0.5))             # sine bump
    candidates.append(project_onto_box_and_sum(sinb, target_sum))

    for _ in range(5):
        candidates.append(random_feasible_vector(n, rng))

    # --------------------------------------------------------------
    # Keep the best among the pool
    best_h = None
    best_c = np.inf
    for cand in candidates:
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()

    # --------------------------------------------------------------
    # Main optimisation loop (subgradient + occasional random moves)
    h = best_h.copy()
    conv = compute_corr_fft(h)
    cur_c = conv.max() * dx
    step0 = 200.0
    it = 0

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.9:
            break

        # 20 % random perturbation, otherwise subgradient step
        if rng.random() < 0.2:
            h_new = perturb_pairwise(h, rng, scale=0.5)
        else:

            grad = dx * (1.0 - shift_array(h, s) - shift_array(h, -s))
            step = step0 / np.sqrt(it + 1.0)
            h_new = h - step * grad

        h_new = project_onto_box_and_sum(h_new, target_sum)
        conv_new = compute_corr_fft(h_new)
        new_c = conv_new.max() * dx

        # Greedy acceptance with a tiny probability for uphill moves
        if new_c < cur_c or rng.random() < 0.001:
            h = h_new
            conv = conv_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()

        it += 1

        # Periodic SLSQP refinement when we have spent a fair amount of time
        if it % 5000 == 0 and elapsed > budget_s * 0.5:
            for beta in (200, 800, 2000, 5000):

                    h_ref = slsqp_optimize(best_h, beta, max_iter=400)

                        best_h = h_ref.copy()
                        h = h_ref.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
