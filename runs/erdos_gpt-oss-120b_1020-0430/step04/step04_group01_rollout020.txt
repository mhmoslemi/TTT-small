def gradient_descent(h0, target_sum, deadline):
    # parameters
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t = 0
    beta = 5.0
    beta_max = 2000.0
    beta_inc = 1.05  # multiplicative increase each iter?
    lr = 0.2
    lr_decay = 0.9995
    momentum = np.zeros_like(h)
    mu = 0.9
    while time.time() < deadline:
        # compute corr
        corr = correlate_full(h)
        # compute weights w (softmax)
        maxc = corr.max()
        expb = np.exp(beta * (corr - maxc))
        w = expb / expb.sum()
        # compute w_mod
        w_mod = w[:len(h)] + np.concatenate(([0.0], w[:len(h)-1]))
        # compute A and B via FFT
        H = np.fft.fft(h)
        W = np.fft.fft(w_mod)
        A = np.real(np.fft.ifft(W * H))
        W_rev = np.fft.fft(w_mod[::-1])
        B = np.real(np.fft.ifft(W_rev * H))
        grad = 1.0 - (A + B)
        # update momentum
        momentum = mu * momentum + grad
        h_new = h - lr * momentum
        # project
        h_new = project_onto_bounds_sum(h_new, target_sum)
        # evaluate
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            # maybe reset momentum?
            # Optionally adapt lr
        # update
        h = h_new
        # schedule updates

import time
import numpy as np

# --------------------------------------------------------------
# Projection onto [0,1] with exact sum constraint
# --------------------------------------------------------------
def project_onto_bounds_sum(y, target, iters=60):
    """Find x = clip(y + lam, 0, 1) with sum(x) == target via bisection on lam."""
    lo, hi = -2.0, 2.0
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random vector projected onto the simplex with upper bound 1."""
    y = rng.random(n)
    return project_onto_bounds_sum(y, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Transfer a random ε from i to j while preserving the sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h.copy()
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


# --------------------------------------------------------------
# Correlation utilities
# --------------------------------------------------------------
def correlate_full(h):
    """Full linear correlation of h with (1 - h) (zero‑padded)."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, L)
    rev = (1 - h)[::-1]
    fr = np.fft.rfft(rev, L)
    conv = np.fft.irfft(fh * fr, L)[: 2 * n - 1]
    return conv


def compute_c5(h):
    """Discrete C₅ for step function h (dx = 2/n)."""
    conv = correlate_full(h)
    return conv.max() * (2.0 / h.size)


# --------------------------------------------------------------
# Helper to fold soft‑max weights to a length‑n periodic vector
# --------------------------------------------------------------
def fold_weights(w):
    """Fold 2n‑1 weights w_k (k = -(n-1)…n‑1) into a length‑n periodic vector."""
    n = (len(w) + 1) // 2
    w_mod = w[n - 1 : n - 1 + n].copy()      # shifts 0…n‑1
    w_mod[1:] += w[: n - 1]                  # add shifts -(n‑1)…-1
    return w_mod


# --------------------------------------------------------------
# Gradient descent on the smooth log‑sum‑exp surrogate
# --------------------------------------------------------------
def gradient_descent(h0, target_sum, deadline):
    """Projected gradient descent on  (1/β) log‑sum‑exp of corr_k."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)

    beta = 5.0
    beta_max = 2000.0
    beta_mul = 1.03

    lr = 0.15
    lr_decay = 0.9995

    momentum = np.zeros_like(h)

    while time.time() < deadline:
        # correlation
        corr = correlate_full(h)                      # length 2n‑1
        # soft‑max weights
        m = corr.max()
        expb = np.exp(beta * (corr - m))
        w = expb / expb.sum()                         # sum w = 1

        # folded weights for cyclic convolutions
        w_mod = fold_weights(w)                       # length n
        w_rev = w_mod[::-1]

        # cyclic convolutions via FFT
        H = np.fft.fft(h)
        A = np.real(np.fft.ifft(H * np.fft.fft(w_mod)))
        B = np.real(np.fft.ifft(H * np.fft.fft(w_rev)))

        grad = 1.0 - (A + B)                           # ∇ of smooth objective

        # momentum update
        momentum = 0.9 * momentum + grad
        h_new = h - lr * momentum

        # projection onto feasible set
        h_new = project_onto_bounds_sum(h_new, target_sum)

        c = compute_c5(h_new)
        if c < best_c:
            best_c, best_h = c, h_new.copy()

        h = h_new
        beta = min(beta_max, beta * beta_mul)
        lr *= lr_decay

    return best_h, best_c


# --------------------------------------------------------------
# Simple binary simulated annealing
# --------------------------------------------------------------
def binary_sa(start_bin, rng, deadline):
    """Swap‑based SA on a binary vector (sum = n/2)."""
    h = start_bin.astype(float).copy()
    n = h.size
    target = n // 2
    best_h = h.copy()
    best_c = compute_c5(h)

    T0 = best_c * 0.02 + 1e-6
    Tf = 1e-6
    max_iters = 30000

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if h[i] == h[j]:
            continue
        # ensure i is 1, j is 0
        if h[i] < h[j]:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - best_c
        frac = it / max_iters
        T = T0 * ((Tf / T0) ** frac)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c


# --------------------------------------------------------------
# Greedy hill‑climb (continuous)
# --------------------------------------------------------------
def hill_climb(h_start, rng, iters=2000, step_scale=0.2):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(iters):
        new_h = perturb_h(h, rng, scale=step_scale)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
    return best_h, best_c
